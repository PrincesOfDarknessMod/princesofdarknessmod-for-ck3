namespace = POD_web_of_knives

POD_web_of_knives.1000 = {
	hidden = yes
	type = character_event
	
	trigger = {
		has_variable = web_gold_reward
		NOT = { has_character_flag = won_bounty_duel }
		is_ai = yes
	}
	
	immediate = {
		
		if = {
			limit = { var:web_gold_reward = 100 }
			create_character = {
				location = root.location
				template = POD_vampire_web_assassin_1
				save_scope_as = assassin
			}	
		}
		
		else_if = {
			limit = { var:web_gold_reward = 250 }
			create_character = {
				location = root.location
				template = POD_vampire_web_assassin_2
				save_scope_as = assassin
			}	
		}
		
		else_if = {
			limit = { var:web_gold_reward = 500 }
			create_character = {
				location = root.location
				template = POD_vampire_web_assassin_3
				save_scope_as = assassin
			}	
		}
		
		else_if = {
			limit = { var:web_gold_reward = 1000 }
			create_character = {
				location = root.location
				template = POD_vampire_web_assassin_4
				save_scope_as = assassin
			}	
		}
		
		else_if = {
			limit = { var:web_gold_reward = 5000 }
			create_character = {
				location = root.location
				template = POD_vampire_web_assassin_5
				save_scope_as = assassin
			}	
		}
		if = {
			limit = { 
				NOT = { var:web_gold_reward = 50 } 
				exists = scope:assassin
			}
			scope:assassin = {
				if = {
					limit = { root = { is_landed = yes } }
					set_employer = root
					return_to_court = yes
				}
				else_if = {
					limit = { exists = root.court_owner }
					set_employer = root.court_owner
					return_to_court = yes
				}
				
				
				add_character_flag = {
					flag = blocked_from_leaving
					years = 2
				}
				random_list = {
					30 = { add_secret = { type = secret_vampgeneration6 } }
					30 = { add_secret = { type = secret_vampgeneration7 } }
					40 = { add_secret = { type = secret_vampgeneration8 } }
					30 = { add_secret = { type = secret_vampgeneration9 } } 
				}
				POD_vampires_look_younger_effect = yes
				POD_character_vampire_bloodpotency_initialization_effect = yes
				POD_character_vampire_predatortype_initialization_effect = yes
				POD_character_vampire_bloodhunger_initialization_effect = yes
				POD_character_initialization_random_background_effect = yes
				root = { add_character_flag = { flag = won_bounty_duel years = 10 } }
				configure_start_single_combat_effect = {
					SC_INITIATOR = scope:assassin
					SC_ATTACKER = scope:assassin
					SC_DEFENDER = root
					FATALITY = always
					FIXED = no
					LOCALE = terrain_scope
					OUTPUT_EVENT = single_combat.1006
					INVALIDATION_EVENT = single_combat.1006	#In a real event, this would obviously be a different even to the output.
				}
				
			}
		}
		
		trigger_event = { id = POD_web_of_knives.1000 years = 20 }
	
		trigger_event = { id = POD_web_of_knives.1001 days = 5 }

	}

}

POD_web_of_knives.2 = {

	type = character_event
	title = POD_web_of_knives.2.t
	desc = POD_web_of_knives.2.desc
	theme = intrigue
	left_portrait = {
		character = root
		animation = schadenfreude
		}
	right_portrait = {
		character = scope:poormortal
		animation = fear
		}		

	immediate = {
		random_courtier = {
			limit = {
				is_landed = no
				has_trait = mortal
				}
			save_scope_as = poormortal	
			}
			
		if = {
			limit = {
				NOT = {
					exists = scope:poormortal 
					}
				}
			hidden_effect = {
				create_character = {
						location = root.capital_province
						template = POD_mortal_hunter_template
						save_scope_as = mortal
					}
					scope:mortal = {
						add_trait = "mortal"
						set_to_lowborn = yes
						add_character_flag = {
							flag = blocked_from_leaving
							years = 2
						}
						add_trait = POD_knows_too_much
						save_scope_as = poormortal
					}
					add_courtier = scope:poormortal
				}
			}
		}			

	option = {
		name = POD_web_of_knives.2.a
		
		stress_impact = {
			compassionate = major_stress_impact_gain
			forgiving = medium_stress_impact_gain
			sadistic = medium_stress_impact_loss
			vengeful = medium_stress_impact_loss
		}		
		
		if = {
			limit = {
				faith = { religion_tag = roadofhumanity_religion }
			}
		add_piety = -200
		}

		unknown_murder_effect = {
			VICTIM = scope:poormortal
			MURDERER = root
			REASON = death_execution
		}		
		
		trigger_event = {
			id = POD_web_of_knives.5
			days = 5
			}
		
		}
		
	option = {
		name = POD_web_of_knives.2.b

		custom_tooltip = web_setback_tooltip
		
		remove_character_flag = wok_trial_flag
		
		}
		
	option = {
		name = POD_web_of_knives.2.c

		custom_tooltip = web_ending_tooltip
				
		}

}
