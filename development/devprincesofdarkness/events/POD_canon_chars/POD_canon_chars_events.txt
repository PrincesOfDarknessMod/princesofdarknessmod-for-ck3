		character = root

		}

	right_portrait = {
		character = character:790512
		animation = love
		}			
		
	trigger = { 
		character:790512 = {
			is_alive = yes
			is_ai = yes
			}
		}	
		
	immediate = {
	if = {
		limit = {
			character:790512 = {
				is_married = yes
				}
		}
		character:790512 = {
			primary_spouse = {
				save_scope_as = kemintirihusband
				}
		}
		}
		
	if = {
		limit = {
			character:790512 = {
				is_married = yes
				}
		}
		character:790512 = { divorce = scope:kemintirihusband }
		}
		
	if = {
		limit = {
			character:505071 = {
				is_married = yes
				}
		}
		character:505071 = {
			primary_spouse = {
				save_scope_as = mithraswife
				}
		}
		}
		
	if = {
		limit = {
			character:505071 = {
				is_married = yes
				}
		}
		character:505071 = { divorce = scope:mithraswife }
		}	
		
		if = {
			limit = {
				character:790512 = {
					is_landed = yes 
					}
				}
				character:790512 = {
					depose = yes
				}
			}
		
		add_courtier = character:790512	
	}
	
	option = 
	{
		name = PODcanon_chars.3.a
		
		add_character_modifier = serpent_garden_modifier	
		
		marry = character:790512
		
		set_relation_soulmate = character:790512
		
		custom_tooltip = PODcanon_chars.mithras	
	}
	
}
