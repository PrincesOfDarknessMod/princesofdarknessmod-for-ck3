namespace = POD_wraith_maintenance

POD_wraith_maintenance.1 = {
	hidden = yes
	type = character_event
	
	trigger = {
		is_ai = no
		num_of_relation_bound_wraith >= 1
	}
	
	immediate = {
		every_relation = {
			type = bound_wraith
			if = {
				limit = {
					OR = {
						NOT = { has_variable = spirit_location }
						var:spirit_location = flag:skinlands
					}
					NOR = {
						has_trait = risen
						has_trait = frozen
					}
				}
				root = { 
					send_interface_toast = {
						title = POD_wraith_maintenance
						left_icon = root
						right_icon = prev
						add_stress = minor_stress_impact_gain
						custom_tooltip = POD_wraith_maintenance.tt
					}
				}
			}
		}
	}
}

POD_wraith_maintenance.20 = {
	hidden = yes
	type = character_event
	
	trigger = {
		has_character_modifier = enfant_modifier
	}
	
	immediate = {
		set_variable = { name = spirit_location value = flag:shadowlands }
		add_character_modifier = traveling_spirit_world_modifier
		add_trait = frozen
		trigger_event = {
			id = POD_wraith_maintenance.21
			years = { 2 5 }
		}
	}
}

POD_wraith_maintenance.21 = {
	hidden = yes
	type = character_event
	
	trigger = {
		has_character_modifier = enfant_modifier
		has_trait = frozen
	}
	
	immediate = {
		remove_trait = frozen
		remove_character_modifier = enfant_modifier
		POD_send_to_dark_kingdom_effect = yes
		if = {
			limit = { culture_has_east_asian_heritage_pillar_trigger = yes }
			random = {
				chance = 20
				modifier = {
					add = 5
					num_sinful_traits >= 1
				}
				modifier = {
					add = 10
					num_sinful_traits >= 2
				}
				modifier = {
					add = 15
					num_sinful_traits >= 3
				}
				modifier = {
					add = 20
					num_sinful_traits >= 4
				}
				modifier = {
					add = 25
					num_sinful_traits >= 5
				}
				trigger_event = {
					id = POD_wraith_maintenance.22
					years = { 2 5 }
				}
			}
		}
	}
}
