namespace = POD_wraith_maintenance

POD_wraith_maintenance.1 = {
	hidden = yes
	type = character_event
	
	trigger = {
		is_ai = no
		num_of_relation_bound_wraith >= 1
	}
	
	immediate = {
		every_relation = {
			type = bound_wraith
			if = {
				limit = {
					NOT = { has_character_modifier = traveling_spirit_world_modifier }
					NOR = {
						has_trait = risen
						has_trait = lament
						has_trait = frozen
					}
				}
				root = { 
					send_interface_toast = {
						title = POD_wraith_maintenance
						left_icon = root
						right_icon = prev
						add_stress = miniscule_stress_impact_gain
						custom_tooltip = POD_wraith_maintenance.tt
					}
				}
			}
		}
	}
}

POD_wraith_maintenance.10 = {
	hidden = yes
	type = character_event
	
	trigger = {
		is_ai = yes
		has_trait = wraith
	}
	
	immediate = {
		if = {
			limit = {
				NOT = { has_trait_with_flag = can_not_cure }
				NOT = { has_character_modifier = eye_of_sakhmet_modifier }
			}
			
			if = {
				limit = {
					has_trait = scarred
					NOT = { has_character_flag = POD_permanent_scarring }
				}
				remove_trait = scarred
				POD_losing_health_level_effect = { COUNT = 1 }
			}
			
			if = {
				limit = {
					has_trait = scarred
					NOT = { has_character_flag = POD_permanent_scarring }
				}
				remove_trait = scarred
				POD_losing_health_level_effect = { COUNT = 1 }
			}
			if = {
				limit = {
					OR = {
						has_trait = wounded_1
						has_trait = wounded_2
						has_trait = wounded_3
					}
				}
				remove_trait = wounded_3
				remove_trait = wounded_2
				remove_trait = wounded_1
				POD_losing_health_level_effect = { COUNT = 1 }
			}
			if = {
				limit = {
					has_trait = blind
				}
				remove_trait = blind
				POD_losing_health_level_effect = { COUNT = 4 }
			}
			if = {
				limit = {
					has_trait = one_eyed
					NOT = { has_character_flag = POD_permanent_one_eyed }
				}
				remove_trait = one_eyed
				POD_losing_health_level_effect = { COUNT = 2 }
			}	

			if = {
				limit = {
					has_trait = one_legged
				}
				remove_trait = one_legged
				POD_losing_health_level_effect = { COUNT = 3 }
			}
			if = {
				limit = {
					has_trait = maimed
				}
				remove_trait = maimed
				POD_losing_health_level_effect = { COUNT = 3 }
			}
			if = {
				limit = {
					has_trait = disfigured
				}
				remove_trait = disfigured
				POD_losing_health_level_effect = { COUNT = 2 }
			}
		}
	}
}
