namespace = POD_wraith_maintenance

POD_wraith_maintenance.1 = {
	hidden = yes
	type = character_event
	
	trigger = {
		is_ai = no
		num_of_relation_bound_wraith >= 1
	}
	
	immediate = {
		every_relation = {
			type = bound_wraith
			if = {
				limit = { 
					has_variable = spirit_location
					OR = { 
						var:spirit_location = flag:shadowlands
						var:spirit_location = flag:dark_kingdom_of_iron
						var:spirit_location = flag:dark_kingdom_of_clay
						var:spirit_location = flag:dark_kingdom_of_ivory
						var:spirit_location = flag:dark_kingdom_of_jade
						var:spirit_location = flag:dark_kingdom_of_obsidian
						var:spirit_location = flag:dark_kingdom_of_sand
						var:spirit_location = flag:swar
					}
				}
				POD_wraith_potency_increase = yes
			}
			else_if = {
				limit = {
					NOR = {
						has_trait = risen
						has_trait = frozen
					}
				}
				root = { 
					send_interface_toast = {
						title = POD_wraith_maintenance
						left_icon = root
						right_icon = prev
						add_stress = minor_stress_impact_gain
						custom_tooltip = POD_wraith_maintenance.tt
					}
				}
			}
			root = { 
				if = {
					limit = { has_trait = mummy }
					send_interface_message = {
						type = event_generic_good
						title = POD_wraith_maintenance
						left_icon = prev
						right_icon = root
						add_hekau_necromancy_lifestyle_xp = medium_lifestyle_xp
					}
				}
			}
		}
	}
}
