namespace = POD_wraith_maintenance

POD_wraith_maintenance.1 = {
	hidden = yes
	type = character_event
	
	trigger = {
		is_ai = no
		num_of_relation_bound_wraith >= 1
	}
	
	immediate = {
		every_relation = {
			type = bound_wraith
			if = {
				limit = {
					NOR = {
						has_trait = risen
						has_trait = frozen
					}
				}
				root = { 
					send_interface_toast = {
						title = POD_wraith_maintenance
						left_icon = root
						right_icon = prev
						add_stress = minor_stress_impact_gain
						custom_tooltip = POD_wraith_maintenance.tt
					}
				}
			}
		}
	}
}

POD_wraith_maintenance.10 = {

		is_imprisoned = no
	}
	
	immediate = {
		POD_reset_pathos_effect = yes
	}
}
