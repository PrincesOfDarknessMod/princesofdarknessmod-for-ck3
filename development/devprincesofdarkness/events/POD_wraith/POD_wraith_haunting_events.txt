namespace = POD_wraith_haunting

POD_wraith_haunting.100 = {
	type = character_event
	title = POD_wraith_haunting.100.t
	desc =  POD_wraith_haunting.100.desc		
	theme = auspex
	right_portrait = scope:actor
	left_portrait = scope:recipient
	
	trigger = {
		scope:recipient = {NOT = {has_trait = wraith}}
		scope:actor = {has_trait = wraith}
		
	}
	
	option = {
		name = POD_wraith_haunting.100.a
		scope:recipient = {
			add_character_modifier = {
				modifier = haunted_willpower
				years = 10
			}
		}
	}
	option = {
		name = POD_wraith_haunting.100.b
		scope:recipient = {
			add_character_modifier = {
				modifier = haunted_sleep
				years = 10
			}
		}
	}
	option = {
		name = POD_wraith_haunting.100.c
		scope:recipient = {
			add_character_modifier = {
				modifier = haunted_schemes
				years = 10
			}
		}
	}
	option = {
		name = POD_wraith_haunting.100.d
		scope:recipient = {
			add_character_modifier = {
				modifier = haunted_rulership
				years = 10
			}
		}
	}
	option = {
		name = POD_wraith_haunting.100.e
		scope:recipient = {
			add_character_modifier = {
				modifier = haunted_war
				years = 10
			}
		}
	}
	option = {
		name = POD_wraith_haunting.100.f
		scope:recipient = {
			add_character_modifier = {
				modifier = haunted_diplomacy
				years = 10
			}
		}
	}
	after = {
		scope:recipient = {trigger_event = POD_wraith_haunting.200}
		scope:actor = {
			if = {
				limit = { has_variable = necromancer }
				scope:necromancer = {
					send_interface_toast = {
						title = POD_wraith_haunting.toast
						left_icon = scope:actor
						right_icon = scope:recipient
					}
				}
			}
		}
	}
}
