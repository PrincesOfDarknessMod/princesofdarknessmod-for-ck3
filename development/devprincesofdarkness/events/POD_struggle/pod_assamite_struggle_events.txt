pod_assamite_struggle.0001 = {
	hidden = yes
	type = empty
	scope = struggle

	immediate = {
		# Activate the catalyst.
		if = {
			limit = {
				exists = global_var:influence_wok
				global_var:influence_wok >= 5
				}
		activate_struggle_catalyst = catalyst_massive_influence_wok
		}
		if = {
			limit = {
				exists = global_var:influence_wok
				global_var:influence_wok <= 0
				}
		activate_struggle_catalyst = catalyst_no_influence_wok
		}
		if = {
			limit = {
				exists = global_var:influence_ashirra
				global_var:influence_wok >= 5
				}
		activate_struggle_catalyst = catalyst_massive_influence_ashirra
		}
		if = {
			limit = {
				exists = global_var:influence_ashirra
				global_var:influence_wok <= 0
				}
		activate_struggle_catalyst = catalyst_no_influence_ashirra
		}
		if = {
			limit = {
				exists = global_var:influence_council
				global_var:influence_wok >= 5
				}
		activate_struggle_catalyst = catalyst_massive_influence_council
		}
		if = {
			limit = {
				exists = global_var:influence_council
				global_var:influence_wok <= 0
				}
		activate_struggle_catalyst = catalyst_no_influence_council
		}
		if = {
			limit = {
				exists = global_var:influence_silsila
				global_var:influence_wok >= 5
				}
		activate_struggle_catalyst = catalyst_massive_influence_silsila
		}
		if = {
			limit = {
				exists = global_var:influence_silsila
				global_var:influence_wok <= 0
				}
		activate_struggle_catalyst = catalyst_no_influence_silsila
		}
		if = {
			limit = {
				exists = global_var:influence_losttribe
				global_var:influence_wok >= 5
				}
		activate_struggle_catalyst = catalyst_massive_influence_losttribe
		}
		if = {
			limit = {
				exists = global_var:influence_losttribe
				global_var:influence_wok <= 0
				}
		activate_struggle_catalyst = catalyst_no_influence_losttribe
		}

		#Simulate the AI doing stuff

		random_list = {
			40 = { }
			
			10 = { 
				if = {
					limit = { exists = global_var:influence_wok }
					change_global_variable = {
						name = influence_wok
						add = 1
					}
				}
			}
			
			10 = { 
				if = {
					limit = { exists = global_var:influence_wok }
					change_global_variable = {
						name = influence_wok
						add = -1
					}
				}
			}			

			20 = { 
				if = {
					limit = { exists = global_var:influence_ashirra }
					change_global_variable = {
						name = influence_ashirra
						add = 1
					}
				}
			}
			
			5 = { 
				if = {
					limit = { exists = global_var:influence_ashirra }
					change_global_variable = {
						name = influence_ashirra
						add = -1
					}
				}
			}
			
			10 = { 
				if = {
					limit = { exists = global_var:influence_council }
					change_global_variable = {
						name = influence_council
						add = 1
					}
				}
			}
			
			10 = { 
				if = {
					limit = { exists = global_var:influence_council }
					change_global_variable = {
						name = influence_council
						add = -1
					}
				}
			}			

			5 = { 
				if = {
					limit = { exists = global_var:influence_silsila }
					change_global_variable = {
						name = influence_silsila
						add = 1
					}
				}
			}
			
			10 = { 
				if = {
					limit = { exists = global_var:influence_silsila }
					change_global_variable = {
						name = influence_silsila
						add = -1
					}
				}
			}
			
			5 = { 
				if = {
					limit = { exists = global_var:influence_losttribe }
					change_global_variable = {
						name = influence_losttribe
						add = 1
					}
				}
			}
			
			15 = { 
				if = {
					limit = { exists = global_var:influence_losttribe }
					change_global_variable = {
						name = influence_losttribe
						add = -1
					}
				}
			}
		}
		# Then queue the event up against for a year hence.
		trigger_event = {
			id = pod_assamite_struggle.0001
			years = 1
		}
			
	}
}

	hidden = yes
	type = empty
	scope = struggle

				}

		}		
	}
}

pod_assamite_struggle.100 = {
	type = character_event
	title = pod_assamite_struggle.100.t
	desc = pod_assamite_struggle.100.desc
	
	theme = realm
	left_portrait = {
		character = root
		animation = war_over_win
	}

	trigger = {
	}

	immediate = {
		save_scope_as = iberian_footholder
		every_held_title = {
			limit = {
				# Any de jure kingdom of Iberia is completely controlled
				struggle_secure_assamite_foothold_assamite_kingdom_trigger = yes
				scope:iberian_kingdom_temp = { title_held_years = fp2_struggle_secure_iberian_foothold_years_value }
			}
			add_to_list = iberian_footholder_list
		}
		hidden_effect  = {
			while = {
				limit = {
					any_in_list = {
						list = iberian_footholder_list
						# Borders held lands in a non-Iberian kingdom you hold
						any_de_jure_county = {
							any_title_to_title_neighboring_and_across_water_county = { struggle_secure_assamite_foothold_outsider_kingdom_trigger = yes }
						}
						NOT = { is_in_list = iberian_footholder_transfer_list }
					}
				}
				random_in_list = {
					list = iberian_footholder_list
					limit = {
						# Borders held lands in a non-Iberian kingdom you hold
						any_de_jure_county = {
							any_title_to_title_neighboring_and_across_water_county = { struggle_secure_assamite_foothold_outsider_kingdom_trigger = yes }
						}
						NOT = { is_in_list = iberian_footholder_transfer_list }
					}
					random_de_jure_county = {
						limit = {
							any_title_to_title_neighboring_and_across_water_county = { struggle_secure_assamite_foothold_outsider_kingdom_trigger = yes }
						}
						random_title_to_title_neighboring_and_across_water_county = {
							limit = { struggle_secure_assamite_foothold_outsider_kingdom_trigger = yes }
							empire = { save_scope_as = transfer_empire }
						}
					}
					add_to_list = iberian_footholder_transfer_list
					set_de_jure_liege_title = scope:transfer_empire
					custom_tooltip = iberian_footholder_transfer_tt
				}
			}
		}
		show_as_tooltip = {
			every_in_list = {
				list = iberian_footholder_transfer_list
				set_de_jure_liege_title = scope:transfer_empire
			}
		}
	}

	option = {
		name = pod_assamite_struggle.100.a
	}

	after = {
		struggle:assamite_struggle = {
			every_involved_ruler = {
				limit = { is_ai = no }
				trigger_event = pod_assamite_struggle.101
			}
		}
		# End Struggle
		struggle:assamite_struggle = {
			end_struggle = secure_assamite_foothold_decision_title
		}
	}
}
