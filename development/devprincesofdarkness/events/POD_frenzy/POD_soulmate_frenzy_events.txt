#First frenzy events, all other events are tied to this one
POD_soulmate_frenzy.001 = {
    type = character_event
	title = POD_soulmate_frenzy.001.t
	desc = POD_soulmate_frenzy.001.desc
	theme = embrace 
	
	left_portrait = {
	    character = scope:frenziedvamp
		animation = grief 
	}
	
	immediate = {
	    save_scope_as = frenziedvamp
		scope:dead_character = {
		    save_scope_as = lost_soulmate
		}
	}
	
	trigger = {
	    is_vampire_trigger = yes
		NOT = { has_character_flag = immune_to_frenzy}
	}
	
	option = {
	    name = POD_soulmate_frenzy.001.option.a
		custom_tooltip = POD_soulmate_frenzy.001.option.a.tt
		
		hidden_effect ={
            random_list = {
			    3 = {
				    trigger_event = {
                        id = POD_soulmate_frenzy.002
						days = 2
					}					
				}
				7 = {
				    trigger_event = {
                        id = POD_soulmate_frenzy.004
						days = 2
					} 
					
					modifier = {
					   add = 6
					   has_trait = brujah
					}
					modifier = {
					    add = 2
						piety_level < 0
					}
					modifier = {
					    add = 2
						has_character_modifier = bloodhunger5_modifier
					}
					modifier = {
					    add = 1
						has_character_modifier = bloodhunger4_modifier
					}
				}
			}
		}
	}
	option = {
	    name = POD_soulmate_frenzy.001.option.b
		custom_tooltip = POD_soulmate_frenzy.001.option.b.tt
		
		hidden_effect = {
		    random_list = {
			    4 = {
				    trigger_event = {
                        id = POD_soulmate_frenzy.003
						days = 2
					}
  					
					modifier = {
					    add = 2
						has_trait = celerity
					}
					modifier = {
					    add = 4
						has_trait = celerityadvanced
					}
				}
				6 = {
				    trigger_event = {
                        id = POD_soulmate_frenzy.004
						days = 2
						
					}

					modifier = {
					   add = 6
					   has_trait = brujah
					}
					modifier = {
					    add = 2
						has_character_modifier = bloodhunger5_modifier
					}
					modifier = {
					    add = 1
						has_character_modifier = bloodhunger4_modifier
					}
				}
			}
		}
	}
	
	option = {
	    name = POD_soulmate_frenzy.001.option.c
		custom_tooltip = POD_soulmate_frenzy.001.option.c.tt
		
		hidden_effect = {
		    trigger_event = {
			    id = POD_soulmate_frenzy.004
				days = 2
			}
		}
	}
}

POD_soulmate_frenzy.103 = {
    type = character_event

	    character = scope:frenziedvamp
		animation = rage
	}
	
	option = {
	    name = POD_soulmate_frenzy.103.option.a
		custom_tooltip = POD_soulmate_frenzy.103.option.a.tt

		    limit = {
			    has_trait = wounded_1

		add_piety = major_piety_loss

		    capital_county = {
			    add_county_modifier = {
				    modifier = POD_frenzy_terrified_populace_modifier
					years = 20
				}
			}
		}
		hidden_effect = {
		    trigger_event = {
			    id = POD_soulmate_frenzy.203
				days = 2
			}
		}
	}
}

POD_soulmate_frenzy.303 = {
    type = character_event
	title = POD_soulmate_frenzy.303.t
	desc = POD_soulmate_frenzy.303.desc
	theme = embrace
	
	left_portrait = {
	    character = scope:frenziedvamp 
		animation = rage
	}
	
	immediate = {
	    if = {
		    limit = {
			    any_vassal_or_below = {
				    OR = {
				        has_relation_friend = scope:frenziedvamp
						has_relation_lover = scope:frenziedvamp
				    }
			    }
		    }
		    random_vassal_or_below = {
				limit = {
				    OR = {
					    has_relation_friend = scope:frenziedvamp
					    has_relation_lover = scope:frenziedvamp
					}
				}
				save_scope_as = friend_savior
		    }
	    }
	}
	option = {
	    name = POD_soulmate_frenzy.303.option.a
		custom_tooltip = POD_soulmate_frenzy.303.option.a.tt
		
		
		duel = {
		    skill = prowess
		    value = 70
		    desc = POD_soulmate_frenzy.303.option.a.desc
			
		    50 = {
		        desc = POD_soulmate_frenzy.303.option.a.success
				
			    compare_modifier = {
				    value = scope:duel_value
				    multiplier = 1.5
			    }
					
				custom_tooltip = POD_soulmate_frenzy.303.option.a.tt.win
				
			    hidden_effect = {	
					trigger_event = {
					    id = POD_soulmate_frenzy.403
					    days = 2
				    }
			    }
		    }	
		  
		    50 = {
		        desc = POD_soulmate_frenzy.303.option.a.failure

			    compare_modifier = {
				    value = scope:duel_value
				    multiplier = -1.5
			    }
					
				custom_tooltip = POD_soulmate_frenzy.303.option.a.tt.lose

			    hidden_effect = {
				    if = {
					    limit = {
						    exists = scope:friend_savior
						}
				        trigger_event = {
						    id = POD_soulmate_frenzy.603
						    days = 2
					    }
			        }
					else = {
					    trigger_event = {
						    id = POD_soulmate_frenzy.703
						    days = 2
					    }  
					}
				}
		    }
	    }
	}
}

	    character = scope:frenziedvamp

	}
	
	immediate = {
	    hidden_effect = {
		    every_courtier_or_guest = {
			    random_list = {
				    1 = {
					    death = {
						    death_reason = death_murder
							killer = scope:frenziedvamp

						    limit = {

								    NOT = { has_trait = murderer }
								}
							}
							scope:frenziedvamp = {
							    add_trait = murderer
							}
						}
						
						
						modifier = {
						    add = 15
							prowess < scope:frenziedvamp.prowess
						}
					}
				     
					9 = {
					    add_opinion = {
						    modifier = liege_attempting_to_murder_me_opinion
							target = scope:frenziedvamp
						}
						
						modifier = {
						    add = 5
						    prowess > scope:frenziedvamp.prowess
						}
						modifier = {
						    add = 10
							OR = {
							    has_trait = beatific
								has_trait = truefaith
							}
						}
					}
				}
			}
		}
	}
	
	option = {
	    name = POD_soulmate_frenzy.104.option.a
		custom_tooltip = POD_soulmate_frenzy.104.option.a.tt
		
		add_piety = major_piety_loss

		hidden_effect = {
		    trigger_event = {
