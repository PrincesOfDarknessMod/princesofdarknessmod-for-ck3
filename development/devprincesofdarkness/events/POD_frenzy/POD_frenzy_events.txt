POD_vampire_frenzy_events.201 = {
	title = POD_vampire_frenzy_events.201.t
	desc = POD_vampire_frenzy_events.201.desc
	type = character_event
	theme = masquerade
	right_portrait = {
		character = root
		animation = ecstasy
	}
	left_portrait = {
		character = scope:mortal
		animation = pain
	}
	
	immediate = {
		hidden_effect = {
			if = {
				limit = {any_courtier_or_guest =	{
							has_trait = mortal
						}	
				}
			
				random_courtier_or_guest = {
					limit = {
						has_trait = mortal
					}
					save_scope_as = mortal
				}
			}
			else = {
				create_character = {
					location = root.capital_province
					template = POD_mortal_hunter_template
					save_scope_as = mortal
				}
				scope:mortal = {
					add_trait = "mortal"
					set_to_lowborn = yes
					add_character_flag = {
						flag = blocked_from_leaving
						years = 2
					}
					add_trait = POD_knows_too_much
					save_scope_as = victim
				}
				add_courtier = scope:mortal
			
			}
		}
		
		scope:mortal = {save_scope_as = victim}
		hidden_effect = {
			scope:victim = {
				#POD_clear_resonance_effect = yes
				POD_set_resonance_effect = yes
			}
		}
		POD_hunger_feeding_drain_effect = yes
		

		
		
	}
	#let them die
	option = {
		name = POD_vampire_frenzy_events.201.kill
		custom_tooltip = POD_vampire_frenzy_events.201.kill.beast
		POD_resonance_feeding_drain_effect = yes
		scope:mortal = { 
			death = { death_reason = death_mysterious }
		}
		
		if = {
			limit = {
				OR = {
					faith = {
						has_doctrine_parameter = vampirism_illegal
					}
					faith = {
						has_doctrine_parameter = vampirism_shunned
					}
					faith = { has_doctrine = tenet_conscience_and_selfcontrol }
					faith = { has_doctrine = tenet_conscience_and_instinct }
				}
			}
			add_piety = massive_piety_loss
		}
		else = {
			add_piety = major_piety_loss
		
		}
		stress_impact = {
			gluttonous = major_stress_impact_loss
			compassionate = major_stress_impact_gain
		}
		POD_moderate_risk_of_increasing_masquerade_exposure = yes
		scope:mortal = {
			every_close_or_extended_family_member = {
				limit = {
					NOT = { root = this }
				}
				add_opinion = {
					modifier = murdered_close_relation_opinion
					target = root
				}
			}
		}
		
	}
	#guilty-embrace them
	option = {
		name = POD_vampire_frenzy_events.201.embrace
		custom_tooltip = POD_vampire_frenzy_events.201.embrace.beast
		POD_resonance_feeding_drain_effect = yes
		POD_hunger_rouse_check_effect = yes
		trigger ={
			NOT = {is_ai = yes}
		}
		root = {
			save_scope_as = vampiresire 
		}
		scope:mortal = {
			save_scope_as = created_vampirechilde
		}
		if = {
			limit = {
				OR = {
					faith = {
						has_doctrine_parameter = vampirism_illegal
					}
					faith = {
						has_doctrine_parameter = vampirism_shunned
					}
					faith = { has_doctrine = tenet_conscience_and_selfcontrol }
					faith = { has_doctrine = tenet_conscience_and_instinct }
				}
			}
			add_piety = medium_piety_loss
		}
		else = {
			add_piety = minor_piety_value
		
		}
		stress_impact = {
			gluttonous = minor_stress_impact_gain
			compassionate = major_stress_impact_loss
		}
		hidden_effect = {
			POD_player_character_create_vampirechilde_effect = yes
		}
	}
	

}

POD_vampire_frenzy_events.301 = {
	type = character_event
	title = POD_vampire_frenzy_events.301.t
	desc = POD_vampire_frenzy_events.301.desc
	theme = intrigue
	right_portrait = {
		character = root
		animation = shame
	}
	left_portrait = {
		character = scope:bonder
		animation = schadenfreude
	
	}
	immediate = {
		if = {
			limit = {
				any_vassal_or_below = {
					is_vampire_trigger = yes
					NOT = { has_character_modifier = bloodhunger5_modifier }
				}
			}
			random_vassal_or_below = {
				limit = {
					is_vampire_trigger = yes
					NOT = { has_character_modifier = bloodhunger5_modifier }
				}
				save_scope_as = bonder
			}
		}
		else = {
			random_courtier_or_guest = {
				limit = {
					is_vampire_trigger = yes
					NOT = { has_character_modifier = bloodhunger5_modifier }
				}
				save_scope_as = bonder
			}
		}
		hidden_effect ={
			scope:bonder = {save_scope_as = victim}
			scope:victim = {
				#POD_clear_resonance_effect = yes
				POD_set_resonance_effect = yes
			}
		}
		scope:bonder = {
			POD_hunger_rouse_check_effect = yes
			#POD_hunger_rouse_check_effect = yes
		}
		
	}
	option = {
		name = POD_vampire_frenzy_events.301.a
		custom_tooltip = POD_vampire_frenzy_events.301.a.beast
		POD_hunger_feeding_drink_effect = yes
		POD_resonance_feeding_drink_effect = yes
		if = {
			limit = {
				NOT = { 
					has_trait = predatortype_bloodleech
				}
			}
			add_secret = {
				type = secret_predatortype_bloodleech
			}		
		}
		#give_predatortype_bloodleech_secret_or_trait_effect = yes
		random_secret = {
			limit = { secret_type = secret_predatortype_bloodleech }
			reveal_to = scope:bonder
		}
		if = {
			limit = {
				scope:bonder = {
					has_hook_of_type = {
						target = root
						type = weak_bloodoath_hook
					}
					NOT = {has_character_flag = blood_bonded }
				}
			}
			scope:bonder = {
				add_hook = {
					target = root
					type = strong_bloodoath_hook
				}	
			}
			scope:root = {
				add_character_flag = blood_bonded
			}
		}
		else = {
			scope:bonder = {
				add_hook = {
					target = root
					type = weak_bloodoath_hook
				}	
			}
		}
		stress_impact = {
			gluttonous = medium_stress_impact_loss
			paranoid = major_stress_impact_gain
			trusting = major_stress_impact_loss
		}
		
		
	}

}
