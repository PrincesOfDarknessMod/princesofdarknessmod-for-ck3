POD_anteshuntabsi.3 = {
	type = character_event
	title = POD_anteshuntabsi.3.t
	desc = POD_anteshuntabsi.3.desc
	theme = nosferatu_coa
	left_portrait = {
		character = root
		animation = worry
	}
	
	right_portrait = {
		character = scope:target
		animation = fear
	}	

	immediate = {
		if = {
			limit = {any_vassal_or_below = { always = yes}}
			random_vassal = {
				limit = {
					NOT = {has_trait = torpor}
					NOT = {has_trait = incapable}
				}
				save_scope_as = target
			}
		}
	}
	
	option = {
		name = POD_anteshuntabsi.3.a
		
		custom_tooltip = anteabsi_nictuku
		
		add_character_flag = hunted_by_nictuku

		scope:target = {
			death = { death_reason = death_murder }
		}
		
		trigger_event = {
				id = POD_anteshuntabsi.5
				days = 30 
				}		
		}	
					
}

		}

		limit = {
			NOT = { has_character_flag = dealt_with_echidna }

		limit = {
			NOT = { has_character_flag = dealt_with_gorgo }

		trigger_event = {
				id = POD_anteshuntabsi.15
				days = 90
				}		
		}	

	#Nuckalavee

		limit = {

			}		
		trigger_event = {
				id = POD_anteshuntabsi.16

				}		
		}	

	#Gayomart
	if = {
		limit = {
			NOT = { has_character_flag = dealt_with_gayomart }
			}		
		trigger_event = {
				id = POD_anteshuntabsi.17
				days = 120
				}		
		}	

	#Ending	
		trigger_event = {
				id = POD_anteshuntabsi.18
				days = 135
				}		
		}
