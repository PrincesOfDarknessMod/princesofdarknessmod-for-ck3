POD_umbra_incarna.0100 = {
	type = character_event
	title = POD_umbra_incarna.0100.t
	desc = POD_umbra_incarna.0100.desc
	
	left_portrait = root
	theme = spirit_world
	
	trigger = {
		has_variable = spirit_location
		var:spirit_location = flag:court_of_eshtarra
	}

	option = {
		name = POD_umbra_incarna.0100.a
		trigger = {
			has_trait = werewolf
			NOT = { has_perk = earth_heal_perk }
			piety >= 750
		}
		show_as_unavailable = {
			piety < 750
		}
		
		add_piety = -750
		if = {
			limit = { NOT = { has_perk = earth_sense_perk } }
			add_perk = earth_sense_perk
		}
		else_if = {
			limit = { NOT = { has_perk = herb_call_perk } }
			add_perk = herb_call_perk
		}
		else_if = {
			limit = { NOT = { has_perk = oaksong_perk } }
			add_perk = oaksong_perk
		}
		else_if = {
			limit = { NOT = { has_perk = lore_of_the_land_perk } }
			add_perk = lore_of_the_land_perk
		}
		else = {
			add_perk = earth_heal_perk
		}
	}

	option = {
		name = POD_umbra_incarna.0100.f
	}
}
