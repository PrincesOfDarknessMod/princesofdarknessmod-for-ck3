POD_spirit_travel.10 = {
	type = character_event
	hidden = yes

	trigger = {
		has_variable = spirit_destination
	}
	
	immediate = {
		send_interface_message = {
			type = character_arrived
			title = spirit_travel_arrived_title
			desc = spirit_travel_arrived_desc
			right_icon = root
			
			set_variable = { name = spirit_location value = var:spirit_destination }
			if = {
				limit = {
					var:spirit_location = flag:skinlands
				}
				remove_character_modifier = traveling_spirit_world_modifier
			}
			else_if = {
				limit = {
					NOT = { has_character_modifier = traveling_spirit_world_modifier }
				}
				add_character_modifier = traveling_spirit_world_modifier
				if = {
					limit = { 
						has_POD_shapeshifted_form_trigger = yes 
						NOT = { has_perk = aspect_of_duat_perk }
					}
					POD_shapeshift_revert_effect = yes
				}
				if = {
					limit = { has_trait = kueijin  }
					if = {
						limit = { var:spirit_location = flag:shadowlands }
						if = {
							limit = { NOT = { has_perk = semblance_of_the_ebon_dragon_perk } }
							send_interface_toast = {
								type = event_generic_bad
								title = paying_for_spirit_travel_toast
								left_icon = root
								custom_tooltip = paying_for_spirit_travel_notification_yin

								remove_short_term_gold = medium_gold_value
							}
						}
					}
					else = {
						if = {
							limit = { NOT = { has_perk = semblance_of_the_scalet_queen_perk } }
							send_interface_toast = {
								type = event_generic_bad
								title = paying_for_spirit_travel_toast
								left_icon = root
								custom_tooltip = paying_for_spirit_travel_notification_yang

								remove_short_term_gold = medium_gold_value
							}
						}
					}
				}
				trigger_event = POD_spirit_travel.1000
			}
			POD_set_local_spirit_list_effect = yes
			if = {
				limit = { 
					var:spirit_location = flag:shadowlands
					NOT = { has_variable_list = spirit_list }
				}
				POD_create_random_local_spirit = yes
				POD_create_random_local_spirit = yes
				POD_create_random_local_spirit = yes
			}
			if = {
				limit = { var:spirit_destination = flag:oblivion }
				trigger_event = POD_spirit_travel.102
			}
			remove_variable = spirit_destination
		}
	}
}
