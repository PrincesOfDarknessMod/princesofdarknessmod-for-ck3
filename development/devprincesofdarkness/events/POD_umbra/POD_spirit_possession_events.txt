namespace = POD_spirit_possession

POD_spirit_possession.1000 = {
	type = character_event
	hidden = yes
	
	trigger = {
		NOT = { game_start_date = 1230.1.4 } #The possessed trees and mechanics are too messy for the first bookmark
		has_variable = spirit_location
		var:spirit_location = flag:wyrm_reaches
		POD_is_possessed_trigger = no
	}

	immediate = {
		duel = {
			skill = learning
			value = medium_skill_rating
		
			25 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = 0.25
				}
				trigger_event = POD_spirit_possession.1001
			}
			25 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = -0.25
				}
			}
		}
	}
}

POD_spirit_possession.1001 = {
	type = character_event
	title = POD_spirit_possession.1001.t
	desc = POD_spirit_possession.1001.desc
	
	left_portrait = root
	theme = spirit_world

	option = {
		name = POD_spirit_possession.1001.a
		
		trigger = {
			NOT = { has_trait = wraith }
			NOT = { has_trait = demon }
		}
		
		show_as_unavailable = { 
			OR = { 
				has_trait = wraith
				has_trait = demon
			}
		}
		
		duel = {
			skill = intrigue
			value = medium_skill_rating
		
			25 = {
				desc = POD_spirit_travel.1100.success
				compare_modifier = {
					value = scope:duel_value
					multiplier = 0.5
				}
				modifier = {
					factor = 0.75
					num_sinful_traits > 0
				}
				modifier = {
					factor = 0.75
					num_sinful_traits > 1
				}
				modifier = {
					factor = 0.75
					num_sinful_traits > 2
				}
				modifier = {
					factor = 1.25
					num_virtuous_traits > 0
				}
				modifier = {
					factor = 1.25
					num_virtuous_traits > 1
				}
				modifier = {
					factor = 1.25
					num_virtuous_traits > 2
				}
				add_trait = fomori
				if = {
					limit = { POD_is_wrym_servant_trigger = no }
					POD_become_wrym_servant_effect = yes
					add_piety = massive_piety_loss
				}
			}
			25 = {
				desc = POD_spirit_travel.1100.fail
				compare_modifier = {
					value = scope:duel_value
					multiplier = -0.5
				}
				add_stress = medium_stress_gain
				add_piety = medium_piety_loss
			}
		}
	}

	option = {
		name = POD_spirit_possession.1001.b
		add_stress = minor_stress_gain
		add_piety = minor_piety_loss
	}
	
	after = {
		trigger_event = POD_spirit_travel.1000
	}
}
