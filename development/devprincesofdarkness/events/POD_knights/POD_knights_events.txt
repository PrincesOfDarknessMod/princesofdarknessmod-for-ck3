namespace = pod_knights

# Some events depending on your knight traits

pod_knights.1 = {
	type = character_event
	title = pod_knights.1.t
	desc = pod_knights.1.desc
	theme = masquerade
	left_portrait = {
		character = root
		animation = dismissal
	}
	right_portrait = {
		character = scope:knight
		animation = scheme
	}	

	trigger = {
		is_vampire_trigger = yes
		is_landed = yes
		is_imprisoned = no
		NOT = {
			has_trait = torpor
			}
		any_knight = {
			is_landed = no
			has_trait = arrogant
			is_vampire_trigger = yes
			NOT = { is_child_of = root }
			}
	}

	immediate = {
		random_knight = {
		limit = {
			is_landed = no
			has_trait = arrogant
			is_vampire_trigger = yes
			NOT = { is_child_of = root }
			}
		save_scope_as = knight	
			}
	}

	option = { 
		name = pod_knights.1.a
		
		scope:knight = { 
			select_and_move_to_pool_effect = yes
				}
 
	}
	option = { 
		name = pod_knights.1.b
		
		POD_masquerade_exposure_increase_effect = yes
	}
}

pod_knights.3 = {
	type = character_event
	title = pod_knights.3.t
	desc = pod_knights.3.desc
	theme = masquerade
	left_portrait = {
		character = root
		animation = stress
	}
	right_portrait = {
		character = scope:knight
		animation = rage
	}	

	trigger = {
		is_vampire_trigger = yes
		is_landed = yes
		is_imprisoned = no
		is_ai = no
		NOT = {
			has_trait = torpor
			}
		any_knight = {
			is_landed = no
			has_trait = wrathful
			is_vampire_trigger = yes
			prowess > 50
			NOT = { is_child_of = root }
			}
	}

	immediate = {
		random_knight = {
		limit = {
			is_landed = no
			has_trait = wrathful
			is_vampire_trigger = yes
			prowess > 40
			NOT = { is_child_of = root }
			}
		save_scope_as = knight	
			}
		if = {
			limit = {any_vassal_or_below = { always = yes}}
			random_vassal = {
				limit = {
					NOT = {has_trait = torpor}
					NOT = {has_trait = incapable}
					NOT = {prowess > 40}
				}
				save_scope_as = target
			}
		}
		else = {
			hidden_effect = {
				create_character = {
						location = root.capital_province
						template = POD_mortal_hunter_template
						save_scope_as = mortal
					}
					scope:mortal = {
						add_trait = "mortal"
						set_to_lowborn = yes
						add_character_flag = {
							flag = blocked_from_leaving
							years = 2
						}
						add_trait = POD_knows_too_much
						save_scope_as = target
					}
					add_courtier = scope:mortal
			}
		}			
	}

	option = { 
		name = pod_knights.3.a
		
		scope:knight = { 
			select_and_move_to_pool_effect = yes
				}
 
	}
	option = { 
		name = pod_knights.3.b
	
		unknown_murder_effect = {
			VICTIM = scope:target
			MURDERER = scope:knight
			REASON = death_execution
		}
	}
}
