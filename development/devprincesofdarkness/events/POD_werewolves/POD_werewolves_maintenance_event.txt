namespace = POD_werewolves_maintenance

POD_werewolves_maintenance.1 = {
	type = character_event
	hidden = yes
	
	trigger = {
		is_adult = no
		has_POD_race_trigger = no
		NOT = { has_character_flag = had_POD_character_initialization }
	}
	
    immediate = {

	if = {
		limit = {
			has_POD_race_trigger = no
			NOT = { has_character_flag = had_POD_character_initialization }
			OR = {
				AND = {
					exists = mother
					mother = { 
						OR = {
							has_trait = kinfolk
							has_trait = werewolf
						}
					}
				}
				AND = {
					exists = father
					father = { 
						OR = {
							has_trait = kinfolk
							has_trait = werewolf
						}
					}
				}
				AND = {
					exists = real_father
					real_father = { 
						OR = {
							has_trait = kinfolk
							has_trait = werewolf
						}
					}
				}
			}
		}
		
		add_trait = kinfolk
			
		add_character_flag = had_POD_character_initialization
			
			
			random = {
				chance = 25
				
				modifier = {
					add = 25
					AND = {
						exists = mother
						mother = { 
							has_trait = werewolf
							}
					}
				}
				modifier = {
					add = 25
					AND = {
						exists = real_father
						real_father = { 
							has_trait = werewolf
						}
					}
				}
				
				trigger_event = { 			
				id = POD_werewolves_maintenance.2
				years = { 13 18 }
				}
			}				
		}			
	}
}

POD_werewolves_maintenance.2 = {
	type = character_event
	title = POD_werewolves_maintenance.2.t
	desc = POD_werewolves_maintenance.2.desc

	theme = faith
	right_portrait = root
	
	trigger = {
		is_vampire_trigger = no
	}
	
	option = {
		name = POD_werewolves_maintenance.2.option.a	
		
			if = {
				limit = {
					has_trait = kinfolk 
				}
				remove_trait = kinfolk
				}
				
			add_trait = werewolf
			add_trait = homid_born	

		random_list = {
			20 = {
				add_trait = ragabash
			}
			20 = {
				add_trait = ahroun
			}
			20 = {
				add_trait = philodox
			}
			20 = {
				add_trait = theurge
			}
			20 = {
				add_trait = galliard
			}			
		}		
	}
}

POD_werewolves_maintenance.3 = {
	type = character_event
	title = POD_werewolves_maintenance.3.t
	desc = POD_werewolves_maintenance.3.desc

	theme = faith
	right_portrait = root
	
	trigger = {
		has_trait = werewolf
	}
	
	immediate = {
		hidden_effect = {
			if = {
				limit = {
					exists = capital_province
				}
				capital_province = { save_scope_as = target_location }
			}
			else = {
				location = { save_scope_as = target_location }
			}
			create_character = {
				location = scope:target_location
				template = lupus_maintenance_character
				save_scope_as = lupus_1
			}
			add_visiting_courtier = scope:lupus_1
		}
		
		random = {			
			chance = 20
			modifier = {
				add = 50
				has_faith = faith:redtalon
			}
			
		hidden_effect = {
			if = {
				limit = {
					exists = capital_province
				}
				capital_province = { save_scope_as = target_location }
			}
			else = {
				location = { save_scope_as = target_location }
			}
			create_character = {
				location = scope:target_location
				template = lupus_maintenance_character
				save_scope_as = lupus_2
			}
			add_visiting_courtier = scope:lupus_2
		}						
		}
		
		random = {			
			chance = 10
			modifier = {

				has_faith = faith:redtalon
			}
			
		hidden_effect = {

				limit = {

				}
				capital_province = { save_scope_as = target_location }
			}
			else = {
				location = { save_scope_as = target_location }

				exists = scope:lupus_2
				}
		recruit_courtier = scope:lupus_2
		}

		if = {
			limit = {
				exists = scope:lupus_3
				}
