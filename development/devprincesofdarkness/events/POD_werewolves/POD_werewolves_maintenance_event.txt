namespace = POD_werewolves_maintenance

POD_werewolves_maintenance.1 = {
	type = character_event
	hidden = yes
	
	trigger = {
		is_adult = no
		has_POD_race_trigger = no
		NOT = { has_character_flag = had_POD_character_initialization }
	}
	
    immediate = {

	if = {
		limit = {
			has_POD_race_trigger = no
			NOT = { has_character_flag = had_POD_character_initialization }
			OR = {
				AND = {
					exists = mother
					mother = { 
						OR = {
							has_trait = kinfolk
							has_trait = werewolf
						}
					}
				}
				AND = {
					exists = father
					father = { 
						OR = {
							has_trait = kinfolk
							has_trait = werewolf
						}
					}
				}
				AND = {
					exists = real_father
					real_father = { 
						OR = {
							has_trait = kinfolk
							has_trait = werewolf
						}
					}
				}
			}
		}
		
		add_trait = kinfolk
			
		add_character_flag = had_POD_character_initialization
			
			
			random = {
				chance = 25
				
				modifier = {
					add = 25
					AND = {
						exists = mother
						mother = { 
							has_trait = werewolf
							}
					}
				}
				modifier = {
					add = 25
					AND = {
						exists = real_father
						real_father = { 
							has_trait = werewolf
						}
					}
				}
				
				trigger_event = { 			
				id = POD_werewolves_maintenance.2
				years = { 13 18 }
				}
			}				
		}			
	}
}
