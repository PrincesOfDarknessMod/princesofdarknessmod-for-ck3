POD_werewolves_maintenance.2 = {
	type = character_event
	title = POD_werewolves_maintenance.2.t
	desc = POD_werewolves_maintenance.2.desc

	theme = faith
	left_portrait = {
		character = root
		animation = shock
	}
	
	trigger = {
		has_trait = kinfolk 
	}
	
	option = {
		name = POD_werewolves_maintenance.2.option.a	
		
				
			if = {
				limit = {
					has_trait = kinfolk 
				}
				remove_trait = kinfolk
				}
				
			if = {
				limit = {
					has_trait = psychic 
				}
				remove_trait = psychic
				refund_perks = psi_lifestyle	
				add_psi_lifestyle_perk_points = -100
				}
			if = {
				limit = {
					OR = {
						has_trait = sorcerer 
						has_trait = theurgist 
					}
				}
				remove_trait = sorcerer
				remove_trait = theurgist
				refund_perks = sorcery_lifestyle
				add_sorcery_lifestyle_perk_points = -100
				}
				
			add_trait = werewolf
			add_trait = homid_born	

		random_list = {
			20 = {
				add_trait = ragabash
			}
			20 = {
				add_trait = ahroun
			}
			20 = {
				add_trait = philodox
			}
			20 = {
				add_trait = theurge
			}
			20 = {
				add_trait = galliard
			}			
		}		
	}
}

# Werewolf Healing

POD_werewolves_maintenance.6 = {
	hidden = yes
	
	trigger = {
		POD_is_werewolf_trigger = yes
		OR = {
			has_trait = wounded_1
			has_trait = wounded_2
			has_trait = wounded_3
			has_trait = one_legged
			has_trait = maimed
			AND = {
				has_trait = eunuch	
				NOT = { has_character_flag = POD_permanent_eunuch_flag }
			}
			AND = {
				has_trait = one_eyed
				NOT = { has_character_flag = POD_permanent_one_eyed }
			}
			has_trait = blind
			has_trait = disfigured
		}	
	}

	weight_multiplier = {
		base = 1
		modifier = {
			add = 1000
			POD_is_werewolf_trigger = yes
		}			
	}

	immediate = {
		if = {
			limit = { has_character_modifier = infected_wound_modifier }
			remove_character_modifier = infected_wound_modifier

			if = {
				limit = { NOT = { has_trait = scarred } }
				add_character_flag = will_get_scar
			}
		}
		else_if = {
			limit = { NOT = { has_trait = scarred } }
			random = {
				chance = 10
				add_character_flag = will_get_scar
			}
		}
		send_interface_toast = {
			title = health.0001.desc

			if = {
				limit = {
					has_trait = wounded_1
				}
				remove_trait = wounded_1
			}

			if = {
				limit = {
					has_trait = wounded_2
				}
				remove_trait = wounded_2
			}
			
			if = {
				limit = {
					has_trait = wounded_3
				}
				remove_trait = wounded_3
			}			

			if = {
				limit = {
					has_trait = maimed
				}
				remove_trait = maimed
			}	

			if = {
				limit = {
					has_trait = one_legged
				}
				remove_trait = one_legged
			}

			if = {
				limit = {
					has_trait = eunuch			
					NOT = { has_character_flag = POD_permanent_eunuch_flag }
				}
				remove_trait = eunuch
			}

			if = {
				limit = {
					has_trait = one_eyed
					NOT = { has_character_flag = POD_permanent_one_eyed }
				}
				remove_trait = one_eyed
			}
			
			if = {
				limit = {
					has_trait = blind
				}
				remove_trait = blind
			}
			if = {
				limit = {
					has_trait = disfigured
				}
				remove_trait = disfigured
			}
			if = {
				limit = { has_character_flag = will_get_scar }
				add_trait = scarred
				remove_character_flag = will_get_scar
			}
		}
	}
}

	hidden = yes

		OR = {
			has_trait = ill
