					name = disaster_aggraved_flag

							}
						}

							limit = { exists = local_var:disaster_earthquake_flag }

								modifier = weather_magic_earthquake_modifier
								years = 5
							}
						}
						if = {
							limit = { exists = local_var:disaster_flood_flag }

								modifier = weather_magic_flood_modifier
								years = 5
							}
						}

						limit = {
							is_ai = yes
						}

							limit = { has_any_nickname = no }

					}
				}
			}
		}

				OR = {
					has_character_flag = goto_stop_the_disaster

						scope:target_title = {
							is_de_jure_liege_or_above_target = prev
						}
					}

			remove_character_flag = goto_stop_the_disaster
			remove_character_flag = goto_aggrav_the_disaster
		}
