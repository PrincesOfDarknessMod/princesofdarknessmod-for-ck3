namespace = POD_charjoin

POD_charjoin.100 = {           
	type = character_event
	title = POD_charjoin.100.t
	desc = POD_charjoin.100.desc
	theme = vampire
	left_portrait = scope:created_vampirechilde
	right_portrait = scope:vampiresire
	
	# Become Vampire
	option = {
		name = POD_charjoin.100.option.a
		remove_trait = mortal
		if = { 
			limit = {
				has_trait = order_member 
				}
				remove_trait = order_member
				add_character_modifier = holy_order_experience_modifier
		}
		
		add_trait_force_tooltip = vampire
		if = {
			limit = {
				scope:vampiresire = {
					is_female = no
				}
			}
			set_father = scope:vampiresire
		}
		else = {
			set_mother = scope:vampiresire
		}
		POD_character_vampire_inherit_clan_effect = yes
		POD_character_vampire_inherit_generation_effect = yes
		set_house = scope:vampiresire.house

		trigger_event = POD_chargen.610
	}
	
	# Become Ghoul
	option = {
		name = POD_charjoin.100.option.b
		remove_trait = mortal
		add_trait_force_tooltip = ghoul
		
		if = {
			limit = {
				has_masquerade_risk_trigger = no
			}
			add_character_modifier = {
				modifier = masquerade0_modifier 
			}
		}

		if = {
			limit = {
				has_blood_hunger_trigger = no
			}
			add_character_modifier = {
				modifier = bloodhunger0_modifier
			}
		}		

		if = {
			limit = {
				has_blood_potency_trigger =  no
			}
			add_character_modifier = {
				modifier = bloodpotency0_modifier 
			}
		}
		POD_character_ghoul_discipline_initialization_effect = yes
		add_character_flag = blood_bonded 
		
		scope:vampiresire = {
			add_hook = {
				target = scope:created_vampirechilde
				type = strong_bloodoath_hook
			}
			POD_hunger_rouse_check_effect = yes
			POD_high_risk_of_increasing_masquerade_exposure = yes	
			
			if = {
				limit = {
					faith = {has_doctrine_parameter = vampireghouling_illegal }
				}
				add_piety = major_piety_loss
			}
			else_if = {
				limit = {
					faith = {has_doctrine_parameter = vampireghouling_shunned } 
				}
				add_piety = medium_piety_loss
			}
			
			POD_give_regnant_secret_effect = yes
			
			add_to_variable_list = {
				name = ghouls
				target = scope:created_vampirechilde
			}
		}
		
		set_variable = { name = regnant value = scope:vampiresire}
	}

	# Regret
	option = {
		name = POD_charjoin.100.option.c
		custom_tooltip = POD_charjoin.100.option.c.t
	}
}

POD_charjoin.300 = {           
	type = character_event
	title = POD_charjoin.300.t
	desc = POD_charjoin.300.desc
	theme = faith
	left_portrait = scope:hunterpupil
	right_portrait = scope:huntermentor
	
	# Become Hunter
	option = {
		name = POD_charjoin.100.option.a
		
		if = {
			limit = {
				exists = scope:huntermentor.dynasty
				scope:huntermentor.dynasty = { has_dynasty_perk = unity_legacy_4 }
			}
			add_martial_skill = 4
		}
		else = {
			add_martial_skill = 2
		}
		remove_trait = mortal
		if = {
			limit = {
				has_trait = bastard
			}
			remove_trait = bastard
		}
		add_trait_force_tooltip = supehunter
		if = {
			limit = {
				scope:huntermentor = { has_trait = order_member }
				scope:hunterpupil = { NOT = { has_trait = order_member } }
			}
			add_trait = order_member
		}
		set_character_faith = scope:huntermentor.faith
		set_house = scope:huntermentor.house
		if = {
			limit = {
			has_orison_trigger = no
			}
			POD_give_orison_effect = yes
		}
		random = {
			chance = 10
			add_trait = truefaith
		}
	}

	# Regret
	option = {
		name = POD_charjoin.100.option.c
		custom_tooltip = POD_charjoin.100.option.c.t
	}
}

POD_charjoin.310 = {           
	type = character_event
	title = POD_charjoin.310.t
	desc = POD_charjoin.310.desc
	theme = faith
	left_portrait = scope:hunterpupil
	right_portrait = scope:huntermentor
	
	# Become Hunter
	option = {
		name = POD_charjoin.100.option.a
		
		if = {
			limit = {
				is_married = yes
			}
			every_spouse = {
				divorce = scope:hunterpupil
			}
		}
		if = {
			limit = {
				exists = betrothed
			}
			break_betrothal = betrothed
		}
		if = {
			limit = {
				is_concubine = yes
			}
			this.concubinist = {
				remove_concubine = scope:hunterpupil
			}
		}
		if = {
			limit = {
				number_of_concubines > 0
			}
			every_concubine = {
				scope:hunterpupil = {
					remove_concubine = prev
				}
			}
		}
		
		if = {
			limit = {
				exists = scope:huntermentor.dynasty
				scope:huntermentor.dynasty = { has_dynasty_perk = unity_legacy_4 }
			}
			add_martial_skill = 4
		}
		else = {
			add_martial_skill = 2
		}
		remove_trait = mortal
		if = {
			limit = {
				has_trait = bastard
			}
			remove_trait = bastard
		}
		add_trait_force_tooltip = supehunter
		if = {
			limit = {
				scope:huntermentor = { has_trait = order_member }
				scope:hunterpupil = { NOT = { has_trait = order_member } }
			}

			}
			POD_give_orison_effect = yes
		}
		random = {
			chance = 10

		}
	}

	# Regret
	option = {
		name = POD_charjoin.100.option.c
		custom_tooltip = POD_charjoin.100.option.c.t
	}

POD_charjoin.510 = {           
	type = character_event
	title = POD_charjoin.510.t
	desc = POD_charjoin.510.desc
	theme = vampire
	left_portrait = scope:kueijindisciple
	right_portrait = scope:kueijinsifu
	
	# Present
	option = {
		name = POD_charjoin.510.option.a

		if = {
			limit = {
				scope:kueijinsifu = { has_trait = dharma9 }
			}
			scope:kueijindisciple = { 
				pay_short_term_gold = {
					target = scope:kueijinsifu
					gold = 1000
				} 
			}
		}
		else_if = {
			limit = {
				scope:kueijinsifu = { has_trait = dharma8 }
			}
			scope:kueijindisciple = { 
				pay_short_term_gold = {
					target = scope:kueijinsifu
					gold = 500
				} 
			}
		}
		else_if = {
			limit = {
				scope:kueijinsifu = { has_trait = dharma7 }
			}
			scope:kueijindisciple = { 
				pay_short_term_gold = {
					target = scope:kueijinsifu
					gold = 300
				} 
			}
		}
		else_if = {
			limit = {
				scope:kueijinsifu = { has_trait = dharma6 }
			}
			scope:kueijindisciple = { 
				pay_short_term_gold = {
					target = scope:kueijinsifu
					gold = 200
				} 
			}
		}
		else = {
			scope:kueijindisciple = { 
				pay_short_term_gold = {
					target = scope:kueijinsifu
					gold = 100
				} 
			}
		}
		
		trigger_event = POD_charjoin.500
	}

	# Lose Prestige
	option = {
		name = POD_charjoin.510.option.a

		if = {
			limit = {
				scope:kueijinsifu = { has_trait = dharma9 }
			}
			scope:kueijindisciple = { add_prestige = -1000 }
		}
		else_if = {
			limit = {
				scope:kueijinsifu = { has_trait = dharma8 }
			}
			scope:kueijindisciple = { add_prestige = -500 }
		}
		else_if = {
			limit = {
				scope:kueijinsifu = { has_trait = dharma7 }
			}
			scope:kueijindisciple = { add_prestige = -300 }
		}
		else_if = {
			limit = {
				scope:kueijinsifu = { has_trait = dharma6 }
			}
			scope:kueijindisciple = { add_prestige = -200 }
		}
		else = {
			scope:kueijindisciple = { add_prestige = -100 }
		}
		
		trigger_event = POD_charjoin.500
	}

	# Give up
	option = {
		name = POD_charjoin.100.option.c
		custom_tooltip = POD_charjoin.100.option.c.t
	}
}
