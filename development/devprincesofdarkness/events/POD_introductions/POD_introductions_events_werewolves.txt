namespace = podintroductionswerewolves

podintroductionswerewolves.1 = {
#Main event handling introductions
#Called from game start on action with a small delay
	hidden = yes
	type = empty
	
	trigger = { 
		}	
	
	immediate =	
	{					
		#Intro and houses modifiers
		trigger_event = podintroductions.1001

		trigger_event = {
				id = podintroductions.1002
				days = 5
			}	

		#Regional intros
		every_player = {				
			if = {
				limit = {												
						is_werewolf_trigger = yes
						capital_province = { geographical_region = custom_germany }	
					}			
				trigger_event = podintroductionswerewolves.2
			}
			else_if = {
				limit = {												
						is_werewolf_trigger = yes
						capital_province = { geographical_region = world_europe_west_iberia }	
					}			
				trigger_event = podintroductionswerewolves.3
			}
			else_if = {
				limit = {												
						is_werewolf_trigger = yes
						capital_province = { geographical_region = world_europe_north }	
					}			
				trigger_event = podintroductionswerewolves.4
			}
			else_if = {
				limit = {												
						is_werewolf_trigger = yes
						capital_province = { geographical_region = world_europe_west_britannia }	
					}			
				trigger_event = podintroductionswerewolves.5
			}		
			else_if = {
				limit = {
						is_werewolf_trigger = yes
						capital_province = { geographical_region = world_steppe_west }	
					}
				trigger_event = podintroductionswerewolves.6
			}
			else_if = {
				limit = {
						is_werewolf_trigger = yes
						capital_province = { geographical_region = custom_roman_syria_palestina }	
					}
				trigger_event = podintroductionswerewolves.7
			}			
			else_if = {
				limit = {
						is_werewolf_trigger = yes
						capital_province = { geographical_region = world_africa_north_west }	
					}
				trigger_event = podintroductionswerewolves.8
			}	
			else_if = {
				limit = {
						is_werewolf_trigger = yes
						capital_province = { geographical_region = world_tibet }	
					}
				trigger_event = podintroductionswerewolves.9
			}				
		}		
	}
	
}
