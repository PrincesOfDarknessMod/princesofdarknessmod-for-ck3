podregionalpersons.1 = {
#Main event handling RP
#Called from specific yearly pulse on action with a small delay
	hidden = yes
	type = empty
	
	trigger = { 
		}	
	
	immediate =	
	{
	if = {
		limit = {
			current_date >= 1235.1.1
			current_date < 1236.1.1
			#NOT = { has_global_variable = spawned_gp_ald }
			any_player = {
				any_realm_province = {
					geographical_region = world_europe_west_germania
				}
			OR = {
			has_trait = supehunter	
			is_vampire_trigger = yes
			}
			is_kueijin_trigger = no	
			}				
			}	
			random_player = {
				limit = {
				any_realm_province = {
					geographical_region = world_europe_west_germania
				}
				OR = {
				has_trait = supehunter	
				is_vampire_trigger = yes
				}
				is_kueijin_trigger = no		
				}				
			trigger_event = {
					id = podregionalpersons.2
				}
			}		
		}
	}
}
