namespace = POD_vampire_bloodpotency_events

POD_vampire_bloodpotency_events.001 = {
    hidden = yes
	type = character_event
	
	trigger = {
	    is_vampire_trigger = yes
		NOT = { has_character_flag = bloodpotency_increased }
		NOT = { has_trait = torpor } 
		NOT = { has_trait = kueijin }
	}
	immediate = {			
		POD_bloodpotency_increase_effect = yes
	    add_character_flag = {
	        flag = bloodpotency_increased
		    years = 87
		}
    }
}

POD_vampire_bloodpotency_events.002 = {
    hidden = yes
	type = character_event
	
	trigger = {
	    is_vampire_trigger = yes
		NOT = { has_character_flag = bloodpotency_decreased }
		NOT = { has_trait = kueijin }
		has_trait = torpor
	}
	immediate = {
	    POD_bloodpotency_decrease_effect = yes
	    add_character_flag = {
	        flag = bloodpotency_decreased
		    years = 50
		}
	}
}
