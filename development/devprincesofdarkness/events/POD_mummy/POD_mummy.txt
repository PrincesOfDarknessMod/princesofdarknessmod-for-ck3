podmummy.20 = {
	hidden = yes
	type = character_event
	
	trigger = {
		is_ai = no
		has_trait = mummy
	}
	
	immediate = {
		every_relation = {
			type = effigy_character
			limit = {
				is_alive = yes
				NOT = { has_trait = frozen }
			}
			root = { 
				send_interface_message = {
					type = event_generic_bad
					title = POD_effigy_maintenance
					left_icon = root
					right_icon = prev
					if = { 
						limit = { energy_points_value >= 1 }
						POD_hekau_cost_effect = { COUNT = 1 }
					}
					else = { 
						prev = { increase_wounds_effect = { REASON = mysterious } }
					}
					custom_tooltip = POD_effigy_maintenance.tt
				}
			}
		}
		every_character_artifact = {
			limit = { 
				has_variable = active_effigy
				var:active_effigy = yes
			}
			root = { 
				send_interface_message = {
					type = event_generic_bad
					title = POD_effigy_maintenance
					right_icon = root
					if = { 
						limit = { 
							prev = { 
								OR = { 
									has_variable = viziers_beast_effigy
									has_variable = kingly_beast_effigy
									has_variable = divine_beast_effigy
								}
							}
							short_term_gold >= tiny_gold_value
						}
						remove_short_term_gold = tiny_gold_value
					}
					else_if = {
						limit = { energy_points_value >= 1 }
						POD_hekau_cost_effect = { COUNT = 1 }
					}
					else = {
						prev = { degrade_mummy_artifact_rarity_effect = yes }
					}
					custom_tooltip = POD_effigy_maintenance.tt
				}
			}
		}
	}
}

podmummy.300 = {
	type = character_event
	title = podmummy.300.t
	desc = podmummy.300.desc
	right_portrait = root
	theme = witchcraft
	
	immediate = {
		if = {
			limit = { has_trait = cabiri }
			set_variable = { name = spirit_location value = flag:hades }
		}
		else_if = {
			limit = { has_trait = wutian }
			set_variable = { name = spirit_location value = flag:gates_of_heaven }
		}
		else_if = {
			limit = { has_trait = wukuei }
			set_variable = { name = spirit_location value = flag:yomi_wan }
		}
		else = {
			set_variable = { name = spirit_location value = flag:shadow_realm }
		}
		remove_variable = spirit_destination
		add_character_modifier = traveling_spirit_world_modifier
	}
	
	option = {
		name = podmummy.300.a
		if = {
			limit = { num_sinful_traits >= 4 }
			add_piety = massive_resurrect_piety_loss
		}
		else_if = {
			limit = { num_sinful_traits >= 3 }
			add_piety = major_resurrect_piety_loss
		}
		else_if = {
			limit = { num_sinful_traits >= 2 }
			add_piety = medium_resurrect_piety_loss
		}
		else_if = {
			limit = { num_sinful_traits >= 1 }
			add_piety = minor_resurrect_piety_loss
		}
		else = {
			add_piety = miniscule_resurrect_piety_loss
		}
		
		trigger_event = podmummy.301
	}
}
