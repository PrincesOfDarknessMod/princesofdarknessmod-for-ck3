podkueijin.350 = {
	type = character_event
	title = podkueijin.350.t
	desc = podkueijin.350.desc
	theme = dynasty 
	
	left_portrait = {
		character = root
		animation = idle
	}
	
	right_portrait = {
		character = scope:optional_child
		animation = idle
	}	

	# Adopt
	option = 
	{
		name = podkueijin.350.a			

		hidden_effect = {
		
			scope:optional_child = {
				if = {
					limit = { root = { is_male = yes } }
					set_father = root
					set_house = father.house
				}
				else = {
					set_mother = root
					set_house = mother.house
				}
			}
		
			reverse_add_opinion = {
				target = scope:optional_child
				modifier = hired_me_opinion
			}
		}
		
		trigger_event = {
			id = podkueijin.400
			months = 6
		}
		
		ai_chance = {
			factor = 100
		}
	}
}

podkueijin.400 = {
	type = character_event
	title = podkueijin.400.t
	desc = podkueijin.400.desc

	theme = vampire
	left_portrait = {
		character = root
		animation = idle
	}
	
	right_portrait = {
		character = scope:optional_child
		animation = idle
	}	
	
	trigger = {
		scope:optional_child = { is_alive = yes }
	}
	
	immediate = {
		scope:optional_child = { 
			if = {
				limit = { has_trait = dharma0 } 
				remove_trait = dharma0
				add_trait = dharma1
			}
		}
	}
	
	# Continue study
	option = {
		name = podkueijin.400.a
		remove_short_term_gold = 100
		custom_tooltip = podkueijin_education

		hidden_effect = {
		random_list = {
			1 = {
				trigger_event = {
					id = podkueijin.400
					years = 1
				}
			}
			1 = {
				trigger_event = {
					id = podkueijin.401
					years = 1
				}
			}
		}
	}
	}
	
	# Finish study
	option = {
		name = podkueijin.400.b
		
		scope:optional_child = { 
			random_list = {
				1 = {
					add_trait = education_intrigue_1
				}
				1 = {
					add_trait = education_diplomacy_1
				}
				1 = {
					add_trait = education_learning_1
				}
				1 = {
					add_trait = education_stewardship_1
				}
				1 = {
					add_trait = education_martial_1
				}
				1 = {
					add_trait = education_martial_prowess_1
				}
			}
		}		
	}
}
