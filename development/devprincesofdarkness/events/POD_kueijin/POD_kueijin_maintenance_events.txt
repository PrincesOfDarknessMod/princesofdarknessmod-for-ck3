podkueijin.20 = {
	hidden = yes
	type = character_event
	
	trigger = {
		is_kueijin_trigger = yes
		is_ai = no
		OR = { 
			has_dharma_trigger = no
			has_yinyang_trigger = no
			has_chi_trigger = no
		}
	}
	
	immediate = {
		trigger_event = {
			id = podkueijin.200
			days = 5
			}
	}
}

podkueijin.100 = {
	type = character_event
	title = podkueijin.100.t

	option = {
		name = podkueijin.100.a	
		
		trigger = { has_trait = yin_imbalanced }
		
		custom_tooltip = podkueijin.100.a.tt
		
		hidden_effect = {
			#Using vanilla Homely / Beautiful trait to represent the process for now
			if = {
				limit = {
					has_trait = beauty_bad_3

				limit = {
					has_trait = beauty_bad_1
				}
				remove_trait = beauty_bad_1
				add_trait = beauty_bad_2
			}
			if = {
				limit = {
					OR = {
						has_trait = beauty_good_1

							OR = {
								has_trait = beauty_good_3

						}
					}
				}
				remove_trait = beauty_good_1
				add_trait = beauty_bad_1			 
			}
			if = {
				limit = {
					has_trait = beauty_good_1
				}
				remove_trait = beauty_good_1
				add_trait = beauty_good_2
			}
