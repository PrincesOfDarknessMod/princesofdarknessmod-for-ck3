podkueijin.10 = {
	hidden = yes
	type = character_event
	
	trigger = {
		is_kueijin_trigger = yes
		is_ai = no
	}
	
	immediate = {
		if = {
			limit = {
				has_dharma_trigger = no
			}
			trigger_event = podkueijin.100
		}
		else_if = {
			limit = {
				has_yinyang_trigger = no
			}
			trigger_event = podkueijin.110
		}
		else_if = {
			limit = {
				has_chi_trigger = no
			}
			trigger_event = podkueijin.120
		}
	}
}

podkueijin.20 = {
	hidden = yes
	type = character_event
	
	trigger = {
		is_kueijin_trigger = yes
		is_ai = no
	}
	
	immediate = {
		if = {
			limit = {
				NOT = { has_trait = yinyang_balanced }
			}
			trigger_event = podkueijin.200
		}
		
		# Change to the fifth age
		if = {
			limit = {
				NOT = { has_character_flag = age_degenerated }
				OR = {
					AND = {
						current_date >= 1250.1.1
						OR = {
							has_trait = dharma1
							has_trait = dharma2
							has_trait = dharma3
						}
					}
					AND = {
						current_date >= 1350.1.1
						has_trait = dharma4
					}
					AND = {
						current_date >= 1450.1.1
						has_trait = dharma5
					}
					AND = {
						current_date >= 1550.1.1
						has_trait = dharma6
					}
					AND = {
						current_date >= 1600.1.1
						has_trait = dharma7
					}
					AND = {
						current_date >= 1650.1.1
						has_trait = dharma8
					}
					# AND = {
						# # Even Bodhisattvas can degenerate in Final Night
						# has_trait = dharma9
					# }
				}
			}
			trigger_event = podkueijin.600
		}
	}
}
