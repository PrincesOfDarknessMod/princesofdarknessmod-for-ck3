podkueijin.20 = {
	hidden = yes
	type = character_event
	
	trigger = {
		is_kueijin_trigger = yes
		is_ai = no
	}
	
	immediate = {
		if = {
			limit = {
				NOT = { has_trait = yinyang_balanced }
			}
			trigger_event = podkueijin.200
		}
		
		# Change to the fifth age
		if = {
			limit = {
				NOT = { has_character_flag = age_degenerated }
				OR = {
					AND = {
						current_date >= 1250.1.1
						OR = {
							has_trait = dharma1
							has_trait = dharma2
							has_trait = dharma3
						}
					}
					AND = {
						current_date >= 1350.1.1
						has_trait = dharma4
					}
					AND = {
						current_date >= 1450.1.1
						has_trait = dharma5
					}
					AND = {
						current_date >= 1550.1.1
						has_trait = dharma6
					}
					AND = {
						current_date >= 1600.1.1
						has_trait = dharma7
					}
					AND = {
						current_date >= 1650.1.1
						has_trait = dharma8
					}
					# AND = {
						# # Even Bodhisattvas can degenerate in Final Night
						# has_trait = dharma9
					# }
				}
			}
			trigger_event = podkueijin.600
		}
	}
}

	type = character_event
	title = podkueijin.1000.t
	desc = podkueijin.1000.desc
	theme = vampire
	
	left_portrait = {
		character = root
		animation = idle
	}

	option = {
		name = podkueijin.1000.a

				months = 3
			}
		}

			limit = {
				has_focus = soul_arts_all_focus
			}
			trigger_event = {
				id = podkueijin.1013
				months = 3
			}
		}
		else_if = {
			limit = {
				has_focus = demon_arts_all_focus
			}
			trigger_event = {
				id = podkueijin.1014
				months = 3
			}
		}
	}
	
	# Treking
	option = {
		name = podkueijin.1000.b
	
		add_character_modifier = regular_treking_modifier
	}
	
	# Living as mortal
	option = {
		name = podkueijin.1000.c
	
		add_internal_flag = special

	}
	
	# Spirit travel
	option = {
		name = podkueijin.1000.d

				wall_weak_enough_trigger = yes

				location = {

				}
