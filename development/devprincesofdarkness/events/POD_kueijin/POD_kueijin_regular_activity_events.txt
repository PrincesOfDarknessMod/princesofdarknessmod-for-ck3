podkueijin.30 = {
	hidden = yes
	type = character_event
	
	trigger = {
		is_kueijin_trigger = yes
		is_ai = no
	}
	
	immediate = {
		
		if = {
			limit = {
				OR = {
					has_character_modifier = traveling_yin_world_modifier
					has_character_modifier = traveling_yang_world_modifier
				}
				NOT = { has_character_flag = has_spirit_travel }
			}
			trigger_event = podkueijin.1600
		}
		else = {
			limit = {
			}
			trigger_event = podkueijin.40
		}
	}
}

# Event case
podkueijin.1000 = {
	type = character_event
	title = podkueijin.1000.t
	desc = podkueijin.1000.desc
	theme = vampire
	
	left_portrait = {
		character = root
		animation = idle
	}
	
	# Meditating
	option = {
		name = podkueijin.1000.a
			
		add_character_modifier = regular_meditating_modifier
		
		show_as_unavailable = {
			OR = {
				has_focus = shintai_all_focus
				has_focus = chi_arts_all_focus
				has_focus = soul_arts_all_focus
				has_focus = demon_arts_all_focus
			}
		}
	}
	
	# Treking
	option = {
		name = podkueijin.1000.b
	
		add_character_modifier = regular_treking_modifier
	}
	
	# Living as mortal
	option = {
		name = podkueijin.1000.c
	
		add_internal_flag = special
		add_character_modifier = regular_mortal_living_modifier
	}
	
	# Spirit travel
	option = {
		name = podkueijin.1000.d
		show_as_unavailable = {
			location = {
				wall_weak_enough_trigger = yes
			}
			OR = {
				location = {
					has_building_or_higher = dragon_nest_01
				}
				has_perk = semblance_of_the_ebon_dragon_perk
				has_perk = semblance_of_the_scalet_queen_perk
			}
		}
		hidden_effect = {
			trigger_event = podkueijin.1020
		}
	}
	
	# Ruling
	option = {
		name = podkueijin.1000.e
	}
}
