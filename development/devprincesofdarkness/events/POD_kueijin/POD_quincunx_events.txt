#0100 - Quincunx convention
#0200 - Disolution of court

PODquincunx.0001 = {
	hidden = yes
	type = empty
	
	immediate = {
		# Letter from lefting black court
		every_player = {
			limit = {
				is_kueijin_trigger = yes
			}
			trigger_event = PODquincunx.0100
		}
		
		# Scalet phonix court convert to golder court (Player not holding)
		if = {
			limit = {
				OR = {
					title:e_scarlet_dragon = { is_title_created = no }
					title:e_scarlet_dragon.holder = { is_ai = yes }
				}
			}
			trigger_event = {
				id = PODquincunx.202
				days = { 2 5 }
			}
		}
		else = {
			every_player = {
				limit = {
					has_title = title:e_scarlet_dragon
				}
				# Let player choose
				trigger_event = {
					id = PODquincunx.201
					days = { 2 5 }
				}
			}
		}
	}
}

PODquincunx.202 = {
	hidden = yes
	type = empty
	
	immediate = {
		title:e_scarlet_dragon = {
			every_in_de_jure_hierarchy = {
				continue = {
					tier > tier_kingdom
				}
				limit = {
					tier = tier_kingdom
				}
				set_de_jure_liege_title = title:e_golden_courts
			}
		}
		
		if = {
			limit = {
				title:e_scarlet_dragon = { is_title_created = yes }
			}
			title:e_scarlet_dragon.holder = {
				save_scope_as = scoped_ruler
			}
			
			create_title_and_vassal_change = {
				type = returned
				save_scope_as = change
				add_claim_on_loss = no
			}
			title:e_golden_courts = {
				change_title_holder = {
					holder = scope:scoped_ruler
					change = scope:change
				}
			}
			resolve_title_and_vassal_change = scope:change
			scope:scoped_ruler = { destroy_title = title:e_scarlet_dragon }
			hidden_effect = {
				set_primary_title_to = title:e_golden_courts
			}
		}
		
		every_player = {
			limit = {
				is_kueijin_trigger = yes
			}
			trigger_event = PODquincunx.200
		}
	}
}

# War to Anda
PODquincunx.250 = {
	hidden = yes
	type = character_event
	
	trigger = {
		title:e_anda = { is_title_created = yes }
		title:e_anda.holder = { faith = { has_doctrine = special_doctrine_is_anda_faith } }
	}
	
	immediate = {
		add_character_flag = POD_quincunx_warrior
		start_war = {
			cb = POD_directed_great_holy_war
			target = title:e_anda.holder
			target_title = title:e_anda
		}
		random_character_war = {
			limit = { using_cb = POD_directed_great_holy_war }
			save_scope_as = quincunxwar
		}
		every_ruler = {
			limit = { 
				is_independent_ruler = yes
				is_landed = yes
				NOT = { this = root }
			}
			if = {
				limit = { faith = { has_doctrine = special_doctrine_against_to_yomi_wan_hostility } }
				scope:quincunxwar = {
					hidden_effect = {
						set_called_to = prev
					}
					add_attacker = prev
				}
			}
			if = {
				limit = { faith = { has_doctrine = special_doctrine_is_anda_faith } }
				scope:quincunxwar = {
					hidden_effect = {
						set_called_to = prev
					}
					add_defender = prev
				}
			}
		}
	}
}
