namespace = pod1230objectives_werewolves

pod1230objectives_werewolves.1 = {
	type = character_event
	title = pod1230objectives_werewolves.1.t
	desc = pod1230objectives_werewolves.1.desc
	theme = hunt_activity
	left_portrait = {
		character = scope:victorious_blackfury
		animation = personality_honorable
	}

	immediate = {
		play_music_cue = "mx_cue_epic_sacral_moment"		
	}

	option = {
		name = pod1230objectives_werewolves.1.a
		
		create_artifact_bronze_labrys_effect = {
			OWNER = scope:victorious_blackfury
		}
		
		root.dynasty = {
			add_dynasty_prestige = 1000
			add_dynasty_prestige_level = 1
		}
		add_prestige = 500
		
	hidden_effect = {
		every_held_title = { #Should shift all dejure of all Empires owned at the time.
			limit = {
				tier = tier_empire
			}
			every_in_de_jure_hierarchy = {
				limit = {
					tier = tier_kingdom
				}
				set_de_jure_liege_title = title:e_black_furies
				
			}
		}
		every_held_title = { #Completely Controlled Kingdoms as well.
			limit = {
				tier = tier_kingdom
				root = { completely_controls = prev }
			}
			set_de_jure_liege_title = title:e_black_furies
		}
	}
	
		create_title_and_vassal_change = {
			type = created
			save_scope_as = change
			add_claim_on_loss = no
		}

	title:e_black_furies = {
		change_title_holder = {
			holder = root
			change = scope:change
		}
	}
	resolve_title_and_vassal_change = scope:change
	hidden_effect = { set_primary_title_to = title:e_black_furies }		

		every_player = {
			limit = {
				NOT = { this = root }
				}
			trigger_event = pod1230objectives_werewolves.2
		}
	}
	
}
