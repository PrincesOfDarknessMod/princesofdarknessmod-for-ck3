POD_underworld_travel.10 = {
	type = character_event
	title = POD_spirit_travel.998.t
	desc = POD_spirit_travel.998.desc
	left_portrait = root
	theme = faith

	trigger = {
		has_supernatural_modifier_beyond_zero_trigger = no
		NOR = {
			has_character_modifier = traveling_yang_world_modifier
			has_character_modifier = traveling_umbra_modifier
		}
	}

	option = {
		name = POD_spirit_travel.998.a
	}

	after = {
		#End the activity
		if = {
			limit = {
				exists = scope:activity
			}
			scope:activity = {
				complete_activity = yes
			}
		}
	}
}

POD_underworld_travel.100 = { # The Shadowlands
	type = character_event
	title = POD_underworld_travel.100.t
	desc = POD_underworld_travel.100.desc
	
	left_portrait = root
	theme = faith
	
	trigger = {
		OR = {
			has_character_modifier = traveling_yin_world_modifier
			has_character_modifier = traveling_underworld_modifier
		}
	}
	
	immediate = {
	    add_character_flag = in_shadowlands
	}

	option = { # Pass Byways throw The Tempest
		name = POD_underworld_travel.100.a
		
		remove_character_flag = in_shadowlands
		
		trigger_event = {
			id = POD_underworld_travel.200
			days = { 20 30 }
		}
	}

	option = { # Change Aspect
		name = POD_underworld_travel.100.b
		
		trigger = {
			OR = {
				has_perk = semblance_of_the_scalet_queen_perk
				has_perk = see_the_lands_of_the_spirit_perk
				has_perk = separate_ba_perk
			}
		}
		
		remove_character_flag = in_shadowlands
		
		hidden_effect = {
			if = {
				limit = {
					exists = scope:activity
				}
				scope:activity = {
					complete_activity = yes
				}
			}
			capital_province = {
				save_scope_as = capital_province
				spawn_activity = {
					type = activity_umbra_travel
					owner = root
				}
			}
		}
	}

	option = { # Stay
		name = POD_underworld_travel.100.c
		
		trigger_event = {
			id = POD_underworld_travel.100
			days = { 20 30 }
		}
	}

	option = { # Return to Earth
		name = POD_underworld_travel.100.d
		
		remove_character_modifier = traveling_yin_world_modifier
		remove_character_modifier = traveling_underworld_modifier
	}
}
