namespace = POD_underworld_travel

POD_underworld_travel.100 = { # The Shadowlands
	type = character_event
	title = POD_underworld_travel.100.t
	desc = POD_underworld_travel.100.desc
	
	left_portrait = root
	theme = spirit_world
		
	trigger = {
		has_character_modifier = traveling_spirit_world_modifier
	}

	option = { # Pass Byways throw The Tempest
		name = POD_underworld_travel.100.a
		
		remove_character_flag = in_shadowlands
	    add_character_flag = in_tempest
		trigger_event = {
			id = POD_underworld_travel.200
			days = { min_spirit_travel_movement_time max_spirit_travel_movement_time }
		}
	}

	option = { # Change Aspect
		name = POD_underworld_travel.100.b
		
		trigger = {
			OR = {
				has_perk = semblance_of_the_scalet_queen_perk
				has_perk = see_the_lands_of_the_spirit_perk
				has_perk = separate_ba_perk
			}
		}
		
		remove_character_flag = in_shadowlands
		add_character_flag = in_penumbra
		trigger_event = {
			id = POD_umbra_travel.100
			days = { min_spirit_travel_movement_time max_spirit_travel_movement_time }
		}
	}

	option = { # Stay
		name = POD_underworld_travel.100.c
		
		trigger_event = {
			id = POD_underworld_travel.100
			days = { min_spirit_travel_movement_time max_spirit_travel_movement_time }
		}
	}

	option = { # Return to Earth
		name = POD_underworld_travel.100.d
	
		remove_character_flag = in_shadowlands
		add_character_flag = trying_to_end_spirit_travel
		trigger_event = {
			on_action = spirit_travel_end_pulse
			days = { 3 7 }
		}

		ai_chance = {
			base = 100
		}
	}
}

POD_underworld_travel.200 = { # Pass Byways throw The Tempest
	type = character_event
	title = POD_underworld_travel.200.t
	desc = POD_underworld_travel.200.desc
	
	left_portrait = root
	theme = spirit_world
		
	trigger = {
		has_character_modifier = traveling_spirit_world_modifier
	}

	option = {
		name = POD_underworld_travel.200.a
		
		remove_character_flag = in_tempest
		random_list = {
			1 = {
				trigger = { culture = { has_cultural_era_or_later = culture_era_final_nights } }
				add_character_flag = in_dark_kingdom_of_clay
				trigger_event = {
					id = POD_underworld_travel.310
					days = { min_spirit_travel_movement_time max_spirit_travel_movement_time }
				}
			}
			1 = {
				trigger = { 
					location = { 
						OR = {
							geographical_region = world_europe 
							geographical_region = world_middle_east
							geographical_region = world_middle_east_persia
							geographical_region = world_asia_minor
						}
					} 
				}
				add_character_flag = in_dark_kingdom_of_iron
				trigger_event = {
					id = POD_underworld_travel.320
					days = { min_spirit_travel_movement_time max_spirit_travel_movement_time }
				}
			}
			1 = {
				trigger = { location = { geographical_region = world_africa } }
				add_character_flag = in_dark_kingdom_of_ivory
				trigger_event = {
					id = POD_underworld_travel.330
					days = { min_spirit_travel_movement_time max_spirit_travel_movement_time }
				}
			}
			1 = {
				trigger = { 
					location = { 
						OR = {
							geographical_region = world_steppe
							geographical_region = world_tibet
							geographical_region = world_india
							geographical_region = world_burma
						}
					} 
				}
				add_character_flag = in_dark_kingdom_of_jade
				trigger_event = {
					id = POD_underworld_travel.340
					days = { min_spirit_travel_movement_time max_spirit_travel_movement_time }
				}
			}
			1 = {
				trigger = { culture = { has_cultural_era_or_later = culture_era_final_nights } }
				add_character_flag = in_dark_kingdom_of_obsidian
				trigger_event = {
					id = POD_underworld_travel.350
					days = { min_spirit_travel_movement_time max_spirit_travel_movement_time }
				}
			}
			1 = {
				trigger = { 
					has_trait = mummy
					is_east_mummy_trigger = no
				}
				add_character_flag = in_dark_kingdom_of_sand
				trigger_event = {
					id = POD_underworld_travel.360
					days = { min_spirit_travel_movement_time max_spirit_travel_movement_time }
				}
			}
			1 = {
				trigger = { 
					NOT = { has_perk = stormwalk_perk }
				}
				trigger_event = {
					id = POD_underworld_travel.201
					days = { min_spirit_travel_movement_time max_spirit_travel_movement_time }
				}
			}
		}
	}

	option = {
		name = POD_underworld_travel.200.b
		
		trigger_event = {
			id = POD_underworld_travel.200
			days = { min_spirit_travel_movement_time max_spirit_travel_movement_time }
		}
	}

	option = {
		name = POD_underworld_travel.200.c
		
		remove_character_flag = in_tempest
		
		random_list = {
			1 = {
				add_character_flag = in_shadowlands
				trigger_event = {
					id = POD_underworld_travel.100
					days = { min_spirit_travel_movement_time max_spirit_travel_movement_time }
				}
			}
			1 = {
				trigger = { 
					NOT = { has_perk = stormwalk_perk }
				}
				trigger_event = {
					id = POD_underworld_travel.201
					days = { min_spirit_travel_movement_time max_spirit_travel_movement_time }
				}
			}
		}

		ai_chance = {
			base = 100
		}
	}
}

POD_underworld_travel.301 = { # Shadow Realm
	type = character_event
	title = POD_underworld_travel.301.t
	desc = POD_underworld_travel.301.desc
	
	left_portrait = root
	theme = spirit_world
		
	trigger = {
		has_character_modifier = traveling_spirit_world_modifier
	}

	option = {
		name = POD_underworld_travel.301.a
		
	    remove_character_flag = in_shadow_realm
		if = {
			limit = {
				is_east_mummy_trigger = yes
			}
			add_character_flag = in_dark_kingdom_of_jade
			trigger_event = {
				id = POD_underworld_travel.340
				days = { min_spirit_travel_movement_time max_spirit_travel_movement_time }
			}
		}
		else = {
			add_character_flag = in_dark_kingdom_of_sand
			trigger_event = {
				id = POD_underworld_travel.360
				days = { min_spirit_travel_movement_time max_spirit_travel_movement_time }
			}
		}
		trigger_event = {
			on_action = spirit_travel_random_pulse
			days = { 20 30 }
		}
	}

	option = {
		name = POD_underworld_travel.301.b
		trigger = {
			has_perk = empower_ba_perk
		}
		show_as_unavailable = {
			NOT = { has_perk = empower_ba_perk }
		}
		add_internal_flag = special
		custom_tooltip = POD_underworld_travel.301.b.tt
	
		remove_character_flag = in_shadow_realm
		add_character_flag = trying_to_end_spirit_travel
		trigger_event = {
			on_action = spirit_travel_end_pulse
			days = { 3 7 }
		}

		ai_chance = {
			base = 100
		}
	}
}
