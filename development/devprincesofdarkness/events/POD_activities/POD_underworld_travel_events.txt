POD_underworld_travel.100 = { # The Shadowlands
	type = character_event
	title = shadowlands_title
	desc = shadowlands_desc
	
	left_portrait = root
	theme = spirit_world
		
	trigger = {
		has_character_modifier = traveling_spirit_world_modifier
	}

	option = { # Pass Byways throw The Tempest
		name = tempest_option
		
		set_variable = { name = spirit_location value = flag:tempest }
		trigger_event = {
			id = POD_underworld_travel.200
			days = { min_spirit_travel_movement_time max_spirit_travel_movement_time }
		}
	}

	option = { # Change Aspect
		name = penumbra_option
		
		trigger = {
			OR = {
				has_perk = semblance_of_the_scalet_queen_perk
				has_perk = see_the_lands_of_the_spirit_perk
				has_perk = separate_ba_perk
				any_equipped_character_artifact = { has_variable = fans_of_ma_yi_yuan }
				has_character_modifier = philtre_of_the_blackened_sands_modifier
			}
		}
		
		set_variable = { name = spirit_location value = flag:penumbra }
		trigger_event = {
			id = POD_umbra_travel.100
			days = { min_spirit_travel_movement_time max_spirit_travel_movement_time }
		}
	}

	option = { # Stay
		name = POD_spirit_travel.100.a
		
		trigger_event = {
			id = POD_underworld_travel.100
			days = { min_spirit_travel_movement_time max_spirit_travel_movement_time }
		}
	}

	option = { # Return to Earth
		name = skinlands_option
	
		set_variable = { name = spirit_location value = flag:skinlands }
		trigger_event = {
			on_action = spirit_travel_end_pulse
			days = { 3 7 }
		}

		ai_chance = {
			base = 100
		}
	}
}

	}

	option = {

				is_east_mummy_trigger = yes
			}
			set_variable = { name = spirit_location value = flag:dark_kingdom_of_jade }

			}
		
		trigger_event = {
			on_action = spirit_travel_random_pulse
			days = { 20 30 }
		}
	}

	option = {
		name = dark_kingdom_of_sand_option
		
		trigger = {
				is_east_mummy_trigger = no
			}
			set_variable = { name = spirit_location value = flag:dark_kingdom_of_sand }
			trigger_event = {
				id = POD_underworld_travel.360
				days = { min_spirit_travel_movement_time max_spirit_travel_movement_time }
			}
		trigger_event = {
			on_action = spirit_travel_random_pulse
			days = { 20 30 }
		}
	}

	option = {
		name = skinlands_option
		trigger = {
			has_perk = empower_ba_perk
		}
		show_as_unavailable = {
			NOT = { has_perk = empower_ba_perk }
		}
		add_internal_flag = special
		custom_tooltip = shadow_realm_to_skinlands_option_tooltip
	
		set_variable = { name = spirit_location value = flag:skinlands }
		trigger_event = {
			on_action = spirit_travel_end_pulse
			days = { 3 7 }
		}

		ai_chance = {
			base = 100
		}
	}
}
