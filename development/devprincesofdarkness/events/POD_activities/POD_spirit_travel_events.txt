POD_spirit_travel.1000 = { # Nothing happen
	hidden = yes
	type = character_event
	
	immediate = {
		if = {
			limit = { NOT = { has_trait = wraith } }
			POD_universal_energy_decrease_multiple_effect = { COUNT = 1 }
		}
		else = {
			POD_wraith_potency_increase = yes
		}
		if = {
			limit = { energy_points_value = 0 }
			add_character_flag = trying_to_end_spirit_travel
			trigger_event = {
				on_action = spirit_travel_end_pulse
				days = { 3 7 }
			}
		}
		if = {
			limit = { NOT = { has_character_flag = trying_to_end_spirit_travel } }
			trigger_event = {
				on_action = spirit_travel_random_pulse
				days = { 20 30 }
			}
		}
	}
}

}

POD_spirit_travel.1002 = {
	type = character_event
	title = POD_spirit_travel.1002.t
	desc = {
		desc = POD_spirit_travel.1002.desc_opening
		first_valid = {
			triggered_desc = {
				trigger = {
					OR = {
						has_trait = dharma9 
						has_trait = dharma10
					}
				}
				desc = POD_spirit_travel.1002.desc_high
			}
			triggered_desc = {
				trigger = {
					OR = {
						has_trait = dharma8
						has_trait = dharma7
						has_trait = dharma6
					}
					OR = {
						has_character_flag = in_shadowlands
						has_character_flag = in_tempest
					}
				}
				desc = POD_spirit_travel.1002.desc_yin
			}
			triggered_desc = {
				trigger = {
					OR = {
						has_trait = dharma8
						has_trait = dharma7
						has_trait = dharma6
					}
				}
				desc = POD_spirit_travel.1002.desc_yang
			}
			desc = POD_spirit_travel.1002.desc_low
		}
		desc = POD_spirit_travel.1002.desc_ending
	}
	
	left_portrait = root
	theme = spirit_world
	
	trigger = {
		has_trait = kueijin
	}
	
	option = {
		name = POD_spirit_travel.1002.a
		if = {
			limit = {
				OR = {
					has_trait = dharma9 
					has_trait = dharma10
				}
			}
			add_piety = major_piety_value
		}
		else_if = {
			limit = {
				OR = {
					has_trait = dharma6
					has_trait = dharma7
					has_trait = dharma8
				}
			}
			add_piety = medium_piety_value
		}
		else = {
			add_piety = minor_piety_value
		}
	}
	
	after = {
		trigger_event = POD_spirit_travel.1000
	}
}

POD_spirit_travel.1101 = {
	type = character_event
	title = POD_spirit_travel.1100.t
	desc = POD_spirit_travel.1100.desc
	
	left_portrait = root
	theme = spirit_world

	option = {
		name = POD_spirit_travel.1100.a
		duel = {
			skill = diplomacy
			value = low_skill_rating
		
			1 = {
				desc = POD_spirit_travel.1100.critical_success
				compare_modifier = {
					value = scope:duel_value
					multiplier = 0.1
				}
				add_diplomacy_lifestyle_xp = medium_lifestyle_xp
			}
			25 = {
				desc = POD_spirit_travel.1100.success
				compare_modifier = {
					value = scope:duel_value
					multiplier = 0.5
				}
				add_diplomacy_lifestyle_xp = miniscule_lifestyle_xp
			}
			25 = {
				desc = POD_spirit_travel.1100.fail
				compare_modifier = {
					value = scope:duel_value
					multiplier = -0.5
				}
			}
			1 = {
				desc = POD_spirit_travel.1100.critical_fail
				compare_modifier = {
					value = scope:duel_value
					multiplier = -0.01
				}
				add_stress = major_stress_gain
			}
		}
	}
	
	after = {

	}
}
