namespace = POD_spirit_travel

# Common event

POD_spirit_travel.1000 = { # Nothing happen
	hidden = yes
	type = character_event
	
	immediate = {
		if = {
			limit = {
				OR = {
					has_character_modifier = traveling_yang_world_modifier
					has_character_modifier = traveling_umbra_modifier
				}
			}
			trigger_event = {
				on_action = umbra_travel_random_pulse
				days = { 14 20 }
			}
		}
		else = {
			trigger_event = {
				on_action = underworld_travel_random_pulse
				days = { 14 20 }
			}
		}
	}
}

	title = POD_spirit_travel.1001.t
	desc = POD_spirit_travel.1001.desc
	
	left_portrait = root
	theme = faith

		has_trait = kueijin
	}

		base = 1
		modifier = {
			add = 1000

		name = POD_spirit_travel.1001.a

			}
		}
		else = {
			trigger_event = {
				on_action = underworld_travel_random_pulse
				days = { 14 20 }
			}
		}
	}
}

POD_spirit_travel.1002 = {
	type = character_event
	title = POD_spirit_travel.1002.t
	desc = {
		desc = POD_spirit_travel.1002.desc_opening
		first_valid = {
			triggered_desc = {
				trigger = {
					OR = {
						has_trait = dharma9 
						has_trait = dharma10
					}
				}
				desc = POD_spirit_travel.1002.desc_high
			}
			triggered_desc = {
				trigger = {
					OR = {
						has_trait = dharma8
						has_trait = dharma7
						has_trait = dharma6
					}
					has_character_modifier = traveling_yin_world_modifier
				}
				desc = POD_spirit_travel.1002.desc_yin
			}
			triggered_desc = {
				trigger = {
					OR = {
						has_trait = dharma8
						has_trait = dharma7
						has_trait = dharma6
					}
					has_character_modifier = traveling_yang_world_modifier
				}
				desc = POD_spirit_travel.1002.desc_yang
			}
			desc = POD_spirit_travel.1002.desc_low
		}
		desc = POD_spirit_travel.1002.desc_ending
	}
	
	left_portrait = root
	theme = faith
	
	trigger = {
		has_trait = kueijin
	}
	
	option = {
		name = POD_spirit_travel.1002.a
		if = {
			limit = {
				OR = {
					has_trait = dharma9 
					has_trait = dharma10
				}
			}
			add_piety = major_piety_value
		}
		else_if = {
			limit = {
				OR = {
					has_trait = dharma6
					has_trait = dharma7
					has_trait = dharma8
				}
			}
			add_piety = medium_piety_value
		}
		else = {
			add_piety = minor_piety_value
		}
	}
	
	after = {
		if = {
			limit = {
				OR = {
					has_character_modifier = traveling_yang_world_modifier
					has_character_modifier = traveling_umbra_modifier
				}
			}
			trigger_event = {
				on_action = umbra_travel_random_pulse
				days = { 14 20 }
			}
		}
		else = {
			trigger_event = {
				on_action = underworld_travel_random_pulse
				days = { 14 20 }
			}
		}
	}
}
