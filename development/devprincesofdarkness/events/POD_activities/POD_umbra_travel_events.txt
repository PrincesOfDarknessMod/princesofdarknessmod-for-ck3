namespace = POD_umbra_travel

POD_umbra_travel.100 = { # Penumbra
	type = character_event
	title = POD_umbra_travel.100.t
	desc = POD_umbra_travel.100.desc
	
	left_portrait = root
	theme = spirit_world
		
	trigger = {
		has_character_modifier = traveling_spirit_world_modifier
	}
	
	immediate = {
	    add_character_flag = in_penumbra
	}

	option = { # Go up to Astra Reaches
		name = POD_umbra_travel.100.a
		
		remove_character_flag = in_penumbra
		
		trigger_event = {
			id = POD_umbra_travel.200
			days = { 20 30 }
		}
	}

	option = { # Go far to Spirit Wilds
		name = POD_umbra_travel.100.b
		
		remove_character_flag = in_penumbra
		
		trigger_event = {
			id = POD_umbra_travel.300
			days = { 20 30 }
		}
	}

	option = { # I see a trod
		name = POD_umbra_travel.100.c
		
		remove_character_flag = in_penumbra
		
		trigger_event = {
			id = POD_umbra_travel.400
			days = { 20 30 }
		}
	}

	option = { # Change Aspect
		name = POD_umbra_travel.100.e
		
		trigger = {
			OR = {
				has_perk = semblance_of_the_ebon_dragon_perk
				has_perk = lesser_spirit_projection_perk
				has_perk = separate_ka_perk
			}
		}
		
		remove_character_flag = in_penumbra
		
		trigger_event = {
			id = POD_underworld_travel.100
			days = { 20 30 }
		}
	}

	option = { # Stay
		name = POD_umbra_travel.100.f
		
		trigger_event = {
			id = POD_umbra_travel.100
			days = { 20 30 }
		}
	}

	option = { # Return to Earth
		name = POD_umbra_travel.100.g
	
		add_character_flag = trying_to_end_spirit_travel
		trigger_event = {
			on_action = spirit_travel_end_pulse
			days = { 3 7 }
		}
	}
}
