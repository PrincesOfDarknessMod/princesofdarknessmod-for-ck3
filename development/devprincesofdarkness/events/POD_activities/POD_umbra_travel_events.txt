namespace = POD_umbra_travel

POD_umbra_travel.100 = { # Penumbra
	type = character_event
	title = penumbra_title
	desc = penumbra_desc
	
	left_portrait = root
	theme = spirit_world
		
	trigger = {
		has_character_modifier = traveling_spirit_world_modifier
	}

	option = { # Go up to Astra Reaches
		name = astral_reaches_option
		
	    set_variable = { name = spirit_location value = flag:astral_reaches }
		trigger_event = {
			id = POD_umbra_travel.200
			days = { min_spirit_travel_movement_time max_spirit_travel_movement_time }
		}
	}

	option = { # Go far to Spirit Wilds
		name = spirit_wilds_option
		
	    set_variable = { name = spirit_location value = flag:spirit_wilds }
		trigger_event = {
			id = POD_umbra_travel.300
			days = { min_spirit_travel_movement_time max_spirit_travel_movement_time }
		}
	}

	option = { # I see a trod
		name = near_dreaming_option
		
		set_variable = { name = spirit_location value = flag:near_dreaming }
		trigger_event = {
			id = POD_umbra_travel.400
			days = { min_spirit_travel_movement_time max_spirit_travel_movement_time }
		}
	}

	option = { # Change Aspect
		name = shadowlands_option
		
		trigger = {
			OR = {
				has_perk = semblance_of_the_ebon_dragon_perk
				has_perk = lesser_spirit_projection_perk
				has_perk = separate_ka_perk
				has_perk = ex_nihilo_perk
				has_trait = wraith
			}
		}
		
		set_variable = { name = spirit_location value = flag:shadowlands }
		trigger_event = {
			id = POD_underworld_travel.100
			days = { min_spirit_travel_movement_time max_spirit_travel_movement_time }
		}
	}

	option = { # Stay
		name = POD_spirit_travel.100.a
		
		trigger_event = {
			id = POD_umbra_travel.100
			days = { min_spirit_travel_movement_time max_spirit_travel_movement_time }
		}
	}

	option = { # Return to Earth
		name = skinlands_option
	
		set_variable = { name = spirit_location value = flag:skinlands }
		trigger_event = {
			on_action = spirit_travel_end_pulse
			days = { 3 7 }
		}

		ai_chance = {
			base = 100
		}
	}
}
