POD_umbra_travel.10 = {
	type = character_event
	title = POD_spirit_travel.998.t
	desc = POD_spirit_travel.998.desc
	left_portrait = root
	theme = faith

	trigger = {
		has_supernatural_modifier_beyond_zero_trigger = no
	}

	option = {
		name = POD_spirit_travel.998.a
	}

	after = {
		#End the activity
		if = {
			limit = {
				exists = scope:activity
			}
			scope:activity = {
				complete_activity = yes
			}
		}
	}
}

POD_umbra_travel.100 = { # Penumbra
	type = character_event
	title = POD_umbra_travel.100.t
	desc = POD_umbra_travel.100.desc
	
	left_portrait = root
	theme = faith
	
	immediate = {
	    add_character_flag = in_penumbra
	}

	option = { # Go up to Astra Reaches
		name = POD_umbra_travel.100.a
		
		remove_character_flag = in_penumbra
		
		trigger_event = {
			id = POD_umbra_travel.200
			days = { 20 30 }
		}
	}

	option = { # Go far to Spirit Wilds
		name = POD_umbra_travel.100.b
		
		remove_character_flag = in_penumbra
		
		trigger_event = {
			id = POD_umbra_travel.300
			days = { 20 30 }
		}
	}

	option = { # I see a trod
		name = POD_umbra_travel.100.c
		
		remove_character_flag = in_penumbra
		
		trigger_event = {
			id = POD_umbra_travel.400
			days = { 20 30 }
		}
	}

	option = { # Change Aspect
		name = POD_umbra_travel.100.e
		
		trigger = {
			OR = {
				has_perk = semblance_of_the_ebon_dragon_perk
				has_perk = lesser_spirit_projection_perk
				has_perk = separate_ka_perk
			}
		}
		
		remove_character_flag = in_penumbra
		
		hidden_effect = {
			if = {
				limit = {
					exists = scope:activity
				}
				scope:activity = {
					complete_activity = yes
				}
			}
			capital_province = {
				save_scope_as = capital_province
				spawn_activity = {
					type = activity_underworld_travel
					owner = root
				}
			}
		}
	}

	option = { # Stay
		name = POD_umbra_travel.100.f
		
		trigger_event = {
			id = POD_umbra_travel.100
			days = { 20 30 }
		}
	}

	option = { # Return to Earth
		name = POD_umbra_travel.100.g
		
		if = {
			limit = { has_character_modifier = traveling_yang_world_modifier }
			remove_character_modifier = traveling_yang_world_modifier
		}
		if = {
			limit = { has_character_modifier = traveling_umbra_modifier }
			remove_character_modifier = traveling_umbra_modifier
		}
	}
}
