#Events for de jure flavorization

namespace = POD_title_flavor

#Event popping at the game start of a vampire dominant bookmark
#Flavorize de jure for vampires
#De jure base is vampire centric, so only needed for non vampire starts/titles
POD_title_flavor.0001 = {
	hidden = yes
	type = empty
	
	trigger = {
		}
	
	immediate = {
	}	

}


#Event checking during a title holder change if the name has to be changed
POD_title_flavor.0002 = {
	hidden = yes
	type = character_event
	
	trigger = {
		}
	
	immediate = {
		if = {
			limit = {
				scope:title = { this = title:e_britannia }
				NOT = { faith = faith:viamithras }
				}
		title:e_britannia = { set_coa = e_britannia_vanilla }
		title:e_britannia = { set_title_name = e_britannia_vanilla }
	}
		else_if = {
			limit = {
				scope:title = { this = title:e_france }
				NOT = { is_vampire_trigger = yes }
				}
		title:e_france = { set_coa = e_france_vanilla }
		title:e_france = { set_title_name = e_france_vanilla }
	}
		else_if = {
			limit = {
				scope:title = { this = title:e_spain }
				NOT = { is_vampire_trigger = yes }
				}
		title:e_spain = { set_title_name = e_spain_vanilla }
	}
		else_if = {
			limit = {
				scope:title = { this = title:e_germany }
				NOT = { is_vampire_trigger = yes }
				}
		title:e_germany = { set_coa = e_germany_vanilla }
		title:e_germany = { set_title_name = e_germany_vanilla }
	}
		else_if = {
			limit = {
				scope:title = { this = title:e_italy }
				NOT = { is_vampire_trigger = yes }
				}
		title:e_italy = { set_title_name = e_italy_vanilla }
	}	
	}
}
