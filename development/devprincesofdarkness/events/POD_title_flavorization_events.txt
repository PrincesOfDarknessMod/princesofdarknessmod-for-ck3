#Event popping at the game start of a vampire dominant bookmark
#Flavorize de jure for vampires
POD_title_flavor.0001 = {
	hidden = yes
	type = empty
	
	trigger = {
		}
	
	immediate = {
		if = {
			limit = {
				OR = {
					game_start_date = 1230.1.4
					game_start_date = 1230.1.5
					game_start_date = 1510.1.3
					}
				}
				
			title:e_britannia = {
				every_in_de_jure_hierarchy = {
					limit = {
						tier = tier_kingdom
					}
					set_de_jure_liege_title = title:e_avalon
				}
			}

			title:e_france = {
				every_in_de_jure_hierarchy = {
					limit = {
						tier = tier_kingdom
					}
					set_de_jure_liege_title = title:e_love
				}
			}			

			title:e_germany = {
				every_in_de_jure_hierarchy = {
					limit = {
						tier = tier_kingdom
					}
					set_de_jure_liege_title = title:e_fiefs_of_the_black_cross
				}
			}			

			title:e_italy = {
				every_in_de_jure_hierarchy = {
					limit = {
						tier = tier_kingdom
					}
					set_de_jure_liege_title = title:e_sea_of_shadows
				}
			}			

			title:e_russia = {
				every_in_de_jure_hierarchy = {
					limit = {
						tier = tier_kingdom
					}
					set_de_jure_liege_title = title:e_shadow_curtain
				}
			}			

			title:e_scandinavia = {
				every_in_de_jure_hierarchy = {
					limit = {
						tier = tier_kingdom
					}
					set_de_jure_liege_title = title:e_midgard
				}
			}			

			title:e_arabia = {
				every_in_de_jure_hierarchy = {
					limit = {
						tier = tier_kingdom
					}
					set_de_jure_liege_title = title:e_ashirra
				}
			}			

			title:e_mongolia = {
				every_in_de_jure_hierarchy = {
					limit = {
						tier = tier_kingdom
					}
					set_de_jure_liege_title = title:e_anda
				}
			}			

			title:e_persia = {
				every_in_de_jure_hierarchy = {
					limit = {
						tier = tier_kingdom
					}
					set_de_jure_liege_title = title:e_assamites
				}
			}			
	
			title:e_rajastan = {
				every_in_de_jure_hierarchy = {
					limit = {
						tier = tier_kingdom
					}
					set_de_jure_liege_title = title:e_ravnos
				}
			}

				every_in_de_jure_hierarchy = {
					limit = {
						tier = tier_kingdom
					}
					set_de_jure_liege_title = title:e_infinite_thunder
				}
			}			

			title:e_carpathia = {
				every_in_de_jure_hierarchy = {
					limit = {
						tier = tier_kingdom
					}
					set_de_jure_liege_title = title:e_voivodate
				}
			}			
																								
			title:k_sicily = { set_de_jure_liege_title = title:e_sea_of_shadows }
			title:e_byzantium = { set_title_name = e_dream }				
			title:e_byzantium = { set_coa = e_dream }					
			title:e_maghreb = { set_title_name = e_carthage }				
			title:e_maghreb = { set_coa = e_carthage }					
			title:k_egypt = { set_de_jure_liege_title = title:e_setite }
			title:k_nubia = { set_de_jure_liege_title = title:e_setite }
			title:k_blemmyia = { set_de_jure_liege_title = title:e_setite }
	}
		if = {
			limit = {
					game_start_date = 1510.1.3
				}	
		title:e_spain = { set_coa = e_cainite_curia }
		title:e_spain = { set_title_name = e_cainite_curia }		
		}
	
	}	
	
}
