#Event checking during a title holder change if the name has to be changed
POD_title_flavor.0002 = {
	hidden = yes
	type = character_event
	
	trigger = {
		}
	
	immediate = {
		if = {
			limit = {
				scope:title = { this = title:e_avalon }
				NOT = { faith = faith:viamithras }
				}
		title:e_avalon = { set_coa = e_britannia }
		title:e_avalon = { set_title_name = e_britannia }
	}
		else_if = {
			limit = {
				scope:title = { this = title:e_love }
				NOT = { is_vampire_trigger = yes }
				}
		title:e_love = { set_coa = e_france }
		title:e_love = { set_title_name = e_france }
	}
		else_if = {
			limit = {
				scope:title = { this = title:e_spain }
				NOT = { faith = { religion_tag = sabbat_religion } }
				game_start_date = 1510.1.3
				}
		title:e_spain = { set_coa = e_spain }
		title:e_spain = { set_title_name = e_spain }		
		}	
		else_if = {
			limit = {
				scope:title = { this = title:e_fiefs_of_the_black_cross }
				NOT = { is_vampire_trigger = yes }
				}
		title:e_fiefs_of_the_black_cross = { set_coa = e_germany }
		title:e_fiefs_of_the_black_cross = { set_title_name = e_germany }
	}
		else_if = {
			limit = {
				scope:title = { this = title:e_sea_of_shadows }
				is_lasombra_trigger = no
				}
		title:e_sea_of_shadows = { set_coa = e_italy }
		title:e_sea_of_shadows = { set_title_name = e_italy }	
	}
		else_if = {
			limit = {
				scope:title = { this = title:e_shadow_curtain }
				NOT = { is_vampire_trigger = yes }
				}
		title:e_shadow_curtain = { set_coa = e_russia }
		title:e_shadow_curtain = { set_title_name = e_russia }
	}
		else_if = {
			limit = {
				scope:title = { this = title:e_midgard }
				NOT = { faith = faith:viaeinherjar }
				}
		title:e_midgard = { set_coa = e_scandinavia }
		title:e_midgard = { set_title_name = e_scandinavia }
	}
		else_if = {
			limit = {
				scope:title = { this = title:e_anda }
				NOT = { is_vampire_trigger = yes }
				}
		title:e_anda = { set_coa = e_mongolia }
		title:e_anda = { set_title_name = e_mongolia }
	}
		else_if = {
			limit = {
				scope:title = { this = title:e_ashirra }
				NOT = { faith = { religion_tag = ashirra_religion } }
				}
		title:e_ashirra = { set_coa = e_arabia }
		title:e_ashirra = { set_title_name = e_arabia }		
		}	
		else_if = {
			limit = {
				scope:title = { this = title:e_assamites }
				is_assamite_trigger = no
				}
		title:e_assamites = { set_coa = e_persia }
		title:e_assamites = { set_title_name = e_persia }		
		}
		else_if = {
			limit = {
				scope:title = { this = title:e_ravnos }
				is_ravnos_trigger = no
				}
		title:e_ravnos = { set_coa = e_rajastan }
		title:e_ravnos = { set_title_name = e_rajastan }		
		}	
		else_if = {
			limit = {
				scope:title = { this = title:e_infinite_thunder }

		title:e_infinite_thunder = { set_coa = e_bengal }
		title:e_infinite_thunder = { set_title_name = e_bengal }		
		}

			limit = {
				scope:title = { this = title:e_voivodate }

				}
		title:e_voivodate = { set_coa = e_carpathia }
		title:e_voivodate = { set_title_name = e_carpathia }

			limit = {
				scope:title = { this = title:e_maghreb }

				}
		title:e_maghreb = { set_coa = e_maghreb }
		title:e_maghreb = { set_title_name = e_maghreb }	
	}
		else_if = {
			limit = {
				scope:title = { this = title:e_byzantium }

			OR = {	
				faith = faith:viaarchangeli
				is_toreador_trigger = yes
