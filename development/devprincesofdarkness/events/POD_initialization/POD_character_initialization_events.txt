	}

	}	
}

POD_character_initialization.1 = {
	hidden = yes
	type = character_event
	
	trigger = {
		is_ai = yes 
		has_POD_race_trigger = no
		NOT = { has_character_flag = had_POD_character_initialization }
		#exists = capital_province
		#is_adult = yes
		#exists = location		
		#capital_province = { has_supernatural_religion_trigger = yes }
		#is_landed = yes
	}
	
	immediate = {
	
	#POD_clear_resonance_effect = yes
	POD_set_resonance_effect = yes
	
	# Exclude Popes
		if = {
			limit = {
				OR = {
					has_title = title:k_papal_state 
					has_title = title:k_orthodox
					has_title = title:d_iconoclast
					has_title = title:d_rabbinism
					#has_title = title:d_marranoism
					has_title = title:d_karaism
					has_title = title:d_haymanot
					has_title = title:d_malabarism
					has_title = title:d_samaritan
					has_title = title:d_kabarism
					has_title = title:d_mazdayasna
					has_title = title:d_zurvanism
					has_title = title:d_gayomarthianism
					has_title = title:d_khurmazta
					has_title = title:d_mazdakism
					has_title = title:d_khurramism
					has_title = title:d_urartuism 
				}
			}
			add_character_flag = {
				flag = had_POD_character_initialization
			}
			
			if = {
				limit = {
					NOT = { has_game_rule = world_of_monsters }
				}			
				add_character_flag = {
					flag = POD_mortal_race_flag
				}
			}	
			if = {
				limit = {
					has_game_rule = world_of_monsters
				}			
				add_trait = mortal
				if = {
					limit = {
						NOT = { religion = religion:hunters_religion }
					}
					set_character_faith = faith:innocence
				}
			}
		}	

		# Merc commanders
		
		if = {
			limit = {
				has_POD_race_trigger = no
				primary_title.tier = tier_duchy
				is_landed = no
			}
			POD_character_initialization_for_merc_commanders_effect = yes
		}
		
		if = {
			limit = {
				has_POD_race_trigger = no
				OR = {
					primary_title.tier = tier_county
					primary_title.tier = tier_duchy
					primary_title.tier = tier_kingdom
					primary_title.tier = tier_empire
				}
				NOT = { has_character_flag = had_POD_character_initialization }
			}
			POD_character_initialization_for_nobles_effect = yes
		}
		
		if = {
			limit = {
				primary_title.tier = tier_barony	
				has_POD_race_trigger = no
			}
			POD_character_initialization_for_barons_effect = yes
			save_scope_as = characterneedsnewdynasty					
			POD_character_new_dynasty_effect = yes
		}
		
		if = {
			limit = {
				is_landed = no
				has_POD_race_trigger = no
				NOT = { has_character_flag = had_POD_character_initialization }
				is_adult = yes
			}
			POD_character_initialization_for_commoners_effect = yes
		}
	}	
}	


# Make canon supernaturals secret

POD_character_initialization.3 = {
	hidden = yes
	type = character_event
	
	trigger = {
		has_POD_race_trait_trigger = yes
		NOT = { has_character_flag = had_POD_character_initialization }
		is_ai = yes 
		#is_adult = yes
		NOT = { has_game_rule = world_of_monsters }	
		exists = location
	}
	
	immediate = {
	
		# make vampires secret
		if = {
			limit = {
				has_trait = vampire 
			}
			
			trigger_event = POD_disappearance.1
			
			remove_trait = vampire
			
			add_secret = {
				type = secret_vampire 
			}
			
			
		}
		
		# make werewolves secret
		if = {
			limit = {
				has_trait = werewolf
			}
			remove_trait = werewolf
			add_secret = {
				type = secret_werewolf 
			}
		}	

		# make mages secret
		if = {
			limit = {
				has_trait = mage
			}
			remove_trait = mage
			add_secret = {
				type = secret_mage
			}
		}

		# make fae secret
		if = {
			limit = {
				has_trait = fae
			}
			remove_trait = fae
			add_secret = {
				type = secret_fae 
			}
		}

		# make demon secret
		if = {
			limit = {
				has_trait = demon
			}
			remove_trait = demon
			add_secret = {
				type = secret_demon 
			}
		}

		# make ghoul secret
		if = {
			limit = {
				has_trait = ghoul
			}
			remove_trait = ghoul
			add_secret = {
				type = secret_ghoul
			}
		}

		# make revenant secret
		if = {
			limit = {
				has_trait = revenant
			}
			remove_trait = revenant
			add_secret = {
				type = secret_revenant
			}
		}

		# make supehunter secret
		if = {
			limit = {
				has_trait = supehunter
			}
			remove_trait = supehunter
			add_secret = {
				type = secret_supehunter 
			}
		}		
		
		add_character_flag = {
			flag = had_POD_character_initialization
		}			
	}	
}
