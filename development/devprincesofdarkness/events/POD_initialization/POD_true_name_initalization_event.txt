namespace = POD_true_name_initalization

POD_true_name_initalization.1 = {
	hidden = yes
	type = empty
	
	immediate = {
		set_global_variable = {
			name = true_name_count
			value = 10000000
		}
		every_living_character = {
			trigger_event = POD_true_name_initalization.100
		}
	}
}

POD_true_name_initalization.100 = {
	type = character_event
	hidden = yes
	
	trigger = {
		NOT = { exists = var:true_name }
	}
	
	immediate = {
		random_list = {
			1 = {
				change_global_variable = {
					name = true_name_count
					subtract = 1
				}
			}
			1 = {
				change_global_variable = {
					name = true_name_count
					subtract = 2
				}
			}
			1 = {
				change_global_variable = {
					name = true_name_count
					subtract = 3
				}
			}
			1 = {
				change_global_variable = {
					name = true_name_count
					subtract = 4
				}
			}
			1 = {
				change_global_variable = {
					name = true_name_count
					subtract = 5
				}
			}
			1 = {
				change_global_variable = {
					name = true_name_count
					subtract = 6
				}
			}
			1 = {
				change_global_variable = {
					name = true_name_count
					subtract = 7
				}
			}
			1 = {
				change_global_variable = {
					name = true_name_count
					subtract = 8
				}
			}
			1 = {
				change_global_variable = {
					name = true_name_count
					subtract = 9
				}
			}
		}
		set_variable = {
			name = true_name
			value = global_var:true_name_count
		}
	}
}
