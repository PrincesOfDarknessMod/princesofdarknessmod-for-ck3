namespace = POD_world_initialization

POD_world_initialization.1 = {
	hidden = yes
	type = empty
	
	trigger = {
		NOT = { has_global_variable = variable_POD_world_county_faith_initialization }
		#NOT = { has_game_rule = world_of_mortals }
		#NOT = { has_game_rule = world_of_mortals_player_knows}
		# has_game_rule = world_of_minions
		#has_game_rule = world_of_minions_and_monsters
		has_game_rule = world_of_monsters
		OR = {
			AND = {
				current_date >= 1230.1.3
				current_date < 1232.1.6
			}
			AND = {
				current_date >= 1375.7.4
				current_date < 1375.7.6
			}		
		}
	}
	
	immediate = {
		
		POD_county_faith_initialization_effect = yes
		
		set_global_variable = {
			name = variable_POD_world_county_faith_initialization
			value = yes
		}
	
		every_independent_ruler = {	
	
			trigger_event = POD_character_initialization.1 
			
			every_vassal_or_below = {
				trigger_event = POD_character_initialization.1 
			}
		}	
		
	}	
}

#Kill characters of unwanted splats depending on game start
POD_world_initialization.2 = {
	hidden = yes
	type = empty
	
	immediate = {
	#WW and hunters and co are killed in vamps startdate
		if = {
			limit = {
				OR = {
					game_start_date = 1230.1.4
					game_start_date = 1510.1.3
					}
				}
		hidden_effect = {
		every_living_character = {
		limit = {
		OR = {
			has_trait = supehunter
			has_trait = werewolf
			has_trait = mummy
			has_demon_trait_trigger = yes
			POD_is_wraith_andco_trigger = yes			
		}
			is_ai = yes
		}
		 death = { death_reason = death_disappearance }
			if = {
				limit = {
					has_trait = kinfolk
				}
				remove_trait = kinfolk
			}
		}
		}	
	}
	#WW are killed in hunters startdate
		if = {
			limit = {
				game_start_date = 1230.1.5
				}
		hidden_effect = {
		every_living_character = {
		limit = {
			OR = {
			has_trait = werewolf
			has_trait = kinfolk
			}
			is_ai = yes
		}
		 death = { death_reason = death_disappearance }
		}
		}	
	}	
	}
	
}
