#Guest vampires are kicked out of Inqui courts
POD_maintenance.4 = {
	hidden = yes
	type = character_event

	trigger = {
		scope:host = {
			has_government = inquisition_government
			}
		scope:guest = { 
			OR = { 
			is_vampire_trigger = yes
			has_trait = revenant
			has_trait = ghoul
				}
			}
		}

	immediate =	
	{	
		scope:guest = { 
			select_and_move_to_pool_effect = yes
				}
	}
}


#AI vampires lose stress traits
POD_maintenance.5 = {
	hidden = yes
	type = character_event

	trigger = {
			is_landed = yes
			is_vampire_trigger = yes
			is_ai = yes			
		}

	immediate =	
	{	
		if = {
			limit = {
				has_trait = drunkard
				}
			remove_trait = drunkard
		}		
		else_if = {
			limit = {
				has_trait = comfort_eater
				}
			remove_trait = comfort_eater
		}
		else_if = {
			limit = {
				has_trait = inappetetic
				}
			remove_trait = inappetetic
		}		
		else_if = {
			limit = {
				has_trait = hashishiyah
				}
			remove_trait = hashishiyah
		}
		else_if = {
			limit = {
				has_trait = rakish
				}
			remove_trait = rakish
		}	
		else_if = {
			limit = {
				has_trait = flagellant
				}
			remove_trait = flagellant
		}
		else_if = {
			limit = {
				has_trait = profligate
				}
			remove_trait = profligate
		}
		else_if = {
			limit = {
				has_trait = improvident
				}
			remove_trait = improvident
		}		
	}
}


#Vampire landholders are ousted when a hunter win a war 
POD_maintenance.6 = {
	hidden = yes
	type = character_event

	trigger = {		
		}

	immediate =	
	{
	every_sub_realm_county = {
		limit = {
				holder = {
					OR = {
						is_vampire_trigger = yes
						has_trait = ghoul
						has_trait = revenant
						}
						is_ai = yes	
					}
				}
			
			save_scope_as = target_title
			add_to_list = target_title_counties
			}
		
		create_title_and_vassal_change = {
			type = revoked
			save_scope_as = change
			add_claim_on_loss = yes
		}	
		
		every_in_list = {
			list = target_title_counties
			limit = {
				holder = {
					target_is_liege_or_above = root
				}
			}
			change_title_holder = {
				holder = root
				change = scope:change
			}
		}
		
		resolve_title_and_vassal_change = scope:change		
	}
}


#Vampire realms get feudal elective if not already active
POD_maintenance.7 = {
	hidden = yes
	type = character_event

	trigger = {	
		is_landed = yes
		highest_held_title_tier >= tier_duchy
		is_vampire_trigger = yes
		}

	immediate =	
	{

		every_held_title = {
			limit = {
			OR = {
				tier = tier_empire
				tier = tier_duchy
				tier = tier_kingdom
				}
				NOT = {
					has_title_law = feudal_elective_succession_law
				}
			}
			add_title_law = feudal_elective_succession_law
		}
	}
}
