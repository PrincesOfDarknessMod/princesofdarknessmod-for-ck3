POD_ghoul_maintenance.100 = {
	type = character_event
	hidden = yes

    immediate = {
		if = {
			limit = {
				is_ai = yes	
				POD_is_ghoul_trigger = yes
			}
			if = {
				limit = {
					has_trait = ghoul 
				}
				remove_trait = ghoul
				add_trait = mortal
			}
			if = {
				limit = {
					any_secret = { secret_type = secret_ghoul }
				}
				random_secret = {
					limit = { secret_type = secret_ghoul }
						remove_secret = yes
				}
			}
		}
	}	
}

POD_ghoul_maintenance.300 = {
	type = character_event
	title = POD_ghoul_maintenance.300.t
	desc = POD_ghoul_maintenance.300.desc

	theme = vampire
	right_portrait = root
	
	trigger = {
		has_character_flag = POD_will_become_revenant
	}
	
	option = {
		name = POD_ghoul_maintenance.300.option.a	
		
		if = {
			limit = {
				NOT = { has_game_rule = world_of_monsters }
			}
			add_secret = {
				type = secret_revenant
			}
			remove_character_flag = POD_mortal_race_flag
		}
		
		if = {
			limit = {
				has_game_rule = world_of_monsters
			}
			hidden_effect = {
				if = {
					limit = {
						has_trait = mortal 
					}
					remove_trait = mortal
				}
			}
			add_trait = revenant			
			POD_character_revenant_faith_initialization_effect = yes
		}		
		POD_character_revenant_discipline_initialization_effect = yes
		if = {
			limit = {
				has_masquerade_risk_trigger = no
			}
			add_character_modifier = {
				modifier = masquerade0_modifier 
			}
		}
		if = {
			limit = {
				has_blood_hunger_trigger = no
			}
			add_character_modifier = {
				modifier = bloodhunger0_modifier 
			}
		}
			
		if = {
			limit = {
				NOT = { has_character_modifier = bloodpotency0_modifier }
			}
			add_character_modifier = {
				modifier = bloodpotency0_modifier 
			}
		}		
	}
}
