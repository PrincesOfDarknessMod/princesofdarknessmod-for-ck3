POD_mesmerize.100 = {
	type = character_event
	title = POD_mesmerize.100.t
	desc =  POD_mesmerize.100.desc		
	theme = dominate
	right_portrait = scope:scheme.scheme_owner
	left_portrait = scope:scheme.scheme_target
	
	immediate = {
		scope:scheme.scheme_target = {
			save_scope_as = target
		}
		scope:scheme.scheme_owner = {
			save_scope_as = actor
		}
	}
	option = {
		name = POD_mesmerize.100.option.a
		trigger = {
			stress < very_high_stress
		}
		
		POD_willpower_rouse_check_effect = yes
		
		save_scope_value_as = {
			name = mental_duel_defender_will
			value = flag:willpower_spent
		}
		
		scope:scheme.scheme_owner = {
			trigger_event = POD_mesmerize.200
		}
		
		ai_chance = {
			base = 95
		}	
	}

	option = {
		name = POD_mesmerize.100.option.b
		save_scope_value_as = {
			name = mental_duel_defender_will
			value = flag:willpower_unspent
		}
		scope:scheme.scheme_owner = {
			trigger_event = POD_mesmerize.200
		}		
	}	
}

POD_mesmerize.300 = {
	type = character_event
	title = POD_mesmerize.300.t
	desc = { 
		desc = POD_mesmerize.300.desc
		triggered_desc = {
			trigger = {
				NOT = { has_perk = submerged_directive_perk }
			}
			desc = POD_mesmerize.300.desc.short_term
		}
	}
	theme = dominate
	right_portrait = scope:scheme.scheme_owner
	left_portrait = scope:scheme.scheme_target

	option = {
		name = POD_mesmerize.300.option.a
		scope:scheme.scheme_owner = {
			if = {
				limit = {
					has_perk = submerged_directive_perk
				}
				add_hook = {
					target = scope:scheme.scheme_target 
					type = weak_dominate_hook
					days = 3650
				}
				trigger_event = { id = POD_mesmerize.700 days = 3650 } 
				
			}
			else = {
				add_hook = {
					target = scope:scheme.scheme_target 
					type = weak_dominate_hook
				}
				trigger_event = { id = POD_mesmerize.700 days = 5 } 
			}
			add_to_variable_list = {
				name = dominated_targets
				target = scope:scheme.scheme_target 
			}
			
		}

		scope:scheme.scheme_target = {
			trigger_event = POD_mesmerize.400
		
		}
		
		
		
		custom_tooltip = POD_mesmerize.300.option.a.tt.beast
	}		
}

POD_mesmerize.700 = {
	type = character_event
	hidden = yes

	immediate = {
		if = {
			limit = { 
				has_variable_list = dominated_targets 
				trigger_if = {
					limit = { has_variable_list = dominated_targets } 
					AND = {
						has_variable_list = dominated_targets
						is_target_in_variable_list = {
							name = dominated_targets
							target = scope:scheme.scheme_target
						}
					}
				}
				
			
			}
			remove_list_variable = {
				name = dominated_targets
				target = scope:scheme.scheme_target
			}
			send_interface_toast = {
				title = POD_mesmerize.700_toast_title
				left_icon = root
				right_icon = scope:scheme.scheme_target
				custom_tooltip = POD_mesmerize.700_toast_ct
				
			}
		}
		
	}
	
}
