POD_inquihunt.2 = {
	type = character_event
	title = POD_inquihunt.2.t
	desc =  POD_inquihunt.2.desc		
	theme = faith
	left_portrait = {

	}
	
	right_portrait = {
		character = scope:scheme.scheme_target
		animation = fear
	}
	
	option = {
		name = POD_inquihunt.2.option.a
		
		add_piety = 200
		add_prestige = 200

				if = {
					limit = {
						scope:scheme.scheme_target = { 
							age >= 1000
						}
						exists = dynasty
					}
					dynasty = {
						add_dynasty_prestige = major_dynasty_prestige_gain
					}
				}
				
				else_if = {
					limit = {
						scope:scheme.scheme_target = { 
							age >= 1000
						}
						exists = liege.dynasty
					}
					liege.dynasty = {
						add_dynasty_prestige = major_dynasty_prestige_gain
					}
				}
				
				if = {
					limit = {
						scope:scheme.scheme_target = { 
							age < 1000
						}
						scope:scheme.scheme_target = { 
							age >= 400
						}
						exists = dynasty
					}
					dynasty = {
						add_dynasty_prestige = medium_dynasty_prestige_gain
					}
				}
				
				else_if = {
					limit = {
						scope:scheme.scheme_target = { 
							age < 1000
						}
						scope:scheme.scheme_target = { 
							age >= 400
						}
						exists = liege.dynasty
					}
					liege.dynasty = {
						add_dynasty_prestige = medium_dynasty_prestige_gain
					}
				}
				
				if = {
					limit = {
						scope:scheme.scheme_target = { 
							age < 400
						}
						scope:scheme.scheme_target = { 
							age >= 200
						}
						exists = dynasty
					}
					dynasty = {
						add_dynasty_prestige = minor_dynasty_prestige_gain
					}
				}
				
				else_if = {
					limit = {
						scope:scheme.scheme_target = { 
							age < 400
						}
						scope:scheme.scheme_target = { 
							age >= 200
						}
						exists = liege.dynasty
					}
					liege.dynasty = {
						add_dynasty_prestige = minor_dynasty_prestige_gain
					}
				}
				
				if = {
					limit = {
						scope:scheme.scheme_target = { 
							age < 200
						}
						exists = dynasty
					}
					dynasty = { 
						add_dynasty_prestige = miniscule_dynasty_prestige_gain
					}
				}
				
				else_if = {
					limit = {
						scope:scheme.scheme_target = { 
							age < 200
						}
						exists = liege.dynasty
					}
					liege.dynasty = { 
						add_dynasty_prestige = miniscule_dynasty_prestige_gain
					}
				}

custom_tooltip = POD_inquihunt_effect_message

hidden_effect = {
	if = {
		limit = {
			OR = {
				has_trait = truefaith
				
				}
			}	
		if = {
			limit = {
				scope:defender = { 
					age >= 1000
					}
				} 
			add_faith_lifestyle_xp = 2000	
		}
		else_if = {
			limit = {
				scope:defender = { 
					age < 1000
				}
				scope:defender = { 
					age >= 400
				}	
			} 
			add_faith_lifestyle_xp = 1000	
		}
		else_if = {
			limit = {
				scope:defender = { 
					age < 400
				}
				scope:defender = { 
					age >= 200
				} 
			}
			add_faith_lifestyle_xp = 500	
		}	
		else = {
			add_faith_lifestyle_xp = 250
		}
	}
}
	scope:defender = {
			death = {
	        	death_reason = death_hunted_diplomacy
				killer = scope:attacker
	        }
		}
		
	scope:attacker = {
			scope:scheme = {
				end_scheme = yes
			}
		}	
	}
}
