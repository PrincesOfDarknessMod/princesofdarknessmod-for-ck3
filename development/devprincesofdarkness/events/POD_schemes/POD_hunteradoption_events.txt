POD_hunteradoption.100 = {
	type = character_event
	title = POD_hunteradoption.100.t
	desc =  POD_hunteradoption.100.desc		
	theme = faith
	left_portrait = {
		character = root
		animation = personality_honorable
	}
	
	right_portrait = {
		character = scope:target
		animation = personality_bold
	}
	
	immediate = {
		scope:target = {
			save_scope_as = victim		 
		}
		
		root = {
			save_scope_as = vampiresire 
		}
		
		scope:victim = { POD_set_parent_effect = { PARENT = scope:vampiresire } }
	}
	
	option = {
		name = POD_hunteradoption.100.option.a
		custom_tooltip = POD_hunteradoption.100.option.a.tt

		
		stress_impact = {
			base = medium_stress_gain
		}

		scope:target = {
			if = {
				limit = {
				exists = scope:vampiresire.dynasty
				scope:vampiresire.dynasty = { has_dynasty_perk = unity_legacy_4 }
				}
			add_martial_skill = 4
			}
			else = {
			add_martial_skill = 2
			}
			remove_trait = mortal
			if = {
				limit = {
					has_trait = bastard
					}
				remove_trait = bastard
				}
			add_trait = supehunter
			if = {
				limit = {
				scope:vampiresire = { has_trait = order_member }
				scope:target = {
				NOT = { has_trait = order_member }
						}
					}
			add_trait = order_member
				}
			set_character_faith = scope:vampiresire.faith
			set_house = scope:vampiresire.house
			if = {
				limit = {
				has_orison_trigger = no
				}
				POD_give_orison_effect = yes
			}
			if = {
				limit = {
					has_trait = sorcerer
				}
			remove_trait = sorcerer
			}
			if = {
				limit = {
					has_trait = theurgist
					NOT = { has_trait = truefaith }
				}	
			remove_trait = theurgist
			add_trait = truefaith
			}
			else_if = {
				limit = {
					has_trait = theurgist
					has_trait = truefaith
				}	
			remove_trait = theurgist
			}
			if = {
				limit = {
					NOT = { has_trait = truefaith }
				}
				random = {
					chance = 10
					add_trait = truefaith
				}
			}
		}		
		
		hidden_effect = {
			scope:scheme = {
				end_scheme = yes
			}
		}	
	}
