namespace = POD_hunteradoption

POD_hunteradoption.100 =
{
	type = character_event
	title = POD_hunteradoption.100.t
	desc =  POD_hunteradoption.100.desc		
	theme = faith
	left_portrait = {
		character = root
		animation = personality_honorable
	}
	
	right_portrait = {
		character = scope:target
		animation = personality_bold
	}
	
	immediate = {
		scope:target = {
			save_scope_as = victim		 
		}
		
		root = {
			save_scope_as = vampiresire 
		}
		
		
		scope:vampiresire = {
			if = {
				limit = {
					is_female = yes
				}
				scope:victim = { 
					set_mother = scope:vampiresire 
				}
			}
			if = {
				limit = {
					is_female = no
				}
				scope:victim = { 
					set_father = scope:vampiresire 
				}
			}			
		}

		scope:target = {
			save_scope_as = created_vampirechilde
		}
	}
	
	option = {
		name = POD_hunteradoption.100.option.a
		custom_tooltip = POD_hunteradoption.100.option.a.tt

		
		stress_impact = {
			base = medium_stress_gain
		}

		scope:target = {
			add_martial_skill = 2	
			remove_trait = mortal
			if = {
				limit = {
					has_trait = bastard
					}
				remove_trait = bastard
				}
			add_trait = supehunter
			if = {
				limit = {
				scope:vampiresire = { has_trait = order_member }
					}
			add_trait = order_member
				}
			set_character_faith = scope:vampiresire.faith
			set_house = scope:vampiresire.house	
			random = {
				chance = 10
				add_trait = truefaith
			}	
		}		
		
		hidden_effect = {
			scope:scheme = {
				end_scheme = yes
			}
		}	
	}
	
	option = {
		name = POD_hunteradoption.100.option.b
		custom_tooltip = POD_hunteradoption.100.option.b.tt

		
		stress_impact = {
			base = medium_stress_gain
		}

		scope:target = {
			add_prowess_skill = 5	
			remove_trait = mortal
			if = {
				limit = {
					has_trait = bastard
					}
				remove_trait = bastard
				}
			add_trait = supehunter	
			if = {
				limit = {
				scope:vampiresire = { has_trait = order_member }
					}
			add_trait = order_member
				}
			set_character_faith = scope:vampiresire.faith
			set_house = scope:vampiresire.house	
			random = {
				chance = 10
				add_trait = truefaith
			}
		}		
		
		hidden_effect = {
			scope:scheme = {
				end_scheme = yes
			}
		}	
	}
	option = {
		name = POD_hunteradoption.100.option.c
		custom_tooltip = POD_hunteradoption.100.option.c.tt

		trigger = {
			dynasty = dynasty:13000006 
		}
		
		stress_impact = {
			base = medium_stress_gain
		}

		scope:target = {
			add_intrigue_skill = 2	
			remove_trait = mortal
			if = {
				limit = {
					has_trait = bastard
					}
				remove_trait = bastard
				}
			add_trait = supehunter
			if = {
				limit = {
				scope:vampiresire = { has_trait = order_member }
					}
			add_trait = order_member
				}	
			set_character_faith = scope:vampiresire.faith
			set_house = scope:vampiresire.house	
			random = {
				chance = 10
				add_trait = truefaith
			}		
		}		
		
		hidden_effect = {
			scope:scheme = {
				end_scheme = yes
			}
		}	
	}	
}
