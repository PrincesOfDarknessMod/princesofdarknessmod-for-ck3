POD_rationalize.100 = {
	type = character_event
	title = POD_rationalize.100.t
	desc =  POD_rationalize.100.desc		
	theme = dominate
	right_portrait = scope:scheme.scheme_owner
	left_portrait = scope:scheme.scheme_target
	
	
	option = {
		name = POD_rationalize.100.option.a
		trigger = {
			stress < very_high_stress
		}
		
		POD_willpower_rouse_check_effect = yes
		
		save_scope_value_as = {
			name = mental_duel_defender_will
			value = flag:willpower_spent
		}
		
		scope:scheme.scheme_owner = {
			trigger_event = POD_rationalize.200
		}
		
		ai_chance = {
			base = 95
		}	
	}

	option = {
		name = POD_rationalize.100.option.b
		save_scope_value_as = {
			name = mental_duel_defender_will
			value = flag:willpower_unspent
		}
		scope:scheme.scheme_owner = {
			trigger_event = POD_rationalize.200
		}		
	}	
}

POD_rationalize.300 = {
	type = character_event
	title = POD_rationalize.300.t
	desc =  POD_rationalize.300.desc		
	theme = dominate
	right_portrait = scope:scheme.scheme_owner
	left_portrait = scope:scheme.scheme_target

	option = {
		name = POD_rationalize.300.option.a
		scope:scheme.scheme_owner = {
			add_hook = {
				target = scope:scheme.scheme_target 
				type = rationalize_hook
			}	
		}

		scope:scheme.scheme_target = {
			trigger_event = POD_rationalize.400
		
		}
		
		
		
		custom_tooltip = POD_rationalize.300.option.a.tt.beast
	}		
}
