namespace = POD_summon

POD_summon.100 = {
	type = character_event
	title = POD_summon.100.t
	desc =  POD_summon.100.desc		
	theme = presence
	right_portrait = scope:scheme.scheme_owner
	left_portrait = scope:scheme.scheme_target
	
	
	option = {
		name = POD_summon.100.option.a
		trigger = {
			stress < very_high_stress
		}
		
		POD_willpower_rouse_check_effect = yes
		
		save_scope_value_as = {
			name = mental_duel_defender_will
			value = flag:willpower_spent
		}
		
		scope:scheme.scheme_owner = {
			trigger_event = POD_summon.200
		}
		
		ai_chance = {
			base = 95
			modifier = {
				add = 1000
				has_trait = paranoid
			}
		}	
	}

	option = {
		name = POD_summon.100.option.b
		save_scope_value_as = {
			name = mental_duel_defender_will
			value = flag:willpower_unspent
		}
		scope:scheme.scheme_owner = {
			trigger_event = POD_summon.200
		}		
	}	
	
	option = {
		name = POD_summon.100.option.c
		scope:scheme.scheme_owner = {
			trigger_event = POD_summon.300
		}
		
		stress_impact = {
			reclusive = minor_stress_gain
		}			
		ai_chance = {
			base = 5
			opinion_modifier = {
				opinion_target = scope:sender
			}			
			modifier = {
				add = -100
				has_trait = paranoid
			}
		}			
	}	
}

# Scheme Success, Summoner Notified

POD_summon.300 = {
	type = character_event
	title = POD_summon.300.t
	desc =  POD_summon.300.desc		
	theme = presence
	right_portrait = scope:scheme.scheme_owner
	left_portrait = scope:scheme.scheme_target

	option = {
		name = POD_summon.300.option.a
		
		scope:scheme.scheme_target = {
			set_employer = scope:scheme.scheme_owner
			return_to_court = yes
			add_character_flag = {
				flag = blocked_from_leaving
				years = 5
			}
			trigger_event = POD_summon.400
		}
		custom_tooltip = POD_summon.300.option.a.tt.beast
	}		
}
