kueijin_scheme.100 = {
	type = character_event
	title = kueijin_scheme.100.t
	desc = kueijin_scheme.100.desc
	theme = vampire

	left_portrait = {
		character = root
		animation = anger
	}
	
	right_portrait = {
		character = scope:scheme.scheme_target
		animation = fear
	}
	
	immediate = {
		scope:scheme.scheme_target = {
			save_scope_as = target	
		}
		
		scope:scheme.scheme_owner= {
			save_scope_as = caster
		}
	}

	# Heal
	option = {
		name = kueijin_scheme.100.a
		
		trigger_event = kueijin_scheme.110
		
		ai_chance = {
			base = 20
			modifier = {
				factor = 2
				has_any_good_relationship_with_character_trigger = {
					CHARACTER = scope:target
				}
			}
			modifier = {
				factor = 2
				has_trait = compassionate
			}
			modifier = {
				factor = 1.5
				is_diablerist_trigger = no
			}
		}
	}

	# Torture
	option = {
		name = kueijin_scheme.100.b
		
		duel = {
			skill = prowess
			value = scope:target.prowess
			desc = POD_diablerize.500.option.a.desc
			
			50 = {
				desc = POD_diablerize.500.option.a.success

				compare_modifier = {
					value = scope:duel_value
					multiplier = high_positive_duel_skill_multiplier
				}
				
				custom_tooltip = POD_diablerize.500.option.a.tt.attacker_wins
		
				trigger_event = kueijin_scheme.120
			}
			50 = {
				desc = POD_diablerize.500.option.a.failure

				compare_modifier = {
					value = scope:duel_value
					multiplier = low_negative_duel_skill_multiplier 
				}
				
				custom_tooltip = POD_diablerize.500.option.a.tt.defender_wins
			}
		}
		
		ai_chance = {
			base = 20
			modifier = {
				factor = 2
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:target
				}
			}
			modifier = {
				factor = 2
				has_trait = sadistic
			}
			modifier = {
				factor = 1.5
				is_diablerist_trigger = no
			}
		}
	}

	# Eat Soul
	option = {
		name = kueijin_scheme.100.c
		trigger = {
			scope:caster = {
				trigger_if = {
					limit = {
						is_parent_of = scope:target
						NOT = {
							has_trait_with_flag = can_murder_own_children
						}
					}
					NOT = { is_parent_of = scope:target }
				}
			}
		}
			
		if = {
			limit = {
				NOR = {
					faith = faith:deviltiger
					faith = faith:wayofthetenthousandscreams
				}
			}
			POD_decrease_dharma_effect = yes
			add_piety = major_piety_loss
		}
		else = {
			add_piety = major_piety_value
		}
		
		configure_start_single_combat_effect = {
			SC_INITIATOR = scope:caster
			SC_ATTACKER = scope:caster
			SC_DEFENDER = scope:target
			FATALITY = no
			FIXED = no
			LOCALE = throne_room
			OUTPUT_EVENT = kueijin_scheme.101
			INVALIDATION_EVENT = tribal.1022
		}
		
		ai_chance = {
			base = 0
			modifier = {
				factor = 2
				is_diablerist_trigger = yes
			}
			modifier = {
				factor = 1.5
				has_trait = ambitious
			}			
		}
	}
}

#Who won the final duel?
kueijin_scheme.101 = {
	hidden = yes

	immediate = {
		# If any clothes were stripped, put 'em back on.
		single_combat_clean_shirtlessness_effect = {
			ATTACKER = scope:caster
			DEFENDER = scope:target
		}
		# Did win?
		if = {
			limit = { scope:sc_victor = scope:caster }
			
			#Reroot to the old diablerie finish
			scope:caster = {
				trigger_event = {
					id = kueijin_scheme.130
					days = 1
				}
			}
		}
		# Did lose?
		else_if = {
			scope:caster = {
				 death = {
	        		 death_reason = death_duel
					 killer = scope:target
	        	 }
			 }
		} 
	}	
}
