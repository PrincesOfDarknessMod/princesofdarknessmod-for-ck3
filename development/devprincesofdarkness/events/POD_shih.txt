namespace = podshih

podshih.100 = {
	type = character_event
	title = podshih.100.t
	desc = podshih.100.desc
	theme = faith
	
	trigger = {
		is_independent_ruler = yes
	}

	option = {
		name = podshih.100.a
		
		add_truce_both_ways = {
			character = scope:shih
			years = 3
			name = TRUCE_SIGNED_SHIH
		}
		
		ai_chance = {
			base = 0
			modifier = {
				add = 50
				scope:shih = { is_ai = no } 
			}

			ai_value_modifier = {
				ai_compassion = 200
				ai_honor = 200
			}
		}
	}
	
	option = {
		name = podshih.100.b
	}
}

podshih.1000 = {
	type = character_event
	title = podshih.1000.t
	desc = podshih.1000.desc
	theme = faith

	option = {
		name = podshih.1000.a
		custom_tooltip = podshih.1000.a.tt
		
		set_global_variable = {
			name = celestial_army_restored
			value = yes
		}
		change_government = celestial_army_government
		hidden_effect = {
			every_vassal_or_below = {
				limit = {
					has_trait = shih
				}
				change_government = celestial_army_government
			}
		}
	}
	
	option = {
		name = podshih.1000.b
	
		trigger = {
			NOT = { culture = { has_cultural_era_or_later = culture_era_final_nights } }
		}
		
		add_character_modifier = shih_peace_modifier
	}
}
