namespace = pod_lasombraevents

#Control event for Creed flavor stuff

pod_lasombraevents.1001 = {
	type = character_event
	hidden = yes
	
	immediate = {
	
		if = {
			limit = {
					is_vampire_trigger = yes
					is_imprisoned = no
					NOT = {
						has_trait = torpor
						}
					has_trait = lasombra
					age > 150
				}
			trigger_event = pod_lasombraevents.1	
		}
	}
}

	type = character_event
	title = pod_lasombraevents.1.n
	desc = {

			triggered_desc = {
				trigger = {
					exists = scope:helper
				}
				desc = pod_lasombraevents.1.loverdesc
			}

			triggered_desc = {
				desc = pod_lasombraevents.1.desc
			}

		}
	}
	theme = lasombra
	left_portrait = {
		character = root
		animation = stress
		triggered_animation = {
			trigger = {
				has_trait = arrogant
			}
			animation = personality_bold
		}
	}	
	right_portrait = {
			character = scope:helper
			animation = personality_compassionate
		}

	trigger = {
	}

	immediate = {
		random_relation = {
			type = lover
			save_scope_as = helper
		}
		
	}

	option = { 
		name = pod_lasombraevents.1.a

		custom_tooltip = pod_lasombraevents.1.a.t
		add_character_modifier = {
            modifier = pareidolia_modifier
            years = 1
        }
        remove_short_term_gold = minor_gold_value
		add_stress = 30
	}

	option = { 
		name = pod_lasombraevents.1.b
		trigger = {
			NOT = { 
				has_trait = arrogant

			}
		}

		custom_tooltip = pod_lasombraevents.1.b.t

            modifier = pareidolia_modifier
            years = 5
        }
		add_stress = 70

	}

	option = {

			has_trait = arrogant
		}

		custom_tooltip = pod_lasombraevents.1.c.t

		}

	option = {
		name = pod_lasombraevents.1.d

		}

		custom_tooltip = pod_lasombraevents.1.d.t
