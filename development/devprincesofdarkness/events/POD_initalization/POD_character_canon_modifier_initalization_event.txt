namespace = POD_character_canon_modifier_initalization

POD_character_canon_modifier_initalization.1 = {
	hidden = yes
	type = empty
	
	trigger = {
	}
	
	immediate = {	
		every_living_character = {
		if = {
		limit = {
			has_POD_vamp_blood_trigger = yes
			has_blood_potency_trigger = no
			has_masquerade_risk_trigger = no
		}
		trigger_event = POD_character_canon_modifier_initalization.100
		}
		}
	}
}

POD_character_canon_modifier_initalization.100 = {
	type = character_event
	hidden = yes
	
	trigger = {
			has_POD_vamp_blood_trigger = yes
		OR = {
			has_blood_potency_trigger = no
			has_masquerade_risk_trigger = no
		}
	}
	
	immediate = {
	
		#POD_clear_resonance_effect = yes
		POD_set_resonance_effect = yes
		
		if = {
			limit = {
				has_blood_potency_trigger = no
			}
			
			if = { 
				limit = {
					POD_is_vampire_trigger = yes 
				}
				POD_character_vampire_bloodpotency_initialization_effect = yes
			}
			if = { 
				limit = {
					OR = {
						POD_is_revenant_trigger = yes
						POD_is_ghoul_trigger = yes
					}
				}
				add_character_modifier = {
					modifier = bloodpotency0_modifier 
				}
			}		
		}
		
		if = {
			limit = {
				has_masquerade_risk_trigger = no
			}
			add_character_modifier = {
				modifier = masquerade0_modifier 
			}	
		}
		
		if = {
			limit = {
				has_blood_hunger_trigger = no
			}
			if = {
				limit = {
					POD_is_vampire_trigger = yes
				}
				add_character_modifier = {
					modifier = bloodhunger1_modifier
				}
			}
			if = {
				limit = {
					OR = {
						POD_is_revenant_trigger = yes
						POD_is_ghoul_trigger = yes
					}
				}
				add_character_modifier = {
					modifier = bloodhunger0_modifier
				}
			}					
		}
		
		if = {
			limit = {
				POD_is_vampire_trigger = yes
				has_predator_type_trigger = no
				is_ai = yes 
			}
			POD_character_vampire_predatortype_initialization_effect = yes
		}

		if = {
			limit = {
				is_ventrue_trigger = yes 
				is_ai = yes
				has_prey_exclusion_trigger = no
			}
			POD_character_ventrue_prey_exclusion_initialization_effect = yes
		}

		if = {
			limit = {
				POD_is_vampire_trigger = yes
				has_predator_type_trigger = no
				is_ai = no
			}
			if = {
				limit = {has_trait = ventrue}
				trigger_event = { 
					id = POD_chargen.1100 
					days = 5
				}
			}
			else = {
				trigger_event = { 
					id = POD_chargen.1020 
					days = 5
				}
			}
		}				
		
		if = {
			limit = {
				is_nosferatu_trigger = yes 
				NOT = { has_character_flag = had_POD_character_nosferatu_looks }
			}
			POD_character_vampire_nosferatu_initialization_effect = yes
		}
		
		#initialize lifestyle perks
		if = {
			limit = {
				NOT = { has_character_flag = had_POD_character_lifestyle }
			}
		POD_character_vampire_lifestyle_initialization_effect = yes
		}		
	}
}
