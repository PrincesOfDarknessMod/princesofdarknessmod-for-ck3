namespace = POD_maintenance

POD_maintenance.1 = {
	hidden = yes
	type = empty

	immediate =	
	{	
		every_ruler = {
			if = {
				limit = {
						is_ai = yes 
						is_vampire_trigger = yes
					}			
				trigger_event = POD_maintenance.2
			}
		}
		every_ruler = {
			if = {
				limit = {
						is_ai = yes 
						has_trait = supehunter
					}			
				trigger_event = POD_maintenance.3
			}
		}	
		every_living_character = {
			if = {
				limit = {
						is_ai = yes 
						is_vampire_trigger = yes
						has_faith = faith:shadowinquisition
					}			
				set_character_faith = faith:viahumanitatis
			}
		}			
	}
}

POD_maintenance.2 = {
	type = character_event
	hidden = yes

	immediate =	
	{	
		every_realm_county = { 
			limit = {
				any_county_province = {
					NOT = { has_holding_type = castle_holding }
				}
			}
			every_county_province = {
				 limit = {
					NOT = { has_holding_type = castle_holding }
					NOT = { has_holding_type = tribal_holding }
				 }
				set_holding_type = tribal_holding
			}
		}
		every_realm_county = { 
			limit = {
				any_county_province = {
					has_holding_type = tribal_holding 
				}
			}
			every_county_province = {
				 limit = {
					has_holding_type = tribal_holding 
				 }
				set_holding_type = castle_holding
			}
		}		
	}
}

#Guest vampires are kicked out of Inqui courts
POD_maintenance.4 = {
	hidden = yes
	type = character_event

	trigger = {
		scope:host = {
			has_government = inquisition_government
			}
		scope:guest = { 
			OR = { 
			is_vampire_trigger = yes
			has_trait = revenant
			has_trait = ghoul
				}
			}
		}

	immediate =	
	{	
		scope:guest = { 
			select_and_move_to_pool_effect = yes
				}
	}
}


#AI vampires lose stress traits
POD_maintenance.5 = {
	hidden = yes
	type = character_event

	trigger = {
			is_landed = yes
			is_vampire_trigger = yes
			is_ai = yes			
		}

	immediate =	
	{	
		if = {
			limit = {
				has_trait = drunkard
				}
			remove_trait = drunkard
		}		
		else_if = {
			limit = {
				has_trait = comfort_eater
				}
			remove_trait = comfort_eater
		}
		else_if = {
			limit = {
				has_trait = inappetetic
				}
			remove_trait = inappetetic
		}		
		else_if = {
			limit = {
				has_trait = hashishiyah
				}
			remove_trait = hashishiyah
		}
		else_if = {
			limit = {
				has_trait = rakish
				}
			remove_trait = rakish
		}	
		else_if = {
			limit = {
				has_trait = flagellant
				}
			remove_trait = flagellant
		}
		else_if = {
			limit = {
				has_trait = profligate
				}
			remove_trait = profligate
		}
		else_if = {
			limit = {
				has_trait = improvident
				}
			remove_trait = improvident
		}		
	}
}


#Vampire landholders are ousted when a hunter win a war 
POD_maintenance.6 = {
	hidden = yes
	type = character_event

	trigger = {		
		}

	immediate =	
	{
	every_sub_realm_county = {
		limit = {
				holder = {
					OR = {
						is_vampire_trigger = yes
						has_trait = ghoul
						has_trait = revenant
						}
						is_ai = yes	
					}
				}
			
			save_scope_as = target_title
			add_to_list = target_title_counties
			}
		
		create_title_and_vassal_change = {
			type = revoked
			save_scope_as = change
			add_claim_on_loss = yes
		}	
		
		every_in_list = {
			list = target_title_counties
			limit = {
				holder = {
					target_is_liege_or_above = root
				}
			}
			change_title_holder = {
				holder = root
				change = scope:change
			}
		}
		
		resolve_title_and_vassal_change = scope:change		
	}
}
