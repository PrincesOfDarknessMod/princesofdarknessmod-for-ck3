namespace = POD_ai_inquisitor_ordination

POD_ai_inquisitor_ordination.1 = {
	type = character_event
	hidden = yes

    trigger = {
		is_landed = yes
        is_vampire_trigger = no
		has_trait = supehunter
		age > 20
		is_imprisoned = no
		is_ai = yes
		OR = {
			any_child = { is_landed = no count < 1 }
			any_child = { is_adult = yes count < 1 }
			any_child = { POD_is_supehunter_trigger = yes count < 1 }
		}
		
		NOT = { dynasty = dynasty:13000004 }
    }

    immediate = {
		save_scope_as = masterhunter
		scope:masterhunter = {
			POD_ai_inquisitor_ordination_effect = yes
		}
	}
}

POD_ai_inquisitor_ordination.2 = {
	type = character_event
	hidden = yes

    trigger = {
        is_vampire_trigger = no
		POD_is_supehunter_trigger = no
		is_adult = yes
		OR = {
			dynasty = dynasty:13000004
			dynasty = dynasty:13000005
		}
    }

    immediate = {
		random = {
			chance = 10
			
			modifier = {
				add = 90
				is_ruler = yes
			}
			
			if = {
				limit = { has_trait = mortal }
				remove_trait = mortal
				add_trait = supehunter
			}
			
			random = {
				chance = 10
				add_trait = truefaith
			}
			
			set_character_faith = faith:shadowinquisition
		}
	}
}
