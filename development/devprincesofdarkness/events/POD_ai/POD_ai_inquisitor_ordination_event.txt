namespace = POD_ai_inquisitor_ordination

POD_ai_inquisitor_ordination.1 = {
	type = character_event
	hidden = yes

    trigger = {
		is_landed = yes
        is_vampire_trigger = no
		has_trait = supehunter
		age > 20
		is_imprisoned = no
		is_ai = yes
		OR = {
			any_child = { is_landed = no count < 1 }
			any_child = { is_adult = yes count < 1 }
			any_child = { POD_is_supehunter_trigger = yes count < 1 }
		}
		
		NOT = { dynasty = dynasty:13000004 }
    }

    immediate = {
		save_scope_as = masterhunter
		scope:masterhunter = {
			POD_ai_inquisitor_ordination_effect = yes	
		}
		scope:apprenticehunter = {
			if = {
				limit = {
					scope:masterhunter = {
					has_trait = order_member
					}
					exists = scope:masterhunter.dynasty
					scope:masterhunter.dynasty = { has_dynasty_perk = meritocracy_legacy_1 }
					is_female = no
					}
				meritocracy_1_effect = yes
			}
		
		if = {
				limit = {
					scope:masterhunter = {
					has_trait = order_member
					}
					exists = scope:masterhunter.dynasty
					scope:masterhunter.dynasty = { has_dynasty_perk = meritocracy_legacy_2 }
					is_female = no
					}
				meritocracy_2_effect_a = yes
			}
		
		if = {
				limit = {
					scope:masterhunter = {
					has_trait = order_member
					}
					exists = scope:masterhunter.dynasty
					scope:masterhunter.dynasty = { has_dynasty_perk = meritocracy_legacy_2 }
					is_female = yes
					}
				meritocracy_2_effect_b = yes
			}
		
		if = {
				limit = {
					scope:masterhunter = {
					has_trait = order_member
					}
					exists = scope:masterhunter.dynasty
					scope:masterhunter.dynasty = { has_dynasty_perk = meritocracy_legacy_3 }
					}
				meritocracy_3_effect = yes
			}
		
		if = {
				limit = {
					scope:masterhunter = {
					has_trait = order_member
					}
					exists = scope:masterhunter.dynasty
					scope:masterhunter.dynasty = { has_dynasty_perk = meritocracy_legacy_4 }
					}
				meritocracy_4_effect = yes
			}
			
		if = {
				limit = {
					scope:masterhunter = {
					has_trait = order_member
					}
					exists = scope:masterhunter.dynasty
					scope:masterhunter.dynasty = { has_dynasty_perk = meritocracy_legacy_5 }
					}
				meritocracy_5_effect = yes
			}
		}
	}
}

POD_ai_inquisitor_ordination.2 = {
	type = character_event
	hidden = yes

    trigger = {
        is_vampire_trigger = no
		POD_is_supehunter_trigger = no
		OR = {
			dynasty = dynasty:13000004
			dynasty = dynasty:13000005
		}
    }

    immediate = {
		random = {
			chance = 10
			
			modifier = {
				add = 90
				is_ruler = yes
			}
			
			if = {
				limit = { has_trait = mortal }
				remove_trait = mortal
				add_trait = supehunter
			}
			
			random = {
				chance = 10
				add_trait = truefaith
			}
			
			set_character_faith = faith:shadowinquisition
		}
	}
}

POD_ai_inquisitor_ordination.3 = {
	type = character_event
	hidden = yes

    trigger = {
        is_vampire_trigger = no
		NOT = { has_trait = supehunter }
		is_adult = yes
				NOR = {
				dynasty = dynasty:13000001
				dynasty = dynasty:13000002
				dynasty = dynasty:13000003
				dynasty = dynasty:13000004
				dynasty = dynasty:13000005
				dynasty = dynasty:13000006
				dynasty = dynasty:13000007
				dynasty = dynasty:13000008
				}
		
				OR = {
					AND = {
						exists = father
						father = { has_trait = supehunter }
							}
					AND = {
						exists = mother
						mother = { has_trait = supehunter }
							}
					AND = {
						exists = liege
						liege = { has_trait = supehunter }
							}
					}
				}
	
	immediate = {
		
			
			remove_trait = mortal
			add_trait = supehunter
			random = {
				chance = 10
				add_trait = truefaith
			}
			if = {
				limit = {
					OR = {
						mother = { faith = faith:shadowinquisition }
						father = { faith = faith:shadowinquisition }
							}
						}
					set_character_faith = faith:shadowinquisition
						}
			else = {
					set_character_faith = faith:independenthunters
					}
	}
}
