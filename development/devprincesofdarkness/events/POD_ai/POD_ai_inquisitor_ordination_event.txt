namespace = POD_ai_inquisitor_ordination

POD_ai_inquisitor_ordination.1 = {
	type = character_event
	hidden = yes

    trigger = {
		is_landed = yes
        is_vampire_trigger = no
		has_trait = supehunter
		age > 20
		is_imprisoned = no
		is_ai = yes
		OR = {
			any_child = { is_landed = no count < 2 }
			any_child = { is_adult = yes count < 2 }
			any_child = { POD_is_supehunter_trigger = yes count < 2 }
		}
		
		NOT = { dynasty = dynasty:13000004 }
    }

    immediate = {
		save_scope_as = masterhunter
		scope:masterhunter = {
			POD_ai_inquisitor_ordination_effect = yes	
		}
		scope:created_apprenticehunter = {
			if = {
				limit = {
					scope:masterhunter = {
					has_trait = order_member
					}
					exists = scope:masterhunter.dynasty
					scope:masterhunter.dynasty = { has_dynasty_perk = meritocracy_legacy_1 }
					}
				meritocracy_1_effect = yes
			}
		
		if = {
				limit = {
					scope:masterhunter = {
					has_trait = order_member
					}
					exists = scope:masterhunter.dynasty
					scope:masterhunter.dynasty = { has_dynasty_perk = meritocracy_legacy_2 }
					is_female = no
					}
				meritocracy_2_effect_a = yes
			}
		
		if = {
				limit = {
					scope:masterhunter = {
					has_trait = order_member
					}
					exists = scope:masterhunter.dynasty
					scope:masterhunter.dynasty = { has_dynasty_perk = meritocracy_legacy_2 }
					is_female = yes
					}
				meritocracy_2_effect_b = yes
			}
		
		if = {
				limit = {
					scope:masterhunter = {
					has_trait = order_member
					}
					exists = scope:masterhunter.dynasty
					scope:masterhunter.dynasty = { has_dynasty_perk = meritocracy_legacy_3 }
					}
				meritocracy_3_effect = yes
			}
		
		if = {
				limit = {
					scope:masterhunter = {
					has_trait = order_member
					}
					exists = scope:masterhunter.dynasty
					scope:masterhunter.dynasty = { has_dynasty_perk = meritocracy_legacy_4 }
					}
				meritocracy_4_effect = yes
			}
			
		if = {
				limit = {
					scope:masterhunter = {
					has_trait = order_member
					}
					exists = scope:masterhunter.dynasty
					scope:masterhunter.dynasty = { has_dynasty_perk = meritocracy_legacy_5 }
					}
				meritocracy_5_effect = yes
			}
		}
	}
}

POD_ai_inquisitor_ordination.2 = {
	type = character_event
	hidden = yes

    trigger = {
        is_vampire_trigger = no
		POD_is_supehunter_trigger = no
		OR = {
			dynasty = dynasty:13000004
			dynasty = dynasty:13000005
			dynasty = dynasty:13000007
		}
    }

    immediate = {
			
			if = {
				limit = { has_trait = mortal }
				remove_trait = mortal
			}
			
			if = {
				limit = { 
				NOT = { has_trait = supehunter }
				}
				add_trait = supehunter
			}
			
			random = {
				chance = 10
				add_trait = truefaith
			}
			
			if = {
				limit = {
					OR = {
					scope:father.faith = faith:shadowinquisition
					scope:mother.faith = faith:shadowinquisition
					}
				}
			set_character_faith = faith:shadowinquisition
			}
			else_if = {
				limit = {
					OR = {
					scope:father.faith = faith:calomenasforsaken
					scope:mother.faith = faith:calomenasforsaken
					}
				}
			set_character_faith = faith:calomenasforsaken
			}
			
			else = {
				limit = {
					OR = {
					scope:father.faith = faith:independenthunters
					scope:mother.faith = faith:independenthunters
					}
				}
			set_character_faith = faith:independenthunters
			}
	}
}

POD_ai_inquisitor_ordination.3 = {
	type = character_event
	hidden = yes

    trigger = {
        is_vampire_trigger = no
		NOR = {
				has_trait = supehunter 
				has_trait = bastard		
					}
		is_adult = yes
				OR = {
					NOR = {
					dynasty = dynasty:13000001
					dynasty = dynasty:13000002
					dynasty = dynasty:13000003
					dynasty = dynasty:13000004
					dynasty = dynasty:13000005
					dynasty = dynasty:13000006
					dynasty = dynasty:13000007
					dynasty = dynasty:13000008
					}
					AND = {
						OR = {
							dynasty = dynasty:13000001
							dynasty = dynasty:13000002
							dynasty = dynasty:13000003
							dynasty = dynasty:13000006
							dynasty = dynasty:13000008
							}
						has_trait = legitimized_bastard
					}
				}
				OR = {
					AND = {
						exists = father
						father = { has_trait = supehunter }
							}
					AND = {
						exists = mother
						mother = { has_trait = supehunter }
							}
					AND = {
						exists = liege
						liege = { has_trait = supehunter }
							}
		}
	}
	
	immediate = {
			if = {
			limit = {
				has_trait = mortal
				}
			remove_trait = mortal
			}
			add_trait = supehunter
			random = {
				chance = 10
				add_trait = truefaith
			}
			if = {
				limit = {
					OR = {
						AND = { 
						exists = father
						father = { faith = faith:shadowinquisition }
							}
						AND = { 
						exists = mother
						mother = { faith = faith:shadowinquisition }
							}
						AND = {
						exists = liege
						liege = { faith = faith:shadowinquisition }
							}
						}
					}
					set_character_faith = faith:shadowinquisition
				}
			else_if = {
				limit = {
						exists = liege
						liege = { faith = faith:calomenasforsaken }
						}
					set_character_faith = faith:calomenasforsaken
				}
			else = {
					set_character_faith = faith:independenthunters
					}
			if = {
				limit = {
					OR = {
					AND = {
						exists = father
						father = { has_trait = order_member }
							}
					AND = {
						exists = mother
						mother = { has_trait = order_member }
							}
					AND = {
						exists = liege
						liege = { has_trait = order_member }
							}
					OR = {
							dynasty = dynasty:13000001
							dynasty = dynasty:13000002
							dynasty = dynasty:13000003
							dynasty = dynasty:13000006
							dynasty = dynasty:13000008
							}
						}
					}
					add_trait = order_member
			}
	}
}
