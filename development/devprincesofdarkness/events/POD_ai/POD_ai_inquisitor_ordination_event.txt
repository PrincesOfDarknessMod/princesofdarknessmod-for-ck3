namespace = POD_ai_inquisitor_ordination

POD_ai_inquisitor_ordination.1 = {
	type = character_event
	hidden = yes

    trigger = {
		is_landed = yes
        is_vampire_trigger = no
		has_trait = supehunter
		age > 20
		is_imprisoned = no
		is_ai = yes 
		NOT = { has_character_flag = had_event_POD_ai_inquisitor_ordination_1 }
		OR = {
			any_child = { is_alive = yes count < 1 }
			count_has_two_childer_or_less_trigger = yes
			duke_has_three_childer_or_less_trigger = yes
			king_has_four_childer_or_less_trigger = yes
			emperor_has_five_childer_or_less_trigger = yes
		}	
		#The Von Murnau make children the old fashioned way
		NOT = { dynasty = dynasty:13000004 } 
    }

    immediate = {
		if = {
			limit = {
				any_child = { is_alive = yes count < 1 }
			}
			add_character_flag = {
				flag = had_event_POD_ai_inquisitor_ordination_1
				years = 1
			}
		}	
		if = {
			limit = {
				any_child = { is_alive = yes count < 2 }
			}
			add_character_flag = {
				flag = had_event_POD_ai_inquisitor_ordination_1
				years = 2
			}
		}		
		
		if = {
			limit = {
				any_child = { is_alive = yes count < 3 }
			}
			add_character_flag = {
				flag = had_event_POD_ai_inquisitor_ordination_1
				years = 5
			}
		}	

		if = {
			limit = {
				any_child = { is_alive = yes count < 4 }
			}
			add_character_flag = {
				flag = had_event_POD_ai_inquisitor_ordination_1
				years = 10
			}
		}		

		if = {
			limit = {
				any_child = { is_alive = yes count < 5 }
			}
			add_character_flag = {
				flag = had_event_POD_ai_inquisitor_ordination_1
				years = 25
			}
		}	
		
		save_scope_as = masterhunter
		scope:masterhunter = {
			POD_ai_inquisitor_ordination_effect = yes		
		}
	}
}

#Von Murnau becoming hunter

POD_ai_inquisitor_ordination.2 = {
	type = character_event
	hidden = yes

    trigger = {
        is_vampire_trigger = no
		NOT = { has_trait = supehunter }
		is_ai = yes 
		is_adult = yes
		dynasty = dynasty:13000004
    }

    immediate = {
		random = {
			chance = 10
					modifier = {
						add = 90
						is_ruler = yes
					}			
			add_trait = supehunter
			set_character_faith = faith:shadowinquisition
		}
	}
}
