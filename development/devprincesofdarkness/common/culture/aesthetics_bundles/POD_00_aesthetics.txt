#######################################
# Ancient Egyptian Divergant Cultures #
#######################################
aesthetics_ancientegyptian = {
	is_shown = {
		scope:character = {
			culture = {
				has_cultural_pillar = heritage_ancientegyptian 
			}
			any_sub_realm_county = {
				any_this_title_or_de_jure_above = {
					this = title:k_egypt
				}
			}
			NOT = {
				culture = culture:ancientegyptian
			}
		}
	}
	
	name_list = name_list_ancientegyptian

	coa_gfx = { arabic_group_coa_gfx }
	building_gfx = { arabic_group_building_gfx mena_building_gfx }
	clothing_gfx = { dde_abbasid_clothing_gfx mena_clothing_gfx }
	unit_gfx = { mena_unit_gfx  }
}
