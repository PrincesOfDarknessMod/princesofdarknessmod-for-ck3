cloak_of_shadows_perk = {
	lifestyle = melancholy_lifestyle
	tree = obfuscate
	position = { 0 0 }
	icon = node_melancholy
	
	can_be_picked = {
		has_obfuscate_trigger = yes
	}		
	
	can_be_auto_selected = { 
		has_character_flag = perkscheckpoint	
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_melancholy_interest_trigger = yes
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = melancholy_obfuscate_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
		hostile_scheme_resistance_add = 10
		personal_scheme_resistance_add = 10
	}
	
	effect = {
		custom_description_no_bullet = {
			text = cloak_of_shadows_perk_stats_effect
		}
		custom_description_no_bullet = { 
			text = NEEDS_DISCIPLINE_TEXT
		}
	}
}
