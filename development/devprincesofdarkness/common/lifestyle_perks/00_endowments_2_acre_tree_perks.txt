acre_catechumen_perk = {
	lifestyle = endowments_lifestyle
	tree = acre
	position = { 0 0 }
	icon = acre_catechumen
	
	can_be_picked = {
		exists = dynasty
		OR = {
		dynasty = dynasty:13000002
		dynasty = dynasty:13000006
		dynasty = dynasty:13000008
		AND = {
			exists = dynasty
			dynasty = { has_dynasty_modifier = custom_knightly_order_modifier }
			}
		}
		has_trait = supehunter
		faith = faith:shadowinquisition
	}		
	
	can_be_auto_selected = {
	}
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_focus = endowments_all_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}
	character_modifier = {
	}
	
	effect = {
		add_trait_force_tooltip = acre_catechumen
	}
}
