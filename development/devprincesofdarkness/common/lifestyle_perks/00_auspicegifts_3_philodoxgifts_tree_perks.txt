	}
	
	can_be_auto_selected = { 
		has_character_flag = perkscheckpoint	
	}	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {

			}
			add = 1000
		}

			limit = {

			}

		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
	}
	
	effect = {
		if = {
			limit = {

					has_trait = philodox
					}
			}

				text = NEEDS_PHILODOX_TEXT
			}
		}
		custom_description_no_bullet = {
			text = digging_for_dirt_perk_effect
		}	
	}
}
