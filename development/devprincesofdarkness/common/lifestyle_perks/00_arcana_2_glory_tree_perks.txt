longevity_perk = {
	lifestyle = arcana_lifestyle
	tree = glory
	position = { 0 1 }
	icon = arcana

	can_be_picked = {
		OR = {
			has_trait = earthbound
			is_a_thrall = yes
		}
	}
	
	can_be_auto_selected = { 
		has_character_flag = perkscheckpoint
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}
	
	effect = {
		custom_description_no_bullet = { 
			text = longevity_perk_effect
		}
	}
}
