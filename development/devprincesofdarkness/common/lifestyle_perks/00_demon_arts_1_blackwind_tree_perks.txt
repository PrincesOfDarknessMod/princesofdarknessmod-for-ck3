black_wind_1_perk = {
	lifestyle = demon_arts_lifestyle
	tree = blackwind
	position = { 1 0 }
	icon = node_demon_arts
	
	can_be_picked = {
		is_kueijin_trigger = yes
		NOR = {
			perks_in_tree = { tree = cultivation value > 0 }
			perks_in_tree = { tree = internalize value > 0 }
			perks_in_tree = { tree = obligation value > 0 }
		}
	}
	
	can_be_auto_selected = {
		has_character_flag = perkscheckpoint
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_focus = demon_arts_all_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}
	
	character_modifier = {
		movement_speed = 0.25
		stress_gain_mult = 0.1
	}
	
	effect = {
		custom_description_no_bullet = {
			text = black_wind_perk_effect
		}
	}
}
