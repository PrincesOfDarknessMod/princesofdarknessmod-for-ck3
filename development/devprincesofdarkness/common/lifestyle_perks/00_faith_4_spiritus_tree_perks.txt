tears_of_the_martyr_perk = {
	lifestyle = faith_lifestyle
	tree = spiritus
	position = { 0 0 }
	icon = node_learning

	can_be_picked = {
		exists = dynasty
		has_trait = supehunter
		OR = {
			dynasty = dynasty:13000001
			dynasty = dynasty:13000003
			AND = {
			exists = dynasty
				OR = {
					dynasty = { has_dynasty_modifier = custom_mendicant_order_modifier }
					dynasty = { has_dynasty_modifier = custom_mendicant_order_modifier }
				}
			}
			has_trait = truefaith
			has_trait = beatific
		}
		faith = faith:shadowinquisition
	}
	
	can_be_auto_selected = { 
		
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}
	
	character_modifier = {
	}

	effect = {
		custom_description_no_bullet = {
			text = tears_of_the_martyr_perk_effect
		}
		if = {
			limit = {
				NOR = {
					dynasty = dynasty:13000001
					dynasty = dynasty:13000003
					AND = {
						exists = dynasty
						OR = {
							dynasty = { has_dynasty_modifier = custom_mendicant_order_modifier }
							dynasty = { has_dynasty_modifier = custom_mendicant_order_modifier }
						}
					}
					has_trait = truefaith
					has_trait = beatific
				}
			}
			custom_description_no_bullet = { 
				text = SPIRITUS_NOT_PRIMARY_TEXT
			}
		}
		if = {
			limit = {
				NAND = { 
					faith = faith:shadowinquisition
					has_trait = supehunter
				}
			}
		custom_description_no_bullet = { 
				text = SHADOW_INQUISITION_ONLY_TEXT
			}
		}
	}
}
