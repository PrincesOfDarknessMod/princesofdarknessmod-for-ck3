no_mind_perk = {
	lifestyle = soul_arts_lifestyle
	tree = internalize
	position = { 0 0 }
	icon = node_soul_arts
	
	can_be_picked = {
		is_kueijin_trigger = yes
		demon_arts_lifestyle_perks = 0
	}
	
	can_be_auto_selected = {
		has_character_flag = perkscheckpoint
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_focus = soul_arts_all_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
		if = {
			limit = {
				demon_arts_lifestyle_perks > 0
			}
			multiply = 0
		}
	}
	
	character_modifier = {
		stress_gain_mult = -0.15
	}
	
	effect = {
		custom_description_no_bullet = {
			text = no_mind_perk_effect
		}
		if = {
			limit = {
				demon_arts_lifestyle_perks > 0
			}
			custom_description_no_bullet = { 
				text = soul_demon_incompatible_text
			}
		}
	}
}
