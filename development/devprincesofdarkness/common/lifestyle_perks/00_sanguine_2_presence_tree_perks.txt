awe_perk = {
	lifestyle = sanguine_lifestyle
	tree = presence
	position = { 0 0 }
	icon = node_sanguine
	
	can_be_picked = {
		has_presence_trigger = yes
	}	
	
	can_be_auto_selected = { 
		has_character_flag = perkscheckpoint	
	}
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_sanguine_interest_trigger = yes
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = sanguine_presence_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}
	### gives access to Befriend scheme
	character_modifier = {
		personal_scheme_power_add = 5
		direct_vassal_opinion = 5
	}
	
	effect = {
		custom_description_no_bullet = { 
			text = awe_perk_stats_effect
		}
		custom_description_no_bullet = { 
			text = NEEDS_DISCIPLINE_TEXT
		}
	}
}
