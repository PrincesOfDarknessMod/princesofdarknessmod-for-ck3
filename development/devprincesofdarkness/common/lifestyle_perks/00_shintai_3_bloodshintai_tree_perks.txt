	tree = bloodshintai

	icon = node_shintai

		OR = {
			has_trait = kueijin
			has_trait = dhampyr
		}
	}
	
	can_be_auto_selected = { 
		has_character_flag = perkscheckpoint	
	}

		value = 11

		}
	}
	
	character_modifier = {
		owned_scheme_secrecy_add = 10
	}
	
	effect = {
		custom_description_no_bullet = {
			text = permeate_perk_effect
		}
	}
}
