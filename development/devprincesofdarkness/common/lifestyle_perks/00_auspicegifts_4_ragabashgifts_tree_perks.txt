liar_face_perk = {
	lifestyle = auspicegifts_lifestyle
	tree = ragabashgifts
	position = { 0 0 }
	icon = node_animal
	
	can_be_picked = {
		has_trait = ragabash
	}
	
	can_be_auto_selected = { 
		has_character_flag = perkscheckpoint	
	}	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_trait = ragabash
			}
			add = 1000
		}
		if = {
			limit = {
				has_focus = auspicegifts_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
		owned_hostile_scheme_success_chance_add = 10
	}
	
	effect = {
		if = {
			limit = {
				NOT = {
					has_trait = ragabash
					}
			}
			custom_description_no_bullet = { 
				text = NEEDS_RAGABASH_TEXT
			}
		}
	}
}
