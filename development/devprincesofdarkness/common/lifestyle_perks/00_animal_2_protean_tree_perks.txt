eyes_of_the_beast_perk = {
	lifestyle = animal_lifestyle
	tree = protean
	position = { 0 0 }
	icon = node_animal
	
	can_be_picked = {
		has_protean_trigger = yes
	}		
	
	can_be_auto_selected = { 
		has_global_variable = POD_learning_starts_now	
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_animal_interest_trigger = yes
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = animal_protean_focus
			}
			multiply = 5
		}
		
		if = {
			limit = {
				NOT = { has_trait = vampire }
			}
			multiply = 0
		}		
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
		enemy_hostile_scheme_success_chance_add = -15
	}
	
	effect = {	
		custom_description_no_bullet = { 
			text = NEEDS_DISCIPLINE_TEXT
		}
	}
}
