cats_grace_perk = {
	lifestyle = choleric_lifestyle
	tree = celerity
	position = { 0 0 }
	icon = node_choleric
	
	can_be_picked = {
		has_celerity_trigger = yes
	}		
	can_be_auto_selected = { 
		has_character_flag = perkscheckpoint	
	}
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_choleric_interest_trigger = yes
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = choleric_celerity_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
		murder_scheme_resistance_add = 5
		abduct_scheme_resistance_add = 5
		diablerize_scheme_resistance_add = 5
	}

	effect = {
		custom_description_no_bullet = {
			text = cats_grace_perk_stats_effect
		}
		if = {
			limit = {
				has_celerity_trigger = no
			}
			custom_description_no_bullet = { 
				text = NEEDS_DISCIPLINE_TEXT
			}
		}
	}
}
