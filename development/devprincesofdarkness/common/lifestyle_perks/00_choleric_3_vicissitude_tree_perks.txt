		has_vicissitude_trigger = yes
	}	
	
	can_be_auto_selected = { 
		has_character_flag = perkscheckpoint	
	}
	auto_selection_weight = {
		value = 11
		if = {
			limit = {

			}
			add = 1989
		}

			multiply = 5
		}

			limit = {

			}
			multiply = 0
		}
	}

	character_modifier = {
		attraction_opinion = 10 #futurely decision to get attactive traits
	}
	
	effect = {
		custom_description_no_bullet = { 
			text = malleable_visage_perk_stats_effect
		}
	}
}
