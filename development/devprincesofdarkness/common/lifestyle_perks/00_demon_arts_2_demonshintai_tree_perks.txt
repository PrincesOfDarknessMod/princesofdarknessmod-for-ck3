	icon = node_demon_arts

		is_kueijin_trigger = yes
		has_soul_lifestyle_perk_except_chiiumuh_assigned_trigger = no
	}
	
	can_be_auto_selected = {
		has_character_flag = perkscheckpoint
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_focus = demon_arts_all_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
		if = {
			limit = {
				has_soul_lifestyle_perk_except_chiiumuh_assigned_trigger = yes
			}
			multiply = 0
		}
	}
	
	character_modifier = {
		advantage = 5
		stress_gain_mult = 0.05
	}
	
	effect = {
		custom_description_no_bullet = {
			text = demon_shintai_perk_effect
		}
		if = {
			limit = {
				has_soul_lifestyle_perk_except_chiiumuh_assigned_trigger = yes
			}
			custom_description_no_bullet = { 
				text = soul_demon_incompatible_text
			}
		}
	}
}
