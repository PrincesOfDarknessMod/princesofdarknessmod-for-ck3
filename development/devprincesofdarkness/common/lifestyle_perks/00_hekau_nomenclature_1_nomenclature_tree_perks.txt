asking_the_trees_perk = {
	lifestyle = hekau_nomenclature_lifestyle
	tree = nomenclature
	position = { 0 0 }
	icon = node_learning

	can_be_picked = {
		has_trait = mummy
		NOT = { has_trait = wutian }
	}	
	
	can_be_auto_selected = { 
		has_character_flag = perkscheckpoint
	}

	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_trait = mummy
				NOT = { has_trait = wutian }
			}
			add = 1989
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}
	effect = {
		custom_description_no_bullet = { text = asking_the_trees_perk_effect }
	}
}
