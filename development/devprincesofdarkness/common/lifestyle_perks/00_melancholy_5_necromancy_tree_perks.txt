insight_perk = {
	lifestyle = melancholy_lifestyle
	tree = melancholy_necromancy
	position = { 1 0 }
	icon = node_melancholy
	
	can_be_picked = {
		has_necromancy_trigger = yes
	}		
	
	can_be_auto_selected = { 
		has_character_flag = perkscheckpoint	
	}
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_melancholy_interest_trigger = yes
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = melancholy_necromancy_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}
	
	effect = {
		custom_description_no_bullet = { 
			text = insight_perk_effect
		}
		if = {
			limit = {
				has_necromancy_trigger = no
			}
			custom_description_no_bullet = { 
				text = NEEDS_DISCIPLINE_TEXT
			}
		}
	}
}
