	lifestyle = shintai_lifestyle
	tree = fleshshintai
	position = { 0 0 }
	icon = node_shintai
	
	can_be_picked = {
		is_kueijin_trigger = yes
	}
	
	can_be_auto_selected = { 
		has_character_flag = perkscheckpoint	
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {

			}
			multiply = 5
		}
		if = {

				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}
	
	character_modifier = {
		prowess = 2
	}
	
	effect = {
		custom_description_no_bullet = {
			text = long_neck_perk_effect
		}
	}
}
