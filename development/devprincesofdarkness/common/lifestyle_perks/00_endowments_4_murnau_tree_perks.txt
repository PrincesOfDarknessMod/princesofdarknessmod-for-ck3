murnau_knowledge_perk = {
	lifestyle = endowments_lifestyle
	tree = murnau
	position = { 0 0 }
	icon = murnau_knowledge
	
	can_be_picked = {
		dynasty = dynasty:13000004
		POD_is_supehunter_trigger = yes
	}		
	
	can_be_auto_selected = { 
		has_character_flag = perkscheckpoint	
	}
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_focus = endowments_all_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}
	effect = {
		add_trait_force_tooltip = murnau_knowledge
		if = {
			limit = {
				NOT = { dynasty = dynasty:13000004 }
			}
			custom_description_no_bullet = { 
				text = murnau_only_text
			}
		}
	}
}
