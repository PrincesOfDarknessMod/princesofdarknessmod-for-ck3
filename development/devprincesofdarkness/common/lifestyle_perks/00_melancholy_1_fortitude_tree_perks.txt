resilence_perk = {
	lifestyle = melancholy_lifestyle
	tree = fortitude
	position = { 0 0 }
	icon = node_melancholy
	
	can_be_picked = {
		has_fortitude_trigger = yes
	}			
	
	can_be_auto_selected = { 
		has_global_variable = POD_learning_starts_now	
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_melancholy_interest_trigger = yes
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = melancholy_fortitude_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
		health = 5
	}
	
	effect = {
		custom_description_no_bullet = { 
			text = NEEDS_DISCIPLINE_TEXT
		}
	}
}
