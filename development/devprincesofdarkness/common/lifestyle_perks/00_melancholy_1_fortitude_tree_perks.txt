resilence_perk = {
	lifestyle = melancholy_lifestyle
	tree = fortitude
	position = { 0 0 }
	icon = node_melancholy
	
	can_be_picked = {
		has_fortitude_trigger = yes
	}			
	
	can_be_auto_selected = { 
			
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_melancholy_interest_trigger = yes
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = melancholy_fortitude_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = unswayable_mind_perk }	
			}
			multiply = 0
		}
	}

	character_modifier = {
		prowess = 1
		diablerize_scheme_resistance_add = 5
		murder_scheme_resistance_add = 5
	}
	
	effect = {
		custom_description_no_bullet = {
			text = resilience_perk_stats_effect
		}
		if = {
			limit = {
				has_fortitude_trigger = no
			}
			custom_description_no_bullet = { 
				text = NEEDS_DISCIPLINE_TEXT
			}
		}
	}
}
