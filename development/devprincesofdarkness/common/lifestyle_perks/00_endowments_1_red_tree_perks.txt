red_vigil_perk = {
	lifestyle = endowments_lifestyle
	tree = red
	position = { 0 0 }
	icon = red_vigil
	
	can_be_picked = {
		dynasty = dynasty:13000001
		POD_is_supehunter_trigger = yes
	}		
	
	can_be_auto_selected = { 
			
	}
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_focus = endowments_all_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}
	effect = {
		add_trait_force_tooltip = red_vigil
		if = {
			limit = {
				NOT = { dynasty = dynasty:13000001 }
			}
			custom_description_no_bullet = { 
				text = red_only_text
			}
		}
	}
}
