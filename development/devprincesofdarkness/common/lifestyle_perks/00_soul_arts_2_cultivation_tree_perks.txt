scrutinize_perk = {
	lifestyle = soul_arts_lifestyle
	tree = cultivation
	position = { 1 0 }
	icon = node_soul_arts
	
	can_be_picked = {
		is_kueijin_trigger = yes
		NOR = {
			perks_in_tree = { tree = blackwind value >= 1 }
			perks_in_tree = { tree = demonshintai value >= 1 }
		}
	}
	
	can_be_auto_selected = {
		has_character_flag = perkscheckpoint
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_focus = soul_arts_all_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
		if = {
			limit = {
				perks_in_tree = { tree = blackwind value >= 1 }
				perks_in_tree = { tree = demonshintai value >= 1 }
			}
			multiply = 0
		}
	}
	
	character_modifier = {
		scheme_success_chance = 10
	}
	
	effect = {
		custom_description_no_bullet = {
			text = scrutinize_perk_effect
		}
	}
}
