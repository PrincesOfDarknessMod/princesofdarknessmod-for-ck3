learning_kaja_decision = {
    picture = "gfx/interface/illustrations/decisions/decision_misc.dds"
    ai_check_interval = 0

    is_shown = {
		is_ai = no
		NOT = { knows_language = language_kaja }
		OR = {
			is_kueijin_trigger = yes
			AND = {
				is_shih_trigger = yes
				has_global_variable = celestial_army_restored
			}
		}
	}

	is_valid_showing_failures_only = {
		NOT = { has_character_modifier = traveling_spirit_world_modifier }
		is_available_adult = yes
		is_at_war = no
	}

	cost = {
		prestige = {
			value = 200
			if = { 
				limit = {
					is_kueijin_trigger = no
				}
				multiply = 10
			}
			if = { 
				limit = {
					has_trait = dharma1
				}
				multiply = 5
			}
			if = { 
				limit = {
					has_trait = dharma2
				}
				multiply = 3
			}
			if = { 
				limit = {
					has_trait = dharma3
				}
				multiply = 2
			}
			if = { 
				limit = {
					has_trait = dharma4
				}
				multiply = 1.5
			}
		}
	}

    effect = {
		learn_language = language_kaja
		
		save_scope_as = player
		every_ruler = {
			limit = {
				knows_language = language_kaja
			}
			if = { 
				limit = {
					scope:player = { is_kueijin_trigger = no }
				}
				scope:player = {
					add_hook = {
						type = threat_hook
						target = prev
						years = 3
					}
				}
			}
			else = {
				add_opinion = {
					target = scope:player
					modifier = friendliness_opinion
					opinion = 20
				}
			}
		}
	}	

	ai_potential = {
		always = no
	}

	ai_will_do = {
		base = 0
	}
}

				base = feast_stress_loss_value
			}

		}

			ai_sociability = 1
			ai_greed = -0.5

			has_trait = yin_imbalanced
		}

			add = 30
			has_trait = yang_imbalanced
		}
		modifier = {
			add = 30 # Has Greed of 20, so this is only (-10 + 30 = +20)
			has_trait = gluttonous
		}
		modifier = {
			add = 12.5 # Has Greed of 5, so this is only (-2.5 + 12.5 = +10)

		}
