full_blood_decision = {
	picture = "gfx/interface/illustrations/decisions/decision_major_religion.dds"
	desc = full_blood_decision_desc
	major = no
	
	ai_check_interval = 0

	is_shown = {
		is_ai = no
		any_character_artifact = {
			OR = {
				has_variable = holy_grail
				has_variable = blood_horn
				}
			}	
		is_vampire_trigger = yes
		NOT = {
			has_character_modifier = bloodhunger0_modifier
			}
		}			
		
	is_valid = {
		}
	
	effect = {
		if = {
			limit = {
				has_character_modifier = bloodhunger1_modifier
			}
			remove_character_modifier = bloodhunger1_modifier
			add_character_modifier = {
				modifier = bloodhunger0_modifier
				}
			set_variable = { name = blood_level value = 5 }
			}
		else_if = {
			limit = {
				has_character_modifier = bloodhunger2_modifier
			}
			remove_character_modifier = bloodhunger2_modifier
			add_character_modifier = {
				modifier = bloodhunger0_modifier
				}
			set_variable = { name = blood_level value = 5 }
			}	
		else_if = {
			limit = {
				has_character_modifier = bloodhunger3_modifier
			}
			remove_character_modifier = bloodhunger3_modifier
			add_character_modifier = {
				modifier = bloodhunger0_modifier
				}
			set_variable = { name = blood_level value = 5 }
			}	
		else_if = {
			limit = {
				has_character_modifier = bloodhunger4_modifier
			}
			remove_character_modifier = bloodhunger4_modifier
			add_character_modifier = {
				modifier = bloodhunger0_modifier
				}
			set_variable = { name = blood_level value = 5 }
			}
		else_if = {
			limit = {
				has_character_modifier = bloodhunger5_modifier
			}
			remove_character_modifier = bloodhunger5_modifier
			add_character_modifier = {
				modifier = bloodhunger0_modifier
				}
			set_variable = { name = blood_level value = 5 }
			}						
	}

	ai_potential = {
		always = no
	}

	ai_will_do = {
		base = 0
	}
}
