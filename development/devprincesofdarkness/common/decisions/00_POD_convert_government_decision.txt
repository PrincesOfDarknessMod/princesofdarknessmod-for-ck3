POD_convert_government_decision = {
	picture = "gfx/interface/illustrations/decisions/decision_realm.dds"
	major = yes
	desc = POD_convert_government_decision_desc
	
	ai_check_interval = 62
	
	is_shown = {
		has_POD_race_trait_trigger = yes
		is_ruler = yes
		NOT = { has_government = mercenary_government }
		NOT = { has_government = holy_order_government }
		trigger_if = {
			limit = {
				is_kueijin_trigger = yes
			}
			NOT = { has_government = kueijin_court_government }
		}
		trigger_else_if = {
			limit = {
				has_POD_public_vamp_blood_trigger = yes
				is_forsaken_or_gantenbein_trigger = no
			}
			NOR = {
				has_government = traditions_government
				has_government = free_state_government
				AND = {
					has_government = free_state_government
					AND = {
						exists = liege
						liege = { has_government = traditions_government }
					}
				}
				AND = {
					has_government = traditions_government
					AND = {
						exists = liege
						liege = { has_government = free_state_government }
					}
				}
			}
		}
		trigger_else_if = {
			limit = {
				is_shih_trigger = yes
				has_global_variable = celestial_army_restored
			}
			NOT = { has_government = celestial_army_government }
		}
		trigger_else_if = {
			limit = {
				OR = {
					POD_is_mortal_trigger = yes
					POD_is_supehunter_trigger = yes
				}
				faith = faith:shadowinquisition
			}
			NOT = { has_government = inquisition_government }
		}
		trigger_else_if = {
			limit = {
				religion = { has_POD_religion_group = yes }
			}
			NOT = { has_government = indep_government }
		}
		trigger_else = {
			always = no
		}
	}
	
	effect = {
		POD_adopt_correct_goverment_effect = yes
		
		#convert_whole_realm_to_feudalism_effect = yes
		#add_realm_law_skip_effects = crown_authority_0
	}
	
	ai_potential = {
		always = yes
	}
	
	ai_will_do = {
		base = 100
	}
}
