POD_perform_alchemy_decision = {
    picture = "gfx/interface/illustrations/decisions/decision_smith.dds"

    ai_check_interval = 0
	
    is_shown = {
		is_ai = no
		has_trait = mummy
		has_variable = alchemy_craft_slot_1
	}

	is_valid_showing_failures_only = {
		trigger_if = {
			limit = { is_east_mummy_trigger = yes }
			NOT = { has_character_modifier = yangchi0_modifier }
		}
		trigger_if = {
			limit = { is_amenti_trigger = yes }
			NOT = { has_character_modifier = sekhem0_modifier }
		}
		trigger_if = {
			limit = { has_character_modifier = traveling_spirit_world_modifier }
			has_perk = shadow_talisman_perk
		}
		is_available_adult = yes
		is_at_war = no
		has_level_1_alchemy_recipe_trigger = yes
		custom_description = {
			text = alchemy_craft_slots_are_full
			OR = { 
				var:alchemy_craft_slot_1 = 0
				var:alchemy_craft_slot_2 = 0
				var:alchemy_craft_slot_3 = 0
				AND ={
					has_level_2_alchemy_recipe_trigger = yes
					var:alchemy_craft_slot_4 = 0
				}
				AND ={
					has_level_3_alchemy_recipe_trigger = yes
					var:alchemy_craft_slot_5 = 0
				}
				AND ={
					has_level_4_alchemy_recipe_trigger = yes
					var:alchemy_craft_slot_6 = 0
				}
				AND ={
					has_level_5_alchemy_recipe_trigger = yes
					var:alchemy_craft_slot_7 = 0
				}
				AND ={
					has_level_6_alchemy_recipe_trigger = yes
					var:alchemy_craft_slot_8 = 0
				}
				AND ={
					has_level_7_alchemy_recipe_trigger = yes
					var:alchemy_craft_slot_9 = 0
				}
				AND ={
					has_level_8_alchemy_recipe_trigger = yes
					var:alchemy_craft_slot_10 = 0
				}
			}
		}
	}
	
	cooldown = { years = 1 }

    effect = {
		capital_province = {
			save_scope_as = capital_province
			spawn_activity = {
				type = activity_alchemy_craft
				owner = root
			}
		}
	}

	ai_will_do = {
		base = 0
	}
}
