practicing_arts_decision = {
	picture = "gfx/interface/illustrations/decisions/decision_personal_religious.dds"
	ai_check_interval = 60

	cooldown = { months = 36 }

	is_shown = {
		is_landed = yes
		is_kueijin_trigger = yes
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
		is_at_war = no
		OR = {
			has_focus = shintai_all_focus
			has_focus = chi_arts_all_focus
			has_focus = soul_arts_all_focus
			has_focus = demon_arts_all_focus
		}
	}

	effect = {
		if = {
			limit = { is_ai = yes }
			add_character_flag = {
				flag = planning_an_activity
				days = 30
			}
			if = {
				limit = { has_focus = shintai_all_focus }
				add_shintai_lifestyle_xp = 100
			}
			else_if = {
				limit = { has_focus = chi_arts_all_focus }
				add_chi_arts_lifestyle_xp = 100
			}
			else_if = {
				limit = { has_focus = soul_arts_all_focus }
				add_soul_arts_lifestyle_xp = 100
			}
			else_if = {
				limit = { has_focus = demon_arts_all_focus }
				add_demon_arts_lifestyle_xp = 100
			}
			add_character_modifier = {
				modifier = regular_meditating_modifier
				days = 30
			}
		}
		else = {
			hidden_effect = {
				capital_province = {
					save_scope_as = capital_province
					spawn_activity = {
						type = activity_kueijin_practice
						owner = root
					}
				}
			}
		}
	}

	ai_potential = {
		is_landed = yes
		is_kueijin_trigger = yes
		OR = {
			has_focus = shintai_all_focus
			has_focus = chi_arts_all_focus
			has_focus = soul_arts_all_focus
			has_focus = demon_arts_all_focus
		}
	}

	ai_will_do = {
		base = 50
	}
}
