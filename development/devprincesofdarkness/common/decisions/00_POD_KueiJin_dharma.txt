initiate_dharma = {
	title = advance_dharma
	selection_tooltip = advance_dharma_tooltip
	picture = "gfx/interface/illustrations/decisions/decision_personal_religious.dds"
	desc = advance_dharma_desc
	confirm_text = advance_dharma_confirm
	
	major = yes

	ai_check_interval = 60

	is_shown = {
		is_ruler = yes
		is_landed = yes
		is_kueijin_trigger = yes
		NOR = {
			has_trait = dharma10
			has_trait = dharma9
			has_trait = dharma8
			has_trait = dharma7
			has_trait = dharma6
			has_trait = dharma5
			has_trait = dharma4
			has_trait = dharma3
			has_trait = dharma2
			has_trait = dharma1
			has_trait = dharma0
		}
	}

	effect = {
		if = {
			limit = {		
				age >= 1000
			}
			add_trait = dharma9
		}
		else_if = {
			limit = {		
				age >= 500
			}
			add_trait = dharma8
		}
		else_if = {
			limit = {		
				age >= 250
			}
			add_trait = dharma7
		}
		else_if = {
			limit = {		
				age >= 100
			}
			add_trait = dharma6
		}
		else = {
			add_trait = dharma5
		}
	}	
	
	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}
