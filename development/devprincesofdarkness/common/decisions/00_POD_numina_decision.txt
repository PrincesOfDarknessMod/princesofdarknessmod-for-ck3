POD_sorcery_become_immortal_decision = {
    picture = "gfx/interface/illustrations/decisions/decision_misc.dds"

    ai_check_interval = 0

	cost = {
		gold = { value = omega_gold_value }
	}
	
    is_shown = {
		is_ai = no
		has_trait = sorcerer
		has_perk = alchemy_perk
		NOT = { has_trait = immortal }
	}
	
	is_valid = {
		custom_description = {
			text = POD_sorcery_become_immortal_decision_custom	
			any_prisoner = {
				count >= 3
				has_trait = mummy
			}
		}
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_at_war = no
	}

    effect = {
		add_trait = immortal
	}

	ai_will_do = {
		base = 0
	}
}

POD_sorcery_become_vampire_decision = {
    picture = "gfx/interface/illustrations/decisions/decision_misc.dds"

    ai_check_interval = 0

	cost = {
		gold = { value = medium_gold_value }
	}
	
    is_shown = {
		is_ai = no
		has_trait = sorcerer
		has_perk = alchemy_perk
		NOT = { has_trait = immortal }
	}
	
	is_valid = {
		custom_description = {
			text = POD_sorcery_become_vampire_decision_custom	
			any_prisoner = {
				count >= 3
				has_trait = vampire
			}
		}
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_at_war = no
	}

    effect = {
		if = {
			limit = { is_pregnant = yes } 
			end_pregnancy = yes
			pregnancy_cleanup_effect = yes
		}
		if = {
			limit = {
				has_trait = psychic 
			}
			remove_trait = psychic
			refund_perks = psi_lifestyle
			add_psi_lifestyle_perk_points = -100
		}
		if = {
			limit = {
				OR = {
					has_trait = sorcerer 
					has_trait = theurgist 
				}
			}
			remove_trait = sorcerer
			remove_trait = theurgist
			refund_perks = sorcery_lifestyle
			add_sorcery_lifestyle_perk_points = -100
		}
		if = { 
			limit = {
				has_trait = supehunter 
			}
			trigger_event = POD_hunterembrace.100
		}
			
		POD_remove_splat_traits_effect = yes
		add_trait = vampire
		
		POD_character_vampire_clan_initialization_effect = yes
		add_secret = { type = secret_vampgeneration12 }
		
		POD_character_vampire_faith_initialization_effect = yes
		change_government = free_state_government
			
		hidden_effect = {
			POD_character_vampire_bloodpotency_initialization_effect = yes
			POD_character_vampire_bloodhunger_initialization_effect = yes
			POD_character_vampire_discipline_initialization_effect = yes
			POD_character_vampire_predatortype_initialization_effect = yes
						
			if = {
				limit = { is_ventrue_trigger = yes }
				POD_character_ventrue_prey_exclusion_initialization_effect = yes
			}
			POD_character_vampire_nosferatu_initialization_effect = yes
			POD_character_vampire_gangrel_beast_looks_initialization_effect = yes
			POD_character_vampire_malkavian_initialization_effect = yes
			POD_character_vampire_tzimisce_initialization_effect = yes
			POD_salubri_caste_new_vampire_initialization_effect = yes
			
			POD_character_initialization_random_background_effect = yes
				
			add_character_flag = immune_to_disease

			POD_commander_trait_initialization_effect = yes
			
			
			clear_variable_list = auto_resurrect_list
		}
	}

	ai_will_do = {
		base = 0
	}
}
