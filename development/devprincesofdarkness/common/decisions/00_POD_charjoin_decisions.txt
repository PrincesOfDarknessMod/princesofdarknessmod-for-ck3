join_clan_decision = {
    picture = "gfx/interface/illustrations/decisions/decision_dynasty_house.dds"

    ai_check_interval = 0

    is_shown = {
		is_ruler = yes
		is_landed = yes
		OR = {
			AND = {
				POD_is_supehunter_trigger = yes
				NOT = { has_trait = shih }
				has_orison_trigger = no
			}
			AND = {
				is_mortal_trigger = yes
				has_trait = POD_knows_too_much
			}
		}
		NOR = { 
			has_character_flag = independent_char_join_clan_flag
			has_character_flag = independent_char_join_order_flag
			has_character_flag = independent_char_join_wu_flag
			has_character_flag = independent_char_join_ghoulbros_flag
		}
		NOT = { has_trait = truefaith }
	}

    effect = {
		custom_tooltip = join_clan_decision_tooltip
		add_character_flag = independent_char_join_clan_flag
	}

	ai_potential = {
		always = no
	}

	ai_will_do = {
		base = 0
	}
}
