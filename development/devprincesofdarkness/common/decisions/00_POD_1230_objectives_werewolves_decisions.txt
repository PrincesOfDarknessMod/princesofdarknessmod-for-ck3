blackfury_objective_decision = {
	picture = "gfx/interface/illustrations/decisions/decision_destiny_goal.dds"
	major = yes

	ai_check_interval = 120

	desc = blackfury_objective_decision_desc

	is_shown = {
		has_trait = werewolf
		is_landed = yes
		faith = { religion_tag = blackfury_religion }
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:blackfury_objective_decision
			}
		}

	}

	is_valid = {
		is_independent_ruler = yes
		completely_controls = title:k_italy
		martial >= 15
		custom_description = {
			text = blackfury_decision_custom
		
		title:b_montbelliard = {
			title_province = { has_building = sun_glory_03 }
			}
		}			
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_at_war = no
	}

	effect = {
		save_scope_as = victorious_blackfury
		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:blackfury_objective_decision
		}
		custom_tooltip = blackfury_objective_decision_effect_tooltip
		trigger_event = pod1230objectives_werewolves.1
	
	}
	
	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}
