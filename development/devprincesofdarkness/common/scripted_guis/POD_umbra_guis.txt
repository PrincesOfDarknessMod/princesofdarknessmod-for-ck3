spirit_travel_moving = {
	scope = character
	is_shown = {
		has_variable = spirit_destination
	}
	is_valid = {
		NOT = { has_variable = spirit_destination }
	}
}

spirit_location_initiate = {
	scope = character
	is_valid = {
		NOT = { has_variable = spirit_location }
	}
	effect = {
		set_variable = { name = spirit_location value = flag:skinlands }
	}
}

spirit_list_initiate = {
	scope = character
	is_valid = {
		NOT = { has_variable_list = spirit_list }
	}
	effect = {
		POD_create_random_local_spirit = yes
		POD_create_random_local_spirit = yes
		POD_create_random_local_spirit = yes
	}
}

spirit_location_skinlands_choose = {
	scope = character
	is_shown = {
		NOT = { has_variable = spirit_destination }
		has_variable = spirit_location
		OR = {
			var:spirit_location = flag:penumbra
			var:spirit_location = flag:shadowlands
			has_character_flag = allow_teleport_to_skinlands
		}
	}
	effect = {
		set_variable = { name = spirit_destination value = flag:skinlands }
		trigger_event = {
			id = POD_spirit_travel.1
			days = { min_spirit_travel_movement_time max_spirit_travel_movement_time }
		}
	}
}

spirit_location_penumbra_choose = {
	scope = character
	is_shown = {
		NOT = { has_variable = spirit_destination }
		has_variable = spirit_location
		OR = {
			var:spirit_location = flag:skinlands
			var:spirit_location = flag:astral_reaches
			var:spirit_location = flag:spirit_wilds
			var:spirit_location = flag:near_dreaming
			var:spirit_location = flag:shadowlands
		}
	}
	is_valid = {
		trigger_if = {
			limit = {
				has_variable = spirit_location
				var:spirit_location = flag:skinlands
			}
			OR = {
				has_trait = werewolf
				has_perk = semblance_of_the_scalet_queen_perk
				has_perk = see_the_lands_of_the_spirit_perk
				has_perk = separate_ba_perk
				any_equipped_character_artifact = { has_variable = fans_of_ma_yi_yuan }
				has_character_modifier = philtre_of_the_blackened_sands_modifier
			}
		}
	}
	effect = {
		set_variable = { name = spirit_destination value = flag:penumbra }
		trigger_event = {
			id = POD_spirit_travel.1
			days = { min_spirit_travel_movement_time max_spirit_travel_movement_time }
		}
	}
}
