spirit_travel_enable = {
	scope = character
	is_valid = {
		OR = {
			has_trait = werewolf
			has_trait = mage
			has_trait = fae
			has_trait = wraith
			has_trait = demon
			has_trait = kueijin
			has_trait = mummy
			# Vamp
			has_perk = ex_nihilo_perk
			has_perk = psychic_projection_perk
			# Periphery
			AND = {
				OR = {
					has_trait = mortal
					has_trait = supehunter
					has_trait = kinfolk
				}
				OR = {
					has_trait = lunatic_1
					has_trait = lunatic_genetic
				}
				has_trait = poet
			}
			# Generic
			has_character_modifier = philtre_of_the_blackened_sands_modifier
			any_equipped_character_artifact = { has_variable = fans_of_ma_yi_yuan }
			capital_county = { has_county_modifier = link_enoch_modifier }
		}
	}
}

spirit_location_skinlands_choose = {
	scope = character
	is_shown = {
		NOT = { has_variable = spirit_destination }
		has_variable = spirit_location
		OR = {
			var:spirit_location = flag:periphery
			var:spirit_location = flag:shadowlands
			var:spirit_location = flag:spirit_wilds
			var:spirit_location = flag:astral_plane
			var:spirit_location = flag:shard_realm_of_prime
			var:spirit_location = flag:dreamtime
			var:spirit_location = flag:near_dreaming
			var:spirit_location = flag:digital_web
			var:spirit_location = flag:hollow_worlds
			var:spirit_location = flag:glen
			
			var:spirit_location = flag:first_city
			has_character_flag = allow_teleport_to_skinlands
		}
	}
	is_valid = {
		OR = {
			NOT = { has_trait = wraith }
			has_trait = risen
			has_perk = shroudshield_perk
		}
	}
	effect = {
		if = {
			limit = {
				has_trait = wraith
				NOT = { has_trait = risen }
			}
			trigger_event = POD_spirit_travel.200
		}
		else = {
			set_variable = { name = spirit_destination value = flag:skinlands }
			trigger_event = POD_spirit_travel.1
		}
	}
}
