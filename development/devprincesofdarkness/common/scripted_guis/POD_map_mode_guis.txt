pod_special_slots_map = {
	effect = {
		every_barony = {
			if = {
				limit = { title_province = { has_special_building_slot = yes } }
				if = {
					limit = { title_province = { has_special_building = yes } }
					set_color_from_title = title:k_kham
				}
				else = {
					set_color_from_title = title:k_pagan
				}
			}
			else = {
				set_color_from_title = title:k_angara
			}
		}
	}
	is_shown = {
		any_player = {
			is_ai = no
		}
	}
	is_valid = {
		always = yes
	}
}

pod_splats_map = {
	effect = {
		every_barony = {
			if = {
				limit = {
					OR = {
						AND = {
							exists = holder
							holder = { has_trait = vampire }
						}
						AND = {
							NOT = { exists = holder }
							barony_controller = { has_trait = vampire }
						}
					}
				}
				set_color_from_title = title:k_kham
			}
			else_if = {
				limit = {
					OR = {
						AND = {
							exists = holder
							holder = { 
								OR = {
									has_trait = ghoul
									has_trait = revenant
									has_trait = bloodgorged
								}
							}
						}
						AND = {
							NOT = { exists = holder }
							barony_controller = { 
								OR = {
									has_trait = ghoul
									has_trait = revenant
									has_trait = bloodgorged
								}
							}
						}
					}
				}
				set_color_from_title = title:k_pagan
			}
			else_if = {
				limit = {
					OR = {
						AND = {
							exists = holder
							holder = { 
								OR = {
									has_trait = mortal
									has_trait = supehunter
								}
							}
						}
						AND = {
							NOT = { exists = holder }
							barony_controller = { 
								OR = {
									has_trait = mortal
									has_trait = supehunter
								}
							}
						}
					}
				}
				set_color_from_title = title:e_shadow_inquisition
			}
			else = {
				set_color_from_title = title:k_angara
			}
		}
	}
	is_shown = {
		any_player = {
			is_ai = no
		}
	}
	is_valid = {
		always = yes
	}
}
