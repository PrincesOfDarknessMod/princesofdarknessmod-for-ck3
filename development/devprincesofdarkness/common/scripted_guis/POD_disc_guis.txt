power_template = {
	scope = character
	saved_scopes = { target }
	is_shown = { always = yes}
	
	is_valid = {
		NOT = {has_character_modifier = bloodhunger5_modifier}
		has_variable = path_of_template_rating
		custom_description = {
			text = I_need_rating
			trigger_if = {
				limit = { has_variable = path_of_template_rating }
				var:path_of_template_rating > 0
				
			}
		}
	}
	effect = {
	
		POD_hunger_rouse_check_effect = yes
		if = {
			limit = {has_character_flag = commits_willpower}
			POD_willpower_rouse_check_effect = yes
		}
		random_list = {
			0 = {
				modifier = {
					add = learning
					always = yes
				}
				modifier = {
					factor = 1.5
					has_character_flag = commits_willpower
				}
				hidden_effect = {
					send_interface_toast = {
						title = power_template.toast.1
						custom_tooltip = power_template.toast.1.ct
						left_icon = root
						right_icon = scope:target
					}
				}
				
				
			}
			10 = {
				custom_tooltip = power_template_fail
				hidden_effect = {
					send_interface_toast = {
						title = power_template_fail.toast.1
						custom_tooltip = power_template_fail.toast.1.ct
						left_icon = root
						right_icon = scope:target
					}
				}
			
			}
		}
	}
}

path_of_blood_choose = {
	scope = character
	is_shown = {
		var:chosen_path = flag:path_of_blood
	}
	effect = {
		set_variable = { name = chosen_path value = flag:path_of_blood }
		if = {
			limit = { NOT = {has_variable = path_of_blood_rating} }
			set_variable = { name = blood_sorcery_cost value = 100 }
		}
		else_if = {
			limit = { var:path_of_blood_rating = 1}
			set_variable = { name = blood_sorcery_cost value = 200 }
		}
		else_if = {
			limit = { var:path_of_blood_rating = 2}
			set_variable = { name = blood_sorcery_cost value = 300 }
		}
	}
}

path_of_blood_buy = {
	scope = character
	is_valid = {
		custom_description = {
			text = NOT_ENOUGH_POINTS
			NOT = { var:blood_sorcery_cost > var:blood_sorcery_points}
		}
		custom_description = {
			text = PATH_MAXED
			trigger_if = {
				limit = { has_variable = path_of_blood_rating }
				3 > var:path_of_blood_rating
			}
		}
		custom_description = {
			text = NOT_ADEPT_ENOUGH
			OR = {
				has_perk = a_taste_of_blood_perk
				has_perk = corrosive_vitae_perk
			}
			trigger_if = {
				limit = { has_variable = path_of_blood_rating}
				OR = {
					AND = {
						var:path_of_blood_rating = 1
						OR = {
							has_perk = blood_of_potency_perk
							has_perk = scorpions_touch_perk
						}
					}
					AND = {
						var:path_of_blood_rating = 2
						OR = {
							has_perk = baals_curse_perk
							has_perk = cauldron_of_blood_perk
						}
					}
				}
			}
		}
	}
	effect = {
		if = {
			limit = {NOT = { has_variable = path_of_blood_rating }}
			set_variable = { name = path_of_blood_rating value = 0}
		}
		change_variable = {
			name = path_of_blood_rating
			add = 1
		}
		change_variable = {
			name = blood_sorcery_points
			subtract = var:blood_sorcery_cost
		}
		change_variable = {
			name = blood_sorcery_cost
			add = 100
		}
		hidden_effect = {
			send_interface_toast = {
				title = path_buy_toast
				
				left_icon = root
			}
		}
		
		
	}
}
