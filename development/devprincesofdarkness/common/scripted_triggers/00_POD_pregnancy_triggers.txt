# Only integrated splat for now
POD_race_pregnancy_compatibility_with_character_trigger = {
	trigger_if = {
		limit = {
			OR = {
				POD_is_thinblooded_trigger = yes
				has_trait = yang_imbalanced
			}
		}
		
		$CHARACTER$ = {
			POD_is_mortal_trigger = yes
		}
	}

	trigger_if = {
		limit = { 
			OR = {
				POD_is_ghoul_trigger = yes
				POD_is_revenant_trigger = yes
				POD_is_bloodgorged_trigger = yes
				POD_is_supehunter_trigger = yes
			}
		}
		
		$CHARACTER$ = {
			OR = {
				POD_is_ghoul_trigger = yes
				POD_is_revenant_trigger = yes
				POD_is_bloodgorged_trigger = yes
				POD_is_supehunter_trigger = yes
				POD_is_mortal_trigger = yes
			}
		}
	}
	
	trigger_if = {
		limit = { 
			POD_is_mortal_trigger = yes
		}
		
		$CHARACTER$ = {
			OR = {
				POD_is_thinblooded_trigger = yes
				has_trait = yang_imbalanced
				POD_is_ghoul_trigger = yes
				POD_is_revenant_trigger = yes
				POD_is_bloodgorged_trigger = yes
				POD_is_supehunter_trigger = yes
				POD_is_mortal_trigger = yes
			}
		}
	}
}
