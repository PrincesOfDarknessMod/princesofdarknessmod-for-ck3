is_my_fetter_trigger = {
	OR = {
		is_target_in_variable_list = { name = fetter_1_list target = $FETTER$ }
		is_target_in_variable_list = { name = fetter_2_list target = $FETTER$ }
		is_target_in_variable_list = { name = fetter_3_list target = $FETTER$ }
		is_target_in_variable_list = { name = fetter_4_list target = $FETTER$ }
		is_target_in_variable_list = { name = fetter_5_list target = $FETTER$ }
	}
}

is_fetter_of_trigger = {
	$OWNER$ = {
		OR = {
			is_target_in_variable_list = { name = fetter_1_list target = prev }
			is_target_in_variable_list = { name = fetter_2_list target = prev }
			is_target_in_variable_list = { name = fetter_3_list target = prev }
			is_target_in_variable_list = { name = fetter_4_list target = prev }
			is_target_in_variable_list = { name = fetter_5_list target = prev }
		}
	}
}
