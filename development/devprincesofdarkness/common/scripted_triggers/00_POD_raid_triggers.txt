#Controls the can_raid rule.

can_raid_trigger = {
	OR = {
		government_has_flag = government_can_raid_rule
		faith = { 
			OR = {
				has_doctrine = unreformed_faith_doctrine
				has_doctrine = roadofbeast_hostility_doctrine
				has_doctrine = roadofmetamorphosis_hostility_doctrine
				has_doctrine  = tenet_warmonger
			}
		}
	}
	NOT = {
		faith = {
			has_doctrine_parameter = holy_wars_forbidden
		}
	}
}
