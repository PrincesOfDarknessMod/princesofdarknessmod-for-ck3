build_find_lair_targets_trigger = {

	any_neighboring_top_liege_realm_owner = {
			count = all
			trigger_if = {
			limit = {
			POD_is_vampire_trigger = yes
			age < 1000
			root = {
				highest_held_title_tier >= 4
				can_start_scheme = {
					target = prev
					type = murder
					}
				}
			}
		add_to_temporary_list = find_lair_targets
		}
	}
	
	any_neighboring_realm_same_rank_owner = {
			count = all
			trigger_if = {
			limit = {
			POD_is_vampire_trigger = yes
			age < 1000
			OR = {
			diplomacy < 5
			martial < 5
			stewardship < 5
			intrigue < 5
			learning < 5
			prowess < 20
			}
			root = {
				can_start_scheme = {
					target = prev
					type = murder
					}
				}
			}
		add_to_temporary_list = find_lair_targets
		}
	}
}
