#Triggers for Shunned and Criminal Secret Types for POD

# SECRET VAMPIRE 
secret_vampire_is_valid_trigger = {
	$OWNER$ = { 
		NOT = { 
			has_trait = vampire 
		} 
	}
}

secret_vampire_is_shunned_trigger = {
	$OWNER$ = {
		OR = {
			faith = { has_doctrine_parameter = vampirism_shunned }
			any_liege_or_above = { faith = { has_doctrine_parameter = vampirism_shunned } }
		}
	}
}

secret_vampire_is_criminal_trigger = {
	$OWNER$ = {
		OR = {
			faith = { has_doctrine_parameter = vampirism_illegal }
			any_liege_or_above = { faith = { has_doctrine_parameter = vampirism_illegal } }
		}
	}
}
