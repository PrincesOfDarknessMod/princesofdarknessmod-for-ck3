POD_discover_true_name = {
	skill = learning
	hostile = yes
	
	cooldown = { months = 12 }
	uses_agents = no
	is_secret = yes
	base_secrecy = 10
	icon = icon_scheme_hostile
	
	power_per_skill_point = 2.0
	resistance_per_skill_point = 1.75
	
	minimum_progress_chance = 10
	maximum_progress_chance = 90
	minimum_success = 5
	maximum_secrecy = 95
	maximum_success = 95
	icon = icon_scheme_hostile

	success_desc = "POD_discover_true_name_SUCCESS_DESC"
	
	allow = {
		scope:owner = { 
			is_imprisoned = no
			has_trait = mummy
			is_east_mummy_trigger = no
		}
	}
	
	valid = {
	}

	base_success_chance = {
		base = 50
		#SCHEME OWNER#
		
		compare_modifier = {
			desc = POD_discover_true_name_my_learning
			target = scope:owner
			value = learning
			multiplier = 3
		}
	}

	on_ready = {
	
		if = {
			limit = { scope:owner = { NOT = { has_hook_of_type = { target = scope:target type = true_name_hook } } } }
			POD_discover_true_name_effect = {
				MASTER = scope:owner
				VICTIM = scope:target
			}
		}
		if = {
			limit = { 
				scope:target = { is_ruler = yes }
				scope:owner = { has_perk = echo_the_lesser_name_perk } 
			}
			scope:target = { 
				every_courtier_or_guest = { 
					limit = { scope:owner = { NOT = { has_hook_of_type = { target = prev type = true_name_hook } } } }
					POD_discover_true_name_effect = {
						MASTER = scope:owner
						VICTIM = this
					}
				}
			}
		}
		if = {
			limit = { 
				scope:target = { is_ruler = yes }
				scope:owner = { has_perk = echo_the_greater_name_perk } 
			}
			scope:target = { 
				every_vassal_or_below = { 
					limit = { scope:owner = { NOT = { has_hook_of_type = { target = prev type = true_name_hook } } } }
					POD_discover_true_name_effect = {
						MASTER = scope:owner
						VICTIM = this
					}
				}
			}
		}
		end_scheme = yes
	}

	on_monthly = {

	}
	
	on_invalidated = {

	}
}
