knighting = {
	
	skill = learning
	hostile = no
	# cooldown = { months = 2 }
	uses_agents = no
	is_secret = no
	power_per_skill_point = 20
	spymaster_power_per_skill_point = 0	
	resistance_per_skill_point = 0
	spymaster_resistance_per_skill_point = 0
	uses_resistance = no
	minimum_progress_chance = 95
	maximum_progress_chance = 99
	
	allow = {
		has_trait = supehunter
		#is_adult = yes
		scope:target = { 
			is_adult = yes 
			has_trait = mortal
			NOT = {
				has_trait = supehunter
				}
		}
		#scope:target.location = scope.owner.location
	}
	
	valid = {
		
	}

	base_success_chance = {
		base = 95
	}


	on_ready = {
		scheme_owner = {
			trigger_event = POD_hunteradoption.100 
		}
	}


	on_invalidated = {
	}

	on_monthly = {
		save_scope_as = scheme
		scheme_owner = { save_scope_as = owner }
		scheme_target = { save_scope_as = target }
		if = {
			limit = {
				scope:owner = { has_trait = order_member }
				scope:target = {
					NOT = { has_trait = order_member }
								}
							}
				scope:target = { add_trait = order_member }
			}
		#TODO_CD: Lifestyle gain?
	}
	success_desc = "KNIGHTING_SUCCESS_DESC"
}
