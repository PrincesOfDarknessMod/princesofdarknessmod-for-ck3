
herdify_mortal = {
	
	skill = diplomacy
	#cooldown = { months = 2 }
	uses_agents = no
	is_secret = no
	power_per_skill_point = 20
	spymaster_power_per_skill_point = 0	
	resistance_per_skill_point = 0
	spymaster_resistance_per_skill_point = 0
	uses_resistance = no
	minimum_progress_chance = 95
	maximum_progress_chance = 99
	
	allow = {
		is_vampire_trigger = yes
		is_adult = yes
		scope:target = { 
			is_adult = yes 
			is_vampire_trigger = no
			NOT = {
				has_opinion_modifier = {
					target  = scope:owner
					modifier = POD_herd_opinion
				
				}
			}
		}
	}
	
	valid = {
		
	}

	base_success_chance = {
		base = 50
		#SCHEME OWNER
		compare_modifier = {
			desc = herdify_my_diplomacy
			target = scope:owner
			value = diplomacy
			multiplier = 2
		}

	}


	on_ready = {
		scheme_owner = {
			trigger_event = POD_herdify.1 
		}
	}


	on_invalidated = {
	}

	on_monthly = {
		save_scope_as = scheme
		scheme_owner = { save_scope_as = owner }
		scheme_target = { save_scope_as = target }

		#TODO_CD: Lifestyle gain?
	}
	success_desc = HERDIFY_SUCCESS_DESC
}
