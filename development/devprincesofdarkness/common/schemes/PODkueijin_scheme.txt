PODsoulbond = {
	skill = intrigue
	hostile = yes
	
	cooldown = { months = 6 }
	uses_agents = no
	is_secret = yes
	base_secrecy = 10
	icon = icon_scheme_hostile
	
	power_per_skill_point = 2.0
	resistance_per_skill_point = 1.75
	
	minimum_progress_chance = 10
	maximum_progress_chance = 90
	minimum_success = 5
	maximum_secrecy = 95
	maximum_success = 95
	icon = icon_scheme_hostile

	success_desc = "PODsoulbond_SUCCESS_DESC"
	
	allow = {
		is_imprisoned = no
		scope:owner = { has_perk = soul_shackles_perk }
		scope:owner = { 
			NOT = { 
				has_hook_of_type = { 
					target = scope:target 
					type = soul_bond_hook 
				}
			}
		}
	}
	
	valid = {
		custom_description = {
			object = scope:target
			text = "I_have_imprisoned_target"
			scope:target = {
				NOT = { imprisoner = scope:owner }
			}
		}
		scope:target = {
			OR = {
				exists = location
				in_diplomatic_range = scope:owner
			}
		}
	}

	base_success_chance = {
		base = 50
		#SCHEME OWNER#
		
		compare_modifier = {
			desc = PODsoulbond_my_intrigue
			target = scope:owner
			value = intrigue
			multiplier = 3
		}
	}

	on_ready = {
		scheme_target = {
			POD_soulbond_effect = {
				MASTER = scheme_owner
				VICTIM = scheme_target
			}
		}
		end_scheme = yes
	}

	on_monthly = {

	}
	
	on_invalidated = {

	}
}
