PODsoulbond = {
	skill = intrigue
	hostile = yes
	
	cooldown = { months = 6 }
	uses_agents = no
	is_secret = yes
	base_secrecy = 10
	
	power_per_skill_point = 2.0
	resistance_per_skill_point = 1.75
	
	minimum_progress_chance = 10
	maximum_progress_chance = 90
	minimum_success = 5
	maximum_secrecy = 95
	maximum_success = 95
	icon = icon_scheme_hostile

	success_desc = "PODsoulbond_SUCCESS_DESC"
	
	allow = {
		is_imprisoned = no
		is_kueijin_trigger = yes
		scope:owner = { has_perk = soul_shackles_perk }
		scope:target = { 
			NOT = { 
				has_hook_of_type = { 
					target = scope:owner 
					type = soul_bond_hook 
				}
			}
		}
	}
	
	valid = {
		custom_description = {
			object = scope:target
			text = "I_have_imprisoned_target"
			scope:target = {
				NOT = { imprisoner = scope:owner }
			}
		}
		scope:target = {
			OR = {
				exists = location
				in_diplomatic_range = scope:owner
			}
		}
	}

	base_success_chance = {
		base = 50
		#SCHEME OWNER#
		
		compare_modifier = {
			desc = PODsoulbond_my_intrigue
			target = scope:owner
			value = intrigue
			multiplier = 3
		}
	}

	on_ready = {
		scheme_target = {
			POD_soulbond_effect = {
				MASTER = scheme_owner
				VICTIM = scheme_target
			}
		}
		end_scheme = yes
	}

	on_monthly = {

	}
	
	on_invalidated = {

	}
}

PODchi_mastery = {
	skill = intrigue
	hostile = yes
	
	cooldown = { years = 2 }
	uses_agents = no
	is_secret = yes
	
	power_per_skill_point = 3.0
	resistance_per_skill_point = 1.0
	
	minimum_progress_chance = 10
	maximum_progress_chance = 90
	minimum_success = 5
	maximum_secrecy = 95
	maximum_success = 95
	
	success_desc = "PODchi_mastery_SUCCESS_DESC"
	
	allow = {
		is_imprisoned = no
		is_kueijin_trigger = yes
		scope:owner = { has_perk = chi_mastery_perk }
	}
	
	valid = {
		custom_description = {
			object = scope:target
			text = "I_have_imprisoned_target"
			scope:target = {
				NOT = { imprisoner = scope:owner }
			}
		}
		scope:target = {
			OR = {
				exists = location
				in_diplomatic_range = scope:owner
			}
		}		
	}
	
	base_success_chance = {
		base = 0
		
		hostile_scheme_base_chance_modifier = yes
	}

	on_ready = {
	
		scheme_owner = {
			POD_chi_decrease_effect = { COUNT = 3 }
			duel = {
				skill = prowess
				value = scope:target.prowess
				desc = PODchi_mastery.duel.desc

				50 = {
					desc = PODchi_mastery.success	

					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					
					custom_tooltip = PODchi_mastery.duel.tt.attacker_wins
					
					scope:target = {
						add_character_modifier = {
							modifier = pod_defiled_modifier
							months = 6
						}
					
						if = {
							limit = {
								is_kueijin_trigger = yes
								has_trait = shih
							}
							POD_clean_chi_effect = yes
							add_character_modifier = yinchi0_modifier
							add_character_modifier = yangchi0_modifier
						}
						else_if = {
							limit = {
								POD_is_mortal_trigger = yes
								POD_is_supehunter_trigger = yes
							}
							POD_losing_health_level_effect = { COUNT = 9 }
						}
						else_if = {
							limit = {
								scope:target = {
									POD_is_vampire_trigger = yes
									POD_is_bloodgorged_trigger = yes
								}
							}
							hidden_effect = {
								remove_character_modifier = bloodhunger0_modifier
								remove_character_modifier = bloodhunger1_modifier
								remove_character_modifier = bloodhunger2_modifier
								remove_character_modifier = bloodhunger3_modifier
								remove_character_modifier = bloodhunger4_modifier
							}
							add_character_modifier = bloodhunger5_modifier
						}
						else_if = {
							limit = {
								scope:target = {
										POD_is_ghoul_trigger = yes
										POD_is_revenant_trigger = yes
								}
							}
							POD_skuldfulfilled_remove_bloodpotency_effect = yes
							POD_skuldfulfilled_remove_masquerade_effect = yes
							POD_skuldfulfilled_remove_disciplines_effect = yes
							POD_skuldfulfilled_remove_hunger_effect = yes
						}
					}
				}
				50 = {
					desc = PODchi_mastery.duel.failure

					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier 
					}
					
					custom_tooltip = PODchi_mastery.duel.tt.attacker_lost
				}
			}
		}
		
		end_scheme = yes
	}

	on_monthly = {

	}
	
	on_invalidated = {

	}
}
