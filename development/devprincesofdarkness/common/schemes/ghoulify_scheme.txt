# Sparc 

POD_ghoulify_mortal = {
	
	skill = diplomacy
	cooldown = { months = 6 }
	uses_agents = no
	is_secret = yes
	power_per_skill_point = 20
	spymaster_power_per_skill_point = 0	
	resistance_per_skill_point = 0
	spymaster_resistance_per_skill_point = 0
	uses_resistance = no
	minimum_progress_chance = 95
	maximum_progress_chance = 99
	
	allow = {
		is_vampire_trigger = yes
		is_adult = yes
		scope:target = {
			is_adult = yes 
			POD_can_be_ghouled_scheme_trigger = yes
			POD_is_supehunter_trigger = no 
			NOT = {
				has_opinion_modifier = {
					target  = scope:owner
					modifier = POD_ghoulthrall_opinion
				
				}
			}
		}
	}
	
	valid = {
		custom_description = {
			object = scope:target
			text = POD_ghoulify_mortal_interaction_limit
			scope:owner = {
				any_courtier_or_guest = {
						has_opinion_modifier ={
							target = scope:actor
							modifier = POD_ghoulthrall_opinion
						}	
						count < 5
				}
			}	
		}
	}

	base_success_chance = {
		base = 50
		#SCHEME OWNER
		compare_modifier = {
			desc = POD_ghoulify_mortal_my_diplomacy
			target = scope:owner
			value = diplomacy
			multiplier = 2
		}
	}


	on_ready = {
		scheme_owner = {
			trigger_event = POD_ghoulify.100
		}
	}


	on_invalidated = {
	}

	on_monthly = {
		save_scope_as = scheme
		scheme_owner = { save_scope_as = owner }
		scheme_target = { save_scope_as = target }

		#TODO_CD: Lifestyle gain?
	}
	success_desc = POD_GHOULIFY_MORTAL_SUCCESS_DESC
}

POD_ghoul_ghoulify = {
	
	skill = diplomacy
	cooldown = { months = 6 }
	uses_agents = no
	is_secret = yes
	power_per_skill_point = 20
	spymaster_power_per_skill_point = 0	
	resistance_per_skill_point = 0
	spymaster_resistance_per_skill_point = 0
	uses_resistance = no
	minimum_progress_chance = 95
	maximum_progress_chance = 99
	
	allow = {
		OR = {
		is_ghoul_trigger = yes
		is_revenant_trigger = yes
		}
		is_adult = yes
		scope:target = {
			is_adult = yes
			has_POD_public_vamp_blood_trigger = no
		}
	}
	
	valid = {
		trigger_if = {
				limit = {
				NOT = { dynasty = dynasty:13000007 }
				}
			scope:target = {
			POD_can_be_ghouled_scheme_trigger = yes
			POD_is_supehunter_trigger = no
				}
			}
		custom_description = {
			object = scope:target
			text = I_have_no_bloodbag
			scope:owner = {
			any_prisoner = {
				POD_is_vampire_trigger = yes
				}
			}
		}
	}

	base_success_chance = {
		base = 50
		#SCHEME OWNER
		compare_modifier = {
			desc = POD_ghoul_ghoulify_my_diplomacy
			target = scope:owner
			value = diplomacy
			multiplier = 2
		}
	}


	on_ready = {
		scheme_owner = {
			trigger_event = POD_ghoulify.200
		}
	}


	on_invalidated = {
	}

	on_monthly = {
		save_scope_as = scheme
		scheme_owner = { save_scope_as = owner }
		scheme_target = { save_scope_as = target }
		#TODO_CD: Lifestyle gain?
	}
	success_desc = POD_GHOUL_GHOULIFY_SUCCESS_DESC
}
