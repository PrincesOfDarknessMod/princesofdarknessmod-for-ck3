knight_discipline_attack = {
	type = knight

	is_valid = {
		scope:combat_side.enemy_side = {
			any_side_knight = {
				is_alive = yes
				this.prowess <= { value = root.prowess multiply = 2 }
				OR = {
					can_be_killed_physically_trigger = yes
					root = { can_hurt_wraith_trigger = yes }
				}
			}
		}
	}

	chance = {
		base = 0
		knight_discipline_attack_wound_perk_modifier = yes
		knight_discipline_attack_severe_injury_perk_modifier = yes
		knight_discipline_attack_maul_perk_modifier = yes
		knight_discipline_attack_maim_perk_modifier = yes
		knight_discipline_attack_kill_perk_modifier = yes
		modifier = {
			factor = 1.5
			has_perk = stalwart_leader_perk
			is_ai = no
		}
		modifier = {
			factor = 1.15
			has_perk = stalwart_leader_perk
			is_ai = yes
		}
		modifier = {
			factor = 1.25
			exists = dynasty
			dynasty = {
				has_dynasty_perk = warfare_legacy_3
			}
		}
		modifier = {
			factor = 0.25
			OR = {
				has_trait = one_legged
				has_trait = disfigured
				has_trait = one_eyed
				has_trait = maimed
			}
		}
		modifier = {
			factor = 2
			has_trait = potenceadvanced
		}
		modifier = {
			factor = 2
			has_trait = celerityadvanced
		}
	}

	effect = {
		random_list = {
			50 = {
				trigger = {
					scope:combat_side.enemy_side = {
						any_side_knight = {
							is_alive = yes
							this.prowess <= { value = root.prowess multiply = 2 }
							OR = {
								can_be_killed_physically_trigger = yes
								root = { can_hurt_wraith_trigger = yes }
							}
						}
					}
				}
				scope:combat_side.enemy_side = {
					random_side_knight = {
						limit = {
							is_alive = yes
							this.prowess <= { value = root.prowess multiply = 2 }
							OR = {
								can_be_killed_physically_trigger = yes
								root = { can_hurt_wraith_trigger = yes }
							}
						}
						save_temporary_scope_as = enemy_knight
					}
				}
				scope:enemy_knight = {
					scope:combat_side = {
						battle_event = {
							key = "knight_wounded_by_enemy"
							left_portrait = scope:enemy_knight
							right_portrait = root
							type = wound
						}
					}
					increase_wounds_effect = { REASON = fight }
				}
			}
			0 = { # Severe injury
				trigger = {
					scope:combat_side.enemy_side = {
						any_side_knight = {
							is_alive = yes
							this.prowess <= { value = root.prowess multiply = 1.5 }
							OR = {
								can_be_killed_physically_trigger = yes

					random_side_knight = {
						limit = {

							this.prowess <= { value = root.prowess multiply = 1.5 }

						save_temporary_scope_as = enemy_knight
					}

							key = "knight_wounded_by_enemy"

							root = { can_hurt_wraith_trigger = yes }
						}
					}
				}
			}
			scope:combat_side.enemy_side = {
				random_side_knight = {
					limit = {
						this.prowess <= { value = root.prowess multiply = 0.8 }
						OR = {
							can_be_killed_physically_trigger = yes
							root = { can_hurt_wraith_trigger = yes }
						}
					}
					save_temporary_scope_as = enemy_knight
				}
			}
			if = {
				limit = {
					root = { is_alive = yes }
					scope:enemy_knight = { is_alive = yes }
				}
				scope:combat_side = {
					battle_event = {
						key = "berserker_rage_killed_enemy"
						left_portrait = root
						right_portrait = scope:enemy_knight
						type = death
						target_right = yes
					}
				}

				add_trait = berserker
				scope:enemy_knight = {
					pod_death_effect = {
						DEATH_REASON = death_head_ripped_off
						KILLER = root
					}
				}
			}
		}
		else = {
			if = {
				limit = {
					root = { is_alive = yes }
				}
				scope:combat_side = {
					battle_event = {
						key = "berserker_rage_no_enemy"
						left_portrait = root
						type = death
						target_right = yes
					}
				}
				add_trait = berserker
			}
		}
	}
}

knight_wounded = {
	type = knight

	is_valid = {
		NOT = {
			has_trait_rank = {
				trait = wounded
				rank = 3
			}
		}
		can_be_killed_physically_trigger = yes
	}

	chance = {
		base = 10
		modifier = {
			factor = { # Prowess directly reduces chance of injury/death
				value = 40 # Value at which 'immortality' happens
				subtract = prowess
				if = {
					limit = {
						has_court_position = garuda_court_position
					}
					add = garuda_prowess_svalue
				}
				divide = 40
				min = 0.01 # Everyone dies.
			}
		}
		modifier = { # Reduce chance when outnumbering the enemy
			factor = {
				value = scope:combat_side.enemy_side.side_strength
				divide = scope:combat_side.side_strength
				multiply = 1.4 # Outnumbering 2-1 reduces chance by 30%
				max = 1
			}
			scope:combat_side.side_strength > scope:combat_side.enemy_side.side_strength
		}
		modifier = {
			factor = 1.05
			culture = {
