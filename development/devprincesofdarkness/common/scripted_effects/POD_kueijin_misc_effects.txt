kueijin_party_default_setup_effect = {
	scope:activity = {
		#Add any special setups here (multiple player characters, etc.), BEFORE the every_participant effect

		activity_owner = {
			if = {
				limit = {
					has_character_flag = kueijin_birthday
				}
				trigger_event = {
					on_action = kueijin_birthday_default_host_event_selection
					days = feast_initial_event_delay
				}
			}
			else_if = {
				limit = {
					has_character_flag = kueijin_dragon_boat_festival
				}
				trigger_event = {
					on_action = kueijin_dragon_boat_festival_default_host_event_selection
					days = feast_initial_event_delay
				}
			}
		}

		every_participant = { #Effect for remaining (most) guests
			limit = {
				NOT = { this = scope:activity.activity_owner }
			}
			if = {
				limit = {
					scope:activity.activity_owner = {
						has_character_flag = kueijin_birthday
					}
				}
				trigger_event = {
					on_action = kueijin_birthday_default_event_selection
					days = feast_initial_event_delay
				}
			}
			else_if = {
				limit = {
					scope:activity.activity_owner = {
						has_character_flag = kueijin_dragon_boat_festival
					}
				}
				trigger_event = {
					on_action = kueijin_dragon_boat_festival_default_event_selection
					days = feast_initial_event_delay
				}
			}
		}
	}
}

			}

			}
		}
	}
	else = {
		if = {
			limit = {
				scope:activity.activity_owner = {
					has_character_flag = kueijin_birthday
				}
			}
			trigger_event = {
				on_action = kueijin_birthday_default_event_selection
				days = { min_feast_event_spacing max_feast_event_spacing }
			}
		}
		else_if = {
			limit = {
				scope:activity.activity_owner = {
					has_character_flag = kueijin_dragon_boat_festival
				}
			}
			trigger_event = {
				on_action = kueijin_dragon_boat_festival_default_event_selection
				days = { min_feast_event_spacing max_feast_event_spacing }
			}
		}
	}
}

kueijin_party_wrap_up_or_fire_random_event_effect = {
	if = {
		limit = {
			NOT = { exists = local_var:num_of_party_events }
		}
		set_local_variable = {
			name = num_of_party_events
			value = 1
		}
		kueijin_party_fire_correct_default_event_effect = yes
	}
	else = {
		change_local_variable = {
			name = num_of_party_events
			add = 1
		}
		if = {
			limit = { # Check the number of events fired
				local_var:num_of_party_events < max_default_feast_events
			}
			# If below the max number check whether to fire more
			if = {
				limit = {
					local_var:num_of_party_events >= min_default_feast_events
				}
				random_list = {
					0 = {
						modifier = { add = chance_of_extra_feast_events }
						kueijin_party_fire_correct_default_event_effect = yes
					}
					100 = {
						modifier = {
							add = {
								subtract = chance_of_extra_feast_events
							}
						}
						kueijin_party_concluded_my_events_effect = yes
					}
				}
			}
			else = {
				kueijin_party_fire_correct_default_event_effect = yes
			}
		}
		else = {
			kueijin_party_concluded_my_events_effect = yes
		}
	}
}

hosted_successful_party_effect = {
	add_prestige = medium_prestige_gain
	scope:activity = {
		every_participant = {
			limit = { NOT = { this = root } }
			custom = every_guest_scope_tt
			add_opinion = {
				target = root
				modifier = kueijin_party_hosted_successful_party
			}
		}
	}
	reveler_lifestyle_rank_up_check_effect = yes
	hidden_effect = {
		if = {
			limit = { exists = scope:spouse }
			add_opinion = { #Your opinion of your Spouse increases, considering they did most of the work
				target = scope:spouse
				modifier = kueijin_party_spouse_hosted_successful_party_opinion
			}
		}
	}
}
