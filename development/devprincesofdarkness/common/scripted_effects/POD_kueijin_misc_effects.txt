POD_kueijin_initialization_effect = {
	if = {
		limit = {
			has_dharma_trigger = no
		}
		POD_set_dharma_effect = yes
	}
	if = {
		limit = {
			has_yinyang_trigger = no
		}
		POD_set_yinyang_effect = yes
	}
	if = {
		limit = {
			has_chi_trigger = no
		}
		POD_set_chi_effect = yes
	}
}

POD_set_dharma_effect = {
	if = {
		limit = {		
			age >= 1000
		}
		add_trait = dharma9
	}
	else_if = {
		limit = {		
			age >= 500
		}
		add_trait = dharma8
	}
	else_if = {
		limit = {		
			age >= 250
		}
		add_trait = dharma7
	}
	else_if = {
		limit = {		
			age >= 100
		}
		add_trait = dharma6
	}
	else = {
		add_trait = dharma5
	}
}

POD_clean_regular_activity_effect = {
	if = {
		limit = {
			has_character_modifier = regular_meditating_modifier
		}
		remove_character_modifier = regular_meditating_modifier
	}
	if = {
		limit = {
			has_character_modifier = regular_treking_modifier
		}
		remove_character_modifier = regular_treking_modifier
	}
	if = {
		limit = {
			has_character_modifier = regular_mortal_living_modifier
		}
		remove_character_modifier = regular_mortal_living_modifier
	}
	if = {
		limit = {
			has_character_modifier = traveling_yin_world_modifier
		}
		remove_character_modifier = traveling_yin_world_modifier
	}
	if = {
		limit = {
			has_character_modifier = traveling_yang_world_modifier
		}
		remove_character_modifier = traveling_yang_world_modifier
	}
}
