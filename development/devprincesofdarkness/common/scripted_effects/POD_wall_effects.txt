POD_set_or_rename_local_shroud_or_wall_modifier_effect = {
	if = {
		limit = {
			OR = {
				is_kueijin_trigger = yes
				POD_lesser_chi_user_trigger = yes
			}
		}
		if = {
			limit = {
				has_wall_trigger = no
			}
			if = {
				limit = {
					has_wall_trigger = yes
				}
				POD_rename_local_wall_modifier_to_shroud_modifier_effect = yes
			}
			else = {
				POD_set_local_wall_modifier_effect = yes
			}
		}
	}
	else = {
		if = {
			limit = {
				has_shroud_trigger = no
			}
			if = {
				limit = {
					has_wall_trigger = yes
				}
				POD_rename_local_shroud_modifier_to_wall_modifier_effect = yes
			}
			else = {
				POD_set_local_shroud_modifier_effect = yes
			}
		}
	}
}
