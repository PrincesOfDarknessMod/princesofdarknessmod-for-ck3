POD_add_missing_race_traits_effect = {
	if = {
		limit = {
			has_POD_race_trait_trigger = no
		}
		random_list = {
			10 = {
				add_trait = vampire
			}
			10 = {
				add_trait = supehunter
			}
			1 = {
				add_trait = kueijin
			}
			1 = {
				add_trait = ghoul
			}
			1 = {
				add_trait = revenant
			}
			1 = {
				add_trait = mummy
			}
			1 = {
				add_trait = wraith
			}
		}
	}
}
	
POD_add_missing_supernatural_traits_effect = {
	if = {
		limit = {
			POD_is_vampire_trigger = yes
		}		
		if = {
			limit = {
				has_POD_vampire_clan_trigger = no
			}
			POD_character_vampire_clan_initialization_effect = yes
		}
		
		if = {
			limit = {
				has_POD_vampire_generation_trigger = no
			}
			add_secret = { type = secret_vampgeneration12 }
		}
	}
	if = {
		limit = {
			is_kueijin_trigger = yes
		}				
		trigger_event = {
			id = podkueijin.10
			days = 3
		}
	}
	if = {
		limit = {
			has_trait = dhampyr
		}		
		POD_dhampyr_initialization_effect = yes
	}
	if = {
		limit = {
			has_trait = shih
		}		
		POD_shih_initialization_effect = yes
	}
	if = {
		limit = {
			has_trait = mummy
		}		
		POD_mummy_initialization_effect = yes
	}
	if = {
		limit = {
			has_trait = wraith
		}		
		POD_wraith_initialization_effect = yes
	}
}
	
POD_remove_wrong_supernatural_traits_effect = {
	if = {
		limit = {
			has_POD_vamp_blood_trigger = no
		}
		if = {
			limit = { has_trait = risen }
			POD_remove_vampire_non_risen_discipline_traits_effect = yes
		}
		else = {	
			if = {
				limit = { has_trait = wraith }
				add_trait = arcanos_embody
			}
			POD_remove_vampire_discipline_traits_effect = yes
		}
	}
	if = {
		limit = {	
			POD_is_vampire_trigger = no
		}		
		POD_remove_vampire_generation_traits_effect = yes
		POD_remove_vampire_clan_traits_effect = yes
		if = {
			limit = { is_kueijin_trigger = no }
			remove_trait = diablerist
		}
	}
	if = {
		limit = {
			is_kueijin_trigger = no
		}
		POD_clean_dharma_effect = yes
		POD_clean_yinyang_effect = yes
	}
	if = {
		limit = {
			POD_is_supehunter_trigger = no
		}
		POD_remove_hunter_traits_effect = yes
	}
	if = {
		limit = {
			POD_is_mummy_trigger = no
		}
		POD_remove_mummy_traits_effect = yes
	}
	if = {
		limit = {
			POD_is_wraith_trigger = no
		}
		POD_remove_wraith_traits_effect = yes
	}
}

POD_adopt_correct_religion_effect = {
	if = {
		limit = {
			has_trait = vampire
			OR = { 
				NOT = { religion = { is_in_family = rf_kindred } }
				religion = religion:calomenasforsaken_religion
			}
		}
		POD_character_vampire_faith_initialization_effect = yes
	}
	else_if = {
		limit = {
			OR = {
				has_trait = kueijin 
				has_trait = dhampyr
			}
			NOT = { religion = religion:kueijin_religion }
		}
		POD_character_vampire_faith_initialization_effect = yes
	}
	else_if = {
		limit = {
			OR = {
				has_trait = ghoul
				has_trait = revenant
				has_trait = bloodgorged
			}
			OR = { 
				NOT = { religion = { is_in_family = rf_kindred } }
				religion = religion:kueijin_religion
			}
		}
		POD_character_revenant_faith_initialization_effect = yes
	}
	else_if = {
		limit = {
			has_trait = supehunter
			NOT = { religion = { is_in_family = rf_hunters } }
		}
		if = {
			limit = {
				has_inquisitor_dynasty_trigger = yes
				NOT = { faith = faith:shadowinquisition }
			}
			set_character_faith = faith:shadowinquisition
		}
		else_if = {
			limit = {
				has_trait = shih
				NOT = { faith = faith:shih_faith }
			}
			set_character_faith = faith:shih_faith
		}
		else = {
			set_character_faith = faith:independenthunters
		}
	}
	else_if = {
		limit = {
			OR = {
				has_trait = werewolf
				has_trait = kinfolk
			}	
			NOT = { 
				OR = {
					religion = { is_in_family = rf_gaia }
					religion = { is_in_family = rf_wyrm }
				}	
			}
		}
		set_character_faith = faith:bonegnawer
	}
	else_if = {
		limit = {
			has_trait = mummy
			NOT = { religion = { is_in_family = rf_mummies } }
		}
		POD_character_mummy_faith_initiation_effect = yes
	}
	else_if = {
		limit = {
			has_trait = wraith
			NOT = { religion = { is_in_family = rf_wraiths } }
		}
		POD_character_wraith_faith_initiation_effect = yes
	}
}

POD_adopt_correct_goverment_effect = {
	if = {
		limit = {
			OR = {
				has_trait = kueijin 
				has_trait = dhampyr
			}
		}
		if = {
			limit = {
				exists = liege
				liege = { has_government = august_court_government }
			}
			change_government = august_court_government
		}
		else = {
			change_government = heretical_court_government
		}
	}
	else_if = {
		limit = {
			has_POD_public_vamp_blood_trigger = yes
			is_forsaken_or_gantenbein_trigger = no
		}
		if = {
			limit = {
				exists = liege
				liege = { has_government = free_state_government }
			}
			change_government = free_state_government
		}
		else = {
			change_government = traditions_government
		}
	}
	else_if = {
		limit = {
			has_trait = shih
			has_global_variable = celestial_army_restored
		}
		change_government = celestial_army_government
	}
	else_if = {
		limit = {
			OR = {
				POD_is_mortal_trigger = yes
				POD_is_supehunter_trigger = yes
			}
			faith = faith:shadowinquisition
		}
		change_government = inquisition_government
	}
	else_if = {
		limit = {
			OR = {
				has_trait = werewolf
				has_trait = kinfolk
			}
		}
		# To not flod log with set illegal gov warning
		capital_province = {
			if = {
				limit = { NOT = { has_holding_type = tribal_holding } }
				set_holding_type = tribal_holding
			}
		}
		change_government = litany_government
	}
	else_if = {
		limit = {
			has_trait = wraith
		}
		if = {
			limit = {
				exists = liege
				liege = { has_government = hierarchy_government }
			}
			change_government = hierarchy_government
		}
		else_if = {
			limit = {
				exists = liege
				liege = { has_government = guild_government }
			}
			change_government = guild_government
		}
		else_if = {
			limit = {
				exists = liege
				liege = { has_government = heretic_government }
			}
			change_government = heretic_government
		}
		else_if = {
			limit = {
				exists = liege
				liege = { has_government = far_shore_government }
			}
			change_government = far_shore_government
		}
		else_if = {
			limit = {
				exists = liege
				liege = { has_government = ferryman_government }
			}
			change_government = ferryman_government
		}
		else = {
			change_government = renegade_government
		}
	}
	else = {
		change_government = indep_government
	}
}
