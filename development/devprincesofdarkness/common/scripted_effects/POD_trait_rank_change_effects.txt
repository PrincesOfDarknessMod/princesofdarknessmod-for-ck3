POD_become_hideous_effect = { 
	if = {
		limit = {
			has_trait = beauty_bad_3
			has_trait = kueijin
			NOT = { has_character_flag = POD_permanent_ching_shih } #Using vanilla Homely / Beautiful trait to represent the process for now
		}
		# Using vanilla disfigured trait to represent Ch'ing shih state for now
		trigger_event = podkueijin.201
	}
	else_if = {
		limit = {
			has_trait = beauty_bad_2
		}
		remove_trait = beauty_bad_2
		add_trait = beauty_bad_3
	}
	else_if = {
		limit = {
			has_trait = beauty_bad_1
		}
		remove_trait = beauty_bad_1
		add_trait = beauty_bad_2
	}
	else_if = {
		limit = {
			has_trait = beauty_good_3
		}
		remove_trait = beauty_good_3
		add_trait = beauty_good_2
	}
	else_if = {
		limit = {
			has_trait = beauty_good_2					   
		}
		remove_trait = beauty_good_2
		add_trait = beauty_good_1
	}
	else_if = {
		limit = {
			has_trait = beauty_good_1
		}
		remove_trait = beauty_good_1		 
	}
	else = {
		add_trait = beauty_bad_1			 
	}
}
