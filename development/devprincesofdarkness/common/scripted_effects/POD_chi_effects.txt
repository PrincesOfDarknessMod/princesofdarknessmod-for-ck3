POD_reset_chi_effect = {
	if = {
		limit = {
			yinchi_lack_1_or_more_trigger = yes
		}
		hidden_effect = {
			POD_clean_yinchi_effect = yes
		}
		if = {
			limit = {
				has_character_flag = devore_chi_even_overdose
			}
			add_character_modifier = yinchi10_modifier	
			POD_yinchi_overdose_effect = yes
		}
		else = {
			POD_set_yinchi_effect = yes
		}
	}
	
	if = {
		limit = {
			yangchi_lack_1_or_more_trigger = yes
		}
		hidden_effect = {
			POD_clean_yangchi_effect = yes
		}
		if = {
			limit = {
				has_character_flag = devore_chi_even_overdose
			}
			add_character_modifier = yangchi10_modifier
			POD_yangchi_overdose_effect = yes
		}
		else = {
			POD_set_yangchi_effect = yes
		}
	}
	
	POD_chi_fill_effect = yes
}

		}
	}

POD_chi_decrease_effect = {
	
	
	if = {
		limit = {
			NOT = { has_character_flag = first_spend_yang }
			OR = {
				has_character_flag = first_spend_yin
				trigger_if = {
					limit = { 
						OR = {
							has_character_modifier = yinchi9_modifier 
							has_character_modifier = yinchi8_modifier
						}
					}
					has_yangchi_9_or_greater_trigger = no
				}
				trigger_else_if = {
					limit = { 
						OR = {
							has_character_modifier = yinchi7_modifier 
							has_character_modifier = yinchi6_modifier
						}
					}
					has_yangchi_7_or_greater_trigger = no
				}
				trigger_else_if = {
					limit = { 
						OR = {
							has_character_modifier = yinchi5_modifier 
							has_character_modifier = yinchi4_modifier
						}
					}
					has_yangchi_5_or_greater_trigger = no
				}
				trigger_else_if = {
					limit = { 
						OR = {
							has_character_modifier = yinchi3_modifier 
							has_character_modifier = yinchi2_modifier
						}
					}
					has_yangchi_3_or_greater_trigger = no
				}
				trigger_else = {
					OR = {
						has_character_modifier = yinchi10_modifier
						has_character_modifier = yangchi0_modifier
					}
				}
			}
		}
		POD_yinchi_decrease_effect = { COUNT = $COUNT$ }
	}
	else = {

	}
