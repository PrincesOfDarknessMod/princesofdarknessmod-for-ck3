	custom_tooltip = POD_chi_decrease_effect.tt
	
	if = {
		limit = {
			has_character_flag = yang_preference
			NOR = {
				has_character_modifier = yinchi0_modifier
				OR = {
					has_character_modifier = yangchi3_modifier
					has_character_modifier = yangchi2_modifier
					has_character_modifier = yangchi1_modifier
				}
			}
		}
		POD_yinchi_decrease_effect = yes
	}
	else_if = {
		limit = {
			has_character_flag = yin_preference
			NOR = {
				has_character_modifier = yangchi3_modifier
				OR = {
					has_character_modifier = yinchi3_modifier
					has_character_modifier = yinchi2_modifier
					has_character_modifier = yinchi1_modifier
				}
			}
		}
		POD_yangchi_decrease_effect = yes
	}
	else = {
		POD_balancedchi_decrease_effect = yes
	}
	POD_balance_yinyang_for_player_effect = yes
}
