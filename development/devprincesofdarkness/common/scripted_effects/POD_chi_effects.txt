###Chi hunger###

POD_chi_fill_effect = {
	if = {
		limit = {
			has_trait = yang_imbalanced
		}
		if = {
			limit = {
				chi_hungry_trigger = no
				has_character_modifier = chi_ravenous_modifier
			}
			remove_character_modifier = chi_ravenous_modifier
		}
	}
	else = {
		if	= {
			limit = {
				chi_ravenous_trigger = no
				has_character_modifier = chi_ravenous_modifier
			}
			hidden_effect = {
				remove_character_modifier = chi_ravenous_modifier
			}
			add_character_modifier = chi_hungry_modifier
		}
		if	= {
			limit = {
				chi_hungry_trigger = no
				has_character_modifier = chi_hungry_modifier
			}
			remove_character_modifier = chi_hungry_modifier
		}
	}
}

					has_trait = yin_imbalanced

					has_trait = yang_imbalanced

				}
			}
		}

				modifier = attuned_to_yin_modifier
				months = 1
			}

		}

			limit = {

			}
			add_character_modifier = {
				modifier = attuned_to_yang_modifier
				months = 1
			}
			POD_yangchi_decrease_effect = { COUNT = 1 }	
		}
		else = {
			random_list = {
				50 = {

				}
				50 = {
					POD_yangchi_decrease_effect = { COUNT = 1 }
