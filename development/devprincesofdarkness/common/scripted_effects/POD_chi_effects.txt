###Chi hunger###

POD_chi_fill_effect = {
	if = {
		limit = {
			has_trait = yang_imbalanced
		}
		if	= {
			limit = {
				chi_hungry_trigger = no
			}
			remove_character_modifier = chi_ravenous_modifier
		}
	}
	else = {
		if	= {
			limit = {
				chi_ravenous_trigger = no
			}
			remove_character_modifier = chi_ravenous_modifier
		}
		if	= {
			limit = {
				chi_hungry_trigger = no
			}
			remove_character_modifier = chi_hungry_modifier
		}
	}
}

POD_chi_hunger_effect = {
	if = {
		limit = {
			chi_hungry_trigger = yes
		}
		if = {
			limit = {
				chi_ravenous_trigger = yes
			}
			if = {
				limit = {
					has_trait = yang_imbalanced
				}
				trigger_event = podkueijin.800
			}
			else = {
				add_character_modifier = chi_ravenous_modifier
			}
		}
		else = {
			if = {
				limit = {
					has_trait = yang_imbalanced
				}
				add_character_modifier = chi_ravenous_modifier
			}
			else = {
				add_character_modifier = chi_hungry_modifier
			}
		}
	}
}

				modifier = attuned_to_yin_modifier
				months = 1
			}

		}

				}

					POD_yangchi_decrease_effect = yes
