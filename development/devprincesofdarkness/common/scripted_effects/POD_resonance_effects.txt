POD_clear_resonance_effect = {

	if = {
		limit = {
				has_character_flag = choleric_resonance
		}
		remove_character_flag = choleric_resonance
	}

	if = {
		limit = {
				has_character_flag = melancholic_resonance
		}
		remove_character_flag = melancholic_resonance
	}
	
	if = {
		limit = {
				has_character_flag = phlegmatic_resonance
		}
		remove_character_flag = phlegmatic_resonance
	}	
	
	if = {
		limit = {
				has_character_flag = sanguine_resonance
		}
		remove_character_flag = sanguine_resonance
	}		

}

POD_resonance_feeding_sip_effect = {
		
		if = {
			limit = {
				scope:victim = {
					has_character_flag = choleric_resonance
				}
			}
			add_choleric_lifestyle_xp = miniscule_lifestyle_xp
		}

		if = {
			limit = {
				scope:victim = {
					has_character_flag = melancholic_resonance
				}
			}
			add_melancholy_lifestyle_xp = miniscule_lifestyle_xp
		}

		if = {
			limit = {
				scope:victim = {
					has_character_flag = phlegmatic_resonance
				}
			}
			add_phlegmatic_lifestyle_xp = miniscule_lifestyle_xp
		}

		if = {
			limit = {
				scope:victim = {
					has_character_flag = sanguine_resonance
				}
			}
			add_sanguine_lifestyle_xp = miniscule_lifestyle_xp
		}
		POD_ventrue_bane_effect = yes
}
