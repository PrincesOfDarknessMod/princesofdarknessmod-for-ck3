	add_character_modifier = wraith_potency_1

POD_bind_wraith_effect = {

	
	if = {
		limit = {
			$NECROMANCER$ = {
				has_hook_of_type = {
					target = $WRAITH$
					type = necromancy_bind_hook
				}	
			}
		}
		$NECROMANCER$ = {
			
			add_hook = {
				target = $WRAITH$
				type = necromancy_bind_strong_hook
			}
			reverse_add_opinion = {
				modifier = POD_binding_strong_opinion
				target = $WRAITH$
			}
		}
		$WRAITH$ = {
			add_character_flag = {flag = wraith_bound days = 3560}
			set_variable = {name = necromancer value = $NECROMANCER$}
			remove_opinion = {
				target = $NECROMANCER$
				modifier = POD_binding_opinion
			}
		}
		
	}
	else_if = {
		limit = {
			$NECROMANCER$ = {
				has_hook_of_type = {
					target = $WRAITH$
					type = necromancy_bind_strong_hook
				}	
			}
		}
		$NECROMANCER$ = {
			remove_hook = {
				target = $WRAITH$
				type = necromancy_bind_strong_hook
			}	
			add_hook = {
				target = $WRAITH$
				type = necromancy_bind_permanent_hook
			}
			reverse_add_opinion = {
				modifier = POD_binding_permanent_opinion
				target = $WRAITH$
			}
		}
		$WRAITH$ = {
			add_character_flag = wraith_bound 
			set_variable = {name = necromancer value = $NECROMANCER$}
			remove_opinion = {
				target = $NECROMANCER$
				modifier = POD_binding_strong_opinion
			}
		}
		
	
	}
	else = {
		$NECROMANCER$ = {
			add_hook = {
				target = $WRAITH$
				type = necromancy_bind_hook
			}
			reverse_add_opinion = {
				modifier = POD_binding_opinion
				target = $WRAITH$
			}
		}
		$WRAITH$ = {
			add_character_flag = {flag = wraith_bound days = 3560}
			set_variable = {name = necromancer value = $NECROMANCER$}
		}
	
	}
	$NECROMANCER$ = {
		add_to_variable_list = {
			name = bound_wraiths
			target = $WRAITH$
		}
	}


}

POD_wraith_weak_possession_effect = {
	$WRAITH$ = {
		add_hook = {
			target = $PUPPET$
			type = possessed_weak_hook
		}
		reverse_add_opinion = {
			modifier = POD_weak_possession_opinion
			target = $PUPPET$
		}
		add_character_flag = { 
			flag = in_possession
			days = 3650
		}
		set_variable = { name = puppet value = $PUPPET$}
		if = {
			limit = {has_character_flag = wraith_bound}
			var:necromancer = {
				send_interface_toast = {
					title = POD_wraith_possession_toast_title
					custom_tooltip = POD_wraith_possession_toast_text
					left_icon = scope:actor
					right_icon = scope:recipient
					add_hook = {
						target = $PUPPET$
						type = necromancy_bind_hook
					}
					reverse_add_opinion = {
						modifier = POD_possessed_by_bound_ghost
						target = $PUPPET$
					}

				}
				
				
			}
		}
		else_if = {
			limit = { has_variable = necromancer }
			var:necromancer = {
				send_interface_toast = {
					title = POD_wraith_possession_toast_title
					custom_tooltip = POD_wraith_possession_toast_text
					left_icon = scope:actor
					right_icon = scope:recipient					
					reverse_add_opinion = {
						modifier = POD_possessed_by_bound_ghost
						target = $PUPPET$
					}

				}
				
				
			}
		}
	}
	$PUPPET$ = {
		set_variable = {name = wraithprowess value = { value = $WRAITH$.prowess divide = 2 } }
		add_prowess_skill = { value = $WRAITH$.prowess divide = 2 }
		set_variable = {name = wraith value = $WRAITH$}
		add_character_flag = { 
			flag = possessed
			days = 3650
		}
	}
	

}

			}

		}

			}
		}

		}

			type = possessed_weak_hook
			target = $PUPPET$
		}

					target = $PUPPET$

				}

					target = $PUPPET$

				}

						target = scope:necromancer
						modifier = POD_possessed_by_bound_ghost
					}
				}
			}
		}
	
	}
