POD_become_wraith_effect = {
	if = {
		limit = { has_POD_immortality_trigger = yes }
		set_local_variable = {
			name = saved_immortal_age
			value = effective_age
		}
	}
	hidden_effect = {
		if = {
			limit = { has_perk = necromancy_possession_perk } 
			add_trait = risen
		}
	}
	hidden_effect = {
		remove_trait = vampire
		remove_trait = ghoul
		remove_trait = revenant
		remove_trait = bloodgorged
		remove_trait = supehunter
		remove_trait = werewolf
		remove_trait = kinfolk

		remove_trait = mortal

		remove_trait = massregnant
		remove_trait = unbondable
		remove_trait = sire
		remove_trait = masssire
		remove_trait = diablerist
		remove_trait = infernalist
		remove_trait = wight
		remove_trait = exposed
		
		POD_remove_vampire_clan_traits_effect = yes
		POD_remove_vampire_generation_traits_effect = yes
		POD_remove_vampire_discipline_traits_effect = yes
		POD_remove_vampire_predatortype_traits_effect = yes
		POD_remove_hunter_traits_effect = yes
		POD_clean_dharma_effect = yes
		POD_clean_yinyang_effect = yes
		POD_remove_mummy_traits_effect = yes
		
		POD_remove_bloodpotency_modifiers_effect = yes
		POD_remove_masquerade_modifiers_effect = yes
		POD_remove_hunger_modifiers_effect = yes
		POD_clean_chi_effect = yes
		POD_clean_sekhem_effect = yes
		
		POD_remove_supernatural_perks_effect = yes
		POD_remove_supernatural_secrets_effect = yes

		save_scope_as = victim
		every_relation = {
			type = possessesing_wraith
			POD_wraith_end_possession = {
				WRAITH = this
				PUPPET = scope:victim
			}
		}
		
		every_relation = {
			type = herd_owner
			remove_hook = {
				target = scope:victim
				type = herd_hook
			}
			scope:victim = {
				remove_relation_herd_owner = prev
				remove_character_flag = blocked_from_leaving
				remove_character_flag = fed_on_auto_sip
				remove_character_flag = fed_on_auto_drink
				remove_character_flag = fed_on_auto_lot
			}
		}
		
		every_relation = {
			type = ghoul_domitor
			remove_hook = {
				target = scope:victim
				type = strong_bloodoath_hook
			}
			scope:victim = {
				remove_relation_ghoul_domitor = prev
				remove_character_flag = blocked_from_leaving
				remove_character_flag = blood_bonded
			}
		}
	}
	
	add_trait = wraith
	if = {
		limit = { has_trait = risen } # Change diplay order
		hidden_effect = { remove_trait = risen }
		add_trait_force_tooltip = risen
	}
	if = {
		limit = { 
			NOR = {
				has_character_modifier = wraith_potency_1
				has_character_modifier = wraith_potency_2
				has_character_modifier = wraith_potency_3
				has_character_modifier = wraith_potency_4
				has_character_modifier = wraith_potency_5
				has_character_modifier = wraith_potency_6
				has_character_modifier = wraith_potency_7
				has_character_modifier = wraith_potency_8
				has_character_modifier = wraith_potency_9
				has_character_modifier = wraith_potency_10
			}
		}
		add_character_modifier = wraith_potency_1
		set_variable = {name = wraith_potency value = 1}
	}
	add_character_flag = immune_to_disease
	if = {
		limit = { exists = local_var:saved_immortal_age }
		set_immortal_age = local_var:saved_immortal_age
	}

POD_bind_wraith_effect = {

	
	if = {
		limit = {
			$NECROMANCER$ = {
				has_hook_of_type = {
					target = $WRAITH$
					type = necromancy_bind_hook
				}	
			}
		}
		$NECROMANCER$ = {
			
			add_hook = {
				target = $WRAITH$
				type = necromancy_bind_strong_hook
			}
			reverse_add_opinion = {
				modifier = POD_binding_strong_opinion
				target = $WRAITH$
			}
		}
		$WRAITH$ = {
			add_character_flag = {flag = wraith_bound days = 3560}
			set_variable = {name = necromancer value = $NECROMANCER$}
			remove_opinion = {
				target = $NECROMANCER$
				modifier = POD_binding_opinion
			}
		}
		
	}
	else_if = {
		limit = {
			$NECROMANCER$ = {
				has_hook_of_type = {
					target = $WRAITH$
					type = necromancy_bind_strong_hook
				}	
			}
		}
		$NECROMANCER$ = {
			remove_hook = {
				target = $WRAITH$
				type = necromancy_bind_strong_hook
			}	
			add_hook = {
				target = $WRAITH$
				type = necromancy_bind_permanent_hook
			}
			reverse_add_opinion = {
				modifier = POD_binding_permanent_opinion
				target = $WRAITH$
			}
		}
		$WRAITH$ = {
			add_character_flag = wraith_bound 
			set_variable = {name = necromancer value = $NECROMANCER$}
			remove_opinion = {
				target = $NECROMANCER$
				modifier = POD_binding_strong_opinion
			}
		}
		
	
	}
	else = {
		$NECROMANCER$ = {
			add_hook = {
				target = $WRAITH$
				type = necromancy_bind_hook
			}
			reverse_add_opinion = {
				modifier = POD_binding_opinion
				target = $WRAITH$
			}
		}
		$WRAITH$ = {
			add_character_flag = {flag = wraith_bound days = 3560}
			set_variable = {name = necromancer value = $NECROMANCER$}
		}
	
	}
	$NECROMANCER$ = {
		add_to_variable_list = {
			name = bound_wraiths
			target = $WRAITH$
		}
		set_relation_bound_wraith = $WRAITH$
	}


}

POD_wraith_weak_possession_effect = {
	$WRAITH$ = {
		add_hook = {
			target = $PUPPET$
			type = possessed_weak_hook
		}
		reverse_add_opinion = {
			modifier = POD_weak_possession_opinion
			target = $PUPPET$
		}
		add_character_flag = { 
			flag = in_possession
			days = 3650
		}
		set_variable = { name = puppet value = $PUPPET$}
		if = {
			limit = {has_character_flag = wraith_bound}
			var:necromancer = {
				send_interface_toast = {
					title = POD_wraith_possession_toast_title
					custom_tooltip = POD_wraith_possession_toast_text
					left_icon = scope:actor
					right_icon = scope:recipient
					add_hook = {
						target = $PUPPET$
						type = necromancy_bind_hook
					}
					reverse_add_opinion = {
						modifier = POD_possessed_by_bound_ghost
						target = $PUPPET$
					}

				}
				
				
			}
		}
		else_if = {
			limit = { has_variable = necromancer }
			var:necromancer = {
				send_interface_toast = {
					title = POD_wraith_possession_toast_title
					custom_tooltip = POD_wraith_possession_toast_text
					left_icon = scope:actor
					right_icon = scope:recipient					
					reverse_add_opinion = {
						modifier = POD_possessed_by_bound_ghost
						target = $PUPPET$
					}

				}
				
				
			}
		}
	}
	$PUPPET$ = {
		set_variable = {name = wraithprowess value = { value = $WRAITH$.prowess divide = 2 } }
		add_prowess_skill = { value = $WRAITH$.prowess divide = 2 }
		set_variable = {name = wraith value = $WRAITH$}
		add_character_flag = { 
			flag = possessed
			days = 3650
		}
		trigger_event = {
			id = POD_wraith_possession.20
			days = 3650
		}
	}
	

}

POD_wraith_increase_possession = {

	if = {
		limit = {
			$WRAITH$ = {
				has_hook_of_type = {
					target = $PUPPET$
					type = possessed_hook 
				}
			}
		}
		POD_wraith_complete_possession_effect = { WRAITH = $WRAITH$ PUPPET = $PUPPET$}
	}
	else_if = {
		limit = {
			$WRAITH$ = {
				has_hook_of_type  = {
					target = $PUPPET$
					type = possessed_weak_hook
				}
			}
		}
		POD_wraith_possession_effect = { WRAITH = $WRAITH$ PUPPET = $PUPPET$}
	
	}
	else = {
		POD_wraith_weak_possession_effect = { WRAITH = $WRAITH$ PUPPET = $PUPPET$}
	
	}

}

POD_wraith_end_possession = {
	if = {
		limit = {
			$PUPPET$ = {
				has_relation_possessesing_wraith = $WRAITH$
			}
		}
		$PUPPET$ = { 
			if = {
				limit = {exists = var:wraithprowess}
				add_prowess_skill = { value = var:wraithprowess multiply = -1}
				remove_variable = wraithprowess
			}
			
			if = {
				limit = {exists = var:wraithdiplomacy}
				add_diplomacy_skill = { value = var:wraithdiplomacy multiply = -0.5}
				remove_variable = wraithdiplomacy
			}
			
			if = {
				limit = {exists = var:wraithstewardship}
				add_stewardship_skill = { value = var:wraithstewardship multiply = -0.5}
				remove_variable = wraithstewardship
			}
			
			if = {
				limit = {exists = var:wraithmartial}
				add_martial_skill = { value = var:wraithmartial multiply = -0.5}
				remove_variable = wraithmartial
			}
			
			if = {
				limit = {exists = var:wraithintrigue}
				add_intrigue_skill = { value = var:wraithintrigue multiply = -0.5}
				remove_variable = wraithintrigue
			}
			
			if = {
				limit = {exists = var:wraithlearning}
				add_learning_skill = { value = var:wraithlearning multiply = -0.5}
				remove_variable = wraithlearning
			}
		}
	}
	else_if = {
		limit = {
			$WRAITH$ = {
				has_hook_of_type = {
					target = $PUPPET$
					type = possessed_hook 
				}
			}
		}
		
		$PUPPET$ = { 
			if = {
				limit = {exists = var:wraithprowess}
				add_prowess_skill = { value = var:wraithprowess multiply = -1}
				remove_variable = wraithprowess
			}
			
			if = {
				limit = {exists = var:wraithdiplomacy}
				add_diplomacy_skill = { value = var:wraithdiplomacy multiply = -0.2}
				remove_variable = wraithdiplomacy
			}
			
			if = {
				limit = {exists = var:wraithstewardship}
				add_stewardship_skill = { value = var:wraithstewardship multiply = -0.2}
				remove_variable = wraithstewardship
			}
			
			if = {
				limit = {exists = var:wraithmartial}
				add_martial_skill = { value = var:wraithmartial multiply = -0.2}
				remove_variable = wraithmartial
			}
			
			if = {
				limit = {exists = var:wraithintrigue}
				add_intrigue_skill = { value = var:wraithintrigue multiply = -0.2}
				remove_variable = wraithintrigue
			}
			
			if = {
				limit = {exists = var:wraithlearning}
				add_learning_skill = { value = var:wraithlearning multiply = -0.2}
				remove_variable = wraithlearning
			}
		}
		
	
	}
	
	else = {
		$PUPPET$ = { 
			if = {
				limit = {exists = var:wraithprowess}
				add_prowess_skill = { value = var:wraithprowess multiply = -1}
				remove_variable = wraithprowess
			}
		}
	}
	
	$PUPPET$ = {
		remove_opinion = {
			target = $WRAITH$
			modifier = POD_weak_possession_opinion
		
		}
		remove_opinion = {
			target = $WRAITH$
			modifier = POD_possession_opinion
		
		}
		remove_opinion = {
			target = $WRAITH$
			modifier = POD_complete_possession_opinion
		
		}
		remove_relation_possessesing_wraith = $WRAITH$
		remove_variable = wraith
		remove_character_flag = possessed
	}
	$WRAITH$ = {
		remove_character_flag = in_possession
		remove_variable = puppet
		remove_hook = {
			type = possessed_hook
			target = $PUPPET$
		}
		remove_hook = {
			type = possessed_weak_hook
			target = $PUPPET$
		}
		if = {
			limit = {has_character_flag = wraith_bound}
			var:necromancer = {
				remove_hook = {
					target = $PUPPET$
					type = necromancy_bind_permanent_hook
				}
				remove_hook = {
					target = $PUPPET$
					type = necromancy_bind_strong_hook
				}
				remove_hook = {
					target = $PUPPET$
					type = necromancy_bind_hook
				}
				save_scope_as = necromancer
				$PUPPET$ = {
					remove_opinion = {
						target = scope:necromancer
						modifier = POD_possessed_by_bound_ghost
					}
				}
			}
		}
	
	}

}

create_soulforged_artifact_effect = {
	create_artifact = {
		name = artifact_soulforged_artifact_name
		creator = scope:creator
		description = artifact_soulforged_artifact_description
		visuals = $VISUALS$
		type = $TYPE$
		modifier = artifact_stress_gain_1_penalty_modifier
		save_scope_as = newly_created_artifact
		wealth = scope:wealth
		quality = scope:quality
		#decaying = no
	}
	scope:newly_created_artifact = {
		if = {
			limit = {
				scope:owner = { has_royal_court = yes }
			}
			add_scaled_artifact_modifier_grandeur_small_effect = yes
		}
		add_scaled_artifact_modifier_minor_prestige_effect = yes
		if = {
			limit = {
				rarity = illustrious
			}
			add_artifact_modifier = artifact_dread_baseline_add_2_modifier
		}
		else_if = {
			limit = {
				rarity = famed
			}
			add_artifact_modifier = artifact_dread_baseline_add_1_modifier
		}
		add_soulforged_artifact_modifier_effect = yes
	}
}

get_soulforged_artifact_quality_effect = {
	hidden_effect = {
		random_list = {
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 0.5
				} 
			}
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 1
				} 
			}
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 2
				} 
			}
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 2.5
				} 
			}
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 3
				} 
			}
		}
	}
	
	# Calculate the final quality of the artifact
	save_scope_value_as = {
		name = quality
		value = {
			value = 0
			
			scope:killed = {
				# Bonus quality from high skills
				add = {	
					if = {
						limit = { scope:armor = yes }
						value = armor_inspiration_average_skill_value
					}
					else_if = {
						limit = { scope:weapon = yes }
						value = weapon_inspiration_average_skill_value
					}
					else_if = {
						limit = { scope:shinies = yes }
						value = smith_inspiration_average_skill_value
					}
					else_if = {
						limit = { scope:book = yes }
						value = book_inspiration_average_skill_value
					}
					else_if = {
						limit = { scope:furniture = yes }
						value = weaver_inspiration_average_skill_value
					}
					else_if = {
						limit = { scope:trinket = yes }
						value = artisan_inspiration_average_skill_value
					}
					multiply = {
						value = quality_bonus_per_skill_level_value
						add = scope:random_inspired_skill_quality_bonus
					}
				}

				add = {	
					if = {
						limit = {
							OR = {
								has_trait = education_martial_4
								has_trait = education_diplomacy_4
								has_trait = education_learning_4
								has_trait = education_stewardship_4
								has_trait = education_intrigue_4
								has_trait = education_martial_prowess_4
							}
						}
						value = 20
					}
					else_if = {
						limit = {
							OR = {
								has_trait = education_martial_3
								has_trait = education_diplomacy_3
								has_trait = education_learning_3
								has_trait = education_stewardship_3
								has_trait = education_intrigue_3
								has_trait = education_martial_prowess_3
							}
						}
						value = 15
					}
					else_if = {
						limit = {
							OR = {
								has_trait = education_martial_2
								has_trait = education_diplomacy_2
								has_trait = education_learning_2
								has_trait = education_stewardship_2
								has_trait = education_intrigue_2
								has_trait = education_martial_prowess_2
							}
						}
						value = 10
					}
					else = {
						value = 5
					}
				}
			}
			
			# Can't have negative quality
			min = 1
		}
	}
}
