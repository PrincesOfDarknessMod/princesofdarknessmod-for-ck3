POD_set_local_spirit_list_effect = {
	clear_variable_list = spirit_list
	every_pool_character = {
		province = capital_province
		limit = { has_trait = wraith }
		POD_move_at_local_spirit_list_effect = { PLAYER = root }
	}
	every_courtier_or_guest = {
		limit = { has_trait = wraith }
		POD_move_at_local_spirit_list_effect = { PLAYER = root }
	}
}

POD_move_at_local_spirit_list_effect = {
	if = {
		limit = { 
			trigger_if = {
				limit = { 
					has_character_modifier = traveling_spirit_world_modifier
					$PLAYER$ = { has_character_modifier = traveling_spirit_world_modifier }
				}
				has_variable = spirit_location
				$PLAYER$ = { has_variable = spirit_location }
				var:spirit_location = $PLAYER$.var:spirit_location
			}
			trigger_else = {
				NOT = { has_character_modifier = traveling_spirit_world_modifier }
				$PLAYER$ = { NOT = { has_character_modifier = traveling_spirit_world_modifier } }
			}
		}
		$PLAYER$ = {
			add_to_variable_list = {
				name = spirit_list
				target = prev
			}
		}
	}
	POD_set_random_dialog_effect = { PLAYER = $PLAYER$ }
	if = {
		limit = { NOT = { has_relation_necromancer = $PLAYER$ } }
		move_to_pool = yes
	}
}

POD_create_random_local_spirit = {
	random_pool_character = {
		province = capital_province
		limit = {
			is_adult = yes
			NOT = { has_trait = wraith }
		}
		POD_become_wraith_effect = yes
		set_variable = { name = random_dialog value = flag:generica }
		set_variable = { name = spirit_location value = root.var:spirit_location }
		if = {
			limit = { NOT = { var:spirit_location = flag:skinlands } }
			add_character_modifier = traveling_spirit_world_modifier
		}
		root = {
			add_to_variable_list = {
				name = spirit_list
				target = prev
			}
		}
	}
}

				SET_SUBJECT = flag:no

			}
		}
		else_if = {
			limit = { scope:furniture = yes }
			random_list = {
				10 = {
					create_artifact_throne_effect = {
						OWNER = scope:owner
						MAKER = scope:owner
					}
				}
				10 = {

						OWNER = scope:owner
						WEAVER = scope:owner

				}
				10 = {
					create_artifact_box_small_effect = {
						OWNER = scope:owner
						CARPENTER = scope:owner
					}
				}
				10 = {
					create_artifact_box_medium_effect = {
						OWNER = scope:owner
						CARPENTER = scope:owner
					}
				}
				10 = {
					create_artifact_goblet_effect = {
						OWNER = scope:owner
						SMITH = scope:owner
					}
				}
				10 = {
					create_artifact_animal_hide_big_effect = {
						OWNER = scope:owner
						HUNTER = scope:owner
					}
				}
				10 = {
					create_artifact_animal_hide_medium_effect = {
						OWNER = scope:owner
						HUNTER = scope:owner
					}
				}
				10 = {
					create_artifact_animal_hide_small_effect = {
						OWNER = scope:owner
						HUNTER = scope:owner
					}
				}
				10 = {
					create_artifact_wall_banner_effect = {
						OWNER = scope:owner
						CREATOR = scope:owner
						TARGET = scope:owner.primary_title
					}
				}
				10 = {
					create_artifact_wall_skull_effect = {
						OWNER = scope:owner
						HUNTER = scope:owner
					}
				}
				10 = {
					create_artifact_sculpture_effect = {
						OWNER = scope:owner
						SCULPTER = scope:owner
					}
				}
				10 = {
					create_artifact_wall_shield_effect = {
						OWNER = scope:owner
						TARGET = scope:owner.house
						CREATOR = scope:owner
					}
				}
			}
		}
		else_if = {
			limit = { scope:trinket = yes }
			create_artifact = {
				name = cuckoldry_psalter_name
				description = cuckoldry_psalter_description
				type = miscellaneous
				visuals = ring
				modifier = artifact_placeholder_modifier
			}
		}
		
		hidden_effect = {
			scope:newly_created_artifact = { clear_artifact_modifiers = yes }
			if = {
				limit = { scope:owner = scope:creator }
				every_character_artifact = {
					limit = {
						has_variable = artifact_pieces
					}
					if = {
						limit = { 
							rarity = illustrious 
							scope:owner = { has_perk = princely_restoration_perk }
						}
						scope:owner = { add_piety = major_piety_loss }
						scope:newly_created_artifact = { copy_artifact_modifiers = prev }
						destroy_artifact = this
					}
					else_if = {
						limit = { 
							rarity = famed 
							scope:owner = { has_perk = greater_restoration_perk }
						}
						scope:owner = { add_piety = medium_piety_loss }
						scope:newly_created_artifact = { copy_artifact_modifiers = prev }
						destroy_artifact = this
					}
					else_if = {
						limit = { 
							rarity = masterwork 
							scope:owner = { has_perk = major_restoration_perk }
						}
						scope:owner = { add_piety = minor_piety_loss }
						scope:newly_created_artifact = { copy_artifact_modifiers = prev }
						destroy_artifact = this
					}
					else = {
						scope:owner = { add_piety = miniscule_piety_loss }
						scope:newly_created_artifact = { copy_artifact_modifiers = prev }
						destroy_artifact = this
					}
				}
			}
			else = {
				every_equipped_character_artifact = {
					limit = {
						artifact_slot_type = miscellaneous
						has_variable = artifact_pieces
					}
					scope:newly_created_artifact = { copy_artifact_modifiers = prev }
					if = {
						limit = { rarity = common }
						scope:owner = { add_piety = miniscule_piety_loss }
					}
					else_if = {
						limit = { rarity = masterwork }
						scope:owner = { add_piety = minor_piety_loss }
					}
					else_if = {
						limit = { rarity = famed }
						scope:owner = { add_piety = medium_piety_loss }
					}
					else = {
						scope:owner = { add_piety = major_piety_loss }
					}
					destroy_artifact = this
				}
			}
		}
	}
}
