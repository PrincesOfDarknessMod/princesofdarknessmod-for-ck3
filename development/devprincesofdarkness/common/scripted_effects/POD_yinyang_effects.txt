POD_set_yinyang_effect = {
	POD_clear_yinyang_effect = yes
	POD_determine_yinyang_effect = yes	
}

POD_clear_yinyang_effect = {

	if = {
		limit = {
			has_trait = yin_imbalanced
		}
		remove_trait = yin_imbalanced
	}	
	
	if = {
		limit = {
			has_trait = yang_imbalanced
		}
		remove_trait = yang_imbalanced
	}
	
	if = {
		limit = {
			has_trait = yinyang_balanced
		}
		remove_trait = yinyang_balanced
	}
}

POD_determine_yinyang_effect = {

		random_list = {
			40 = { 
				modifier = {
					add = 50
					has_faith = faith:songoftheshadow
				}
				add_trait = yin_imbalanced
			}
			40 = { 
				modifier = {
					add = 50
					has_faith = faith:thrashingdragon
				}
				add_trait = yang_imbalanced
			}
			20 = { 
				modifier = {
					add = 70
					has_faith = faith:thousandwhispers
				}
				add_trait = yinyang_balanced
			}	
		}
}
