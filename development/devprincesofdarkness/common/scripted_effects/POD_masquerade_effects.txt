POD_piety_loss_from_massembrace = {

	if = { 
		limit = {
			OR = {
				faith = { has_doctrine_parameter = vampirism_illegal }
				faith = { has_doctrine = tenet_conscience_and_selfcontrol }
				faith = { has_doctrine = tenet_conscience_and_instinct }
				faith = { has_doctrine_parameter = vampirethemasquerade_consent }
				faith = { has_doctrine_parameter = vampirethemasquerade_strict }
			}
		}
		add_piety = massive_piety_loss
	}

	else_if = { 
		limit = {
			OR = {
				faith = { has_doctrine_parameter = vampirethemasquerade_family }
				faith = { has_doctrine_parameter = vampirism_shunned }
			}
		}
		add_piety = major_piety_loss
	}
	
	else_if = { 
		limit = {
			OR = {
				faith = { has_doctrine_parameter = vampirethemasquerade_lax }
				faith = { has_doctrine_parameter = vampirethemasquerade_cult }
			}
		}
		add_piety = medium_piety_loss
	}
}

	if = { 
		limit = {

			}

		add_piety = minor_piety_loss
	}
