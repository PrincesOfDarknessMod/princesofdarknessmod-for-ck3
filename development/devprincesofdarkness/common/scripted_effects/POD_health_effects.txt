no_health_left_effect = {
	if = {
		limit = {
			has_trait = vampire
		}
		add_trait = torpor
		if = {
			limit = { is_landed = yes }
			depose = yes
		}
	}
	else = {
		death = {
			death_reason = death_lost_all_life_force
		}
	}
}
