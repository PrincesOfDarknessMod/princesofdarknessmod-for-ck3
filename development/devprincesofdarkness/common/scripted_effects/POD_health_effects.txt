POD_losing_health_level_effect = {
	if = {
		limit = {
			has_character_modifier = health_major_loss_modifier
		}
		hidden_effect = {
			remove_character_modifier = health_major_loss_modifier
		}
		add_trait = torpor
	}
	else_if = {
		limit = {
			has_character_modifier = health_moderate_loss_modifier
		}
		hidden_effect = {
			remove_character_modifier = health_moderate_loss_modifier
		}
		add_character_modifier = {
			modifier = health_major_loss_modifier
			years = 1
		}
	}
	else_if = {
		limit = {
			has_character_modifier = health_minor_loss_modifier
		}
		hidden_effect = {
			remove_character_modifier = health_minor_loss_modifier
		}
		add_character_modifier = {
			modifier = health_moderate_loss_modifier
			years = 1
		}
	}
	else = {
		add_character_modifier = {
			modifier = health_minor_loss_modifier
			years = 1
		}
	}
}
