find_title_for_giangaleazzo_spawn_effect = {
	if = { # The important thing is to not mess too much with players who can't resist
		limit = {
			title:c_lombardia = {
				holder = { is_ai = yes }
				OR = {
					NOT = { exists = duchy.holder }
					duchy.holder = { is_ai = yes }
				}
			}
		}
		title:c_lombardia = {
			save_scope_as = giangaleazzo_county
		}
	}
	else = {
		title:k_italy = {
			random_in_de_jure_hierarchy = {
				continue = { tier > tier_county }
				limit = {
					tier = tier_county
					holder = { is_ai = yes } # Let's not murder players
					save_temporary_scope_as = potential_giangaleazzo_county
					# If someone holds k_italy, put Gian in their realm
					trigger_if = {
						limit = { exists = title:k_italy.holder }
						title:k_italy.holder = {
							OR = {
								this = scope:potential_giangaleazzo_county.holder
								any_sub_realm_county = {
									this = scope:potential_giangaleazzo_county
								}
							}
						}
					}
					# Either this should be the holder's only title, or it should NOT be their capital
					trigger_if = {
						limit = {
							holder = {
								any_sub_realm_county = { count > 1 }
							}
						}
						NOT = {
							this = holder.capital_county
						}
					}
				}
				weight = {
					base = 1
					# Massively prefer to avoid a player-held Duchies if possible
					modifier = {
						factor = 0.01
						exists = duchy.holder
						duchy.holder = { is_ai = no }
					}
				}
				save_scope_as = giangaleazzo_county
			}
		}
	}

	if = { # Discard stuff from here on out
		limit = { NOT = { exists = scope:giangaleazzo_county } }
		debug_log = "No Giangaleazzo title found, Giangaleazzo won't spawn"
	}
	# Convert the County and warn relevant people
	else = {
		scope:giangaleazzo_county = {
			set_county_faith = faith:viaregalis
		}
		scope:giangaleazzo_county.holder = {
			if = {
				limit = {
					is_ai = no
				}
				trigger_event = PODadventurers.1001
			}
			every_liege_or_above = {
				limit = { is_ai = no }
				trigger_event = PODadventurers.1001
			}
		}
		trigger_event = { # Random amount of time to prepare
			id = PODadventurers.0002
			days = { 5 10 }
		}
	}
}

create_giangaleazzo_effect = {
	create_character = {
			name = "Giangaleazzo" 
			#dna = 790456_giangaleazzo
			age = 32
			gender = male
			dynasty = none
			intrigue = 7			
			faith = faith:viaregalis
			culture = culture:italian
			trait = education_intrigue_4
			trait = cynical
			trait = deceitful
			trait = ambitious
			trait = vampire
			trait = schemer
			trait = lasombra
			trait = presence
			trait = oblivion
			trait = potence
			random_traits = no		
			location = title:c_lombardia.title_province
			save_scope_as = giangaleazzo
			after_creation = {
			give_nickname = nick_model_prince
			add_character_flag = {
				flag = immune_to_disease
			}

			add_secret = { 
				type = secret_vampgeneration7
			}			
			}
	}
	scope:giangaleazzo = {
		create_story = {
			type = story_giangaleazzo_invasion
			save_scope_as = story
		}
		add_pressed_claim = title:k_italy
		add_pressed_claim = title:d_lombardia
		add_pressed_claim = title:d_friuli
		add_pressed_claim = title:d_verona	
		find_giangaleazzo_titles_and_assign_to_giangaleazzo_effect = yes
		giangaleazzo_become_independent_effect = yes
		add_gold = monumental_gold_max_value
		add_prestige = 1000	
		#POD_clear_resonance_effect = yes
		POD_set_resonance_effect = yes
	}
	scope:story = {
		set_variable = {
			name = giangaleazzo
			value = scope:giangaleazzo
		}
		# set_variable = {
			# name = giangaleazzo_county
			# value = scope:giangaleazzo_county
		# }
	}
		random_player = {
				limit = {
				any_realm_province = {
					geographical_region = ghw_region_italy
				}
				is_vampire_trigger = yes
				highest_held_title_tier > tier_county		
				}	
			trigger_event = {
				id = PODadventurers.1002
				days = 5
			}
		}
	}


find_giangaleazzo_titles_and_assign_to_giangaleazzo_effect = {

	create_title_and_vassal_change = change
	scope:change = { set_title_and_vassal_change_type = usurped }
	title:c_lombardia = {
		change_title_holder = {
			holder = scope:giangaleazzo
			change = scope:change
			take_baronies = yes
		}
	}
	resolve_title_and_vassal_change = scope:change
	
	spawn_first_giangaleazzo_troops_effect = yes

}

giangaleazzo_become_independent_effect = {
	hidden_effect = {
		if = {
			limit = { scope:giangaleazzo = { is_imprisoned = yes } }
			scope:giangaleazzo = { release_from_prison = yes }
		}
	}
	create_title_and_vassal_change = going_independent
	scope:going_independent = { set_title_and_vassal_change_type = independency }
	scope:giangaleazzo = {
		becomes_independent = {
			change = scope:going_independent
		}
	}

	resolve_title_and_vassal_change = scope:going_independent
}

spawn_first_giangaleazzo_troops_effect = {
	spawn_army = {
		uses_supply = yes
		inheritable = no
		name = first_giangaleazzo_event_troops
		levies = {
			value = 20000
		}
		men_at_arms = {
			type = picchieri
			stacks = 5
		}		
		location = scope:giangaleazzo.capital_province
	}
}
