POD_diablerie_unconquered_soul_effect = {

	if = {
		limit = {
			OR = {
				scope:victim = { is_vampgeneration1_trigger = yes }
				scope:victim = { is_vampgeneration2_trigger = yes }
				scope:victim = { is_vampgeneration3_trigger = yes }
				scope:victim = { is_vampgeneration4_trigger = yes }
				scope:victim = { is_vampgeneration5_trigger = yes }
				scope:victim = { is_vampgeneration6_trigger = yes }
				scope:victim = { is_vampgeneration7_trigger = yes }
				scope:victim = { is_vampgeneration8_trigger = yes }
				scope:victim = { is_vampgeneration9_trigger = yes }
			}		
		}

		random = {
			chance = 1
			modifier = {
				add = 4
				scope:victim = { 
					is_vampgeneration6_trigger = yes
				}
			}			
			modifier = {
				add = 9
					scope:victim = { 
						is_vampgeneration5_trigger = yes
				}
			}		
			modifier = {
				add = 24
					scope:victim = { 
						is_vampgeneration4_trigger = yes
				}
			}				
			modifier = {
				add = 49
					scope:victim = { 
						is_vampgeneration3_trigger = yes
				}
			}				
			modifier = {
				add = 74
					scope:victim = { 
						is_vampgeneration2_trigger = yes
				}
			}
			modifier = {
				add = 95
					scope:victim = { 
						is_vampgeneration1_trigger = yes
					}
			}
					
		hidden_effect = {
				scope:victim = {
					save_scope_as = unconquered_soul
				}
				add_character_flag = POD_in_unconquered_soul_event
				trigger_event = { 
					id = POD_diablerize.800 days = { 5 3650 }
				}
			}	
		}
	}
}
