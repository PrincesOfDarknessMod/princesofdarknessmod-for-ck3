
POD_give_massregnant_secret_effect = {
	save_temporary_scope_as = massregnant_character
	if = {
		limit = { #Not already a massregnant
			NOR = {
				has_trait = massregnant
				any_secret = { secret_type = secret_massregnant }
			}	
		}
		if = {
			limit = {
				NOT = { has_trait = massregnant }
			}
			add_secret = {
				type = secret_massregnant
			}
		}
	}	
}
