pod_death_effect = {
	$KILLER$ = { 
		pod_killed_person_count_increase_effect = { VICTIM = prev }
	}
	if = {
		limit = { 
			has_POD_resurrection_trigger = yes
		}
		pod_resurrection_effect = { KILLER = $KILLER$ }
	}
	else = {
		show_as_tooltip = {
			death = {
				death_reason = $DEATH_REASON$
				killer = $KILLER$
			}
		}
		hidden_effect = {
			if = {
				limit = { 
					can_convert_to_wraith_trigger = yes
					any_player = { has_character_modifier = soul_beacon_modifier }
				}
				pod_soul_beacon_effect = yes
			}
			random_list = {
				1 = {
					trigger = {
						can_convert_to_wraith_trigger = yes
					}
					pod_return_from_death_effect = yes
					modifier = {
						factor = 100
						has_trait = dhampyr
					}
				}
				1000 = {
					trigger = {
						#NOT = { has_perk = inurement_perk }
						NOT = { exists = var:marked_for_return }
					}
					death = {
						death_reason = $DEATH_REASON$
						killer = $KILLER$
					}
				}
			}
		}
	}
}

pod_killed_person_count_increase_effect = {
	if = {
		limit = {
			is_ai = no
			exists = var:pod_killed_person_count
		}
		change_variable = {
			name = pod_killed_person_count
			add = 1
		}
	}
	if = {
		limit = {
			is_ai = no
			faith = faith:faces_of_apophis
			has_relation_soulmate = $VICTIM$
		}
		add_character_flag = killed_a_soulmate
	}
	if = {
		limit = {
			is_ai = no
			faith = faith:thousandwhispers
			is_parent_of = $VICTIM$
			$VICTIM$ = { 
				is_adult = yes
				has_trait = dhampyr
			}
			exists = var:pod_killed_dhampyrchildren_count
		}
		change_variable = {
			name = pod_killed_dhampyrchildren_count
			add = 1
		}
	}
}

pod_return_from_death_effect = {
	if  = {
		limit = { 
			is_ruler = yes 
			is_ai = yes
		}
		depose = yes
	}
	if = {
		limit = { is_imprisoned = yes }
		release_from_prison = yes
	}
	if = {
		limit = { exists = var:marked_for_return }
		var:marked_for_return = {
			save_scope_as = necromancer
		}
	}
	POD_become_wraith_effect = yes
	if = {
		limit = { exists = scope:necromancer }
		if = {
			limit = { NOT = { location = scope:necromancer.location } } 
			set_employer = scope:necromancer
			return_to_court = yes
			add_character_flag = {
				flag = blocked_from_leaving
				years = 5
			}
		}
		scope:necromancer = {
			send_interface_toast = {
				title = marked_wraith_has_arrive
				left_icon = prev
				right_icon = scope:necromancer
				if = {
					limit = { has_trait = mummy }
					add_hekau_necromancy_lifestyle_xp = medium_lifestyle_xp
				}
			}
			add_to_variable_list = {
				name = spirit_list
				target = prev
			}
		}
		remove_variable = marked_for_return
	}
}

pod_resurrection_effect = {
	save_scope_as = resurrected_victim 
	add_opinion = {
		target = $KILLER$
		modifier = POD_killed_me_opinion
	}
	if = {
		limit = { is_ai = no }
		set_variable = {
			name = my_killer
			value = $KILLER$
		}
	}
	if = {
		limit = { is_imprisoned = yes }
		release_from_prison = yes
	}
	hidden_effect = {
		$KILLER$ = { 
			send_interface_toast = {
				title = victim_has_resurrected_toast
				left_icon = $KILLER$
				right_icon = scope:resurrected_victim

				show_as_tooltip = { scope:resurrected_victim = { pod_resurrection_cost_effect = yes } }
			}
		}
		scope:resurrected_victim = { 
			send_interface_toast = {
				title = you_have_resurrected_toast
				left_icon = this
				right_icon = $KILLER$

				show_as_tooltip = { pod_resurrection_cost_effect = yes }
			}
		}
		# Send a message to all other players.
		every_player = {
			limit = {
				NOR = {
					this = scope:resurrected_victim
					this = $KILLER$
				}
				has_any_relation_trigger = { CHARACTER = scope:resurrected_victim }
			}
			send_interface_toast = {
				title = victim_has_resurrected_toast
				left_icon = scope:resurrected_victim
				right_icon = $KILLER$

				show_as_tooltip = { scope:resurrected_victim = { pod_resurrection_cost_effect = yes } }
			}
		}
		pod_resurrection_travel_effect = yes
	}
	pod_resurrection_cost_effect = yes
}

pod_resurrection_from_accident_effect = {
	save_scope_as = resurrected_victim
	hidden_effect = {
		send_interface_toast = {
			title = you_have_resurrected_toast
			left_icon = this

			show_as_tooltip = { pod_resurrection_cost_effect = yes }
		}
		# Send a message to all other players.
		every_player = {
			limit = {
				NOT = { this = scope:resurrected_victim }
				has_any_relation_trigger = { CHARACTER = scope:resurrected_victim }
			}
			send_interface_toast = {
				title = victim_has_resurrected_toast
				left_icon = scope:resurrected_victim

				show_as_tooltip = { scope:resurrected_victim = { pod_resurrection_cost_effect = yes } }
			}
		}
		pod_resurrection_travel_effect = yes
	}
	pod_resurrection_cost_effect = yes
}

pod_resurrection_travel_effect = {
	if = {
		limit = {
			has_trait = mummy
			is_ai = no
		}
		trigger_event = podmummy.300
	}
}

pod_resurrection_cost_effect = {
	if = {
		limit = { has_character_modifier = extra_life_4_modifier }
		remove_character_modifier = extra_life_4_modifier
		add_character_modifier = extra_life_3_modifier
	}
	else_if = {
		limit = { has_character_modifier = extra_life_3_modifier }
		remove_character_modifier = extra_life_3_modifier
		add_character_modifier = extra_life_2_modifier
	}
	else_if = {
		limit = { has_character_modifier = extra_life_2_modifier }
		remove_character_modifier = extra_life_2_modifier
		add_character_modifier = extra_life_1_modifier
	}
	else_if = {
		limit = { has_character_modifier = extra_life_1_modifier }
		remove_character_modifier = extra_life_1_modifier
	}
	else_if = {
		limit = { has_trait = wraith }
		POD_become_spectre_effect = yes
	}
	else_if = {
		limit = { has_trait = truefallen }
		trigger_event = POD_demon_maintenance.20 
	}
	else_if = {
		limit = { has_perk = ashes_of_the_phoenix_perk }
		remove_perk = ashes_of_the_phoenix_perk
	}
	else_if = {
		limit = { has_perk = sleep_of_the_hero_perk }
		remove_perk = sleep_of_the_hero_perk
		add_trait = torpor
		depose = yes
	}
	else_if = {
		limit = { has_perk = rebirthing_perk }
		remove_perk = rebirthing_perk
		remove_trait = werewolf
		POD_remove_werewolf_traits_effect = yes
		refund_all_perks = yes
		add_trait = kinfolk
		add_trait = reincarnation
		set_age = 1
		trigger_event = { 			
			id = POD_werewolves_maintenance.2
			years = { 13 18 }
		}
	}
	else_if = {
		limit = { 
			any_player = {
				has_perk = healing_perk
				is_target_in_variable_list = { name = auto_resurrect_list target = prev }
			}
		}
		random_player = {
			limit = { is_target_in_variable_list = { name = auto_resurrect_list target = prev } }
			add_piety =  massive_piety_loss	
		}
		hidden_effect = { 
			increase_wounds_no_death_effect = { REASON = treatment }
			increase_wounds_no_death_effect = { REASON = treatment }
			increase_wounds_no_death_effect = { REASON = treatment }
		}
	}
	
	if = {
		limit = { has_trait = asekh_sen }
		POD_become_hideous_effect = yes
	}
}

pod_soul_beacon_effect = {
	save_scope_as = interest
	random_player = {
		limit = {
			NOT = { this = scope:interest }
			OR = {
				AND = {
					has_character_flag = flag_soul_beacon_diplomacy
					scope:interest = { diplomacy > 25 }
				}
				AND = {
					has_character_flag = flag_soul_beacon_martial
					scope:interest = { martial > 25 }
				}
				AND = {
					has_character_flag = flag_soul_beacon_stewardship
					scope:interest = { stewardship > 25 }
				}
				AND = {
					has_character_flag = flag_soul_beacon_intrigue
					scope:interest = { intrigue > 25 }
				}
				AND = {
					has_character_flag = flag_soul_beacon_learning
					scope:interest = { learning > 25 }
				}
				AND = {
					has_character_flag = flag_soul_beacon_artifact
					scope:interest = { 
						any_character_artifact = {
							OR = {
								rarity = illustrious
								rarity = famed
							}
						}
					}
				}
				AND = {
					has_character_flag = flag_soul_beacon_gold
					scope:interest = { gold > 1500 }
				}
			}
		}
		set_variable = {
			name = marked_for_return
			value = scope:actor
			days = 10
		}
	}
}
