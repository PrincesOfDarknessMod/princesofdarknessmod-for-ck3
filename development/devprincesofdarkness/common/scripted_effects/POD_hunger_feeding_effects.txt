POD_hunger_feeding_sip_effect = {

		# Resolve Blood Hunger Slaking from Sip
		if = {
			limit = {
				OR = {
					has_character_modifier = bloodpotency0_modifier
					has_character_modifier = bloodpotency1_modifier
					has_character_modifier = bloodpotency2_modifier
					has_character_modifier = bloodpotency3_modifier
				}
				OR = {
					has_character_modifier = bloodhunger2_modifier
					has_character_modifier = bloodhunger3_modifier
					has_character_modifier = bloodhunger4_modifier
					has_character_modifier = bloodhunger5_modifier
				}
			}
			if = {
				limit = {
					has_character_modifier = bloodhunger2_modifier
				}
					hidden_effect = {
						remove_character_modifier = bloodhunger2_modifier
					}
					add_character_modifier = {
						modifier = bloodhunger1_modifier					
					}
					#set_variable = { name = blood_level value = 4 }
			}
			if = {
				limit = {
					has_character_modifier = bloodhunger3_modifier
				}
					remove_character_modifier = bloodhunger3_modifier
					add_character_modifier = {
						modifier = bloodhunger2_modifier
					}
					#set_variable = { name = blood_level value = 3 }
			}	
			if = {
				limit = {
					has_character_modifier = bloodhunger4_modifier
				}
					remove_character_modifier = bloodhunger4_modifier
					add_character_modifier = {
						modifier = bloodhunger3_modifier						
					}
					#set_variable = { name = blood_level value = 2 }
			}	
			if = {
				limit = {
					has_character_modifier = bloodhunger5_modifier
				}
					remove_character_modifier = bloodhunger5_modifier
					add_character_modifier = {
						modifier = bloodhunger4_modifier					
					}
					#set_variable = { name = blood_level value = 1 }
			}					
		}	

}
