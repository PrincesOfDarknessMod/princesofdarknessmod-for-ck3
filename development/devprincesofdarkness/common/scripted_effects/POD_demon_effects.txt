POD_demon_initialization_effect = {
	if = {
		limit = { has_demon_trait_trigger = no }
		add_trait = truefallen
	}
	if = {
		limit = { NOT = { religion = { is_in_family = rf_demons } } }
		POD_character_demon_faith_initiation_effect = yes
	}
	if = {
		limit = { arcana_lifestyle_perks = 0 }
		POD_character_demon_lifestyle_initialization_effect = yes
	}
}

POD_character_demon_faith_initiation_effect = {
	religion:demons_religion = {
		random_faith = {
			save_scope_as = this_faith
		}
	}
	set_character_faith = scope:this_faith
}
