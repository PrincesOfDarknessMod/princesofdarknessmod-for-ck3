POD_mummy_initialization_effect = {
	if = {
		limit = {
			has_trait = mummy
			NOT = { faith.religion = { is_in_family = rf_mummies } }
		}
		POD_character_mummy_faith_initiation_effect = yes
	}
	if = {
		limit = {
			has_mummy_trait_trigger = no
		}
		if = {
			limit = { faith = { has_doctrine  = special_doctrine_is_cabiri_faith } }
			add_trait = cabiri
		}
		else_if = {
			limit = { faith = { has_doctrine  = special_doctrine_allied_to_apophis_hostility } }
			add_trait = banemummy
		}
		else_if = {
			limit = { faith = { has_doctrine  = special_doctrine_against_to_yomi_wan_hostility } }
			add_trait = wutian
		}
		else_if = {
			limit = { faith = { has_doctrine  = special_doctrine_allied_to_yomi_wan_hostility } }
			add_trait = wukuei
		}
		else = {
			add_trait = shemsuheru
		}
	}
	if = {
		limit = {
			is_east_mummy_trigger = no
			has_sekhem_trigger = no
		}
		POD_set_sekhem_effect = yes
	}
	if = {
		limit = {
			is_east_mummy_trigger = yes
			has_chi_trigger = no
		}
		add_character_modifier = yinvirtue0_modifier
		add_character_modifier = yangvirtue10_modifier
		POD_set_chi_effect = yes
	}
	if = {
		limit = {
			has_alchemy_craft_slot_trigger = no
		}
		POD_set_alchemy_craft_slot_effect = yes
	}
	if = {
		limit = {
			is_ai = no
			NOT = { exists = var:pod_killed_person_count }
		}
		set_variable = {
			name = pod_killed_person_count
			value = 0
		}		
	}
}

POD_character_mummy_faith_initiation_effect = {
	if = {
		limit = {
			is_east_mummy_trigger = yes
			NOT = { faith = faith:ten_mandates_of_heven }
		}
		set_character_faith = faith:ten_mandates_of_heven
	}
	else_if = {
		limit = {
			has_trait = cabiri
			NOT = { faith = faith:cabiri }
		}
		set_character_faith = faith:cabiri
	}
	else_if = {
		limit = {
			has_trait = banemummy
			NOT = { faith = faith:faces_of_apophis }
		}
		set_character_faith = faith:faces_of_apophis
	}
	else = {
		random_list = {
			1 = { set_character_faith = faith:code_of_horus }
			1 = { set_character_faith = faith:ishmaelite }
		}
	}
}

		creator = scope:owner
		description = placeholder #properly set later
		visuals = necklace
		type = miscellaneous
		modifier = artifact_placeholder_modifier
		wealth = scope:wealth
		quality = scope:quality
		save_scope_as = newly_created_artifact
	}

	hidden_effect_new_artifact = {
		scope:newly_created_artifact = {
			#Loc
			remove_artifact_modifier = artifact_placeholder_modifier

			set_variable = {
				name = quality
				value = scope:quality
			}
			set_variable = {
				name = wealth
				value = scope:wealth
			}
		}
	}
}

pod_mummy_amulets_craft_quality_effect = {

	# Calculate the final quality of the artifact
	save_scope_value_as = {
		name = quality
		value = {
			value = 0

			scope:owner = {
				add = {	
					value = learning
					multiply = {
						value = var:amulets_craft_outcome
					}
				}
			}

			# Can't have negative quality
			min = 1
		}
	}
}

pod_mummy_amulets_craft_wealth_effect = {
	save_scope_value_as = {
		name = wealth
		value = {
			if = {
				# If our artifact already has a quality, set base wealth equal to 1/4th quality value.
				limit = { exists = scope:quality }
				add = scope:quality
				multiply = 0.25
				round = yes

			}
			else = {
				add = 10 # Otherwise, set base wealth value to 10.
			}

			if = {
				limit = {
					exists = scope:owner
					scope:owner.highest_held_title_tier >= 0
				}
				add = {
					# Add 10 wealth per character tier (up to a max of 50 wealth for emperors)
					value = scope:owner.highest_held_title_tier
					multiply = 10
				}

				# Add up to 25 wealth, scaling with capital's development
				if = {
					limit = {
						exists = scope:owner.capital_county
					}
					scope:owner.capital_county = {
						add = {
							value = development_level
							multiply = 0.25
						}
					}
				}
			}
		}
	}
}
