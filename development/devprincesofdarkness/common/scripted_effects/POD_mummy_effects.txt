POD_mummy_initialization_effect = {
	if = {
		limit = {
			NOT = { has_trait = wutian }
			has_sekhem_trigger = no
		}
		POD_set_sekhem_effect = yes
	}
	if = {
		limit = {
			is_ai = no
			has_alchemy_craft_slot_trigger = no
		}
		POD_set_alchemy_craft_slot_effect = yes
	}
}

POD_set_sekhem_effect = {
	add_character_modifier = sekhem10_modifier
}

POD_reset_sekhem_effect = {
	POD_clean_sekhem_effect = yes
	add_character_modifier = sekhem10_modifier
}

POD_mummy_random_perk_initialization_for_noble_effect = {
	random_list = {
		1 = {
			trigger = { perks_in_tree = { tree = alchemy value = 0 } }
			assign_alchemy_effect = yes
		}
		1 = {
			trigger = { perks_in_tree = { tree = amulets value = 0 } }
			assign_amulets_effect = yes
		}
		1 = {
			trigger = { perks_in_tree = { tree = celestial value = 0 } }
			assign_celestial_effect = yes
		}
		1 = {
			trigger = { perks_in_tree = { tree = effigy value = 0 } }
			assign_effigy_effect = yes
		}
		1 = {
			trigger = { perks_in_tree = { tree = necromancy value = 0 } }
			assign_necromancy_effect = yes
		}
		1 = {
			trigger = {
				NOT = { has_trait = wutian } 
				perks_in_tree = { tree = nomenclature value = 0 } 
			}
			assign_nomenclature_effect = yes
		}
		1 = {
			trigger = { 
				has_trait = wutian
				perks_in_tree = { tree = feng_tan value = 0 } 
			}
			assign_feng_tan_effect = yes
		}
		1 = {
			trigger = { 
				has_trait = wutian
				perks_in_tree = { tree = lung_tan value = 0 } 
			}
			assign_lung_tan_effect = yes
		}
	}
}
