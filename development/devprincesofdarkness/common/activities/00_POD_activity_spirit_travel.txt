activity_umbra_travel = {
	solo = yes
	color = { 0.253906 0.867188 0.105469 1 }
	on_spawn = {
		invite_character_to_activity = activity_owner
		accept_invitation_for_character = activity_owner
		activity_owner = {
			if = {
				limit = {
					OR = {
						is_kueijin_trigger = yes
						POD_lesser_chi_user_trigger = yes
					}
				}
				add_character_modifier = traveling_yang_world_modifier
			}
			else = {
				add_character_modifier = traveling_umbra_modifier
			}
			trigger_event = {
				on_action = umbra_travel_start_events
			}
		}
	}
	on_activate = {
		activity_owner = {
			trigger_event = {
				on_action = umbra_travel_random_pulse
				days = { 10 20 }
			}
		}
	}
	on_complete = {
		activity_owner = {
			remove_character_modifier = traveling_yang_world_modifier
			remove_character_modifier = traveling_umbra_modifier
		
			remove_character_flag = in_penumbra
			remove_character_flag = in_astral_reaches
			remove_character_flag = in_spirit_wilds
			remove_character_flag = in_near_dreaming
		}
	}
}
