activity_spirit_travel = {
	solo = yes
	color = { 0.253906 0.867188 0.105469 1 }
	on_spawn = {
		invite_character_to_activity = activity_owner
		accept_invitation_for_character = activity_owner
		activity_owner = {
			add_character_modifier = traveling_spirit_world_modifier
			if = {
				limit = { has_character_flag = in_shadowlands }
				trigger_event = POD_underworld_travel.100
			}
			else = {
				trigger_event = POD_umbra_travel.100
			}
		}
	}
	on_activate = {
		activity_owner = {
			trigger_event = {
				on_action = spirit_travel_random_pulse
				days = { 10 20 }
			}
		}
	}
	on_complete = {
		activity_owner = {
			remove_character_modifier = traveling_spirit_world_modifier
			remove_character_flag = trying_to_end_spirit_travel
		
			remove_character_flag = in_penumbra
			remove_character_flag = in_astral_reaches
			remove_character_flag = in_spirit_wilds
			remove_character_flag = in_near_dreaming
			remove_character_flag = in_shadowlands
			remove_character_flag = in_tempest
		}
	}
}
