activity_spirit_travel = {
	solo = yes
	color = { 0.253906 0.867188 0.105469 1 }
	on_spawn = {
		invite_character_to_activity = activity_owner
		accept_invitation_for_character = activity_owner
		activity_owner = {
			add_character_modifier = traveling_spirit_world_modifier
			if = {
				limit = { var:spirit_location = flag:shadow_realm }
				trigger_event = podmummy.300
			}
			else_if = {
				limit = { var:spirit_location = flag:shadowlands }
				trigger_event = POD_underworld_travel.100
			}
			else = {
				trigger_event = POD_umbra_travel.100
			}
		}
	}
	on_activate = {
		activity_owner = {
			if = {
				limit = { NOT = { var:spirit_location = flag:shadow_realm } }
				trigger_event = {
					on_action = spirit_travel_random_pulse
					days = { 20 30 }
				}
			}
		}
	}
	on_complete = {
		activity_owner = {
			remove_character_modifier = traveling_spirit_world_modifier
		
			set_variable = { name = spirit_location value = flag:skinlands }
		}
	}
}
