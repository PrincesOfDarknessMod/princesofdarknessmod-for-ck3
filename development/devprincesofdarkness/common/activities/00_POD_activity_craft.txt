activity_alchemy_craft = {
	expiration_days = 300
	has_activity_window = yes
	days_until_auto_start = 20
	color = { 0.253906 0.867188 0.105469 1 }
	valid = {
		activity_owner = {
			is_alive = yes
			is_imprisoned = no
		}
	}
	on_spawn = {
		invite_character_to_activity = activity_owner
		accept_invitation_for_character = activity_owner
		activity_owner = {
			set_variable = {
				name = alchemy_craft_outcome
				value = 0
				days = 400
			}
			trigger_event = {
				on_action = alchemy_craft_start_events
			}
		}
	}
	on_activate = {
		activity_owner = {
			trigger_event = {
				on_action = alchemy_craft_identification_events
				days = { 10 20 }
			}
		}
	}
	on_complete = {
		activity_owner = {
			remove_variable = alchemy_craft_outcome
		}
	}
}
