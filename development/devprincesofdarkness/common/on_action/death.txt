# character just about to die in root scope
# if a killer is know, it's set as scope:killer
#Triggered by code

on_death = {
	effect = {
		play_music_cue = "mx_cue_death"

		add_character_flag = { flag = about_to_die_but_not_dead_yet_flag } # Used in perk calc
		
		if = {
			limit = { 
				any_in_list = {
					variable = contractor
					exists = yes
				}
			}
			every_in_list = {
				variable = contractor 
				save_scope_as = contractor
			}
			remove_list_global_variable  = {
				name = global_web_of_knives_targets
				target = this
			}
			every_player = {
				remove_list_variable = {
					name = web_of_knives_targets
					target = prev
				}
			}
			if = {
				limit = { 
					exists = scope:killer
					NOT = { scope:killer = scope:contractor }
					scope:killer = {
						web_of_knives_member_trigger = yes
					}
				}
				
				if = {
					limit = { 
						has_variable = web_gold_reward
						OR = {
							scope:killer = {
								has_trait = assamite
							}
							NOR = {
								dynasty:10000003 = {
									any_dynasty_member = {
										exists = yes
									}
								}
								title:k_webofknives = {
									holder = {
										exists = yes
									}
								}
							}
						}
					}
					scope:killer = {
						send_interface_toast = {
							right_icon = scope:contractor
							left_icon = root
							title = kill_reward
							scope:contractor = { 
								pay_short_term_gold = { target = scope:killer gold = root.var:web_gold_reward}
							} 
						}
					}
					scope:contractor = {
						send_interface_toast = {
							title = target_killed_pay
							right_icon = scope:killer
							left_icon = root
							show_as_tooltip = {
								pay_short_term_gold = { target = scope:killer gold = root.var:web_gold_reward}
							} 
						}
					
					}
				}
				
				
				else_if = {
					limit = { 
						has_variable = web_gold_reward
						OR = {
							scope:killer = {
								web_of_knives_member_trigger = yes
							}
							 
							title:k_webofknives = {
								holder = {
									exists = yes
								}
							}
							
						}
					}
					scope:killer = {
						send_interface_toast = {
							right_icon = scope:contractor
							left_icon = root
							title = kill_reward
							scope:contractor = { 
								pay_short_term_gold = { target = scope:killer gold =  { value = root.var:web_gold_reward multiply = 0.75 } }
								pay_short_term_gold = { target = title:k_webofknives.holder gold = { value = root.var:web_gold_reward multiply = 0.25 } }
							} 
						}
					}
					scope:contractor = {
						send_interface_toast = {
							title = target_killed_pay
							right_icon = scope:killer
							left_icon = root
							show_as_tooltip = {
								pay_short_term_gold = { target = scope:killer gold = { value = root.var:web_gold_reward multiply = 0.75 } }
								pay_short_term_gold = { target = title:k_webofknives.holder gold = { value = root.var:web_gold_reward multiply = 0.25 } }
							} 
						}
					
					}
				}
				else = {
					scope:killer = {
						send_interface_toast = {
							right_icon = scope:contractor
							left_icon = root
							title = kill_reward
							scope:contractor = { 
								pay_short_term_gold = { target = scope:killer gold = { value = root.var:web_gold_reward multiply = 0.75 } }
								pay_short_term_gold = { target = dynasty:10000003.dynast gold = { value = root.var:web_gold_reward multiply = 0.25 } }
							} 
						}
					}
					scope:contractor = {
						send_interface_toast = {
							title = target_killed_pay
							right_icon = scope:killer
							left_icon = root
							show_as_tooltip = {
								pay_short_term_gold = { target = scope:killer gold = { value = root.var:web_gold_reward multiply = 0.75 } }
								pay_short_term_gold = { target = dynasty:10000003.dynast gold = { value = root.var:web_gold_reward multiply = 0.25 } }
							} 
						}
					
					}
				}
				
			}
			
			
			#every_in_list = {
			#	variable = contractor 
			#	send_interface_toast = {
			#		title = target_killed_pay
			#		add_gold = -50
			#	}
			#}
		}
		
		if = {
			limit = {
				any_in_list = {
					variable = web_of_knives_contracts
					exists = yes
				}
			}
			every_in_list = {
				variable = web_of_knives_contracts
				remove_list_global_variable = {
					name = global_web_of_knives_targets
					target = this
				}
				remove_list_variable = {
					name = contractor
					target = root
				}
				every_player = {
					remove_list_variable = {
						name = web_of_knives_targets
						target = prev
					}
				}
				
				remove_variable = web_gold_reward
			}
		
		}
		
		
		if = { 
			limit = { has_variable = necromancer }
			var:necromancer = {
				
				remove_list_variable = {
					name = bound_wraiths
					target = root
				}
			}
		}
		if = { 
			limit = { has_variable = regnant }
			
			var:regnant = {
				remove_list_variable = {
					name = ghouls
					target = root
				}
				
			}
			
		}
		if = { 
			limit = { has_variable = owner }
			var:owner = {
				remove_list_variable = {
					name = herd
					target = root
				}
			}
			
		}
		# if = {
			# limit = {
				# OR = {
					# has_variable = puppet
					# has_variable = wraith
				# }
			# }
			# # This event doesn't exist 
			# trigger_event = POD_wraith_possession.10
		# }		
		
		if = {
			limit = {
				is_ai = no
				exists = player_heir
			}
			player_heir = {
				add_character_flag = {
					flag = tutorial_reactive_advice_succession
				}
			}
		}
		if = {
			limit = {
				exists = global_var:mongol_empire_was_broken_up
				exists = global_var:handed_out_mongolia_in_mongol_succession # If Mongolia wasn't properly granted, something has gone wrong, and the primary_heir should receive the Mongol Empire as expected
				any_held_title = {
					this = title:e_mongol_empire
				}
			}
			destroy_title = title:e_mongol_empire
		}
		#Moncada special content
		if = {
			limit = {
				this = character:790178
				character:505154 = { 
					is_alive = yes
					is_ai = no	
					}
			}
			character:505154 = {
				trigger_event = {
					id =  PODcanon_chars.20
					days = 1
					}
				}
		}		

		#Hardestadt's takeover
		if = {
			limit = {
				this = character:505050
				any_child = { 
					is_alive = yes
					is_female = no	
					}
				}
			character:505050 = {
				random_child = {
					limit = {
						is_female = no
						}
				trigger_event = {
					id =  PODcanon_chars.58
					days = 1
					}
				}
			}
		}

		#cap
		if = {
			limit = {
				has_POD_resurrection_trigger = yes
			}
			POD_create_resurrection_effect = yes
		}
		invalidate_claimant_factions_on_death_effect = yes
	}
	events = {
		death_management.0096	# Updates marriage opinions of other spouses if relevant.
		death_management.0097	# Sets a flag if the dead one was someone you loved (to make suicide available)
		death_management.0098 	# Removes rejected_from_marriage_bed_modifier if spouse dies
		death_management.0099 	# If spouse is pregnant, save as variable
		death_management.0001 	# Finds characters which would care about death for notifications etc.
		intrigue_dread.4012		# Interrupts this event chain for the imprisoner when their prisoner dies.
		stewardship_duty.1065 	# Removes the patron buff from their employer when this character dies.
		stewardship_duty.1067 	# When you die removes your clients fron your court.
		great_holy_war.0024		# Replace recipient for ongoing GHW.
		great_holy_war.0028 	# Beneficiary died, reset and replace.
		great_holy_war.0080 	# ghw_sponsor died, replace.
		great_holy_war.0083		# ghw fallback beneficiary dies.
		religious_decision.0301 # ancestor died and needs to be buried.
		religious_decision.0311 # Dynasty member has died and you can raise a runestone
		game_rule.2				# Exclave Independence
		martial_authority.2057	# Right-Hand Person Dies 
		martial_authority.2055  # Right-handed Person (person with right-hand) Died
		death_management.1002	# Sound Foundations stat removal
		death_management.1003	# Friendly Counsel stat removal
		fp1_yearly.0052			# FP1: Rumoured Draugr
		fp1_yearly.1061			# FP1: Ship Burial
		fp1_other_decisions.0062	# FP1: someone intending to HumSac you has died
		fp1_other_decisions.0063	# FP1: designated HumSac has died
		fp1_other_decisions.0064	# FP1: remove counties offended by HumSac modifier		
	}
}

		}

		}
	}
