on_death = {
		if = { 
			limit = { has_variable = necromancer }
			var:necromancer = {
				
				every_in_list = {
					variable = bound_wraiths
					limit = { this = root}
					remove_from_list = bound_wraiths
				}
			}
		}
		if = {
			limit = {
				OR = {
					has_variable = puppet
					has_variable = wraith
				}
			}
			trigger_event = POD_wraith_possession.10
		}
}
