on_set_relation_demonic_master = {
	effect = {
		scope:target = {
			if = {
				limit = { has_perk = usurpation_of_flesh_perk }
				add_hook_no_toast = {
					type = usurpation_of_flesh_perk_hook
					target = prev
				}
			}
			add_arcana_lifestyle_xp = 100
		}
	}
}

on_remove_demonic_master = {
	effect = {
		scope:target = {
			if = {
				limit = {
					has_hook_of_type = {
						target = prev
						type = usurpation_of_flesh_perk_hook
					}
				}
				remove_hook = {
					target = prev
					type = usurpation_of_flesh_perk_hook
				}
			}
		}
	}
}
