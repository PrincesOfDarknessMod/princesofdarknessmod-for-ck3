POD_player_power_maintenance_quarterly_pulse = {
	trigger = {
		is_ai = no
		is_imprisoned = no
	}
	on_actions = {
		POD_player_power_maintenance_monthly_pulse
		POD_umbra_location_maintenance_monthly_pulse
		delay = { months = 1 }
		POD_player_power_maintenance_monthly_pulse
		POD_umbra_location_maintenance_monthly_pulse
		delay = { months = 2 }
		POD_player_power_maintenance_monthly_pulse
		POD_umbra_location_maintenance_monthly_pulse
	}
}

POD_player_power_maintenance_monthly_pulse = {
	trigger = {
		is_ai = no
		is_imprisoned = no
	}
	effect = {
		if = {
			limit = {
				POD_lesser_chi_user_trigger = yes
				is_east_mummy_trigger = no
				NOT = { has_character_modifier = celestial_barrier_modifier }
				NOT = { has_character_modifier = traveling_spirit_world_modifier }
			}
			if = {
				limit = {
					has_yinchi_5_or_greater_trigger = no
				}
				POD_yinchi_increase_1_effect = yes
			}
			if = {
				limit = {
					has_yangchi_5_or_greater_trigger = no
				}
				POD_yangchi_increase_1_effect = yes
			}
		}
		if = {
			limit = {
				is_east_mummy_trigger = yes
				NOT = { has_character_modifier = celestial_barrier_modifier }
				NOT = { has_character_modifier = yangchi10_modifier }
				NOT = { has_character_modifier = traveling_spirit_world_modifier }
			}
			POD_yangchi_increase_1_effect = yes
		}
		if = {
			limit = {
				is_amenti_trigger = yes
				NOT = { has_character_modifier = sekhem10_modifier }
				NOT = { has_character_modifier = traveling_spirit_world_modifier }
				exists = location
				OR = {
					location = { in_land_of_faith_trigger = yes }
					AND = {
						has_trait = cabiri
						location = { 
							OR = {
								geographical_region = world_europe
								geographical_region = world_asia_minor
							}
						}
					}
				}
			}
			POD_sekhem_increase_effect = yes
		}
	}
}

POD_umbra_location_maintenance_monthly_pulse = {
	trigger = {
		has_variable = spirit_location
	}
	effect = {
		if = {
			limit = {
				var:spirit_location = flag:skinlands
				has_trait = wraith
				NOT = { has_trait = risen }
				num_of_relation_necromancer = 0
			}
			send_interface_toast = {
				title = POD_wraith_maintenance
				left_icon = root
				add_stress = medium_stress_impact_gain
				custom_tooltip = POD_wraith_maintenance.tt
			}
		}
		if = {
			limit = {
				var:spirit_location = flag:shadowlands
				has_trait = wraith
			}
			random = {
				chance = {
					value = 0
					add = lost_fetter_size
				}
				send_interface_toast = {
					title = POD_wraith_maintenance
					left_icon = root
					add_stress = medium_stress_impact_gain
					custom_tooltip = shadowlands_fetter_maintenance_tooltip
				}
			}
		}
		if = {
			limit = {
				var:spirit_location = flag:glen
			}
			add_stress = minor_stress_loss
		}
		if = {
			limit = {
				var:spirit_location = flag:abyss
			}
			add_stress = major_stress_gain
		}
		if = {
			limit = {
				var:spirit_location = flag:first_city
			}
			add_prestige = minor_prestige_gain
		}
		if = {
			limit = {
				POD_is_wrym_servant_trigger = no
			}
			if = {
				limit = {
					var:spirit_location = flag:labyrinth
				}
				add_stress = minor_stress_gain
			}
			if = {
				limit = {
					OR = {
						var:spirit_location = flag:well_of_the_void
						var:spirit_location = flag:malfeas
					}
				}
				add_stress = medium_stress_gain
			}
			if = {
				limit = {
					var:spirit_location = flag:black_spiral_labyrinth
				}
				add_stress = major_stress_gain
			}
		}
		if = {
			limit = {
				is_amenti_trigger = yes
				NOT = { has_character_modifier = sekhem10_modifier }
				var:spirit_location = flag:dark_kingdom_of_sand
			}
			POD_sekhem_increase_effect = yes
		}
	}
}
