# RANDOM PULSE #
# For events that can be used in any pulse.

umbra_travel_random_pulse = {
	trigger = {
		exists = scope:activity
		has_supernatural_modifier_beyond_zero_trigger = yes
		OR = {
			has_character_modifier = traveling_yang_world_modifier
			has_character_modifier = traveling_umbra_modifier
		}
	}
	effect = {
		POD_universal_energy_decrease_multiple_effect = { COUNT = 1 }
	}
	random_events = {
		1000 = POD_spirit_travel.1000 # Nothing Happen
		5 = POD_spirit_travel.1001
		100 = POD_spirit_travel.1002
	}
	fallback = umbra_travel_end_pulse #To complete the activity
}

umbra_travel_end_pulse = {
	events = {
		POD_umbra_travel.10
		POD_umbra_travel.20
	}
}
