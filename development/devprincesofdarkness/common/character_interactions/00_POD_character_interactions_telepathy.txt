POD_telepathy_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	interface_priority = 88
	desc = POD_telepathy_interaction_desc
	
	

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		scope:actor = { has_POD_vamp_blood_trigger = yes }
		scope:actor = {
			OR = {
				has_trait = auspexadvanced
				has_perk = telepathy_perk
			}
		
		}
		scope:recipient = {
			any_secret = {
				NOT = {is_known_by = scope:actor}
			}
		}
		scope:recipient = { 
			NOT = {
				has_trait = torpor
			}
		}
	}

	is_valid_showing_failures_only = {
			scope:actor = { NOT = { has_character_modifier = bloodhunger5_modifier } }
	}


	on_accept = {
		if = {
			limit = { scope:recipient = { is_vampire_trigger = yes } }
			scope:actor = {POD_willpower_rouse_check_effect = yes}
		}
		scope:actor = { POD_hunger_rouse_check_effect = yes }
		scope:recipient = { trigger_event = POD_auspex.100}
		scope:actor = { add_character_flag = used_telepathy }
	}
	
	#ai_potential = {
	#	always = no
	#}
	
	ai_targets = {
		ai_recipients = peer_vassals
		ai_recipients = liege
		ai_recipients = neighboring_rulers
		ai_recipients = vassals
	}
	ai_targets = {
		ai_recipients = realm_characters
		max = 10
	}
	ai_frequency = 60
	auto_accept = yes
	
	ai_will_do = {
		base = 0
		
		opinion_modifier = {
			opinion_target = scope:recipient
			multiplier = -0.25
		}
		
		
		
		modifier = {
			add = 10
			scope:recipient = liege 			
		}
		
		modifier = {
			add = 10
			scope:recipient = liege
			OR = {
				has_trait = ambitious 
				has_trait = fickle
				has_trait = callous
				has_trait = deceitful
				has_trait = craven
			}
		}
		
		modifier = {
			add = -10
			scope:recipient = liege
			OR = {
				has_trait = content
				has_trait = just
				has_trait = honest
				has_trait = lazy
				has_trait = humble
			}
		}
		
		modifier = {
			add = 10
			scope:recipient = { is_powerful_vassal = yes }
		}
		
		modifier = {
			add = 10
			scope:recipient = { is_powerful_vassal = yes }
			OR = {
				has_trait = arbitrary
				has_trait = callous
				has_trait = arrogant
				has_trait = deceitful
				
				
			}
		}
		
		
		modifier = {
			factor = 0.5
			OR = {
				scope:recipient = { has_trait = fortitude }
				scope:recipient = { has_trait = fortitudeadvanced }
			}
		}
		
		modifier = {
			add = 25
			ai_diplomacy_stance = { target = scope:recipient  stance = threat }
		
		}
		
		modifier = {
			factor = 0.5
			scope:recipient.intrigue > intrigue
		}
		
		
	}
}
