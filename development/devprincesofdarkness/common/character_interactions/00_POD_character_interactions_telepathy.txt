POD_telepathy_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	interface_priority = 88
	desc = POD_telepathy_interaction_desc
	
	ai_frequency = 0

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		scope:actor = { has_POD_vamp_blood_trigger = yes }
		scope:actor = { is_ai = no }
		scope:actor = {
			OR = {
				has_trait = auspexadvanced
				has_perk = telepathy_perk
			}
		
		}
		scope:recipient = {
			any_secret = {
				NOT = {is_known_by = scope:actor}
			}
		}
		scope:recipient = { 
			NOT = {
				has_trait = torpor
			}
		}
	}

	is_valid_showing_failures_only = {
			scope:actor = { NOT = { has_character_modifier = bloodhunger5_modifier } }
	}


	on_accept = {
		if = {
			limit = { scope:recipient = { is_vampire_trigger = yes } }
			scope:actor = {POD_willpower_rouse_check_effect = yes}
		}
		POD_hunger_rouse_check_effect = yes
		scope:recipient = { trigger_event = POD_auspex.100}
	}
	
	#ai_potential = {
	#	always = no
	#}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
