firebombs_interaction = {
	category = interaction_category_specialweapons
	interface_priority = 60
	common_interaction = yes
	desc = firebombs_interaction_desc

	greeting = positive
	notification_text = FIREBOMBS_PROPOSAL

	answer_accept_key = FIREBOMBS_ACCEPT
	answer_reject_key = FIREBOMBS_REJECT

	is_shown = {
		NOT = {
			scope:recipient = scope:actor
		}
	scope:actor = { has_trait = supehunter }
	scope:recipient = { has_trait = supehunter }
	scope:actor.dynasty = {
			 	has_dynasty_modifier = masters_of_fire
			}
	}

	is_valid_showing_failures_only = {
		scope:actor.gold >= 25
		scope:recipient = { is_diplomatically_available = yes }
		custom_description = {
			text = has_fireweapons
			scope:recipient = {
				NOR = {	
						has_character_modifier = firebombs_modifier
						has_character_modifier = firespewer_modifier
				}
			}
		}
	}
	
	cost = { gold = 25 }

	on_accept = {
		scope:actor = {
			send_interface_message = {
				type = event_generic_neutral
				title = gavefirebombs_notification
				right_icon = scope:recipient
					}
				}
		scope:recipient = {
			send_interface_message = {
				type = event_generic_neutral
				title = gotfirebombs_notification
				right_icon = scope:actor
				add_character_modifier = firebombs_modifier
			}
		}
	}
	
	ai_accept = {
		base = 100
	}
	
	ai_min_reply_days = 1
	ai_max_reply_days = 1
	
	ai_will_do = {
		base = 0
	}
}
