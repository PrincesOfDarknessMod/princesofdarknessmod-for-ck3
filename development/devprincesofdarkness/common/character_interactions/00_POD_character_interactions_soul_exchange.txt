POD_soul_exchange_interaction = {
	category = interaction_category_vampire

	desc = POD_soul_exchange_interaction_desc
	
	cooldown = { months = 12 }	

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { is_ai = no }
		scope:actor = { 
			OR = { 
				has_perk = soul_exchange_perk 
				has_perk = necromancy_soul_exchange_perk 
			}
		}
		POD_is_in_same_plane_trigger = yes
	}

	is_valid_showing_failures_only = {
		custom_description = {
            object = scope:actor
            text = "I_must_be_at_the_same_location"
            scope:actor = { location = scope:recipient.location }
        }
	}

	populate_recipient_list = {
		scope:actor = {
			every_courtier_or_guest = {
				limit = { 
					NOR = {
						has_trait = torpor
						has_trait = frozen
						has_trait = effigy	
						has_character_modifier = traveling_spirit_world_modifier					
					}
				}
				add_to_list = characters
			}
		}
		scope:recipient = {
			every_courtier_or_guest = {
				limit = { 
					NOR = {
						has_trait = torpor
						has_trait = frozen
						has_trait = effigy	
						has_character_modifier = traveling_spirit_world_modifier					
					}
				}
				add_to_list = characters
			}
		}
	}
	
	cost = {
		piety = major_piety_value
	}

	on_accept = {
		scope:actor = {
			POD_universal_energy_decrease_multiple_effect = { COUNT = 2 }
			POD_willpower_rouse_check_effect = yes
		}
		scope:recipient = {
			add_opinion = {
				target = scope:actor
				modifier = hurt_opinion
				opinion = -20
			}
		}
		scope:secondary_recipient = {
			add_opinion = {
				target = scope:actor
				modifier = hurt_opinion
				opinion = -20
			}
		}
		
		
		show_as_tooltip = { 
			scope:recipient = { POD_custom_copy_vanilla_traits_effect = { ORIGIN = secondary_recipient } }
			scope:recipient = { POD_remove_vanilla_traits_effect = yes }
			scope:secondary_recipient = { POD_custom_copy_vanilla_traits_effect = { ORIGIN = recipient } } 
			scope:secondary_recipient = { POD_remove_vanilla_traits_effect = yes }
		}
		
		hidden_effect = {
			create_character = {
				save_scope_as = dummyclone
				age = 16
				faith = faith:innocence
				gender = female
				random_traits = no
				dynasty = none
				employer = scope:actor
			}
			scope:dummyclone = { POD_custom_copy_vanilla_traits_effect = { ORIGIN = recipient } }
			scope:recipient = { POD_remove_vanilla_traits_effect = yes }
			scope:recipient = { POD_custom_copy_vanilla_traits_effect = { ORIGIN = secondary_recipient } }
			scope:secondary_recipient = { POD_remove_vanilla_traits_effect = yes }
			scope:secondary_recipient = { POD_custom_copy_vanilla_traits_effect = { ORIGIN = dummyclone } }
			scope:dummyclone = { death = { death_reason = death_vanished } }
		}
	}
	
	auto_accept = yes

	ai_frequency = 0
	
	#ai_potential = {
	#	always = no
	#}
	
	ai_will_do = {
		base = 0
	}
}
