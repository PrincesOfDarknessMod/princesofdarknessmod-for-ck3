POD_dragon_chariot_interaction = {
	category = interaction_category_vampire

	desc = POD_dragon_chariot_interaction_desc

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { 
			OR = {
				has_trait = kueijin 
				has_trait = dhampyr
			}
		}
		scope:actor = { is_ai = no }
		scope:actor = { has_perk = ride_the_dragon_perk }
		scope:actor = { is_commanding_army = yes }
		scope:recipient = { exists = location }
		POD_is_in_same_plane_trigger = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			wall_too_strong_trigger = no
		}
		scope:actor = { 
			OR = {
				has_perk = dragon_chariot_perk
				custom_description = {
					text = POD_army_size_less_then_thousand 
					commanding_army = {
						army_size <= 1000
					}
				}
			}
		}
		custom_description = {
			text = not_has_dragon_nest_in_your_location
			scope:actor.location = {
				OR = {
					has_building_or_higher = dragon_nest_01
					has_province_modifier = temporary_dragon_nest_modifier
					AND = {
						exists = barony
						barony = { is_holy_site = yes } # Natural Dragon Nest?
					}
				}
			}
		}
		custom_description = {
			text = not_has_dragon_nest_in_target_location
			scope:recipient.location = {
				OR = {
					has_building_or_higher = dragon_nest_01
					has_province_modifier = temporary_dragon_nest_modifier
					AND = {
						exists = barony
						barony = { is_holy_site = yes } # Natural Dragon Nest?
					}
				}
			}
		}
	}

	on_accept = {
		scope:actor = { 		
			POD_chi_decrease_effect = { COUNT = 5 }
			POD_willpower_rouse_check_effect = yes
			commanding_army = {
				set_army_location = scope:recipient.location
			}
		}
	}

	ai_frequency = 0
	
	#ai_potential = {
	#	always = no
	#}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
