POD_distant_death_kata_interaction = {
	category = interaction_category_vampire
	interface_priority = 90
	desc = POD_distant_death_kata_interaction_desc
	
	cooldown = { months = 3 }	
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_perk = distant_death_kata_perk }
		scope:actor = { is_kueijin_trigger = yes }
		scope:actor = { is_ai = no }
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
			prowess >= 10
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
		}
	}

	on_accept = {
		scope:actor = { POD_chi_decrease_effect = { COUNT = 1 } }	
		scope:actor = { POD_willpower_rouse_check_effect = yes }
		
		# Reversed since increase_wounds_multiple_effect don't give any wound if doesn't have one
		if = {
			limit = {
				scope:actor = { prowess >= 100 }
			}
			scope:recipient = { POD_losing_health_level_effect = { COUNT = 10 } }
		}
		else_if = {
			limit = {
				scope:actor = { prowess >= 90 }
			}
			scope:recipient = { POD_losing_health_level_effect = { COUNT = 9 } }
		}
		else_if = {
			limit = {
				scope:actor = { prowess >= 80 }
			}
			scope:recipient = { POD_losing_health_level_effect = { COUNT = 8 } }
		}
		else_if = {
			limit = {
				scope:actor = { prowess >= 70 }
			}
			scope:recipient = { POD_losing_health_level_effect = { COUNT = 7 } }
		}
		else_if = {
			limit = {
				scope:actor = { prowess >= 60 }
			}
			scope:recipient = { POD_losing_health_level_effect = { COUNT = 6 } }
		}
		else_if = {
			limit = {
				scope:actor = { prowess >= 50 }
			}
			scope:recipient = { POD_losing_health_level_effect = { COUNT = 5 } }
		}
		else_if = {
			limit = {
				scope:actor = { prowess >= 40 }
			}
			scope:recipient = { POD_losing_health_level_effect = { COUNT = 4 } }
		}
		else_if = {
			limit = {
				scope:actor = { prowess >= 30 }
			}
			scope:recipient = { POD_losing_health_level_effect = { COUNT = 3 } }
		}
		else_if = {
			limit = {
				scope:actor = { prowess >= 20 }
			}
			scope:recipient = { POD_losing_health_level_effect = { COUNT = 2 } }
		}
		else = {
			scope:recipient = { POD_losing_health_level_effect = { COUNT = 1 } }
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
