POD_distant_death_kata_interaction = {
	category = interaction_category_vampire
	#interface_priority = 90
	desc = POD_distant_death_kata_interaction_desc
	
	cooldown = { months = 9 }	
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_perk = distant_death_kata_perk }
		scope:actor = { 
			OR = {
				has_trait = kueijin 
				has_trait = dhampyr
			}
		}
		scope:actor = { is_ai = no }
		POD_is_in_same_plane_trigger = yes
	}
	
	is_valid_showing_failures_only = {
		hurt_wraith_trigger = yes
		scope:actor = {
			prowess >= POD_prowess_low_skill_rating
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
		}
		custom_description = {
            object = scope:actor
            text = "soul_demon_incompatible"
            scope:actor = { using_demon_arts_trigger = no }
        }
	}

	on_accept = {	
		scope:actor = { POD_willpower_rouse_check_effect = yes }
		
		if = {
			limit = {
				scope:actor = { prowess >= POD_prowess_extremely_high_skill_rating }
			}
			scope:actor = { add_piety = -1000 }
			scope:actor = { POD_chi_decrease_effect = { COUNT = 10 } }
			increase_wounds_multiple_effect = { REASON = fight AMOUNT = 4 }
		}
		else_if = {
			limit = {
				scope:actor = { prowess >= POD_prowess_high_skill_rating }
			}
			scope:actor = { add_piety = -500 }
			scope:actor = { POD_chi_decrease_effect = { COUNT = 5 } }
			increase_wounds_multiple_effect = { REASON = fight AMOUNT = 3 }
		}
		else_if = {
			limit = {
				scope:actor = { prowess >= POD_prowess_medium_skill_rating }
			}
			scope:actor = { add_piety = -300 }
			scope:actor = { POD_chi_decrease_effect = { COUNT = 3 } }
			increase_wounds_multiple_effect = { REASON = fight AMOUNT = 2 }
		}
		else = {
			scope:actor = { add_piety = -100 }
			scope:actor = { POD_chi_decrease_effect = { COUNT = 1 } }
			increase_wounds_multiple_effect = { REASON = fight AMOUNT = 1 }
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
