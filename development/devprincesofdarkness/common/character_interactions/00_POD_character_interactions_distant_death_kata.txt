POD_distant_death_kata_interaction = {
	category = interaction_category_vampire
	interface_priority = 90
	desc = POD_distant_death_kata_interaction_desc
	
	cooldown = { months = 1 }	
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_perk = distant_death_kata_perk }
		scope:actor = { is_kueijin_trigger = yes }
		scope:actor = { is_ai = no }
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
			prowess >= 10
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
		}
	}

	on_accept = {
		scope:actor = { POD_chi_decrease_effect = { COUNT = 1 } }	
		scope:actor = { POD_willpower_rouse_check_effect = yes }
		
		if = {
			limit = {
				scope:actor = { prowess >= 90 }
			}
			scope:recipient = {
				increase_wounds_multiple_effect = {
					REASON = fight
					AMOUNT = 4
				}
			}
		}
		else_if = {
			limit = {
				scope:actor = { prowess >= 60 }
			}
			scope:recipient = {
				increase_wounds_multiple_effect = {
					REASON = fight
					AMOUNT = 3
				}
			}
		}
		else_if = {
			limit = {
				scope:actor = { prowess >= 30 }
			}
			scope:recipient = {
				increase_wounds_multiple_effect = {
					REASON = fight
					AMOUNT = 2
				}
			}
		}
		else = {
			scope:recipient = {
				increase_wounds_multiple_effect = {
					REASON = fight
					AMOUNT = 1
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
