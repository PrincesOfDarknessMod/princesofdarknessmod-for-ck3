POD_harmony_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	interface_priority = 88
	desc = POD_harmony_interaction_desc
	
	cooldown = { months = 12 }	
	
	is_shown = {
		scope:actor = scope:recipient
		scope:recipient = { is_adult = yes }
		scope:actor = { has_POD_vamp_blood_trigger = yes }
		scope:actor = { has_perk = harmony_with_the_all_perk }
		scope:recipient = { 
			NOT = {
				has_trait = torpor
			}
		}
		scope:actor = { is_ai = no }
	}

	is_valid_showing_failures_only = {
			scope:actor = { NOT = { has_character_modifier = bloodhunger5_modifier } }
	}

	on_accept = {
		if = {
			limit = { scope:recipient = { is_vampire_trigger = yes } }
			scope:actor = { POD_willpower_rouse_check_effect = yes}
		}
		scope:recipient = { add_stress = -150 }
		scope:recipient = { POD_hunger_rouse_check_effect = yes }
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
