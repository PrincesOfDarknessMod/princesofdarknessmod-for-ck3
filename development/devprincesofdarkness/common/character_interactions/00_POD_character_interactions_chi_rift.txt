POD_chi_rift_interaction = {
	category = interaction_category_vampire
	#common_interaction = yes
	#interface_priority = 90
	
	desc = POD_chi_rift_interaction_desc
	
	cooldown = { years = 5 }
	
	auto_accept = yes

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { is_ai = no  }
		scope:actor = { 
			OR = {
				has_trait = kueijin 
				has_trait = dhampyr
			}
		}
		scope:actor = { has_perk = chi_rift_perk }
		scope:recipient = { is_landed = yes }
		NOT = { scope:recipient.liege = scope:actor }
		scope:recipient.highest_held_title_tier > tier_barony
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
			wall_too_strong_trigger = no
		}
	}

	on_accept = {
	   custom_tooltip = POD_chi_rift_interaction.tt
	   scope:actor = {
			add_piety = -1500
			POD_chi_decrease_effect = { COUNT = 3 }
			POD_willpower_rouse_check_effect = yes
			duel = {
				skill = learning
				value = 10
				desc = POD_chi_rift.duel.desc
			
				50 = {
					desc = POD_chi_rift.duel.success
					
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					
					modifier = {
						add = -300
						scope:recipient = { 
							has_perk = celestial_needle_perk	
						}
					}
					
					custom_tooltip = POD_chi_rift.duel.tt.attacker_wins
					
					
					hidden_effect = {
						scope:recipient = {
							send_interface_toast = {
								title = POD_whole_realm_was_havocked_toast
								left_icon = scope:recipient
								right_icon = scope:actor
							}
							every_sub_realm_county = {
								add_county_modifier = {
									modifier = chi_rift_power_modifier
									years = 5
								}
							}
							add_opinion = {
								target = scope:actor
								modifier = POD_havocked_my_realm
							}						
						}
						
						scope:actor = {
							send_interface_toast = {
								title = POD_havocked_enemy_realm_toast
								left_icon = scope:actor
								right_icon = scope:recipient
							}
						}
					}
				}
				
				50 = {
					desc = POD_chi_rift.duel.failure

					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier
					}
					
					modifier = {
						add = 10
						scope:recipient = { 
							has_trait = obtenebrationadvanced	
						}
					}
					modifier = {
						add = 10
						scope:recipient = { 
							has_perk = chi_rift_perk	
						}
					}
					modifier = {
						add = 5
						scope:recipient = { 
							has_trait = truefaith	
						}
					}
					modifier = {
						add = 10
						scope:recipient = { 
							has_trait = beatific	
						}
					}
					
					custom_tooltip = POD_chi_rift.duel.tt.attacker_lost
					hidden_effect = {
						scope:actor = {
							send_interface_toast = {
								title = POD_chi_rift_failed_toast
								left_icon = scope:actor
								right_icon = scope:recipient
							}
						}
					}
				}
			}
	   }
    }
	
	ai_frequency = 360
	
	ai_targets = {
		ai_recipients = neighboring_rulers
	}
	
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 45
			scope:actor = {
			    is_at_war_with = scope:recipient
			}
			piety > 1500
		}
		modifier = {
		    add = 10
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
	}  	
}

POD_celestial_needle_interaction = {
	category = interaction_category_vampire
	#common_interaction = yes
	#interface_priority = 90
	
	desc = POD_celestial_needle_interaction_desc
	
	cooldown = { months = 3 }
	
	auto_accept = yes

	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { is_ai = no }
		scope:actor = { 
			OR = {
				has_trait = kueijin 
				has_trait = dhampyr
			}
		}
		scope:actor = { has_perk = celestial_needle_perk }
	}
	
	is_valid_showing_failures_only = {
		custom_description = {
			text = not_have_chi_rift_in_location
			location.county = {
				has_county_modifier = chi_rift_power_modifier 
			}
		}
		scope:actor = {
			wall_too_strong_trigger = no
		}
	}

	on_accept = {
		location.county = {
			remove_county_modifier = chi_rift_power_modifier 
		}
    }
	
	ai_frequency = 0
	
	ai_will_do = {
		base = 0
	}  	
}

	   }
    }
