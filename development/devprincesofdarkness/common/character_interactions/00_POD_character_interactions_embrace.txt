# POD Schemes
# Embrace Childer Vampire
# Sparc

embrace_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	interface_priority = 83
	scheme = embrace
	
	ai_frequency = 0

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		#scope:actor = { location = scope:recipient.location }
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		scope:actor = { is_vampire_trigger = yes  }
		scope:recipient = { is_vampire_trigger = no }
		scope:actor = { is_ai = no }
		scope:recipient = { POD_can_be_embraced_scheme_trigger = yes }
		scope:actor = { NOT = { has_trait = kueijin } }
		scope:actor = { NOT = { has_trait = wraith } }
		trigger_if = {
			limit = {
				scope:recipient = { POD_is_supehunter_trigger = yes }
					}
			scope:recipient = { is_imprisoned_by = scope:actor }
		}
	}

	is_valid_showing_failures_only = {
		custom_description = {
			object = scope:actor
			text = "I_can_only_plot_one_embrace_at_time"
			scope:actor = {
				NOT = {
					any_scheme = {
						scheme_type = embrace
					}
				}
			}
		}
		custom_description = {
			object = scope:recipient
			text = "Blood_is_holy"
			scope:recipient = {
				NOR = {
					has_trait = truefaith
					has_trait = beatific
					has_trait = orison_sanguine
					}
				}
		}
		custom_description = {
			object = scope:actor
			text = "I_must_be_at_the_same_location"
			scope:actor = { location = scope:recipient.location }
		}	
		scope:actor = {
			can_start_scheme = {
				type = embrace
				target = scope:recipient
			}
		}
		scope:recipient = {
		is_landed = no
		}
	}

	desc = {
		triggered_desc = {
			trigger = { scope:recipient.age >= scope:actor.age }
			desc = embrace_childe_older_than_you_tt
		}
		triggered_desc = {
			trigger = {
				scope:actor = {
					NOT = { is_at_war_with = scope:recipient }
					can_start_scheme = {
						type = embrace
						target = scope:recipient
					}

				}
			}
			desc = scheme_interaction_tt_embrace_approved
		}

	}

	on_accept = {
		hidden_effect = {
			scope:actor = {
				start_scheme = {
					type = embrace
					target = scope:recipient
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
