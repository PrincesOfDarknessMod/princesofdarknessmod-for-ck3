POD_choose_sire_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	interface_priority = 100
	use_diplomatic_range = no

	is_shown = {
		scope:actor = {
			has_character_flag = chargen_choosing_sire_flag
			NOT = {
				scope:actor = scope:recipient
				is_vampgeneration16_trigger = yes
			}
		}
		scope:recipient = {
			is_vampire_trigger = yes
			NOT = {
				is_kueijin_trigger = yes
			}
		}
	}
	auto_accept = yes

	desc = POD_choose_sire_interaction_desc

	on_accept = {
		scope:recipient = {
			save_scope_as = vampiresire
		}
		scope:actor = {
			save_scope_as = created_vampirechilde
			trigger_event = POD_chargen.600
		}
	}
}
