# POD Personal Character Interactions
#  Bind famulus

POD_bind_famulus_interaction = {
	category = interaction_category_vampire
	#interface_priority= 91
	icon = POD_animalism_interaction
	desc = POD_bind_famulus_interaction_desc
	options_heading = POD_bind_famulus_interaction_options_heading

	send_option = {
		flag = POD_bind_famulus_interaction_horse 
		localization = POD_bind_famulus_interaction_horse	
	}

	send_option = {
		flag = POD_bind_famulus_interaction_dog
		localization = POD_bind_famulus_interaction_dog	
	} 
	
	send_option = {
		flag = POD_bind_famulus_interaction_cat
		localization = POD_bind_famulus_interaction_cat		
	} 

	send_options_exclusive = yes
	
	auto_accept = yes
	ai_frequency = 120
	use_diplomatic_range = no
	
	ai_targets = {
		ai_recipients = self
	}
	
	cooldown = { months = 60 }
	
	is_shown = {
		scope:actor = scope:recipient 
		scope:recipient = { is_adult = yes }
		scope:actor = { has_POD_vamp_blood_trigger = yes  }	
		scope:actor = {
				NOT = { has_character_modifier = cat_story_modifier }
				NOT = { has_character_modifier = horse_story_modifier }
				NOT = { has_character_modifier = dog_story_modifier }
				NOT = { has_character_modifier = horse_story_modifier }
				NOT = { has_character_modifier = horse_strength_modifier }
				NOT = { has_character_modifier = horse_agility_modifier }
				NOT = { has_character_modifier = horse_endurance_modifier }
				NOT = { has_character_modifier = horse_fierceness_modifier }
				NOT = { has_character_modifier = horse_strength_strength_modifier }
				NOT = { has_character_modifier = horse_agility_strength_modifier }
				NOT = { has_character_modifier = horse_agility_agility_modifier }
				NOT = { has_character_modifier = horse_endurance_strength_modifier }
				NOT = { has_character_modifier = horse_endurance_agility_modifier }
				NOT = { has_character_modifier = horse_endurance_endurance_modifier }
				NOT = { has_character_modifier = horse_fierceness_strength_modifier }
				NOT = { has_character_modifier = horse_fierceness_agility_modifier }
				NOT = { has_character_modifier = horse_fierceness_endurance_modifier } 
				NOT = { has_character_modifier = horse_fierceness_fierceness_modifier }
				NOT = { has_character_modifier = horse_practice_modifier }
				NOT = { has_character_modifier = cursed_horse_modifier }			
		}	
		scope:actor = {
			OR = {
				has_trait = animalismadvanced
				has_perk = bond_famulus_perk
			}
		}
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
			NOT = { has_character_modifier = bloodhunger5_modifier }			
		}	
	}	

	on_accept = {
		scope:actor = {
		
			if = {
				limit = {
					scope:POD_bind_famulus_interaction_cat = yes
				}
				add_character_flag = bound_famulus_cat
				POD_hunger_rouse_check_effect = yes
				start_cat_story_cycle_effect = yes	
			}	
			
			if = {
				limit = {
					scope:POD_bind_famulus_interaction_dog = yes
				}
				add_character_flag = bound_famulus_dog
				POD_hunger_rouse_check_effect = yes
				start_dog_story_cycle_effect = yes				
			}	
			
			if = {
				limit = {
					scope:POD_bind_famulus_interaction_horse = yes
				}
				scope:actor = {
					remove_short_term_gold = medium_gold_value
				}	
				add_character_flag = bound_famulus_horse
				POD_hunger_rouse_check_effect = yes
				scope:actor = { 
					hidden_effect = { start_warhorse_story_cycle_effect = yes }
					add_character_modifier = {
						modifier = horse_story_modifier
					}
					add_character_flag = had_warhorse_story
				}
				random_owned_story = {
					limit = { story_type = story_cycle_martial_lifestyle_warhorse }
					assign_horse_gender_effect = { GENDER = female }						
				}
			}	
			
		}	
	}

	ai_will_do = {
		base = 0
		modifier = {
			add = 100
			NOT = {

				scope:actor = { has_character_modifier =  cat_story_modifier  }
				scope:actor = { has_character_modifier =  dog_story_modifier  }
			}
		}
	}
}
