POD_train_psi_interaction = {
	category = interaction_category_holyarts
	icon = POD_psi_interaction
    desc = POD_train_psi_interaction_desc
	common_interaction = yes
	
	cooldown = { months = 30 }
	cost = { piety = medium_piety_value }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_trait = psychic }
		scope:actor = { has_perk = synergy_perk }
		scope:recipient = { has_trait = psychic }
	}

	is_valid_showing_failures_only = {
		custom_description = {
            object = scope:actor
            text = "I_must_be_at_the_same_location"
            scope:actor = { location = scope:recipient.location }
        }
	    scope:actor = {
			psi_lifestyle_perk_points >= 1
		}
		scope:recipient = { is_imprisoned = no }
	}
	
	on_accept = {
	    scope:actor = { 
			add_psi_lifestyle_perk_points = -1
			stress_impact = {
				base = medium_stress_gain
			}
		}			
	    scope:recipient = { add_psi_lifestyle_perk_points = 1 }	
	}

	ai_frequency = 0
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
