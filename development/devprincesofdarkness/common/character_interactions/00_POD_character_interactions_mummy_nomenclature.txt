POD_discover_true_name_interaction = {
	category = interaction_category_mummy
	scheme = POD_discover_true_name
	use_diplomatic_range = no
	
	desc = {
		triggered_desc = {
			trigger = {
				scope:actor = {
					NOT = { is_at_war_with = scope:recipient }
					can_start_scheme = {
						type = POD_discover_true_name
						target = scope:recipient
					}

				}
			}
			desc = scheme_interaction_tt_POD_discover_true_name_approved
		}
	}

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_trait = mummy }
		scope:actor = { is_east_mummy_trigger = no }
		scope:actor = { is_ai = no }
		scope:recipient = { POD_is_clone_trigger = no }
		scope:actor = { 
			NOT = { 
				has_hook_of_type = {
					target = scope:recipient
					type = true_name_hook
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			can_start_scheme = {
				type = POD_discover_true_name
				target = scope:recipient
			}
		}
	}

	on_accept = {
		hidden_effect = {
			scope:actor = {
				start_scheme = {
					type = POD_discover_true_name
					target = scope:recipient
				}
			}
		}
	}
	
	ai_frequency = 0
	
	#ai_potential = {
	#	always = no
	#}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
