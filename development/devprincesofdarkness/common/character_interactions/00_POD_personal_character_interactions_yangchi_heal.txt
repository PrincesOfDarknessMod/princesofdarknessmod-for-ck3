# POD Personal Character Interactions

#  Chi Heal
POD_chi_heal_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	interface_priority = 110
	desc = POD_chi_heal_interaction_desc
	options_heading = POD_chi_heal_interaction_options_heading	

	use_diplomatic_range = no

	cooldown = { months = 3 }	
	
	is_shown = {
		scope:actor = scope:recipient 
		scope:actor = { 
			is_kueijin_trigger = yes 
			energy_points_value >= 1
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			NOR = {
				has_trait_with_flag = can_not_cure
				has_character_modifier = eye_of_sakhmet_modifier
			}
			OR = {
				AND = {
					has_trait = scarred
					NOT = { has_character_flag = POD_permanent_scarring }
				}
				
				has_trait = wounded_1
				has_trait = wounded_2
				has_trait = wounded_3
				
				has_trait = maimed
				has_trait = blind
				has_trait = one_legged
				AND = {
					has_trait = one_eyed
					NOT = { has_character_flag = POD_permanent_one_eyed }
					#NOT = { has_character_modifier = eye_of_hazmiel }
				}
	
				AND = {
					has_trait = disfigured
					NOT = { has_character_flag = POD_permanent_disfigured }
				}
			}
		}		
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = scarred
				NOT = { has_character_flag = POD_permanent_scarring }
			}
		}	
		flag = scarred
		localization = POD_yangchi_heal_scarred
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				OR = {
					has_trait = wounded_1
					has_trait = wounded_2
					has_trait = wounded_3
				}
			}
		}	
		flag = wounded
		localization = POD_yangchi_heal_wounded
	}

	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = blind
			}
		}	
		flag = blind
		localization = POD_yangchi_heal_blind
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = maimed
			}
		}	
		flag = maimed
		localization = POD_yangchi_heal_maimed
	}	
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = one_legged
			}
		}	
		flag = one_legged
		localization = POD_yangchi_heal_one_legged
	}

	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = one_eyed
				NOT = { has_character_flag = POD_permanent_one_eyed }
			}
		}	
		flag = one_eyed
		localization = POD_yangchi_heal_one_eyed
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = disfigured
				NOT = { has_character_flag = POD_permanent_disfigured }
			}
		}	
		flag = disfigured
		localization = POD_yangchi_heal_disfigured
	}
	
	on_accept = {
		scope:actor = { 
			POD_willpower_rouse_check_effect = yes
		}
		scope:recipient = {
			play_music_cue = "mx_cue_peace_ensues"

			if = {
				limit = {
					scope:scarred = yes
				}
				scope:recipient = {
					remove_trait = scarred
				}
				scope:actor = { POD_chi_decrease_effect = { COUNT = 1 } }
			}	
			
			if = {
				limit = {
					scope:wounded = yes
				}
				scope:recipient = {
					remove_trait = wounded_3
					remove_trait = wounded_2
					remove_trait = wounded_1
				}
				scope:actor = { POD_chi_decrease_effect = { COUNT = 1 } }
			}		

			if = {
				limit = {
					scope:one_eyed = yes
					NOT = { has_character_flag = POD_permanent_one_eyed }
				}
				scope:recipient = {
					remove_trait = one_eyed
				}
				scope:actor = { POD_chi_decrease_effect = { COUNT = 2 } }
			}	

			if = {
				limit = {
					scope:blind = yes
				}
				scope:recipient = {
					remove_trait = blind
					add_trait = one_eyed
				}
				scope:actor = { POD_chi_decrease_effect = { COUNT = 2 } }
			}	

			if = {
				limit = {
					scope:one_legged = yes
				}
				scope:recipient = {
					remove_trait = one_legged
				}
				scope:actor = { POD_chi_decrease_effect = { COUNT = 3 } }
			}	

			if = {
				limit = {
					scope:maimed = yes
				}
				scope:recipient = {
					remove_trait = maimed
				}
				scope:actor = { POD_chi_decrease_effect = { COUNT = 3 } }
			}

			if = {
				limit = {
					scope:disfigured = yes
				}
				scope:recipient = {
					remove_trait = disfigured
				}
				scope:actor = { POD_chi_decrease_effect = { COUNT = 2 } }
			}	
		}
	}
	
	auto_accept = yes
	ai_targets = {
		ai_recipients = self
	}
	ai_frequency = 12
	ai_will_do = {
		base = 100
		
	}
}
