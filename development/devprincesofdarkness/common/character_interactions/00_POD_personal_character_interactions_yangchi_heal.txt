# POD Personal Character Interactions
#  Yang Chi Heal

POD_yangchi_heal_interaction = {
	category = interaction_category_vampire
	interface_priority = 110
	desc = POD_yangchi_heal_interaction_desc
	options_heading = POD_yangchi_heal_interaction_options_heading	

	use_diplomatic_range = no

	cooldown = { months = 3 }	
	
	is_shown = {
		scope:actor = scope:recipient 
		#scope:recipient = { is_adult = yes }
		scope:actor = { is_kueijin_trigger = yes  }
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			OR = {
				has_trait = wounded_1
				has_trait = wounded_2
				has_trait = wounded_3
				has_trait = maimed
				has_trait = blind
				has_trait = disfigured
				AND = {
					has_trait = scarred
					NOT = {
						has_character_flag = POD_permanent_scarring
					}
				}
				AND = {
					has_trait = one_eyed
					OR = {
						NOT = { has_character_flag = POD_permanent_one_eyed }
					}
					#NOT = { has_character_modifier = eye_of_hazmiel }
				}
				has_trait = one_legged
			}
		}		
	}

	send_option = {
		is_shown = {
			scope:actor = { 
				has_trait = blind
				is_kueijin_trigger = yes
			}
		}	
		flag = blind
		localization = POD_yangchi_heal_blind
		#current_description = POD_yangchi_heal_blind
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				has_trait = maimed
			}
		}	
		flag = maimed
		localization = POD_yangchi_heal_maimed
		#current_description = POD_yangchi_heal_maimed
	}	
	
	send_option = {
		is_shown = {
			scope:actor = { 
				has_trait = one_legged
				is_kueijin_trigger = yes
			}
		}	
		flag = one_legged
		localization = POD_yangchi_heal_one_legged
		#current_description = POD_yangchi_heal_one_legged
	}

	send_option = {
		is_shown = {
			scope:actor = { 
				has_trait = one_eyed
				is_kueijin_trigger = yes
				NOT = { has_character_flag = POD_permanent_one_eyed }
			}
		}	
		flag = one_eyed
		localization = POD_yangchi_heal_one_eyed
		#current_description = POD_yangchi_heal_one_eyed
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				OR = {
					has_trait = wounded_1
					has_trait = wounded_2
					has_trait = wounded_3
				}
			}
		}	
		flag = wounded
		localization = POD_yangchi_heal_wounded
		#current_description = POD_yangchi_heal_wounded
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				has_trait = scarred
				NOT = { 
					has_character_flag = POD_permanent_scarring
				}
			}
		}	
		flag = scarred
		localization = POD_yangchi_heal_scarred
		#current_description = POD_yangchi_heal_scarred
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				has_trait = disfigured
				NOT = { 
					has_character_flag = POD_permanent_disfigured
				}
			}
		}	
		flag = disfigured
		localization = POD_yangchi_heal_disfigured
		#current_description = POD_yangchi_heal_disfigured
	}
	
	on_accept = {
		scope:actor = {
			POD_yangchi_decrease_effect = yes
			
			if = {
				limit = {
					scope:blind = yes
				}
				scope:actor = {
					remove_trait = blind
					add_trait = one_eyed
				}
			}	

			if = {
				limit = {
					scope:maimed = yes
				}
				scope:actor = {
					remove_trait = maimed
					POD_yangchi_decrease_effect = yes
				}
			}		

			if = {
				limit = {
					scope:one_legged = yes
				}
				scope:actor = {
					remove_trait = one_legged
					POD_yangchi_decrease_effect = yes
				}
			}				

			if = {
				limit = {
					scope:disfigured = yes
				}
				scope:actor = {
					remove_trait = disfigured
					POD_yangchi_decrease_effect = yes
				}
			}

			if = {
				limit = {
					scope:one_eyed = yes
					NOT = { has_character_flag = POD_permanent_one_eyed }
				}
				scope:actor = {
					remove_trait = one_eyed
				}
			}
			
			if = {
				limit = {
					scope:wounded = yes
				}
				scope:actor = {
					remove_trait = wounded_3
					remove_trait = wounded_2
					remove_trait = wounded_1
				}
			}

			if = {
				limit = {
					scope:scarred = yes
				}
				scope:actor = {
					remove_trait = scarred
				}
			}				
		}
	}
	
	auto_accept = yes
	ai_targets = {
		ai_recipients = self
	}
	ai_frequency = 12
	ai_will_do = {
		base = 100
		
	}
}
