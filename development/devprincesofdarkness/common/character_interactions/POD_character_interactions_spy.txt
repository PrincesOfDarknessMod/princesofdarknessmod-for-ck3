POD_kueijin_spy_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	interface_priority = 88
	
	cooldown = { months = 6 }
	
	auto_accept = yes
	use_diplomatic_range = no
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { is_ai = no }
		scope:actor = { is_kueijin_trigger = yes }
		scope:actor = {
			OR = {
				has_perk = evaluate_perk
				has_perk = scrutinize_perk
				has_perk = permeate_perk
				has_perk = detach_limb_perk
				# has_perk = placate_the_earth_dragon_perk
				has_perk = pelesit_perk
			}
		}
		
		scope:recipient = {
			OR = {
				any_secret = {
					NOT = {is_known_by = scope:actor}
				}
				has_owned_scheme = yes
			}
		}
	}
	
	is_valid_showing_failures_only = {
		scope:recipient = { 
			NOT = {
				has_trait = torpor
			}
		}
	}

	options_heading = POD_kueijin_spy_interaction_options_heading

	# Level 1
	send_option = {
		is_shown = {
			scope:actor = {
				has_perk = evaluate_perk
			}
			scope:recipient = {
				any_secret = {
					NOT = {is_known_by = scope:actor}
				}
			}
		}	
		flag = evaluate
		localization = POD_kueijin_spy_interaction_evaluate
	}

	# Level 1
	send_option = {
		is_shown = {
			scope:actor = {
				has_perk = scrutinize_perk
			}
			scope:recipient = {
				has_owned_scheme = yes
			}
		}	
		flag = scrutinize
		localization = POD_kueijin_spy_interaction_scrutinize
	}

	# Level 1
	send_option = {
		is_shown = {
			scope:actor = {
				has_perk = permeate_perk
				NOT = { has_perk = placate_the_earth_dragon_perk }
			}
			scope:recipient = {
				any_secret = {
					NOT = {is_known_by = scope:actor}
				}
			}
		}	
		flag = permeate
		localization = POD_kueijin_spy_interaction_permeate
	}

	# Level 2
	send_option = {
		is_shown = {
			scope:actor = {
				has_perk = detach_limb_perk
			}
			scope:recipient = {
				has_owned_scheme = yes
			}
		}	
		flag = detach_limb
		localization = POD_kueijin_spy_interaction_detach_limb
	}

	# Level 1+3
	send_option = {
		is_shown = {
			scope:actor = {
				has_perk = permeate_perk
				has_perk = placate_the_earth_dragon_perk
			}
			scope:recipient = {
				any_secret = {
					NOT = {is_known_by = scope:actor}
				}
			}
		}	
		flag = permeate_plus
		localization = POD_kueijin_spy_interaction_permeate_plus
	}

	# Level 4
	send_option = {
		is_shown = {
			scope:actor = {
				has_perk = pelesit_perk
			}
			scope:recipient = {
				has_owned_scheme = yes
			}
		}	
		flag = pelesit
		localization = POD_kueijin_spy_interaction_pelesit
	}
	
	send_options_exclusive = yes

	on_accept = {
		scope:actor = { POD_willpower_rouse_check_effect = yes }
		if = {
			limit = {
				scope:evaluate = yes
			}
			duel = {
				skill = learning
				value = scope:recipient.learning
				70 = {
					desc = POD_kueijin_spy.duel.success
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					# Success 
					custom_tooltip = POD_kueijin_spy.duel.tt.evaluate
					scope:recipient = {
						random_secret = {
							reveal_to = scope:actor
						}
					}
				}
				30 = {
					desc = POD_kueijin_spy.duel.failure
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier 
					}
					# Failure
					custom_tooltip = POD_kueijin_spy.duel.tt.attacker_lost
				}
			}
			scope:actor = { POD_chi_decrease_effect = { COUNT = 1 } }
		}
		if = {
			limit = {
				scope:scrutinize = yes
			}
			duel = {
				skill = learning
				value = scope:recipient.learning
				70 = {
					desc = POD_kueijin_spy.duel.success
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					# Success 
					custom_tooltip = POD_kueijin_spy.duel.tt.scrutinize
					scope:recipient = {
						every_scheme = {
							expose_scheme = yes
						}
					}
				}
				30 = {
					desc = POD_kueijin_spy.duel.failure
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier 
					}
					# Failure
					custom_tooltip = POD_kueijin_spy.duel.tt.attacker_lost
				}
			}
			scope:actor = { POD_chi_decrease_effect = { COUNT = 1 } }
		}
		if = {
			limit = {
				scope:permeate = yes
			}
			duel = {
				skill = intrigue
				value = scope:recipient.intrigue
				70 = {
					desc = POD_kueijin_spy.duel.success
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					# Success 
					custom_tooltip = POD_kueijin_spy.duel.tt.permeate
					scope:recipient = {
						random_secret = {
							reveal_to = scope:actor
						}
					}
				}
				30 = {
					desc = POD_kueijin_spy.duel.failure
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier 
					}
					# Failure
					custom_tooltip = POD_kueijin_spy.duel.tt.attacker_lost
				}
			}
			scope:actor = { POD_chi_decrease_effect = { COUNT = 1 } }
		}	
		if = {
			limit = {
				scope:detach_limb = yes
			}
			duel = {
				skill = intrigue
				value = scope:recipient.intrigue
				60 = {
					desc = POD_kueijin_spy.duel.success
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					# Success 
					custom_tooltip = POD_kueijin_spy.duel.tt.detach_limb
					scope:recipient = {
						every_scheme = {
							add_scheme_progress = scheme_progress_loss
						}
					}
				}
				40 = {
					desc = POD_kueijin_spy.duel.failure
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier 
					}
					# Failure
					custom_tooltip = POD_kueijin_spy.duel.tt.attacker_lost
				}
			}
			scope:actor = { POD_chi_decrease_effect = { COUNT = 2 } }
		}	
		if = {
			limit = {
				scope:permeate_plus = yes
			}
			duel = {
				skill = intrigue
				value = scope:recipient.intrigue
				50 = {
					desc = POD_kueijin_spy.duel.success
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					# Success 
					custom_tooltip = POD_kueijin_spy.duel.tt.permeate_plus
					scope:recipient = {
						every_secret = {
							reveal_to = scope:actor
						}
					}
				}
				50 = {
					desc = POD_kueijin_spy.duel.failure
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier 
					}
					# Failure
					custom_tooltip = POD_kueijin_spy.duel.tt.attacker_lost
				}
			}
			scope:actor = {
				add_piety = minor_piety_loss
				POD_chi_decrease_effect = { COUNT = 4 } 
			}
		}	
		if = {
			limit = {
				scope:pelesit = yes
			}
			duel = {
				skill = intrigue
				value = scope:recipient.intrigue
				50 = {
					desc = POD_kueijin_spy.duel.success
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					# Success 
					custom_tooltip = POD_kueijin_spy.duel.tt.pelesit
					scope:recipient = {
						every_scheme = {
							end_scheme = yes
						}
					}
				}
				50 = {
					desc = POD_kueijin_spy.duel.failure
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier 
					}
					# Failure
					custom_tooltip = POD_kueijin_spy.duel.tt.attacker_lost
				}
			}
			scope:actor = {
				POD_losing_health_level_effect = { COUNT = 1 }
				POD_chi_decrease_effect = { COUNT = 4 } 
			}
		}
	}
	
	ai_frequency = 0

	ai_will_do = {
		base = 0
		modifier = {
			add = 10
			scope:recipient = liege 			
		}
		modifier = {
			add = 10
			scope:recipient = liege
			OR = {
				has_trait = ambitious 
				has_trait = fickle
				has_trait = callous
				has_trait = deceitful
				has_trait = craven
			}
		}
		modifier = {
			add = -10
			scope:recipient = liege
			OR = {
				has_trait = content
				has_trait = just
				has_trait = honest
				has_trait = lazy
				has_trait = humble
			}
		}
		modifier = {
			add = 10
			scope:recipient = { is_powerful_vassal = yes }
		}
		
		modifier = {
			add = 10
			scope:recipient = { is_powerful_vassal = yes }
			OR = {
				has_trait = arbitrary
				has_trait = callous
				has_trait = arrogant
				has_trait = deceitful
			}
		}
		modifier = {
			add = 25
			ai_diplomacy_stance = { target = scope:recipient  stance = threat }
		}
		modifier = {
		    factor = 2
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
		modifier = {
			factor = 0.5
			scope:recipient.intrigue > intrigue
		}
	}
}
