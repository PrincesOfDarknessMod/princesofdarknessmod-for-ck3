POD_activate_effigy_interaction = {
	category = interaction_category_mummy
	desc = POD_activate_effigy_interaction_desc
	options_heading = POD_activate_effigy_interaction_options_heading
	
	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { is_ai = no }
		scope:actor = { has_trait = mummy }
		scope:actor = { 
			any_character_artifact = {
				has_variable = active_effigy
				var:active_effigy = no
			}
		}
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				any_character_artifact = {
					has_variable = servitor_effigy
					NOT = { has_variable = active_effigy }
				}
			}
		}
		flag = POD_servitor_effigy
		localization = POD_servitor_effigy
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				any_character_artifact = {
					has_variable = overseer_effigy
					NOT = { has_variable = active_effigy }
				}
			}
		}
		flag = POD_overseer_effigy
		localization = POD_overseer_effigy
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				any_character_artifact = {
					has_variable = guard_effigy
					NOT = { has_variable = active_effigy }
				}
			}
		}
		flag = POD_guard_effigy
		localization = POD_guard_effigy
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				any_character_artifact = {
					has_variable = the_crawling_eye_effigy
					NOT = { has_variable = active_effigy }
				}
			}
		}
		flag = POD_the_crawling_eye_effigy
		localization = POD_the_crawling_eye_effigy
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				any_character_artifact = {
					has_variable = stalking_hound_of_typhon_effigy
					NOT = { has_variable = active_effigy }
				}
			}
		}
		flag = POD_stalking_hound_of_typhon_effigy
		localization = POD_stalking_hound_of_typhon_effigy
	}
	
	send_option = {
		is_shown = {
			always = no
		}
		flag = POD_creature_effigy
		localization = POD_servitor_effigy
	}
	
	send_option = {
		is_shown = {
			always = no
		}
		flag = POD_chattel_effigy
		localization = POD_servitor_effigy
	}

	on_accept = {
		scope:actor = {
			POD_hekau_cost_effect = { COUNT = 1 }
		}
		scope:recipient = {
			if = {
				limit = { scope:POD_servitor_effigy = yes }
				random_character_artifact = {
					limit = { 
						has_variable = servitor_effigy
						NOT = { has_variable = active_effigy }
					} 
					set_variable = {
						name = active_effigy
						value = yes
					}
				}
			}
			if = {
				limit = { scope:POD_overseer_effigy = yes }
				random_character_artifact = {
					limit = { 
						has_variable = overseer_effigy
						NOT = { has_variable = active_effigy }
					} 
					set_variable = {
						name = active_effigy
						value = yes
					}
				}
			}
			if = {
				limit = { scope:POD_guard_effigy = yes }
				random_character_artifact = {
					limit = { 
						has_variable = guard_effigy
						NOT = { has_variable = active_effigy }
					} 
					set_variable = {
						name = active_effigy
						value = yes
					}
				}
			}
			if = {
				limit = { scope:POD_the_crawling_eye_effigy = yes }
				random_character_artifact = {
					limit = { 
						has_variable = the_crawling_eye_effigy
						NOT = { has_variable = active_effigy }
					} 
					set_variable = {
						name = active_effigy
						value = yes
					}
				}
			}
			if = {
				limit = { scope:POD_stalking_hound_of_typhon_effigy = yes }
				random_character_artifact = {
					limit = { 
						has_variable = stalking_hound_of_typhon_effigy
						NOT = { has_variable = active_effigy }
					} 
					set_variable = {
						name = active_effigy
						value = yes
					}
				}
			}
			if = {
				limit = { scope:POD_creature_effigy = yes }
				random_character_artifact = {
					limit = { 
						OR = { 
							has_variable = simple_creature_effigy
							has_variable = lesser_creature_effigy
							has_variable = major_creature_effigy
							has_variable = superior_creature_effigy
							has_variable = princely_creature_effigy
							has_variable = viziers_beast_effigy
							has_variable = kingly_beast_effigy
							has_variable = divine_beast_effigy
						}
						NOT = { has_variable = active_effigy }
					} 
					set_variable = {
						name = active_effigy
						value = yes
					}
				}
			}
			if = {
				limit = { scope:POD_chattel_effigy = yes }
				random_character_artifact = {
					limit = { 
						OR = { 
							has_variable = lesser_chattel_effigy
							has_variable = complex_chattel_effigy
							has_variable = superior_chattel_effigy
							has_variable = princely_chattel_effigy
						}
						NOT = { has_variable = active_effigy }
					} 
					set_variable = {
						name = active_effigy
						value = yes
					}
				}
			}
		}
	}
	
	auto_accept = yes
	
	
	ai_will_do = {
		base = 0
	}
}

POD_deactivate_effigy_interaction = {
	category = interaction_category_mummy
	desc = POD_deactivate_effigy_interaction_desc
	
	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { is_ai = no }
		scope:actor = { has_trait = mummy }
		scope:actor = {
			any_character_artifact = { 
				has_variable = active_effigy
				var:active_effigy = yes
			} 
		}
	}

	on_accept = {
		scope:recipient = {
			custom_tooltip = POD_deactivate_effigy_interaction_desc
			every_character_artifact = {
				limit = { has_variable = active_effigy }
				set_variable = {
					name = active_effigy
					value = no
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_command_effigy_interaction = {
	category = interaction_category_mummy
	desc = POD_command_effigy_interaction_desc
	options_heading = POD_command_effigy_interaction_options_heading
	
	cooldown = { months = 6 }
	
	is_shown = {
		scope:actor = { is_ai = no }
		scope:actor = { has_trait = mummy }
		scope:actor = {
			any_character_artifact = { 
				has_variable = active_effigy
				var:active_effigy = yes
			} 
		}
	}
	
	send_option = {
		is_shown = {
			scope:actor = scope:recipient
		}
		flag = POD_command_effigy_scheme
		localization = POD_command_effigy_scheme
	}
	
	send_option = {
		is_shown = {
			scope:actor = scope:recipient
		}
		flag = POD_command_effigy_lifestyle
		localization = POD_command_effigy_lifestyle
	}
	
	send_option = {
		is_shown = {
			scope:actor = scope:recipient
		}
		flag = POD_command_effigy_gold
		localization = POD_command_effigy_gold
	}
	
	send_option = {
		is_shown = {
			scope:actor = scope:recipient
		}
		flag = POD_command_effigy_prestige
		localization = POD_command_effigy_prestige
	}
	
	send_option = {
		is_shown = {
			scope:actor = scope:recipient
		}
		flag = POD_command_effigy_piety
		localization = POD_command_effigy_piety
	}
	
	send_option = {
		is_shown = {
			scope:actor = scope:recipient
		}
		flag = POD_command_effigy_renown
		localization = POD_command_effigy_renown
	}
	
	send_option = {
		is_shown = {
			NOT = { scope:actor = scope:recipient }
			scope:actor = {
				any_character_artifact = {
					has_variable = the_crawling_eye_effigy
					has_variable = active_effigy
				}
			}
			scope:recipient = { any_secret = { NOT = { is_known_by = scope:actor } } }
		}
		flag = POD_command_effigy_secret
		localization = POD_command_effigy_secret
	}
	send_option = {
		is_shown = {
			NOT = { scope:actor = scope:recipient }
			scope:actor = {
				any_character_artifact = {
					has_variable = stalking_hound_of_typhon_effigy
					has_variable = active_effigy
				}
			}
			scope:recipient = { is_ruler = yes }
		}
		flag = POD_command_effigy_wound
		localization = POD_command_effigy_wound
	}

	on_accept = {
		scope:actor = {
			POD_hekau_cost_effect = { COUNT = 1 }
			POD_willpower_rouse_check_effect = yes
			if = {
				limit = {
					scope:POD_command_effigy_secret = yes
				}
				duel = {
					skill = intrigue
					target = scope:recipient
					70 = {
						desc = POD_command_effigy.success
						compare_modifier = {
							value = scope:duel_value
							multiplier = high_positive_duel_skill_multiplier
						}
						send_interface_toast = {
							title = POD_command_effigy.success
							left_icon = scope:actor
							right_icon = scope:recipient
							
							custom_tooltip = POD_command_effigy.the_crawling_eye.tt
						}
						hidden_effect = {
							every_character_artifact = {
								limit = { 
									has_variable = active_effigy 
									has_variable = the_crawling_eye_effigy
								}
								if = {
									limit = { scope:recipient = { any_secret = { NOT = { is_known_by = scope:actor } } } }
									scope:recipient = {
										random_secret = {
											reveal_to = scope:actor
										}
									}
								}
							}
						}
					}
					30 = {
						desc = POD_command_effigy.fail
						compare_modifier = {
							value = scope:duel_value
							multiplier = low_negative_duel_skill_multiplier 
						}
						send_interface_toast = {
							title = POD_command_effigy.fail
							left_icon = scope:actor
							right_icon = scope:recipient
							
							custom_tooltip = POD_command_effigy.fail_result
						}
					}
				}
			}
			if = {
				limit = {
					scope:POD_command_effigy_wound = yes
				}
				duel = {
					skill = martial
					target = scope:recipient
					50 = {
						desc = POD_command_effigy.success
						compare_modifier = {
							value = scope:duel_value
							multiplier = high_positive_duel_skill_multiplier
						}
						send_interface_toast = {
							title = POD_command_effigy.success
							left_icon = scope:actor
							right_icon = scope:recipient
							
							every_character_artifact = {
								limit = { 
									has_variable = active_effigy 
									has_variable = stalking_hound_of_typhon_effigy
								}
								scope:recipient = { 
									if = {
										limit = { any_knight = { has_any_nickname = no } }
										random_knight = {
											limit = { has_any_nickname = no }
											pod_death_effect = {
												DEATH_REASON = death_battle
												KILLER = scope:actor
											}
										}
									}
								}
							}
						}
					}
					50 = {
						desc = POD_command_effigy.fail
						compare_modifier = {
							value = scope:duel_value
							multiplier = low_negative_duel_skill_multiplier 
						}
						send_interface_toast = {
							title = POD_command_effigy.fail
							left_icon = scope:actor
							right_icon = scope:recipient
							
							custom_tooltip = POD_command_effigy.fail_result
						}
					}
				}
			}
		}
		scope:recipient = {
			if = {
				limit = { 
					any_character_artifact = { 
						has_variable = active_effigy 
						has_variable = overseer_effigy 
					} 
				}
				if = {
					limit = { scope:POD_command_effigy_scheme = yes }
					every_character_artifact = {
						limit = { has_variable = active_effigy }
						if = {
							limit = { has_variable = servitor_effigy }
							scope:recipient = { random_scheme = { add_scheme_progress = 1 } }
						}
						if = {
							limit = { has_variable = guard_effigy }
							scope:recipient = { random_scheme = { add_scheme_progress = 2 } }
						}
					}
				}
				if = {
					limit = { scope:POD_command_effigy_lifestyle = yes }
					every_character_artifact = {
						limit = { has_variable = active_effigy }
						if = {
							limit = { has_variable = servitor_effigy }
							scope:recipient = { add_focus_progress = standard_lifestyle_focus_progress }
						}
						if = {
							limit = { has_variable = guard_effigy }
							scope:recipient = { add_focus_progress = standard_lifestyle_focus_progress }
						}
					}
				}
				if = {
					limit = { scope:POD_command_effigy_gold = yes }
					every_character_artifact = {
						limit = { has_variable = active_effigy }
						if = {
							limit = { has_variable = servitor_effigy }
							scope:recipient = { add_gold = tiny_gold_value }
						}
						if = {
							limit = { has_variable = guard_effigy }
							scope:recipient = { add_gold = minor_gold_value }
						}
					}
				}
				if = {
					limit = { scope:POD_command_effigy_prestige = yes }
					every_character_artifact = {
						limit = { has_variable = active_effigy }
						if = {
							limit = { has_variable = servitor_effigy }
							scope:recipient = { add_prestige = miniscule_prestige_value }
						}
						if = {
							limit = { has_variable = guard_effigy }
							scope:recipient = { add_prestige = minor_prestige_value }
						}
					}
				}
				if = {
					limit = { scope:POD_command_effigy_piety = yes }
					every_character_artifact = {
						limit = { has_variable = active_effigy }
						if = {
							limit = { has_variable = servitor_effigy }
							scope:recipient = { add_piety = miniscule_piety_value }
						}
						if = {
							limit = { has_variable = guard_effigy }
							scope:recipient = { add_piety = minor_piety_value }

					}

							scope:recipient = { add_dynasty_prestige = miniscule_dynasty_prestige_value }
						}
						if = {
							limit = { has_variable = guard_effigy }
							scope:recipient = { add_dynasty_prestige = minor_dynasty_prestige_value }

					}

					limit = { scope:POD_command_effigy_lifestyle = yes }
					random_character_artifact = {
						limit = { has_variable = active_effigy }
						if = {
							limit = { has_variable = servitor_effigy }
							scope:recipient = { add_focus_progress = standard_lifestyle_focus_progress }
						}
						if = {
							limit = { has_variable = guard_effigy }
							scope:recipient = { add_focus_progress = standard_lifestyle_focus_progress }
						}
					}
				}
				if = {
					limit = { scope:POD_command_effigy_gold = yes }
					random_character_artifact = {
						limit = { has_variable = active_effigy }
						if = {
							limit = { has_variable = servitor_effigy }
							scope:recipient = { add_gold = tiny_gold_value }
						}
						if = {
							limit = { has_variable = guard_effigy }
							scope:recipient = { add_gold = minor_gold_value }
						}
					}
				}
				if = {
					limit = { scope:POD_command_effigy_prestige = yes }
					random_character_artifact = {
						limit = { has_variable = active_effigy }
						if = {
							limit = { has_variable = servitor_effigy }
							scope:recipient = { add_prestige = miniscule_prestige_value }
						}
						if = {
							limit = { has_variable = guard_effigy }
							scope:recipient = { add_prestige = minor_prestige_value }
						}
					}
				}
				if = {
					limit = { scope:POD_command_effigy_piety = yes }
					random_character_artifact = {
						limit = { has_variable = active_effigy }
						if = {
							limit = { has_variable = servitor_effigy }
							scope:recipient = { add_piety = miniscule_piety_value }
						}
						if = {
							limit = { has_variable = guard_effigy }
							scope:recipient = { add_piety = minor_piety_value }

					}

				if = {
					limit = { scope:POD_command_effigy_renown = yes }
					random_character_artifact = {
						limit = { has_variable = active_effigy }
						if = {
							limit = { has_variable = servitor_effigy }
							scope:recipient = { add_dynasty_prestige = miniscule_dynasty_prestige_value }
						}
						if = {
							limit = { has_variable = guard_effigy }
							scope:recipient = { add_dynasty_prestige = minor_dynasty_prestige_value }
						}
					}
				}
			}

	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
