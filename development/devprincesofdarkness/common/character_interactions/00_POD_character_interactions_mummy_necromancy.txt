POD_fertility_of_osiris_interaction = {
	category = interaction_category_mummy
	desc = POD_fertility_of_osiris_interaction_desc
	options_heading = POD_fertility_of_osiris_interaction_options_heading
	
	is_shown = {
		scope:actor = { is_ai = no }
		scope:actor = { has_trait = mummy }
	}
	
	is_valid_showing_failures_only = {
		scope:recipient = {
			is_female = yes
			is_pregnant = no
			custom_description = {
				text = pod_cannot_pregnant
				is_visibly_fertile = yes
			}
			custom_description = {
				text = has_any_dead_spouse
				any_spouse = {
					even_if_dead = yes
					is_alive = no
					is_female = no
				}
			}
		}
	}
	
	send_option = {
		flag = spouse_1
		localization = POD_fertility_of_osiris_spouse_1
	}
	
	send_option = {
		is_shown = {
			scope:recipient = {
				any_spouse = {
					even_if_dead = yes
					is_alive = no
					is_female = no
					count >= 2
				}
			}
		}
		flag = spouse_2
		localization = POD_fertility_of_osiris_spouse_2
	}
	
	send_option = {
		is_shown = {
			scope:recipient = {
				any_spouse = {
					even_if_dead = yes
					is_alive = no
					is_female = no
					count >= 3
				}
			}
		}
		flag = spouse_3
		localization = POD_fertility_of_osiris_spouse_3
	}
	
	send_option = {
		is_shown = {
			scope:recipient = {
				any_spouse = {
					even_if_dead = yes
					is_alive = no
					is_female = no
					count >= 4
				}
			}
		}
		flag = spouse_4
		localization = POD_fertility_of_osiris_spouse_4
	}
	
	send_option = {
		is_shown = {
			scope:recipient = {
				any_spouse = {
					even_if_dead = yes
					is_alive = no
					is_female = no
					count >= 5
				}
			}
		}
		flag = spouse_5
		localization = POD_fertility_of_osiris_spouse_5
	}

	on_accept = {
		scope:actor = { POD_hekau_cost_effect = { COUNT = 2 } }
		scope:recipient = {
			every_spouse = {
				even_if_dead = yes
				limit = {
					is_alive = no
					is_female = no
				}
				add_to_list = dead_spouse
			}
		}

		if = {
			limit = { scope:spouse_1 = yes }
			ordered_in_list = {
				list = dead_spouse
				position = 0
				order_by = age
				save_scope_as = target_spouse
			}
		}
		if = {
			limit = { scope:spouse_2 = yes }
			ordered_in_list = {
				list = dead_spouse
				position = 1
				order_by = age
				save_scope_as = target_spouse
			}
		}
		if = {
			limit = { scope:spouse_3 = yes }
			ordered_in_list = {
				list = dead_spouse
				position = 2
				order_by = age
				save_scope_as = target_spouse
			}
		}
		if = {
			limit = { scope:spouse_4 = yes }
			ordered_in_list = {
				list = dead_spouse
				position = 3
				order_by = age
				save_scope_as = target_spouse
			}
		}
		if = {
			limit = { scope:spouse_5 = yes }
			ordered_in_list = {
				list = dead_spouse
				position = 4
				order_by = age
				save_scope_as = target_spouse
			}
		}
		show_as_tooltip = {
			scope:recipient = {
				make_pregnant = { father = scope:target_spouse }
			}
		}
		hidden_effect = {
			scope:target_spouse =  {
				save_scope_as = cloned_character
				
				create_character = {
					save_scope_as = clone
					age = scope:cloned_character.age
					faith = scope:cloned_character.faith
					gender = scope:cloned_character
					random_traits = no
					dynasty = none
					employer = scope:actor
				}
				
				scope:clone = {
					copy_traits = scope:cloned_character
					copy_inheritable_appearance_from = scope:cloned_character
					POD_copy_base_skills_effect = {
						CLONED_CHAR = scope:cloned_character
						CLONE = scope:clone
					}
				}
			
				scope:recipient = {
					make_pregnant = { father = scope:clone }
					scope:clone = { silent_disappearance_effect = yes }
					set_variable = {
						name = fertility_of_osiris
						value = scope:cloned_character
						days = 730
					}
				}
			}
		}
	}
	
	auto_accept = yes
	
	
	ai_will_do = {
		base = 0
	}
}
