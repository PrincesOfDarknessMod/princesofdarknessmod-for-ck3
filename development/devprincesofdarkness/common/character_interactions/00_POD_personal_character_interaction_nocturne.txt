POD_nocturne_on_personal_interaction = {
	category = interaction_category_vampire
	#common_interaction = yes
	#interface_priority = 90
    desc = POD_nocturne_on_personal_interaction_desc
	
	ai_frequency = 20

	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { is_adult = yes }
		scope:actor = { is_vampire_trigger = yes  }
		scope:actor = { 
		    OR = {
			    has_perk = nocturne_perk
				has_trait = obtenebrationadvanced
			}
		}
		scope:actor = { NOT = { has_character_modifier = nocturne_power_modifier } }
		scope:actor = { NOT = { has_trait = kueijin } }
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
		    NOT = { has_character_modifier = bloodhunger5_modifier }
		}
	}

	on_accept = {
	   custom_tooltip = POD_nocturne_on_personal_interaction.tt
	   scope:actor = {
	       add_character_modifier = nocturne_power_modifier
		   POD_hunger_rouse_check_effect = yes
		   if = {
		       limit = {
			       has_perk = nocturne_perk
				   NOT = { has_perk = the_darkness_within_perk }
				   NOT = { has_trait = obtenebrationadvanced }
			   }
			   POD_small_risk_of_increasing_masquerade_exposure = yes
		   }
		   else_if = {
		       limit = {
			       has_perk = nocturne_perk
				   OR = {
                       has_perk = the_darkness_within_perk
					   has_trait = obtenebrationadvanced
				   }
				   NOT = { has_perk = devour_all_hope_perk }
			   }
			   POD_moderate_risk_of_increasing_masquerade_exposure = yes
		   }
		   else_if = {
		       limit = {
			       has_perk = devour_all_hope_perk
			   }
			   POD_hunger_rouse_check_effect = yes
			   POD_very_high_risk_of_increasing_masquerade_exposure = yes
		   }
	   }
	}
	
	ai_targets = {
		ai_recipients = self
	}
	
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 45
			is_at_war = yes
			piety > 450
		}
		
		modifier = {
			add = 100
			is_commanding_army = yes
			piety > 250
		}
	}  	
}
