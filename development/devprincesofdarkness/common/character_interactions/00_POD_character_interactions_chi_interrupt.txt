POD_chi_interrupt_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	interface_priority = 88
	desc = POD_chi_interrupt_interaction_desc
	
	cooldown = { months = 1 }	
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { is_adult = yes }
		scope:recipient = { is_adult = yes }
		scope:actor = { has_trait = kueijin }
		scope:actor = { has_perk = chi_interrupt_perk }
		scope:recipient = {
			NOT = {
				has_trait = torpor
			}
		}
		scope:actor = { is_ai = no }
	}
	
	on_accept = {
		scope:actor = { POD_chi_decrease_effect = yes }
		
		scope:recipient = {
			add_character_flag = {
				flag = is_being_tortured
				months = 6
			}
			add_character_modifier = {
				modifier = risky_wound_treatment_failure_modifier
				months = 2
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
