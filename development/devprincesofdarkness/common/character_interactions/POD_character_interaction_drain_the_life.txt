POD_drain_the_life_interaction = {
	category = interaction_category_vampire
	#common_interaction = yes
	#interface_priority = 90
    desc = POD_drain_the_life_interaction_desc
	
	ai_frequency = 360

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { is_adult = yes }
		scope:actor = { is_vampire_trigger = yes  }
		scope:actor = { 
		    OR = {
			    has_perk = drain_the_life_perk
				has_trait = obtenebrationadvanced
			}
		}
		scope:actor = { NOT = { has_trait = kueijin } }
		scope:recipient = { is_landed = yes }
		NOT = { scope:recipient.liege = scope:actor }
		scope:recipient.highest_held_title_tier > tier_barony
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
		    OR = {
		        NOT = { has_character_modifier = bloodhunger4_modifier }
				NOT = { has_character_modifier = bloodhunger5_modifier }
			}
		}
	}

	on_accept = {
	   custom_tooltip = POD_drain_the_life_interaction.tt
	   scope:actor = {
	       trigger_event = POD_drain_the_life.001
	   }
    }
	
	ai_targets = {
		ai_recipients = neighboring_rulers
	}
	
	auto_accept = yes
	
	cooldown = { years = 5 }
	
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 45
			scope:actor = {
			    is_at_war_with = scope:recipient
			}
			piety > 300
		}
		modifier = {
		    add = 10
			scope:actor = {
			    has_relation_rival = scope:recipient
			}
		}
	}  	
}
