POD_set_thrall_interaction = {
	category = interaction_category_demon
	common_interaction = yes
	interface_priority = 100
	desc = POD_set_thrall_interaction_desc

	ai_min_reply_days = 4
	ai_max_reply_days = 9
	
	cost = { prestige = minor_prestige_value }
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			is_landed = yes
			has_perk = enthrall_perk
		}
		scope:recipient = { 
			is_a_thrall = no 
			NOT = { has_trait = demon }
			NOT = { has_trait = effigy }
			NOT = { has_trait = fae } #Changeling when?
			NOT = { has_trait = wraith }
		}
		POD_is_in_same_plane_trigger = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			NOT =  { is_at_war_with = scope:recipient }
		}
		scope:recipient = {
			OR = {
				is_courtier_of = scope:actor
				is_vassal_of = scope:actor
			}
			is_diplomatically_available = yes
			NOT = { has_strong_hook = scope:actor }
		}
	}

	#Use hook
	send_option = {
		is_valid = {
			scope:actor = {
				has_usable_hook = scope:recipient
			}
		}
		flag = hook
		localization = SCHEME_HOOK
	}
	should_use_extra_icon = {
		scope:actor = { has_usable_hook = scope:recipient }
	}
	extra_icon = "gfx/interface/icons/character_interactions/hook_icon.dds"

	send_options_exclusive = no

	on_accept = {
		scope:actor = {
			POD_resolve_decrease_effect = yes
		}
		scope:recipient = {
			add_trait = infernalist
			set_relation_demonic_master = scope:actor
				
			if = {
				limit = {
					has_trait = supehunter 
				}
				scope:actor = { save_scope_as = vampiresire }
				POD_hunterembrace_effect = yes
			}
		}
	}
	
	ai_targets = {
		ai_recipients = vassals
	}
	
	ai_frequency = 72
	
	ai_potential = {
		primary_title.tier > tier_county
		prestige > minor_prestige_value
	}
	
	ai_accept = {
		base = 0 # Try to make it 0 for most interactions

		modifier = {
			scope:hook = yes
			add = 50
			desc = SCHEME_WEAK_HOOK_USED
		}
		
		opinion_modifier = { # Opinion Factor
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.5
			desc = AI_SIMPLE_OPINION_REASON
		}
		
		modifier = {
			add = {
				value = ai_zeal
				divide = 2
				multiply = -1
			}
			ai_zeal > 0
			desc = TAKE_THE_VOWS_ZEALOUS
		}

		modifier = {
			add = -50
			has_relation_rival = scope:recipient
		}
		modifier = {
			add = -80
			has_relation_nemesis = scope:recipient
		}
	}
	
	ai_will_do = {
		base = 100
		
		modifier = { # The AI will only use a Hook if they couldn't otherwise do this
			scope:hook = yes
			add = -1
		}
	}
}

POD_remove_thrall_interaction = {
	category = interaction_category_demon
	desc = POD_remove_thrall_interaction_desc
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_trait = demon }
		scope:recipient = { has_relation_demonic_master = scope:actor }
	}

	on_accept = {
		scope:recipient = {
			remove_relation_demonic_master = scope:actor
			POD_remove_resolve_modifiers_effect = yes
			if = {
				limit = {
					exists = house
					house = scope:actor.house
					can_execute_decision = create_cadet_branch_decision
				}
				execute_decision = create_cadet_branch_decision
				show_as_tooltip = { create_cadet_branch = yes }
			}
			if = {
				limit = {
					has_trait = mortal
					has_game_rule = world_of_monsters
					can_hold_title_as_mortal_trigger = no
				}
				depose = yes
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_designate_vessel_interaction = {
	category = interaction_category_demon
	common_interaction = yes
	interface_priority = 120
	desc = POD_designate_vessel_interaction_desc

	is_shown = {
		NOT = { scope:actor = scope:recipient }

		NOT = { scope:actor = { any_child = { is_alive = yes } } }
		scope:recipient = { can_be_demonic_possesed = yes }
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			has_relation_demonic_master = scope:actor
			NOT = { is_child_of = scope:actor }
		}
	}

	on_accept = {
		scope:recipient = { 
			POD_set_parent_effect = { PARENT = scope:actor }
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_revelation_interaction = {
	category = interaction_category_demon
	common_interaction = yes
	interface_priority = 120
	desc = POD_revelation_interaction_desc
	
	cooldown = { months = 1 }
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_trait = demon }
		scope:actor = { has_perk = revelation_perk }
		scope:recipient = { 
			OR = { 
				POD_can_be_vamphunted_scheme_trigger = yes 
				is_a_thrall = yes
			}
		}
	}

	is_valid_showing_failures_only = {
		custom_description = {
            object = scope:actor
            text = "I_must_be_at_the_same_location"
			scope:actor = { exists = location }
			scope:recipient = { exists = location }
            scope:actor = { location = scope:recipient.location }
        }
		scope:actor = { 
			trigger_if = {
				limit = {
					NOT = { has_character_modifier = POD_shapeshift_form_apocalyptic_angelic_modifier }
					NOT = { has_character_modifier = POD_shapeshift_form_apocalyptic_damned_modifier }
				}
				has_character_modifier = POD_shapeshift_form_apocalyptic_angelic_modifier
			}
			trigger_if = {
				limit = {
					NOT = { has_character_modifier = POD_shapeshift_form_apocalyptic_angelic_modifier }
					NOT = { has_character_modifier = POD_shapeshift_form_apocalyptic_damned_modifier }
				}
				has_character_modifier = POD_shapeshift_form_apocalyptic_damned_modifier
			}
		}
		scope:recipient = { piety > 0 }
	}

	on_accept = {
		random_list = {
			10 = {
				scope:actor = { POD_resolve_increase_multiple_effect = { COUNT = 4 } }
				scope:recipient = { 
					add_stress = massive_stress_impact_gain 
					add_piety = massive_piety_loss
				}
			}
			20 = {
				scope:actor = { POD_resolve_increase_multiple_effect = { COUNT = 3 } }
				scope:recipient = { 
					add_stress = major_stress_impact_gain 
					add_piety = major_piety_loss
				}
			}
			30 = {
				scope:actor = { POD_resolve_increase_multiple_effect = { COUNT = 2 } }
				scope:recipient = { 
					add_stress = medium_stress_impact_gain 
					add_piety = medium_piety_loss
				}
			}
			40 = {
				scope:actor = { POD_resolve_increase_multiple_effect = { COUNT = 1 } }
				scope:recipient = { 
					add_stress = minor_stress_impact_gain 
					add_piety = minor_piety_loss
				}
			}
		}
		scope:recipient = { 
			if = { 
				limit = { has_trait = mortal }
				add_trait = POD_knows_too_much 
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
