
POD_disciplines_interaction = {
	category = interaction_category_vampire
	#common_interaction = yes
	#interface_priority = 83
	desc = POD_disciplines_interaction_desc
	icon = icon_culture

	ai_frequency = 0

	is_shown = {
		scope:actor = { is_ai = no }		
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		scope:actor = { 
			OR = { 
				has_POD_vamp_blood_trigger = yes
				has_trait = risen
			}
		}
		scope:recipient = { has_POD_vamp_blood_trigger = yes }
	}
	
	cooldown = { years = 5 } 

	is_valid_showing_failures_only = {
	custom_description = {
			object = scope:actor
			text = "He_must_be_my_friend"
		scope:recipient = { 
			OR = {
		has_relation_lover = scope:actor
		has_relation_friend = scope:actor
		has_relation_best_friend = scope:actor
		has_relation_soulmate = scope:actor
			}
		}
		}
		scope:actor = {
			NOT = { is_at_war_with = scope:recipient }
			}		
	}

	on_accept = {
		scope:actor = {
			custom_tooltip = POD_disciplines_interaction_accept
		}
		hidden_effect = {
			scope:actor = {
				trigger_event = POD_mentoring_disciplines.0001
		}
	}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_blink_interaction = {
	category = interaction_category_vampire
	#common_interaction = yes
	#interface_priority = 83
	desc = POD_blink_interaction_desc
	icon = icon_combat

	ai_frequency = 0

	is_shown = {
		scope:actor = { is_ai = no }		
		scope:actor = scope:recipient 
		scope:actor = { has_perk = blink_perk }
	}
	
	cooldown = { years = 5 } 

	is_valid_showing_failures_only = {		
	}

	on_accept = {
		scope:actor = {
			add_character_modifier = {
				modifier = POD_blink_modifier
				years = 1
			}
		if = {
			limit = { has_trait = risen }
			POD_pathos_decrease_multiple_effect = { COUNT = 1 }
		}
		else = {
			POD_hunger_rouse_check_effect = yes
		}
		if = {
			limit = {
				NOT = { has_perk = lightning_strike_perk }
			}
			add_character_modifier = {
				modifier = POD_drained_modifier
				years = 5
			}
		}
		if = {
			limit = {
				NOT = { has_perk = split_second_perk }
			}
		add_stress = major_stress_gain	
		}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
