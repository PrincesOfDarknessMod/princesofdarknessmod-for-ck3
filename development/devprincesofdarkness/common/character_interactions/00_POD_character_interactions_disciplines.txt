POD_blink_interaction = {
	category = interaction_category_vampire
	#common_interaction = yes
	#interface_priority = 83
	desc = POD_blink_interaction_desc
	icon = icon_combat

	ai_frequency = 0

	is_shown = {
		scope:actor = { is_ai = no }		
		scope:actor = scope:recipient 
		scope:actor = { has_perk = blink_perk }
	}
	
	cooldown = { years = 5 } 

	is_valid_showing_failures_only = {		
	}

	on_accept = {
		scope:actor = {
			add_character_modifier = {
				modifier = POD_blink_modifier
				years = 1
			}
		if = {
			limit = { has_trait = risen }
			POD_pathos_decrease_multiple_effect = { COUNT = 1 }
		}
		else = {
			POD_hunger_rouse_check_effect = yes
		}
		if = {
			limit = {
				NOT = { has_perk = lightning_strike_perk }
			}
			add_character_modifier = {
				modifier = POD_drained_modifier
				years = 5
			}
		}
		if = {
			limit = {
				NOT = { has_perk = split_second_perk }
			}
		add_stress = major_stress_gain	
		}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
