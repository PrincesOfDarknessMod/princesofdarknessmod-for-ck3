POD_awake_spirit_interaction = {
	category = interaction_category_vampire
	#common_interaction = yes
	#interface_priority = 88
	desc = POD_awake_spirit_interaction_desc
	
	cooldown = { months = 1 }	
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { is_adult = yes }
		scope:recipient = { is_adult = yes }
		scope:actor = { has_trait = kueijin }
		scope:actor = { has_perk = blood_awakening_perk }
		scope:recipient = { POD_is_wraith_trigger = yes }
		scope:actor = { is_ai = no }
	}
	
	on_accept = {
		scope:actor = { POD_yangchi_decrease_effect = { COUNT = 1 }	 }
		scope:recipient = {
			remove_trait = incapable	
		}
	}
	
	#ai_potential = {
	#	always = no
	#}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
