POD_awake_spirit_interaction = {
	category = interaction_category_wraith
	icon = POD_command_wraith_interaction
	desc = POD_awake_spirit_interaction_desc
	
	cooldown = { months = 1 }	
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { is_ai = no }
		scope:actor = { 
			OR = {
				has_trait = kueijin 
				has_trait = dhampyr
			}
		}
		scope:actor = { has_perk = blood_awakening_perk }
		scope:recipient = { has_trait = frozen }
		scope:recipient = { POD_is_wraith_trigger = yes }
		POD_is_in_same_plane_trigger = yes
	}
	
	is_valid_showing_failures_only = {
		scope:recipient = { has_relation_necromancer = scope:actor }
	}
	
	on_accept = {
		scope:actor = { POD_yangchi_decrease_effect = { COUNT = 1 }	 }
		scope:recipient = {
			remove_trait = frozen	
		}
	}
	
	#ai_potential = {
	#	always = no
	#}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
