knighting_interaction = {
	category = interaction_category_friendly
	common_interaction = yes
	interface_priority = 83
	scheme = knighting
	
	ai_frequency = 0

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		scope:actor = { has_trait = supehunter  }
		scope:recipient = { NOT = { has_trait = supehunter } }
		scope:actor = { is_ai = no }
		scope:recipient = { has_trait = mortal }
		scope:actor = {
		OR = {	
		dynasty = dynasty:13000002
		dynasty = dynasty:13000006 
		dynasty = dynasty:13000008
		}
		}
	}

is_valid_showing_failures_only = {
		custom_description = {
			subject = scope:actor
			text =  i_can_only_hunterize_one_at_a_time
			trigger_if = {
			scope:actor = {
				NOT = {
					any_scheme = {
						scheme_type = knighting
					}
				}
			}
			}
		}
		custom_description = {
			object = scope:actor
			text = "I_must_be_at_the_same_location"
			scope:actor = { location = scope:recipient.location }
		}
		scope:recipient = { is_married = no }
		custom_description = {
			object = scope:actor
			text = "I_must_be_older"
			scope:actor = {
				age > scope:recipient.age
			}
		}			
		
		
		scope:actor = {
			can_start_scheme = {
				type = knighting
				target = scope:recipient
			}
		}
	}

	desc = {
		triggered_desc = {
			trigger = {
				scope:actor = {
					NOT = { is_at_war_with = scope:recipient }
					can_start_scheme = {
						type = knighting
						target = scope:recipient
					}

				}
			}
			desc = scheme_interaction_tt_knighting_approved
		}

	}

	on_accept = {
		hidden_effect = {
			scope:actor = {
				start_scheme = {
					type = knighting
					target = scope:recipient
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

inquihunt_interaction = {
	category = interaction_category_hostile
	common_interaction = yes
	interface_priority = 82
	scheme = inquihunt

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		scope:actor = {
		OR = {
			has_trait = supehunter
			AND = {
			dynasty = dynasty:13000007
			faith = faith:calomenasforsaken
			}
		}
		}
		scope:recipient = { is_vampire_trigger = yes }
		scope:actor = { is_ai = no }
		scope:recipient = { is_ai = yes }
		scope:recipient = { 
			NOT = {
				has_trait = torpor
			}
		}
		NAND = {
			has_game_rule = no_players_multiplayer_murder_schemes
			scope:actor = {
				is_ai = no
			}
			scope:recipient = {
				is_ai = no
			}
		}

		NAND = {
			has_game_rule = no_player_families_multiplayer_murder_schemes
			scope:actor = {
				is_ai = no
			}
			scope:recipient = {
				OR = {
					is_ai = no
					any_close_family_member = {
						AND = {
							is_ai = no
							NOT = { this = scope:actor }
						}
					}
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		custom_description = {
			object = scope:actor
			text = "I_can_only_plot_one_inquihunt_at_time"
			scope:actor = {
				NOT = {
					any_scheme = {
						scheme_type = inquihunt
					}
				}
			}
		}
			
		scope:actor = {
			can_start_scheme = {
				type = inquihunt
				target = scope:recipient
			}
		}
	}

	desc = {
		triggered_desc = {
			trigger = {
				scope:actor = {
					NOT = { is_at_war_with = scope:recipient }
					can_start_scheme = {
						type = inquihunt
						target = scope:recipient
					}

				}
			}
			desc = scheme_interaction_tt_inquihunt_approved
		}

	}

	on_accept = {
		hidden_effect = {
			scope:actor = {
				start_scheme = {
					type = inquihunt
					target = scope:recipient
				}
			}
		}
	}
	
	auto_accept = yes
}
