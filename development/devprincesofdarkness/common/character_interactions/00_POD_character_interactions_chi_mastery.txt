POD_chi_mastery_interaction = {
	category = interaction_category_vampire
	desc = POD_chi_mastery_interaction_desc

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { 
			OR = {
				has_trait = kueijin 
				has_trait = dhampyr
			}
		}
		scope:actor = { has_perk = chi_mastery_perk }
		scope:recipient = { POD_is_clone_trigger = no }
		POD_is_in_same_plane_trigger = yes
	}

	is_valid_showing_failures_only = {
		scope:recipient = { 
			can_be_killed_physically_trigger = yes
			NOT = { has_trait = torpor }
		}
		scope:actor = {
			wall_too_strong_trigger = no
		}
	}
	
	send_option = {
		is_shown = {
			NOR = {
				scope:actor = { 
					has_hook_of_type = {
						target = scope:recipient
						type = strong_soul_shackles_hook
					}
				}
				scope:recipient = { is_imprisoned_by = scope:actor }
				scope:recipient = { has_relation_necromancer = scope:actor }
			}
		}
		is_valid = {
			scope:actor = {
				can_start_scheme = {
					type = PODchi_mastery
					target = scope:recipient
				}
			}
		}
		flag = scheme_attack
		localization = POD_interaction_scheme_attack
	}
	
	send_option = {
		is_shown = {
			NOR = {
				scope:actor = { 
					has_hook_of_type = {
						target = scope:recipient
						type = strong_soul_shackles_hook
					}
				}
				scope:recipient = { is_imprisoned_by = scope:actor }
				scope:recipient = { has_relation_necromancer = scope:actor }
			}
		}
		is_valid = {
			custom_description = {
				object = scope:actor
				text = "I_must_be_at_the_same_location"
				scope:actor = { location = scope:recipient.location }
			}
		}
		flag = duel_attack
		localization = POD_interaction_duel_attack
	}
	
	send_option = {
		is_valid = {
			OR = {
				scope:actor = { 
					has_hook_of_type = {
						target = scope:recipient
						type = strong_soul_shackles_hook
					}
				}
				scope:recipient = { is_imprisoned_by = scope:actor }
				scope:recipient = { has_relation_necromancer = scope:actor }
			}
		}
		flag = automatic
		localization = POD_interaction_automatic
	}
	
	on_accept = {
		if = {
			limit = {
				scope:scheme_attack = yes
			}
			scope:actor = {
				start_scheme = {
					type = PODchi_mastery
					target = scope:recipient
				}
			}
		}
	
		if = {
			limit = {
				scope:duel_attack = yes
			}
			scope:actor = {
				configure_start_single_combat_effect = {
					SC_INITIATOR = scope:actor
					SC_ATTACKER = scope:actor
					SC_DEFENDER = scope:recipient
					FATALITY = no
					FIXED = no
					LOCALE = throne_room
					OUTPUT_EVENT = kueijin_scheme.400
					INVALIDATION_EVENT = tribal.1022
				}
			}
		}
		
		if = {
			limit = {
				scope:automatic = yes
			}
			scope:actor = { trigger_event = kueijin_scheme.400 }
		}
	}
	
	auto_accept = yes
	
	ai_frequency = 360
	
	ai_targets = {
		ai_recipients = neighboring_rulers
	}
	
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 45
			scope:actor = {
			    is_at_war_with = scope:recipient
			}
		}
		modifier = {
		    add = 10
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
	}
}
