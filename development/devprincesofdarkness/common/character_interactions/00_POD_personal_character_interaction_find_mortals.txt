# POD Personal Character Interactions
# Find Mortals

POD_find_mortals_interaction = {
	category = interaction_category_vampire
	interface_priority = 96
	#icon = "gfx/interface/icons/character_interactions/blood_surge_interaction.dds"
	desc = POD_find_mortals_interaction_desc
	options_heading = POD_find_mortals_interaction_options_heading

	send_option = {
		flag = POD_find_mortals_peasants
		localization = POD_find_mortals_interaction_peasants
	} 

	send_option = {
		flag = POD_find_mortals_intrigue
		localization = POD_find_mortals_interaction_intrigue
	} 
	
	send_option = {
		flag = POD_find_mortals_prowess
		localization = POD_find_mortals_interaction_prowess
	} 

	send_option = {
		flag = POD_find_mortals_merchants
		localization = POD_find_mortals_interaction_merchants
	}

	send_option = {
		flag = POD_find_mortals_clergy
		localization = POD_find_mortals_interaction_clergy
	}

	send_option = {
		is_shown = {
			scope:actor = {
				OR = {
					has_perk = scry_the_soul_perk
					has_trait = auspexadvanced
				}
				NOT = { has_character_modifier = bloodhunger5_modifier }	
			}
		}	
		flag = POD_find_mortals_choleric 
		localization = POD_find_mortals_interaction_choleric
	}	
	
	send_option = {
		is_shown = {
			scope:actor = {
				OR = {
					has_perk = scry_the_soul_perk
					has_trait = auspexadvanced
				}
				NOT = { has_character_modifier = bloodhunger5_modifier }	
			}
		}	
		flag = POD_find_mortals_melancholic 
		localization = POD_find_mortals_interaction_melancholic
	}	

	send_option = {
		is_shown = {
			scope:actor = {
				OR = {
					has_perk = scry_the_soul_perk
					has_trait = auspexadvanced
				}
				NOT = { has_character_modifier = bloodhunger5_modifier }	
			}
		}	
		flag = POD_find_mortals_phlegmatic
		localization = POD_find_mortals_interaction_phlegmatic
	}		
	
	send_option = {
		is_shown = {
			scope:actor = {
				OR = {
					has_perk = scry_the_soul_perk
					has_trait = auspexadvanced
				}
				NOT = { has_character_modifier = bloodhunger5_modifier }	
			}
		}		
		flag = POD_find_mortals_sanguine
		localization = POD_find_mortals_interaction_sanguine
	}	

	send_options_exclusive = yes
	
	auto_accept = yes
	ai_frequency = 0
	use_diplomatic_range = no
	
	cooldown = { months = 3 }
	
	is_shown = {
		scope:actor = scope:recipient 
		scope:recipient = { is_adult = yes }
		scope:actor = { is_vampire_trigger = yes  }
		scope:actor = { is_ai = no }
	}
	
	is_valid_showing_failures_only = {
	}	

	on_accept = {
		scope:actor = {
			if = {
				limit = {
					scope:POD_find_mortals_choleric = yes
				}
				custom_tooltip = POD_find_mortals_interaction_choleric.tt
				scope:actor = {
					POD_hunger_rouse_check_effect = yes
				}
			}
			if = {
				limit = {
					scope:POD_find_mortals_melancholic = yes
				}
				custom_tooltip = POD_find_mortals_interaction_melancholic.tt
				scope:actor = {
					POD_hunger_rouse_check_effect = yes
				}
			}		
			if = {
				limit = {
					scope:POD_find_mortals_phlegmatic = yes
				}
				custom_tooltip = POD_find_mortals_interaction_phlegmatic.tt
				scope:actor = {
					POD_hunger_rouse_check_effect = yes
				}
			}		
			if = {
				limit = {
					scope:POD_find_mortals_sanguine = yes
				}
				custom_tooltip = POD_find_mortals_interaction_sanguine.tt
				scope:actor = {
					POD_hunger_rouse_check_effect = yes
				}
			}			
			if = {
				limit = {
					scope:POD_find_mortals_prowess = yes
				}
				scope:actor = {
					add_prestige = minor_prestige_loss
				}
			}		
			if = {
				limit = {
					scope:POD_find_mortals_merchants = yes
				}
				scope:actor = {
					remove_short_term_gold = minor_gold_value
				}
			}
			if = {
				limit = {
					scope:POD_find_mortals_clergy = yes
				}
				scope:actor = {
					add_piety = minor_piety_loss
				}
			}	

			if = {
				limit = {
					scope:POD_find_mortals_intrigue = yes
				}
				scope:actor = {
					random_list = {
						25 = {
							scope:actor = {
								add_piety = minor_piety_loss
							}
						}
						25 = {
							scope:actor = {
								add_prestige = minor_prestige_loss
							}
						}
						25 = {
							scope:actor = {
								remove_short_term_gold = minor_gold_value
							}
						}
						#25 = { }
					}
				}
			}	
			
			if = {
				limit = {
					scope:POD_find_mortals_peasants = yes
				}
			}			
			
			trigger_event = {
				id = POD_find_mortals.100 
				days = { 1 14 }
			} 
		}
	}

	ai_will_do = {
		base = 0
	}
}
