POD_divide_interaction = {
	category = interaction_category_werewolf
	#common_interaction = yes
	#interface_priority = 90

	ai_frequency = 0

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_perk = divide_perk }
		scope:recipient = { is_adult = yes }
		scope:recipient = { is_landed = yes }
		scope:actor = { is_ai = no }			
	}

	is_valid_showing_failures_only = {
		scope:recipient = { is_imprisoned = no }
	}

	desc = POD_divide_interaction_desc
	
	cost = { piety = 150 }
	
	cooldown = { months = 6 }

	on_accept = {
		scope:recipient = {
			add_character_modifier = {
			modifier = divide_modifier
			years = 5
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_inspiration_interaction = {
	category = interaction_category_werewolf
	#common_interaction = yes
	#interface_priority = 90

	ai_frequency = 0

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		scope:actor = { is_ai = no }
		scope:actor = { 
			OR = {
				has_perk = inspiration_perk 
				has_perk = pack_tactics_perk
				has_perk = strenght_will_perk		
				}
		}
		scope:recipient = { 
			OR = {
				has_trait = werewolf
				has_trait = kinfolk
			}
		}			
	}

	is_valid_showing_failures_only = {
		scope:recipient = { is_imprisoned = no }
	}

	desc = POD_inspiration_interaction_desc
	
	cost = { prestige = 50 }
	
	cooldown = { months = 6 }

	on_accept = {
	
	if = {
		limit = {
			scope:actor = {
				has_perk = strenght_will_perk
				}
			}
		scope:recipient = {
			add_character_modifier = {
			modifier = strength_will_modifier
			years = 5
			}
		}
	}	
	else_if = {
		limit = {
			scope:actor = {
				has_perk = pack_tactics_perk
				}
			}
		scope:recipient = {
			add_character_modifier = {
			modifier = pack_tactics_modifier
			years = 5
			}
		}
	}
	else = {
		scope:recipient = {
			add_character_modifier = {
			modifier = inspiration_modifier
			years = 5
			}
		}
	}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

		scope:actor = { has_trait = werewolf }
		scope:recipient = { has_trait = werewolf }		
		NOT = { scope:actor = scope:recipient }

		NOT = {
			scope:actor = {
				has_relation_packmember = scope:recipient
			}
		}
		
		scope:actor = {
			num_of_relation_packmember < 4
		}
	}

	is_valid_showing_failures_only = {
		
	custom_tooltip = {
	scope:recipient = {
	opinion = {
		target = scope:actor
		value >= 25
	}
	}
	text = WW_PACK_OPINION_TT
	}
	
	}

	desc = POD_packmember_interaction_desc
	

	on_accept = {
		scope:recipient = {
			set_relation_packmember = scope:actor
		}
	}

	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_burden_of_knowledge_interaction = {
	category = interaction_category_werewolf
	desc = POD_burden_of_knowledge_interaction_desc

	ai_frequency = 0
	cooldown = ( years = 3 )

	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { has_trait = werewolf }
		scope:actor = { has_perk = burden_of_knowledge_perk }
	}

	is_valid_showing_failures_only = {
		scope:actor = { 
			NOT = { has_character_modifier = POD_gnosis_0_modifier }
		}
	}
	
	on_accept = {
		scope:actor = {
			POD_gnosis_loss_check_effect = yes	
			duel = {
				skill = learning
				target = scope:recipient
				50 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					scope:recipient = {
						add_stress = 100
							
						random_list = {
							20 = {
								add_character_modifier = {
								   modifier = murderous_frenzy_modifier
								   years = 10
								}
							}
							9 = {
								add_character_modifier = {
								   modifier = change_for_better_modifier
								   years = 10
								}
							}
							1 = {
								death = {
								   death_reason = death_suicide
								}
							}
						}
					}
				}
				50 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier 
					}
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
