POD_divide_interaction = {
	category = interaction_category_hostile
	#common_interaction = yes
	#interface_priority = 90

	ai_frequency = 0

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_perk = divide_perk }
		scope:recipient = { is_adult = yes }
		scope:recipient = { is_landed = yes }
		scope:actor = { is_ai = no }			
	}

	is_valid_showing_failures_only = {
		scope:recipient = { is_imprisoned = no }
	}

	desc = POD_divide_interaction_desc
	
	cost = { piety = 150 }
	
	cooldown = { months = 6 }

	on_accept = {
		scope:recipient = {
			add_character_modifier = {
			modifier = divide_modifier
			years = 5
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_inspiration_interaction = {
	category = interaction_category_friendly
	#common_interaction = yes
	#interface_priority = 90

	ai_frequency = 0

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		scope:actor = { is_ai = no }
		scope:actor = { 
			OR = {
				has_perk = inspiration_perk 
				has_perk = pack_tactics_perk
				has_perk = strenght_will_perk		
				}
		}
		scope:recipient = { 
			OR = {
				has_trait = werewolf
				has_trait = kinfolk
			}
		}			
	}

	is_valid_showing_failures_only = {
		scope:recipient = { is_imprisoned = no }
	}

	desc = POD_inspiration_interaction_desc
	
	cost = { prestige = 50 }
	
	cooldown = { months = 6 }

	on_accept = {
	
	if = {
		limit = {
			scope:actor = {
				has_perk = strenght_will_perk
				}
			}
		scope:recipient = {
			add_character_modifier = {
			modifier = strength_will_modifier
			years = 5
			}
		}
	}	
	else_if = {
		limit = {
			scope:actor = {
				has_perk = pack_tactics_perk
				}
			}
		scope:recipient = {
			add_character_modifier = {
			modifier = pack_tactics_modifier
			years = 5
			}
		}
	}
	else = {
		scope:recipient = {
			add_character_modifier = {
			modifier = inspiration_modifier
			years = 5
			}
		}
	}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
