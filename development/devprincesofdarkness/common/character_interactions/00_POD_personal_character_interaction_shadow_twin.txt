POD_shadow_twin_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	interface_priority = 91
    desc = POD_shadow_twin_interaction_desc
	
	ai_frequency = 360

	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { is_adult = yes }
		scope:actor = { is_vampire_trigger = yes  }
		scope:actor = { has_perk = shadow_twin_perk }
		scope:actor = { NOT = { has_character_flag = has_shadow_twin_flag } }
		scope:actor = { NOT = { has_trait = kueijin } }
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
		    OR = {
		        NOT = { has_character_modifier = bloodhunger4_modifier }
				NOT = { has_character_modifier = bloodhunger5_modifier }
			}
		}
		custom_description = {
		    object = scope:actor 
			text = already_has_shadow_twin
			scope:actor = {
		        num_of_relation_shadow_twin_servant < 1
		    }
		}
	}

	on_accept = {
	   custom_tooltip = POD_shadow_twin_interaction.tt
	   scope:actor = {
	       trigger_event = POD_shadow_twin.001
	   }
    }
	
	ai_targets = {
		ai_recipients = self
	}
	
	auto_accept = yes
	
	cooldown = { years = 20 }
	
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 5
			piety > 800
		}
		modifier = {
		    add = 10
			is_at_war = yes
			piety > 800
		}
	}  	
}
