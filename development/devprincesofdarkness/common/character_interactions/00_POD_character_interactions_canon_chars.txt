POD_infiltrate_interaction = {
	category = interaction_category_hostile
	common_interaction = yes
	interface_priority = 90

	ai_frequency = 0

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { this = character:790179 }
		scope:recipient = { is_adult = yes }
		scope:recipient = { is_landed = yes }
		scope:recipient = { is_vampire_trigger = yes }
		scope:actor = { is_ai = no }	
		is_target_in_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:helena_decision
		}		
	}

	is_valid_showing_failures_only = {
		scope:recipient = { is_imprisoned = no }
	}

	desc = POD_infiltrate_interaction_desc
	
	cost = { gold = 150 }
	
	cooldown = { months = 6 }

	on_accept = {
		scope:recipient = {
			add_character_modifier = {
			modifier = infiltrate_modifier
			years = 5
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

		scope:recipient = { 
			NOT = {
				has_character_modifier = bjorn_crew_modifier
				}
			}
	}

	is_valid_showing_failures_only = {
		custom_description = {
			text = POD_bjorn_crew_interaction_limit
			scope:actor = {
				any_courtier_or_guest = {
						has_opinion_modifier ={
							target = scope:actor
							modifier = POD_bjorn_crew_opinion
						}	
						count < 5 #for some reason less than 6 means "it's alright to get the 6th one too"
				}
			}	
		}	
		scope:recipient = { is_landed = no }
		
		custom_description = {
		text = POD_bjorn_opinion_limit
		scope:recipient = { opinion = { target = character:790438 value >=0 } }
		}		
	}			

	desc = POD_bjorn_interaction_desc
	
	cost = { prestige = 150 }
	cooldown = ( years = 1 )
	

	on_accept = {
	scope:recipient = {
			add_character_modifier = bjorn_crew_modifier
			add_opinion = {
				modifier = POD_bjorn_crew_opinion
