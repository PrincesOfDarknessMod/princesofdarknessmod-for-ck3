POD_bless_interaction = {
	category = interaction_category_friendly
	common_interaction = yes
	interface_priority = 87
	
	ai_frequency = 0

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		scope:actor = { is_ai = no }
		scope:actor = { 
			OR = {
				has_perk = dominus_vobiscum_perk 
				has_trait = beatific
				}
		}
		scope:recipient = { 
			OR = {
				has_trait = mortal
				has_trait = supehunter
			}
		}
	}

	is_valid_showing_failures_only = {
	}
	
	cooldown = { months = 6 }
	
	cost = { piety = 100 }

	desc = POD_bless_interaction_desc
	
	on_accept = {
		scope:actor = {
			hidden_effect = {
				send_interface_toast = {
					title = POD_bless_interaction_toast
					left_icon = scope:actor
					right_icon = scope:recipient
					add_piety = -100
					}
				}	
			}	
		scope:recipient = {
			add_character_modifier = {
			modifier = blessed_by_truefaith_modifier
			years = 5
			}
		}
	}
	
	#ai_potential = {
	#	always = no
	#}

	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_holy_heal_interaction = {
	category = interaction_category_friendly
	interface_priority = 110
	desc = POD_holy_heal_interaction_desc
	options_heading = POD_holy_heal_interaction_options_heading	

	use_diplomatic_range = no

	cooldown = { months = 3 }	
	
	cost = { piety = 100 }
	
	is_shown = {
		scope:actor = scope:recipient 
		scope:actor = { has_trait = supehunter }
		scope:actor = { is_ai = no }
		scope:actor = { 
			OR = {
				has_perk = lay_of_hands_perk 
				has_trait = beatific
				}
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			OR = {
				has_trait = wounded_1
				has_trait = wounded_2
				has_trait = wounded_3
				has_trait = maimed
				has_trait = blind
				has_trait = disfigured
				AND = {
					has_trait = scarred
					NOT = {
						has_character_flag = POD_permanent_scarring
					}
				}
				AND = {
					has_trait = one_eyed
					OR = {
						NOT = { has_character_flag = POD_permanent_one_eyed }
					}
					#NOT = { has_character_modifier = eye_of_hazmiel }
				}
				has_trait = one_legged
			}
		}		
	}

	send_option = {
		is_shown = {
			scope:actor = { 
				has_trait = blind
				has_trait = supehunter
			}
		}	
		flag = blind
		localization = POD_blood_heal_blind
		#current_description = POD_blood_heal_blind
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				has_trait = maimed
			}
		}	
		flag = maimed
		localization = POD_blood_heal_maimed
		#current_description = POD_blood_heal_maimed
	}	
	
	send_option = {
		is_shown = {
			scope:actor = { 
				has_trait = one_legged
				has_trait = supehunter
			}
		}	
		flag = one_legged
		localization = POD_blood_heal_one_legged
		#current_description = POD_blood_heal_one_legged
	}

	send_option = {
		is_shown = {
			scope:actor = { 
				has_trait = one_eyed
				has_trait = supehunter
				NOT = { has_character_flag = POD_permanent_one_eyed }
			}
		}	
		flag = one_eyed
		localization = POD_blood_heal_one_eyed
		#current_description = POD_blood_heal_one_eyed
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				OR = {
					has_trait = wounded_1
					has_trait = wounded_2
					has_trait = wounded_3
				}
			}
		}	
		flag = wounded
		localization = POD_blood_heal_wounded
		#current_description = POD_blood_heal_wounded
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				has_trait = scarred
				NOT = { 
					has_character_flag = POD_permanent_scarring
				}
			}
		}	
		flag = scarred
		localization = POD_blood_heal_scarred
		#current_description = POD_blood_heal_scarred
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				has_trait = disfigured
				NOT = { 
					has_character_flag = POD_permanent_disfigured
				}
			}
		}	
		flag = disfigured
		localization = POD_blood_heal_disfigured
		#current_description = POD_blood_heal_disfigured
	}
	
	on_accept = {
		scope:actor = {
			
			if = {
				limit = {
					scope:blind = yes
				}
				scope:actor = {
					remove_trait = blind
					add_trait = one_eyed
				}
			}	

			if = {
				limit = {
					scope:maimed = yes
				}
				scope:actor = {
					remove_trait = maimed
				}
			}		

			if = {
				limit = {
					scope:one_legged = yes
				}
				scope:actor = {
					remove_trait = one_legged
				}
			}				

			if = {
				limit = {
					scope:disfigured = yes
				}
				scope:actor = {
					remove_trait = disfigured
				}
			}

			if = {
				limit = {
					scope:one_eyed = yes
					NOT = { has_character_flag = POD_permanent_one_eyed }
				}
				scope:actor = {
					remove_trait = one_eyed
				}
			}
			
			if = {
				limit = {
					scope:wounded = yes
				}
				scope:actor = {
		if = {
			limit = {
				has_trait = wounded_3
			}
			remove_trait = wounded_3
		}
		
		if = {
			limit = {
				has_trait = wounded_2
			}
			remove_trait = wounded_2
		}
		
		if = {
			limit = {
				has_trait = wounded_1
			}
			remove_trait = wounded_1
		}	
				}
			}

			if = {
				limit = {
					scope:scarred = yes
				}
				scope:actor = {
					remove_trait = scarred
				}
			}				
		}
	}
	
	auto_accept = yes

	ai_will_do = {
		base = 0
	}
}
