POD_bless_interaction = {
	category = interaction_category_holyarts
	common_interaction = yes
	interface_priority = 111
	
	ai_frequency = 0

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		scope:actor = { is_ai = no }
		scope:actor = { 
			OR = {
				has_perk = dominus_vobiscum_perk 
				has_trait = beatific
				}
		}
		scope:recipient = { 
			OR = {
				has_trait = mortal
				has_trait = supehunter
			}
		}
	}

	is_valid_showing_failures_only = {
	    scope:actor = {
		    NOT = { has_character_modifier = crisis_of_faith_modifier }
		}
	}
	
	cooldown = { months = 6 }
	
	cost = { piety = 100 }

	desc = POD_bless_interaction_desc
	
	on_accept = {
		scope:actor = {
			hidden_effect = {
				send_interface_toast = {
					title = POD_bless_interaction_toast
					left_icon = scope:actor
					right_icon = scope:recipient
					}
				}	
			}	
		scope:recipient = {
			add_character_modifier = {
			modifier = blessed_by_truefaith_modifier
			years = 5
			}
		}
	}
	
	#ai_potential = {
	#	always = no
	#}

	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
