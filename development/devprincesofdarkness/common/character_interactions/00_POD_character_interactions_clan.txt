
POD_gangrel_challenge_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	interface_priority = 88
	desc = POD_gangrel_challenge_interaction_desc
	
	is_shown = {
		scope:actor = { has_trait = gangrel }
		NOT = { scope:actor = scope:recipient }
		scope:actor = { 
			is_ai = no
			is_ruler = yes
		}
	
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
			can_add_hook = {
				target = scope:recipient
				type = proven_beta
			}
		}
		scope:actor = {
			NOT = { has_hook =  scope:recipient  }
		}
	}
	
	on_accept = {
		scope:actor = {
			duel = {
				skill = prowess
				value = scope:recipient.prowess
				0 = {
					desc = POD_gangrel_challenge_interaction_succ
					compare_modifier = {
						value = scope:actor.prowess
						multiplier = high_positive_duel_skill_multiplier
					}
					show_as_tooltip = {
						add_hook = { target = scope:recipient type = proven_beta }						
						
					}
					custom_tooltip = target_gains
					scope:recipient = {
						show_as_tooltip = {
							add_character_modifier = { modifier = POD_proven_beta_modifier years = 10 }
						}
							
					}
					
					hidden_effect = {
						send_interface_toast = {
							title = POD_gangrel_challenge_interaction_succ_toast			
							left_icon = scope:actor
							right_icon = scope:recipient
							add_hook = { target = scope:recipient type = proven_beta }						
							scope:recipient = {
								add_character_modifier = { modifier = POD_proven_beta_modifier years = 10 }
							}
						}
					}
				}
				0 = {
					desc = POD_gangrel_challenge_interaction_lose
					compare_modifier = {
						value = scope:recipient.prowess
						multiplier = medium_positive_duel_skill_multiplier
					}
					show_as_tooltip = {
						scope:recipient = {
							add_hook = { target = scope:actor type = proven_beta }						
						}
						add_character_modifier = { modifier = POD_proven_beta_modifier years = 10 }
					}
					hidden_effect = {
						send_interface_toast = {
							title = POD_gangrel_challenge_interaction_lose_toast			
							left_icon = scope:actor
							right_icon = scope:recipient
							scope:recipient = {
								add_hook = { target = scope:actor type = proven_beta }						
							}
							add_character_modifier = { modifier = POD_proven_beta_modifier years = 10 }
						}
					}
					
					
				}
			}
		}
	
	}
	
	auto_accept = {
		scope:recipient = { has_trait = gangrel }
	}
	
	ai_accept = {
		base = -10
		
		opinion_modifier = { # More likely to accept if Recipient likes the Actor
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = -0.2
			desc = AI_OPINION_REASON
		}
		modifier = {
			add = -5
			scope:recipient = {
				has_dread_level_towards = {
					target = scope:actor
					level = 1
				}
			}
			desc = INTIMIDATED_REASON
		}
		modifier = {
			add = -10
			scope:recipient = {
				has_dread_level_towards = {
					target = scope:actor
					level = 2
				}
			}
			desc = COWED_REASON
		}
		modifier = {
			add = -15
			scope:recipient.prowess < scope:actor.prowess
			
		}
		modifier = {
			add = 10
			scope:recipient = {
				has_trait = arrogant	
			}
			desc = PERSONALITY
		}
		modifier = {
			add = 5
			scope:recipient = {
				has_trait = wrathful	
			}
			desc = PERSONALITY
		}
		modifier = {
			add = { value = scope:recipient.ai_vengefulness divide = 10}
			always = yes
			desc = PERSONALITY
		}
		modifier = {
			add = { value = scope:recipient.ai_boldness divide = 10 }
			always = yes
		}
		modifier = {
			add = 10
			scope:recipient = { has_trait = tzimisce }
			desc = PERSONALITY
		}
		modifier = {
			add = 10
			scope:recipient = { has_trait = brujah }
			desc = PERSONALITY
		}
		modifier = {
			add = 5
			scope:recipient = { has_trait = impatient }
			desc = PERSONALITY
			
		}
		
	
	}

}
