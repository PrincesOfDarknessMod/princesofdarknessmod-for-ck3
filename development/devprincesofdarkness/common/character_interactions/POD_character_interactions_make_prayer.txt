make_prayer_interaction = {
	category = interaction_category_specialweapons
	#common_interaction = yes
	#interface_priority = 87
	
	ai_frequency = 0
	
	cooldown = { months = 1 }

	cost = { gold = 50 }

	is_shown = {
		scope:actor = { is_ai = no }
		scope:actor = { has_perk = celestial_punishment_perk }
		scope:recipient = { 
			NOR = {
				has_character_modifier = warning_prayer_modifier
				has_character_modifier = glowing_prayer_modifier
				has_character_modifier = bursing_prayer_modifier
				has_character_modifier = indenfensible_prayer_modifier
				has_character_modifier = explosive_prayer_modifier
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:actor.gold >= 50
		scope:recipient = { is_diplomatically_available = yes }
	}
	
	on_accept = {
		if = {
			limit = {
				scope:actor = {
					has_perk = heavens_thunder_perk
				}
			}
			scope:recipient = {
				add_character_modifier = {
					modifier = explosive_prayer_modifier
					months = 3
				}
			}
		}
		else_if = {
			limit = {
				scope:actor = {
					has_perk = celestial_fires_perk
				}
			}
			scope:recipient = {
				add_character_modifier = {
					modifier = indenfensible_prayer_modifier
					months = 3
				}
			}
		}
		else_if = {
			limit = {
				scope:actor = {
					has_perk = heavenly_flames_perk
				}
			}
			scope:recipient = {
				add_character_modifier = {
					modifier = bursing_prayer_modifier
					months = 3
				}
			}
		}
		else_if = {
			limit = {
				scope:actor = {
					has_perk = divine_bindings_perk
				}
			}
			scope:recipient = {
				add_character_modifier = {
					modifier = glowing_prayer_modifier
					months = 3
				}
			}
		}
		else_if = {
			limit = {
				scope:actor = {
					has_perk = celestial_punishment_perk
				}
			}
			scope:recipient = {
				add_character_modifier = {
					modifier = warning_prayer_modifier
					months = 3
				}
			}
		}
	}
	
	#ai_potential = {
	#	always = no
	#}

	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
