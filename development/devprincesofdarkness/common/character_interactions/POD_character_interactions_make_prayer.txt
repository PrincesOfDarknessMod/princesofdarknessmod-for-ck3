make_prayer_interaction = {
	category = interaction_category_specialweapons
	desc = make_prayer_interaction_desc
	
	cooldown = { months = 1 }

	cost = { gold = 20 }

	is_shown = {
		scope:actor = { has_perk = celestial_punishment_perk }
	}

	is_valid_showing_failures_only = {
		scope:recipient = { is_diplomatically_available = yes }
		scope:recipient = { 
			NOR = {
				has_character_modifier = warning_prayer_modifier
				has_character_modifier = glowing_prayer_modifier
				has_character_modifier = bursing_prayer_modifier
				has_character_modifier = indenfensible_prayer_modifier
				has_character_modifier = explosive_prayer_modifier
				has_character_modifier = burn_out_prayer_modifier
				has_character_modifier = cut_off_prayer_modifier
			}
		}
	}

	on_accept = {
		if = {
			limit = {
				scope:actor = {
					has_perk = divine_prohibition_perk
				}
			}
			scope:recipient = {
				send_interface_toast = {
					title = gotprayer_notification
					add_character_modifier = {
						modifier = cut_off_prayer_modifier
						years = 5
					}
				}
			}
			scope:actor = { POD_chi_decrease_effect = { COUNT = 5 } }
		}
		else_if = {
			limit = {
				scope:actor = {
					has_perk = heavens_judgment_perk
				}
			}
			scope:recipient = {
				send_interface_toast = {
					title = gotprayer_notification
					add_character_modifier = {
						modifier = burn_out_prayer_modifier
						years = 5
					}
				}
			}
			scope:actor = { POD_chi_decrease_effect = { COUNT = 4 } }
		}
		else_if = {
			limit = {
				scope:actor = {
					has_perk = heavens_thunder_perk
				}
			}
			scope:recipient = {
				send_interface_toast = {
					title = gotprayer_notification
					add_character_modifier = {
						modifier = explosive_prayer_modifier
						years = 5
					}
				}
			}
			scope:actor = { POD_chi_decrease_effect = { COUNT = 3 } }
		}
		else_if = {
			limit = {
				scope:actor = {
					has_perk = celestial_fires_perk
				}
			}
			scope:recipient = {
				send_interface_toast = {
					title = gotprayer_notification
					add_character_modifier = {
						modifier = indenfensible_prayer_modifier
						years = 5
					}
				}
			}
			scope:actor = { POD_chi_decrease_effect = { COUNT = 3 } }
		}
		else_if = {
			limit = {
				scope:actor = {
					has_perk = heavenly_flames_perk
				}
			}
			scope:recipient = {
				send_interface_toast = {
					title = gotprayer_notification
					add_character_modifier = {
						modifier = bursing_prayer_modifier
						years = 5
					}
				}
			}
			scope:actor = { POD_chi_decrease_effect = { COUNT = 2 } }
		}
		else_if = {
			limit = {
				scope:actor = {
					has_perk = divine_bindings_perk
				}
			}
			scope:recipient = {
				send_interface_toast = {
					title = gotprayer_notification
					add_character_modifier = {
						modifier = glowing_prayer_modifier
						years = 5
					}
				}
			}
			scope:actor = { POD_chi_decrease_effect = { COUNT = 2 } }
		}
		else_if = {
			limit = {
				scope:actor = {
					has_perk = celestial_punishment_perk
				}
			}
			scope:recipient = {
				send_interface_toast = {
					title = gotprayer_notification
					add_character_modifier = {
						modifier = warning_prayer_modifier
						years = 5
					}
				}
			}
			scope:actor = { POD_chi_decrease_effect = { COUNT = 1 } }
		}
		
		if = {
			limit = {
				NOT = { scope:actor = scope:recipient }
			}
			scope:recipient = {
				add_opinion = {
					target = scope:actor
					modifier = gift_opinion
					opinion = 20
				}
			}
		}
	}

	auto_accept = yes
	
	ai_frequency = 36

	ai_potential = {
		# The basic requirements
		is_imprisoned = no 
		is_landed = yes 
		number_of_knights > 0
		primary_title.tier > tier_barony
		
		short_term_gold >= 50
	}
	ai_targets = {
		ai_recipients = vassals
		max = 10
	}
	ai_targets = {
		ai_recipients = courtiers
		max = 10
	}
	
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 100
			scope:recipient = {
				is_knight_of = scope:actor
			}
		}
	}
}
