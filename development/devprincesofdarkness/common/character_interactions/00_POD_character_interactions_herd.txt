POD_drink_herd = {
	category = interaction_category_vampire
	common_interaction = yes
	#interface_priority = 88
	desc = POD_drink_herd_desc
	ai_frequency = 0
	
	is_shown = {
		scope:actor = {
			has_POD_vamp_blood_trigger = yes
		}
		scope:actor = { is_kueijin_trigger = no }
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		scope:actor = { is_ai = no }
		scope:recipient = {NOT = {has_character_modifier = bloodhunger5_modifier}}
		scope:recipient = {
			has_relation_herd_owner = scope:actor
			# has_opinion_modifier = {
				# modifier = POD_herd_opinion 
				# target = scope:actor
			# }
		}
	}
	
	is_valid_showing_failures_only = {
		custom_description = {
            object = scope:actor
            text = "I_must_be_at_the_same_location"
            scope:actor = { location = scope:recipient.location }
        }
	}
	
	send_option = {
		is_shown = {
			NOT = { scope:recipient = {has_character_modifier = bloodhunger5_modifier} }
		}	
		flag = sip
		localization = POD_drink_herd_sip_option
		

	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				NOR = {
					has_character_modifier = majorbloodloss_modifier
					has_character_modifier = bloodhunger4_modifier
					has_character_modifier = bloodhunger5_modifier
				}
			}
		}	
		flag = drink
		localization = POD_drink_herd_drink_option
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				NOR = {
					has_character_modifier = majorbloodloss_modifier
					has_character_modifier = smallbloodloss_modifier
					has_character_modifier = bloodhunger3_modifier
					has_character_modifier = bloodhunger4_modifier
					has_character_modifier = bloodhunger5_modifier
				}
			}
		}	
		flag = alot
		localization = POD_drink_herd_alot_option
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				NOR = {
					has_character_modifier = majorbloodloss_modifier
					has_character_modifier = smallbloodloss_modifier
					has_character_modifier = slightbloodloss_modifier
					has_character_modifier = bloodhunger2_modifier
					has_character_modifier = bloodhunger3_modifier
					has_character_modifier = bloodhunger4_modifier
					has_character_modifier = bloodhunger5_modifier
				}
			}
		}	
		flag = drain
		localization = POD_drink_herd_drain_option
	}
	
	on_accept = {
		
		scope:recipient = {save_scope_as = victim}
		hidden_effect = {
			scope:victim = {
				if = {
					limit = {
						NOT = {	has_character_flag = POD_resonance_negligible_flag }
						NOT = {	has_character_flag = POD_resonance_fleeting_flag }
						NOT = {	has_character_flag = POD_resonance_intense_flag }
						NOT = {	has_character_flag = POD_resonance_acute_flag }
						NOT = {	has_character_flag = choleric_resonance }
						NOT = {	has_character_flag = melancholic_resonance }
						NOT = {	has_character_flag = phlegmatic_resonance }
						NOT = {	has_character_flag = sanguine_resonance }
					}
					POD_set_resonance_effect = yes
				}
			}
		}
		if = {
			limit = {scope:sip = yes}
			scope:actor = {POD_resonance_feeding_herd_sip_effect = yes}
			POD_hunger_feeding_bloodloss_effect = {
				VICTIM = scope:victim
				COUNT = 1
				VAMPIRE = scope:actor
			
			
			}
			
			scope:actor = {POD_hunger_feeding_sip_effect = yes}
			
		}
		if = {
			limit = {scope:drink = yes}
			
			scope:actor = {POD_resonance_feeding_herd_drink_effect = yes}
			POD_hunger_feeding_bloodloss_effect = {
				VICTIM = scope:victim
				COUNT = 2
				VAMPIRE = scope:actor
			
			}

			
			scope:actor = {POD_hunger_feeding_drink_effect = yes}
		}
		if = {
			limit = {scope:alot = yes}
			
			scope:actor = {POD_resonance_feeding_herd_lot_effect = yes}
			POD_hunger_feeding_bloodloss_effect = {
				VICTIM = scope:victim
				COUNT = 3
				VAMPIRE = scope:actor
			
			}

			
			scope:actor = {POD_hunger_feeding_lot_effect = yes}
		}
		if = {
			limit = {scope:drain = yes}
			scope:actor = {POD_resonance_feeding_herd_drain_effect = yes}
			POD_hunger_feeding_bloodloss_effect = {
				VICTIM = scope:victim
				COUNT = 4
				VAMPIRE = scope:actor
			
			}
			
			scope:actor = {POD_hunger_feeding_drain_effect = yes}
		}
		
	}
	auto_accept = yes
}
