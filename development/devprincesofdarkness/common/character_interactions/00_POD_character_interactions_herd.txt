POD_drink_herd = {
	category = interaction_category_vampire
	common_interaction = yes
	interface_priority = 88
	desc = POD_drink_herd_desc
	ai_frequency = 0
	
	is_shown = {
		scope:actor = {
			has_POD_vamp_blood_trigger = yes
		}
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		scope:actor = { is_ai = no }
		scope:recipient = {NOT = {has_character_modifier = bloodhunger5_modifier}}
		scope:recipient = {
			has_opinion_modifier = {
				modifier = POD_herd_opinion 
				target = scope:actor
			}
		}
	}
	
	is_valid_showing_failures_only = {
			scope:actor = { location = scope:recipient.location }
	}
	
	send_option = {
		is_shown = {
			NOT = { scope:recipient = {has_character_modifier = bloodhunger5_modifier} }
		}	
		flag = sip
		localization = POD_drink_herd_sip_option
		

	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				NOR = {
					has_character_modifier = majorbloodloss_modifier
					has_character_modifier = bloodhunger4_modifier
					has_character_modifier = bloodhunger5_modifier
				}
			}
		}	
		flag = drink
		localization = POD_drink_herd_drink_option
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				NOR = {
					has_character_modifier = majorbloodloss_modifier
					has_character_modifier = smallbloodloss_modifier
					has_character_modifier = bloodhunger3_modifier
					has_character_modifier = bloodhunger4_modifier
					has_character_modifier = bloodhunger5_modifier
				}
			}
		}	
		flag = alot
		localization = POD_drink_herd_alot_option
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				NOR = {
					has_character_modifier = majorbloodloss_modifier
					has_character_modifier = smallbloodloss_modifier
					has_character_modifier = slightbloodloss_modifier
					has_character_modifier = bloodhunger2_modifier
					has_character_modifier = bloodhunger3_modifier
					has_character_modifier = bloodhunger4_modifier
					has_character_modifier = bloodhunger5_modifier
				}
			}
		}	
		flag = drain
		localization = POD_drink_herd_drain_option
	}
	
	on_accept = {
		
		scope:recipient = {save_scope_as = victim}
		hidden_effect = {
			scope:victim = {
				POD_clear_resonance_effect = yes
				POD_set_resonance_effect = yes
			}
		}
		if = {
			limit = {scope:sip = yes}
			scope:actor = {POD_resonance_feeding_sip_effect = yes}
			POD_hunger_feeding_bloodloss_effect = {
				VICTIM = scope:victim
				COUNT = 1
				VAMPIRE = scope:actor
			
			
			}
			
			scope:actor = {POD_hunger_feeding_sip_effect = yes}
			
		}
		if = {
			limit = {scope:drink = yes}
			
			scope:actor = {POD_resonance_feeding_drink_effect = yes}
			POD_hunger_feeding_bloodloss_effect = {
				VICTIM = scope:victim
				COUNT = 2
				VAMPIRE = scope:actor
			
			}

			
			scope:actor = {POD_hunger_feeding_drink_effect = yes}
		}
		if = {
			limit = {scope:alot = yes}
			
			scope:actor = {POD_resonance_feeding_lot_effect = yes}
			POD_hunger_feeding_bloodloss_effect = {
				VICTIM = scope:victim
				COUNT = 3
				VAMPIRE = scope:actor
			
			}

			
			scope:actor = {POD_hunger_feeding_lot_effect = yes}
		}
		if = {
			limit = {scope:drain = yes}
			scope:actor = {POD_resonance_feeding_drain_effect = yes}
			POD_hunger_feeding_bloodloss_effect = {
				VICTIM = scope:victim
				COUNT = 4
				VAMPIRE = scope:actor
			
			}
			
			scope:actor = {POD_hunger_feeding_drain_effect = yes}
		}
		
	}
	auto_accept = yes
}

POD_herdify_mortal_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	interface_priority = 88
	scheme = herdify_mortal
	
	ai_frequency = 0

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		scope:actor = { location = scope:recipient.location }
		scope:actor = { is_vampire_trigger = yes  }
		scope:recipient = { is_vampire_trigger = no  }
		scope:recipient = { POD_is_supehunter_trigger = no }
		scope:actor = { is_ai = no }
		scope:recipient = {
			NOT = {
				has_opinion_modifier ={
					target = scope:actor
					modifier = POD_herd_opinion
				}
			}
		}
		
		
		
		
	}

	is_valid_showing_failures_only= {
		scope:actor = {
			can_start_scheme = {
				type = herdify_mortal
				target = scope:recipient
			}
		}
		custom_description = {
			text = POD_herdify_mortal_interaction_limit
			scope:actor = {
				any_courtier_or_guest = {
						has_opinion_modifier ={
							target = scope:actor
							modifier = POD_herd_opinion
						}
						
						count < 5
				}

			}
		}
	}

	desc = POD_herdify_mortal_interaction_desc

	on_accept = {
	
		hidden_effect = {
			scope:actor = {
				start_scheme = {
					type = herdify_mortal
					target = scope:recipient
				}
			}
		}
		
		if = {
			limit = {
				OR = {
					scope:actor = {has_relation_best_friend = scope:recipient}
					scope:actor = {has_relation_friend = scope:recipient}
					scope:actor = {has_relation_lover = scope:recipient}
				}
			}
			scope:actor = {
				stress_impact = {
					compassionate = medium_stress_impact_gain
				}
			}
		}
		
		if = {
			limit = {
					scope:actor = {
						faith = {has_doctrine_parameter = vampirism_illegal }
					}
			}
			scope:actor = {add_piety = major_piety_loss}
		}
		
		else_if = {
			limit = {
				scope:actor = {
					faith = {has_doctrine_parameter = vampirism_shunned} 
				}
			}
			scope:actor = {add_piety = medium_piety_loss}
		}
	}
	
	#ai_potential = {
	#	always = no
	#}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
