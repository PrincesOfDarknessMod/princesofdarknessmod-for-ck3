POD_project_negative_psychic_energy_interaction = {
	category = interaction_category_vampire
	#common_interaction = yes
	#interface_priority = 88
	desc = POD_project_negative_psychic_energy_interaction_desc
	
	cooldown = { months = 3 }	
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { is_kueijin_trigger = yes }
		scope:actor = { has_perk = thousand_hell_stare_perk }
		scope:actor = { is_ai = no }
	}

	is_valid_showing_failures_only = {
		scope:recipient = { 
			NOT = {
				has_trait = torpor
			}
		}
		custom_description = {
            object = scope:actor
            text = "soul_demon_incompatible"
            scope:actor = { using_demon_arts_trigger = no }
        }
	}
	
	on_accept = {
		scope:actor = { POD_chi_decrease_effect = { COUNT = 1 } }
		scope:actor = { POD_willpower_rouse_check_effect = yes }
		scope:recipient = {
			add_opinion = {
				target = scope:actor
				modifier = hurt_opinion
				opinion = -20
			}
			random_list = {
				20 = {
					POD_willpower_rouse_check_effect = yes
				}
				20 = {
					POD_willpower_rouse_check_effect = yes
					add_character_modifier = {
						modifier = intrigue_broken_modifier
						months = 1
					}
				}
				35 = {
					add_stress = scope:actor.stress
					add_character_modifier = {
						modifier = intrigue_broken_modifier
						months = 6
					}
				}
				25 = {
					add_stress = scope:actor.stress
					add_character_modifier = {
						modifier = intrigue_broken_modifier
						years = 1
					}
					random_list = {
						40 = {
							trigger = {
								NOR = { 
									has_trait = depressed_1
									has_trait = depressed_genetic 
								}
							}
							add_trait = depressed_genetic
						}
						40 = {
							trigger = {
								NOR = { 
									has_trait = lunatic_1
									has_trait = lunatic_genetic 
								}
							}
							add_trait = lunatic_genetic
						}
						20 = {
							trigger = {
								NOT = { has_trait = dull }
							}
							add_trait = dull
						}
					}
				}	
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_purification_interaction = {
	category = interaction_category_vampire
	#common_interaction = yes
	#interface_priority = 88
	desc = POD_purification_interaction_desc
	
	cooldown = { months = 3 }	
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { is_kueijin_trigger = yes }
		scope:actor = { has_perk = purification_perk }
		scope:actor = { is_ai = no }
	}

	is_valid_showing_failures_only = {
		scope:recipient = { 
			NOT = {
				has_trait = torpor
			}
		}
	}
	
	on_accept = {
		scope:actor = { POD_chi_decrease_effect = { COUNT = 1 } }
		scope:recipient = {
			add_opinion = {
				target = scope:actor
				modifier = pleased_opinion
				opinion = 15
			}
			if = {
				limit = {
					has_trait = depressed_1
				}
				remove_trait = depressed_1
			}
			if = {
				limit = {
					has_trait = depressed_genetic
				}
				remove_trait = depressed_genetic
			}
			if = {
				limit = {
					has_trait = lunatic_1
					is_malkavian_trigger = no
				}
				remove_trait = lunatic_1
			}
			if = {
				limit = {
					has_trait = lunatic_genetic
				}
				remove_trait = lunatic_genetic
			}
			if = {
				limit = {
					has_trait = dull
				}
				remove_trait = dull
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
