POD_project_negative_psychic_energy_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	interface_priority = 88
	desc = POD_project_negative_psychic_energy_interaction_desc
	
	cooldown = { months = 12 }	
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		scope:actor = { has_POD_vamp_blood_trigger = yes }
		scope:actor = { has_perk = thousand_hell_stare_perk }
		scope:recipient = { 
			NOT = {
				has_trait = torpor
			}
		}
		scope:actor = { is_ai = no }
	}
	
	on_accept = {
		if = {
			limit = { scope:recipient = { is_vampire_trigger = yes } }
			scope:actor = { POD_willpower_rouse_check_effect = yes}
		}
		scope:actor = { add_stress = -50 }
		scope:recipient = { add_stress = 50 }
		scope:recipient = { POD_hunger_rouse_check_effect = yes }
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
