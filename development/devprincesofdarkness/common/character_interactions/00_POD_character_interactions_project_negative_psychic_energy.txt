POD_project_negative_psychic_energy_interaction = {
	category = interaction_category_vampire
	desc = POD_project_negative_psychic_energy_interaction_desc
	
	cooldown = { months = 3 }	
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { is_kueijin_trigger = yes }
		scope:actor = { has_perk = thousand_hell_stare_perk }
		scope:recipient = { POD_is_clone_trigger = no }
	}

	is_valid_showing_failures_only = {
		custom_description = {
            object = scope:actor
            text = "soul_demon_incompatible"
            scope:actor = { using_demon_arts_trigger = no }
        }
		scope:recipient = { 
			NOR = {
				has_trait = torpor
				has_trait = effigy
				has_character_modifier = traveling_spirit_world_modifier
			}
		}
	}
	
	send_option = {
		is_shown = {
			NOR = {
				scope:actor = { 
					has_hook_of_type = {
						target = scope:recipient
						type = strong_soul_shackles_hook
					}
				}
				scope:recipient = { is_imprisoned_by = scope:actor }
				scope:recipient = { has_relation_necromancer = scope:actor }
			}
		}
		is_valid = {
			scope:actor = {
				can_start_scheme = {
					type = POD_project_negative_psychic_energy
					target = scope:recipient
				}
			}
		}
		flag = scheme_attack
		localization = POD_interaction_scheme_attack
	}
	
	send_option = {
		is_shown = {
			NOR = {
				scope:actor = { 
					has_hook_of_type = {
						target = scope:recipient
						type = strong_soul_shackles_hook
					}
				}
				scope:recipient = { is_imprisoned_by = scope:actor }
				scope:recipient = { has_relation_necromancer = scope:actor }
			}
		}
		is_valid = {
			custom_description = {
				object = scope:actor
				text = "I_must_be_at_the_same_location"
				scope:actor = { location = scope:recipient.location }
			}
		}
		flag = duel_attack
		localization = POD_interaction_duel_attack
	}
	
	send_option = {
		is_valid = {
			OR = {
				scope:actor = { 
					has_hook_of_type = {
						target = scope:recipient
						type = strong_soul_shackles_hook
					}
				}
				scope:recipient = { is_imprisoned_by = scope:actor }
				scope:recipient = { has_relation_necromancer = scope:actor }
			}
		}
		flag = automatic
		localization = POD_interaction_automatic
	}
	
	on_accept = {
		scope:recipient = {
			custom_tooltip = POD_project_negative_psychic_energy.tt
		}
		
		if = {
			limit = {
				scope:scheme_attack = yes
			}
			scope:actor = {
				start_scheme = {
					type = POD_project_negative_psychic_energy
					target = scope:recipient
				}
			}
		}
	
		if = {
			limit = {
				scope:duel_attack = yes
			}
			scope:actor = {
				configure_start_single_combat_effect = {
					SC_INITIATOR = scope:actor
					SC_ATTACKER = scope:actor
					SC_DEFENDER = scope:recipient
					FATALITY = no
					FIXED = no
					LOCALE = throne_room
					OUTPUT_EVENT = kueijin_scheme.300
					INVALIDATION_EVENT = tribal.1022
				}
			}
		}
		
		if = {
			limit = {
				scope:automatic = yes
			}
			scope:actor = { trigger_event = kueijin_scheme.300 }
		}
	}
	
	auto_accept = yes
	
	ai_frequency = 360
	
	ai_targets = {
		ai_recipients = neighboring_rulers
	}
	
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 45
			scope:actor = {
			    is_at_war_with = scope:recipient
			}
		}
		modifier = {
		    add = 10
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
	}
}

POD_purification_interaction = {
	category = interaction_category_vampire
	desc = POD_purification_interaction_desc
	
	cooldown = { months = 3 }	
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { is_kueijin_trigger = yes }
		scope:actor = { has_perk = purification_perk }
		scope:actor = { is_ai = no }
	}

	is_valid_showing_failures_only = {
		scope:recipient = { 
			NOT = {
				has_trait = torpor
			}
			OR = {
				has_trait = depressed_1
				has_trait = depressed_genetic
				has_trait = lunatic_1
				has_trait = lunatic_genetic
				has_trait = possessed_1
				has_trait = possessed_genetic
			}
		}
	}
	
	on_accept = {
		scope:actor = { POD_chi_decrease_effect = { COUNT = 2 } }
		scope:recipient = {
			add_opinion = {
				target = scope:actor
				modifier = pleased_opinion
				opinion = 15
			}
			if = {
				limit = {
					has_trait = depressed_1
				}
				remove_trait = depressed_1
			}
			if = {
				limit = {
					has_trait = depressed_genetic
				}
				remove_trait = depressed_genetic
			}
			if = {
				limit = {
					has_trait = lunatic_1
					is_malkavian_trigger = no
				}
				remove_trait = lunatic_1
			}
			if = {
				limit = {
					has_trait = lunatic_genetic
					is_malkavian_trigger = no
				}
				remove_trait = lunatic_genetic
			}
			if = {
				limit = {
					has_trait = possessed_1
				}
				remove_trait = possessed_1
			}
			if = {
				limit = {
					has_trait = possessed_genetic
				}
				remove_trait = possessed_genetic
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
