# POD Personal Character Interactions
#  Drain Animal

POD_drain_animal_interaction = {
	category = interaction_category_vampire
	#interface_priority= 95
	#icon = "gfx/interface/icons/character_interactions/blood_surge_interaction.dds"
	desc = POD_drain_animal_interaction_desc
	options_heading = POD_drain_animal_interaction_options_heading

	send_option = {
		flag = POD_drain_animal_interaction_my_horse
		localization = POD_drain_animal_interaction_my_horse
		is_shown = {
			scope:actor = {
				OR = {
					has_character_modifier = horse_story_modifier
					has_character_modifier = horse_strength_modifier
					has_character_modifier = horse_agility_modifier
					has_character_modifier = horse_endurance_modifier
					has_character_modifier = horse_fierceness_modifier
					has_character_modifier = horse_strength_strength_modifier 
					has_character_modifier = horse_agility_strength_modifier
					has_character_modifier = horse_agility_agility_modifier
					has_character_modifier = horse_endurance_strength_modifier
					has_character_modifier = horse_endurance_agility_modifier
					has_character_modifier = horse_endurance_endurance_modifier
					has_character_modifier = horse_fierceness_strength_modifier
					has_character_modifier = horse_fierceness_agility_modifier 
					has_character_modifier = horse_fierceness_endurance_modifier 
					has_character_modifier = horse_fierceness_fierceness_modifier
					has_character_modifier = horse_practice_modifier
					has_character_modifier = cursed_horse_modifier 
				}
			}
		}		
	}

	send_option = {
		flag = POD_drain_animal_interaction_my_dog
		localization = POD_drain_animal_interaction_my_dog
		is_shown = {
			scope:actor = {
				has_character_modifier = dog_story_modifier
			}
		}		
	} 
	
	send_option = {
		flag = POD_drain_animal_interaction_my_cat
		localization = POD_drain_animal_interaction_my_cat
		is_shown = {
			scope:actor = {
				has_character_modifier = cat_story_modifier
			}
		}		
	} 

	send_option = {
		flag = POD_drain_animal_interaction_rats
		localization = POD_drain_animal_interaction_rats
		is_shown = {
			OR = {
				scope:actor = { has_character_modifier = bloodpotency0_modifier }
				scope:actor = { has_character_modifier = bloodpotency1_modifier }
				AND = {
					scope:actor = { has_perk = animal_succulence_perk }
					OR = {
						scope:actor = { has_character_modifier = bloodpotency2_modifier }
						scope:actor = { has_character_modifier = bloodpotency3_modifier }
					}
				}
			}
		}		
	} 	

	send_option = {
		flag = POD_drain_animal_interaction_stray
		localization = POD_drain_animal_interaction_stray
	} 
	
	send_option = {
		flag = POD_drain_animal_interaction_cow
		localization = POD_drain_animal_interaction_cow
	} 	

	send_option = {
		flag = POD_drain_animal_interaction_deer
		localization = POD_drain_animal_interaction_deer
	} 

	send_option = {
		flag = POD_drain_animal_interaction_auto
		localization = POD_drain_animal_interaction_auto
		is_shown = {
			scope:actor = 
				{ NOT = { has_character_flag = autodrinkanimalsflag } }
			}	
	} 
	
	send_option = {
		flag = POD_drain_animal_interaction_auto_stop
		localization = POD_drain_animal_interaction_auto_stop
		is_shown = {
			scope:actor = 
				{ has_character_flag = autodrinkanimalsflag }
			}
	} 	

	send_options_exclusive = yes
	
	auto_accept = yes
	ai_frequency = 0
	use_diplomatic_range = no
	
	cooldown = { months = 3 }
	
	is_shown = {
		scope:actor = scope:recipient 
		scope:recipient = { is_adult = yes }
		scope:actor = { is_vampire_trigger = yes  }
		scope:actor = { is_ai = no }			
	}
	
	is_valid_showing_failures_only = {		
	}	

	on_accept = {
		scope:actor = {
		
			if = {
				limit = {
					scope:POD_drain_animal_interaction_rats = yes
				}
				custom_tooltip = POD_drain_animal_interaction_rats.tt
				POD_minimal_risk_of_increasing_masquerade_exposure = yes
				random = {
					chance = 5
					scope:actor = {
						add_prestige = minor_prestige_loss
					}
				}
				POD_hunger_feeding_small_animal_effect = yes
			}

			if = {
				limit = {
					scope:POD_drain_animal_interaction_stray = yes
				}
				custom_tooltip = POD_drain_animal_interaction_stray.tt
				POD_small_risk_of_increasing_masquerade_exposure = yes
				POD_hunger_feeding_small_animal_effect = yes
				random = {
					chance = 25
					add_animal_lifestyle_xp = minor_lifestyle_xp
				}					
			}	

			if = {
				limit = {
					scope:POD_drain_animal_interaction_cow = yes
				}
				custom_tooltip = POD_drain_animal_interaction_cow.tt
				scope:actor = {
					remove_short_term_gold = minor_gold_value
				}
				POD_small_risk_of_increasing_masquerade_exposure = yes
				POD_hunger_feeding_large_animal_effect = yes				
			}

			if = {
				limit = {
					scope:POD_drain_animal_interaction_deer = yes
				}
				custom_tooltip = POD_drain_animal_interaction_deer.tt
				random_list = {
					90 = {
						custom_tooltip = POD_hunting_success_tt
						scope:actor = { POD_hunger_feeding_large_animal_effect = yes }
						modifier = {
							add = 1
							has_trait = hunter_1
						}
						modifier = {
							add = 2
							has_trait = hunter_2
						}
						modifier = {
							add = 3
							has_trait = hunter_3
						}
						modifier = {
							add = 1
							any_secret = { 
								secret_type = secret_predatortype_farmer
							}
						}
						modifier = {
							add = 1
							has_trait = predatortype_farmer
						}					
					}
					
					9 = {
						scope:actor = { increase_wounds_effect = { REASON = hunting_accident } }
						custom_tooltip = POD_hunting_accident_tt
					}	
					
					1 = {
						scope:actor = {
							custom_tooltip = POD_werewolf_teaser_hunting_tt
							hidden_effect = {
								if = { 
									limit = {
										NOT = { has_character_flag = had_event_POD_werewolf_teaser_100_flag }
									}
									trigger_event = {
										id = POD_werewolf_teaser.100
										days = 7
									}
								}
								if = { 
									limit = {
										NOT = { has_character_flag = had_event_POD_werewolf_teaser_100_flag }
									}								
									scope:actor = { increase_wounds_effect = { REASON = hunting_accident } }
								}
							}
						}
					}
					
				}
				scope:actor = { POD_minimal_risk_of_increasing_masquerade_exposure = yes }			
			}

			if = {
				limit = {
					scope:POD_drain_animal_interaction_my_horse = yes
				}
				custom_tooltip = POD_drain_animal_interaction_my_horse.tt
				POD_minimal_risk_of_increasing_masquerade_exposure = yes
				if = {
					limit = { 
						scope:actor = {
							OR = {
								AND = {
									has_perk = animal_succulence_perk 
									has_perk = bond_famulus_perk
								}
								has_trait = animalismadvanced
							}
						}
					}
					POD_hunger_feeding_famulus_effect = yes
				}
				
				else_if = {
					POD_hunger_feeding_large_animal_effect = yes
				}
	
				if = {
					limit = {
						OR = {
							faith = {
								has_doctrine = doctrine_enlightenment_virtues_conscience_and_selfcontrol
							}
							faith = {
								has_doctrine = doctrine_enlightenment_virtues_conscience_and_instinct
							}
						}
					}
					add_piety = minor_piety_loss
				}				
				stress_impact = {
					gluttonous = minor_stress_impact_loss
					compassionate = minor_stress_impact_gain
				}
				scope:actor = { trigger_event = warhorse.8003 }
				hidden_effect = { 
					scope:actor = { 
						remove_horse_story_modifiers_effect = yes 						
					}					
				}
				scope:actor = {
					if = { 
						limit = {
							has_character_modifier = horse_story_modifier 
						}
						remove_character_modifier = horse_story_modifier 
					}
					if = { 
						limit = {
							has_character_modifier = horse_strength_modifier
						}
						remove_character_modifier = horse_strength_modifier
					}
					if = { 
						limit = {
							has_character_modifier = horse_agility_modifier
						}
						remove_character_modifier = horse_agility_modifier
					}
					if = { 
						limit = {
							has_character_modifier = horse_endurance_modifier
						}
						remove_character_modifier = horse_endurance_modifier
					}
					if = { 
						limit = {
							has_character_modifier = horse_fierceness_modifier
						}
						remove_character_modifier = horse_fierceness_modifier
					}
					if = { 
						limit = {
							has_character_modifier = horse_strength_strength_modifier
						}
						remove_character_modifier = horse_strength_strength_modifier 
					}
					if = { 
						limit = {
							has_character_modifier = horse_agility_strength_modifier
						}
						remove_character_modifier = horse_agility_strength_modifier
					}
					if = { 
						limit = {
							has_character_modifier = horse_agility_agility_modifier
						}
						remove_character_modifier = horse_agility_agility_modifier
					}
					if = { 
						limit = {
							has_character_modifier = horse_endurance_strength_modifier
						}
						remove_character_modifier = horse_endurance_strength_modifier
					}
					if = { 
						limit = {
							has_character_modifier = horse_endurance_agility_modifier
						}
						remove_character_modifier = horse_endurance_agility_modifier
					}
					if = { 
						limit = {
							has_character_modifier = horse_endurance_endurance_modifier
						}
						remove_character_modifier = horse_endurance_endurance_modifier
					}
					if = { 
						limit = {
							has_character_modifier = horse_fierceness_strength_modifier
						}
						remove_character_modifier = horse_fierceness_strength_modifier
					}
					if = { 
						limit = {
							has_character_modifier = horse_fierceness_agility_modifier 
						}
						remove_character_modifier = horse_fierceness_agility_modifier
					}
					if = { 
						limit = {
							has_character_modifier = horse_fierceness_endurance_modifier 
						}
						remove_character_modifier = horse_fierceness_endurance_modifier 
					}
					if = { 
						limit = {
							has_character_modifier = horse_fierceness_fierceness_modifier
						}
						remove_character_modifier = horse_fierceness_fierceness_modifier
					}
					if = { 
						limit = {
							has_character_modifier = horse_practice_modifier
						}
						remove_character_modifier = horse_practice_modifier
					}
					if = { 
						limit = {
							has_character_modifier = cursed_horse_modifier 
						}
						remove_character_modifier = cursed_horse_modifier 
					}
				}
				scope:actor = { 
					scope:story_cycle_martial_lifestyle_warhorse = {
						hidden_effect = { end_story = yes }
					}
				}
			}	

			if = {
				limit = {
					scope:POD_drain_animal_interaction_my_dog = yes
				}
				custom_tooltip = POD_drain_animal_interaction_my_dog.tt
				if = {
					limit = {
						scope:actor = {
							OR = {
								AND = {
									has_perk = animal_succulence_perk 
									has_perk = bond_famulus_perk
								}
								has_trait = animalismadvanced
							}
						}
					}
					POD_hunger_feeding_famulus_effect = yes
				}
				else_if = {
					POD_hunger_feeding_small_animal_effect = yes
					random_list = {
						20 = { add_animal_lifestyle_xp = miniscule_lifestyle_xp }
						40 = { add_animal_lifestyle_xp = minor_lifestyle_xp }
						40 = { add_animal_lifestyle_xp = medium_lifestyle_xp }
					}					
				}
				if = {
					limit = {
						OR = {
							faith = {
								has_doctrine = doctrine_enlightenment_virtues_conscience_and_selfcontrol
							}
							faith = {
								has_doctrine = doctrine_enlightenment_virtues_conscience_and_instinct
							}
						}
					}
					add_piety = minor_piety_loss
				}				
				stress_impact = {
					gluttonous = minor_stress_impact_loss
					compassionate = minor_stress_impact_gain
				}				
				scope:actor = { trigger_event = pet_animal.1199 }
				remove_dog_story_modifiers_effect = yes
			}	

			if = {
				limit = {
					scope:POD_drain_animal_interaction_my_cat = yes
				}
				custom_tooltip = POD_drain_animal_interaction_my_cat.tt
				if = {
					limit = { 
						scope:actor = {
							OR = {
								AND = {
									has_perk = animal_succulence_perk 
									has_perk = bond_famulus_perk
								}
								has_trait = animalismadvanced
							}
						}
					}
					POD_hunger_feeding_famulus_effect = yes
					
				}
				else_if = {
					POD_hunger_feeding_small_animal_effect = yes
					random = {
						chance = 25
						add_animal_lifestyle_xp = minor_lifestyle_xp
					}					
				}
				if = {
					limit = {
						OR = {
							faith = {
								has_doctrine = doctrine_enlightenment_virtues_conscience_and_selfcontrol
							}
							faith = {
								has_doctrine = doctrine_enlightenment_virtues_conscience_and_instinct
							}
						}
					}
					add_piety = minor_piety_loss
				}				
				stress_impact = {
					gluttonous = minor_stress_impact_loss
					compassionate = minor_stress_impact_gain
				}
				scope:actor = { trigger_event = pet_animal.0199 }
				remove_cat_story_modifiers_effect = yes
			}	
			
			if = {
				limit = {
					scope:POD_drain_animal_interaction_auto = yes
				}
				custom_tooltip = POD_drain_animal_interaction_auto.tt			

				scope:actor = { trigger_event = POD_herd_automation.4010  }
			}	

			if = {
				limit = {
					scope:POD_drain_animal_interaction_auto_stop = yes
				}
				custom_tooltip = POD_drain_animal_interaction_auto_stop.tt			

				remove_character_flag = autodrinkanimalsflag
			}				
			
		}	
	}

	ai_will_do = {
		base = 0
	}
}
