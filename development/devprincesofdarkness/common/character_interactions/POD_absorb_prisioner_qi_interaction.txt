POD_absorb_prisioner_yinchi_interaction = {
	interface_priority = 88

	category = interaction_category_prison
	
	cooldown = { months = 3 }	

	is_shown = {
		scope:recipient = {
			is_imprisoned_by = scope:actor
			OR = {
				POD_is_wraith_trigger = yes
				has_trait = yin_imbalanced
				has_trait = yinyang_balanced
			}
		}
		scope:actor = {
			is_adult = yes
			is_ai = no
			has_trait = kueijin
		}
	}

	on_accept = {
		scope:recipient = {
			if = {
				limit = {
					scope:actor = { 
						OR = { 
							has_trait = dharma5
							has_trait = dharma4
							has_trait = dharma3
							has_trait = dharma2
							has_trait = dharma1
						}
					}
				}
				add_character_flag = {
					flag = is_being_tortured
					months = 6
				}
			}
			if = {
				limit = {
					scope:actor = { has_trait = dharma3 }
				}
				add_trait = wounded_1
			}
			if = {
				limit = {
					scope:actor = { has_trait = dharma2 }
				}
				add_trait = wounded_2
			}
			if = {
				limit = {
					scope:actor = { has_trait = dharma1 }
				}
				add_trait = wounded_3
			}
		}

		scope:actor = {
			random_list = {
				80 = { 
					POD_turn_to_yin_effect = yes
				}
				20 = { 
					POD_turn_to_balanced_effect = yes
				}
			}
		}
	}
	
	#ai_potential = {
	#	always = no
	#}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
