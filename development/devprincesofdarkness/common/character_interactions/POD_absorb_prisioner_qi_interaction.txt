absorb_prisioner_yinqi_interaction = {
	interface_priority = 30

	category = interaction_category_prison

	is_shown = {
		scope:recipient = {
		is_imprisoned_by = scope:actor
		POD_is_wraith_trigger = yes
		}
		scope:actor = {
			is_adult = yes
			is_ai = no
			has_trait = kueijin
		}
	}

	on_accept = {
		scope:recipient = {
			add_character_modifier = {
				modifier = recently_tortured
				months = 6
			}
		}
		scope:actor = {
			custom_tooltip = prisoner_yin_tooltip
			hidden_effect = {
			if = {
				limit = {
					has_trait = yinyang_balanced
				}
				remove_trait = yinyang_balanced
				add_trait = yin_imbalanced
			}
			if = {
				limit = {
					has_trait = yang_imbalanced
				}
				remove_trait = yang_imbalanced
				random_list = {
					80 = { 
						add_trait = yin_imbalanced
					}
					20 = { 
						add_trait = yinyang_balanced
					}	
				}
			}
		}
		}
	}
	
	#ai_potential = {
	#	always = no
	#}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
