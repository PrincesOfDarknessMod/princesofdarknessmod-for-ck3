POD_adjust_balance_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	interface_priority = 88
	desc = POD_adjust_balance_interaction_desc
	
	cooldown = { months = 1 }	
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { is_kueijin_trigger = yes }
		scope:actor = { has_perk = adjust_balance_perk }
		scope:recipient = {
			NOT = {
				has_trait = torpor
			}
		}
		scope:actor = { is_ai = no }
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
			wall_too_strong_trigger = no
		}		
	}
	
	send_option = {
		flag = yin
		localization = POD_adjust_balance_interaction_to_yin
	}

	send_option = {
		flag = yang
		localization = POD_adjust_balance_interaction_to_yang
	}	

	send_option = {
		flag = balance
		localization = POD_adjust_balance_interaction_to_balanced
	}

	on_accept = {
		scope:actor = { POD_chi_decrease_effect = { COUNT = 1 } }
		if = {
			limit = {
				scope:yin = yes
			}
			scope:recipient = {
				if = {
					limit = {
						is_kueijin_trigger = yes
					}
					POD_yinvirtue_increase_effect = yes
					POD_yangvirtue_decrease_effect = yes
					POD_check_yinyang_effect = yes
				}
				add_character_modifier = {
					modifier = attuned_to_yin_modifier
					months = 1
				}
			}
		}
		if = {
			limit = {
				scope:yang = yes
			}
			scope:recipient = {
				if = {
					limit = {
						is_kueijin_trigger = yes
					}
					POD_yinvirtue_decrease_effect = yes
					POD_yangvirtue_increase_effect = yes
					POD_check_yinyang_effect = yes
				}
				add_character_modifier = {
					modifier = attuned_to_yang_modifier
					months = 1
				}
			}
		}
		if = {
			limit = {
				scope:balance = yes
			}
			scope:recipient = {
				if = {
					limit = {
						is_kueijin_trigger = yes
					}
					if = {
						limit = {
							has_trait = yin_imbalanced
						}
						POD_yinvirtue_increase_effect = yes
						POD_yangvirtue_decrease_effect = yes
					}
					if = {
						limit = {
							has_trait = yang_imbalanced
						}
						POD_yinvirtue_decrease_effect = yes
						POD_yangvirtue_increase_effect = yes
					}
					POD_check_yinyang_effect = yes
				}
				add_character_modifier = {
					modifier = state_of_balance_modifier
					months = 1
				}
				remove_trait = ill
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_shift_balance_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	interface_priority = 88
	desc = POD_shift_balance_interaction_desc
	
	cooldown = { months = 1 }	
	
	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { is_kueijin_trigger = yes }
		scope:actor = { is_ai = no }
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
			NOT = { has_trait = torpor }
			wall_too_strong_trigger = no
		}		
	}
	
	send_option = {
		is_shown = {
			scope:actor = {
				OR = {
					has_trait = yinyang_balanced
					has_trait = yang_imbalanced
				}
			}
		}	
		flag = yin
		localization = POD_shift_balance_interaction_to_yin
	}
	
	send_option = {
		is_shown = {
			scope:actor = {
				OR = {
					has_trait = yinyang_balanced
					has_trait = yin_imbalanced
				}
			}
		}
		flag = yang
		localization = POD_shift_balance_interaction_to_yang
	}

	send_option = {
		is_shown = {
			scope:actor = {
				OR = {
					has_trait = yin_imbalanced
					has_trait = yang_imbalanced
				}
			}
		}	
		flag = balance
		localization = POD_shift_balance_interaction_to_balanced
	}
	
	send_option = {
		is_shown = {
			scope:actor = {
				OR = {
					has_trait = yinyang_balanced
					has_trait = yang_imbalanced
				}
				scope:actor = { has_perk = shift_the_balance_perk }
			}
		}	
		flag = yin_instant
		localization = POD_shift_balance_interaction_to_yin_instant
	}
	
	send_option = {
		is_shown = {
			scope:actor = {
				OR = {
					has_trait = yinyang_balanced
					has_trait = yin_imbalanced
				}
				scope:actor = { has_perk = shift_the_balance_perk }
			}
		}
		flag = yang_instant
		localization = POD_shift_balance_interaction_to_yang_instant
	}

	send_option = {
		is_shown = {
			scope:actor = {
				OR = {
					has_trait = yin_imbalanced
					has_trait = yang_imbalanced
				}
				scope:actor = { has_perk = shift_the_balance_perk }
			}
		}	
		flag = balance_instant
		localization = POD_shift_balance_interaction_to_balanced_instant
	}

	on_accept = {
		scope:actor = { POD_willpower_rouse_check_effect = yes }
		if = {
			limit = {
				scope:yin = yes
			}
			scope:recipient = {
				POD_yinvirtue_increase_effect = yes
				POD_yangvirtue_decrease_effect = yes
				POD_check_yinyang_effect = yes
			}
		}
		
		if = {
			limit = {
				scope:yin_instant = yes
			}
			scope:recipient = {
				POD_clean_yinvirtue_effect = yes
				POD_clean_yangvirtue_effect = yes
				POD_turn_to_yin_effect = yes
				POD_set_chivirtue_effect = yes
				POD_chi_hunger_effect = yes
				POD_chi_fill_effect = yes
				POD_yinchi_overdose_effect = yes
				POD_yangchi_overdose_effect = yes
			}
		}
		
		if = {
			limit = {
				scope:yang = yes
			}
			scope:recipient = {
				POD_yinvirtue_decrease_effect = yes
				POD_yangvirtue_increase_effect = yes
				POD_check_yinyang_effect = yes
			}
		}
		
		if = {
			limit = {
				scope:yang_instant = yes
			}
			scope:recipient = {
				POD_clean_yinvirtue_effect = yes
				POD_clean_yangvirtue_effect = yes
				POD_turn_to_yang_effect = yes
				POD_set_chivirtue_effect = yes
				POD_chi_hunger_effect = yes
				POD_chi_fill_effect = yes
				POD_yinchi_overdose_effect = yes
				POD_yangchi_overdose_effect = yes
			}
		}
		
		if = {
			limit = {
				scope:balance = yes
			}
			scope:recipient = {
				if = {
					limit = {
						has_trait = yin_imbalanced
					}
					POD_yinvirtue_increase_effect = yes
					POD_yangvirtue_decrease_effect = yes
				}
				if = {
					limit = {
						has_trait = yang_imbalanced
					}
					POD_yinvirtue_decrease_effect = yes
					POD_yangvirtue_increase_effect = yes
				}
				POD_check_yinyang_effect = yes
			}
		}
		
		
		if = {
			limit = {
				scope:balance_instant = yes
			}
			scope:recipient = {
				POD_clean_yinvirtue_effect = yes
				POD_clean_yangvirtue_effect = yes
				POD_turn_to_balanced_effect = yes
				POD_set_chivirtue_effect = yes
				POD_chi_hunger_effect = yes
				POD_chi_fill_effect = yes
				POD_yinchi_overdose_effect = yes
				POD_yangchi_overdose_effect = yes
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
