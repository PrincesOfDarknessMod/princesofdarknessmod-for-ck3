POD_adjust_balance_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	interface_priority = 88
	desc = POD_adjust_balance_interaction_desc
	
	cooldown = { months = 3 }	
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { is_adult = yes }
		scope:recipient = { is_adult = yes }
		scope:actor = { has_trait = kueijin }
		scope:actor = { has_perk = adjust_balance_perk }
		scope:recipient = {
			NOT = {
				has_trait = torpor
			}
			has_trait = kueijin
		}
		scope:actor = { is_ai = no }
	}

	is_valid_showing_failures_only = {
			scope:actor = { NOT = { has_character_modifier = bloodhunger5_modifier } }
	}
	
	send_option = {
		is_shown = {
			scope:actor = {
				OR = {
					has_trait = yinyang_balanced
					has_trait = yang_imbalanced
				}
			}
		}	
		flag = yin
		localization = POD_adjust_balance_interaction_to_yin
	}

	send_option = {
		is_shown = {
			scope:actor = {
				OR = {
					has_trait = yinyang_balanced
					has_trait = yin_imbalanced
				}
			}
		}
		flag = yang
		localization = POD_adjust_balance_interaction_to_yang
	}	

	send_option = {
		is_shown = {
			scope:actor = {
				OR = {
					has_trait = yin_imbalanced
					has_trait = yang_imbalanced
				}
			}
		}	
		flag = balance
		localization = POD_adjust_balance_interaction_to_balanced
	}

	on_accept = {
		if = {
			limit = { scope:recipient = { is_vampire_trigger = yes } }
			scope:actor = { POD_willpower_rouse_check_effect = yes }
		}
		if = {
			limit = {
				scope:yin = yes
			}
			scope:recipient = {
				POD_turn_to_yin_effect = yes
			}
		}
		if = {
			limit = {
				scope:yang = yes
			}
			scope:recipient = {
				POD_turn_to_yang_effect = yes
			}
		}
		if = {
			limit = {
				scope:balance = yes
			}
			scope:recipient = {
				POD_turn_to_balanced_effect = yes
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
