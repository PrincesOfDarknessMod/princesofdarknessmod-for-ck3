POD_animal_control_interaction = {
	category = interaction_category_possessed
	desc = POD_animal_control_interaction_desc
	options_heading = POD_animal_control_interaction_options_heading
	
	cooldown = { months = 30 }
	cost = { piety = medium_piety_value }
	
	is_shown = {
		scope:actor = scope:recipient 
		scope:actor = { POD_is_possessed_trigger = yes }
		scope:actor = { has_perk = animal_control_perk }
		scope:actor = {
			NAND = {
				any_owned_story = { story_type = story_cycle_pet_dog }
				any_owned_story = { story_type = story_cycle_pet_cat }
				any_owned_story = { story_type = story_cycle_martial_lifestyle_warhorse }
			}
		}
	}

	send_option = {
		is_shown = {
			scope:actor = { NOT = { any_owned_story = { story_type = story_cycle_martial_lifestyle_warhorse } } }
		}
		flag = horse 
		localization = POD_control_creature_interaction_horse	
	}

	send_option = {
		is_shown = {
			scope:actor = { NOT = { any_owned_story = { story_type = story_cycle_pet_dog } } }
		}
		flag = dog
		localization = POD_control_creature_interaction_dog	
	} 
	
	send_option = {
		is_shown = {
			scope:actor = { NOT = { any_owned_story = { story_type = story_cycle_pet_cat } } }
		}
		flag = cat
		localization = POD_control_creature_interaction_cat		
	}
	
	send_option = {
		is_shown = {
			scope:actor = { NOT = { has_character_modifier = exotic_pet_wolf } }
		}
		flag = wolf
		localization = POD_control_creature_interaction_wolf	
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				NOT = { has_character_modifier = exotic_pet_bear } 
				# Not outside of Europe or Asian black bear territory
				location = { 
					OR = {
						geographical_region = world_europe
						geographical_region = world_tibet
						geographical_region = world_himalaya
						geographical_region = world_burma
						geographical_region = world_middle_east_persia
					}
				}
			}
		}
		flag = bear
		localization = POD_control_creature_interaction_bear	
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				NOT = { has_character_modifier = exotic_pet_lion } 
				# Not outside of Africa
				location = { 
					geographical_region = world_africa
				}
			}
		}
		flag = lion
		localization = POD_control_creature_interaction_lion	
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				NOT = { has_character_modifier = exotic_pet_tiger } 
				# Not outside of India or Asia
				location = { 
					OR = {
						geographical_region = world_india
						geographical_region = world_tibet
						geographical_region = world_himalaya
						geographical_region = world_burma
					}
				}
			}
		}
		flag = tiger
		localization = POD_control_creature_interaction_tiger	
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				NOT = { has_character_modifier = prison_pet_rat_modifier } 
			}
		}
		flag = rat
		localization = POD_control_creature_interaction_rat
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				NOT = { has_character_modifier = prison_pet_cockroach_modifier } 
			}
		}
		flag = cockroach
		localization = POD_control_creature_interaction_cockroach
	}

	send_options_exclusive = yes

	on_accept = {
		scope:actor = { POD_willpower_rouse_check_effect = yes }
		scope:actor = {
			if = {
				limit = {
					scope:cat = yes
				}
				start_cat_story_cycle_effect = yes	
			}	
			
			if = {
				limit = {
					scope:dog = yes
				}
				start_dog_story_cycle_effect = yes
				
			}	
			
			if = {
				limit = {
					scope:horse = yes
				}
				start_warhorse_story_cycle_effect = yes
			}
			
			if = {
				limit = {
					scope:wolf = yes
				}
				add_character_modifier = {
					modifier = exotic_pet_wolf
					years = 10
				}
			}
			
			if = {
				limit = {
					scope:bear = yes
				}
				add_character_modifier = {
					modifier = exotic_pet_bear
					years = 10
				}
			}
			
			if = {
				limit = {
					scope:lion = yes
				}
				add_character_modifier = {
					modifier = exotic_pet_lion
					years = 10
				}
			}
			
			if = {
				limit = {
					scope:tiger = yes
				}
				add_character_modifier = {
					modifier = exotic_pet_tiger
					years = 10
				}
			}
			
			if = {
				limit = {
					scope:rat = yes
				}
				add_character_modifier = {
					modifier = prison_pet_rat_modifier
					years = 10
				}
			}
			
			if = {
				limit = {
					scope:cockroach = yes
				}
				add_character_modifier = {
					modifier = prison_pet_cockroach_modifier
					years = 10
				}
			}
		}	
	}
	
	auto_accept = yes
	
	ai_frequency = 120
	
	ai_targets = {
		ai_recipients = self
	}

	ai_will_do = {
		base = 0
		modifier = {
			add = 100
			NOT = {
				scope:actor = { has_character_modifier =  cat_story_modifier  }
				scope:actor = { has_character_modifier =  dog_story_modifier  }
			}
		}
	}
}

POD_train_possessed_power_interaction = {
	category = interaction_category_possessed
    desc = POD_train_possessed_power_interaction_desc
	common_interaction = yes
	
	cooldown = { months = 12 }
	cost = { piety = medium_piety_value }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		OR = {
			AND = {
				scope:actor = { has_trait = fomori }
				scope:recipient = { has_trait = fomori }
			}
			AND = {
				scope:actor = { has_trait = drone }
				scope:recipient = { has_trait = drone }
			}
			AND = {
				scope:actor = { has_trait = gorgon }
				scope:recipient = { has_trait = gorgon }
			}
			AND = {
				scope:actor = { has_trait = kami }
				scope:recipient = { has_trait = kami }
			}
		}
	}

	is_valid_showing_failures_only = {
		custom_description = {
            object = scope:actor
            text = "I_must_be_at_the_same_location"
			scope:actor = { exists = location }
			scope:recipient = { exists = location }
            scope:actor = { location = scope:recipient.location }
        }
	    scope:actor = {
			possessedpower_lifestyle_perk_points >= 1
		}
		scope:recipient = { is_imprisoned = no }
	}
	
	on_accept = {
	    scope:actor = { 
			add_possessedpower_lifestyle_perk_points = -1
			stress_impact = {
				base = medium_stress_gain
			}
		}			
	    scope:recipient = { 
			add_possessedpower_lifestyle_perk_points = 1 
			add_opinion = {
				target = scope:actor
				modifier = mentored_me_opinion
			}
		}	
	}

	ai_frequency = 0
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
