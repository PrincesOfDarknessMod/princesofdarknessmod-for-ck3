POD_healing_bandages_interaction = {
	category = interaction_category_mummy
	common_interaction = yes
	interface_priority = 110
	desc = POD_healing_bandages_interaction_desc
	options_heading = POD_healing_bandages_interaction_options_heading	

	use_diplomatic_range = no

	cooldown = { months = 3 }	
	
	is_shown = {
		scope:actor = { 
			has_trait = mummy
			energy_points_value >= 1
			has_perk = wrapping_of_imhotep_perk
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			NOR = {
				has_trait_with_flag = can_not_cure
				has_character_modifier = eye_of_sakhmet_modifier
			}
			OR = {
				AND = {
					has_trait = scarred
					NOT = { has_character_flag = POD_permanent_scarring }
				}
				
				has_trait = wounded_1
				has_trait = wounded_2
				has_trait = wounded_3
	
				AND = {
					has_trait = disfigured
					NOT = { has_character_flag = POD_permanent_disfigured }
				}
			}
		}		
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				has_perk = wrapping_of_imhotep_perk
			}
			scope:recipient = { 
				has_trait = scarred
				NOT = { has_character_flag = POD_permanent_scarring }
			}
		}	
		flag = scarred
		localization = POD_yangchi_heal_scarred
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				has_perk = wrapping_of_imhotep_perk
			}
			scope:recipient = { 
				OR = {
					has_trait = wounded_1
					has_trait = wounded_2
				}
			}
		}	
		flag = wounded
		localization = POD_yangchi_heal_wounded
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				has_perk = wrappings_of_osiris_perk
			}
			scope:recipient = { 
				has_trait = wounded_3
			}
		}	
		flag = very_wounded
		localization = POD_yangchi_heal_wounded
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				has_perk = wrappings_of_osiris_perk
			}
			scope:recipient = { 
				has_trait = disfigured
				is_kueijin_trigger = yes
				NOT = { has_character_flag = POD_permanent_disfigured }
			}
		}	
		flag = disfigured
		localization = POD_yangchi_heal_disfigured
	}
	
	on_accept = {
		scope:actor = { 
			POD_willpower_rouse_check_effect = yes
		}
		scope:recipient = {
			play_music_cue = "mx_cue_peace_ensues"

			if = {
				limit = {
					scope:scarred = yes
				}
				scope:recipient = {
					remove_trait = scarred
				}
				scope:actor = { POD_hekau_cost_effect = { COUNT = 1 } }
			}	
			
			if = {
				limit = {
					OR = {
						scope:wounded = yes
						scope:very_wounded = yes
					}
				}
				scope:recipient = {
					remove_trait = wounded_3
					remove_trait = wounded_2
					remove_trait = wounded_1
				}
				scope:actor = { POD_hekau_cost_effect = { COUNT = 2 } }
			}		

			if = {
				limit = {
					scope:disfigured = yes
				}
				scope:recipient = {
					remove_trait = disfigured
				}
				scope:actor = { POD_hekau_cost_effect = { COUNT = 3 } }
			}	
		}
	}
	
	auto_accept = yes
	ai_targets = {
		ai_recipients = self
	}
	ai_frequency = 12
	ai_will_do = {
		base = 100
		
	}
}
