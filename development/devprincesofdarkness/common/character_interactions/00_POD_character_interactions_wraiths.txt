POD_possess_interaction = {
	category = interaction_category_wraith
	cooldown = { years = 1 }
	desc = POD_possess_interaction_desc
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { is_adult = yes }
		scope:actor = {has_trait = wraith} 
		scope:actor = { is_ai = no }
		NOT = {
			scope:recipient = {
				has_opinion_modifier = {
					target = scope:actor
					modifier = POD_complete_possession_opinion
				}
			}
		}
		trigger_if = {
			limit = { scope:actor = { has_character_flag = in_possession } }
			scope:actor = { var:puppet = scope:recipient }
		}
			
		trigger_if = {
			limit = { scope:recipient = { has_character_flag = possessed } }
			scope:recipient = { var:wraith = scope:actor }
		}
	}
	
	is_valid = {
		#scope:actor = {
		##	can_start_scheme = { 
		#		type = possess 
		#		target = scope:recipient
		#	}
		#}
		
	}
	on_accept = {
		scope:recipient = {
			# This event exists 
			trigger_event = POD_wraith_possession.100
		}
		custom_tooltip = POD_possess_interaction_desc
	
	}
	
	auto_accept = yes

}

	scheme = convert_to_wraith

		scope:actor = { is_adult = yes }

			scope:actor ={ has_trait = oblivionadvanced }
			scope:actor = { has_perk = touch_of_oblivion_perk }
		}

    		NOT = { has_character_modifier = shadow_twin_modifier }

		scope:recipient = {

			start_scheme = {
				type = convert_to_wraith
				target = scope:recipient
			}
		}
		custom_tooltip = POD_convert_to_wraith_interaction_desc
	
	}
	
	auto_accept = yes

}

POD_command_wraith_interaction = {
	cooldown_against_recipient = { years = 2 }
	desc = POD_command_wraith_interaction_desc
	common_interaction = yes
	interface_priority = 90
	category = interaction_category_vampire
	
	is_shown = {
		scope:actor = { is_adult = yes }
		scope:actor = { is_ai = no }
		scope:actor = {
			OR = {
				has_trait = wraith
				has_trait = oblivionadvanced
				has_perk = oblivions_sight_perk
			}
		
		}
		scope:actor = {
			OR = {
				any_in_list = {
					variable = bound_wraiths
					exists = yes
				
				}
				any_courtier_or_guest = {
					has_trait = wraith
				}
			}
		}
		scope:recipient = {
			NOT = {has_trait = wraith}
		}
		
	
	}
	
	populate_actor_list = {
		scope:actor = {
			every_courtier_or_guest = {
				limit = {
					has_trait = wraith
				}
				add_to_list = characters
			
			}
			every_in_list = {
				limit = {is_alive = yes}
				variable = bound_wraiths
				add_to_list = characters
			}
		}
	
	
	}
	
	is_valid = {
		always = yes
	}
	
	send_option = {
		is_shown = {
			exists = scope:secondary_actor
			
		}
		is_valid = {
			scope:haunt = no
			scope:spy = no

			trigger_if = {
				limit = {scope:secondary_actor = { has_character_flag = in_possession } }
				scope:secondary_actor = { var:puppet = scope:recipient }
			}
				
			trigger_if = {
				limit = { scope:recipient = {has_character_flag = possessed } }
				scope:recipient = { var:wraith = scope:secondary_actor }
			}
			NOT = {
				scope:recipient = {
					has_opinion_modifier = {
						target = scope:secondary_actor
						modifier = POD_complete_possession_opinion
					}
				}
			}
			
		}
		flag = possess
		localization = command_wraith_interaction.possess
	
	}
	send_option = {
		is_shown = { exists = scope:secondary_actor }
		is_valid = {
			scope:spy = no
			scope:possess = no
		}
		flag = haunt
		localization = command_wraith_interaction.haunt
	
	}
	send_option = {
		is_shown = { exists = scope:secondary_actor }
		is_valid = {
			scope:haunt = no
			scope:possess = no
			scope:recipient = {
				any_secret = {
					NOT = { is_known_by = scope:actor }
				}
			}
		}
		flag = spy
		localization = command_wraith_interaction.spy
	}
	send_option = {
		is_shown = {
			exists = scope:secondary_actor
		}
		is_valid = {
			scope:actor = {
				has_usable_hook = scope:secondary_actor
			}
			OR = {
				scope:haunt = yes
				scope:possess = yes
				scope:spy = yes
			}
		}
		flag = hook
		localization = command_wraith_interaction.hook
	}

	
	send_options_exclusive = no
	
	on_accept = {
		scope:actor = { save_scope_as = necromancer }
		scope:secondary_actor = { save_scope_as = actor }
		scope:actor = { set_variable = { name = necromancer value = scope:necromancer} }
		if = {
			limit = { scope:possess = yes }
			scope:recipient = {

				trigger_event = POD_wraith_possession.100
				custom_tooltip = POD_command_wraith_possess_desc
			}
		}
		if = {
			limit = { scope:spy = yes }
			scope:recipient = {
				
				trigger_event = POD_wraith_spying.100
				custom_tooltip = POD_command_wraith_spy_desc
			}
		}
		if = {
			limit = { scope:haunt = yes }
			scope:actor = {

				trigger_event = POD_wraith_haunting.100
				custom_tooltip = POD_command_wraith_haunt_desc
			}
		}
		if = {
			limit = { scope:hook = yes }
			scope:necromancer = {
				use_hook = scope:actor
			}
		}
		if = {
			limit = { scope:secondary_actor = { is_ruler = no } }
		
			hidden_effect = {
				scope:secondary_actor = {
					set_employer = scope:recipient
					return_to_court = yes
				}
			}
			custom_tooltip = POD_command_wraith_interaction.movecourt
		
		}
		scope:necromancer = {
			add_to_variable_list = {
				name = bound_wraiths
				target = scope:secondary_actor
			}
		}
		
		
		
	}
	can_send = {
		OR = {
			scope:haunt = yes
			scope:spy = yes
			scope:possess = yes
		}
	
	}
	auto_accept = {
		custom_description = {
			text = "spending_hook"
			subject = scope:actor
			object = scope:secondary_actor
			scope:hook = yes
			scope:actor = { has_strong_hook = scope:secondary_actor }
		}
	
	}
	
	ai_accept = {
		base = -50
		opinion_modifier = { # More likely to accept if Recipient likes the Actor
			who = scope:secondary_actor
			opinion_target = scope:actor
			multiplier = 1.0
			desc = AI_OPINION_REASON
		}
		modifier = {
			add = 50
			scope:secondary_actor = {
				has_dread_level_towards = {
					target = scope:actor
					level = 1
				}
			}
			desc = INTIMIDATED_REASON
		}
		modifier = {
			add = 100
			scope:secondary_actor = {
				has_dread_level_towards = {
					target = scope:actor
					level = 2
				}
			}
			desc = COWED_REASON
		}
		modifier = {
			scope:secondary_actor.house = scope:actor.house
			add = 50
			desc = IS_SAME_HOUSE_REASON
		}
	}

}
