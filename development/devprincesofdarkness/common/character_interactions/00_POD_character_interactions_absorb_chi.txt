POD_absorb_chi_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	interface_priority = 100
	
	is_shown = {
		scope:actor = {
			is_kueijin_trigger = yes
			is_ai = no
			OR = {
				yinchi_lack_1_or_more_trigger = yes
				yangchi_lack_1_or_more_trigger = yes
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			is_imprisoned_by = scope:actor
			NOT = { has_trait = torpor }
		}
	}
	
	send_option = {
		is_shown = {
			scope:actor = {
				has_midhigh_dharma_trigger = yes
			}
			scope:recipient = {
				has_chi_reserve_trigger = yes
				# Vampire race for now
				has_vampire_blood_race_trigger = yes
			}
		}
		flag = breath
		localization = POD_absorb_chi_interaction_breath
	}
				
	# Devore flesh
	send_option = {
		flag = flesh
		localization = POD_absorb_chi_interaction_flesh
	}

	on_accept = {
		if = {
			limit = {
				scope:actor = {
					yinchi_lack_1_or_more_trigger = yes
				}
			}
			if = {
				limit = {
					scope:flesh = yes
				}
				if = {
					limit = {
						has_less_chi_reserve_than_chi_lack_from_flesh_yin_trigger = yes
					}
					POD_absorb_yinchi_from_flesh_by_reserve_effect = yes
				}
				else = {
					POD_absorb_yinchi_from_flesh_by_lack_effect = yes
				}
			}
			else_if = {
				limit = {
					scope:recipient = { is_kueijin_trigger = yes }
				}
				if = {
					limit = {
						has_less_chi_reserve_than_chi_lack_from_chipool_yin_trigger = yes
					}
					POD_absorb_yinchi_from_chipool_by_reserve_effect = yes
				}
				else = {
					POD_absorb_yinchi_from_chipool_by_lack_effect = yes
				}
			}
			else = {
				if = {
					limit = {
						has_less_chi_reserve_than_chi_lack_from_breath_yin_trigger = yes
					}
					POD_absorb_yinchi_from_breath_by_reserve_effect = yes
				}
				else = {
					POD_absorb_yinchi_from_breath_by_lack_effect = yes
				}
			}
		}
		
		if = {
			limit = {
				scope:actor = {
					yangchi_lack_1_or_more_trigger = yes
				}
			}
			if = {
				limit = {
					scope:flesh = yes
				}
				if = {
					limit = {
						has_less_chi_reserve_than_chi_lack_from_flesh_yang_trigger = yes
					}
					POD_absorb_yangchi_from_flesh_by_reserve_effect = yes
				}
				else = {
					POD_absorb_yangchi_from_flesh_by_lack_effect = yes
				}
			}
			else_if = {
				limit = {
					scope:recipient = { is_kueijin_trigger = yes }
				}
				if = {
					limit = {
						has_less_chi_reserve_than_chi_lack_from_chipool_yang_trigger = yes
					}
					POD_absorb_yangchi_from_chipool_by_reserve_effect = yes
				}
				else = {
					POD_absorb_yangchi_from_chipool_by_lack_effect = yes
				}
			}
			else = {
				if = {
					limit = {
						has_less_chi_reserve_than_chi_lack_from_breath_yang_trigger = yes
					}
					POD_absorb_yangchi_from_breath_by_reserve_effect = yes
				}
				else = {
					POD_absorb_yangchi_from_breath_by_lack_effect = yes
				}
			}
		}
	}
	
	#ai_potential = {
	#	always = no
	#}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

		scope:actor = {
			is_kueijin_trigger = yes

			OR = {
				NOT = { has_character_modifier = yinchi10_modifier }
				NOT = { has_character_modifier = yangchi10_modifier }

	}

	is_valid_showing_failures_only = {
