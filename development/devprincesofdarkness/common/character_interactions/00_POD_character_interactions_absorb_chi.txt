POD_absorb_yinchi_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	interface_priority = 88
	desc = POD_absorb_yinchi_interaction_desc
	
	cooldown = { months = 3 }	

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { is_adult = yes }
		scope:recipient = { is_adult = yes }
		scope:actor = { has_trait = kueijin }
		scope:actor = { has_perk = chi_mastery_perk }
		scope:recipient = {
			OR = {
				POD_is_wraith_trigger = yes
				has_trait = yin_imbalanced
				has_trait = yinyang_balanced
			}
		}
		scope:recipient = {
			NOT = {
				has_trait = torpor
			}
		}
		scope:actor = { is_ai = no }
	}
	
	on_accept = {
		scope:recipient = {
			if = {
				limit = {
					scope:actor = { 
						OR = { 
							has_trait = dharma5
							has_trait = dharma4
							has_trait = dharma3
							has_trait = dharma2
							has_trait = dharma1
						}
					}
				}
				add_character_flag = {
					flag = is_being_tortured
					months = 6
				}
			}
			if = {
				limit = {
					scope:actor = { has_trait = dharma3 }
				}
				add_trait = wounded_1
			}
			if = {
				limit = {
					scope:actor = { has_trait = dharma2 }
				}
				add_trait = wounded_2
			}
			if = {
				limit = {
					scope:actor = { has_trait = dharma1 }
				}
				add_trait = wounded_3
			}
		}

		scope:actor = {
			random_list = {
				80 = { 
					POD_turn_to_yin_effect = yes
				}
				20 = { 
					POD_turn_to_balanced_effect = yes
				}
			}
		}
	}
	
	#ai_potential = {
	#	always = no
	#}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
