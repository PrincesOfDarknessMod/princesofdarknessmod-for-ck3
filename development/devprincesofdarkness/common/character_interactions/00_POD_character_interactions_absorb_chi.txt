POD_absorb_chi_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	interface_priority = 100

	category = interaction_category_prison
	
	is_shown = {
		scope:recipient = {
			is_imprisoned_by = scope:actor
			NOT = { has_trait = torpor }
		}
		scope:actor = {
			is_kueijin_trigger = yes
			is_ai = no
			NOR = {
				has_character_modifier = yinchi10_modifier
				has_character_modifier = yangchi10_modifier
			}
		}
	}
	
	send_option = {
		is_shown = {
			scope:actor = {
				yinchi_lack_1_or_more_trigger = yes
				yangchi_lack_1_or_more_trigger = yes
				has_midhigh_dharma_trigger = yes
			}
			scope:recipient = {
				has_chi_reserve_trigger = yes
			}
		}
		flag = breath
		localization = POD_absorb_chi_interaction_breath
	}
				
	# Devore chi even overdose
	send_option = {
		is_shown = {
			scope:actor = {
				has_midhigh_dharma_trigger = yes
			}
			scope:recipient = {
				has_chi_reserve_trigger = yes
			}
		}
		flag = breath_overdose
		localization = POD_absorb_chi_interaction_breath_overdose
	}
				
	# Devore flesh
	send_option = {
		is_shown = {
			scope:actor = {
				yinchi_lack_1_or_more_trigger = yes
				yangchi_lack_1_or_more_trigger = yes
			}
		}
		flag = flesh
		localization = POD_absorb_chi_interaction_flesh
	}
				
	# Devore flesh
	send_option = {
		flag = flesh_overdose
		localization = POD_absorb_chi_interaction_flesh_overdose
	}

	on_accept = {
		if = {
			limit = {
				OR = {
					scope:breath_overdose = yes
					scope:flesh_overdose = yes
				}
			}
			scope:actor = {
				add_character_flag = deveore_chi_even_overdose
			}
		}
		
		if = {
			limit = {
				OR = {
					scope:flesh = yes
					scope:flesh_overdose = yes
				}
			}
			scope:recipient = {
				add_character_flag = deveored_chi_from_flesh
			}
		}
		
		POD_absorb_chi_effect = yes
	}
	
	#ai_potential = {
	#	always = no
	#}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
