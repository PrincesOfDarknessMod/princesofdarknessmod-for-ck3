POD_absorb_chi_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	interface_priority = 100
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			is_kueijin_trigger = yes
			is_ai = no
			OR = {
				yinchi_lack_1_or_more_trigger = yes
				yangchi_lack_1_or_more_trigger = yes
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			NOT = {
				has_character_modifier = celestial_barrier_modifier
			}
		}
		scope:recipient = {
			is_imprisoned_by = scope:actor
			NOR = { 
				has_trait = torpor
				has_perk = frozen_river_perk
			}
		}
	}
	
	send_option = {
		is_shown = {
			scope:actor = {
				has_midhigh_dharma_trigger = yes
			}
			scope:recipient = {
				has_chi_reserve_trigger = yes
				# Vampire race for now
				has_vampire_blood_race_trigger = yes
			}
		}
		flag = breath
		localization = POD_absorb_chi_interaction_breath
	}
				
	# Devore flesh
	send_option = {
		flag = flesh
		localization = POD_absorb_chi_interaction_flesh
	}

	on_accept = {
		if = {
			limit = {
				scope:actor = {
					yinchi_lack_1_or_more_trigger = yes
				}
			}
			if = {
				limit = {
					scope:flesh = yes
				}
				if = {
					limit = {
						has_less_chi_reserve_than_chi_lack_from_flesh_yin_trigger = yes
					}
					POD_absorb_yinchi_from_flesh_by_reserve_effect = yes
				}
				else = {
					POD_absorb_yinchi_from_flesh_by_lack_effect = yes
				}
			}
			else_if = {
				limit = {
					scope:recipient = { is_kueijin_trigger = yes }
				}
				if = {
					limit = {
						has_less_chi_reserve_than_chi_lack_from_chipool_yin_trigger = yes
					}
					POD_absorb_yinchi_from_chipool_by_reserve_effect = yes
				}
				else = {
					POD_absorb_yinchi_from_chipool_by_lack_effect = yes
				}
			}
			else = {
				if = {
					limit = {
						has_less_chi_reserve_than_chi_lack_from_breath_yin_trigger = yes
					}
					POD_absorb_yinchi_from_breath_by_reserve_effect = yes
				}
				else = {
					POD_absorb_yinchi_from_breath_by_lack_effect = yes
				}
			}
		}
		
		if = {
			limit = {
				scope:actor = {
					yangchi_lack_1_or_more_trigger = yes
				}
			}
			if = {
				limit = {
					scope:flesh = yes
				}
				if = {
					limit = {
						has_less_chi_reserve_than_chi_lack_from_flesh_yang_trigger = yes
					}
					POD_absorb_yangchi_from_flesh_by_reserve_effect = yes
				}
				else = {
					POD_absorb_yangchi_from_flesh_by_lack_effect = yes
				}
			}
			else_if = {
				limit = {
					scope:recipient = { is_kueijin_trigger = yes }
				}
				if = {
					limit = {
						has_less_chi_reserve_than_chi_lack_from_chipool_yang_trigger = yes
					}
					POD_absorb_yangchi_from_chipool_by_reserve_effect = yes
				}
				else = {
					POD_absorb_yangchi_from_chipool_by_lack_effect = yes
				}
			}
			else = {
				if = {
					limit = {
						has_less_chi_reserve_than_chi_lack_from_breath_yang_trigger = yes
					}
					POD_absorb_yangchi_from_breath_by_reserve_effect = yes
				}
				else = {
					POD_absorb_yangchi_from_breath_by_lack_effect = yes
				}
			}
		}
	}
	
	auto_accept = yes
	
	
	ai_will_do = {
		base = 0
	}
}

POD_absorb_chi_ambient_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	interface_priority = 100
	
	is_shown = {
		scope:actor = scope:recipient
		scope:actor = {
			is_kueijin_trigger = yes
			is_ai = no
			OR = {
				yinchi_lack_1_or_more_trigger = yes
				yangchi_lack_1_or_more_trigger = yes
			}
			has_wall_trigger = yes
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			has_high_dharma_trigger = yes
			wall_too_strong_trigger = no
			NOT = { has_character_modifier = celestial_barrier_modifier }
		}
	}
	
	#send_option = {
		#flag = ambient
		#localization = POD_absorb_chi_interaction_ambient
	#}
				
	on_accept = {
		if = {
			limit = {
				NOT = { 
					AND = { 
						has_perk = reaping_the_dragons_road_perk
						location = { has_building_or_higher = dragon_nest_01 }
					}
				}
			}
			POD_strengthening_the_wall_effect = yes
		}
		
		if = {
			limit = {
				yinchi_lack_1_or_more_trigger = yes
			}
			POD_clean_yinchi_effect = yes
			if = {
				limit = {
					has_character_flag = devore_chi_even_overdose
				}
				add_character_modifier = yinchi10_modifier	
				POD_yinchi_overdose_effect = yes
			}
			else = {
				POD_set_yinchi_effect = yes
			}
		}
		
		if = {
			limit = {
				yangchi_lack_1_or_more_trigger = yes
			}
			POD_clean_yangchi_effect = yes
			if = {
				limit = {
					has_character_flag = devore_chi_even_overdose
				}
				add_character_modifier = yangchi10_modifier
				POD_yangchi_overdose_effect = yes
			}
			else = {
				POD_set_yangchi_effect = yes
			}
		}
		
		POD_chi_fill_effect = yes
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_steal_chi_interaction = {
	category = interaction_category_hostile
	common_interaction = yes
	interface_priority = 100
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			has_perk = clutch_of_the_wan_kuei_perk
		}
		scope:recipient = {
			OR = {
				has_yinchi_1_or_greater_trigger = yes
				has_yangchi_1_or_greater_trigger = yes
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			NOT = {
				has_character_modifier = celestial_barrier_modifier
			}
		}
		scope:recipient = {
			is_imprisoned_by = scope:actor
			NOR = { 
				has_trait = torpor
				has_perk = frozen_river_perk
			}
		}
	}

	on_accept = {
		scope:recipient = {
			hidden_effect = {				
				POD_clean_chi_effect = yes
			}
			add_character_modifier = yinchi0_modifier
			add_character_modifier = yangchi0_modifier
		}
		if = {
			limit = {
				scope:actor = {
					has_perk = bite_of_the_mu_perk
				}
			}
			scope:actor = {
				hidden_effect = {				
					POD_clean_chi_effect = yes
					POD_set_chi_effect = yes
				}
				POD_chi_fill_effect = yes
			}
		}
	}
	
	auto_accept = yes
	
	
	ai_will_do = {
		base = 0
	}
}

POD_celestial_barrier_interaction = {
	category = interaction_category_hostile
	common_interaction = yes
	interface_priority = 100
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			has_perk = celestial_barrier_perk
		}
		scope:recipient = {
			has_chi_trigger = yes
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			NOT = {
				has_character_modifier = celestial_barrier_modifier
			}
		}
		scope:recipient = {
			NOT = { 
				has_trait = torpor
			}
		}
	}

	on_accept = {
		scope:recipient = {
			add_character_modifier = {
				modifier = celestial_barrier_modifier
				months = 1
			}
		}
		scope:actor = {
			POD_chi_decrease_effect = { COUNT = 2 }
		}
	}
	
	auto_accept = yes
	
	
	ai_will_do = {
		base = 0
	}
}
