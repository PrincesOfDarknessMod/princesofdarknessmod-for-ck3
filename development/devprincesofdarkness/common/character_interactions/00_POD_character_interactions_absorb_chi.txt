POD_absorb_chi_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	interface_priority = 100
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			is_kueijin_trigger = yes
			is_ai = no
			OR = {
				yinchi_lack_1_or_more_trigger = yes
				yangchi_lack_1_or_more_trigger = yes
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			NOT = {
				has_character_modifier = celestial_barrier_modifier
			}
		}
		scope:recipient = {
			is_imprisoned_by = scope:actor
			NOR = { 
				has_trait = torpor
				has_perk = frozen_river_perk
			}
		}
	}
	
	send_option = {
		is_shown = {
			scope:actor = {
				has_midhigh_dharma_trigger = yes
			}
			scope:recipient = {
				has_chi_reserve_trigger = yes
			}
		}
		flag = breath
		localization = POD_absorb_chi_interaction_breath
	}
				
	# Devore flesh
	send_option = {
		flag = flesh
		localization = POD_absorb_chi_interaction_flesh
	}

	on_accept = {
		if = {
			limit = {
				scope:flesh = yes
			}
			POD_absorb_chi_from_flesh_or_blood_effect = yes
		}
		else = {
			POD_absorb_chi_from_breath_effect = yes
		} 
	}
	
	auto_accept = yes
	
	
	ai_will_do = {
		base = 0
	}
}

POD_absorb_chi_ambient_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	interface_priority = 100
	
	is_shown = {
		scope:actor = scope:recipient
		scope:actor = {
			is_kueijin_trigger = yes
			is_ai = no
			OR = {
				yinchi_lack_1_or_more_trigger = yes
				yangchi_lack_1_or_more_trigger = yes
			}
			has_wall_trigger = yes
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			has_high_dharma_trigger = yes
			wall_too_strong_trigger = no
			NOT = { has_character_modifier = celestial_barrier_modifier }
		}
	}
	
	#send_option = {
		#flag = ambient
		#localization = POD_absorb_chi_interaction_ambient
	#}
				
	on_accept = {
		if = {
			limit = {
				NOT = { 
					AND = { 
						has_perk = reaping_the_dragons_road_perk
						location = { has_building_or_higher = dragon_nest_01 }
					}
				}
			}
			POD_strengthening_the_wall_effect = yes
		}
		
		if = {
			limit = {
				yinchi_lack_1_or_more_trigger = yes
			}
			hidden_effect = {
				POD_clean_yinchi_effect = yes
			}
			if = {
				limit = {
					has_character_flag = devore_chi_even_overdose
				}
				add_character_modifier = yinchi10_modifier	
				POD_yinchi_overdose_effect = yes
			}
			else = {
				POD_set_yinchi_effect = yes
			}
		}
		
		if = {
			limit = {
				yangchi_lack_1_or_more_trigger = yes
			}
			hidden_effect = {
				POD_clean_yangchi_effect = yes
			}
			if = {
				limit = {
					has_character_flag = devore_chi_even_overdose
				}
				add_character_modifier = yangchi10_modifier
				POD_yangchi_overdose_effect = yes
			}
			else = {
				POD_set_yangchi_effect = yes
			}
		}
		
		POD_chi_fill_effect = yes
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_steal_chi_interaction = {
	category = interaction_category_hostile
	common_interaction = yes
	interface_priority = 100
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			has_perk = clutch_of_the_wan_kuei_perk
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			NOT = {
				has_character_modifier = celestial_barrier_modifier
			}
		}
		scope:recipient = {
			is_imprisoned_by = scope:actor
			NOR = { 
				has_trait = torpor
				has_perk = frozen_river_perk
			}
		}
		scope:recipient = {
			has_chi_reserve_trigger = yes
		}
	}
	
	send_option = {
		flag = weapon
		localization = POD_steal_chi_interaction_weapon
	}

	send_option = {
		is_shown = {
			scope:actor = {
				has_perk = bite_of_the_mu_perk
				OR = {
					yinchi_lack_1_or_more_trigger = yes
					yangchi_lack_1_or_more_trigger = yes
				}
			}
		}
		flag = self
		localization = POD_steal_chi_interaction_self
	}

	on_accept = {
		if = {
			limit = {
				scope:weapon = yes
			}
			scope:recipient = { 
				if = {
					limit = {
						OR = { 
							is_kueijin_trigger = yes 
							POD_lesser_chi_user_trigger = yes
						}
					}
					POD_chi_decrease_effect = { COUNT = 10 }
				}
				else = {
					POD_decrease_chi_other_effect = { COUNT = 10 }
				}
			}
			
			scope:actor = {
				add_character_modifier = {
					modifier = chi_enhanced_weapon_modifier
					years = 1
				}
			}
		}
		else = {
			POD_absorb_chi_from_breath_effect = yes
		}
	}
	
	auto_accept = yes
	
	
	ai_will_do = {
		base = 0
	}
}

POD_celestial_barrier_interaction = {
	category = interaction_category_hostile
	common_interaction = yes
	interface_priority = 100
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			has_perk = celestial_barrier_perk
		}
		scope:recipient = {
			has_chi_trigger = yes
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			NOT = {
				has_character_modifier = celestial_barrier_modifier
			}
		}
		scope:recipient = {
			NOR = { 
				has_trait = torpor
				has_character_modifier = celestial_barrier_modifier
			}
		}
	}

	on_accept = {
		scope:recipient = {
			add_character_modifier = {
				modifier = celestial_barrier_modifier
				years = 1
			}
		}
		scope:actor = {
			POD_chi_decrease_effect = { COUNT = 2 }
		}
	}
	
	auto_accept = yes
	
	
	ai_will_do = {
		base = 0
	}
}
