# POD Schemes
# Summon
# Sparc

POD_summon_interaction = {
	category = interaction_category_vampire
	#common_interaction = yes
	#interface_priority = 87
	scheme = PODsummon
	
	ai_frequency = 0

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_adult = yes }
		scope:recipient = { is_ruler = no } 
		scope:actor = { is_adult = yes }
		scope:actor = { 
			OR = { 
				has_POD_vamp_blood_trigger = yes 
				has_trait = kueijin
				has_trait = dhampyr
			}
		}
		NOT = { scope:actor = { location = scope:recipient.location } }
		scope:actor = { is_ai = no }
		scope:actor = { 
			OR = {
				has_perk = summon_perk 
				has_trait = presenceadvanced
			}
		}
		POD_is_in_same_plane_trigger = yes
	}

	is_valid_showing_failures_only = {
		scope:recipient = { 
			NOT = { has_trait = torpor }
			NOT = { has_trait = effigy }
		}
		scope:actor = {
			can_start_scheme = {
				type = PODsummon
				target = scope:recipient
			}
			NOT = { has_character_modifier = bloodhunger5_modifier }
		}
		custom_description = {
            object = scope:recipient
            text = "strong_mental_defenses_trigger"
            scope:recipient = { strong_mental_defenses_trigger = no }
        }				
	}

	desc = {
		triggered_desc = {
			trigger = {
				scope:actor = {
					NOT = { is_at_war_with = scope:recipient }
					can_start_scheme = {
						type = PODsummon
						target = scope:recipient
					}

				}
			}
			desc = scheme_interaction_tt_PODsummon_approved
		}

	}
	
	on_accept = {
		scope:actor = {
			if  = {
				limit = { is_kueijin_trigger = no }
				POD_hunger_rouse_check_effect = yes
			}
			else_if = {
				limit = {
					OR = {
						has_trait = kueijin 
						has_trait = dhampyr
					}
				}
				POD_chi_decrease_effect = { COUNT = 1 }
			}
		}	
		hidden_effect = {
			scope:actor = {
				start_scheme = {
					type = PODsummon
					target = scope:recipient
				}
			}
		}
	}
	
	#ai_potential = {
	#	always = no
	#}

	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
