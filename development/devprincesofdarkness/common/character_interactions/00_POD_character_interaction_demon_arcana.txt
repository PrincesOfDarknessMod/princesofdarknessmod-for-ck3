POD_regeneration_interaction = {
	category = interaction_category_demon
	common_interaction = yes
	interface_priority = 110
	icon = POD_arcana_interaction
	desc = POD_regeneration_interaction_desc
	options_heading = POD_regeneration_interaction_options_heading

	cooldown = { months = 3 }	
	
	is_shown = {
		scope:actor = scope:recipient	
		scope:actor = { has_trait = wraith }
		scope:recipient = {
			OR = {
				AND = {
					has_trait = scarred
					NOT = { has_character_flag = POD_permanent_scarring }
				}
				
				has_trait = wounded_1
				has_trait = wounded_2
				has_trait = wounded_3
				
				has_trait = maimed
				has_trait = blind
				has_trait = one_legged
				AND = {
					has_trait = one_eyed
					NOT = { has_character_flag = POD_permanent_one_eyed }
				}
	
				AND = {
					has_trait = disfigured
					NOT = { has_character_flag = POD_permanent_disfigured }
				}
			}
		}	
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			NOT = { has_trait_with_flag = can_not_cure }
			NOT = { has_character_modifier = eye_of_sakhmet_modifier }
		}
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = scarred
				NOT = { has_character_flag = POD_permanent_scarring }
			}
		}	
		flag = scarred
		localization = POD_holy_heal_scarred
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				OR = {
					has_trait = wounded_1
					has_trait = wounded_2
					has_trait = wounded_3
				}
			}
		}	
		flag = wounded
		localization = POD_holy_heal_wounded
	}

	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = blind
			}
		}
		is_valid = {
			trigger_if = {
				limit = { scope:actor = scope:recipient }
				scope:actor = { has_perk = lizard_tail_perk }
			}
		}
		flag = blind
		localization = POD_holy_heal_blind
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = maimed
			}
		}
		is_valid = {
			trigger_if = {
				limit = { scope:actor = scope:recipient }
				scope:actor = { has_perk = lizard_tail_perk }
			}
		}
		flag = maimed
		localization = POD_holy_heal_maimed
	}	
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = one_legged
			}
		}	
		is_valid = {
			trigger_if = {
				limit = { scope:actor = scope:recipient }
				scope:actor = { has_perk = lizard_tail_perk }
			}
		}
		flag = one_legged
		localization = POD_holy_heal_one_legged
	}

	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = one_eyed
				NOT = { has_character_flag = POD_permanent_one_eyed }
			}
		}	
		is_valid = {
			trigger_if = {
				limit = { scope:actor = scope:recipient }
				scope:actor = { has_perk = lizard_tail_perk }
			}
		}
		flag = one_eyed
		localization = POD_holy_heal_one_eyed
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = disfigured
				NOT = { has_character_flag = POD_permanent_disfigured }
			}
		}	
		flag = disfigured
		localization = POD_holy_heal_disfigured
	}
	
	on_accept = {
		scope:actor = { 
			POD_wraith_willpower_check_effect = yes
		}
		scope:recipient = {
			play_music_cue = "mx_cue_peace_ensues"

			if = {
				limit = {
					scope:scarred = yes
				}
				scope:recipient = {
					remove_trait = scarred
				}
				scope:actor = { add_stress = minor_stress_impact_gain }
			}	
			
			if = {
				limit = {
					scope:wounded = yes
				}
				scope:recipient = {
					remove_trait = wounded_3
					remove_trait = wounded_2
					remove_trait = wounded_1
				}
				scope:actor = { add_stress = minor_stress_impact_gain }
			}		

			if = {
				limit = {
					scope:one_eyed = yes
				}
				scope:recipient = {
					remove_trait = one_eyed
				}
				scope:actor = { add_stress = medium_stress_impact_gain }
			}	

			if = {
				limit = {
					scope:blind = yes
				}
				scope:recipient = {
					remove_trait = blind
					add_trait = one_eyed
				}
				scope:actor = { add_stress = medium_stress_impact_gain }
			}	

			if = {
				limit = {
					scope:one_legged = yes
				}
				scope:recipient = {
					remove_trait = one_legged
				}
				scope:actor = { add_stress = major_stress_impact_gain }
			}	

			if = {
				limit = {
					scope:maimed = yes
				}
				scope:recipient = {
					remove_trait = maimed
				}
				scope:actor = { add_stress = major_stress_impact_gain }
			}

			if = {
				limit = {
					scope:disfigured = yes
				}
				scope:recipient = {
					remove_trait = disfigured
				}
				scope:actor = { add_stress = medium_stress_impact_gain }
			}	
		}
	}
	
	auto_accept = yes
	ai_targets = {
		ai_recipients = self
	}
	ai_frequency = 12
	ai_will_do = {
		base = 100
		
	}
}
