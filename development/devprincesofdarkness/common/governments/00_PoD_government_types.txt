inquisition_government = {
	create_cadet_branches = no
	rulers_should_have_dynasty = yes
	#uses_opinion_of_liege = yes

	can_get_government = { 
		OR = {
		has_trait = supehunter
		has_trait = mortal
		}
	}

	primary_holding = city_holding
	valid_holdings = { church_holding castle_holding tribal_holding }
	required_county_holdings = { castle_holding city_holding church_holding }

	preferred_religions = { 
		hunters_religion
	}
	
	vassal_contract = { clan_government_obligations }
	
	opinion_of_liege = {
		scope:vassal = {
			if = {
				limit = {
					NOT = {
						is_allied_to = scope:liege
					}
				}
				if = {
					limit = {
						is_powerful_vassal = yes
					}
					value = -30
				}
				else = {
					value = -15
				}
			}
		}
	}
	opinion_of_liege_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:vassal = {
						NOT = {
							is_allied_to = scope:liege
						}
						is_powerful_vassal = yes
					}
				}
				desc = "GOVERNMENT_CLAN_NOT_ALLIED_POWERFUL"
			}
			triggered_desc = {
				trigger = {
					scope:vassal = {
						NOT = {
							is_allied_to = scope:liege
						}
						is_powerful_vassal = no
					}
				}
				desc = "GOVERNMENT_CLAN_NOT_ALLIED"
			}
		}
	}
	
	court_generate_spouses = no
	court_generate_commanders = no
	affected_by_development = yes
	color = hsv{ 0.0 0.0  0.77 }
}
