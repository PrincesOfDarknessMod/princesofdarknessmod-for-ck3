mummy_ressurection_cost_value = {
	value = -3000

	if = {
		limit = {
			exists = dynasty
				dynasty = {
					has_dynasty_perk = ba_mummy_legacy_1
				}
			}
		add = 500
	}
	
	if = {
		limit = {
			exists = dynasty
				dynasty = {
					has_dynasty_perk = ba_mummy_legacy_2
				}
			}
		add = 500
	}	

	if = {
		limit = {
			exists = dynasty
				dynasty = {
					has_dynasty_perk = ba_mummy_legacy_3
				}
			}
		add = 500
	}

	if = {
		limit = {
			exists = dynasty
				dynasty = {
					has_dynasty_perk = ba_mummy_legacy_4
				}
			}
		add = 500
	}

	if = {
		limit = {
			exists = dynasty
				dynasty = {
					has_dynasty_perk = ba_mummy_legacy_5
				}
			}
		add = 500
	}	
}

miniscule_resurrect_piety_loss = {
	value = mummy_ressurection_cost_value
	if = {
		limit = {
			has_perk = effigy_of_sin_perk
		}
		subtract = {
			value = mummy_ressurection_cost_value
			multiply = 0.4
		}
	}
	if = {
		limit = {
			has_perk = eating_the_heart_perk
		}
		subtract = {
			value = mummy_ressurection_cost_value
			multiply = 0.2
		}
	}
	ceiling = yes
}
minor_resurrect_piety_loss = {
	value = miniscule_resurrect_piety_loss
	multiply = 2
}
medium_resurrect_piety_loss = {
	value = minor_resurrect_piety_loss
	multiply = 2
}
major_resurrect_piety_loss = {
	value = medium_resurrect_piety_loss
	multiply = 2
}
massive_resurrect_piety_loss = {
	value = major_resurrect_piety_loss
	multiply = 2
}
