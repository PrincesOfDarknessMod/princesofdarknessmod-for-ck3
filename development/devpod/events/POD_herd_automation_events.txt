namespace = POD_herd_automation

POD_herd_automation.1000 = {
	hidden = yes
	immediate = {
		if = {
			limit = {
				scope:victim = {
					OR = {
						has_character_modifier = slightbloodloss_modifier
						has_character_modifier = smallbloodloss_modifier
						has_character_modifier = majorbloodloss_modifier
						
					}
				}
			}
			trigger_event = { id = POD_herd_automation.1000 months = 1}
		}
		else_if = { 
			limit = {
				scope:victim = { has_character_flag = fed_on_auto_sip }
			}
			scope:actor = {POD_resonance_feeding_herd_sip_effect = yes}
			POD_hunger_feeding_bloodloss_effect = {
				VICTIM = scope:victim
				COUNT = 1
				VAMPIRE = scope:actor
			
			
			}
			scope:actor = {POD_hunger_feeding_sip_effect = yes}
			trigger_event = { id = POD_herd_automation.1000 months = 1}
			
		
		}
	}

}

POD_herd_automation.4000 = {
	hidden = yes
	
	immediate = {
		if = {
			limit = {
				has_variable = resonance_count 			
			}
			if = {
				limit = {  resonance_max > { value = var:resonance_count multiply = resonance_vector }  } 
				change_variable = {
					name = resonance_count multiply = resonance_vector
				}

			}
			else = {
				set_variable = { name = resonance_count value = resonance_max }
			}
			
		}
		else = {
			set_variable = { name = resonance_count value = 2 }
		}
		trigger_event = { id = POD_herd_automation.4000 months = 1 }
	}
}

#Auto drain animals
POD_herd_automation.4010 = {
	hidden = yes
	
	immediate = {
		if = {
			limit = {
				NOT = { has_character_flag = autodrinkanimalsflag }
				}
			add_character_flag = autodrinkanimalsflag
			}
			
			send_interface_toast = {
				title = POD_herd_automation.4010.toast
				left_icon = root
			random_list = {
				1 = { add_animal_lifestyle_xp = miniscule_lifestyle_xp }
				10 = { add_animal_lifestyle_xp = minor_lifestyle_xp }
				40 = { add_animal_lifestyle_xp = medium_lifestyle_xp }
				}
			add_stress = 10
				}
				
		trigger_event = { id = POD_herd_automation.4011 months = 6 }
	}
}
