namespace = POD_gift_of_melancholia

POD_gift_of_melancholia.100 = {
	type = character_event 
	title = POD_gift_of_melancholia.100.t
	desc = POD_gift_of_melancholia.100.desc
	theme = cappadocian_coa
	
	left_portrait = {
	    character = scope:attacker
		animation = scheme
	
	}
	
	right_portrait = {
	    character = scope:target
		animation = sick
	}

	immediate = {
	    scope:actor = {
		    save_scope_as = attacker
		}
		scope:recipient = {
		    save_scope_as = target
		}
	}

	option = {
		name = POD_gift_of_melancholia.100.option.a
		POD_hunger_rouse_check_effect = yes
		POD_very_high_risk_of_increasing_masquerade_exposure = yes

		random_list = {
			10 = {
				modifier = {
					add = scope:attacker.learning
				}
				custom_tooltip = POD_gift_of_melancholia_interaction_success
				set_variable = { name = gift_of_melancholia_waves_remaining value = 5 }

				trigger_event = POD_gift_of_melancholia.200

				scope:actor = {
					hidden_effect = {
						scope:attacker = {
	     					send_interface_toast = {
	        					title = POD_gift_of_melancholia_actor_toast
	        					left_icon = scope:attacker
	        					right_icon = scope:target
	        					custom_tooltip = POD_gift_of_melancholia_actor_toast.desc
	    					}
						}
					}
				}
				
			}
			0 = {
				modifier = {
					add = scope:target.learning
				}
				custom_tooltip = POD_gift_of_melancholia_interaction_failure
				scope:actor = {
					hidden_effect = {
						send_interface_toast = {
							title = POD_gift_of_melancholia_failure_toast
							left_icon = scope:attacker
							right_icon = scope:target
							custom_tooltip = POD_gift_of_melancholia_failure_toast.desc
					
						}
					}
				}
			
			}
		}
	}

	option = {
		name = POD_gift_of_melancholia.100.option.b
		custom_tooltip = POD_gift_of_melancholia.100.option.b.tt
	}
}


POD_gift_of_melancholia.200 = {
	hidden = yes
	type = character_event
	
	immediate = {
		scope:target = {
	        capital_county = {
                add_county_modifier = {
			        modifier = POD_gift_of_melancholia_power_modifier
				    days = { 100 150 }
			    }
		    }
			send_interface_toast = {
			    title = POD_gift_of_melancholia_enemy_toast
				left_icon = scope:target
		        right_icon = scope:attacker
		    }
        }
        trigger_event = {
			id = POD_gift_of_melancholia.300
			days = { 25 35 }
		}
	}
}

POD_gift_of_melancholia.300 = {
	hidden = yes
	type = character_event

	immediate = {
		if = {
			limit = {
				var:gift_of_melancholia_waves_remaining > 0
			}
			every_county = {
				limit = {
					any_neighboring_county = { has_county_modifier = POD_gift_of_melancholia_power_modifier }
					NOT = { has_county_modifier = POD_gift_of_melancholia_power_modifier }
				}
				random_list = {
					10 = {
						add_county_modifier = {
			    			modifier = POD_gift_of_melancholia_power_modifier
							days = { 100 150 }
		    			}
		    		}
		    		30 = {
						add_county_modifier = {
			    			modifier = POD_gift_of_melancholia_power_modifier
							days = { 90 120 }
		    			}
		    		}
		    		20 = {
						add_county_modifier = {
			    			modifier = POD_gift_of_melancholia_power_modifier
							days = { 60 90 }
		    			}
		    		}
		    		40 = {

		    		}
		    	}
			}
			change_variable = {
				name = gift_of_melancholia_waves_remaining
				subtract = 1
			}
			trigger_event = {
				id = POD_gift_of_melancholia.300
				days = { 25 35 }
			}
		}
		else = {
			remove_variable = gift_of_melancholia_waves_remaining
		}
	}
}
