POD_gehenna_seer.100 = {
	type = character_event
	title = POD_gehenna_seer.100.t
	desc = POD_gehenna_seer.100.desc
	theme = greatperson
	
	left_portrait = {
		character = scope:gehenna_seer
		animation = personality_irrattional
	}

	trigger = {
		has_trait = vampire
		is_imprisoned = no
		is_ai = no
		NOT = { has_trait = torpor }
		NOT = { has_character_flag = denied_gehenna_seer }
		NOT = {
			any_courtier = {
				POD_is_gehenna_seer_trigger = yes
				#is_available_ai = yes
			}
		}
	}

	immediate = {
	
		save_scope_as = gehenna_seer_liege
		
		hidden_effect = {
			POD_create_gehenna_seer_effect = yes
		}
	}

	option = { #Ok...
		name = POD_gehenna_seer.100.a
		stress_impact = {
 			base = minor_stress_impact_gain
			lunatic_1 = medium_stress_impact_gain
 			paranoid = major_stress_impact_gain
 		}
		scope:gehenna_seer = {
			set_employer = root
			add_character_flag = {
				flag = blocked_from_leaving
			}
			add_character_flag = {
				flag = POD_gehenna_seer_flag
			}
		}
 		ai_chance = {
 			base = 100
 		}
	}

	option = { # NOT Ok...
		name = POD_gehenna_seer.100.b
 		ai_chance = {
 			base = 100
			ai_value_modifier = {
				ai_rationality = 1.5
			}
 		}
		hidden_effect = {
			root = {
				add_character_flag = {
					flag = denied_gehenna_seer
					years = 50
				}
			}
			scope:gehenna_seer = {
				death = {
					death_reason = death_disappearance
				}
			}
		}
	}
}

	theme = gehennaseer
	left_portrait = {
		character = scope:gehenna_seer
		animation = personality_irrattional
	}

	immediate = {
		random_courtier = {
			limit = {
				POD_is_gehenna_seer_trigger = yes
				#is_available_ai = yes
			}
			save_scope_as = gehenna_seer
		}
	}

	option = { # Accept Prophecy
		name = POD_gehenna_seer.gen.a
		stress_impact = {
 			base = minor_stress_impact_gain
			lunatic_1 = medium_stress_impact_gain
 			paranoid = major_stress_impact_gain
 		}
		scope:gehenna_seer = {
			add_opinion = {
				target = root
				modifier = friendliness_opinion
				opinion = 10
			}
		}
		add_character_modifier = {
			modifier = gehenna_seer_03_modifier
			years = 5
		}
 		ai_chance = {
 			base = 100
 		}
	}

	option = { # Reject Prophecy
		name = POD_gehenna_seer.gen.b
 		ai_chance = {
 			base = 100
			ai_value_modifier = {
				ai_rationality = 1.5
			}
 		}
		scope:gehenna_seer = {
			add_opinion = {
				target = root
				modifier = disappointed_opinion
				opinion = -10
			}
		}
	}
}
