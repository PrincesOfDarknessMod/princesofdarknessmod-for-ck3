POD_enforce_masquerade_interaction = {
	category = interaction_category_vampire
	#common_interaction = yes
	#interface_priority = 93
    desc = POD_enforce_masquerade_interaction_desc
	
	ai_frequency = 360

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		scope:actor = { is_vampire_trigger = yes  }
		scope:recipient = { is_vampire_trigger = yes }
		scope:recipient = { is_imprisoned = no }
		scope:recipient = { liege = scope:actor }
		scope:actor.dynasty = {
		    has_dynasty_perk = ventrue_legacy_3
		}
	}
	
	is_valid_showing_failures_only = {
		custom_description = {
			object = scope:recipient
			text = "has_high_masquerade_exposure"
			scope:recipient = {
			    OR = {
				    has_character_modifier = masquerade1_modifier
					has_character_modifier = masquerade2_modifier
					has_character_modifier = masquerade3_modifier
					has_character_modifier = masquerade4_modifier
					has_character_modifier = masquerade5_modifier
				}
			}
		}
	}

	cooldown = { years = 15 }

	on_accept = {
	    scope:recipient = {
		    save_scope_as = target
		}	
		scope:actor = {
			save_scope_as = enforcer
		}
	    scope:actor = {
		    trigger_event = {
			    id = POD_masquerade_enforcement.001
				days = 2
			}
        }	
	}
	
	ai_targets = {
		ai_recipients = peer_vassals
		ai_recipients = vassals
	}
	
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 20
			scope:recipient = {
				has_character_modifier = masquerade2_modifier
			}
		}
		
		modifier = {
			add = 40
			scope:recipient = {
				has_character_modifier = masquerade3_modifier
			}
		}
		
		modifier = {
			add = 60
			scope:recipient = {
				has_character_modifier = masquerade4_modifier
			}
		}
		
		modifier = {
			add = 80
			scope:recipient = {
				has_character_modifier = masquerade5_modifier
			}
		}
		
		modifier = {
		    factor = 0
			scope:recipient = { is_ai = no }
		}
	}  	
}
