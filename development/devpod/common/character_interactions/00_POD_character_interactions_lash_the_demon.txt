POD_lash_the_demon_interaction = {
	category = interaction_category_vampire
	desc = POD_lash_the_demon_interaction_desc
	
	cooldown = { days = 1825 }	

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { is_ai = no }
		scope:actor = { 
			OR = {
				has_trait = kueijin 
				has_trait = dhampyr
			}
		}
		scope:actor = { has_perk = lash_the_demon_perk }
		scope:recipient = { cant_frenzy_trigger = no }
	}
	
	is_valid_showing_failures_only = {
		custom_description = {
            object = scope:actor
            text = "soul_demon_incompatible"
            scope:actor = { using_demon_arts_trigger = no }
        }
		custom_description = {
            object = scope:actor
            text = "I_must_be_at_the_same_location"
			scope:actor = { exists = location }
			scope:recipient = { exists = location }
            scope:actor = { location = scope:recipient.location }
        }
		custom_description = {
            object = scope:recipient
            text = "strong_mental_defenses_trigger"
            scope:recipient = { strong_mental_defenses_trigger = no }
        }
	}
	
	on_accept = {
		scope:actor = { POD_chi_decrease_effect = { COUNT = 2 } }	
		scope:actor = { POD_willpower_rouse_check_effect = yes }
		scope:recipient = {
			custom_tooltip = POD_forced_to_frenzy.tt
			if = {
				limit = {
					has_trait = kueijin
				}
				add_character_flag = forced_to_frenzy
				trigger_event = stress_threshold.1741
				remove_character_flag = forced_to_frenzy
			}
			else_if = {
				limit = {
					has_trait = vampire
				}
				add_character_flag = forced_to_frenzy
				trigger_event = POD_vampire_frenzy_events.001
				remove_character_flag = forced_to_frenzy
			}
			else = {
				trigger_event = { on_action = stress_threshold_level_1_event days = 0 }
			}
		}
	}
	
	auto_accept = yes
	
	#ai_potential = {
	#	always = no
	#}
	
	ai_will_do = {
		base = 0
	}
}

POD_po_manipulation_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	interface_priority = 100
	desc = POD_po_manipulation_interaction_desc
	options_heading = POD_po_manipulation_interaction_options_heading
	
	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { is_ai = no }
		scope:actor = { 
			OR = {
				has_trait = kueijin 
				has_trait = dhampyr
			}
		}
		scope:actor = { has_perk = cleave_the_demon_perk }
	}

	is_valid_showing_failures_only = {
		custom_description = {
            object = scope:actor
            text = "soul_demon_incompatible"
            scope:actor = { using_demon_arts_trigger = no }
        }
	}
	
	send_option = {
		flag = cleave_the_demon
		is_shown = {
			scope:actor = { has_perk = yoke_the_demon_perk }
			scope:actor = { NOT = { has_character_modifier = POD_cleave_the_demon_modifier } }
		}
		localization = POD_po_manipulation_interaction_cleave_the_demon
	}
	
	send_option = {
		is_shown = {
			scope:actor = { has_perk = yoke_the_demon_perk }
			scope:actor = { NOT = { has_character_modifier = POD_yoke_the_demon_modifier } }
		}
		flag = yoke_the_demon
		localization = POD_po_manipulation_interaction_yoke_the_demon
	}
	
	send_option = {
		is_shown = {
			scope:actor = { has_perk = two_become_one_perk }
			scope:actor = { NOT = { has_character_modifier = POD_two_become_one_modifier } }
		}
		flag = two_become_one
		localization = POD_po_manipulation_interaction_two_become_one
	}
	
	send_option = {
		is_shown = {
			scope:actor = { has_perk = strengthening_the_demon_perk }
		}
		is_valid = {
			custom_description = {
				object = scope:actor
				text = "I_am_full_of_chi"
				scope:actor = {
					OR = {
						yinchi_lack_1_or_more_trigger = yes
						yangchi_lack_1_or_more_trigger = yes
					}
				}
			}
		}
		flag = strengthening_the_demon
		localization = POD_po_manipulation_interaction_strengthening_the_demon
	}
	
	send_options_exclusive = yes

	on_accept = {
		scope:actor = { POD_willpower_rouse_check_effect = yes }
		
		if = {
			limit = {
				scope:cleave_the_demon = yes
			}
			scope:actor = { POD_chi_decrease_effect = { COUNT = 1 } }	
			scope:recipient = {
				add_character_modifier = {
					modifier = POD_cleave_the_demon_modifier
					days = 1825					
				}
			}
		}
		
		if = {
			limit = {
				scope:yoke_the_demon = yes
			}
			scope:actor = { POD_chi_decrease_effect = { COUNT = 2 } }	
			scope:recipient = {
				add_character_modifier = {
					modifier = POD_yoke_the_demon_modifier
					days = 1825					
				}
			}
		}
		
		if = {
			limit = {
				scope:two_become_one = yes
			}
			scope:actor = { POD_chi_decrease_effect = { COUNT = 3 } }	
			scope:recipient = {
				add_character_modifier = {
					modifier = POD_two_become_one_modifier
					days = 1825					
				}
			}
		}
		
		if = {
			limit = {
				scope:strengthening_the_demon = yes
			}
			scope:actor = { add_stress = 250 }
			scope:recipient = {
				POD_reset_chi_effect = yes
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
