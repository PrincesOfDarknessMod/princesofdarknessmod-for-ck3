# POD Schemes
# Blood Bond, for simplicity sake it is a major hook that lasts 6 months
# Thaumaturge

POD_bloodbond_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	#interface_priority = 88
	scheme = PODbloodbond
	
	ai_frequency = 0

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		scope:actor = { is_vampire_trigger = yes  }
		scope:actor = { POD_is_thinblooded_trigger = no }
		scope:recipient = { has_POD_vamp_blood_trigger = yes } 
		scope:actor = { is_ai = no }
		POD_is_in_same_plane_trigger = yes
		scope:recipient = { 
			NOT = {
				has_trait = torpor
			}
		}
		scope:recipient = { POD_is_clone_trigger = no }
		scope:actor = { 
			NOT = { 
				has_hook_of_type = {
					target = scope:recipient
					type = strong_bloodoath_hook
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		custom_description = {
            object = scope:actor
            text = "I_must_be_at_the_same_location"
			scope:actor = { exists = location }
			scope:recipient = { exists = location }
            scope:actor = { location = scope:recipient.location }
        }
		custom_description = {
            object = scope:recipient
            text = "strong_mental_defenses_trigger"
            scope:recipient = { strong_mental_defenses_trigger = no }
        }		
		scope:actor = {
			can_start_scheme = {
				type = PODbloodbond
				target = scope:recipient
			}
			NOT = { has_character_modifier = bloodhunger5_modifier }
		}
		can_bloodbond_trigger = {
			REGENT = scope:actor
			THRALL = scope:recipient
		}
		scope:recipient = { is_landed = no }		
	}

	desc = {
		triggered_desc = {
			trigger = {
				scope:actor = {
					NOT = { is_at_war_with = scope:recipient }
					can_start_scheme = {
						type = PODbloodbond
						target = scope:recipient
					}

				}
			}
			desc = scheme_interaction_tt_POD_bloodbond_approved
		}

	}

	on_accept = {
	
		hidden_effect = {
			scope:actor = {
				start_scheme = {
					type = PODbloodbond
					target = scope:recipient
				}
			}
		}
		
		if = {
			limit = {
				OR = {
					scope:actor = {has_relation_best_friend = scope:recipient}
					scope:actor = {has_relation_friend = scope:recipient}
					scope:actor = {has_relation_lover = scope:recipient}
				}
			}
			scope:actor = {
				stress_impact = {
					compassionate = medium_stress_impact_gain
				}
			}
		}
		
		if = {
			limit = {
					scope:actor = {
						faith = {has_doctrine_parameter = vampirebloodoath_illegal }
					}
			}
			scope:actor = {add_piety = major_piety_loss}
		}
		
		else_if = {
			limit = {
				scope:actor = {
					faith = {has_doctrine_parameter = vampirebloodoath_shunned} 
				}
			}
			scope:actor = {add_piety = medium_piety_loss}
		}
	}
	
	#ai_potential = {
	#	always = no
	#}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
