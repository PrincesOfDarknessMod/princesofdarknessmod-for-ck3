	on_setup = {
		set_global_variable = {
			name = spawned_giangaleazzo
			value = yes
		}
	}

	on_end = {
		debug_log = "Giangaleazzo story ended on:"
		debug_log_date = yes
	}

			end_story = yes
		}

}
