POD_blood_surge_prowess_effect = {
	if = {
		limit = {
			POD_is_unable_to_surge_trigger = no
		}
		if = {
			 limit = {
				OR = {
					has_character_modifier = bloodpotency10_modifier
					has_character_modifier = bloodpotency9_modifier
				}
			}
			add_character_modifier = {
				modifier = blood_surge_prowess_5_modifier 
				days = 365
				
			}	
			POD_moderate_risk_of_increasing_masquerade_exposure = yes
		}

		if = {
			 limit = {
				OR = {
					has_character_modifier = bloodpotency8_modifier
					has_character_modifier = bloodpotency7_modifier
				}
			}
			add_character_modifier = {
				modifier = blood_surge_prowess_4_modifier 
				days = 365
			}	
			POD_small_risk_of_increasing_masquerade_exposure = yes
		}	

		if = {
			 limit = {
				OR = {
					has_character_modifier = bloodpotency6_modifier
					has_character_modifier = bloodpotency5_modifier
				}
			}
			
			add_character_modifier = {
				modifier = blood_surge_prowess_3_modifier 
				days = 365
			}
			POD_minimal_risk_of_increasing_masquerade_exposure = yes
		}

		if = {
			 limit = {
				OR = {
					has_character_modifier = bloodpotency4_modifier
					has_character_modifier = bloodpotency3_modifier
				}
			}
			
			add_character_modifier = {
				modifier = blood_surge_prowess_2_modifier 
				days = 365
			}	
		}

		if = {
			 limit = {
				OR = {
					has_character_modifier = bloodpotency2_modifier
					has_character_modifier = bloodpotency1_modifier
				}
			}
			
			add_character_modifier = {
				modifier = blood_surge_prowess_1_modifier 
				days = 365
			}	
		}
	}	
}
