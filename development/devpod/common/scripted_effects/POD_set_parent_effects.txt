POD_set_parent_effect = {
	hidden_effect = {
			if = {
				limit = { age > $PARENT$.age }
				set_local_variable = {
				name = saved_age
				value = this.age
			}
			set_age = 16
		}
	}
	
	if = {
		limit = {
			$PARENT$ = {
				is_female = no
			}
		}
		set_father = $PARENT$
	}
	else = {
		set_mother = $PARENT$
	}
	hidden_effect = { 
		if = {
			limit = { exists = local_var:saved_age }
			set_age = local_var:saved_age
		}
	}
	set_house = $PARENT$.house
}
