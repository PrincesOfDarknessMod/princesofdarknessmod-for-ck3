	african_low_nobility = {
		dna_modifiers = {
			accessory = {
				mode = add
				gene = special_accessories_earrings
				template = african_low_nobility
				range = { 0 1 } # For the randomness to work correctly
			}
		}
		weight = {
			base = 0
			modifier = {
				add = 101
				exists = this
				NOT = { is_male = yes }
				portrait_african_clothing_trigger = yes
				OR = {
					AND = {
						highest_held_title_tier >= tier_county # an above
						NOR = {
							has_dynasty = no
							AND = {
								has_government = republic_government
								primary_title.tier > 1
							}
							AND = { is_ruler = yes has_government = theocracy_government }
						}
					}
					AND = {
						exists = primary_spouse
						OR = {
							is_ruler = no
							highest_held_title_tier >= tier_barony # an above
						}
						primary_spouse = {
							highest_held_title_tier >= tier_county # an above
							NOR = {
								has_dynasty = no
								AND = {
									has_government = republic_government
									primary_title.tier > 1
								}
								AND = { is_ruler = yes has_government = theocracy_government }
							}
						}
					}
				}
			}
		}
	}
