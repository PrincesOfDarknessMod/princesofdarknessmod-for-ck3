set_arete_effect = {
	hidden_effect = { remove_arete_effect = yes }
	add_character_modifier = arete$ARETE$_modifier
}

remove_arete_effect = {
	if = {
		limit = { has_character_modifier = arete1_modifier }
		remove_character_modifier = arete1_modifier
	}
	if = {
		limit = { has_character_modifier = arete2_modifier }
		remove_character_modifier = arete2_modifier
	}
	if = {
		limit = { has_character_modifier = arete3_modifier }
		remove_character_modifier = arete3_modifier
	}
	if = {
		limit = { has_character_modifier = arete4_modifier }
		remove_character_modifier = arete4_modifier
	}
	if = {
		limit = { has_character_modifier = arete5_modifier }
		remove_character_modifier = arete5_modifier
	}
	if = {
		limit = { has_character_modifier = arete6_modifier }
		remove_character_modifier = arete6_modifier
	}
	if = {
		limit = { has_character_modifier = arete7_modifier }
		remove_character_modifier = arete7_modifier
	}
	if = {
		limit = { has_character_modifier = arete8_modifier }
		remove_character_modifier = arete8_modifier
	}
	if = {
		limit = { has_character_modifier = arete9_modifier }
		remove_character_modifier = arete9_modifier
	}
	if = {
		limit = { has_character_modifier = arete10_modifier }
		remove_character_modifier = arete10_modifier
	}
}

increase_arete_effect = {
	if = {
		limit = { arete_level = 9 }
		set_arete_effect = { ARETE = 10 }
	}
	else_if = {
		limit = { arete_level = 8 }
		set_arete_effect = { ARETE = 9 }
	}
	else_if = {
		limit = { arete_level = 7 }
		set_arete_effect = { ARETE = 8 }
	}
	else_if = {
		limit = { arete_level = 6 }
		set_arete_effect = { ARETE = 7 }
	}
	else_if = {
		limit = { arete_level = 5 }
		set_arete_effect = { ARETE = 6 }
	}
	else_if = {
		limit = { arete_level = 4 }
		set_arete_effect = { ARETE = 5 }
	}
	else_if = {
		limit = { arete_level = 3 }
		set_arete_effect = { ARETE = 4 }
	}
	else_if = {
		limit = { arete_level = 2 }
		set_arete_effect = { ARETE = 3 }
	}
	else_if = {
		limit = { arete_level = 1 }
		set_arete_effect = { ARETE = 2 }
	}
	else = {
		limit = { arete_level = 0 }
		set_arete_effect = { ARETE = 1 }
	}
	
}
