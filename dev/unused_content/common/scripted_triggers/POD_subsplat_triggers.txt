POD_check_subsplat_dual_triggers = {
    trigger_if = {
		limit = { 
			$ACTOR$ = {
                POD_check_subsplat_secondary_trigger_not = { SPLAT_TRIGGER = $SPLAT_TRIGGER$ SECONDARY_TRIGGER = $SECONDARY_TRIGGER$ }
			}
		}
        POD_check_subsplat_secondary_trigger_not = { SPLAT_TRIGGER = $SPLAT_TRIGGER$ SECONDARY_TRIGGER = $SECONDARY_TRIGGER$ }
	}
	trigger_if = {
		limit = { 
			$ACTOR$ = {
                POD_check_subsplat_secondary_trigger_yes = { SPLAT_TRIGGER = $SPLAT_TRIGGER$ SECONDARY_TRIGGER = $SECONDARY_TRIGGER$ }
			}
		}
        POD_check_subsplat_secondary_trigger_yes = { SPLAT_TRIGGER = $SPLAT_TRIGGER$ SECONDARY_TRIGGER = $SECONDARY_TRIGGER$ }
	}
}
POD_check_subsplat_secondary_trigger_yes = {
    $SPLAT_TRIGGER$ = yes
	$SECONDARY_TRIGGER$ = yes
}
POD_check_subsplat_secondary_trigger_not = {
    $SPLAT_TRIGGER$ = yes
	$SECONDARY_TRIGGER$ = no
}

# If there isn't a pre-existing trigger like is_ahrimane_trigger, the following are used
# Call within a trigger like POD_subsplat_check_religion_tag_trigger_laibon which can then be used by the injectors
POD_check_subsplat_religion_tag_trigger = {
    faith = { religion_tag = $RELIGION_TAG$ }
}
POD_check_subsplat_faith_trigger = {
    faith = faith:$FAITH$
}
POD_check_subsplat_faith_doctrine_trigger = {
    faith = { has_doctrine = $DOCTRINE$ }
}
POD_check_subsplat_culture_trigger = {
    culture = culture:$CULTURE$
}
POD_check_subsplat_culture_pillar_trigger = {
    culture = { has_cultural_pillar = $PILLAR$ }
}
