POD_prince_city_district = {
	primary_building = prince_city_01
	color = district_duchy
	
	# generic placeholder backgrounds
	background = {
		trigger = {
			culture ?= { has_graphical_east_asia_culture_group_trigger = yes }
			barony_controller ?= { POD_day_background_trigger = yes }
		}
		reference = @alley_day_asia
	}
	background = {
		trigger = { culture ?= { has_graphical_east_asia_culture_group_trigger = yes } }
		reference = @alley_night_asia
	}
	background = {
		trigger = { barony_controller ?= { POD_day_background_trigger = yes } }
		reference = @alley_day
	}
	background = {
		reference = @alley_night
	}

	type = duchy_capital

	can_be_replaced = yes

	is_enabled = {
		scope:holder = { POD_duchy_building_requirement_duchy_princedom = yes }
	}

	can_construct_potential = {
		scope:holder = { POD_duchy_building_requirement_duchy_princedom = yes }
	}
}
