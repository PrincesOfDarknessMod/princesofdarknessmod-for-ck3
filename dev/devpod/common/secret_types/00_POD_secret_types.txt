#Note! Should only be added with "give_diablerist_secret_or_trait_effect"
secret_diablerist = {
	category = amaranth

	is_valid = {
		secret_diablerist_is_valid_trigger = {
			OWNER = scope:secret_owner
		}
	}

	is_shunned = {
		secret_diablerist_is_shunned_trigger = {
			OWNER = scope:secret_owner
		}
	}

	is_criminal = {
		secret_diablerist_is_criminal_trigger = {
			OWNER = scope:secret_owner
		}
	}

	on_expose = {
		save_scope_as = secret
		
		secret_exposed_notification_effect = yes
		
		scope:secret_owner = {
			if = {
				limit = { NOT = { this = scope:secret_exposer } }
				trigger_event = POD_secrets.0001
			}
			
			if = {
				limit = { has_perk = fomori_deception_perk }
				remove_trait = diablerist
				add_secret = { TYPE = secret_diablerist }
			}
		}
	}
}

secret_diablerist_specific = {
	category = amaranth

	is_valid = {
		always = yes
	}

	is_shunned = {
		secret_diablerist_is_shunned_trigger = {
			OWNER = scope:secret_owner
		}
	}

	is_criminal = {
		secret_diablerist_is_criminal_trigger = {
			OWNER = scope:secret_owner
		}
	}

	on_discover = {
		scope:secret_target = { save_scope_as = victim }
		scope:secret_owner = { save_scope_as = diablerist }

		if = {
			limit = { scope:victim = { is_alive = no } }
			scope:victim = { add_knows_of_killer = scope:discoverer }
		}

		scope:discoverer = {
			if = {
				limit = {
					NOT = {
						has_character_flag = block_secret_murder_discovery_event
					}
				}

				#EVENT FOR DISCOVERER WHO IS CLOSE FAMILY/SPOUSE OF VICTIM (can get crime reason on expose)#
				if = {
					limit = {
						OR = {
							is_spouse_of_even_if_dead = scope:victim
							is_close_family_of = scope:victim
						}
						NOT = { this = scope:diablerist }
					}
					trigger_event = POD_secrets.0201
				}
			}
			else = { remove_character_flag = block_secret_murder_discovery_event }
		}
	}

	on_expose = {
		save_scope_as = secret
		scope:secret_target = {
			save_scope_as = victim
		}
		scope:secret_owner = {
			save_scope_as = diablerist
		}

		###EXPOSURE EVENTS/EFFECTS###
		
		if = {
			limit = { NOT = { exists = var:known_diablerie } }
			scope:diablerist = { trigger_event = POD_secrets.0001 } #Applies consequence effects
			scope:victim = { set_killer_public = yes }
		}
		else = {
			scope:diablerist = {
				secret_exposed_owner_effects_effect = { SECRET = scope:secret POV = scope:secret_exposer }
				secret_exposed_owner_opinion_effects_effect = yes
			}
		}
		POD_diablerie_specific_opinion_known_effect = {
			VICTIM = scope:victim
			DIABLERIST = scope:diablerist
			SECRET = scope:secret
		}
		secret_exposed_notification_effect = yes
		
		###Set murderer & death_reason###
		scope:victim = {
			set_death_reason = {
				death_reason = death_diablerie
				killer = scope:diablerist
			}
		}
		scope:secret_owner = {
			if = {
				limit = { NOT = { has_trait = diablerist } }
				add_trait = diablerist
			}
		}
	}
}
