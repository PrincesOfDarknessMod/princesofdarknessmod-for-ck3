#NOTE!
#When you add new secrets, make sure they're properly included in these scritped triggers:
# trait_is_shunned_in_faith_trigger
# trait_is_criminal_in_faith_trigger
# secret_is_shunned_in_faith_trigger
# secret_is_criminal_in_faith_trigger

#Also, all secrets should be added through their associated scripted effect/effects, not by the add_secret effect

#Note! Should only be added with "give_diablerist_secret_or_trait_effect"
secret_diablerist = {
	category = amaranth

	is_valid = {
		secret_diablerist_is_valid_trigger = {
			OWNER = scope:secret_owner
		}
	}

	is_shunned = {
		secret_diablerist_is_shunned_trigger = {
			OWNER = scope:secret_owner
		}
	}

	is_criminal = {
		secret_diablerist_is_criminal_trigger = {
			OWNER = scope:secret_owner
		}
	}

	on_expose = {
		save_scope_as = secret
		
		secret_exposed_notification_effect = yes
		
		scope:secret_owner = {
			trigger_event = POD_secrets.0001
			
			if = {
				limit = { has_perk = fomori_deception_perk }
				remove_trait = diablerist
				add_secret = { TYPE = secret_diablerist }
			}
		}
	}
}

			OWNER = scope:secret_owner

	}
