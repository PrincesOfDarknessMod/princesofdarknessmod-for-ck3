has_demon_trait_trigger = {
	OR = { 
		has_trait = truefallen
		has_trait = earthbound
	}
}

is_a_thrall = {
	num_of_relation_demonic_master > 0
}

is_a_diabolist = {
	num_of_relation_bound_demon > 0
}

is_a_host = {
	has_character_modifier = possessed_demonic_modifier
}

has_demonic_liege_trigger = {
	exists = liege
	liege = { 
		OR = {
			has_trait = demon
			is_a_thrall = yes 
		}
	} 
}

can_be_thrall = {
	NOT = { has_trait = demon }
	NOT = { has_trait = fae }
	NOT = { has_trait = wraith }
}

can_be_demonic_possesed = {
	NOT = { has_trait = demon }
	NOT = { has_trait = wraith } # As spirits, ghosts cannot suffer demonic possession.
	NOT = { has_trait = shapeshifter } # The souls of werewolves blaze with spiritual power, leaving no room for demons to enter and seize control.
	NOT = { has_trait = vampire } # The blood of Caine carries a curse imposed by God upon that monstrous race.
	NOT = { has_trait = fae } # Demons generally find the true fae too alien to possess under any circumstances.
	NOT = { has_trait = mummy }
	NOT = { has_trait = kueijin }
	NOT = { has_trait = fomori }
	NOT = { has_trait = drone }
	NOT = { has_trait = gorgon }
	NOT = { has_trait = kami }
}

has_eternal_sin_modifier_trigger = {
	exists = house
	house = {
		OR = {
			has_house_modifier = eternal_avarice_modifier
			has_house_modifier = eternal_envy_modifier
			has_house_modifier = eternal_gluttony_modifier
			has_house_modifier = eternal_lust_modifier
			has_house_modifier = eternal_pride_modifier
			has_house_modifier = eternal_sloth_modifier
			has_house_modifier = eternal_wrath_modifier
		}
	}
}

can_be_tempted_trigger = {
	trigger_if = {
		limit = {
			scope:actor = { has_faith = faith:avarice }
		}
		scope:recipient = { NOT = { has_trait = greedy } }
	}		
	trigger_else_if = {
		limit = {
			scope:actor = { has_faith = faith:envy }
		}
		scope:recipient = { NOT = { has_trait = vengeful } }
	}				
	trigger_else_if = {
		limit = {
			scope:actor = { has_faith = faith:gluttony }
		}
		scope:recipient = { NOT = { has_trait = gluttonous } }
	}			
	trigger_else_if = {
		limit = {
			scope:actor = { has_faith = faith:lust }
		}
		scope:recipient = { NOT = { has_trait = lustful } }
	}
	trigger_else_if = {
		limit = {
			scope:actor = { has_faith = faith:pride }
		}
		scope:recipient = { NOT = { has_trait = arrogant } }
	}
	trigger_else_if = {
		limit = {
			scope:actor = { has_faith = faith:sloth }
		}
		scope:recipient = { NOT = { has_trait = lazy } }
	}
	trigger_else_if = {
		limit = {
			scope:actor = { has_faith = faith:wrath }
		}
		scope:recipient = { NOT = { has_trait = wrathful } }
	}
	trigger_else = {
		always = no
	}	
}
