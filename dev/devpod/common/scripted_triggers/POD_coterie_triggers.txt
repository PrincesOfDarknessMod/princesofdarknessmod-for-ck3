# Character scope

POD_has_coterie_trigger = {
	custom_tooltip = {
		text = POD_has_coterie_trigger_tt
		exists = var:coterie
	}
}

POD_has_coterie_members_trigger = {
	custom_tooltip = {
		text = POD_has_coterie_members_trigger_tt
		var:coterie ?= {
			variable_list_size = {
				name = members
				value > 1
			}
		}
	}
}

POD_has_coterie_players_trigger = {
	var:coterie ?= {
		any_in_list = {
			variable = members
			is_ai = no
		}
	}
}

POD_has_coterie_members_present_trigger = {
	save_temporary_scope_as = coterie_leader
	var:coterie ?= {
		any_in_list = {
			variable = members
			count > 0
			NOT = { this = scope:coterie_leader }
			POD_is_in_same_journey_trigger = { TARGET = scope:coterie_leader }
		}
	}
}

POD_is_in_same_coterie_trigger = {
	exists = var:coterie
	$TARGET$ = { exists = var:coterie }
	var:coterie = $TARGET$.var:coterie
}

# For checking if a character is participating in someone else's journey
# Use in that character's scope. TARGET is the Journey leader (e.g. root, for Journey events)
POD_is_in_same_journey_trigger = {
	OR = {
		$TARGET$ = { has_variable = traveling_with_coterie }
		this = $TARGET$
	}
	OR = {
		current_travel_plan ?= { this = $TARGET$.current_travel_plan }
		NOT = { exists = $TARGET$.current_travel_plan }
	}
}

POD_coterie_at_max_size_trigger = {
	coterie_size >= coterie_max_size
}

POD_is_valid_coterie_member_trigger = {
	is_adult = yes
	POD_is_not_incapable = yes
	POD_is_not_unreachable_trigger = yes
}

# for character interactions (needs scope:actor)
POD_coterie_interaction_requirements_trigger = {
	custom_tooltip = {
		text = POD_coterie_disabled_on_travel_phase_tt
		scope:actor = {
			NOT = { has_character_flag = POD_is_in_travel_phase }
		}
	}
}

	OR = {
		secret_type = secret_sorcerer
		secret_type = secret_demonology
		secret_type = secret_infernalist
		secret_type = secret_cultist
		secret_type = secret_unbondable
	}
