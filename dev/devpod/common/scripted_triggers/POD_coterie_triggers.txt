# Character scope

POD_has_coterie_trigger = {
	has_variable = coterie
}

POD_has_coterie_members_trigger = {
	var:coterie ?= {
		variable_list_size = {
			name = members
			value > 1
		}
	}
}

POD_coterie_at_max_size_trigger = {
	coterie_size >= coterie_max_size
}

POD_is_valid_coterie_member_trigger = {
	is_adult = yes
	NOT = { has_trait = statue }
	NOT = { has_trait = wight }
	NOT = { has_trait = barghest }
	POD_is_not_clone_trigger = yes
	POD_is_not_trapped_in_umbra_basic_trigger = yes
	
	NAND = {
		has_trait = earthbound
		is_ai = yes
	}
}

# for character interactions (needs scope:actor)
POD_coterie_interaction_requirements_trigger = {
	custom_tooltip = {
		text = POD_coterie_disabled_on_travel_phase_tt
		scope:actor = {
			NOT = { has_character_flag = POD_is_in_travel_phase }
		}
	}
}
