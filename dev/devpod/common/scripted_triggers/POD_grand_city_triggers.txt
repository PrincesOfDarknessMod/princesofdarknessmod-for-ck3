POD_is_grand_city_regular_district = {
	has_building_with_flag = grand_city_primary_building
}


POD_save_grand_city_province = {
	save_temporary_scope_value_as = {
		name  = gc_type
		value = $TYPE$
	}
	"global_variable_map(grand_cities|scope:gc_type)" ?= {
		save_temporary_scope_as = $SAVE_AS$
	}
}

POD_save_grand_city_county = {
	save_temporary_scope_value_as = {
		name  = gc_type
		value = $TYPE$
	}
	"global_variable_map(grand_cities|scope:gc_type)" ?= {
		county = { save_temporary_scope_as = $SAVE_AS$ }
	}
}

# Province scope
POD_is_below_grand_city_cap = {
	custom_description = {
		text = is_below_grand_city_cap_trigger
		trigger_if = {
			limit = {
				NOT = { has_game_rule = pod_grand_city_limit_disabled }
				# circumvent the requirement while switching grand city types
				POD_can_only_be_built_in_script = no
			}
			barony_controller = {
				any_held_title = {
					count <= 1
					title_tier = barony
					title_province ?= { POD_is_grand_city_location = yes }
				}
			}
		}
		trigger_else = {
			always = yes
		}
	}
}

			employs_court_position = $POSITION$

				type = $POSITION$

					court_position = $POSITION$
					value >= $APTITUDE$
				}
			}
		}
	}

POD_can_use_constantinople = {
	barony_controller = {
		NOT = {
			has_government = fera_tribe_government
		}
	}
}

POD_province_is_plundered_constantinople = {
	has_building_or_higher = constantinople_plundered_constantinople_01
}

POD_building_requirement_plundered_constantinople = {
	POD_is_grand_city_location_constantinople = yes
	POD_can_use_constantinople = yes
	barony_controller = {
		OR = {
			has_character_flag = POD_gc_keep
			culture = {
				OR = {
					AND = {
						has_cultural_pillar = heritage_latin
						NOT = { this = culture:roman }
					}
					has_cultural_pillar = heritage_frankish
					has_cultural_pillar = heritage_iberian
					has_cultural_pillar = heritage_central_germanic
					has_cultural_pillar = heritage_west_germanic
				}
			}
		}
	}
	NOT = { has_province_modifier = POD_constantinople_plundered_modifier }
}

POD_province_is_rebuild_constantinople = {
	has_building_or_higher = constantinople_rebuild_constantinople_01
}

POD_building_requirement_rebuild_constantinople = {
	POD_is_grand_city_location_constantinople = yes
	POD_can_use_constantinople = yes
	barony_controller = {
		OR = {
			culture = { has_cultural_pillar = heritage_byzantine }
			has_culture = culture:roman
			has_culture = culture:russian
			culture = { has_cultural_pillar = heritage_ancient_greek }
			has_character_flag = POD_gc_keep
		}
	}
}

POD_province_is_reform_constantinople = {
	has_building_or_higher = constantinople_reform_constantinople_01
}

POD_building_requirement_reform_constantinople = {
	POD_is_grand_city_location_constantinople = yes
	POD_can_use_constantinople = yes
	
	OR = {
		has_province_modifier = POD_constantinople_plundered_modifier 
		
		barony_controller = {
			OR = {
				NOT = {
					culture = { 	
						OR = {
							has_cultural_pillar = heritage_frankish
							has_cultural_pillar = heritage_iberian
							has_cultural_pillar = heritage_central_germanic
							has_cultural_pillar = heritage_west_germanic
							has_cultural_pillar = heritage_byzantine
							has_cultural_pillar = heritage_ancient_greek 
							AND = {
								has_cultural_pillar = heritage_latin
								this != culture:roman
							}
							this = culture:russian 	
						}
					}
				}
				has_character_flag = POD_gc_keep
			}
		}
	}
}

	NOR = {
		AND = {
			POD_province_is_chivalric_paris = yes

				limit = { has_building_or_higher = paris_night_tournaments_01 }

					is_ventrue_trigger = no
					OR = {
						is_toreador_trigger = yes

			}

				limit = { has_building_or_higher = paris_alexander_court_01 }
				barony_controller = character:505082 #Alexander
			}
		}
		AND = {
			POD_province_is_umbral_paris = yes
			POD_building_requirement_umbral_paris = yes
		}
		AND = {
			POD_province_is_free_paris = yes
			POD_building_requirement_free_paris = yes
			trigger_if = {
				limit = { has_building_or_higher = paris_st_dennis_01 }

			}
			trigger_if = {
				limit = { has_building_or_higher = paris_free_chatelet_01 }
