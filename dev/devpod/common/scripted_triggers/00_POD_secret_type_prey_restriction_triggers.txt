ventrue_weakness_compatible_trigger = {
	trigger_if = {
		limit = {
			$RECIPIENT$ = {
				POD_is_vampire_trigger = yes
			}
		}
		always = yes
	}
	trigger_else_if = {
		limit = {
			$ACTOR$ = {
				any_secret = { secret_type = secret_prey_diplomacy }
			}
		}
		$RECIPIENT$ = {
			OR = {
				has_trait = education_diplomacy
				has_trait = diplomatic_court
				diplomacy > 20
			}
		}
	}
	trigger_else_if = {
		limit = {
			$ACTOR$ = {
				any_secret = { secret_type = secret_prey_intrigue }
			}
		}
		$RECIPIENT$ = {
			OR = {
				has_trait = education_intrigue
				has_trait = intrigue_court
				intrigue > 20
			}
		}
	}		
	trigger_else_if = {
		limit = {
			$ACTOR$ = {
				any_secret = { secret_type = secret_prey_martial }
			}
		}
		$RECIPIENT$ = {
			OR = {
				has_trait = education_martial
				has_trait = warlike_court
				martial > 20
			}
		}
	}		
	trigger_else_if = {
		limit = {

				any_secret = { secret_type = secret_prey_stewardship }
			}

			}
		}

			is_male = yes

		}
	}

		limit = {
			$ACTOR$ = {
				any_secret = { secret_type = secret_prey_middle_aged_men }
			}
		}

			is_male = yes
			effective_age > 30
			effective_age < 50
		}
	}
	trigger_else_if = {
		limit = {
			$ACTOR$ = {
				any_secret = { secret_type = secret_prey_old_men }
			}
		}

			is_male = yes

		}
	}

		limit = {

				any_secret = { secret_type = secret_prey_young_women }
			}
		}

			is_male = no

	}
