ventrue_weakness_compatible_trigger = {
	trigger_if = {
		limit = {
			$RECIPIENT$ = {
				POD_is_vampire_or_abomination_trigger = yes
			}
		}
		always = yes
	}
	trigger_else_if = {
		limit = {
			$ACTOR$ = {
				any_secret = { secret_type = secret_prey_diplomacy }
			}
		}
		$RECIPIENT$ = {
			OR = {
				has_trait = education_diplomacy
				has_trait = diplomatic_court
				diplomacy > 20
			}
		}
	}
	trigger_else_if = {
		limit = {
			$ACTOR$ = {
				any_secret = { secret_type = secret_prey_intrigue }

			OR = {
				has_trait = education_intrigue
				has_trait = intrigue_court
				intrigue > 20
			}
		}
	}		
	trigger_else_if = {
		limit = {
			$ACTOR$ = {
				any_secret = { secret_type = secret_prey_martial }
			}
		}
		$RECIPIENT$ = {
			OR = {
				has_trait = education_martial
				has_trait = education_martial_prowess
				has_trait = warlike_court
				martial > 20
			}
		}
	}		
	trigger_else_if = {
		limit = {
			$ACTOR$ = {
