# The wound was dealt by swinging or firing something, so the attacker's
# equipped weapon gets a say in what kind of damage it was. Read by the
# weapon-artifact arms of POD_damage_type_is_aggravated and by the wielded-bane
# list, which is what keeps a bane sword from aggravating a poisoning.
#
# Six of these are death-only in the current content (nothing deals them as a
# wound REASON): botched_decapitation, chopped_to_pieces, cloven_in_half,
# decapitated, tournament_melee_throat, while_assassinating. They are listed
# anyway so the set reads as "weapon-shaped" rather than "whatever happens to
# be wired today", and so a future wound caller inherits the behaviour.
POD_is_death_by_weapon = {
	OR = {
		$DEATH_REASON$ = flag:death_attacked
		$DEATH_REASON$ = flag:death_battle
		$DEATH_REASON$ = flag:death_botched_decapitation
		$DEATH_REASON$ = flag:death_chopped_to_pieces
		$DEATH_REASON$ = flag:death_cloven_in_half
		$DEATH_REASON$ = flag:death_contest_duel_accident
		$DEATH_REASON$ = flag:death_contest_melee_accident
		$DEATH_REASON$ = flag:death_decapitated
		$DEATH_REASON$ = flag:death_duel
		$DEATH_REASON$ = flag:death_fight
		$DEATH_REASON$ = flag:death_piteously_cut_down
		$DEATH_REASON$ = flag:death_tournament_melee_throat
		$DEATH_REASON$ = flag:death_tournament_pas_darmes_challenge
		$DEATH_REASON$ = flag:death_tournament_pas_darmes_defend
		$DEATH_REASON$ = flag:death_while_assassinating
	}
}
