	OR = {

		has_trait = asekh_sen

			has_doctrine = special_doctrine_set_pillars_kindred
		}
		faith = {
			has_doctrine = special_doctrine_set_corrupted_kindred
		}
		faith = {

		}
	}
}

POD_can_use_or_get_hekau_level_trigger = {
	trigger_if = {
		limit = { has_trait = mummy }
		POD_mummy_memory_level >= $LEVEL$
	}
	trigger_else = {
		OR = {
			POD_splat_has_access_to_hekau = yes
			POD_splat_has_access_to_hekau_necromancy = yes
			POD_splat_has_access_to_hekau_nomenclature = yes
		}
	}
}

	}
}

POD_is_in_homeland_trigger = {
	trigger_if = {
		limit = { has_trait = shemsuheru }
		location = { in_land_of_faith_trigger = yes }
	}
	trigger_else_if = {
		limit = { has_trait = cabiri }
		OR = {
			location = { geographical_region = POD_region_greece }
			AND = {
				location = { in_land_of_faith_trigger = yes }
				is_target_in_global_variable_list = {
					name = unavailable_unique_decisions
					target = flag:contunuing_greek_tradition_decision
				}
			}
		}
	}
	trigger_else_if = {
		limit = { has_trait = wutian }
		POD_is_in_middle_kingdom_trigger = yes
	}
	trigger_else = { always = no }
}

POD_spell_of_life_can_afford = {
	piety >= {
		value = POD_mummy_spell_of_life_piety_cost
		multiply = $FACTOR$
		multiply = -1
	}
	trigger_if = {
		limit = { 
			has_trait = mummy
			has_trait = integrity
		}
		has_trait_xp = {
			trait = integrity
			value >= {
				value = POD_mummy_spell_of_life_integrity_cost
				multiply = $FACTOR$
				multiply = -1
				min = POD_mummy_spell_of_life_integrity_cost
				floor = yes
			}
		}
	}
	trigger_if = {
		limit = { 
			has_trait = mummy
		}
		has_trait_xp = {
			trait = mummy
			value >= {
				value = POD_mummy_spell_of_life_memory_cost
				multiply = $FACTOR$
				multiply = -1
				min = POD_mummy_spell_of_life_memory_cost
				floor = yes
			}
		}
	}
}

POD_spell_of_life_can_understand = {
	OR = {
		learning >= 30
		has_trait_rank = {
			trait = education_learning
			rank >= 4
		}
		POD_is_scholarly = yes
		faith = {
			has_doctrine = tenet_knowledge_above_all
		}
		has_trait = eternal_rebirth_seeker
	}
}

POD_spell_of_life_cares_about = {
	OR = {
		is_amenti_trigger = yes
		is_setite_trigger = yes
		faith = {
			has_doctrine = special_doctrine_osiris_league
		}
		POD_is_aligned_with_set = yes
		POD_is_scholarly = yes
		is_tremere_trigger = yes
		religion = religion:talmahera_religion
		AND = {
			faith = {
				has_doctrine = tenet_knowledge_above_all
			}
			exists = capital_county
			capital_county.title_province = { geographical_region = pod_lands_of_faith }
		}
		AND = {
			has_trait = supehunter
			faith = faith:judge
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:judge_research_origins_mummies_objective_decision
			}
		}
		AND = {
			has_trait = supehunter
			faith = faith:ikhwan_al_safa
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:ikhwan_al_safa_egypt_objective_decision
			}
		}
	}
}
