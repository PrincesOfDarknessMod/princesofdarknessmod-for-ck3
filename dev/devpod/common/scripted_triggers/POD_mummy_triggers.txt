		has_trait = shemsuheru
		has_trait = banemummy
		has_trait = cabiri
		has_trait = wutian
		has_trait = wukuei
	}
}

	OR = {
		has_trait = wutian
		has_trait = wukuei
	}
}

can_use_necromantic_sekhem_trigger = {
	this = character:792605
}

is_amenti_trigger = {
	OR = {
		has_trait = shemsuheru
		has_trait = banemummy
		has_trait = cabiri
	}
}

has_amenti_liege_trigger = {
	exists = liege
	liege = {
		OR = {
			is_amenti_trigger = yes
			is_a_neteru = yes
		}
	}
}

is_a_neteru = {
	num_of_relation_worshipped_reborn > 0
}

in_land_of_faith_trigger = {
	geographical_region = pod_lands_of_faith
}

POD_has_integrity = {
	has_trait = integrity
}

POD_can_use_or_get_hekau_level_trigger = {
	trigger_if = {
		limit = { has_trait = mummy }
		POD_mummy_memory_level >= $LEVEL$
	}
	trigger_else = {
		OR = {
			POD_splat_has_access_to_hekau = yes
			POD_splat_has_access_to_hekau_necromancy = yes
		}
	}
}

POD_should_have_integrity_trigger = {
	OR = {
		has_trait = shemsuheru
		has_trait = wutian
	}
}

POD_is_in_homeland_trigger = {
	trigger_if = {
		limit = { has_trait = shemsuheru }
		location = { in_land_of_faith_trigger = yes }
	}
	trigger_else_if = {
		limit = { has_trait = cabiri }
		OR = {
			location = { geographical_region = POD_region_greece }
			AND = {
				location = { in_land_of_faith_trigger = yes }
				is_target_in_global_variable_list = {
					name = unavailable_unique_decisions
					target = flag:contunuing_greek_tradition_decision
				}
			}
		}
	}
	trigger_else_if = {
		limit = { has_trait = wutian }
		POD_is_in_middle_kingdom_trigger = yes
	}
	trigger_else = { always = no }
}
