has_mummy_trait_trigger = {
	OR = {
		has_trait = shemsuheru
		has_trait = banemummy
		has_trait = cabiri
		has_trait = wutian
		has_trait = wukuei
	}
}

is_east_mummy_trigger = {
	OR = {
		has_trait = wutian
		has_trait = wukuei
	}
}

can_use_necromantic_sekhem_trigger = {
	this = character:792605
}

is_amenti_trigger = {
	OR = {
		has_trait = shemsuheru
		has_trait = banemummy
		has_trait = cabiri
	}
}

has_amenti_liege_trigger = {
	exists = liege
	liege = {
		OR = {
			is_amenti_trigger = yes
			is_a_neteru = yes
		}
	}
}

is_a_neteru = {
	num_of_relation_worshipped_reborn > 0
}

in_land_of_faith_trigger = {
	geographical_region = pod_lands_of_faith
}

POD_can_use_or_get_hekau_level_trigger = {
	OR = {
		POD_mummy_memory_level >= $LEVEL$
		has_character_modifier = wepauwet_modifier
		uses_cairo_osiris_main_reward = yes
	}
}
POD_should_have_integrity_trigger = {
	OR = {
		has_trait = shemsuheru
		has_trait = wutian
	}
}
POD_is_in_homeland_trigger = {
	trigger_if = { limit = { has_trait = shemsuheru } in_land_of_faith_trigger =  yes }
	trigger_else_if = { limit = { has_trait = cabiri } geographical_region = POD_region_greece }
	trigger_else_if = { limit = { has_trait = wutian } geographical_region = world_asia_minor }
	trigger_else = { always = no }
}
