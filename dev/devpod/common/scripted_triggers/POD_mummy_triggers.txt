can_use_judge_of_maat_interaction = {
	has_trait = judge_of_maat
	custom_tooltip = { text = "is_not_horus" NOT = { scope:recipient = character:792593 } }
	scope:recipient = { 
		has_trait = mummy 
		OR = {
			has_trait = shemsuheru
			has_trait = cabiri
		}
	}
}

can_use_yomi_mummy_interaction = {
	scope:recipient = {
		OR = {
			AND = {
				has_trait = cultist
				has_trait = kueijin
			}
			has_trait = wukuei
		}
	}
	faith = { has_doctrine = mummy_special_kingdom_yomi_doctrine }
}

	trigger_else = { always = no }
