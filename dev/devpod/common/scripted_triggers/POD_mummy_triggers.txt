	trigger_if = {
		limit = { has_trait = shemsuheru }
		location = { in_land_of_faith_trigger = yes }
	}
	trigger_else_if = {
		limit = { has_trait = cabiri }
		OR = {
			location = { geographical_region = POD_region_greece }
			AND = {
				location = { in_land_of_faith_trigger = yes }
				is_target_in_global_variable_list = {
					name = unavailable_unique_decisions
					target = flag:contunuing_greek_tradition_decision
				}
			}
		}
	}
	trigger_else_if = {
		limit = { has_trait = wutian }
		POD_is_in_middle_kingdom_trigger = yes
	}
	trigger_else_if = {
		limit = { has_trait = custom_mummy_dynasty }
		POD_in_sol_custom_homeland_trigger = yes
	}
	trigger_else = { always = no }
}

		multiply = -1
	}
	trigger_if = {
		limit = { 
			has_trait = mummy

		}
		has_trait_xp = {

POD_can_select_custom_religion_tenet = {
	$TENET$ = { save_temporary_scope_as = tenet_considered }
	
	trigger_if = {
		limit = { scope:tenet_considered.var:type = flag:homeland }
		scope:tenet_considered.var:homeland = { save_temporary_scope_as = homeland_region }
		location = { save_temporary_scope_as = current_location }
		culture = { save_temporary_scope_as = current_culture }
		custom_tooltip = {
			text = POD_can_select_custom_religion_tenet_tt
			OR = {
				location = { geographical_region = scope:homeland_region }
				POD_culture_matches_homeland = { HOMELAND = scope:tenet_considered.var:id }
			}
		}
	}
	trigger_else = {
		always = yes
	}
}
