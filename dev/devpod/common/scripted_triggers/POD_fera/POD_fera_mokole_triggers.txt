
POD_mokole_has_stream_trigger = {
	OR = {
		POD_mokole_is_mokole_mbembe_trigger = yes
		POD_mokole_is_gumagan_trigger = yes
		POD_mokole_is_makara_trigger = yes
		POD_mokole_is_zhong_lung_trigger = yes
	}
}

POD_mokole_is_mokole_mbembe_trigger = {
	custom_tooltip = {
		text = POD_mokole_is_mokole_mbembe_trigger_tooltip
		OR = {
			var:mokole_stream ?= flag:mokole_mbembe
			dynasty.var:mokole_stream ?= flag:mokole_mbembe
		}
	}
}

POD_mokole_has_assigned_archid_colour = {
	OR = {
		has_character_flag = POD_mokole_werecroc_form_01_flag
		has_character_flag = POD_mokole_werecroc_form_02_flag
		has_character_flag = POD_mokole_werecroc_form_03_flag
		has_character_flag = POD_mokole_werecroc_form_04_flag
		has_character_flag = POD_mokole_werecroc_form_05_flag

POD_mokole_can_access_streamgifts_trigger = { 
	OR = {
		POD_mokole_is_$STREAM$_trigger = yes
		has_character_flag = unlocked_$STREAM$_gifts
	}
}



POD_mokole_has_all_man_perks = { mnesis_lifestyle_perks >= 11 }
POD_mokole_has_all_beast_perks = { mnesis_lifestyle_perks >= 22 }
