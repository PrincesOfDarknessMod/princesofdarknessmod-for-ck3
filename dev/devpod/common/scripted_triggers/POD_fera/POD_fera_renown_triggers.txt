POD_fera_uses_renown_glory = {
	OR = {
		AND = {
			POD_is_werewolf_trigger = yes
			NOT = { has_religion = religion:blackspiraldancer_religion }
		}
		POD_is_corax_trigger = yes
		POD_is_kitsune_trigger = yes #They call it Chie
		POD_is_mokole_trigger = yes
		POD_is_nuwisha_trigger = yes
	}
}

#### Helper functions

POD_has_trait_xp_for_fera_rankup = {
	POD_trait_injector_trigger_fera_with_value = {
		TRIGGER = POD_has_trait_xp
		VALUE   = POD_xp_requirement_for_next_fera_rank
	}
}

POD_is_max_fera_rank = {
	POD_fera_rank_xp >= 100
}

POD_has_spendable_fera_renown_trigger = {
	trigger_if = {
		limit = { POD_fera_rank_value = 6 }
		custom_tooltip = POD_has_spendable_fera_renown_trigger_tooltip_max
	}
	trigger_else = {
		custom_tooltip = POD_has_spendable_fera_renown_trigger_tooltip
	}
	POD_spendable_fera_renown_value >= $COUNT$
}
