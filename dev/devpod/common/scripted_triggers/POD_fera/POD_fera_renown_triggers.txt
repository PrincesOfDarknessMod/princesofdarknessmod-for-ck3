


#####How to use Fera Renown:#####
##This file contains all the triggers that control which fera type uses what renown types
##
##
##Notes:
##Nagah need their own system since their method is vastly different from other fera
##The above may also apply to Ananasi but they can still use the default renown system without breaking any lore
##Hengeyokai specific renown has not been added yet. Whether it is needed or not relies on how Hengeyokai and the Beast Courts get designed in the future
##Apis, Camazotz, and Grondr have no info on their renown types, they are also extinct

POD_fera_uses_renown_glory = {
	OR = {
		AND = {
			POD_is_werewolf_trigger = yes
			NOT = { has_religion = religion:blackspiraldancer_religion }
		}
		POD_is_corax_trigger = yes
		POD_is_kitsune_trigger = yes #They call it Chie
		POD_is_mokole_trigger = yes
		POD_is_nuwisha_trigger = yes
	}
}
