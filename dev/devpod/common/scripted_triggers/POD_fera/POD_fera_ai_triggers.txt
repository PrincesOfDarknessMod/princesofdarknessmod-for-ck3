

POD_fera_ai_can_spirit_pact = {
	has_trait = fera
	is_playable_character = yes
	POD_fera_rank_value >= 3
	num_of_relation_spirit_companion <= 4
	is_ai = yes
}

POD_fera_ai_can_spirit_pact_ancestors = {
	POD_fera_ai_can_spirit_pact = yes
	
	#only 1 before rank 6
	trigger_if = {
		limit = { POD_fera_rank_value < 6 }
		NOT = {
			any_relation = {
				type = spirit_companion
				POD_is_ancestor_spirit_trigger = yes
			}
		}
	}
	#you can have 2 at rank 6
	trigger_else = {
		NOT = {
			any_relation = {
				type = spirit_companion
				count = 2
				POD_is_ancestor_spirit_trigger = yes
			}
		}
	}
}

POD_fera_ai_can_summon_spirit_pact = {
	POD_fera_ai_can_spirit_pact = yes
	POD_can_summon_spirit_trigger = yes
}

POD_fera_ai_domicile_spirit_boost = {
	domicile ?= {
		OR = {
			has_domicile_building_or_higher = mokole_spirit_guides_01
			has_domicile_building_or_higher = mokole_spirit_infuser_01
		}
	}
}


POD_fera_ai_can_be_invited_to_coterie = {
	scope:coterie_inviter = {
		POD_coterie_at_max_size_trigger = no
		POD_has_coterie_players_trigger = no
		NOT = { is_vassal_or_below_of = prev }
	}

	POD_has_coterie_trigger = no
	POD_is_valid_coterie_member_trigger = yes
	has_trait = fera

	OR = {
		top_liege = scope:coterie_inviter.top_liege
		any_neighboring_and_across_water_top_liege_realm_owner = { this = scope:coterie_inviter.top_liege }
	}
}
