
POD_splat_can_have_wyrm_taint = {
	has_trait = fera
	religion = { 
		NOR = {
			is_in_family = rf_wyrm
			is_in_family = rf_demons
			this = religion:thousandhells_religion
		}
	}
}

POD_has_wyrm_taint_value = {
	trigger_if = {
		limit = { POD_splat_can_have_wyrm_taint = yes }
		custom_description = {
			text = POD_has_wyrm_taint
			value = $COUNT$
			wyrm_taint_value >= $COUNT$
		}
	}
}
