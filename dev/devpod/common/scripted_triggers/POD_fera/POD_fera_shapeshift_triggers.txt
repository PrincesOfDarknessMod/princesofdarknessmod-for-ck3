
POD_can_shapeshift_into_fera_forms = {
	POD_has_fera_type = { TRAIT = $TRAIT$ }
	POD_splat_is_valid_for_fera_shapeshifting = yes
}

		POD_is_shapeshifted_beastform_trigger = yes

		POD_is_shapeshifted_primalbeastform_trigger = yes
		POD_is_shapeshifted_fera_giftform_trigger = yes
	}
}
