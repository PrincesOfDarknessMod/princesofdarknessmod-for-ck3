
POD_has_fera_type = {
	OR = {
		has_trait = $TRAIT$
		var:fera_type ?= $TRAIT$
	}
}

POD_has_access_to_fera_gifts = {
	POD_has_fera_type = { TRAIT = $TRAIT$ }
	POD_splat_has_access_to_any_fera_gifts = yes
}

POD_splat_has_access_to_any_fera_gifts = {
	OR = {
		has_trait = fera
		has_trait = kinfolk
		has_trait = abomination
	}
}

POD_has_inheritable_fera_type = {
	POD_has_fera_type = { TRAIT = $TRAIT$ }
	POD_splat_has_any_fera_genetics = yes
}

POD_splat_has_any_fera_genetics = {
	OR = {
		has_trait = fera
		has_trait = kinfolk
	}
}


POD_splat_is_valid_for_fera_shapeshifting = {
	OR = {
		has_trait = fera
		has_trait = abomination
		has_trait = spirit
	}
}

POD_should_have_fera_type_variable = {
	NOT = { has_trait = fera }
	OR = {
		has_trait = kinfolk
		has_trait = abomination
		has_trait = spirit
	}
}

POD_has_any_inheritable_fera_type = {
	OR = {
		has_trait_with_flag = fera_type
		AND = {
			#exists = this
			#is_alive = yes
			exists = var:fera_type
			POD_splat_has_any_fera_genetics = yes
		}
	}
}

POD_has_all_spirit_quest_artifacts = {
	POD_has_all_common_quest_artifacts = yes
	POD_has_all_masterwork_quest_artifacts = yes
	POD_has_all_famed_quest_artifacts = yes
	POD_has_all_illustrious_quest_artifacts = yes
}

POD_has_all_common_quest_artifacts = {
	any_character_artifact = { has_variable = lepers_rags }
	any_character_artifact = { has_variable = trail_branch }
	any_character_artifact = { has_variable = feather_of_the_waters }
	any_character_artifact = { has_variable = heart_of_fire }
	any_character_artifact = { has_variable = stone_of_wealth }

	trigger_if = {
		limit = { POD_is_mokole_trigger = yes }
		any_character_artifact = { has_variable = dragon_crystal }
		any_character_artifact = { has_variable = hatchlings_tail }
		any_character_artifact = { has_variable = egg_basket }
	}
}

POD_has_all_masterwork_quest_artifacts = {
	any_character_artifact = { has_variable = fish_swatter }
	any_character_artifact = { has_variable = leaf_armor }
	any_character_artifact = { has_variable = bent_vine }
	any_character_artifact = { has_variable = gnostic_bag }
	any_character_artifact = { has_variable = spirit_jar }
	any_character_artifact = { has_variable = seeing_boomerang }
	any_character_artifact = { has_variable = kurdaitcha }
	any_character_artifact = { has_variable = warshell }
	any_character_artifact = { has_variable = beggars_mantle }
	
	trigger_if = {
		limit = { POD_is_werewolf_trigger = yes }
		any_character_artifact = { has_variable = horn_of_distress }
	}
}

POD_has_all_famed_quest_artifacts = {
	any_character_artifact = { has_variable = tokoyanga }

	trigger_if = {
		limit = { 
			OR = {
				POD_is_mokole_trigger = yes
				POD_is_corax_trigger = yes
			}
		}
		any_character_artifact = { has_variable = mirror_amaterasu_omikami }
	}
}

POD_has_all_illustrious_quest_artifacts = {

	trigger_if = {
		limit = { POD_is_mokole_trigger = yes }
		any_character_artifact = { has_variable = echo_of_ages }
		any_character_artifact = { has_variable = bowl_of_mnesis }
	}
