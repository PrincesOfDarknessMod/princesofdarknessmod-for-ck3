
POD_splat_can_be_fleshcrafted = {
	POD_is_not_incorporeal_spirit = yes
}

# to be used from inside the GUI (assumes that var:fleshcraft_target is set)
POD_can_fleshcraft = {
	var:fleshcraft_target ?= {
		is_alive = yes
		POD_is_available_for_events = yes
		POD_splat_can_be_fleshcrafted = yes
	}
	POD_has_enough_blood_for_discipline = {
		AMOUNT = POD_fleshcrafting_blood_cost
	}
}

# saves the current fleshcrafting complexity as scope:complexity
# trigger defaults to "yes"
POD_fleshcrafting_save_current_complexity = {
	save_temporary_scope_value_as = {
		name  = complexity
		value = 0
	}
	
	POD_fleshcrafting_layer_injector_trigger = {
		TRIGGER = POD_fleshcrafting_add_layer_to_complexity
	}
}

POD_fleshcrafting_add_layer_to_complexity = {
	var:$LAYER$ = { save_temporary_scope_as = fc_perk_to_add }
	POD_perk_injector_trigger_$LAYER$ = {
		TRIGGER = POD_fleshcrafting_add_perk_to_complexity
	}
	#clear_saved_scope = fc_perk_to_add
}

POD_fleshcrafting_add_perk_to_complexity = {
	trigger_if = {
		limit = { scope:fc_perk_to_add = flag:$PERK$ }
		save_temporary_scope_value_as = {
			name = complexity
			value = {
				value = scope:complexity
				add = $PERK$_complexity
			}
		}
	}
}

POD_fleshcrafting_target_has_all_selected_perks = {
	custom_tooltip = {
		text = POD_fleshcrafting_need_to_select_perk
		POD_perk_injector_trigger_fleshcrafting = {
			TRIGGER = POD_fleshcrafting_target_has_selected_perk
		}
	}
}

POD_fleshcrafting_target_has_selected_perk = {
	trigger_if = {
		limit = {
			is_target_in_variable_list = {
				name = fc_all_selected_perks
				target = flag:$PERK$
			}
		}
		var:fleshcraft_target = { has_perk = $PERK$ }
	}
	trigger_else = {
		var:fleshcraft_target = { NOT = { has_perk = $PERK$ } }
	}
}

POD_has_no_fleshcrafting_perks = {
	POD_fleshcrafting_layer_injector_trigger = {
		TRIGGER = POD_has_no_fleshcrafting_perks_in_layer
	}
}

# arguments: LAYER (one of the ones from POD_fleshcrafting_layer_injector)
POD_has_no_fleshcrafting_perks_in_layer = {
	POD_perk_injector_trigger_$LAYER$ = {
		TRIGGER = POD_lacks_fleshcrafting_perk
	}
}

POD_lacks_fleshcrafting_perk = {
	NOT = { has_perk = $PERK$ }
}

POD_has_fleshcrafting_visual_perk = {
	trigger_if = {
		limit = {
			exists = this
			is_alive = yes
			has_variable = fleshcrafted_by
		}
		var:fleshcrafted_by = {
			any_in_list = {
				variable = fc_all_selected_perks
				this = flag:$PERK$
			}
		}
	}
	trigger_else = {
		has_perk = $PERK$
	}
}
