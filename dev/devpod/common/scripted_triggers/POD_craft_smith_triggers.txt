POD_artifact_empower_trigger = {
	trigger_if = {
		limit = { var:current_recipe = flag:artifact_empower }
		trigger_if = {
			limit = { is_target_in_variable_list = { name = ingredient_list target = flag:masterwork_artifact_piece } }
			var:craft_creator = { has_perk = greater_restoration_perk }
		}
		trigger_else_if = {
			limit = { is_target_in_variable_list = { name = ingredient_list target = flag:famed_artifact_piece } }
			var:craft_creator = { has_perk = princely_restoration_perk }
		}
		trigger_else = {
			var:craft_creator = { has_perk = major_restoration_perk }
		}
	}
}
