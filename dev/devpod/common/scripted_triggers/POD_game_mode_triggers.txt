POD_is_storyteller_mode_enabled = {
	has_game_rule = pod_lore_restrictions_more
}

POD_is_roguelike_mode_enabled_trigger = {
	OR = {
		has_game_rule = POD_roguelike_mode_enabled_randomized_vampire
		has_game_rule = POD_roguelike_mode_enabled
		has_game_rule = POD_roguelike_mode_enabled_no_permadeath
	}
}

POD_is_roguelike_randomization_enabled_trigger = {
	has_game_rule = POD_roguelike_mode_enabled_randomized_vampire
}

POD_has_roguelike_curse = {
	OR = {
		has_character_flag = POD_roguelike_cursed
		has_character_flag = POD_roguelike_all_curses
	}
}

POD_blocked_from_learning_trigger = {
	has_character_modifier = POD_roguelike_no_curiosity
}

POD_is_valid_roguelike_ruler_trigger = {
	sub_realm_size = 1
	trigger_if = {
		limit = { has_global_variable = roguelike_immortality_filter }
		is_immortal = yes
	}
	has_POD_resurrection_trigger = no
	is_landed = yes
	is_ai = yes
}
