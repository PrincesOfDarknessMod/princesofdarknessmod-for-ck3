POD_error_suppression_unused_flag_trigger = {
	trigger_if = {
		limit = { always = no }
		trigger_if = {
			limit = { this = flag:$FLAG$ }
		}
	}
}

POD_error_suppression_unset_flag_trigger = {
	trigger_if = {
		limit = { always = no }
		save_temporary_scope_value_as = {
			name  = target
			value = flag:$FLAG$
		}
	}
}
