can_be_thrall = {
	NOT = { has_trait = demon }
	NOT = { has_trait = fae }
	NOT = { has_trait = wraith }
	NOT = { has_trait = spirit }
}

POD_splat_can_be_infernalist = {
	NOT = { has_trait = demon }
}

POD_character_can_use_investments = {
	OR = {
		has_character_modifier = decani_boon_modifier
		has_character_flag = access_to_investiture
	}
	POD_splat_can_be_infernalist = yes
}

is_a_cultist = {
	num_of_relation_worshipped_demon > 0
}

is_a_thrall = {
	num_of_relation_demonic_master > 0
}

POD_has_pact_with_obligation_trigger = {
	any_in_list = {
		variable = demonic_pacts
		var:obligation = $OBLIGATION$
		save_temporary_scope_as = pact_detail
	}
	exists = scope:pact_detail
}

POD_pact_obligation_is_prohibitory_trigger = {
	OR = {
		$OBLIGATION$ = flag:pact_of_non_aggression
		$OBLIGATION$ = flag:soul_of_glass
		$OBLIGATION$ = flag:power_of_war
		$OBLIGATION$ = flag:power_of_pain
	}
}
