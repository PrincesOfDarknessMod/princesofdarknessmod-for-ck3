POD_save_diabolism_artifact_rarity_tooltip = {
	trigger_if = {
		limit = { POD_diabolist_perks_gained >= 6 }
		POD_save_temporary_tooltip = { LOC = POD_rarity_illustrious }
	}
	trigger_else_if = {
		limit = { POD_diabolist_perks_gained = 5 }
		POD_save_temporary_tooltip = { LOC = POD_rarity_famed }
	}
	trigger_else_if = {
		limit = { POD_diabolist_perks_gained = 4 }
		POD_save_temporary_tooltip = { LOC = POD_rarity_masterwork }
	}
	trigger_else = {
		POD_save_temporary_tooltip = { LOC = POD_rarity_common }
	}
}

# artifact scope
# arguments: DEMON
POD_artifact_can_contain_specific_demon = {
	$DEMON$ = {
		POD_save_diabolism_artifact_rarity_tooltip = yes
	}
	trigger_if = {
		limit = { scope:saved_tooltip = flag:POD_rarity_illustrious }
		rarity = illustrious
	}
	trigger_else_if = {
		limit = { scope:saved_tooltip = flag:POD_rarity_famed }
		OR = {
			rarity = illustrious
			rarity = famed
		}
	}
	trigger_else_if = {
		limit = { scope:saved_tooltip = flag:POD_rarity_masterwork }
		OR = {
			rarity = illustrious
			rarity = famed
			rarity = masterwork
		}
	}
	trigger_else_if = {
		limit = { scope:saved_tooltip = flag:POD_rarity_common } # just to make sure the saved flag is used
		always = yes
	}
	trigger_else = {
		always = yes
	}
}

# scoped to the demon
# arguments: DIABOLIST
POD_demon_can_sabotage = {
	POD_is_demon_trigger = yes
	NOT = { has_relation_diabolist = $DIABOLIST$ }
	has_opinion_modifier = {
		modifier = POD_demon_binding_opinion
		target = $DIABOLIST$
	}
	POD_is_available = yes
	has_relation_rival = $DIABOLIST$
}

# note that these will break if the macro arguments are context-dependent (like var:)
POD_has_pact_with_obligation_trigger = {
	any_in_list = {
		variable = demonic_pacts
		var:obligation = $OBLIGATION$
	}
}
POD_has_pact_with_frequency_trigger = {
	any_in_list = {
		variable = demonic_pacts
		var:frequency = $FREQUENCY$
	}
}
POD_has_pact_with_consequence_trigger = {
	any_in_list = {
		variable = demonic_pacts
		var:consequence = $CONSEQUENCE$
	}
}

POD_pact_obligation_is_prohibitory_trigger = {
	OR = {
		$OBLIGATION$ = flag:pact_of_non_aggression
		$OBLIGATION$ = flag:soul_of_glass
		$OBLIGATION$ = flag:power_of_war
		$OBLIGATION$ = flag:power_of_pain
	}
}

POD_can_select_pact_stipulation = {
	$STIPULATION$ = { save_temporary_scope_as = stipulation }
	
	trigger_if = {
		limit = {
			POD_pact_obligation_is_prohibitory_trigger = {
				OBLIGATION = scope:stipulation.var:id
			}
		}
		custom_tooltip = {
			text = demonic_pact_creator_prohibitions_must_be_unique
			NOT = {
				POD_has_pact_with_obligation_trigger = {
					OBLIGATION = scope:stipulation.var:id
				}
			}
		}
	}
	
	custom_tooltip = {
		text = demonic_pact_creator_only_one_life
		NAND = {
			scope:stipulation.var:id = flag:life
			POD_has_pact_with_consequence_trigger = {
				CONSEQUENCE = flag:life
			}
		}
	}
	
	trigger_if = {
		limit = {
			exists = var:selected_obligation
			POD_pact_obligation_is_prohibitory_trigger = {
				OBLIGATION = var:selected_obligation.var:id
			}
		}
		custom_tooltip = {
			text = demonic_pact_creator_prohibitory_frequency
			NOT = { scope:stipulation.var:type = flag:frequency }
		}
	}
	
	trigger_if = {
		limit = { scope:stipulation.var:id = flag:power_of_war }
		is_at_war = yes
	}
	trigger_else_if = {
		limit = { scope:stipulation.var:id = flag:power_of_pain }
		POD_health_has_any_wound_trigger = yes
	}
	trigger_else = {
		always = yes
	}
}

POD_can_finalize_pact_in_gui = {
	trigger_if = {
		limit = { exists = var:selected_obligation }
		POD_can_select_pact_stipulation = { STIPULATION = var:selected_obligation }
	}
	trigger_else = {
		custom_tooltip = {
			text = demonic_pact_creator_finish_script_obligation
			always = no
		}
	}
	
	trigger_if = {
		limit = { exists = var:selected_frequency }
		POD_can_select_pact_stipulation = { STIPULATION = var:selected_frequency }
	}
	trigger_else = {
		custom_tooltip = {
			text = demonic_pact_creator_finish_script_frequency
			# no need to select a frequency for prohibitory obligations
			exists = var:selected_obligation
			POD_pact_obligation_is_prohibitory_trigger = {
				OBLIGATION = var:selected_obligation.var:id
			}
		}
	}
	
	trigger_if = {
		limit = { exists = var:selected_consequence }
		POD_can_select_pact_stipulation = { STIPULATION = var:selected_consequence }
	}
	trigger_else = {
		custom_tooltip = {
			text = demonic_pact_creator_finish_script_the_fine_print
			always = no
		}
	}
}
