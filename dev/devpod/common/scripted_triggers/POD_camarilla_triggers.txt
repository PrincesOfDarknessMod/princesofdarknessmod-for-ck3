POD_can_be_justicar_candidate_trigger = {
	#faith = faith:camahumanity
	POD_is_inner_circle_trigger = no
	has_trait = vampire
	NOT = { has_trait = torpor }
	POD_is_anathema_trigger = no
	is_alive = yes
}

POD_has_higher_camarilla_status_trigger = {
	custom_description = {
		text = has_higher_camarilla_status
		object = $CHARACTER$
		camarilla_status > $CHARACTER$.camarilla_status
	}
}

POD_is_justicar_trigger = {
	custom_description = {
		text = is_justicar
		faith:camahumanity = {
			is_target_in_variable_list = { 
				name = justicars
				target = prev
			}
		}
	}
}
