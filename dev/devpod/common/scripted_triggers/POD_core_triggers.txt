# For comparing the current date with something that isn't formatted like YYYY.MM.DD
# e.g. variables or scope values
POD_compare_current_date_with_value = {
	save_temporary_scope_value_as = {
		name  = date_for_comparison
		value = current_date
	}
	scope:date_for_comparison $COMPARATOR$ $VALUE$
}
