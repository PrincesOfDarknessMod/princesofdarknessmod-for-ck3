POD_set_temporary_scope_value_from_trigger = {
	trigger_if = {
		limit = {
			$TRIGGER$
		}
		save_temporary_scope_value_as = {
			name = $NAME$
			value = yes
		}
	}
	trigger_else = {
		save_temporary_scope_value_as = {
			name = $NAME$
			value = no
		}
	}
}

# save a localization key to be used later
# e.g. with custom_tooltip = POD_saved_tooltip
POD_save_temporary_tooltip = {
	save_temporary_scope_value_as = {
		name = saved_tooltip
		value = flag:$LOC$
	}
	POD_error_suppression_unused_flag_trigger = { FLAG = $LOC$ }
}

# For comparing the current date with something that isn't formatted like YYYY.MM.DD
# e.g. variables or scope values
POD_compare_current_date_with_value = {
	save_temporary_scope_value_as = {
		name  = date_for_comparison
		value = current_date
	}
	scope:date_for_comparison $COMPARATOR$ $VALUE$
}

POD_can_receive_landed_titles = {
	trigger_if = {
		limit = { faith = { has_doctrine = doctrine_spiritual_head } }
		NOT = { any_held_title = { is_head_of_faith = yes } }
	}
	trigger_if = {
		limit = { is_ruler = yes }
		primary_title = {
			is_mercenary_company = no
			is_holy_order = no
		}
	}
	trigger_else = {
		is_diarch = no
	}
	NOT = { has_faith = faith:viaservilis }
}

POD_character_supports_you_for_objectives = {
	OR = {
		this = $CHARACTER$
		has_relation_friend = $CHARACTER$
		has_relation_best_friend = $CHARACTER$
		has_relation_lover = $CHARACTER$
		has_relation_soulmate = $CHARACTER$
		has_relation_coterie_member = $CHARACTER$
	}
}

POD_county_supports_you_for_objectives = {
	trigger_if = {
		limit = { 
			is_landless_adventurer = yes
		}
		custom_description = {
			text = POD_county_supports_you_for_objectives_desc
			object = $COUNTY$
			$COUNTY$.holder ?= { has_faith = root.faith }
		}
	}
	trigger_else = {
		completely_controls = $COUNTY$
	}
}

is_councillor_with_position_skill = {
	custom_description = {
		text = is_councillor_with_position_skill_tooltip
		value = $SKILL$
		OR = {
			AND = {
				has_council_position = councillor_chancellor
				diplomacy >= $SKILL$
			}
			AND = {
				has_council_position = councillor_steward
				stewardship >= $SKILL$
			}
			AND = {
				has_council_position = councillor_marshal
				martial >= $SKILL$
			}
			AND = {
				has_council_position = councillor_spymaster
				intrigue >= $SKILL$
			}
			AND = {
				has_council_position = councillor_court_chaplain
				learning >= $SKILL$
			}
		}
	}
}

POD_has_trait_xp = {
	trigger_if = {
		limit = { has_trait = $TRAIT$ }
		has_trait_xp = {
			trait = $TRAIT$
			value >= $VALUE$
		}
	}
}

##To show date requirements in game tooltips
#Using it as is displays dates after. use NOT = {} for dates before
POD_current_date = {
	custom_description = {
		text = POD_current_date_trigger
		value = $DATE$
		current_date >= $DATE$
	}
}
POD_current_year = {
	custom_description = {
		text = POD_current_date_trigger
		value = $DATE$
		current_year >= $DATE$
	}
}

POD_has_total_friends_trigger = {
	custom_description = {
		text = POD_has_total_friends
		value = $AMOUNT$
		save_temporary_scope_value_as = {
			name = total_friends
			value = {
				value = num_of_relation_friend
				add = num_of_relation_best_friend
			}
		}
		scope:total_friends >= $AMOUNT$
	}
}
