# The link must have been saved via POD_save_new_artifact_as_link
# need to pass LINK
POD_character_owns_linked_artifact = {
	trigger_if = {
		limit = {
			POD_artifact_link_exists = {
				LINK = $LINK$
			}
		}
		character:POD_wraith_01.var:$LINK$ = {
			custom_description = {
				text = POD_book_of_nod_decision_custom
				artifact_owner = prev
			}
		}
	}
}

POD_artifact_link_exists = {
	character:POD_wraith_01 ?= {
		exists = var:$LINK$
	}
}

POD_can_use_non_CFP_throne_rooms = {
	trigger_if = {
		limit = { exists = global_var:cfp_is_loaded }
		NOR = { 
			POD_CFP_has_graphical_papal_culture_group_trigger = yes # CFP papal
			POD_CFP_has_graphical_hre_culture_group_trigger = yes # CFP hre
			POD_CFP_has_graphical_byzantine_culture_group_trigger = yes # CFP byzantine/byzantine_mena/byzantine_pagan
			POD_CFP_has_graphical_norman_culture_group_trigger = yes # CFP norman
			POD_CFP_has_graphical_goidelic_culture_group_trigger = yes # CFP goidelic
			AND = { # CFP norse
				POD_CFP_has_graphical_norse_culture_group_trigger = yes
				NOT = { religion = religion:codeoftremere_religion }
			}
			POD_CFP_has_graphical_iberian_culture_group_trigger = yes # CFP iberian/andalusian
			AND = { # CFP west_african
				POD_CFP_has_graphical_west_african_culture_group_trigger = yes
				faith = { is_POD_muslim_religion_trigger = no }
			}
			AND = { # CFP east_african
				POD_CFP_has_graphical_east_african_culture_group_trigger = yes
				faith = { is_POD_muslim_religion_trigger = no }
			}
		}
	}
	trigger_else = {
		always = yes
	}
}

POD_uses_chinese_throne_room = {
	always = no # disabled pending a nighttime version
	has_tgp_dlc_trigger = yes
	OR = {
		capital_county ?= {
			OR = {
				culture = { has_graphical_chinese_culture_group_trigger = yes }
				# Also used for East Asian throne rooms
				title_province = { 
					geographical_region = world_asia
					NOT = { geographical_region = world_asia_japan }
					NOT = { geographical_region = world_asia_southeast }
				}
			}
		}
	}
}

###### Custom idle animations

# for animations that don't need fat/dwarf variants
POD_has_generic_idle_animation = {
	is_alive = yes
	var:custom_idle ?= $FLAG$
	NOT = { has_trait = pregnant }
}

# for animations that have fat/dwarf variants
POD_has_default_idle_animation = {
	is_alive = yes
	var:custom_idle ?= $FLAG$
	should_use_fat_animation = no
	should_use_fat_dwarf_animation = no
	NOT = { has_trait = pregnant }
}

POD_has_fat_idle_animation = {
	is_alive = yes
	var:custom_idle ?= $FLAG$
	should_use_fat_animation = yes
	NOT = { has_trait = pregnant }
}

POD_has_fat_dwarf_idle_animation = {
	is_alive = yes
	var:custom_idle ?= $FLAG$
	should_use_fat_dwarf_animation = yes
	NOT = { has_trait = pregnant }
}

POD_has_fat_or_dwarf_idle_animation = {
	is_alive = yes
	var:custom_idle ?= $FLAG$
	OR = {
		should_use_fat_animation = yes
		should_use_fat_dwarf_animation = yes
	}
	NOT = { has_trait = pregnant }
}

POD_has_potential_mortal_herd_trigger = {
	POD_is_vampire_trigger = yes
	num_of_relation_herd_member < max_herd_cap
	NOT = { any_scheme = { scheme_type = herdify_mortal } }
	any_courtier_or_guest = {
		prev = {
			is_character_interaction_valid = {
				recipient = prev
				interaction = POD_herdify_mortal_interaction
			}
		}
	}
}

##### Triggers for domicile building asset = {} blocks
# These are in descending priority

# Priority 1
POD_domicile_uses_landed_yurt_assets = {
	owner = { is_landed_or_landless_administrative = yes }
	owner = { POD_khans_of_the_steppe_event_bg_trigger = yes }
}

# Priority 2
POD_domicile_uses_landed_mena_assets = {
	owner = { is_landed_or_landless_administrative = yes }
	owner.culture = {
		OR = {
			has_graphical_iranian_culture_group_trigger = yes
			has_graphical_mena_culture_group_trigger = yes
			has_graphical_steppe_culture_group_trigger = yes
			has_graphical_african_culture_group_trigger = yes
			has_graphical_india_culture_group_trigger = yes
			has_graphical_iberian_culture_group_trigger = yes
		}
	}
}

# Priority 3
POD_domicile_uses_landed_western_assets = {
	owner = { is_landed_or_landless_administrative = yes }
	owner.culture = {
		OR = {
			has_graphical_western_culture_group_trigger = yes
			has_graphical_norse_culture_group_trigger = yes
		}
	}
}

# Priority 4
# used as a fallback for landed domiciles
POD_domicile_uses_landed_byzantine_assets = {
	owner = { is_landed_or_landless_administrative = yes }
}
