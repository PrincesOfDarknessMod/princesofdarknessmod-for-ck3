# The link must have been saved via POD_save_new_artifact_as_link
# need to pass LINK and CHARACTER
POD_character_owns_linked_artifact = {
	trigger_if = {
		limit = {
			POD_artifact_link_exists = {
				LINK = $LINK$
			}
		}
		character:POD_wraith_01.var:$LINK$ = {
			custom_description = {
				text = POD_book_of_nod_decision_custom
				artifact_owner = $CHARACTER$
			}
		}
	}
}

POD_artifact_link_exists = {
	character:POD_wraith_01 ?= {
		exists = var:$LINK$
	}
}

POD_use_nighttime_event_bg = {
	OR = {
		POD_day_background_trigger = no
		trigger_if = {
			limit = { exists = scope:bg_override_char }
			scope:bg_override_char = { POD_day_background_trigger = no }
		}
	}
}

POD_has_potential_mortal_herd_trigger = {
	POD_is_vampire_trigger = yes
	num_of_relation_herd_member < max_herd_cap
	NOT = { any_scheme = { scheme_type = herdify_mortal } }
	any_courtier_or_guest = {
		prev = {
			is_character_interaction_valid = {
				recipient = prev
				interaction = POD_herdify_mortal_interaction
			}
		}
	}
}

POD_has_potential_dreamer_trigger = {
	can_gain_glamour_from_dreamers_trigger = yes

				recipient = prev
