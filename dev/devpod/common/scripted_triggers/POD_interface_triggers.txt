POD_has_potential_mortal_herd_trigger = {
	POD_is_vampire_trigger = yes
	num_of_relation_herd_member < max_herd_cap
	NOT = { any_scheme = { scheme_type = herdify_mortal } }
	any_courtier_or_guest = {
		prev = {
			is_character_interaction_valid = {
				recipient = prev
				interaction = POD_herdify_mortal_interaction
			}
		}
	}
}
