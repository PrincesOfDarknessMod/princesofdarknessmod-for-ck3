	NOT = { has_trait = theurgist }
	trigger_if = {
		limit = {
			NOR = {
				has_religion = religion:roadofheaven_religion
				has_religion = religion:ashirra_religion

				has_religion = religion:hunters_religion
				faith = { has_doctrine = doctrine_fae_agenda_constantinian }
				faith = { has_doctrine = tenet_constantinian }
			}
		}
		faith = { NOT = { has_doctrine = doctrine_magicraft_crime } }
	}
}
