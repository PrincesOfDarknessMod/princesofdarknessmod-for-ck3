	POD_blocked_from_learning_trigger = no
	NOT = { has_trait = theurgist }
	trigger_if = {
		limit = { POD_is_samuel_haight_mode_enabled = yes }
		always = yes
	}
	trigger_else_if = {
		limit = {
			faith = { 
				NOR = {
					is_POD_christian_religion_trigger = yes
					is_POD_muslim_religion_trigger = yes
					is_POD_jewish_religion_trigger = yes
				}
			}
		}
		faith = { NOT = { has_doctrine = doctrine_magicraft_crime } }
	}
	trigger_else = {
		always = yes
	}
}
