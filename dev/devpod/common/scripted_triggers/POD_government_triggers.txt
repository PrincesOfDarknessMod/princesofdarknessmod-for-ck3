has_POD_government = {
	OR = {
		has_government = fera_tribe_government
		has_government = infernal_court_government
		has_government = cult_government
		has_government = inquisition_government
		has_government = cell_government
		has_government = traditions_government
		has_government = august_court_government
		has_government = household_government
		has_government = dark_kingdom_government
	}
}

POD_title_is_haven_landed_domicile = {
	is_nomad_title = yes
	POD_title_is_flagged_as_haven = yes
}

POD_title_is_flagged_as_haven = {
	has_variable = haven_government
}


# to check whether someone is a "pure" half-splat
POD_has_no_politically_relevant_splat = {
	# Fera Tribe
	POD_splat_is_fera_trigger = no
	NOT = { has_trait = spirit }
	
	# Infernal Court
	POD_is_demon_trigger = no
	
	# Mummy Cult
	POD_is_mummy_trigger = no
	
	# Traditions
	POD_is_vampire_trigger = no
	POD_is_revenant_trigger = no
	
	# Inquisition/Cell
	POD_is_supehunter_trigger = no
	
	# TODO: MAGE PLACEHOLDER
	#NOT = { has_trait = mage }
	
	# August Court
	is_kueijin_trigger = no
	
	# Households
	POD_is_fae_trigger = no
	POD_is_changeling_trigger = no
	
	# Dark Kingdom
	POD_is_wraith_trigger = no
}


POD_can_participate_in_fera_politics = {
	OR = {
		POD_splat_is_fera_trigger = yes
		has_trait = spirit
		AND = {
			POD_is_kinfolk_trigger = yes
			POD_has_no_politically_relevant_splat = yes
		}
	}
}

POD_can_participate_in_demon_politics = {
	OR = {
		POD_is_demon_trigger = yes
		AND = {
			OR = {
				has_trait = laham
				has_trait = nephilim
				is_a_cultist = yes
			}
			POD_has_no_politically_relevant_splat = yes
		}
	}
}

POD_can_participate_in_mummy_politics = {
	OR = {
		POD_is_mummy_trigger = yes
		AND = {
			is_a_neteru = yes
			POD_has_no_politically_relevant_splat = yes
		}
	}
}

# like has_POD_vamp_blood_trigger, but excluding things like ghoulbros or ghouled werewolves
POD_can_participate_in_vampire_politics = {
	OR = {
		POD_is_vampire_trigger = yes
		POD_is_revenant_trigger = yes
		AND = {
			OR = {
				POD_is_ghoul_trigger = yes
				has_trait = dhampir
			}
			POD_has_no_politically_relevant_splat = yes
		}
	}
}

POD_can_participate_in_kj_politics = {
	OR = {
		is_kueijin_trigger = yes
		AND = {
			has_trait = dhampyr
			POD_has_no_politically_relevant_splat = yes
		}
	}
}

POD_can_participate_in_fae_politics = {
	OR = {
		POD_is_fae_trigger = yes
		POD_is_changeling_trigger = yes
		AND = {
			has_trait = kinain
			POD_has_no_politically_relevant_splat = yes
		}
	}
}


POD_splat_is_valid_for_fera_tribe_government = {
	POD_can_participate_in_fera_politics = yes
}

POD_splat_is_valid_for_infernal_court_government = {
	POD_can_participate_in_demon_politics = yes
}

POD_splat_is_valid_for_cult_government = {
	POD_can_participate_in_mummy_politics = yes
}

POD_splat_is_valid_for_inquisition_government = {
	faith = faith:shadowinquisition
}

POD_splat_is_valid_for_cell_government = {
	OR = {
		POD_is_supehunter_trigger = yes
		is_child_of_supehunter_trigger = yes
	}
	POD_splat_is_valid_for_inquisition_government = no
}

POD_splat_is_valid_for_traditions_government = {
	POD_can_participate_in_vampire_politics = yes
}

POD_splat_is_valid_for_august_court_government = {
	POD_can_participate_in_kj_politics = yes
}

POD_splat_is_valid_for_household_government = {
	POD_can_participate_in_fae_politics = yes
}

POD_splat_is_valid_for_dark_kingdom_government = {
	POD_is_wraith_trigger = yes # no wraith half-splats
}


POD_government_is_valid_for_character = {
	trigger_if = {
		limit = { $GOVERNMENT$ = government_type:landless_adventurer_government }
		always = yes
	}
	trigger_else_if = {
		limit = { $GOVERNMENT$ = government_type:fera_tribe_government }
		POD_splat_is_valid_for_fera_tribe_government = yes
	}
	trigger_else_if = {
		limit = { $GOVERNMENT$ = government_type:infernal_court_government }
		POD_splat_is_valid_for_infernal_court_government = yes
	}
	trigger_else_if = {
		limit = { $GOVERNMENT$ = government_type:cult_government }
		POD_splat_is_valid_for_cult_government = yes
	}
	trigger_else_if = {
		limit = { $GOVERNMENT$ = government_type:inquisition_government }
		POD_splat_is_valid_for_inquisition_government = yes
	}
	trigger_else_if = {
		limit = { $GOVERNMENT$ = government_type:cell_government }
		POD_splat_is_valid_for_cell_government = yes
	}
	trigger_else_if = {
		limit = { $GOVERNMENT$ = government_type:traditions_government }
		POD_splat_is_valid_for_traditions_government = yes
	}
	trigger_else_if = {
		limit = { $GOVERNMENT$ = government_type:august_court_government }
		POD_splat_is_valid_for_august_court_government = yes
	}
	trigger_else_if = {
		limit = { $GOVERNMENT$ = government_type:household_government }
		POD_splat_is_valid_for_household_government = yes
	}
	trigger_else_if = {
		limit = { $GOVERNMENT$ = government_type:dark_kingdom_government }
		POD_splat_is_valid_for_dark_kingdom_government = yes
	}
	trigger_else = {
		always = yes
	}
}

POD_has_illegal_government_trigger = {
	trigger_if = {
		limit = { exists = government_type }
		NOT = {
			POD_government_is_valid_for_character = {
				GOVERNMENT = this.government_type
			}
		}
	}
	trigger_else = {
		always = no
	}
}

POD_can_convert_government_trigger = {
	$RECIPIENT$ = {
		POD_government_is_valid_for_character = {
			GOVERNMENT = $ACTOR$.government_type
		}
	}
}
