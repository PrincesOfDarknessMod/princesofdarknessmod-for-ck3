POD_CFP_has_graphical_byzantine_culture_group_trigger = {
	exists = global_var:cfp_is_loaded
	OR = {
		has_title = title:e_byzantium
		has_title = title:h_roman_empire
		has_title = title:e_latin_empire
		# has_title = title:e_dream
		AND = {
			has_title = title:k_thessalonika
			has_title = title:d_thrace
			has_title = title:c_byzantion
		}
		AND = {
			highest_held_title_tier = tier_empire
			capital_county.kingdom = title:k_thessalonika
		}
	}
}
