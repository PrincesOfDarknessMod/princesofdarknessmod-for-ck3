
POD_can_progress_journey_trigger = {
	custom_tooltip = {
		text = POD_journey_requirements_notravel
		NOT = { has_character_flag = POD_is_in_travel_phase }
	}
	is_imprisoned = no
	#is_in_army = no
	POD_is_not_trapped_in_umbra_basic_trigger = yes
	custom_tooltip = {
		text = POD_journey_requirements_notravel_generic
		NOT = { exists = current_travel_plan }
	}
	custom_tooltip = {
		text = POD_journey_requirements_noactivity
		NOT = { exists = involved_activity }
	}
}

POD_can_join_journey_trigger = {
	is_imprisoned = no
	POD_is_valid_coterie_member_trigger = yes
}

POD_not_already_on_journey_trigger = {
	custom_tooltip = {
		text = POD_not_already_on_journey_tt
		NOR = {
			var:current_journey ?= {
				var:journey_type = flag:$TYPE$
			}
			any_in_list = {
				variable = paused_journeys
				count > 0
				var:journey_type = flag:$TYPE$
			}
		}
	}
}

# Used for character interactions (needs actor and recipient scopes)
POD_cant_target_coterie_trigger = {
	custom_tooltip = {
		text = POD_cant_target_coterie_tt
		scope:actor = {
			OR = {
				NOT = { exists = var:coterie }
				var:coterie = {
					any_in_list = {
						variable = members
						count = 0
						this = scope:recipient
					}
				}
			}
		}
	}
}
