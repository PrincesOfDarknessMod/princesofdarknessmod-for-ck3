
POD_can_start_planning_phase_trigger = {
	custom_tooltip = {
		text = POD_journey_requirements_notravel
		NOT = { has_character_flag = POD_is_in_travel_phase }
		trigger_if = {
			limit = { has_variable = current_journey }
			NOT = { var:current_journey.var:phase = flag:travel_phase }
		}
	}
}

POD_can_start_travel_phase_trigger = {
	trigger_if = {
		limit = { exists = involved_activity }
		custom_tooltip = {
			text = POD_journey_requirements_noactivity
			always = no
		}
	}
	trigger_else_if = {
		limit = { exists = current_travel_plan }
		custom_tooltip = {
			text = POD_journey_requirements_notravel_generic
			always = no
		}
	}
	trigger_else = {
		always = yes
	}
	is_adult = yes
	is_imprisoned = no
	#is_in_army = no
	POD_is_not_trapped_in_umbra_basic_trigger = yes
}

POD_can_join_journey_trigger = {
	is_imprisoned = no
	POD_is_valid_coterie_member_trigger = yes
}

POD_is_in_travel_phase_trigger = {
	var:current_journey ?= {
		var:phase = flag:travel_phase
	}
}

POD_is_on_journey_trigger = {
	OR = {
		var:current_journey ?= {
			var:journey_type = flag:$TYPE$
		}
		any_in_list = {
			variable = paused_journeys
			count > 0
			var:journey_type = flag:$TYPE$
		}
	}
}

POD_not_already_on_journey_trigger = {
	custom_tooltip = {
		text = POD_not_already_on_journey_tt
		NOR = {
			var:current_journey ?= {
				var:journey_type = flag:$TYPE$
			}
			any_in_list = {
				variable = paused_journeys
				count > 0
				var:journey_type = flag:$TYPE$
			}
		}
	}
}

# Used for character interactions (needs actor and recipient scopes)
POD_cant_target_coterie_trigger = {
	custom_tooltip = {
		text = POD_cant_target_coterie_tt
		scope:actor = {
			OR = {
				NOT = { exists = var:coterie }
				var:coterie = {
					any_in_list = {
						variable = members
						count = 0
						this = scope:recipient
					}
				}
			}
		}
	}
}

POD_provoke_vatican_available_trigger = {
	NOR = {
		current_date >= 1430.1.1
		POD_provoke_vatican_completed_trigger = yes
		is_target_in_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:destroy_inquisition_objective_decision
		}
		is_target_in_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:constantinian_objective_decision
		}
	}
}

POD_provoke_vatican_completed_trigger = {
	has_global_variable = premature_inquisition_year
}

# If the player provoked the Inquisition, adjust the dates accordingly
POD_reached_endgame_date_trigger = {
	trigger_if = {
		limit = { POD_provoke_vatican_completed_trigger = yes }
		character:POD_wraith_01 = { # current_year needs a scope, i guess?
			current_year >= {
				value = global_var:premature_inquisition_year
				add = $YEAR$
				subtract = 1430 # canonical Inquisition date
			}
		}
	}
	trigger_else = {
		current_date >= $YEAR$.1.1
	}
}
