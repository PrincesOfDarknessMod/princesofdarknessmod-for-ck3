### Character scope

POD_can_start_planning_phase_trigger = {
	custom_tooltip = {
		text = POD_journey_requirements_notravel
		NOT = { has_character_flag = POD_is_in_travel_phase }
		trigger_if = {
			limit = { has_variable = current_journey }
			NOT = { var:current_journey.var:phase = flag:travel_phase }
		}
	}
}

POD_can_start_travel_phase_trigger = {
	trigger_if = {
		limit = { exists = involved_activity }
		custom_tooltip = {
			text = POD_journey_requirements_noactivity
			always = no
		}
	}
	trigger_else_if = {
		limit = { exists = current_travel_plan }
		custom_tooltip = {
			text = POD_journey_requirements_notravel_generic
			always = no
		}
	}
	trigger_else = {
		always = yes
	}
	is_adult = yes
	is_imprisoned = no
	is_in_army = no
	POD_is_not_trapped_in_umbra_trigger = yes
}

	is_landless_adventurer = yes

POD_is_in_travel_phase_trigger = {
	var:current_journey ?= {
		var:phase = flag:travel_phase
	}
}

# skip travel into event if traveling for a journey since they have their own
POD_skip_first_travel_event_trigger = {
	var:current_journey ?= { exists = var:travel_plan }
	current_travel_plan ?= var:current_journey.var:travel_plan
}

POD_is_on_any_journey_trigger = {
	OR = {
		has_variable = current_journey
		has_variable_list = paused_journeys
	}
}

POD_is_on_journey_trigger = {
	OR = {
		var:current_journey ?= {
			var:journey_type = flag:$TYPE$
		}
		any_in_list = {
			variable = paused_journeys
			count > 0
			var:journey_type = flag:$TYPE$
		}
	}
}

POD_any_journey_fulfills_trigger = {
	OR = {
		var:current_journey ?= {
			$TRIGGER$
		}
		any_in_list = {
			variable = paused_journeys
			count > 0
			$TRIGGER$
		}
	}
}

		}
	}

# Used for character interactions (needs actor and recipient scopes)
POD_cant_target_coterie_trigger = {
	custom_tooltip = {
		text = POD_cant_target_coterie_tt
		scope:actor = {
			OR = {
				NOT = { exists = var:coterie }
				var:coterie = {
					any_in_list = {
						variable = members
						count = 0
						this = scope:recipient
					}
				}
			}
		}
	}
}

# If the player provoked the Inquisition, adjust the dates accordingly
POD_reached_endgame_date_trigger = {
	character:POD_wraith_01 ?= { # current_year needs a scope, i guess?
		current_year >= {
			value = inquisition_spawn_year
			add = $YEAR$
			subtract = 1430 # canonical Inquisition date
		}
	}
}
