# If POD_gore_censorship_trigger = yes, disable the gore. If no, enable the gore.
POD_gore_censorship_trigger = {
	OR = {
		#has_global_variable = POD_gore_disallowed
		should_show_disturbing_portrait_modifiers = no
		#always = yes
	}
}

# If POD_nudity_censorship_trigger = yes, disable the nudity. If no, enable the nudity.

	OR = {
		#has_global_variable = POD_nudity_disallowed
		should_show_nudity = no
		#always = yes
	}
}
