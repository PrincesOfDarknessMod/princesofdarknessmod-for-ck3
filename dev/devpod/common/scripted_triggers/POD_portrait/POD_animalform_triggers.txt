######################################################################
# Gates for the naturalistic animal portrait replacements

#
# Removing a hook is deleting one line.
######################################################################

###### WOLF ##########################################################
# Nuwisha take the wolf mesh as the closest available coyote.

POD_animalform_wolf_trigger = {
	OR = {
		has_character_modifier = POD_shapeshift_form_wolf_modifier
		has_character_modifier = POD_nuwisha_form_latrani_modifier
		POD_werewolf_lupus_form_trigger = yes
		POD_shapeshift_pooka_wolf_mien_trigger = yes
		POD_shapeshift_pooka_coyote_mien_trigger = yes
	}
}

###### HYENA #########################################################

POD_animalform_hyena_trigger = {
	OR = {
		POD_ajaba_form_hyaenid_trigger = yes
		POD_shapeshift_pooka_hyena_mien_trigger = yes
	}
}

###### BEAR ##########################################################
# Urus is the Gurahl true-bear form.

POD_animalform_bear_trigger = {
	OR = {
		POD_gurahl_form_urus_trigger = yes
		POD_shapeshift_pooka_bear_mien_trigger = yes
		POD_has_spirit_type = { TYPE = spirit_bear }
	}
}

###### LIONESS #######################################################
# The generic big-cat silhouette.

POD_animalform_lioness_trigger = {
	always = no
}

###### LION ##########################################################
# Simba is the lion breed.  The gene template branches on sex, 
# so a female Simba gets the lioness
# mesh from this same trigger without a separate entry.

POD_animalform_lion_trigger = {
	OR = {
		has_character_modifier = POD_shapeshift_form_lion_modifier
		AND = {
			POD_bastet_form_feline_trigger = yes
			faith = faith:simba
		}
	}
}

###### TIGER #########################################################
# Khan is the tiger breed.

POD_animalform_tiger_trigger = {
	OR = {
		AND = {
			POD_bastet_form_feline_trigger = yes
			faith = faith:khan
		}
		POD_has_spirit_type = { TYPE = spirit_tiger }
	}
}

###### ALLIGATOR #####################################################
# Mokole suchid is picked per varna and none of those modifiers is consulted by any
# PoD portrait trigger. Gharial, makara and suchus are the crocodilian varna left here.

POD_animalform_alligator_trigger = {
	OR = {
		has_character_modifier = POD_mokole_form_suchid_gharial_modifier
		has_character_modifier = POD_mokole_form_suchid_makara_modifier
		has_character_modifier = POD_mokole_form_suchid_suchus_modifier
		POD_shapeshift_pooka_crocodile_mien_trigger = yes
	}
}

###### KOMODO DRAGON #################################################
# Ora is literally the Komodo dragon varna; karna is the monitor lizard, close
# enough to share the mesh.

POD_animalform_komodo_trigger = {
	OR = {
		has_character_modifier = POD_mokole_form_suchid_ora_modifier
		has_character_modifier = POD_mokole_form_suchid_karna_modifier
	}
}

###### WOLF, SECOND MODEL ############################################
# A second wolf mesh, deliberately NOT wired to any live PoD state. The first wolf
# already claims every wolf-shaped hook, and the portrait group picks exactly one
# entry, so giving both the same triggers would just make the winner depend on
# declaration order.
#
# To use this one for real, move a line out of POD_animalform_wolf_trigger and into
# here - for instance give Protean's shape-of-the-beast this mesh and leave Garou
# Lupus on the other, or swap them wholesale.

POD_animalform_wolf2_trigger = {
	always = no
}

###### SPIDER ########################################################

POD_animalform_spider2_trigger = {
	OR = {
		POD_ananasi_form_primalbeastform_trigger = yes
		POD_shapeshift_pooka_spider_mien_trigger = yes
	}
}

###### BLACK WIDOW ###################################################
# A spider mesh, deliberately claiming no live PoD state - spider2 already holds
# the Ananasi Pithus and Pooka spider-mien hooks, and the portrait group picks exactly
# one entry, so sharing triggers would only make the winner depend on declaration
# order. It is the obvious candidate for something species-specific if PoD ever
# distinguishes Ananasi breeds; until then, move a line out of
# POD_animalform_spider2_trigger to give it real work.

POD_animalform_blackwidow_trigger = {
	always = no
}

###### NILE CROCODILE ################################################
# Champsa is the Nile crocodile varna, so it belongs on an actual Nile crocodile
# rather than on the American alligator that was standing in for it. Moving it is a
# one-line change: delete the champsa line here and put it back in
# POD_animalform_alligator_trigger.

POD_animalform_crocodile_trigger = {
	has_character_modifier = POD_mokole_form_suchid_champsa_modifier
}

###### BLACK PANTHER ################################################
# Bagheera are the black-panther breed by name, so they move off the generic lioness
# silhouette onto an actual black leopard. One line: delete the faith check here and
# put faith:bagheera back in POD_animalform_lioness_trigger to undo it.
#
# Bubasti seemed close enough as well.

POD_animalform_panther_trigger = {
	OR = {
		AND = {
			POD_bastet_form_feline_trigger = yes
			OR = {
				faith = faith:bagheera
				faith = faith:bubasti
			}
		}
	}
}

###### RAT ##########################################################
# POD_shapeshift_form_rat_modifier is one of PoD's GENERIC form keys: nine separate
# interactions grant it (vampire, fae, Fera, mummy, Kuei-jin, demon, possessed,
# psychic, and a beast_cloak artifact that checks no splat at all). 
# Rodens and the Pooka mien instead fold into the anthro wererat.

POD_animalform_rat_trigger = {
	OR = {
		has_character_modifier = POD_shapeshift_form_rat_modifier
		has_character_modifier = POD_ratkin_form_rodens_modifier
		POD_shapeshift_pooka_rat_mien_trigger = yes
		POD_has_spirit_type = { TYPE = spirit_mouse }
	}
}

###### TITANOBOA ####################################################
# The giant-serpent tier only. Kept apart from the rattlesnake below because PoD
# distinguishes the two sizes and the meshes differ by an order of magnitude.

POD_animalform_titanoboa_trigger = {
	has_character_modifier = POD_shapeshift_form_giantserpent_modifier
}

###### JAGUAR #######################################################

	AND = {
		POD_bastet_form_feline_trigger = yes

}

###### DOMESTIC CAT #################################################

POD_animalform_cat_trigger = {
	OR = {
		AND = {
			POD_bastet_form_feline_trigger = yes
			OR = {
				faith = faith:ceilican
				faith = faith:hellcats
			}
		}

		has_character_modifier = POD_shapeshift_form_mooncat_modifier
		POD_shapeshift_pooka_cat_mien_trigger = yes
	}
}
