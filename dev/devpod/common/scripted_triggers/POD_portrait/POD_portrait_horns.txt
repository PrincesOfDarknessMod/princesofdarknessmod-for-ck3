POD_portrait_monster_antler_trigger = {
	AND = {
		OR = {
			AND = {
				has_character_modifier = POD_pooka_deer_affinity_modifier 
				is_male = yes # Female antelope and reindeer also have antlers, so a more complex trigger is needed eventually
			}
			has_character_flag = POD_portrait_monster_antler_flag
		}
		POD_portrait_trait_override_trigger = no
		NOT = { has_character_modifier = POD_shapeshift_pooka_deer_mien_modifier }
		POD_portrait_no_human_body_trigger = no
	}
}

POD_portrait_monster_antenna_trigger = { # bee antenna
	AND = {
		OR = {
			has_character_modifier = POD_pooka_beetle_affinity_modifier 
			has_character_flag = POD_portrait_monster_antenna_flag
			has_character_modifier = demon_regalia_antennae_modifier
			AND = {
				has_character_modifier = POD_gargoyle_caste_fairy_modifier
				POD_suppress_visual_clan_curse_trigger = no
			}
		}
		POD_portrait_trait_override_trigger = no
		NOT = { has_character_modifier = POD_shapeshift_pooka_beetle_mien_modifier }
		NOT = { has_character_modifier = POD_shapeshift_pooka_ant_mien_modifier }
		NOT = { has_character_modifier = POD_shapeshift_pooka_mantis_mien_modifier }
		NOT = { has_character_modifier = POD_shapeshift_pooka_millipede_mien_modifier }
		NOT = { has_character_modifier = POD_shapeshift_pooka_bee_mien_modifier }
		NOT = { has_character_modifier = POD_shapeshift_pooka_butterfly_mien_modifier }
	}
	POD_portrait_no_human_body_trigger = no
}
