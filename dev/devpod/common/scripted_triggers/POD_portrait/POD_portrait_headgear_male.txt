POD_jester_hat_trigger = {
	is_female = no
	OR = {
		AND = {
			is_malkavian_trigger = yes
			has_trait = lunatic
			portrait_western_clothing_trigger = yes
			is_ai = yes
			is_ruler = no
		}
		has_character_flag = POD_portrait_male_jester_hat_flag
	}
}

POD_chaperon_hat_trigger = {
	current_date >= 1300.1.1
	OR = {
		AND = {
			
			AND = {
				highest_held_title_tier <= tier_county
				current_date >= 1300.1.1
				is_ai = yes
				is_female = no
			}
			OR = {
				house = house:house_giovanni
				house = house:house_canale
				is_toreador_trigger = yes
				#is_predatortype_gentry_trigger = yes
			}
			OR = {
				portrait_western_clothing_trigger = yes
				portrait_byzantine_clothing_trigger = yes
				portrait_northern_clothing_trigger = yes
			}	
		}
		has_character_flag = POD_portrait_chaperone_flag
	}
}
