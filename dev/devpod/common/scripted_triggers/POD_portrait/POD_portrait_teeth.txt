# MONSTER TEETH

POD_portrait_monster_fangs_teeth_trigger = {
	has_character_flag = POD_portrait_monster_fangs_teeth_flag
	POD_portrait_no_clothing_form_trigger = no
	POD_portrait_no_human_body_trigger = no
	NOT = { has_character_flag = POD_portrait_monster_double_fangs_teeth_flag }
	NOT = { has_character_flag = POD_portrait_monster_snekfangs_teeth_flag }
	NOT = { has_character_flag = POD_portrait_monster_farfangs_teeth_flag }
	NOT = { has_character_flag = POD_portrait_monster_orlok_teeth_flag }
	NOT = { has_character_flag = POD_portrait_monster_gnarly_teeth_flag }
	NOT = { has_character_flag = POD_portrait_monster_no_teeth_flag }
	NOT = { has_character_flag = POD_portrait_monster_gummy_teeth_flag }
	NOT = { has_character_flag = POD_portrait_monster_normal_teeth_flag }
	
}

POD_portrait_monster_double_fangs_teeth_trigger = {
	has_character_flag = POD_portrait_monster_double_fangs_teeth_flag
	POD_portrait_no_clothing_form_trigger = no
	POD_portrait_no_human_body_trigger = no
}

POD_portrait_monster_farfangs_teeth_trigger = {
	OR = {
		has_character_flag = POD_portrait_monster_farfangs_teeth_flag
		has_trait = djedi
	}
	POD_portrait_no_clothing_form_trigger = no
	POD_portrait_no_human_body_trigger = no
}

POD_portrait_monster_snekfangs_teeth_trigger = {
	has_character_flag = POD_portrait_monster_snekfangs_teeth_flag
	POD_portrait_no_clothing_form_trigger = no
	POD_portrait_no_human_body_trigger = no
}

POD_portrait_monster_orlok_teeth_trigger = {
	has_character_flag = POD_portrait_monster_orlok_teeth_flag
	POD_portrait_no_clothing_form_trigger = no
	POD_portrait_no_human_body_trigger = no
}

POD_portrait_monster_gnarly_teeth_trigger = {
	OR = {
		has_character_flag = POD_portrait_monster_gnarly_teeth_flag
		has_character_modifier = demon_regalia_poisoned_fangs_modifier
		POD_is_absimiliard_trigger = yes
		AND = {
			has_trait = shaytan
			POD_portrait_trait_override_trigger = no
		}
	}
	POD_portrait_no_clothing_form_trigger = no
	POD_portrait_no_human_body_trigger = no
}

POD_portrait_monster_no_teeth_trigger = {
	OR = {
		has_character_flag = POD_portrait_monster_no_teeth_flag
		POD_portrait_no_clothing_form_trigger = yes
		POD_portrait_no_human_body_trigger = yes
	}
}

POD_portrait_monster_gummy_trigger = {
	has_character_flag = POD_portrait_monster_gummy_teeth_flag
	POD_portrait_no_clothing_form_trigger = no
	POD_portrait_no_human_body_trigger = no
}
POD_portrait_monster_normal_teeth_trigger = {
	has_character_flag = POD_portrait_monster_normal_teeth_flag
	POD_portrait_no_clothing_form_trigger = no
	POD_portrait_no_human_body_trigger = no
}
