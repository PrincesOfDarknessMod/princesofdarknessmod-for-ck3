POD_portrait_pharaoh_clothing_trigger = {
	OR = {
		has_character_flag = POD_portrait_pharaoh_clothing_flag
		AND = {
			has_trait = setite
			has_trait = paragon
			has_faith = faith:viaset
		}
		POD_is_sutekh_trigger = yes	
		POD_is_osiris_trigger = yes
		this = character:792593 # Horus
		has_character_modifier = POD_shapeshift_form_pharaoh_modifier
	}
}
