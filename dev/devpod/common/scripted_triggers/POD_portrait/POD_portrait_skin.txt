# POD skin colors

has_POD_obsidian_skin_trigger = {
	OR = {
		AND = {
			has_game_rule = banu_haqim_default_darken 
			NOT = { has_character_flag = POD_portrait_no_obsidian_skin_flag }
			is_assamite_trigger = yes
		}
		POD_is_Ur_Shulgi_trigger = yes
		AND = {
			has_trait = kuino
			POD_portrait_trait_override_trigger = no
		}
		has_character_flag = POD_portrait_obsidian_skin_flag	
	}
	
}
