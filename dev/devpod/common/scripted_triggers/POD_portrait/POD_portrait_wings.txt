POD_portrait_monster_bone_wings_trigger = {
	OR = {
		has_character_flag = POD_portrait_monster_bone_wings_flag
		AND = {
			has_character_modifier = POD_gargoyle_caste_bone_modifier
			POD_suppress_visual_clan_curse_trigger = no
		}
		has_character_modifier = POD_shapeshift_form_apocalyptic_ereshkigal_modifier
		POD_is_lausombheu_trigger = yes
	}
}
