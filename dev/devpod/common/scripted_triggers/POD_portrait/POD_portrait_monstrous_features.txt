# For features that don't fit into any of the other categories
# (not eyes, ears, horns, wings, teeth)

POD_portrait_monster_hoof_legs_trigger = {
	OR = {
		has_character_modifier = POD_pooka_horse_affinity_modifier 
		has_character_modifier = POD_pooka_bovine_affinity_modifier 
		has_character_modifier = POD_pooka_deer_affinity_modifier 
		has_character_flag = POD_portrait_monster_hoof_legs_flag
	}
	POD_portrait_trait_override_trigger = no
	POD_portrait_no_human_body_trigger = no
	NOT = { has_character_modifier = POD_shapeshift_pooka_horse_mien_modifier }
	NOT = { has_character_modifier = POD_shapeshift_pooka_bovine_mien_modifier }
	NOT = { has_character_modifier = POD_shapeshift_pooka_deer_mien_modifier }
}

# Halo
POD_portrait_monster_halo_trigger = {
	OR = {
		has_character_flag = POD_portrait_halo_flag
		has_trait_xp = {
			trait = beatific
			value >= 100
		}
		POD_is_danu_trigger = yes
		POD_is_michael_trigger = yes
		AND = {
			has_trait = demon
			is_alive = no
			death_reason = death_redemption 
		}
	}
}
