POD_cappadocian_skeletal_form_trigger = {
	POD_gore_censorship_trigger = no
	POD_suppress_visual_clan_curse_trigger = no
	NOT = { has_game_rule = cappadocian_never_skeleton }
	is_cappadocian_trigger = yes
	age > 100
	is_giovanni_trigger = no 
	is_lamia_trigger = no
	POD_is_apacia_trigger = no # She's already hideious enough
	#POD_is_cappadocius_trigger = no
	NOT = { has_character_flag = POD_no_cappadocian_look_flag }
	OR = {
		POD_is_cappadocius_trigger = yes
		has_trait = wight
		AND = {
			OR = {
				has_character_modifier = bloodhunger5_modifier
				has_trait = torpor
			}
			OR = {
				has_game_rule = cappadocian_hunger_skeleton
				has_game_rule = cappadocian_masquerade_skeleton
				has_game_rule = cappadocian_stress_skeleton
				has_game_rule = cappadocian_piety_skeleton
			}
		}
		AND = {
			has_character_modifier = masquerade5_modifier
			OR = {
				has_game_rule = cappadocian_masquerade_skeleton
				has_game_rule = cappadocian_stress_skeleton
				has_game_rule = cappadocian_piety_skeleton
			}
		}		
		AND = {
			stress_level >= 3
			OR = {
				has_game_rule = cappadocian_stress_skeleton
				has_game_rule = cappadocian_piety_skeleton
			}
		}		
		AND = {
			piety < 0
			has_game_rule = cappadocian_piety_skeleton
		}
		has_game_rule = cappadocian_all_skeleton
		is_alive = no
		AND = {
			POD_is_troglodytia_trigger = yes
			OR = {
				has_character_modifier = bloodhunger3_modifier
				has_character_modifier = bloodhunger4_modifier
			}
			OR = {
				has_game_rule = cappadocian_hunger_skeleton
				has_game_rule = cappadocian_masquerade_skeleton
				has_game_rule = cappadocian_stress_skeleton
				has_game_rule = cappadocian_piety_skeleton
			}
		}
	}
}

			NOT = { has_trait = risen }
			NOT = { has_trait = shadow }
		}
		AND = {
			is_kuntilanak_trigger = yes
			POD_suppress_visual_clan_curse_trigger = no
		}
		POD_is_cappadocius_trigger = yes
		POD_is_charon_trigger = yes
	}
}
