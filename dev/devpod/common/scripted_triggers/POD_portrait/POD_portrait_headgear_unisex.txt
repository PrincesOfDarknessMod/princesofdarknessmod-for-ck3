portrait_mithrasist_trigger = {
	OR = {
		AND = {
			OR = {
				faith = faith:viamithras
				has_character_flag = POD_portrait_mithrasist_flag
			}
			has_trait = zealous
		}
		OR = {
			POD_wearing_red_cap_trigger = yes
		}
	}
	NOT = { has_trait = disfigured }
