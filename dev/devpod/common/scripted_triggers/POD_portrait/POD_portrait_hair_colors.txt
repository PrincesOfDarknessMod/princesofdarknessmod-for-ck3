POD_pure_black_hair_form_trigger = {
	OR = {
		AND = {
			has_trait = sluagh
			POD_portrait_trait_override_trigger = no
		}
		POD_is_dagda_trigger = yes
		has_character_flag = POD_portrait_pure_black_hair_flag
	}
}
