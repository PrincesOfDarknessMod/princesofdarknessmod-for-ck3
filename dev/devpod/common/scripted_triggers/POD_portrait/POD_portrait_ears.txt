POD_portrait_monster_long_ear_trigger = {
	OR = {
		POD_is_absimiliard_trigger = yes
		has_character_flag = POD_portrait_long_ears_flag
		AND = {
			OR = {
				has_trait = sidhe
				has_trait = nocker
				has_trait = troll
				has_trait = satyr
				has_trait = kubera
				has_trait = ondine
				has_trait = paroseme
				has_trait = solimond
			}
			POD_portrait_trait_override_trigger = no
		}
		AND = {
			has_character_modifier = POD_gargoyle_caste_fairy_modifier
			POD_suppress_visual_clan_curse_trigger = no
		}
	}
	POD_portrait_no_human_body_trigger = no
}
