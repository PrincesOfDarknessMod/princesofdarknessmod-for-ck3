POD_umbra_travel_valid_trigger = {
	trigger_if = {
		limit = {
			OR = {
				NOT = { has_variable = spirit_location }
				NOT = { var:spirit_location = $LOCATION$ }
			}
		}
		POD_is_not_trapped_in_umbra_basic_trigger = yes
		trigger_if = {
			limit = {
				OR = {
					in_penumbra_trigger = { LOCATION = $LOCATION$ } 
					in_zone_trigger = { LOCATION = $LOCATION$ } 
				}
			}
		}
		trigger_else_if = {
			limit = { in_underworld_trigger = { LOCATION = $LOCATION$ } }
			has_access_to_underworld_trigger = yes
			trigger_if = {
				limit = { location ?= { is_sea_province = no } }
				trigger_if = {
					limit = { has_trait = entomos }
					shroud_too_weak_trigger = no
				}
				trigger_else = {
					shroud_too_strong_trigger = no
				}
			}
		}
		trigger_else = {
			trigger_if = {
				limit = { in_dreaming_trigger = { LOCATION = $LOCATION$ } }
				has_access_to_dreaming_trigger = yes
			}
			trigger_if = {
				limit = { in_spirit_wilds_trigger = { LOCATION = $LOCATION$ } }
				has_access_to_spirit_wilds_trigger = yes
				trigger_if = {
					limit = { POD_is_bastet_trigger = yes }

						text = bastet_spirit_wilds_access_tooltip
						OR = {
							has_faith = faith:swara
							has_trait_xp = { trait = bastet value >= 20 }
						}
					}
				}
			}
			trigger_if = {
				limit = { in_astral_reaches_trigger = { LOCATION = $LOCATION$ } }
				has_access_to_astral_reaches_trigger = yes
			}
			trigger_if = {
				limit = { in_deep_umbra_trigger = { LOCATION = $LOCATION$ } }
				has_access_to_deep_umbra_trigger = yes
			}
			trigger_if = {
				limit = { location ?= { is_sea_province = no } }
				trigger_if = {
					limit = { has_trait = entomos }
					gauntlet_too_weak_trigger = no
				}
				trigger_else = {
					gauntlet_too_strong_trigger = no
				}
			}
		}

					}
				}
			}

				trigger_if = {
					limit = { has_trait = demon }
					custom_tooltip = {
						text = breakdown_the_abyss_objective_decision_finished
						has_global_variable = abyss_broken
					}
				}
			}
			flag:digital_web = {
				has_trait = mage
			}
			flag:umi = {
				location = {
					trigger_if = {
						limit = { is_sea_province = no }
						is_coastal = yes
					}
				}
			}
			flag:hollow_worlds = {
				location = {
					custom_tooltip = {

						OR = {
							has_building_or_higher = chaw_gully_mine_01

				OR = {

			limit = { has_trait = mokole is_ai = yes }
