has_character_material_enhance_trigger = {
	var:current_method ?= {
		OR = {
			this = flag:impress
			this = flag:empower_jade
			this = flag:bound_spirit
		}
	}
}

POD_enhance_create_trigger = {
	switch = {
		trigger = var:current_method
		flag:enhance_object = {
			var:craft_creator = { has_perk = enhance_object_perk }
			var:target_artifact = {
				custom_tooltip = {
					text = enhanced_artifact_tt
					NOT = { has_variable = enhanced_artifact } 
				}
			}
		}
		flag:the_forgers_hammer = {
			var:craft_creator = { has_perk = the_forgers_hammer_perk }
			var:target_artifact = {
				custom_tooltip = {
					text = no_weapon_chosen
					artifact_slot_type = primary_armament
				}
				custom_tooltip = {
					text = reforged_artifact_tt
					NOT = { has_variable = reforged_artifact } 
				}
			}
		}
		flag:aid_from_spirits = {
			var:craft_creator = { has_perk = aid_from_spirits_perk }
			var:target_artifact = {
				custom_tooltip = {
					text = possessed_artifact_tt
					NOT = { has_variable = possessed_artifact } 
					NOT = { has_variable = soultaker } # Probably not a good idea
					NOT = { has_variable = soulforge } # Probably not a good idea
				}
			}
		}
		flag:sword_of_the_righteous = {
			var:craft_creator = { has_perk = sword_of_the_righteous_perk }
			var:target_artifact = {
				custom_tooltip = {
					text = no_weapon_chosen
					artifact_slot_type = primary_armament
				}
				custom_tooltip = {
					text = enchanted_artifact_tt
					NOT = { has_variable = enchanted_artifact } 
				}
			}
		}
		flag:rune = {
			var:craft_creator = { has_perk = rune_perk }
			var:target_artifact = {
				custom_tooltip = {
					text = enhanced_artifact_tt
					NOT = { has_variable = enhanced_artifact } 
				}
			}
		}
		flag:channel_yin = {
			var:craft_creator = { has_perk = yin_mantle_perk }
			var:target_artifact = {
				custom_tooltip = {
					text = channelled_artifact_tt
					NOT = { has_variable = channelled_artifact } 
				}
			}
		}
		flag:inspire_life = {
			var:craft_creator = { has_perk = the_spark_of_life_perk }
			var:target_artifact = {
				custom_tooltip = {
					text = possessed_artifact_tt
					NOT = { has_variable = possessed_artifact } 
					NOT = { has_variable = soultaker } # Probably not a good idea
					NOT = { has_variable = soulforge } # Probably not a good idea
				}
			}
		}
		flag:bless = {
			var:craft_creator = { 
				has_trait = beatific 
				has_trait_xp = { trait = beatific value >= 40 }
			}
			var:target_artifact = {
				custom_tooltip = {
					text = blessed_artifact_tt
					NOT = { has_variable = blessed_artifact } 
				}
			}
		}
		flag:sorcerers_blade = {
			var:craft_creator = { has_perk = sorcerers_blade_perk }
			var:target_artifact = {
				custom_tooltip = {
					text = no_weapon_chosen
					artifact_slot_type = primary_armament
				}
				custom_tooltip = {
					text = enchanted_artifact_tt
					NOT = { has_variable = enchanted_artifact } 
				}
			}
		}
		flag:blessing_the_blade = {
			var:craft_creator = { has_perk = blessing_the_blade_perk }
			var:target_artifact = {
				custom_tooltip = {
					text = no_weapon_chosen
					artifact_slot_type = primary_armament
				}
				custom_tooltip = {
					text = blessed_artifact_tt
					NOT = { has_variable = blessed_artifact } 
				}
			}
		}
		flag:sanctify = {
			var:craft_creator = { has_perk = sanctify_perk }
			var:target_artifact = {
				custom_tooltip = {
					text = no_weapon_chosen
					artifact_slot_type = primary_armament
				}
				custom_tooltip = {
					text = possessed_artifact_tt
					NOT = { has_variable = possessed_artifact } 
					NOT = { has_variable = soultaker } # Probably not a good idea
					NOT = { has_variable = soulforge } # Probably not a good idea
				}
			}
		}
		flag:bound_spirit = {
			var:craft_creator = { 
				custom_tooltip = {
					text = creator_need_shapeshifter_large_tt
					POD_can_equip_fetish_trigger = yes
				}
			}
			trigger_if = {
				limit = { has_variable = character_material }
				var:character_material = {
					has_trait = spirit
					trigger_if = {
						limit = { NOT = { this = root } }
						has_relation_spirit_companion = root
					}
					custom_tooltip = {
						text = possessing_artifact_tt
						NOT = { has_variable = stone_prison }
					}
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = no_character_material
					always = no
				}
			}
			var:target_artifact = {
				custom_tooltip = {
					text = possessed_artifact_tt
					NOT = { has_variable = possessed_artifact } 
					NOT = { has_variable = soultaker } # Probably not a good idea
					NOT = { has_variable = soulforge } # Probably not a good idea
				}
			}
		}
		flag:impress = {
			var:craft_creator = { has_perk = impress_perk }
			trigger_if = {
				limit = { has_variable = character_material }
				custom_tooltip = { 
					text = commit_suicide_decision 
					NOT = { var:craft_creator = var:character_material } 
				}
				var:character_material = { 
					has_trait = wraith
					is_imprisoned_by = root
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = no_character_material
					always = no
				}
			}
			var:target_artifact = {
				custom_tooltip = {
					text = possessed_artifact_tt
					NOT = { has_variable = possessed_artifact } 
					NOT = { has_variable = soultaker } # Probably not a good idea
					NOT = { has_variable = soulforge } # Probably not a good idea
				}
			}
		}
		flag:strengthen = {
			var:craft_creator = { has_perk = reliquary_perk }
			var:target_artifact = {
				custom_tooltip = {
					text = strengthened_artifact_tt
					NOT = { has_variable = strengthened_artifact } 
				}
			}
		}
		flag:empower_jade = {
			var:craft_creator = { has_perk = empower_jade_perk }
			trigger_if = {
				limit = { has_variable = character_material }
				custom_tooltip = { 
					text = commit_suicide_decision 
					NOT = { var:craft_creator = var:character_material } 
				}
				var:character_material = { 
					has_trait = wraith
					is_imprisoned_by = root
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = no_character_material
					always = no
				}
			}
			var:target_artifact = {
				custom_tooltip = {
					text = possessed_artifact_tt
					NOT = { has_variable = possessed_artifact } 
					NOT = { has_variable = soultaker } # Probably not a good idea
					NOT = { has_variable = soulforge } # Probably not a good idea
				}
			}
		}
		flag:rework = {
			var:craft_creator = { has_perk = rework_relics_perk }
			var:target_artifact = {
				custom_tooltip = {
					text = reworked_artifact_tt
					NOT = { has_variable = reworked_artifact } 
				}
			}
			trigger_if = {
				limit = { has_variable = artifact_material }
				custom_tooltip = {
					text = "is_wraith_jade"
					var:artifact_material = { has_variable = wraith_jade }
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = no_artifact_material
					always = no
				}
			}
		}
	}
}

has_enhance_recipe_trigger = {
	OR = {
		has_perk = enhance_object_perk
		has_perk = the_forgers_hammer_perk
		has_perk = aid_from_spirits_perk
		has_perk = sword_of_the_righteous_perk
		has_perk = rune_perk
		has_perk = yin_mantle_perk
		has_perk = the_spark_of_life_perk
		has_perk = sorcerers_blade_perk
		has_perk = blessing_the_blade_perk
		has_perk = sanctify_perk
		has_perk = impress_perk
		has_perk = reliquary_perk
		has_perk = empower_jade_perk
		has_perk = rework_relics_perk
		has_trait = beatific
		POD_can_equip_fetish_trigger = yes
	}
}
