POD_can_pick_elder_perk_trigger = {
	OR = {
		vampgeneration_less_or_equal_to_trigger = { LEVEL = $GEN$ }
		has_character_modifier = antediluvian_dominor_modifier
		has_trait_with_flag = elder_perk_flag
	}
	POD_is_not_clone_trigger = yes
}

POD_is_hungerless_trigger = {
	has_trait_with_flag = hungerless_flag
	has_POD_vamp_blood_trigger = yes
}

POD_has_infinite_blood_trigger = {
	OR = {
		POD_is_hungerless_trigger = yes
		has_character_flag = drunk_grail_flag
		any_character_artifact = { has_variable = blood_horn }
	}
	has_POD_vamp_blood_trigger = yes
}

# GUI only, for determining whether to show/hide red beast dialogue
# for frenzy, use cant_frenzy_trigger
POD_has_a_beast_trigger = {
	has_POD_vamp_blood_trigger = yes
	NOR = {
		has_trait = golconda
		has_character_modifier = heart_of_darkness_modifier
		has_character_flag = drunk_grail_flag
		has_character_modifier = ritual_purge_the_inner_demon_modifier
	}
}

POD_knows_about_golconda_trigger = {
	OR = {
		has_character_flag = knows_about_golconda
		is_salubri_trigger = yes
		POD_has_golconda_trait_trigger = yes
		any_character_artifact = {
			has_variable = golconda_lore
		}
	}
}

POD_knows_about_golconda_coterie_trigger = {
	OR = {
		POD_knows_about_golconda_trigger = yes
		var:coterie ?= {
			any_in_list = {
				variable = members
				POD_knows_about_golconda_trigger = yes
			}
		}
	}
}

POD_is_on_golconda_trigger = {
	OR = {
		POD_has_golconda_trait_trigger = yes
		POD_is_on_golconda_journey_trigger = yes
	}
}

POD_has_golconda_trait_trigger = {
	OR = {
		has_trait = golconda
		has_trait = golconda_seeker
	}
}

POD_is_on_golconda_journey_trigger = {
	OR = {
		POD_is_on_journey_trigger = { TYPE = golconda_2 }
		POD_is_on_journey_trigger = { TYPE = golconda_3 }
	}
}
