
POD_enabled_node = {
	NOT = { is_target_in_variable_list = { name = disabled_nodes target = flag:$NODE$ } }
}

POD_artifact_can_be_enhanced = {
	custom_tooltip = { text = POD_artifact_can_be_enhanced artifact_enhance_amount < artifact_enhance_limit }

POD_crafting_has_artifact_shards_matching_rarity = {
	trigger_if = {
		limit = { rarity = illustrious }
		scope:owner = { num_illustrious_artifact_pieces_trigger = { VALUE = $VALUE$ } }
	}
	trigger_else_if = {
		limit = { rarity = famed }
		scope:owner = { num_famed_artifact_pieces_trigger = { VALUE = $VALUE$ } }
	}
	trigger_else_if = {
		limit = { rarity = masterwork }
		scope:owner = { num_masterwork_artifact_pieces_trigger = { VALUE = $VALUE$ } }
	}
	trigger_else = { scope:owner = { num_common_artifact_pieces_trigger = { VALUE = $VALUE$ } } }
}

num_common_artifact_pieces_trigger = {
	custom_description = {
		text = has_common_artifact_pieces
		value = $VALUE$
		common_artifact_piece_reserve >= $VALUE$
	}
}
num_masterwork_artifact_pieces_trigger = {
	custom_description = {
		text = has_masterwork_artifact_pieces
		value = $VALUE$
		masterwork_artifact_piece_reserve >= $VALUE$
	}
}
num_famed_artifact_pieces_trigger = {
	custom_description = {
		text = has_famed_artifact_pieces
		value = $VALUE$
		famed_artifact_piece_reserve >= $VALUE$
	}
}
num_illustrious_artifact_pieces_trigger = {
	custom_description = {
		text = has_illustrious_artifact_pieces
		value = $VALUE$
		illustrious_artifact_piece_reserve >= $VALUE$
	}
}
