
has_character_material_forge_trigger = {
	var:current_method ?= {
		OR = {
			this = flag:fleshcraft
			this = flag:shape_flesh
			this = flag:catgut
			this = flag:mold
			this = flag:soulforge
			this = flag:fetishsmith_forced
			this = flag:fetishsmith
			this = flag:enchant_object
		}
	}
}

POD_forge_create_trigger = {
	switch = {
		trigger = var:current_method
		flag:fleshcraft = {
			var:craft_creator = { has_perk = living_testudo_perk }
			trigger_if = {
				limit = { has_variable = character_material }
				custom_tooltip = { 
					text = commit_suicide_decision
					NOT = { var:craft_creator = var:character_material } 
				}
				var:character_material ?= { 
					POD_is_not_clone_trigger = yes
					POD_is_not_spirit_trigger = yes
					not_has_POD_resurrection_trigger = yes
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = no_character_material
					always = no
				}
			}
		}
		flag:shape_flesh = {
			var:craft_creator = { has_perk = shape_flesh_perk }
			trigger_if = {
				limit = { has_variable = character_material }
				custom_tooltip = { 
					text = commit_suicide_decision
					NOT = { var:craft_creator = var:character_material } 
				}
				var:character_material ?= { 
					POD_is_not_clone_trigger = yes
					POD_is_not_spirit_trigger = yes
					not_has_POD_resurrection_trigger = yes
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = no_character_material
					always = no
				}
			}
		}
		flag:mold = {
			var:craft_creator = { 
				has_perk = mold_perk
				has_perk = realm_nature_3_perk
			}
			trigger_if = {
				limit = { has_variable = character_material }
				custom_tooltip = { 
					text = commit_suicide_decision
					NOT = { var:craft_creator = var:character_material } 
				}
				var:character_material = { 
					POD_is_not_clone_trigger = yes
					POD_is_not_spirit_trigger = yes
					not_has_POD_resurrection_trigger = yes
				}
				realm_actor_custom = {
					ACTOR = var:craft_creator
					RECIPIENT = var:character_material
				}
				realm_fae_custom = {
					ACTOR = var:craft_creator
					RECIPIENT = var:character_material
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = no_character_material
					always = no
				}
			}
		}
		flag:catgut = {
			var:craft_creator = { has_perk = catgut_perk }
			trigger_if = {
				limit = { has_variable = character_material }
				custom_tooltip = { 
					text = commit_suicide_decision
					NOT = { var:craft_creator = var:character_material } 
				}
				var:character_material = { 
					has_trait = shapeshifter
					POD_is_not_clone_trigger = yes
					POD_is_not_spirit_trigger = yes
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = no_character_material
					always = no
				}
			}
		}
		flag:soulforge = {
			trigger_if = {
				limit = { var:current_recipe = flag:white_jade_forge }
				var:craft_creator = { has_perk = create_white_jade_perk }
			}
			trigger_else_if = {
				limit = {
					OR = {
						is_book_forge_trigger = { RECIPE = var:current_recipe } 
						is_furniture_forge_trigger = { RECIPE = var:current_recipe } 
					}
				}
				var:craft_creator = { has_trait = arcanos_moliate }
			}
			trigger_else_if = {
				limit = { var:craft_creator = { has_perk = ectoplasmic_generation_5_perk } }
			}
			trigger_else = {
				var:craft_creator = { has_trait = arcanos_flux }
			}
			trigger_if = {
				limit = { has_variable = character_material }
				custom_tooltip = { 
					text = commit_suicide_decision
					NOT = { var:craft_creator = var:character_material } 
				}
				var:character_material = { 
					has_trait = wraith 
					custom_description = {
						text = POD_not_in_possession
						num_of_relation_possessed_target = 0
					}
					# trigger_if = {
					# 	limit = { NOT = { is_imprisoned_by = root } }
					# 	var:craft_creator = { has_perk = ectoplasmic_generation_5_perk }
					# }
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = no_character_material
					always = no
				}
			}
		}
		flag:fetishsmith_forced = {
			var:craft_creator = { 
				custom_tooltip = {
					text = creator_need_shapeshifter_large_tt
					POD_can_equip_fetish_trigger = yes

				}
			}
			trigger_if = {
				limit = { has_variable = character_material }
				custom_tooltip = { 
					text = commit_suicide_decision 
					NOT = { var:craft_creator = var:character_material } 
				}
				var:character_material = { 
					has_trait = spirit
					custom_tooltip = {
						text = possessing_artifact_tt
						NOT = { has_variable = stone_prison }
					}
					is_imprisoned_by = root
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = no_character_material
					always = no
				}
			}
		}
		flag:fetishsmith = {
			var:craft_creator = { 
				custom_tooltip = {
					text = creator_need_shapeshifter_large_tt
					POD_can_equip_fetish_trigger = yes
				} 
			}
			trigger_if = {
				limit = { has_variable = character_material }
				trigger_if = {
					limit = { NOT = { root = var:character_material } }
					has_relation_spirit_companion = var:character_material
				}
				var:character_material = { 
					has_trait = spirit 
					custom_tooltip = {
						text = possessing_artifact_tt
						NOT = { has_variable = stone_prison }
					}
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = no_character_material
					always = no
				}
			}
		}
		flag:shape_true_jade = {
			var:craft_creator = { has_perk = shape_true_jade_perk }
			trigger_if = {
				limit = { has_variable = artifact_material }
				custom_tooltip = {
					text = "is_wraith_jade"
					var:artifact_material = { has_variable = wraith_jade }
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = no_artifact_material
					always = no
				}
			}
		}
		flag:frost_facsimile = {
			var:craft_creator = { 
				has_perk = frost_facsimile_perk 
				has_perk = realm_prop_1_perk 
			}
		}
		flag:eightfold_yin_mantle = {
			var:craft_creator = { has_perk = eightfold_yin_mantle_perk }
		}
		flag:corrupt_relic = {
			var:craft_creator = { has_perk = corrupt_relic_perk }
		}
		flag:tainted_relic = {
			var:craft_creator = { has_perk = tainted_relic_perk }
			custom_tooltip = {
				text = "one_tainted_relic"
				NOT = { any_character_artifact = { has_variable = tainted_relic } }
			}
		}
		flag:horrid_blade = {
			var:craft_creator = { has_perk = horrid_blade_of_the_demons_perk }
		}
		flag:blood_blade = {
			var:craft_creator = { has_perk = blood_blade_perk }
		}
		flag:goblin_sword = {
			var:craft_creator = { has_perk = goblin_sword_perk }
		}
		flag:blade_of_stone = {
			var:craft_creator = { has_perk = blade_of_stone_perk }
		}
		flag:demon_weapon = {
			var:craft_creator = { has_character_modifier = demonic_feature_demon_weapon_modifier }
			custom_tooltip = {
				text = "one_demon_weapon"
				NOT = { any_character_artifact = { has_variable = demon_weapon } }
			}
		}
		flag:reweave_common = {
			var:craft_creator = {
				custom_tooltip = {
					text = can_reweave_tt
					OR = {
						has_perk = major_restoration_perk 
						has_perk = beckon_relic_perk 
						has_perk = create_object_perk 
						has_trait = arcanos_flux
					}
				}
			}
		}
		flag:reweave_masterwork = {
			var:craft_creator = {
				custom_tooltip = {
					text = can_reweave_tt
					OR = {
						has_perk = major_restoration_perk 
						has_perk = beckon_relic_perk 
						has_perk = create_object_perk 
						has_trait = arcanos_flux
					}
				}
			}
		}
		flag:reweave_famed = {
			var:craft_creator = { 
				OR = {
					has_perk = greater_restoration_perk 
					has_perk = relic_dowsing_perk 
				}
			}
		}
		flag:reweave_illustrious = {
			var:craft_creator = { 
				OR = {
					has_perk = princely_restoration_perk 
					has_perk = reintegrate_relics_perk 
				}
			}
		}
		flag:enchant_object = {
			var:craft_creator = { has_perk = enchant_object_perk }
			trigger_if = {
				limit = { has_variable = character_material }
				# var:craft_creator = { has_perk = anchor_the_soul_perk } # When player can get other lore?
				custom_tooltip = { 
					text = commit_suicide_decision
					NOT = { var:craft_creator = var:character_material } 
				}
				var:character_material = {
					custom_description = {
						text = POD_not_in_possession
						num_of_relation_possessed_target = 0
					}
					trigger_if = {
						limit = { NOT = { this = root } }
						is_imprisoned_by = root
					}
					NOT = { has_trait = automaton }
				}
			}
		}
		flag:ectoplasmic_artifact = {
			var:craft_creator = { has_perk = ectoplasmic_generation_3_perk }
		}
	}
}
