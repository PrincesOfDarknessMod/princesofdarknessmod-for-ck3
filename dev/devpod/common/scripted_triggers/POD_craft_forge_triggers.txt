
has_character_material_forge_trigger = {
	var:current_method ?= {
		OR = {
			this = flag:fleshcraft
			this = flag:catgut
			this = flag:mold
			this = flag:soulforge
			this = flag:fetishsmith_forced
			this = flag:fetishsmith
		}
	}
}

has_artifact_material_forge_trigger = {
	var:current_method ?= flag:shape_true_jade
}

is_armor_forge_trigger = {
	$RECIPE$ = {
		OR = {
			this = flag:armor_plate_forge
			this = flag:armor_scale_forge
			this = flag:armor_lamellar_forge
			this = flag:armor_laminar_forge
			this = flag:armor_brigandine_forge
			this = flag:armor_mail_forge
		}
	}
}

is_weapon_forge_trigger = {
	$RECIPE$ = {
		OR = {
			this = flag:axe_forge
			this = flag:hammer_forge
			this = flag:spear_forge
			this = flag:mace_forge
			this = flag:dagger_forge
			this = flag:sword_forge
		}
	}
}

is_shinies_forge_trigger = {
	$RECIPE$ = {
		OR = {
			this = flag:regalia_forge
			this = flag:crown_forge
			this = flag:brooch_forge
		}
	}
}

is_book_forge_trigger = {
	$RECIPE$ = flag:book_forge
}

is_furniture_forge_trigger = {
	$RECIPE$ = {
		OR = {
			this = flag:goblet_forge
			this = flag:statue_forge
			this = flag:tapestry_forge
			this = flag:throne_forge
			this = flag:diamond_forge
		}
	}
}

is_trinket_forge_trigger = {
	$RECIPE$ = {
		OR = {
			this = flag:ring_forge
			this = flag:necklace_forge
		}
	}
}

POD_forge_create_trigger = {
	switch = {
		trigger = var:current_method
		flag:fleshcraft = {
			var:craft_creator = { has_perk = living_testudo_perk }
			trigger_if = {
				limit = { has_variable = character_material }
				custom_tooltip = { 
					text = commit_suicide_decision
					NOT = { var:craft_creator = var:character_material } 
				}
				var:character_material ?= { 
					POD_is_not_clone_trigger = yes
					POD_is_not_spirit_trigger = yes
					not_has_POD_resurrection_trigger = yes
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = no_character_material
					always = no
				}
			}
		}
		flag:mold = {
			var:craft_creator = { 
				has_perk = mold_perk
				has_perk = realm_nature_3_perk
			}
			trigger_if = {
				limit = { has_variable = character_material }
				custom_tooltip = { 
					text = commit_suicide_decision
					NOT = { var:craft_creator = var:character_material } 
				}
				var:character_material = { 
					POD_is_not_clone_trigger = yes
					POD_is_not_spirit_trigger = yes
					not_has_POD_resurrection_trigger = yes
				}
				realm_actor_custom = {
					ACTOR = var:craft_creator
					RECIPIENT = var:character_material
				}
				realm_fae_custom = {
					ACTOR = var:craft_creator
					RECIPIENT = var:character_material
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = no_character_material
					always = no
				}
			}
		}
		flag:catgut = {
			var:craft_creator = { has_perk = catgut_perk }
			trigger_if = {
				limit = { has_variable = character_material }
				custom_tooltip = { 
					text = commit_suicide_decision
					NOT = { var:craft_creator = var:character_material } 
				}
				var:character_material = { 
					has_trait = shapeshifter
					POD_is_not_clone_trigger = yes
					POD_is_not_spirit_trigger = yes
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = no_character_material
					always = no
				}
			}
		}
		flag:soulforge = {
			trigger_if = {
				limit = {
					OR = {
						is_book_forge_trigger = { RECIPE = var:current_recipe } 
						is_furniture_forge_trigger = { RECIPE = var:current_recipe } 
					}
				}
				var:craft_creator = { has_trait = arcanos_moliate }
			}
			trigger_else = {
				var:craft_creator = { has_trait = arcanos_inhabit }
			}
			trigger_if = {
				limit = { var:current_recipe = flag:white_jade_forge }
				var:craft_creator = { has_perk = create_white_jade_perk }
			}
			trigger_if = {
				limit = { has_variable = character_material }
				custom_tooltip = { 
					text = commit_suicide_decision
					NOT = { var:craft_creator = var:character_material } 
				}
				var:character_material = { 
					has_trait = wraith 
					custom_description = {
						text = POD_not_in_possession
						num_of_relation_possessed_target = 0
					}
					trigger_if = {
						limit = { NOT = { is_imprisoned_by = root } }
						var:craft_creator = { has_perk = ectoplasmic_generation_perk }
					}
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = no_character_material
					always = no
				}
			}
		}
		flag:fetishsmith_forced = {
			var:craft_creator = { 
				custom_tooltip = {
					text = creator_need_shapeshifter_large_tt
					POD_can_equip_fetish_trigger = yes
				}
			}
			trigger_if = {
				limit = { has_variable = character_material }
				custom_tooltip = { 
					text = commit_suicide_decision 
					NOT = { var:craft_creator = var:character_material } 
				}
				var:character_material = { 
					has_trait = spirit
					is_imprisoned_by = root
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = no_character_material
					always = no
				}
			}
		}
		flag:fetishsmith = {
			var:craft_creator = { 
				custom_tooltip = {
					text = creator_need_shapeshifter_large_tt
					POD_can_equip_fetish_trigger = yes
				} 
			}
			trigger_if = {
				limit = { has_variable = character_material }
				trigger_if = {
					limit = { NOT = { root = var:character_material } }
					has_relation_spirit_companion = var:character_material
				}
				var:character_material = { has_trait = spirit }
			}
			trigger_else = {
				custom_tooltip = {
					text = no_character_material
					always = no
				}
			}
		}
		flag:corrupt_relic = {
			var:craft_creator = { 
				has_trait = earthbound 
				has_perk = corrupt_relic_perk 
			}
		}
		flag:tainted_relic = {
			var:craft_creator = { 
				has_trait = shadow
				has_perk = tainted_relic_perk 
			}
			custom_tooltip = {
				text = "one_tainted_relic"
				NOT = { any_character_artifact = { has_variable = tainted_relic } }
			}
		}
		flag:shape_true_jade = {
			var:craft_creator = { has_perk = shape_true_jade_perk }
			trigger_if = {
				limit = { has_variable = artifact_material }
				custom_tooltip = {
					text = "is_wraith_jade"
					var:artifact_material = { has_variable = wraith_jade }
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = no_artifact_material
					always = no
				}
			}
		}
	}
}
