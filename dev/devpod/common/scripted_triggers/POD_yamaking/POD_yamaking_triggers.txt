POD_is_middle_kingdom_breached_trigger = {
	is_target_in_global_variable_list = {
		name = unavailable_unique_decisions
		target = flag:POD_yamaking_breaching_the_middle_kingdom_objective_decision
	}
}
POD_has_enough_concession_trigger = {
	custom_description = {
		text = POD_has_enough_concession
		value = $AMOUNT$
		defiled_concession_value >= $AMOUNT$
	}
}
POD_concession_check_trigger = {
	save_temporary_scope_value_as = { name = concession_required value = 0 }
	switch = {
		trigger = $LEVEL$
		1 = { save_temporary_scope_value_as = { name = concession_required value = 0 } }
		2 = { save_temporary_scope_value_as = { name = concession_required value = 5 } }
		3 = { save_temporary_scope_value_as = { name = concession_required value = 10 } }
		4 = { save_temporary_scope_value_as = { name = concession_required value = 20 } }
		5 = { save_temporary_scope_value_as = { name = concession_required value = 30 } }
		6 = { save_temporary_scope_value_as = { name = concession_required value = 50 } }
	}
	trigger_if = {
		limit = {
			has_variable = main_defilement
			NOT = { var:main_defilement ?= flag:$DEFILEMENT$ }
		}
		save_temporary_scope_value_as = {
			name = concession_required
			value = {
				value = scope:concession_required
				multiply = 2
				min = 5
			}
		}
	}
	POD_has_enough_concession_trigger = { AMOUNT = scope:concession_required }
}
POD_defilements_has_enough_to_grant_defilement_generalized_trigger = {
	trigger_if = {
		limit = {
			OR = {
				panoplies_lifestyle_perk_points >= 1
				defilement_of_$TYPE$_granting_cost = 1000
			}
		}
		panoplies_lifestyle_perk_points >= 1
	}
	trigger_else = {
		panoplies_lifestyle_xp >= defilement_of_$TYPE$_granting_cost
	}
}
POD_defilements_has_enough_to_grant_defilement_trigger = {
	switch = {
		trigger = flag:$TYPE$
		flag:taint = { POD_defilements_has_enough_to_grant_defilement_generalized_trigger = { TYPE = taint } }
		flag:war = { POD_defilements_has_enough_to_grant_defilement_generalized_trigger = { TYPE = war } }
		flag:greed = { POD_defilements_has_enough_to_grant_defilement_generalized_trigger = { TYPE = greed } }
		flag:shadows = { POD_defilements_has_enough_to_grant_defilement_generalized_trigger = { TYPE = shadows } }
		flag:rage = { POD_defilements_has_enough_to_grant_defilement_generalized_trigger = { TYPE = rage } }
	}
}
