### Aye Tier Triggers ###
# Track scale: 0-100, tiers at 25/50/75

has_aye_tier_1 = {
	has_trait = laibon_spirit
	has_trait_xp = {
		trait = laibon_spirit
		track = aye
		value >= POD_laibon_spirit_xp_for_rank_1
	}
	NOT = {
		has_trait_xp = {
			trait = laibon_spirit
			track = aye
			value >= POD_laibon_spirit_xp_for_rank_2
		}
	}
}

has_aye_tier_2 = {
	has_trait = laibon_spirit
	has_trait_xp = {

		track = aye
		value >= POD_laibon_spirit_xp_for_rank_2
	}
	NOT = {
		has_trait_xp = {
			trait = laibon_spirit
			track = aye
			value >= POD_laibon_spirit_xp_for_rank_3
		}
	}
}

has_aye_tier_3 = {
