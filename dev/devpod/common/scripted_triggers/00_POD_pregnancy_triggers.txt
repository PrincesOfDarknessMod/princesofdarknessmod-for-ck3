# Only integrated splat for now
POD_race_pregnancy_compatibility_with_character_trigger = {
	trigger_if = {
		limit = { POD_race_pregnancy_trigger = yes } 
		$CHARACTER$ = { POD_race_fertility_care_trigger = yes }
	}

	trigger_if = {
		limit = { POD_race_fertility_care_trigger = yes }
		$CHARACTER$ = { POD_race_pregnancy_trigger = yes }
	}
}
