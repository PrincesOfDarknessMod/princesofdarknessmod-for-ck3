# For example, Gifts have a Gnosis cost.
# But since the triggers are splat-agnostic, any other resource can be used in its place.
# (e.g. Blood for Abominations)

# also see the priority list in common\scripted_effects\POD_energy_effects.txt

### Triggers for selecting the right energy type per splat
# (DO NOT USE THESE OUTSIDE THIS FILE, use the wrappers instead)

POD_has_enough_energy_max_5 = {
	save_temporary_scope_value_as = {
		name = amount_max_10
		value = {
			value = $AMOUNT$
			multiply = 2
		}
	}
	POD_has_enough_energy = {
		AMOUNT_MAX_5  = $AMOUNT$
		AMOUNT_MAX_10 = scope:amount_max_10
	}
}

POD_has_enough_energy_max_10 = {
	save_temporary_scope_value_as = {
		name = amount_max_5
		value = {
			value = $AMOUNT$
			multiply = 0.5
			ceiling = yes
		}
	}
	POD_has_enough_energy = {
		AMOUNT_MAX_5  = scope:amount_max_5
		AMOUNT_MAX_10 = $AMOUNT$
	}
}

POD_has_enough_energy = {
	trigger_if = {
		limit = { POD_splat_has_hunger_trigger = yes } # max 5
		POD_has_enough_energy_as_blood = { AMOUNT = $AMOUNT_MAX_5$ }
	}
	trigger_else_if = {
		limit = { POD_splat_has_angst_trigger = yes } # max 5
		POD_has_enough_energy_as_angst = { AMOUNT = $AMOUNT_MAX_5$ }
	}
	trigger_else_if = {
		limit = { POD_splat_has_pathos_trigger = yes } # max 5
		POD_has_enough_energy_as_pathos = { AMOUNT = $AMOUNT_MAX_5$ }
	}
	trigger_else_if = {
		limit = { POD_splat_has_essence_trigger = yes } # max 10
		POD_has_enough_energy_as_essence = { AMOUNT = $AMOUNT_MAX_10$ }
	}
	trigger_else_if = {
		limit = { POD_splat_has_resolve_trigger = yes } # max 10
		POD_has_enough_energy_as_resolve = { AMOUNT = $AMOUNT_MAX_10$ }
	}
	trigger_else_if = {
		limit = { POD_splat_has_gnosis_trigger = yes } # max 5
		POD_has_enough_energy_as_gnosis = { AMOUNT = $AMOUNT_MAX_5$ }
	}
	trigger_else_if = {
		limit = { POD_splat_has_glamour_trigger = yes } # max 5
		POD_has_enough_energy_as_glamour = { AMOUNT = $AMOUNT_MAX_5$ }
	}
	trigger_else_if = {
		limit = { POD_splat_has_dread_as_resource_trigger = yes } # scaled to max 10 for simplicity
		POD_has_enough_energy_as_dread = { AMOUNT = $AMOUNT_MAX_10$ }
	}
	trigger_else = {
		# static stress as fallback (scaled to max 10 for simplicity)
		POD_has_enough_energy_as_willpower = { AMOUNT = $AMOUNT_MAX_10$ }
	}
}


### Base triggers for checking a splat's energy reserves

# using a scale of 0-5
POD_has_enough_energy_as_blood = {
	trigger_if = {
		limit = { POD_has_infinite_blood_trigger = yes }
		always = yes
	}
	trigger_else = {
		custom_description = {
			text = POD_has_total_blood_reserves
			value = $AMOUNT$
			total_blood_reserves >= $AMOUNT$
		}
	}
}

# using a scale of 0-10, e.g. AMOUNT = 4 means 40 dread
POD_has_enough_energy_as_dread = {
	save_temporary_scope_value_as = {
		name = dread_overflow
		value = {
			value = $AMOUNT$
			multiply = 10 # each "energy point" equivalent to 10 dread
			subtract = total_stored_splat_energy
			min = 0
		}
	}
	dread >= scope:dread_overflow # TODO: check if this one's missing vanilla loc too
}

# using a scale of 0-10, e.g. AMOUNT = 4 means 40 static_stress
POD_has_enough_energy_as_willpower = {
	save_temporary_scope_value_as = {
		name = willpower_overflow
		value = {
			value = $AMOUNT$
			multiply = 10 # each "energy point" equivalent to 10 stress
			subtract = total_stored_splat_energy
			min = 0
			# match the discounted spend and its whole-stress rounding.
			multiply = pod_combat_kueijin_energy_cost_multiplier
			ceiling = yes
		}
	}
	trigger_if = {
		limit = { scope:willpower_overflow > 0 }
		save_temporary_scope_value_as = {
			name = max_allowed_stress
			value = {
				value = stress_cap_for_willpower
				subtract = scope:willpower_overflow
			}
		}
		stress < scope:max_allowed_stress
	}
	trigger_else = {
		always = yes
	}
}
