### Generic triggers

# using a scale of 0-10
# each "energy point" equivalent to 10 stress
POD_has_enough_willpower = {
	save_temporary_scope_value_as = {
		name = willpower_overflow
		value = {
			value = $AMOUNT$
			subtract = total_stored_energy
			min = 0
			multiply = 10
		}
	}
	trigger_if = {
		limit = { scope:willpower_overflow > 0 }
		save_temporary_scope_value_as = {
			name = max_allowed_stress
			value = {
				value = 300 # upper cap for stress
				subtract = scope:willpower_overflow
			}
		}
		stress < scope:max_allowed_stress
	}
	trigger_else = {
		always = yes
	}
}
