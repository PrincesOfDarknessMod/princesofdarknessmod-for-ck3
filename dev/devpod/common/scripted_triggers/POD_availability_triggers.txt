# Note that these triggers are used in A LOT of places in script.
# Adding complexity to these will have a big negative effect on performance.
# Optimize wherever possible.


########## basic triggers
########## (using NOT = {} to avoid complex triggers)

# excludes characters in deep sleep that can't do anything
# (can be targeted by interactions that wake this character up)
POD_is_not_torpid = {
	NOT = { has_trait = torpor }
}

# excludes feral characters who can't think or speak or follow orders
POD_is_not_wight = {
	NOT = { has_trait = wight }
	NOT = { has_trait = barghest } # TODO: merge into one trait?
}

# excludes robots who can perform simple tasks
POD_is_not_automaton = {
	NOT = { has_trait = automaton }
}

# excludes literal zombies
POD_is_not_zombu = {
	NOT = { has_trait = zombu }
}

# excludes literal spirits from the spirit wilds, who shouldn't show up in regular events
POD_is_not_spirit = {
	NOT = { has_trait = spirit }
}

# excludes remote-controlled constructs, e.g. made of shadow or glamour
POD_is_not_a_mindless_clone = {
	NOT = { has_character_flag = mindless_clone }
}

# excludes lore characters that shouldn't show up anywhere: charon, ur-shulgi, etc
POD_is_not_apocalyptic_background_character = {
	custom_description = {
		text = POD_do_not_disturb_text
		NOT = { has_character_flag = POD_do_not_disturb_flag }
	}
}


########## triggers for is_valid_showing_failures_only
########## (using NOT = {} to avoid complex triggers)

# excludes characters that can't do any tasks at all
# basically like the is_incapable trigger from vanilla
# (as of patch 1.15, is_incapable doesn't always seem to work for traits with incapacitating = yes)
POD_is_not_incapable = {
	POD_is_not_torpid = yes
	POD_is_not_wight = yes
}

# excludes characters that can only do simple tasks
# (not complex/intelligent/autonomous ones)
POD_is_not_robotic = {
	POD_is_not_automaton = yes
	POD_is_not_zombu = yes
	POD_is_not_a_mindless_clone = yes
}

# both of the above, to check for sapient characters
POD_is_not_mindless = {
	POD_is_not_incapable = yes
	POD_is_not_robotic = yes
}

POD_is_not_unreachable_trigger = {
	POD_is_not_apocalyptic_background_character = yes
	POD_is_not_trapped_in_umbra_trigger = yes
}

POD_is_not_blocked_from_regular_events = {
	POD_is_not_spirit = yes
}


########## triggers for events

# the generic "should this character show up in events and be able to talk?" trigger
POD_is_available = {
	POD_is_not_mindless = yes
	POD_is_not_unreachable_trigger = yes
	POD_is_not_blocked_from_regular_events = yes
}

POD_is_busy_with_ritual = {
	custom_tooltip = {
		text = POD_is_casting_ritual
		exists = var:current_journey
		OR = {
			var:current_journey = { var:journey_type = flag:ritual_blood_sorcery_casting }
			var:current_journey = { var:journey_type = flag:ritual_necromagic_casting }
		}
	}
}

POD_vampire_can_show_up_in_events = {
	NOT = { POD_vampire_is_defeated = { CHARACTER = $CHARACTER$ } }
	NOT = { this = $CHARACTER$ }
}

# for the "must be dead or in torpor" checks
POD_vampire_is_defeated = {
	trigger_if = {
		# hide objective if the character to be checked is you
		limit = { NOT = { this = $CHARACTER$ } }
		$CHARACTER$ = {
			custom_description = {
				text = POD_vampire_is_defeated_desc
				subject = this
				OR = {
					is_alive = no
					# torpor/wight
					POD_is_not_mindless = no
					# being turned into another splat also disqualifies you
					POD_is_vampire_trigger = no
				}
			}
		}
	}
}

# more generic version for other splats
# (don't use for wraiths, since getting turned into one counts as defeat)
POD_character_is_defeated = {
	trigger_if = {
		# hide objective if the character to be checked is you
		limit = { NOT = { this = $CHARACTER$ } }
		$CHARACTER$ = {
			custom_description = {
				text = POD_character_is_defeated_desc
				subject = this
				OR = {
					is_alive = no
					POD_is_not_mindless = no
					POD_is_wraith_trigger = yes
					POD_is_mortal_trigger = yes
				}
			}
		}
	}
}

	POD_hostile_interaction_is_shown_custom = {
		ACTOR = $ACTOR$
		RECIPIENT = $RECIPIENT$
	}
	NOT = {
		POD_custom_incompatible_splats_trigger = {
			ACTOR = $ACTOR$
			RECIPIENT = $RECIPIENT$
		}
	}
	$RECIPIENT$ = {
		POD_is_not_wight = yes
		POD_is_not_robotic = yes
	}
}
