POD_has_any_book_of_nod_tale_trigger = {
	any_character_artifact = {
		OR = {
			has_variable = book_of_nod
			has_variable = chronicle_of_caine
			has_variable = chronicle_of_shadows
			has_variable = chronicle_of_secrets
			has_variable = first_of_times
			has_variable = coming_of_lilith
			has_variable = liliths_magick
			has_variable = temptation_of_caine
			has_variable = zillahs_tale
			has_variable = tale_of_the_crone
			has_variable = tale_of_the_first_city
		}
	}
}
