##### 00_artifact_interactions

POD_gift_artifact_interaction_is_valid = {
	trigger_if = {
		limit = {
			exists = scope:target
			scope:actor = { NOT = { has_trait = wraith } }
			scope:recipient = { has_trait = wraith }
		}
		OR = {
			scope:actor = {
				trigger_if = {
					limit = { scope:target = { rarity = illustrious } }
					has_perk = sanctify_chattel_perk
				}
				trigger_else_if = {
					limit = { scope:target = { rarity = famed } }
					has_perk = princely_reflection_perk
				}
				trigger_else_if = {
					limit = { scope:target = { rarity = masterwork } }
					has_perk = majestic_reflection_perk
				}
				trigger_else = {
					has_perk = modest_reflection_perk
				}
			}
			scope:recipient = { has_perk = obliviate_perk }
		}
	}
	trigger_if = {
		limit = {
			scope:actor = { has_trait = wraith }
			scope:recipient = { NOT = { has_trait = wraith } }
		}
		scope:actor = { has_perk = cache_perk }
	}
}

##### 00_prison_interactions

POD_imprison_interaction_is_shown = {
	#Give time for very precarious start position to flourish instead of insta prison
	trigger_if = {
		limit = {
			years_from_game_start < 15
			POD_incompatible_splats_trigger = no
		}
		scope:actor = {
			is_ai = no
		}
	}
}

POD_imprison_interaction_is_valid_showing_failures_only = {
	POD_damage_spirit_trigger = yes
}

POD_move_to_dungeon_interaction_is_valid_showing_failures_only = {
	scope:actor = {
		custom_tooltip = {
			text = house_arrest_only_tooltip
			NOT = { faith = { has_doctrine_parameter = house_arrest_only } }
		}
	}
}

POD_ransom_interaction_is_shown = {
	NOT = {
		POD_inquisitor_on_unholy_trigger = { ACTOR = scope:actor RECIPIENT = scope:recipient }
	}
}

POD_release_from_prison_interaction_is_shown = {
	NAND = {
		scope:actor = { is_ai = yes }
		POD_inquisitor_on_unholy_trigger = { ACTOR = scope:actor RECIPIENT = scope:recipient }
	}
}

POD_execute_prisoner_interaction_is_shown = {
	scope:actor = {
		trigger_if = {
			limit = { is_ai = yes }
			POD_is_willing_to_kill_trigger = yes
		}
	}
}

POD_torture_interaction_is_shown = {
	scope:actor = {
		trigger_if = {
			limit = { is_ai = yes }
			POD_is_willing_to_commit_heinous_acts_trigger = yes
		}
	}
}
