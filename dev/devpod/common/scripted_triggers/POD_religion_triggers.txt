is_wrong_religion_splat_gui_trigger = {
	trigger_if = {
		limit = { has_trait = supehunter }
		NOT = { scope:new_faith = { religion = { is_in_family = rf_hunters } } }
	}
	trigger_else = {
		OR = {
			AND = {
				is_vampire_trigger = yes
				NOT = { scope:new_faith  = { religion = { is_in_family = rf_kindred } } }
			}
			AND = {
				trigger_if = {
					limit = { POD_is_samuel_haight_mode_enabled = yes }
					has_trait = dharma
				}
				trigger_else = {
					has_trait = kueijin
				}
				NOT = { scope:new_faith = { religion = { is_in_family = rf_kueijin } } }
			}
			AND = {
				has_trait =  shapeshifter
				NOR = {
					scope:new_faith = { religion = { is_in_family = rf_gaia } }
					scope:new_faith = { religion = { is_in_family = rf_wyrm } }
				}
			}
			AND = {
				has_trait = mummy
				NOT = { scope:new_faith  = { religion = { is_in_family = rf_mummies } } }
			}
			AND = {
				has_trait = wraith
				NOT = { scope:new_faith  = { religion = { is_in_family = rf_wraiths } } }
			}
			AND = {
				has_trait = demon
				NOT = { scope:new_faith  = { religion = { is_in_family = rf_demons } } }
			}
			AND = {
				has_trait = fae
				NOT = { scope:new_faith = { religion = { is_in_family = rf_fae } } }
			}
			AND = {
				POD_is_any_kind_of_ghoul_trigger = yes
				NOR = {
					scope:new_faith = { religion = { is_in_family = rf_kindred } }
					AND = {
						OR = {
							has_trait = kinfolk
							has_trait = shapeshifter
						}
						OR = {
							scope:new_faith = { religion = { is_in_family = rf_gaia } }
							scope:new_faith = { religion = { is_in_family = rf_wyrm } }
						}
					}
					AND = {
						OR = {
							has_trait = kinain
							has_trait = changeling
						}
						scope:new_faith = { religion = { is_in_family = rf_fae } }
					}
					AND = {
						is_a_neteru = yes
						scope:new_faith = { religion = { is_in_family = rf_mummies } }
					}
					AND = {
						is_a_cultist = yes
						scope:new_faith = { religion = { is_in_family = rf_demons } }
					}
				}		
			}
			AND = {
				POD_is_changeling_trigger = yes
				NOR = {
					scope:new_faith = { religion = { is_in_family = rf_fae } }
					AND = {
						has_trait = kinfolk
						OR = {
							scope:new_faith = { religion = { is_in_family = rf_gaia } }
							scope:new_faith = { religion = { is_in_family = rf_wyrm } }
						}
					}
					AND = {
						POD_is_any_kind_of_ghoul_trigger = yes
						scope:new_faith = { religion = { is_in_family = rf_kindred } }
					}
					AND = {
						is_a_neteru = yes
						scope:new_faith = { religion = { is_in_family = rf_mummies } }
					}
					AND = {
						is_a_cultist = yes
						scope:new_faith = { religion = { is_in_family = rf_demons } }
					}
				}		
			}

POD_has_illegal_religion_trigger = {
	trigger_if = {
		limit = {
			has_supernatural_race_trigger = no
		}
		always = no
	}
	trigger_else_if = {
		limit = {
			religion = { is_in_family = rf_ignorance }
		}
		always = yes # already has_supernatural_race_trigger = no
	}
	trigger_else_if = {
		limit = {
			religion = { is_in_family = rf_kindred }
		}
		has_POD_vamp_blood_trigger = no
	}
	trigger_else_if = {
		limit = {
			religion = { is_in_family = rf_kueijin }
		}
		trigger_if = {
			limit = { POD_is_samuel_haight_mode_enabled = yes }
			NOT = { has_trait = dharma }
		}
		trigger_else = {
			NOT = { has_trait = kueijin }
		}
	}
	trigger_else_if = {
		limit = {
			religion = { is_in_family = rf_hunters }
		}
		POD_is_any_kind_of_ghoul_trigger = no
		NOT = { has_trait = supehunter }
	}
	trigger_else_if = {
		limit = {
			religion = { 
				OR = {
					is_in_family = rf_gaia 
					is_in_family = rf_wyrm 
				}
			}
		}
		NOT = { has_trait = shapeshifter }
		NOT = { has_trait = kinfolk }
	}	
	trigger_else_if = {
		limit = {
			religion = { is_in_family = rf_wraiths }
		}
		NOT = { has_trait = wraith }
	}
	trigger_else_if = {
		limit = {
			religion = { is_in_family = rf_fae }
		}
		has_POD_fae_blood_trigger = no
	}
	trigger_else_if = {
		limit = {
			religion = { is_in_family = rf_mummies }
		}
		NOT = { has_trait = mummy }
	}
	trigger_else = {
		NOT = { religion = { is_in_family = rf_demons } } #Demons can corrupt everyone as part of their gameplay, no need to force rollback the religion if someone is of a demon faith	
		has_game_rule = world_of_monsters # Allow vanilla religion?
	}
}
