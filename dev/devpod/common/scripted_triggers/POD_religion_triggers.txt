pod_convert_trigger = {	
	trigger_if = {
		limit = {
			NAND = {
				$FAITH$.religion = { is_in_family = rf_demons }
				$CHARACTER$ = { is_a_cultist = yes }
			}
			$CHARACTER$ = { POD_is_spirit_trigger = no }
		}
		pod_convert_splat_trigger = { CHARACTER = $CHARACTER$ FAITH = $FAITH$ }
		pod_convert_half_splat_trigger = { CHARACTER = $CHARACTER$ FAITH = $FAITH$ }
	}
	trigger_if = {
		limit = {
			$FAITH$.religion = { is_in_family = rf_demons }
			$CHARACTER$ = {
				is_a_cultist = no
				NOT = { has_trait = demon }
			}
		}
		custom_tooltip = {
			text = demon_faith_requires_cultist_tooltip
			always = no
		}
	}
	custom_tooltip = {
		text = spirit_cant_convert_tooltip
		$CHARACTER$ = { POD_is_spirit_trigger = no }
	}
	custom_tooltip = {
		text = viaservilis_cant_convert_tooltip
		NOT = { $CHARACTER$.faith = faith:viaservilis } #Via servilis gargoyles can't be converted or given land
		NOT = { $FAITH$ = faith:viaservilis } #Via servilis gargoyles can't be converted or given land
	}
}

POD_has_illegal_religion_custom_trigger = {
	trigger_if = {
		limit = {
			$FAITH$ = { religion = { is_in_family = rf_ignorance } }
		}
		$CHARACTER$ = { NOT = { has_trait = mortal } }
	}
	trigger_else_if = {
		limit = {
			$FAITH$ = { religion = { is_in_family = rf_kindred } }
		}
		$CHARACTER$ = { has_POD_vamp_blood_trigger = no }
	}
	trigger_else_if = {
		limit = {
			$FAITH$ = { religion = { is_in_family = rf_kueijin } }
		}
		$CHARACTER$ = { 
			NOT = { has_trait = kueijin } 
			NOT = { has_trait = dhampyr }
		}
	}
	trigger_else_if = {
		limit = {
			$FAITH$ = { religion = { is_in_family = rf_hunters } }
		}
		$CHARACTER$ = { 
			NOT = { has_trait = supehunter }
			is_child_of_supehunter_trigger = no
		}
	}
	trigger_else_if = {
		limit = {
			$FAITH$ = { 
				OR = {
					religion = { is_in_family = rf_gaia }
					religion = { is_in_family = rf_wyrm }
				}
				is_spirit_faith_trigger = no
			}
		}
		$CHARACTER$ = { 
			NOT = { has_trait = fera }
			NOT = { has_trait = kinfolk }
		}
	}
	trigger_else_if = {
		limit = {
			$FAITH$ = { religion = { is_in_family = rf_wraiths } }
		}
		$CHARACTER$ = { NOT = { has_trait = wraith } }
	}
	trigger_else_if = {
		limit = {
			$FAITH$ = { religion = { is_in_family = rf_fae } }
		}
		$CHARACTER$ = { has_POD_fae_blood_trigger = no }
	}
	trigger_else_if = {
		limit = {
			$FAITH$ = { religion = { is_in_family = rf_mummies } }
		}
		$CHARACTER$ = { 
			NOT = { has_trait = mummy } 
			is_a_neteru = no
		}
	}
	trigger_else_if = {
		limit = {
			$FAITH$ = { religion = { is_in_family = rf_demons } }
		}
		$CHARACTER$ = { 
			NOT = { has_trait = demon } 
			is_a_cultist = no
		}
	}
	trigger_else_if = {
		limit = { 
			$FAITH$ = { is_spirit_faith_trigger = yes }
		}
		POD_is_spirit_trigger = no
	}
	trigger_else = {
		has_game_rule = world_of_monsters # Allow vanilla religion?
	}
}
