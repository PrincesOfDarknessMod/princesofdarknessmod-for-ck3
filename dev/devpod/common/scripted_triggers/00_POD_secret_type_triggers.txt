#Triggers for Shunned and Criminal Secret Types for POD

# SECRET DIABLERIST 

secret_diablerist_is_valid_trigger = {
	$OWNER$ = { 
		NOT = { 
			has_trait = diablerist 
		} 
	}
}

secret_diablerist_is_shunned_trigger = {
	$OWNER$ = {
		OR = {
			faith = { has_doctrine_parameter = vampirediablerie_shunned }
			any_liege_or_above = { faith = { has_doctrine_parameter = vampirediablerie_shunned } }
		}
	}
}

secret_diablerist_is_criminal_trigger = {
	$OWNER$ = {
		OR = {
			faith = { has_doctrine_parameter = vampirediablerie_illegal }
			any_liege_or_above = { faith = { has_doctrine_parameter = vampirediablerie_illegal } }
		}
	}
}

# SECERT DIABLERIST - END
