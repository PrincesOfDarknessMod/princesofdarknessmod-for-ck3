POD_has_any_terrestrial_aspect_trigger = {
	OR = {
		has_trait = terrestrial_aspect_air
		has_trait = terrestrial_aspect_earth
		has_trait = terrestrial_aspect_fire
		has_trait = terrestrial_aspect_water
		has_trait = terrestrial_aspect_wood
	}
}

POD_can_afford_charm_perk_trigger = {
	save_temporary_scope_value_as = {
		name = charm
		value = flag:$CHARM$
	}
	save_temporary_scope_value_as = {
		name = level
		value = $LEVEL$
	}
	charms_lifestyle_perk_points >= POD_charm_perk_cost
}

POD_is_charm_favorite_trigger = {
	switch = {
		trigger = $CHARM$
		flag:air_charms = { has_trait = terrestrial_aspect_air }
		flag:earth_charms = { has_trait = terrestrial_aspect_earth }
		flag:fire_charms = { has_trait = terrestrial_aspect_fire }
		flag:water_charms = { has_trait = terrestrial_aspect_water }
		flag:wood_charms = { has_trait = terrestrial_aspect_wood }
	}
}

POD_passive_charm_cost_macro_trigger = {
	OR = {
		any_in_list = {
			variable = possible_charms
			var:flag_name = flag:$FLAG$
		}
		has_perk = charm_$FLAG$_perk 
	}
}

POD_has_enough_environmental_essence_trigger = {
	custom_description = {
		text = POD_has_enough_environmental_essence
		value = $VALUE$
		POD_unallocated_environmental_essence_value >= $VALUE$
	}
}

POD_the_seed_of_a_dragon_macro1 = {
	trigger_if = { # to remove localization `All of These`
		limit = { always = yes }
		POD_is_terrestrial_exalted_trigger = yes 
		opinion = { target = ROOT value >= 100 }
		is_ruler = no
		POD_has_exalted_rating_equal_or_above_trigger = { VALUE = 5 }
	}
}
POD_the_seed_of_a_dragon_macro2 = {
	trigger_if = {
		limit = { flag:$DRAGON$ = flag:mela }
		AND = {
			has_trait = terrestrial_aspect_air
			diplomacy >= 20
		}
	}
	trigger_if = {
		limit = { flag:$DRAGON$ = flag:pasiap }
		AND = {
			has_trait = terrestrial_aspect_earth
			stewardship >= 20
		}
	}
	trigger_if = {
		limit = { flag:$DRAGON$ = flag:hesiesh }
		AND = {
			has_trait = terrestrial_aspect_fire
			martial >= 20
		}
	}
	trigger_if = {
		limit = { flag:$DRAGON$ = flag:daana_d }
		AND = {
			has_trait = terrestrial_aspect_water
			intrigue >= 20
		}
	}
	trigger_if = {
		limit = { flag:$DRAGON$ = flag:sextes_jylis }
		AND = {
			has_trait = terrestrial_aspect_wood
			learning >= 20
		}
	}
}
POD_the_seed_of_a_dragon_macro = {
	POD_the_seed_of_a_dragon_macro1 = yes
	POD_the_seed_of_a_dragon_macro2 = { DRAGON = $DRAGON$ }
}
