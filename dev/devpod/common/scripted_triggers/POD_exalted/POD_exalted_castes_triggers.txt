POD_general_terrestrial_condition_trigger = {
	always = no
}
POD_terrestrial_aspect_air_condition_trigger = {
	OR = {
		POD_general_terrestrial_condition_trigger = yes
		capital_province.duchy = {
			any_in_de_jure_hierarchy = {
				tier = tier_barony
				title_province = {
					OR = {
						terrain = mountains
						terrain = desert_mountains
					}
				}
			}
		}
	}
}
