POD_knows_ancient_sorcery_trigger = {
	$SORCERY$ = { save_temporary_scope_as = sorcery_flag }
	POD_ancient_sorcery_flag_and_perk_trigger_injector = { TRIGGER = POD_knows_ancient_sorcery_macro }
}
POD_knows_ancient_sorcery_macro = {
	trigger_if = {
		limit = { scope:sorcery_flag ?= flag:$FLAG$ }
		has_perk = $PERK$
	}
}
