has_animal_resonance_trigger = {
	OR = {
		has_trait = abombwe
		has_trait = animalism
		has_trait = protean
		has_trait = serpentis
		has_trait = spiritus
		has_trait = abombweadvanced
		has_trait = animalismadvanced
		has_trait = proteanadvanced
		has_trait = serpentisadvanced
		has_trait = spiritusadvanced
		has_trait = kinfolk
		has_trait = shapeshifter
	}
}

POD_has_enough_blood_for_discipline = {
	custom_description = {
		text = POD_has_total_blood_reserves
		value = $AMOUNT$
		total_blood_reserves >= $AMOUNT$
	}
}

POD_has_mundane_blood_trigger = {
	OR = {
		has_main_supernatural_race_trigger = no
		POD_is_mage_trigger = yes
	}
}

POD_will_die_from_feeding_trigger = {
	trigger_if = {
		limit = { is_vampire_trigger = no }
		drinkable_blood_left < $AMOUNT$
	}
	trigger_else = {
		always = no
	}
}

POD_bloodloss_causes_piety_loss_trigger = {
	trigger_if = {
		limit = { has_supernatural_race_trigger = no }
		1 >= { value = drinkable_blood_left subtract = $AMOUNT$ }
	}
	trigger_else = {
		always = no
	}
}

POD_has_enough_hunger = {
	trigger_if = {
		limit = { $COUNT$ = 5 }
		NOT = { has_character_modifier = bloodhunger5_modifier }
		NOT = { has_character_modifier = bloodhunger4_modifier }
		NOT = { has_character_modifier = bloodhunger3_modifier }
		NOT = { has_character_modifier = bloodhunger2_modifier }
		NOT = { has_character_modifier = bloodhunger1_modifier }
	}
	trigger_else_if = {
		limit = { $COUNT$ = 4 }
		NOT = { has_character_modifier = bloodhunger5_modifier }
		NOT = { has_character_modifier = bloodhunger4_modifier }
		NOT = { has_character_modifier = bloodhunger3_modifier }
		NOT = { has_character_modifier = bloodhunger2_modifier }
	}
	trigger_else_if = {
		limit = { $COUNT$ = 3 }
		NOT = { has_character_modifier = bloodhunger5_modifier }
		NOT = { has_character_modifier = bloodhunger4_modifier }
		NOT = { has_character_modifier = bloodhunger3_modifier }
	}
	trigger_else_if = {
		limit = { $COUNT$ = 2 }
		NOT = { has_character_modifier = bloodhunger5_modifier }
		NOT = { has_character_modifier = bloodhunger4_modifier }

	}
	trigger_else_if = {
		limit = { $COUNT$ = 1 }
		NOT = { has_character_modifier = bloodhunger5_modifier }
	}
	trigger_else = {
		always = yes
	}
}
