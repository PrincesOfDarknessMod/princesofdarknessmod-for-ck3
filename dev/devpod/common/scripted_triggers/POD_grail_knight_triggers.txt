		value >= 75
	}

POD_lacks_gkboons_perk = {
	NOT = { has_perk = $PERK$ }
}

POD_has_no_gkboons_perks = {
	POD_gkboons_type_injector_trigger = {
		TRIGGER = POD_lacks_gkboons_perk
	}
}

POD_has_no_gkboons_perks_of_type = {
	POD_perk_injector_trigger_$TYPE$ = {
		TRIGGER = POD_lacks_gkboons_perk
	}
}

POD_has_no_gkboons_perks_of_type_tier = {
	POD_perk_injector_trigger_$TYPE$_tier$TIER$ = {
		TRIGGER = POD_lacks_gkboons_perk
	}
}
