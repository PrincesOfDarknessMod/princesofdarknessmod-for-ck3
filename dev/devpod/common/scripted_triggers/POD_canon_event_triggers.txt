POD_is_valid_for_regional_event = {
	trigger_if = {
		limit = { is_landed = yes }
		any_realm_province = {
			geographical_region = $REGION$
		}
	}
	trigger_else = {
		domicile ?= { # for laamps and administrative
			domicile_location = {
				geographical_region = $REGION$
			}
		}
	}
}

POD_inquisition_is_blocked_from_spawning = {
	is_target_in_global_variable_list = {
		name = unavailable_unique_decisions
		target = flag:destroy_inquisition_objective_decision
	}
}

POD_bookmark_has_flaying_plague = {
	POD_bookmark_has_fera = yes
	NOT = { has_game_rule = pod_flaying_plague_disabled }
	game_start_date <= 1240.1.1
}
