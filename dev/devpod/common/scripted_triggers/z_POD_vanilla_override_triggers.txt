##### common\scripted_triggers\00_religious_triggers.txt

# these are rewritten to use trait scopes instead of flags so they're more reusable

trait_is_shunned_in_faith_trigger = {
	trait_is_shunned_in_faith_use_scope_trigger = {
		TRAIT = trait:$TRAIT$
		FAITH = $FAITH$
		GENDER_CHARACTER = $GENDER_CHARACTER$
	}
}

	}
}

has_any_shunned_or_criminal_trait_in_faith_trigger = {
	$CHARACTER$ = {
		any_character_trait = {
			OR = {
				trait_is_shunned_in_faith_use_scope_trigger = {
					TRAIT = this
					FAITH = $FAITH$
					GENDER_CHARACTER = $CHARACTER$
				}
				trait_is_criminal_in_faith_use_scope_trigger = {
					TRAIT = this
					FAITH = $FAITH$
					GENDER_CHARACTER = $CHARACTER$
				}
			}
		}
	}
}
