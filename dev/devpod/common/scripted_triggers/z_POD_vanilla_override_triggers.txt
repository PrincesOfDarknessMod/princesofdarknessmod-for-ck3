# This file contains triggers from vanilla that were changed extensively.
# Since these changes would make the Winmerge compatch process way too messy, they instead get overwritten entirely here.
# This works because the name of this file goes last (z_) alphabetically.

# If you add new triggers here, please add a comment noting which file they came from.


##### common\scripted_triggers\00_religious_triggers.txt

has_any_shunned_or_criminal_trait_in_faith_trigger = {
	$CHARACTER$ = {
		any_character_trait = {
			OR = {
				trait_is_shunned_in_faith_trigger = {
					TRAIT = this
					FAITH = $FAITH$
					GENDER_CHARACTER = $CHARACTER$
				}
				trait_is_criminal_in_faith_trigger = {
					TRAIT = this
					FAITH = $FAITH$
					GENDER_CHARACTER = $CHARACTER$
				}
			}
		}
	}
}
