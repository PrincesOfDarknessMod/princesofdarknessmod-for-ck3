

POD_has_enough_gnosis_for_gift = {
	custom_description = {
		text = POD_has_total_gnosis_reserves
		value = $COUNT$
		total_gnosis_reserves >= $COUNT$
	}
}



#Shapeshifts
POD_is_shapeshifted_werewolf_trigger = {
	OR = {
		has_character_modifier = POD_werewolf_form_crinos_modifier
		has_character_modifier = POD_werewolf_form_glabro_modifier
		has_character_modifier = POD_werewolf_form_hispo_modifier
		has_character_modifier = POD_werewolf_form_lupus_modifier
	}
}
POD_is_shapeshifted_bastet_trigger = {
	OR = {
		has_character_modifier = POD_bastet_form_crinos_modifier
		has_character_modifier = POD_bastet_form_sokto_modifier
		has_character_modifier = POD_bastet_form_chatro_modifier
		has_character_modifier = POD_bastet_form_feline_modifier
	}
}
