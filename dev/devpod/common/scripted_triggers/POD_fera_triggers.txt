
POD_has_fera_type_variable = {
	OR = {
		has_trait = $TRAIT$
		#AND = {
			#exists = this
			#is_alive = yes
			var:fera_type ?= $TRAIT$
		#}
	}
}

POD_has_access_to_fera_gifts = {
	POD_has_fera_type_variable = { TRAIT = $TRAIT$ }
	POD_splat_has_access_to_any_fera_gifts = yes
}

POD_splat_has_access_to_any_fera_gifts = {
	OR = {
		has_trait = shapeshifter
		has_trait = kinfolk
		has_trait = abomination
	}
}

POD_has_inheritable_fera_type = {
	POD_has_fera_type_variable = { TRAIT = $TRAIT$ }
	POD_splat_has_any_fera_genetics = yes
}

POD_splat_has_any_fera_genetics = {
	OR = {
		has_trait = shapeshifter
		has_trait = kinfolk
	}
}

POD_can_shapeshift_into_fera_forms = {
	POD_has_fera_type_variable = { TRAIT = $TRAIT$ }
	POD_splat_is_valid_for_fera_shapeshifting = yes
}

POD_splat_is_valid_for_fera_shapeshifting = {
	OR = {
		has_trait = shapeshifter
		has_trait = abomination
		has_trait = spirit
	}
}

POD_should_have_fera_type_variable = {
	NOT = { has_trait = shapeshifter }
	OR = {
		has_trait = kinfolk
		has_trait = abomination
		has_trait = spirit
	}
}

POD_has_any_inheritable_fera_type = {
	OR = {
		has_trait_with_flag = fera_type
		AND = {
			#exists = this
			#is_alive = yes
			exists = var:fera_type
			POD_splat_has_any_fera_genetics = yes
		}
	}
}




POD_has_enough_gnosis_for_gift = {
	custom_description = {
		text = POD_has_total_gnosis_reserves
		value = $COUNT$
		total_gnosis_reserves >= $COUNT$
	}
}
