POD_artifact_is_unavailable_trigger = {
	has_variable = unavailable_artifact
}

POD_artifact_is_destructible_trigger = {
	NOR = {
		has_variable = historical_unique_artifact
		artifact_type = animal
		AND = {
			has_variable = fetter_owner
			var:fetter_owner = artifact_owner
		}
	}
}

cant_frenzy_trigger = {
	OR = {
		any_equipped_character_artifact = {
			OR = {
				has_variable = blade_of_enoch
				has_variable = hand_nikolas
			}	
		}
		has_character_flag = drunk_grail_flag
		has_trait_with_flag = immune_to_frenzy_flag
		has_character_modifier = ritual_purge_the_inner_demon_modifier
		has_character_flag = suppressed_beast_flag
		has_character_modifier = armor_of_diamond_serenity_modifier
		#has_character_flag = immune_to_frenzy # from old bloodsorcery
	}
}
