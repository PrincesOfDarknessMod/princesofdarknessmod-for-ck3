POD_artifact_is_destructible_trigger = {
	always = yes
}

# Whether the artifact *should* be able to get damaged in events
POD_artifact_is_destructible_by_events_trigger = {
	POD_artifact_is_destructible_trigger = yes
	NOR = {
		has_variable = historical_unique_artifact
		artifact_type = animal
		AND = {
			has_variable = fetter_owner
			var:fetter_owner = artifact_owner
		}
	}
}

# Don't touch this
compare_artifact_rank_injector = {
    trigger_if = {
        limit = { $CURRENT_RANK$ >= $TARGET_RANK$ }
        has_artifact_modifier = artifact_$MOD$_$CURRENT_RANK$_modifier
    }
	trigger_else = {
		always = no
	}
}
compare_artifact_no_rank_injector = {
    trigger_if = {
        limit = { $CURRENT_RANK$ >= $TARGET_RANK$ }
        has_artifact_modifier = artifact_$MOD$_modifier
    }
	trigger_else = {
		always = no
	}
}
