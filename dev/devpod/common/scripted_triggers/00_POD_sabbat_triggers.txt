POD_can_be_cardinal_candidate_trigger = {
	religion = religion:sabbat_religion
	POD_is_sabbat_regent_trigger = no
	POD_is_cardinal_trigger = no
	has_trait = vampire
	NOT = { has_trait = torpor }
	NOT = { has_trait = excommunicated }
	is_alive = yes
}

POD_is_sabbat_regent_trigger = {
	custom_description = {
		text = is_sabbat_regent
		religion:sabbat_religion.var:regent ?= this
	}
}

POD_liege_is_sabbat_regent_trigger = {
	custom_tooltip = {
		text = liege_is_sabbat_regent
		NOT = { this = liege }
		OR = {
			religion:sabbat_religion.var:regent ?= liege
			religion:sabbat_religion.var:regent ?= liege.liege
			religion:sabbat_religion.var:regent ?= liege.liege.liege
		}
	}
}

POD_is_cardinal_trigger = {
	custom_description = {
		text = is_cardinal
		religion:sabbat_religion = {
			is_target_in_variable_list = { 
				name = cardinals
				target = prev
			}
		}
	}
}

POD_is_black_hand_trigger = {
	has_character_modifier = black_hand_membership_modifier
}

POD_is_sabbat_region_trigger = {
	save_temporary_scope_value_as = { 
		name = region_loca 
		value = flag:$REGION$ 
	}
	custom_tooltip = {
		text = cardinal_sabbat_region_tt
		religion:sabbat_religion = {
			any_in_list = {
				variable = cardinals
				var:cardinal_region = flag:$REGION$
			}
		}
	}
}

POD_potential_sabbat_region_trigger = {
	custom_tooltip = {
		text = sabbat_$REGION$_tt
		any_county_in_region = {
			region = $REGION$
			percent >= 0.33
			POD_is_sabbat_county_trigger = yes
		}
	}
}

POD_is_sabbat_county_trigger = {
	OR = {
		religion = religion:sabbat_religion
		holder = { religion = religion:sabbat_religion }
	}
}
