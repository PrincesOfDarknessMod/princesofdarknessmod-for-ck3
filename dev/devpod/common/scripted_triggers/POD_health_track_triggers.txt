POD_has_wrong_wound_trait = {
	save_temporary_scope_value_as = {
		name  = trait_xp
		value = POD_health_track_percentage
	}
	trigger_if = {
		limit = { scope:trait_xp >= POD_wound_3_xp_threshold }
		NOT = { has_trait = wounded_3 }
	}
	trigger_else_if = {
		limit = { scope:trait_xp >= POD_wound_2_xp_threshold }
		NOT = { has_trait = wounded_2 }
	}
	trigger_else_if = {
		limit = { scope:trait_xp >= POD_wound_1_xp_threshold }
		NOT = { has_trait = wounded_1 }
	}
	trigger_else_if = {
		limit = { scope:trait_xp >= 1 }
		NOT = { has_trait = POD_wounded }
	}
	trigger_else = { has_trait = wounded }
}

# need to pass DEATH_REASON
POD_has_immunity_to_damage_type = {
	trigger_if = {
		limit = { POD_is_death_by_fire = { DEATH_REASON = $DEATH_REASON$ } }
		POD_immune_to_fire_trigger = yes
	}
	# Combat Mini-game: splat-default immunity for the matrix '-'
	# cells. These reasons are only ever dealt cross-splat by the
	# combat minigame, so the branches are inert outside it.
	trigger_else_if = {
		limit = { $DEATH_REASON$ = flag:death_sun }
		# Sun burns only vampires, abominations, and kuei-jin; every other splat is
		# immune. The perk-based sun immunity still exempts those three.
		OR = {
			NOR = { POD_is_vampire_or_abomination_trigger = yes  is_kueijin_trigger = yes }
			POD_immune_to_sun_damage_trigger = yes
		}
	}
	trigger_else_if = {
		limit = { $DEATH_REASON$ = flag:death_true_faith }
		# True faith harms only the damned and the fae; changelings take it superficial
		# (so are NOT immune - excluded here); the mundane are immune.
		NOR = {
			POD_is_vampire_or_abomination_trigger = yes
			POD_is_demon_trigger = yes
			is_kueijin_trigger = yes
			POD_is_wraith_trigger = yes
			POD_is_fae_trigger = yes
			POD_is_changeling_trigger = yes
		}
	}
	trigger_else_if = {
		limit = { $DEATH_REASON$ = flag:death_poison }
		# The undead cannot be poisoned.
		OR = {
			POD_is_vampire_or_abomination_trigger = yes
			is_kueijin_trigger = yes
			POD_is_wraith_trigger = yes
			POD_is_spirit_trigger = yes
		}
	}
	trigger_else = {
		always = no
	}
}

# need to pass DEATH_REASON
POD_damage_type_is_aggravated = {
	OR = {
		# Combat Mini-game: 
		# Each elemental death_reason is aggravated only against the splats the vulnerability
		# matrix marks 'A'. The non-matrix reasons further down stay unconditional.
		# Fire burns all flesh but only scorches the incorporeal (wraith/spirit).
		AND = {
			POD_is_death_by_fire = { DEATH_REASON = $DEATH_REASON$ }
			NOR = { POD_is_wraith_trigger = yes  POD_is_spirit_trigger = yes }
		}
		# Sun is a vampire / abomination / kuei-jin bane (others immune - see immunity);
		# the ash-and-fire shintai downgrades it to superficial.
		AND = {
			$DEATH_REASON$ = flag:death_sun
			OR = { POD_is_vampire_or_abomination_trigger = yes  is_kueijin_trigger = yes }
			POD_takes_sun_damage_as_superficial = no
		}
		# Lightning arcs through flesh and ectoplasm alike - aggravated against
		# every splat, with no immunity cell in the matrix.
		$DEATH_REASON$ = flag:death_lightning
		# Cold bites mortals aggravated; supernaturals endure it.
		AND = { $DEATH_REASON$ = flag:death_cold       POD_is_mortal_trigger = yes }
		# Poison rots living mortals; the undead are immune (immunity trigger).
		AND = { $DEATH_REASON$ = flag:death_poison  POD_weak_to_poison_trigger = yes }
		$DEATH_REASON$ = flag:death_self_mutilation
		$DEATH_REASON$ = flag:death_duel_aggravated
		$DEATH_REASON$ = flag:death_consequence_of_pact
		AND = {
			$DEATH_REASON$ = flag:death_silver
			POD_weak_to_silver_trigger = yes
		}
		AND = {
			$DEATH_REASON$ = flag:death_gold
			POD_weak_to_gold_trigger = yes
		}
		# Cold iron is the fae bane (true fae + changelings).
		AND = {
			$DEATH_REASON$ = flag:death_cold_iron
			POD_weak_to_cold_iron_trigger = yes
		}
		$DEATH_REASON$ = flag:death_sea_creatures
		# Balefire is the Wyrm's fire - unlike ordinary fire it sears the
		# incorporeal too, so wraiths and spirits take it aggravated as well.
		$DEATH_REASON$ = flag:death_balefire
		$DEATH_REASON$ = flag:death_fleshcrafting
		# Combat Mini-game addition: true faith burns the damned - vampires and
		# abominations, demons,
		# kuei-jin, wraiths, and the fae - as aggravated (holy damage class ->
		# REASON = true_faith). Changelings take it superficial; the mundane are
		# immune (POD_has_immunity_to_damage_type).
		AND = {
			$DEATH_REASON$ = flag:death_true_faith
			OR = {
				POD_is_vampire_or_abomination_trigger = yes
				POD_is_demon_trigger = yes
				is_kueijin_trigger = yes
				POD_is_wraith_trigger = yes
				POD_is_fae_trigger = yes
			}
		}
		# Combat Mini-game addition: the combat minigame's change_type bonus on
		# an elemental move. The attacker carries a transient marker so the
		# elemental death_reason (immunity checks intact) still classifies as
		# aggravated. Set/removed around one dispatch in
		# pod_combat_apply_single_hp_damage_effect.
		trigger_if = {
			limit = { exists = scope:wound_attacker }
			scope:wound_attacker = { has_variable = pod_combat_force_agg }
		}
		trigger_if = {
			limit = { exists = scope:wound_attacker }
			POD_applies_superficial_as_aggravated_damage = {
				ATTACKER = scope:wound_attacker
				VICTIM = this
			}
		}
		has_character_modifier = POD_quest_of_deed_modifier
		#$DEATH_REASON$ = flag:death_drained_of_quintessence # hmm. maybe too harsh
	}
}

POD_applies_superficial_as_aggravated_damage = {
	# Temp scope renamed victim -> pod_wound_victim.
	# CK3 refuses to create a temporary target that shadows an existing PERMANENT
	# one: it logs "trying to add the temporary target 'victim'..." and DISCARDS
	# the temp, so every read below silently resolved to whatever the parent
	# event chain had saved instead of the character being wounded.
	# 96 sites in the tree save a permanent scope:victim (incl. the duel-defender
	# alias in pod_combat_sc_context_effects.txt), so the collision is routine -
	# it just happens to pick the right character in a 1v1, where defender ==
	# victim. Do not rename back to a common noun.
	$VICTIM$ = { save_temporary_scope_as = pod_wound_victim }
	$ATTACKER$ = {
		OR = {
			has_character_modifier = demon_arts_infernal_glory_modifier
			exists = scope:apply_as_aggravated
			has_character_modifier = POD_talons_of_the_plague_crow_modifier
			AND = {
				has_character_modifier = archid_ancient_strength_modifier
				scope:pod_wound_victim = {
					OR = {
						POD_weak_to_silver_trigger = yes
						POD_weak_to_gold_trigger = yes
					}
				}
			}
			# The wielded-artifact list moved to
			# POD_weapon_artifact_forces_agg_trigger so combat can consult the
			# same set. In the minigame it only applies to a move that actually
			# swings the weapon (the marker is set around that dispatch), so a
			# sword stops aggravating spat venom and mind blasts; everywhere
			# else the list stays unconditional.
			AND = {
				trigger_if = {
					# scope:combat_host, not the pod_combat_duel_active modifier:
					# that modifier carries a 30-day safety duration, so a fight
					# abandoned without teardown would keep suppressing the list
					# for weeks of ordinary wounds. The saved scope exists only
					# inside a live combat damage dispatch, which is exactly the
					# window this rule is about.
					limit = { exists = scope:combat_host }
					has_variable = pod_combat_weapon_attack
				}
				POD_weapon_artifact_forces_agg_trigger = { VICTIM = scope:pod_wound_victim }
			}
		}
	}
}
