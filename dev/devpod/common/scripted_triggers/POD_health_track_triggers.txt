POD_has_wrong_wound_trait = {
	save_temporary_scope_value_as = {
		name  = trait_xp
		value = POD_health_track_percentage
	}
	trigger_if = {
		limit = { scope:trait_xp >= POD_wound_3_xp_threshold }
		NOT = { has_trait = wounded_3 }
	}
	trigger_else_if = {
		limit = { scope:trait_xp >= POD_wound_2_xp_threshold }
		NOT = { has_trait = wounded_2 }
	}
	trigger_else_if = {
		limit = { scope:trait_xp >= POD_wound_1_xp_threshold }
		NOT = { has_trait = wounded_1 }
	}
	trigger_else_if = {
		limit = { scope:trait_xp >= 1 }
		NOT = { has_trait = POD_wounded }
	}
	trigger_else = { has_trait = wounded }
}

# Victim scope. DEATH_REASON: death-reason flag.
POD_has_immunity_to_damage_type = {
	trigger_if = {
		limit = { POD_is_death_by_fire = { DEATH_REASON = $DEATH_REASON$ } }
		POD_immune_to_fire_trigger = yes
	}
	# Combat Mini-game: elemental immunities.
	trigger_else_if = {
		limit = { $DEATH_REASON$ = flag:death_drowned }
		POD_immune_to_drowning_trigger = yes
	}
	trigger_else_if = {
		limit = { $DEATH_REASON$ = flag:death_sun }
		OR = {
			NOR = { POD_is_vampire_or_abomination_trigger = yes  is_kueijin_trigger = yes }
			POD_immune_to_sun_damage_trigger = yes
		}
	}
	trigger_else_if = {
		limit = { $DEATH_REASON$ = flag:death_true_faith }
		NOR = {
			POD_is_vampire_or_abomination_trigger = yes
			POD_is_demon_trigger = yes
			is_kueijin_trigger = yes
			POD_is_wraith_trigger = yes
			POD_is_fae_trigger = yes
			POD_is_changeling_trigger = yes
		}
	}
	trigger_else_if = {
		limit = { $DEATH_REASON$ = flag:death_poison }
		OR = {
			POD_is_vampire_or_abomination_trigger = yes
			is_kueijin_trigger = yes
			POD_is_wraith_trigger = yes
			POD_is_spirit_trigger = yes
		}
	}
	trigger_else = {
		always = no
	}
}

# Victim scope. DEATH_REASON: death-reason flag.
# Explicit false without wound_attacker prevents a skipped trigger_if from satisfying the caller's OR.
POD_wound_attacker_forces_agg_trigger = {
	trigger_if = {
		limit = { exists = scope:wound_attacker }
		OR = {
			# pod_combat_apply_single_hp_damage_effect sets and removes this variable
			# around the wound effect. Keeping the elemental death reason preserves immunity checks.
			scope:wound_attacker = { has_variable = pod_combat_force_agg }
			POD_applies_superficial_as_aggravated_damage = {
				ATTACKER = scope:wound_attacker
				VICTIM = this
				DEATH_REASON = $DEATH_REASON$
			}
		}
	}
	trigger_else = { always = no }
}

# Victim scope. DEATH_REASON: death-reason flag.
POD_damage_type_is_aggravated = {
	OR = {
		AND = {
			POD_is_death_by_fire = { DEATH_REASON = $DEATH_REASON$ }
			POD_weak_to_fire_trigger = yes
		}
		AND = {
			$DEATH_REASON$ = flag:death_sun
			POD_weak_to_sun_trigger = yes
		}
		$DEATH_REASON$ = flag:death_lightning
		AND = { $DEATH_REASON$ = flag:death_cold       POD_is_mortal_trigger = yes }
		AND = { $DEATH_REASON$ = flag:death_poison  POD_weak_to_poison_trigger = yes }
		$DEATH_REASON$ = flag:death_self_mutilation
		$DEATH_REASON$ = flag:death_duel_aggravated
		$DEATH_REASON$ = flag:death_consequence_of_pact
		AND = {
			$DEATH_REASON$ = flag:death_silver
			POD_weak_to_silver_trigger = yes
		}
		AND = {
			$DEATH_REASON$ = flag:death_gold
			POD_weak_to_gold_trigger = yes
		}
		AND = {
			$DEATH_REASON$ = flag:death_cold_iron
			POD_weak_to_cold_iron_trigger = yes
		}
		$DEATH_REASON$ = flag:death_sea_creatures
		$DEATH_REASON$ = flag:death_balefire
		$DEATH_REASON$ = flag:death_fleshcrafting
		AND = {
			$DEATH_REASON$ = flag:death_true_faith
			POD_weak_to_holy_trigger = yes
		}
		POD_wound_attacker_forces_agg_trigger = { DEATH_REASON = $DEATH_REASON$ }
		# Outside combat, weapon damage classes can aggravate weapon wounds.
		# Failed immunity or weakness checks leave the original wound type unchanged.
		# Combat applies weapon damage classes in pod_combat_apply_single_hp_damage_effect.
		AND = {
			NOT = { exists = scope:combat_host }
			POD_is_death_by_weapon = { DEATH_REASON = $DEATH_REASON$ }
			trigger_if = {
				limit = { exists = scope:wound_attacker }
				# Reject a stale wound_attacker that refers to this victim.
				NOT = { this = scope:wound_attacker }
				OR = {
					AND = {
						scope:wound_attacker = { POD_wound_attacker_wields_class_trigger = { CLASS = fire } }
						NOT = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_fire } }
						POD_weak_to_fire_trigger = yes
					}
					AND = {
						scope:wound_attacker = { POD_wound_attacker_wields_class_trigger = { CLASS = cold } }
						NOT = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_cold } }
						POD_is_mortal_trigger = yes
					}
					AND = {
						scope:wound_attacker = { POD_wound_attacker_wields_class_trigger = { CLASS = lightning } }
						NOT = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_lightning } }
					}
					AND = {
						scope:wound_attacker = { POD_wound_attacker_wields_class_trigger = { CLASS = sun } }
						NOT = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_sun } }
						POD_weak_to_sun_trigger = yes
					}
					AND = {
						scope:wound_attacker = { POD_wound_attacker_wields_class_trigger = { CLASS = silver } }
						NOT = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_silver } }
						POD_weak_to_silver_trigger = yes
					}
					AND = {
						scope:wound_attacker = { POD_wound_attacker_wields_class_trigger = { CLASS = gold } }
						NOT = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_gold } }
						POD_weak_to_gold_trigger = yes
					}
					AND = {
						scope:wound_attacker = { POD_wound_attacker_wields_class_trigger = { CLASS = holy } }
						NOT = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_true_faith } }
						POD_weak_to_holy_trigger = yes
					}
					AND = {
						scope:wound_attacker = { POD_wound_attacker_wields_class_trigger = { CLASS = balefire } }
						NOT = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_balefire } }
					}
					AND = {
						scope:wound_attacker = { POD_wound_attacker_wields_class_trigger = { CLASS = poison } }
						NOT = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_poison } }
						POD_weak_to_poison_trigger = yes
					}
					AND = {
						scope:wound_attacker = { POD_wound_attacker_wields_class_trigger = { CLASS = cold_iron } }
						NOT = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_cold_iron } }
						POD_weak_to_cold_iron_trigger = yes
					}
				}
			}
			trigger_else = { always = no }
		}
		has_character_modifier = POD_quest_of_deed_modifier
	}
	POD_applies_aggravated_as_superficial_damage = no
}

POD_applies_superficial_as_aggravated_damage = {
	# ATTACKER and VICTIM: character scopes. DEATH_REASON: death-reason flag;
	# only the weapon-artifact condition uses it.
	# A temporary scope cannot shadow a scope saved with save_scope_as.
	# Uses pod_wound_victim to avoid collisions with callers' scope:victim.
	$VICTIM$ = { save_temporary_scope_as = pod_wound_victim }
	$ATTACKER$ = {
		OR = {
			has_character_modifier = demon_arts_infernal_glory_modifier
			exists = scope:apply_as_aggravated
			has_character_modifier = POD_talons_of_the_plague_crow_modifier
			AND = {
				has_character_modifier = archid_ancient_strength_modifier
				scope:pod_wound_victim = {
					OR = {
						POD_weak_to_silver_trigger = yes
						POD_weak_to_gold_trigger = yes
					}
				}
			}
			# Weapon artifacts require a weapon attack; the other conditions do not.
			AND = {
				trigger_if = {
					# Use combat_host for combat context. pod_combat_duel_active can remain
					# for its 30-day duration if cleanup does not run.
					limit = { exists = scope:combat_host }
					has_variable = pod_combat_weapon_attack
				}
				trigger_else = {
					POD_is_death_by_weapon = { DEATH_REASON = $DEATH_REASON$ }
					# Check for a stale attacker here too. This block runs in ATTACKER scope,
					# so compare with the victim.
					NOT = { this = scope:pod_wound_victim }
				}
				POD_weapon_artifact_forces_agg_trigger = { VICTIM = scope:pod_wound_victim }
			}
			AND = {
				has_perk = charm_life_swelling_sap_strike_perk
				scope:victim = { 
					OR = {
						has_trait = vampire 
						has_trait = kueijin 
					}
				}
			}
		}
	}
}

	}
}
