has_dharma_greater_than_trigger = {
	OR = {
		has_trait = dhampyr
		has_trait_rank = {
			trait = dharma
			rank >= $LEVEL$
		}
		has_perk = unfolding_scarlet_lotus_perk
		#has_global_variable = wan_xian_wan_kuei
		has_character_modifier = yama_king_modifier
		has_character_modifier = higher_shih_modifier
	}
}
has_dharma_lesser_than_trigger = {
	OR = {
		NOT = { has_trait = dhampyr }
		has_trait_rank = {
			trait = dharma
			rank <= $LEVEL$
		}
		NOT = { has_perk = unfolding_scarlet_lotus_perk }
		#NOT = { has_global_variable = wan_xian_wan_kuei }
		NOT = { has_character_modifier = yama_king_modifier }
		NOT = { has_character_modifier = higher_shih_modifier }
	}
}
