has_dharma_greater_than_trigger = {
	OR = {
		has_trait = dhampyr
		has_trait_rank = {
			trait = dharma
			rank >= $LEVEL$
		}
		has_perk = unfolding_scarlet_lotus_perk
		#has_global_variable = wan_xian_wan_kuei
	}
}
has_dharma_lesser_than_trigger = {
	OR = {
		NOT = { has_trait = dhampyr }
		has_trait_rank = {
			trait = dharma
			rank <= $LEVEL$
		}
		NOT = { has_perk = unfolding_scarlet_lotus_perk }
		#NOT = { has_global_variable = wan_xian_wan_kuei }
	}
}

	trigger_if = {
		limit = { $LEVEL$ <= 4 }
		has_dharma_greater_than_trigger = { LEVEL = $LEVEL$ }
	}
	trigger_else = {
		trigger_if = {
			limit = { dharma_level < shinta_mastery_dharma_requirement }
			has_dharma_greater_than_trigger = { LEVEL = shinta_mastery_dharma_requirement }
		}
	}
}
