has_koldunic_bloodmagic_trigger = {
    custom_tooltip = {
        text = has_koldunic_bloodmagic_trigger_tooltip
        has_trait = bloodsorcerydiscipline
        OR = {
            has_character_flag = has_koldunic_bloodmagic_flag
            AND = {
                is_tzimisce_trigger = yes
                NOT = { has_character_flag = remove_clan_bloodmagic }
            }
            AND = {
                is_chibo_trigger = yes
                NOT = { has_character_flag = remove_clan_bloodmagic }
            }
        }
    }
    
}

is_in_empowered_kraina_location = {
    has_perk = genius_loci_5_perk
    trigger_if = {
        limit = {
            any_character_artifact = {
				has_variable = incunabulum_kupalam 
				prev = { can_benefit_from_artifact = prev }
			}
        }
        always = yes
    }
    trigger_else = {
        switch = {
            trigger = flag:$KRAINA$

            flag:transylvania = {
                location ?= { geographical_region = PoD_region_transylvania_kraina }
            }

            flag:black_sea = {
                location ?= { geographical_region = PoD_region_black_sea_kraina }
            }

            flag:bialowieza = {
                location ?= { geographical_region = PoD_region_bialowieza_kraina }
            }

            flag:well = {
                location ?= {
                    OR = {
                        has_building_or_higher = land_nexus_01
                        has_building = sanctuary_of_flesh_01
                        has_building_or_higher = cathedral_of_flesh_01
                        has_building_or_higher = well_of_sacrifice_01
                    }
                }
            }

            flag:enoch = {
                location ?= {
                    is_sea_province = no
                    county = { development_level >= 40 }
                }
            }
        }
    }
}
