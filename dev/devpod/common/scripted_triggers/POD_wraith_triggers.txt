

can_convert_to_wraith_trigger = { # Even Forced
	NOT = { has_trait = wraith }
	NOT = { has_trait = spirit }
	NOT = { has_trait = automaton }
	NOT = { has_trait = mummy } # Technically they are wraith in resurrection
	NOT = { has_trait = demon } 
	NOT = { has_trait = fae } 
	is_adult = yes
	POD_is_clone_trigger = no
}

can_become_wraith_trigger = { # Naturally
	NOT = { has_trait = mage } # Mages by and large reincarnate
	NOT = { has_trait = fera } # Garou, like mages, reincarnate
	NOT = { has_trait = changeling } # Changelings are only susceptible to becoming one of the Restless Dead if killed with cold iron.
}

not_has_fetter_trigger = {
	OR = {
		NOT = { has_variable_list = owned_fetter_list }
		variable_list_size = {
			name = owned_fetter_list
			value = 0
		}
	}
}

is_my_fetter_trigger = {
	is_target_in_variable_list = { name = owned_fetter_list target = $FETTER$ }
}

is_fetter_of_actor_trigger = {
	has_variable = fetter_owner
	var:fetter_owner = scope:actor
}

is_fetter_of_trigger = {
	has_variable = fetter_owner
	var:fetter_owner = $OWNER$
}


POD_has_fetter_of_type = {
	has_variable_list = owned_fetter_list
	any_in_list = {
		variable = owned_fetter_list
		var:type = flag:$TYPE$
	}
}
POD_has_fetter_of_type_and_level = {
	has_variable_list = owned_fetter_list
	any_in_list = {
		variable = owned_fetter_list
		var:type = flag:$TYPE$
		var:level = $LEVEL$
	}
}

POD_is_fetter_at_max_level = {
	custom_tooltip = {
		text = POD_is_fetter_at_max_level
		has_variable_list = owned_fetter_list
		any_in_list = {
			variable = owned_fetter_list
			var:type = flag:$TYPE$
			var:level = 5
		}
	}
}
