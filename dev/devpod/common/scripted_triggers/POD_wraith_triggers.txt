can_be_killed_physically_trigger = {
	trigger_if = {
		limit = { NOT = { has_trait = risen } }
		NOT = { has_trait = wraith }
	}
	NOT = { has_trait = spirit }
	NOT = { has_trait = demon }
	is_in_umbra_trigger = no
}

hurt_wraith_trigger = {
	trigger_if = {
		limit = {
			scope:recipient = { 
				can_be_killed_physically_trigger = no
			}
		}
		scope:actor = { can_hurt_wraith_trigger = yes }
	}
}

can_hurt_wraith_trigger = {
	custom_tooltip = {
		text = can_hurt_wraith
		OR = {
			has_perk = torment_perk
			has_perk = spirit_scepter_perk
			has_perk = rend_perk
			has_perk = spirit_strike_perk
			has_perk = fomori_spirit_rending_perk
			
			has_trait_with_flag = can_hurt_spirit
			has_trait_with_flag = can_imprison_spirit
			
			any_equipped_character_artifact = { has_variable = can_hurt_spirit }
			
			is_in_umbra_trigger = yes
		}
	}
}

eat_wraith_custom_trigger = {
	trigger_if = {
		limit = {
			$RECIPIENT$ = { 
				can_be_killed_physically_trigger = no
			}
		}
		$ACTOR$ = { can_eat_wraith_trigger = yes }
	}
}

can_eat_wraith_trigger = {
	trigger_if = {
		limit = { has_POD_vamp_blood_trigger = yes }
		has_perk = soul_feasting_perk
	}
	trigger_else_if = {
		limit = { has_POD_kj_blood_trigger = yes }
		has_perk = spirit_eating_perk
	}
	trigger_else = {
		NOT = { has_trait = kinfolk }
		NOT = { has_trait = changeling }
		NOT = { has_trait = kinain }
		NOT = { has_trait = laham }
		NOT = { has_trait = nephilim }

		NOT = { has_trait = asekh_sen }
		NOT = { has_trait = automaton }
		NOT = { has_trait = mage }
		NOT = { has_trait = mortal }
		NOT = { has_trait = dhampir }
		NOT = { has_trait = alien }
	}
}

POD_physically_move_trigger = {
	OR = {
		AND = {
			this = scope:sc_defender
			OR = {
				can_hurt_wraith_trigger = yes
				scope:sc_attacker = { can_be_killed_physically_trigger = yes }
			}
		}
		AND = {
			this = scope:sc_attacker
			OR = {
				can_hurt_wraith_trigger = yes
				scope:sc_defender = { can_be_killed_physically_trigger = yes }
			}
		}
	}
}
