

can_convert_to_wraith_trigger = { # Even Forced
	NOT = { has_trait = wraith }
	NOT = { has_trait = spirit }
	NOT = { has_trait = automaton }
	NOT = { has_trait = mummy } # Technically they are wraith in resurrection
	NOT = { has_trait = demon } 
	NOT = { has_trait = fae } 
	is_adult = yes
	POD_is_clone_trigger = no
}

can_become_wraith_trigger = { # Naturally
	NOT = { has_trait = mage } # Mages by and large reincarnate
	NOT = { has_trait = shapeshifter } # Garou, like mages, reincarnate
	NOT = { has_trait = changeling } # Changelings are only susceptible to becoming one of the Restless Dead if killed with cold iron.
}
