not_has_fetter_trigger = {
	OR = {
		NOT = { has_variable_list = owned_fetter_list }
		variable_list_size = {
			name = owned_fetter_list
			value = 0
		}
	}
}

	is_target_in_variable_list = { name = owned_fetter_list target = $FETTER$ }

	has_variable = fetter_owner
	var:fetter_owner = scope:actor
}

	var:fetter_owner = $OWNER$
