is_grand_city_location_trigger = {
	OR = {
		barony = title:b_roma
		barony = title:b_constantinople
		barony = title:b_paris
		barony = title:b_cairo
		# insert future grand cities here
	}
}

is_below_grand_city_cap_trigger = {
	custom_description = {
		text = is_below_grand_city_cap_trigger
		trigger_if = {
			limit = { NOT = { has_game_rule = pod_grand_city_limit_disabled } }
			barony_controller = {
				NOT = {
					calc_true_if = {
						amount > 1
						has_title = title:c_byzantion
						has_title = title:c_roma
						has_title = title:c_ile_de_france
						has_title = title:c_cairo
					}
				}
			}
		}
		trigger_else = {
			always = yes
		}
	}
}

can_use_rome_trigger = {
	barony_controller = {
		NOT = {
			has_government = litany_government
		}
	}
}

is_roma_aeterna_trigger = {
	has_building_or_higher = rome_roma_aeterna_01
}

building_requirement_roma_aeterna = {
	barony = title:b_roma
	can_use_rome_trigger = yes
	barony_controller = {
		POD_is_vampire_trigger = yes
	}
}

is_st_peters_city_trigger = {
	has_building_or_higher = rome_st_peters_city_01
}

building_requirement_st_peters_city = {
	barony = title:b_roma
	can_use_rome_trigger = yes
	barony_controller = {
		faith = faith:shadowinquisition
	}

}

is_eternal_rome_trigger = {
	has_building_or_higher = rome_eternal_rome_01
}

building_requirement_eternal_rome = {
	barony = title:b_roma
	can_use_rome_trigger = yes
	barony_controller = {
		POD_is_vampire_trigger = no
		NOT = { faith = faith:shadowinquisition }
	}

}

is_san_michele_built_trigger = {
	title:c_roma = {
		has_county_modifier = san_michele_modifier
	}
}

rome_has_wrong_holder_trigger = {
	title:b_roma.title_province = {
		OR = {
			AND = {
				is_roma_aeterna_trigger = yes
				building_requirement_roma_aeterna = no
			}
			AND = {
				is_st_peters_city_trigger = yes
				building_requirement_st_peters_city = no
			}
			AND = {
				is_eternal_rome_trigger = yes
				building_requirement_eternal_rome = no
			}
			AND = {
				has_holding_type = tribal_holding
				can_use_rome_trigger = yes
			}
		}
	}
}

##########################
can_use_constantinople_trigger = {
	barony_controller = {
		NOT = {
			has_government = litany_government
		}
	}
}

is_plundered_constantinople_trigger = {
	has_building_or_higher = constantinople_plundered_constantinople_01
}

building_plundered_constantinople = {
	barony = title:b_constantinople
	can_use_constantinople_trigger = yes
	barony_controller = {
		OR = {
			has_character_flag = POD_gc_keep
			culture = {
				OR = {
					AND = {
						has_cultural_pillar = heritage_latin
						NOT = { this = culture:roman }
					}
				has_cultural_pillar = heritage_frankish
				has_cultural_pillar = heritage_iberian
				has_cultural_pillar = heritage_central_germanic
				has_cultural_pillar = heritage_west_germanic			
				}
			}
		}
	}
	NOT = { has_province_modifier = POD_constantinople_plundered_modifier }
}

is_rebuild_constantinople_trigger = {
	has_building_or_higher = constantinople_rebuild_constantinople_01
}

building_rebuild_constantinople = {
	barony = title:b_constantinople
	can_use_constantinople_trigger = yes
	barony_controller = {
		OR = {
			culture = { has_cultural_pillar = heritage_byzantine }
			has_culture = culture:roman
			has_culture = culture:russian
			culture = { has_cultural_pillar = heritage_ancient_greek }
			has_character_flag = POD_gc_keep
		}
	}
}

is_reform_constantinople_trigger = {
	has_building_or_higher = constantinople_reform_constantinople_01
}

	barony = title:b_constantinople
	can_use_constantinople_trigger = yes
	
	OR = {
    has_province_modifier = POD_constantinople_plundered_modifier 
	
	barony_controller = {
		OR = {
			NOT = {
			culture = {

			has_cultural_pillar = heritage_frankish
			has_cultural_pillar = heritage_iberian
