is_grand_city_location_trigger = {
	OR = {
		barony = title:b_roma
		barony = title:b_constantinople
		# insert future grand cities here
	}
}

is_below_grand_city_cap_trigger = {
	custom_tooltip = {
		text = is_below_grand_city_cap_trigger
		barony_controller = {
			NOT = {
				calc_true_if = {
					amount > 1
					has_title = title:c_byzantion
					has_title = title:c_roma
				}
			}
		}
	}
}

can_use_rome_trigger = {
	barony_controller = {
		NOT = {
			has_government = litany_government
		}
	}
}

can_use_constantinople_trigger = {
	barony_controller = {
		NOT = {
			has_government = litany_government
		}
	}
}

is_roma_aeterna_trigger = {
	has_building_or_higher = rome_roma_aeterna_01
}

building_requirement_roma_aeterna = {
	building_requirement_roma_aeterna_ignore_cap = yes
	is_below_grand_city_cap_trigger = yes
}

building_requirement_roma_aeterna_ignore_cap = {
	barony = title:b_roma
	can_use_rome_trigger = yes
	barony_controller = {
		POD_is_vampire_trigger = yes
	}
}

is_st_peters_city_trigger = {
	has_building_or_higher = rome_st_peters_city_01
}

building_requirement_st_peters_city = {
	building_requirement_st_peters_city_ignore_cap = yes
	is_below_grand_city_cap_trigger = yes
}

building_requirement_st_peters_city_ignore_cap = {
	barony = title:b_roma
	can_use_rome_trigger = yes
	barony_controller = {
		faith = faith:shadowinquisition
	}

}

is_eternal_rome_trigger = {
	has_building_or_higher = rome_eternal_rome_01
}

building_requirement_eternal_rome = {
	building_requirement_eternal_rome_ignore_cap = yes
	is_below_grand_city_cap_trigger = yes
}

building_requirement_eternal_rome_ignore_cap = {
	barony = title:b_roma
	can_use_rome_trigger = yes
	barony_controller = {
		POD_is_vampire_trigger = no
		NOT = { faith = faith:shadowinquisition }
	}

}

is_san_michele_built_trigger = {
	title:c_roma = {
		has_county_modifier = san_michele_modifier
	}
}

rome_has_wrong_holder_trigger = {
	title:b_roma.title_province = {
		OR = {
			AND = {
				is_roma_aeterna_trigger = yes
				building_requirement_roma_aeterna_ignore_cap = no
			}
			AND = {
				is_st_peters_city_trigger = yes
				building_requirement_st_peters_city_ignore_cap = no
			}
			AND = {
				is_eternal_rome_trigger = yes
				building_requirement_eternal_rome_ignore_cap = no
			}
			AND = {
				has_holding_type = tribal_holding
				can_use_rome_trigger = yes
			}
		}
	}
}

is_plundered_constantinople_trigger = {
	has_building_or_higher = constantinople_plundered_constantinople_01
}

building_plundered_constantinople = {
	building_plundered_constantinople_ignore_cap = yes
	is_below_grand_city_cap_trigger = yes
}

building_plundered_constantinople_ignore_cap = {
	barony = title:b_constantinople
	can_use_constantinople_trigger = yes
	barony_controller = {
		culture = {
			has_cultural_pillar = heritage_latin
		}
	}
	NOT = { has_province_modifier = POD_constantinople_plundered_modifier }
}

is_rebuild_constantinople_trigger = {
	has_building_or_higher = constantinople_rebuild_constantinople_01
}

building_rebuild_constantinople = {
	building_rebuild_constantinople_ignore_cap = yes
	is_below_grand_city_cap_trigger = yes
}

building_rebuild_constantinople_ignore_cap = {
	barony = title:b_constantinople
	can_use_constantinople_trigger = yes
	barony_controller = {
		culture = { has_cultural_pillar = heritage_byzantine }
	}
}

is_reform_constantinople_trigger = {
	has_building_or_higher = constantinople_reform_constantinople_01
}

building_reform_constantinople = {
	building_reform_constantinople_ignore_cap = yes
	is_below_grand_city_cap_trigger = yes
}

building_reform_constantinople_ignore_cap = {
	barony = title:b_constantinople
	can_use_constantinople_trigger = yes
	barony_controller = {
		OR = {
			NOT = { culture = { has_cultural_pillar = heritage_byzantine } }
			OR = {
				prev = { has_province_modifier = POD_constantinople_plundered_modifier }
				NOT = { culture = { has_cultural_pillar = heritage_latin } }
			}
		}
	}
}

constantinople_has_wrong_holder_trigger = {
	title:b_constantinople.title_province = {
		OR = {
			AND = {
				is_plundered_constantinople_trigger = yes
				building_plundered_constantinople_ignore_cap = no
			}
			AND = {
				is_rebuild_constantinople_trigger = yes
				building_rebuild_constantinople_ignore_cap = no
			}
			AND = {
				is_reform_constantinople_trigger = yes
				building_reform_constantinople_ignore_cap = no
			}
			AND = {
				has_holding_type = tribal_holding
				can_use_constantinople_trigger = yes
			}
		}
	}
}
