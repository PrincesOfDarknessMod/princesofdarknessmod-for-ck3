	}

# based on valid_laamp_basic_trigger, but more general (valid for all governments and splats)
POD_valid_contract_basic_trigger = {
	custom_description = {
		text = adventurer_not_famous_or_patron_trigger_tt
		object = $EMPLOYER$
		value = $EMPLOYER$.highest_held_title_tier
		OR = {
			$EMPLOYER$.highest_held_title_tier <= tier_county
			$EMPLOYER$.highest_held_title_tier <= $LAAMP$.prestige_level
			$EMPLOYER$ = { is_contact_of = $LAAMP$ }
		}
	}
	$EMPLOYER$ = {
		# Players shouldn't get tangled up in contracts, ever.
		is_ai = yes
		# Plus some standard stuff.
		is_alive = yes
		is_adult = yes
	}
	$EMPLOYER$ = { is_incapable = no }
	# Laamps shouldn't offer contracts in their local area, ever.
	$EMPLOYER$ = {
		OR = {
			employer = $LAAMP$
			NOT = { government_has_flag = government_is_landless_adventurer }
		}
	}
	# Don't offer to work with those who've wronged us.
	## No one we're at war with.
	$EMPLOYER$ = {
		NOT = { is_at_war_with = $LAAMP$ }
	}
	$LAAMP$ = {
		trigger_if = {
			limit = { is_at_war = no }
			is_commanding_army = no
		}
	}
}

POD_can_manipulate_disaster = {
	$CONTRACT$ ?= { save_temporary_scope_as = disaster_contract }
	trigger_if = {
		limit = {
			scope:disaster_contract.var:disaster_type ?= flag:pod_weather_flood
		}
		OR = {
			POD_can_manipulate_floods = yes
			any_relation = {
				type = coterie_member
				POD_can_manipulate_floods = yes
			}
		}
	}
	trigger_else_if = {
		limit = {
			scope:disaster_contract.var:disaster_type ?= flag:pod_weather_earthquake
		}
		OR = {
			POD_can_manipulate_earthquakes = yes
			any_relation = {
				type = coterie_member
				POD_can_manipulate_earthquakes = yes
			}
		}
	}
	trigger_else_if = {
		limit = {
			scope:disaster_contract.var:disaster_type ?= flag:pod_weather_storm
		}
		OR = {
			POD_can_manipulate_storms = yes
			any_relation = {
				type = coterie_member
				POD_can_manipulate_storms = yes
			}
		}
	}
	trigger_else = {
		always = no
	}
}
