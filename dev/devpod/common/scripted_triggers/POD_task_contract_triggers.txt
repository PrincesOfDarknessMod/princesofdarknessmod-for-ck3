POD_can_manipulate_disaster = {
	$CONTRACT$ ?= { save_temporary_scope_as = disaster_contract }
	trigger_if = {
		limit = {
			scope:disaster_contract.var:disaster_type ?= flag:POD_weather_magic_flood
		}
		OR = {
			POD_can_manipulate_floods = yes
			any_relation = {
				type = coterie_member
				POD_can_manipulate_floods = yes
			}
		}
	}
	trigger_else_if = {
		limit = {
			scope:disaster_contract.var:disaster_type ?= flag:POD_weather_magic_earthquake
		}
		OR = {
			POD_can_manipulate_earthquakes = yes
			any_relation = {
				type = coterie_member
				POD_can_manipulate_earthquakes = yes
			}
		}
	}
	trigger_else_if = {
		limit = {
			scope:disaster_contract.var:disaster_type ?= flag:POD_weather_magic_storm
		}
		OR = {
			POD_can_manipulate_storms = yes
			any_relation = {
				type = coterie_member
				POD_can_manipulate_storms = yes
			}
		}
	}
	trigger_else = {
		always = no
	}
}
