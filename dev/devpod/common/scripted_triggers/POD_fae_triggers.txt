realm_scene = {
	scope:actor = { exists = location }
	scope:recipient = { exists = location }
	trigger_if = { 
		limit = { scope:actor = scope:recipient }
	}
	trigger_else_if = { 
		limit = { scope:actor.location = { squared_distance = { target = scope:recipient.location value > squared_distance_huge } } }
		scope:actor = { has_perk = realm_scene_5_perk }
	}
	trigger_else_if = { 
		limit = { scope:actor.location = { squared_distance = { target = scope:recipient.location value > squared_distance_large } } }
		scope:actor = { has_perk = realm_scene_4_perk }
	}
	trigger_else_if = { 
		limit = { scope:actor.location = { squared_distance = { target = scope:recipient.location value > squared_distance_medium } } }
		scope:actor = { has_perk = realm_scene_3_perk }
	}
	trigger_else_if = { 
		limit = { scope:actor.location = { squared_distance = { target = scope:recipient.location value > squared_distance_small } } }
		scope:actor = { has_perk = realm_scene_2_perk }
	}
	trigger_else_if = { 
		limit = { NOT = { scope:actor.location = scope:recipient.location } }
		scope:actor = { has_perk = realm_scene_1_perk }
	}
	trigger_else = {
	}
}

realm_actor_custom = {
	trigger_if = { 
		limit = {  $ACTOR$ = $RECIPIENT$ }
	}
	trigger_else_if = { 
		limit = { 
			$RECIPIENT$ = { 
				OR = {
					has_relation_rival = $ACTOR$
					is_at_war_with = $ACTOR$
				}
			}
		}
		$ACTOR$ = { has_perk = realm_actor_5_perk }
	}
	trigger_else_if = { 
		limit = { 
			$RECIPIENT$ = { 
				OR = {
					has_any_bad_relationship_with_character_trigger = { CHARACTER = $ACTOR$ } 
					existing_feud_against_target_trigger = { TARGET = $ACTOR$ }
				}
			}
		}
		$ACTOR$ = { has_perk = realm_actor_4_perk }
	}
	trigger_else_if = { 
		limit = { 
			$RECIPIENT$ = { 
				OR = {
					is_imprisoned_by = $ACTOR$
					has_truce = $ACTOR$
					NOR = { 
						has_any_good_relationship_with_character_trigger = { CHARACTER = $ACTOR$ }
						is_courtier_of = $ACTOR$
						is_foreign_court_or_pool_guest_of = $ACTOR$
						is_vassal_or_below_of = $ACTOR$
						is_liege_or_above_of = $ACTOR$
						is_allied_to = $ACTOR$
						dynasty ?= $ACTOR$.dynasty
						any_secret = {
							is_known_by = $ACTOR$
						}
						any_held_title = {
							$ACTOR$ = { has_claim_on = prev }
						}
						any_character_artifact = {
							$ACTOR$ = { has_personal_artifact_claim = prev }
						}
						$ACTOR$ = { has_hook = prev }
					}
				}
			}
		}
		$ACTOR$ = { has_perk = realm_actor_3_perk }
	}
	trigger_else_if = { 
		limit = { 
			$RECIPIENT$ = { 
				NOR = { 
					has_any_moderate_good_relationship_with_character_trigger = { CHARACTER = $ACTOR$ }
					is_courtier_of = $ACTOR$
					is_foreign_court_or_pool_guest_of = $ACTOR$
					is_vassal_or_below_of = $ACTOR$
					is_liege_or_above_of = $ACTOR$
					dynasty ?= $ACTOR$.dynasty
					any_secret = {
						is_known_by = $ACTOR$
					}
					any_held_title = {
						$ACTOR$ = { has_claim_on = prev }
					}
					$ACTOR$ = { has_hook = prev }
				}
			}
		}
		$ACTOR$ = { has_perk = realm_actor_2_perk }
	}
	trigger_else_if = { 
		limit = { 
			$RECIPIENT$ = { 
				NOR = {
					has_any_best_good_relationship_with_character_trigger = { CHARACTER = $ACTOR$ }
					is_close_family_of = $ACTOR$
					is_knight_of = $ACTOR$
					AND = {
						is_courtier_of = $ACTOR$
						has_employed_any_court_position = yes
					}
				}
			}
		}
		$ACTOR$ = { has_perk = realm_actor_1_perk }
	}
	trigger_else = {
	}
}

realm_prop = {
	trigger_if = { 
		limit = { 
			NOT = { exists = scope:target }
		}
	}
	trigger_else_if = { 
		limit = { 
			scope:target = { 
				rarity = illustrious 
				is_unique = yes
			}
		}
		scope:actor = { has_perk = realm_prop_5_perk }
	}
	trigger_else_if = { 
		limit = { 
			scope:target = { 
				rarity = famed 
				is_unique = yes
			}
		}
		scope:actor = { has_perk = realm_prop_4_perk }
	}
	trigger_else_if = { 
		limit = { 
			scope:target = { 
				OR = {
					rarity = illustrious 
					rarity = famed 
				}
			}
		}
		scope:actor = { has_perk = realm_prop_3_perk }
	}
	trigger_else_if = { 
		limit = { 
			scope:target = { 
				OR = {
					artifact_slot_type = primary_armament
					artifact_slot_type = armor
					artifact_slot_type = regalia
					artifact_slot_type = helmet
				}
			}
		}
		scope:actor = { has_perk = realm_prop_2_perk }
	}
	trigger_else_if = { 
		limit = { 
			scope:target = { 
				NOR = {
					has_variable = ai_to_ai_destroy_in_sieges 
					has_variable = unwanted_artifact 
					has_variable = cursed_artifact
				}
			}
		}
		scope:actor = { has_perk = realm_prop_1_perk }
	}
	trigger_else = {
	}
}
