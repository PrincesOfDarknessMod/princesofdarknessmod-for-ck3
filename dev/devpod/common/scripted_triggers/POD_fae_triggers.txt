realm_scene = {
	scope:actor = { exists = location }
	scope:recipient = { exists = location }
	trigger_if = { 
		limit = { 
			OR = {
				scope:actor = scope:recipient
				scope:actor = { has_perk = realm_scene_5_perk }
			}
		}
	}
	trigger_else_if = { 
		limit = { scope:actor = { has_perk = realm_scene_4_perk } }
		custom_description = {
			text = "I_must_be_within_a_great_distance"
			scope:actor.location = { squared_distance = { target = scope:recipient.location value <= squared_distance_huge } }
		}
	}
	trigger_else_if = { 
		limit = { scope:actor = { has_perk = realm_scene_3_perk } }
		custom_description = {
			text = "I_must_be_within_a_long_distance"
			scope:actor.location = { squared_distance = { target = scope:recipient.location value <= squared_distance_large } }
		}
	}
	trigger_else_if = { 
		limit = { scope:actor = { has_perk = realm_scene_2_perk } }
		custom_description = {
			text = "I_must_be_within_a_medium_distance"
			scope:actor.location = { squared_distance = { target = scope:recipient.location value <= squared_distance_medium } }
		}
	}
	trigger_else_if = { 
		limit = { scope:actor = { has_perk = realm_scene_1_perk } }
		custom_description = {
			text = "I_must_be_within_a_short_distance"
			scope:actor.location = { squared_distance = { target = scope:recipient.location value <= squared_distance_small } }
		}
	}
	trigger_else = { 
		custom_description = {
			text = "same_location"
			scope:actor.location = scope:recipient.location
		}
	}
}

		}
	}

		limit = { $ACTOR$ = { has_perk = realm_actor_4_perk } }

			NOT = { has_relation_rival = $ACTOR$ }
			NOT = { is_at_war_with = $ACTOR$ }
		}

		limit = { $ACTOR$ = { has_perk = realm_actor_3_perk } }

					text = "has_any_bad_relationship"

				OR = { 
					has_any_moderate_good_relationship_with_character_trigger = { CHARACTER = $ACTOR$ }

					is_foreign_court_or_pool_guest_of = $ACTOR$

realm_prop = {
	trigger_if = { 
		limit = { 
			OR = {
				NOT = { exists = scope:target }
				scope:actor = { has_perk = realm_prop_5_perk } 
			}
		}
	}
	trigger_else_if = { 
		limit = { scope:actor = { has_perk = realm_prop_4_perk } }
		scope:target = { 
			trigger_if = { 
				limit = { rarity = illustrious }
				NOT = { 
					custom_description = {
						text = POD_unique_artifact 
						is_unique = yes
					}
				}
			}
		}
	}
	trigger_else_if = { 
		limit = { scope:actor = { has_perk = realm_prop_3_perk } }
		scope:target = { 
			trigger_if = { 
				limit = { 
					OR = {
						rarity = illustrious 
						rarity = famed 
					}
				}
				NOT = { 
					custom_description = {
						text = POD_unique_artifact 
						is_unique = yes
					}
				}
			}
		}
	}
	trigger_else_if = { 
		limit = { scope:actor = { has_perk = realm_prop_2_perk } }
		scope:target = { 
			NOT = { rarity = illustrious }
			NOT = { rarity = famed }
		}
	}
	trigger_else_if = { 
		limit = { scope:actor = { has_perk = realm_prop_1_perk } }
		scope:target = { 
			NOT = { rarity = illustrious }
			NOT = { rarity = famed }
			NOT = { artifact_slot_type = primary_armament }
			NOT = { artifact_slot_type = armor }
			NOT = { artifact_slot_type = regalia }
			NOT = { artifact_slot_type = helmet }
		}
	}
	trigger_else = { 
		scope:target = { 
			custom_description = {
				text = POD_trash_artifact
				OR = { 
					has_variable = ai_to_ai_destroy_in_sieges 
					has_variable = unwanted_artifact 
					has_variable = cursed_artifact
				}
			}
		}
	}
}
