realm_scene = {
	scope:actor = { exists = location }
	scope:recipient = { exists = location }
	trigger_if = { 
		limit = { scope:actor = { has_perk = realm_scene_5_perk } }
	}
	trigger_else_if = { 
		limit = { scope:actor = { has_perk = realm_scene_4_perk } }
		custom_description = {
			text = "I_must_be_within_a_great_distance"
			scope:actor.location = { squared_distance = { target = scope:recipient.location value <= squared_distance_huge } }
		}
	}
	trigger_else_if = { 
		limit = { scope:actor = { has_perk = realm_scene_3_perk } }
		custom_description = {
			text = "I_must_be_within_a_long_distance"
			scope:actor.location = { squared_distance = { target = scope:recipient.location value <= squared_distance_large } }
		}
	}
	trigger_else_if = { 
		limit = { scope:actor = { has_perk = realm_scene_2_perk } }
		custom_description = {
			text = "I_must_be_within_a_medium_distance"
			scope:actor.location = { squared_distance = { target = scope:recipient.location value <= squared_distance_medium } }
		}
	}
	trigger_else_if = { 
		limit = { scope:actor = { has_perk = realm_scene_1_perk } }
		custom_description = {
			text = "I_must_be_within_a_short_distance"
			scope:actor.location = { squared_distance = { target = scope:recipient.location value <= squared_distance_small } }
		}
	}
	trigger_else = { 
		custom_description = {
			text = "I_must_be_at_the_same_location"
			scope:actor.location = scope:recipient.location
		}
	}
}

realm_actor = {
	trigger_if = { 
		limit = { scope:actor = { has_perk = realm_actor_5_perk } }
	}
	trigger_else_if = { 
		limit = { scope:actor = { has_perk = realm_actor_4_perk } }
		scope:recipient = { 
			NOT = { has_relation_rival = scope:actor }
			NOT = { is_at_war_with = scope:actor }
		}
	}
	trigger_else_if = { 
		limit = { scope:actor = { has_perk = realm_actor_3_perk } }
		scope:recipient = { 
			NOT = { 
				custom_description = {
					text = "has_any_bad_relationship"
					has_any_bad_relationship_with_character_trigger = { CHARACTER = scope:actor } 
				}
			}
			NOT = { is_at_war_with = scope:actor }
			custom_description = {
				text = "realm_actor_3_valid"
				OR = { 
					has_any_good_relationship_with_character_trigger = { CHARACTER = scope:actor }
					is_courtier_of = scope:actor
					is_foreign_court_or_pool_guest_of = scope:actor
					is_vassal_or_below_of = scope:actor
					is_liege_or_above_of = scope:actor
					is_allied_to = scope:actor
					AND = {
						exists = dynasty
						dynasty = scope:actor.dynasty
					}
					any_secret = {
						is_known_by = scope:actor
					}
					any_held_title = {
						scope:actor = { has_claim_on = prev }
					}
					any_character_artifact = {
						scope:actor = { has_personal_artifact_claim = prev }
					}
					scope:actor = { has_hook = prev }
				}
			}
		}
	}
	trigger_else_if = { 
		limit = { scope:actor = { has_perk = realm_actor_2_perk } }
		scope:recipient = { 
			NOT = { 
				custom_description = {
					text = "has_any_bad_relationship"
					has_any_bad_relationship_with_character_trigger = { CHARACTER = scope:actor } 
				}
			}
			NOT = { is_imprisoned_by = scope:actor }
			NOT = { is_at_war_with = scope:actor }
			NOT = { has_truce = scope:actor }
			custom_description = {
				text = "realm_actor_2_valid"
				OR = { 
					has_any_moderate_good_relationship_with_character_trigger = { CHARACTER = scope:actor }
					is_courtier_of = scope:actor
					is_foreign_court_or_pool_guest_of = scope:actor
					is_vassal_or_below_of = scope:actor
					is_liege_or_above_of = scope:actor
					AND = {
						exists = dynasty
						dynasty = scope:actor.dynasty
					}
					any_secret = {
						is_known_by = scope:actor
					}
					any_held_title = {
						scope:actor = { has_claim_on = prev }
					}
					scope:actor = { has_hook = prev }
				}
			}
		}
	}
	trigger_else_if = { 
		limit = { scope:actor = { has_perk = realm_actor_1_perk } }
		scope:recipient = { 
			NOT = { 
				custom_description = {
					text = "has_any_bad_relationship"
					has_any_bad_relationship_with_character_trigger = { CHARACTER = scope:actor } 
				}
			}
			NOT = { is_imprisoned_by = scope:actor }
			NOT = { is_at_war_with = scope:actor }
			NOT = { has_truce = scope:actor }
			custom_description = {
				text = "realm_actor_1_valid"
				OR = { 
					has_any_moderate_good_relationship_with_character_trigger = { CHARACTER = scope:actor }
					is_courtier_of = scope:actor
					is_vassal_of = scope:actor
					AND = {
						exists = house
						house = scope:actor.house
					}
					any_secret = {
						is_criminal_for = scope:recipient
						is_known_by = scope:actor
					}
					any_held_title = {
						scope:actor = { has_strong_claim_on = prev }
					}
					scope:actor = { has_strong_hook = prev }
				}
			}
		}
	}
	trigger_else = { 
		scope:recipient = { 
			NOT = { 
				custom_description = {
					text = "has_any_bad_relationship"
					has_any_bad_relationship_with_character_trigger = { CHARACTER = scope:actor } 
				}
			}
			NOT = { is_imprisoned_by = scope:actor }
			NOT = { is_at_war_with = scope:actor }
			NOT = { has_truce = scope:actor }
			custom_description = {
				text = "realm_actor_0_valid"
				OR = {
					has_any_best_good_relationship_with_character_trigger = { CHARACTER = scope:actor }
					is_close_family_of = scope:actor
					is_knight_of = scope:actor
					AND = {
						is_courtier_of = scope:actor
						has_employed_any_court_position = yes
					}
				}
			}
		}
	}
}
