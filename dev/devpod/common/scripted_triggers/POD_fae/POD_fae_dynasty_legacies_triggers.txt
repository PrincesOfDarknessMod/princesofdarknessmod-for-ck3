POD_dynasty_is_a_fae_noble_house_trigger = {
	OR = {
		dynasty ?= dynasty:POD_fae_dynasty_aesin
		dynasty ?= dynasty:POD_fae_dynasty_ailil
		dynasty ?= dynasty:POD_fae_dynasty_balor
		dynasty ?= dynasty:POD_fae_dynasty_beaumayn
		dynasty ?= dynasty:POD_fae_dynasty_tuatha_de_danaan
		dynasty ?= dynasty:POD_fae_dynasty_danaan
		dynasty ?= dynasty:POD_fae_dynasty_daireann
		dynasty ?= dynasty:POD_fae_dynasty_dougal
		dynasty ?= dynasty:POD_fae_dynasty_eiluned
		dynasty ?= dynasty:POD_fae_dynasty_fiona
		dynasty ?= dynasty:POD_fae_dynasty_gwydion
		dynasty ?= dynasty:POD_fae_dynasty_leanhaun
		dynasty ?= dynasty:POD_fae_dynasty_liam
		dynasty ?= dynasty:POD_fae_dynasty_scathach
		dynasty ?= dynasty:POD_fae_dynasty_varich
		dynasty ?= { has_variable = POD_fae_noble_house }
	}
}
