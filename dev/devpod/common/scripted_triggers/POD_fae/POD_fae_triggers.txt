POD_aquis_interaction_heal_wound_trigger = {
	scope:recipient = { 
		trigger_if = { limit = { scope:heal = yes } 
			has_trait = $TRAIT$
			POD_is_trait_not_permanent_trigger = { TRAIT = $TRAIT$ } 
		}
		trigger_else = { 
			NOT = { has_trait = $TRAIT$ }
			any_character_trait = { NOT = { is_opposite_of_trait = $TRAIT$ } }
		}
		POD_aquis_interaction_option_trigger = { OPTION = $TRAIT$ }
	}
}

		}

	}
}
