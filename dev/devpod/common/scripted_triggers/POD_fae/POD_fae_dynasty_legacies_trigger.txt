POD_can_obtain_fae_dynasty_perk_trigger = {
	trigger_if = {
		limit = { is_ai = no }
		switch = {
			trigger = flag:$DYNASTY_LEGACY$
			flag:aesin = { POD_can_obtain_aesin_dynasty_perk_trigger = { PERK_LEVEL = $PERK_LEVEL$ } }
			flag:ailil = { POD_can_obtain_ailil_dynasty_perk_trigger = { PERK_LEVEL = $PERK_LEVEL$ } }
			flag:balor = { POD_can_obtain_balor_dynasty_perk_trigger = { PERK_LEVEL = $PERK_LEVEL$ } }
			flag:beaumayn = { POD_can_obtain_beaumayn_dynasty_perk_trigger = { PERK_LEVEL = $PERK_LEVEL$ } }
			flag:danaan = { POD_can_obtain_danaan_dynasty_perk_trigger = { PERK_LEVEL = $PERK_LEVEL$ } }
		}
	}
}

POD_can_obtain_aesin_dynasty_perk_trigger = {
	save_temporary_scope_value_as = {
		name = completed_hunts_required
		value = {
			value = 2
			multiply = $PERK_LEVEL$
		}
	}
	POD_has_completed_hunts_trigger = { VALUE = scope:completed_hunts_required }
	POD_has_succeeded_hunts_trigger = { VALUE = $PERK_LEVEL$ }
}
POD_can_obtain_ailil_dynasty_perk_trigger = {
	save_temporary_scope_value_as = {
		name = completed_battles_required
		value = {
			value = 3
			multiply = $PERK_LEVEL$
		}
	}
	POD_has_won_battles_as_commander_trigger = { VALUE = scope:completed_battles_required }
	POD_has_successful_wars_trigger = { VALUE = $PERK_LEVEL$ }
}
POD_can_obtain_balor_dynasty_perk_trigger = {
	save_temporary_scope_value_as = {
		name = completed_battles_required
		value = {
			value = 2
			multiply = $PERK_LEVEL$
		}
	}
	trigger_if = {
		limit = { is_landed = no }
		POD_has_completed_criminal_contracts_trigger = { VALUE = scope:completed_battles_required }
	}
	POD_has_completed_hostile_scheme_trigger = { VALUE = $PERK_LEVEL$ }
}
POD_can_obtain_beaumayn_dynasty_perk_trigger = {
	piety_level >= $PERK_LEVEL$
	POD_num_of_visited_holy_sited_trigger = { VALUE = $PERK_LEVEL$ }
}
POD_can_obtain_danaan_dynasty_perk_trigger = {
	save_temporary_scope_value_as = {
		name = percentage_needed
		value = {
			value = 0.2
			multiply = $PERK_LEVEL$
		}
	}
	POD_percentage_of_dreaming_explored_trigger = { PERCENTAGE = scope:percentage_needed }
	custom_description = {
		text = POD_number_of_strong_non_sidhe_kithain_vassals
		VALUE = $PERK_LEVEL$
		any_vassal = {
			count >= $PERK_LEVEL$
			is_powerful_vassal = yes
			has_trait = kithain
			NOT = { has_trait = sidhe }
		}
	}
}
