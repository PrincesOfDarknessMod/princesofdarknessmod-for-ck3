POD_can_obtain_danaan_dynasty_perk_trigger = {
	save_temporary_scope_value_as = {
		name = percentage_needed
		value = {
			value = 0.2
			multiply = $PERK_LEVEL$
		}
	}
	POD_percentage_of_dreaming_explored_trigger = { PERCENTAGE = scope:percentage_needed }
	custom_description = {
		text = POD_number_of_strong_non_sidhe_kithain_vassals
		VALUE = $PERK_LEVEL$
		any_vassal = {
			count >= $PERK_LEVEL$
			is_powerful_vassal = yes
			has_trait = kithain
			NOT = { has_trait = sidhe }
		}
	}
}

POD_can_obtain_fiona_dynasty_perk_trigger = {
	save_temporary_scope_value_as = {
		name = completed_battles_required
		value = {
			value = 2
			multiply = $PERK_LEVEL$
		}
	}
	trigger_if = {
		limit = { $PERK_LEVEL$ >= 5 }
		custom_tooltip = {
			text = POD_can_obtain_fiona_dynasty_perk_soulmate
			num_of_relation_soulmate >= 1
		}
	}
	save_temporary_scope_value_as = {
		name = required_lovers
		value = $PERK_LEVEL$
	}
	custom_tooltip = {
		text = POD_can_obtain_fiona_dynasty_perk_loves
		num_of_relation_lover >= scope:required_lovers
	}
	POD_has_won_battles_as_commander_trigger = { VALUE = scope:completed_battles_required }
}
