	}
	save_temporary_scope_value_as = {
		name = required_ruler_friends
		value = {
			value = $PERK_LEVEL$
		}
	}
	custom_tooltip = {
		text = POD_can_obtain_gwydion_dynasty_perk_vassal
		any_vassal = { count >= scope:required_vassals }
	}

		name = independent_ruler_friends

				type = friend

				add = 1
			}
			every_relation = {
				type = best_friend
				limit = { is_independent_ruler = yes }
				add = 1
			}
		}
	}
	custom_tooltip = {
		text = POD_can_obtain_gwydion_dynasty_perk
		scope:independent_ruler_friends >= scope:required_ruler_friends
	}
}
POD_can_obtain_leanhaun_dynasty_perk_trigger = {
	save_temporary_scope_value_as = {
		name = required_friends
		value = $PERK_LEVEL$
	}
	save_temporary_scope_value_as = {
		name = required_great_person_relation
		value = {
			value = 0 
			if = { limit = { $PERK_LEVEL$ = 4 } value = 1 }
			if = { limit = { $PERK_LEVEL$ = 5 } value = 2 }
		}
	}
	custom_tooltip = {
		text = POD_can_obtain_leanhaun_dynasty_perk_friends
		num_of_relation_friend >= scope:required_friends
	}
	save_temporary_scope_value_as = {
		name = great_relations
		value = {
			every_relation = {
				type = friend
				limit = { has_character_flag = POD_great_person_flag }
				add = 1
			}
			every_relation = {
				type = best_friend
				limit = { has_character_flag = POD_great_person_flag }
				add = 1
			}
			every_relation = {
				type = lover
				limit = { has_character_flag = POD_great_person_flag }
				add = 1
			}
			every_relation = {
				type = soulmate
				limit = { has_character_flag = POD_great_person_flag }
				add = 1
			}
		}
	}
	custom_tooltip = {
		text = POD_can_obtain_fiona_dynasty_perk_great_person_relation
		scope:great_relations >= scope:required_great_person_relation
	}
}
