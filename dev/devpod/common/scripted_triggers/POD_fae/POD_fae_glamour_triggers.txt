POD_has_any_glamour_modifier = {
	OR = {
		has_character_modifier = glamour0_modifier
		has_character_modifier = glamour1_modifier
		has_character_modifier = glamour2_modifier
		has_character_modifier = glamour3_modifier
		has_character_modifier = glamour4_modifier
		has_character_modifier = glamour5_modifier
	}
}
POD_can_use_epiphanies = {
	POD_splat_has_glamour_exclusively = yes
}
POD_can_use_reverie = {
	POD_splat_has_glamour_exclusively = yes
	NOT = { has_character_modifier = POD_useless_dreamers }
	NOT = { has_trait_with_flag = cannot_use_reverie }
}
POD_can_use_revelry = {
	POD_splat_has_glamour_exclusively = yes
	NOT = { has_trait_with_flag = cannot_use_revelry }
}
POD_can_use_ravaging = {
	POD_splat_has_glamour_exclusively = yes
	NOT = { has_trait_with_flag = cannot_use_ravaging }
}

POD_improved_ravaging_trigger = {
	dynasty ?= {
		OR = {
			has_dynasty_perk = balor_legacy_5
			has_dynasty_perk = leanhaun_legacy_5
			has_dynasty_perk = varich_legacy_5
		}
	}
}
POD_improved_reverie_trigger = {
	dynasty ?= {
		OR = {
			has_dynasty_perk = beaumayn_legacy_4
			has_dynasty_perk = danaan_legacy_4
			has_dynasty_perk = dougal_legacy_4
			has_dynasty_perk = liam_legacy_5
		}
	}
}
