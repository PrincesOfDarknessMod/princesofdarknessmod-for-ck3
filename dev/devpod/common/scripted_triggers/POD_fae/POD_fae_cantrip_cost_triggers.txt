POD_has_enough_glamour_for_cantrip_interaction = {
	save_temporary_scope_as = cantrip_actor
	save_temporary_scope_value_as = {
		name = nature_level
		value = $NATURE_LEVEL$
	}
	$RECIPIENT$ = { save_temporary_scope_as = cantrip_recipient }
	trigger_if = {
		limit = { exists = $TARGET$ }
		$TARGET$ = { save_temporary_scope_as = cantrip_target }
	}
	trigger_if = {
		limit = { exists = $PROVINCE$ }
		$PROVINCE$ = { save_temporary_scope_as = cantrip_province }
	}
	save_temporary_scope_value_as = {
		name = cantrip_cost
		value = {
			value = $AMOUNT$
			if = {
				limit = { always = $REALM_ACTOR$ }
				add = cantrip_interaction_additional_realm_actor_cost
			}
			if = {
				limit = { always = $REALM_FAE$ }
				add = cantrip_interaction_additional_realm_fae_cost
			}
			if = {
				limit = { always = $REALM_NATURE$ }
				add = cantrip_interaction_additional_realm_nature_cost
			}
			if = {
				limit = { always = $REALM_PROP$ }
				add = cantrip_interaction_additional_realm_prop_cost
			}
			if = {
				limit = { always = $REALM_SCENE$ }
				add = cantrip_interaction_additional_realm_scene_cost
			}
			if = {
				limit = { always = $REALM_SCENE_AREA$ }
				add = cantrip_interaction_additional_realm_scene_area_cost
			}
			if = {
				limit = { always = $REALM_SCENE_PROVINCE$ }
				add = cantrip_interaction_additional_realm_scene_province_cost
			}
		}
	}
	trigger_if = {
		limit = { scope:cantrip_cost > $AMOUNT$ }
		
		POD_set_temporary_scope_value_from_trigger = {
			NAME = hide_tooltip
			TRIGGER = "POD_has_enough_glamour_for_fae_art = { AMOUNT = scope:cantrip_cost }"
		}
		
		trigger_if = {
			limit = { always = $REALM_ACTOR$ }
			POD_additional_realm_actor_cost_trigger_tt = {
				HIDE_TOOLTIP = scope:hide_tooltip
			}
		}
		trigger_if = {
			limit = { always = $REALM_FAE$ }
			POD_additional_realm_fae_cost_trigger_tt = {
				HIDE_TOOLTIP = scope:hide_tooltip
			}
		}
		trigger_if = {
			limit = { always = $REALM_NATURE$ }
			POD_additional_realm_nature_cost_trigger_tt = {
				HIDE_TOOLTIP = scope:hide_tooltip
			}
		}
		trigger_if = {
			limit = { always = $REALM_PROP$ }
			POD_additional_realm_prop_cost_trigger_tt = {
				HIDE_TOOLTIP = scope:hide_tooltip
			}
		}
		trigger_if = {
			limit = { always = $REALM_PROP$ }
			POD_additional_realm_prop_cost_trigger_tt = {
				HIDE_TOOLTIP = scope:hide_tooltip
			}
		}
		trigger_if = {
			limit = { always = $REALM_SCENE$ }
			POD_additional_realm_scene_cost_trigger_tt = {
				HIDE_TOOLTIP = scope:hide_tooltip
			}
		}
		trigger_if = {
			limit = { always = $REALM_SCENE_AREA$ }
			POD_additional_realm_scene_area_cost_trigger_tt = {
				HIDE_TOOLTIP = scope:hide_tooltip
			}
		}
		trigger_if = {
			limit = { always = $REALM_SCENE_PROVINCE$ }
			POD_additional_realm_scene_province_cost_trigger_tt = {
				HIDE_TOOLTIP = scope:hide_tooltip
			}
		}
	}
	POD_has_enough_glamour_for_fae_art = { AMOUNT = scope:cantrip_cost }
}

# for Primal
POD_has_enough_glamour_for_primal_eldritch_prime_interaction_custom = {
	save_temporary_scope_value_as = {
		name = cantrip_cost
		value = {
			switch = {
				trigger = yes
				scope:county = { value = 1 }
				scope:duchy = { value = 2 }
				scope:kingdom = { value = 3 }
				scope:empire = { value = 4 }
				scope:continent = { value = 5 }
			}
		}
	}
	save_temporary_scope_value_as = {
		name = cantrip_nature_level
		value = {
			switch = {
				trigger = yes
				scope:abundant_flora = { value = 2 }
				scope:dead_flora = { value = 2 }
				scope:agitated_animals = { value = 3 }
				scope:calm_animals = { value = 3 }
				scope:flow = { value = 4 }
				scope:earthquake = { value = 4 }
				scope:meteor_shower = { value = 5 }
				scope:storm = { value = 5 }
				scope:drought = { value = 5 }
				scope:flood = { value = 4 }
			}
		}
	}
	switch = {
		trigger = scope:$OPTION$
		scope:county = {
			scope:recipient.location.county = { save_temporary_scope_as = cantrip_target }
		}
		scope:duchy = {
			scope:recipient.location.duchy = { save_temporary_scope_as = cantrip_target }
		}
		scope:kingdom = {
			scope:recipient.location.kingdom = { save_temporary_scope_as = cantrip_target }
		}
		scope:empire = {
			scope:recipient.location.empire = { save_temporary_scope_as = cantrip_target }
		}
		scope:continent	= {
			scope:recipient.location = { save_temporary_scope_as = cantrip_target }
			POD_flag_cantrip_area_as_continent_for_trigger = yes
		}
	}
	save_temporary_scope_value_as = {
		name = $VAR$
		value = $VAL$
	}
	trigger_if = {
		limit = { scope:$OPTION$ = no }
		POD_has_enough_glamour_for_cantrip_interaction = {
			AMOUNT		= scope:cantrip_cost
			RECIPIENT	= scope:recipient # needs to exist but gets ignored
			TARGET		= scope:target
			PROVINCE	= scope:ignore_this_scope
			NATURE_LEVEL	= scope:cantrip_nature_level
			REALM_ACTOR		= no
			REALM_FAE		= no
			REALM_NATURE	= yes
			REALM_PROP		= no
			REALM_SCENE		= yes
			REALM_SCENE_AREA 	 = yes
			REALM_SCENE_PROVINCE = no
		}
	}
}

POD_additional_realm_scene_area_cost_trigger_tt = {
	trigger_if = {
		limit = { cantrip_interaction_additional_realm_scene_area_cost > 0 }
		
		trigger_if = {
			limit = { exists = scope:cantrip_target_continent }
			
			custom_description = {
				text = POD_additional_realm_scene_continent_cost_trigger
				subject = scope:cantrip_actor
				always = $HIDE_TOOLTIP$
			}
		}
		trigger_else_if = {
			limit = { exists = scope:cantrip_target }
			
			custom_description = {
				text = POD_additional_realm_scene_area_cost_trigger
				subject = scope:cantrip_actor
				object = scope:cantrip_target
				always = $HIDE_TOOLTIP$
			}
		}
		trigger_else = {}
	}
}
