POD_strong_mental_defenses_trigger = {
	POD_strong_mental_defenses_custom_trigger = {
		ACTOR = scope:actor
		RECIPIENT = scope:recipient
	}
}

POD_strong_mental_defenses_custom_trigger = {
	trigger_if = {
		limit = {
			$RECIPIENT$ = { POD_has_strong_mental_defenses_trigger = yes }
		}
		
		$ACTOR$ = {
			custom_description = {
				object = $RECIPIENT$
				text = "POD_has_strong_mental_defenses_trigger"
				can_bypass_mental_defenses_trigger = yes
			}
		}
	}
	trigger_else_if = {
		limit = { $RECIPIENT$ = { POD_has_conditional_strong_mental_defenses_trigger = yes } }
		trigger_if = {
			limit = {
				$RECIPIENT$ = { has_perk = aegis_alighieri_perk }
				$ACTOR$ = { POD_is_demonic_trigger = yes }
			}
			$ACTOR$ = {
				custom_description = {
					object = $RECIPIENT$
					text = "POD_has_strong_mental_defenses_trigger"
					can_bypass_mental_defenses_trigger = yes
				}
			}
		}
	}
	trigger_else = { always = yes }
}

# to be used with POD_select_clan_disciplines_for_character
POD_save_clan_disciplines = {
	trait:$TRAIT1$ = { save_temporary_scope_as = discipline_1 }
	trait:$TRAIT2$ = { save_temporary_scope_as = discipline_2 }
	trait:$TRAIT3$ = { save_temporary_scope_as = discipline_3 }
}
POD_save_advanced_clan_disciplines = {
	trait:$TRAIT1$advanced = { save_temporary_scope_as = discipline_1 }
	trait:$TRAIT2$advanced = { save_temporary_scope_as = discipline_2 }
	trait:$TRAIT3$advanced = { save_temporary_scope_as = discipline_3 }
}

POD_is_clan_discipline = {
	POD_select_clan_disciplines_for_character = {
		TRIGGER = POD_save_clan_disciplines
	}
	OR = {
		scope:discipline_1 ?= $TRAIT$
		scope:discipline_2 ?= $TRAIT$
		scope:discipline_3 ?= $TRAIT$
		# trait isn't part of the original spread of 3 disciplines
		AND = {
			$TRAIT$ = trait:visceratika
			is_gargoyle_trigger = yes
		}
	}
}

# character scope
POD_save_current_resonance_perk_points_for_trait = {
	trigger_if = {
		limit = { $TRAIT$ = { has_trait_flag = animal_discipline } }
		save_temporary_scope_value_as = {
			name = $SAVE_AS$
			value = animal_lifestyle_perk_points
		}
	}
	trigger_else_if = {
		limit = { $TRAIT$ = { has_trait_flag = choleric_discipline } }
		save_temporary_scope_value_as = {
			name = $SAVE_AS$
			value = choleric_lifestyle_perk_points
		}
	}
	trigger_else_if = {
		limit = { $TRAIT$ = { has_trait_flag = melancholy_discipline } }
		save_temporary_scope_value_as = {
			name = $SAVE_AS$
			value = melancholy_lifestyle_perk_points
		}
	}
	trigger_else_if = {
		limit = { $TRAIT$ = { has_trait_flag = phlegmatic_discipline } }
		save_temporary_scope_value_as = {
			name = $SAVE_AS$
			value = phlegmatic_lifestyle_perk_points
		}
	}
	trigger_else_if = {
		limit = { $TRAIT$ = { has_trait_flag = sanguine_discipline } }
		save_temporary_scope_value_as = {
			name = $SAVE_AS$
			value = sanguine_lifestyle_perk_points
		}
	}
	# trigger_else_if chain always needs to end in a trigger_else
	trigger_else = { always = yes }
}

POD_can_afford_discipline_perk_trigger = {
	POD_save_discipline_perk_cost_values = {
		LEVEL = $LEVEL$
		TRAIT = trait:$DISCIPLINE$

	POD_save_current_resonance_perk_points_for_trait = {
		TRAIT = trait:$DISCIPLINE$
		SAVE_AS = current_perk_points
	}
	scope:current_perk_points >= POD_discipline_perk_cost
}
