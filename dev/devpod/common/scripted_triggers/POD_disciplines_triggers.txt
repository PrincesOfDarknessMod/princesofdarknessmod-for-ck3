strong_mental_defenses_trigger = {
	OR = {
		has_trait_with_flag = strong_mental_defenses

		has_perk = holy_bulwark_perk
		has_perk = steel_soul_perk
		has_perk = shield_of_the_soul_perk
		has_perk = wisdom_of_isis_perk
		has_perk = mountain_meditation_perk
		has_perk = mind_shield_perk
		has_perk = shell_perk
		
		has_perk = inviolate_soul_perk
		has_perk = release_from_bondage_perk
		any_equipped_character_artifact = {
			has_variable = ancestor_talisman
		}		
	}
}

can_bypass_mental_defenses_trigger = {
	has_trait_with_flag = bypass_mental_defenses
}

strong_mental_defenses_interaction_trigger = {
	OR = {
		scope:actor = { can_bypass_mental_defenses_trigger = yes }
		scope:recipient = { strong_mental_defenses_trigger = no }
	}
}

strong_mental_defenses_interaction_secondary_trigger = {
	OR = {
		scope:secondary_actor = { can_bypass_mental_defenses_trigger = yes }
		scope:recipient = { strong_mental_defenses_trigger = no }
	}
}

strong_mental_defenses_art_interaction_trigger = {
	trigger_if = {
		limit = { scope:recipient = { NOT = { has_trait = enchanted } } } # Do ai ever get it?
		strong_mental_defenses_interaction_trigger = yes
	}
}
