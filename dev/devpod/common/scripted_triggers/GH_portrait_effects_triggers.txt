GH_shader_effect_macro = {
	OR = {
		POD_shapeshifted_$EFFECT$_form_trigger = yes
		AND = {
			POD_natural_$EFFECT$_form_trigger = yes
			POD_has_shapeshifted_shader_form_trigger = no
		}
	}
}
