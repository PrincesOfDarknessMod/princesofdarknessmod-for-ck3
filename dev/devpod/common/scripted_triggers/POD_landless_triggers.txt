# need to pass CAMP_OWNER
POD_can_be_invited_to_camp = {
	OR = {
		has_relation_coterie_member = $CAMP_OWNER$
		POD_has_subservient_relationship_to = { TARGET = $CAMP_OWNER$ }
		$CAMP_OWNER$ = { POD_has_powerful_control_via_hook_on = { TARGET = prev } }
		var:former_camp_leader ?= $CAMP_OWNER$
	}
}
