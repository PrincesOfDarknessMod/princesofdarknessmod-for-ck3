has_giovanni_necromancy_trigger = {
	OR = {
		is_giovanni_trigger = yes
		has_character_flag = has_giovanni_necromancy_flag
	}
}

has_cappadocian_necromancy_trigger = {
	OR = {
		AND = {
			is_cappadocian_trigger = yes
			is_giovanni_trigger = no
			is_lamia_trigger = no
		}
		is_harbingersofskulls_trigger = yes
		has_character_flag = has_cappadocian_necromancy_flag
	}
}

has_harbinger_necromancy_trigger = {
	OR = {
		is_harbingersofskulls_trigger = yes
		has_character_flag = has_harbinger_necromancy_flag
	}
}

has_lamia_necromancy_trigger = {
	OR = {
		is_lamia_trigger = yes
		has_character_flag = has_lamia_necromancy_flag
	}
}

has_impundulu_necromancy_trigger = {
	OR = {
		is_impundulu_trigger = yes
		has_character_flag = has_impundulu_necromancy_flag
	}
}

has_nagaraja_necromancy_trigger = {
	OR = {
		is_nagaraja_trigger = yes
		has_character_flag = has_nagaraja_necromancy_flag
	}
}

has_any_necromancy_path = {
	OR = {
		has_giovanni_necromancy_trigger = yes
		has_cappadocian_necromancy_trigger = yes
		has_harbinger_necromancy_trigger = yes
		has_lamia_necromancy_trigger = yes
		has_impundulu_necromancy_trigger = yes
		has_nagaraja_necromancy_trigger = yes
	}
}
