
##### Character scope

POD_is_on_heist_trigger = {
	has_variable = current_heist
}

##### Story scope

POD_heist_is_node_connected_trigger = {
	$TARGET$ = { save_temporary_scope_as = target_node }
	any_in_list = {
		variable = lines
		OR = {
			var:start_node = scope:target_node
			var:end_node   = scope:target_node
		}
	}
}

POD_heist_is_node_exit_trigger = {
	var:type ?= flag:node_exit
}

POD_heist_uses_suspicion_trigger = {
	has_variable = suspicion
}
