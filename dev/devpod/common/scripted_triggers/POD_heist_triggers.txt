	POD_incompatible_splats_trigger = no # WIP: no hostile infiltrations yet
	scope:recipient = { # WIP: suspicion events only for landed characters
		is_landed = yes
		highest_held_title_tier >= tier_county
	}
}

##### Character scope

POD_is_on_heist_trigger = {

}

##### Story scope

POD_heist_is_node_connected_trigger = {
	$TARGET$ = { save_temporary_scope_as = target_node }
	any_in_list = {
		variable = lines
		OR = {
			var:start_node = scope:target_node
			var:end_node   = scope:target_node
		}
	}
}
