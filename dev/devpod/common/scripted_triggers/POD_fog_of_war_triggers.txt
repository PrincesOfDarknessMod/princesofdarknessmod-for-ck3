POD_is_default_fog_of_war_enabled = {
	has_game_rule = pod_fog_of_war_enabled
}

POD_is_custom_fog_of_war_enabled = {
	has_game_rule = pod_fog_of_war_custom
}

POD_show_random_outcomes_as_percentage = {
	trigger_if = {
		limit = { POD_is_custom_fog_of_war_enabled = yes }
		has_game_rule = pod_random_outcomes_percentages
	}
	trigger_else = { POD_is_default_fog_of_war_enabled = no }
}
