POD_has_only_one_full_time_court_position = {
	custom_tooltip = {
		text = none_does_not_have_a_full_time_job
		trigger_if = {
			limit = { NOT = { has_court_position = $EXCEPTION$ } }
			court_position_does_not_already_have_a_job_trigger = yes # This positions is considered a job, and cannot be held at the same time as another similar position
		}
	}
}


##### Primogen

can_employ_brujah_primogen_trigger = {
	always = yes
}

can_employ_cappadocian_primogen_trigger = {
	#The trait is used post purge only, useful to make sure that clan Giovanni doesn't use Cappa primogens
	#NOT = {
	#	has_trait = giovanni
	#}
	always = yes
}

can_employ_lasombra_primogen_trigger = {
	always = yes
}

can_employ_toreador_primogen_trigger = {
	always = yes
}

can_employ_tzimisce_primogen_trigger = {
	#No Tzimisce primogen for the Tremere
	is_tremere_trigger = no # Banpaia don't care
}

can_employ_ventrue_primogen_trigger = {
	always = yes
}

can_employ_assamite_primogen_trigger = {
	#Assamite primogens only for Assamites, Ashirra or if you are centered in the ME
	OR = {
		is_assamite_trigger = yes
		has_religion = religion:ashirra_religion
		AND = {
			is_playable_character = yes
			capital_province = {
				geographical_region = world_middle_east
			}
		}
	}
}

can_employ_gangrel_primogen_trigger = {
	always = yes
}

can_employ_malkavian_primogen_trigger = {
	always = yes
}

can_employ_nosferatu_primogen_trigger = {
	always = yes
}

can_employ_ravnos_primogen_trigger = {
	OR = {
		is_ravnos_clan_or_bloodline_trigger = yes
		AND = {
			is_playable_character = yes
			capital_province = {
				geographical_region = world_india
			}
		}
	}
}

can_employ_setite_primogen_trigger = {
	#Everybody hates the Setites except themselves	
	OR = {
		is_setite_trigger = yes
		faith = { 
			OR = {
				has_doctrine = special_doctrine_set_pillars_kindred
				has_doctrine = special_doctrine_set_corrupted_kindred
			}
		}
	}
}

can_employ_tremere_primogen_trigger = {
	#No Tzimisce or Salubri ruler would employ a Tremere primo	
	is_tzimisce_trigger = no
	is_salubri_trigger = no	
	#Almost no tremere in ashirra lands, and disliked there as well
	is_assamite_trigger = no
	NOR = {
		has_religion = religion:ashirra_religion
		AND = {
			is_playable_character = yes
			capital_province = {
				geographical_region = world_middle_east
			}
		}
	}
}

can_employ_salubri_primogen_trigger = {
	#Salubri only for now
	is_salubri_trigger = yes
}

can_employ_gargoyle_primogen_trigger = {
	OR = {
		AND = {
			is_gargoyle_trigger = yes
			NOT = { faith = faith:viaservilis }
		}
		this = character:790870
		is_banpaia_trigger = yes
	}
	liege = {
		OR = {
			NOT = { has_religion = religion:codeoftremere_religion }
			this = character:790870
		}
	}
	top_liege = {
		OR = {
			NOT = { has_religion = religion:codeoftremere_religion }
			this = character:790870
		}
	}		
}
