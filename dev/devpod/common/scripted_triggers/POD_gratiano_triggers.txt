######################################
# ARC STATE
######################################

POD_gratiano_arc_active_trigger = {
	any_owned_story = { story_type = POD_gratiano_story }
	NOT = { has_global_variable = variable_POD_lasombra_dead }
}

# Warning only; duel odds use POD_gratiano_assault_edge_value.
POD_gratiano_strike_doomed_trigger = {
	NOT = {
		AND = {
			has_character_flag = POD_gratiano_giovanni_cleared
			has_character_flag = POD_gratiano_moncada_dealt
			has_character_flag = POD_gratiano_nahir_dealt
			has_character_flag = POD_gratiano_ventrue_cleared
			has_character_flag = POD_gratiano_assamite_pact
		}
	}
}

# LEVER & COTERIE VALIDITY

# Born to Rule removes bad_gratiano_modifier, unlocking the Courtier levers.
POD_gratiano_lever_actor_trigger = {
	this = character:505144
	is_ai = no
	has_character_flag = POD_gratiano_started
	NOT = { has_character_modifier = bad_gratiano_modifier }
	NOT = { has_global_variable = variable_POD_lasombra_dead }
}

# Held until the chosen rival chain completes; map chains release when their sphere clears.
POD_gratiano_rival_focus_free_trigger = {
	NOT = { has_character_flag = POD_gratiano_rival_in_focus }
}

# Gratiano scope; $CHAR$ is the prospective pool member.
POD_gratiano_pool_member_trigger = {
	is_target_in_variable_list = {
		name = POD_gratiano_recruit_pool
		target = $CHAR$
	}
}

# Gratiano scope; $TARGET$ is the prospective coterie member.
POD_gratiano_can_induct_trigger = {
	POD_coterie_at_max_size_trigger = no
	$TARGET$ = {
		is_alive = yes
		POD_is_valid_coterie_member_trigger = yes
		is_landed = no		# Landed conspirators remain pool allies.
		NOT = { has_relation_coterie_member = root }
	}
}

# Reserve these characters for their dedicated arcs. Scope: prospective recruit.
POD_gratiano_recruit_excluded_trigger = {
	OR = {
		this = character:505154	# Moncada
		this = character:790178	# Lucita
		this = character:505143	# Nahir
		this = character:505140	# Sybil
		this = character:505070	# Montano
	}
}

######################################
# REGION & COMPLETION
######################################

# County scope. Includes Venice and Ravenna; kingdom membership survives the Hecata metaplot.
POD_gratiano_italian_sphere_trigger = {
	OR = {
		target_is_de_jure_liege_or_above = title:k_italy
		target_is_de_jure_liege_or_above = title:k_venice
		target_is_de_jure_liege_or_above = title:k_romagna
	}
}

###### Objective completion ######

# Vassalization counts as clearing a rival; restrict realm checks to independent rulers.

# County scope. Include the Adriatic shore so fleeing Italy cannot clear the Giovanni objective.
POD_gratiano_giovanni_sphere_trigger = {
	OR = {
		POD_gratiano_italian_sphere_trigger = yes
		target_is_de_jure_liege_or_above = title:k_croatia
	}
}

POD_gratiano_giovanni_cleared_trigger = {
	NOT = {
		any_independent_ruler = {
			is_giovanni_trigger = yes
			any_realm_county = { POD_gratiano_giovanni_sphere_trigger = yes }
		}
	}
}

POD_gratiano_ventrue_cleared_trigger = {
	NOT = {
		any_independent_ruler = {
			is_ventrue_trigger = yes
			any_realm_county = { POD_gratiano_italian_sphere_trigger = yes }
		}
	}
}

POD_gratiano_map_arcs_active_trigger = {
	is_ai = no
	has_character_flag = POD_gratiano_started
	NOT = { has_global_variable = variable_POD_lasombra_dead }
	NOT = {
		is_target_in_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:POD_gratiano_strike_the_castle
		}
	}
}

# Use this for completed Plan I objectives: phase only advances, while cleared flags can be revoked.
POD_gratiano_plan_i_sealed_trigger = {
	any_owned_story = {
		story_type = POD_gratiano_story
		var:phase >= 3
	}
}

# Frame targets: Montano, Alfonzo, Guilelmo and Boukephos. Scope: prospective target.

	OR = {
		this = character:505070
		this = character:505098
		this = character:505099
		this = character:505149
	}
}

# Gratiano scope; $ELDER$ is the target. Purged characters also count as neutralized.
POD_gratiano_elder_neutralized_trigger = {
	trigger_if = {
		limit = { exists = $ELDER$ }
		OR = {
			is_target_in_variable_list = {
				name = POD_gratiano_framed_elders
				target = $ELDER$
			}
			POD_vampire_is_defeated = { CHARACTER = $ELDER$ }
		}
	}
	trigger_else = { always = yes }
}

# Gratiano scope, which owns the framed-elder list.
POD_gratiano_elders_framed_live_trigger = {
	calc_true_if = {
		amount >= 3	# [TEAM-TUNABLE] N of the 4-elder slate
		POD_gratiano_elder_neutralized_trigger = { ELDER = character:505070 }
		POD_gratiano_elder_neutralized_trigger = { ELDER = character:505098 }
		POD_gratiano_elder_neutralized_trigger = { ELDER = character:505099 }
		POD_gratiano_elder_neutralized_trigger = { ELDER = character:505149 }
	}
}

# Accept the live elder count before its cached flag updates. Identity and one-shot checks stay on the decision.
POD_gratiano_strike_gate_trigger = {
	has_character_flag = POD_gratiano_giovanni_cleared
	has_character_flag = POD_gratiano_moncada_dealt
	has_character_flag = POD_gratiano_nahir_dealt
	has_character_flag = POD_gratiano_ventrue_cleared
	has_character_flag = POD_gratiano_assamite_pact
	custom_tooltip = {
		text = POD_gratiano_gate_elders_tt
		OR = {
			has_character_flag = POD_gratiano_elders_framed
			POD_gratiano_elders_framed_live_trigger = yes
		}
	}
	custom_tooltip = {
		text = POD_gratiano_gate_conspiracy_tt
		has_variable = POD_gratiano_conspiracy
		var:POD_gratiano_conspiracy >= 80
	}
	completely_controls = title:c_siracusa
	NOT = { has_global_variable = variable_POD_lasombra_dead }
}
