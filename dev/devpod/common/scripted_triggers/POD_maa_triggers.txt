### Generic (character scope)

POD_can_recruit_basic_vanilla_maa = {
	OR = {
		has_trait = mortal
		has_trait = wraith
		has_trait = supehunter
		has_trait = spirit
		has_government = mercenary_government 
	}
}

POD_can_recruit_basic_vampire_maa = {
	POD_can_participate_in_vampire_politics = yes
}

POD_can_recruit_basic_eastern_maa = {
	OR = {
		has_trait = kueijin
		has_trait = wutian
		has_trait = wukuei
		has_trait = shih
	}
}

POD_can_recruit_basic_fera_maa = {
	OR = {
		has_trait = fera
		has_trait = kinfolk
	}
}

POD_can_recruit_basic_mummy_maa = {
	has_trait = mummy
	NOT = { has_trait = wutian }
	NOT = { has_trait = wukuei }
}

POD_can_recruit_special_mummy_maa = {
	POD_can_recruit_basic_mummy_maa = yes
}

POD_can_recruit_special_wu_tian_maa = {
	has_trait = mummy
	OR = {
		has_trait = wutian
		has_trait = wukuei
	}
}

POD_can_recruit_basic_demon_maa = {
	has_trait = demon
}

POD_can_recruit_basic_fae_maa = {
	OR = {
		has_trait = fae
		has_trait = changeling
	}
}

### Helper function (character scope)

POD_valid_for_maa_trigger = {
	OR = {
		AND = {
			is_landless_adventurer = yes
			has_perk = planned_cultivation_perk
			location.county.culture ?= { $TRIGGER$ = yes }
		}
		culture = { $TRIGGER$ = yes }
	}
}


### Cultural (culture scope)

POD_culture_unlocks_enochian_guards_maa = {
	has_cultural_pillar = heritage_enochian
}

POD_culture_unlocks_maryannu_maa = {
	OR = {
		has_cultural_pillar = heritage_ancientegyptian
		has_cultural_pillar = heritage_coptic
	}
}

POD_culture_unlocks_romani_scouts_maa = {
	AND = {
		has_cultural_pillar = heritage_indo_aryan
		has_cultural_pillar = language_domari
	}
}

POD_culture_unlocks_hoplites_maa = {
	has_cultural_pillar = heritage_ancient_greek
}

POD_culture_unlocks_companion_cavalry_maa = {
	this = culture:macedonian
