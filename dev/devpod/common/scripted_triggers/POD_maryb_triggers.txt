# True when Mary's "Tanith's Protection" arrangement is no longer in force.
# three invalidation conditions, any one of which is sufficient:
#
#   1. Tanith herself no longer exists (her character record is gone).
#   2. Tanith is dead.
#   3. Tanith is alive but no longer likes Mary (opinion < 0).
#
# Used by POD_maryb.50 (the yearly maintenance dispatcher) to decide
# whether to fire A Lingering Debt (POD_maryb.10). Lives as a named
# trigger so any future event or decision can ask "is the protection
# still good?" without duplicating the OR block.
#
POD_maryb_tanith_protection_invalidated_trigger = {
	OR = {
		NOT = { exists = character:505095 }
		character:505095 = { is_alive = no }
		AND = {
			character:505095 = { is_alive = yes }
			opinion = { target = character:505095 value < 0 }
		}
	}
}

# True once Mary has reached one of the three endings of the Deal with Michael.
# The tablet-gathering / balance "becoming" phase is over: the journey panel's
# progress counters are retired and the resolution effects own the panel from
# here, so the generic story sync must stop mirroring (and re-creating) them.
POD_maryb_resolved_trigger = {
	OR = {
		has_character_flag = mary_resolved_black_dream
		has_character_flag = mary_resolved_second_dream
		has_character_flag = mary_resolved_ash_and_gold
	}
}
