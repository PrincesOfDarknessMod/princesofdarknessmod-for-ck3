######################################################
# Asakku State Checks
######################################################

POD_asakku_has_asakku_trigger = {
	OR = {
		has_trait = POD_asakku_asakku
		has_trait = POD_asakku_iryi_ascendant
	}
}

POD_asakku_has_any_asakku_trait_trigger = {
	OR = {
		has_trait = POD_asakku_soul_infection
		has_trait = POD_asakku_asakku
		has_trait = POD_asakku_iryi_ascendant
	}
}

# for complex triggers (is_valid_showing_failures_only)
POD_asakku_has_no_asakku_trait_trigger = {
	NOT = { has_trait = POD_asakku_soul_infection }
	NOT = { has_trait = POD_asakku_asakku }
	NOT = { has_trait = POD_asakku_iryi_ascendant }
}

POD_asakku_is_andeleon_trigger = {
	this = character:790612
}

######################################################
# Depth Threshold Triggers
######################################################

POD_has_asakku_depth = {
	save_temporary_scope_value_as = {
		name  = assaku_depth_requirement
		value = $DEPTH$
	}
	custom_tooltip = {
		text = POD_asakku_depth_trigger
		POD_asakku_depth >= scope:assaku_depth_requirement
	}
}
