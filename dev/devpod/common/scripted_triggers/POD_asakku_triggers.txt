POD_asakku_has_soul_infection_trigger = {
	has_trait = POD_asakku_soul_infection
}

POD_asakku_has_asakku_trigger = {
	OR = {
		has_trait = POD_asakku_asakku
		has_trait = POD_asakku_iryi_ascendant
	}
}

POD_asakku_has_iryi_ascendant_trigger = {
	has_trait = POD_asakku_iryi_ascendant
}

POD_asakku_has_elioud_trigger = {
	has_trait = POD_asakku_elioud
}

POD_asakku_has_any_asakku_trait_trigger = {
	OR = {
		has_trait = POD_asakku_soul_infection
		has_trait = POD_asakku_asakku
		has_trait = POD_asakku_iryi_ascendant
	}
}

# Target Validity Triggers
######################################################

POD_asakku_is_valid_soul_infection_target_trigger = {
	is_alive = yes
	POD_is_vampire_trigger = yes
	OR = {
		has_vicissitude_trigger = yes
		Has_trait = fleshcrafted
	}
	NOT = { POD_asakku_has_any_asakku_trait_trigger = yes }
	NOT = { has_character_modifier = POD_asakku_soul_infection_resistant }
	NOT = { has_trait = true_faith }
}

POD_asakku_is_valid_blood_inception_target_trigger = {
	is_alive = yes
	POD_is_vampire_trigger = yes
	is_imprisoned = yes
	NOT = { POD_asakku_has_asakku_trigger = yes }
	NOT = { POD_asakku_has_iryi_ascendant_trigger = yes }
	NOT = { has_trait = true_faith }
}

POD_asakku_is_valid_kamut_recruitment_target_trigger = {
	is_alive = yes
	faith = { religion = religion:talmahera_religion }
	NOT = { POD_asakku_has_elioud_trigger = yes }
	NOT = { POD_asakku_has_any_asakku_trait_trigger = yes }
	NOT = { has_character_flag = POD_asakku_kamut_recruitment_cooldown }
}
