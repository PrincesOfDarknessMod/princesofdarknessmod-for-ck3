POD_asakku_has_asakku_trigger = {
	OR = {
		has_trait = POD_asakku_asakku
		has_trait = POD_asakku_iryi_ascendant
	}
}

POD_asakku_has_any_asakku_trait_trigger = {
	OR = {
		has_trait = POD_asakku_soul_infection
		has_trait = POD_asakku_asakku
		has_trait = POD_asakku_iryi_ascendant
	}
}
