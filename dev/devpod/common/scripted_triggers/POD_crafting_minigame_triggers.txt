POD_node_is_starting = {
	OR = {
		var:node_type ?= flag:starting_node_amulet
		var:node_type ?= flag:starting_node_crown
		var:node_type ?= flag:starting_node_weapon
		var:node_type ?= flag:starting_node_regalia
		var:node_type ?= flag:starting_node_armor
	}
}
POD_node_requirement = {
	switch = {
		trigger = $FLAG$
		flag:basic_node_durability_bonus = { $OWNER$ = { gold >= tiny_gold_max_value } }
		flag:basic_node_decorations = { $OWNER$ = { gold >= tiny_gold_max_value } }
		flag:basic_node_sharpen_edge = {
			custom_tooltip = { text = "POD_only_weapons" $ARTIFACT$ = { artifact_slot_type = primary_armament } }
			$OWNER$ = { gold >= tiny_gold_max_value }
		}
		flag:node_fleshcraft = {
			$OWNER$ = {
				custom_tooltip = { text = "POD_crafting_minigame_fleshcraft_requirement"
					any_prisoner = {
						count >= 1
						NOT = { has_trait_with_flag = is_spirit }
					}
				}
			}
		}
		flag:node_soulforge = {
			$OWNER$ = {
				custom_tooltip = { text = "POD_crafting_minigame_soulforge_requirement"
					any_prisoner = {
						count >= 1
						has_trait = wraith
					}
				}
			}
		}
		flag:node_fetishsmith = {
			$OWNER$ = {
				custom_tooltip = { text = "POD_crafting_minigame_fetishsmith_requirement"
					any_prisoner = {
						count >= 1
						has_trait = spirit
					}
				}
			}
		}
		flag:node_frost_facsimile = { $CRAFTER$ = { NOT = { has_character_modifier = glamour0_modifier } } }
		flag:node_corrupt_relic = { $CRAFTER$ = { NOT = { has_character_modifier = resolve0_modifier } } }
		flag:node_horrid_blade = {
			custom_tooltip = { text = "POD_only_weapons" $ARTIFACT$ = { artifact_slot_type = primary_armament } }
			$CRAFTER$ = { POD_has_enough_blood_for_discipline = { AMOUNT = 2 } }
		}


		flag:node_reweave_common = { $CRAFTER$ = { custom_tooltip = { text = "POD_crafting_minigame_not_enough_energy" energy_points_value >= 2 } } }
		flag:node_reweave_masterwork = { $CRAFTER$ = { custom_tooltip = { text = "POD_crafting_minigame_not_enough_energy" energy_points_value >= 4 } } }
		flag:node_reweave_famed = { $CRAFTER$ = { custom_tooltip = { text = "POD_crafting_minigame_not_enough_energy" energy_points_value >= 6 } } }
		flag:node_reweave_illustrious = { $CRAFTER$ = { custom_tooltip = { text = "POD_crafting_minigame_not_enough_energy" energy_points_value >= 8 } } }

		flag:node_enchant_object = { $CRAFTER$ = { piety >= 250 stress_level < 3 } }
		flag:node_ectoplasmic_artifact = { $CRAFTER$ = { piety >= 250 stress_level < 3 } }


		flag:node_mummy_ward = {
			custom_tooltip = { text = "POD_crafting_minigame_not_enough_energy" $CRAFTER$ = { energy_points_value >= 2 } }
			$OWNER$ = { gold >= tiny_gold_max_value }
			custom_tooltip = { text = "POD_crafting_minigame_too_low_durability" $ARTIFACT$ ?= { artifact_max_durability >= 20 } }
		}
		flag:node_mummy_talisman = {
			custom_tooltip = { text = "POD_crafting_minigame_not_enough_energy" $CRAFTER$ = { energy_points_value >= 5 } }
			$OWNER$ = { gold >= tiny_gold_max_value }
			custom_tooltip = { text = "POD_crafting_minigame_too_low_durability" $ARTIFACT$ ?= { artifact_max_durability >= 20 } }
		}
		flag:node_mummy_full_talisman = {
			custom_tooltip = { text = "POD_crafting_minigame_not_enough_energy" $CRAFTER$ = { energy_points_value >= 5 } }
			$OWNER$ = { gold >= tiny_gold_max_value }
			custom_tooltip = { text = "POD_crafting_minigame_too_low_durability" $ARTIFACT$ ?= { artifact_max_durability >= 20 } }
		}
		flag:node_mummy_inner_strength = {
			custom_tooltip = { text = "POD_crafting_minigame_not_enough_energy" $CRAFTER$ = { energy_points_value >= 5 } }
			$OWNER$ = { gold >= tiny_gold_max_value }
			custom_tooltip = { text = "POD_crafting_minigame_too_low_durability" $ARTIFACT$ ?= { artifact_max_durability >= 20 } }
		}
		flag:node_mummy_veil = {
			custom_tooltip = { text = "POD_crafting_minigame_not_enough_energy" $CRAFTER$ = { energy_points_value >= 3 } }
			$OWNER$ = { gold >= tiny_gold_max_value }
			custom_tooltip = { text = "POD_crafting_minigame_too_low_durability" $ARTIFACT$ ?= { artifact_max_durability >= 20 } }
		}
		flag:node_mummy_whisper = {
			custom_tooltip = { text = "POD_crafting_minigame_not_enough_energy" $CRAFTER$ = { energy_points_value >= 4 } }
			$OWNER$ = { gold >= tiny_gold_max_value }
			custom_tooltip = { text = "POD_crafting_minigame_too_low_durability" $ARTIFACT$ ?= { artifact_max_durability >= 20 } }
		}
		flag:node_mummy_sekhem_vessel = {
			custom_tooltip = { text = "POD_crafting_minigame_not_enough_energy" $CRAFTER$ = { energy_points_value >= 10 } }
			$OWNER$ = { gold >= tiny_gold_max_value }
			custom_tooltip = { text = "POD_crafting_minigame_too_low_durability" $ARTIFACT$ ?= { artifact_max_durability >= 20 } }
		}

		flag:node_blade_of_stone = {
			custom_tooltip = { text = "POD_only_weapons" $ARTIFACT$ = { artifact_slot_type = primary_armament } }
			$CRAFTER$ = { stress_level <= 3 }
		}
		#ENHANCING
		flag:node_enhance_object = { $CRAFTER$ = { piety >= 250 } }
		flag:node_the_forgers_hammer = { $CRAFTER$ = { POD_has_enough_blood_for_discipline = { AMOUNT = 2 } } }
		flag:node_aid_from_spirits = { $CRAFTER$ = { POD_has_enough_blood_for_discipline = { AMOUNT = 1 } } }
		flag:node_sword_of_the_righteous = { $CRAFTER$ = { POD_has_enough_blood_for_discipline = { AMOUNT = 3 } } }
		flag:node_rune = { $CRAFTER$ = { NOT = { has_character_modifier = glamour0_modifier } } }
		flag:node_bless = { $CRAFTER$ = { piety >= 250 } }
		flag:node_sorcerers_blade = { $CRAFTER$ = { NOT = { has_character_modifier = POD_gnosis_0_modifier } } }
		flag:node_bound_spirit = {
			$OWNER$ = {
				custom_tooltip = { text = "POD_crafting_minigame_fetishsmith_requirement"
					any_prisoner = {
						count >= 1
						has_trait = spirit
					}
				}
			}
		}
		flag:node_impress = {
			$OWNER$ = {
				custom_tooltip = { text = "POD_crafting_minigame_soulforge_requirement"
					any_prisoner = {
						count >= 1
						has_trait = spirit
					}
				}
			}
		}

		flag:node_empower_jade = { $CRAFTER$ = { POD_has_enough_pathos_for_arcanos = { AMOUNT = 1 } } }
		flag:node_conforma = {
			$CRAFTER$ = {
				trigger_if = { limit = { has_perk = scinde_perk } POD_has_enough_blood_for_discipline = { AMOUNT = 2 } }
				trigger_else = { POD_has_enough_blood_for_discipline = { AMOUNT = 3 } }
			}
		}
		flag:node_imbue_object = {
			$OWNER$ = {
				custom_tooltip = { text = "POD_crafting_minigame_imbue_object_requirement"
					any_prisoner = {
						count >= 1
					}
				}
			}
		}
		flag:node_blood_blade = {
			custom_tooltip = { text = "POD_only_weapons" $ARTIFACT$ = { artifact_slot_type = primary_armament } }
			$CRAFTER$ = { stress_level <= 3 }
		}

		flag:node_enchantment_prowess = {
			$CRAFTER$ = { stress_level <= 3 } $OWNER$ = { gold >= tiny_gold_max_value }
			custom_tooltip = { text = "POD_only_weapons" $ARTIFACT$ = { artifact_slot_type = primary_armament } }
		}
		flag:node_enchantment_durability = { $CRAFTER$ = { stress_level <= 3 } $OWNER$ = { gold >= tiny_gold_max_value } }
		flag:node_enchantment_attribute = {
			$CRAFTER$ = { stress_level <= 3 }
			trigger_if = { limit = { $CRAFTER$ = { has_perk = numina_enchantment_6_perk } } }
			trigger_else_if = { limit = { $CRAFTER$ = { has_perk = numina_enchantment_5_perk } } custom_tooltip = { text = POD_crafting_minigame_too_many_enchantment_2 NOT = { var:enchantment_value ?= 2 } } }
			trigger_else = { custom_tooltip = { text = POD_crafting_minigame_too_many_enchantment_1 NOT = { has_variable = enchantment_value } } }
			$OWNER$ = { gold >= tiny_gold_max_value }
		}

		flag:node_enchantment_enhance_prowess = {
			$CRAFTER$ = { stress_level <= 3 } $OWNER$ = { gold >= minor_gold_max_value }
			custom_tooltip = { text = "POD_only_weapons" $ARTIFACT$ = { artifact_slot_type = primary_armament } }
		}
		flag:node_enchantment_enhance_durability = { $CRAFTER$ = { stress_level <= 3 } $OWNER$ = { gold >= minor_gold_max_value } }
		flag:node_enchantment_enhance_attribute = {
			$CRAFTER$ = { stress_level <= 3 }
			trigger_if = { limit = { $CRAFTER$ = { has_perk = numina_enchantment_6_perk } } custom_tooltip = { text = POD_crafting_minigame_too_many_enchantment_2 NOT = { var:enchantment_value ?= 2 } } }
			trigger_else = { custom_tooltip = { text = POD_crafting_minigame_too_many_enchantment_1 NOT = { has_variable = enchantment_value } } }
			$OWNER$ = { gold >= minor_gold_max_value }
		}
		#BREWING
		flag:basic_node_sweetener = { $OWNER$ = { gold >= tiny_gold_max_value } }
		flag:basic_node_longevity = { $OWNER$ = { gold >= tiny_gold_max_value } }
		flag:basic_node_long_effects = { $OWNER$ = { gold >= tiny_gold_max_value } }

		flag:node_mild_poison = { $OWNER$ = { gold >= minor_gold_max_value } }
		flag:node_simple_elixir = { $OWNER$ = { gold >= minor_gold_max_value } }
		flag:node_simple_philtre = { $OWNER$ = { gold >= minor_gold_max_value } }
		flag:node_potion_of_resilience = { $OWNER$ = { gold >= minor_gold_max_value } }
		flag:node_simple_tonic = {
			$OWNER$ = {
				trigger_if = { limit = { has_perk = water_of_life_perk } gold >= tiny_gold_max_value }
				trigger_else = { gold >= minor_gold_max_value }
			}
		}

		flag:node_drink_of_seven_days_rest = { $OWNER$ = { gold >= minor_gold_max_value } }
		flag:node_philosophers_stone = { $OWNER$ = { gold >= minor_gold_max_value } }
		flag:node_panacea = { $OWNER$ = { gold >= minor_gold_max_value } }

		flag:node_alchemy_antidepresants = { $OWNER$ = { gold >= minor_gold_max_value } }
		flag:node_alchemy_aphrodisiac = { $OWNER$ = { gold >= minor_gold_max_value } }
		flag:node_alchemy_poison = { $OWNER$ = { gold >= minor_gold_max_value } }
		flag:node_alchemy_antibiotics = { $OWNER$ = { gold >= minor_gold_max_value } }
		flag:node_alchemy_stimulant = { $OWNER$ = { gold >= minor_gold_max_value } }
		flag:node_alchemy_health_booster = { $OWNER$ = { gold >= minor_gold_max_value } }

		flag:node_alchemy_immortal = {
			$OWNER$ = {
				gold >= medium_gold_max_value
				has_trait_xp = {
					trait = pod_crafter
					track = brewing
					value >= 70
				}
			}
		}

		flag:node_alchemy_vampire = {
			$OWNER$ = {
				custom_tooltip = { text = "POD_crafting_minigame_alchemy_vampire_requirement"
					any_prisoner = {
						count >= 1
						has_trait = vampire
					}
				}
			}
		}
		flag:node_alchemy_kinfolk = {
			$OWNER$ = {
				custom_tooltip = { text = "POD_crafting_minigame_alchemy_kinfolk_requirement"
					any_prisoner = {
						count >= 1
						POD_splat_is_shifter_trigger = yes
					}
				}
			}
		}
		flag:node_alchemy_kinain = {
			$OWNER$ = {
				custom_tooltip = { text = "POD_crafting_minigame_alchemy_kinain_requirement"
					any_prisoner = {
						count >= 1
						has_POD_fae_blood_trigger = yes
					}
				}
			}
		}
		fallback = {  }
	}
}

POD_fleshcraft_requirements = {
	OR = {
		has_perk = living_testudo_perk
		has_perk = shape_flesh_perk
		has_perk = catgut_perk
		AND = {
			has_perk = mold_perk
			has_perk = realm_nature_3_perk
		}
	}
}
POD_soulforge_requirements = {
	OR = {
		has_trait = wraith
		has_perk = ectoplasmic_generation_5_perk
		has_perk = numina_necromancy_4_perk
		has_perk = shape_true_jade_perk #FOR NOW
	}
}

POD_enabled_node = {
	NOT = { is_target_in_variable_list = { name = disabled_nodes target = flag:$NODE$ } }
}
POD_artifact_can_be_enhanced = {
	custom_tooltip = { text = POD_artifact_can_be_enhanced NOT = { var:enhance_amount ?= artifact_enhance_limit } }
