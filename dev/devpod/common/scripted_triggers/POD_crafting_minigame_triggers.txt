POD_node_is_starting = {
        OR = {
                var:node_type ?= flag:starting_node_amulet
                var:node_type ?= flag:starting_node_crown
                var:node_type ?= flag:starting_node_weapon
                var:node_type ?= flag:starting_node_regalia
                var:node_type ?= flag:starting_node_armor
        }
}
POD_node_requirement = {
        switch = {
                trigger = scope:story.var:node_type
                flag:basic_node_durability_bonus = { PREV = { gold >= tiny_gold_max_value } }
                flag:basic_node_decorations = { PREV = { gold >= tiny_gold_max_value } }
                flag:basic_node_sharpen_edge = { 
                        custom_tooltip = { text = "POD_only_weapons" var:dummy_artifact = { artifact_slot_type = primary_armament } } 
                        PREV = { gold >= tiny_gold_max_value } 
                }
                flag:basic_node_sweetener = { PREV = { gold >= tiny_gold_max_value } }
                flag:basic_node_longevity = { PREV = { gold >= tiny_gold_max_value } }
                flag:basic_node_long_effects = { PREV = { gold >= tiny_gold_max_value } }
                flag:node_fleshcraft = { 
                        PREV = { 
                                custom_tooltip = { text = "POD_crafting_minigame_fleshcraft_requirement" 
                                        any_prisoner = {
                                                count >= 1
                                                NOT = { has_trait_with_flag = is_spirit }
                                        }
                                }
                        }
                }
                flag:node_soulforge = { 
                        PREV = { 
                                custom_tooltip = { text = "POD_crafting_minigame_soulforge_requirement" 
                                        any_prisoner = {
                                                count >= 1
                                                has_trait = wraith
                                        }
                                }
                        }
                }
                flag:node_fetishsmith = { 
                        PREV = { 
                                custom_tooltip = { text = "POD_crafting_minigame_fetishsmith_requirement" 
                                        any_prisoner = {
                                                count >= 1
                                                has_trait = spirit
                                        }
                                }
                        }
                }
                flag:node_frost_facsimile = { var:crafter = { NOT = { has_character_modifier = glamour0_modifier } } }
                flag:node_eightfold_yin_mantle = { 
                        var:crafter = { custom_tooltip = {  text = "i_am_empty_of_chi" NAND = { has_character_modifier = yinchi0_modifier has_character_modifier = yangchi0_modifier } } }
                }
                flag:node_corrupt_relic = { var:crafter = { NOT = { has_character_modifier = resolve0_modifier } } }
                flag:node_horrid_blade = { 
                        custom_tooltip = { text = "POD_only_weapons" var:dummy_artifact = { artifact_slot_type = primary_armament } } 
                        var:crafter = { POD_has_enough_blood_for_discipline = { AMOUNT = 2 } } }
                flag:node_blood_blade = { 
                        custom_tooltip = { text = "POD_only_weapons" var:dummy_artifact = { artifact_slot_type = primary_armament } } 
                        var:crafter = { custom_tooltip = {  text = "i_am_empty_of_chi" NAND = { has_character_modifier = yinchi0_modifier has_character_modifier = yangchi0_modifier } } }
                }
                flag:node_goblin_sword = { 
                        custom_tooltip = { text = "POD_only_weapons" var:dummy_artifact = { artifact_slot_type = primary_armament } } 
                        var:crafter = { custom_tooltip = {  text = "i_am_empty_of_chi" NAND = { has_character_modifier = yinchi0_modifier has_character_modifier = yangchi0_modifier } } }
                }
                flag:node_blade_of_stone = { 
                        custom_tooltip = { text = "POD_only_weapons" var:dummy_artifact = { artifact_slot_type = primary_armament } } 
                        var:crafter = { custom_tooltip = {  text = "i_am_empty_of_chi" NAND = { has_character_modifier = yinchi0_modifier has_character_modifier = yangchi0_modifier } } }
                }

                flag:node_reweave_common = { var:crafter = { energy_points_value >= 2 } }
                flag:node_reweave_masterwork = { var:crafter = { energy_points_value >= 4 } }
                flag:node_reweave_famed = { var:crafter = { energy_points_value >= 6 } }
                flag:node_reweave_illustrious = { var:crafter = { energy_points_value >= 8 } }

                flag:node_enchant_object = { var:crafter = { piety >= 250 stress_level < 3 } }
                flag:node_ectoplasmic_artifact = { var:crafter = { piety >= 250 stress_level < 3 } }
                
                fallback = {  }
        }
}
