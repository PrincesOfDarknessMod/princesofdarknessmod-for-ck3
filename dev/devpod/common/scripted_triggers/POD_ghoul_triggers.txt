POD_is_bound_ghoul_trigger = {
	has_trait = ghoul
	num_of_relation_ghoul_domitor >= 1
}
POD_is_masterless_ghoul_trigger = {
	has_trait = ghoul
	num_of_relation_ghoul_domitor = 0
}

POD_has_ghoul_rebellion_strength_trigger = {
	save_temporary_scope_value_as = {
		name = required_rebellion_strength
		value = $VALUE$
	}
	custom_tooltip = {
		text = POD_has_ghoul_rebellion_strength_trigger
		POD_ghoul_objective_rebellion_strength >= $VALUE$
	}
}
