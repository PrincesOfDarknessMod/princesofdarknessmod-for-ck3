POD_is_bound_ghoul_trigger = {
	has_trait = ghoul
	num_of_relation_ghoul_domitor >= 1
}
POD_is_masterless_ghoul_trigger = {
	has_trait = ghoul
	num_of_relation_ghoul_domitor = 0
}
POD_is_bound_ghoul_of = {
	has_trait = ghoul
	has_relation_ghoul_domitor = $DOMITOR$
}

# task_contract scope
POD_ghoul_mission_can_be_finished = {
	save_temporary_scope_as = ghoul_contract
	task_contract_employer ?= { is_alive = yes }
	task_contract_taker = {
		POD_is_bound_ghoul_of = { DOMITOR = prev.task_contract_employer }
		# safeguard since contract scopes don't get deleted immediately after invalidation
		any_character_active_contract = { scope:ghoul_contract ?= this }
	}
	POD_contract_has_not_expired = yes
}

POD_has_ghoul_rebellion_strength_trigger = {
	save_temporary_scope_value_as = {
		name = required_rebellion_strength
		value = $VALUE$
	}
	custom_tooltip = {
		text = POD_has_ghoul_rebellion_strength_trigger
		POD_ghoul_objective_rebellion_strength >= $VALUE$
	}
}
