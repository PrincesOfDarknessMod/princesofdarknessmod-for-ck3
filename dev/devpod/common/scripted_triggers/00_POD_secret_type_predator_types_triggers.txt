#Triggers for Shunned and Criminal Secret Types for POD

# Predator Types

# Consensualist

secret_predatortype_consensualist_is_valid_trigger = {
	$OWNER$ = { 
		NOT = { 
			has_trait = predatortype_consensualist
		}
	}
}

secret_predatortype_consensualist_is_shunned_trigger = {
	$OWNER$ = {
		OR = {
			faith = { has_doctrine_parameter = vampirethemasquerade_lax }
			any_liege_or_above = { faith = { has_doctrine_parameter = vampirethemasquerade_lax } }
			faith = { has_doctrine_parameter = vampirethemasquerade_cult }
			any_liege_or_above = { faith = { has_doctrine_parameter = vampirethemasquerade_cult } }
		}
	}
}

secret_predatortype_consensualist_is_criminal_trigger = {
	$OWNER$ = {
		OR = {
			faith = { has_doctrine_parameter = vampirethemasquerade_strict }
			any_liege_or_above = { faith = { has_doctrine_parameter = vampirethemasquerade_strict } }
		}
	}
}

# Is Triggers, Known Triggers 

is_predatortype_consensualist_trigger = {
	custom_description = {
		text = is_a_predatortype_consensualist
		subject = this
		OR = {
			has_trait = predatortype_consensualist
			any_secret = { secret_type = secret_predatortype_consensualist }
		}
	}
}

is_predatortype_consensualist_known_by_trigger = {
	OR = {
		has_trait = predatortype_consensualist
		custom_description = {
			text = has_a_predatortype_consensualist_secret_known_by_character
			subject = this
			object = $CHARACTER$
			any_secret = { 
				secret_type = secret_predatortype_consensualist
				is_known_by = $CHARACTER$
			}
		}
	}
}
