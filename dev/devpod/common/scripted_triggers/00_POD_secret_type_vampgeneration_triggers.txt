secret_vampgeneration14_is_shunned_trigger = {
	$OWNER$ = {
		OR = {
			faith = { has_doctrine_parameter = vampirethinblood_shunned  }
			any_liege_or_above = { faith = { has_doctrine_parameter = vampirethinblood_shunned  } }
		}
	}
}
