######################################################
######   POD COMBAT MOVE EXECUTION TRIGGERS
######################################################
# The single generic trigger that validates ANY combat move by reading
# attributes off its story_cycle scope, replacing the per-move
# `pod_combat_can_use_{key}_trigger` pattern.
#
# Called from:
#   - pod_combat_execute_move SGUI `is_valid` — mid-combat PC action
#     button enable gate; full runtime check (unlock + AP + energy +
#     targeting + per-resource).
#   - pod_combat_ai_select_and_execute_move_effect — AI move selector;
#     used as the `trigger` filter inside the random_in_list weight
#     block so weighted-but-unaffordable picks are excluded. The
#     selector uses random_in_list (proportional weighted pick), NOT
#     ordered_in_list — see the comment in pod_combat_ai_effects.txt
#     for the rationale.
# NOT called from the loadout-editor SGUIs: pod_combat_loadout_option
# / pod_combat_loadout_toggle gate VISIBILITY and CLICKABILITY on
# `can_execute_decision = scope:story.var:decision` directly, which is
# a narrower unlock-only subset of this trigger (no AP / energy /
# targeting checks — those don't apply outside an active combat).
#
# Called on character scope (the combat host, where the workspace
# lives). The ACTOR parameter specifies which character's traits the
# unlock check runs against — for the PC this is the same character,
# but for AI turns the combat host is still the PC while the actor is
# the AI NPC.


POD_can_execute_combat_move = {
	# $STORY$ = the POD_combat_move story_cycle
	# $ACTOR$ = the character whose splat / trait / perk must satisfy
	#          the hidden decision's unlock gate. Workspace reads
	#          (AP, energy) resolve against the caller's root scope
	#          (the combat host — may differ from the actor on AI
	#          turns).

	# Combat must not already be resolved.
	NOT = { pod_combat_is_over_trigger = yes }

	# Unlock/splat/trait/perk gate. Run against the actor, NOT the
	# combat host — during AI turns the host is the PC but the unlock
	# requirements live on the AI NPC (their traits, their perks).
	# Without this scope switch every trait-gated move was silently
	# rejected for every AI actor because the check was evaluating
	# the PC's traits instead of the AI's.
	$ACTOR$ = {
		can_execute_decision = $STORY$.var:decision
	}

	# AP affordability. Reads the workspace-side current AP value.
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = ap_cost }
		}
		pod_combat_can_afford_ap_trigger = { COST = $STORY$.var:ap_cost }
	}

	# Energy affordability. PoD's splat-agnostic trigger figures out
	# which pool (blood/gnosis/pathos/angst/resolve/glamour/dread/
	# willpower) applies and compares it to the current reserve — but
	# ONLY on the scope it's evaluated against. Combat host = PC, so
	# calling it bare would check the PC's pool for every AI turn.
	# Wrap in $ACTOR$ so the reserve comparison hits the acting
	# character. The move's required_trait gate (enforced above via
	# can_execute_decision) guarantees the actor has a matching pool,
	# so this comparison is always meaningful.
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = energy_cost }
		}
		$ACTOR$ = {
			POD_has_enough_splat_agnostic_energy_reserves = { AMOUNT = $STORY$.var:energy_cost }
		}
	}

	# CK3 resource costs are PC-only — only slot 0 (the player) spends
	# these. AI combatants get the move for free on the CK3 side; the
	# trigger reflects this by gating these checks on the player turn.
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = gold_cost }
			pod_combat_is_player_turn_trigger = yes
		}
		gold >= $STORY$.var:gold_cost
	}

	trigger_if = {
		limit = {
			$STORY$ = { has_variable = piety_cost }
			pod_combat_is_player_turn_trigger = yes
		}
		piety >= $STORY$.var:piety_cost
	}

	trigger_if = {
		limit = {
			$STORY$ = { has_variable = prestige_cost }
			pod_combat_is_player_turn_trigger = yes
		}
		prestige >= $STORY$.var:prestige_cost
	}

	# stress_cost adds stress on execution — it's always "affordable"

	# Targeting validity during the player's turn. The PC has explicitly
	# picked a slot via pod_combat_pc_target; the move's targeting flag
	# dictates which range is acceptable. Outside of the player turn
	# (AI eval), target selection happens via the AI swap effects AFTER
	# move choice, so this check is skipped.
	trigger_if = {
		limit = {
			pod_combat_is_player_turn_trigger = yes
			$STORY$ = { var:targeting = flag:single_enemy }
		}
		has_variable = pod_combat_pc_target
		OR = {
			AND = { var:pod_combat_pc_target = 5 has_variable = pod_combat_slot_5 var:pod_combat_slot_5 = { has_variable = slot_char var:slot_alive = 1 } }
			AND = { var:pod_combat_pc_target = 6 has_variable = pod_combat_slot_6 var:pod_combat_slot_6 = { has_variable = slot_char var:slot_alive = 1 } }
			AND = { var:pod_combat_pc_target = 7 has_variable = pod_combat_slot_7 var:pod_combat_slot_7 = { has_variable = slot_char var:slot_alive = 1 } }
			AND = { var:pod_combat_pc_target = 8 has_variable = pod_combat_slot_8 var:pod_combat_slot_8 = { has_variable = slot_char var:slot_alive = 1 } }
			AND = { var:pod_combat_pc_target = 9 has_variable = pod_combat_slot_9 var:pod_combat_slot_9 = { has_variable = slot_char var:slot_alive = 1 } }
		}
	}

	trigger_if = {
		limit = {
			pod_combat_is_player_turn_trigger = yes
			$STORY$ = { var:targeting = flag:single_ally }
		}
		has_variable = pod_combat_pc_target
		OR = {
			AND = { var:pod_combat_pc_target = 1 has_variable = pod_combat_slot_1 var:pod_combat_slot_1 = { has_variable = slot_char var:slot_alive = 1 } }
			AND = { var:pod_combat_pc_target = 2 has_variable = pod_combat_slot_2 var:pod_combat_slot_2 = { has_variable = slot_char var:slot_alive = 1 } }
			AND = { var:pod_combat_pc_target = 3 has_variable = pod_combat_slot_3 var:pod_combat_slot_3 = { has_variable = slot_char var:slot_alive = 1 } }
			AND = { var:pod_combat_pc_target = 4 has_variable = pod_combat_slot_4 var:pod_combat_slot_4 = { has_variable = slot_char var:slot_alive = 1 } }
		}
	}

	trigger_if = {
		limit = {
			pod_combat_is_player_turn_trigger = yes
			$STORY$ = { var:targeting = flag:single_pc }
		}
		# Protect-the-PC moves target slot 0 (the player character).
		has_variable = pod_combat_slot_0
		var:pod_combat_slot_0 = { has_variable = slot_char var:slot_alive = 1 }
	}

	trigger_if = {
		limit = {
			pod_combat_is_player_turn_trigger = yes
			$STORY$ = { var:targeting = flag:team_enemy }
		}
		# At least one alive enemy slot.
		OR = {
			AND = { has_variable = pod_combat_slot_5 var:pod_combat_slot_5 = { has_variable = slot_char var:slot_alive = 1 } }
			AND = { has_variable = pod_combat_slot_6 var:pod_combat_slot_6 = { has_variable = slot_char var:slot_alive = 1 } }
			AND = { has_variable = pod_combat_slot_7 var:pod_combat_slot_7 = { has_variable = slot_char var:slot_alive = 1 } }
			AND = { has_variable = pod_combat_slot_8 var:pod_combat_slot_8 = { has_variable = slot_char var:slot_alive = 1 } }
			AND = { has_variable = pod_combat_slot_9 var:pod_combat_slot_9 = { has_variable = slot_char var:slot_alive = 1 } }
		}
	}

	# AI-side single_ally gate. Without this, an AI with a single_ally
	# heal/buff move equipped (e.g. an enemy NPC who unlocks a clan
	# heal) and no alive same-team allies (1v1, or all allies KO'd)
	# would still pass selection. The ally-swap effect's if/else_if
	# chain wouldn't find a match, the workspace would stay on the
	# enemy target, and the heal/buff would land on the PC. This
	# gate makes single_ally moves visible to the AI selector only
	# when there's actually an ally to target.
	trigger_if = {
		limit = {
			NOT = { pod_combat_is_player_turn_trigger = yes }
			$STORY$ = { var:targeting = flag:single_ally }
		}
		pod_combat_ai_has_available_ally_trigger = yes
	}

	# team_ally and self are always satisfiable during the player turn —
	# team_ally includes the acting PC (slot 0 is always alive during
	# their own turn), and self is the actor. Same on the AI 
	# side: team_ally always includes the acting AI itself, and
	# self is self. single_enemy is vacuously satisfied because
	# combat ends (victory check) when no enemies remain, so the AI
	# never reaches selection without at least one valid enemy.
}
