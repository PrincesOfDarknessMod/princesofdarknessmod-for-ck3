######################################################
######   POD COMBAT MOVE EXECUTION TRIGGERS
######################################################
# The single generic trigger that validates ANY combat move by reading
# attributes off its story_cycle scope, replacing the per-move
# `pod_combat_can_use_{key}_trigger` pattern.
#
# Called from:
#   - pod_combat_execute_move SGUI `is_valid` — mid-combat PC action
#     button enable gate; full runtime check (unlock + AP + energy +
#     targeting + per-resource).
#   - pod_combat_ai_select_and_execute_move_effect — AI move selector;
#     used as the `trigger` filter inside the random_in_list weight
#     block so weighted-but-unaffordable picks are excluded. The
#     selector uses random_in_list (proportional weighted pick), NOT
#     ordered_in_list — see the comment in pod_combat_ai_effects.txt
#     for the rationale.
# NOT called from the loadout-editor SGUIs: pod_combat_loadout_option
# / pod_combat_loadout_toggle gate VISIBILITY and CLICKABILITY on
# `can_execute_decision = scope:story.var:decision` directly, which is
# a narrower unlock-only subset of this trigger (no AP / energy /
# targeting checks — those don't apply outside an active combat).
#
# Called on character scope (the combat host, where the workspace
# lives). The ACTOR parameter specifies which character's traits the
# unlock check runs against — for the PC this is the same character,
# but for AI turns the combat host is still the PC while the actor is
# the AI NPC.


POD_can_execute_combat_move = {
	# $STORY$ = the POD_combat_move story_cycle
	# $ACTOR$ = the character whose splat / trait / perk must satisfy
	#          the hidden decision's unlock gate. Workspace reads
	#          (AP, energy) resolve against the caller's root scope
	#          (the combat host — may differ from the actor on AI
	#          turns).

	# Combat must not already be resolved.
	NOT = { pod_combat_is_over_trigger = yes }

	# Unlock/splat/trait/perk gate. Run against the actor, NOT the
	# combat host — during AI turns the host is the PC but the unlock
	# requirements live on the AI NPC (their traits, their perks).
	# Without this scope switch every trait-gated move was silently
	# rejected for every AI actor because the check was evaluating
	# the PC's traits instead of the AI's.
	$ACTOR$ = {
		can_execute_decision = $STORY$.var:decision
	}

	# AP affordability. Reads the workspace-side current AP value.
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = ap_cost }
		}
		pod_combat_can_afford_ap_trigger = { COST = $STORY$.var:ap_cost }
	}

	# Energy affordability. PoD's splat-agnostic trigger figures out
	# which pool (blood/gnosis/pathos/angst/resolve/glamour/dread/
	# willpower) applies and compares it to the current reserve — but
	# ONLY on the scope it's evaluated against. Combat host = PC, so
	# calling it bare would check the PC's pool for every AI turn.
	# Wrap in $ACTOR$ so the reserve comparison hits the acting
	# character. The move's required_trait gate (enforced above via
	# can_execute_decision) guarantees the actor has a matching pool,
	# so this comparison is always meaningful.
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = energy_cost }
		}
		$ACTOR$ = {
			POD_has_enough_splat_agnostic_energy_reserves = { AMOUNT = $STORY$.var:energy_cost }
		}
	}

	# CK3 resource costs are PC-only — only slot 0 (the player) spends
	# these. AI combatants get the move for free on the CK3 side; the
	# trigger reflects this by gating these checks on the player turn.
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = gold_cost }
			pod_combat_is_player_turn_trigger = yes
		}
		gold >= $STORY$.var:gold_cost
	}

	trigger_if = {
		limit = {
			$STORY$ = { has_variable = piety_cost }
			pod_combat_is_player_turn_trigger = yes
		}
		piety >= $STORY$.var:piety_cost
	}

	trigger_if = {
		limit = {
			$STORY$ = { has_variable = prestige_cost }
			pod_combat_is_player_turn_trigger = yes
		}
		prestige >= $STORY$.var:prestige_cost
	}

	# stress_cost adds stress on execution — it's always "affordable"

	# Duplicate-stance / reaction guard — reject moves that try to
	# apply a stance / reaction the actor already has. Mirrors the
	# AI selector's NAND block in pod_combat_ai_effects.txt; without
	# the matching guard on the PC side, the player could spam stealth
	# while already stealthed and double-bake the +30 hit-chance bonus
	# (slot_hit_chance saved with +30, second stealth adds another +30,
	# now the actor sits at baseline+60 until two separate consumes
	# revert it).
	#
	# NAND, NOT NOR: "exclude when BOTH the move applies the stance
	# AND the actor already has it" — so a stance-applying move stays
	# valid for an actor without that stance, and a non-stance move
	# stays valid regardless of the actor's stance state. (Plain NOT
	# would blacklist every other move as soon as one stance is on.)
	NAND = {
		$STORY$ = { has_variable = set_guard }
		has_variable = pod_combat_actor_is_guarding
	}
	NAND = {
		$STORY$ = { has_variable = dodge }
		has_variable = pod_combat_actor_is_dodging
	}
	NAND = {
		$STORY$ = { has_variable = stealth }
		has_variable = pod_combat_actor_is_stealthed
	}
	NAND = {
		$STORY$ = { has_variable = retaliate_if_hit }
		has_variable = pod_combat_actor_retaliate_hit
	}
	NAND = {
		$STORY$ = { has_variable = retaliate_if_missed }
		has_variable = pod_combat_actor_retaliate_miss
	}

	# Tier-redundancy guard — reject moves that would apply a tier
	# buff/debuff at or below the actor/target's current tier. Mirrors
	# the AI selector's redundancy gate in pod_combat_ai_effects.txt
	# §random_in_list/limit so PC move buttons match the AI's
	# "upgrade-only" selection contract. Without this, a tier-1 buff
	# move stays clickable after a tier-3 buff of the same stat
	# landed; firing it downgrades the actor's tier instead of being
	# a no-op.
	#
	# Each gate is a single trigger_if (not NAND) so the variable read
	# only happens when both the move attribute AND the workspace var
	# exist — saves a "variable not set" warning when either is absent.
	# Strict `<` matches the AI guard (a tier-2 candidate against a
	# tier-2 current is rejected; same-tier reapplication is wasted AP).
	#
	# Coverage: workspace single-target tiers only (actor's self buffs
	# and target's debuffs). AoE moves can still downgrade a non-actor
	# slot whose tier is higher than the move's — handled at write time
	# inside POD_combat_apply_team_ally_buffs / _team_enemy_debuffs.
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = soak_buff_round_tier }
			has_variable = pod_combat_actor_soak_buff_round_tier
		}
		var:pod_combat_actor_soak_buff_round_tier < $STORY$.var:soak_buff_round_tier
	}
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = soak_buff_combat_tier }
			has_variable = pod_combat_actor_soak_buff_combat_tier
		}
		var:pod_combat_actor_soak_buff_combat_tier < $STORY$.var:soak_buff_combat_tier
	}
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = hit_buff_round_tier }
			has_variable = pod_combat_actor_hit_buff_round_tier
		}
		var:pod_combat_actor_hit_buff_round_tier < $STORY$.var:hit_buff_round_tier
	}
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = hit_buff_combat_tier }
			has_variable = pod_combat_actor_hit_buff_combat_tier
		}
		var:pod_combat_actor_hit_buff_combat_tier < $STORY$.var:hit_buff_combat_tier
	}
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = soak_debuff_round_tier }
			has_variable = pod_combat_target_soak_debuff_round_tier
		}
		var:pod_combat_target_soak_debuff_round_tier < $STORY$.var:soak_debuff_round_tier
	}
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = soak_debuff_combat_tier }
			has_variable = pod_combat_target_soak_debuff_combat_tier
		}
		var:pod_combat_target_soak_debuff_combat_tier < $STORY$.var:soak_debuff_combat_tier
	}
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = hit_debuff_round_tier }
			has_variable = pod_combat_target_hit_debuff_round_tier
		}
		var:pod_combat_target_hit_debuff_round_tier < $STORY$.var:hit_debuff_round_tier
	}
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = hit_debuff_combat_tier }
			has_variable = pod_combat_target_hit_debuff_combat_tier
		}
		var:pod_combat_target_hit_debuff_combat_tier < $STORY$.var:hit_debuff_combat_tier
	}

	# Targeting validity during the player's turn. The PC has explicitly
	# picked a slot via pod_combat_pc_target; the move's targeting flag
	# dictates which range is acceptable. Outside of the player turn
	# (AI eval), target selection happens via the AI swap effects AFTER
	# move choice, so this check is skipped.
	trigger_if = {
		limit = {
			pod_combat_is_player_turn_trigger = yes
			$STORY$ = { var:targeting = flag:single_enemy }
		}
		# Player must have an alive enemy currently selected via the
		# combat window's portrait-pick. custom_description swap-in
		# means the disabled-button tooltip renders "Select an enemy
		# target" instead of the raw trigger text — useful when every
		# equipped move greys out because the player accidentally
		# picked an ally slot.
		custom_description = {
			text = pod_combat_targeting_need_enemy_tt
			has_variable = pod_combat_pc_target
			OR = {
				AND = { var:pod_combat_pc_target = 5 has_variable = pod_combat_slot_5 var:pod_combat_slot_5 = { has_variable = slot_char var:slot_alive = 1 } }
				AND = { var:pod_combat_pc_target = 6 has_variable = pod_combat_slot_6 var:pod_combat_slot_6 = { has_variable = slot_char var:slot_alive = 1 } }
				AND = { var:pod_combat_pc_target = 7 has_variable = pod_combat_slot_7 var:pod_combat_slot_7 = { has_variable = slot_char var:slot_alive = 1 } }
				AND = { var:pod_combat_pc_target = 8 has_variable = pod_combat_slot_8 var:pod_combat_slot_8 = { has_variable = slot_char var:slot_alive = 1 } }
				AND = { var:pod_combat_pc_target = 9 has_variable = pod_combat_slot_9 var:pod_combat_slot_9 = { has_variable = slot_char var:slot_alive = 1 } }
			}
		}
	}

	trigger_if = {
		limit = {
			pod_combat_is_player_turn_trigger = yes
			$STORY$ = { var:targeting = flag:single_ally }
		}
		custom_description = {
			text = pod_combat_targeting_need_ally_tt
			has_variable = pod_combat_pc_target
			OR = {
				AND = { var:pod_combat_pc_target = 1 has_variable = pod_combat_slot_1 var:pod_combat_slot_1 = { has_variable = slot_char var:slot_alive = 1 } }
				AND = { var:pod_combat_pc_target = 2 has_variable = pod_combat_slot_2 var:pod_combat_slot_2 = { has_variable = slot_char var:slot_alive = 1 } }
				AND = { var:pod_combat_pc_target = 3 has_variable = pod_combat_slot_3 var:pod_combat_slot_3 = { has_variable = slot_char var:slot_alive = 1 } }
				AND = { var:pod_combat_pc_target = 4 has_variable = pod_combat_slot_4 var:pod_combat_slot_4 = { has_variable = slot_char var:slot_alive = 1 } }
			}
		}
	}

	trigger_if = {
		limit = {
			pod_combat_is_player_turn_trigger = yes
			$STORY$ = { var:targeting = flag:single_pc }
		}
		# Protect-the-PC moves target slot 0 (the player character).
		custom_description = {
			text = pod_combat_targeting_need_pc_alive_tt
			has_variable = pod_combat_slot_0
			var:pod_combat_slot_0 = { has_variable = slot_char var:slot_alive = 1 }
		}
	}

	trigger_if = {
		limit = {
			pod_combat_is_player_turn_trigger = yes
			$STORY$ = { var:targeting = flag:team_enemy }
		}
		# At least one alive enemy slot.
		OR = {
			AND = { has_variable = pod_combat_slot_5 var:pod_combat_slot_5 = { has_variable = slot_char var:slot_alive = 1 } }
			AND = { has_variable = pod_combat_slot_6 var:pod_combat_slot_6 = { has_variable = slot_char var:slot_alive = 1 } }
			AND = { has_variable = pod_combat_slot_7 var:pod_combat_slot_7 = { has_variable = slot_char var:slot_alive = 1 } }
			AND = { has_variable = pod_combat_slot_8 var:pod_combat_slot_8 = { has_variable = slot_char var:slot_alive = 1 } }
			AND = { has_variable = pod_combat_slot_9 var:pod_combat_slot_9 = { has_variable = slot_char var:slot_alive = 1 } }
		}
	}

	# AI-side single_ally gate. Without this, an AI with a single_ally
	# heal/buff move equipped (e.g. an enemy NPC who unlocks a clan
	# heal) and no alive same-team allies (1v1, or all allies KO'd)
	# would still pass selection. The ally-swap effect's if/else_if
	# chain wouldn't find a match, the workspace would stay on the
	# enemy target, and the heal/buff would land on the PC. This
	# gate makes single_ally moves visible to the AI selector only
	# when there's actually an ally to target.
	trigger_if = {
		limit = {
			NOT = { pod_combat_is_player_turn_trigger = yes }
			$STORY$ = { var:targeting = flag:single_ally }
		}
		pod_combat_ai_has_available_ally_trigger = yes
	}

	# team_ally and self are always satisfiable during the player turn —
	# team_ally includes the acting PC (slot 0 is always alive during
	# their own turn), and self is the actor. Same on the AI
	# side: team_ally always includes the acting AI itself, and
	# self is self. single_enemy is vacuously satisfied because
	# combat ends (victory check) when no enemies remain, so the AI
	# never reaches selection without at least one valid enemy.
}


######################################################
###### RESULT-CODE CONDITION TRIGGERS
######################################################
# Three primitive triggers that read pod_combat_last_result on the host
# scope and partition the result flag values by outcome class. Both
# the rider condition gate and the retaliation gate compose against
# these so the result → outcome mapping lives in one place.
#
# Partition membership (see COMBAT_REFERENCE.info §4 for the full map):
#   "hp_damage" — flag:hp_damage / hp_damage_retal / soaked_retal
#   "contact"   — hp_damage partition + flag:guard_blocked / soaked
#   "miss"      — flag:missed / dodged / stealthed
# Outcomes outside the three partitions: flag:utility, flag:morale_damage,
# flag:hp_healed, flag:morale_healed.

# The original attack landed real wound boxes on the target. The
# retaliation overlay flags (hp_damage_retal, soaked_retal) won't
# appear at rider/retaliation gate time on the SAME move (retaliation
# hasn't fired yet) but listing them keeps the partition complete and
# future-safe for any downstream code that calls this trigger after §15.
pod_combat_result_code_indicates_hp_damage_trigger = {
	has_variable = pod_combat_last_result
	OR = {
		var:pod_combat_last_result = flag:hp_damage
		var:pod_combat_last_result = flag:hp_damage_retal
		var:pod_combat_last_result = flag:soaked_retal
	}
}

# Anything that made physical contact: hp_damage flags plus blocked /
# soaked. Lets designers author "fires on touch even if absorbed"
# riders (e.g. a poison that delivers on the strike whether or not
# wounds were inflicted).
pod_combat_result_code_indicates_contact_trigger = {
	has_variable = pod_combat_last_result
	OR = {
		var:pod_combat_last_result = flag:hp_damage
		var:pod_combat_last_result = flag:guard_blocked
		var:pod_combat_last_result = flag:soaked
		var:pod_combat_last_result = flag:hp_damage_retal
		var:pod_combat_last_result = flag:soaked_retal
	}
}

# Anything that evaded the attack — straight miss, dodge consumed,
# stealth auto-miss. Used by retaliate_if_missed and by
# rider_condition = on_miss (e.g. a clumsy whiff that wounds the
# actor on the overswing).
pod_combat_result_code_indicates_miss_trigger = {
	has_variable = pod_combat_last_result
	OR = {
		var:pod_combat_last_result = flag:missed
		var:pod_combat_last_result = flag:dodged
		var:pod_combat_last_result = flag:stealthed
	}
}


######################################################
###### RIDER CONDITION GATE
######################################################
# Generic gate for rider effects (heal / morale damage / self-damage).
# A "rider" is a payload that piggybacks on the move's primary HP-damage
# step — e.g. a vampire bite that heals when it lands, a brutal swing
# that damages morale alongside HP, or a reckless charge that wounds
# the actor only on miss. Each rider lives behind its own condition var
# on the move story (heal_condition / morale_damage_condition /
# self_damage_condition). The §6 HP-damage outcome (read off
# pod_combat_last_result on the host) decides whether the rider
# fires.
#
# Recognized condition flags (must match RIDER_CONDITIONS in the
# generator):
#   always       — fire unconditionally (default when var unset)
#   on_hp_damage — flag:hp_damage / hp_damage_retal / soaked_retal
#   on_contact   — hp_damage partition + flag:guard_blocked / soaked
#   on_miss      — flag:missed / dodged / stealthed
#
# Caller scope: the combat host (where pod_combat_last_result lives).
# $STORY$ is the move story; $RIDER_VAR$ is the variable name on the
# story that holds the condition flag.
#
# Drain moves: the generator emits heal_condition = on_hp_damage for
# any move with the `drain` token in its `special` column, so the
# runtime gate handles "only heal on a landed hit" through the same
# rider machinery as every other condition rather than a special case.
pod_combat_rider_condition_passes_trigger = {
	# $STORY$     = the POD_combat_move story_cycle
	# $RIDER_VAR$ = the condition variable name on the story (e.g. heal_condition)
	# Default: var unset OR var = always → pass.
	trigger_if = {
		limit = {
			OR = {
				NOT = { $STORY$ = { has_variable = $RIDER_VAR$ } }
				$STORY$ = { var:$RIDER_VAR$ = flag:always }
			}
		}
		always = yes
	}
	trigger_else_if = {
		limit = { $STORY$ = { var:$RIDER_VAR$ = flag:on_hp_damage } }
		pod_combat_result_code_indicates_hp_damage_trigger = yes
	}
	trigger_else_if = {
		limit = { $STORY$ = { var:$RIDER_VAR$ = flag:on_contact } }
		pod_combat_result_code_indicates_contact_trigger = yes
	}
	trigger_else_if = {
		limit = { $STORY$ = { var:$RIDER_VAR$ = flag:on_miss } }
		pod_combat_result_code_indicates_miss_trigger = yes
	}
	# Unknown flag value — fail safely. Generator validates rider-
	# condition values against RIDER_CONDITIONS so unknown flags should
	# never reach runtime, but the closing trigger_else avoids a silent
	# pass on misconfigured data.
	trigger_else = {
		always = no
	}
}


######################################################
###### AOE TEAM-SIDE FILTERS
######################################################
# Slot-membership filters used by the AoE dispatchers in
# pod_combat_move_effects.txt. Each iterator runs in slot-story scope
# (every_in_list switches into pod_combat_slots entries) and needs to
# decide "is this slot on the same team as the actor or the opposite
# team?". Both filters read scope:host (the combat host save the
# dispatcher establishes) for the actor's slot index and compare
# against the iterated slot's slot_index.
#
# Slot layout: indices 0-4 are the PC team, 5-9 the opponent team.
# - Same-team: actor < 5 + slot < 5  OR  actor >= 5 + slot >= 5
# - Opposite-team: actor < 5 + slot >= 5  OR  actor >= 5 + slot < 5
#
# Both filters live here as a single source of truth so any future
# AoE addition can reuse the team decision without re-deriving the
# index split.
#
# Caller contract:
#   - Implicit scope must be the iterated slot story (var:slot_index
#     lives there).
#   - scope:host must be saved to the combat host (where
#     pod_combat_current_actor is read).

pod_combat_aoe_team_enemy_filter_trigger = {
	# Slot belongs to the team OPPOSITE the current actor.
	OR = {
		AND = {
			scope:host = { var:pod_combat_current_actor < 5 }
			var:slot_index >= 5
		}
		AND = {
			scope:host = { var:pod_combat_current_actor >= 5 }
			var:slot_index < 5
		}
	}
}

pod_combat_aoe_team_ally_filter_trigger = {
	# Slot belongs to the SAME team as the current actor (includes the
	# actor's own slot — team_ally heals/buffs cover self).
	OR = {
		AND = {
			scope:host = { var:pod_combat_current_actor < 5 }
			var:slot_index < 5
		}
		AND = {
			scope:host = { var:pod_combat_current_actor >= 5 }
			var:slot_index >= 5
		}
	}
}
