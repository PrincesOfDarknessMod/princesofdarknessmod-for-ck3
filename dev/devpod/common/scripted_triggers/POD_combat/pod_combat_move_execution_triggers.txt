######   POD COMBAT MOVE EXECUTION TRIGGERS

# Generic move-validation triggers: read move attributes off the
# story_cycle scope, called on the combat-host (PC) scope.
# POD_can_execute_combat_move (baked pc_can_use unlock flag + dynamic gates)
# serves every PC/UI path (execute_move SGUI is_valid); the AI runtime uses the
# dynamic-only mirror pod_combat_move_is_usable_now_trigger below. The loadout
# editor's pool (pod_combat_loadout_option) also reads the baked pc_can_use flag.
# The unlock is baked once per fight / per editor-open (running the FULL 
# dispatcher per move - pod_combat_bake_*_unlocks_effect in
# pod_combat_group_effects.txt) so the expensive switch is off the per-frame path

# once_per usage limiter: yes if $ACTOR$'s slot has already used $STORY$ within its
# once_per window (the round-list or combat-list, per the move's once_per duration).
# Match the slot by its occupant character (var:slot_char) - stable and
# scope-independent, unlike the host current-actor pointers. Shared by both gates.
# is_target_in_variable_list on a never-created list reads as "not present", so no
# list-existence guard is needed.
pod_combat_actor_has_used_move_trigger = {
	any_in_list = {
		variable = pod_combat_active_slots
		limit = { var:slot_char = { this = $ACTOR$ } }
		trigger_if = {
			limit = { $STORY$ = { var:once_per = flag:round } }
			is_target_in_variable_list = { name = slot_moves_used_round target = $STORY$ }
		}
		trigger_else = {
			is_target_in_variable_list = { name = slot_moves_used_combat target = $STORY$ }
		}
	}
}

# Gates mirrored minus the unlock dispatcher by pod_combat_move_is_usable_now_trigger (below) - keep gate lists in lockstep.
POD_can_execute_combat_move = {
	# $STORY$ = the POD_combat_move story_cycle
	# $ACTOR$ = character whose unlock gate (splat/trait/perk) is tested.
	#          On AI turns this differs from the host (PC); workspace
	#          reads (AP, energy) still resolve against root = the host.

	NOT = { pod_combat_is_over_trigger = yes }

	# Unlock gate: read the per-fight baked pc_can_use flag instead of
	# re-running the unlock dispatcher every GUI frame (the documented
	# PC-turn hotspot; unlocks are static mid-combat).
	# ($ACTOR$ is still consumed by the energy gate below.)
	$STORY$ = { has_variable = pc_can_use }

	trigger_if = {
		limit = {
			$STORY$ = { has_variable = ap_cost }
		}
		pod_combat_can_afford_ap_trigger = { COST = $STORY$.var:ap_cost }
	}

	# Energy: the splat-agnostic reserve check resolves against the scope
	# it runs on, so wrap in $ACTOR$ to hit the acting character's pool
	# (bare it would read the host/PC pool on AI turns).
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = energy_cost }
		}
		$ACTOR$ = {
			POD_has_enough_splat_agnostic_energy_reserves = { AMOUNT = $STORY$.var:energy_cost }
		}
	}

	# CK3 resource costs are PC-only — AI combatants spend nothing, so
	# gold/piety/prestige gates apply only on the player turn.
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = gold_cost }
			pod_combat_is_player_turn_trigger = yes
		}
		gold >= $STORY$.var:gold_cost
	}

	trigger_if = {
		limit = {
			$STORY$ = { has_variable = piety_cost }
			pod_combat_is_player_turn_trigger = yes
		}
		piety >= $STORY$.var:piety_cost
	}

	trigger_if = {
		limit = {
			$STORY$ = { has_variable = prestige_cost }
			pod_combat_is_player_turn_trigger = yes
		}
		prestige >= $STORY$.var:prestige_cost
	}

	# State-condition gate (cond_mode = gate): usable only when the condition
	# holds. Dynamic, both PC and AI.
	trigger_if = {
		limit = { $STORY$ = { has_variable = cond_mode  var:cond_mode = flag:gate } }
		pod_combat_move_condition_met_trigger = { STORY = $STORY$ ACTOR = $ACTOR$ }
	}

	# once_per usage limiter: block re-use within the move's round / combat
	# window. 
	trigger_if = {
		limit = { $STORY$ = { has_variable = once_per } }
		NOT = { pod_combat_actor_has_used_move_trigger = { STORY = $STORY$  ACTOR = $ACTOR$ } }
	}

	# stress_cost adds stress on execution — it's always "affordable"

	# Duplicate-stance / reaction guard — reject re-applying a stance the
	# actor already holds (mirror: pod_combat_ai_effects.txt,
	# pod_combat_ai_select_and_execute_move_effect's NAND block). Without
	# it the bonus double-bakes and needs two separate consumes to revert.
	NAND = {
		$STORY$ = { has_variable = set_guard }
		has_variable = pod_combat_actor_is_guarding
	}
	NAND = {
		$STORY$ = { has_variable = dodge }
		has_variable = pod_combat_actor_is_dodging
	}
	NAND = {
		$STORY$ = { has_variable = stealth }
		has_variable = pod_combat_actor_is_stealthed
	}
	NAND = {
		$STORY$ = { has_variable = retaliate_if_hit }
		has_variable = pod_combat_actor_retaliate_hit
	}
	NAND = {
		$STORY$ = { has_variable = retaliate_if_missed }
		has_variable = pod_combat_actor_retaliate_miss
	}

	# Tier-redundancy guard — reject a tier buff/debuff at or below the
	# current tier (upgrade-only contract; see the §11 rationale in
	# pod_combat_move_effects.txt). Strict `<` matches the AI guard.
	# Gate mechanics: pod_combat_tier_upgrade_only_trigger (below).
	# Covers workspace single-target tiers only; AoE downgrades are
	# guarded at write time in pod_combat_apply_team_ally_buffs /
	# _team_enemy_debuffs.
	pod_combat_tier_upgrade_only_trigger = { STORY = $STORY$ SIDE = actor TIER_VAR = soak_buff_round_tier }
	pod_combat_tier_upgrade_only_trigger = { STORY = $STORY$ SIDE = actor TIER_VAR = soak_buff_combat_tier }
	pod_combat_tier_upgrade_only_trigger = { STORY = $STORY$ SIDE = actor TIER_VAR = hit_buff_round_tier }
	pod_combat_tier_upgrade_only_trigger = { STORY = $STORY$ SIDE = actor TIER_VAR = hit_buff_combat_tier }
	pod_combat_tier_upgrade_only_trigger = { STORY = $STORY$ SIDE = target TIER_VAR = soak_debuff_round_tier }
	pod_combat_tier_upgrade_only_trigger = { STORY = $STORY$ SIDE = target TIER_VAR = soak_debuff_combat_tier }
	pod_combat_tier_upgrade_only_trigger = { STORY = $STORY$ SIDE = target TIER_VAR = hit_debuff_round_tier }
	pod_combat_tier_upgrade_only_trigger = { STORY = $STORY$ SIDE = target TIER_VAR = hit_debuff_combat_tier }

	# Heal-utility gate — block self-targeted heal-only moves (no damage
	# rider) when the actor is already at the cap the heal can't exceed
	# (mirrors the AI selector's gate in pod_combat_ai_effects.txt).
	# Scoped to targeting = flag:self (and untargeted heals, which the
	# generator defaults to actor-self). single_ally / team_ally heals
	# stay clickable — the actor's HP isn't the target, and the player
	# may intentionally heal a wounded ally while full.
	trigger_if = {
		limit = {
			$STORY$ = {
				OR = {
					has_variable = hp_heal
					has_variable = morale_heal
				}
				NOT = { has_variable = hp_damage }
				NOT = { has_variable = morale_damage }
				OR = {
					NOT = { has_variable = targeting }
					var:targeting = flag:self
				}
			}
		}
		custom_description = {
			text = pod_combat_cant_heal_already_full_desc
			OR = {
				# Superficial heals stop at sup_ceiling (= max_hp −
				# agg_boxes): aggravated wounds pin the track top, so
				# this can be a no-op even while hp < max_hp.
				AND = {
					$STORY$ = {
						has_variable = hp_heal
						var:heal_type = flag:superficial
					}
					pod_combat_actor_hp_value < pod_combat_actor_sup_ceiling_value
				}
				# Aggravated heals any wound (aggravated first), capped at max_hp.
				AND = {
					$STORY$ = {
						has_variable = hp_heal
						var:heal_type = flag:aggravated
					}
					pod_combat_actor_hp_value < pod_combat_actor_max_hp_value
				}
				AND = {
					$STORY$ = { has_variable = morale_heal }
					var:pod_combat_actor_morale < pod_combat_max_morale
				}
			}
		}
	}

	# Targeting validity, player turn only — the PC picked a slot via
	# pod_combat_pc_target_*. AI eval skips this (the AI swap effects
	# choose the target after move choice).
	trigger_if = {
		limit = {
			pod_combat_is_player_turn_trigger = yes
			$STORY$ = { var:targeting = flag:single_enemy }
		}
		# custom_description swaps the disabled-button tooltip to "Select
		# an enemy target" instead of raw trigger text.
		# save_temporary_scope_as captures the host before any_in_list
		# flips scope into per-slot iteration, so pod_combat_pc_target_*
		# stays readable in the loop body. Same pattern as
		# pod_combat_actor_has_alive_team_trigger in pod_combat_triggers.txt.
		save_temporary_scope_as = host
		custom_description = {
			text = pod_combat_targeting_need_enemy_tt
			has_variable = pod_combat_pc_target_team
			any_in_list = {
				variable = pod_combat_active_slots
				var:combat_team = scope:host.var:pod_combat_pc_target_team
				var:team_position = scope:host.var:pod_combat_pc_target_position
				pod_combat_slot_is_enemy_trigger = yes
				pod_combat_slot_is_alive_trigger = yes
			}
		}
	}

	trigger_if = {
		limit = {
			pod_combat_is_player_turn_trigger = yes
			$STORY$ = { var:targeting = flag:single_ally }
		}
		# Mirror of the single_enemy branch above; team_position >= 1
		# narrows to ally slots (player team minus the PC).
		save_temporary_scope_as = host
		custom_description = {
			text = pod_combat_targeting_need_ally_tt
			has_variable = pod_combat_pc_target_team
			any_in_list = {
				variable = pod_combat_active_slots
				var:combat_team = scope:host.var:pod_combat_pc_target_team
				var:team_position = scope:host.var:pod_combat_pc_target_position
				pod_combat_slot_is_player_team_trigger = yes
				var:team_position >= 1
				pod_combat_slot_is_alive_trigger = yes
			}
		}
	}

	trigger_if = {
		limit = {
			pod_combat_is_player_turn_trigger = yes
			$STORY$ = { var:targeting = flag:single_pc }
		}
		# Protect-the-PC moves target player/0 — direct pointer check
		# rather than any_in_list, since it's a single-slot probe.
		custom_description = {
			text = pod_combat_targeting_need_pc_alive_tt
			has_variable = pod_combat_pc_slot
			var:pod_combat_pc_slot = { has_variable = slot_char pod_combat_slot_is_alive_trigger = yes }
		}
	}

	trigger_if = {
		limit = {
			pod_combat_is_player_turn_trigger = yes
			$STORY$ = { var:targeting = flag:team_enemy }
		}
		# At least one alive enemy slot. No pc_target dependency → no
		# scope save (all filters are on the slot story itself).
		any_in_list = {
			variable = pod_combat_active_slots
			pod_combat_slot_is_enemy_trigger = yes
			pod_combat_slot_is_alive_trigger = yes
		}
	}

	# AI-side single_ally gate: only offer a single_ally move when an
	# ally exists to target. Without it the ally-swap finds no match,
	# the workspace stays on the enemy, and the heal/buff lands on the PC.
	trigger_if = {
		limit = {
			NOT = { pod_combat_is_player_turn_trigger = yes }
			$STORY$ = { var:targeting = flag:single_ally }
		}
		pod_combat_ai_has_available_ally_trigger = yes
	}

	# AI-side single_ally HEAL need-gate: beyond "an ally exists", only
	# pick a single_ally heal if some ally actually needs it (the
	# ally-swap picks flat among living allies). Checked against ally
	# state here rather than the actor-keyed redundancy gate in
	# pod_combat_ai_effects.txt, since the actor's own state is
	# irrelevant to an ally-targeted heal. Heal-bearing moves only.
	trigger_if = {
		limit = {
			NOT = { pod_combat_is_player_turn_trigger = yes }
			$STORY$ = {
				var:targeting = flag:single_ally
				OR = {
					has_variable = hp_heal
					has_variable = morale_heal
				}
			}
		}
		pod_combat_ai_has_ally_needing_heal_trigger = { STORY = $STORY$ }
	}

	# C26: summon-capacity gate. A summons_combatant move can't seat a new combatant
	# once the caster's side is at pod_combat_max_team_size (the glue pre-checks and
	# no-ops, so without this the button looks live but the click wastes AP/energy).
	# current_actor_team = the caster's side on either turn; has_variable-guarded since
	# the combat window can re-evaluate is_valid between rounds when it is briefly unset.
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = summons_combatant }
			has_variable = pod_combat_current_actor_team
		}
		custom_description = {
			text = pod_combat_cant_summon_team_full_tt
			OR = {
				AND = {
					var:pod_combat_current_actor_team = flag:player
					var:pod_combat_player_team_size < pod_combat_max_team_size
				}
				AND = {
					var:pod_combat_current_actor_team = flag:enemy
					var:pod_combat_enemy_team_size < pod_combat_max_team_size
				}
			}
		}
	}

	# team_ally and self need no gate (both always include the live
	# actor). single_enemy is vacuous for the AI too: the victory check
	# ends combat when no enemies remain, so selection never runs
	# without a valid enemy.
}

# Mirror of POD_can_execute_combat_move minus the unlock dispatcher re-proof
# (pod_combat_actor_unlocks_move_trigger) - keep gate lists in lockstep.
# The AI selector limit and its post-pick recheck call this instead of the
# full validator: the AI picks from pod_combat_loadout, and membership
# already proves the unlock (POD_apply_ai_full_loadout filters the catalogue
# through the same dispatcher once, at fight finalize / mid-fight seat;
# unlocks are static trait/perk facts and the combat window pauses the sim).
# Skipping the per-candidate dispatcher walk is the dominant AI-turn perf win. 
# Gate rationale comments live on the canonical copies above; only deltas are commented.
pod_combat_move_is_usable_now_trigger = {
	# $STORY$ / $ACTOR$: same contract as POD_can_execute_combat_move.

	# AP gate hoisted first (cheapest): AP-exhausted tail tries fail every
	# candidate right here instead of after the pricier gates.
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = ap_cost }
		}
		pod_combat_can_afford_ap_trigger = { COST = $STORY$.var:ap_cost }
	}

	trigger_if = {
		limit = {
			$STORY$ = { has_variable = energy_cost }
		}
		$ACTOR$ = {
			POD_has_enough_splat_agnostic_energy_reserves = { AMOUNT = $STORY$.var:energy_cost }
		}
	}

	NOT = { pod_combat_is_over_trigger = yes }

	# PC-only CK3 resource costs.
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = gold_cost }
			pod_combat_is_player_turn_trigger = yes
		}
		gold >= $STORY$.var:gold_cost
	}

	trigger_if = {
		limit = {
			$STORY$ = { has_variable = piety_cost }
			pod_combat_is_player_turn_trigger = yes
		}
		piety >= $STORY$.var:piety_cost
	}

	trigger_if = {
		limit = {
			$STORY$ = { has_variable = prestige_cost }
			pod_combat_is_player_turn_trigger = yes
		}
		prestige >= $STORY$.var:prestige_cost
	}

	# State-condition gate (cond_mode = gate): usable only when the condition
	# holds. Dynamic, both PC and AI. 
	trigger_if = {
		limit = { $STORY$ = { has_variable = cond_mode  var:cond_mode = flag:gate } }
		pod_combat_move_condition_met_trigger = { STORY = $STORY$ ACTOR = $ACTOR$ }
	}

	# once_per usage limiter (lockstep with POD_can_execute_combat_move).
	trigger_if = {
		limit = { $STORY$ = { has_variable = once_per } }
		NOT = { pod_combat_actor_has_used_move_trigger = { STORY = $STORY$  ACTOR = $ACTOR$ } }
	}

	# Duplicate-stance / reaction guard.
	NAND = {
		$STORY$ = { has_variable = set_guard }
		has_variable = pod_combat_actor_is_guarding
	}
	NAND = {
		$STORY$ = { has_variable = dodge }
		has_variable = pod_combat_actor_is_dodging
	}
	NAND = {
		$STORY$ = { has_variable = stealth }
		has_variable = pod_combat_actor_is_stealthed
	}
	NAND = {
		$STORY$ = { has_variable = retaliate_if_hit }
		has_variable = pod_combat_actor_retaliate_hit
	}
	NAND = {
		$STORY$ = { has_variable = retaliate_if_missed }
		has_variable = pod_combat_actor_retaliate_miss
	}

	# Tier-redundancy guard (upgrade-only contract).
	pod_combat_tier_upgrade_only_trigger = { STORY = $STORY$ SIDE = actor TIER_VAR = soak_buff_round_tier }
	pod_combat_tier_upgrade_only_trigger = { STORY = $STORY$ SIDE = actor TIER_VAR = soak_buff_combat_tier }
	pod_combat_tier_upgrade_only_trigger = { STORY = $STORY$ SIDE = actor TIER_VAR = hit_buff_round_tier }
	pod_combat_tier_upgrade_only_trigger = { STORY = $STORY$ SIDE = actor TIER_VAR = hit_buff_combat_tier }
	pod_combat_tier_upgrade_only_trigger = { STORY = $STORY$ SIDE = target TIER_VAR = soak_debuff_round_tier }
	pod_combat_tier_upgrade_only_trigger = { STORY = $STORY$ SIDE = target TIER_VAR = soak_debuff_combat_tier }
	pod_combat_tier_upgrade_only_trigger = { STORY = $STORY$ SIDE = target TIER_VAR = hit_debuff_round_tier }
	pod_combat_tier_upgrade_only_trigger = { STORY = $STORY$ SIDE = target TIER_VAR = hit_debuff_combat_tier }

	# Heal-utility gate (self-targeted heal-only moves).
	trigger_if = {
		limit = {
			$STORY$ = {
				OR = {
					has_variable = hp_heal
					has_variable = morale_heal
				}
				NOT = { has_variable = hp_damage }
				NOT = { has_variable = morale_damage }
				OR = {
					NOT = { has_variable = targeting }
					var:targeting = flag:self
				}
			}
		}
		custom_description = {
			text = pod_combat_cant_heal_already_full_desc
			OR = {
				AND = {
					$STORY$ = {
						has_variable = hp_heal
						var:heal_type = flag:superficial
					}
					pod_combat_actor_hp_value < pod_combat_actor_sup_ceiling_value
				}
				AND = {
					$STORY$ = {
						has_variable = hp_heal
						var:heal_type = flag:aggravated
					}
					pod_combat_actor_hp_value < pod_combat_actor_max_hp_value
				}
				AND = {
					$STORY$ = { has_variable = morale_heal }
					var:pod_combat_actor_morale < pod_combat_max_morale
				}
			}
		}
	}

	# Targeting validity, player turn only.
	trigger_if = {
		limit = {
			pod_combat_is_player_turn_trigger = yes
			$STORY$ = { var:targeting = flag:single_enemy }
		}
		save_temporary_scope_as = host
		custom_description = {
			text = pod_combat_targeting_need_enemy_tt
			has_variable = pod_combat_pc_target_team
			any_in_list = {
				variable = pod_combat_active_slots
				var:combat_team = scope:host.var:pod_combat_pc_target_team
				var:team_position = scope:host.var:pod_combat_pc_target_position
				pod_combat_slot_is_enemy_trigger = yes
				pod_combat_slot_is_alive_trigger = yes
			}
		}
	}

	trigger_if = {
		limit = {
			pod_combat_is_player_turn_trigger = yes
			$STORY$ = { var:targeting = flag:single_ally }
		}
		save_temporary_scope_as = host
		custom_description = {
			text = pod_combat_targeting_need_ally_tt
			has_variable = pod_combat_pc_target_team
			any_in_list = {
				variable = pod_combat_active_slots
				var:combat_team = scope:host.var:pod_combat_pc_target_team
				var:team_position = scope:host.var:pod_combat_pc_target_position
				pod_combat_slot_is_player_team_trigger = yes
				var:team_position >= 1
				pod_combat_slot_is_alive_trigger = yes
			}
		}
	}

	trigger_if = {
		limit = {
			pod_combat_is_player_turn_trigger = yes
			$STORY$ = { var:targeting = flag:single_pc }
		}
		custom_description = {
			text = pod_combat_targeting_need_pc_alive_tt
			has_variable = pod_combat_pc_slot
			var:pod_combat_pc_slot = { has_variable = slot_char pod_combat_slot_is_alive_trigger = yes }
		}
	}

	trigger_if = {
		limit = {
			pod_combat_is_player_turn_trigger = yes
			$STORY$ = { var:targeting = flag:team_enemy }
		}
		any_in_list = {
			variable = pod_combat_active_slots
			pod_combat_slot_is_enemy_trigger = yes
			pod_combat_slot_is_alive_trigger = yes
		}
	}

	# AI-side single_ally availability + heal-need gates.
	trigger_if = {
		limit = {
			NOT = { pod_combat_is_player_turn_trigger = yes }
			$STORY$ = { var:targeting = flag:single_ally }
		}
		pod_combat_ai_has_available_ally_trigger = yes
	}

	trigger_if = {
		limit = {
			NOT = { pod_combat_is_player_turn_trigger = yes }
			$STORY$ = {
				var:targeting = flag:single_ally
				OR = {
					has_variable = hp_heal
					has_variable = morale_heal
				}
			}
		}
		pod_combat_ai_has_ally_needing_heal_trigger = { STORY = $STORY$ }
	}

	# C26: summon-capacity gate (mirror of POD_can_execute_combat_move; plain trigger,
	# no PC tooltip). Don't let the AI pick a summon when its side is full - the glue
	# would no-op and waste the turn.
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = summons_combatant }
			has_variable = pod_combat_current_actor_team
		}
		OR = {
			AND = {
				var:pod_combat_current_actor_team = flag:player
				var:pod_combat_player_team_size < pod_combat_max_team_size
			}
			AND = {
				var:pod_combat_current_actor_team = flag:enemy
				var:pod_combat_enemy_team_size < pod_combat_max_team_size
			}
		}
	}
}


######################################################
###### TIER UPGRADE-ONLY GUARD (parametric)
######################################################

# Passes unless the move's $TIER_VAR$ is at or below the live workspace tier
# (upgrade-only contract; rationale at section 11 of pod_combat_move_effects.txt).
# Single trigger_if (not NAND) so the var reads only happen when both the move
# attribute and the workspace var exist, avoiding "variable not set" warnings.
pod_combat_tier_upgrade_only_trigger = {
	# $STORY$ = move story; $SIDE$ = actor|target; $TIER_VAR$ = tier var name.
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = $TIER_VAR$ }
			has_variable = pod_combat_$SIDE$_$TIER_VAR$
		}
		var:pod_combat_$SIDE$_$TIER_VAR$ < $STORY$.var:$TIER_VAR$
	}
}


######################################################
###### RESULT-CODE CONDITION TRIGGERS
######################################################
# Primitive partitions of attack_result by outcome class, read off the
# host scope. The rider-condition and retaliation gates compose against
# these so the mapping lives in one place (canonical map:
# COMBAT_REFERENCE.info §4):
#   "hp_damage" — attack_result=hp_damage
#   "contact"   — attack_result ∈ (hp_damage, guard_blocked, soaked)
#   "miss"      — attack_result ∈ (missed, dodged, stealthed)
# Each gates on outcome=attack first — attack_result is only meaningful
# then, so non-attack outcomes fail every partition by design.

# The attack landed real wound boxes on the target.
# Single-target path reads the live host last_* pair (the AND branch). AoE drains
# loop enemies and the per-target push helper clears that pair, so the team_enemy
# damage loop instead latches pod_combat_aoe_any_hp_damage when any target took HP
# damage; the OR lets the §9 on_hp_damage drain rider fire once after the loop. A
# fully soaked/missed AoE latches nothing, so the drain correctly does not heal.
pod_combat_result_code_indicates_hp_damage_trigger = {
	OR = {
		AND = {
			has_variable = pod_combat_last_outcome
			var:pod_combat_last_outcome = flag:attack
			has_variable = pod_combat_last_attack_result
			var:pod_combat_last_attack_result = flag:hp_damage
		}
		has_variable = pod_combat_aoe_any_hp_damage
	}
}

# Physical contact: hp_damage plus blocked / soaked. For "fires on
# touch even if absorbed" riders (e.g. poison delivered on the strike).
pod_combat_result_code_indicates_contact_trigger = {
	has_variable = pod_combat_last_outcome
	var:pod_combat_last_outcome = flag:attack
	has_variable = pod_combat_last_attack_result
	OR = {
		var:pod_combat_last_attack_result = flag:hp_damage
		var:pod_combat_last_attack_result = flag:guard_blocked
		var:pod_combat_last_attack_result = flag:soaked
	}
}

# Evaded the attack — miss, dodge consumed, stealth auto-miss. Used by
# retaliate_if_missed and rider_condition = on_miss.
pod_combat_result_code_indicates_miss_trigger = {
	has_variable = pod_combat_last_outcome
	var:pod_combat_last_outcome = flag:attack
	has_variable = pod_combat_last_attack_result
	OR = {
		var:pod_combat_last_attack_result = flag:missed
		var:pod_combat_last_attack_result = flag:dodged
		var:pod_combat_last_attack_result = flag:stealthed
	}
}

# Miss partition plus guard_blocked: every zero-damage outcome. Gates
# the combat-log damage-source icon (no elemental marker where nothing
# landed).
#
# Excludes `soaked` deliberately: soak doesn't always zero damage, and
# even on full-soak rows we keep the source flavour (something WAS
# rolled and survived as residue / aura).
#
# Parametric on $RESULT_VAR$ — two call sites read the enum off
# different vars:
#   - host scope (move execution)  — RESULT_VAR = pod_combat_last_attack_result
#   - log-entry scope (post-action) — RESULT_VAR = attack_result
pod_combat_result_indicates_no_damage_dealt_trigger = {
	has_variable = $RESULT_VAR$
	OR = {
		var:$RESULT_VAR$ = flag:missed
		var:$RESULT_VAR$ = flag:dodged
		var:$RESULT_VAR$ = flag:stealthed
		var:$RESULT_VAR$ = flag:guard_blocked
	}
}


######################################################
###### RIDER CONDITION GATE
######################################################
# Generic gate for rider effects (heal / morale damage / self-damage)
# that piggyback on a move's HP-damage step — e.g. a bite that heals on
# a landed hit. Each rider's condition var on the move story holds one
# of: always (default when unset), on_hp_damage, on_contact, on_miss —
# resolved against the partition triggers above. Caller scope = host.
#
# Drain moves: the generator emits heal_condition = on_hp_damage for any
# `drain`-token move, so "heal only on a landed hit" flows through this
# rider machinery rather than a special case.
pod_combat_rider_condition_passes_trigger = {
	# $STORY$     = the POD_combat_move story_cycle
	# $RIDER_VAR$ = the condition variable name on the story (e.g. heal_condition)
	# Default: var unset OR var = always → pass.
	trigger_if = {
		limit = {
			OR = {
				NOT = { $STORY$ = { has_variable = $RIDER_VAR$ } }
				$STORY$ = { var:$RIDER_VAR$ = flag:always }
			}
		}
		always = yes
	}
	trigger_else_if = {
		limit = { $STORY$ = { var:$RIDER_VAR$ = flag:on_hp_damage } }
		pod_combat_result_code_indicates_hp_damage_trigger = yes
	}
	trigger_else_if = {
		limit = { $STORY$ = { var:$RIDER_VAR$ = flag:on_contact } }
		pod_combat_result_code_indicates_contact_trigger = yes
	}
	trigger_else_if = {
		limit = { $STORY$ = { var:$RIDER_VAR$ = flag:on_miss } }
		pod_combat_result_code_indicates_miss_trigger = yes
	}
	# Unknown flag value — fail closed. The generator validates against
	# RIDER_CONDITIONS, so this guards only misconfigured data.
	trigger_else = {
		always = no
	}
}

######################################################
###### STATE CONDITION (gate / bonus)
######################################################
# $STORY$ = the move; $ACTOR$ = the acting
# character (the "user"). Evaluated in HOST scope: round / morale / stance read
# host workspace vars; the user is $ACTOR$, the target is var:pod_combat_target_char
# (morale via the pod_combat_*_morale mirrors). cond_target routes single-subject
# predicates to user vs target; a target predicate with no loaded target fails.
# Phase 1 predicates (combat state) only - character / relational / outcome cases
# are added later (this is built to absorb the outcome-rider predicates so the two
# condition axes can merge). Guarded for unset workspace vars (tooltip walks).
pod_combat_move_condition_met_trigger = {
	switch = {
		trigger = $STORY$.var:cond_predicate

		# global (host round)
		flag:first_round = {
			trigger_if = { limit = { has_variable = pod_combat_round } var:pod_combat_round = 1 }
			trigger_else = { always = no }
		}
		flag:round_min = {
			trigger_if = { limit = { has_variable = pod_combat_round } var:pod_combat_round >= $STORY$.var:cond_threshold }
			trigger_else = { always = no }
		}

		# HP, live from the character's PoD wound track
		flag:undamaged = {
			trigger_if = {
				limit = { $STORY$ = { var:cond_target = flag:target }  has_variable = pod_combat_target_char }
				var:pod_combat_target_char = { POD_empty_damage_boxes = POD_health_track_length }
			}
			trigger_else_if = { limit = { $STORY$ = { var:cond_target = flag:target } } always = no }
			trigger_else = { $ACTOR$ = { POD_empty_damage_boxes = POD_health_track_length } }
		}
		flag:hp_missing = {
			trigger_if = {
				limit = { $STORY$ = { var:cond_target = flag:target }  has_variable = pod_combat_target_char }
				var:pod_combat_target_char = { POD_empty_damage_boxes <= { value = POD_health_track_length  subtract = $STORY$.var:cond_threshold } }
			}
			trigger_else_if = { limit = { $STORY$ = { var:cond_target = flag:target } } always = no }
			trigger_else = { $ACTOR$ = { POD_empty_damage_boxes <= { value = POD_health_track_length  subtract = $STORY$.var:cond_threshold } } }
		}

		# morale, via the host workspace mirror
		flag:full_morale = {
			trigger_if = {
				limit = { $STORY$ = { var:cond_target = flag:target }  has_variable = pod_combat_target_morale }
				var:pod_combat_target_morale = pod_combat_max_morale
			}
			trigger_else_if = { limit = { $STORY$ = { var:cond_target = flag:target } } always = no }
			trigger_else_if = { limit = { has_variable = pod_combat_actor_morale } var:pod_combat_actor_morale = pod_combat_max_morale }
			trigger_else = { always = no }
		}
		flag:morale_missing = {
			trigger_if = {
				limit = { $STORY$ = { var:cond_target = flag:target }  has_variable = pod_combat_target_morale }
				var:pod_combat_target_morale <= { value = pod_combat_max_morale  subtract = $STORY$.var:cond_threshold }
			}
			trigger_else_if = { limit = { $STORY$ = { var:cond_target = flag:target } } always = no }
			trigger_else_if = { limit = { has_variable = pod_combat_actor_morale } var:pod_combat_actor_morale <= { value = pod_combat_max_morale  subtract = $STORY$.var:cond_threshold } }
			trigger_else = { always = no }
		}

		# stance flags (has_variable is unset-safe)
		flag:has_guard = {
			trigger_if = { limit = { $STORY$ = { var:cond_target = flag:target } } has_variable = pod_combat_target_is_guarding }
			trigger_else = { has_variable = pod_combat_actor_is_guarding }
		}
		flag:has_dodge = {
			trigger_if = { limit = { $STORY$ = { var:cond_target = flag:target } } has_variable = pod_combat_target_is_dodging }
			trigger_else = { has_variable = pod_combat_actor_is_dodging }
		}
		flag:has_stealth = {
			trigger_if = { limit = { $STORY$ = { var:cond_target = flag:target } } has_variable = pod_combat_target_is_stealthed }
			trigger_else = { has_variable = pod_combat_actor_is_stealthed }
		}

		# Character status (vanilla character triggers; scope-switch to the char by
		# cond_target, same routing as the HP case above). cond_threshold is the
		# minimum: prestige_level / piety_level are 0-5 ranks, dread is 0-100.
		flag:prestige_level_min = {
			trigger_if = {
				limit = { $STORY$ = { var:cond_target = flag:target }  has_variable = pod_combat_target_char }
				var:pod_combat_target_char = { prestige_level >= $STORY$.var:cond_threshold }
			}
			trigger_else_if = { limit = { $STORY$ = { var:cond_target = flag:target } } always = no }
			trigger_else = { $ACTOR$ = { prestige_level >= $STORY$.var:cond_threshold } }
		}
		flag:piety_level_min = {
			trigger_if = {
				limit = { $STORY$ = { var:cond_target = flag:target }  has_variable = pod_combat_target_char }
				var:pod_combat_target_char = { piety_level >= $STORY$.var:cond_threshold }
			}
			trigger_else_if = { limit = { $STORY$ = { var:cond_target = flag:target } } always = no }
			trigger_else = { $ACTOR$ = { piety_level >= $STORY$.var:cond_threshold } }
		}
		flag:dread_min = {
			trigger_if = {
				limit = { $STORY$ = { var:cond_target = flag:target }  has_variable = pod_combat_target_char }
				var:pod_combat_target_char = { dread >= $STORY$.var:cond_threshold }
			}
			trigger_else_if = { limit = { $STORY$ = { var:cond_target = flag:target } } always = no }
			trigger_else = { $ACTOR$ = { dread >= $STORY$.var:cond_threshold } }
		}

		# Relational (opinion): the USER ($ACTOR$) judges the TARGET. cond_target is
		# flag:relational. The target is reached via root.var (root = host on entry,
		# the §0 invariant) so it survives the $ACTOR$ scope-switch. cond_threshold is
		# the opinion magnitude; the negative side flips the sign in the compare.
		flag:relation_positive = {
			trigger_if = {
				limit = { has_variable = pod_combat_target_char }
				$ACTOR$ = { opinion = { target = root.var:pod_combat_target_char  value >= $STORY$.var:cond_threshold } }
			}
			trigger_else = { always = no }
		}
		flag:relation_negative = {
			trigger_if = {
				limit = { has_variable = pod_combat_target_char }
				$ACTOR$ = { opinion = { target = root.var:pod_combat_target_char  value <= { value = $STORY$.var:cond_threshold  multiply = -1 } } }
			}
			trigger_else = { always = no }
		}

		# Equipped artifact by slot type (any_character_artifact iterates the char's
		# artifacts; artifact_slot_type is the equipment slot - primary_armament = the
		# weapon slot, armor = the armor slot). Boolean, so cond_threshold is unused.
		flag:has_weapon_artifact = {
			trigger_if = {
				limit = { $STORY$ = { var:cond_target = flag:target }  has_variable = pod_combat_target_char }
				var:pod_combat_target_char = { any_character_artifact = { is_equipped = yes  artifact_slot_type = primary_armament } }
			}
			trigger_else_if = { limit = { $STORY$ = { var:cond_target = flag:target } } always = no }
			trigger_else = { $ACTOR$ = { any_character_artifact = { is_equipped = yes  artifact_slot_type = primary_armament } } }
		}
		flag:has_armor_artifact = {
			trigger_if = {
				limit = { $STORY$ = { var:cond_target = flag:target }  has_variable = pod_combat_target_char }
				var:pod_combat_target_char = { any_character_artifact = { is_equipped = yes  artifact_slot_type = armor } }
			}
			trigger_else_if = { limit = { $STORY$ = { var:cond_target = flag:target } } always = no }
			trigger_else = { $ACTOR$ = { any_character_artifact = { is_equipped = yes  artifact_slot_type = armor } } }
		}

		# Identifier predicates: route to the char (has_trait) / run in $ACTOR$
		# (target_known_secret), then a sub-dispatch switches on cond_param
		# to the literal trait / secret_type (pod_combat_condition_dispatch_triggers.txt).
		flag:has_trait = {
			trigger_if = {
				limit = { $STORY$ = { var:cond_target = flag:target }  has_variable = pod_combat_target_char }
				var:pod_combat_target_char = { pod_combat_cond_has_trait_trigger = { STORY = $STORY$ } }
			}
			trigger_else_if = { limit = { $STORY$ = { var:cond_target = flag:target } } always = no }
			trigger_else = { $ACTOR$ = { pod_combat_cond_has_trait_trigger = { STORY = $STORY$ } } }
		}
		flag:target_known_secret = {
			trigger_if = {
				limit = { has_variable = pod_combat_target_char }
				$ACTOR$ = { pod_combat_cond_known_secret_trigger = { STORY = $STORY$ } }
			}
			trigger_else = { always = no }
		}

		# Unknown predicate - fail closed (generator validates cond_predicate).
		fallback = { always = no }
	}
}

###### COMBAT-LOG SEGMENT BURST GATE

# Move-level log segments render once per move, not once per AoE target
# row: pass on a burst's first row (index 0) or on non-AoE rows, which
# never get an aoe_burst_index. Caller scope = the POD_combat_log_entry
# story; used by the segment-visibility bake in pod_combat_push_log_effect
# (pod_combat_effects.txt).
pod_combat_log_seg_burst_gate_trigger = {
	OR = {
		NOT = { has_variable = aoe_burst_index }
		var:aoe_burst_index = 0
	}
}

# AI move-selection helper: true iff the acting AI's loadout holds a strictly-
# larger-tier move of the SAME stat var ($STAT_VAR$) than the candidate, whose
# AP cost still fits the AP it has left this action. Used by the undersized-buff
# downgrade in pod_combat_ai_effects.txt to steer the AI off a smaller buff when
# it could afford the bigger one of the same kind. scope:ai_actor (loadout),
# scope:combat_host (live AP), and scope:ai_candidate_move (the scored move) are
# all saved by the AI selector. AP affordability only: buffs are light on energy,
# and a per-move energy check would need an element capture inside the any_in_list
# that the trigger context can't do cleanly; the bigger move's own energy gate
# still applies when it is actually scored, so at worst the smaller buff keeps a
# slightly lower weight (never unpickable - the weight floors at @ai_weight_floor).
pod_combat_ai_has_affordable_larger_buff_tier_trigger = {
	scope:ai_actor = {
		any_in_list = {
			variable = pod_combat_loadout
			has_variable = $STAT_VAR$
			scope:ai_candidate_move = { has_variable = $STAT_VAR$ }
			# Guard BOTH vars the compare reads. A var: comparison evaluates even when a
			# preceding has_variable is false (CK3 no-short-circuit), so the trigger_if
			# limit must cover the element's $STAT_VAR$ (LHS) AND the candidate's (RHS).
			# The candidate guard is load-bearing: the calling modifier's
			# `AND = { has_variable=X  <this trigger STAT_VAR=X> }` branches also do NOT
			# short-circuit, so this trigger runs for stats the candidate lacks; without
			# the candidate guard the RHS errored (Invalid right side during comparison
			# 'var'). The two hard has_variable gates above keep the match semantics.
			trigger_if = {
				limit = {
					has_variable = $STAT_VAR$
					scope:ai_candidate_move = { has_variable = $STAT_VAR$ }
				}
				var:$STAT_VAR$ > scope:ai_candidate_move.var:$STAT_VAR$
				trigger_if = {
					limit = { has_variable = ap_cost }
					scope:combat_host.var:pod_combat_current_ap >= var:ap_cost
				}
			}
		}
	}
}
