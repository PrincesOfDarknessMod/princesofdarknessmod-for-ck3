######################################################
######   POD COMBAT MOVE EXECUTION TRIGGERS
######################################################
# Host-scoped move validation; attributes live on the move story.
# Player/UI paths use unlock membership; AI loadout proves unlocks.
# Both paths share pod_combat_move_dynamic_gates_trigger.

# $ACTOR$ = character; $STORY$ = move. Match usage by slot occupant.
# An absent list reads as unused.
pod_combat_actor_has_used_move_trigger = {
	var:pod_combat_session ?= {
		any_in_list = {
			variable = pod_combat_active_slots
			var:slot_char = { this = $ACTOR$ }
			trigger_if = {
				limit = { $STORY$ = { var:once_per = flag:round } }
				is_target_in_variable_list = { name = slot_moves_used_round target = $STORY$ }
			}
			trigger_else = {
				is_target_in_variable_list = { name = slot_moves_used_combat target = $STORY$ }
			}
		}
	}
}


######################################################
###### SHARED USABILITY TRIGGERS
######################################################
# Host-scoped scripted triggers shared by live gates and failure tooltips.
# LOCKSTEP: each new player-facing block needs a scripted trigger, a usability check and a
# pod_combat_move_usability_tt line in pod_combat_move_guis.txt.

# AI combatants spend no currencies; callers gate these on human-controlled turns.
pod_combat_can_afford_move_gold_trigger = {
	gold >= $STORY$.var:gold_cost
}

pod_combat_can_afford_move_piety_trigger = {
	piety >= $STORY$.var:piety_cost
}

pod_combat_can_afford_move_prestige_trigger = {
	prestige >= $STORY$.var:prestige_cost
}

pod_combat_actor_has_baked_unlock_trigger = {
	$ACTOR$ = { is_target_in_variable_list = { name = pod_combat_unlocked_moves target = $STORY$ } }
}

# Only blocks pure riders; callers first test pod_combat_move_has_independent_payload_trigger.
pod_combat_move_payload_useful_trigger = {
	trigger_if = {
		limit = { $STORY$ = { OR = { var:targeting = flag:team_ally  var:targeting = flag:team_enemy  var:targeting = flag:team_all } } }
		pod_combat_team_utility_move_admissible_trigger = { STORY = $STORY$ }
	}
	# AI chooses an ally after move selection, so test candidate recipients here.
	trigger_else_if = {
		limit = {
			NOT = { pod_combat_is_player_turn_trigger = yes }
			$STORY$ = { var:targeting = flag:single_ally }
		}
		pod_combat_ai_has_ally_benefiting_trigger = { STORY = $STORY$ }
	}
	trigger_else = {
		trigger_if = {
			limit = {
				pod_combat_is_player_turn_trigger = yes
				$STORY$ = { var:targeting = flag:single_ally }
			}
			NAND = {
				$STORY$ = { has_variable = set_guard }
				var:pod_combat_session ?= { has_variable = pod_combat_target_is_guarding }
			}
			NAND = {
				$STORY$ = { has_variable = dodge }
				var:pod_combat_session ?= { has_variable = pod_combat_target_is_dodging }
			}
			NAND = {
				$STORY$ = { has_variable = stealth }
				var:pod_combat_session ?= { has_variable = pod_combat_target_is_stealthed }
			}
		}
		trigger_if = {
			limit = {
				$STORY$ = { NOR = { var:targeting = flag:single_ally  var:targeting = flag:team_ally } }
			}
			NAND = {
				$STORY$ = { has_variable = set_guard }
				var:pod_combat_session ?= { has_variable = pod_combat_actor_is_guarding }
			}
			NAND = {
				$STORY$ = { has_variable = dodge }
				var:pod_combat_session ?= { has_variable = pod_combat_actor_is_dodging }
			}
			NAND = {
				$STORY$ = { has_variable = stealth }
				var:pod_combat_session ?= { has_variable = pod_combat_actor_is_stealthed }
			}
		}
		NAND = {
			$STORY$ = { has_variable = retaliate_if_hit }
			var:pod_combat_session ?= { has_variable = pod_combat_actor_retaliate_hit }
		}
		NAND = {
			$STORY$ = { has_variable = retaliate_if_missed }
			var:pod_combat_session ?= { has_variable = pod_combat_actor_retaliate_miss }
		}
		NAND = {
			$STORY$ = { has_variable = retaliate_if_attacked }
			var:pod_combat_session ?= { has_variable = pod_combat_actor_retaliate_any }
		}

		# Equal tiers cannot refresh duration.
		# Hit tiers read the fight workspace; soak tiers read character bridge vars.
		# Player ally buffs use the selected target; AI ally checks branch above.
		trigger_if = {
			limit = {
				pod_combat_is_player_turn_trigger = yes
				$STORY$ = { var:targeting = flag:single_ally }
			}
			pod_combat_soak_tier_upgrade_only_trigger = { STORY = $STORY$ SIDE = target TIER_VAR = soak_buff_round_tier  BRIDGE_VAR = pod_combat_tier_soak_buff_round }
			pod_combat_soak_tier_upgrade_only_trigger = { STORY = $STORY$ SIDE = target TIER_VAR = soak_buff_combat_tier BRIDGE_VAR = pod_combat_tier_soak_buff_combat }
			pod_combat_tier_upgrade_only_trigger = { STORY = $STORY$ SIDE = target TIER_VAR = hit_buff_round_tier }
			pod_combat_tier_upgrade_only_trigger = { STORY = $STORY$ SIDE = target TIER_VAR = hit_buff_combat_tier }
		}
		trigger_if = {
			limit = {
				$STORY$ = { NOR = { var:targeting = flag:single_ally  var:targeting = flag:team_ally } }
			}
			pod_combat_soak_tier_upgrade_only_trigger = { STORY = $STORY$ SIDE = actor TIER_VAR = soak_buff_round_tier  BRIDGE_VAR = pod_combat_tier_soak_buff_round }
			pod_combat_soak_tier_upgrade_only_trigger = { STORY = $STORY$ SIDE = actor TIER_VAR = soak_buff_combat_tier BRIDGE_VAR = pod_combat_tier_soak_buff_combat }
			pod_combat_tier_upgrade_only_trigger = { STORY = $STORY$ SIDE = actor TIER_VAR = hit_buff_round_tier }
			pod_combat_tier_upgrade_only_trigger = { STORY = $STORY$ SIDE = actor TIER_VAR = hit_buff_combat_tier }
		}
		# A selected target's tier can still provide value for the rest of the team even it already has it.
		trigger_if = {
			limit = { $STORY$ = { NOR = { var:targeting = flag:team_enemy  var:targeting = flag:team_all } } }
			pod_combat_soak_tier_upgrade_only_trigger = { STORY = $STORY$ SIDE = target TIER_VAR = soak_debuff_round_tier  BRIDGE_VAR = pod_combat_tier_soak_debuff_round }
			pod_combat_soak_tier_upgrade_only_trigger = { STORY = $STORY$ SIDE = target TIER_VAR = soak_debuff_combat_tier BRIDGE_VAR = pod_combat_tier_soak_debuff_combat }
			pod_combat_tier_upgrade_only_trigger = { STORY = $STORY$ SIDE = target TIER_VAR = hit_debuff_round_tier }
			pod_combat_tier_upgrade_only_trigger = { STORY = $STORY$ SIDE = target TIER_VAR = hit_debuff_combat_tier }
		}
	}
}

# Self-heal-only moves; ally healing must test recipient state.
pod_combat_move_is_self_only_heal_trigger = {
	$STORY$ = {
		OR = {
			has_variable = hp_heal
			has_variable = morale_heal
		}
		NOT = { has_variable = hp_damage }
		NOT = { has_variable = morale_damage }
		OR = {
			NOT = { has_variable = targeting }
			var:targeting = flag:self
		}
	}
}

pod_combat_self_heal_has_effect_trigger = {
	OR = {
		# Superficial healing cannot clear aggravated boxes.
		AND = {
			$STORY$ = {
				has_variable = hp_heal
				var:heal_type = flag:superficial
			}
			pod_combat_actor_hp_value < pod_combat_actor_sup_ceiling_value
		}
		AND = {
			$STORY$ = {
				has_variable = hp_heal
				var:heal_type = flag:aggravated
			}
			pod_combat_actor_hp_value < pod_combat_actor_max_hp_value
		}
		AND = {
			$STORY$ = { has_variable = morale_heal }
			# Keep the comparison host-scoped: the max-morale SV follows the session
			# pointer itself and would fall back to the base cap in fight scope.
			trigger_if = {
				limit = { var:pod_combat_session ?= { has_variable = pod_combat_actor_morale } }
				var:pod_combat_session.var:pod_combat_actor_morale < pod_combat_actor_max_morale_value
			}
			trigger_else = { always = no }
		}
	}
}

# Save the host before slot iteration so target pointers resolve from inside the loop.
pod_combat_player_has_enemy_target_trigger = {
	save_temporary_scope_as = host
	has_variable = pod_combat_my_control_slot
	var:pod_combat_my_control_slot ?= { has_variable = slot_target_team }
	scope:host.var:pod_combat_session ?= {
		any_in_list = {
			variable = pod_combat_active_slots
			var:combat_team = scope:host.var:pod_combat_my_control_slot.var:slot_target_team
			var:team_position = scope:host.var:pod_combat_my_control_slot.var:slot_target_position
			pod_combat_slot_is_enemy_trigger = yes
			pod_combat_slot_is_alive_trigger = yes
		}
	}
}

pod_combat_player_has_ally_target_trigger = {
	save_temporary_scope_as = host
	has_variable = pod_combat_my_control_slot
	var:pod_combat_my_control_slot ?= { has_variable = slot_target_team }
	scope:host.var:pod_combat_session ?= {
		any_in_list = {
			variable = pod_combat_active_slots
			var:combat_team = scope:host.var:pod_combat_my_control_slot.var:slot_target_team
			var:team_position = scope:host.var:pod_combat_my_control_slot.var:slot_target_position
			pod_combat_slot_is_player_team_trigger = yes
			NOT = { var:team_position = scope:host.var:pod_combat_my_control_slot.var:team_position }
			pod_combat_slot_is_alive_trigger = yes
		}
	}
}

pod_combat_pc_slot_alive_trigger = {
	var:pod_combat_session ?= { has_variable = pod_combat_pc_slot }
	var:pod_combat_session.var:pod_combat_pc_slot ?= { has_variable = slot_char pod_combat_slot_is_alive_trigger = yes }
}

# Check the resolved side: summon_hostile seats opposite the caster.
pod_combat_summon_capacity_available_trigger = {
	OR = {
		AND = {
			pod_combat_summon_destination_is_player_team_trigger = { MOVE_STORY = $STORY$ }
			var:pod_combat_session ?= { var:pod_combat_player_team_size < pod_combat_max_team_size }
		}
		AND = {
			pod_combat_summon_destination_is_enemy_team_trigger = { MOVE_STORY = $STORY$ }
			var:pod_combat_session ?= { var:pod_combat_enemy_team_size < pod_combat_max_team_size }
		}
	}
}


######################################################
###### SHARED DYNAMIC GATES
######################################################

# Host scope; $STORY$ = move story; $ACTOR$ = acting character.
# Calling scripted triggers check AP/energy; the player trigger also checks the unlocked-move list.
# Workspace reads require pod_combat_session. Wrong-scope trigger_if limits can
# silently skip gates. Keep new player blocks aligned with pod_combat_move_usability_tt.
pod_combat_move_dynamic_gates_trigger = {
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = gold_cost }
			pod_combat_is_player_turn_trigger = yes
		}
		pod_combat_can_afford_move_gold_trigger = { STORY = $STORY$ }
	}

	trigger_if = {
		limit = {
			$STORY$ = { has_variable = piety_cost }
			pod_combat_is_player_turn_trigger = yes
		}
		pod_combat_can_afford_move_piety_trigger = { STORY = $STORY$ }
	}

	trigger_if = {
		limit = {
			$STORY$ = { has_variable = prestige_cost }
			pod_combat_is_player_turn_trigger = yes
		}
		pod_combat_can_afford_move_prestige_trigger = { STORY = $STORY$ }
	}

	# AI single-ally gates test candidate slots because ally selection follows move choice.
	trigger_if = {
		limit = { $STORY$ = { has_variable = cond_mode  var:cond_mode = flag:gate } }
		trigger_if = {
			limit = {
				NOT = { pod_combat_is_player_turn_trigger = yes }
				$STORY$ = {
					var:targeting = flag:single_ally
					var:cond_target = flag:target
				}
			}
			pod_combat_ai_has_available_ally_trigger = { STORY = $STORY$ }
		}
		trigger_else = {
			pod_combat_move_condition_met_trigger = { STORY = $STORY$ ACTOR = $ACTOR$ PREDICATE = $STORY$.var:cond_predicate }
		}
	}

	trigger_if = {
		limit = { $STORY$ = { has_variable = once_per } }
		NOT = { pod_combat_actor_has_used_move_trigger = { STORY = $STORY$  ACTOR = $ACTOR$ } }
	}

	# stress_cost adds stress on execution - it's always "affordable"

	# Reject redundant pure riders only; hybrids retain their independent payload.
	# AI ally recipients and team utility are checked by the shared scripted trigger.
	trigger_if = {
		limit = { NOT = { pod_combat_move_has_independent_payload_trigger = { STORY = $STORY$ } } }
		custom_description = {
			text = pod_combat_move_redundant_tt
			pod_combat_move_payload_useful_trigger = { STORY = $STORY$ }
		}
	}

	# Only self-heal moves can be rejected from the actor's health alone.
	trigger_if = {
		limit = { pod_combat_move_is_self_only_heal_trigger = { STORY = $STORY$ } }
		custom_description = {
			text = pod_combat_cant_heal_already_full_desc
			pod_combat_self_heal_has_effect_trigger = { STORY = $STORY$ }
		}
	}

	# Human turns use the standing target; AI chooses after move selection.
	trigger_if = {
		limit = {
			pod_combat_is_player_turn_trigger = yes
			$STORY$ = { var:targeting = flag:single_enemy }
		}
		custom_description = {
			text = pod_combat_targeting_need_enemy_tt
			pod_combat_player_has_enemy_target_trigger = yes
		}
	}

	trigger_if = {
		limit = {
			pod_combat_is_player_turn_trigger = yes
			$STORY$ = { var:targeting = flag:single_ally }
		}
		custom_description = {
			text = pod_combat_targeting_need_ally_tt
			pod_combat_player_has_ally_target_trigger = yes
		}
	}

	trigger_if = {
		limit = {
			pod_combat_is_player_turn_trigger = yes
			$STORY$ = { var:targeting = flag:single_pc }
		}
		custom_description = {
			text = pod_combat_targeting_need_pc_alive_tt
			pod_combat_pc_slot_alive_trigger = yes
		}
	}

	trigger_if = {
		limit = {
			pod_combat_is_player_turn_trigger = yes
			$STORY$ = { var:targeting = flag:team_enemy }
		}
		var:pod_combat_session ?= {
			any_in_list = {
				variable = pod_combat_active_slots
				pod_combat_slot_is_enemy_trigger = yes
				pod_combat_slot_is_alive_trigger = yes
			}
		}
	}

	trigger_if = {
		limit = {
			pod_combat_is_player_turn_trigger = yes
			$STORY$ = { var:targeting = flag:team_all }
		}
		# A live fight includes the caster; an empty tooltip-build roster fails closed.
		var:pod_combat_session ?= {
			any_in_list = {
				variable = pod_combat_active_slots
				pod_combat_slot_is_alive_trigger = yes
			}
		}
	}

	# Without an ally, the AI swap leaves the enemy workspace loaded.
	trigger_if = {
		limit = {
			NOT = { pod_combat_is_player_turn_trigger = yes }
			$STORY$ = { var:targeting = flag:single_ally }
		}
		pod_combat_ai_has_available_ally_trigger = { STORY = $STORY$ }
	}

	# AI ally utility tests recipient benefit, not actor state.
	trigger_if = {
		limit = {
			NOT = { pod_combat_is_player_turn_trigger = yes }
			$STORY$ = {
				var:targeting = flag:single_ally
				OR = {
					has_variable = hp_heal
					has_variable = morale_heal
					has_variable = set_guard
					has_variable = dodge
					has_variable = stealth
					has_variable = soak_buff_round_tier
					has_variable = soak_buff_combat_tier
					has_variable = hit_buff_round_tier
					has_variable = hit_buff_combat_tier
				}
			}
		}
		pod_combat_ai_has_ally_benefiting_trigger = { STORY = $STORY$ }
	}

	# Check the duel ban before capacity so a full team does not hide the ban reason.
	trigger_if = {
		limit = { $STORY$ = { has_variable = summons_combatant } }
		custom_description = {
			text = pod_combat_cant_summon_duel_barred_tt
			NOT = { pod_combat_duel_bans_summons_trigger = yes }
		}
	}

	# Recheck capacity before spending AP/energy. Actor team may be unset between rounds.
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = summons_combatant }
			var:pod_combat_session ?= { has_variable = pod_combat_current_actor_team }
		}
		custom_description = {
			text = pod_combat_cant_summon_team_full_tt
			pod_combat_summon_capacity_available_trigger = { STORY = $STORY$ }
		}
	}

}

# PC/UI triggers: per-character unlock membership + resource gates
# + the shared dynamic gate list above.
POD_can_execute_combat_move_trigger = {
	# $STORY$ = move story; $ACTOR$ = driven combatant, potentially distinct from root.
	# AP is fight-owned; both characters resolve the same session.

	NOT = { pod_combat_is_over_trigger = yes }

	# Unlocks are per character at fight start to avoid evaluating the unlock scripted trigger every frame.
	pod_combat_actor_has_baked_unlock_trigger = { STORY = $STORY$ ACTOR = $ACTOR$ }

	trigger_if = {
		limit = {
			$STORY$ = { has_variable = ap_cost }
		}
		pod_combat_can_afford_ap_trigger = { COST = $STORY$.var:ap_cost }
	}

	# Energy must resolve on the acting character, not the host/client.
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = energy_cost }
		}
		$ACTOR$ = {
			POD_has_enough_splat_agnostic_energy_reserves = { AMOUNT = $STORY$.var:energy_cost }
		}
	}

	pod_combat_move_dynamic_gates_trigger = { STORY = $STORY$ ACTOR = $ACTOR$ }
}

}


######################################################
###### TIER UPGRADE-ONLY GUARD (parametric)
######################################################

# Host scope. Compare the move tier against the fight workspace.
# Equal tiers cannot refresh round/combat duration; writers require a strict upgrade.
# Missing fight, workspace or rider passes. Guard both reads inside trigger_if.
pod_combat_tier_upgrade_only_trigger = {
	# $STORY$ = move story; $SIDE$ = actor|target; $TIER_VAR$ = tier var name.
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = $TIER_VAR$ }
			var:pod_combat_session ?= { has_variable = pod_combat_$SIDE$_$TIER_VAR$ }
		}
		var:pod_combat_session.var:pod_combat_$SIDE$_$TIER_VAR$ < $STORY$.var:$TIER_VAR$
	}
	trigger_else = { always = yes }
}

# Soak tiers live on the combatant, not the workspace. An absent bridge means tier 0.
# $BRIDGE_VAR$ = pod_combat_tier_soak_{buff,debuff}_{round,combat}.
pod_combat_soak_tier_upgrade_only_trigger = {
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = $TIER_VAR$ }
			var:pod_combat_session ?= { has_variable = pod_combat_$SIDE$_char }
			exists = var:pod_combat_session.var:pod_combat_$SIDE$_char
		}
		# Nest the bridge read: character scope are unset between rounds.
		var:pod_combat_session.var:pod_combat_$SIDE$_char ?= {
			trigger_if = {
				limit = { has_variable = $BRIDGE_VAR$ }
				var:$BRIDGE_VAR$ < $STORY$.var:$TIER_VAR$
			}
			trigger_else = { always = yes }
		}
	}
}

# $STORY$ = move story. Independent payloads must survive redundant-rider checks.
pod_combat_move_has_independent_payload_trigger = {
	$STORY$ = {
		OR = {
			has_variable = hp_damage
			has_variable = morale_damage
			has_variable = hp_heal
			has_variable = morale_heal
			has_variable = ap_gain
			has_variable = ap_drain
			has_variable = initiative_debuff
			has_variable = dot_amount
			has_variable = mind_control
			has_variable = remove_guard
			has_variable = summons_combatant
			has_variable = steal_gold
			has_variable = gain_dread
			has_variable = energy_gain
			has_variable_list = bonus_clauses
			has_variable = marker_modifier
			has_variable = remove_marker
		}
	}
}


######################################################
###### RESULT-CODE CONDITION TRIGGERS
######################################################
# Host-scoped result partitions; see COMBAT_REFERENCE.info section 4.
# attack_result is meaningful only when outcome=attack.

# AoE log pushes clear the per-target result, so damage loops retain any landed
# HP damage in pod_combat_aoe_any_hp_damage for the later drain rider.
pod_combat_result_code_indicates_hp_damage_trigger = {
	var:pod_combat_session ?= {
		OR = {
			AND = {
				has_variable = pod_combat_last_outcome
				var:pod_combat_last_outcome = flag:attack
				has_variable = pod_combat_last_attack_result
				var:pod_combat_last_attack_result = flag:hp_damage
			}
			has_variable = pod_combat_aoe_any_hp_damage
		}
	}
}

# Contact includes guarded/soaked hits for on-touch riders.
pod_combat_result_code_indicates_contact_trigger = {
	var:pod_combat_session ?= {
		has_variable = pod_combat_last_outcome
		var:pod_combat_last_outcome = flag:attack
		has_variable = pod_combat_last_attack_result
		OR = {
			var:pod_combat_last_attack_result = flag:hp_damage

# Host scope. Rider conditions: always (also the unset default), on_hp_damage,
# on_contact or on_miss. Drain moves use heal_condition=on_hp_damage.
pod_combat_rider_condition_passes_trigger = {
	# $STORY$     = the POD_combat_move story_cycle
	# $RIDER_VAR$ = the condition variable name on the story (e.g. heal_condition)
	# Outcome conditions ignore ACTOR, so root is a valid placeholder.
	trigger_if = {
		limit = { NOT = { $STORY$ = { has_variable = $RIDER_VAR$ } } }
		always = yes
	}
	trigger_else = {
		pod_combat_move_condition_met_trigger = {
			STORY = $STORY$
			ACTOR = root
			PREDICATE = $STORY$.var:$RIDER_VAR$
		}
	}
}

######################################################
###### STATE CONDITION (gate / bonus)
######################################################
# Host scope. $STORY$ is the move; $ACTOR$ is the acting character.
# cond_target selects $ACTOR$ or var:pod_combat_target_char for character conditions;
# target conditions fail when no target is loaded. Morale uses the host's mirrored variables.
# $PREDICATE$ selects a state or outcome condition. State checks pass cond_predicate;
# rider checks pass their condition variable. Outcome conditions read host result
# variables set in execution section 6 and ignore $ACTOR$.
pod_combat_move_condition_met_trigger = {
	switch = {
		trigger = $PREDICATE$

		# global (host round)
		flag:first_round = {
			trigger_if = { limit = { var:pod_combat_session ?= { has_variable = pod_combat_round } } var:pod_combat_session ?= { var:pod_combat_round = 1 } }
			trigger_else = { always = no }
		}
		flag:round_min = {
			trigger_if = { limit = { var:pod_combat_session ?= { has_variable = pod_combat_round } } var:pod_combat_session ?= { var:pod_combat_round >= $STORY$.var:cond_threshold } }
			trigger_else = { always = no }
		}

		# HP, live from the character's PoD wound track
		flag:undamaged = {
			trigger_if = {
				limit = { $STORY$ = { var:cond_target = flag:target }  var:pod_combat_session ?= { has_variable = pod_combat_target_char } }
				var:pod_combat_session.var:pod_combat_target_char ?= { POD_empty_damage_boxes = POD_health_track_length }
			}
			trigger_else_if = { limit = { $STORY$ = { var:cond_target = flag:target } } always = no }
			trigger_else = { $ACTOR$ = { POD_empty_damage_boxes = POD_health_track_length } }
		}
		flag:hp_missing = {
			trigger_if = {
				limit = { $STORY$ = { var:cond_target = flag:target }  var:pod_combat_session ?= { has_variable = pod_combat_target_char } }
				var:pod_combat_session.var:pod_combat_target_char ?= { POD_empty_damage_boxes <= { value = POD_health_track_length  subtract = $STORY$.var:cond_threshold } }
			}
			trigger_else_if = { limit = { $STORY$ = { var:cond_target = flag:target } } always = no }
			trigger_else = { $ACTOR$ = { POD_empty_damage_boxes <= { value = POD_health_track_length  subtract = $STORY$.var:cond_threshold } } }
		}

		# morale, via the host workspace mirror
		flag:full_morale = {
			trigger_if = {
				limit = { $STORY$ = { var:cond_target = flag:target }  var:pod_combat_session ?= { has_variable = pod_combat_target_morale } }
				var:pod_combat_session.var:pod_combat_target_morale = pod_combat_target_max_morale_value
			}
			trigger_else_if = { limit = { $STORY$ = { var:cond_target = flag:target } } always = no }
			trigger_else_if = { limit = { var:pod_combat_session ?= { has_variable = pod_combat_actor_morale } } var:pod_combat_session.var:pod_combat_actor_morale = pod_combat_actor_max_morale_value }
			trigger_else = { always = no }
		}
		flag:morale_missing = {
			trigger_if = {
				limit = { $STORY$ = { var:cond_target = flag:target }  var:pod_combat_session ?= { has_variable = pod_combat_target_morale } }
				var:pod_combat_session.var:pod_combat_target_morale <= { value = pod_combat_target_max_morale_value  subtract = $STORY$.var:cond_threshold }
			}
			trigger_else_if = { limit = { $STORY$ = { var:cond_target = flag:target } } always = no }
			trigger_else_if = { limit = { var:pod_combat_session ?= { has_variable = pod_combat_actor_morale } } var:pod_combat_session.var:pod_combat_actor_morale <= { value = pod_combat_actor_max_morale_value  subtract = $STORY$.var:cond_threshold } }
			trigger_else = { always = no }
		}

		# stance flags (has_variable is unset-safe)
		flag:has_guard = {
			trigger_if = { limit = { $STORY$ = { var:cond_target = flag:target } } var:pod_combat_session ?= { has_variable = pod_combat_target_is_guarding } }
			trigger_else = { var:pod_combat_session ?= { has_variable = pod_combat_actor_is_guarding } }
		}
		flag:has_dodge = {
			trigger_if = { limit = { $STORY$ = { var:cond_target = flag:target } } var:pod_combat_session ?= { has_variable = pod_combat_target_is_dodging } }
			trigger_else = { var:pod_combat_session ?= { has_variable = pod_combat_actor_is_dodging } }

		flag:has_stealth = {
			trigger_if = { limit = { $STORY$ = { var:cond_target = flag:target } } var:pod_combat_session ?= { has_variable = pod_combat_target_is_stealthed } }
			trigger_else = { var:pod_combat_session ?= { has_variable = pod_combat_actor_is_stealthed } }
		}

		# cond_threshold: prestige/piety ranks 0-5, dread 0-100.
		flag:prestige_level_min = {
			trigger_if = {
				limit = { $STORY$ = { var:cond_target = flag:target }  var:pod_combat_session ?= { has_variable = pod_combat_target_char } }
				var:pod_combat_session ?= { var:pod_combat_target_char = { prestige_level >= $STORY$.var:cond_threshold } }
			}

			trigger_else = { $ACTOR$ = { prestige_level >= $STORY$.var:cond_threshold } }
		}
		flag:piety_level_min = {
			trigger_if = {
				limit = { $STORY$ = { var:cond_target = flag:target }  var:pod_combat_session ?= { has_variable = pod_combat_target_char } }
				var:pod_combat_session ?= { var:pod_combat_target_char = { piety_level >= $STORY$.var:cond_threshold } }
			}
			trigger_else_if = { limit = { $STORY$ = { var:cond_target = flag:target } } always = no }
			trigger_else = { $ACTOR$ = { piety_level >= $STORY$.var:cond_threshold } }

		flag:dread_min = {
			trigger_if = {
				limit = { $STORY$ = { var:cond_target = flag:target }  var:pod_combat_session ?= { has_variable = pod_combat_target_char } }
				var:pod_combat_session ?= { var:pod_combat_target_char = { dread >= $STORY$.var:cond_threshold } }

		}

		# $ACTOR$ judges the target; root=host preserves target access after switching.
		# cond_threshold is the magnitude; negative opinion negates it.
		flag:relation_positive = {
			trigger_if = {
				limit = { var:pod_combat_session ?= { has_variable = pod_combat_target_char } }
				$ACTOR$ = { opinion = { target = root.var:pod_combat_session.var:pod_combat_target_char  value >= $STORY$.var:cond_threshold } }
			}
			trigger_else = { always = no }
		}
		flag:relation_negative = {
			trigger_if = {
				limit = { var:pod_combat_session ?= { has_variable = pod_combat_target_char } }
				$ACTOR$ = { opinion = { target = root.var:pod_combat_session.var:pod_combat_target_char  value <= { value = $STORY$.var:cond_threshold  multiply = -1 } } }
			}
			trigger_else = { always = no }
		}

		flag:has_weapon_artifact = {
			trigger_if = {
				limit = { $STORY$ = { var:cond_target = flag:target }  var:pod_combat_session ?= { has_variable = pod_combat_target_char } }
				var:pod_combat_session ?= { var:pod_combat_target_char = { any_character_artifact = { is_equipped = yes  artifact_slot_type = primary_armament } } }
			}
			trigger_else_if = { limit = { $STORY$ = { var:cond_target = flag:target } } always = no }
			trigger_else = { $ACTOR$ = { any_character_artifact = { is_equipped = yes  artifact_slot_type = primary_armament } } }
		}
		flag:has_armor_artifact = {
			trigger_if = {
				limit = { $STORY$ = { var:cond_target = flag:target }  var:pod_combat_session ?= { has_variable = pod_combat_target_char } }
				var:pod_combat_session ?= { var:pod_combat_target_char = { any_character_artifact = { is_equipped = yes  artifact_slot_type = armor } } }
			}
			trigger_else_if = { limit = { $STORY$ = { var:cond_target = flag:target } } always = no }
			trigger_else = { $ACTOR$ = { any_character_artifact = { is_equipped = yes  artifact_slot_type = armor } } }
		}

		# Resolve cond_param to literal identifiers in pod_combat_condition_dispatch_triggers.txt.
		flag:has_trait = {
			trigger_if = {
				limit = { $STORY$ = { var:cond_target = flag:target }  var:pod_combat_session ?= { has_variable = pod_combat_target_char } }
				var:pod_combat_session ?= { var:pod_combat_target_char = { pod_combat_cond_has_trait_trigger = { STORY = $STORY$ } } }
			}
			trigger_else_if = { limit = { $STORY$ = { var:cond_target = flag:target } } always = no }
			trigger_else = { $ACTOR$ = { pod_combat_cond_has_trait_trigger = { STORY = $STORY$ } } }
		}
		flag:target_known_secret = {
			trigger_if = {
				limit = { var:pod_combat_session ?= { has_variable = pod_combat_target_char } }
				$ACTOR$ = { pod_combat_cond_known_secret_trigger = { STORY = $STORY$ } }
			}
			trigger_else = { always = no }
		}
		flag:has_modifier = {
			trigger_if = {
				limit = { $STORY$ = { var:cond_target = flag:target }  var:pod_combat_session ?= { has_variable = pod_combat_target_char } }
				var:pod_combat_session ?= { var:pod_combat_target_char = { pod_combat_cond_has_modifier_trigger = { STORY = $STORY$ } } }
			}
			trigger_else_if = { limit = { $STORY$ = { var:cond_target = flag:target } } always = no }
			trigger_else = { $ACTOR$ = { pod_combat_cond_has_modifier_trigger = { STORY = $STORY$ } } }
		}
		flag:has_perk = {
			trigger_if = {
				limit = { $STORY$ = { var:cond_target = flag:target }  var:pod_combat_session ?= { has_variable = pod_combat_target_char } }
				var:pod_combat_session ?= { var:pod_combat_target_char = { pod_combat_cond_has_perk_trigger = { STORY = $STORY$ } } }
			}
			trigger_else_if = { limit = { $STORY$ = { var:cond_target = flag:target } } always = no }
			trigger_else = { $ACTOR$ = { pod_combat_cond_has_perk_trigger = { STORY = $STORY$ } } }
		}

		flag:fera_form = {
			trigger_if = {
				limit = { $STORY$ = { var:cond_target = flag:target }  var:pod_combat_session ?= { has_variable = pod_combat_target_char } }
				var:pod_combat_session ?= { var:pod_combat_target_char = { pod_combat_cond_fera_form_trigger = { STORY = $STORY$ } } }

			trigger_else_if = { limit = { $STORY$ = { var:cond_target = flag:target } } always = no }
			trigger_else = { $ACTOR$ = { pod_combat_cond_fera_form_trigger = { STORY = $STORY$ } } }

		flag:wyrm_servant = {
			trigger_if = {
				limit = { $STORY$ = { var:cond_target = flag:target }  var:pod_combat_session ?= { has_variable = pod_combat_target_char } }
				var:pod_combat_session ?= { var:pod_combat_target_char = { POD_is_wyrm_servant_trigger = yes } }
			}
			trigger_else_if = { limit = { $STORY$ = { var:cond_target = flag:target } } always = no }
			trigger_else = { $ACTOR$ = {POD_is_wyrm_servant_trigger = yes } }
		}

			trigger_if = {
				limit = { $STORY$ = { var:cond_target = flag:target }  var:pod_combat_session ?= { has_variable = pod_combat_target_char } }
				var:pod_combat_session ?= { var:pod_combat_target_char = { POD_is_fera_trigger = yes } }
			}
			trigger_else_if = { limit = { $STORY$ = { var:cond_target = flag:target } } always = no }
			trigger_else = { $ACTOR$ = { POD_is_fera_trigger = yes } }
		}

		# Outcome conditions read execution section 6 results and ignore $ACTOR$.
		flag:always = { always = yes }
		flag:on_hp_damage = { pod_combat_result_code_indicates_hp_damage_trigger = yes }
		flag:on_contact = { pod_combat_result_code_indicates_contact_trigger = yes }
		flag:on_miss = { pod_combat_result_code_indicates_miss_trigger = yes }


		# Team conditions ignore cond_target. Boolean conditions also ignore cond_threshold.
		# Save $ACTOR$ before slot iteration: callers may pass a host-relative dereference.
		flag:outnumbered = {
			$ACTOR$ = { save_temporary_scope_as = pod_combat_cond_actor }

				limit = { var:pod_combat_session ?= { any_in_list = { variable = pod_combat_active_slots  var:slot_char = { this = scope:pod_combat_cond_actor }  var:combat_team = flag:player } } }
				pod_combat_alive_player_count_value < pod_combat_alive_enemy_count_value

		}
		# Allies exclude the actor. Whole-team count SVs require a one-member offset;
		# the actor is alive at all evaluation sites.
		flag:alive_enemies_min = {
			$ACTOR$ = { save_temporary_scope_as = pod_combat_cond_actor }

				limit = { var:pod_combat_session ?= { any_in_list = { variable = pod_combat_active_slots  var:slot_char = { this = scope:pod_combat_cond_actor }  var:combat_team = flag:enemy } } }
				pod_combat_alive_player_count_value >= $STORY$.var:cond_threshold

			$ACTOR$ = { save_temporary_scope_as = pod_combat_cond_actor }

				pod_combat_alive_enemy_count_value <= $STORY$.var:cond_threshold
			}
			trigger_else_if = {
				limit = { var:pod_combat_session ?= { any_in_list = { variable = pod_combat_active_slots  var:slot_char = { this = scope:pod_combat_cond_actor }  var:combat_team = flag:enemy } } }
				pod_combat_alive_player_count_value <= $STORY$.var:cond_threshold
			}
			trigger_else = { always = no }
		}
		flag:alive_allies_min = {
			$ACTOR$ = { save_temporary_scope_as = pod_combat_cond_actor }
			trigger_if = {
				limit = { var:pod_combat_session ?= { any_in_list = { variable = pod_combat_active_slots  var:slot_char = { this = scope:pod_combat_cond_actor }  var:combat_team = flag:player } } }
				pod_combat_alive_player_count_value >= { value = $STORY$.var:cond_threshold  add = 1 }
			}
			trigger_else_if = {
				limit = { var:pod_combat_session ?= { any_in_list = { variable = pod_combat_active_slots  var:slot_char = { this = scope:pod_combat_cond_actor }  var:combat_team = flag:enemy } } }
				pod_combat_alive_enemy_count_value >= { value = $STORY$.var:cond_threshold  add = 1 }
			}
			trigger_else = { always = no }
		}
		flag:alive_allies_max = {
			$ACTOR$ = { save_temporary_scope_as = pod_combat_cond_actor }
			trigger_if = {
				limit = { var:pod_combat_session ?= { any_in_list = { variable = pod_combat_active_slots  var:slot_char = { this = scope:pod_combat_cond_actor }  var:combat_team = flag:player } } }
				pod_combat_alive_player_count_value <= { value = $STORY$.var:cond_threshold  add = 1 }
			}
			trigger_else_if = {
				limit = { var:pod_combat_session ?= { any_in_list = { variable = pod_combat_active_slots  var:slot_char = { this = scope:pod_combat_cond_actor }  var:combat_team = flag:enemy } } }
				pod_combat_alive_enemy_count_value <= { value = $STORY$.var:cond_threshold  add = 1 }
			}
			trigger_else = { always = no }
		}
		# Unknown condition - fail closed (generator validates cond_predicate).
		fallback = { always = no }
	}
}

# Slot-scoped benefit checks run before execution, while slot vars are current.
pod_combat_slot_stance_rider_useful_trigger = {
	OR = {
		AND = { $STORY$ = { has_variable = set_guard } NOT = { has_variable = slot_is_guarding } }
		AND = { $STORY$ = { has_variable = dodge }     NOT = { has_variable = slot_is_dodging } }
		AND = { $STORY$ = { has_variable = stealth }   NOT = { has_variable = slot_is_stealthed } }
	}
}

pod_combat_slot_tier_rider_upgrades_trigger = {
	# $TIER_VAR$ = move tier; absent slot tier = 0.
	$STORY$ = { has_variable = $TIER_VAR$ }
	trigger_if = {
		# Guard both operands here; the sibling presence test does not short-circuit.
		limit = { has_variable = slot_$TIER_VAR$  $STORY$ = { has_variable = $TIER_VAR$ } }
		var:slot_$TIER_VAR$ < $STORY$.var:$TIER_VAR$
	}
	trigger_else = { always = yes }
}

# Read soak from the occupant's character bridge; no occupant means no benefit.
pod_combat_slot_soak_tier_rider_upgrades_trigger = {
	$STORY$ = { has_variable = $TIER_VAR$ }
	trigger_if = {
		limit = { exists = var:slot_char }
		# Nest the character-pointer guard before reading the bridge.
		var:slot_char = {
			trigger_if = {
				limit = { has_variable = $BRIDGE_VAR$ }
				var:$BRIDGE_VAR$ < $STORY$.var:$TIER_VAR$
			}
			trigger_else = { always = yes }
		}
	}
	# A slot whose occupant is gone cannot benefit.
	trigger_else = { always = no }
}

# Host scope. Pure team utility is useful if any affected living slot benefits.
# Missing actor team passes. This shared GUI/AI gate scans live state because
# player checks cannot use an AI-action cache.
pod_combat_team_utility_move_admissible_trigger = {
	save_temporary_scope_as = host
	trigger_if = {
		limit = { NOT = { var:pod_combat_session ?= { has_variable = pod_combat_current_actor_team } } }
		always = yes
	}
	trigger_else_if = {
		limit = { $STORY$ = { var:targeting = flag:team_ally } }
		scope:host.var:pod_combat_session ?= {
			any_in_list = {
				variable = pod_combat_active_slots
				pod_combat_slot_is_alive_trigger = yes
				pod_combat_aoe_team_ally_filter_trigger = yes
				OR = {
					pod_combat_slot_stance_rider_useful_trigger = { STORY = $STORY$ }
					pod_combat_slot_soak_tier_rider_upgrades_trigger = { STORY = $STORY$ TIER_VAR = soak_buff_round_tier  BRIDGE_VAR = pod_combat_tier_soak_buff_round }
					pod_combat_slot_soak_tier_rider_upgrades_trigger = { STORY = $STORY$ TIER_VAR = soak_buff_combat_tier BRIDGE_VAR = pod_combat_tier_soak_buff_combat }
					pod_combat_slot_tier_rider_upgrades_trigger = { STORY = $STORY$ TIER_VAR = hit_buff_round_tier }
					pod_combat_slot_tier_rider_upgrades_trigger = { STORY = $STORY$ TIER_VAR = hit_buff_combat_tier }
				}
			}
		}
	}
	trigger_else_if = {
		# team_all debuffs include the caster.
		limit = { $STORY$ = { var:targeting = flag:team_all } }
		scope:host.var:pod_combat_session ?= {
			any_in_list = {
				variable = pod_combat_active_slots
				pod_combat_slot_is_alive_trigger = yes
				OR = {
					pod_combat_slot_soak_tier_rider_upgrades_trigger = { STORY = $STORY$ TIER_VAR = soak_debuff_round_tier  BRIDGE_VAR = pod_combat_tier_soak_debuff_round }
					pod_combat_slot_soak_tier_rider_upgrades_trigger = { STORY = $STORY$ TIER_VAR = soak_debuff_combat_tier BRIDGE_VAR = pod_combat_tier_soak_debuff_combat }
					pod_combat_slot_tier_rider_upgrades_trigger = { STORY = $STORY$ TIER_VAR = hit_debuff_round_tier }
					pod_combat_slot_tier_rider_upgrades_trigger = { STORY = $STORY$ TIER_VAR = hit_debuff_combat_tier }
				}
			}
		}
	}
	trigger_else_if = {
		limit = { $STORY$ = { var:targeting = flag:team_enemy } }
		scope:host.var:pod_combat_session ?= {
			any_in_list = {
				variable = pod_combat_active_slots
				pod_combat_slot_is_alive_trigger = yes
				pod_combat_aoe_team_enemy_filter_trigger = yes
				OR = {
					pod_combat_slot_soak_tier_rider_upgrades_trigger = { STORY = $STORY$ TIER_VAR = soak_debuff_round_tier  BRIDGE_VAR = pod_combat_tier_soak_debuff_round }
					pod_combat_slot_soak_tier_rider_upgrades_trigger = { STORY = $STORY$ TIER_VAR = soak_debuff_combat_tier BRIDGE_VAR = pod_combat_tier_soak_debuff_combat }
					pod_combat_slot_tier_rider_upgrades_trigger = { STORY = $STORY$ TIER_VAR = hit_debuff_round_tier }
					pod_combat_slot_tier_rider_upgrades_trigger = { STORY = $STORY$ TIER_VAR = hit_debuff_combat_tier }
				}
			}
		}
	}
	trigger_else = { always = yes }
}

# scope:ai_candidate_move = scored move; scope:combat_host = host character.
# Penalize redundant hybrid riders using the same tier tests.
# Round tiers already have @ai_round_buff_spam / @ai_round_buff_lateap penalties;
# soak uses @ai_soak_buff_capped instead.
pod_combat_ai_candidate_has_redundant_rider_trigger = {
	OR = {
		AND = { scope:ai_candidate_move = { has_variable = set_guard } scope:combat_host.var:pod_combat_session ?= { has_variable = pod_combat_actor_is_guarding } }
		AND = { scope:ai_candidate_move = { has_variable = dodge }     scope:combat_host.var:pod_combat_session ?= { has_variable = pod_combat_actor_is_dodging } }
		AND = { scope:ai_candidate_move = { has_variable = stealth }   scope:combat_host.var:pod_combat_session ?= { has_variable = pod_combat_actor_is_stealthed } }
		AND = { scope:ai_candidate_move = { has_variable = retaliate_if_hit }    scope:combat_host.var:pod_combat_session ?= { has_variable = pod_combat_actor_retaliate_hit } }
		AND = { scope:ai_candidate_move = { has_variable = retaliate_if_missed } scope:combat_host.var:pod_combat_session ?= { has_variable = pod_combat_actor_retaliate_miss } }
		AND = { scope:ai_candidate_move = { has_variable = retaliate_if_attacked } scope:combat_host.var:pod_combat_session ?= { has_variable = pod_combat_actor_retaliate_any } }
		# Host scope required. NOT passes only for a present, non-upgrading tier.
		scope:combat_host = { NOT = { pod_combat_tier_upgrade_only_trigger = { STORY = scope:ai_candidate_move SIDE = actor TIER_VAR = hit_buff_combat_tier } } }
		scope:combat_host = { NOT = { pod_combat_tier_upgrade_only_trigger = { STORY = scope:ai_candidate_move SIDE = target TIER_VAR = hit_debuff_combat_tier } } }
	}
}

# Log-entry scope. Render move-level segments once: first AoE row or non-AoE row.
pod_combat_log_seg_burst_gate_trigger = {
	OR = {
		NOT = { has_variable = aoe_burst_index }
		var:aoe_burst_index = 0
	}
	# Failed pure-control delivery must not advertise unapplied payloads.
	# Guard move/outcome reads for system rows that lack them.
	trigger_if = {
		limit = {
			has_variable = move
			has_variable = outcome
		}
		NAND = {
			var:move = { has_variable = requires_control_contact }
			var:outcome = flag:attack
		}
	}
	trigger_else = { always = yes }
}

# AI scoring scopes: ai_actor, combat_host and ai_candidate_move.
# Uses the per-action maximum usable $STAT_VAR$ tier, not just AP affordability;
# energy-starved or once-per-spent alternatives must not penalize a usable move.
pod_combat_ai_has_usable_larger_buff_tier_trigger = {
	# Guard both operands: tooltip walks can precede the tier precompute.
	trigger_if = {
		limit = {
			scope:ai_candidate_move = { has_variable = $STAT_VAR$ }
			scope:combat_host.var:pod_combat_session ?= { has_variable = pod_combat_ai_maxtier_$STAT_VAR$ }
		}
		scope:combat_host.var:pod_combat_session.var:pod_combat_ai_maxtier_$STAT_VAR$ > scope:ai_candidate_move.var:$STAT_VAR$
	}
	trigger_else = { always = no }
}

# Candidate-move scope; $TIER$ = 1/2/3. Tests all buff/debuff tier attributes.
pod_combat_ai_candidate_has_tier_trigger = {
	OR = {
		AND = { has_variable = soak_buff_round_tier var:soak_buff_round_tier = $TIER$ }
		AND = { has_variable = soak_buff_combat_tier var:soak_buff_combat_tier = $TIER$ }
		AND = { has_variable = soak_debuff_round_tier var:soak_debuff_round_tier = $TIER$ }
		AND = { has_variable = soak_debuff_combat_tier var:soak_debuff_combat_tier = $TIER$ }
		AND = { has_variable = hit_buff_round_tier var:hit_buff_round_tier = $TIER$ }
		AND = { has_variable = hit_buff_combat_tier var:hit_buff_combat_tier = $TIER$ }
		AND = { has_variable = hit_debuff_round_tier var:hit_debuff_round_tier = $TIER$ }
		AND = { has_variable = hit_debuff_combat_tier var:hit_debuff_combat_tier = $TIER$ }
	}
}

# A larger usable tier of the same stat exists; delegate operand guards to the shared check.
pod_combat_ai_candidate_has_undersized_buff_trigger = {
	OR = {
		AND = { has_variable = soak_buff_round_tier pod_combat_ai_has_usable_larger_buff_tier_trigger = { STAT_VAR = soak_buff_round_tier } }
		AND = { has_variable = soak_buff_combat_tier pod_combat_ai_has_usable_larger_buff_tier_trigger = { STAT_VAR = soak_buff_combat_tier } }

		AND = { has_variable = hit_buff_combat_tier pod_combat_ai_has_usable_larger_buff_tier_trigger = { STAT_VAR = hit_buff_combat_tier } }
		AND = { has_variable = hit_debuff_round_tier pod_combat_ai_has_usable_larger_buff_tier_trigger = { STAT_VAR = hit_debuff_round_tier } }

	}
