######################################################
######   POD COMBAT MOVE EXECUTION TRIGGERS
######################################################
# Generic move-validation trigger: reads move attributes off the
# story_cycle scope. Called on the combat-host (PC) scope from the
# execute_move SGUI is_valid and the AI move selector. The loadout
# editor does NOT call this — it gates on can_execute_decision alone
# (unlock-only subset; no AP/energy/targeting outside live combat).


POD_can_execute_combat_move = {
	# $STORY$ = the POD_combat_move story_cycle
	# $ACTOR$ = character whose unlock gate (splat/trait/perk) is tested.
	#          On AI turns this differs from the host (PC); workspace
	#          reads (AP, energy) still resolve against root = the host.

	NOT = { pod_combat_is_over_trigger = yes }

	# Unlock gate runs against the actor, not the host: on AI turns the
	# host is the PC but the unlock requirements live on the AI NPC.
	$ACTOR$ = {
		can_execute_decision = $STORY$.var:decision
	}

	trigger_if = {
		limit = {
			$STORY$ = { has_variable = ap_cost }
		}
		pod_combat_can_afford_ap_trigger = { COST = $STORY$.var:ap_cost }
	}

	# Energy: the splat-agnostic reserve check resolves against the scope
	# it runs on, so wrap in $ACTOR$ to hit the acting character's pool
	# (bare it would read the host/PC pool on AI turns).
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = energy_cost }
		}
		$ACTOR$ = {
			POD_has_enough_splat_agnostic_energy_reserves = { AMOUNT = $STORY$.var:energy_cost }
		}
	}

	# CK3 resource costs are PC-only — AI combatants spend nothing, so
	# gold/piety/prestige gates apply only on the player turn.
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = gold_cost }
			pod_combat_is_player_turn_trigger = yes
		}
		gold >= $STORY$.var:gold_cost
	}

	trigger_if = {
		limit = {
			$STORY$ = { has_variable = piety_cost }
			pod_combat_is_player_turn_trigger = yes
		}
		piety >= $STORY$.var:piety_cost
	}

	trigger_if = {
		limit = {
			$STORY$ = { has_variable = prestige_cost }
			pod_combat_is_player_turn_trigger = yes
		}
		prestige >= $STORY$.var:prestige_cost
	}

	# stress_cost adds stress on execution — it's always "affordable"

	# Duplicate-stance / reaction guard — reject re-applying a stance the
	# actor already holds (mirrors the AI selector's NAND block in
	# pod_combat_ai_effects.txt). Without it the bonus double-bakes and
	# needs two separate consumes to revert.
	NAND = {
		$STORY$ = { has_variable = set_guard }
		has_variable = pod_combat_actor_is_guarding
	}
	NAND = {
		$STORY$ = { has_variable = dodge }
		has_variable = pod_combat_actor_is_dodging
	}
	NAND = {
		$STORY$ = { has_variable = stealth }
		has_variable = pod_combat_actor_is_stealthed
	}
	NAND = {
		$STORY$ = { has_variable = retaliate_if_hit }
		has_variable = pod_combat_actor_retaliate_hit
	}
	NAND = {
		$STORY$ = { has_variable = retaliate_if_missed }
		has_variable = pod_combat_actor_retaliate_miss
	}

	# Tier-redundancy guard — reject a tier buff/debuff at or below the
	# current tier (upgrade-only contract; see the §11 rationale in
	# pod_combat_move_effects.txt). Strict `<` matches the AI guard.
	# Each gate is a single trigger_if (not NAND) so the var reads only
	# happen when both the move attribute and the workspace var exist —
	# avoids a "variable not set" warning when either is absent.
	# Covers workspace single-target tiers only; AoE downgrades are
	# guarded at write time in POD_combat_apply_team_ally_buffs /
	# _team_enemy_debuffs.
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = soak_buff_round_tier }
			has_variable = pod_combat_actor_soak_buff_round_tier
		}
		var:pod_combat_actor_soak_buff_round_tier < $STORY$.var:soak_buff_round_tier
	}
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = soak_buff_combat_tier }
			has_variable = pod_combat_actor_soak_buff_combat_tier
		}
		var:pod_combat_actor_soak_buff_combat_tier < $STORY$.var:soak_buff_combat_tier
	}
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = hit_buff_round_tier }
			has_variable = pod_combat_actor_hit_buff_round_tier
		}
		var:pod_combat_actor_hit_buff_round_tier < $STORY$.var:hit_buff_round_tier
	}
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = hit_buff_combat_tier }
			has_variable = pod_combat_actor_hit_buff_combat_tier
		}
		var:pod_combat_actor_hit_buff_combat_tier < $STORY$.var:hit_buff_combat_tier
	}
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = soak_debuff_round_tier }
			has_variable = pod_combat_target_soak_debuff_round_tier
		}
		var:pod_combat_target_soak_debuff_round_tier < $STORY$.var:soak_debuff_round_tier
	}
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = soak_debuff_combat_tier }
			has_variable = pod_combat_target_soak_debuff_combat_tier
		}
		var:pod_combat_target_soak_debuff_combat_tier < $STORY$.var:soak_debuff_combat_tier
	}
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = hit_debuff_round_tier }
			has_variable = pod_combat_target_hit_debuff_round_tier
		}
		var:pod_combat_target_hit_debuff_round_tier < $STORY$.var:hit_debuff_round_tier
	}
	trigger_if = {
		limit = {
			$STORY$ = { has_variable = hit_debuff_combat_tier }
			has_variable = pod_combat_target_hit_debuff_combat_tier
		}
		var:pod_combat_target_hit_debuff_combat_tier < $STORY$.var:hit_debuff_combat_tier
	}

	# Heal-utility gate — block self-targeted heal-only moves (no damage
	# rider) when the actor is already at the cap the heal can't exceed
	# (mirrors the AI selector's gate in pod_combat_ai_effects.txt).
	# Scoped to targeting = flag:self (and untargeted heals, which the
	# generator defaults to actor-self). single_ally / team_ally heals
	# stay clickable — the actor's HP isn't the target, and the player
	# may intentionally heal a wounded ally while full.
	trigger_if = {
		limit = {
			$STORY$ = {
				OR = {
					has_variable = hp_heal
					has_variable = morale_heal
				}
				NOT = { has_variable = hp_damage }
				NOT = { has_variable = morale_damage }
				OR = {
					NOT = { has_variable = targeting }
					var:targeting = flag:self
				}
			}
		}
		custom_description = {
			text = pod_combat_cant_heal_already_full_desc
			OR = {
				# Superficial heals stop at sup_ceiling (= max_hp −
				# agg_boxes): aggravated wounds pin the track top, so
				# this can be a no-op even while hp < max_hp.
				AND = {
					$STORY$ = {
						has_variable = hp_heal
						var:heal_type = flag:superficial
					}
					pod_combat_actor_hp_value < pod_combat_actor_sup_ceiling_value
				}
				# Aggravated heals any wound (aggravated first), capped at max_hp.
				AND = {
					$STORY$ = {
						has_variable = hp_heal
						var:heal_type = flag:aggravated
					}
					pod_combat_actor_hp_value < pod_combat_actor_max_hp_value
				}
				AND = {
					$STORY$ = { has_variable = morale_heal }
					var:pod_combat_actor_morale < pod_combat_max_morale
				}
			}
		}
	}

	# Targeting validity, player turn only — the PC picked a slot via
	# pod_combat_pc_target_*. AI eval skips this (the AI swap effects
	# choose the target after move choice).
	trigger_if = {
		limit = {
			pod_combat_is_player_turn_trigger = yes
			$STORY$ = { var:targeting = flag:single_enemy }
		}
		# custom_description swaps the disabled-button tooltip to "Select
		# an enemy target" instead of raw trigger text.
		# save_temporary_scope_as captures the host before any_in_list
		# flips scope into per-slot iteration, so pod_combat_pc_target_*
		# stays readable in the loop body. Same pattern as
		# pod_combat_actor_has_alive_team_trigger in pod_combat_triggers.txt.
		save_temporary_scope_as = host
		custom_description = {
			text = pod_combat_targeting_need_enemy_tt
			has_variable = pod_combat_pc_target_team
			any_in_list = {
				variable = pod_combat_active_slots
				var:combat_team = scope:host.var:pod_combat_pc_target_team
				var:team_position = scope:host.var:pod_combat_pc_target_position
				pod_combat_slot_is_enemy_trigger = yes
				pod_combat_slot_is_alive_trigger = yes
			}
		}
	}

	trigger_if = {
		limit = {
			pod_combat_is_player_turn_trigger = yes
			$STORY$ = { var:targeting = flag:single_ally }
		}
		# Mirror of the single_enemy branch above; team_position >= 1
		# narrows to ally slots (player team minus the PC).
		save_temporary_scope_as = host
		custom_description = {
			text = pod_combat_targeting_need_ally_tt
			has_variable = pod_combat_pc_target_team
			any_in_list = {
				variable = pod_combat_active_slots
				var:combat_team = scope:host.var:pod_combat_pc_target_team
				var:team_position = scope:host.var:pod_combat_pc_target_position
				pod_combat_slot_is_player_team_trigger = yes
				var:team_position >= 1
				pod_combat_slot_is_alive_trigger = yes
			}
		}
	}

	trigger_if = {
		limit = {
			pod_combat_is_player_turn_trigger = yes
			$STORY$ = { var:targeting = flag:single_pc }
		}
		# Protect-the-PC moves target player/0 — direct pointer check
		# rather than any_in_list, since it's a single-slot probe.
		custom_description = {
			text = pod_combat_targeting_need_pc_alive_tt
			has_variable = pod_combat_pc_slot
			var:pod_combat_pc_slot = { has_variable = slot_char pod_combat_slot_is_alive_trigger = yes }
		}
	}

	trigger_if = {
		limit = {
			pod_combat_is_player_turn_trigger = yes
			$STORY$ = { var:targeting = flag:team_enemy }
		}
		# At least one alive enemy slot. No pc_target dependency → no
		# scope save (all filters are on the slot story itself).
		any_in_list = {
			variable = pod_combat_active_slots
			pod_combat_slot_is_enemy_trigger = yes
			pod_combat_slot_is_alive_trigger = yes
		}
	}

	# AI-side single_ally gate: only offer a single_ally move when an
	# ally exists to target. Without it the ally-swap finds no match,
	# the workspace stays on the enemy, and the heal/buff lands on the PC.
	trigger_if = {
		limit = {
			NOT = { pod_combat_is_player_turn_trigger = yes }
			$STORY$ = { var:targeting = flag:single_ally }
		}
		pod_combat_ai_has_available_ally_trigger = yes
	}

	# AI-side single_ally HEAL need-gate: beyond "an ally exists", only
	# pick a single_ally heal if some ally actually needs it (the
	# ally-swap picks flat among living allies). Checked against ally
	# state here rather than the actor-keyed redundancy gate in
	# pod_combat_ai_effects.txt, since the actor's own state is
	# irrelevant to an ally-targeted heal. Heal-bearing moves only.
	trigger_if = {
		limit = {
			NOT = { pod_combat_is_player_turn_trigger = yes }
			$STORY$ = {
				var:targeting = flag:single_ally
				OR = {
					has_variable = hp_heal
					has_variable = morale_heal
				}
			}
		}
		pod_combat_ai_has_ally_needing_heal_trigger = { STORY = $STORY$ }
	}

	# team_ally and self need no gate (both always include the live
	# actor). single_enemy is vacuous for the AI too: the victory check
	# ends combat when no enemies remain, so selection never runs
	# without a valid enemy.
}


######################################################
###### RESULT-CODE CONDITION TRIGGERS
######################################################
# Primitive partitions of attack_result by outcome class, read off the
# host scope. The rider-condition and retaliation gates compose against
# these so the mapping lives in one place (canonical map:
# COMBAT_REFERENCE.info §4):
#   "hp_damage" — attack_result=hp_damage
#   "contact"   — attack_result ∈ (hp_damage, guard_blocked, soaked)
#   "miss"      — attack_result ∈ (missed, dodged, stealthed)
# Each gates on outcome=attack first — attack_result is only meaningful
# then, so non-attack outcomes fail every partition by design.

# The attack landed real wound boxes on the target.
pod_combat_result_code_indicates_hp_damage_trigger = {
	has_variable = pod_combat_last_outcome
	var:pod_combat_last_outcome = flag:attack
	has_variable = pod_combat_last_attack_result
	var:pod_combat_last_attack_result = flag:hp_damage
}

# Physical contact: hp_damage plus blocked / soaked. For "fires on
# touch even if absorbed" riders (e.g. poison delivered on the strike).
pod_combat_result_code_indicates_contact_trigger = {
	has_variable = pod_combat_last_outcome
	var:pod_combat_last_outcome = flag:attack
	has_variable = pod_combat_last_attack_result
	OR = {
		var:pod_combat_last_attack_result = flag:hp_damage
		var:pod_combat_last_attack_result = flag:guard_blocked
		var:pod_combat_last_attack_result = flag:soaked
	}
}

# Evaded the attack — miss, dodge consumed, stealth auto-miss. Used by
# retaliate_if_missed and rider_condition = on_miss.
pod_combat_result_code_indicates_miss_trigger = {
	has_variable = pod_combat_last_outcome
	var:pod_combat_last_outcome = flag:attack
	has_variable = pod_combat_last_attack_result
	OR = {
		var:pod_combat_last_attack_result = flag:missed
		var:pod_combat_last_attack_result = flag:dodged
		var:pod_combat_last_attack_result = flag:stealthed
	}
}

# Miss partition plus guard_blocked: every zero-damage outcome. Gates
# the combat-log damage-source icon (no elemental marker where nothing
# landed).
#
# Excludes `soaked` deliberately: soak doesn't always zero damage, and
# even on full-soak rows we keep the source flavour (something WAS
# rolled and survived as residue / aura).
#
# Parametric on $RESULT_VAR$ — two call sites read the enum off
# different vars:
#   - host scope (move execution)  — RESULT_VAR = pod_combat_last_attack_result
#   - log-entry scope (post-action) — RESULT_VAR = attack_result
pod_combat_result_indicates_no_damage_dealt_trigger = {
	has_variable = $RESULT_VAR$
	OR = {
		var:$RESULT_VAR$ = flag:missed
		var:$RESULT_VAR$ = flag:dodged
		var:$RESULT_VAR$ = flag:stealthed
		var:$RESULT_VAR$ = flag:guard_blocked
	}
}


######################################################
###### RIDER CONDITION GATE
######################################################
# Generic gate for rider effects (heal / morale damage / self-damage)
# that piggyback on a move's HP-damage step — e.g. a bite that heals on
# a landed hit. Each rider's condition var on the move story holds one
# of: always (default when unset), on_hp_damage, on_contact, on_miss —
# resolved against the partition triggers above. Caller scope = host.
#
# Drain moves: the generator emits heal_condition = on_hp_damage for any
# `drain`-token move, so "heal only on a landed hit" flows through this
# rider machinery rather than a special case.
pod_combat_rider_condition_passes_trigger = {
	# $STORY$     = the POD_combat_move story_cycle
	# $RIDER_VAR$ = the condition variable name on the story (e.g. heal_condition)
	# Default: var unset OR var = always → pass.
	trigger_if = {
		limit = {
			OR = {
				NOT = { $STORY$ = { has_variable = $RIDER_VAR$ } }
				$STORY$ = { var:$RIDER_VAR$ = flag:always }
			}
		}
		always = yes
	}
	trigger_else_if = {
		limit = { $STORY$ = { var:$RIDER_VAR$ = flag:on_hp_damage } }
		pod_combat_result_code_indicates_hp_damage_trigger = yes
	}
	trigger_else_if = {
		limit = { $STORY$ = { var:$RIDER_VAR$ = flag:on_contact } }
		pod_combat_result_code_indicates_contact_trigger = yes
	}
	trigger_else_if = {
		limit = { $STORY$ = { var:$RIDER_VAR$ = flag:on_miss } }
		pod_combat_result_code_indicates_miss_trigger = yes
	}
	# Unknown flag value — fail closed. The generator validates against
	# RIDER_CONDITIONS, so this guards only misconfigured data.
	trigger_else = {
		always = no
	}
}


######################################################
###### AOE TEAM-SIDE FILTERS
######################################################
# Same-team / opposite-team slot filters for the AoE dispatchers in
# pod_combat_move_effects.txt.
#
# Caller contract:
#   - Implicit scope must be the iterated slot story (var:combat_team
#     lives there).
#   - scope:host must be saved to the combat host (where
#     pod_combat_current_actor_team is read).

pod_combat_aoe_team_enemy_filter_trigger = {
	NOT = { var:combat_team = scope:host.var:pod_combat_current_actor_team }
}

pod_combat_aoe_team_ally_filter_trigger = {
	# Includes the actor's own slot — team_ally heals/buffs cover self.
	var:combat_team = scope:host.var:pod_combat_current_actor_team
}
