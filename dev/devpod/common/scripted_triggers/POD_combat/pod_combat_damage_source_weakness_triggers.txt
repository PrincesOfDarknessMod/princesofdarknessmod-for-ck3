######################################################
######   POD COMBAT DAMAGE-SOURCE WEAKNESS TRIGGERS
######################################################
# Single-source-of-truth for "does damage source X exploit a weakness
# of character Y?" Used by the AI selector's weakness-exploitation
# weight modifier in pod_combat_ai_select_and_execute_move_effect to
# bias toward moves whose damage_source matches the workspace target's
# splat vulnerability.
#
# Add or change a weakness here by editing the matching per-source
# trigger. New damage sources require adding both a per-source trigger
# AND a branch in pod_combat_move_exploits_target_weakness_trigger
# below.
#
# Each per-source trigger is called on a CHARACTER scope (the target)
# and returns yes when that character takes special damage from the
# source — typically because PoD's wound pipeline promotes the wound
# to aggravated or doubles the box count. Sources that auto-aggravate
# against everyone (fire / lightning / balefire) or are universally
# superficial (cold) don't appear here because they aren't
# target-specific weaknesses; the AI's aggravated-vs-soak modifier
# already biases toward those when soak is high.
#
# Cross-reference: the concept cards in pod_combat_concepts.txt and
# their pod_combat_static_l_english.yml entries (game_concept_pod_combat_damage_*)
# describe each source's PoD behavior — keep the trigger logic and
# the player-facing concept text aligned.


# Silver-weak fera — werewolves and most shifter breeds. Ananasi,
# corax, and nuwisha aren't silver-weak; a wyrmflesh artifact also
# suppresses the weakness. Delegates to PoD's authoritative trigger
# (/common/scripted_triggers/POD_health_track_triggers.txt).
pod_combat_target_is_weak_to_silver_trigger = {
	POD_weak_to_silver_trigger = yes
}

# Gold-weak — mokole and corax. Wyrmflesh artifact suppresses.
# Delegates to PoD's authoritative trigger.
pod_combat_target_is_weak_to_gold_trigger = {
	POD_weak_to_gold_trigger = yes
}

# POD_wound_multiplier_from_splat_weaknesses
pod_combat_target_is_weak_to_sun_trigger = {
	OR = {
		has_trait = setite
		is_bushi_trigger = yes
	}
	POD_suppress_clan_curse_trigger = no
	NOT = { POD_takes_sun_damage_as_superficial = yes }
}

# POD_wound_multiplier_from_splat_weaknesses
pod_combat_target_is_weak_to_fire_trigger = {
	is_bushi_trigger = yes
	POD_suppress_clan_curse_trigger = no
}

# POD_wound_multiplier_from_splat_weaknesses
# POD_damage_type_is_aggravated
pod_combat_target_is_weak_to_holy_trigger = {
	OR = {
		
		POD_is_demon_trigger = yes
		POD_is_wraith_trigger = yes
		POD_is_vampire_trigger = yes
		#is_baali_trigger = yes
		is_kueijin_trigger = yes
	}
	POD_suppress_clan_curse_trigger = no
}


######################################################
###### BRIDGE TRIGGER (called from the AI selector)
######################################################
# Returns yes when the candidate move's damage_source matches a
# weakness of the AI's current workspace target.
#
# Called inside a modifier inside the random_in_list weight block in
# pod_combat_ai_select_and_execute_move_effect. The implicit scope at
# the call site is the iterated candidate move story_cycle, so
# `var:damage_source` reads directly off the move. The target
# character is reached via scope:combat_host.var:pod_combat_target_char.
#
# To add a new damage_source → weakness mapping: add a per-source
# trigger above, then add one AND branch to the OR below pairing the
# flag with the trigger.
pod_combat_move_exploits_target_weakness_trigger = {
	has_variable = damage_source
	OR = {
		AND = {
			var:damage_source = flag:silver
			scope:combat_host.var:pod_combat_target_char = {
				pod_combat_target_is_weak_to_silver_trigger = yes
			}
		}
		AND = {
			var:damage_source = flag:gold
			scope:combat_host.var:pod_combat_target_char = {
				pod_combat_target_is_weak_to_gold_trigger = yes
			}
		}
		AND = {
			var:damage_source = flag:sun
			scope:combat_host.var:pod_combat_target_char = {
				pod_combat_target_is_weak_to_sun_trigger = yes
			}
		}
		AND = {
			var:damage_source = flag:fire
			scope:combat_host.var:pod_combat_target_char = {
				pod_combat_target_is_weak_to_fire_trigger = yes
			}
		}
		AND = {
			var:damage_source = flag:holy
			scope:combat_host.var:pod_combat_target_char = {
				pod_combat_target_is_weak_to_holy_trigger = yes
			}
		}
	}
}
