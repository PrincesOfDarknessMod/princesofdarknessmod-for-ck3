###### CORE COMBAT STATE TRIGGERS ######

pod_combat_is_active_trigger = {
	has_variable = pod_combat_active
}

pod_combat_is_over_trigger = {
	# The pod_combat_is_over flag is set by pod_combat_check_group_victory_effect
	# whenever any of its three conditions trips: all enemies KO'd
	# (player wins), PC KO'd (player loses even with surviving allies,
	# the player has no way to continue the fight), or all allies KO'd
	# (player loses). All combats (including 1v1) go through the same
	# victory check, so this single flag is authoritative.
	has_variable = pod_combat_is_over
}

pod_combat_is_player_turn_trigger = {
	# Returns yes ONLY when the PC themselves are the current actor
	# NOT just when it's the player team's turn. Every caller
	# semantically wants "PC is acting", and the older
	# `current_turn_side = 0` check incorrectly fired on ally AI turns
	# (slots 1-4 also share player_turn_side = 0), which let ally AI
	# move selection inherit PC-only targeting / resource gates.
	has_variable = pod_combat_current_actor
	var:pod_combat_current_actor = 0
}

###### AP AFFORDABILITY TRIGGER ######

pod_combat_can_afford_ap_trigger = {
	pod_combat_current_ap_value >= $COST$
}

###### INCAPACITATED CHECK ######
# Matches any character in a state that takes them out of tactical
# combat entirely
#
# Used by the alive-check OR blocks alongside is_alive = no and
# morale <= 0 to flag a combatant as KO.
#
# Hardcoded list of the incapacitating traits, 
# kept separate from is_incapable so the minigame 
# doesn't neeed to change PoD's is_incapable contract.

pod_combat_is_incapacitated_trigger = {
	OR = {
		has_trait = incapable    
		has_trait = torpor       
		# used to catch mummies when waiting to resurrect
		has_character_modifier = trapped_in_shadow_realm_modifier
		# used to catch demons when they have been banished
		has_character_modifier = trapped_in_abyss_modifier
	}
}


###### LOADOUT TRIGGERS ######

pod_combat_no_moves_equipped_trigger = {
	# List-based loadout: the count SV iterates the list and returns the
	# number of equipped moves. 
	pod_combat_equipped_count = 0
}

###### TEAM-ALIVE TRIGGERS ######

pod_combat_ai_has_available_ally_trigger = {
	# True if the current actor has at least one alive non-self
	# same-team ally. Used by POD_can_execute_combat_move's AI gate to
	# filter single_ally moves out of the selector pool when no ally is
	# available.
	#
	# Implementation: walk pod_combat_active_slots (filled-slot list —
	# pod_combat_set_slot_char_effect adds on create,
	# pod_combat_clear_slot_stories_effect removes at combat end; KO'd
	# slots STAY in the list with slot_alive = 0 until that final
	# teardown, so callers that need "living" slots add the
	# `var:slot_alive = 1` filter explicitly, as the iteration body
	# below does). Check whether any entry is same-team and not the actor.
	# Team membership is fixed by slot index (0-4 = player team,
	# 5-9 = enemy team), matching the filter in
	# pod_combat_ai_swap_to_ally_target_effect.
	#
	# save_temporary_scope_as captures the caller scope (the combat
	# host) BEFORE any_in_list flips into per-slot iteration. Inside
	# the loop body, scope:host reaches back to read pod_combat_current_actor.
	# Same pattern as PoD's POD_has_coterie_members_present_trigger.
	save_temporary_scope_as = host
	any_in_list = {
		variable = pod_combat_active_slots
		var:slot_alive = 1
		NOT = { var:slot_index = scope:host.var:pod_combat_current_actor }
		OR = {
			AND = { scope:host = { var:pod_combat_current_actor <= 4 }  var:slot_index <= 4 }
			AND = { scope:host = { var:pod_combat_current_actor >= 5 }  var:slot_index >= 5 }
		}
	}
}
