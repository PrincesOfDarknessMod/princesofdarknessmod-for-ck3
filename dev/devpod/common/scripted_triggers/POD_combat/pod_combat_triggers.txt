###### CORE COMBAT STATE TRIGGERS ######

pod_combat_is_active_trigger = {
	has_variable = pod_combat_active
}

pod_combat_is_over_trigger = {
	# The pod_combat_is_over flag is set by pod_combat_check_group_victory_effect
	# when an entire team is wiped. All combats (including 1v1) go through the
	# same victory check, so this single flag is authoritative.
	has_variable = pod_combat_is_over
}

pod_combat_is_player_turn_trigger = {
	has_variable = pod_combat_current_turn_side
	var:pod_combat_current_turn_side = 0
}

###### AP AFFORDABILITY TRIGGER ######

pod_combat_can_afford_ap_trigger = {
	pod_combat_current_ap_value >= $COST$
}

###### INCAPACITATED CHECK ######
# Matches any character in a state that takes them out of tactical
# combat entirely — severe wounds, vampire torpor, wight withering,
# wraith Shadow domination, kuei-jin pre-dharma, or zombified state.
# Used by the alive-check OR blocks alongside is_alive = no and
# morale <= 0 to flag a combatant as KO.
#
# Hardcoded list of the incapacitating traits. As is, it can be replaced
# with is_incapable. Hard coded currently since we might actually
# want to allow wight, dharma0, shadow, & zombu partipate but that is 
# left for a future decision.

pod_combat_is_incapacitated_trigger = {
	OR = {
		has_trait = incapable    # vanilla — severe-wound result
		has_trait = torpor       # PoD — vampire coma
		has_trait = wight        # PoD — bloodless husk
		has_trait = dharma0      # PoD — kuei-jin pre-dharma state
		has_trait = shadow       # PoD — wraith Shadow-dominated
		has_trait = zombu        # PoD — zombie / necromancy thrall
	}
}

###### TEAM-ALIVE TRIGGERS ######

pod_combat_ai_has_available_ally_trigger = {
	# True if the current actor has at least one alive non-self
	# same-team ally available as a single-ally target. Slot story
	# read pattern: outer has_variable on the pointer var, then
	# chained `var:pod_combat_slot_$N$ = { has_variable = slot_char
	# var:slot_alive = 1 }` for the filled-and-alive gate.
	OR = {
		AND = {
			has_variable = pod_combat_slot_0
			var:pod_combat_slot_0 = { has_variable = slot_char var:slot_alive = 1 }
			var:pod_combat_current_actor <= 4
			NOT = { var:pod_combat_current_actor = 0 }
		}
		AND = {
			has_variable = pod_combat_slot_1
			var:pod_combat_slot_1 = { has_variable = slot_char var:slot_alive = 1 }
			var:pod_combat_current_actor <= 4
			NOT = { var:pod_combat_current_actor = 1 }
		}
		AND = {
			has_variable = pod_combat_slot_2
			var:pod_combat_slot_2 = { has_variable = slot_char var:slot_alive = 1 }
			var:pod_combat_current_actor <= 4
			NOT = { var:pod_combat_current_actor = 2 }
		}
		AND = {
			has_variable = pod_combat_slot_3
			var:pod_combat_slot_3 = { has_variable = slot_char var:slot_alive = 1 }
			var:pod_combat_current_actor <= 4
			NOT = { var:pod_combat_current_actor = 3 }
		}
		AND = {
			has_variable = pod_combat_slot_4

			var:pod_combat_current_actor <= 4
