###### CORE COMBAT STATE TRIGGERS ######

# Character scope. Includes proxy player; use pod_combat_is_seated_trigger
# for checks that require participation.
pod_combat_is_active_trigger = {
	trigger_if = {
		limit = { has_variable = pod_combat_session }
		var:pod_combat_session ?= { has_variable = pod_combat_active }
	}
	trigger_else = { always = no }
}

# Character scope. Checks roster membership, excluding unseated proxy player's chracter.
# Save the character before the slot iteration; this inside the loop is the slot.
pod_combat_is_seated_trigger = {
	save_temporary_scope_as = pod_seat_asker
	trigger_if = {
		limit = { has_variable = pod_combat_session }
		var:pod_combat_session ?= {
			any_in_list = {
				variable = pod_combat_active_slots
				var:slot_char ?= { this = scope:pod_seat_asker }
			}
		}
	}
	trigger_else = { always = no }
}

pod_combat_is_over_trigger = {
	# End flag set by pod_combat_check_group_victory_effect.
	trigger_if = {
		limit = { has_variable = pod_combat_session }
		var:pod_combat_session ?= { has_variable = pod_combat_is_over }
	}
	trigger_else = { always = no }
}

# Host scope, from the shell option with its inherited sc_* scopes.
# Reject cards orphaned by host death so they cannot dispatch a later fight
# with stale scopes. Compare through the live scope; the old story has ended.
pod_combat_shell_owns_live_fight_trigger = {
	exists = scope:sc_attacker
	exists = scope:sc_defender
	# Keep identity checks inside the scope guard: the card can outlive its fight.
	var:pod_combat_session ?= {
		has_variable = pod_combat_active
		var:attacker = scope:sc_attacker
		var:defender = scope:sc_defender
	}
}

# Turn-engine roster check. GUI gates use the hand-off latch instead.
# Nest the actor-presence check so the matcher cannot dereference an unset actor.
pod_combat_current_actor_is_controlled_trigger = {
	save_temporary_scope_as = pod_cac_host
	trigger_if = {
		limit = { var:pod_combat_session ?= { has_variable = pod_combat_current_actor_team } }
		var:pod_combat_session ?= {
			any_in_list = {
				variable = pod_combat_active_slots
				var:combat_team = scope:pod_cac_host.var:pod_combat_session.var:pod_combat_current_actor_team
				var:team_position = scope:pod_cac_host.var:pod_combat_session.var:pod_combat_current_actor_position
				has_variable = slot_human_controller
			}
		}
	}
	trigger_else = { always = no }
}

# Tests human control, regardless of which character owns the turn.
# Command buttons must use pod_combat_is_my_turn_trigger instead.
# Avoid dereferencing the control slot here: it can be unset between turns.
pod_combat_is_player_turn_trigger = {
	var:pod_combat_session ?= {
		has_variable = pod_combat_control_turn_live
	}
}

# Character scope; root is the clicking character in a scripted_gui.
# Keep character ownership checks out of turn flow to avoid character-dependent results.
# The engine uses pod_combat_current_actor_is_controlled_trigger.
pod_combat_is_my_turn_trigger = {
	save_temporary_scope_as = pod_my_turn_asker
	trigger_if = {
		limit = { var:pod_combat_session ?= { has_variable = pod_combat_control_turn_live } }
		var:pod_combat_session.var:pod_combat_active_control_slot ?= {
			var:slot_human_controller = scope:pod_my_turn_asker
		}
	}
	trigger_else = { always = no }
}

# Character scope. Tests the acting character, which can differ from a player character.
# The control-slot scope is written with the turn latch and outlives it.
pod_combat_char_is_live_control_actor_trigger = {
	save_temporary_scope_as = pod_control_actor_asker
	trigger_if = {
		limit = { var:pod_combat_session ?= { has_variable = pod_combat_control_turn_live } }
		var:pod_combat_session.var:pod_combat_active_control_slot ?= {
			var:slot_char = { this = scope:pod_control_actor_asker }
		}
	}
	trigger_else = { always = no }
}

# Host scope. GUI Advance gate; the hand-off latch avoids a roster scan.
pod_combat_can_advance_trigger = {
	pod_combat_is_active_trigger = yes
	NOT = { pod_combat_is_over_trigger = yes }
	var:pod_combat_session ?= {
		has_variable = pod_combat_current_actor_team
		NOT = { has_variable = pod_combat_control_turn_live }
	}
}

# Host scope. Round settlement waits for every living controlled slot to act.
# Exclude KO slots: initiative resets slot_has_acted only for living slots.
pod_combat_pc_has_acted_this_round_trigger = {
	save_temporary_scope_as = pod_has_acted_host
	trigger_if = {
		limit = { has_variable = pod_combat_session }
		var:pod_combat_session ?= {
			NOT = {
				any_in_list = {
					variable = pod_combat_active_slots
					has_variable = slot_human_controller
					pod_combat_slot_is_alive_trigger = yes
					pod_combat_slot_has_not_acted_trigger = yes
				}
			}
		}
	}
	trigger_else = { always = no }
}

###### DUEL RULE: SUMMONING ######
# Applies to summons_combatant moves on both sides, not event-driven arrivals.
# The builder writes var:summons from literal flags to preserve its type.
# Scripted triggers support fight, participant-character and slot-story callers.

pod_combat_fight_bans_summons_trigger = {
	# Fight-story scope.
	trigger_if = {
		limit = { has_variable = summons }
		var:summons = flag:banned
	}
	# Missing fight/rule allows summoning, including tooltip builds.
	trigger_else = { always = no }
}

pod_combat_duel_bans_summons_trigger = {
	# Participant scope.
	trigger_if = {
		limit = { has_variable = pod_combat_session }
		var:pod_combat_session ?= { pod_combat_fight_bans_summons_trigger = yes }
	}
	trigger_else = { always = no }
}

pod_combat_slot_summons_allowed_trigger = {
	# Slot scope.
	trigger_if = {
		limit = { has_variable = fight }
		NOT = { var:fight ?= { pod_combat_fight_bans_summons_trigger = yes } }
	}
	trigger_else = { always = yes }
}

# Host scope. Nested gates reject successful/rejected picks before the roster proof
# and guard every variable read against absence.
pod_combat_ai_retry_context_valid_trigger = {
	trigger_if = {
		limit = { has_variable = pod_combat_session }
		var:pod_combat_session ?= {
			trigger_if = {
				limit = {
					has_variable = pod_combat_active
					NOT = { has_variable = pod_combat_is_over }
					NOT = { has_variable = pod_combat_ai_picked }
					NOT = { has_variable = pod_combat_ai_action_executed }
					NOT = { has_variable = pod_combat_turn_control_fizzled }
					has_variable = pod_combat_ai_admitted_n
					has_variable = pod_combat_ai_intent
					has_variable = pod_combat_current_ap
					has_variable = pod_combat_actor_char
					has_variable = pod_combat_target_char
					has_variable = pod_combat_current_actor_team
					has_variable = pod_combat_current_actor_position
					has_variable = pod_combat_selected_target_team
					has_variable = pod_combat_selected_target_position
					has_variable = pod_combat_actor_morale
					has_variable = pod_combat_actor_hit_chance
					has_variable = pod_combat_target_morale
					has_variable = pod_combat_target_hit_chance
				}
				trigger_if = {
					limit = {
						var:pod_combat_ai_admitted_n = 0
						exists = var:pod_combat_actor_char
						exists = var:pod_combat_target_char
						var:pod_combat_current_ap >= 0
						NOR = {
							var:pod_combat_ai_intent = flag:none
							var:pod_combat_ai_intent = flag:steal
						}
					}
					var:pod_combat_actor_char ?= { is_alive = yes }
					var:pod_combat_target_char ?= { is_alive = yes }
					scope:host = { NOT = { pod_combat_current_actor_is_down_trigger = yes } }
				}
				trigger_else = { always = no }
			}
			trigger_else = { always = no }
		}
	}
	trigger_else = { always = no }
}

# Slot scope. Typed equality preserves retaliation damage-kind flags; never lift
# them through save_scope_value_as. Absent on both sides is conservatively equal.
pod_combat_ai_retry_value_matches_trigger = {
	save_temporary_scope_as = pod_ai_retry_value_slot
	trigger_if = {
		limit = { scope:host = { has_variable = pod_combat_session } }
		scope:host.var:pod_combat_session ?= {
			trigger_if = {
				limit = { has_variable = $WORKSPACE_FIELD$ }
				trigger_if = {
					limit = { scope:pod_ai_retry_value_slot = { has_variable = $SLOT_FIELD$ } }
					var:$WORKSPACE_FIELD$ = scope:pod_ai_retry_value_slot.var:$SLOT_FIELD$
				}
				trigger_else = { always = no }
			}
			trigger_else = { scope:pod_ai_retry_value_slot = { NOT = { has_variable = $SLOT_FIELD$ } } }
		}
	}
	trigger_else = { always = no }
}

# Host scope, after the ordinary save. Consume ready before entering either retry.
pod_combat_ai_can_reuse_preparation_trigger = {
	trigger_if = {
		limit = { exists = scope:pod_ai_retry_ready }
		trigger_if = {
			limit = {
				scope:pod_ai_retry_ready = 1
				exists = scope:pod_ai_retry_fight
				exists = scope:pod_ai_retry_actor
				exists = scope:pod_ai_retry_target
				exists = scope:pod_ai_retry_ap
				has_variable = pod_combat_session
			}
			pod_combat_ai_retry_context_valid_trigger = yes
			var:pod_combat_session ?= {
				this = scope:pod_ai_retry_fight
				trigger_if = {
					limit = {
						has_variable = pod_combat_actor_char
						has_variable = pod_combat_target_char
						has_variable = pod_combat_current_ap
					}
					var:pod_combat_actor_char = scope:pod_ai_retry_actor
					var:pod_combat_target_char = scope:pod_ai_retry_target
					var:pod_combat_current_ap = scope:pod_ai_retry_ap
				}
				trigger_else = { always = no }
			}
		}
		trigger_else = { always = no }
	}
	trigger_else = { always = no }
}

	trigger_if = {

		var:pod_combat_session = {

# Host scope; requires scope:ai_candidate_move and scope:ai_actor.
# Run after intent selection, target-summary precompute and actor workspace load.
# Missing prerequisites can skip trigger_if gates and silently admit moves.
# pod_combat_ai_score_candidate_effect caches this result as pod_combat_ai_admitted.
# The intent counter runs before selection, so it retains a separate membership test.
pod_combat_ai_move_admissible_trigger = {
	# Basic usability comes from full preparation or a proven unchanged retry.
	# No gameplay mutation may intervene before admission; see the tier precompute.
	scope:ai_candidate_move = { has_variable = pod_combat_ai_usable_now }

	# Stolen turns permit only single-target offense.
	trigger_if = {
		limit = { var:pod_combat_session ?= { has_variable = pod_combat_turn_is_controlled } }
		scope:ai_candidate_move = { var:targeting = flag:single_enemy }
		scope:ai_candidate_move = {
			OR = {
				has_variable = hp_damage
				has_variable = morale_damage
				has_variable = dot_amount
				has_variable = ap_drain
				has_variable = initiative_debuff
				has_variable = remove_guard
				has_variable = soak_debuff_round_tier
				has_variable = soak_debuff_combat_tier
				has_variable = hit_debuff_round_tier
				has_variable = hit_debuff_combat_tier
			}
		}
		# Prevent chained domination and summons joining the actor's original team.
		NOR = {
			scope:ai_candidate_move = { has_variable = mind_control }
			scope:ai_candidate_move = { has_variable = summons_combatant }
		}
	}
	trigger_if = {
		limit = { scope:ai_candidate_move = { has_variable = mind_control } }
		pod_combat_ai_mind_control_is_useful_trigger = { STORY = scope:ai_candidate_move }
	}
	trigger_if = {
		limit = { scope:ai_candidate_move = { has_variable = ap_drain } }
		pod_combat_ai_ap_drain_is_useful_trigger = { STORY = scope:ai_candidate_move }
	}

	# single_pc needs a loadable PC slot: without one the swap is a no-op and the
	# move would fire at the still-selected enemy.
	trigger_if = {
		limit = { scope:ai_candidate_move = { var:targeting = flag:single_pc } }
		pod_combat_pc_slot_is_loadable_trigger = yes
	}

	# Exclude redundant pure DoTs; hybrids retain their other payload and take a penalty.
	# Target summaries are initialized by the per-action precompute.
	trigger_if = {
		limit = {
			scope:ai_candidate_move = {
				has_variable = dot_amount
				NOR = { has_variable = hp_damage  has_variable = morale_damage  has_variable = initiative_debuff  has_variable = ap_drain  has_variable = mind_control  has_variable = remove_guard has_variable = soak_debuff_round_tier  has_variable = soak_debuff_combat_tier  has_variable = hit_debuff_round_tier  has_variable = hit_debuff_combat_tier }
			}
		}
		OR = {
			# HP and morale DoTs are not interchangeable; compare strength only within a kind.
			AND = {
				scope:ai_candidate_move = { has_variable = dot_morale }
				var:pod_combat_session ?= { var:pod_combat_ai_tgt_dot_morale = 0 }
			}
			AND = {
				scope:ai_candidate_move = { NOT = { has_variable = dot_morale } }
				var:pod_combat_session ?= { var:pod_combat_ai_tgt_dot_morale = 1 }
			}
			var:pod_combat_session ?= { var:pod_combat_ai_tgt_dot_amount < scope:ai_candidate_move.var:dot_amount }
			var:pod_combat_session ?= { var:pod_combat_ai_tgt_dot_rounds < scope:ai_candidate_move.var:dot_rounds }
		}
	}

	# AI-only heal gate. Shared dynamic gates handle tier/stance redundancy.
	# Keep the usefulness tests aligned with the heal scoring arms.
	trigger_if = {
		limit = {
			scope:ai_candidate_move = {
				OR = {
					has_variable = hp_heal
					has_variable = morale_heal
				}
				NOT = { has_variable = hp_damage }
				NOT = { has_variable = morale_damage }
				# Ally heals test recipient slots separately, including the actor for team_ally.
				NOT = { var:targeting = flag:single_ally }
				NOT = { var:targeting = flag:team_ally }
			}
		}
		OR = {
			AND = {
				scope:ai_candidate_move = {
					has_variable = hp_heal
					var:heal_type = flag:superficial
				}
				var:pod_combat_session ?= { var:pod_combat_ai_actor_hp < var:pod_combat_ai_actor_sup_ceiling }
			}
			AND = {
				scope:ai_candidate_move = {
					has_variable = hp_heal
					var:heal_type = flag:aggravated
				}
				var:pod_combat_session ?= { var:pod_combat_ai_actor_hp < var:pod_combat_ai_actor_max_hp }
			}
			# Keep this comparison host-scoped: the max-morale SV follows the session
			# pointer itself. The workspace mirror is absent before the first load.
			AND = {
				scope:ai_candidate_move = { has_variable = morale_heal }
				trigger_if = {
					limit = { var:pod_combat_session ?= { has_variable = pod_combat_actor_morale } }
					var:pod_combat_session.var:pod_combat_actor_morale < pod_combat_actor_max_morale_value
				}
				trigger_else = { always = no }
			}
		}
	}

	# Team heals include the actor; use the slot-scoped healing benefit trigger.
	trigger_if = {
		limit = {
			scope:ai_candidate_move = {
				OR = {
					has_variable = hp_heal
					has_variable = morale_heal
				}
				NOT = { has_variable = hp_damage }
				NOT = { has_variable = morale_damage }
				var:targeting = flag:team_ally
			}
		}
		pod_combat_ai_has_team_member_needing_heal_trigger = { STORY = scope:ai_candidate_move }
	}

	###### TACTICAL INTENT DISPATCH ######
	# Membership follows payload, not authored category. Keep this list aligned with
	# pod_combat_ai_fold_usable_move_effect and the upstream membership verifier.
	# Control waste is already filtered above; the counter fold must test it itself.
	# Zero-AP moves remain available under every intent.
	trigger_if = {
		limit = { var:pod_combat_session ?= { var:pod_combat_ai_intent = flag:pressure } }
		scope:ai_candidate_move = {
			OR = {
				var:ap_cost = 0
				has_variable = hp_damage
				AND = {
					has_variable = dot_amount
					NOT = { has_variable = dot_morale }
				}
			}
		}
	}
	trigger_else_if = {
		limit = { var:pod_combat_session ?= { var:pod_combat_ai_intent = flag:break } }
		scope:ai_candidate_move = {
			OR = {
				var:ap_cost = 0
				has_variable = morale_damage
				AND = {
					has_variable = dot_amount
					has_variable = dot_morale
				}
			}
		}
	}
	trigger_else_if = {
		limit = { var:pod_combat_session ?= { var:pod_combat_ai_intent = flag:setup } }
		scope:ai_candidate_move = {
			OR = {
				var:ap_cost = 0
				has_variable = soak_buff_round_tier
				has_variable = soak_buff_combat_tier
				has_variable = soak_debuff_round_tier
				has_variable = soak_debuff_combat_tier
				has_variable = hit_buff_round_tier
				has_variable = hit_buff_combat_tier
				has_variable = hit_debuff_round_tier
				has_variable = hit_debuff_combat_tier
				has_variable = mind_control
				has_variable = ap_drain
				has_variable = initiative_debuff
				has_variable = remove_guard
				has_variable = bypass_dodge
				has_variable = bypass_stealth
				has_variable = drain
				has_variable = summons_combatant
				has_variable = steal_gold
				has_variable = gain_dread
				has_variable = energy_gain
				has_variable_list = bonus_clauses
				has_variable = marker_modifier
				has_variable = remove_marker
				has_variable = retaliate_damage
				has_variable = retaliate_morale
			}
		}
	}
	trigger_else_if = {
		limit = { var:pod_combat_session ?= { var:pod_combat_ai_intent = flag:stabilize } }
		scope:ai_candidate_move = {
			OR = {
				var:ap_cost = 0
				has_variable = hp_heal
				has_variable = morale_heal
				has_variable = set_guard
				has_variable = dodge
				has_variable = stealth
				has_variable = soak_buff_round_tier
				has_variable = soak_buff_combat_tier
			}
		}
	}
	# Unrestricted intents and a locked pass admit the full pool.
	trigger_else = { always = yes }
}

						AND = {
							is_ruler = no

						AND = {
							is_ruler = no
							exists = host
							scope:sc_defender.host = { government_has_flag = government_is_tribal }
						}
					}
				}
			}
		}
	}

# Move-story scope; $ACTOR$ = viewer. Display filter only; unlocks use

###### MID-FIGHT DELIRIUM ######
# Scripted triggers for pod_combat_delirium_effects.txt, which re-derives PoD's Delirium
# when a combatant changes form after slot init has already run the knight effects.

pod_combat_char_has_any_delirium_trigger = {
	# Character scope. Existing Delirium prevents repeat affliction and log rows.
	OR = {
		has_character_modifier = POD_delirium_modifier
		has_character_modifier = POD_reduced_delirium_modifier
	}
}

pod_combat_delirium_caster_ready_trigger = {
	# Character scope. Skip the roster scan without an inducing form; PoD rechecks each victim.
	is_alive = yes
	POD_inflicts_delirium_trigger = yes
}

pod_combat_slot_delirium_candidate_trigger = {
	# Slot scope. PoD owns immunity and bypass checks. Test character liveness,
	# not slot KO: init also afflicts living characters with full wound tracks.
	pod_combat_slot_char_is_alive_trigger = yes
	NOT = { var:slot_char = { pod_combat_char_has_any_delirium_trigger = yes } }
}
