###### CORE COMBAT STATE TRIGGERS ######

pod_combat_is_active_trigger = {
	has_variable = pod_combat_active
}

pod_combat_is_over_trigger = {
	# Authoritative end flag — set by pod_combat_check_group_victory_effect; all combats route through it.
	has_variable = pod_combat_is_over
}

pod_combat_is_player_turn_trigger = {
	# PC is the current actor specifically, not just the player team's turn — ally AI shares the team side.
	has_variable = pod_combat_current_actor_team
	var:pod_combat_current_actor_team = flag:player
	var:pod_combat_current_actor_position = 0
}

# Gates the mid-step Advance entry point. Advanceable
# iff combat is live, not over, an actor is pending, and that actor is NOT the PC
# (any AI, including an ally AI at player-team position > 0). Host (PC) scope.
# The NOR wraps the single PC-match AND: a multi-child NOT would parse as an
# OR-of-children, so the team+position pair must sit inside one AND under NOR.
pod_combat_can_advance_trigger = {
	pod_combat_is_active_trigger = yes
	NOT = { has_variable = pod_combat_is_over }
	has_variable = pod_combat_current_actor_team
	NOR = {
		AND = {
			var:pod_combat_current_actor_team = flag:player
			var:pod_combat_current_actor_position = 0
		}
	}
}

# True once the PC has committed its turn this round
# (the prologue in process_group_turns set the PC slot's slot_has_acted). Routes
# pod_combat_advance_turn between the in-round settle (PC already acted -> draining
# post-PC AI) and the pre-PC settle (PC not yet acted -> combat-start/round-start
# sequence). Host (PC) scope; pod_combat_pc_slot is the player/0 slot story.
pod_combat_pc_has_acted_this_round_trigger = {
	has_variable = pod_combat_pc_slot
	var:pod_combat_pc_slot = { var:slot_has_acted = 1 }
}

###### AP AFFORDABILITY TRIGGER ######

pod_combat_can_afford_ap_trigger = {
	pod_combat_current_ap_value >= $COST$
}

###### INCAPACITATED CHECK ######
# Separate from is_incapable so the minigame doesn't have to alter PoD's is_incapable contract.

pod_combat_is_incapacitated_trigger = {
	OR = {
		has_trait = incapable
		has_trait = torpor
		has_character_modifier = trapped_in_shadow_realm_modifier	# mummy awaiting resurrection
		has_character_modifier = trapped_in_abyss_modifier			# banished demon
	}
}


###### LOADOUT TRIGGERS ######

pod_combat_no_moves_equipped_trigger = {
	pod_combat_equipped_count = 0
}

###### TEAM-ALIVE TRIGGERS ######

pod_combat_ai_has_available_ally_trigger = {
	# Actor has an alive non-self same-team ally — gates single_ally moves out of the AI pool when none exists.
	# save_temporary_scope_as captures the host BEFORE any_in_list flips scope to per-slot, so the body
	# can reach back via scope:host for the actor's team/position. (KO'd slots linger in the list, hence the explicit alive filter.)
	save_temporary_scope_as = host
	any_in_list = {
		variable = pod_combat_active_slots
		pod_combat_slot_is_alive_trigger = yes
		NOT = {
			AND = {
				var:combat_team = scope:host.var:pod_combat_current_actor_team
				var:team_position = scope:host.var:pod_combat_current_actor_position
			}
		}
		var:combat_team = scope:host.var:pod_combat_current_actor_team
	}
}

pod_combat_ai_has_ally_needing_heal_trigger = {
	# $STORY$ = a single_ally heal move. True if some alive non-self ally would actually benefit (below full HP /
	# below morale cap), so the AI doesn't pick an ally-heal nobody needs — actor self-state is irrelevant here.
	# scope:host capture mirrors pod_combat_ai_has_available_ally_trigger above.
	save_temporary_scope_as = host
	any_in_list = {
		variable = pod_combat_active_slots
		pod_combat_slot_is_alive_trigger = yes
		NOT = {
			AND = {
				var:combat_team = scope:host.var:pod_combat_current_actor_team
				var:team_position = scope:host.var:pod_combat_current_actor_position
			}
		}
		var:combat_team = scope:host.var:pod_combat_current_actor_team
		OR = {
			AND = {
				$STORY$ = { has_variable = hp_heal }
				var:slot_char = { POD_empty_damage_boxes < POD_health_track_length }
			}
			AND = {
				$STORY$ = { has_variable = morale_heal }
				var:slot_morale < pod_combat_max_morale
			}
		}
	}
}

pod_combat_slot_should_be_marked_ko_trigger = {
	# Slot-scoped. Shared KO predicate for all post-action sites.
	# Reads var:slot_morale, so callers MUST flush post-action morale to the slot field first.
	# KO sources: death, incapacitation, wound-track-full (non-fatal duels where is_alive stays yes), morale-broken.
	OR = {
		var:slot_char = { is_alive = no }
		var:slot_char = { pod_combat_is_incapacitated_trigger = yes }
		var:slot_char = { POD_total_damage_boxes >= POD_health_track_length }
		var:slot_morale <= 0
	}
}

###### SLOT PREDICATE TRIGGERS ######
# Slot-scoped: the implicit scope must be a POD_combat_slot story (inside an
# every_in_list/any_in_list over pod_combat_active_slots, or a
# var:pod_combat_*_slot = { } deref). Encapsulating the identity/state vars
# keeps call sites readable and confines a future identity-model change here.

pod_combat_slot_is_alive_trigger = {
	var:slot_alive = 1
}

pod_combat_slot_is_ko_trigger = {
	# Stored flag — distinct from pod_combat_slot_should_be_marked_ko_trigger,
	# which recomputes from char + morale at writeback.
	var:slot_alive = 0
}

pod_combat_slot_is_enemy_trigger = {
	var:combat_team = flag:enemy
}

pod_combat_slot_is_player_team_trigger = {
	# The whole player side (PC + allies), not the PC alone.
	var:combat_team = flag:player
}

pod_combat_slot_is_pc_trigger = {
	var:combat_team = flag:player
	var:team_position = 0
}

pod_combat_slot_is_leader_trigger = {
	# Position 0 of either team.
	var:team_position = 0
}

pod_combat_slot_has_not_acted_trigger = {
	var:slot_has_acted = 0
}
