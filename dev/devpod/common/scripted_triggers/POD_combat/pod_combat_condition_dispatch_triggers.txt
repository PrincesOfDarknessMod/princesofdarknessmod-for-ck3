# Sub-dispatches for the identifier condition predicates: a switch on the
# move's cond_param selects the literal trait / secret_type key. Called from
# pod_combat_move_condition_met_trigger. Unknown cond_param -> always=no.
#   $STORY$ = the POD_combat_move story_cycle (carries cond_param)
pod_combat_cond_has_trait_trigger = {
	switch = {
		trigger = $STORY$.var:cond_param
		flag:brave = { has_trait = brave }
		fallback = { always = no }
	}
}
pod_combat_cond_known_secret_trigger = {
	switch = {
		trigger = $STORY$.var:cond_param
		flag:secret_diablerist = { any_known_secret = { secret_type = secret_diablerist  secret_target ?= root.var:pod_combat_target_char } }
		fallback = { always = no }
	}
}
