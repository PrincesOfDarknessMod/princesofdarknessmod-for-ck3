# Sub-dispatches for the identifier condition predicates: a switch on the
# move's cond_param selects the literal trait / secret_type / modifier key.
# Called from pod_combat_move_condition_met_trigger. Unknown cond_param -> always=no.
#   $STORY$ = the POD_combat_move story_cycle (carries cond_param)
pod_combat_cond_has_trait_trigger = {
	switch = {
		trigger = $STORY$.var:cond_param
		flag:vampire = { has_trait = vampire }
		flag:wraith = { has_trait = wraith }
		flag:spirit = { has_trait = spirit }
		flag:psychic = { has_trait = psychic }
		flag:brave = { has_trait = brave }
		fallback = { always = no }
	}
}
