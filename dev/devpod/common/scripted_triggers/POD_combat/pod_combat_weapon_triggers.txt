# Reads of the actor's equipped weapon, used by the uses_weapon override
# path. The shared dataset of artifacts that force aggravated damage lives
# one level up in POD_weapon_artifact_triggers.txt (PoD core reads it too).
# System internals: COMBAT_REFERENCE.info section 2.

# Host scope. $STORY$ = the move story. True when this action swings the
# actor's own weapon: either the static uses_weapon flag, or the uses_weapon
# conditional bonus granted by a met clause at execution §2b.
pod_combat_move_swings_weapon_trigger = {
	trigger_if = {
		limit = { $STORY$ = { has_variable = uses_weapon } }
		always = yes
	}
	trigger_else_if = {
		# ?= does not skip a block in a TRIGGER, so the nested has_variable is
		# the guard: with no fight in scope this arm simply answers no.
		limit = { var:pod_combat_session ?= { has_variable = pod_combat_bonus_uses_weapon } }
		always = yes
	}
	trigger_else = { always = no }
}

# Character scope. can_benefit_from_artifact keeps a weapon the character
# cannot actually wield (wrong culture / faith / body) from empowering
# anything - the older artifact-aggravation list predates that check and
# still omits it, so the two disagree for an unusable weapon. Deliberate:
# tightening the old list would change non-combat wounds.
pod_combat_actor_is_armed_trigger = {
	any_equipped_character_artifact = {
		artifact_slot_type = primary_armament
		prev = { can_benefit_from_artifact = prev }
	}
}

# Character scope. True when the equipped weapon carries ANY override the
# uses_weapon path can act on. The list itself lives in
# POD_weapon_artifact_has_override_trigger so this, the tooltip heading and the
# tooltip's trailing break all read one copy; recognising a weapon by NAME
# anywhere in this path is what the var-only pass removed, so a new weapon
# needs one set_variable at its creation effect and no edit here. Used by the
# move card and by the AI to value a weapon move it can actually empower.
pod_combat_actor_weapon_has_override_trigger = {
	any_equipped_character_artifact = {
		artifact_slot_type = primary_armament
		prev = { can_benefit_from_artifact = prev }
		POD_weapon_artifact_has_override_trigger = yes
	}
}

# Host scope. $VICTIM$ = the character about to be wounded.
# True when the equipped weapon's retype would land on a splat that is IMMUNE
# to that element, which deals literally nothing - a sun-forged bow against a
# werewolf, say. Without this the weapon would be strictly worse than no weapon
# at all, since the move's own superficial damage is what it replaced. The
# dispatch falls back to the authored class whenever this answers yes.
# Self-guarding: answers no when no retype is live, so callers need not order
# their limit terms (neither AND nor OR short-circuits in this engine).
pod_combat_weapon_retype_is_wasted_trigger = {
	trigger_if = {
		limit = { var:pod_combat_session ?= { has_variable = pod_combat_weapon_class } }
		OR = {
			AND = {
				var:pod_combat_session ?= { var:pod_combat_weapon_class = flag:fire }
				$VICTIM$ = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_fire } }
			}
			AND = {
				var:pod_combat_session ?= { var:pod_combat_weapon_class = flag:cold }
				$VICTIM$ = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_cold } }
			}
			AND = {
				var:pod_combat_session ?= { var:pod_combat_weapon_class = flag:lightning }
				$VICTIM$ = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_lightning } }
			}
			AND = {
				var:pod_combat_session ?= { var:pod_combat_weapon_class = flag:sun }
				$VICTIM$ = {
					OR = {
						POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_sun }
						# Ash-and-Fire Kuei-jin: not immune, but the packet is capped at 1
						# box (POD_damage_cap_per_death_reason) AND downgraded to
						# superficial, so the retype is strictly worse than the plain duel
						# damage it replaced. The parent perk alone caps without the
						# downgrade (one aggravated box) - a trade, not waste, so it is
						# deliberately not tested here.
						POD_takes_sun_damage_as_superficial = yes
					}
				}
			}
			AND = {
				var:pod_combat_session ?= { var:pod_combat_weapon_class = flag:silver }
				$VICTIM$ = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_silver } }
			}
			AND = {
				var:pod_combat_session ?= { var:pod_combat_weapon_class = flag:gold }
				$VICTIM$ = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_gold } }
			}
			AND = {
				var:pod_combat_session ?= { var:pod_combat_weapon_class = flag:holy }
				$VICTIM$ = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_true_faith } }
			}
			AND = {
				var:pod_combat_session ?= { var:pod_combat_weapon_class = flag:balefire }
				$VICTIM$ = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_balefire } }
			}
			AND = {
				var:pod_combat_session ?= { var:pod_combat_weapon_class = flag:poison }
				$VICTIM$ = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_poison } }
			}
			AND = {
				var:pod_combat_session ?= { var:pod_combat_weapon_class = flag:cold_iron }
				$VICTIM$ = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_cold_iron } }
			}
		}
	}
	trigger_else = { always = no }
}

# Host scope. The deferred-fire twin of the trigger above, for a kind that was
# STAMPED earlier rather than read live. A counter's kind is set at arm time
# from the counter owner's weapon, but the counter fires during a LATER
# attacker's dispatch - by then var:pod_combat_weapon_class belongs to the
# attacker, so the live trigger above would test the wrong weapon. Same ten
# arms, same self-guarding shape; a morale or aggravated kind matches no arm
# and answers no.
# It also serves the change_class element, which §2b stamps on the fight story
# as pod_combat_bonus_class. Because the kind is resolved through the session
# pointer, every caller must invoke this from HOST scope: from a slot or a move
# story there is no pointer, so it would answer "not wasted" for every victim.
# $KIND_VAR$ = the FIGHT var holding the stamped kind, reached through the
# host's session pointer. $VICTIM$ = who is about to be wounded.
pod_combat_stamped_type_is_wasted_trigger = {
	trigger_if = {
		limit = { var:pod_combat_session ?= { has_variable = $KIND_VAR$ } }
		OR = {
			AND = {
				var:pod_combat_session.var:$KIND_VAR$ = flag:fire
				$VICTIM$ = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_fire } }
			}
			AND = {
				var:pod_combat_session.var:$KIND_VAR$ = flag:cold
				$VICTIM$ = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_cold } }
			}
			AND = {
				var:pod_combat_session.var:$KIND_VAR$ = flag:lightning
				$VICTIM$ = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_lightning } }
			}
			AND = {
				var:pod_combat_session.var:$KIND_VAR$ = flag:sun
				$VICTIM$ = {
					OR = {
						POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_sun }
						# Same cap-plus-downgrade case as the trigger above.
						POD_takes_sun_damage_as_superficial = yes
					}
				}
			}
			AND = {
				var:pod_combat_session.var:$KIND_VAR$ = flag:silver
				$VICTIM$ = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_silver } }

			AND = {

				$VICTIM$ = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_gold } }
			}
			AND = {
				var:pod_combat_session.var:$KIND_VAR$ = flag:holy
				$VICTIM$ = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_true_faith } }
			}
			AND = {
				var:pod_combat_session.var:$KIND_VAR$ = flag:balefire
				$VICTIM$ = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_balefire } }
			}
			AND = {
				var:pod_combat_session.var:$KIND_VAR$ = flag:poison
				$VICTIM$ = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_poison } }
			}
			AND = {
				var:pod_combat_session.var:$KIND_VAR$ = flag:cold_iron
				$VICTIM$ = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_cold_iron } }
			}
		}
	}
	trigger_else = { always = no }
}
