# Reads of the actor's equipped weapon, used by the uses_weapon override
# path. The shared dataset of artifacts that force aggravated damage lives
# one level up in POD_weapon_artifact_triggers.txt (PoD core reads it too).
# System internals: COMBAT_REFERENCE.info section 2.

# Character scope. can_benefit_from_artifact keeps a weapon the character
# cannot actually wield (wrong culture / faith / body) from empowering
# anything - the older artifact-aggravation list predates that check and
# still omits it, so the two disagree for an unusable weapon. Deliberate:
# tightening the old list would change non-combat wounds.
pod_combat_actor_is_armed_trigger = {
	any_equipped_character_artifact = {
		artifact_slot_type = primary_armament
		prev = { can_benefit_from_artifact = prev }
	}
}

# Character scope. True when the equipped weapon carries ANY override the
# uses_weapon path can act on. Reads the four generic vars and nothing else -
# recognising a weapon by NAME anywhere in this path is what the var-only pass
# removed, so a new weapon needs one set_variable at its creation effect and no
# edit here. Used by the move card and by the AI to value a weapon move it can
# actually empower.
pod_combat_actor_weapon_has_override_trigger = {
	any_equipped_character_artifact = {
		artifact_slot_type = primary_armament
		prev = { can_benefit_from_artifact = prev }
		OR = {
			has_variable = artifact_weapon_damage_class
			has_variable = artifact_weapon_bonus_damage
			has_variable = artifact_weapon_undodgeable
			has_variable = artifact_deals_aggravated
		}
	}
}

# Host scope. $VICTIM$ = the character about to be wounded.
# True when the equipped weapon's retype would land on a splat that is IMMUNE
# to that element, which deals literally nothing - a sun-forged bow against a
# werewolf, say. Without this the weapon would be strictly worse than no weapon
# at all, since the move's own superficial damage is what it replaced. The
# dispatch falls back to the authored class whenever this answers yes.
# Self-guarding: answers no when no retype is live, so callers need not order
# their limit terms (neither AND nor OR short-circuits in this engine).
pod_combat_weapon_retype_is_wasted_trigger = {
	trigger_if = {
		limit = { has_variable = pod_combat_weapon_class }
		OR = {
			AND = {
				var:pod_combat_weapon_class = flag:fire
				$VICTIM$ = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_fire } }
			}
			AND = {
				var:pod_combat_weapon_class = flag:cold
				$VICTIM$ = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_cold } }
			}
			AND = {
				var:pod_combat_weapon_class = flag:lightning
				$VICTIM$ = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_lightning } }
			}
			AND = {
				var:pod_combat_weapon_class = flag:sun
				$VICTIM$ = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_sun } }
			}
			AND = {
				var:pod_combat_weapon_class = flag:silver
				$VICTIM$ = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_silver } }
			}
			AND = {
				var:pod_combat_weapon_class = flag:gold
				$VICTIM$ = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_gold } }
			}
			AND = {
				var:pod_combat_weapon_class = flag:holy
				$VICTIM$ = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_true_faith } }
			}
			AND = {
				var:pod_combat_weapon_class = flag:balefire
				$VICTIM$ = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_balefire } }
			}
			AND = {
				var:pod_combat_weapon_class = flag:poison
				$VICTIM$ = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_poison } }
			}
			AND = {
				var:pod_combat_weapon_class = flag:cold_iron
				$VICTIM$ = { POD_has_immunity_to_damage_type = { DEATH_REASON = flag:death_cold_iron } }
			}
		}
	}
	trigger_else = { always = no }
}
