######################################################
###### SHARED BOSS TRIGGERS AND LAUNCH GATES
######################################################

# Character scope. STORY is a POD_combat_move story; the list is written by POD_combat_boss_grant_move_effect.
pod_combat_boss_move_granted_trigger = {
	trigger_if = {
		limit = { has_variable_list = pod_combat_boss_moves }
		is_target_in_variable_list = { name = pod_combat_boss_moves target = $STORY$ }
	}
	trigger_else = { always = no }
}

# Character scope; LEADER is the recruiting combat host.
pod_combat_boss_ally_available_trigger = {
	is_alive = yes
	is_adult = yes
	is_ai = yes
	is_imprisoned = no
	NOT = { this = $LEADER$ }
	in_diplomatic_range = $LEADER$
	POD_is_available = yes
	pod_combat_is_incapacitated_trigger = no
	POD_empty_damage_boxes > 0
	NOT = { has_variable = pod_combat_session }
	NOT = { has_variable = pod_combat_intro_pending }
}

# Character scope.
pod_combat_boss_launch_ready_trigger = {
	is_ai = no
	is_alive = yes
	pod_combat_is_incapacitated_trigger = no
	NOT = { has_variable = pod_combat_session }
	NOT = { has_variable = pod_combat_intro_pending }
	has_game_rule = pod_combat_enabled
	pod_combat_any_fight_active_trigger = no
}
