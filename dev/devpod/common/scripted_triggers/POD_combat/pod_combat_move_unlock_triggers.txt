# Each move has one character-scoped unlock rule. Hidden
# decisions call that rule for failure text, while the story dispatcher
# calls it for the loadout editor and runtime combat. The trait/faith/perk
# predicate therefore exists in exactly one location.
#
# can_execute_decision is deliberately not runtime authority: implicit
# decision-taker requirements reject generic unlanded combatants, which
# previously emptied AI loadouts and made those actors pass every turn.
#
# The two-level dispatcher first switches on config_section, then
# skill_key. Every CSV move has an explicit branch, including universal
# moves whose shared rule is always = yes. Unknown hand-registered manual
# moves retain the historical permissive fallback for compatibility.

######################################################
######   RUNTIME STORY-TO-RULE DISPATCHER
######################################################
#   $STORY$ = the POD_combat_move story_cycle
#   $ACTOR$ = character whose unlock (trait/perk) is tested
pod_combat_actor_unlocks_move_trigger = {
	switch = {
		trigger = $STORY$.var:config_section
		flag:base_physical = {
			switch = {
				trigger = $STORY$.var:skill_key
				flag:strike = {
					$ACTOR$ = {
						pod_combat_move_strike_unlock_trigger = yes
					}
				}
				flag:heavy_blow = {
					$ACTOR$ = {
						pod_combat_move_heavy_blow_unlock_trigger = yes
					}
				}
				flag:quick_jab = {
					$ACTOR$ = {
						pod_combat_move_quick_jab_unlock_trigger = yes
					}
				}
				flag:reckless_charge = {
					$ACTOR$ = {
						pod_combat_move_reckless_charge_unlock_trigger = yes
					}
				}
				fallback = { always = yes }
			}
		}
		flag:base_tactical = {
			switch = {
				trigger = $STORY$.var:skill_key
