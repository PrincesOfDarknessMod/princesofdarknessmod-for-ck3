POD_has_wrong_type_of_building_trigger = {
	OR = {
		AND = { 
			has_building_with_flag = vampireonly
			root = { NOT = { POD_is_vampire_or_abomination_trigger = yes } }
		}
		AND = { 
			has_building_with_flag = vampire
			root = { building_requirement_vampire = no }
		}
		AND = { 
			has_building_with_flag = vamp_blood
			root = { 
				OR = { 
					building_requirement_vamp_blood = no
					building_requirement_ghoulbro = yes
				}
			}
		}
		AND = { 
			has_building_with_flag = ghoulonly
			root = { 
				OR = { 
					POD_is_any_kind_of_ghoul_trigger = no
					building_requirement_ghoulbro = yes
				}
			}
		}
		AND = { 
			has_building_with_flag = gantenbeins
			root = { building_requirement_ghoulbro = no }
		}
		AND = { 
			has_building_with_flag = kueijin
			root = { building_requirement_kj = no }
		}
		AND = { 
			has_building_with_flag = inquisitor
			root = { building_requirement_inquisitor = no }
		}
		AND = { 
			has_building_with_flag = hunter
			root = { building_requirement_hunter = no }
		}
		AND = { 
			has_building_with_flag = shih
			root = { building_requirement_shih = no }
		}
		AND = { 
			has_building_with_flag = mummy
			root = { 
				NOT = { has_trait = mummy }
				is_a_neteru = no
			}
		}
		AND = { 
			has_building_with_flag = demon
			root = { 
				NOT = { has_trait = demon }
				NOT = { has_trait = laham }
				NOT = { has_trait = nephilim }
				is_a_cultist = no
			}
		}
		AND = { 
			has_building_with_flag = werewolfonly
			root = { 
				NOT = { has_trait = werewolf } 
				NOT = { has_trait = kinfolk } 
			}
		}
		AND = { 
			has_building_with_flag = bastetonly
			root = { 
				NOT = { has_trait = bastet } 
				NOT = { has_trait = kinfolk } 
			}
		}
		AND = { 
			has_building_with_flag = mokoleonly
			root = { 
				NOT = { has_trait = mokole }
				NOT = { has_trait = kinfolk }
			}
		}
		AND = { 
			has_building_with_flag = fera
			root = { 
				NOT = { has_trait = fera } 
				NOT = { has_trait = kinfolk } 
			}
		}
		AND = { 
			has_building_with_flag = wraith
			root = { building_requirement_wraith = no }
		}
		AND = { 
			has_building_with_flag = fae
			root = { building_requirement_fae = no }
		}
		AND = { 
			has_building_with_flag = rome
			rome_has_wrong_holder_trigger = yes
		}
		AND = { 
			has_building_with_flag = constantinople
			constantinople_has_wrong_holder_trigger = yes
		}
		AND = { 
			has_building_with_flag = paris
			paris_has_wrong_holder_trigger = yes
		}
		AND = { 
			has_building_with_flag = cairo
			cairo_has_wrong_holder_trigger = yes
		}
		AND = {
			has_building_with_flag = jerusalem
			jerusalem_has_wrong_holder_trigger = yes
		}
		AND = {
			has_building_with_flag = baghdad
			baghdad_has_wrong_holder_trigger = yes
		}
		AND = {
			has_building_with_flag = bjarkarey
			bjarkarey_has_wrong_holder_trigger = yes
		}
		AND = {
			has_building_with_flag = mogadishu
			mogadishu_has_wrong_holder_trigger = yes
		}
	}
}
