POD_has_wrong_type_of_building_trigger = {
	trigger_if = {
		limit = { has_building_with_flag = vampireonly }
		barony.holder ?= { NOT = { has_trait = vampire } }
	}
	trigger_if = {
		limit = { has_building_with_flag = vampire }
		barony.holder ?= { building_requirement_vampire = no }
	}
	trigger_if = {
		limit = { has_building_with_flag = vamp_blood }
		barony.holder ?= { 
			OR = { 
				building_requirement_vamp_blood = no
				building_requirement_ghoulbro = yes
			}
		}
	}
	trigger_if = {
		limit = { has_building_with_flag = ghoulonly }
		barony.holder ?= { 
			OR = { 
				POD_is_any_kind_of_ghoul_trigger = no
				building_requirement_ghoulbro = yes
			}
		}
	}
	trigger_if = {
		limit = { has_building_with_flag = gantenbeins }
		barony.holder ?= { 
			building_requirement_ghoulbro = no
		}
	}
	trigger_if = {
		limit = { has_building_with_flag = kueijin }
		barony.holder ?= { building_requirement_kj = no }
	}
	trigger_if = {
		limit = { has_building_with_flag = mortal }
		barony.holder ?= { 
			NOT = { has_trait = mortal }
		}
	}
	trigger_if = {
		limit = { has_building_with_flag = inquisitor }
		barony.holder ?= { 
			building_requirement_inquisitor = no
		}
	}
	trigger_if = {
		limit = { has_building_with_flag = hunter }
		barony.holder ?= { 
			building_requirement_hunter = no
		}
	}	
	trigger_if = {
		limit = { has_building_with_flag = shih }
		barony.holder ?= { building_requirement_shih = no }
	}
	trigger_if = {
		limit = { has_building_with_flag = mummy }
		barony.holder ?= { 
			NOT = { has_trait = mummy }
			is_a_neteru = no
		}
	}
	trigger_if = {
		limit = { has_building_with_flag = demon }
		barony.holder ?= { 
			NOT = { has_trait = demon }
			NOT = { has_trait = laham }
			NOT = { has_trait = nephilim }
			is_a_cultist = no
		}
	}
	trigger_if = {
		limit = { has_building_with_flag = shapeshifter }
		barony.holder ?= { 
			NOT = { has_trait = shapeshifter } 
			NOT = { has_trait = kinfolk } 
		}
	}
	trigger_if = {
		limit = { has_building_with_flag = wraith }
		barony.holder ?= { 
			NOT = { has_trait = wraith } 
		}
	}
	trigger_if = {
		limit = { has_building_with_flag = fae }
		barony.holder ?= { has_POD_fae_blood_trigger = no }
	}
	trigger_if = {
		limit = { has_building_with_flag = rome }
		rome_has_wrong_holder_trigger = yes
	}
}
