POD_can_learn_true_name_trigger = {
	OR = {
		has_perk = arcana_invocation_of_name_perk
		has_perk = via_geniorum_2_perk
		is_demonologist_trigger = yes 
		POD_can_banish_demons_trigger = yes
		has_perk = reap_the_well_perk
		has_perk = saining_perk
		has_perk = command_thought_perk
		character:easteregg_flandre ?= this
	}
}

POD_true_name_is_learnable = {
	NOT = { has_character_flag = ate_his_true_name }
	NOT = { has_perk = cloud_the_name_perk }
}

	}
}

				has_trait_flag = reweaving_lifestyle_trait

		}
	}
