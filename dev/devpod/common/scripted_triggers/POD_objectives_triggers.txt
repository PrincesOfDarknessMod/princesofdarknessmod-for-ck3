# Jerusalem Objectives

POD_can_maintain_the_sleeper_in_jerusalem = {
	OR = {
		house = house:house_moloch
		has_faith = faith:viahyron
	}
}

POD_can_wake_the_sleeper_in_jerusalem = {
	OR = {
		house = house:house_nergal
		house = house:house_angellisater
		is_infernalist_trigger = yes
	}
}

# Impundulu Objectives

### Checks if character has completed the major objective (either branch)
### Used by Legacy Tier 5 prerequisite
POD_has_completed_withering_blood_trigger = {
	custom_tooltip = {
		text = POD_impundulu_has_completed_withering_blood_tt
		OR = {
			dynasty = { has_dynasty_modifier = covenant_of_life_and_death_modifier }
			has_character_modifier = ishologu_modifier
		}
	}
}

### Checks if a character is a valid target for necromancy knowledge exchange
### Must be a vampire with necromancy, different bloodline, positive opinion
# arguments: ACTOR
POD_is_valid_knowledge_exchange_target_for = {
	is_vampire_trigger = yes
	has_trait = necromancydiscipline
	is_impundulu_trigger = no
	opinion = {
		target = $ACTOR$
		value >= 20
	}
	POD_is_available_for_events = yes
	OR = {
		AND = {
			$ACTOR$ = { has_giovanni_necromancy_trigger = no }
			has_giovanni_necromancy_trigger = yes
		}
		AND = {
			$ACTOR$ = { has_cappadocian_necromancy_trigger = no }
			has_cappadocian_necromancy_trigger = yes
		}
		AND = {
			$ACTOR$ = { has_harbinger_necromancy_trigger = no }
			has_harbinger_necromancy_trigger = yes
		}
		AND = {
			$ACTOR$ = { has_lamia_necromancy_trigger = no }
			has_lamia_necromancy_trigger = yes
		}
		AND = {
			$ACTOR$ = { has_nagaraja_necromancy_trigger = no }
			has_nagaraja_necromancy_trigger = yes
		}
		AND = {
			$ACTOR$ = { has_rosselini_necromancy_trigger = no }
			has_rosselini_necromancy_trigger = yes
		}
	}
}
