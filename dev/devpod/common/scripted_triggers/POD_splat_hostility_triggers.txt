POD_supe_on_vamp_trigger = {
	AND = {
		$ACTOR$ ?= { 
			POD_is_supehunter_trigger = yes
		}
		$RECIPIENT$ ?= { POD_is_vampire_or_abomination_trigger = yes }
	}
	NAND = {
		$ACTOR$.faith = faith:sanbenito
		$RECIPIENT$.religion = religion:roadofhumanity_religion
	}
}
POD_inquisitor_on_unholy_trigger = {
	AND = {
		$ACTOR$ ?= {
			POD_is_supehunter_trigger = yes 
			faith = faith:shadowinquisition
		}
		$RECIPIENT$ ?= { POD_is_unholy_trigger = yes }
	}
}
POD_werewolf_on_vampire_trigger = {
	AND = {
		$ACTOR$ ?= { 
			has_trait = fera
			religion = { is_in_family = rf_gaia }
		}
		$RECIPIENT$ ?= { 
			has_POD_public_vamp_blood_trigger = yes
			NOR = { 
				has_trait = fera
				has_trait = kinfolk
			}
			NOT = { has_perk = fomori_deception_perk }
		}
	}
}

POD_werewolf_on_hunter_trigger = {
	AND = {
		$ACTOR$ ?= { 
			has_trait = fera		
		}
		$RECIPIENT$ ?= { 
			POD_is_supehunter_trigger = yes 
			NOT = { faith = faith:qorjin_baghatur }
		}
	}
}

POD_osiris_league_on_apophis_servant_trigger = {
	AND = {
		$ACTOR$ ?= { faith = { has_doctrine  = special_doctrine_against_to_apophis_hostility } }
		$RECIPIENT$ ?= { faith = { has_doctrine  = special_doctrine_allied_to_apophis_hostility } }
	}
}

POD_supe_on_demon_trigger = {
	AND = {
		$ACTOR$ ?= { 
			has_trait = supehunter
		}
		$RECIPIENT$ ?= { has_trait = demon }
	}
}

POD_supe_on_fae_trigger = {
	NOT = { has_global_variable = fae_is_holy }
	AND = {
		$ACTOR$ ?= { 
			has_trait = supehunter
		}
		$RECIPIENT$ ?= { 
			OR = { 
				has_trait = fae 
				has_trait = changeling 
			}
		}
	}
}

POD_vampire_on_fae_trigger = {
	AND = {
		$ACTOR$ ?= { 
			POD_is_vampire_or_abomination_trigger = yes
			NOT = { is_malkavian_trigger = yes }
			NOT = { has_trait = maeghar }
			NOT = { dynasty = dynasty:10000021 }
			NOT = { has_perk = fomori_deception_perk }
		}
		$RECIPIENT$ ?= { 
			#has_trait = kithain 
			has_kith_trigger = yes #(Surely it shouldn't matter which type of fae it is?)
			faith = { has_doctrine = doctrine_fae_morality_seelie }
		}
	}
}

POD_supe_on_bane_trigger = {
	AND = {
		$ACTOR$ ?= { 
			has_trait = supehunter
		}
		$RECIPIENT$ ?= { 
			OR = { 
				has_trait = banemummy 
				has_trait = wukuei 
			}
		}
	}
}

POD_supe_on_ghoul_trigger = {
	AND = {
		$ACTOR$ ?= { 
			POD_is_supehunter_trigger = yes 
		}
		$RECIPIENT$ ?= { POD_is_any_kind_of_ghoul_trigger = yes }
	}
	NAND = {
		$ACTOR$ ?= { religion = religion:calomenasforsaken_religion }
		$RECIPIENT$ ?= { religion = religion:calomenasforsaken_religion }
	}
	NAND = {
		NOT = { has_global_variable = fae_is_holy }
		$RECIPIENT$ ?= { has_POD_fae_blood_trigger = yes }
	}
}

POD_gaia_on_wyrm_trigger = {
	AND = {
		$ACTOR$ ?= { 
			POD_is_gaia_splat_trigger = yes
		}
		$RECIPIENT$ ?= { 
			POD_is_wyrm_splat_trigger = yes 
			NOT = { has_perk = fomori_deception_perk }
			NOT = { has_perk = fomori_triatic_scent_perk }
		}
	}
}

POD_wyld_on_weaver_trigger = {
	AND = {
		$ACTOR$ ?= { 
			POD_is_wyld_splat_trigger = yes 
			NOT = { has_perk = gorgon_triatic_scent_perk }
		}
		$RECIPIENT$ ?= { 
			POD_is_weaver_splat_trigger = yes 
			NOT = { has_perk = drone_triatic_scent_perk }
		}
	}
}

POD_osiris_league_on_league_trigger = {
	NAND = {
		$ACTOR$ ?= { faith = { has_doctrine = special_doctrine_osirian_league } }
		$RECIPIENT$ ?= { faith = { has_doctrine = special_doctrine_osirian_league } }
	}
}

	NAND = {
		$ACTOR$ ?= { faith = { has_doctrine = golconda_repentance } }
		$RECIPIENT$ ?= { faith = { has_doctrine = golconda_repentance } }
	}
}

POD_set_corrupted_on_corrupted_trigger = {
	NAND = {
		OR = {
			$ACTOR$ ?= { faith = { has_doctrine = special_doctrine_set_pillars_kindred } }
			$ACTOR$ ?= { faith = { has_doctrine = special_doctrine_set_corrupted_kindred } }
		}
		OR = {
			$RECIPIENT$ ?= { faith = { has_doctrine = special_doctrine_set_pillars_kindred } }
			$RECIPIENT$ ?= { faith = { has_doctrine = special_doctrine_set_corrupted_kindred } }
		}
	}
}

	POD_custom_incompatible_splats_trigger = { ACTOR = scope:actor RECIPIENT = scope:recipient }
}
