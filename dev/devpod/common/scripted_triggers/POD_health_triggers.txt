POD_is_trait_not_permanent_trigger = {
	trait:$TRAIT$ ?= { save_temporary_scope_as = trait }
	custom_description = {
		text = POD_is_trait_permanent
		object = scope:trait
		NOT = { has_character_flag = POD_permanent_$TRAIT$_flag }
	}
}

POD_has_any_healable_maim_trigger = {
	custom_tooltip = {
		text = POD_has_any_healable_maim_trigger_tooltip
		OR = {
			AND = { has_trait = blind      POD_is_trait_not_permanent_trigger = { TRAIT = blind } }
			AND = { has_trait = scarred    POD_is_trait_not_permanent_trigger = { TRAIT = scarred } }
			AND = { has_trait = disfigured POD_is_trait_not_permanent_trigger = { TRAIT = disfigured } }
			AND = { has_trait = maimed     POD_is_trait_not_permanent_trigger = { TRAIT = maimed } }
			AND = { has_trait = one_legged POD_is_trait_not_permanent_trigger = { TRAIT = one_legged } }
			AND = { has_trait = one_eyed   POD_is_trait_not_permanent_trigger = { TRAIT = one_eyed } }
			AND = { has_trait = eunuch_1   POD_is_trait_not_permanent_trigger = { TRAIT = eunuch_1 } }
		}
	}
}
