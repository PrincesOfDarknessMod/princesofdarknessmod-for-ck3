
# this trigger is based on can_contract_disease_trigger in 20_health_triggers.txt
# but it takes a disease scope so it's more flexible and reusable than paradox' version
can_contract_scoped_disease_trigger = {
	$DISEASE$ = { save_temporary_scope_as = disease_type_for_trigger }
	save_temporary_scope_as = sick_character_check

	NOT = { has_trait = torpor }
	NOT = { has_character_flag = POD_do_not_disturb_flag }
	is_in_umbra_trigger = no
		
	trigger_if = {
		limit = { $DISEASE$ = trait:demonic_plague }
		POD_is_not_incorporeal_spirit = yes
		NOT = { has_trait = automaton }
	}
	trigger_else_if = {
		limit = { $DISEASE$ = trait:scourge }
		has_trait = vampire
	}
	trigger_else_if = {
		limit = { $DISEASE$ = trait:flaying_plague }
		OR = {
			has_trait = fera
			has_trait = kinfolk
		}
	}
	trigger_else_if = {
		limit = { $DISEASE$ = trait:gift_of_melancholia }
		OR = {
			has_POD_vamp_blood_trigger = yes
			NOT = { has_character_flag = immune_to_disease }
		}
	}
	trigger_else_if = {
		limit = { $DISEASE$ = trait:chulorviosis }
		OR = {
			has_trait = fera
			NOT = { has_character_flag = immune_to_disease }
		}
	}
	trigger_else_if = {
		limit = { $DISEASE$ = trait:crimson_fever }
		OR = {
			NOT = { has_character_flag = immune_to_disease }
		}
	}
	trigger_else = {
		# Is not immune to disease
		#OR = {
			##### old bloodsorcery
			#has_character_modifier = malady_modifier
			NOT = { has_character_flag = immune_to_disease }
		#}
	}
	
	#Doesn't already have the trait
	NOT = { has_trait = $DISEASE$ }

	#Only get measles once
	trigger_if = {
		limit = { $DISEASE$ = trait:measles }
		NOT = { has_character_flag = measles_immunity }
	}

	#Only get bubonic plague once
	trigger_if = {
		limit = { $DISEASE$ = trait:bubonic_plague }
		NOT = { has_character_flag = bubonic_immunity }
	}

	#Only get pod plague once
	trigger_if = {
		limit = { $DISEASE$ = trait:demonic_plague }
		NOT = { has_character_flag = demonic_plague_immunity }
	}
	trigger_if = {
		limit = { $DISEASE$ = trait:flaying_plague }
		NOT = { has_character_flag = flaying_plague_immunity }
	}
	trigger_if = {
		limit = { $DISEASE$ = trait:scourge }
		NOT = { has_character_flag = scourge_immunity }
	}

	trigger_if = { #Lover's pox can only be contracted if you are sexually active
		limit = {
			scope:disease_type_for_trigger = trait:lovers_pox
			is_imprisoned = yes
		}
		imprisoner = {
			OR = {
				AND = {
					scope:sick_character_check = { is_in_prison_type = dungeon }
					any_prisoner = {
						OR = {
							has_relation_lover = scope:sick_character_check
							any_consort = {
								this = scope:sick_character_check
							}
						}
					}
				}
				AND = {
					scope:sick_character_check = { is_in_prison_type = house_arrest }
					OR = {
						any_prisoner = {
							OR = {
								has_relation_lover = scope:sick_character_check
								any_consort = {
									this = scope:sick_character_check
								}
							}
						}
						any_courtier = {
							OR = {
								has_relation_lover = scope:sick_character_check
								any_consort = {
									this = scope:sick_character_check
								}
							}
						}
					}
				}
			}
		}
	}

	# Check to see that there is someone for you to have sex with if you are traveling
	trigger_if = {
		limit = {
			scope:disease_type_for_trigger = trait:lovers_pox
			is_travelling = yes
			# Do not block Goliardic Shenanigans!
			involved_activity ?= {
				NOT = { has_activity_type = activity_adult_education }
			}
		}
		current_travel_plan = {
			any_entourage_character = {
				OR = {
					has_relation_lover = scope:sick_character_check
					any_consort = {
						this = scope:sick_character_check
					}
				}
			}
		}
	}

	#Secondary requirements (mutually exclusive diseases or immunity)
	OR = {
		NOR = { #No secondary requirement
			scope:disease_type_for_trigger = trait:ill
			scope:disease_type_for_trigger = trait:smallpox
			scope:disease_type_for_trigger = trait:bubonic_plague
			
			scope:disease_type_for_trigger = trait:demonic_plague
			scope:disease_type_for_trigger = trait:flaying_plague
			scope:disease_type_for_trigger = trait:scourge
		}
		AND = {
			scope:disease_type_for_trigger = trait:ill
			NOT = { has_trait = pneumonic }
		}
		AND = {
			scope:disease_type_for_trigger = trait:smallpox
			NOT = { has_character_flag = disease_immunity_smallpox }
		}
		AND = {
			scope:disease_type_for_trigger = trait:bubonic_plague
			NOT = { has_character_flag = disease_immunity_bubonic_plague }
		}
		AND = {
			scope:disease_type_for_trigger = trait:demonic_plague
			NOT = { has_character_flag = disease_immunity_demonic_plague }
		}
		AND = {
			scope:disease_type_for_trigger = trait:flaying_plague
			NOT = { has_character_flag = disease_immunity_flaying_plague }
		}
		AND = {
			scope:disease_type_for_trigger = trait:scourge
			NOT = { has_character_flag = disease_immunity_scourge }
		}
	}

	#No disease stacking
	OR = {
		can_get_non_epidemic_disease_trigger = yes
		OR = {
			scope:disease_type_for_trigger = trait:bubonic_plague
			scope:disease_type_for_trigger = trait:smallpox
			
			scope:disease_type_for_trigger = trait:demonic_plague
			scope:disease_type_for_trigger = trait:flaying_plague
			scope:disease_type_for_trigger = trait:scourge
		}
		AND = {
			scope:disease_type_for_trigger = trait:pneumonic
			has_trait = ill
		}	
	}

	#Cooldown is over
	NOR = {
		has_character_flag = contraction_cooldown_general #Haven't had a disease for 2 years
		#No specific cooldowns:
		#Ill, you can get it often
		#Smallpox and bubonic plague, after one time you are immune
		#Lover's pox and leper because you never recover
		AND = {
			has_character_flag = contraction_cooldown_pneumonic
			scope:disease_type_for_trigger = trait:pneumonic
		}
		AND = {
			has_character_flag = contraction_cooldown_gout_ridden
			scope:disease_type_for_trigger = trait:gout_ridden
		}
		AND = {
			has_character_flag = contraction_cooldown_typhus
			scope:disease_type_for_trigger = trait:typhus
		}
		AND = {
			has_character_flag = contraction_cooldown_consumption
			scope:disease_type_for_trigger = trait:consumption
		}
		AND = {
			has_character_flag = contraction_cooldown_cancer
			scope:disease_type_for_trigger = trait:cancer
		}
		AND = {
			has_character_flag = contraction_cooldown_great_pox
			scope:disease_type_for_trigger = trait:great_pox
		}
		
		AND = {
			has_character_flag = contraction_cooldown_crimson_fever
			scope:disease_type_for_trigger = trait:crimson_fever
		}
		AND = {
			has_character_flag = contraction_cooldown_gift_of_melancholia
			scope:disease_type_for_trigger = trait:gift_of_melancholia
		}
	}
}

POD_health_has_normal_wound_trigger = {
	custom_description = {
		text = POD_health_has_normal_wound_trigger
		POD_health_normal_damage >= 1
	}
}
POD_health_has_aggravated_wound_trigger = {
	custom_description = {
		text = POD_health_has_aggravated_wound_trigger
		POD_health_aggravated_damage >= 1
	}
}

POD_can_be_healed_trigger = {
	NOT = { has_trait_with_flag = can_not_cure }
	NOT = { has_character_modifier = eye_of_sakhmet_modifier }
	NOT = { has_character_modifier = thin_blood_perk_modifier }
	NOT = { has_character_modifier = ritual_infirm_inert_modifier }
	NOT = { has_character_modifier = nar_iman_modifier }
}

POD_can_naturaly_heal_trigger = {
	POD_can_be_healed_trigger = yes
	NOT = { has_trait = vampire }
}
