# used for contract_disease_notify_effect, which seems to ignore splat immunity sometimes
# (TODO GDP: investigate further)
POD_can_contract_mundane_diseases_in_events = {
	OR = {
		has_trait = mortal
		has_trait = kinfolk
	}
}

# for use in contract_disease_effect
POD_should_receive_disease_notification_events = {
	NOT = { is_in_list = disase_notification_list }
	is_immortal = no
}

# arguments: FLAG
POD_is_contagious_supernatural_disease_flag = {
	OR = {
		$FLAG$ = flag:crimson_fever
		$FLAG$ = flag:chulorviosis
		$FLAG$ = flag:demonic_plague
		$FLAG$ = flag:gift_of_melancholia
		$FLAG$ = flag:flaying_plague
		$FLAG$ = flag:scourge
		$FLAG$ = flag:entoban_plague
	}
}

# arguments: TRAIT
POD_is_contagious_supernatural_disease_trait = {
	OR = {
		$TRAIT$ = trait:crimson_fever
		$TRAIT$ = trait:chulorviosis
		$TRAIT$ = trait:demonic_plague
		$TRAIT$ = trait:gift_of_melancholia
		$TRAIT$ = trait:flaying_plague
		$TRAIT$ = trait:scourge
		$TRAIT$ = trait:entoban_plague
	}
}

POD_is_trait_not_permanent_trigger = {
	trait:$TRAIT$ ?= { save_temporary_scope_as = trait }
	custom_description = {
		text = POD_is_trait_permanent
		object = scope:trait
		NOT = { has_character_flag = POD_permanent_$TRAIT$_flag }
	}
}

POD_has_any_healable_maim_trigger = {
	custom_tooltip = {
		text = POD_has_any_healable_maim_trigger_tooltip
		OR = {
			AND = { has_trait = blind      POD_is_trait_not_permanent_trigger = { TRAIT = blind } }
			AND = { has_trait = scarred    POD_is_trait_not_permanent_trigger = { TRAIT = scarred } }
			AND = { has_trait = disfigured POD_is_trait_not_permanent_trigger = { TRAIT = disfigured } }
			AND = { has_trait = maimed     POD_is_trait_not_permanent_trigger = { TRAIT = maimed } }
			AND = { has_trait = one_legged POD_is_trait_not_permanent_trigger = { TRAIT = one_legged } }
			AND = { has_trait = one_eyed   POD_is_trait_not_permanent_trigger = { TRAIT = one_eyed } }
			AND = { has_trait = eunuch_1   POD_is_trait_not_permanent_trigger = { TRAIT = eunuch_1 } }
		}
	}
}
