POD_automaton_create_trigger = {
	switch = {
		trigger = var:current_recipe
		flag:simple_servitor = { has_perk = simple_servitor_perk }
		flag:overseer = { has_perk = effigy_overseer_perk }
		flag:guard = { has_perk = guard_perk }
		flag:animantis = { 
			has_perk = animantis_perk 
			custom_tooltip = { 
				text = no_creating_animantis
				NOT = { has_variable = character_creator_points }
			}
		}
	}
}

has_automaton_recipe_trigger = {
	OR = {
		has_perk = simple_servitor_perk
		has_perk = animantis_perk
	}
}

POD_automaton_trigger = {
	switch = {
		trigger = scope:recipe
		flag:simple_servitor = { has_perk = simple_servitor_perk }
		flag:overseer = { has_perk = effigy_overseer_perk }
		flag:guard = { has_perk = guard_perk }
		flag:animantis = { has_perk = animantis_perk }
	}
}
