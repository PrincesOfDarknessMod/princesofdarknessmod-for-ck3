POD_has_baghatur_military_strength_trigger = {
	save_temporary_scope_value_as = {
		name = required_military_strength
		value = $VALUE$
	}
	custom_tooltip = {
		text = POD_has_baghatur_military_strength_trigger
		POD_baghatur_objective_military_strength >= $VALUE$
	}
}

POD_has_happy_vassal_of_religion_trigger = {
	$RELIGION$ = { save_temporary_scope_as = vassal_religion }
    custom_tooltip = {
        text = POD_has_happy_vassal_of_religion_tt
		any_vassal = {
			NOT = { primary_title.tier = tier_barony }	
			religion = scope:vassal_religion
			opinion = {
				target = prev
				value >= 1
			}
		}
    }
}

akritai_is_canonization_candidate_trigger = {
	is_alive = no
	NOT = { has_trait = saint }
	faith ?= faith:akritai
	# Saintly life and miracle worker
	num_virtuous_traits = {
		value >= 1
		target = this.rite
	}
	num_sinful_traits = {
		value = 0
		target = this.rite
	}
	has_trait = true_faith
}

POD_story_akritai_pulled_in_two_directions_resolved_trigger = {
	exists = global_var:pod_akritai_chosen_faction
}

POD_akritai_can_use_fetishes_trigger = {
	AND = {
		has_faith = faith:akritai
		is_target_in_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:akritai_joining_old_alliances_decision
		}
	}
}
