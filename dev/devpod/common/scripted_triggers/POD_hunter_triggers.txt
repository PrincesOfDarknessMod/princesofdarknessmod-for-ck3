POD_has_baghatur_military_strength_trigger = {
	save_temporary_scope_value_as = {
		name = required_military_strength
		value = $VALUE$
	}
	custom_tooltip = {
		text = POD_has_baghatur_military_strength_trigger
		POD_baghatur_objective_military_strength >= $VALUE$
	}
}

POD_has_happy_vassal_of_religion_trigger = {
	$RELIGION$ = { save_temporary_scope_as = vassal_religion }
    custom_tooltip = {
        text = POD_has_happy_vassal_of_religion_tt
		any_vassal = {
			NOT = { primary_title.tier = tier_barony }	
			religion = scope:vassal_religion
			opinion = {
				target = prev
				value >= 1
			}
		}
    }
}
