has_enough_material_trigger = {
	trigger_if = {
		limit = { is_target_in_variable_list = { name = ingredient_list target = flag:$ALCHEMY$ } }
		$ALCHEMY$_reserve >= $ALCHEMY$_amount
	}
}

can_use_alchemical_product_trigger = {
	trigger_if = {
		limit = { exists = scope:target }
		trigger_if = {
			limit = { scope:target = { var:alchemy = flag:elixir_immortality } }
			is_immortal = no
		}
        trigger_else_if = {
			limit = { scope:target = { var:alchemy = flag:kinain_elixir } }
			NOR = {
				has_main_supernatural_race_trigger = yes
				has_POD_fae_blood_trigger = yes
			}
		}
		trigger_else_if = {
			limit = { scope:target = { var:alchemy = flag:kinfolk_elixir } }
			NOR = {
				has_main_supernatural_race_trigger = yes
				POD_is_kinfolk_trigger = yes
			}
		}
		trigger_else_if = {
			limit = { scope:target = { var:alchemy = flag:vampire_concoction } }
			POD_can_be_embraced_scheme_trigger = yes
		}
		trigger_else_if = {
			limit = { scope:target = { var:alchemy = flag:foul_food } }
			POD_eat_human_food_trigger = yes
		}
		trigger_else_if = {
			limit = { scope:target = { is_risen_counterfeit_alchemy_recipe_trigger = { RECIPE = var:alchemy } } }
			OR = {
				has_trait = risen
				has_POD_vamp_blood_trigger = yes
			}
		}
		trigger_else_if = {
			limit = { scope:target = { is_counterfeit_alchemy_recipe_trigger = { RECIPE = var:alchemy } } }
			has_POD_vamp_blood_trigger = yes
		}
		trigger_else = { }
	}
}

can_afford_alchemy_product_with_opinion_trigger = {
	$SELLER$ = { save_temporary_scope_as = seller }
	save_temporary_scope_value_as = {
		name = alchemy_value
		value = {
			if = {
				limit = { $PRODUCT$ = { is_level_1_alchemy_recipe_trigger = { RECIPE = var:alchemy } } }
				value = miniscule_alchemy_product_gold_value
			}
			else_if = {
				limit = { $PRODUCT$ = { is_level_2_alchemy_recipe_trigger = { RECIPE = var:alchemy } } }
				value = minor_alchemy_product_gold_value
			}
			else_if = {
				limit = { $PRODUCT$ = { is_level_3_alchemy_recipe_trigger = { RECIPE = var:alchemy } } }
				value = medium_alchemy_product_gold_value
			}
			else_if = {
				limit = { $PRODUCT$ = { is_level_4_alchemy_recipe_trigger = { RECIPE = var:alchemy } } }
				value = major_alchemy_product_gold_value
			}
			else_if = {
				limit = { $PRODUCT$ = { is_level_5_alchemy_recipe_trigger = { RECIPE = var:alchemy } } }
				value = massive_alchemy_product_gold_value
			}
			else_if = {
				limit = { $PRODUCT$ = { is_level_6_alchemy_recipe_trigger = { RECIPE = var:alchemy } } }
				value = monumental_alchemy_product_gold_value
			}
			else_if = {
				limit = { $PRODUCT$ = { is_level_7_alchemy_recipe_trigger = { RECIPE = var:alchemy } } }
				value = excessive_alchemy_product_gold_value
			}
			else_if = {
				limit = { $PRODUCT$ = { is_level_8_alchemy_recipe_trigger = { RECIPE = var:alchemy } } }
				value = omega_alchemy_product_gold_value
			}
			else = {
				value = miniscule_alchemy_product_gold_value
			}
			multiply = $BUYER$.alchemy_price_bonuses
		}
	}
	$BUYER$ = {
		gold >= scope:alchemy_value
	}
}
