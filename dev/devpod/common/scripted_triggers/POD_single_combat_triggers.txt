POD_repeat_moves_trigger = {
	OR = {
		AND = {
			exists = scope:sc_defender_last_move
			this = scope:sc_defender
			scope:sc_defender_last_move = flag:POD_$MOVE$
		}
		AND = {
			exists = scope:sc_attacker_last_move
			this = scope:sc_attacker
			scope:sc_attacker_last_move = flag:POD_$MOVE$
		}
	}
}

POD_repeat_moves_triple_trigger = {
	OR = {
		AND = {
			exists = scope:sc_defender_last_move
			this = scope:sc_defender
			OR = {
				scope:sc_defender_last_move = flag:POD_$MOVE$_1
				scope:sc_defender_last_move = flag:POD_$MOVE$_2
				scope:sc_defender_last_move = flag:POD_$MOVE$_3
			}
		}
		AND = {
			exists = scope:sc_attacker_last_move
			this = scope:sc_attacker
			OR = {
				scope:sc_attacker_last_move = flag:POD_$MOVE$_1
				scope:sc_attacker_last_move = flag:POD_$MOVE$_2
				scope:sc_attacker_last_move = flag:POD_$MOVE$_3
			}
		}
	}
}
