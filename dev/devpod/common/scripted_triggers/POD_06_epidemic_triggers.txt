# these differ from the other epidemic detection triggers by checking the live epidemic followed by the province (which works) instead of the other way around (source of weak inference error upon multiple identical epidemic collision)

province_has_live_epidemic_intensity = {
	any_epidemic = {
		outbreak_intensity >= $INTENSITY$
		prev = {
			# After returning to the province, PREV is the live epidemic.
			province_infection_rate = {
				target = prev
				value > 0
			}
		}
	}
}

# replaces any_province_epidemic = { }

province_has_live_epidemic_trigger = {
	OR = {
		province_has_live_epidemic_intensity = { INTENSITY = minor }
		province_has_live_epidemic_intensity = { INTENSITY = major }
		province_has_live_epidemic_intensity = { INTENSITY = apocalyptic }
	}
}

province_has_live_contractible_epidemic = {
	any_epidemic = {
		# PREV is the province.
		prev = {
			province_infection_rate = {
				target = prev
				value > 0
			}
		}
		root = {
			can_contract_scoped_disease_trigger = {
				DISEASE = prev.epidemic_type.epidemic_trait
			}
		}
	}
}

province_has_live_epidemic_trait = {
	any_epidemic = {
		epidemic_type.epidemic_trait = $TRAIT$
		prev = {
			province_infection_rate = {
				target = prev
				value > 0
			}
		}
	}
}
