POD_is_roguelike_mode_enabled_trigger = {
	OR = {
		has_game_rule = POD_roguelike_mode_enabled
		has_game_rule = POD_roguelike_mode_enabled_no_permadeath
	}
}

POD_has_roguelike_curse = {
	OR = {
		has_character_flag = POD_roguelike_cursed
		has_character_flag = POD_roguelike_all_curses
	}
}

POD_blocked_from_learning_trigger = {
	has_character_modifier = POD_roguelike_no_curiosity
}
