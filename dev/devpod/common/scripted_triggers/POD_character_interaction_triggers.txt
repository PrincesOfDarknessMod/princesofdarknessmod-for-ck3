# used by POD_mass_demand_payment_interaction and its associated script_values
POD_can_demand_payment_trigger = {
	trigger_if = {
		limit = { scope:exclude_partial_payments = no }
		gold >= 1
	}
	trigger_else = {
		gold > golden_obligation_value
	}

	trigger_if = {
		limit = { scope:strong_hooks_only = yes }
		scope:actor = { has_strong_usable_hook = prev }
	}
	trigger_else = {
		scope:actor = { has_usable_hook = prev }
	}

	trigger_if = {
		limit = { scope:exclude_house_members = yes }
		NAND = {
			exists = scope:actor.house
			house ?= { this = scope:actor.house }
		}
	}

	trigger_if = {
		limit = { scope:exclude_necromancy_bind_hooks = yes }
		NOT = {
			scope:actor = {
				has_hook_of_type = {
					target = prev
					type = necromancy_bind_hook
				}
			}
		}
	}

	trigger_if = {
		limit = { scope:exclude_mesmerize_hooks = yes }
		NOT = {
			scope:actor = {
				has_hook_of_type = {
					target = prev
					type = mesmerize_hook
				}
			}
		}
	}

	is_ai = yes
	is_imprisoned = no
	NOT = { is_at_war_with = scope:actor }

	in_diplomatic_range = scope:actor

	# If payment was unlocked via struggle, check if both characters are in the same struggle
	trigger_if = {
		limit = {
			scope:actor = {
				NOT = { has_perk = golden_obligations_perk }
				any_character_struggle = {
					involvement = involved
					has_struggle_phase_parameter = unlocks_demand_payments_for_all
				}
			}
		}
		any_character_struggle = {
			involvement = involved
			has_struggle_phase_parameter = unlocks_demand_payments_for_all
			is_secondary_character_involvement_either_trigger = {
				CHAR = scope:actor
			}
		}
	}
}
