POD_can_embrace = {
	$SIRE$ = {
		#is_vampire_trigger = yes # checked elsewhere
		is_adult = yes
		is_vampgeneration16_trigger = no
		NOT = { has_trait = abomination }
		trigger_if = {
			limit = { has_trait = setite }
			POD_can_setite_be_embraced_trigger = yes
		}
		trigger_if = {
			# Characters who cannot embrace
			limit = {
				OR = {
					this = character:791232 # Catubodua
				}
			}
			custom_tooltip = {
				text = POD_this_character_cannot_embrace
				always = no
			}
		}
	}
	$CHILDE$ = {
		is_adult = yes
		# better to check elsewhere since some abilities let you embrace other splats (like wraiths)
		#POD_can_be_embraced_scheme_trigger = yes
		POD_is_clone_trigger = no
		NOT = { has_character_modifier = boon_of_anubis_perk_modifier }
	}
	custom_description = {
		object = $CHILDE$
		text = "blood_is_holy"
		$CHILDE$ = {
			NOR = {
				POD_has_true_faith_trigger = yes
				has_trait = orison_sanguine
			}
		}
	}
	trigger_if = {
		# Characters who only embrace women
		limit = {
			OR = {
				$SIRE$ = character:791706 # Adisa
			}
			$CHILDE$ = { is_female = no }
		}
		$SIRE$ = {
			custom_tooltip = {
				text = POD_this_character_only_embraces_women
				always = no
			}
		}
	}
}
