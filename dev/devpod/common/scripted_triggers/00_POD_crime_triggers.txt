	
###TRIGGER LIST

is_diablerist_trigger = {
	custom_description = {
		text = is_a_diablerist
		subject = this
		OR = {
			has_trait = diablerist
			any_secret = { secret_type = secret_diablerist }
		}
	}
}

is_diablerist_known_by_trigger = {
	OR = {
		has_trait = diablerist
		custom_description = {
			text = has_a_diablerist_secret_known_by_character
			subject = this
			object = $CHARACTER$
			any_secret = { 
				secret_type = secret_diablerist
				is_known_by = $CHARACTER$
			}
		}
	}
}

is_infernalist_trigger = {
	custom_description = {
		text = is_a_infernalist
		subject = this

			has_trait = baali

			any_secret = { secret_type = secret_infernalist }
