POD_is_not_spirit_trigger = { # Avoid complex trigger
	trigger_if = {
		limit = { 
			NOT = { has_trait = risen } 
		}
		NOT = { has_trait_with_flag = is_spirit }
	}
}

can_be_killed_physically_trigger = {
	POD_is_not_spirit_trigger = yes
	NOT = { has_perk = umbral_body_perk }
}

POD_physically_move_trigger = {
	OR = {
		AND = {
			this = scope:sc_defender
			damage_spirit_custom_trigger = {
				RECIPIENT = scope:sc_attacker
				ACTOR = scope:sc_defender
			}
		}
		AND = {
			this = scope:sc_attacker
			damage_spirit_custom_trigger = {
				RECIPIENT = scope:sc_defender
				ACTOR = scope:sc_attacker
			}
		}
	}
}

damage_spirit_trigger = {
	damage_spirit_custom_trigger = {
		RECIPIENT = scope:recipient
		ACTOR = scope:actor
	}
}

damage_spirit_custom_trigger = {
	trigger_if = {
		limit = { 
			$RECIPIENT$ = { 
				can_be_killed_physically_trigger = no 
				is_in_umbra_trigger = no
			}
		}
		$ACTOR$ = { can_damage_spirit_trigger = yes }
	}
	trigger_if = {
		limit = {
			$ACTOR$ = { 
				NOT = { has_perk = spirit_the_body_away_perk } 
				NOT = { has_perk = walk_the_matres_mind_perk }
				is_in_umbra_trigger = no
			}
			$RECIPIENT$ = { is_in_umbra_trigger = yes }
		}
		$RECIPIENT$ = {
			NOT = { has_perk = empower_ka_perk } # Protection
			custom_tooltip = {
				text = physical_spirit_travel_desc
				NOT = { has_variable = physical_spirit_travel }
				NOT = { has_trait_with_flag = is_spirit }
			}
		}
	}
}

imprison_spirit_trigger = {
	imprison_spirit_custom_trigger = {
		RECIPIENT = scope:recipient
		ACTOR = scope:actor
	}
}

imprison_spirit_custom_trigger = {
	trigger_if = {
		limit = {
			$RECIPIENT$ = {
				OR = {
					has_perk = cets_shell_perk
					can_be_killed_physically_trigger = no
				}
			}
		}
		$ACTOR$ = { can_damage_spirit_trigger = yes }
	}
}

eat_spirit_trigger = {
	eat_spirit_custom_trigger = {
		RECIPIENT = scope:recipient
		ACTOR = scope:actor
	}
}

eat_spirit_custom_trigger = {
	trigger_if = {
		limit = { $RECIPIENT$ = { can_be_killed_physically_trigger = no } }
		$ACTOR$ = { can_eat_spirit_trigger = yes }
	}
}
