		has_perk = fomori_spirit_sense_perk # Possessed
		has_perk = gorgon_spirit_sense_perk # Possessed
		has_perk = drone_spirit_sense_perk # Possessed
		has_perk = kami_spirit_sense_perk # Possessed
	}
}

POD_is_not_spirit_trigger = { # Avoid complex trigger
	trigger_if = {
		limit = { 
			NOT = { has_trait = risen } 
			NOT = { has_trait = fallen }
		}
		NOT = { has_trait_with_flag = is_spirit }
	}
}

can_be_killed_physically_trigger = {
	POD_is_not_spirit_trigger = yes
	is_in_umbra_trigger = no
}

POD_physically_move_trigger = {
	OR = {
		AND = {
			this = scope:sc_defender
			damage_spirit_custom_trigger = {
				RECIPIENT = scope:sc_attacker
				ACTOR = scope:sc_defender
			}
		}
		AND = {
			this = scope:sc_attacker
			damage_spirit_custom_trigger = {
				RECIPIENT = scope:sc_defender
				ACTOR = scope:sc_attacker
			}
		}
	}
}

damage_spirit_trigger = {
	damage_spirit_custom_trigger = {
		RECIPIENT = scope:recipient
		ACTOR = scope:actor
	}
}

damage_spirit_custom_trigger = {
	trigger_if = {
		limit = {
			$RECIPIENT$ = {
				OR = {
					can_be_killed_physically_trigger = no
					has_perk = umbral_body_perk
				}
			}
		}
		$ACTOR$ = { can_damage_spirit_trigger = yes }
	}
}

can_damage_spirit_trigger = {
	custom_tooltip = {
		text = can_hurt_wraith
		OR = {
			has_perk = torment_perk
			has_perk = spirit_scepter_perk
			has_perk = spirit_strike_perk
			has_perk = fomori_spirit_rending_perk
			
			has_trait_with_flag = can_hurt_spirit
			has_trait_with_flag = can_imprison_spirit
			
			any_equipped_character_artifact = { has_variable = can_hurt_spirit }
			
			is_in_umbra_trigger = yes
		}
	}
}

	trigger_if = {
		limit = { has_POD_vamp_blood_trigger = yes }
		has_perk = soul_feasting_perk
	}

		limit = { has_POD_kj_blood_trigger = yes }
