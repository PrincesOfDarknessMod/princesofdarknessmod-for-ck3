

POD_is_not_incorporeal_spirit = {
	trigger_if = {
		limit = {
			NOT = { has_trait = risen }
		}
		NOT = { has_trait_with_flag = is_spirit }
	}
}

POD_can_be_killed_physically_trigger = {
	POD_is_not_incorporeal_spirit = yes
}


POD_has_spirit_type = {
	trigger_if = {
		limit = { is_alive = yes }
		var:spirit_type ?= flag:$TYPE$
	}
	trigger_else = {
		has_dead_character_variable = $TYPE$
	}
}

POD_knows_spirit_language_trigger = {
	POD_knows_spirit_language_custom_trigger = {
		ACTOR = scope:actor
		RECIPIENT = scope:recipient
	}
}

POD_knows_spirit_language_custom_trigger = {
	trigger_if = {
		limit = {
			$RECIPIENT$ = {
				POD_is_spirit_trigger = yes
				has_variable = spirit_type
			}
		}
		trigger_if = {
			limit = { $RECIPIENT$ = { POD_spirit_uses_harmony_language_trigger = yes } }
			$ACTOR$ = { knows_language = language_harmonys_melody }
		}
		trigger_else_if = {
			limit = { $RECIPIENT$ = { POD_spirit_uses_conflict_language_trigger = yes } }
			$ACTOR$ = { knows_language = language_conflicts_drum }
		}
		trigger_else_if = {
			limit = { $RECIPIENT$ = { POD_spirit_uses_artisan_language_trigger = yes } }
			$ACTOR$ = { knows_language = language_artisans_anthem }
		}
		trigger_else_if = {
			limit = { $RECIPIENT$ = { POD_spirit_uses_sedition_language_trigger = yes } }
			$ACTOR$ = { knows_language = language_seditions_lullaby }
		}
		trigger_else_if = {
			limit = { $RECIPIENT$ = { POD_spirit_uses_wisdom_language_trigger = yes } }
			$ACTOR$ = { knows_language = language_wisdoms_hymn }
		}
		trigger_else_if = {
			limit = { $RECIPIENT$ = { POD_spirit_uses_wyrmspeak_language_trigger = yes } }
			$ACTOR$ = { knows_language = language_wyrmspeak }
		}
		trigger_else_if = {
			limit = { $RECIPIENT$ = { POD_spirit_uses_corrupting_voice_language_trigger = yes } }
			$ACTOR$ = { knows_language = language_corrupting_voice }
		}
		trigger_else_if = {
			limit = { $RECIPIENT$ = { POD_spirit_uses_tainted_speech_language_trigger = yes } }
			$ACTOR$ = { knows_language = language_tainted_speech }
		}
		trigger_else_if = {
			limit = { $RECIPIENT$ = { POD_spirit_uses_ancestor_language_trigger = yes } }
			trigger_if = {
				limit = { $RECIPIENT$ = { var:spirit_type = flag:spirit_werewolf } }
				$ACTOR$ = { knows_language = language_ancestor_werewolf }
			}
			trigger_else_if = {
				limit = { $RECIPIENT$ = { var:spirit_type = flag:spirit_bastet } }
				$ACTOR$ = { knows_language = language_ancestor_bastet }
			}
			trigger_else_if = {
				limit = { $RECIPIENT$ = { var:spirit_type = flag:spirit_mokole } }
				$ACTOR$ = { knows_language = language_ancestor_mokole }
			}
			trigger_else = {
				always = no
			}
		}
		trigger_else = {
			always = no
		}
	}
}

POD_can_damage_spirit_trigger = {
	custom_tooltip = {
		text = cannot_hurt_spirit_tt
		OR = {
			POD_can_naturally_damage_spirit_trigger = yes

			has_perk = ns_torment_perk
			has_perk = ns_touch_of_the_nether_perk
			has_perk = spirit_scepter_perk
			has_perk = fomori_spirit_rending_perk
			has_perk = duality_perk
			has_perk = numina_via_necromantiae_4_perk
			has_perk = numina_ephemera_4_perk
			has_perk = arcana_spirit_strike_perk
			has_perk = burning_ghosts_perk
			has_perk = soul_arts_the_touch_of_ghost_metal_perk
			has_perk = heed_the_hell_bound_heart_perk
			
			any_equipped_character_artifact = { has_artifact_modifier = artifact_can_damage_spirit_modifier }
		}
	}
}
POD_can_naturally_damage_spirit_trigger = {
	OR = {
		has_trait = fera
		has_trait = spirit
		has_trait = fae
		has_trait = wraith
		has_trait = true_faith
	}
}

POD_can_summon_gaian_spirit_trigger = {
	OR = {
		has_character_modifier = wutian_ninth_immortal #Wutian Objective
		has_perk = spectral_authority_perk #White Howlers

		#Mokole can do it only if they additionally have 1 of 2 auspice perks

			limit = { has_perk = dream_the_spirit_perk }
			OR = {
				has_perk = call_water_spirit_perk
				has_perk = call_to_duty_perk
			}
		}
	}
}

	}
}

POD_spirit_can_possess_trigger = {
	$TARGET$ = { POD_can_be_possessed_trigger = yes }
	trigger_if = {
		limit = { $SPIRIT$ = { POD_is_wyrm_spirit_trigger = yes } }
		$TARGET$ = { POD_is_not_gaia_servant_trigger = yes }
	}
	trigger_else = { always = no }
}

POD_can_activate_fetish_trigger = {
	POD_can_equip_fetish_trigger = yes
	trigger_if = {
		limit = { is_ai = no }
		trigger_if = {
			limit = { has_trait = fera }
			custom_tooltip = {
				text = fetish_no_gnosis
				gnosis_regen_chance >= 5
			}
		}
		trigger_else = {
			custom_tooltip = {
				text = fetish_no_companion
				POD_equipped_fetish_artifacts_value <= this.num_of_relation_spirit_companion
			}
		}
	}
