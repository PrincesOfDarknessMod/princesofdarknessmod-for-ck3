

POD_is_not_incorporeal_spirit = {
	trigger_if = {
		limit = { 
			NOT = { has_trait = risen } 
		}
		NOT = { has_trait_with_flag = is_spirit }
	}
}

POD_can_be_killed_physically_trigger = {
	POD_is_not_incorporeal_spirit = yes
}

#Tooltip is negative because this only shows up ingame when you can't damage a spirit
POD_can_damage_spirit_trigger = {
	custom_tooltip = {
		text = cannot_hurt_spirit_tt
		OR = {
			POD_can_naturally_damage_spirit_trigger = yes

			has_perk = ns_torment_perk
			has_perk = ns_touch_of_the_nether_perk
			has_perk = spirit_scepter_perk
			has_perk = fomori_spirit_rending_perk
			has_perk = via_necromantiae_2_perk
			has_perk = duality_perk
			has_perk = numina_ephemera_4_perk
			has_perk = arcana_spirit_strike_perk
			has_perk = burning_ghosts_perk
			has_perk = soul_arts_the_touch_of_ghost_metal_perk
			
			
			any_equipped_character_artifact = { has_artifact_modifier = artifact_can_damage_spirit_modifier }
		}
	}
}

POD_can_naturally_damage_spirit_trigger = {
	OR = {
		has_trait = fera
		has_trait = spirit
		has_trait = fae
		has_trait = wraith
		has_trait = lament
		has_trait = true_faith
	}
}

POD_damage_spirit_trigger = {
	POD_damage_spirit_custom_trigger = {
		RECIPIENT = scope:recipient
		ACTOR = scope:actor
	}
}

POD_damage_spirit_custom_trigger = {
	trigger_if = {
		limit = {
			$RECIPIENT$ = {
				POD_can_be_killed_physically_trigger = no
			}
		}
		$ACTOR$ = { POD_can_damage_spirit_trigger = yes }
		POD_cross_umbra_valid_trigger = { ACTOR = $ACTOR$ RECIPIENT = $RECIPIENT$ }
	}
}

POD_knows_spirit_language_trigger = {
	trigger_if = {
		limit = { $RECIPIENT$ = { POD_spirit_uses_harmony_language_trigger = yes } }
		$ACTOR$ = { knows_language = language_harmonys_melody }
	}
	trigger_else_if = {
		limit = { $RECIPIENT$ = { POD_spirit_uses_conflict_language_trigger = yes } }
		$ACTOR$ = { knows_language = language_harmonys_melody }
	}
	trigger_else_if = {
		limit = { $RECIPIENT$ = { POD_spirit_uses_artisan_language_trigger = yes } }
		$ACTOR$ = { knows_language = language_harmonys_melody }
	}
	trigger_else_if = {
		limit = { $RECIPIENT$ = { POD_spirit_uses_sedition_language_trigger = yes } }
		$ACTOR$ = { knows_language = language_harmonys_melody }
	}
	trigger_else_if = {
		limit = { $RECIPIENT$ = { POD_spirit_uses_wisdom_language_trigger = yes } }
		$ACTOR$ = { knows_language = language_harmonys_melody }
	}
	trigger_else_if = {
		limit = { $RECIPIENT$ = { POD_spirit_uses_chaos_language_trigger = yes } }
		$ACTOR$ = { knows_language = language_harmonys_melody }
	}
	trigger_else_if = {
		limit = { $RECIPIENT$ = { POD_spirit_uses_ancestral_language_trigger = yes } }
		trigger_if = {
			limit = { $RECIPIENT$ = { has_character_flag = werewolf_spirit } }
			$ACTOR$ = { knows_language = language_ancestral_werewolf }
		}
		trigger_else_if = {
			limit = { $RECIPIENT$ = { has_character_flag = bastet_spirit } }
			$ACTOR$ = { knows_language = language_ancestral_bastet }
		}
		trigger_else_if = {
			limit = { $RECIPIENT$ = { has_character_flag = mokole_spirit } }
			$ACTOR$ = { knows_language = language_ancestral_mokole }
		}
	}
}
