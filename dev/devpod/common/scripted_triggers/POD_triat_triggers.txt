POD_is_gaia_servant_trigger = {
	OR = {
		AND = {
			religion = { is_in_family = rf_gaia }
			NOT = { faith = { has_doctrine = doctrine_triat_alignment_wyrm } }
		}
		POD_is_gaia_splat_trigger = yes
	}
}

POD_is_not_gaia_servant_trigger = { # for complex trigger
	trigger_if = {
		limit = { NOT = { faith = { has_doctrine = doctrine_triat_alignment_wyrm } } }
		religion = { is_in_family = rf_gaia }
	}
	POD_is_not_gaia_splat_trigger = yes
}

POD_is_gaia_splat_trigger = {
	has_trait = kami
}

POD_is_not_gaia_splat_trigger = { # for complex trigger
	NOT = { has_trait = kami }
}

		POD_is_wyrm_splat_trigger = yes

	POD_is_not_wyrm_splat_trigger = yes

POD_is_wyld_servant_trigger = {
	OR = {
		faith = { has_doctrine = doctrine_triat_alignment_wyld }
		POD_is_wyld_splat_trigger = yes
	}
}

POD_is_not_wyld_servant_trigger = { # for complex trigger
	NOT = { faith = { has_doctrine = doctrine_triat_alignment_wyld } }
	POD_is_not_wyld_splat_trigger = yes
}

POD_is_wyld_splat_trigger = {
	OR = {
		AND = {
			OR = {
				has_trait = changeling
				has_trait = fae
			}
			NOR = {
				has_trait = phyla
				has_trait = hsien
			}
		}
		has_trait = gorgon
	}
}

POD_is_not_wyld_splat_trigger = { # for complex trigger
	trigger_if = {
		limit = { 
			NOT = { has_trait = hsien } 
			NOT = { has_trait = phyla }
		}
		NOT = { has_trait = changeling } 
		NOT = { has_trait = fae }
	}
	NOT = { has_trait = gorgon }
}
