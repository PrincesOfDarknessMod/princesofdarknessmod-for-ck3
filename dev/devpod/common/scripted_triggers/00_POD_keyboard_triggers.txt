POD_keyboard_entry_set = {
	NOT = { var:keyboard_entry_$N$ = flag:__ }
}

POD_keyboard_can_submit_answer = {
	trigger_if = {
		limit = {
			POD_keyboard_entry_set = { N = 5 } 
		}
		always = yes
	}
	trigger_else = {
		custom_tooltip = {
			text = POD_keyboard_can_submit_answer_letter_count_requirement
			always = no
		}
	}

	trigger_if = {
		limit = {
			POD_wordle_guess_count < 6
		}
		always = yes
	}
	trigger_else = {
		custom_tooltip = {
			text = POD_keyboard_can_submit_answer_guess_count_requirement
			always = no
		}
	}
}
