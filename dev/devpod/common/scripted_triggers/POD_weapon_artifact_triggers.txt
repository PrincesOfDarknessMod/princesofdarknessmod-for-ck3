######################################################
######   POD WEAPON ARTIFACT TRIGGERS
######################################################
# The wielded artifacts that promote the damage their bearer deals to
# aggravated. Read by PoD's wound classifier
# (POD_applies_superficial_as_aggravated_damage, POD_health_track_triggers.txt)
# and by the combat minigame, so the set is defined once.
#
# ADDING A WEAPON: do NOT add an arm here. Stamp
# `set_variable = artifact_deals_aggravated` on the artifact in its creation
# effect and the generic arm below picks it up - that one edit is the whole
# contract, and it is also what the move card and the AI read. The named arms
# that remain are the ones whose aggravation is CONDITIONAL (on the victim, or
# on the weapon's power), which is logic rather than a tag and has nowhere else
# to live.
#
# Attacker = the calling scope. $VICTIM$ = the character being wounded,
# needed by the arms that only bite against a particular kind of prey.
#
# Outside the minigame the classifier consults this for weapon-shaped wounds
# only (POD_is_death_by_weapon), so a bane sword no longer aggravates a
# poisoning. Inside it, the marker that reaches this list is only set around a
# weapon move's dispatch, so it no longer aggravates spat venom or a mind
# blast either - see pod_combat_apply_single_hp_damage_effect.

# Artifact scope: does this weapon carry ANY override the uses_weapon path
# acts on? One list, three readers - the character-scope wrapper below, the
# artifact tooltip's heading and its trailing break. Keeping them on one
# trigger is what stops a new override from rendering its line with no
# heading, which is exactly what happened when the header had its own copy.
POD_weapon_artifact_has_override_trigger = {
	OR = {
		has_variable = artifact_weapon_damage_class
		has_variable = artifact_weapon_bonus_damage
		has_variable = artifact_weapon_undodgeable
		has_variable = artifact_deals_aggravated
		has_variable = artifact_weapon_heal_on_damage
		has_variable = artifact_weapon_energy_on_damage
		has_variable = artifact_weapon_soak_penalty
		# The one override still keyed by artifact name rather than a tag: its
		# condition is a power comparison (see the forces-agg trigger below),
		# which is logic and has nowhere else to live. Listed here so the card, the AI
		# and the tooltip heading all recognise the blade.
		has_variable = blade_of_the_kings
		POD_weapon_artifact_has_conditional_bane_trigger = yes
	}
}

# Artifact scope: does this item carry damage class $CLASS$? The nested
# trigger_if keeps the value read off items with no class at all, since
# neither AND nor OR short-circuits here.
POD_weapon_artifact_is_class_trigger = {
	trigger_if = {
		limit = { has_variable = artifact_weapon_damage_class }
		var:artifact_weapon_damage_class = flag:$CLASS$
	}
	trigger_else = { always = no }
}

# Character scope: does this character strike with damage class $CLASS$?
# The armament answers first - a lightning katana is lightning whatever else
# you are carrying. Only when the weapon has no class of its own does carried
# kit get a say, which is how a quiver of silvered arrows makes a plain bow
# strike silver. Both halves require can_benefit_from_artifact, so gear the
# bearer cannot actually use colours nothing.
POD_wound_attacker_wields_class_trigger = {
	trigger_if = {
		limit = {
			any_equipped_character_artifact = {
				artifact_slot_type = primary_armament
				prev = { can_benefit_from_artifact = prev }
				has_variable = artifact_weapon_damage_class
			}
		}
		any_equipped_character_artifact = {
			artifact_slot_type = primary_armament
			prev = { can_benefit_from_artifact = prev }
			POD_weapon_artifact_is_class_trigger = { CLASS = $CLASS$ }
		}
	}
	trigger_else = {
		any_equipped_character_artifact = {
			prev = { can_benefit_from_artifact = prev }
			POD_weapon_artifact_is_class_trigger = { CLASS = $CLASS$ }
		}
	}
}

# Artifact scope: does this weapon bite only a particular kind of prey?
# Composable presence tags - a weapon sets one or more and they OR together,
# so Ivory Bow (vampires AND werewolves) is two lines at its creation effect
# and no code anywhere. artifact_agg_except_werewolf is the one inverse in the
# set: klaives savage everything that is NOT a werewolf.
POD_weapon_artifact_has_conditional_bane_trigger = {
	OR = {
		has_variable = artifact_agg_vs_vampire
		has_variable = artifact_agg_vs_werewolf
		has_variable = artifact_agg_vs_demon
		has_variable = artifact_agg_vs_fera
		has_variable = artifact_agg_vs_wyrm_tainted
		has_variable = artifact_agg_except_werewolf
	}
}

# Attacker scope. $VICTIM$ = the character being wounded. True when any tag on
# the wielded weapon matches this victim. Each arm pairs the tag with the same
# splat predicate the old per-artifact arm used, so the conversion was
# behaviour-preserving.
POD_weapon_artifact_bane_applies_trigger = {
	OR = {
		AND = {
			any_equipped_character_artifact = { has_variable = artifact_agg_vs_vampire }
			$VICTIM$ = { POD_is_vampire_or_abomination_trigger = yes }
		}
		AND = {
			any_equipped_character_artifact = { has_variable = artifact_agg_vs_werewolf }
			$VICTIM$ = { has_trait = werewolf }
		}
		AND = {
			any_equipped_character_artifact = { has_variable = artifact_agg_vs_demon }
			$VICTIM$ = { has_trait = demon }
		}
		AND = {
			any_equipped_character_artifact = { has_variable = artifact_agg_vs_fera }
			$VICTIM$ = { has_trait = fera }
		}
		AND = {
			any_equipped_character_artifact = { has_variable = artifact_agg_vs_wyrm_tainted }
			$VICTIM$ = { POD_is_wyrm_servant_trigger = yes }
		}
		AND = {
			any_equipped_character_artifact = { has_variable = artifact_agg_except_werewolf }
			$VICTIM$ = { NOT = { has_trait = werewolf } }
		}
	}
}

POD_weapon_artifact_forces_agg_trigger = {
	OR = {
		any_equipped_character_artifact = {
			has_variable = blade_of_the_kings
			# Nested, for the same reason as POD_weapon_artifact_is_class_trigger
			# above: neither AND nor OR short-circuits, so a bare sibling read would
			# fire on every equipped artifact of every attacker. artifact_power only
			# exists on forged crafting artifacts.
			trigger_if = {
				limit = { has_variable = artifact_power }
				var:artifact_power >= $VICTIM$.prowess
			}
			trigger_else = { always = no }
		}
		# The conditional opt-in: a weapon that only bites a particular kind of
		# prey tags itself with artifact_agg_vs_* and this one arm resolves it,
		# so a new bane weapon needs no edit in this file either.
		AND = {
			any_equipped_character_artifact = {
				POD_weapon_artifact_has_conditional_bane_trigger = yes
			}
			POD_weapon_artifact_bane_applies_trigger = { VICTIM = $VICTIM$ }
		}
		# The generic opt-in: an artifact carrying this var joins the list
		# without needing its own arm here.
		any_equipped_character_artifact = {
			has_variable = artifact_deals_aggravated
		}
	}
}
