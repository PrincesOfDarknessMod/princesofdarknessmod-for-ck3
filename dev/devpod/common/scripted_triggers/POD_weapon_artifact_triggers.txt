######################################################
######   POD WEAPON ARTIFACT TRIGGERS
######################################################
# The wielded artifacts that promote the damage their bearer deals to
# aggravated. Read by PoD's wound classifier
# (POD_applies_superficial_as_aggravated_damage, POD_health_track_triggers.txt)
# and by the combat minigame, so the set is defined once.
#
# ADDING A WEAPON: do NOT add an arm here. Stamp
# `set_variable = artifact_deals_aggravated` on the artifact in its creation
# effect and the generic arm below picks it up - that one edit is the whole
# contract, and it is also what the move card and the AI read. The named arms
# that remain are the ones whose aggravation is CONDITIONAL (on the victim, or
# on the weapon's power), which is logic rather than a tag and has nowhere else
# to live.
#
# Attacker = the calling scope. $VICTIM$ = the character being wounded,
# needed by the arms that only bite against a particular kind of prey.
#
# Outside the minigame the classifier consults this for every wound the
# bearer deals. Inside it, the marker that reaches this list is only set
# around a weapon move's dispatch, so a sword no longer aggravates spat
# venom or a mind blast - see pod_combat_apply_single_hp_damage_effect.

POD_weapon_artifact_forces_agg_trigger = {
	OR = {
		any_equipped_character_artifact = {
			has_variable = blade_of_the_kings
			var:artifact_power >= $VICTIM$.prowess
		}
		# The generic opt-in: an artifact carrying this var joins the list
		# without needing its own arm here.
		any_equipped_character_artifact = {
			has_variable = artifact_deals_aggravated
		}
		AND = {
			any_equipped_character_artifact = {
				has_variable = unicorn_ire
			}
			$VICTIM$ = {
				faith = { has_doctrine = wyrm_tainted_doctrine }
			}
		}
		AND = {
			any_equipped_character_artifact = {
				OR = {
					has_variable = grand_klaive
					has_variable = yuri_klaive
				}
			}
			$VICTIM$ = {
				NOT = { has_trait = werewolf }
			}
		}
		AND = {
			any_equipped_character_artifact = {
				has_variable = ivory_bow
			}
			$VICTIM$ = {
				OR = {
					POD_is_vampire_or_abomination_trigger = yes
					has_trait = werewolf
				}
			}
		}
		AND = {
			any_equipped_character_artifact = {
				has_variable = flagrum_passion
			}
			$VICTIM$ = {
				OR = {
					POD_is_vampire_or_abomination_trigger = yes
					has_trait = demon
				}
			}
		}
		AND = {
			OR = {
				any_equipped_character_artifact = {
					has_variable = sword_arthur
				}
				any_equipped_character_artifact = {
					has_variable = eternal_love
				}
			}
			$VICTIM$ = {
				POD_is_vampire_or_abomination_trigger = yes
			}
		}
		AND = {
			any_equipped_character_artifact = {
				has_variable = sword_of_shu_mo_pan
			}
			$VICTIM$ = {
				has_trait = fera
			}
		}
		AND = {
			any_equipped_character_artifact = {
				has_variable = quiver_silvered_arrows
			}
			$VICTIM$ = {
				POD_weak_to_silver_trigger = yes
			}
		}
		AND = {
			any_equipped_character_artifact = { has_variable = toomler_femur }
			$VICTIM$ = { POD_is_vampire_or_abomination_trigger = yes }
		}
	}
}
