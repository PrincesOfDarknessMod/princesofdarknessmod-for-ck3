POD_possessed_trait_is_preferred_victim = {
	has_perk = possessed_slasher_perk
	$TRAIT$ = { save_temporary_scope_as = victim_trait_check }
	any_in_list = {
		variable = preferred_victim_traits
		this = scope:victim_trait_check
	}
}

# arguments: SLASHER, TARGET
# TODO PMVP: customdesc (with different perspective so it works for puppets)
POD_possessed_character_is_preferred_victim = {
	trigger_if = {
		limit = {
			exists = $TARGET$
			$TARGET$ != $SLASHER$
		}
		$TARGET$ = { save_temporary_scope_as = victim_char }
		custom_description = {
			text = POD_possessed_character_is_preferred_victim_tt
			subject = $SLASHER$
			object = $TARGET$
			
			$SLASHER$ = {
				any_in_list = {
					variable = preferred_victim_traits
					scope:victim_char = { has_trait = prev }
				}
			}
		}
	}
	trigger_else = {
		always = no
	}
}

POD_possessed_is_preferred_victim_of_local_player = {
	exists = this
	is_alive = yes
	any_player = {
		is_local_player = yes
		has_perk = possessed_slasher_perk
		POD_possessed_character_is_preferred_victim = {
			SLASHER = this
			TARGET = prev
		}
	}
}
