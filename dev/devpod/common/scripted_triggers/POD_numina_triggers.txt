POD_has_enough_perk_points_for_new_path = {
	sorcery_lifestyle_perk_points >= {
		value = 1
		add = POD_numina_sorcery_additional_perk_cost
	}
}

pod_sorcery_6th_perk_requirement = {
	has_perk = numina_$PATH$_5_perk
	OR = {
		AND = {
			has_game_rule = pod_sorcery_6th_perk_requirements_unlockable
			has_character_flag = numina_$PATH$_6_perk_flag_requirement
		}
		has_game_rule = pod_sorcery_6th_perk_requirements_always_unlocked
		has_character_flag = POD_ignore_archsorcery_requirement
	}
	NOT = { has_game_rule = pod_sorcery_6th_perk_requirements_disabled }
}

POD_splat_is_valid_for_sorcery = {
	trigger_if = {
		limit = { POD_sorcery_unrestricted = yes }
		always = yes
	}
	trigger_else = {
		NOT = { has_trait = asekh_sen }
		OR = {
			has_trait = bastet
			has_trait = kitsune
			has_trait = rune_wise
			has_main_supernatural_race_trigger = no
			trigger_if = {
				limit = { has_trait = cabiri }
				is_target_in_global_variable_list = {
					name = unavailable_unique_decisions
					target = flag:new_theourgia_arts_objective_decision
				}
			}
			has_character_flag = POD_ignore_archsorcery_requirement # No need to create 2 flags, this can be enough
		}
	}
}

POD_can_gain_perks_on_path_trigger = {
	trigger_if = {
		limit = { 
			has_variable = eiluned_sorcery_path
			NOT = { has_trait = sorcerer }
			NOT = { any_secret = { secret_type = secret_sorcerer } }
		}
		custom_tooltip = {
			text = numina_sorcery_unaccessible_path
			var:eiluned_sorcery_path ?= flag:$PATH$
		}
	}
	trigger_if = {
		limit = {
			OR = {
				flag:$PATH$ = flag:alchemy
				flag:$PATH$ = flag:conjuration
				flag:$PATH$ = flag:conveyance
				flag:$PATH$ = flag:hellfire
			}
			NOT = { has_trait = sorcerer }
			NOT = { any_secret = { secret_type = secret_sorcerer } }
		}
		NOT = { has_trait = rune_wise }
	}
}

POD_splat_is_valid_for_psychic = {
	trigger_if = {
		limit = { POD_splat_mixing_haight_mode = no }
		has_main_supernatural_race_trigger = no
	}
}
