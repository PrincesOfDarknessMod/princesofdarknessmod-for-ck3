POD_has_enough_perk_points_for_new_path = {
	OR = {
		NOT = { has_variable = POD_numina_sorcery_path_num }
		trigger_if = { limit = { has_variable = POD_numina_sorcery_path_num } sorcery_lifestyle_perk_points >= var:POD_numina_sorcery_path_num }
	}
}

pod_sorcery_6th_perk_requirement = {
	has_perk = numina_$PATH$_5_perk
	OR = {
		AND = {
			has_game_rule = pod_sorcery_6th_perk_requirements_unlockable
			has_character_flag = numina_$PATH$_6_perk_flag_requirement
		}
		has_game_rule = pod_sorcery_6th_perk_requirements_always_unlocked
	}
	NOT = { has_game_rule = pod_sorcery_6th_perk_requirements_disabled }
}

POD_splat_is_valid_for_sorcery = {
	trigger_if = {
		limit = { POD_sorcery_unrestricted = yes }
		always = yes
	}
	trigger_else = {
		OR = {
			has_trait = bastet
			has_trait = kitsune
			has_main_supernatural_race_trigger = no
		}
	}
}

POD_splat_is_valid_for_psychic = {
	trigger_if = {
		limit = { POD_splat_mixing_haight_mode = no }
		has_main_supernatural_race_trigger = no
	}
}
