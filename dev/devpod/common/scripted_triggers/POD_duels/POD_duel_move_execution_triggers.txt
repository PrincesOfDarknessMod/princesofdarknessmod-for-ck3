# this is just a quick and dirty example of a script that reads duel move attributes.
# you need three of these:
# 1. a trigger to check if you can use the move (e.g. do you have enough energy)
# 2. an effect to execute the move (e.g. do damage to the target)
# 3. another effect for the AI to evaluate duel moves to use
#
# only 3 scripts instead of 3 times 400+ is a pretty good deal
#
# character scope
# arguments: STORY (the duel move story_cycle)
POD_can_execute_duel_move = {
	can_execute_decision = $STORY$.var:decision
	
	trigger_if = {
		limit = {
			$STORY$ = {
				has_variable = ap_cost
				save_temporary_scope_value_as = {
					name = ap_cost
					value = var:ap_cost
				}
			}
		}
		# [pseudocode] check if the player has enough AP
	}
	
	trigger_if = {
		limit = {
			$STORY$ = {
				has_variable = energy_cost
				save_temporary_scope_value_as = {
					name = energy_cost
					value = var:energy_cost
				}
			}
		}
		# works for any splat
		POD_has_enough_splat_agnostic_energy_reserves = {
			AMOUNT = scope:energy_cost
		}
	}
	
	trigger_if = {
		limit = {
			$STORY$ = {
				has_variable = piety_cost
			}
		}
		piety >= $STORY$.var:piety_cost
	}
}
