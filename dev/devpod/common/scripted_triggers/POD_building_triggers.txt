# Building type scope
# args: HOLDER (the building's holder)
POD_building_type_is_incompatible = {
	OR = {
		AND = { 
			building_type_has_flag = vampireonly
			$HOLDER$ = { NOT = { POD_is_vampire_or_abomination_trigger = yes } }
		}
		AND = { 
			building_type_has_flag = vampire
			$HOLDER$ = { POD_building_requirement_vampire = no }
		}
		AND = { 
			building_type_has_flag = vamp_blood
			$HOLDER$ = { 
				OR = { 
					POD_building_requirement_vamp_blood = no
					POD_building_requirement_ghoulbro = yes
				}
			}
		}
		AND = { 
			building_type_has_flag = ghoulonly
			$HOLDER$ = { 
				OR = { 
					POD_is_any_kind_of_ghoul_trigger = no
					POD_building_requirement_ghoulbro = yes
				}
			}
		}
		AND = { 
			building_type_has_flag = gantenbeins
			$HOLDER$ = { POD_building_requirement_ghoulbro = no }
		}
		AND = { 
			building_type_has_flag = kueijin
			$HOLDER$ = { POD_building_requirement_kj = no }
		}
		AND = { 
			building_type_has_flag = inquisitor
			$HOLDER$ = { POD_building_requirement_inquisitor = no }
		}
		AND = { 
			building_type_has_flag = hunter
			$HOLDER$ = { POD_building_requirement_hunter = no }
		}
		AND = { 
			building_type_has_flag = shih
			$HOLDER$ = { POD_building_requirement_shih = no }
		}
		AND = { 
			building_type_has_flag = mummy
			$HOLDER$ = { POD_building_requirement_mummy = no }
		}
		AND = { 
			building_type_has_flag = demon
			$HOLDER$ = { POD_building_requirement_demon = no }
		}
		AND = { 
			building_type_has_flag = werewolfonly
			$HOLDER$ = { 
				NOT = { has_trait = werewolf } 
				NOT = { has_trait = kinfolk } 
			}
		}
		AND = { 
			building_type_has_flag = bastetonly
			$HOLDER$ = { 
				NOT = { has_trait = bastet } 
				NOT = { has_trait = kinfolk } 
			}
		}
		AND = { 
			building_type_has_flag = mokoleonly
			$HOLDER$ = { 
				NOT = { has_trait = mokole }
				NOT = { has_trait = kinfolk }
			}
		}
		AND = { 
			building_type_has_flag = fera
			$HOLDER$ = { 
				NOT = { has_trait = fera } 
				NOT = { has_trait = kinfolk } 
			}
		}
		AND = { 
			building_type_has_flag = wraith
			$HOLDER$ = { POD_building_requirement_wraith = no }
		}
		AND = { 
			building_type_has_flag = fae
			$HOLDER$ = { POD_building_requirement_fae = no }
		}
		AND = { 
			has_building_with_flag = exalted_only
			root = { POD_has_exalted_blood_trigger = no }
		}
		AND = { 
			has_building_with_flag = exalted_terrestrial_only
			root = { POD_has_terrestrial_exalted_blood_trigger = no }
		}
	}
}
