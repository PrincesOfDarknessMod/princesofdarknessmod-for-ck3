POD_has_resurrection = {
	OR = {
		POD_has_guaranteed_resurrection = yes
		POD_has_random_resurrection = yes
	}
}

POD_has_guaranteed_resurrection = {
	OR = {
		has_trait = mummy
		has_trait = demon
		has_trait = wraith
		has_trait = hsien
		has_trait = siochain
		trigger_if = {
			limit = { stress < 300 } 
			has_perk = ashes_of_the_phoenix_perk
		}
		has_perk = rebirthing_perk
		has_perk = sleep_of_the_hero_perk
		has_perk = draught_of_phoenix_blood_perk
		has_perk = the_fleeing_scarab_perk
		has_character_modifier = extra_life_4_modifier
		has_character_modifier = extra_life_3_modifier
		has_character_modifier = extra_life_2_modifier
		has_character_modifier = extra_life_1_modifier
		has_character_modifier = green_man_modifier
		has_character_modifier = life_warden_modifier
		has_character_modifier = ritual_the_curse_belated_modifier
		has_character_modifier = elemental_ascension_decision_modifier
		has_character_modifier = POD_immaculate_dragon_modifier
		has_character_modifier = most_beloved_of_gaia_the_primordial_modifier
	}
}

POD_has_random_resurrection = {
	OR = {
		POD_can_become_ancestor_spirit_trigger = yes
	}
}

POD_has_natural_death_second_chance_trigger = {
	OR = {
		has_variable = fetter_owner
		has_trait = hsien
		POD_will_intercept_death_reason_on_natural_death = yes
	}
}

POD_will_intercept_death_reason_on_natural_death = {
	OR = {
		has_perk = rebirthing_perk
		POD_can_become_ancestor_spirit_trigger = yes
		POD_will_resurrect_as_wraith = yes
	}
}

POD_will_resurrect_as_wraith = {
	OR = {
		exists = var:marked_for_return
		has_perk = fc_arm_r_necrotic_arm_perk
		has_perk = fc_arm_l_necrotic_arm_perk
	}
	can_convert_to_wraith_trigger = yes
}
