# Checks if any line exists containing the pair of given nodes
POD_map_has_line = {
	save_temporary_scope_value_as = {
		name = start
		value = var:start
	}
	save_temporary_scope_value_as = {
		name = end
		value = var:end
	}
	
	AND = {
		scope:start = {
			any_in_list = {
				variable = lines
				OR = {
					var:start = scope:end
					var:end = scope:end
				}
			}
		}
		scope:end = {
			any_in_list = {
				variable = lines
				OR = {
					var:start = scope:start
					var:end = scope:end
				}
			}
		}
	}
}
