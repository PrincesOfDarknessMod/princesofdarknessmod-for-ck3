
# "Scripted" contracts means these aren't meant to be part of random pools.
# These are created manually via other scripts, to be used as one-offs or historical events.


##### Some explanations for the jank:
#
# travel = yes straight up doesn't work for landed characters.
# You will travel to accept the contract, but on_accepted doesn't fire when you arrive.
# So instead, here on_accept fires an event where you can choose whether to travel.
# And the travel plan is handled by that event.
#
# There is no way to un-accept a contract after you've accepted it.
# So if you quit out of the travel plan or otherwise abort the journey, there needs to be a fallback.
# The fallback is in on_invalidated.
# Put simply, if the contract invalidates, it gets deleted and recreated as a new copy.
# For that to work properly, all internal variables are copied to the new contract during on_invalidated.
#
#
# Multiple timekeeping variables are necessary to make the expiration date work:
#
##### time_remaining
# Saves the remaining time in a format that localization/GUI can understand.
# This is done via GetVarTimeRemaining and either GetGameTimeDifferenceForDiffDays or GetCurrentDateWithDiff.
#
##### end_date
# Saves the remaining time in a format that script can understand.
# Necessary for recreating the contract.
#
##### POD_task_contracts.1
# An event that deletes the contract (defined in events\POD_task_contracts\POD_task_contract_events.txt)
# Necessary to make the contract expire on an exact date.
# Otherwise, the date would get fuzzy (valid_to_continue and valid_to_keep are only checked once a month).
#
##### POD_days_remaining_for_task_contract
# A script value (defined in common\script_values\POD_task_contract_values.txt)
# Necessary because you can't use the inline value = {} syntax for time delays.
#
#
# Weirdly, abandoning or completing a contract doesn't actually delete it immediately.
# It gets hidden in the GUI, but scripts can still access the scope if it was saved anywhere.
# Hence, events will need safeguards to check whether the scoped contract is still "active".
# E.g. via any_character_active_contract = { scope:task_contract ?= this }
#
# What seems to happen is that abandoning or completing a contract silently invalidates it after a month.
# Which also fires on_invalidated.
# Hence, I also set a contract_completed variable on completion, and check for it during on_invalidated.
# So that the contract doesn't get recreated after successful completion.
#
#
# All of this jank would become obsolete if travel = yes worked properly for landed characters.
# Fingers crossed?

		}

		save_scope_as = task_contract
		task_contract_taker = { trigger_event = POD_natural_disasters.0003 }
	}
	on_create = {
		scope:contract = {
			set_variable = retain_despite_distance
			# other variables are set when first creating the contract
			# in POD_natural_disasters.0002
		}
	}
	
	on_completed = {
		set_variable = contract_completed
		# travel_plan variable is set in POD_task_contracts.3
		var:travel_plan ?= {
			reroute_to_home = yes
		}
	}
	
	on_invalidated = {
		# For creating a new copy of the contract if it was abandoned (e.g. from getting imprisoned)
		if = {
			limit = {
				POD_compare_current_date_with_value = {
					COMPARATOR = "<"
					VALUE = var:end_date
				}
				NOT = { has_variable = contract_completed }
			}
			hidden_effect_new_object = {
				task_contract_taker = {
					create_task_contract = {
						task_contract_type = POD_natural_disaster
						task_contract_tier = prev.task_contract_tier
						location = prev.task_contract_location
						save_scope_as = task_contract
					}
					scope:task_contract = {
						set_variable = {
							name  = disaster_loc
							value = root.var:disaster_loc
						}
						set_variable = {
							name  = disaster_type
							value = root.var:disaster_type
						}
						set_variable = {
							name  = end_date
							value = root.var:end_date
						}
						set_variable = {
							name  = time_remaining
							value = yes
							# needs to be a script value since inline values don't work here
							days  = POD_days_remaining_for_task_contract
						}
						trigger_event = {
							id   = POD_task_contracts.1
							days = POD_days_remaining_for_task_contract
						}
						set_variable = {
							name  = target_title
							value = root.var:target_title
						}
					}
				}
			}
		}
		# travel_plan variable is set in POD_task_contracts.3
		var:travel_plan ?= {
			reroute_to_home = yes
		}
	}
