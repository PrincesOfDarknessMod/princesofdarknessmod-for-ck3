@pod_slayer_contract_time_limit = 1 # in years

POD_assault_lair = {
	group = POD_contracts_slayers
	icon = "gfx/interface/icons/traits/supehunter.dds"

	use_diplomatic_range = yes
	travel = no

	valid_to_create = {
		POD_splat_can_inquihunt_trigger = yes
		POD_valid_target_for_inquihunt_trigger = {
			HUNTER = root
			HUNTED = scope:employer
		}
		POD_valid_contract_basic_trigger = {
			EMPLOYER = scope:employer
			LAAMP = root
		}
	}
	valid_to_accept = {
		is_available_allow_travelling = yes
		POD_valid_contract_basic_trigger = {
			EMPLOYER = scope:employer
			LAAMP = root
		}
	}
	valid_to_continue = {
		POD_contract_has_not_expired = yes
	}
	valid_to_keep = {
		POD_contract_has_not_expired = yes
	}

	weight = {
		value = task_contract_weight_default_value
		# Let's make sure that contracts with other Inquisitors are prioritized (for now)
		#add = POD_contracts_weight_up_slayers_value
		scope:employer = { multiply = task_contract_weight_by_tier_value }
	}

	# On_actions
	on_accepted = {
		save_scope_as = task_contract
		task_contract_taker = { trigger_event = POD_inquihunt.7010 }
	}
	on_create = {
		scope:contract = {
			#set_variable = retain_despite_distance
			set_variable = recreate_on_travel_abort
			set_variable = hide_employer
			POD_setup_timekeeping_vars_for_contract = {
				YEARS_REMAINING = @pod_slayer_contract_time_limit
			}
		}
	}
	
	on_completed = {
		set_variable = do_not_recreate
		# travel_plan variable is set in POD_task_contracts.4
		if = {
			limit = { exists = var:travel_plan }
			task_contract_taker = { return_home = yes }
		}
	}
	
	on_invalidated = {
		# For creating a new copy of the contract if it was abandoned (e.g. from getting imprisoned)
		if = {
			limit = {
				POD_contract_has_not_expired = yes
				NOT = { has_variable = do_not_recreate }
			}
			POD_recreate_contract_with_timekeeping_vars = {
				CONTRACT_TYPE = POD_assault_lair
				SAVE_AS = task_contract
			}
		}
		# travel_plan variable is set in POD_task_contracts.4
		if = {
			limit = { exists = var:travel_plan }
			task_contract_taker = { return_home = yes }
		}
	}

	# Rewards
	task_contract_reward = {
		success_standard = {
			effect = {
				task_contract_taker = {
					play_sound_effect = "event:/DLC/EP3/SFX/UI/Contracts/ep3_ui_contracts_success_on_contract"
					custom_tooltip = POD_assault_lair_contract_success_tt
					# TODO: show tooltip?
				}
			}
		}
		failure_abandoned_hunt = {
			positive = no
			effect = {
				task_contract_taker = {
					play_sound_effect = "event:/DLC/EP3/SFX/UI/Contracts/ep3_ui_contracts_failure_on_contract"
					custom_tooltip = NO_EFFECT
				}
			}
		}
		failure_failed_hunt = {
			positive = no
			effect = {
				task_contract_taker = {
					play_sound_effect = "event:/DLC/EP3/SFX/UI/Contracts/ep3_ui_contracts_failure_on_contract"
					show_as_tooltip = {
						pod_death_no_killer = {
							death_reason = death_murder
						}
					}
				}
			}
		}
	}
}
