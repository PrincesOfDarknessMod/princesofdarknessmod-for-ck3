POD_ghoul_mission_kill_hunter = {
	group = POD_ghoul_missions
	icon = "gfx/interface/icons/traits/ghoul.dds"

	use_diplomatic_range = yes
	travel = no

	valid_to_create = {
		POD_is_bound_ghoul_of = { DOMITOR = scope:employer }
	}
	valid_to_accept = {
		# always valid since the contract gets auto-accepted
	}
	valid_to_continue = {
		POD_ghoul_mission_can_be_finished = yes
	}
	valid_to_keep = {
		# irrelevant since contract gets auto-accepted
	}

	on_create = {
		scope:contract = {
			POD_save_local_culture_for_task_contract = {
				SAVE_AS = culture
			}
			create_character = {
				age = { 16 24 }
				location = task_contract_location
				culture = scope:culture
				template = POD_predatorstories_hunter_agent_template
				gender_female_chance = 50
				save_scope_as = mortal
				after_creation = {
					add_character_flag = wear_armor
					set_variable = {
						name = ghoul_mission_targeted_by
						value = scope:contract
					}
				}
			}
			set_variable = {
				name  = target
				value = scope:mortal
			}
		}
		POD_ghoul_mission_task_contract_on_create = yes
	}

	on_completed = {
		POD_ghoul_mission_task_contract_on_completed = yes
	}

	on_invalidated = {
		POD_ghoul_mission_task_contract_on_invalidated = yes
	}

	task_contract_reward = {
		success_standard = {
			effect = {
				POD_ghoul_mission_task_contract_success = yes
			}
		}
		failure_standard = {
			positive = no
			effect = {
				POD_ghoul_mission_task_contract_failure = yes
			}
		}
	}
}

POD_ghoul_mission_abduct_desirable_mortal = {

	icon = "gfx/interface/icons/traits/ghoul.dds"

	use_diplomatic_range = yes
	travel = no

	valid_to_create = {
		POD_is_bound_ghoul_of = { DOMITOR = scope:employer }
	}
	valid_to_accept = {
		# always valid since the contract gets auto-accepted
	}
	valid_to_continue = {
		POD_ghoul_mission_can_be_finished = yes
	}
	valid_to_keep = {
		# irrelevant since contract gets auto-accepted
	}

	on_create = {
		scope:contract = {
			POD_save_local_culture_for_task_contract = {
				SAVE_AS = culture
			}
			create_character = {
				location = task_contract_location
				culture = scope:culture
				template = POD_ghoul_mission_desirable_mortal_character
				gender_female_chance = 50
				save_scope_as = mortal
				after_creation = {
					set_variable = {
						name = ghoul_mission_targeted_by
						value = scope:contract
					}
				}
			}
			set_variable = {
				name  = target
				value = scope:mortal
			}
		}
		POD_ghoul_mission_task_contract_on_create = yes
	}

	on_completed = {
		POD_ghoul_mission_task_contract_on_completed = yes
	}

	on_invalidated = {
		POD_ghoul_mission_task_contract_on_invalidated = yes
	}

	task_contract_reward = {
		success_standard = {
			effect = {
				POD_ghoul_mission_task_contract_success = yes
				var:target = {
					if = {
						limit = { is_imprisoned = yes }
						release_from_prison = yes
					}
				}
				if = {
					limit = { task_contract_employer = { is_ruler = yes } }
					task_contract_employer = {
						add_courtier = prev.var:target
					}
					var:target = {
						hidden_effect = { return_to_court = yes }
					}
				}
				else = {
					hidden_effect = {
						var:target = { death = { death_reason = death_disappearance } }
					}
				}
			}
		}
		failure_standard = {
			positive = no
			effect = {
				POD_ghoul_mission_task_contract_failure = yes
			}
		}
	}
}
