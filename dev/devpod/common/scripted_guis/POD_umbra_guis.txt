spirit_travel_enable = {
	scope = character
	saved_scopes = { player }
	is_shown = {
		scope:player ?= {
			is_alive = yes
			OR = {
				has_access_to_astral_reaches_trigger = yes
				has_access_to_spirit_wilds_trigger = yes
				has_access_to_dreaming_trigger = yes
				has_access_to_underworld_trigger = yes
				has_access_to_deep_umbra_trigger = yes
				# Enoch
				capital_county ?= { has_county_modifier = link_enoch_modifier }
				
				is_in_umbra_trigger = yes
			}
		}
	}
}

umbra_initiate = {
	scope = character
	effect = {
		if = {
			limit = { NOT = { has_variable_list = known_umbra_realms_list } }
			POD_create_umbra_location = { NAME = flag:earth }
		}
		if = {
			limit = { NOT = { POD_umbra_every_neighboring_realm_known_trigger = { LOCATION = flag:earth } } }
			POD_umbra_set_neighboring_realm_known_effect = { LOCATION = flag:earth }
		}
	}
}

umbra_diplomacy_train_start_choose = {
	scope = character
	saved_scopes = { explorer }
	confirm_title = umbra_diplomacy_train_start_confirm
	confirm_text = umbra_quarterly_action_start
	is_shown = {
		NOT = { has_variable = umbra_diplomacy_train }
	}
	is_valid = {
		custom_tooltip = {
			text = umbra_action_no_explorer_tooltip
			exists = scope:explorer
		}
		custom_tooltip = {
			text = umbra_action_cooldown_tooltip
			NOT = {
				is_target_in_variable_list = {
					name = umbra_action_cooldown
					target = flag:umbra_diplomacy_train_start
				}
			}
		}
	}
	effect = {
		add_to_variable_list = {
			name = umbra_action_cooldown
			target = flag:umbra_diplomacy_train_start
		}
		set_variable = {
			name = umbra_diplomacy_train
			value = 1
		}
		custom_tooltip = umbra_quarterly_action_start
		add_character_modifier = umbra_action_diplomacy_train_modifier
		show_as_tooltip = { umbra_train_effect = { SKILL = diplomacy LEVEL = 1 } }
		trigger_event = POD_umbra_monthly.1001
	}
}

umbra_diplomacy_train_next_choose = {
	scope = character
	saved_scopes = { explorer }
	is_shown = {
		has_variable = umbra_diplomacy_train
		var:umbra_diplomacy_train <= 6
	}
	is_valid = {
		custom_tooltip = {
			text = umbra_action_no_explorer_tooltip
			exists = scope:explorer
		}
		trigger_if = {
			limit = { var:umbra_diplomacy_train = 6 }
			custom_description = {
				text = level_of_umbra_knowledge
				value = 5
				explorer_umbra_level >= 5
			}
		}
		trigger_else_if = {
			limit = { var:umbra_diplomacy_train = 5 }
			custom_description = {
				text = level_of_umbra_knowledge
				value = 4
				explorer_umbra_level >= 4
			}
		}
		trigger_else_if = {
			limit = { var:umbra_diplomacy_train = 4 }
			custom_description = {
				text = level_of_umbra_knowledge
				value = 3
				explorer_umbra_level >= 3
			}
		}
		trigger_else_if = {
			limit = { var:umbra_diplomacy_train = 3 }
			custom_description = {
				text = level_of_umbra_knowledge
				value = 2
				explorer_umbra_level >= 2
			}
		}
		trigger_else_if = {
			limit = { var:umbra_diplomacy_train = 2 }
			custom_description = {
				text = level_of_umbra_knowledge
				value = 1
				explorer_umbra_level >= 1
			}
		}
		trigger_else = {
		}
	}
	effect = {
		change_variable = {
			name = umbra_diplomacy_train
			add = 1
		}
		save_scope_value_as = {
			name = next_level
			value = {
				value = var:umbra_diplomacy_train
				add = 1
			}
		}
		hidden_effect = { remove_character_modifier = umbra_action_diplomacy_train_modifier }
		add_character_modifier = umbra_action_diplomacy_train_modifier
		custom_tooltip = umbra_quarterly_action_start
		show_as_tooltip = { umbra_train_effect = { SKILL = diplomacy LEVEL = scope:next_level } }
	}
}

			}
			root = {
				add_to_variable_list = {
					name = potential_umbra_explorer
					target = prev
				}
			}
		}
	}
}

umbra_expedition_choose = {
	saved_scopes = { location }
	scope = character
	is_shown = {}
	is_valid = {
		custom_tooltip = {
			text = umbra_cooldown_tt
			NOT = { any_memory = { has_memory_type = umbra_expedition_memory } }
		}
		custom_tooltip = {
			text = umbra_ongoing_tt
			NOT = { has_variable = umbra_expedition_postion }
		}
		POD_umbra_travel_valid_trigger = { LOCATION = scope:location }
	}
	effect = {
		if = {
			limit = { has_variable = spirit_location }
			set_variable = {
				name = umbra_origin_location
				value = var:spirit_location
			}
		}
		if = {
			limit = {
				OR = {
					NOT = { has_variable = spirit_location }
					NOT = { var:spirit_location = scope:location }
				}
			}
			set_variable = { name = spirit_destination value = scope:location }
			show_as_tooltip = { POD_umbra_send_effect = { LOCATION = scope:location } }
			if = {
				limit = { POD_umbra_bypass_trigger = { LOCATION = scope:location } }
				trigger_event = { on_action = spirit_travel_movement_end_pulse }
			}
			else = {
				trigger_event = { on_action = spirit_travel_movement_start_pulse }
			}
		}
		if = {
			limit = { var:umbra_stationed_realm.var:name ?= scope:location }
			set_variable = { name = current_umbra_discovery value = var:umbra_discovery_type }
			show_as_tooltip = { POD_umbra_trigger_expedition_discovery_effect = { DISCOVERY = var:umbra_discovery_type } }
		}
		else = {
			random_relation = {
				type = umbra_explorer
				limit = { scope:location = var:umbra_stationed_realm.var:name }
				root = { 
					set_variable = { name = current_umbra_discovery value = prev.var:umbra_discovery_type } 
					show_as_tooltip = { POD_umbra_trigger_expedition_discovery_effect = { DISCOVERY = prev.var:umbra_discovery_type } }
				}
			}
		}
		custom_tooltip = {
			text = umbra_expedition_tooltip
			POD_umbra_expedition_initiate_effect = yes
		}
	}
}
