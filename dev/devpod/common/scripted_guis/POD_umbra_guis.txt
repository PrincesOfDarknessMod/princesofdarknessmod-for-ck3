spirit_travel_enable = {
	scope = character
	saved_scopes = { player }
	is_shown = {
		scope:player ?= {
			is_alive = yes
			OR = {
				has_access_to_astral_reaches_trigger = yes
				has_access_to_spirit_wilds_trigger = yes
				has_access_to_dreaming_trigger = yes
				has_access_to_underworld_trigger = yes
				has_access_to_deep_umbra_trigger = yes
				
				has_variable_list = known_umbra_realms_list
				is_in_umbra_trigger = yes
			}
		}
	}
}

umbra_initiate = {
	scope = character
	effect = {
		if = {
			limit = { NOT = { has_global_variable_list = umbra_type_list } }
			add_to_global_variable_list = { name = umbra_type_list target = flag:all }
			add_to_global_variable_list = { name = umbra_type_list target = flag:stationed }
			add_to_global_variable_list = { name = umbra_type_list target = flag:unstationed }
			add_to_global_variable_list = { name = umbra_type_list target = flag:clear }
			add_to_global_variable_list = { name = umbra_type_list target = flag:unclear }
			add_to_global_variable_list = { name = umbra_type_list target = flag:discovery }
			add_to_global_variable_list = { name = umbra_type_list target = flag:realm }
			add_to_global_variable_list = { name = umbra_type_list target = flag:astral_reaches }
			add_to_global_variable_list = { name = umbra_type_list target = flag:spirit_wilds }
			add_to_global_variable_list = { name = umbra_type_list target = flag:underworld }
			add_to_global_variable_list = { name = umbra_type_list target = flag:dreaming }
			add_to_global_variable_list = { name = umbra_type_list target = flag:deep_umbra }
			add_to_global_variable_list = { name = umbra_type_list target = flag:penumbra }
			add_to_global_variable_list = { name = umbra_type_list target = flag:zone }
		}
		if = {
			limit = { NOT = { has_variable_list = known_umbra_realms_list } }
			POD_create_umbra_location = { NAME = flag:earth }
		}
		if = {
			limit = { NOT = { POD_umbra_every_neighboring_realm_known_trigger = { LOCATION = flag:earth } } }
			POD_umbra_set_neighboring_realm_known_effect = { LOCATION = flag:earth }
		}
		if = {
			limit = { NOT = { has_variable = umbra_filter } }
			set_variable = { name = umbra_filter value = flag:all }
		}
	}
}

POD_filter_umbra_realms = {
	scope = character
	saved_scopes = { realm } 
	is_shown = {
		trigger_if = {
			limit = { var:umbra_filter = flag:all }
		}
		trigger_else_if = {
			limit = { var:umbra_filter = flag:stationed }
			scope:realm = { has_variable = explorer }
		}
		trigger_else_if = {
			limit = { var:umbra_filter = flag:unstationed }
			scope:realm = { NOT = { has_variable = explorer } }
		}
		trigger_else_if = {
			limit = { var:umbra_filter = flag:clear }
			POD_umbra_every_neighboring_realm_known_trigger = { LOCATION = scope:realm.var:name }
		}
		trigger_else_if = {
			limit = { var:umbra_filter = flag:unclear }
			NOT = { POD_umbra_every_neighboring_realm_known_trigger = { LOCATION = scope:realm.var:name } }
		}
		trigger_else_if = {
			limit = { var:umbra_filter = flag:discovery }
			scope:realm = { has_variable = umbra_discovery_type }
		}
		trigger_else_if = {
			limit = { var:umbra_filter = flag:realm }
			in_umbra_realm_trigger = { LOCATION = scope:realm.var:name }
		}
		trigger_else = {
			scope:realm.var:type = var:umbra_filter
		}
	} 
}

umbra_diplomacy_train_start_choose = {
	scope = character
	saved_scopes = { explorer }
	confirm_title = umbra_diplomacy_train_start_confirm
	confirm_text = umbra_quarterly_action_start
	is_shown = {
		NOT = { has_variable = umbra_diplomacy_train }
	}
	is_valid = {
		custom_tooltip = {
			text = umbra_action_no_explorer_tooltip
			exists = scope:explorer
		}
		custom_tooltip = {
			text = umbra_action_cooldown_tooltip
			NOT = {
				is_target_in_variable_list = {
					name = umbra_action_cooldown
					target = flag:umbra_diplomacy_train_start
				}
			}
		}
	}
	effect = {
		add_to_variable_list = {
			name = umbra_action_cooldown
			target = flag:umbra_diplomacy_train_start
		}
		set_variable = {
			name = umbra_diplomacy_train
			value = 1
		}
		custom_tooltip = umbra_quarterly_action_start
		add_character_modifier = umbra_action_diplomacy_train_modifier
		show_as_tooltip = { umbra_train_effect = { SKILL = diplomacy LEVEL = 1 } }
		trigger_event = POD_umbra_monthly.1001
	}
}

			limit = { var:umbra_diplomacy_train = 2 }

		}

		}
	}
	effect = {
		if = {
			limit = { var:umbra_diplomacy_train = 6 }

		}
		else_if = {

potential_umbra_explorer_initiate = {
	scope = character
	saved_scopes = { explorer realm }
	effect = {
		if = {
			limit = { exists = scope:explorer }
			if = {
				limit = { scope:explorer.var:umbra_stationed_realm = scope:realm }
				set_variable = { name = local_umbra_explorer value = scope:explorer }
			}
			else = {
				scope:realm = { remove_variable = explorer }
			}
		}
		set_variable = { name = local_umbra_realm value = scope:realm }
		add_to_variable_list = {
			name = potential_umbra_explorer
			target = this
		}
		every_courtier_or_guest = {
			limit = {
				OR = {
					has_access_to_astral_reaches_trigger = yes 
					has_access_to_spirit_wilds_trigger = yes 
					has_access_to_underworld_trigger = yes 
					has_access_to_dreaming_trigger = yes 
					has_access_to_deep_umbra_trigger = yes 
				}
			}
			root = {
				add_to_variable_list = {
					name = potential_umbra_explorer
					target = prev
				}
			}
		}
		if = {
			limit = { POD_has_coterie_members_trigger = yes }
			var:coterie = {
				every_in_list = {
					variable = members
					limit = {
						is_ai = yes
					}
					root = {
						add_to_variable_list = {
							name = potential_umbra_explorer
							target = prev
						}
					}
				}
			}
		}
	}
}

umbra_expedition_choose = {
	saved_scopes = { realm }
	scope = character
	is_shown = {}
	is_valid = {
		custom_tooltip = {
			text = umbra_cooldown_tt
			NOT = { any_memory = { has_memory_type = umbra_expedition_memory } }
		}
		custom_tooltip = {
			text = umbra_ongoing_tt
			NOT = { has_variable = umbra_map }
		}
		POD_umbra_travel_valid_trigger = { LOCATION = scope:realm.var:name }
	}
	effect = {
		if = {
			limit = { has_variable = spirit_location }
			set_variable = {
				name = umbra_origin_location
				value = var:spirit_location
			}
		}
		if = {
			limit = {
				OR = {
					NOT = { has_variable = spirit_location }
					NOT = { var:spirit_location = scope:realm.var:name }
				}
			}
			set_variable = { name = spirit_destination value = scope:realm.var:name }
			show_as_tooltip = { POD_umbra_travel_effect = { LOCATION = scope:realm.var:name } }
			if = {
				limit = { POD_umbra_bypass_trigger = { LOCATION = scope:realm.var:name } }
				trigger_event = { on_action = spirit_travel_movement_end_pulse }
			}
			else = {
				trigger_event = { on_action = spirit_travel_movement_start_pulse }
			}
		}
		set_variable = { name = current_umbra_discovery value = scope:realm.var:umbra_discovery_type }
		show_as_tooltip = { POD_umbra_trigger_expedition_discovery_effect = { DISCOVERY = scope:realm.var:umbra_discovery_type LOCATION = scope:realm.var:name } }
		custom_tooltip = {
			text = umbra_expedition_tooltip
			if = {
				limit = {
					POD_is_currently_in_umbra_expedition_trigger = no
				}
				POD_create_umbra_expedition = yes
			}
		}
	}
}
