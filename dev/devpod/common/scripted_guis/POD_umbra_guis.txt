potential_umbra_explorer_initiate = {
	scope = character
	effect = {
		clear_variable_list = potential_umbra_explorer
		add_to_variable_list = {
			name = potential_umbra_explorer
			target = this
		}
		if = {
			limit = { is_in_astral_reaches_trigger = yes }
			every_courtier_or_guest = {
				limit = { has_access_to_astral_reaches_trigger = yes }
				root = {
					add_to_variable_list = {
						name = potential_umbra_explorer
						target = prev
					}
				}
			}
		}
		else_if = {
			limit = { is_in_spirit_wilds_trigger = yes }
			every_courtier_or_guest = {
				limit = { has_access_to_spirit_wilds_trigger = yes }
				root = {
					add_to_variable_list = {
						name = potential_umbra_explorer
						target = prev
					}
				}
			}
		}
		else_if = {
			limit = { is_in_underworld_trigger = yes }
			every_courtier_or_guest = {
				limit = { has_access_to_underworld_trigger = yes }
				root = {
					add_to_variable_list = {
						name = potential_umbra_explorer
						target = prev
					}
				}
			}
		}
		else_if = {
			limit = { is_in_dreaming_trigger = yes }
			every_courtier_or_guest = {
				limit = { has_access_to_dreaming_trigger = yes }
				root = {
					add_to_variable_list = {
						name = potential_umbra_explorer
						target = prev
					}
				}
			}
		}
		else_if = {
			limit = { is_in_deep_umbra_trigger = yes }
			every_courtier_or_guest = {
				limit = { has_access_to_deep_umbra_trigger = yes }
				root = {
					add_to_variable_list = {
						name = potential_umbra_explorer
						target = prev
					}
				}
			}
		}
	}
}

assign_umbra_explorer_choose = {
	scope = character
	is_shown = {
		NOT = { has_variable = local_umbra_explorer }
	}
	is_valid = {
		can_assign_umbra_explorer_trigger = yes
		scope:courtier = { can_be_umbra_explorer_trigger = yes }
	}
	effect = {
		if = {
			limit = { 
				exists = scope:courtier 
				NOT = { root = scope:courtier } 
			} 
			set_relation_umbra_explorer = scope:courtier
			scope:courtier = { 
				set_variable = { name = umbra_stationed_location value = prev.var:spirit_location }
				set_variable = { name = umbra_exploration_progress value = 0 }
			}
			set_variable = { name = local_umbra_explorer value = scope:courtier }
		}
		else = {
			set_variable = { name = umbra_stationed_location value = var:spirit_location }
			set_variable = { name = umbra_exploration_progress value = 0 }
			set_variable = { name = local_umbra_explorer value = root }
		}
	}
}

swap_umbra_explorer_choose = {
	scope = character
	is_shown = {
		has_variable = local_umbra_explorer
		trigger_if = {
			limit = { 
				exists = scope:courtier 
				NOT = { root = scope:courtier } 
			}
			NOT = { var:local_umbra_explorer = scope:courtier }
		}
		trigger_else = {
			NOT = { var:local_umbra_explorer = root }
		}
	}
	is_valid = {
		var:local_umbra_explorer = {
			custom_tooltip = {
				text = umbra_interest_memory
				NOT = { any_memory = { has_memory_type = umbra_interest_memory } }
			}
		}
	}
	effect = {
		if = {
			limit = { 
				exists = scope:courtier 
				NOT = { root = scope:courtier } 
			}
			if = { 
				limit = { scope:courtier = { has_variable = umbra_stationed_location } }
				var:local_umbra_explorer = { 
					set_variable = { name = umbra_stationed_location value = scope:courtier.var:umbra_stationed_location } 
					set_variable = { name = umbra_exploration_progress value = 0 }
				}
			}
			else = {
				set_relation_umbra_explorer = scope:courtier
				POD_remove_umbra_explorer_effect = { TARGET = var:local_umbra_explorer }
			}
			scope:courtier = { 
				set_variable = { name = umbra_stationed_location value = root.var:spirit_location }
				set_variable = { name = umbra_exploration_progress value = 0 }
			}
			set_variable = { name = local_umbra_explorer value = scope:courtier }
		}
		else = {
			if = { 
				limit = { has_variable = umbra_stationed_location }
				var:local_umbra_explorer = { 
					set_variable = { name = umbra_stationed_location value = root.var:umbra_stationed_location } 
					set_variable = { name = umbra_exploration_progress value = 0 }
				}
			}
			else = {
				POD_remove_umbra_explorer_effect = { TARGET = var:local_umbra_explorer }
			}
			set_variable = { name = umbra_stationed_location value = var:spirit_location }
			set_variable = { name = umbra_exploration_progress value = 0 }
			set_variable = { name = local_umbra_explorer value = root }
		}
	}
}

umbra_expedition_choose = {
	scope = character
	is_shown = {
		POD_umbra_interest_trigger = { LOCATION = root.var:spirit_location }
		NOT = { has_variable = spirit_destination }
	}
	is_valid = {
		custom_tooltip = {
			text = umbra_cooldown_tt
			NOT = { any_memory = { has_memory_type = umbra_expedition_memory } }
		}
		POD_is_not_trapped_in_umbra_basic_trigger = yes
	}
	effect = {
		set_variable = { name = spirit_destination value = var:spirit_location }
		if = {
			limit = { var:umbra_stationed_location ?= var:spirit_location }
			set_variable = { name = current_umbra_interest value = var:umbra_interest_type }
			show_as_tooltip = { POD_umbra_trigger_expedition_interest_effect = { INTEREST = var:umbra_interest_type } }
		}
		else = {
			random_relation = {
				type = umbra_explorer
				limit = { root.var:spirit_location = var:umbra_stationed_location }
				root = { 
					set_variable = { name = current_umbra_interest value = prev.var:umbra_interest_type } 
					show_as_tooltip = { POD_umbra_trigger_expedition_interest_effect = { INTEREST = prev.var:umbra_interest_type } }
				}
			}
		}
		custom_tooltip = {
			text = umbra_expedition_tooltip
			POD_umbra_adv_initiate_effect = yes
		}
	}
}

umbra_map = {
	scope = character
	is_shown = {
		NOT = { has_variable = spirit_destination }
		has_variable = spirit_location 
		POD_is_not_trapped_in_umbra_trigger = yes
		is_in_shadow_realm_trigger = no
	}
	effect = {
		clear_variable_list = stationed_locations_list
		if = {
			limit = { has_variable = umbra_stationed_location }
			add_to_variable_list = { name = stationed_locations_list target = var:umbra_stationed_location }
		}
		every_relation = {
			type = umbra_explorer
			root = { add_to_variable_list = { name = stationed_locations_list target = prev.var:umbra_stationed_location } }
		}
	}
}
