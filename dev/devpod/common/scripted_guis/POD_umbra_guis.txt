spirit_travel_enable = {
	scope = character
	is_valid = {
		OR = {
			has_access_to_astral_reaches_trigger = yes
			has_access_to_spirit_wilds_trigger = yes
			has_access_to_dreaming_trigger = yes
			has_access_to_underworld_trigger = yes
			# Periphery
			AND = {
				has_supernatural_race_trigger = no
				has_trait = lunatic
				has_trait = poet
			}
			# Enoch
			capital_county = { has_county_modifier = link_enoch_modifier }
			
			has_trait = demon
			is_in_umbra_trigger = yes
		}
	}
}

# Explorer

potential_umbra_explorer_initiate = {
	scope = character
	effect = {
		clear_variable_list = potential_umbra_explorer
		add_to_variable_list = {
			name = potential_umbra_explorer
			target = this
		}
		if = {
			limit = { is_in_astral_reaches_trigger = yes }
			every_courtier = {
				limit = { has_access_to_astral_reaches_trigger = yes }
				root = {
					add_to_variable_list = {
						name = potential_umbra_explorer
						target = prev
					}
				}
			}
		}
		else_if = {
			limit = { is_in_astral_reaches_trigger = yes }
			every_courtier = {
				limit = { has_access_to_spirit_wilds_trigger = yes }
				root = {
					add_to_variable_list = {
						name = potential_umbra_explorer
						target = prev
					}
				}
			}
		}
		else_if = {
			limit = { is_in_underworld_trigger = yes }
			every_courtier = {
				limit = { has_access_to_underworld_trigger = yes }
				root = {
					add_to_variable_list = {
						name = potential_umbra_explorer
						target = prev
					}
				}
			}
		}
		else_if = {
			limit = { is_in_dreaming_trigger = yes }
			every_courtier = {
				limit = { has_access_to_dreaming_trigger = yes }
				root = {
					add_to_variable_list = {
						name = potential_umbra_explorer
						target = prev
					}
				}
			}
		}
	}
}
