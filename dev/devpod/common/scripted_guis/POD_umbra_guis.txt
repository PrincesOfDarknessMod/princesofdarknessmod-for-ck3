spirit_travel_enable = {
	scope = character
	is_valid = {
		OR = {
			has_access_to_astral_reaches_trigger = yes
			has_access_to_spirit_wilds_trigger = yes
			has_access_to_dreaming_trigger = yes
			has_access_to_underworld_trigger = yes
			# Periphery
			AND = {
				OR = {
					has_trait = mortal
					has_trait = supehunter
					has_trait = kinfolk
				}
				OR = {
					has_trait = lunatic_1
					has_trait = lunatic_genetic
				}
				has_trait = poet
			}
			# Enoch
			capital_county = { has_county_modifier = link_enoch_modifier }
			
			has_character_modifier = traveling_spirit_world_modifier
		}
	}
}

				text = number_of_grace

		}

	effect = {

				text = remove_grace

				}
			}
		}
		if = {

				text = remove_grace
