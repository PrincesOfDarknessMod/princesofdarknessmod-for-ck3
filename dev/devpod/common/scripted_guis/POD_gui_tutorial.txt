### SGUI for the GUI tutorial.
# For the rest of the tutorial, check out gui\event_window_widgets\POD_event_window_widget_character_selection.gui
# 
# You can access the SGUI with the GetScriptedGui() datafunction, passing its name as the argument.
# So for this one: GetScriptedGui('POD_character_list_select')
POD_character_list_select = {
	# You need to set an expected scope type for the root scope.
	# In most cases, this will probably be character.
	# But there are other valid options - basically the same ones as customizable_localization.
	# You could also not set a scope type. But then you'd only be able to use global and scope-agnostic effects.
	scope = character
	# For the contents, you can add effect, is_shown, and/or is_valid.
	# effect is an effect block, and the others are trigger blocks.
	# For a more detailed explanation, also check out Agami's wiki:
	# https://ck3.paradoxwikis.com/Interface#Scripted_GUIs
	effect = {
		# Now we can use ol' reliable, the CK3 scripting language.
		set_variable = {
			name  = selected_event_character
			# Note that saved scopes don't exist here unless you added them with the datafunction call.
			# The tutorial GUI file has a more detailed explanation of this.
			value = scope:selection
		}
	}
}
