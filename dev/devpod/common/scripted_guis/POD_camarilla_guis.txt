caitiff_is_member = {
	scope = faith
	is_valid = {
		AND = {
			has_variable_list = member_clan
			OR = {
				is_target_in_variable_list = { 
					name = member_clan
					target = doctrine:doctrine_vampirethinblood_justicar
				}
				is_target_in_variable_list = { 
					name = member_clan
					target = doctrine:doctrine_vampirethinblood_accepted
				}
			}
		}
	}
}

red_list_blood_hunt_war = {
	scope = character
	saved_scopes = { hunted }

	is_shown = {
	}
	is_valid = {
		NOT = { is_at_war_with = scope:hunted }
		scope:hunted = { is_imprisoned = no }
		scope:hunted = { is_landed = yes }
	}
	effect = {
		start_war = {
			cb = POD_blood_hunt_war
			target = scope:hunted
		}
	}
}

red_list_blood_hunt_duel = {
	scope = character
	saved_scopes = { hunted }

	is_shown = {
	}
	is_valid = {
		scope:hunted = { is_landed = no }
	}
	effect = {
		custom_tooltip = fp1_jomsvikings.0041.a.tt
		configure_start_single_combat_effect = {
			SC_INITIATOR = root
			SC_ATTACKER = root
			SC_DEFENDER = scope:hunted
			FATALITY = always
			FIXED = no
			LOCALE = terrain_scope
			OUTPUT_EVENT = single_combat.1006
			INVALIDATION_EVENT = tribal.1022
		}
	}
}

		}
	}
}
