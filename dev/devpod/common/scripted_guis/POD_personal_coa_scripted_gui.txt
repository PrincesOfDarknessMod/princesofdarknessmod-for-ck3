# Personal COA

POD_has_personal_coa = {
	scope = character

	is_shown = {
		OR = {
			POD_can_have_personal_arms_trigger = yes
			any_relation = {
				type = pod_personal_coa_slave
			}
		}
	}	
}

POD_send_away_coa_slave = {
	scope = character
	is_valid = {
		has_character_flag = POD_has_personal_coa_flag
	}
	effect = {
		random_relation = {
			type = pod_personal_coa_slave
			save_scope_as = exiled_dummy
		}
		trigger_event = { id = POD_personal_coat_of_arms.100 days = 1 }
	}	
}

POD_summon_personal_coa_slave = {
	scope = character
	is_valid = {}
	effect = {
		random_relation = {
			type = pod_personal_coa_slave
			save_scope_as = exiled_dummy
		}
		if = {
			limit = {
				NOT = { scope:exiled_dummy = { is_courtier_of = root } }
			}
			add_courtier = scope:exiled_dummy
		}
	}
}
