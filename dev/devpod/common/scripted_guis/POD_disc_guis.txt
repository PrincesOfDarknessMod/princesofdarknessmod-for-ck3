path_of_blood_choose = {
	scope = character
	is_shown = {
		var:chosen_path = flag:path_of_blood
	}
	is_valid = {
		
	}
	effect = {
		set_variable = { name = chosen_path value = flag:path_of_blood }
		if = {
			limit = { NOT = {has_variable = path_of_blood_rating} }
			set_variable = { name = blood_sorcery_cost value = 100 }
		}
		else_if = {
			limit = { var:path_of_blood_rating = 1}
			set_variable = { name = blood_sorcery_cost value = 200 }
		}
		else_if = {
			limit = { var:path_of_blood_rating = 2}
			set_variable = { name = blood_sorcery_cost value = 300 }
		}
	}
}

path_of_music_of_the_shperes_buy = {
	scope = character
	is_valid = {
		custom_description = {
			text = NOT_ENOUGH_POINTS
			NOT = { var:blood_sorcery_cost > var:blood_sorcery_points }
		}
		custom_description = {
			text = PATH_MAXED
			trigger_if = {
				limit = { has_variable = path_of_music_of_the_shperes_rating }
				var:path_of_music_of_the_shperes_rating < 3 
			}
		}
		custom_description = {
			text = NOT_ADEPT_ENOUGH
			OR = {
				has_perk = a_taste_of_blood_perk
				has_perk = corrosive_vitae_perk
			}
			trigger_if = {
				limit = { has_variable = path_of_music_of_the_shperes_rating}
				OR = {
					AND = {
						var:path_of_music_of_the_shperes_rating = 1
						OR = {
							has_perk = blood_of_potency_perk
							has_perk = scorpions_touch_perk
						}
					}
					AND = {
						var:path_of_music_of_the_shperes_rating = 2
						OR = {
							has_perk = baals_curse_perk
							has_perk = cauldron_of_blood_perk
						}
					}
				}
			}
		}
	}
	effect = {
		if = {
			limit = {NOT = { has_variable = path_of_music_of_the_shperes_rating }}
			set_variable = { name = path_of_music_of_the_shperes_rating value = 0}
		}
		change_variable = {
			name = path_of_music_of_the_shperes_rating
			add = 1
		}
		change_variable = {
			name = blood_sorcery_points
			subtract = var:blood_sorcery_cost
		}
		change_variable = {
			name = blood_sorcery_cost
			add = 100
		}
		hidden_effect = {
			send_interface_toast = {
				title = path_buy_toast
				
				left_icon = root
			}
		}
		
		
	}
}

	saved_scopes = { target }
	is_shown = { always = yes}
	
	is_valid = {
		NOT = { has_character_modifier = bloodhunger5_modifier }
		NOT = { scope:target = root	}	
		NOT = { has_relation_best_friend = scope:target}
		custom_description = {
			text = I_need_rating
			has_variable = path_of_music_of_the_shperes_rating
			trigger_if = {
				limit = { has_variable = path_of_music_of_the_shperes_rating }
				var:path_of_music_of_the_shperes_rating > 0
				
			}
		}
		custom_description = {
			text = can_only_be_used_once_on_target
			trigger_if = {
				limit = {
					scope:target = { has_variable_list = songs_of_venus }
				}
				is_target_in_variable_list = {
					name = songs_of_venus
					target = scope:target
				}
			}
		}
		custom_tooltip = {
			text = song_of_jupiter_botch.ct
			NOT = { has_character_flag = song_of_jupiter_botch }
		}
	}
	effect = {
	
		POD_hunger_increase_effect = yes
		if = {
			limit = {has_character_flag = commits_willpower}
			POD_willpower_rouse_check_effect = yes
		}
		random_list = {
			0 = {
				modifier = {
					add = learning
					always = yes
				}
				modifier = {
					factor = 1.5
					has_character_flag = commits_willpower
				}
				custom_tooltip = song_of_venus_works
				hidden_effect = {
					send_interface_toast = {
						title = song_of_venus.toast.1
						left_icon = root
						
						scope:target = {
							add_to_variable_list = {
								name = songs_of_venus
								target = root
							}
						}
						if = {
							limit = {
								has_relation_nemesis = scope:target
							}
							hidden_effect = {
								remove_relation_nemesis = scope:target
								set_relation_rival = scope:target
							}
							custom_tooltip = song_of_venus_rival
							
						}
						else_if = {
							limit = {
								has_relation_rival = scope:target
							}
							hidden_effect = {
								remove_relation_rival = scope:target
							}
							set_relation_potential_rival = scope:target
							custom_tooltip = song_of_venus_potential_rival
						}
						else_if = {
							limit = {
								has_relation_potential_rival = scope:target
							}

							remove_relation_potential_rival = scope:target
							custom_tooltip = song_of_venus_neutral
							
						}
						else_if = {
							limit = {
								has_relation_friend = scope:target
								can_set_relation_friend_trigger = { CHARACTER = scope:target }
							}
							set_relation_best_friend = scope:target
						}
						#If potential friend, set friend!
						else_if = {
							limit = {
								has_relation_potential_friend = scope:target
								can_set_relation_friend_trigger = { CHARACTER = scope:target }
							}
							set_relation_friend = scope:target
						}
						# If not...
						else = {

							# ... set potential relation if we're not already friends!
							if = {
								limit = { can_set_relation_potential_friend_trigger = { CHARACTER = scope:target } }
								set_relation_potential_friend = scope:target
							}
					
								
							
						}
						
					}
					change_variable = {
						name = blood_sorcery_points
						add = 1 
					}
				}
				
				
			}
			10 = {
				custom_tooltip = song_of_venus_fail
				hidden_effect = {
					send_interface_toast = {
						title = song_of_venus_fail.toast.1
						custom_tooltip = song_of_venus_fail.toast.1.ct
						left_icon = root

					}
				}
			
			}
			
			5 = {
				custom_tooltip = path_of_music_of_the_shperes_botch_ct
				show_as_tooltip = {
					add_character_modifier = { 
						modifier = song_of_venus_botch_modifier
						years = 1
					}
				}
				hidden_effect = {					
					send_interface_toast = {
						title = spell_botch
						
						left_icon = root
						add_character_modifier = { 
							modifier = song_of_venus_botch_modifier
							years = 1
						}
					}
				
				
				}
			
			}
		}
	}
}
