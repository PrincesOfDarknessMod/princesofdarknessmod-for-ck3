	scope = character
	is_shown = {
		trigger_if = {
			limit = { 
				exists = this
				NOT = { has_variable = invalid_faiths_shown }
			}
			NOT = { POD_has_illegal_religion_custom_trigger = { CHARACTER = root FAITH = scope:new_faith } }
		}

	}
}

	scope = character
	is_shown = {
		NOT = { has_variable = invalid_faiths_shown }
	}

		if = {
			limit = { has_variable = invalid_faiths_shown }
			remove_variable = invalid_faiths_shown
		}
	}

	scope = character
	is_shown = {

	}

		}
	}

		}

		arcana_lifestyle_perk_points >= 1
	}
	effect = {
		show_as_tooltip = {
			add_arcana_lifestyle_perk_points = {
				value = 0
				add = arcana_lifestyle_perk_points
				max = mass_command_gold_potential_target_value
				multiply = -1
			}
			add_gold = mass_command_gold_value
		}
		ordered_relation = {
			type = worshipping_cultist
			limit = { 
				gold > 0
				NOT = { has_relation_possessesing_wraith = root }
			}
			order_by = gold
			max = mass_command_gold_thrall_list_max_size
			run_interaction = {
				interaction = POD_command_thrall_gold_interaction
				actor = root
				recipient = this
				execute_threshold = accept
			}
		}
	}
}

	scope = artifact
	saved_scopes = { player }

	effect = {
		if = {
			limit = { artifact_owner = scope:player }

			if = {
				limit = {
					has_variable = biblia_negra
					NOT = {
						any_in_list = {
							variable = cursed_characters
							this = scope:player
						}
					}
				}
				# on a delay since otherwise adding all the edge cases for royal court buttons is messy
				scope:player = {
					trigger_event = {
						id = POD_maintenance.666
						days = 1
					}
				}
			}
		}
	}
}
