construct_tribal_holding = {
	scope = province
	
	effect = {
		begin_create_holding = {
			type = tribal_holding
			refund_cost = { gold = 0 }
		}
	}

}

can_construct_tribal_holding = {
	scope = character
	
	is_shown = {
		OR = { 
			has_government = tribal_government
			has_government = litany_government
		}
	}
}

	scope = character
	
	is_shown = {
		OR = {
			POD_splat_is_shifter_trigger = yes
			AND = {
				has_trait = kinfolk
				OR = { 
					religion = { is_in_family = rf_gaia } 
					religion = { is_in_family = rf_wyrm } 
				}
			}
		}
	}
}

	scope = character
	
	is_shown = {

					religion = { is_in_family = rf_wyrm } 
				}
			}
			POD_is_fae_trigger = yes
			AND = {

			}
			POD_is_mage_trigger = yes
			POD_is_wraith_trigger = yes
		}
	}
