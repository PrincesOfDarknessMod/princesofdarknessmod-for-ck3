	scope = faith

		trigger_if = {
			limit = {
				OR = {

					this = faith:ten_mandates_of_hell
				}
			}
			current_year > 1990
		}
		trigger_if = {
			limit = {
				this = faith:sanbenito
			}
			current_year > 1658
		}
		trigger_if = {
			limit = {
				this = { is_patron_faith_trigger = yes } #Hijacked to always hide patron faiths
			}
			POD_bookmark_is_patron_war = yes
		}
		trigger_if = {
			limit = { this = faith:pathofheliopolis }
			has_global_variable = pathofheliopolis_active
		}
		trigger_if = { #Hide Gurahl faiths until splat is fully implemented
			limit = {
				OR = {
					this = faith:riverkeepers
					this = faith:icestalkers
				}
			}
			always = no
		}
	}
}

			run_interaction = {

				recipient = this

			}
		}
	}
