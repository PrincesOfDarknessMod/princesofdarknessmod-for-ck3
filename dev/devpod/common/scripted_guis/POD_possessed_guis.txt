	saved_scopes = { story }

		custom_description_no_bullet = {
			text = POD_possessed_story_header
			object = scope:story.story_owner
		}
		custom_description_no_bullet = {
			text = POD_possessed_story_explanation
			object = scope:story.story_owner
		}
		POD_linebreak_effect = yes

			text = POD_possessed_story_diminishing_returns
			object = scope:story.story_owner

			limit = { scope:story.story_owner = { has_perk = possessed_slasher_perk } }
			POD_linebreak_effect = yes
			custom_description_no_bullet = {
				text = POD_possessed_story_preferred_victim
				object = scope:story.story_owner
			}
		}
	}
}

POD_target_char_is_preferred_victim = {
	scope = character
	saved_scopes = { target }
	is_shown = {
		has_perk = possessed_slasher_perk
		POD_possessed_character_is_preferred_victim = {
			SLASHER = this
			TARGET = scope:target
		}
		scope:target = { is_alive = yes }
	}
	effect = {
		every_in_list = {
			variable = preferred_victim_traits
			limit = { scope:target = { has_trait = prev } }
			save_scope_as = victim_trait
			prev = { custom_tooltip = POD_possessed_preferred_victim_trait_tt }
		}
	}
}

POD_stalk_change_handicaps = {
	scope = character
	effect = {
		save_scope_value_as = {
			name = max_handicaps
			value = {
				value = prowess
				divide = {
					value = POD_stalk_prowess_per_handicap
					multiply = -1
				}
				floor = yes
			}
		}
		set_variable = {
			name = stalk_handicaps_var
			value = {
				value = stalk_handicaps
				add = {
					value = scope:delta
					# it'd be better engineering to handle shiftclicks purely in the gui
					# but the nested datafunctions would get 10 times as long and look horrendous, so
					if = {
						limit = { always = scope:shiftclick }
						multiply = POD_stalk_handicap_shiftclick_mult
					}
				}
				min = 0
				max = scope:max_handicaps
			}
		}
	}
}
