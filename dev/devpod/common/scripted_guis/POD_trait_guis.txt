trait_change_window_open = {
	scope = character
	saved_scopes = { player }
	is_shown = {
		scope:player = {
			has_variable = reweaving_recipient
		}
	}
	effect = {
		remove_variable = reweaving_recipient
	}
}

apply_trait_change = {
	scope = character 
	saved_scopes = { player }
	is_shown = {}
	is_valid = {
		# custom_tooltip = {
			# text = TRAIT_CHANGE_WINDOW_CURRENT_TRAIT
			# POD_trait_change_value > POD_base_value
		# }
		scope:player.piety >= POD_trait_change_value
	}
	effect = {
		scope:player = { 
			add_piety = {
				value = prev.POD_trait_change_value
				multiply = -1
			}
		}
		every_in_list = { 
			variable = trait_to_change_list
			root = {
				if = {
					limit = { has_trait = prev }
					remove_trait = prev
				}
				else = {
					add_trait_force_tooltip = prev 
				}
			}
		}
		clear_variable_list = trait_to_change_list
	}
}

cancel_trait_change = {
	scope = character

	is_valid = {}

	}
}

trait_change_personality_select = {
	scope = character 
	saved_scopes = { trait }
	is_shown = {
		is_target_in_variable_list = {
			name = trait_to_change_list 
			target = scope:trait
		} 
	}
	is_valid = {
		custom_tooltip = {
			text = TRAIT_CHANGE_WINDOW_PERSONALITY_TRAIT_LIMIT
			trigger_if = {
				limit = { 
					NOT = { has_trait = scope:trait } 
					NOT = {
						is_target_in_variable_list = {
							name = trait_to_change_list 
							target = scope:trait
						} 
					}
				}
				POD_current_number_of_personality_traits < personality_trait_limit
			}
		}
		custom_tooltip = {
			text = TRAIT_CHANGE_WINDOW_OPPOSITE_TRAIT
			NOT = {
				any_in_list = {
					variable = trait_to_change_list
					root = { NOT = { has_trait = prev } }
					is_opposite_of_trait = scope:trait
				}
			}
			NOT = {
				any_character_trait = {
					root = {
						NOT = {
							is_target_in_variable_list = {
								name = trait_to_change_list 
								target = prev
							} 
						} 
					}
					is_opposite_of_trait = scope:trait
				}
			}
		}
	}
	effect = {
		if = {
			limit = {
				is_target_in_variable_list = {
					name = trait_to_change_list 
					target = scope:trait
				} 
			}
			remove_list_variable = {
				name = trait_to_change_list 
				target = scope:trait
			}
		}
		else = {
			add_to_variable_list = {
				name = trait_to_change_list 
				target = scope:trait
			}
		}
	}
}
