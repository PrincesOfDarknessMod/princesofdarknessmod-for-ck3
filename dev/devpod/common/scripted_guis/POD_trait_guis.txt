apply_trait_change = {
	scope = character 
	saved_scopes = { player }
	is_shown = {}
	is_valid = {
		trigger_if = {
			limit = { scope:player.var:reweaving_ability ?= flag:gilgul }
			custom_tooltip = {
				text = fp1_yearly.1062.d
				POD_trait_change_value <= 1000
			}
			scope:player.gold >= POD_trait_change_value
		}
		trigger_else = {
			scope:player.piety >= POD_trait_change_value
		}
		scope:player = {
			trigger_if = {
				limit = { exists = var:reweaving_ability }
				custom_tooltip = {
					text = has_reweave_cooldown
					NOT = { has_variable = reweave_cooldown }
				}
			}
			# TODO: Only the blood triggers are properly done
			# The other types of energy will need a refactor
			# (also account for Wraiths with flag:moliate)
			trigger_if = {
				limit = { var:reweaving_ability ?= flag:vicissitude }
				POD_has_enough_blood_for_discipline = { AMOUNT = 5 }
			}
			trigger_else_if = {
				limit = {
					OR = {
						var:reweaving_ability ?= flag:gilgul
						var:reweaving_ability ?= flag:naming_art
					}
				}
				NOT = { has_character_modifier = glamour0_modifier }
			}
			trigger_else_if = {
				limit = { var:reweaving_ability ?= flag:nomenclature }
				trigger_if = {
					limit = { POD_chi_user_trigger = yes }
					NOT = { has_character_modifier = yangchi0_modifier }
				}
				trigger_else = {
					NOT = { has_character_modifier = sekhem0_modifier }
				}
			}
			trigger_else_if = {
				limit = { var:reweaving_ability ?= flag:lore }
				NOT = { has_character_modifier = resolve0_modifier }
			}
			trigger_else = {}
		}
	}
	effect = {
		if = {
			limit = { has_variable_list = trait_to_change_list }
			scope:player = {
				POD_willpower_rouse_check_effect = yes
				POD_universal_energy_decrease_multiple_effect = { COUNT = 10 }
			}
			if = {
				limit = { scope:player.var:reweaving_ability ?= flag:gilgul }
				scope:player = { 
					remove_short_term_gold = prev.POD_trait_change_value
				}
			}
			else = {
				scope:player = { 
					add_piety = {
						value = prev.POD_trait_change_value
						multiply = -1
					}
				}
			}
			if = {
				limit = { POD_splat_has_masquerade_trigger = yes }
				POD_moderate_risk_of_increasing_masquerade_exposure = yes
			}
			every_in_list = { 
				variable = trait_to_change_list
				root = {
					if = {
						limit = { has_trait = prev }
						remove_trait = prev
					}
					else = {
						add_trait_force_tooltip = prev 
					}
				}
			}
			clear_variable_list = trait_to_change_list
			
			scope:player = {
				if = {
					limit = {
						OR = {
							var:reweaving_ability ?= flag:gilgul
							var:reweaving_ability ?= flag:naming_art
						}
						realm_time_yearly_cooldown > 0
					}
					set_variable = {
						name = reweave_cooldown
						value = yes
						years = realm_time_yearly_cooldown
					}
				}
				else_if = {
					limit = { var:reweaving_ability ?= flag:vicissitude }
					set_variable = {
						name = reweave_cooldown
						value = yes
						years = 5
					}
				}
				else_if = {
					limit = { var:reweaving_ability ?= flag:ten_thousand_forms }
					set_variable = {
						name = reweave_cooldown
						value = yes
						years = 10
					}
				}
				else_if = {
					limit = {
						OR = {
							var:reweaving_ability ?= flag:nomenclature
							var:reweaving_ability ?= flag:moliate
							var:reweaving_ability ?= flag:numina_sorcery
							var:reweaving_ability ?= flag:lore
						}
					}
					set_variable = {
						name = reweave_cooldown
						value = yes
						years = 3
					}
				}
			}
		}
		remove_character_flag = physical_trait_only
		remove_character_flag = optional_child
	}
}

trait_change_personality_select = {
	scope = character 
	saved_scopes = { trait player }
	is_shown = {
		is_target_in_variable_list = {
			name = trait_to_change_list 
			target = scope:trait
		} 
	}
	is_valid = {
		custom_tooltip = {
			text = TRAIT_CHANGE_WINDOW_PERSONALITY_TRAIT_LIMIT
			trigger_if = {
				limit = {
					OR = {
						AND = {
							has_trait = scope:trait
							is_target_in_variable_list = {
								name = trait_to_change_list 
								target = scope:trait
							}
						}
						AND = {
							NOT = { has_trait = scope:trait }
							NOT = { 
								is_target_in_variable_list = {
									name = trait_to_change_list 
									target = scope:trait
								}
							}
						} 
					}
				}
				POD_current_number_of_personality_traits < personality_trait_limit
			}
		}
		POD_trait_change_valid_trigger = yes
	}
	effect = {
		if = {
			limit = {
				is_target_in_variable_list = {
					name = trait_to_change_list 
					target = scope:trait
				} 
			}
			remove_list_variable = {
				name = trait_to_change_list 
				target = scope:trait
			}
		}
		else = {
			add_to_variable_list = {
				name = trait_to_change_list 
				target = scope:trait
			}
		}
	}
}

				is_target_in_variable_list = {
					name   = selected_traits
					target = scope:trait
				}
			}
			remove_list_variable = {
				name   = selected_traits
				target = scope:trait
			}
		}
		else = {
			add_to_variable_list = {
				name   = selected_traits
				target = scope:trait
			}
		}
	}
}
