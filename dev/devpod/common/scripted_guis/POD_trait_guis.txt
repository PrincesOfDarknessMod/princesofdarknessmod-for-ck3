apply_trait_change = {
	scope = character 
	saved_scopes = { player }
	is_shown = {}
	is_valid = {
		scope:player.piety >= POD_trait_change_value
		scope:player = {
			trigger_if = {
				limit = { exists = var:reweaving_ability }
				custom_tooltip = {
					text = has_reweave_cooldown
					NOT = { has_variable = reweave_cooldown }
				}
			}
			trigger_if = {
				limit = {
					OR = {
						var:reweaving_ability ?= flag:naming_art
					}
				}
				POD_has_enough_glamour_for_character_cantrip_interaction_custom = {
					AMOUNT = 5
					RECIPIENT = prev
				}
			}
			trigger_else = {
				POD_has_enough_splat_agnostic_energy_reserves = {
					AMOUNT = 10
				}
			}
		}
	}
	effect = {
		if = {
			limit = { has_variable_list = trait_to_change_list }
			scope:player = {
				POD_willpower_rouse_check_effect = yes
				if = {
					limit = {
						OR = {
							var:reweaving_ability ?= flag:naming_art
						}
					}
					POD_character_cantrip_interaction_cost_custom = {
						COUNT = 5
						RECIPIENT = prev
					}
				}
				else = {
					POD_splat_agnostic_energy_cost = { COUNT = 10 }
				}
				
				add_piety = {
					value = prev.POD_trait_change_value
					multiply = -1
				}
			}
			if = {
				limit = { POD_splat_has_masquerade_trigger = yes }
				POD_moderate_risk_of_increasing_masquerade_exposure = yes
			}
			every_in_list = { 
				variable = trait_to_change_list
				root = {
					if = {
						limit = { has_trait = prev }
						remove_trait = prev
					}
					else = {
						add_trait_force_tooltip = prev 
					}
				}
			}
			clear_variable_list = trait_to_change_list
			
			scope:player = {
				if = {
					limit = {
						OR = {
							var:reweaving_ability ?= flag:naming_art
						}
						realm_time_yearly_cooldown > 0
					}
					set_variable = {
						name = reweave_cooldown
						value = yes
						years = realm_time_yearly_cooldown
					}
				}
				else_if = {
					limit = { var:reweaving_ability ?= flag:vicissitude }
					set_variable = {
						name = reweave_cooldown
						value = yes
						years = 5
					}
				}
				else_if = {
					limit = { var:reweaving_ability ?= flag:flowers_of_flesh }
					if = { limit = { has_perk = shintai_the_shintai_of_flesh_perk } }
					else_if = {
						limit = { has_perk = shintai_a_thousand_forms_of_flesh_perk }
						set_variable = {
							name = reweave_cooldown
							value = yes
							years = 5
						}
					}
					else = {
						set_variable = {
							name = reweave_cooldown
							value = yes
							years = 10
						}
					}
				}
				else_if = {
					limit = {
						OR = {
							var:reweaving_ability ?= flag:moliate
							var:reweaving_ability ?= flag:numina_sorcery
							var:reweaving_ability ?= flag:arcana
							var:reweaving_ability ?= flag:lore
							var:reweaving_ability ?= flag:whispers_to_the_body
						}
					}
					set_variable = {
						name = reweave_cooldown
						value = yes
						years = 3
					}
				}
			}
		}
		remove_character_flag = physical_trait_only
		remove_character_flag = optional_child
		remove_character_flag = exclude_personality_change
		remove_character_flag = exclude_education_change
		remove_character_flag = exclude_commander_change
		remove_character_flag = exclude_lifestyle_change
		remove_character_flag = exclude_fame_change
		remove_character_flag = exclude_stress_change
		remove_character_flag = exclude_health_change
	}
}

trait_change_personality_select = {
	scope = character 
	saved_scopes = { trait player }
	is_shown = {
		is_target_in_variable_list = {
			name = trait_to_change_list 
			target = scope:trait
		} 
	}
	is_valid = {
		custom_tooltip = {
			text = TRAIT_CHANGE_WINDOW_PERSONALITY_TRAIT_LIMIT
			trigger_if = {
				limit = {
					OR = {
						AND = {
							has_trait = scope:trait
							is_target_in_variable_list = {
								name = trait_to_change_list 
								target = scope:trait
							}
						}
						AND = {
							NOT = { has_trait = scope:trait }
							NOT = { 
								is_target_in_variable_list = {
									name = trait_to_change_list 
									target = scope:trait
								}
							}
						} 
					}
				}
				POD_current_number_of_personality_traits < personality_trait_limit
			}
		}
		POD_trait_change_valid_trigger = yes
	}
	effect = {
		if = {
			limit = {
				is_target_in_variable_list = {
					name = trait_to_change_list 
					target = scope:trait
				} 
			}
			remove_list_variable = {
				name = trait_to_change_list 
				target = scope:trait
			}
		}
		else = {
			add_to_variable_list = {
				name = trait_to_change_list 
				target = scope:trait
			}
		}
	}
}
