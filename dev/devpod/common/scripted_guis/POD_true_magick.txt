true_magick = {
	scope = character

	is_shown = {
		has_variable = true_magick_open
	}
	is_valid = {
		always = yes
	}
	effect = {
		remove_variable = true_magick_open
	}

}

	}
	is_valid = {
		always = yes
	}
	effect = {
		remove_variable = active_spells_on_you_open
	}

}

active_spells_on_character_open = {
	scope = character
	
	saved_scopes = { target }
	
	is_shown = {
	}
	is_valid = {
		always = yes
	}
	effect = {
		if = {
			limit = {NOT={has_variable = active_spells_on_you_open}}
			set_variable = active_spells_on_you_open
			set_variable = { name = spells_on_character_perspective value = scope:target }
		}
	}

}

active_spells_on_character_gui = {
	scope = character
	
	saved_scopes = { target }
	
	is_shown = {
		has_trait = mage
		scope:target = { has_variable_list = spells_on_character }
	}
}

is_modifier = {
	scope = character
	saved_scopes = { flag }
	
	is_shown = {
		
			any_in_global_list = {
				variable = true_magick_modifiers
				this = scope:flag
			}
		
	}
}

choose_target = {
	scope = title
	saved_scopes = { new_target }
	
	effect = {
		set_variable = { name = target value = scope:new_target }
	}
	
}

choose_spell_name = {
	scope = character 
	
	effect = {
		trigger_event = true_magick.2000
	}	
}

remove_target = {
	scope = character
	saved_scopes = { target }
	
	effect = {
		remove_list_variable = {
			name = spell_targets
			target = scope:target
		}
	}
	
	is_shown = {
		NOT = { scope:target = var:target } 
	}
}

true_magick_give_mods = {
	scope = character
	saved_scopes = { primary_target }
	
	is_valid = {
		custom_description = {
			text = true_magic_can_only_cast_one_spell_tooltip
			any_in_list = {
				variable = casting_spells
				count < 1
			}
		}
	}

	effect = {
		save_spell_effect = yes
		trigger_event = true_magick.4100
		
		remove_variable = true_magick_open
	}
}

	}
}

true_magick_duration_months = {
	scope = character
	
	is_shown = {
		has_variable = duration_type
		trigger_if = {
			limit = { has_variable = duration_type }
			var:duration_type = flag:months
		}
		
	}
	
	effect = {
		set_variable = { name = duration_type value = flag:months }
	}
}

true_magick_duration_increase = {
	scope = character
	
	effect = {
		change_variable = { name = duration add = 1 }
	}
}


true_magick_duration_decrease = {
	scope = character
	
	effect = {
		change_variable = { name = duration subtract = 1 }
		if = {
			limit = {
				var:duration < 1
			}
			set_variable = { name = duration value = 1 }
		}
	}
}

spell_detail_open = {
	scope = character
	
	saved_scopes = { target }
	
	is_shown = {
		has_variable = spell_detail_open
	}
	is_valid = {
		always = yes
	}
	effect = {
		if = {
			limit = {NOT={has_variable = spell_detail_open}}
			set_variable = spell_detail_open
		}
		set_variable = { name = examined_spell value = scope:target }
	}

}

spell_detail_close = {
	scope = character
	
	saved_scopes = { target }
	
	is_shown = {
	}
	is_valid = {
		always = yes
	}
	effect = {
		remove_variable = examined_spell
		remove_variable = spell_detail_open
	}

}

spell_detail_effect_display = {
	scope = character

	effect = {
		var:examined_spell = {
			save_scope_as = spell
		}
		enact_spell_effect = yes
	}
	
}

cancel_spell_gui = {
	scope = character
	
	effect = {
		remove_list_variable = {
			name = casting_spells
			target = var:examined_spell
		}
	}
}
