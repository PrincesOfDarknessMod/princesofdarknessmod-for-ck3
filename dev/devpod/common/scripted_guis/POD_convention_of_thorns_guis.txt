vote_tenet = {
	saved_scopes = { elector }
	is_shown = {
		scope:elector = {
			is_target_in_variable_list = { 
				name = voted_tenet 
				target = root 
			} 
		}
	}
	is_valid = {
	}
	effect = {
		scope:elector = {
			if = {
				limit = { 
					is_target_in_variable_list = { 
						name = voted_tenet 
						target = root 
					} 
				}
				unvote_tenet_effect = { TENET = root }
			}
			else = {
				vote_tenet_effect = { TENET = root }
			}
		}
		faith:camahumanity = { calcule_voted_tenet_effect = yes }
	}
}

apply_voted_tenet = {
	saved_scopes = { player }
	is_shown = {}
	is_valid = {}
	effect = {
		clear_variable_list = tenet_voted_15_list
		clear_variable_list = tenet_voted_14_list
		clear_variable_list = tenet_voted_13_list
		clear_variable_list = tenet_voted_12_list
		clear_variable_list = tenet_voted_11_list
		clear_variable_list = tenet_voted_10_list
		clear_variable_list = tenet_voted_9_list
		clear_variable_list = tenet_voted_8_list
		clear_variable_list = tenet_voted_7_list
		clear_variable_list = tenet_voted_6_list
		clear_variable_list = tenet_voted_5_list
		clear_variable_list = tenet_voted_4_list
		clear_variable_list = tenet_voted_3_list
		clear_variable_list = tenet_voted_2_list
		clear_variable_list = tenet_voted_1_list
		clear_variable_list = tenet_voted_0_list
		
		every_in_list = {
			variable = thorn_elector_list
			if = {
				limit = { has_variable_list = voted_tenet }
				if = {
					limit = {
						NOT = { is_target_in_variable_list = { name = voted_tenet target = root.var:tenet_1 } }
						NOT = { is_target_in_variable_list = { name = voted_tenet target = root.var:tenet_2 } }
						NOT = { is_target_in_variable_list = { name = voted_tenet target = root.var:tenet_3 } }
					}
					custom_description = {
						text = "unhappy_thorn_elector_desc"
						object = this
						add_character_flag = { 
							flag = unhappy_thorn_elector
							years = 1
						}
					}
				}
				else = {
					custom_description = {
						text = "happy_thorn_elector_desc"
						object = this
						add_character_flag = { 
							flag = happy_thorn_elector
							years = 1
						}
					}
				}
				clear_variable_list = voted_tenet
			}
		}
		
		add_doctrine = var:tenet_1
		add_doctrine = var:tenet_2
		add_doctrine = var:tenet_3
		remove_variable = tenet_1
		remove_variable = tenet_2
		remove_variable = tenet_3
		remove_variable = tenet_1_pos
		remove_variable = tenet_2_pos
		remove_variable = tenet_3_pos
		
		scope:player = { 
			remove_variable = custom_convention_of_thorns 
			trigger_event = PODthorns.1003
		}
	}
}

vote_doctrine = {
	saved_scopes = { elector }
	is_shown = {
		faith:camahumanity = {
			is_target_in_variable_list = { 
				name = final_doctrine_list 
				target = root 
			}
		}
	}
	is_valid = {
		scope:elector = {
			NOT = {
				is_target_in_variable_list = { 
					name = voted_doctrine
					target = root 
				} 
			}
		}
	}
	effect = {
		scope:elector = { vote_doctrine_effect = { DOCTRINE = root } }
		faith:camahumanity = { calcule_voted_doctrine_effect = yes }
	}
}

apply_voted_doctrine = {
	saved_scopes = { player }
	is_shown = {}
	is_valid = {}
	effect = {
		every_in_list = {
			variable = thorn_elector_list
			if = {
				limit = { has_variable_list = voted_doctrine }
				if = {
					limit = { passed_voted_doctine_value < 10 }
					custom_description = {
						text = "unhappy_thorn_elector_desc"
						object = this
						add_character_flag = { 
							flag = opposing_thorn_elector
							years = 1
						}
					}
				}
				else = {
					custom_description = {
						text = "happy_thorn_elector_desc"
						object = this
						add_character_flag = { 
							flag = approves_thorn_elector
							years = 1
						}
					}
				}
				clear_variable_list = voted_doctrine
			}
		}
		clear_variable_list = thorn_elector_list
		
		every_in_list = {
			variable = final_doctrine_list
			root = { add_doctrine = prev }
		}
		clear_variable_list = final_doctrine_list
		
		scope:player = { 
			remove_variable = custom_convention_of_thorns_doctrine 
		}
	}
}
