POD_gambling_finish_script = {
	is_valid = {
		custom_tooltip = {
			text = POD_event_window_widget_gambling_requirement_selected
			exists = var:pod_gambling_money_gambled
			var:pod_gambling_money_gambled > 0
		}
		gold >= pod_gambling_money_gambled_val
	}
	effect = {
		remove_short_term_gold = pod_gambling_money_gambled_val
		# Produces 1:2 payout so you win 1G for every 2G staked
		random_list = {
			25 = {
				desc = pod1230objectives_judges.26.tt.critical_success
				show_chance = no
				send_interface_toast = {
					type = event_toast_effect_good
					title = pod1230objectives_judges.26.tt.critical_success
					left_icon = scope:actor

					add_gold = {
						value = pod_gambling_money_gambled_val
						multiply = 3
					}
				}
			}
			25 = {
				desc = pod1230objectives_judges.26.tt.success
				show_chance = no
				send_interface_toast = {
					type = event_toast_effect_good
					title = pod1230objectives_judges.26.tt.success
					left_icon = scope:actor

					add_gold = {
						value = pod_gambling_money_gambled_val
						multiply = 2
					}
				}
			}
			25 = {
				desc = pod1230objectives_judges.26.tt.middling
				show_chance = no
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = pod1230objectives_judges.26.tt.middling
					left_icon = scope:actor
					add_gold = {
						value = pod_gambling_money_gambled_val
					}
				}
			}
			25 = {
				desc = pod1230objectives_judges.26.tt.failure
				show_chance = no
				send_interface_toast = {
					type = event_toast_effect_bad
					title = pod1230objectives_judges.26.tt.failure
					left_icon = scope:actor
				}
			}
		}
		remove_variable = pod_gambling_money_gambled
	}
}

pod_gambling_money_gambled_edit = {
	scope = character
	effect = { 
		if = {
			limit = { NOT = { exists = var:pod_gambling_money_gambled } }
			set_variable = { name = pod_gambling_money_gambled value = pod_gambling_money_gambled_val }
		}
		change_variable = { name = pod_gambling_money_gambled add = scope:delta }
		if = {
			limit = { var:pod_gambling_money_gambled < 1 }
			set_variable = { name = pod_gambling_money_gambled value = 10 }
		}
		if = {
			limit = { var:pod_gambling_money_gambled > 5000 }
			set_variable = { name = pod_gambling_money_gambled value = 5000 }
		}
	}
}
