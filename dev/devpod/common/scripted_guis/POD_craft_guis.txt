craft_initiate = {
	scope = character
	effect = {
		if = {
			limit = {
				OR = {
					NOT = { has_variable = craft_creator } 
					var:craft_creator = { is_alive = no } 
				}
			}
			set_variable = {
				name = craft_creator
				value = root
			}
		}
		if = {
			limit = { NOT = { has_global_variable_list = common_material_list } }
			add_to_global_variable_list = { name = common_material_list target = flag:water }
			add_to_global_variable_list = { name = common_material_list target = flag:fuel }
			add_to_global_variable_list = { name = common_material_list target = flag:herb }
			add_to_global_variable_list = { name = common_material_list target = flag:venom }
			add_to_global_variable_list = { name = common_material_list target = flag:mineral }
		}
		if = {
			limit = { NOT = { has_global_variable_list = uncommon_material_list } }
			add_to_global_variable_list = { name = uncommon_material_list target = flag:exotic_water }
			add_to_global_variable_list = { name = uncommon_material_list target = flag:exotic_fuel }
			add_to_global_variable_list = { name = uncommon_material_list target = flag:exotic_herb }
			add_to_global_variable_list = { name = uncommon_material_list target = flag:exotic_mineral }
		}
		if = {
			limit = { NOT = { has_global_variable_list = rare_material_list } }
			add_to_global_variable_list = { name = rare_material_list target = flag:magic_water }
			add_to_global_variable_list = { name = rare_material_list target = flag:magic_fuel }
			add_to_global_variable_list = { name = rare_material_list target = flag:magic_herb }
			add_to_global_variable_list = { name = rare_material_list target = flag:magic_mineral }
		}
		if = {
			limit = { NOT = { has_global_variable_list = special_material_list } }
			add_to_global_variable_list = { name = special_material_list target = flag:common_artifact_piece }
			add_to_global_variable_list = { name = special_material_list target = flag:masterwork_artifact_piece }
			add_to_global_variable_list = { name = special_material_list target = flag:famed_artifact_piece }
			add_to_global_variable_list = { name = special_material_list target = flag:illustrious_artifact_piece }
		}
		
		hidden_effect = {
			if = {
				limit = { NOT = { has_global_variable = common_artifact_to_create } }
				character:POD_wraith_01 = { create_artifact_to_create_effect = { RARITY = common } }
			}
			if = {
				limit = { NOT = { has_global_variable = masterwork_artifact_to_create } }
				character:POD_wraith_01 = { create_artifact_to_create_effect = { RARITY = masterwork } }
			}
			if = {
				limit = { NOT = { has_global_variable = famed_artifact_to_create } }
				character:POD_wraith_01 = { create_artifact_to_create_effect = { RARITY = famed } }
			}
			if = {
				limit = { NOT = { has_global_variable = illustrious_artifact_to_create } }
				character:POD_wraith_01 = { create_artifact_to_create_effect = { RARITY = illustrious } }
			}
		}
	}
}

craft_clear = {
	scope = character
	is_shown = {}

character_material_initiate = {
	scope = character
	is_shown = {
		OR = {
			has_character_material_enhance_trigger = yes
			has_character_material_forge_trigger = yes
			has_vampire_blood_ingredient_trigger = yes
			
			is_target_in_variable_list = { name = ingredient_list target = flag:shapeshifter_blood }
			is_target_in_variable_list = { name = ingredient_list target = flag:fae_blood }
		}
	}
	effect = {
		clear_variable_list = character_material
		add_to_variable_list = {
			name = character_material
			target = this
		}
		every_prisoner = {
			root = {
				add_to_variable_list = {
					name = character_material
					target = prev
				}
			}
		}
		every_relation = {
			type = bound_wraith
			root = {
				add_to_variable_list = {
					name = character_material
					target = prev
				}
			}
		}
		every_relation = {
			type = spirit_companion
			root = {
				add_to_variable_list = {
					name = character_material
					target = prev
				}
			}
		}
	}
}

assign_character_material_choose = {
	scope = character
	saved_scopes = { prisoner }
	is_shown = {
	}
	is_valid = {
		scope:prisoner = { is_alive = yes }
	}
	effect = {
		if = {
			limit = { 
				exists = scope:prisoner 
				NOT = { root = scope:prisoner } 
			}
			set_variable = { name = character_material value = scope:prisoner }
		}
		else = {
			set_variable = { name = character_material value = root }
		}
		if = {
			limit = {
				has_variable = character_material
				has_variable = current_method 
			}
			POD_get_method_quality_chance_effect = { METHOD = var:current_method }
		}
	}
}

		}
		trigger_if = {
			limit = { has_variable = target_artifact }
			POD_enhance_create_trigger = yes
		}

			custom_tooltip = {
				text = no_artifact_chosen

			}
		}

		}
	}

		if = {
			limit = { 
				has_variable = target_artifact 
				has_variable = current_recipe 
			}
			POD_enhance_create_effect = yes
		}
	}
}

		}
		
		if = {
			limit = {

					OR = {
						NOT = { has_variable = character_material }
						var:character_material = { is_alive = no }
					}
				}
			}
			POD_enhance_generate_artifact_spirit_effect = yes
		}
	}
