craft_initiate = {
	scope = character
	effect = {
		if = {
			limit = { NOT = { has_variable_list = common_material_list } }
			add_to_variable_list = { name = common_material_list target = flag:water }
			add_to_variable_list = { name = common_material_list target = flag:fuel }
			add_to_variable_list = { name = common_material_list target = flag:herb }
			add_to_variable_list = { name = common_material_list target = flag:venom }
		}
		if = {
			limit = { NOT = { has_variable_list = uncommon_material_list } }
			add_to_variable_list = { name = uncommon_material_list target = flag:exotic_water }
			set_variable = {
				name = exotic_water_reserve
				value = 0
			}
			add_to_variable_list = { name = uncommon_material_list target = flag:exotic_fuel }
			set_variable = {
				name = exotic_fuel_reserve
				value = 0
			}
			add_to_variable_list = { name = uncommon_material_list target = flag:exotic_herb }
			set_variable = {
				name = exotic_herb_reserve
				value = 0
			}
		}
		if = {
			limit = { NOT = { has_variable_list = rare_material_list } }
			add_to_variable_list = { name = rare_material_list target = flag:magic_water }
			set_variable = {
				name = magic_water_reserve
				value = 0
			}
			add_to_variable_list = { name = rare_material_list target = flag:magic_fuel }
			set_variable = {
				name = magic_fuel_reserve
				value = 0
			}
			add_to_variable_list = { name = rare_material_list target = flag:magic_herb }
			set_variable = {
				name = magic_herb_reserve
				value = 0
			}
		}
	}
}

character_material_initiate = {
	scope = character
	is_shown = {
		OR = {
			has_character_material_forge_trigger = yes
			has_vampire_blood_ingredient_trigger = yes
		}
	}
	effect = {
		clear_variable_list = character_material
		add_to_variable_list = {
			name = character_material
			target = this
		}
		every_prisoner = {
			root = {
				add_to_variable_list = {
					name = character_material
					target = prev
				}
			}
		}
		every_relation = {
			type = bound_wraith
			root = {
				add_to_variable_list = {
					name = character_material
					target = prev
				}
			}
		}
		every_relation = {
			type = spirit_companion
			root = {
				add_to_variable_list = {
					name = character_material
					target = prev
				}
			}
		}
	}
}

assign_character_material_choose = {
	scope = character
	saved_scopes = { prisoner }
	is_shown = {
	}
	is_valid = {
	}
	effect = {
		if = {
			limit = { 
				exists = scope:prisoner 
				NOT = { root = scope:prisoner } 
			}
			set_variable = { name = character_material value = scope:prisoner }
		}
		else = {
			set_variable = { name = character_material value = root }
		}
		if = {
			limit = {
				has_variable = character_material
				has_variable = current_method 
			}
			POD_get_method_quality_chance_effect = { METHOD = var:current_method }
		}
	}
}
