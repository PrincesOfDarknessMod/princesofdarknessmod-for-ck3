######################################################
######   POD COMBAT CONTROL SGUIS
######################################################
#
# Slot-card target selection. scope:target_slot = the clicked slot's
# POD_combat_slot story, passed as a saved scope.
#
# One parametric SGUI rather than per-slot ones because datamodel
# iteration in the panel can't build an SGUI name dynamically; a
# saved scope works for any slot.

pod_combat_select_target = {
	scope = character
	saved_scopes = { target_slot }

	is_shown = {
		exists = scope:target_slot
		# No click-to-target surface on the PC's own card.
		scope:target_slot = {
			has_variable = combat_team
			NOT = { pod_combat_slot_is_pc_trigger = yes }
		}
	}

	is_valid = {
		pod_combat_is_active_trigger = yes
		pod_combat_is_player_turn_trigger = yes
		exists = scope:target_slot
		scope:target_slot = {
			has_variable = slot_char
			pod_combat_slot_is_alive_trigger = yes
			has_variable = combat_team
			NOT = { pod_combat_slot_is_pc_trigger = yes }
		}
	}

	effect = {
		set_variable = {
			name  = pod_combat_pc_target_team
			value = scope:target_slot.var:combat_team
		}
		set_variable = {
			name  = pod_combat_pc_target_position
			value = scope:target_slot.var:team_position
		}
		# Feeds the top-row "Target: X" readout.
		set_variable = {
			name  = pod_combat_pc_target_char
			value = scope:target_slot.var:slot_char
		}
		set_variable = {
			name  = pod_combat_selected_target_team
			value = scope:target_slot.var:combat_team
		}
		set_variable = {
			name  = pod_combat_selected_target_position
			value = scope:target_slot.var:team_position
		}
		pod_combat_load_workspace_effect = yes
		# Round-duration hit-buff tiers carry over a target swap - they
		# expire at round end, so recalc reapplies them, not resets them.
		pod_combat_recalc_pc_hit_chance_effect = yes
	}
}
