######################################################
######   POD COMBAT CONTROL SGUIS
######################################################
#
# Slot-card target selection. Invoked from each slot's "select
# target" button in the combat window's slot panel via:
#   GuiScope.SetRoot(GetPlayer.MakeScope)
#       .AddScope('target_slot', Scope.Story.MakeScope)
#
# The slot story is passed as scope:target_slot. The SGUI reads
# slot_index off it, then writes pod_combat_pc_target +
# pod_combat_selected_target on the host (PC), reloads the
# workspace target slot into the OPPONENT side, and recalcs the
# PC's hit chance against the new opponent's prowess.
#
# Replaced an earlier set of nine per-slot SGUIs
# (pod_combat_select_target_1 through _9) when the datamodel
# iteration replaced the per-slot panel blocks. The per-slot
# variants were called by name from the panel blocks; the
# datamodel iteration can't construct an SGUI name dynamically,
# so a single SGUI taking the slot story as a saved scope
# replaces all nine.

pod_combat_select_target = {
	scope = character
	saved_scopes = { target_slot }

	is_shown = {
		exists = scope:target_slot
		# Don't render the button on the PC's own slot card.
		scope:target_slot = {
			has_variable = slot_index
			NOT = { var:slot_index = 0 }
		}
	}

	is_valid = {
		# Combat is live, it's the player's turn, and the slot is alive.
		pod_combat_is_active_trigger = yes
		pod_combat_is_player_turn_trigger = yes
		exists = scope:target_slot
		scope:target_slot = {
			has_variable = slot_char
			var:slot_alive = 1
			has_variable = slot_index
			NOT = { var:slot_index = 0 }
		}
	}

	effect = {
		# Save scope:target_slot's slot_index as the PC's persistent
		# target AND the workspace's selected_target, then reload
		# the workspace so the OPPONENT side mirrors the new slot.
		set_variable = {
			name  = pod_combat_pc_target
			value = scope:target_slot.var:slot_index
		}
		set_variable = {
			name  = pod_combat_selected_target
			value = scope:target_slot.var:slot_index
		}
		# Reload the workspace's opponent_* fields from the new slot.
		pod_combat_load_workspace_effect = yes
		# Hit chance depends on the target's prowess — recompute the
		# PC's base hit and reapply tier mods so the move-bar tooltips
		# reflect the new matchup. Round-duration hit-buff tiers
		# carry over (expire at round end, not on target swap).
		pod_combat_recalc_actor_hit_chance_effect = yes
	}
}
