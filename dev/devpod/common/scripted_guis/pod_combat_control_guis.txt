######################################################
######   POD COMBAT CONTROL SGUIS
######################################################
#
# Slot-card target selection. Invoked from each slot's "select
# target" button in the combat window's slot panel via:
#   GuiScope.SetRoot(GetPlayer.MakeScope)
#       .AddScope('target_slot', Scope.Story.MakeScope)
#
# The slot story is passed as scope:target_slot. The SGUI reads
# combat_team + team_position off it, then writes the two-field
# pod_combat_pc_target_* and pod_combat_selected_target_* pointers on
# the host (PC), reloads the workspace target slot into the target
# side, and recalcs the PC's hit chance against the new target's prowess.
#
# A single parametric SGUI rather than nine per-slot ones because
# datamodel iteration in the panel blocks can't construct an SGUI
# name dynamically. Passing the slot story as a saved scope works
# for any slot.

pod_combat_select_target = {
	scope = character
	saved_scopes = { target_slot }

	is_shown = {
		exists = scope:target_slot
		# Don't render the button on the PC's own slot card.
		scope:target_slot = {
			has_variable = combat_team
			NOT = { AND = { var:combat_team = flag:player var:team_position = 0 } }
		}
	}

	is_valid = {
		# Combat is live, it's the player's turn, and the slot is alive.
		pod_combat_is_active_trigger = yes
		pod_combat_is_player_turn_trigger = yes
		exists = scope:target_slot
		scope:target_slot = {
			has_variable = slot_char
			var:slot_alive = 1
			has_variable = combat_team
			NOT = { AND = { var:combat_team = flag:player var:team_position = 0 } }
		}
	}

	effect = {
		# Save scope:target_slot's (combat_team, team_position) as the PC's
		# persistent target AND the workspace's selected_target, then reload
		# the workspace so the target side mirrors the new slot.
		set_variable = {
			name  = pod_combat_pc_target_team
			value = scope:target_slot.var:combat_team
		}
		set_variable = {
			name  = pod_combat_pc_target_position
			value = scope:target_slot.var:team_position
		}
		# Mirror the target's character for the top-row "Target: X" display.
		set_variable = {
			name  = pod_combat_pc_target_char
			value = scope:target_slot.var:slot_char
		}
		set_variable = {
			name  = pod_combat_selected_target_team
			value = scope:target_slot.var:combat_team
		}
		set_variable = {
			name  = pod_combat_selected_target_position
			value = scope:target_slot.var:team_position
		}
		# Reload the workspace's target_* fields from the new slot.
		pod_combat_load_workspace_effect = yes
		# Hit chance depends on the target's prowess — recompute the
		# PC's base hit and reapply tier mods so the move-bar tooltips
		# reflect the new matchup. Round-duration hit-buff tiers
		# carry over (expire at round end, not on target swap).
		pod_combat_recalc_pc_hit_chance_effect = yes
	}
}
