######################################################
######   POD COMBAT CONTROL SGUIS
######################################################
#
# Slot-card target selection. scope:target_slot = the clicked slot's
# POD_combat_slot story, passed as a saved scope.
#
# One parametric SGUI rather than per-slot ones because datamodel
# iteration in the panel can't build an SGUI name dynamically; a
# saved scope works for any slot.

pod_combat_select_target = {
	scope = character
	saved_scopes = { target_slot }

	is_shown = {
		exists = scope:target_slot
		# No click-to-target surface on the PC's own card.
		scope:target_slot = {
			has_variable = combat_team
			NOT = { pod_combat_slot_is_pc_trigger = yes }
		}
	}

	is_valid = {
		pod_combat_is_active_trigger = yes
		pod_combat_is_player_turn_trigger = yes
		exists = scope:target_slot
		scope:target_slot = {
			has_variable = slot_char
			pod_combat_slot_is_alive_trigger = yes
			has_variable = combat_team
			NOT = { pod_combat_slot_is_pc_trigger = yes }
		}
	}

	effect = {
		set_variable = {
			name  = pod_combat_pc_target_team
			value = scope:target_slot.var:combat_team
		}
		set_variable = {
			name  = pod_combat_pc_target_position
			value = scope:target_slot.var:team_position
		}
		# Feeds the top-row "Target: X" readout.
		set_variable = {
			name  = pod_combat_pc_target_char
			value = scope:target_slot.var:slot_char
		}
		set_variable = {
			name  = pod_combat_selected_target_team
			value = scope:target_slot.var:combat_team
		}
		set_variable = {
			name  = pod_combat_selected_target_position
			value = scope:target_slot.var:team_position
		}
		pod_combat_load_workspace_effect = yes
		# Round-duration hit-buff tiers carry over a target swap - they
		# expire at round end, so recalc reapplies them, not resets them.
		pod_combat_recalc_pc_hit_chance_effect = yes
	}
}

pod_char_has_no_human_body = {

	}
}

pod_combat_display_team_name = {
	scope = character
	saved_scopes = { is_enemy }
	effect = {
		if = {
			limit = {
				scope:is_enemy = 0
				pod_combat_has_custom_player_team_name = yes
			}
			custom_description_no_bullet = {
				text = POD_combat_custom_player_team_name
			}
		}
		else_if = {
			limit = {
				scope:is_enemy = 1
				pod_combat_has_custom_enemy_team_name = yes
			}
			custom_description_no_bullet = {
				text = POD_combat_custom_enemy_team_name
			}
		}
		else = {
			ordered_in_list = {
				variable = pod_combat_active_slots
				limit = {
					pod_combat_slot_is_leader_trigger = yes
					pod_combat_slot_is_enemy_value = scope:is_enemy
				}
				var:slot_char ?= {
					save_scope_as = target
				}
			}
			if = {
				limit = { exists = scope:target }
				if = {
					limit = {
						scope:target = {
							POD_has_coterie_trigger = yes
						}
						any_in_list = {
							variable = pod_combat_active_slots
							pod_combat_slot_is_enemy_value = scope:is_enemy
							var:slot_char ?= {
								this != scope:target
								add_to_temporary_list = teammates
							}
						}
						any_in_list = {
							list = teammates
							count = all
							has_relation_coterie_member = scope:target
						}
					}
					custom_description_no_bullet = {
						text = target_character_coteriename
						object = scope:target
					}
				}
				else = {
					custom_description_no_bullet = {
						text = target_character_getfirstnamenotooltip
						object = scope:target
					}
				}
			}
		}
	}
}

pod_combat_player_big_portrait_is_shown = {
	scope = character
	saved_scopes = { slot_story }
	is_shown = {
		scope:slot_story = {
			exists = var:slot_char
			pod_combat_slot_is_leader_trigger = yes
			pod_combat_slot_is_alive_trigger = yes
		}
		exists = var:pod_combat_player_team_size
		var:pod_combat_player_team_size <= 2
	}
}
