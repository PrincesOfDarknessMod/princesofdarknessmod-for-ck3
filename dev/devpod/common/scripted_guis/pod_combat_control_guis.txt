######   POD COMBAT CONTROL GUIS
##########################################
# Static button handlers used by the combat window's chrome:
#   - End Turn / End Combat
#   - Auto-Resolve
#   - Help / Overview (triggers pod_combat.0001)
#   - Forfeit / Close (header X)
#   - Target Selection (9 slots)
#
# Scope: ROOT = player character (via GuiScope.SetRoot(GetPlayer.MakeScope).End)

###### END TURN BUTTON ######

pod_combat_end_turn = {
	is_valid = {
		pod_combat_is_active_trigger = yes
		# Valid during player's turn OR when combat is over (to dismiss)
		OR = {
			pod_combat_is_player_turn_trigger = yes
			pod_combat_is_over_trigger = yes
		}
	}
	effect = {
		# If combat is over, run cleanup and close the window
		if = {
			limit = { pod_combat_is_over_trigger = yes }
			pod_combat_end_combat_effect = yes
		}
		else = {
			pod_combat_process_group_turns_effect = yes
		}
	}
}

###### AUTO-RESOLVE BUTTON ######

pod_combat_auto_resolve = {
	is_valid = {
		pod_combat_is_active_trigger = yes
		NOT = { pod_combat_is_over_trigger = yes }
	}
	effect = {
		pod_combat_auto_resolve_effect = yes
	}
}

###### HELP / OVERVIEW BUTTON ######

pod_combat_help = {
	is_valid = {
		pod_combat_is_active_trigger = yes
	}
	effect = {
		trigger_event = pod_combat.0001
	}
}

###### FORFEIT / CLOSE BUTTON ######

pod_combat_forfeit = {
	is_valid = {
		pod_combat_is_active_trigger = yes
	}
	effect = {
		# Player forfeits — enemy team leader wins.
		# Use slot 5's char (the canonical enemy team leader, set once
		# at combat start in pod_combat_group_init) rather than
		# var:pod_combat_opponent. In NvN the workspace pod_combat_opponent
		# gets reassigned every time the player clicks a different target
		# button, so by forfeit time it could be an ally (heal target), a
		# KO'd enemy, or unset entirely. Slot 5 is invariant for the
		# duration of the combat and works for both 1v1 (slot 5 is the
		# only enemy) and NvN (slot 5 is the team leader the duel was
		# nominally between).
		if = {
			limit = { has_variable = pod_combat_c5_char }
			set_variable = { name = pod_combat_winner value = var:pod_combat_c5_char }
		}
		else_if = {
			limit = { has_variable = pod_combat_opponent }
			set_variable = { name = pod_combat_winner value = var:pod_combat_opponent }
		}
		set_variable = { name = pod_combat_loser value = root }
		set_variable = { name = pod_combat_player_morale value = 0 }
		set_variable = { name = pod_combat_is_over value = yes }
		# Combat window stays open — player clicks End Combat to dismiss
	}
}

###### TARGET SELECTION (UNIFIED) ######
# Called from the Target button on any non-PC card in the combat window.
# Sets the PC's persistent target (pod_combat_pc_target) and mirrors it
# into pod_combat_selected_target so the workspace dispatch loads the
# correct slot into the opponent workspace. Each slot 1-9 gets its own
# GUI; slot 0 (the PC) has no target button — self-only skills use
# targeting = self.
#
# Hit-chance handling differs by team:
#   - Slots 1-4 (allies): no attack will be made against them, so the
#     SGUI just mirrors the actor's current hit_chance forward without
#     recalculating against the ally's prowess. This avoids a meaningless
#     ally-vs-actor comparison polluting the workspace.
#   - Slots 5-9 (enemies): calls pod_combat_recalc_player_hit_chance_effect
#     to derive a fresh hit_chance against the targeted enemy's prowess
#     for the damage pipeline.
# A reader expecting symmetric behavior should NOT extract the recalc
# call into a shared helper without first auditing every per-slot SGUI
# below to see which team it targets.

pod_combat_select_target_1 = {
	is_valid = {
		pod_combat_is_active_trigger = yes
		pod_combat_is_player_turn_trigger = yes
		has_variable = pod_combat_c1_char
		var:pod_combat_c1_alive = 1
		NOT = { pod_combat_is_over_trigger = yes }
	}
	effect = {
		set_variable = { name = pod_combat_pc_target value = 1 }
		set_variable = { name = pod_combat_selected_target value = 1 }
		pod_combat_load_target_from_slot_effect = { SLOT = 1 }
		# Ally target — leave player_hit_chance as-is (no attack against allies)
		set_variable = { name = pod_combat_current_hit_chance value = var:pod_combat_player_hit_chance }
	}
}
