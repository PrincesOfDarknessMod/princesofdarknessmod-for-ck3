quincunx_enable = {
	scope = character
	saved_scopes = { player }

	is_shown = {
		scope:player ?= { has_trait = kueijin }
		has_global_variable = quincunx_started
	}
	is_valid = { 
		scope:player = { 
			custom_tooltip = {
				text = heretical_court_government
				has_government = august_court_government 
			}
		}
	}
	effect = {
		clear_global_variable_list = ancestor_list
		every_august_ruler = {
			limit = { highest_held_title_tier = tier_empire }
			add_to_global_variable_list = { name = ancestor_list target = this }
		}
	}
}
