
# ScriptedGui.BuildTooltip lets you output the effect of an SGUI as a string.
# this SGUI displays the dynamic tooltips for duel moves.
# it makes sure to only show the mechanics that are relevant.
POD_duel_move_desc = {
	# global scope
	saved_scopes = { story }
	effect = {
		if = {
			limit = { scope:story = { has_variable = energy_cost } }
			custom_tooltip = POD_duel_move_energy_cost
		}
		if = {
			limit = { scope:story = { has_variable = piety_cost } }
			custom_tooltip = POD_duel_move_piety_cost
		}
		if = {
			limit = { scope:story = { has_variable = damage_value } }
			custom_tooltip = POD_duel_move_damage
		}
		if = {
			limit = { scope:story = { has_variable = healing_value } }
			custom_tooltip = POD_duel_move_healing
		}
		if = {
			limit = { scope:story = { has_variable = undodgeable } }
			custom_tooltip = POD_duel_move_undodgeable
		}
	}
}


##### SGUIs for the loadout picker

POD_duel_close_loadout_window = {
	scope = character
	is_valid = {
		has_variable = pod_combat_config_open
	}
	effect = {
		remove_variable = pod_combat_config_open
	}
}


POD_duel_loadout_show_all_moves = {
	scope = character
	saved_scopes = { story }
	effect = {
		POD_toggle_variable = { VAR = show_all_duel_moves }
	}
}

POD_duel_loadout_option = {
	scope = character
	saved_scopes = { story }
	is_shown = {
		trigger_if = {
			limit = { has_variable = show_all_duel_moves }
			always = yes
		}
		trigger_else = {
			can_execute_decision = scope:story.var:decision
		}
	}
	is_valid = {
		can_execute_decision = scope:story.var:decision
		OR = {
			is_target_in_variable_list = {
				name = POD_duel_moves
				target = scope:story
			}
			POD_duel_moves_in_loadout < POD_max_duel_moves_in_loadout
		}
	}
}

POD_duel_loadout_toggle = {
	scope = character
	saved_scopes = { story }
	is_shown = {
		is_target_in_variable_list = {
			name = POD_duel_moves
			target = scope:story
		}
	}
	effect = {
		if = {
			limit = {
				is_target_in_variable_list = {
					name = POD_duel_moves
					target = scope:story
				}
			}
			remove_list_variable = {
				name = POD_duel_moves
				target = scope:story
			}
		}
		else = {
			add_to_variable_list = {
				name = POD_duel_moves
				target = scope:story
			}
		}
	}
}
