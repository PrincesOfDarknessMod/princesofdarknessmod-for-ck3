# Muscularity
vicissitude_muscularity_add = {
	scope = character
	is_shown = {}
	is_valid = {
		has_variable = vicissitude_muscular
		var:vicissitude_muscular < 1
	}
	effect = {
		change_variable = { name = vicissitude_muscular add = 0.1 }
	}
}

vicissitude_muscularity_subtract = {
	scope = character
	is_shown = {}
	is_valid = {
		has_variable = vicissitude_muscular
		var:vicissitude_muscular > -1
	}
	effect = {
		change_variable = { name = vicissitude_muscular subtract = 0.1 }
	}
}

vicissitude_muscularity_reset = {
	scope = character
	is_shown = {}
	is_valid = {}
	effect = {
		set_variable = { name = vicissitude_muscular value = 0 }
	}
}
