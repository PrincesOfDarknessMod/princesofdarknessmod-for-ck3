earth_location = {
	scope = character
	confirm_title = umbra_cooldown_title
	confirm_text = umbra_cooldown_text
	is_valid = {
		POD_umbra_travel_valid_trigger = { LOCATION = earth }
		has_access_to_skinlands_trigger = yes
	}
	effect = { 
		POD_umbra_choose_effct = { LOCATION = earth } 
		custom_tooltip = line_break
		custom_description_no_bullet = {
			text = umbra_cooldown_desc
			value = umbra_cooldown_value
			add_character_flag = {
				flag = umbra_cooldown
				years = umbra_cooldown_value
			}
		}
	}
}
