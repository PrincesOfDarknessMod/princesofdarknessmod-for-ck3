skinlands_location = {
	scope = character
	is_shown = {
		NOT = { has_variable = spirit_destination }
		has_variable = spirit_location
		OR = {
			var:spirit_location = flag:periphery
			var:spirit_location = flag:shadowlands
			var:spirit_location = flag:spirit_wilds
			var:spirit_location = flag:astral_plane
			var:spirit_location = flag:shard_realm_of_prime
			var:spirit_location = flag:dreamtime
			var:spirit_location = flag:near_dreaming
			var:spirit_location = flag:digital_web
			
			var:spirit_location = flag:first_city
			var:spirit_location = flag:null_zone
		}
	}
	is_valid = {
		has_access_to_skinlands_trigger = yes
	}
	effect = {
		POD_umbra_choose_effct = { LOCATION = skinlands }
	}
}

astral_plane_location = {
	scope = character
	is_shown = {
		NOT = { has_variable = spirit_destination }
		has_variable = spirit_location
		OR = {
			var:spirit_location = flag:skinlands
			var:spirit_location = flag:spirit_wilds
			var:spirit_location = flag:vulgate
			var:spirit_location = flag:periphery
			var:spirit_location = flag:the_homeland_of_the_stargazers
			var:spirit_location = flag:mirror_zone
			var:spirit_location = flag:calumn
		}
	}
	is_valid = {
		trigger_if = {
			limit = {
				has_variable = spirit_location
				NOT = { var:spirit_location = flag:vulgate }
			}
			NOT = { location = { has_province_modifier = gauntlet10_modifier } }
			has_local_gauntlet_modifier_trigger = yes
			has_access_to_astral_reaches_trigger = yes
		}
	}
	effect = {
		POD_umbra_choose_effct = { LOCATION = astral_plane }
	}
}
