local_umbra_enable = {
	scope = character
	saved_scopes = { player }
	is_valid = {
		scope:player ?= {
			OR = {
				has_access_to_astral_reaches_trigger = yes
				has_access_to_spirit_wilds_trigger = yes
				has_access_to_dreaming_trigger = yes
				has_access_to_underworld_trigger = yes
				has_trait = demon
			}
		}
	}
}

umbra_buff_list_initiate = {
	scope = character
	effect = {
		if = {
			limit = { NOT = { has_character_flag = had_umbra_buff_list_initiation } }
			add_to_variable_list = {
				name = spirit_wilds_buff_list
				target = flag:umbra_scheme_resistance
			}
			add_to_variable_list = {
				name = spirit_wilds_buff_list
				target = flag:umbra_levy_maintenance
			}
			add_to_variable_list = {
				name = spirit_wilds_buff_list
				target = flag:umbra_fort_level
			}
			add_to_variable_list = {
				name = shadowlands_buff_list
				target = flag:umbra_tax_contribution_reduction
			}
			add_to_variable_list = {
				name = shadowlands_buff_list
				target = flag:umbra_mercenary_hire_cost
			}
			add_to_variable_list = {
				name = shadowlands_buff_list
				target = flag:umbra_county_control_change
			}
			add_character_flag = had_umbra_buff_list_initiation
		}
	}
}

apply_umbra_buff_change = {
	scope = province
	effect = {
		county = {
			if = {
				limit = {
					AND = {
						has_variable = spirit_wilds_buff_choice
						OR = {
							NOT = { has_variable = local_spirit_wilds_buff }
							NOT = { var:local_spirit_wilds_buff = var:spirit_wilds_buff_choice }
						}
					}
				}
				if = {
					limit = { has_variable = local_spirit_wilds_buff }
					POD_remove_local_umbra_buff_effect = {
						BUFF = var:local_spirit_wilds_buff
						VICTIM = var:spirit_wilds_controller
					}
					POD_remove_local_umbra_buff_side_effect = { #Player only
						BUFF = var:local_spirit_wilds_buff
						VICTIM = var:spirit_wilds_controller
					}
				}
				POD_add_local_umbra_buff_effect = {
					BUFF = var:spirit_wilds_buff_choice
					VICTIM = var:spirit_wilds_controller
				}
				POD_add_local_umbra_buff_side_effect = { #Player only
					BUFF = var:spirit_wilds_buff_choice
					VICTIM = var:spirit_wilds_controller
				}
				if = {
					limit = { is_temporal_umbra_buff_trigger = { BUFF = var:spirit_wilds_buff_choice } }
					set_variable = { 
						name = local_spirit_wilds_buff 
						value = var:spirit_wilds_buff_choice 
						years = 5
					}
				}
				else = {
					set_variable = { name = local_spirit_wilds_buff value = var:spirit_wilds_buff_choice }
				}
				var:spirit_wilds_controller = {
					POD_anger_the_local_spirit_effect = yes
				}
			}
			if = {
				limit = {
					AND = {
						has_variable = shadowlands_buff_choice
						OR = {
							NOT = { has_variable = local_shadowlands_buff }
							NOT = { var:local_shadowlands_buff = var:shadowlands_buff_choice }
						}
					}
				}
				if = {
					limit = { has_variable = local_shadowlands_buff }
					POD_remove_local_umbra_buff_effect = {
						BUFF = var:local_shadowlands_buff
						VICTIM = var:shadowlands_controller
					}
					POD_remove_local_umbra_buff_side_effect = { #Player only
						BUFF = var:local_shadowlands_buff
						VICTIM = var:shadowlands_controller
					}
				}
				POD_add_local_umbra_buff_effect = {
					BUFF = var:shadowlands_buff_choice
					VICTIM = var:shadowlands_controller
				}
				POD_add_local_umbra_buff_side_effect = { #Player only
					BUFF = var:shadowlands_buff_choice
					VICTIM = var:shadowlands_controller
				}
				if = {
					limit = { is_temporal_umbra_buff_trigger = { BUFF = var:shadowlands_buff_choice } }
					set_variable = { 
						name = local_shadowlands_buff 
						value = var:shadowlands_buff_choice 
						years = 5
					}
				}
				else = {

	scope = province
	saved_scopes = { player }
	is_shown = {
	}
	is_valid = {
		scope:player ?= {
			OR = {
				has_access_to_spirit_wilds_trigger = yes 
				has_access_to_dreaming_trigger = yes
				has_access_to_astral_reaches_trigger = yes
			}
		}
	}
	effect = {
		county = {
			save_scope_as = target_county
			if = {
				limit = { var:spirit_wilds_controller ?= { is_alive = yes } }
				var:spirit_wilds_controller = { save_scope_as = victim }
				scope:player ?= { trigger_event = POD_umbra_local.301 }
			}
			else = {
				scope:player ?= { trigger_event = POD_umbra_local.201 }
			}
		}
	}
}
