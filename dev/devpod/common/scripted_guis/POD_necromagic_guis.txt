spell_haunting_script = {
	scope = character
	is_shown = { }
	is_valid = {
		POD_has_enough_blood_for_discipline = { AMOUNT = 3 }
                num_of_relation_bound_wraith >= 1
		has_perk = ns_haunting_perk
		NOT = { scope:target = ROOT }
                scope:target = { capital_county ?= { NOT = { has_county_modifier = spell_haunting_modifier } } }
	}
	effect = {
		POD_willpower_rouse_check_effect = yes
		POD_static_stress_gain = { COUNT = 25 }
		random_list = {
			0 = {
				modifier = { add = POD_necromagic_success_chance }
				send_interface_toast = { 
					title = POD_shared_loca_success
					scope:target = {
                                                location = {
                                                        county = {
                                                                add_county_modifier = { modifier = spell_haunting_modifier years = 5 }
                                                        }
                                                }
                                        }
				}
			}
			100 = {
				modifier = { add = POD_necromagic_success_chance_negated }
				send_interface_toast = { 
					title = POD_shared_loca_failure
					POD_static_stress_gain = { COUNT = 50 }
				}
			}
		}
		POD_discipline_cost_effect = { COUNT = 3 }
	}
}
