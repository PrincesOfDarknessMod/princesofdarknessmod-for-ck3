POD_crafting_minigame_reroll = {
	is_valid = {
		custom_tooltip = { text = "POD_crafting_minigame_all_rerolls_used" NOT = { crafting_minigame_reloads = 0 } }
	}
	effect = {
		hidden_effect = {

			var:crafting_minigame = {
				save_scope_as = crafting_minigame
				destroy_artifact = var:dummy_artifact
				var:node_map ?= { end_story = yes }
				remove_variable = node_map
				remove_variable = current_location
				remove_variable = moves
				remove_variable = immortal_number
				remove_variable = vampire_number
				remove_variable = kinfolk_number
				remove_variable = kinain_number

				trigger = var:crafting_minigame.var:type
				flag:forge = { POD_crafting_create_forge = yes }
				flag:brew = { POD_crafting_create_brew = yes }
				flag:enhance= { POD_crafting_create_enhance = yes }
			}
			character:POD_wraith_01 = {
				save_scope_as = owner
				set_artifact_rarity_common = yes
				create_artifact = {
					name = dummy_artifact_loca  description = dummy_artifact_loca
					type = armor_plate visuals = armor
					wealth = 1 quality = 1
					modifier = artifact_placeholder_modifier
					generate_history = no
					save_scope_as = newly_created_artifact
				}
				scope:newly_created_artifact = { remove_artifact_modifier = artifact_placeholder_modifier }
			}
			var:crafting_minigame = {
				if = { limit = { has_variable = reloads } change_variable = { name = reloads add = 1 } }
				else = { set_variable = { name = reloads value = 1 } }
				set_variable = { name = dummy_artifact value = scope:newly_created_artifact }
			}
		}
	}
}

POD_crafting_minigame_node = {
	scope = character
	saved_scopes = {
		story
	}
	is_shown = {
		var:crafting_minigame.var:current_location ?= scope:story
	}
	is_valid = {
		save_temporary_scope_as = owner
		var:crafting_minigame = {
			trigger_if = { limit = { var:type ?= flag:forge NOT = { has_variable = current_location } }
				custom_tooltip = { text = "POD_crafting_minigame_node_is_starting" scope:story = { POD_node_is_starting = yes } }
			}
			trigger_else = {
				custom_tooltip = { text = "POD_crafting_minigame_node_is_not_starting" scope:story = { POD_node_is_starting = no } }
			}
			trigger_if = { limit = { has_variable = current_location NOT = { var:current_location ?= scope:story } }
				custom_tooltip = { text = "POD_crafting_minigame_node_too_far" POD_line_exists_between_nodes = { NODE1 = var:current_location NODE2 = scope:story } }
			}
			custom_tooltip = { text = "POD_crafting_minigame_choose_sacrifice" NOT = { has_variable = open_sacrifice_gui } }
			custom_tooltip = { text = "POD_crafting_minigame_used_all_moves" NOT = { PREV.crafting_minigame_moves = 0 } }
			POD_node_requirement = {
                                OWNER = scope:owner
                                ARTIFACT = var:dummy_artifact
                                FLAG = scope:story.var:node_type
                                CRAFTER = var:crafter
                        }
		}
		custom_tooltip = {  text = "POD_crafting_minigame_node_used" scope:story = { NOT = { has_variable = node_used } } }
	}
	effect = {
                save_scope_as = owner
		var:crafting_minigame = {
			var:current_location ?= { 
				set_variable = node_used 
				every_in_list = {
					variable = lines
					save_scope_as = line
					limit = {
						OR = {
							var:start_node = scope:story
							var:end_node = scope:story
						}
						
					}
					set_variable = { name = opacity value = 1 }
				}
			}
			set_variable = { name = current_location value = scope:story }
			POD_crafting_minigame_execute_node = {
                                OWNER = scope:owner
                                ARTIFACT = var:dummy_artifact
                                FLAG = scope:story.var:node_type
                                CRAFTER = var:crafter
                        }

			if = { limit = { has_variable = moves } change_variable = { name = moves add = 1 } }
			else = { set_variable = { name = moves value = 0 } }

			switch = {
				trigger = var:type
				flag:forge = { var:crafter ?= { POD_crafter_xp_gain = { TRACK = forging } } }
				flag:brew = { var:crafter ?= { POD_crafter_xp_gain = { TRACK = brewing } } }
				flag:enhance = { var:crafter ?= { POD_crafter_xp_gain = { TRACK = enhancing } } }
			}
		}
	}
}

POD_crafting_minigame_end = {
	is_valid = {
		custom_tooltip = { text = "POD_cannot_end_without_starting" var:crafting_minigame = { has_variable = current_location } }
		custom_tooltip = { text = "POD_crafting_minigame_choose_sacrifice" var:crafting_minigame = { NOT = { has_variable = open_sacrifice_gui } } }
	}
	effect = {
		save_scope_as = owner
		switch = {
			trigger = var:crafting_minigame.var:type
			flag:forge = { set_variable = { name = forge_cooldown value = yes years = 5 } }
			flag:brew = { set_variable = { name = brew_cooldown value = yes years = 5 } }
			flag:enhance = { set_variable = { name = enhance_cooldown value = yes years = 10 } }
		}
		var:crafting_minigame = {
			save_scope_as = minigame
			var:crafter ?= { save_scope_as = crafter }
			if = { limit = { var:type ?= flag:brew } var:artifact_target ?= { reforge_artifact = { template = alchemy_artifact_template type = miscellaneous visuals = flask } } } # Safe Check
			if = { limit = { var:type ?= flag:enhance } var:artifact_target ?= { copy_artifact_modifiers = scope:minigame.var:dummy_artifact } }
			else = {
				var:dummy_artifact = {
					set_artifact_rarity_famed = yes
					set_owner = scope:owner
					save_scope_as = artifact
					set_variable = { name = crafter value = scope:crafter }
					set_artifact_name = custom_artifact_name
					set_artifact_description = custom_artifact_desc
				}
			}
			var:node_map ?= { end_story = yes }
			end_story = yes
		}
		remove_variable = crafting_minigame
	}
}

	effect = {
		if = { limit = { has_variable = possible_sacrifice } remove_variable = possible_sacrifice }

			limit = {
				trigger_if = { limit = { PREV.var:open_sacrifice_gui ?= flag:fleshcraft }
					NOT = { has_trait_with_flag = is_spirit }
				}
				trigger_if = { limit = { PREV.var:open_sacrifice_gui ?= flag:soulforge }
					has_trait = wraith
				}
				trigger_if = { limit = { PREV.var:open_sacrifice_gui ?= flag:fetishsmith }
					has_trait = spirit
				}
				trigger_if = { limit = { PREV.var:open_sacrifice_gui ?= flag:vampire }
					has_trait = vampire
				}
				trigger_if = { limit = { PREV.var:open_sacrifice_gui ?= flag:kinfolk }
					POD_splat_is_shifter_trigger = yes
				}
				trigger_if = { limit = { PREV.var:open_sacrifice_gui ?= flag:kinain }
					has_POD_fae_blood_trigger = yes
				}
				trigger_if = { limit = { PREV.var:open_sacrifice_gui ?= flag:imbue_object } }
				is_alive = yes
			}
			save_scope_as = sacrifice
			PREV = { add_to_variable_list = { name = possible_sacrifice target = scope:sacrifice } }
		}
		if = { limit = { PREV.var:open_sacrifice_gui ?= flag:fetishsmith }
			every_courtier = {
				limit = {
					opinion = {
						target = PREV
						value >= 50
					}
				}
				save_scope_as = sacrifice
				PREV = { add_to_variable_list = { name = possible_sacrifice target = scope:sacrifice } }
			}
		}
	}
}

	effect = {
		save_scope_as = killer
		var:crafting_minigame = {
			var:dummy_artifact ?= {
				save_scope_as = dummy_artifact
			}
			var:sacrifice ?= {
				if = {
					limit = { NOT = { PREV.var:open_sacrifice_gui ?= flag:vampire } }
					POD_crafting_give_modifier_based_on_character = { ARTIFACT = scope:dummy_artifact }
				}
				death = {
					death_reason = death_fleshcrafting
					killer = scope:killer
				}
				if = { limit = { PREV.var:open_sacrifice_gui ?= flag:vampire }
					PREV.var:dummy_artifact ?= {
						set_variable = { name = sacrifice_used value = PREV }
						POD_crafting_brewing_add_effect = { TYPE = alchemy_vampire }
					}
				}
				if = { limit = { PREV.var:open_sacrifice_gui ?= flag:kinain }
					PREV.var:dummy_artifact ?= {
						set_variable = { name = sacrifice_used value = PREV }
						POD_crafting_brewing_add_effect = { TYPE = alchemy_kinain }
					}
				}
				if = { limit = { PREV.var:open_sacrifice_gui ?= flag:kinfolk }
					PREV.var:dummy_artifact ?= {
						set_variable = { name = sacrifice_used value = PREV }
						POD_crafting_brewing_add_effect = { TYPE = alchemy_kinfolk }
					}
				}
			}
			remove_variable = open_sacrifice_gui
			remove_variable = sacrifice
		}
	}
	is_valid = {
		custom_tooltip = { text = "POD_crafting_minigame_must_sacrifice" var:crafting_minigame = { has_variable = sacrifice } }
		var:crafting_minigame.var:sacrifice ?= { is_alive = yes }
	}
