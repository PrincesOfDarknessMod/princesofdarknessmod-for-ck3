
POD_reload_crafting_minigame = {
	scope = character
	is_valid = {
		custom_tooltip = { text = "POD_used_all_reloads" NOT = { var:crafting_minigame.var:reloads ?= var:crafting_minigame.var:crafter.crafting_minigame_reloads_max } }
	}
	effect = {
		if = { limit = { scope:tooltip ?= 1 } 
		
		}
		else = {
			var:crafting_minigame = {
				save_scope_as = crafting_minigame
			}
			POD_crafting_minigame_generate_map = {
				WIDTH = scope:crafting_minigame.var:width
				HEIGHT = scope:crafting_minigame.var:height
				ROWS = scope:crafting_minigame.var:rows
				COLUMNS = scope:crafting_minigame.var:columns
				PADDING = scope:crafting_minigame.var:padding
				NODE_CHANCE = scope:crafting_minigame.var:node_chance
				SAVE_AS = node_map
			}
			var:crafting_minigame = {
				var:node_map = {
					end_story = yes
				}
				set_variable = {
					name = node_map
					value = scope:node_map
				}
				remove_variable = current_location
				remove_variable = moves
				if = { limit = { has_variable = reloads } change_variable = { name = reloads add = 1 } }
				else = { set_variable = { name = reloads value = 1 } }
			}
		}
	}
}

POD_open_crafting_minigame = {
	scope = character
	effect = {
		if = { limit = { scope:creating ?= 1 } POD_crafting_minigame_create = yes }
                set_variable = POD_open_crafting_minigame
	}
}

POD_crafting_minigame_is_current_location = {
	scope = character
	saved_scopes = {
		story
	}
	is_shown = {
                trigger_if = { limit = { var:crafting_minigame = { has_variable = current_location } } 
		        scope:story = var:crafting_minigame.var:current_location
                }
                trigger_else = {
                        always = no
                }
	}
}

POD_crafting_minigame_node = {
	scope = character
	saved_scopes = {
		story
	}
	is_shown = {
		scope:story.var:include = yes
	}
	is_valid = {
                custom_tooltip = { 
                        text = "POD_already_used_node"
		        scope:story = { NOT = { has_variable_list = lines } }
                }
                trigger_if = {
                        limit = { NOT = { var:crafting_minigame.var:current_location ?= scope:story } }
                        custom_tooltip = {
                                text = "POD_too_far"
                                
                                var:crafting_minigame = { 
                                        trigger_if = { limit = { has_variable = current_location } 
                                                scope:story = { has_variable = possible_node }
                                        }
                                        trigger_else = {
                                                always = yes
                                        }
                                }
                        }
                }
		trigger_if = {
                        limit = { NOT = { PREV = { has_variable = target_artifact } } NOT = { var:crafting_minigame = { has_variable = current_location } } }
			custom_tooltip = {
				text = "POD_must_be_starting_node"
				scope:story = {
					POD_node_is_starting = yes
				}
			}
		}
		trigger_else = {
			custom_tooltip = {
				text = "POD_cannot_be_starting_node"
				scope:story = {
					POD_node_is_starting = no
				}
			}
		}
		trigger_if = { limit = { var:crafting_minigame = { has_variable = artifact_type } } 
			scope:story = {
				POD_node_requirement = { MINIGAME = PREV.var:crafting_minigame }
			}
		}
		trigger_if = { limit = { var:crafting_minigame = { has_variable = moves } } 
			custom_tooltip = { text = "POD_used_all_moves" NOT = { var:crafting_minigame.var:moves = crafting_minigame_max_moves } }
		}
	}
	effect = {
		var:crafting_minigame = {
                        var:node_map = {
                                every_in_list = {
                                        variable = nodes
                                        limit = { has_variable = possible_node }
                                        remove_variable = possible_node
                                }
                        }
                        if = { limit = { has_variable = current_location } 
                                var:current_location ?= {
                                        save_scope_as = current_location
                                        set_variable = possible_node
                                }
                                POD_create_line = {
                                        MAP = var:node_map
                                        START = scope:story
                                        END = scope:current_location
                                        SAVE_AS = line
                                }
                        }
                        set_variable = {
                                name = current_location
                                value = scope:story
                        }
			if = { limit = { scope:story = { POD_node_is_starting = yes } } set_variable = { name = artifact_type value = scope:story.var:type } }
                        var:node_map = {
                                every_in_list = {
                                        variable = nodes
					POD_node_squared_distance = { NODE1 = THIS NODE2 = scope:story SAVE_AS = distance }

					if = { limit = { scope:distance <= 50000 } set_variable = possible_node }
                                        
                                }
                        }
                        POD_crafting_minigame_apply_modifier_from_node = yes
			if = { limit = { has_variable = moves } change_variable = { name = moves add = 1 } }
			else = { set_variable = { name = moves value = 0 } }
                }
	}
}

POD_crafting_minigame_close_window = {
	scope = character
	is_valid = {
		trigger_if = { limit = { scope:finishing ?= 1 } 
			custom_tooltip = { text = POD_cannot_end_without_starting var:crafting_minigame = { has_variable = current_location } }
		}
		trigger_else = {
			always = yes
		}
	}
	effect = {
		if = { limit = { scope:finishing ?= 1 } 
			if = { limit = { var:crafting_minigame = { has_variable = current_location } } 
				save_scope_as = owner
				var:crafting_minigame = {
					save_scope_as = minigame
					if = { limit = { has_variable = target_artifact }
						var:target_artifact ?= {
							copy_artifact_modifiers = scope:minigame.var:dummy_artifact
						}
					}
					else = {
						var:dummy_artifact = { 
							set_artifact_rarity_famed = yes 
							set_owner = scope:owner 
							reforge_artifact = { name = custom_artifact_name description = custom_artifact_desc } 
						}
						
						switch = {
							trigger = var:artifact_type
							flag:node_amulet = { var:dummy_artifact = { reforge_artifact = { type = miscellaneous visuals = necklace } } }
							flag:node_crown = { var:dummy_artifact = { reforge_artifact = { type = helmet visuals = crown } } }
							flag:node_weapon = { var:dummy_artifact = { reforge_artifact = { type = sword visuals = sword } } }
							flag:node_regalia = { var:dummy_artifact = { reforge_artifact = { type = regalia visuals = regalia } } }
							flag:node_armor = { var:dummy_artifact = { reforge_artifact = { type = armor_plate visuals = armor } } }
						}
					}
					var:node_map = {
						end_story = yes
					}
				}
				remove_variable = crafting_minigame
			}
		}
		remove_variable = POD_open_crafting_minigame
	}
}

POD_crafting_minigame_crafter_init = {
	effect = {
		if = { limit = { has_variable = possible_crafters } remove_variable = possible_crafters }
		add_to_variable_list = { name = possible_crafters target = THIS }
		every_vassal = {
			limit = { stewardship >= 10 }
			save_scope_as = crafter
			PREV = { add_to_variable_list = { name = possible_crafters target = scope:crafter } }
		} 
		every_courtier = {
			limit = { stewardship >= 10 }
			save_scope_as = crafter
			PREV = { add_to_variable_list = { name = possible_crafters target = scope:crafter } }
		} 
	}
}
