POD_crafting_minigame_node = {
	scope = character
	saved_scopes = {
		story
	}
	is_shown = {
		var:crafting_minigame.var:current_location ?= scope:story
	}
	is_valid = {
		save_temporary_scope_as = owner
		var:crafting_minigame = {
			trigger_if = { limit = { var:type ?= flag:forge NOT = { has_variable = current_location } }
				custom_tooltip = { text = "POD_crafting_minigame_node_is_starting" scope:story = { POD_node_is_starting = yes } }
			}
			trigger_else = {
				custom_tooltip = { text = "POD_crafting_minigame_node_is_not_starting" scope:story = { POD_node_is_starting = no } }
			}
			trigger_if = { limit = { has_variable = current_location NOT = { var:current_location ?= scope:story } }
				custom_tooltip = { text = "POD_crafting_minigame_node_too_far" POD_line_exists_between_nodes = { NODE1 = var:current_location NODE2 = scope:story } }
			}
			custom_tooltip = { text = "POD_crafting_minigame_choose_sacrifice" NOT = { has_variable = open_sacrifice_gui } }
			custom_tooltip = { text = "POD_crafting_minigame_used_all_moves" NOT = { PREV.crafting_minigame_moves = 0 } }
			POD_node_requirement = {
				OWNER = scope:owner
				ARTIFACT = var:dummy_artifact
				FLAG = scope:story.var:node_type
				CRAFTER = var:crafter
			}
		}
		custom_tooltip = { text = "POD_crafting_minigame_node_used" scope:story = { NOT = { has_variable = node_used } } }
	}
	effect = {
		save_scope_as = owner
		var:crafting_minigame = {
			var:current_location ?= {
				set_variable = node_used
				every_in_list = {
					variable = lines
					save_scope_as = line
					limit = {
						OR = {
							var:start_node = scope:story
							var:end_node = scope:story
						}

					}
					set_variable = { name = opacity value = 1 }
				}
			}
			set_variable = { name = current_location value = scope:story }
			POD_crafting_minigame_execute_node = {
				OWNER = scope:owner
				ARTIFACT = var:dummy_artifact
				FLAG = scope:story.var:node_type
				CRAFTER = var:crafter
			}

			if = { limit = { has_variable = moves } change_variable = { name = moves add = 1 } }
			else = { set_variable = { name = moves value = 0 } }
		}
	}
}

POD_crafting_minigame_end = {
	is_valid = {
		custom_tooltip = { text = "POD_cannot_end_without_starting" var:crafting_minigame = { has_variable = current_location } }
		custom_tooltip = { text = "POD_crafting_minigame_choose_sacrifice" var:crafting_minigame = { NOT = { has_variable = open_sacrifice_gui } } }
	}
	effect = {
		save_scope_as = owner
		switch = {
			trigger = var:crafting_minigame.var:type
			flag:forge = { 
				#if = { limit = { has_character_modifier = POD_one_time_forge_boost } remove_character_modifier = POD_one_time_forge_boost }
				#remove_variable = POD_one_time_forge_boost
				set_variable = { name = forge_cooldown value = yes years = 5 } 
				var:crafting_minigame.var:crafter = { POD_crafter_xp_gain = { TRACK = forging } }
			}
			flag:brew = { 
				#if = { limit = { has_character_modifier = POD_one_time_brew_boost } remove_character_modifier = POD_one_time_brew_boost }
				#remove_variable = POD_one_time_brew_boost
				set_variable = { name = brew_cooldown value = yes years = 5 } 
				var:crafting_minigame.var:crafter = { POD_crafter_xp_gain = { TRACK = brewing } }
			}
			flag:enhance = { 
				#if = { limit = { has_character_modifier = POD_one_time_enhance_boost } remove_character_modifier = POD_one_time_enhance_boost }
				#remove_variable = POD_one_time_enhance_boost
				set_variable = { name = enhance_cooldown value = yes years = 10 } 
				var:crafting_minigame.var:crafter = { POD_crafter_xp_gain = { TRACK = enhancing } }
			}
		}
		var:crafting_minigame = {
			save_scope_as = minigame
			var:crafter ?= { save_scope_as = crafter }
			if = { limit = { var:type ?= flag:brew } var:artifact_target ?= { reforge_artifact = { template = alchemy_artifact_template type = miscellaneous visuals = flask } } } # Safe Check
			if = { 
				limit = { var:type ?= flag:enhance }
				var:artifact_target ?= { 
					copy_artifact_modifiers = scope:minigame.var:dummy_artifact 
					if = { limit = { has_variable = enhance_amount } change_variable = { name = enhance_amount add = 1 } }
					else = { set_variable = { name = enhance_amount value = 1 } }
				} 
			}
			else = {
				var:dummy_artifact = {
					POD_crafting_set_rarity_effect = yes
					set_owner = scope:owner
					save_scope_as = artifact
					set_variable = { name = crafter value = scope:crafter }
					if = {
						limit = { has_variable = fetish_artifact }
						set_artifact_name = custom_fetish_artifact_name
					}
					else = { set_artifact_name = custom_artifact_name }
					set_artifact_description = custom_artifact_desc
				}
			}
			scope:crafter = {

	effect = {
		clear_variable_list = possible_sacrifice
		var:crafting_minigame.var:open_sacrifice_gui = { save_scope_as = sacrifice_gui }
		every_prisoner = {

					NOT = { has_trait_with_flag = is_spirit }

					has_trait = wraith
				}

					has_trait = spirit
				}
				trigger_if = { limit = { scope:sacrifice_gui ?= flag:vampire }
					has_trait = vampire
				}
				trigger_if = { limit = { scope:sacrifice_gui ?= flag:kinfolk }
					OR = {
						POD_splat_is_fera_trigger = yes
						has_trait = kinfolk
					}
				}
				trigger_if = { limit = { scope:sacrifice_gui ?= flag:kinain }
					has_POD_fae_blood_trigger = yes
				}
				is_alive = yes
			}
			save_scope_as = sacrifice
			PREV = { add_to_variable_list = { name = possible_sacrifice target = scope:sacrifice } }
		}
		if = { limit = { scope:sacrifice_gui ?= flag:fetish_smith }
			every_courtier = {
				limit = {
					opinion = {
						target = PREV
						value >= 10
					}
					has_trait = spirit
				}
				save_scope_as = sacrifice
				PREV = { add_to_variable_list = { name = possible_sacrifice target = scope:sacrifice } }
			}
		}
	}

POD_crafting_minigame_sacrifice_gui_execute = {
	effect = {
		save_scope_as = killer
		var:crafting_minigame = {
			var:dummy_artifact ?= {
				save_scope_as = dummy_artifact
			}
			var:sacrifice ?= {
				if = {
					limit = { 
						NOR = { # All sacrifice methods that doesn't give modifiers
							PREV.var:open_sacrifice_gui ?= flag:vampire 
							PREV.var:open_sacrifice_gui ?= flag:kinain 
							PREV.var:open_sacrifice_gui ?= flag:kinfolk
							PREV.var:open_sacrifice_gui ?= flag:fetish_smith
						} 
					}
					POD_crafting_give_modifier_based_on_character = { ARTIFACT = scope:dummy_artifact }
				}
				if = {
					limit = { PREV.var:open_sacrifice_gui ?= flag:fetish_smith }
					POD_crafting_fetish_smithing_effect = { ARTIFACT = scope:dummy_artifact }
					if = {
						limit = { num_of_relation_human_companion >= 1 }
						every_relation = {
							type = human_companion

							trigger_event = { id = POD_spirit_maintenance.01 days = 1 }
						}
					}
					if = {
						limit = { is_imprisoned = yes }
						death = { death_reason = death_fetish_smithing_forced killer = scope:killer }	
					}
					else = { death = { death_reason = death_fetish_smithing } }
				}
				else = {
					death = {
						death_reason = death_fleshcrafting
						killer = scope:killer
					}
				}
				if = { limit = { PREV.var:open_sacrifice_gui ?= flag:vampire }
					PREV.var:dummy_artifact ?= {
						set_variable = { name = sacrifice_used value = PREV }
						POD_crafting_brewing_add_effect = { TYPE = alchemy_vampire }
					}
				}
				if = { limit = { PREV.var:open_sacrifice_gui ?= flag:kinain }
					PREV.var:dummy_artifact ?= {
						set_variable = { name = sacrifice_used value = PREV }
						POD_crafting_brewing_add_effect = { TYPE = alchemy_kinain }
					}
				}
				if = { limit = { PREV.var:open_sacrifice_gui ?= flag:kinfolk }
					PREV.var:dummy_artifact ?= {
						set_variable = { name = sacrifice_used value = PREV }
						POD_crafting_brewing_add_effect = { TYPE = alchemy_kinfolk }
					}
				}
			}
			remove_variable = open_sacrifice_gui
			remove_variable = sacrifice
		}
	}
	is_valid = {
		custom_tooltip = { text = "POD_crafting_minigame_must_sacrifice" var:crafting_minigame = { save_temporary_scope_as = minigame has_variable = sacrifice } }
		var:crafting_minigame.var:sacrifice ?= { 
			trigger_if = { limit = { scope:minigame.var:open_sacrifice_gui ?= flag:fleshcraft }
				NOT = { has_trait_with_flag = is_spirit }
			}
			trigger_if = { limit = { scope:minigame.var:open_sacrifice_gui ?= flag:soulforge }
				has_trait = wraith
			}
			trigger_if = { limit = { scope:minigame.var:open_sacrifice_gui ?= flag:fetish_smith }
				has_trait = spirit
			}
			trigger_if = { limit = { scope:minigame.var:open_sacrifice_gui ?= flag:vampire }
				has_trait = vampire
			}
			trigger_if = { limit = { scope:minigame.var:open_sacrifice_gui ?= flag:kinfolk }
				OR = {
					POD_splat_is_fera_trigger = yes
					has_trait = kinfolk
				}
			}
			trigger_if = { limit = { scope:minigame.var:open_sacrifice_gui ?= flag:kinain }
				has_POD_fae_blood_trigger = yes
			}
			is_alive = yes 
		}
	}
}
