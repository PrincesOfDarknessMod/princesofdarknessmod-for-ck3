POD_passive_charm_open = {
	effect = {
		if = {
			limit = { POD_has_enough_essence_for_charm = { AMOUNT = 1 } }
			set_variable = POD_open_passive_charms_gui
		}
	}
}

POD_passive_charms_innit = {
	effect = {
		POD_passive_charms_close_effect = yes # Safeguard
		POD_passive_charm_injector = { EFFECT = POD_passive_charms_innit_effect }
		
		### Filters
		if = {
			limit = { NOT = { has_variable = POD_passive_charms_type_filter } }
			set_variable = { name = POD_passive_charms_type_filter value = flag:any }
		}
		add_to_variable_list = { name = POD_passive_charms_type_filter_list target = flag:any }
		add_to_variable_list = { name = POD_passive_charms_type_filter_list target = flag:terrestrial_air }
		add_to_variable_list = { name = POD_passive_charms_type_filter_list target = flag:terrestrial_earth }
		add_to_variable_list = { name = POD_passive_charms_type_filter_list target = flag:terrestrial_fire }
		add_to_variable_list = { name = POD_passive_charms_type_filter_list target = flag:terrestrial_water }
		add_to_variable_list = { name = POD_passive_charms_type_filter_list target = flag:terrestrial_wood }

		if = {
			limit = { NOT = { has_variable = POD_passive_charms_active_filter } }
			set_variable = { name = POD_passive_charms_active_filter value = flag:any }
		}
		add_to_variable_list = { name = POD_passive_charms_active_filter_list target = flag:any }
		add_to_variable_list = { name = POD_passive_charms_active_filter_list target = flag:yes }
		add_to_variable_list = { name = POD_passive_charms_active_filter_list target = flag:no }

		### Sort-By
		if = {
			limit = { NOT = { has_variable = POD_passive_charms_sort_by } }
			set_variable = { name = POD_passive_charms_sort_by value = flag:default }
		}
		add_to_variable_list = { name = POD_passive_charms_sort_by_list target = flag:default }
		add_to_variable_list = { name = POD_passive_charms_sort_by_list target = flag:cost }
		add_to_variable_list = { name = POD_passive_charms_sort_by_list target = flag:active }

		POD_passive_charm_list_refresh_effect = yes
	}
}
POD_passive_charms_close = {
	effect = {
		POD_passive_charms_close_effect = yes
		remove_variable = POD_open_passive_charms_gui
		clear_variable_list = POD_passive_charms_type_filter_list
		clear_variable_list = POD_passive_charms_active_filter_list
		clear_variable_list = POD_passive_charms_sort_by_list
	}
}

POD_passive_charms_select = {
	saved_scopes = { charm }
	is_shown = {
		scope:charm.var:activated ?= flag:yes
	}
	effect = {
		scope:charm = {
			switch = {
				trigger = var:activated
				flag:yes = { set_variable = { name = activated value = flag:no } }
				flag:no = { set_variable = { name = activated value = flag:yes } }
			}
		}
	}
}

POD_passive_charm_update = {
	is_valid = {
		custom_tooltip = {
			text = POD_passive_charm_update_cooldown
			NOT = { has_variable = POD_passive_charm_update_cooldown }
		}
	}
	effect = {
		POD_passive_charm_injector = { EFFECT = POD_passive_charms_update_effect }
		custom_tooltip = {
			text = POD_passive_charm_update_cooldown_warning
			set_variable = {
				name = POD_passive_charm_update_cooldown
				value = yes
				years = 1
			}
		}
	}
}

POD_passive_charms_set_sort_by = {
	saved_scopes = { type }
	effect = {
		set_variable = {
			name = POD_passive_charms_sort_by
			value = scope:type
		}

		POD_passive_charm_list_refresh_effect = yes
	}
}

POD_passive_charms_reverse_sort_order = {
	effect = {
		if = {
			limit = { has_variable = POD_passive_charms_sort_by_reverse }
			remove_variable = POD_passive_charms_sort_by_reverse
		}
		else = {
			set_variable = POD_passive_charms_sort_by_reverse
		}

		POD_passive_charm_list_refresh_effect = yes
	}
}

POD_passive_charms_save_layout = {
	effect = {
		trigger_event = POD_exalted_maintenance_events.3
		ordered_in_list = {
			variable = charm_list
			limit = { var:activated = flag:yes }
			max = 1000
			check_range_bounds = no
			custom_tooltip = POD_exalted_maintenance_events.3.a.tt
		}
	}
}
POD_passive_charms_select_layout = {
	saved_scopes = { layout }
	effect = {
		scope:layout = {
			every_in_list = {
				variable = POD_passive_charms
				custom_tooltip = POD_passive_charms_select_layout_contain
			}
			custom_tooltip = POD_passive_charms_select_layout_rest
		}
		every_in_list = {
			variable = charm_list
			var:charm_perk = { save_scope_as = charm }
			if = {
				limit = {
					var:activated = flag:no
					scope:layout = { 
						is_target_in_variable_list = { 
							name = POD_passive_charms 
							target = scope:charm
						} 
					}
				}
				set_variable = { name = activated value = flag:yes }
			}
			if = {
				limit = {
					var:activated = flag:yes
					scope:layout = { 
						NOT = {
							is_target_in_variable_list = { 
								name = POD_passive_charms 
								target = scope:charm
							} 
						}
					}
				}
				set_variable = { name = activated value = flag:no }
			}
		}
	}
}
