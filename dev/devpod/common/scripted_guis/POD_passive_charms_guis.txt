POD_passive_charms_innit = {
	effect = {
		POD_passive_charms_close_effect = yes # Safeguard
		POD_passive_charm_injector = { EFFECT = POD_passive_charms_innit_effect }
	}
}
POD_passive_charms_close = {
	effect = {
		POD_passive_charms_close_effect = yes
		remove_variable = POD_open_passive_charms_gui
	}
}

POD_passive_charms_select = {
	saved_scopes = { charm }
	is_shown = {
		scope:charm.var:activated = flag:yes
	}
	effect = {
		scope:charm = {
			switch = {
				trigger = var:activated
				flag:yes = { set_variable = { name = activated value = flag:no } }
				flag:no = { set_variable = { name = activated value = flag:yes } }
			}
		}
	}
}

POD_passive_chars_update = {
	effect = {
		POD_passive_charm_injector = { EFFECT = POD_passive_charms_update_effect }
	}
}
