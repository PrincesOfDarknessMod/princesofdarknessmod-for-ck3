# For debugging, reroll support
POD_umbra_expedition_reload = {
	scope = character
	effect = {
		POD_create_umbra_expedition = yes
	}
}

POD_umbra_expedition_is_current_location = {
	scope = character
	saved_scopes = {
		story
	}
	is_shown = {
		scope:story = var:umbra_map.var:current_location
	}
}

POD_umbra_expedition_is_start_node = {
	scope = character
	saved_scopes = {
		story
	}
	is_shown = {
		scope:story = var:umbra_map.var:start_node
	}
}

POD_umbra_expedition_line = {
	scope = character
	saved_scopes = {
		story
	}
	is_shown = {
		scope:story.var:start_node = var:umbra_map.var:current_location
	}
}

POD_umbra_expedition_node = {
	scope = character
	saved_scopes = {
		story
	}
	is_shown = {
		scope:story.var:include = yes
	}
	is_valid = {
		custom_tooltip = {
			text = out_of_reach
			OR = {
				var:umbra_map.var:start_node = scope:story
				var:umbra_map.var:current_location = {
					any_in_list = {
						variable = lines
						var:end_node = scope:story
					}
				}
			}
		}
	}
	effect = {
		if = {
			limit = {
				OR = {
					scope:story != var:umbra_map.var:current_location
					scope:story = var:umbra_map.var:start_node
				}
			}
			if = {
				# when you click on the start node, don't move there
				limit = {
					scope:story != var:umbra_map.var:start_node
				}
				var:umbra_map = {
					set_variable = {
						name = current_location
						value = scope:story
					}
				}
			}
			switch = {
				trigger = scope:story.var:type
				flag:node_end = {
					POD_umbra_end_effect = yes
				}
				flag:node_start = {
					POD_umbra_retreat_effect = yes
				}
				flag:node_event = {
					POD_umbra_event_effect = yes
				}
				flag:node_prowess = {
					POD_umbra_prowess_challenge_effect = yes
				}
				flag:node_elite = {
					POD_umbra_elite_challenge_effect = yes
				}
				flag:node_shop = {
					POD_umbra_shop_effect = yes
				}
				fallback = {
					POD_umbra_skill_challenge_effect = yes
				}
			}
		}
	}
}
