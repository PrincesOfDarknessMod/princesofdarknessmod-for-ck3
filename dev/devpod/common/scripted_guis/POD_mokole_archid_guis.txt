
POD_archid_previous_choice_button = {
	scope = character

	is_valid = {
		POD_has_archid_flag = { FLAG = scope:flag }
	}
}

POD_archid_current_choice_button = {
	scope = character

	is_valid = {
		is_target_in_variable_list = { name = CurrentArchidGuiOptions target = scope:flag }
	}
}

POD_archid_current_choice_toggle_button = {
	scope = character
	
	effect = {
		#No more toggle if you have 0 options left
		if = {
			limit = { 
				mokole_available_archid_GUI_options = 0
				NOT = { is_target_in_variable_list = { name = CurrentArchidGuiOptions target = scope:flag } }
			}
		}
		#Can only toggle off if you have 0 options left
		else_if = {
			limit = {
				mokole_available_archid_GUI_options = 0
				is_target_in_variable_list = { name = CurrentArchidGuiOptions target = scope:flag }
			}
			remove_list_variable = { name = CurrentArchidGuiOptions target = scope:flag }
		}
		#If you have options left and you never picked this before, enable it
		else_if = {
			limit = {
				mokole_available_archid_GUI_options > 0
				NOT = { POD_has_archid_flag = { FLAG = scope:flag } }
				NOT = { is_target_in_variable_list = { name = CurrentArchidGuiOptions target = scope:flag } }
			}
			add_to_variable_list = { name = CurrentArchidGuiOptions target = scope:flag }
		}
		#If you have options left and you never picked this before, disable it
		else_if = {
			limit = {
				mokole_available_archid_GUI_options > 0
				NOT = { POD_has_archid_flag = { FLAG = scope:flag } }
				is_target_in_variable_list = { name = CurrentArchidGuiOptions target = scope:flag }
			}
			remove_list_variable = { name = CurrentArchidGuiOptions target = scope:flag }
		}
	}
}
