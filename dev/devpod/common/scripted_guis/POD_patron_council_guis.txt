patron_council_enable = {
	scope = character
	saved_scopes = { player }

	is_shown = {
		scope:player ?= { POD_is_patron_trigger = yes }
		has_global_variable = patron_council_enabled
	}
	is_valid = { 
		scope:player = { NOT = { faith = { has_doctrine = special_doctrine_fae_militants_hostility } } }
	}
	effect = {}
}

austere_conclave_enabled = {
	scope = character

	is_shown = {
	}
	is_valid = {
		POD_is_austere_conclave_trigger = yes
	}
	effect = {
	}
}

patron_council_edict_dreaming_progress = {
	scope = character
	is_shown = { }
	is_valid = {
		custom_description = {
			text = number_of_patron_council_influence
			value = 1000
			patron_council_zeal >= 1000
		}
	}
	effect = {
		custom_description = {
			text = remove_patron_council_influence
			value = 1000
			change_global_variable = {
				name = patron_council_influence
				add = -1000
			}
		}
		save_scope_as = player
		custom_tooltip = {
			text = patron_council_edict_dreaming_progress_effect_tooltip
			if = {
				limit = {
					has_variable = umbra_stationed_realm
					var:umbra_stationed_realm.var:type = flag:dreaming
				} 
				POD_gain_umbra_progress_effect = { VICTIM = root VALUE = 100 }
			}
			every_relation = {
				type = umbra_explorer
				limit = { var:umbra_stationed_realm.var:type = flag:dreaming }
				scope:player = { POD_gain_umbra_progress_effect = { VICTIM = prev VALUE = 100 } }
			}
		}
	}
}
