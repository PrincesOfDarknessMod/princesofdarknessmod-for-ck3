POD_journey_requirements = {
	scope = character
	saved_scopes = { story }
	
	is_shown = {
		NOR = {
			var:current_journey ?= {
				var:journey_type = scope:story.var:journey_type
			}
			any_in_list = {
				variable = paused_journeys
				count > 0
				var:journey_type = scope:story.var:journey_type
			}
		}
	}
	
	is_valid = {
		POD_can_start_planning_phase_trigger = yes
		
		trigger_if = {
			limit = { scope:story = { has_variable = gold_cost } } # do not check for paused journeys
			
			custom_tooltip = {
				text = POD_journey_requirements_gold
				OR = {
					scope:story.var:gold_cost = 0
					gold >= scope:story.var:gold_cost
				}
			}
			custom_tooltip = {
				text = POD_journey_requirements_prestige
				save_temporary_scope_value_as = {
					name = adjusted_prestige_cost
					value = {
						value = scope:story.var:prestige_cost
						if = {
							limit = { scope:story = { POD_is_seek_knowledge_journey_trigger = yes } }
							add = seek_knowledge_added_cost
						}
					}
				}
				OR = {
					scope:adjusted_prestige_cost = 0
					prestige >= scope:adjusted_prestige_cost
				}
			}
			custom_tooltip = {
				text = POD_journey_requirements_piety
				OR = {
					scope:story.var:piety_cost = 0
					piety >= scope:story.var:piety_cost
				}
			}
		}

		# Safeguards for targeted Journeys
		trigger_if = {
			limit = { scope:story = { has_variable = target } }
			custom_tooltip = {
				text = POD_journey_requirements_target_dead
				scope:story.var:target = { is_alive = yes }
			}
		}

		trigger_if = {
			limit = { scope:story = { has_variable = target_artifact } }
			custom_tooltip = {
				text = POD_journey_requirements_target_artifact_destroyed
				scope:story = { exists = var:target_artifact }
			}
		}
	}
}

POD_journey_on_start = {
	scope = character
	saved_scopes = { story }
	effect = {
		# Pay upfront costs
		remove_short_term_gold = scope:story.var:gold_cost
		if = {
			limit = { scope:story = { POD_is_seek_knowledge_journey_trigger = yes } }
			add_prestige = {
				value = 0
				subtract = scope:story.var:prestige_cost
				subtract = seek_knowledge_added_cost
			}
		}
		else = {
			add_prestige = { value = 0 subtract = scope:story.var:prestige_cost }
		}
		add_piety    = { value = 0 subtract = scope:story.var:piety_cost }
		
		remove_variable = journey_detail_window
		
		# hunt_in_capital has a separate effect to determine costs based on the type of mortal
		if = {
			limit = { scope:story.var:journey_type = flag:hunt_in_capital }
			POD_hunt_in_capital_cost_effect = yes
		}

		if = {
			limit = { scope:story = { POD_is_seek_knowledge_journey_trigger = yes } }
			POD_increment_seek_knowledge_cost_effect = yes
		}
		
		if = {
			limit = { scope:story.var:journey_type = flag:fera_spirit_quest }
			trigger_event = { id = POD_fera_spirit_quest.1 }
		}
		else = {
			POD_start_journey_from_template_effect = {
				TEMPLATE = scope:story
			}
		}
	}
}
