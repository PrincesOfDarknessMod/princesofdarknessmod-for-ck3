pod_combat_execute_move = {
	scope = character
	# root = clicking player; actor = control character. Checks control and costs separately.
	saved_scopes = { story actor }

	is_shown = {
		exists = scope:story
		pod_combat_is_active_trigger = yes
		NOT = { pod_combat_is_over_trigger = yes }
	}

	is_valid = {
		# Rejects unavailable commands before evaluating the per-move validator against the current workspace.
		trigger_if = {
			limit = { pod_combat_is_active_trigger = yes }
			trigger_if = {
				limit = {
					NOT = { pod_combat_is_over_trigger = yes }
					pod_combat_is_my_turn_trigger = yes
				}
				trigger_if = {
					# KO blocks move commands only. End Turn must remain usable
					limit = { NOT = { pod_combat_my_slot_is_ko_trigger = yes } }
					trigger_if = {
						limit = {
							exists = scope:story
							exists = scope:actor
						}
						# The validator splices ACTOR into scope switches, so pass the saved
						# combatant scope, while control above stays on the clicking player.
						POD_can_execute_combat_move_trigger = {
							STORY = scope:story
							ACTOR = scope:actor
						}
					}
					trigger_else = { always = no }
				}
				trigger_else = { always = no }
			}
			trigger_else = { always = no }
		}
		trigger_else = { always = no }
	}

	effect = {
		POD_execute_combat_move = { STORY = scope:story }
		pod_combat_save_workspace_effect = yes
	}
}

# Render-only via IsValidTooltip; never bind IsValid or Execute to this SGUI.
# Order: turn ownership, resources, move state.
# LOCKSTEP: mirror player-facing blocks in pod_combat_move_dynamic_gates_trigger
# and its wrappers. Window-lifetime, AI-only and scope-resolution gates need no text.
pod_combat_move_usability_tt = {
	scope = character
	# Required scopes match pod_combat_execute_move: root = client, actor = driven combatant.
	saved_scopes = { story actor }

	is_shown = { always = yes }

	is_valid = {
		# Show turn-level failures alone; per-move checks would read another actor's workspace.
		# Check is_over first: victory leaves both the window and the player's turn latch live.
		trigger_if = {
			limit = { pod_combat_is_over_trigger = yes }
			custom_tooltip = { text = pod_combat_usable_fight_over_tt  always = no }
		}
		trigger_else_if = {
			limit = { NOT = { pod_combat_is_my_turn_trigger = yes } }
			custom_tooltip = { text = pod_combat_usable_not_your_turn_tt  always = no }
		}
		trigger_else_if = {
			limit = { pod_combat_my_slot_is_ko_trigger = yes }
			custom_tooltip = { text = pod_combat_usable_actor_down_tt  always = no }
		}
		trigger_else = {
			# Has_variable checks do not protect optional reads.
			trigger_if = {
				limit = { scope:story = { has_variable = ap_cost } }
				trigger_if = {
					limit = { NOT = { pod_combat_can_afford_ap_trigger = { COST = scope:story.var:ap_cost } } }
					custom_tooltip = { text = pod_combat_usable_need_ap_tt  always = no }
				}
			}

			# Check energy on the controlled charactor. PoD already has its own splat-specific failure text;
			# the inverted limit suppresses successful checks.
			trigger_if = {
				limit = { scope:story = { has_variable = energy_cost } }
				trigger_if = {
					limit = {
						NOT = {
							scope:actor = {
								POD_has_enough_splat_agnostic_energy_reserves = { AMOUNT = scope:story.var:energy_cost }
							}
						}
					}
					scope:actor = {
						POD_has_enough_splat_agnostic_energy_reserves = { AMOUNT = scope:story.var:energy_cost }
					}
				}
			}

			# CK3 currencies are PC-only, matching the live gates' player-turn limits.
			trigger_if = {
				limit = {
					scope:story = { has_variable = gold_cost }
					pod_combat_is_player_turn_trigger = yes
				}
				trigger_if = {
					limit = { NOT = { pod_combat_can_afford_move_gold_trigger = { STORY = scope:story } } }
					custom_tooltip = { text = pod_combat_usable_need_gold_tt  always = no }
				}
			}

			trigger_if = {
				limit = {
					scope:story = { has_variable = piety_cost }
					pod_combat_is_player_turn_trigger = yes
				}
				trigger_if = {
					limit = { NOT = { pod_combat_can_afford_move_piety_trigger = { STORY = scope:story } } }
					custom_tooltip = { text = pod_combat_usable_need_piety_tt  always = no }
				}
			}

			trigger_if = {
				limit = {
					scope:story = { has_variable = prestige_cost }
					pod_combat_is_player_turn_trigger = yes
				}
				trigger_if = {
					limit = { NOT = { pod_combat_can_afford_move_prestige_trigger = { STORY = scope:story } } }
					custom_tooltip = { text = pod_combat_usable_need_prestige_tt  always = no }
				}
			}

			# LOCKSTEP with pod_combat_actor_has_used_move_trigger: non-round means whole fight.
			trigger_if = {
				limit = { scope:story = { has_variable = once_per } }
				trigger_if = {
					limit = { pod_combat_actor_has_used_move_trigger = { STORY = scope:story  ACTOR = scope:actor } }
					trigger_if = {
						limit = { scope:story = { var:once_per = flag:round } }
						custom_tooltip = { text = pod_combat_usable_used_this_round_tt  always = no }
					}
					trigger_else = {
						custom_tooltip = { text = pod_combat_usable_used_this_combat_tt  always = no }
					}
				}
			}

			# Targeting - one presence gate covering all three arms.
			trigger_if = {
				limit = { scope:story = { has_variable = targeting } }
				trigger_if = {
					limit = {
						pod_combat_is_player_turn_trigger = yes
						scope:story = { var:targeting = flag:single_enemy }
						NOT = { pod_combat_player_has_enemy_target_trigger = yes }
					}
					custom_tooltip = { text = pod_combat_targeting_need_enemy_tt  always = no }
				}
				trigger_if = {
					limit = {
						pod_combat_is_player_turn_trigger = yes
						scope:story = { var:targeting = flag:single_ally }
						NOT = { pod_combat_player_has_ally_target_trigger = yes }
					}
					custom_tooltip = { text = pod_combat_targeting_need_ally_tt  always = no }
				}
				trigger_if = {
					limit = {
						pod_combat_is_player_turn_trigger = yes
						scope:story = { var:targeting = flag:single_pc }
						NOT = { pod_combat_pc_slot_alive_trigger = yes }
					}
					custom_tooltip = { text = pod_combat_targeting_need_pc_alive_tt  always = no }
				}
			}

			# The stat card already has the condition; guards both fields before dispatch.
			trigger_if = {
				limit = { scope:story = { has_variable = cond_mode  has_variable = cond_predicate } }
				trigger_if = {
					limit = {
						scope:story = { var:cond_mode = flag:gate }
						NOT = {
							pod_combat_move_condition_met_trigger = {
								STORY = scope:story ACTOR = scope:actor PREDICATE = scope:story.var:cond_predicate
							}
						}
					}
					custom_tooltip = { text = pod_combat_usable_condition_unmet_tt  always = no }
				}
			}

			# The heal and redundancy checks carry their own presence guards
			trigger_if = {
				limit = {
					pod_combat_move_is_self_only_heal_trigger = { STORY = scope:story }
					NOT = { pod_combat_self_heal_has_effect_trigger = { STORY = scope:story } }
				}
				custom_tooltip = { text = pod_combat_cant_heal_already_full_desc  always = no }
			}

			trigger_if = {
				limit = {
					NOT = { pod_combat_move_has_independent_payload_trigger = { STORY = scope:story } }
					NOT = { pod_combat_move_payload_useful_trigger = { STORY = scope:story } }
				}
				custom_tooltip = { text = pod_combat_move_redundant_tt  always = no }
			}

			trigger_if = {
				limit = { scope:story = { has_variable = summons_combatant } }
				trigger_if = {
					limit = { pod_combat_duel_bans_summons_trigger = yes }
					custom_tooltip = { text = pod_combat_cant_summon_duel_barred_tt  always = no }
				}
				trigger_else_if = {
					limit = {
						var:pod_combat_session ?= { has_variable = pod_combat_current_actor_team }
						NOT = { pod_combat_summon_capacity_available_trigger = { STORY = scope:story } }
					}
					custom_tooltip = { text = pod_combat_cant_summon_team_full_tt  always = no }
				}
				trigger_else = { always = yes }
			}

			# Fallback for a missed unlock bake; detailed requirements stay in the editor.
			trigger_if = {
				limit = { NOT = { pod_combat_actor_has_baked_unlock_trigger = { STORY = scope:story ACTOR = scope:actor } } }
				custom_tooltip = { text = pod_combat_usable_not_unlocked_tt  always = no }
			}
		}
	}
}

# Per-player auto-advance preference; apply changes to the current fight immediately.
# Presence means manual stepping; absence means auto-advance.
pod_combat_toggle_auto_advance = {
	scope = character

	is_shown = {
		pod_combat_is_active_trigger = yes
	}

	is_valid = {
		pod_combat_is_active_trigger = yes
	}

	effect = {
		if = {
			limit = { has_variable = pod_combat_manual_step_pref }
			remove_variable = pod_combat_manual_step_pref
		}
		else = {
			set_variable = { name = pod_combat_manual_step_pref value = yes }
		}
		# Re-seed the live fight's step_mode from the now-updated pref.
		if = {
			limit = { has_variable = pod_combat_manual_step_pref }
			var:pod_combat_session ?= { set_variable = { name = pod_combat_step_mode value = yes } }
		}
		else = {
			var:pod_combat_session ?= { remove_variable = pod_combat_step_mode }
		}
	}
}

# Advance does not commit the controlled slot or save its workspace.
# Only pod_combat_process_group_turns_effect owns that.
# Respects current mode so a toggle to auto takes effect on the next click.
pod_combat_advance_turn = {
	scope = character

	is_shown = {
		pod_combat_is_active_trigger = yes
	}

	is_valid = {
		pod_combat_can_advance_trigger = yes
	}

	effect = {
		if = {
			limit = { var:pod_combat_session ?= { has_variable = pod_combat_step_mode } }
			pod_combat_advance_one_ai_turn_effect = yes
			if = {
				limit = { pod_combat_pc_has_acted_this_round_trigger = yes }
				pod_combat_after_step_settle_effect = yes
			}
			else = {
				pod_combat_after_pre_pc_step_settle_effect = yes
			}
		}
		else = {
			# Keep pre-/post-controlled-turn routes separate: re-running setup on a live
			# turn would reset AP and clear round buffs.
			if = {
				limit = { pod_combat_pc_has_acted_this_round_trigger = yes }
				pod_combat_run_remaining_ai_turns_effect = yes
				if = {
					limit = {
						NOT = { pod_combat_current_actor_is_controlled_trigger = yes }
						NOT = { pod_combat_is_over_trigger = yes }
					}
					pod_combat_group_end_round_effect = yes
				}
			}
			else = {
				pod_combat_run_pre_pc_ai_turns_effect = yes
			}
		}
	}
}

# is_shown supplies the equipped checkbox state; unequipping remains allowed at capacity.
pod_combat_loadout_toggle = {
	scope = character
	saved_scopes = { story }

	is_shown = {
		exists = scope:story
		is_target_in_variable_list = {
			name   = pod_combat_loadout
			target = scope:story
		}
	}

	is_valid = {
		exists = scope:story
		OR = {
			# Equipped -> unequip depends ONLY on current loadout membership.
			is_target_in_variable_list = {
				name   = pod_combat_loadout
				target = scope:story
			}
			# Not equipped -> require the move be UNLOCKED and its category have room.
			AND = {
				# Use the unlock trigger: can_execute_decision implicitly requires a ruler
				# and rejects unlanded players.
				pod_combat_actor_unlocks_move_trigger = { STORY = scope:story ACTOR = root }
				pod_combat_loadout_room_left_trigger = yes
				# Cap check per combat_category. custom_description surfaces a friendly
				# "Attack slots full: 4/4" tooltip; the trigger_if chain keeps it singular.
				trigger_if = {
					limit = { scope:story = { var:combat_category = flag:ATK } }
					custom_description = {
						text = pod_combat_loadout_atk_full_tt
						pod_combat_atk_slots_free_trigger = yes
					}
				}
				trigger_else_if = {
					limit = { scope:story = { var:combat_category = flag:DEF } }
					custom_description = {
						text = pod_combat_loadout_def_full_tt
						pod_combat_def_slots_free_trigger = yes
					}
				}
				trigger_else_if = {
					limit = { scope:story = { var:combat_category = flag:CTL } }
					custom_description = {
						text = pod_combat_loadout_ctl_full_tt
						pod_combat_ctl_slots_free_trigger = yes
					}
				}
				trigger_else_if = {
					limit = { scope:story = { var:combat_category = flag:PWR } }
					custom_description = {
						text = pod_combat_loadout_pwr_full_tt
						pod_combat_pwr_slots_free_trigger = yes
					}
				}
				trigger_else = {
					# Required terminator for the trigger_else_if chain.
					# always = no = don't equip an unknown-category move.
					always = no
				}
			}
		}
	}

	effect = {
		if = {
			limit = {
				is_target_in_variable_list = {
					name   = pod_combat_loadout
					target = scope:story
				}
			}
			remove_list_variable = {
				name   = pod_combat_loadout
				target = scope:story
			}
		}
		else = {
			add_to_variable_list = {
				name   = pod_combat_loadout
				target = scope:story
			}
		}
		# Keep the loadout-count display cache in step with the change so the pip
		# strips stay correct without re-walking the loadout per frame.
		pod_combat_refresh_loadout_counts_effect = yes
		# Equipping under the sheet's equipped-only filter changes the row
		# set itself, so rebuild; other filters are equip-agnostic.
		if = {
			limit = {
				has_variable = pod_combat_sheet_view
				has_variable = pod_combat_sheet_f_equipped
			}
			pod_combat_sheet_rebuild_effect = yes
		}
	}
}

# Row-capacity gate uses baked unlocks. Do not bind toggle.IsValid here:

	saved_scopes = { story }

				}

					limit = { scope:story = { var:combat_category = flag:DEF } }
					pod_combat_def_slots_free_trigger = yes
				}
				trigger_else_if = {
					limit = { scope:story = { var:combat_category = flag:CTL } }
					pod_combat_ctl_slots_free_trigger = yes
				}
				trigger_else_if = {
					limit = { scope:story = { var:combat_category = flag:PWR } }
					pod_combat_pwr_slots_free_trigger = yes
				}
				trigger_else = { always = no }
			}
			trigger_else = { always = no }
		}
	}
}

# Render-only capacity text for unlocked, unequipped rows.
# Locked rows show unlock requirements; equipped rows can always be removed.
pod_combat_loadout_block_tt = {
	scope = character
	saved_scopes = { story }

	is_shown = { always = yes }

	is_valid = {
		trigger_if = {
			limit = {
				exists = scope:story
				NOT = {
					is_target_in_variable_list = {
						name   = pod_combat_loadout
						target = scope:story
					}
				}
				root = {
					is_target_in_variable_list = {
						name   = pod_combat_unlocked_moves
						target = scope:story
					}
				}
			}
			# Category first; the independent total cap matters when caps are retuned.
			trigger_if = {
				limit = { scope:story = { has_variable = combat_category } }
				trigger_if = {
					limit = {
						scope:story = { var:combat_category = flag:ATK }
						NOT = { pod_combat_atk_slots_free_trigger = yes }
					}
					custom_tooltip = { text = pod_combat_loadout_atk_full_tt  always = no }
				}
				trigger_else_if = {
					limit = {
						scope:story = { var:combat_category = flag:DEF }
						NOT = { pod_combat_def_slots_free_trigger = yes }
					}
					custom_tooltip = { text = pod_combat_loadout_def_full_tt  always = no }
				}
				trigger_else_if = {
					limit = {
						scope:story = { var:combat_category = flag:CTL }
						NOT = { pod_combat_ctl_slots_free_trigger = yes }
					}
					custom_tooltip = { text = pod_combat_loadout_ctl_full_tt  always = no }
				}
				trigger_else_if = {
					limit = {
						scope:story = { var:combat_category = flag:PWR }
						NOT = { pod_combat_pwr_slots_free_trigger = yes }
					}
					custom_tooltip = { text = pod_combat_loadout_pwr_full_tt  always = no }
				}
				trigger_else = {
					# Required terminator (CK3 rejects an open trigger_else_if chain at
					# load). Nothing to say: this row's category has room.
					always = yes
				}
			}
			trigger_if = {
				limit = { NOT = { pod_combat_loadout_room_left_trigger = yes } }
				custom_tooltip = { text = pod_combat_loadout_total_full_tt  always = no }
			}
		}
	}
}

# Per-row visibility uses editor-open baked unlocks. Show-all adds own-splat moves;
# show-other-splats adds the full catalogue. Neither changes equip eligibility.
pod_combat_loadout_option = {
	scope = character
	saved_scopes = { story }

	is_shown = {
		exists = scope:story
		OR = {
			root = { is_target_in_variable_list = { name = pod_combat_unlocked_moves target = scope:story } }
			AND = {
				has_variable = pod_combat_show_all_moves
				OR = {
					has_variable = pod_combat_show_other_splats
					scope:story = {
						pod_combat_move_group_matches_actor_trigger = { ACTOR = root }
					}
				}
			}
		}
	}

	is_valid = {
		exists = scope:story
		root = { is_target_in_variable_list = { name = pod_combat_unlocked_moves target = scope:story } }
	}
}

		}
	}
}

# Combined filter check: yes if no filter is active, or the move's
# combat_category matches an active filter.
pod_combat_loadout_category_match = {
	scope = character
	saved_scopes = { story }

	is_shown = {
		exists = scope:story
		OR = {
			# No filter active - show everything.
			AND = {
				NOT = { has_variable = pod_combat_filter_atk }
				NOT = { has_variable = pod_combat_filter_def }
				NOT = { has_variable = pod_combat_filter_ctl }
				NOT = { has_variable = pod_combat_filter_pwr }
			}
			# Or filter active and matches.
			AND = {
				has_variable = pod_combat_filter_atk
				scope:story = { var:combat_category = flag:ATK }
			}
			AND = {
				has_variable = pod_combat_filter_def
				scope:story = { var:combat_category = flag:DEF }
			}
			AND = {
				has_variable = pod_combat_filter_ctl
				scope:story = { var:combat_category = flag:CTL }
			}
			AND = {
				has_variable = pod_combat_filter_pwr
				scope:story = { var:combat_category = flag:PWR }
			}
		}
	}
}

					scope:story = {
						has_variable = self_damage_condition
						var:self_damage_condition = flag:on_miss
					}
				}
				custom_tooltip = pod_combat_desc_self_damage_on_miss
			}
			else = {
				custom_tooltip = pod_combat_desc_self_damage
			}
		}

		# Standalone healing; drains are included in the damage line.
		if = {
			limit = {
				scope:story = {
					has_variable = hp_heal
					NOT = { has_variable = drain }
				}
			}
			custom_tooltip = pod_combat_desc_hp_heal
		}
		# Standalone drain (no hp_damage to inline onto)
		if = {
			limit = {
				scope:story = {
					has_variable = drain
					NOT = { has_variable = hp_damage }
				}
			}
			custom_tooltip = pod_combat_desc_drain
		}
		if = {
			limit = { scope:story = { has_variable = morale_heal } }
			custom_tooltip = pod_combat_desc_morale_heal
		}
		# Restore display scales the agnostic amount to the viewer's resource.
		if = {
			limit = { scope:story = { has_variable = energy_gain } }
			custom_tooltip = pod_combat_desc_energy_gain
		}

		# Missing capability flags describe damage moves only.
		if = {
			limit = {
				scope:story = {
					has_variable = hp_damage
					NOT = { has_variable = requires_hit_roll }
				}
			}
			custom_tooltip = pod_combat_desc_always_hits
		}
		if = {
			limit = {
				scope:story = {
					has_variable = hp_damage
					NOT = { has_variable = checks_guard }
				}
			}
			custom_tooltip = pod_combat_desc_bypass_guard
		}
		if = {
			limit = {
				scope:story = {
					has_variable = hp_damage
					NOT = { has_variable = checks_stance }
				}
			}
			custom_tooltip = pod_combat_desc_undodgeable
		}
		if = {
			limit = {
				scope:story = {
					has_variable = hp_damage
					NOT = { has_variable = checks_spirit_gate }
				}
			}
			custom_tooltip = pod_combat_desc_hits_spirits
		}
		# Viewer-relative weapon display. Save pod_weapon before tooltips that read it;
		# match pod_combat_actor_is_armed_trigger so the armed branch always has a scope.
		if = {
			limit = { scope:story = { has_variable = uses_weapon } }
			if = {
				limit = { pod_combat_actor_is_armed_trigger = yes }
				random_equipped_character_artifact = {
					limit = {
						artifact_slot_type = primary_armament
						prev = { can_benefit_from_artifact = prev }
					}
					save_scope_as = pod_weapon
				}
				if = {
					limit = { scope:pod_weapon = { POD_weapon_artifact_has_override_trigger = yes } }
					custom_tooltip = pod_combat_desc_weapon_empowered
				}
				else = {
					custom_tooltip = pod_combat_desc_uses_weapon_armed
				}
			}
			else = {
				custom_tooltip = pod_combat_desc_uses_weapon
			}
		}

		# Strip flags consume stances before damage; Undodgeable only ignores them.
		if = {
			limit = { scope:story = { has_variable = bypass_dodge } }
			custom_tooltip = pod_combat_desc_bypass_dodge
		}
		if = {
			limit = { scope:story = { has_variable = bypass_stealth } }
			custom_tooltip = pod_combat_desc_bypass_stealth
		}

		# Match runtime buff recipients: ally targets use ally wording; all others use self.
		# Debuffs always use target wording.
		if = {
			limit = { scope:story = { has_variable = soak_buff_round_tier } }
			if = {
				limit = { scope:story = { pod_combat_move_targets_ally_trigger = yes } }
				custom_tooltip = pod_combat_desc_soak_buff_round_other
			}
			else = { custom_tooltip = pod_combat_desc_soak_buff_round }
		}
		if = {
			limit = { scope:story = { has_variable = soak_buff_combat_tier } }
			if = {
				limit = { scope:story = { pod_combat_move_targets_ally_trigger = yes } }
				custom_tooltip = pod_combat_desc_soak_buff_combat_other
			}
			else = { custom_tooltip = pod_combat_desc_soak_buff_combat }
		}
		if = {
			limit = { scope:story = { has_variable = soak_debuff_round_tier } }
			custom_tooltip = pod_combat_desc_soak_debuff_round
		}
		if = {
			limit = { scope:story = { has_variable = soak_debuff_combat_tier } }
			custom_tooltip = pod_combat_desc_soak_debuff_combat
		}
		if = {
			limit = { scope:story = { has_variable = hit_buff_round_tier } }
			if = {
				limit = { scope:story = { pod_combat_move_targets_ally_trigger = yes } }
				custom_tooltip = pod_combat_desc_hit_buff_round_other
			}
			else = { custom_tooltip = pod_combat_desc_hit_buff_round }
		}
		if = {
			limit = { scope:story = { has_variable = hit_buff_combat_tier } }
			if = {
				limit = { scope:story = { pod_combat_move_targets_ally_trigger = yes } }
				custom_tooltip = pod_combat_desc_hit_buff_combat_other
			}
			else = { custom_tooltip = pod_combat_desc_hit_buff_combat }
		}
		if = {
			limit = { scope:story = { has_variable = hit_debuff_round_tier } }
			custom_tooltip = pod_combat_desc_hit_debuff_round
		}
		if = {
			limit = { scope:story = { has_variable = hit_debuff_combat_tier } }
			custom_tooltip = pod_combat_desc_hit_debuff_combat
		}

		if = {
			limit = { scope:story = { has_variable = set_guard } }
			if = {
				limit = { scope:story = { pod_combat_move_targets_ally_trigger = yes } }
				custom_tooltip = pod_combat_desc_set_guard_other
			}
			else = { custom_tooltip = pod_combat_desc_set_guard }
		}
		if = {
			limit = { scope:story = { has_variable = dodge } }
			if = {
				limit = { scope:story = { pod_combat_move_targets_ally_trigger = yes } }
				custom_tooltip = pod_combat_desc_dodge_other
			}
			else = { custom_tooltip = pod_combat_desc_dodge }
		}
		if = {
			limit = { scope:story = { has_variable = stealth } }
			if = {
				limit = { scope:story = { pod_combat_move_targets_ally_trigger = yes } }
				custom_tooltip = pod_combat_desc_stealth_other
			}
			else = { custom_tooltip = pod_combat_desc_stealth }
		}
		if = {
			limit = { scope:story = { has_variable = remove_guard } }
			custom_tooltip = pod_combat_desc_remove_guard
		}

		# Match retaliation arming: explicit amount > hp_damage > 1;
		# kind: morale > retaliate_agg > move element > superficial.
		# Elemental classification guards damage_type reads.
		if = {
			limit = { scope:story = { has_variable = retaliate_if_hit } }
			if = {
				limit = { scope:story = { has_variable = retaliate_damage } }
				if = {
					limit = { scope:story = { has_variable = retaliate_morale } }
					custom_tooltip = pod_combat_desc_retaliate_if_hit_explicit_morale
				}
				else_if = {
					limit = { scope:story = { has_variable = retaliate_agg } }
					custom_tooltip = pod_combat_desc_retaliate_if_hit_explicit_agg
				}
				else_if = {
					limit = { scope:story = { pod_combat_move_class_is_elemental_trigger = yes } }
					custom_tooltip = pod_combat_desc_retaliate_if_hit_explicit_elemental
				}
				else = {
					custom_tooltip = pod_combat_desc_retaliate_if_hit_explicit_sup
				}
			}
			else_if = {
				limit = { scope:story = { has_variable = hp_damage } }
				if = {
					limit = { scope:story = { has_variable = retaliate_agg } }
					custom_tooltip = pod_combat_desc_retaliate_if_hit_reuse_agg
				}
				else = {
					custom_tooltip = pod_combat_desc_retaliate_if_hit_reuse_sup
				}
			}
			else = {
				custom_tooltip = pod_combat_desc_retaliate_if_hit
			}
		}
		if = {
			limit = { scope:story = { has_variable = retaliate_if_missed } }
			if = {
				limit = { scope:story = { has_variable = retaliate_damage } }
				if = {
					limit = { scope:story = { has_variable = retaliate_morale } }
					custom_tooltip = pod_combat_desc_retaliate_if_missed_explicit_morale
				}
				else_if = {
					limit = { scope:story = { has_variable = retaliate_agg } }
					custom_tooltip = pod_combat_desc_retaliate_if_missed_explicit_agg
				}
				else_if = {
					limit = { scope:story = { pod_combat_move_class_is_elemental_trigger = yes } }
					custom_tooltip = pod_combat_desc_retaliate_if_missed_explicit_elemental
				}
				else = {
					custom_tooltip = pod_combat_desc_retaliate_if_missed_explicit_sup
				}
			}
			else_if = {
				limit = { scope:story = { has_variable = hp_damage } }
				if = {
					limit = { scope:story = { has_variable = retaliate_agg } }
					custom_tooltip = pod_combat_desc_retaliate_if_missed_reuse_agg
				}
				else = {
					custom_tooltip = pod_combat_desc_retaliate_if_missed_reuse_sup
				}
			}
			else = {
				custom_tooltip = pod_combat_desc_retaliate_if_missed
			}
		}
		if = {
			limit = { scope:story = { has_variable = retaliate_if_attacked } }
			if = {
				limit = { scope:story = { has_variable = retaliate_damage } }
				if = {
					limit = { scope:story = { has_variable = retaliate_morale } }
					custom_tooltip = pod_combat_desc_retaliate_if_attacked_explicit_morale
				}
				else_if = {
					limit = { scope:story = { has_variable = retaliate_agg } }
					custom_tooltip = pod_combat_desc_retaliate_if_attacked_explicit_agg
				}
				else_if = {
					limit = { scope:story = { pod_combat_move_class_is_elemental_trigger = yes } }
					custom_tooltip = pod_combat_desc_retaliate_if_attacked_explicit_elemental
				}
				else = {
					custom_tooltip = pod_combat_desc_retaliate_if_attacked_explicit_sup
				}
			}
			else_if = {
				limit = { scope:story = { has_variable = hp_damage } }
				if = {
					limit = { scope:story = { has_variable = retaliate_agg } }
					custom_tooltip = pod_combat_desc_retaliate_if_attacked_reuse_agg
				}
				else = {
					custom_tooltip = pod_combat_desc_retaliate_if_attacked_reuse_sup
				}
			}
			else = {
				custom_tooltip = pod_combat_desc_retaliate_if_attacked
			}
		}

		# AP / initiative parametric
		if = {
			limit = { scope:story = { has_variable = ap_gain } }
			custom_tooltip = pod_combat_desc_ap_gain
		}
		if = {
			limit = { scope:story = { has_variable = ap_drain } }
			custom_tooltip = pod_combat_desc_ap_drain
		}
		if = {
			limit = { scope:story = { has_variable = initiative_debuff } }
			custom_tooltip = pod_combat_desc_initiative_debuff
		}
		if = {
			limit = { scope:story = { has_variable = dot_rounds } }
			custom_tooltip = pod_combat_desc_dot
		}
		if = {
			limit = { scope:story = { has_variable = mind_control } }
			custom_tooltip = pod_combat_desc_mind_control
		}
		if = {
			limit = { scope:story = { has_variable = summons_combatant } }
			custom_tooltip = pod_combat_desc_summon
		}
		if = {
			limit = { scope:story = { has_variable = marker_modifier } }
			custom_tooltip = pod_combat_desc_marker_modifier
		}
		# Strip before application, matching execution order
		# (12b-marker runs before 12c).
		if = {
			limit = { scope:story = { has_variable = remove_marker } }
			if = {
				limit = { scope:story = { var:remove_marker_target = flag:target } }
				custom_tooltip = pod_combat_desc_remove_marker_target
			}
			else = { custom_tooltip = pod_combat_desc_remove_marker }
		}

		# Render every bonus clause from its own story.
		if = {
			limit = { scope:story = { has_variable_list = bonus_clauses } }
			# Iterate on the move, emit on the viewer for perk/weapon reads.
			# Bind each clause as pod_clause; rebinding story persists beyond the loop.
			scope:story = {
				every_in_list = {
					variable = bonus_clauses
					save_scope_as = pod_clause
					root = { pod_combat_card_clause_line_effect = yes }
				}
			}
		}
		}
	}
}
