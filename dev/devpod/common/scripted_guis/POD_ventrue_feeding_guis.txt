# Root is the target, need to supply the player as scope:actor
is_ventrue_clan_curse_applicable_gui = {
	scope = character
	is_shown = {
		is_alive = yes
		POD_is_not_apocalyptic_background_character = yes
		
		OR = {
			POD_can_be_herd_trigger = yes
			POD_is_vampire_or_abomination_trigger = yes
		}
		
		scope:actor ?= {
			is_ventrue_trigger = yes
		}
	}
}
