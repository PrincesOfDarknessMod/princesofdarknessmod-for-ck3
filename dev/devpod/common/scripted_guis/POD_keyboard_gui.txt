POD_keyboard_press_key = {
	is_valid = {
		#var:keyboard_entry_5 = flag:__
		always = yes
	}
	effect = {
		# TODO: move conditions over into a scripted_trigger and turn this into a switch
		if = {
			limit = { scope:letter = flag:BKSP }

			if = {
				limit = { NOT = { var:keyboard_entry_5 = flag:__ } }
				set_variable = { name = keyboard_entry_5 value = flag:__ }
			}
			else_if = {
				limit = { NOT = { var:keyboard_entry_4 = flag:__ } }
				set_variable = { name = keyboard_entry_4 value = flag:__ }
			}
			else_if = {
				limit = { NOT = { var:keyboard_entry_3 = flag:__ } }
				set_variable = { name = keyboard_entry_3 value = flag:__ }
			}
			else_if = {
				limit = { NOT = { var:keyboard_entry_2 = flag:__ } }
				set_variable = { name = keyboard_entry_2 value = flag:__ }
			}
			else_if = {
				limit = { NOT = { var:keyboard_entry_1 = flag:__ } }
				set_variable = { name = keyboard_entry_1 value = flag:__ }
			}
		}
		else_if = {
			limit = { NOT = { var:keyboard_entry_5 = flag:__ } }
			custom_tooltip = "Can't enter more letters"
		}
		else_if = {
			limit = { NOT = { var:keyboard_entry_4 = flag:__ } }
			set_variable = { name = keyboard_entry_5 value = scope:letter }
		}
		else_if = {
			limit = { NOT = { var:keyboard_entry_3 = flag:__ } }
			set_variable = { name = keyboard_entry_4 value = scope:letter }
		}
		else_if = {
			limit = { NOT = { var:keyboard_entry_2 = flag:__ } }
			set_variable = { name = keyboard_entry_3 value = scope:letter }
		}
		else_if = {
			limit = { NOT = { var:keyboard_entry_1 = flag:__ } }
			set_variable = { name = keyboard_entry_2 value = scope:letter }
		}
		else = {
			set_variable = { name = keyboard_entry_1 value = scope:letter }
		}
	}
}

POD_keyboard_preview_script = {
	effect = {
		POD_keyboard_construct_preview_script = yes
	}
}
