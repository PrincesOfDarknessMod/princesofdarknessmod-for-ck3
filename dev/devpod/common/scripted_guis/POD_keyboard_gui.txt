POD_keyboard_press_key = {
	is_valid = {
		always = yes
	}
	effect = {
		if = {
			limit = { scope:letter = flag:BKSP }

			if = {
				limit = { POD_keyboard_entry_set = { N = 5 } }
				set_variable = { name = keyboard_entry_5 value = flag:__ }
			}
			else_if = {
				limit = { POD_keyboard_entry_set = { N = 4 } }
				set_variable = { name = keyboard_entry_4 value = flag:__ }
			}
			else_if = {
				limit = { POD_keyboard_entry_set = { N = 3 } }
				set_variable = { name = keyboard_entry_3 value = flag:__ }
			}
			else_if = {
				limit = { POD_keyboard_entry_set = { N = 2 } }
				set_variable = { name = keyboard_entry_2 value = flag:__ }
			}
			else_if = {
				limit = { POD_keyboard_entry_set = { N = 1 } }
				set_variable = { name = keyboard_entry_1 value = flag:__ }
			}
		}
		else_if = {
			limit = { POD_keyboard_entry_set = { N = 5 } }
			custom_tooltip = POD_keyboard_max_char_count_hit
		}
		else_if = {
			limit = { POD_keyboard_entry_set = { N = 4 } }
			set_variable = { name = keyboard_entry_5 value = scope:letter }
		}
		else_if = {
			limit = { POD_keyboard_entry_set = { N = 3 } }
			set_variable = { name = keyboard_entry_4 value = scope:letter }
		}
		else_if = {
			limit = { POD_keyboard_entry_set = { N = 2 } }
			set_variable = { name = keyboard_entry_3 value = scope:letter }
		}
		else_if = {
			limit = { POD_keyboard_entry_set = { N = 1 } }
			set_variable = { name = keyboard_entry_2 value = scope:letter }
		}
		else = {
			set_variable = { name = keyboard_entry_1 value = scope:letter }
		}
	}
}

POD_keyboard_preview_script = {
	effect = {
		POD_keyboard_construct_preview_script = yes
	}
}

POD_keyboard_finish_script = {
	is_valid = {
		POD_keyboard_can_submit_answer = yes
	}
	effect = {
		if = {
			limit = { POD_keyboard_can_submit_answer = yes }

			custom_description_no_bullet = { text = POD_keyboard_can_submit_answer_ready }

			if = {
				limit = {
					var:wordle_minigame = {
						POD_wordle_green_check = { N = 1 L = scope:player.var:keyboard_entry_1 }
						POD_wordle_green_check = { N = 2 L = scope:player.var:keyboard_entry_2 }
						POD_wordle_green_check = { N = 3 L = scope:player.var:keyboard_entry_3 }
						POD_wordle_green_check = { N = 4 L = scope:player.var:keyboard_entry_4 }
						POD_wordle_green_check = { N = 5 L = scope:player.var:keyboard_entry_5 }
					}
				}
				POD_wordle_correct_guess_effect = yes
			}
			else = {
				POD_create_wordle_guess_keyboard = { PLAYER = scope:player }

				POD_wordle_incorrect_guess_effect = yes
			}
		}
	}
}
