blood_hunger_hint = {
	scope = character
	is_shown = {
		var:resource_hint_window ?= flag:blood
	}
	effect = {
		set_variable = {
			name  = resource_hint_window
			value = flag:blood
		}

		POD_clear_resource_hint_vars_effect = yes

		every_in_global_list = {
			variable = journey_GUI_list
			limit = { var:journey_type = flag:hunt_in_capital }
			prev = {
				set_variable = {
					name  = hunt_in_capital_story
					value = prev
				}
			}
		}
		ordered_relation = {
			type = herd_member
			limit = { drinkable_blood_left > 0 }
			order_by = drinkable_blood_left
			position = 0
			prev = {
				set_variable = {
					name  = suggested_herdmember
					value = prev
				}
			}
		}
	}
}

blood_hunger_hint_herd_mortal = {
	scope = character
	is_shown = {
		trigger_if = {
			limit = { is_vampire_trigger = yes }
			POD_has_potential_mortal_herd_trigger = yes
		}
		trigger_else = {
			always = no
		}
	}
	effect = {
		POD_save_potential_mortal_herd_effect = yes
		open_interaction_window = {
			interaction = POD_herdify_mortal_interaction
			actor = root
			recipient = scope:potential_mortal_herd
		}
	}
}
