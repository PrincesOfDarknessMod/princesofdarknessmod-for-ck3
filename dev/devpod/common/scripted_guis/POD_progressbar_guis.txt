blood_hunger_hint = {
	scope = character
	is_shown = {
		var:resource_hint_window ?= flag:blood
	}
	effect = {
		set_variable = {
			name  = resource_hint_window
			value = flag:blood
		}

		POD_clear_resource_hint_vars_effect = yes

		POD_find_journey_effect = {
			TYPE = hunt_in_capital
			SAVE_AS = hunt_in_capital_story
		}
		set_variable = {
			name  = hunt_in_capital_story
			value = scope:hunt_in_capital_story
		}
		ordered_relation = {
			type = herd_member
			limit = { drinkable_blood_left > 0 }
			order_by = drinkable_blood_left
			position = 0
			prev = {
				set_variable = {
					name  = suggested_herdmember
					value = prev
				}
			}
		}

		ordered_prisoner = {
			limit = { 
				POD_is_vampire_or_abomination_trigger = yes
				is_adult = yes
				POD_is_thinblooded_trigger = no
			}
			order_by = {
				value = resonance_rating_value
				if = { limit = { has_character_flag = POD_was_drunk_from } divide = 10 }
			}
			position = 0
			prev = {
				set_variable = {
					name  = suggested_prisoner
					value = prev
				}
			}
		}
	}
}
