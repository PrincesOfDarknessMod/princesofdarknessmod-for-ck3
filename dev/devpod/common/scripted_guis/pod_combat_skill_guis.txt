# Mid-combat move button. The save persists per-action state (slot
# _alive flips, soak / hit changes, log entries) back to the slot
# stories before the next turn evaluates.
pod_combat_execute_move = {
	scope = character
	# root  = the CLICKING client, which is who may command.
	# actor = the combatant they are driving, which is who acts. Those are the
	#         same character only while the one controlled slot is the host's own,
	#         and the two questions have to be asked of different characters.
	saved_scopes = { story actor }

	is_shown = {
		exists = scope:story
		pod_combat_is_active_trigger = yes
		NOT = { pod_combat_is_over_trigger = yes }
	}

	is_valid = {
		pod_combat_is_active_trigger = yes
		# Authority, not mode: this is a command surface, so it must ask whether the
		# live turn belongs to the CLICKING client. Identical answer today.
		pod_combat_is_my_turn_trigger = yes
		# ...and that the body being played is still standing. The AI path has had
		# this since its own turns "kept spending a downed actor's remaining
		# actions" (pod_combat_current_actor_is_down_trigger, gating
		# pod_combat_ai_take_turn_effect and pod_combat_ai_try_action_effect); the
		# human command surface never got the equivalent. Retaliation, recoil and a
		# self-including AoE can all kill the actor mid-turn, and nothing between
		# there and the commit drops the latch, so without this the bar stays lit on
		# a slot with slot_alive = 0 and the leftover AP is spendable out of a
		# corpse. In a FATAL duel the occupant can be genuinely dead, and the next
		# click then errors on var mutation against a dead reference.
		#
		# Invisible with one marked slot, which is why it lasted: there its KO IS
		# victory check 2, so is_over fires in the same effect and the bar dies on
		# the same frame. With a second controlled body still alive, check 2 does
		# not fire and the turn just sits there.
		#
		# End Turn deliberately does NOT get this guard - a downed body's turn must
		# stay committable or the round cannot advance.
		#
		# On the COMMAND SURFACE ONLY. Do not auto-commit from the KO site
		# instead: pod_combat_mark_slot_ko_effect runs inside an every_in_list over
		# the roster mid-move-chain, and pod_combat_process_group_turns_effect saves
		# the workspace BEFORE it drops the latch, so committing from there recurses
		# into its own prologue.
		NOT = { pod_combat_my_slot_is_ko_trigger = yes }
		# The unlock proof inside this trigger fails CLOSED, so the actor must be a
		# character whose pod_combat_unlocked_moves has been baked - see the bake
		# loop in POD_combat_finalize_duel_effect, which now covers every marked
		# slot for exactly this reason. A saved scope, not a var chain: the
		# validator re-splices $ACTOR$ into a scope switch and again into
		# pod_combat_move_dynamic_gates_trigger.
		#
		# trigger_if, not sibling exists checks: CK3 trigger siblings do not
		# short-circuit, so a plain `exists` next to the call READS like a guard
		# while gating nothing. Harmless from the four move-bar buttons (their
		# datamodel is the same actor chain, so a broken link yields zero buttons
		# rather than one with an unresolved scope) - but the first caller that is
		# not a datamodel item would splice both scopes unresolved into ~40 derefs.
		trigger_if = {
			limit = {
				exists = scope:story
				exists = scope:actor
			}
			POD_can_execute_combat_move_trigger = {
				STORY = scope:story
				ACTOR = scope:actor
			}
		}
		trigger_else = { always = no }
	}

	effect = {
		POD_execute_combat_move = { STORY = scope:story }
		pod_combat_save_workspace_effect = yes
	}
}

# RENDER-ONLY companion to pod_combat_execute_move - never a gate. Its is_valid
# exists purely to be rendered by IsValidTooltip on the move bar's buttons, so a
# greyed-out move says WHY. Never bind IsValid / Execute to this SGUI: it has no
# effect block, and its answer is deliberately narrower than the real gate's.
#
# Failures-only: every clause is a trigger_if whose limit is the INVERTED
# predicate, so a passing condition emits nothing at all and a usable move
# renders an empty string. That matches what the loadout editor already does for
# unlock requirements (is_valid_showing_failures_only on the hidden decision) and
# keeps the shipped failure-phrased strings readable - as a green "passed" line
# "Select an enemy target" would read as nonsense. custom_tooltip, not
# custom_description: the proven form for text rendered out of an SGUI is_valid
# (POD_journey_requirements -> PlayerGuiIsValidTooltip).
#
# Every boolean is the same named predicate the live validators call
# (pod_combat_move_execution_triggers.txt, SHARED GATE PREDICATES), so the reason
# and the grey-out cannot disagree. Order is most-superseding first: turn
# ownership beats resources, resources beat per-move state.
#
# Deliberately unvoiced: pod_combat_is_active_trigger (the WINDOW gates on
# pod_combat_active, so there is no bar to hover), team_enemy / team_all (
# vacuous during a live fight - the victory check ends the fight when the last
# enemy dies), the AI-only ally gates, and the scope-resolution trigger_else
# arms (not player-meaningful). is_over is NOT in this list: see below.
#
# LOCKSTEP: mirrors pod_combat_move_dynamic_gates_trigger plus the two wrapper
# gates. A blocking gate added there with no line here greys the button with no
# reason shown.
pod_combat_move_usability_tt = {
	scope = character
	# Same contract as pod_combat_execute_move: root = the clicking client,
	# scope:actor = the combatant being driven. Both are mandatory at every call
	# site because they are declared here.
	saved_scopes = { story actor }

	is_shown = { always = yes }

	is_valid = {
		# "You cannot act at all" renders ALONE - a trigger_else chain, not
		# siblings. CK3 trigger siblings do not short-circuit, so as siblings
		# these would stack under every per-move reason, and worse, the per-move
		# gates would price themselves against whatever combatant's workspace is
		# currently loaded. On someone else's turn that is not this move's cost
		# or target.
		#
		# is_over is voiced, and first. The window survives victory - it gates on
		# pod_combat_active, and the End Combat button is shown BY is_over - so the
		# bar sits there with every move greyed. Victory is also detected inside
		# the player's own move execution, so the turn latch is still held and
		# "It is not your turn" would not fire either: without this arm a
		# single_enemy move reports "Select an enemy target" (its target is dead)
		# and a self-targeted one reports nothing at all.
		trigger_if = {
			limit = { pod_combat_is_over_trigger = yes }
			custom_tooltip = { text = pod_combat_usable_fight_over_tt  always = no }
		}
		trigger_else_if = {
			limit = { NOT = { pod_combat_is_my_turn_trigger = yes } }
			custom_tooltip = { text = pod_combat_usable_not_your_turn_tt  always = no }
		}
		trigger_else_if = {
			limit = { pod_combat_my_slot_is_ko_trigger = yes }
			custom_tooltip = { text = pod_combat_usable_actor_down_tt  always = no }
		}
		trigger_else = {
			# Every clause below nests presence OUTSIDE the read. CK3 walks every
			# sub-trigger in a block, so a has_variable sitting BESIDE a var: read
			# does not gate it - the read runs anyway and errors when the var is
			# absent (DUEL_REFACTOR_LESSONS.md, "NOT { has_variable = X; var:X >= ...
			# } evaluates var:X regardless"). It matters more here than in the live
			# gate: only a dozen of the ~559 authored moves carry a currency cost, so
			# a sibling guard would spam error.log for all the rest on every hovered
			# frame. An unentered trigger_if returns true, so a move without the var
			# simply prints nothing.
			trigger_if = {
				limit = { scope:story = { has_variable = ap_cost } }
				trigger_if = {
					limit = { NOT = { pod_combat_can_afford_ap_trigger = { COST = scope:story.var:ap_cost } } }
					custom_tooltip = { text = pod_combat_usable_need_ap_tt  always = no }
				}
			}

			# Energy resolves against the scope it runs on, so wrap in scope:actor to
			# read the driven combatant's pool rather than the clicking client's.
			#
			# No custom_tooltip of our own here: called bare, PoD's energy check prints
			# its own splat-correct line ("You have less than 3 Blood" / "... Gnosis"),
			# from the *_NOT trigger_localization variants on
			# POD_has_total_<splat>_reserves. The enclosing trigger_if is what keeps it
			# failures-only - the identical call in the limit means the body runs, and
			# so prints, only when the reserve is actually short.
			trigger_if = {
				limit = { scope:story = { has_variable = energy_cost } }
				trigger_if = {
					limit = {
						NOT = {
							scope:actor = {
								POD_has_enough_splat_agnostic_energy_reserves = { AMOUNT = scope:story.var:energy_cost }
							}
						}
					}
					scope:actor = {
						POD_has_enough_splat_agnostic_energy_reserves = { AMOUNT = scope:story.var:energy_cost }
					}
				}
			}

			# CK3 currencies are PC-only, matching the live gates' player-turn limits.
			trigger_if = {
				limit = {
					scope:story = { has_variable = gold_cost }
					pod_combat_is_player_turn_trigger = yes
				}
				trigger_if = {
					limit = { NOT = { pod_combat_can_afford_move_gold_trigger = { STORY = scope:story } } }
					custom_tooltip = { text = pod_combat_usable_need_gold_tt  always = no }
				}
			}

			trigger_if = {
				limit = {
					scope:story = { has_variable = piety_cost }
					pod_combat_is_player_turn_trigger = yes
				}
				trigger_if = {
					limit = { NOT = { pod_combat_can_afford_move_piety_trigger = { STORY = scope:story } } }
					custom_tooltip = { text = pod_combat_usable_need_piety_tt  always = no }
				}
			}

			trigger_if = {
				limit = {
					scope:story = { has_variable = prestige_cost }
					pod_combat_is_player_turn_trigger = yes
				}
				trigger_if = {
					limit = { NOT = { pod_combat_can_afford_move_prestige_trigger = { STORY = scope:story } } }
					custom_tooltip = { text = pod_combat_usable_need_prestige_tt  always = no }
				}
			}

			# once_per: presence, then "was it used", then the window wording.
			# LOCKSTEP with the round/else dispatch inside
			# pod_combat_actor_has_used_move_trigger - anything not flag:round is held
			# for the whole fight there, so the else arm must say "this fight".
			trigger_if = {
				limit = { scope:story = { has_variable = once_per } }
				trigger_if = {
					limit = { pod_combat_actor_has_used_move_trigger = { STORY = scope:story  ACTOR = scope:actor } }
					trigger_if = {
						limit = { scope:story = { var:once_per = flag:round } }
						custom_tooltip = { text = pod_combat_usable_used_this_round_tt  always = no }
					}
					trigger_else = {
						custom_tooltip = { text = pod_combat_usable_used_this_combat_tt  always = no }
					}
				}
			}

			# Targeting - one presence gate covering all three arms.
			trigger_if = {
				limit = { scope:story = { has_variable = targeting } }
				trigger_if = {
					limit = {
						pod_combat_is_player_turn_trigger = yes
						scope:story = { var:targeting = flag:single_enemy }
						NOT = { pod_combat_player_has_enemy_target_trigger = yes }
					}
					custom_tooltip = { text = pod_combat_targeting_need_enemy_tt  always = no }
				}
				trigger_if = {
					limit = {
						pod_combat_is_player_turn_trigger = yes
						scope:story = { var:targeting = flag:single_ally }
						NOT = { pod_combat_player_has_ally_target_trigger = yes }
					}
					custom_tooltip = { text = pod_combat_targeting_need_ally_tt  always = no }
				}
				trigger_if = {
					limit = {
						pod_combat_is_player_turn_trigger = yes
						scope:story = { var:targeting = flag:single_pc }
						NOT = { pod_combat_pc_slot_alive_trigger = yes }
					}
					custom_tooltip = { text = pod_combat_targeting_need_pc_alive_tt  always = no }
				}
			}

			# One generic line for all the condition predicates: the stat card
			# directly above already spells out WHICH condition (its "Requires:"
			# dispatch), so naming it again here would only duplicate. cond_predicate
			# is its own var, so the presence gate covers both before the switch is
			# dispatched on it.
			trigger_if = {
				limit = { scope:story = { has_variable = cond_mode  has_variable = cond_predicate } }
				trigger_if = {
					limit = {
						scope:story = { var:cond_mode = flag:gate }
						NOT = {
							pod_combat_move_condition_met_trigger = {
								STORY = scope:story ACTOR = scope:actor PREDICATE = scope:story.var:cond_predicate
							}
						}
					}
					custom_tooltip = { text = pod_combat_usable_condition_unmet_tt  always = no }
				}
			}

			# The heal and redundancy predicates carry their own presence guards -
			# they are the live gate's own bodies, lifted verbatim.
			trigger_if = {
				limit = {
					pod_combat_move_is_self_only_heal_trigger = { STORY = scope:story }
					NOT = { pod_combat_self_heal_has_effect_trigger = { STORY = scope:story } }
				}
				custom_tooltip = { text = pod_combat_cant_heal_already_full_desc  always = no }
			}

			trigger_if = {
				limit = {
					NOT = { pod_combat_move_has_independent_payload_trigger = { STORY = scope:story } }
					NOT = { pod_combat_move_payload_useful_trigger = { STORY = scope:story } }
				}
				custom_tooltip = { text = pod_combat_move_redundant_tt  always = no }
			}

			# Two reasons a summon can be refused, and they are mutually exclusive
			# in the wording: an else-chain, so a barred move in a full team states
			# the rule once instead of stacking both lines. The trailing
			# trigger_else terminates the chain (CK3 rejects an open one at load).
			trigger_if = {
				limit = { scope:story = { has_variable = summons_combatant } }
				trigger_if = {
					limit = { pod_combat_duel_bans_summons_trigger = yes }
					custom_tooltip = { text = pod_combat_cant_summon_duel_barred_tt  always = no }
				}
				trigger_else_if = {
					limit = {
						var:pod_combat_session ?= { has_variable = pod_combat_current_actor_team }
						NOT = { pod_combat_summon_capacity_available_trigger = { STORY = scope:story } }
					}
					custom_tooltip = { text = pod_combat_cant_summon_team_full_tt  always = no }
				}
				trigger_else = { always = yes }
			}

			# Failsafe, last: the bake is meant to cover every marked slot, so this
			# speaks only if one was missed. Generic wording - the per-move
			# requirement text lives in the loadout editor, not mid-fight.
			trigger_if = {
				limit = { NOT = { pod_combat_actor_has_baked_unlock_trigger = { STORY = scope:story ACTOR = scope:actor } } }
				custom_tooltip = { text = pod_combat_usable_not_unlocked_tt  always = no }
			}
		}
	}
}

# Close-skills button. The loadout window's `visible` binding watches
# pod_combat_config_open; removing it hides the window.
pod_combat_close_skills = {
	scope = character

	is_shown = { always = yes }
	is_valid = { always = yes }

	effect = {
		remove_variable = pod_combat_config_open
		# Sheet scratch list is editor-session state; it holds one story ref
		# per catalogue move, which would otherwise sit in the save. Rebuilt
		# on next open if sheet view is still selected.
		clear_variable_list = pod_combat_sheet_rows
	}
}

# End-turn button. PC commits; AI turns then run via
# pod_combat_process_group_turns_effect. Disabled mid-AI-turn so a
# stray click can't compress two PC turns into one round.
pod_combat_end_turn = {
	scope = character

	is_shown = {
		pod_combat_is_active_trigger = yes
	}

	is_valid = {
		pod_combat_is_active_trigger = yes
		# Authority, not mode: this is a command surface, so it must ask whether the
		# live turn belongs to the CLICKING client. Identical answer today.
		pod_combat_is_my_turn_trigger = yes
		NOT = { pod_combat_is_over_trigger = yes }
	}

	effect = {
		pod_combat_process_group_turns_effect = yes
	}
}

# End-combat button. Replaces end-turn once pod_combat_is_over is set.

	}

	is_valid = {
		pod_combat_is_active_trigger = yes
		pod_combat_is_over_trigger = yes
	}

	effect = {
		pod_combat_end_combat_effect = yes
	}
}

# Auto-advance toggle (the combat-window checkbox). Flips the toggling PLAYER's
# persistent preference pod_combat_manual_step_pref (present = manual stepping;
# absent = auto-advance, the default), then re-seeds the live fight's
# pod_combat_step_mode host var so the CURRENT combat responds immediately.
# Per-player, NOT a shared global: one player toggling never changes
# another player's live fight or default. The combat host IS the toggling PC
# (root = GetPlayer from the checkbox onclick), so the pref and the host
# step_mode var live on the same character.
pod_combat_toggle_auto_advance = {
	scope = character

	is_shown = {
		pod_combat_is_active_trigger = yes
	}

	is_valid = {
		pod_combat_is_active_trigger = yes
	}

	effect = {
		if = {
			limit = { has_variable = pod_combat_manual_step_pref }
			remove_variable = pod_combat_manual_step_pref
		}
		else = {
			set_variable = { name = pod_combat_manual_step_pref value = yes }
		}
		# Re-seed the live fight's step_mode from the now-updated pref.
		if = {
			limit = { has_variable = pod_combat_manual_step_pref }
			var:pod_combat_session ?= { set_variable = { name = pod_combat_step_mode value = yes } }
		}
		else = {
			var:pod_combat_session ?= { remove_variable = pod_combat_step_mode }
		}
	}
}

# Advance-one-AI-turn button (mid-step entry point). PROLOGUE-FREE by design:
# never marks the PC acted or saves the PC workspace, so it is safe to fire
# repeatedly while an AI is pending. The PC-commit prologue lives ONLY in
# pod_combat_process_group_turns_effect (via pod_combat_end_turn).
#
# Mode-aware so a mid-step toggle-to-auto drains on the next click; pc_has_acted
# routes the settle in both the STEP and AUTO branches below.
pod_combat_advance_turn = {
	scope = character

	is_shown = {
		pod_combat_is_active_trigger = yes
	}

	is_valid = {
		pod_combat_can_advance_trigger = yes
	}

	effect = {
		if = {
			limit = { var:pod_combat_session ?= { has_variable = pod_combat_step_mode } }
			pod_combat_advance_one_ai_turn_effect = yes
			if = {
				limit = { pod_combat_pc_has_acted_this_round_trigger = yes }
				pod_combat_after_step_settle_effect = yes
			}
			else = {
				pod_combat_after_pre_pc_step_settle_effect = yes
			}
		}
		else = {
			# AUTO. Route on pc_has_acted exactly like the STEP branch above and
			# pod_combat_process_group_turns_effect: already committed -> drain the
			# remaining post-PC AI, then end the round; NOT yet acted (pre-PC phase,
			# e.g. auto-advance toggled on while an AI leads off) -> walk the AI up
			# to the controlled slot via the pre-PC pump, which runs setup_pc_turn
			# inline.
			#
			# The split must stay. Both pumps now stop at a controlled slot, but the
			# pre-PC state is one where setup_pc_turn has ALREADY run and the turn is
			# live: run_remaining there would stop on pass 0 and re-run setup on a
			# turn already in progress, re-seeding AP and stripping round buffs.
			if = {
				limit = { pod_combat_pc_has_acted_this_round_trigger = yes }
				pod_combat_run_remaining_ai_turns_effect = yes
				if = {
					limit = {
						NOT = { pod_combat_current_actor_is_controlled_trigger = yes }
						NOT = { pod_combat_is_over_trigger = yes }
					}
					pod_combat_group_end_round_effect = yes
				}
			}
			else = {
				pod_combat_run_pre_pc_ai_turns_effect = yes
			}
		}
	}
}

# Equip / unequip a move. Unequip depends only on current loadout membership
# (always allowed for an equipped move); equip requires the move be unlocked
# (the dispatcher, not can_execute_decision) AND its category have
# room (cap = pod_combat_max_per_category). The GUI's `checked` binding reads
# is_shown = "currently equipped".
pod_combat_loadout_toggle = {
	scope = character
	saved_scopes = { story }

	is_shown = {
		exists = scope:story
		is_target_in_variable_list = {
			name   = pod_combat_loadout
			target = scope:story
		}
	}

	is_valid = {
		exists = scope:story
		OR = {
			# Equipped -> unequip depends ONLY on current loadout membership.
			is_target_in_variable_list = {
				name   = pod_combat_loadout
				target = scope:story
			}
			# Not equipped -> require the move be UNLOCKED and its category have room.
			AND = {
				# Unlock gate via the dispatcher, NOT can_execute_decision:
				# decisions carry the implicit ruler taker-requirement an unlanded PC
				# fails wholesale. Runtime and the hidden decision both delegate to
				# one per-move rule, so authority and failure text cannot drift.
				# COMBAT_REFERENCE.info §9 file map, pod_combat_move_decisions.txt
				# entry (unlock gating).
				pod_combat_actor_unlocks_move_trigger = { STORY = scope:story ACTOR = root }
				# Capacity. All three loadout surfaces - this click gate, the
				# row-enabled gate (pod_combat_loadout_room) and the reason text
				# (pod_combat_loadout_block_tt) - share these predicates, so a greyed
				# row and its tooltip cannot disagree about which cap is full.
				pod_combat_loadout_room_left_trigger = yes
				# Cap check per combat_category. custom_description surfaces a friendly
				# "Attack slots full: 4/4" tooltip; the trigger_if chain keeps it singular.
				trigger_if = {
					limit = { scope:story = { var:combat_category = flag:ATK } }
					custom_description = {
						text = pod_combat_loadout_atk_full_tt
						pod_combat_atk_slots_free_trigger = yes
					}
				}

# Per-row visibility: does the PC meet the move's unlock requirements? Reads the

	}
}

######################################################
# Dynamic per-attribute stat block, BuildTooltip-rendered from the
# PODCombatMoveEffect macro. The macro saves the move's story_cycle as
# the GuiScope scope `story`; each pod_combat_desc_X loc reads its vars
# via [story.MakeScope.Var('Y')...]. AP cost lives in the move title,
# not here, so this block starts at resource costs.
#
# Dispatch order: costs -> PC resource gains (steal_gold / gain_dread) ->
# condition gate + once_per -> HP damage (per damage class x drain) ->
# morale/self -> healing -> capability callouts -> stance-strip -> tier
# buffs/debuffs -> stance flags -> retaliation ->
# AP/initiative/DoT/mind-control/summon -> conditional bonus.
#
# To surface a NEW move attribute: add a pod_combat_desc_<attr> key in
# POD_combat_static_l_english.yml (DYNAMIC MOVE TOOLTIP section; #medium;outline
# prefix + $EFFECT_LIST_BULLET$ suffix), then add an
#   if = { limit = { scope:story = { has_variable = <attr> } } custom_tooltip = pod_combat_desc_<attr> }
# branch in the matching dispatch group below.
pod_combat_move_desc = {
	scope = character
	# `nohide` is bound only by PODCombatMoveEffectStatic, the combat-log hover
	# card; every other caller leaves it unset and the clause hide gate reads it
	# through exists = scope:nohide. OPEN IN-GAME CHECK: no other in-tree SGUI
	# declares a saved scope it is not always handed, so whether an unsupplied
	# declared scope loads cleanly has no precedent to lean on.
	saved_scopes = { story nohide }
	effect = {
		# Every branch below does scope:story = { ... }. Callers reach this SGUI
		# through the PODCombatMoveEffect data-binding macro, which can hand us an
		# unresolvable story (a combat-log row whose entry story was end_story'd -
		# see pod_combat_host_died_release_effect). Without this guard that logs a
		# 'Failed context switch' per top-level branch while building the tooltip.
		# Body left un-indented: script is whitespace-insensitive and re-indenting
		# ~600 lines would bury the real diff.
		if = {
			limit = { exists = scope:story }
		# Combined cost line. The OR-of-five gate suppresses the
		# dispatcher (and its dangling "Costs:" prefix) when no cost
		# var is set.
		#
		# PARSER QUIRK: do NOT collapse to a single SelectLocalization
		# call. The data-binding parser treats `#` as a comment marker
		# and rejects nested `[X]` substitutions inside function-arg
		# strings, so the `#bold`/`[Var]` cost tokens must live in a
		# customizable_localization loc string, not a function arg.
		if = {
			limit = {
				scope:story = {
					OR = {
						has_variable = energy_cost
						has_variable = gold_cost
						has_variable = piety_cost
						has_variable = prestige_cost
						has_variable = stress_cost
					}
				}
			}
			custom_tooltip = pod_combat_desc_costs
		}
		# Phase 4 PC-only resource gains (steal_gold transfers, gain_dread adds).
		if = {
			limit = { scope:story = { has_variable = steal_gold } }
			custom_tooltip = pod_combat_desc_steal_gold

				}
			}
			else_if = {
				limit = { scope:story = { has_variable = hp_damage } }
				if = {
					limit = { scope:story = { has_variable = retaliate_agg } }
					custom_tooltip = pod_combat_desc_retaliate_if_attacked_reuse_agg
				}
				else = {
					custom_tooltip = pod_combat_desc_retaliate_if_attacked_reuse_sup
				}
			}
			else = {
				custom_tooltip = pod_combat_desc_retaliate_if_attacked
			}
		}

		# AP / initiative parametric
		if = {
			limit = { scope:story = { has_variable = ap_gain } }
			custom_tooltip = pod_combat_desc_ap_gain
		}
		if = {
			limit = { scope:story = { has_variable = ap_drain } }
			custom_tooltip = pod_combat_desc_ap_drain
		}
		if = {
			limit = { scope:story = { has_variable = initiative_debuff } }
			custom_tooltip = pod_combat_desc_initiative_debuff
		}
		if = {
			limit = { scope:story = { has_variable = dot_rounds } }
			custom_tooltip = pod_combat_desc_dot
		}
		if = {
			limit = { scope:story = { has_variable = mind_control } }
			custom_tooltip = pod_combat_desc_mind_control
		}
		if = {
			limit = { scope:story = { has_variable = summons_combatant } }
			custom_tooltip = pod_combat_desc_summon
		}
		if = {
			limit = { scope:story = { has_variable = marker_modifier } }
			custom_tooltip = pod_combat_desc_marker_modifier
		}
		# The strip, above its apply twin so the card reads in execution order
		# (12b-marker runs before 12c).
		if = {
			limit = { scope:story = { has_variable = remove_marker } }
			if = {
				limit = { scope:story = { var:remove_marker_target = flag:target } }
				custom_tooltip = pod_combat_desc_remove_marker_target
			}
			else = { custom_tooltip = pod_combat_desc_remove_marker }
		}

		# Conditional bonus clauses. Each POD_combat_move_clause story on the
		# move's bonus_clauses list gets its own "Bonus if <X>: <effect>" pair,
		# rendered below the base effect; same-effect clauses stack at execution.
		# The move-level cond_* / bonus_* scalars mirror clause 1, so this list is
		# the ONLY bonus surface on the card - a row-level ladder beside it would
		# print clause 1 twice.
		if = {
			limit = { scope:story = { has_variable_list = bonus_clauses } }
			# The walk has to run in story scope - this SGUI is scope = character, so
			# a bare every_in_list iterates nothing - while each line has to be
			# emitted back on the viewer, whose perks and weapon those lines read.
			# That round-trip is the POD_house_display shape (POD_general_guis.txt),
			# with root standing in for its prev because prev here is the move story
			# rather than a character.
			#
			# The clause binds as pod_clause. Never rebind `story`: the rebind would
			# outlive the loop and send every later scope:story read to a clause.
			scope:story = {
				every_in_list = {
					variable = bonus_clauses
					save_scope_as = pod_clause
					root = { pod_combat_card_clause_line_effect = yes }
				}
			}
		}
		}
	}
}
