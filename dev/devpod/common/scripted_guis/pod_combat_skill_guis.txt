######################################################
######   POD COMBAT SCRIPTED GUIS
######################################################
#
# All SGUIs invoked by pod_combat_window.gui (the mid-combat window)
# and pod_combat_skills_window.gui (the loadout editor). Three
# families:
#
#   1. Combat-window action SGUIs (execute_move, select_target,
#      forfeit, close_skills)
#   2. Loadout-editor SGUIs (toggle, option, show_all_moves)
#   3. Filter SGUIs for the loadout editor (toggle_filter_*,
#      clear_filters, is_*_move, loadout_category_match)
#
# Convention: every SGUI is invoked from the GUI with
# `GuiScope.SetRoot(GetPlayer.MakeScope)`, so root = the PC. Per-
# move SGUIs additionally pass `AddScope('story', Scope.Story.MakeScope)`,
# which surfaces inside the SGUI as `scope:story` (= the move's
# POD_combat_move story_cycle scope).
#
# Filter state lives as character variables on the PC:
#   pod_combat_filter_atk / _def / _ctl / _pwr — presence = filter active
# When NO filter is active, every category is visible (default state).
# When at least one is active, only matching categories are shown.


######################################################
###### COMBAT-WINDOW ACTION SGUIS
######################################################

# Mid-combat move button. Wraps the runtime-side
# POD_can_execute_combat_move trigger (unlock + AP + energy +
# targeting + per-resource costs) for is_valid, and
# POD_execute_combat_move + pod_combat_save_workspace_effect for
# the click action. The save ensures all per-action state changes
# (slot _alive flips, soak / hit changes, log entries) persist back
# to the slot stories before the next turn evaluates.
pod_combat_execute_move = {
	scope = character
	saved_scopes = { story }

	is_shown = {
		exists = scope:story
		has_variable = pod_combat_active
		NOT = { has_variable = pod_combat_is_over }
	}

	is_valid = {
		# Combat must be live and it has to be the player's turn —
		# AI turns auto-execute via pod_combat_ai_take_turn_effect,
		# the button only meaningfully fires on PC turns.
		pod_combat_is_active_trigger = yes
		pod_combat_is_player_turn_trigger = yes
		POD_can_execute_combat_move = {
			STORY = scope:story
			ACTOR = root
		}
	}

	effect = {
		POD_execute_combat_move = { STORY = scope:story }
		pod_combat_save_workspace_effect = yes
	}
}

# Forfeit button (top-right X). Surrenders the current combat —
# the PC concedes, the opposing team wins. Sets winner/loser the
# same way pod_combat_auto_resolve_effect does (PC = loser by
# unconditional decision rather than by prowess compare), flips
# is_over so the combat window's End Combat button takes over.
# The standard combat-end pipeline (pod_combat_end_combat_effect)
# runs when the player dismisses, firing the SC follow-up event
# and cleaning up.
pod_combat_forfeit = {
	scope = character

	is_shown = {
		has_variable = pod_combat_active
	}

	is_valid = {
		pod_combat_is_active_trigger = yes
		NOT = { has_variable = pod_combat_is_over }
	}

	effect = {
		save_scope_as = combat_pc
		# PC concedes — opposing primary (slot 5's char) wins.
		if = {
			limit = {
				has_variable = pod_combat_slot_5
				var:pod_combat_slot_5 = { has_variable = slot_char }
			}
			set_variable = {
				name  = pod_combat_winner
				value = var:pod_combat_slot_5.var:slot_char
			}
		}
		set_variable = { name = pod_combat_loser value = scope:combat_pc }
		# Mark combat over; the window's End Combat button calls
		# pod_combat_end_combat_effect to fire the SC follow-up
		# and clean up.
		set_variable = { name = pod_combat_is_over value = yes }
	}
}

# Close-skills button on the loadout-editor window. Removes the
# pod_combat_config_open flag — the window's visibility binding at
# pod_combat_skills_window.gui:37 watches this var, so removing it
# hides the window. Set by the pod_combat_configure_skills decision
# when the player opens the editor.
pod_combat_close_skills = {
	scope = character

	is_shown = { always = yes }
	is_valid = { always = yes }

	effect = {
		remove_variable = pod_combat_config_open
	}
}

# End-turn button. PC commits their turn — AI turns then run via
# pod_combat_process_group_turns_effect (which marks the PC as acted,
# saves the workspace, iterates the remaining actors in initiative
# order, and rolls the round-end recalc when everyone has gone).
# Disabled mid-AI-turn so a stray click can't compress two PC turns
# into one round.
pod_combat_end_turn = {
	scope = character

	is_shown = {
		has_variable = pod_combat_active
	}

	is_valid = {
		pod_combat_is_active_trigger = yes
		pod_combat_is_player_turn_trigger = yes
		NOT = { has_variable = pod_combat_is_over }
	}

	effect = {
		pod_combat_process_group_turns_effect = yes
	}
}

# End-combat button. Replaces the end-turn button at the bottom of
# the combat window once pod_combat_is_over is set (victory, defeat,
# auto-resolve outcome, or forfeit). Fires the SC follow-up event,
# applies non-fatal-mode wound caps, and runs the cleanup pipeline
# that strips workspace/slot stories and dismisses the window.
#
# is_valid intentionally requires pod_combat_is_over — the inverse
# of pod_combat_end_turn — so the two SGUIs partition the lifetime
# of the combat window cleanly: end_turn during play, end_combat
# after.
pod_combat_end_combat = {
	scope = character

	is_shown = {
		has_variable = pod_combat_active
		has_variable = pod_combat_is_over
	}

	is_valid = {
		has_variable = pod_combat_active
		has_variable = pod_combat_is_over
	}

	effect = {
		pod_combat_end_combat_effect = yes
	}
}

# Auto-resolve button. Skips the rest of combat — pod_combat_auto_resolve_effect
# runs a prowess-with-jitter compare to pick a winner, sets is_over,
# leaves the window open for the player to dismiss via End Combat.
# Available even mid-AI-turn so the player can bail out of a long
# combat at any point.
pod_combat_auto_resolve = {
	scope = character

	is_shown = {
		has_variable = pod_combat_active
	}

	is_valid = {
		pod_combat_is_active_trigger = yes
		NOT = { has_variable = pod_combat_is_over }
	}

	effect = {
		pod_combat_auto_resolve_effect = yes
	}
}

# Help button. Fires the combat-system overview event
# (pod_combat.0001) which renders the rules / window legend in a
# normal CK3 event popup. No combat-state change.
pod_combat_help = {
	scope = character

	is_shown = { always = yes }
	is_valid = { always = yes }

	effect = {
		trigger_event = pod_combat.0001
	}
}

# Equip / unequip a move. is_valid permits both directions:
#   - If the move is already equipped, the toggle removes it
#     (always allowed — unequipping never violates the cap).
#   - If the move is not equipped, the toggle adds it only when
#     the actor's category cap allows (4 per ATK/DEF/CTL/PWR).
# `checked` in the GUI binds to IsShown which mirrors "currently
# equipped" so the button shows the equipped state visually.
pod_combat_loadout_toggle = {
	scope = character
	saved_scopes = { story }

	is_shown = {
		# Equipped check — used by the button's `checked` binding
		# to render the equipped vs unequipped visual state.
		exists = scope:story
		is_target_in_variable_list = {
			name   = pod_combat_loadout
			target = scope:story
		}
	}

	is_valid = {
		exists = scope:story
		# Trait / perk gate from the move's hidden decision.
		can_execute_decision = scope:story.var:decision
		OR = {
			# Already equipped — unequip is always allowed.
			is_target_in_variable_list = {
				name   = pod_combat_loadout
				target = scope:story
			}
			# Not equipped — equip only if the move's category
			# still has room.
			trigger_if = {
				limit = { scope:story = { var:combat_category = flag:ATK } }
				pod_combat_atk_count < pod_combat_max_per_category
			}
			trigger_if = {
				limit = { scope:story = { var:combat_category = flag:DEF } }
				pod_combat_def_count < pod_combat_max_per_category
			}
			trigger_if = {
				limit = { scope:story = { var:combat_category = flag:CTL } }
				pod_combat_ctl_count < pod_combat_max_per_category
			}
			trigger_if = {
				limit = { scope:story = { var:combat_category = flag:PWR } }
				pod_combat_pwr_count < pod_combat_max_per_category
			}
		}
	}

	effect = {
		if = {
			limit = {
				is_target_in_variable_list = {
					name   = pod_combat_loadout
					target = scope:story
				}
			}
			# Currently equipped — unequip.
			remove_list_variable = {
				name   = pod_combat_loadout
				target = scope:story
			}
		}
		else = {
			# Not equipped — equip.
			add_to_variable_list = {
				name   = pod_combat_loadout
				target = scope:story
			}
		}
	}
}

# Per-row visibility check in the loadout editor: does this character
# meet the move's unlock requirements (trait / faith / perk)? Reads
# the move's hidden decision and runs can_execute_decision against
# the PC. Returns yes when the move is available.
#
# is_shown also passes when the player has toggled the "Show all
# moves" / "Compact view" overrides — both flip the gate to "show
# every move regardless of unlock state" so the player can browse
# the full catalogue. is_valid stays strict: the toggle button is
# enabled only when the PC could actually equip the move; show-all
# is purely a visibility override, not a permissions bypass.
pod_combat_loadout_option = {
	scope = character
	saved_scopes = { story }

	is_shown = {
		exists = scope:story
		OR = {
			can_execute_decision = scope:story.var:decision
			has_variable = show_all_combat_moves
			# Compact view also implies show-all (state-coupled
			# in pod_combat_skills_window.gui's `checked` binding).
		}
	}

	is_valid = {
		exists = scope:story
		can_execute_decision = scope:story.var:decision
	}
}

# "Show all moves" toggle — flips the show_all_combat_moves PC var.
# When set, pod_combat_loadout_option's is_shown bypasses the
# trait/perk gate so every move in the global catalogue appears in
# the editor (the player can see what they could equip if their
# splat / faith / perks were different). When unset (default),
# the editor only shows moves the PC can currently use. The GUI's
# `checked` binding watches this same var, so the visual toggle
# state stays in sync.
pod_combat_loadout_show_all_moves = {
	scope = character

	is_shown = { always = yes }
	is_valid = { always = yes }

	effect = {
		if = {
			limit = { has_variable = show_all_combat_moves }
			remove_variable = show_all_combat_moves
		}
		else = {
			set_variable = show_all_combat_moves
		}
	}
}

# Combined filter check. Returns yes if either:
#   1. NO filter is active (show every category), OR
#   2. The move's combat_category matches an active filter.
# Used by the loadout-editor row visibility on the unfiltered move
# list so toggling a filter narrows the visible set.
pod_combat_loadout_category_match = {
	scope = character
	saved_scopes = { story }

	is_shown = {
		exists = scope:story
		OR = {
			# No filter active — show everything.
			AND = {
				NOT = { has_variable = pod_combat_filter_atk }
				NOT = { has_variable = pod_combat_filter_def }
				NOT = { has_variable = pod_combat_filter_ctl }
				NOT = { has_variable = pod_combat_filter_pwr }
			}
			# Or filter active and matches.
			AND = {
				has_variable = pod_combat_filter_atk
				scope:story = { var:combat_category = flag:ATK }
			}
			AND = {
				has_variable = pod_combat_filter_def
				scope:story = { var:combat_category = flag:DEF }
			}
			AND = {
				has_variable = pod_combat_filter_ctl
				scope:story = { var:combat_category = flag:CTL }
			}
			AND = {
				has_variable = pod_combat_filter_pwr
				scope:story = { var:combat_category = flag:PWR }
			}
		}
	}
}

###### MOVE STAT BLOCK (BuildTooltip-rendered)
######################################################
# Renders the dynamic per-attribute stat block for a combat move.
# Invoked by the PODCombatMoveEffect data-binding macro via
# `GetScriptedGui('pod_combat_move_desc').BuildTooltip(...)`. The
# macro attaches the move's story_cycle scope as the GuiScope
# saved scope `story`; BuildTooltip carries it into this SGUI's
# effect evaluation, and each pod_combat_desc_X loc string reads
# the move's variables via [story.MakeScope.Var('Y')...] binding.
#
# Each line is a separate `if = { limit = { has_variable } custom_tooltip }`
# pair. custom_tooltip emits the loc unconditionally inside the
# branch; the limit gates the whole branch. Order roughly mirrors
# the original PODCombatMoveStats customizable_loc:
#   1. AP cost (always; variant by attack_type / hp_damage)
#   2. Resource costs (energy, stress, gold, piety, prestige)
#   3. HP damage (one branch per damage_source × drain combo)
#   4. Morale damage / self-damage
#   5. Healing (standalone hp_heal, standalone drain, morale_heal)
#   6. Tier buffs/debuffs (8 variants)
#   7. Stance flags (set_guard, dodge, stealth, remove_guard)
#   8. Bypass flags (bypass_guard, bypass_dodge, bypass_stealth)
#   9. Retaliation (retaliate_if_hit, retaliate_if_missed)
#  10. AP / initiative parametric (ap_gain, ap_drain, initiative_debuff)
#  11. Targeting hint (skip single_enemy default)
#
# History note: an earlier implementation used a customizable_loc
# chain (PODCombatMoveStats) that emitted a composite loc with
# nested [PODCombatStat(...)] calls. Saved scopes attached via
# GuiScope.AddScope don't survive nested .Custom invocations
# inside the resulting loc string, so every per-line custom
# rendered empty. This BuildTooltip-on-SGUI approach matches
# DUEL-Refactor's POD_duel_move_desc and works.
pod_combat_move_desc = {
	scope = character
	saved_scopes = { story }
	effect = {
		# AP cost — always rendered, variant by attack_type
		if = {
			limit = {
				scope:story = {
					has_variable = attack_type
					var:attack_type = flag:supernatural
				}
			}
			custom_tooltip = pod_combat_desc_ap_cost_supernatural
		}
		else_if = {
			limit = {
				scope:story = {
					has_variable = attack_type
					var:attack_type = flag:mental
				}
			}
			custom_tooltip = pod_combat_desc_ap_cost_mental
		}
		else_if = {
			# hp_damage present without explicit attack_type → physical default
			limit = { scope:story = { has_variable = hp_damage } }
			custom_tooltip = pod_combat_desc_ap_cost_physical
		}
		else = {
			custom_tooltip = pod_combat_desc_ap_cost
		}

		# Resource costs
		if = {
			limit = { scope:story = { has_variable = energy_cost } }
			custom_tooltip = pod_combat_desc_energy_cost
		}
		if = {
			limit = { scope:story = { has_variable = stress_cost } }
			custom_tooltip = pod_combat_desc_stress_cost
		}
		if = {
			limit = { scope:story = { has_variable = gold_cost } }
			custom_tooltip = pod_combat_desc_gold_cost
		}
		if = {
			limit = { scope:story = { has_variable = piety_cost } }
			custom_tooltip = pod_combat_desc_piety_cost
		}
		if = {
			limit = { scope:story = { has_variable = prestige_cost } }
			custom_tooltip = pod_combat_desc_prestige_cost
		}

		# HP damage — combined drain variants first, then plain.
		# trigger_if/trigger_else not used; explicit if/else_if chain
		# walks the most-specific branch first.
		if = {
			limit = {
				scope:story = {
					has_variable = hp_damage
					has_variable = drain
					has_variable = damage_source
					var:damage_source = flag:fire
				}
			}
			custom_tooltip = pod_combat_desc_hp_damage_fire_drain
		}
		else_if = {
			limit = {
				scope:story = {
					has_variable = hp_damage
					has_variable = drain
					has_variable = damage_source
					var:damage_source = flag:cold
				}
			}
			custom_tooltip = pod_combat_desc_hp_damage_cold_drain
		}
		else_if = {
			limit = {
				scope:story = {
					has_variable = hp_damage
					has_variable = drain
					has_variable = damage_source
					var:damage_source = flag:lightning
				}
			}
			custom_tooltip = pod_combat_desc_hp_damage_lightning_drain
		}
		else_if = {
			limit = {
				scope:story = {
					has_variable = hp_damage
					has_variable = drain
					has_variable = damage_source
					var:damage_source = flag:sun
				}
			}
			custom_tooltip = pod_combat_desc_hp_damage_sun_drain
		}
		else_if = {
			limit = {
				scope:story = {
					has_variable = hp_damage
					has_variable = drain
					has_variable = damage_source
					var:damage_source = flag:silver
				}
			}
			custom_tooltip = pod_combat_desc_hp_damage_silver_drain
		}
		else_if = {
			limit = {
				scope:story = {
					has_variable = hp_damage
					has_variable = drain
					has_variable = damage_source
					var:damage_source = flag:gold
				}
			}
			custom_tooltip = pod_combat_desc_hp_damage_gold_drain
		}
		else_if = {
			limit = {
				scope:story = {

		}

				scope:story = {
					has_variable = hp_damage
					has_variable = drain
				}
			}
			custom_tooltip = pod_combat_desc_hp_damage_basic_drain
		}
		else_if = {
			limit = {
				scope:story = {
					has_variable = hp_damage
					has_variable = damage_source
					var:damage_source = flag:fire
				}
			}
			custom_tooltip = pod_combat_desc_hp_damage_fire
		}
		else_if = {

		else_if = {

					has_variable = hp_damage

					var:damage_source = flag:sun
				}
			}
			custom_tooltip = pod_combat_desc_hp_damage_sun
		}
		else_if = {

					has_variable = hp_damage
					has_variable = damage_source
					var:damage_source = flag:silver
				}
			}
			custom_tooltip = pod_combat_desc_hp_damage_silver
		}
		else_if = {
			limit = {
				scope:story = {
					has_variable = hp_damage
					has_variable = damage_source
					var:damage_source = flag:gold
				}
			}
			custom_tooltip = pod_combat_desc_hp_damage_gold
		}
		else_if = {
			limit = {
				scope:story = {
					has_variable = hp_damage
					has_variable = damage_source
					var:damage_source = flag:holy
				}
			}
			custom_tooltip = pod_combat_desc_hp_damage_holy
		}
		else_if = {
			limit = {
				scope:story = {
					has_variable = hp_damage
					has_variable = damage_source
					var:damage_source = flag:balefire
				}
			}
			custom_tooltip = pod_combat_desc_hp_damage_balefire
		}
		else_if = {
			limit = { scope:story = { has_variable = hp_damage } }
			custom_tooltip = pod_combat_desc_hp_damage_basic
		}

		# Morale / self damage
		if = {
			limit = { scope:story = { has_variable = morale_damage } }
			custom_tooltip = pod_combat_desc_morale_damage
		}
		if = {
			limit = { scope:story = { has_variable = self_damage } }
			custom_tooltip = pod_combat_desc_self_damage
		}

		# Healing
		# Standalone hp_heal (no drain — drain readout is inlined
		# on the damage line variants above when both are present).
		if = {
			limit = {
				scope:story = {
					has_variable = hp_heal
					NOT = { has_variable = drain }
				}
			}
			custom_tooltip = pod_combat_desc_hp_heal
		}
		# Standalone drain (no hp_damage to inline onto)
		if = {
			limit = {
				scope:story = {
					has_variable = drain
					NOT = { has_variable = hp_damage }
				}
			}
			custom_tooltip = pod_combat_desc_drain
		}
		if = {
			limit = { scope:story = { has_variable = morale_heal } }
			custom_tooltip = pod_combat_desc_morale_heal
		}

		# Tier buffs / debuffs
		if = {
			limit = { scope:story = { has_variable = soak_buff_round_tier } }
			custom_tooltip = pod_combat_desc_soak_buff_round
		}
		if = {
			limit = { scope:story = { has_variable = soak_buff_combat_tier } }
			custom_tooltip = pod_combat_desc_soak_buff_combat
		}
		if = {
			limit = { scope:story = { has_variable = soak_debuff_round_tier } }
			custom_tooltip = pod_combat_desc_soak_debuff_round
		}
		if = {
			limit = { scope:story = { has_variable = soak_debuff_combat_tier } }
			custom_tooltip = pod_combat_desc_soak_debuff_combat
		}
		if = {
			limit = { scope:story = { has_variable = hit_buff_round_tier } }
			custom_tooltip = pod_combat_desc_hit_buff_round
		}
		if = {
			limit = { scope:story = { has_variable = hit_buff_combat_tier } }
			custom_tooltip = pod_combat_desc_hit_buff_combat
		}
		if = {
			limit = { scope:story = { has_variable = hit_debuff_round_tier } }
			custom_tooltip = pod_combat_desc_hit_debuff_round
		}
		if = {
			limit = { scope:story = { has_variable = hit_debuff_combat_tier } }
			custom_tooltip = pod_combat_desc_hit_debuff_combat
		}

		# Stance flags
		if = {
			limit = { scope:story = { has_variable = set_guard } }
			custom_tooltip = pod_combat_desc_set_guard
		}
		if = {
			limit = { scope:story = { has_variable = dodge } }
			custom_tooltip = pod_combat_desc_dodge
		}
		if = {
			limit = { scope:story = { has_variable = stealth } }
			custom_tooltip = pod_combat_desc_stealth
		}
		if = {
			limit = { scope:story = { has_variable = remove_guard } }
			custom_tooltip = pod_combat_desc_remove_guard
		}

		# Bypass flags
		if = {
			limit = { scope:story = { has_variable = bypass_guard } }
			custom_tooltip = pod_combat_desc_bypass_guard
		}
		if = {
			limit = { scope:story = { has_variable = bypass_dodge } }
			custom_tooltip = pod_combat_desc_bypass_dodge
		}
		if = {
			limit = { scope:story = { has_variable = bypass_stealth } }
			custom_tooltip = pod_combat_desc_bypass_stealth
		}

		# Retaliation stances
		if = {
			limit = { scope:story = { has_variable = retaliate_if_hit } }
			custom_tooltip = pod_combat_desc_retaliate_if_hit
		}
		if = {
			limit = { scope:story = { has_variable = retaliate_if_missed } }
			custom_tooltip = pod_combat_desc_retaliate_if_missed
		}

		# AP / initiative parametric
		if = {
			limit = { scope:story = { has_variable = ap_gain } }
			custom_tooltip = pod_combat_desc_ap_gain
		}
		if = {
			limit = { scope:story = { has_variable = ap_drain } }
			custom_tooltip = pod_combat_desc_ap_drain
		}
		if = {
			limit = { scope:story = { has_variable = initiative_debuff } }
			custom_tooltip = pod_combat_desc_initiative_debuff
		}

		# Targeting hint (skip single_enemy default; render every other
		# targeting flag explicitly so AoE / self / ally / PC moves
		# advertise their scope).
		if = {
			limit = { scope:story = { var:targeting = flag:team_enemy } }
			custom_tooltip = pod_combat_desc_target_team_enemy
		}
		else_if = {
			limit = { scope:story = { var:targeting = flag:team_ally } }
			custom_tooltip = pod_combat_desc_target_team_ally
		}
		else_if = {
			limit = { scope:story = { var:targeting = flag:single_ally } }
			custom_tooltip = pod_combat_desc_target_single_ally
		}
		else_if = {
			limit = { scope:story = { var:targeting = flag:single_pc } }
			custom_tooltip = pod_combat_desc_target_single_pc
		}
		else_if = {
			limit = { scope:story = { var:targeting = flag:self } }
			custom_tooltip = pod_combat_desc_target_self
		}

		# Designer-facing flavor line from the CSV `notes` column is
		# NOT rendered here — the SGUI's BuildTooltip context can't
		# resolve Decision.GetCommandDescForPlayer (the data-binding
		# evaluator inside custom_tooltip doesn't expose Decision
		# methods on Var('decision')). Instead the flavor is appended
		# to the top-level POD_combat_move_tooltip / POD_combat_log_move_tooltip
		# loc strings via [AddLine( ... GetCommandDescForPlayer )],
		# matching DUEL-Refactor's POD_duel_move_tooltip pattern.
	}
}
