# Mid-combat move button. The save persists per-action state (slot
# _alive flips, soak / hit changes, log entries) back to the slot
# stories before the next turn evaluates.
pod_combat_execute_move = {
	scope = character
	saved_scopes = { story }

	is_shown = {
		exists = scope:story
		has_variable = pod_combat_active
		NOT = { has_variable = pod_combat_is_over }
	}

	is_valid = {
		pod_combat_is_active_trigger = yes
		pod_combat_is_player_turn_trigger = yes
		POD_can_execute_combat_move_trigger = {
			STORY = scope:story
			ACTOR = root
		}
	}

	effect = {
		POD_execute_combat_move = { STORY = scope:story }
		pod_combat_save_workspace_effect = yes
	}
}

# Close-skills button. The loadout window's `visible` binding watches
# pod_combat_config_open; removing it hides the window.
pod_combat_close_skills = {
	scope = character

	is_shown = { always = yes }
	is_valid = { always = yes }

	effect = {
		remove_variable = pod_combat_config_open
		# Sheet scratch list is editor-session state; it holds one story ref
		# per catalogue move, which would otherwise sit in the save. Rebuilt
		# on next open if sheet view is still selected.
		clear_variable_list = pod_combat_sheet_rows
	}
}

# Auto-advance toggle (the combat-window checkbox). Flips the toggling PLAYER's
# persistent preference pod_combat_manual_step_pref (present = manual stepping;
# absent = auto-advance, the default), then re-seeds the live fight's
# pod_combat_step_mode host var so the CURRENT combat responds immediately.
# Per-player, NOT a shared global: one player toggling never changes
# another player's live fight or default. The combat host IS the toggling PC
# (root = GetPlayer from the checkbox onclick), so the pref and the host
# step_mode var live on the same character.
pod_combat_toggle_auto_advance = {
	scope = character

	is_shown = {
		has_variable = pod_combat_active
	}

	is_valid = {
		pod_combat_is_active_trigger = yes
	}

	effect = {
		if = {
			limit = { has_variable = pod_combat_manual_step_pref }
			remove_variable = pod_combat_manual_step_pref
		}
		else = {
			set_variable = { name = pod_combat_manual_step_pref value = yes }
		}
		# Re-seed the live fight's step_mode from the now-updated pref.
		if = {
			limit = { has_variable = pod_combat_manual_step_pref }
			set_variable = { name = pod_combat_step_mode value = yes }
		}
		else = {
			remove_variable = pod_combat_step_mode
		}
	}
}

# Advance-one-AI-turn button (mid-step entry point). PROLOGUE-FREE by design:
# never marks the PC acted or saves the PC workspace, so it is safe to fire
# repeatedly while an AI is pending. The PC-commit prologue lives ONLY in
# pod_combat_process_group_turns_effect (via pod_combat_end_turn).
#
# Mode-aware so a mid-step toggle-to-auto drains on the next click; pc_has_acted
# routes the settle in both the STEP and AUTO branches below.
pod_combat_advance_turn = {
	scope = character

	is_shown = {
		has_variable = pod_combat_active
	}

	is_valid = {
		pod_combat_can_advance_trigger = yes
	}

	effect = {
		if = {
			limit = { has_variable = pod_combat_step_mode }
			pod_combat_advance_one_ai_turn_effect = yes
			if = {
				limit = { pod_combat_pc_has_acted_this_round_trigger = yes }
				pod_combat_after_step_settle_effect = yes
			}
			else = {
				pod_combat_after_pre_pc_step_settle_effect = yes
			}
		}
		else = {
			# AUTO. Route on pc_has_acted exactly like the STEP branch above and
			# pod_combat_process_group_turns_effect: PC already committed -> drain
			# the remaining post-PC AI, then end the round; PC NOT yet acted
			# (pre-PC phase, e.g. auto-advance toggled on while an AI leads off) ->
			# walk the AI up to the PC via the PC-stopping drain (which runs
			# setup_pc_turn inline). Calling run_remaining with the PC unacted hangs:
			# its while-limit stays true on the alive+unacted PC, which
			# advance_one_ai_turn skips without ever marking acted.
			if = {
				limit = { pod_combat_pc_has_acted_this_round_trigger = yes }
				pod_combat_run_remaining_ai_turns_effect = yes
				if = {
					limit = { NOT = { has_variable = pod_combat_is_over } }
					pod_combat_group_end_round_effect = yes
				}
			}
			else = {
				pod_combat_run_pre_pc_ai_turns_effect = yes
			}
		}
	}
}

# Equip / unequip a move. Unequip depends only on current loadout membership
# (always allowed for an equipped move); equip requires the move be unlocked
# (the dispatcher, not can_execute_decision) AND its category have
# room (cap = pod_combat_max_per_category). The GUI's `checked` binding reads
# is_shown = "currently equipped".
pod_combat_loadout_toggle = {
	scope = character
	saved_scopes = { story }

	is_shown = {
		exists = scope:story
		is_target_in_variable_list = {
			name   = pod_combat_loadout
			target = scope:story
		}
	}

	is_valid = {
		exists = scope:story
		OR = {
			# Equipped -> unequip depends ONLY on current loadout membership.
			is_target_in_variable_list = {
				name   = pod_combat_loadout
				target = scope:story
			}
			# Not equipped -> require the move be UNLOCKED and its category have room.
			AND = {
				# Unlock gate via the dispatcher, NOT can_execute_decision:
				# decisions carry the implicit ruler taker-requirement an unlanded PC
				# fails wholesale. Same unlock check the runtime uses; the generator
				# builds both surfaces from one gate-line fn, so they cannot drift.
				# COMBAT_REFERENCE.info §9 file map, pod_combat_move_decisions.txt
				# entry (unlock gating).
				pod_combat_actor_unlocks_move_trigger = { STORY = scope:story ACTOR = root }
				# Enforce the TOTAL equipped cap independently of the per-category
				# caps. Redundant today (4 x 4 = 16 == total), but a backstop if the caps
				# are retuned so categories no longer sum to pod_combat_max_equipped.
				# LOCKSTEP: the skills-window pip strips bake fixed pip counts, so retuning
				# the caps means editing those GUI literals too.
				pod_combat_equipped_count < pod_combat_max_equipped
				# Cap check per combat_category. custom_description surfaces a friendly
				# "Attack slots full: 4/4" tooltip; the trigger_if chain keeps it singular.
				trigger_if = {
					limit = { scope:story = { var:combat_category = flag:ATK } }
					custom_description = {
						text = pod_combat_loadout_atk_full_tt
						pod_combat_atk_count < pod_combat_max_per_category
					}
				}
				trigger_else_if = {
					limit = { scope:story = { var:combat_category = flag:DEF } }
					custom_description = {
						text = pod_combat_loadout_def_full_tt
						pod_combat_def_count < pod_combat_max_per_category
					}
				}
				trigger_else_if = {
					limit = { scope:story = { var:combat_category = flag:CTL } }
					custom_description = {
						text = pod_combat_loadout_ctl_full_tt
						pod_combat_ctl_count < pod_combat_max_per_category
					}
				}
				trigger_else_if = {
					limit = { scope:story = { var:combat_category = flag:PWR } }
					custom_description = {
						text = pod_combat_loadout_pwr_full_tt
						pod_combat_pwr_count < pod_combat_max_per_category
					}
				}
				trigger_else = {
					# Required terminator for the trigger_else_if chain.
					# always = no = don't equip an unknown-category move.
					always = no
				}
			}
		}
	}

	effect = {
		if = {
			limit = {
				is_target_in_variable_list = {
					name   = pod_combat_loadout
					target = scope:story
				}
			}
			remove_list_variable = {
				name   = pod_combat_loadout
				target = scope:story
			}
		}
		else = {
			add_to_variable_list = {
				name   = pod_combat_loadout
				target = scope:story
			}
		}
		# Keep the loadout-count display cache in step with the change so the pip
		# strips stay correct without re-walking the loadout per frame.
		pod_combat_refresh_loadout_counts_effect = yes
		# Equipping under the sheet's equipped-only filter changes the row
		# set itself, so rebuild; other filters are equip-agnostic.
		if = {
			limit = {
				has_variable = pod_combat_sheet_view
				has_variable = pod_combat_sheet_f_equipped
			}
			pod_combat_sheet_rebuild_effect = yes
		}
	}
}

# Per-row visibility: does the PC meet the move's unlock requirements? Reads the
# per-character pod_combat_unlocked_moves list (built when the editor opens by
# pod_combat_bake_pc_catalogue_unlocks_effect) instead of can_execute_decision
# per row per frame (a measured per-frame hotspot; COMBAT_PERF_NOTES.md). The show-all override forces every row
# visible; is_valid stays strict (show-all is a visibility override, not a
# permissions bypass). Compact view is independent: it only changes the row
# layout, not which moves are listed.
pod_combat_loadout_option = {
	scope = character
	saved_scopes = { story }

	is_shown = {
		exists = scope:story
		OR = {
			root = { is_target_in_variable_list = { name = pod_combat_unlocked_moves target = scope:story } }
			has_variable = pod_combat_show_all_moves
		}
	}

	is_valid = {
		exists = scope:story
		root = { is_target_in_variable_list = { name = pod_combat_unlocked_moves target = scope:story } }
	}
}

# "Show all moves" toggle - flips pod_combat_show_all_moves, which

		}
	}
}

# Combined filter check: yes if no filter is active, or the move's
# combat_category matches an active filter.
pod_combat_loadout_category_match = {
	scope = character
	saved_scopes = { story }

	is_shown = {
		exists = scope:story
		OR = {
			# No filter active - show everything.
			AND = {
				NOT = { has_variable = pod_combat_filter_atk }
				NOT = { has_variable = pod_combat_filter_def }
				NOT = { has_variable = pod_combat_filter_ctl }
				NOT = { has_variable = pod_combat_filter_pwr }
			}
			# Or filter active and matches.
			AND = {
				has_variable = pod_combat_filter_atk
				scope:story = { var:combat_category = flag:ATK }
			}
			AND = {
				has_variable = pod_combat_filter_def
				scope:story = { var:combat_category = flag:DEF }
			}
			AND = {
				has_variable = pod_combat_filter_ctl
				scope:story = { var:combat_category = flag:CTL }
			}
			AND = {

				limit = { scope:story = { pod_combat_move_targets_ally_trigger = yes } }
				custom_tooltip = pod_combat_desc_set_guard_other
			}
			else = { custom_tooltip = pod_combat_desc_set_guard }
		}
		if = {
			limit = { scope:story = { has_variable = dodge } }
			if = {
				limit = { scope:story = { pod_combat_move_targets_ally_trigger = yes } }
				custom_tooltip = pod_combat_desc_dodge_other
			}
			else = { custom_tooltip = pod_combat_desc_dodge }
		}
		if = {
			limit = { scope:story = { has_variable = stealth } }
			if = {
				limit = { scope:story = { pod_combat_move_targets_ally_trigger = yes } }
				custom_tooltip = pod_combat_desc_stealth_other
			}
			else = { custom_tooltip = pod_combat_desc_stealth }
		}
		if = {
			limit = { scope:story = { has_variable = remove_guard } }
			custom_tooltip = pod_combat_desc_remove_guard
		}

		# Retaliation stances. Variant picks the loc key from the runtime
		# fallback chain (explicit retaliate_damage > reuse hp_damage >
		# default 1) and the counter kind, matching the arming precedence
		# in pod_combat_arm_retaliation_effect: morale > retaliate_agg >
		# the move's elemental damage class > superficial. Bare
		# _if_hit/_missed are the no-amount fallback for the
		# designer-error case where neither amount var was set.
		# The elemental arms below (if_hit and if_missed alike) type the
		# counter off the move's own class, and damage_type is guaranteed
		# set once pod_combat_move_class_is_elemental_trigger passes.
		if = {
			limit = { scope:story = { has_variable = retaliate_if_hit } }
			if = {
				limit = { scope:story = { has_variable = retaliate_damage } }
				if = {
					limit = { scope:story = { has_variable = retaliate_morale } }
					custom_tooltip = pod_combat_desc_retaliate_if_hit_explicit_morale
				}
				else_if = {
					limit = { scope:story = { has_variable = retaliate_agg } }
					custom_tooltip = pod_combat_desc_retaliate_if_hit_explicit_agg
				}
				else_if = {
					limit = { scope:story = { pod_combat_move_class_is_elemental_trigger = yes } }
					if =      { limit = { scope:story = { var:damage_type = flag:fire } }      custom_tooltip = pod_combat_desc_retaliate_if_hit_explicit_fire }
					else_if = { limit = { scope:story = { var:damage_type = flag:cold } }      custom_tooltip = pod_combat_desc_retaliate_if_hit_explicit_cold }
					else_if = { limit = { scope:story = { var:damage_type = flag:lightning } } custom_tooltip = pod_combat_desc_retaliate_if_hit_explicit_lightning }
					else_if = { limit = { scope:story = { var:damage_type = flag:sun } }       custom_tooltip = pod_combat_desc_retaliate_if_hit_explicit_sun }
					else_if = { limit = { scope:story = { var:damage_type = flag:silver } }    custom_tooltip = pod_combat_desc_retaliate_if_hit_explicit_silver }
					else_if = { limit = { scope:story = { var:damage_type = flag:gold } }      custom_tooltip = pod_combat_desc_retaliate_if_hit_explicit_gold }
					else_if = { limit = { scope:story = { var:damage_type = flag:holy } }      custom_tooltip = pod_combat_desc_retaliate_if_hit_explicit_holy }
					else_if = { limit = { scope:story = { var:damage_type = flag:balefire } }  custom_tooltip = pod_combat_desc_retaliate_if_hit_explicit_balefire }
					else_if = { limit = { scope:story = { var:damage_type = flag:poison } }    custom_tooltip = pod_combat_desc_retaliate_if_hit_explicit_poison }
					else_if = { limit = { scope:story = { var:damage_type = flag:cold_iron } } custom_tooltip = pod_combat_desc_retaliate_if_hit_explicit_cold_iron }
				}
				else = {
					custom_tooltip = pod_combat_desc_retaliate_if_hit_explicit_sup
				}
			}
			else_if = {
				limit = { scope:story = { has_variable = hp_damage } }
				if = {
					limit = { scope:story = { has_variable = retaliate_agg } }
					custom_tooltip = pod_combat_desc_retaliate_if_hit_reuse_agg
				}
				else = {
					custom_tooltip = pod_combat_desc_retaliate_if_hit_reuse_sup
				}
			}
			else = {
				custom_tooltip = pod_combat_desc_retaliate_if_hit
			}
		}
		if = {
			limit = { scope:story = { has_variable = retaliate_if_missed } }
			if = {
				limit = { scope:story = { has_variable = retaliate_damage } }
				if = {
					limit = { scope:story = { has_variable = retaliate_morale } }
					custom_tooltip = pod_combat_desc_retaliate_if_missed_explicit_morale
				}
				else_if = {
					limit = { scope:story = { has_variable = retaliate_agg } }
					custom_tooltip = pod_combat_desc_retaliate_if_missed_explicit_agg
				}
				else_if = {
					limit = { scope:story = { pod_combat_move_class_is_elemental_trigger = yes } }
					if =      { limit = { scope:story = { var:damage_type = flag:fire } }      custom_tooltip = pod_combat_desc_retaliate_if_missed_explicit_fire }
					else_if = { limit = { scope:story = { var:damage_type = flag:cold } }      custom_tooltip = pod_combat_desc_retaliate_if_missed_explicit_cold }
					else_if = { limit = { scope:story = { var:damage_type = flag:lightning } } custom_tooltip = pod_combat_desc_retaliate_if_missed_explicit_lightning }
					else_if = { limit = { scope:story = { var:damage_type = flag:sun } }       custom_tooltip = pod_combat_desc_retaliate_if_missed_explicit_sun }
					else_if = { limit = { scope:story = { var:damage_type = flag:silver } }    custom_tooltip = pod_combat_desc_retaliate_if_missed_explicit_silver }
					else_if = { limit = { scope:story = { var:damage_type = flag:gold } }      custom_tooltip = pod_combat_desc_retaliate_if_missed_explicit_gold }
					else_if = { limit = { scope:story = { var:damage_type = flag:holy } }      custom_tooltip = pod_combat_desc_retaliate_if_missed_explicit_holy }
					else_if = { limit = { scope:story = { var:damage_type = flag:balefire } }  custom_tooltip = pod_combat_desc_retaliate_if_missed_explicit_balefire }
					else_if = { limit = { scope:story = { var:damage_type = flag:poison } }    custom_tooltip = pod_combat_desc_retaliate_if_missed_explicit_poison }
					else_if = { limit = { scope:story = { var:damage_type = flag:cold_iron } } custom_tooltip = pod_combat_desc_retaliate_if_missed_explicit_cold_iron }
				}
				else = {
					custom_tooltip = pod_combat_desc_retaliate_if_missed_explicit_sup
				}
			}
			else_if = {
				limit = { scope:story = { has_variable = hp_damage } }
				if = {
					limit = { scope:story = { has_variable = retaliate_agg } }
					custom_tooltip = pod_combat_desc_retaliate_if_missed_reuse_agg
				}
				else = {
					custom_tooltip = pod_combat_desc_retaliate_if_missed_reuse_sup
				}
			}
			else = {
				custom_tooltip = pod_combat_desc_retaliate_if_missed
			}
		}

		# AP / initiative parametric
		if = {
			limit = { scope:story = { has_variable = ap_gain } }
			custom_tooltip = pod_combat_desc_ap_gain
		}
		if = {
			limit = { scope:story = { has_variable = ap_drain } }
			custom_tooltip = pod_combat_desc_ap_drain
		}
		if = {
			limit = { scope:story = { has_variable = initiative_debuff } }
			custom_tooltip = pod_combat_desc_initiative_debuff
		}
		if = {
			limit = { scope:story = { has_variable = dot_rounds } }
			custom_tooltip = pod_combat_desc_dot
		}
		if = {
			limit = { scope:story = { has_variable = mind_control } }
			custom_tooltip = pod_combat_desc_mind_control
		}
		if = {
			limit = { scope:story = { has_variable = summons_combatant } }
			custom_tooltip = pod_combat_desc_summon
		}

		# Conditional bonus (cond_mode = flag:bonus), rendered below the base
		# effect. Two lines per move: the "Bonus if <X>:" condition, then the effect.
		if = {
			limit = { scope:story = { has_variable = cond_mode  var:cond_mode = flag:bonus } }
			# Condition line (per predicate, like the gate dispatch). Pairs with the
			# bonus effect line below to read "Bonus if <X>: <effect>".
			if      = { limit = { scope:story = { var:cond_predicate = flag:first_round } } custom_tooltip = pod_combat_desc_bonus_cond_first_round }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:round_min } } custom_tooltip = pod_combat_desc_bonus_cond_round_min }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:undamaged  var:cond_target = flag:user } } custom_tooltip = pod_combat_desc_bonus_cond_undamaged_user }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:undamaged  var:cond_target = flag:target } } custom_tooltip = pod_combat_desc_bonus_cond_undamaged_target }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:hp_missing  var:cond_target = flag:user } } custom_tooltip = pod_combat_desc_bonus_cond_hp_missing_user }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:hp_missing  var:cond_target = flag:target } } custom_tooltip = pod_combat_desc_bonus_cond_hp_missing_target }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:full_morale  var:cond_target = flag:user } } custom_tooltip = pod_combat_desc_bonus_cond_full_morale_user }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:full_morale  var:cond_target = flag:target } } custom_tooltip = pod_combat_desc_bonus_cond_full_morale_target }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:morale_missing  var:cond_target = flag:user } } custom_tooltip = pod_combat_desc_bonus_cond_morale_missing_user }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:morale_missing  var:cond_target = flag:target } } custom_tooltip = pod_combat_desc_bonus_cond_morale_missing_target }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:has_guard  var:cond_target = flag:user } } custom_tooltip = pod_combat_desc_bonus_cond_has_guard_user }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:has_guard  var:cond_target = flag:target } } custom_tooltip = pod_combat_desc_bonus_cond_has_guard_target }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:has_dodge  var:cond_target = flag:user } } custom_tooltip = pod_combat_desc_bonus_cond_has_dodge_user }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:has_dodge  var:cond_target = flag:target } } custom_tooltip = pod_combat_desc_bonus_cond_has_dodge_target }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:has_stealth  var:cond_target = flag:user } } custom_tooltip = pod_combat_desc_bonus_cond_has_stealth_user }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:has_stealth  var:cond_target = flag:target } } custom_tooltip = pod_combat_desc_bonus_cond_has_stealth_target }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:prestige_level_min  var:cond_target = flag:user } } custom_tooltip = pod_combat_desc_bonus_cond_prestige_level_min_user }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:prestige_level_min  var:cond_target = flag:target } } custom_tooltip = pod_combat_desc_bonus_cond_prestige_level_min_target }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:piety_level_min  var:cond_target = flag:user } } custom_tooltip = pod_combat_desc_bonus_cond_piety_level_min_user }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:piety_level_min  var:cond_target = flag:target } } custom_tooltip = pod_combat_desc_bonus_cond_piety_level_min_target }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:dread_min  var:cond_target = flag:user } } custom_tooltip = pod_combat_desc_bonus_cond_dread_min_user }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:dread_min  var:cond_target = flag:target } } custom_tooltip = pod_combat_desc_bonus_cond_dread_min_target }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:relation_positive } } custom_tooltip = pod_combat_desc_bonus_cond_relation_positive }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:relation_negative } } custom_tooltip = pod_combat_desc_bonus_cond_relation_negative }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:outnumbered } } custom_tooltip = pod_combat_desc_bonus_cond_outnumbered }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:ally_knocked_out } } custom_tooltip = pod_combat_desc_bonus_cond_ally_knocked_out }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:has_weapon_artifact  var:cond_target = flag:user } } custom_tooltip = pod_combat_desc_bonus_cond_has_weapon_artifact_user }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:has_weapon_artifact  var:cond_target = flag:target } } custom_tooltip = pod_combat_desc_bonus_cond_has_weapon_artifact_target }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:has_armor_artifact  var:cond_target = flag:user } } custom_tooltip = pod_combat_desc_bonus_cond_has_armor_artifact_user }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:has_armor_artifact  var:cond_target = flag:target } } custom_tooltip = pod_combat_desc_bonus_cond_has_armor_artifact_target }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:has_trait  var:cond_target = flag:user } } custom_tooltip = pod_combat_desc_bonus_cond_has_trait_user }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:has_trait  var:cond_target = flag:target } } custom_tooltip = pod_combat_desc_bonus_cond_has_trait_target }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:target_known_secret } } custom_tooltip = pod_combat_desc_bonus_cond_target_known_secret }
			if =      { limit = { scope:story = { var:bonus_effect = flag:self_heal } }     custom_tooltip = pod_combat_desc_bonus_self_heal }
			else_if = { limit = { scope:story = { var:bonus_effect = flag:heal_target } }   custom_tooltip = pod_combat_desc_bonus_heal_target }
			else_if = { limit = { scope:story = { var:bonus_effect = flag:morale_damage } } custom_tooltip = pod_combat_desc_bonus_morale_damage }
			# Bonus-damage mark: elemental classes show NO box (the main damage
			# line already names the element - mirrors PODCombatCLDamage), so
			# the elemental check must precede the always-aggravated one or
			# fire/lightning/balefire would take the agg box. After it, the only
			# always-aggravated class left is `aggravated` (box); superficial
			# shows its box.
			else_if = {
				limit = { scope:story = { var:bonus_effect = flag:damage  pod_combat_move_class_is_elemental_trigger = yes } }
				custom_tooltip = pod_combat_desc_bonus_damage_plain
			}
			else_if = {
				limit = { scope:story = { var:bonus_effect = flag:damage  pod_combat_class_always_aggravated_trigger = yes } }
				custom_tooltip = pod_combat_desc_bonus_damage_agg
			}
			else_if = { limit = { scope:story = { var:bonus_effect = flag:damage } }      custom_tooltip = pod_combat_desc_bonus_damage }
			else_if = { limit = { scope:story = { var:bonus_effect = flag:change_type } } custom_tooltip = pod_combat_desc_bonus_change_type }
			else_if = { limit = { scope:story = { var:bonus_effect = flag:hit } }         custom_tooltip = pod_combat_desc_bonus_hit }
			else_if = { limit = { scope:story = { var:bonus_effect = flag:ap_refund } }     custom_tooltip = pod_combat_desc_bonus_ap_refund }
			else_if = { limit = { scope:story = { var:bonus_effect = flag:stance_guard } }   custom_tooltip = pod_combat_desc_bonus_stance_guard }
			else_if = { limit = { scope:story = { var:bonus_effect = flag:stance_dodge } }   custom_tooltip = pod_combat_desc_bonus_stance_dodge }
			else_if = { limit = { scope:story = { var:bonus_effect = flag:stance_stealth } } custom_tooltip = pod_combat_desc_bonus_stance_stealth }
			else_if = { limit = { scope:story = { var:bonus_effect = flag:ap_drain } }      custom_tooltip = pod_combat_desc_bonus_ap_drain }
			else_if = { limit = { scope:story = { var:bonus_effect = flag:energy_gain } }  custom_tooltip = pod_combat_desc_bonus_energy_gain }
			else_if = { limit = { scope:story = { var:bonus_effect = flag:energy_drain } } custom_tooltip = pod_combat_desc_bonus_energy_drain }
			else_if = { limit = { scope:story = { var:bonus_effect = flag:dot } }          custom_tooltip = pod_combat_desc_bonus_dot }
		}
		}
	}
}
