# Mid-combat move button. The save persists per-action state (slot
# _alive flips, soak / hit changes, log entries) back to the slot
# stories before the next turn evaluates.
pod_combat_execute_move = {
	scope = character
	saved_scopes = { story }

	is_shown = {
		exists = scope:story
		has_variable = pod_combat_active
		NOT = { has_variable = pod_combat_is_over }
	}

	is_valid = {
		pod_combat_is_active_trigger = yes
		pod_combat_is_player_turn_trigger = yes
		POD_can_execute_combat_move = {
			STORY = scope:story
			ACTOR = root
		}
	}

	effect = {
		POD_execute_combat_move = { STORY = scope:story }
		pod_combat_save_workspace_effect = yes
	}
}

# X button in the combat-window header. Two behaviours behind one
# target: combat in progress → concede; combat resolved → close the
# window (else the X no-ops post-combat and reads as a stuck window).
pod_combat_close_window = {
	scope = character

	is_shown = {
		has_variable = pod_combat_active
	}

	is_valid = {
		has_variable = pod_combat_active
	}

	effect = {
		if = {
			limit = { has_variable = pod_combat_is_over }
			pod_combat_end_combat_effect = yes
		}
		else = {
			# Concede: PC = loser, the first ALIVE enemy = winner (the enemy
			# leader may already be KO'd - C16). Atomic winner+loser pair (C17).
			# is_over flip hands off to the bottom-bar End Combat for cleanup.
			pod_combat_set_outcome_pair_effect = { PC_WON = no }
			# Conceding side = loser: zero its leader slot_morale so the portrait
			# reads defeated and the leader accrues full Mental Strain at combat
			# end (same shared slot-find as the auto-resolve path).
			pod_combat_zero_loser_leader_morale_effect = yes
			set_variable = { name = pod_combat_is_over value = yes }
			pod_combat_push_system_log_effect = { MESSAGE_KEY = combat_ended_defeat }
		}
	}
}

# Auto-advance toggle (the combat-window checkbox). Flips the persistent
# preference global_var:pod_combat_auto_advance (present = auto = the
# all-at-once behavior), then re-seeds the live fight's pod_combat_step_mode
# host var to match so the CURRENT combat responds immediately. The preference
# (a global) and the host re-seed (a per-fight var) are distinct, so both are
# written explicitly on both branches.
pod_combat_toggle_auto_advance = {
	scope = character

	is_shown = {
		has_variable = pod_combat_active
	}

	is_valid = {
		pod_combat_is_active_trigger = yes
	}

	effect = {
		if = {
			limit = { has_global_variable = pod_combat_auto_advance }
			remove_global_variable = pod_combat_auto_advance
		}
		else = {
			set_global_variable = { name = pod_combat_auto_advance value = yes }
		}
		# Re-seed the host var from the now-updated preference.
		if = {
			limit = { has_global_variable = pod_combat_auto_advance }
			remove_variable = pod_combat_step_mode
		}
		else = {
			set_variable = { name = pod_combat_step_mode value = yes }
		}
	}
}

# Advance-one-AI-turn button (mid-step entry point). PROLOGUE-FREE by design:
# never marks the PC acted or saves the PC workspace, so it is safe to fire
# repeatedly while an AI is pending. The PC-commit prologue lives ONLY in
# pod_combat_process_group_turns_effect (via pod_combat_end_turn).
#
# Mode-aware so a mid-step toggle-to-auto drains on the next click; pc_has_acted
# routes the settle in both the STEP and AUTO branches below.
pod_combat_advance_turn = {
	scope = character

	is_shown = {
		has_variable = pod_combat_active
	}

	is_valid = {
		pod_combat_can_advance_trigger = yes
	}

	effect = {
		if = {
			limit = { has_variable = pod_combat_step_mode }
			pod_combat_advance_one_ai_turn_effect = yes
			if = {
				limit = { pod_combat_pc_has_acted_this_round_trigger = yes }
				pod_combat_after_step_settle_effect = yes
			}
			else = {
				pod_combat_after_pre_pc_step_settle_effect = yes
			}
		}
		else = {
			# AUTO. Route on pc_has_acted exactly like the STEP branch above and
			# pod_combat_process_group_turns_effect: PC already committed -> drain
			# the remaining post-PC AI, then end the round; PC NOT yet acted
			# (pre-PC phase, e.g. auto-advance toggled on while an AI leads off) ->
			# walk the AI up to the PC via the PC-stopping drain (which runs
			# setup_pc_turn inline). Calling run_remaining with the PC unacted hangs:
			# its while-limit stays true on the alive+unacted PC, which
			# advance_one_ai_turn skips without ever marking acted. (C01 fix.)
			if = {
				limit = { pod_combat_pc_has_acted_this_round_trigger = yes }
				pod_combat_run_remaining_ai_turns_effect = yes
				if = {
					limit = { NOT = { has_variable = pod_combat_is_over } }
					pod_combat_group_end_round_effect = yes
				}
			}
			else = {
				pod_combat_run_pre_pc_ai_turns_effect = yes
			}
		}
	}
}

# Equip / unequip a move. Unequip depends only on current loadout membership
# (always allowed for an equipped move); equip requires the move be unlocked
# (the dispatcher, not can_execute_decision) AND its category have
# room (cap = pod_combat_max_per_category). The GUI's `checked` binding reads
# is_shown = "currently equipped".
pod_combat_loadout_toggle = {
	scope = character
	saved_scopes = { story }

	is_shown = {
		exists = scope:story
		is_target_in_variable_list = {
			name   = pod_combat_loadout
			target = scope:story
		}
	}

	is_valid = {
		exists = scope:story
		OR = {
			# Equipped -> unequip depends ONLY on current loadout membership.
			is_target_in_variable_list = {
				name   = pod_combat_loadout
				target = scope:story
			}
			# Not equipped -> require the move be UNLOCKED and its category have room.
			AND = {
				# Unlock gate via the dispatcher, NOT can_execute_decision:
				# decisions carry the implicit ruler taker-requirement an unlanded PC
				# fails wholesale. Same unlock check the runtime uses; the generator
				# builds both surfaces from one gate-line fn, so they cannot drift.
				# COMBAT_REFERENCE.info §7 (unlock gating).
				pod_combat_actor_unlocks_move_trigger = { STORY = scope:story ACTOR = root }
				# Enforce the TOTAL equipped cap independently of the per-category
				# caps. Redundant today (4 x 4 = 16 == total), but a backstop if the caps
				# are retuned so categories no longer sum to pod_combat_max_equipped.
				# LOCKSTEP: the skills-window pip strips bake fixed pip counts, so retuning
				# the caps means editing those GUI literals too.
				pod_combat_equipped_count < pod_combat_max_equipped
				# Cap check per combat_category. custom_description surfaces a friendly
				# "Attack slots full: 4/4" tooltip; the trigger_if chain keeps it singular.
				trigger_if = {
					limit = { scope:story = { var:combat_category = flag:ATK } }
					custom_description = {
						text = pod_combat_loadout_atk_full_tt
						pod_combat_atk_count < pod_combat_max_per_category
					}
				}
				trigger_else_if = {
					limit = { scope:story = { var:combat_category = flag:DEF } }
					custom_description = {
						text = pod_combat_loadout_def_full_tt
						pod_combat_def_count < pod_combat_max_per_category
					}
				}
				trigger_else_if = {
					limit = { scope:story = { var:combat_category = flag:CTL } }
					custom_description = {
						text = pod_combat_loadout_ctl_full_tt
						pod_combat_ctl_count < pod_combat_max_per_category
					}
				}
				trigger_else_if = {
					limit = { scope:story = { var:combat_category = flag:PWR } }
					custom_description = {
						text = pod_combat_loadout_pwr_full_tt
						pod_combat_pwr_count < pod_combat_max_per_category
					}
				}
				trigger_else = {
					# Required terminator for the trigger_else_if chain.
					# always = no = don't equip an unknown-category move.
					always = no
				}
			}
		}
	}

	effect = {
		if = {
			limit = {
				is_target_in_variable_list = {
					name   = pod_combat_loadout
					target = scope:story
				}
			}
			remove_list_variable = {
				name   = pod_combat_loadout
				target = scope:story
			}
		}
		else = {
			add_to_variable_list = {
				name   = pod_combat_loadout
				target = scope:story
			}
		}
		# Keep the loadout-count display cache in step with the change so the pip
		# strips stay correct without re-walking the loadout per frame.
		pod_combat_refresh_loadout_counts_effect = yes
	}
}

# Per-row visibility: does the PC meet the move's unlock requirements? Reads the
# baked pc_can_use flag (set when the editor opens by
# pod_combat_bake_pc_catalogue_unlocks_effect) instead of can_execute_decision
# per row per frame - G06 perf fix. The show-all override forces every row
# visible; is_valid stays strict (show-all is a visibility override, not a
# permissions bypass). Compact view is independent: it only changes the row
# layout, not which moves are listed.
pod_combat_loadout_option = {
	scope = character
	saved_scopes = { story }

	is_shown = {
		exists = scope:story
		OR = {
			scope:story = { has_variable = pc_can_use }
			has_variable = show_all_combat_moves
		}
	}

	is_valid = {
		exists = scope:story
		scope:story = { has_variable = pc_can_use }
	}
}

# Combined filter check: yes if no filter is active, or the move's
# combat_category matches an active filter.
pod_combat_loadout_category_match = {
	scope = character
	saved_scopes = { story }

	is_shown = {
		exists = scope:story
		OR = {
			# No filter active — show everything.
			AND = {
				NOT = { has_variable = pod_combat_filter_atk }
				NOT = { has_variable = pod_combat_filter_def }
				NOT = { has_variable = pod_combat_filter_ctl }
				NOT = { has_variable = pod_combat_filter_pwr }
			}
			# Or filter active and matches.
			AND = {
				has_variable = pod_combat_filter_atk
				scope:story = { var:combat_category = flag:ATK }
			}
			AND = {
				has_variable = pod_combat_filter_def
				scope:story = { var:combat_category = flag:DEF }
			}
			AND = {
				has_variable = pod_combat_filter_ctl
				scope:story = { var:combat_category = flag:CTL }
			}
			AND = {
				has_variable = pod_combat_filter_pwr
				scope:story = { var:combat_category = flag:PWR }
			}
		}
	}
}

# Dynamic per-attribute stat block, BuildTooltip-rendered from the
# PODCombatMoveEffect macro. The macro saves the move's story_cycle as
# the GuiScope scope `story`; each pod_combat_desc_X loc reads its vars
# via [story.MakeScope.Var('Y')...]. AP cost lives in the move title,
# not here, so this block starts at resource costs.
#
# Dispatch order: costs → HP damage (per damage_source × drain) →
# morale/self → healing → capability callouts → stance-strip → tier
# buffs/debuffs → stance flags → retaliation → AP/initiative.
pod_combat_move_desc = {
	scope = character
	saved_scopes = { story }
	effect = {
		# Combined cost line. The OR-of-five gate suppresses the
		# dispatcher (and its dangling "Costs:" prefix) when no cost
		# var is set.
		#
		# PARSER QUIRK: do NOT collapse to a single SelectLocalization
		# call. The data-binding parser treats `#` as a comment marker
		# and rejects nested `[X]` substitutions inside function-arg
		# strings, so the `#bold`/`[Var]` cost tokens must live in a
		# customizable_localization loc string, not a function arg.
		if = {
			limit = {
				scope:story = {
					OR = {
						has_variable = energy_cost
						has_variable = gold_cost
						has_variable = piety_cost
						has_variable = prestige_cost
						has_variable = stress_cost
					}
				}
			}
			custom_tooltip = pod_combat_desc_costs
		}
		# Phase 4 PC-only resource gains (steal_gold transfers, gain_dread adds).
		if = {
			limit = { scope:story = { has_variable = steal_gold } }
			custom_tooltip = pod_combat_desc_steal_gold
		}
		if = {
			limit = { scope:story = { has_variable = gain_dread } }
			custom_tooltip = pod_combat_desc_gain_dread
		}

		# State-condition gate requirement (cond_mode = flag:gate). One Requires:
		# line, dispatched per predicate. 
		if = {
			limit = { scope:story = { has_variable = cond_mode  var:cond_mode = flag:gate } }
			if = { limit = { scope:story = { var:cond_predicate = flag:first_round } } custom_tooltip = pod_combat_desc_cond_first_round }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:round_min } } custom_tooltip = pod_combat_desc_cond_round_min }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:undamaged  var:cond_target = flag:user } } custom_tooltip = pod_combat_desc_cond_undamaged_user }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:undamaged  var:cond_target = flag:target } } custom_tooltip = pod_combat_desc_cond_undamaged_target }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:hp_missing  var:cond_target = flag:user } } custom_tooltip = pod_combat_desc_cond_hp_missing_user }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:hp_missing  var:cond_target = flag:target } } custom_tooltip = pod_combat_desc_cond_hp_missing_target }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:full_morale  var:cond_target = flag:user } } custom_tooltip = pod_combat_desc_cond_full_morale_user }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:full_morale  var:cond_target = flag:target } } custom_tooltip = pod_combat_desc_cond_full_morale_target }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:morale_missing  var:cond_target = flag:user } } custom_tooltip = pod_combat_desc_cond_morale_missing_user }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:morale_missing  var:cond_target = flag:target } } custom_tooltip = pod_combat_desc_cond_morale_missing_target }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:has_guard  var:cond_target = flag:user } } custom_tooltip = pod_combat_desc_cond_has_guard_user }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:has_guard  var:cond_target = flag:target } } custom_tooltip = pod_combat_desc_cond_has_guard_target }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:has_dodge  var:cond_target = flag:user } } custom_tooltip = pod_combat_desc_cond_has_dodge_user }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:has_dodge  var:cond_target = flag:target } } custom_tooltip = pod_combat_desc_cond_has_dodge_target }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:has_stealth  var:cond_target = flag:user } } custom_tooltip = pod_combat_desc_cond_has_stealth_user }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:has_stealth  var:cond_target = flag:target } } custom_tooltip = pod_combat_desc_cond_has_stealth_target }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:prestige_level_min  var:cond_target = flag:user } } custom_tooltip = pod_combat_desc_cond_prestige_level_min_user }
			else_if = { limit = { scope:story = { var:cond_predicate = flag:prestige_level_min  var:cond_target = flag:target } } custom_tooltip = pod_combat_desc_cond_prestige_level_min_target }

			else_if = { limit = { scope:story = { var:bonus_effect = flag:damage } }      custom_tooltip = pod_combat_desc_bonus_damage }
			else_if = { limit = { scope:story = { var:bonus_effect = flag:change_type } } custom_tooltip = pod_combat_desc_bonus_change_type }
			else_if = { limit = { scope:story = { var:bonus_effect = flag:hit } }         custom_tooltip = pod_combat_desc_bonus_hit }
			else_if = { limit = { scope:story = { var:bonus_effect = flag:ap_refund } }     custom_tooltip = pod_combat_desc_bonus_ap_refund }
			else_if = { limit = { scope:story = { var:bonus_effect = flag:stance_guard } }   custom_tooltip = pod_combat_desc_bonus_stance_guard }
			else_if = { limit = { scope:story = { var:bonus_effect = flag:stance_dodge } }   custom_tooltip = pod_combat_desc_bonus_stance_dodge }
			else_if = { limit = { scope:story = { var:bonus_effect = flag:stance_stealth } } custom_tooltip = pod_combat_desc_bonus_stance_stealth }
			else_if = { limit = { scope:story = { var:bonus_effect = flag:ap_drain } }      custom_tooltip = pod_combat_desc_bonus_ap_drain }
		}
	}
}
