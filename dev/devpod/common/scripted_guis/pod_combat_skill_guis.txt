# Mid-combat move button. The save persists per-action state (slot
# _alive flips, soak / hit changes, log entries) back to the slot
# stories before the next turn evaluates.
pod_combat_execute_move = {
	scope = character
	saved_scopes = { story }

	is_shown = {
		exists = scope:story
		has_variable = pod_combat_active
		NOT = { has_variable = pod_combat_is_over }
	}

	is_valid = {
		pod_combat_is_active_trigger = yes
		pod_combat_is_player_turn_trigger = yes
		POD_can_execute_combat_move = {
			STORY = scope:story
			ACTOR = root
		}
	}

	effect = {
		POD_execute_combat_move = { STORY = scope:story }
		pod_combat_save_workspace_effect = yes
	}
}

# X button in the combat-window header. Two behaviours behind one
# target: combat in progress → concede; combat resolved → close the
# window (else the X no-ops post-combat and reads as a stuck window).
pod_combat_close_window = {
	scope = character

	is_shown = {
		has_variable = pod_combat_active
	}

	is_valid = {
		has_variable = pod_combat_active
	}

	effect = {
		if = {
			limit = { has_variable = pod_combat_is_over }
			pod_combat_end_combat_effect = yes
		}
		else = {
			# Concede inline: PC = loser unconditionally, enemy/0's
			# char wins. is_over flip hands off to the bottom-bar End
			# Combat for the standard cleanup path.
			save_scope_as = combat_pc
			if = {
				limit = {
					has_variable = pod_combat_enemy_leader_slot
					var:pod_combat_enemy_leader_slot = { has_variable = slot_char }
				}
				set_variable = {
					name  = pod_combat_winner
					value = var:pod_combat_enemy_leader_slot.var:slot_char
				}
			}
			set_variable = { name = pod_combat_loser value = scope:combat_pc }
			set_variable = { name = pod_combat_is_over value = yes }
			pod_combat_push_system_log_effect = { MESSAGE_KEY = combat_ended_defeat }
		}
	}
}

# Equip / unequip a move. Unequip is always allowed; equip only if
# the move's category has room (cap = pod_combat_max_per_category).
# The GUI's `checked` binding reads is_shown = "currently equipped".
pod_combat_loadout_toggle = {
	scope = character
	saved_scopes = { story }

	is_shown = {
		exists = scope:story
		is_target_in_variable_list = {
			name   = pod_combat_loadout
			target = scope:story
		}
	}

	is_valid = {
		exists = scope:story
		can_execute_decision = scope:story.var:decision
		OR = {
			# Equipped — unequip always allowed.
			is_target_in_variable_list = {
				name   = pod_combat_loadout
				target = scope:story
			}
			# Not equipped — cap check per combat_category. custom_description
			# surfaces a friendly "Attack slots full: 4/4" tooltip instead
			# of the raw trigger; the trigger_if chain keeps it singular.
			trigger_if = {
				limit = { scope:story = { var:combat_category = flag:ATK } }
				custom_description = {
					text = pod_combat_loadout_atk_full_tt
					pod_combat_atk_count < pod_combat_max_per_category
				}
			}
			trigger_else_if = {
				limit = { scope:story = { var:combat_category = flag:DEF } }
				custom_description = {
					text = pod_combat_loadout_def_full_tt
					pod_combat_def_count < pod_combat_max_per_category
				}
			}
			trigger_else_if = {
				limit = { scope:story = { var:combat_category = flag:CTL } }
				custom_description = {
					text = pod_combat_loadout_ctl_full_tt
					pod_combat_ctl_count < pod_combat_max_per_category
				}
			}
			trigger_else_if = {
				limit = { scope:story = { var:combat_category = flag:PWR } }
				custom_description = {
					text = pod_combat_loadout_pwr_full_tt
					pod_combat_pwr_count < pod_combat_max_per_category
				}
			}
			trigger_else = {
				# Required terminator for the trigger_else_if chain.
				# always = no = don't equip an unknown-category move.
				always = no
			}
		}
	}

	effect = {
		if = {
			limit = {
				is_target_in_variable_list = {
					name   = pod_combat_loadout
					target = scope:story
				}
			}
			remove_list_variable = {
				name   = pod_combat_loadout
				target = scope:story
			}
		}
		else = {
			add_to_variable_list = {
				name   = pod_combat_loadout
				target = scope:story
			}
		}
	}
}

# Per-row visibility: does the PC meet the move's unlock requirements
# (can_execute_decision on the hidden decision)? Show-all / compact
# overrides force every row visible. is_valid stays strict — show-all
# is a visibility override, not a permissions bypass.
pod_combat_loadout_option = {
	scope = character
	saved_scopes = { story }

	is_shown = {
		exists = scope:story
		OR = {
			can_execute_decision = scope:story.var:decision
			has_variable = show_all_combat_moves
			# Compact view implies show-all.
			has_variable = pod_combat_loadout_compact_view
		}
	}

	is_valid = {
		exists = scope:story
		can_execute_decision = scope:story.var:decision
	}
}

# "Show all moves" toggle — flips show_all_combat_moves, which
# pod_combat_loadout_option's is_shown reads to bypass the unlock gate.
pod_combat_loadout_show_all_moves = {
	scope = character

	is_shown = { always = yes }
	is_valid = { always = yes }

	effect = {
		if = {
			limit = { has_variable = show_all_combat_moves }
			remove_variable = show_all_combat_moves
		}
		else = {
			set_variable = show_all_combat_moves
		}
	}
}

# Combined filter check: yes if no filter is active, or the move's
# combat_category matches an active filter.
pod_combat_loadout_category_match = {
	scope = character
	saved_scopes = { story }

	is_shown = {
		exists = scope:story
		OR = {
			# No filter active — show everything.
			AND = {
				NOT = { has_variable = pod_combat_filter_atk }
				NOT = { has_variable = pod_combat_filter_def }
				NOT = { has_variable = pod_combat_filter_ctl }
				NOT = { has_variable = pod_combat_filter_pwr }
			}
			# Or filter active and matches.
			AND = {
				has_variable = pod_combat_filter_atk
				scope:story = { var:combat_category = flag:ATK }
			}
			AND = {
				has_variable = pod_combat_filter_def
				scope:story = { var:combat_category = flag:DEF }
			}
			AND = {
				has_variable = pod_combat_filter_ctl
				scope:story = { var:combat_category = flag:CTL }
			}
			AND = {
				has_variable = pod_combat_filter_pwr
				scope:story = { var:combat_category = flag:PWR }
			}
		}
	}
}

# Dynamic per-attribute stat block, BuildTooltip-rendered from the
# PODCombatMoveEffect macro. The macro saves the move's story_cycle as
# the GuiScope scope `story`; each pod_combat_desc_X loc reads its vars
# via [story.MakeScope.Var('Y')...]. AP cost lives in the move title,
# not here, so this block starts at resource costs.
#
# Dispatch order: costs → HP damage (per damage_source × drain) →
# morale/self → healing → capability callouts → stance-strip → tier
# buffs/debuffs → stance flags → retaliation → AP/initiative.
pod_combat_move_desc = {
	scope = character
	saved_scopes = { story }
	effect = {
		# Combined cost line. The OR-of-five gate suppresses the
		# dispatcher (and its dangling "Costs:" prefix) when no cost
		# var is set.
		#
		# PARSER QUIRK: do NOT collapse to a single SelectLocalization
		# call. The data-binding parser treats `#` as a comment marker
		# and rejects nested `[X]` substitutions inside function-arg
		# strings, so the `#bold`/`[Var]` cost tokens must live in a
		# customizable_localization loc string, not a function arg.
		if = {
			limit = {
				scope:story = {
					OR = {
						has_variable = energy_cost
						has_variable = gold_cost
						has_variable = piety_cost
						has_variable = prestige_cost
						has_variable = stress_cost
					}
				}
			}
			custom_tooltip = pod_combat_desc_costs
		}

		# HP damage. Most-specific first: drain variants, then plain.
		if = {
			limit = {
				scope:story = {
					has_variable = hp_damage
					has_variable = drain
					has_variable = damage_source
					var:damage_source = flag:fire
				}
			}
			custom_tooltip = pod_combat_desc_hp_damage_fire_drain
		}
		else_if = {
			limit = {
				scope:story = {
					has_variable = hp_damage
					has_variable = drain
					has_variable = damage_source
					var:damage_source = flag:cold
				}
			}
			custom_tooltip = pod_combat_desc_hp_damage_cold_drain
		}
		else_if = {

					var:damage_source = flag:lightning

		else_if = {

					has_variable = drain
					has_variable = damage_source
					var:damage_source = flag:gold
				}
			}
			custom_tooltip = pod_combat_desc_hp_damage_gold_drain
		}
		else_if = {
			limit = {
				scope:story = {
					has_variable = hp_damage
					has_variable = drain
					has_variable = damage_source
					var:damage_source = flag:holy
				}
			}
			custom_tooltip = pod_combat_desc_hp_damage_holy_drain
		}
		else_if = {
			limit = {
				scope:story = {
					has_variable = hp_damage
					has_variable = drain
					has_variable = damage_source
					var:damage_source = flag:balefire

					has_variable = hp_damage
					has_variable = drain
				}
			}
			custom_tooltip = pod_combat_desc_hp_damage_basic_drain

					has_variable = damage_source
					var:damage_source = flag:fire

		else_if = {

					has_variable = damage_source
					var:damage_source = flag:sun
				}
			}
			custom_tooltip = pod_combat_desc_hp_damage_sun
		}
		else_if = {
			limit = {
				scope:story = {
					has_variable = hp_damage
					has_variable = damage_source
					var:damage_source = flag:silver
				}
			}
			custom_tooltip = pod_combat_desc_hp_damage_silver
		}
		else_if = {
			limit = {
				scope:story = {
					has_variable = hp_damage
					has_variable = damage_source
					var:damage_source = flag:gold
				}
			}
			custom_tooltip = pod_combat_desc_hp_damage_gold
		}
		else_if = {
			limit = {
				scope:story = {
					has_variable = hp_damage
					has_variable = damage_source
					var:damage_source = flag:holy
				}
			}
			custom_tooltip = pod_combat_desc_hp_damage_holy
		}
		else_if = {
			limit = {
				scope:story = {
					has_variable = hp_damage
					has_variable = damage_source
					var:damage_source = flag:balefire
				}
			}
			custom_tooltip = pod_combat_desc_hp_damage_balefire
		}
		else_if = {
			limit = { scope:story = { has_variable = hp_damage } }
			custom_tooltip = pod_combat_desc_hp_damage_basic
		}

		# Morale / self damage
		if = {
			limit = { scope:story = { has_variable = morale_damage } }
			custom_tooltip = pod_combat_desc_morale_damage
		}
		if = {
			limit = { scope:story = { has_variable = self_damage } }
			custom_tooltip = pod_combat_desc_self_damage
		}

		# Healing
		# Standalone hp_heal (no drain — drain readout is inlined
		# on the damage line variants above when both are present).
		if = {
			limit = {
				scope:story = {
					has_variable = hp_heal
					NOT = { has_variable = drain }
				}
			}
			custom_tooltip = pod_combat_desc_hp_heal
		}
		# Standalone drain (no hp_damage to inline onto)
		if = {
			limit = {
				scope:story = {
					has_variable = drain
					NOT = { has_variable = hp_damage }
				}
			}
			custom_tooltip = pod_combat_desc_drain
		}
		if = {
			limit = { scope:story = { has_variable = morale_heal } }
			custom_tooltip = pod_combat_desc_morale_heal
		}

		# Capability callouts. Each fires when its flag is ABSENT —
		# capability flags are present-when-true, so missing
		# requires_hit_roll means the move auto-hits. Gated on hp_damage
		# so non-damage moves don't surface contact-resolver properties.
		if = {
			limit = {
				scope:story = {
					has_variable = hp_damage
					NOT = { has_variable = requires_hit_roll }
				}
			}
			custom_tooltip = pod_combat_desc_always_hits
		}
		if = {
			limit = {
				scope:story = {
					has_variable = hp_damage
					NOT = { has_variable = checks_guard }
				}
			}
			custom_tooltip = pod_combat_desc_bypass_guard
		}
		if = {
			limit = {
				scope:story = {
					has_variable = hp_damage
					NOT = { has_variable = checks_stance }
				}
			}
			custom_tooltip = pod_combat_desc_undodgeable
		}
		if = {
			limit = {
				scope:story = {
					has_variable = hp_damage
					NOT = { has_variable = checks_spirit_gate }
				}
			}
			custom_tooltip = pod_combat_desc_hits_spirits
		}
		# Stance-strip flags. bypass_dodge / bypass_stealth CONSUME the
		# target's stance pre-damage (move_effects §5), unlike the
		# Undodgeable callout which only ignores it — so a follow-up
		# move from another actor also catches the now-stanceless target.
		if = {
			limit = { scope:story = { has_variable = bypass_dodge } }
			custom_tooltip = pod_combat_desc_bypass_dodge
		}
		if = {
			limit = { scope:story = { has_variable = bypass_stealth } }
			custom_tooltip = pod_combat_desc_bypass_stealth
		}

		# Tier buffs / debuffs
		if = {
			limit = { scope:story = { has_variable = soak_buff_round_tier } }
			custom_tooltip = pod_combat_desc_soak_buff_round
		}
		if = {
			limit = { scope:story = { has_variable = soak_buff_combat_tier } }
			custom_tooltip = pod_combat_desc_soak_buff_combat
		}
		if = {
			limit = { scope:story = { has_variable = soak_debuff_round_tier } }
			custom_tooltip = pod_combat_desc_soak_debuff_round
		}
		if = {
			limit = { scope:story = { has_variable = soak_debuff_combat_tier } }
			custom_tooltip = pod_combat_desc_soak_debuff_combat
		}
		if = {
			limit = { scope:story = { has_variable = hit_buff_round_tier } }
			custom_tooltip = pod_combat_desc_hit_buff_round
		}
		if = {
			limit = { scope:story = { has_variable = hit_buff_combat_tier } }
			custom_tooltip = pod_combat_desc_hit_buff_combat
		}
		if = {
			limit = { scope:story = { has_variable = hit_debuff_round_tier } }
			custom_tooltip = pod_combat_desc_hit_debuff_round
		}
		if = {
			limit = { scope:story = { has_variable = hit_debuff_combat_tier } }
			custom_tooltip = pod_combat_desc_hit_debuff_combat
		}

		# Stance flags
		if = {
			limit = { scope:story = { has_variable = set_guard } }
			custom_tooltip = pod_combat_desc_set_guard
		}
		if = {
			limit = { scope:story = { has_variable = dodge } }
			custom_tooltip = pod_combat_desc_dodge
		}
		if = {
			limit = { scope:story = { has_variable = stealth } }
			custom_tooltip = pod_combat_desc_stealth
		}
		if = {
			limit = { scope:story = { has_variable = remove_guard } }
			custom_tooltip = pod_combat_desc_remove_guard
		}

		# Retaliation stances. Variant picks the loc key from the runtime
		# fallback chain (explicit retaliate_damage > reuse hp_damage >
		# default 1) and the retaliate_agg flag (aggravated vs
		# superficial). Bare _if_hit/_missed are the no-amount fallback
		# for the designer-error case where neither amount var was set.
		if = {
			limit = { scope:story = { has_variable = retaliate_if_hit } }
			if = {
				limit = { scope:story = { has_variable = retaliate_damage } }
				if = {
					limit = { scope:story = { has_variable = retaliate_morale } }
					custom_tooltip = pod_combat_desc_retaliate_if_hit_explicit_morale
				}
				else_if = {
					limit = { scope:story = { has_variable = damage_source } }
					custom_tooltip = pod_combat_desc_retaliate_if_hit_explicit_source
				}
				else_if = {
					limit = { scope:story = { has_variable = retaliate_agg } }
					custom_tooltip = pod_combat_desc_retaliate_if_hit_explicit_agg
				}
				else = {
					custom_tooltip = pod_combat_desc_retaliate_if_hit_explicit_sup
				}
			}
			else_if = {
				limit = { scope:story = { has_variable = hp_damage } }
				if = {
					limit = { scope:story = { has_variable = retaliate_agg } }
					custom_tooltip = pod_combat_desc_retaliate_if_hit_reuse_agg
				}
				else = {
					custom_tooltip = pod_combat_desc_retaliate_if_hit_reuse_sup
				}
			}
			else = {
				custom_tooltip = pod_combat_desc_retaliate_if_hit
			}
		}
		if = {
			limit = { scope:story = { has_variable = retaliate_if_missed } }
			if = {
				limit = { scope:story = { has_variable = retaliate_damage } }
				if = {
					limit = { scope:story = { has_variable = retaliate_morale } }
					custom_tooltip = pod_combat_desc_retaliate_if_missed_explicit_morale
				}
				else_if = {
					limit = { scope:story = { has_variable = damage_source } }
					custom_tooltip = pod_combat_desc_retaliate_if_missed_explicit_source
				}
				else_if = {
					limit = { scope:story = { has_variable = retaliate_agg } }
					custom_tooltip = pod_combat_desc_retaliate_if_missed_explicit_agg
				}
				else = {
					custom_tooltip = pod_combat_desc_retaliate_if_missed_explicit_sup
				}
			}
			else_if = {
				limit = { scope:story = { has_variable = hp_damage } }
				if = {
					limit = { scope:story = { has_variable = retaliate_agg } }
					custom_tooltip = pod_combat_desc_retaliate_if_missed_reuse_agg
				}
				else = {
					custom_tooltip = pod_combat_desc_retaliate_if_missed_reuse_sup
				}
			}
			else = {
				custom_tooltip = pod_combat_desc_retaliate_if_missed
			}
		}

		# AP / initiative parametric
		if = {
			limit = { scope:story = { has_variable = ap_gain } }
			custom_tooltip = pod_combat_desc_ap_gain
		}
		if = {
			limit = { scope:story = { has_variable = ap_drain } }
			custom_tooltip = pod_combat_desc_ap_drain
		}
		if = {
			limit = { scope:story = { has_variable = initiative_debuff } }
			custom_tooltip = pod_combat_desc_initiative_debuff
		}
		if = {
			limit = { scope:story = { has_variable = dot_rounds } }
			custom_tooltip = pod_combat_desc_dot
		}
		if = {
			limit = { scope:story = { has_variable = mind_control } }
			custom_tooltip = pod_combat_desc_mind_control
		}
	}
}
