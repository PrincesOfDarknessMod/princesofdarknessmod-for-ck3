##########################################
######   POD COMBAT SKILL GUIS   ######
##########################################

# Scripted GUIs for the skill configuration window.
# Each toggle equips/unequips a single combat move.
# Equipping is blocked when the category is at pod_combat_max_per_category (4).
# Full loadout: 16 skills (4 ATK + 4 CTL + 4 DEF + 4 PWR).
# 414 active skills: 8 base + 3 innate vampire + 403 discipline.

###### WINDOW CONTROLS ######

pod_combat_close_skills = {
	is_valid = {
		has_variable = pod_combat_config_open
	}
	effect = {
		remove_variable = pod_combat_config_open
	}
}

###### BASE PHYSICAL TOGGLES ######
# Available to all characters. Equip blocked at max cap. Remove always allowed.

pod_combat_toggle_strike = {
	is_valid = {
		has_variable = pod_combat_config_open
		trigger_if = {
			limit = { NOT = { has_variable = pod_combat_equipped_strike } }
			pod_combat_atk_count < pod_combat_max_per_category
		}
	}
	effect = {
		if = {
			limit = { has_variable = pod_combat_equipped_strike }
			remove_variable = pod_combat_equipped_strike
		}
		else = {
			set_variable = { name = pod_combat_equipped_strike value = yes }
		}
	}
}
