talmahera_hand_add = {
	scope = religion
	saved_scopes = { player }
	is_shown = { var:hand_status < 3 }
	is_valid = {
		custom_tooltip = {
			text = hand_exposed_tt
			NOT = { has_variable = hand_exposed }
		}
	}
	effect = {
		increase_hand_modifier_effect = yes
		change_variable = { 
			name = hand_status
			add = 1
		}
	}
}

talmahera_aralu_nergal = {
	scope = religion
	saved_scopes = { player }
	is_shown = { any_faith = { has_doctrine = talmahera_aralu_nergal_doctrine } }
	is_valid = {
		trigger_if = {
			limit = { NOT = { any_faith = { has_doctrine = talmahera_aralu_nergal_doctrine } } }
			custom_description = {
				text = number_of_talmahera_influence
				value = 2500
				talmahera_zeal >= 2500
			}
		}
	}
	effect = {
		if = {
			limit = { NOT = { any_faith = { has_doctrine = talmahera_aralu_nergal_doctrine } } }
			custom_description = {
				text = remove_talmahera_influence
				value = 2500
				change_variable = {
					name = talmahera_influence
					add = -2500
				}
			}
		}
		every_faith = {
			if = {
				limit = { has_doctrine = talmahera_aralu_ninmug_doctrine }
				remove_doctrine = talmahera_aralu_ninmug_doctrine
			}
			if = {
				limit = { has_doctrine = talmahera_aralu_loz_doctrine }
				remove_doctrine = talmahera_aralu_loz_doctrine
			}
			if = {
				limit = { has_doctrine = talmahera_aralu_al_marhi_doctrine }
				remove_doctrine = talmahera_aralu_al_marhi_doctrine
			}
			if = {
				limit = { NOT = { has_doctrine = talmahera_aralu_nergal_doctrine } }
				add_doctrine = talmahera_aralu_nergal_doctrine
			}
		}
	}
}

talmahera_edict_underworld_progress = {
	scope = religion
	saved_scopes = { player }
	is_shown = { }
	is_valid = {
		custom_description = {
			text = number_of_talmahera_influence
			value = 1000
			talmahera_zeal >= 1000
		}
	}
	effect = {
		custom_description = {
			text = remove_talmahera_influence
			value = 1000
			change_variable = {
				name = talmahera_influence
				add = -1000
			}
		}
		scope:player = {
			custom_tooltip = {
				text = talmahera_edict_underworld_progress_effect_tooltip
				if = {
					limit = { 
						has_variable = umbra_stationed_location 
						in_underworld_trigger = { LOCATION = var:umbra_stationed_location }
					} 
					POD_gain_umbra_progress_effect = { VICTIM = root VALUE = 100 }
				}
				every_relation = {
					type = umbra_explorer
					limit = { in_underworld_trigger = { LOCATION = var:umbra_stationed_location } }
					scope:player = { POD_gain_umbra_progress_effect = { VICTIM = prev VALUE = 100 } }
				}
			}
		}
	}
}
