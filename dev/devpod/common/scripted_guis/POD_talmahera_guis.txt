talmahera_hand_add = {
	scope = religion
	saved_scopes = { player }
	is_shown = {}
	is_valid = { 
		var:hand_status < 3
		NOT = { has_variable = hand_exposed }
		scope:player = { has_title = title:e_talmahera }
	}
	effect = {
		change_variable = { 
			name = hand_status
			add = 1
		}
		refresh_hand_modifier_effect = yes
	}
}

talmahera_hand_subtract = {
	scope = religion
	saved_scopes = { player }
	is_shown = {}
	is_valid = { 
		var:hand_status >= 1
		scope:player = { has_title = title:e_talmahera }
	}
	effect = {
		change_variable = { 
			name = hand_status
			subtract = 1
		}
	}
}

talmahera_edict_gain_fervor = {
	scope = religion
	saved_scopes = { player }
	is_shown = { any_faith = { fervor < 100 } }
	is_valid = {
		custom_description = {
			text = number_of_talmahera_influence
			value = 500
			var:talmahera_influence >= 500
		}
	}
	effect = {
		custom_description = {
			text = remove_talmahera_influence
			value = 500
			change_variable = {
				name = talmahera_influence
				add = -500
			}
		}
		every_faith = {
			change_fervor = {
				value = medium_fervor_gain
				desc = fervor_gain_talmahera_edict
			}
		}
	}
}

talmahera_edict_underworld_progress = {
	scope = religion
	saved_scopes = { player }
	is_shown = { }
	is_valid = {
		custom_description = {
			text = number_of_talmahera_influence
			value = 1000
			var:talmahera_influence >= 1000
		}
		scope:player = { has_title = title:e_talmahera }
	}
	effect = {
		custom_description = {
			text = remove_talmahera_influence
			value = 1000
			change_variable = {
				name = talmahera_influence
				add = -1000
			}
		}
		scope:player = {
			custom_tooltip = {
				text = talmahera_edict_underworld_progress_effect_tooltip
				if = {
					limit = { 
						has_variable = umbra_stationed_location 
						in_underworld_trigger = { LOCATION = var:umbra_stationed_location }
					} 
					POD_gain_umbra_progress_effect = { VICTIM = root VALUE = 100 }
				}
				every_relation = {
					type = umbra_explorer
					limit = { in_underworld_trigger = { LOCATION = var:umbra_stationed_location } }
					scope:player = { POD_gain_umbra_progress_effect = { VICTIM = prev VALUE = 100 } }
				}
			}
		}
	}
}
