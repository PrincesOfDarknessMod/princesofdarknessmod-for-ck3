	}

demonic_pact_creator_select_detail_script = {
	saved_scopes = { detail_id }
	is_shown = {
		any_in_list = {
			variable = pact_detail_list
			var:id = scope:detail_id
			save_temporary_scope_as = pact_detail
		}
		switch = {
			trigger = scope:pact_detail.var:type
			flag:frequency = { 
				has_variable = selected_frequency
				var:selected_frequency.var:id ?= scope:detail_id
			}
			flag:obligation = { 
				has_variable = selected_obligation
				var:selected_obligation.var:id ?= scope:detail_id
			}
			flag:consequence = { 
				has_variable = selected_consequence
				var:selected_consequence.var:id ?= scope:detail_id
			}
		}
	}
	effect = { 
		POD_pact_select_detail_effect = { ID = scope:detail_id }
	}
}

demonic_pact_creator_finish_script = {
	is_valid = {
		custom_tooltip = {
			text = demonic_pact_creator_finish_script_frequency
			has_variable = selected_frequency
		}
		custom_tooltip = {
			text = demonic_pact_creator_finish_script_obligation
			has_variable = selected_obligation
		}
		custom_tooltip = {
			text = demonic_pact_creator_finish_script_consequence
			has_variable = selected_consequence
		}
	}
	effect = {
		if = {
			limit = {
				has_variable = selected_frequency
				has_variable = selected_obligation
				has_variable = selected_consequence
			}
			save_scope_value_as = {
				name = true_frequency
				value = {
					value = var:selected_frequency.var:level
					multiply = 4
				}
			}

			if = { limit = { NOT = { has_relation_worshipped_demon = var:demonic_pact_demon } } set_relation_worshipped_demon = var:demonic_pact_demon }
			if = { limit = { NOT = { has_relation_demonic_master = var:demonic_pact_demon } } set_relation_demonic_master = var:demonic_pact_demon }
			POD_give_and_expose_witch_trait_effect = yes
			if = { 
				limit = { has_trait = supehunter } 
				var:demonic_pact_demon = { save_scope_as = vampiresire } 
				POD_hunter_enthrall_effect = yes 
			}
			var:demonic_pact_demon = { 
				add_opinion = {
					target = PREV
					modifier = pleased_opinion
					opinion = 15
				}
			}
			POD_create_pact_effect = {
				DEMON = var:demonic_pact_demon
				OBLIGATION = var:selected_obligation.var:id
				OBLIGATION_LEVEL = var:selected_obligation.var:level
				FREQUENCY = scope:true_frequency
				CONSEQUENCE = var:selected_consequence.var:id
				CONSEQUENCE_LEVEL = var:selected_consequence.var:level
			}
		}
	}
}
