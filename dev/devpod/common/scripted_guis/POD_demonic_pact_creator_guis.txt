demonic_pact_creator_select_detail_script = {
	saved_scopes = { selected_stipulation }
	is_shown = {
		OR = {
			var:selected_obligation  ?= { var:id = scope:selected_stipulation.var:id }
			var:selected_frequency   ?= { var:id = scope:selected_stipulation.var:id }
			var:selected_consequence ?= { var:id = scope:selected_stipulation.var:id }
		}
	}
	is_valid = {
		POD_can_select_pact_stipulation = {
			STIPULATION = scope:selected_stipulation
		}
	}
	effect = { 
		POD_pact_gui_select_stipulation = { STIPULATION = scope:selected_stipulation }
	}
}

demonic_pact_creator_preview_script = {
	saved_scopes = { demon }
	is_valid = {
		POD_can_finalize_pact_in_gui = yes
	}
	effect = {
		if = {
			limit = { POD_can_finalize_pact_in_gui = yes }
			
			POD_create_pact_from_stipulation_stories = {
				DEMON = scope:demon
				OBLIGATION_STORY  = var:selected_obligation
				FREQUENCY_STORY   = var:selected_frequency
				CONSEQUENCE_STORY = var:selected_consequence
			}
		}
	}
}

demonic_pact_creator_finish_script = {
	saved_scopes = { demon }
	is_valid = {
		POD_can_finalize_pact_in_gui = yes
	}
	effect = {
		if = {
			limit = { POD_can_finalize_pact_in_gui = yes }

			scope:demon = { 
				add_opinion = {
					target = prev
					modifier = pleased_opinion
					opinion = 15
				}
			}
			POD_become_demonic_thrall_of = {
				DEMON = scope:demon
			}
			custom_description_no_bullet = {
				text = POD_create_demonic_pact
				object = scope:demon
			}
			POD_create_pact_from_stipulation_stories = {
				DEMON = scope:demon
				OBLIGATION_STORY  = var:selected_obligation
				FREQUENCY_STORY   = var:selected_frequency
				CONSEQUENCE_STORY = var:selected_consequence
			}
		}
	}
}
