POD_journey_detail_open = {
	scope = character
	is_shown = {
		exists = var:journey_detail_window
	}
	effect = {
		remove_variable = journey_detail_window_widget
		remove_variable = journey_detail_window_subtype
		
		if = {
			limit = { var:journey_detail_window ?= scope:journey }
			remove_variable = journey_detail_window
		}
		else = {
			set_variable = {
				name = journey_detail_window
				value = scope:journey
			}
		}
		
		if = {
			limit = { var:journey_detail_window.var:journey_type = flag:hunt_in_capital }
			POD_hunt_in_capital_create_list_effect = yes
		}
		
		switch = {
			trigger = var:journey_detail_window.var:journey_type
			
			flag:seek_knowledge = {
				set_variable = {
					name = journey_detail_window_widget
					value = flag:seek_knowledge
				}
			}
			
			flag:find_artifact = {
				set_variable = {
					name = journey_detail_window_widget
					value = flag:find_artifact
				}
			}
			
			fallback = {
				set_variable = {
					name = journey_detail_window_subtype
					value = var:journey_detail_window
				}
			}
		}
		
	}
}

POD_journey_detail_close = {
	scope = character
	effect = {
		remove_variable = journey_detail_window
		remove_variable = journey_detail_window_widget
		remove_variable = journey_detail_window_subtype
		if = {
			limit = { has_variable_list = hunt_in_capital_types }
			clear_variable_list = hunt_in_capital_types
		}
	}
}

POD_journey_requirements = {
	scope = character
	
	is_shown = {
		NOR = {
			var:current_journey ?= {
				var:journey_type = scope:journey.var:journey_type
			}
			any_in_list = {
				variable = paused_journeys
				count > 0
				var:journey_type = scope:journey.var:journey_type
			}
		}
	}
	
	is_valid = {
		POD_can_progress_journey_trigger = yes
		
		trigger_if = {
			limit = { scope:journey = { has_variable = gold_cost } } # do not check for paused journeys
			
			custom_tooltip = {
				text = POD_journey_requirements_gold
				OR = {
					scope:journey.var:gold_cost = 0
					gold >= scope:journey.var:gold_cost
				}
			}
			custom_tooltip = {
				text = POD_journey_requirements_prestige
				OR = {
					scope:journey.var:prestige_cost = 0
					prestige >= scope:journey.var:prestige_cost
				}
			}
			custom_tooltip = {
				text = POD_journey_requirements_piety
				OR = {
					scope:journey.var:piety_cost = 0
					piety >= scope:journey.var:piety_cost
				}
			}
		}
	}
}

POD_journey_on_start = {
	scope = character
	effect = {
		# Pay upfront costs
		remove_short_term_gold = scope:journey.var:gold_cost
		add_prestige = { value = 0 subtract = scope:journey.var:prestige_cost }
		add_piety    = { value = 0 subtract = scope:journey.var:piety_cost }
		
		remove_variable = journey_detail_window
		
		# hunt_in_capital has a separate effect to determine costs based on the type of mortal
		if = {
			limit = { scope:journey.var:journey_type = flag:hunt_in_capital }
			POD_hunt_in_capital_cost_effect = yes
		}
		
		POD_start_journey_effect = {
			TEMPLATE = scope:journey
		}
	}
}
