umbra_adv_start = {
	scope = character
	is_valid = { 
		custom_tooltip = {
			text = passing_gauntlet_toast
			var:spirit_location ?= { NOT = { this = flag:earth } }
		}
		custom_tooltip = {
			text = umbra_retreat_cooldown_tt
			NOT = { any_memory = { has_memory_type = umbra_retreat_memory } }
		}
	}
	effect = {
		random = {
			chance = umbra_escape_chance
			custom_tooltip = {
				text = umbra_retreat_tt
				POD_umbra_adv_clean_effect = yes
			}
		}
		POD_umbra_retreat_cooldown_effect = yes
		add_prestige = miniscule_prestige_loss
		stress_impact = {
			base = minor_stress_gain
			brave = minor_stress_impact_gain
			craven = minor_stress_impact_loss
		}
	}
}

umbra_adv_end = { 
	scope = character 
	is_valid = { 
		custom_tooltip = {
			text = out_of_reach
			OR = {
				var:umbra_adv_postion = flag:d7
				var:umbra_adv_postion = flag:c7
				var:umbra_adv_postion = flag:b7
			}
		}
	}
	effect = { 
		POD_umbra_adv_move_effect = { POSITION = c8 }
		if = {
			limit = { faith = faith:inwardway }
			POD_kj_advance_dharma_effect = yes
		}
		change_variable = { name = umbra_difficulty add = 1 }
		trigger_event = { on_action = spirit_travel_movement_end_pulse }
	}
}
