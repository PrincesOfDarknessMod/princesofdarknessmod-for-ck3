spell_mutate_script = {
	scope = character
	is_shown = {

	}
	is_valid = {
		POD_has_enough_blood_for_discipline = { AMOUNT = 1 }
		has_perk = mutate_perk
		NOT = { stress_level >= 3 }
	}
	effect = {
		custom_tooltip = spell_mutate_tooltip
		POD_willpower_rouse_check_effect = yes
		POD_static_stress_gain = { COUNT = 25 }
		random_list = {
			0 = {
				modifier = { add = POD_bloodmagic_success_chance }
				send_interface_toast = { 
					title = POD_shared_loca_success
					if = { limit = { has_perk = transmutate_perk } 
						remove_short_term_gold = 40
						scope:target = { add_gold = 140 }
					}
					else = {
						remove_short_term_gold = 80
						scope:target = { add_gold = 120 }
					}
				}
			}
			100 = {
				modifier = { add = POD_bloodmagic_success_chance_negated }
				send_interface_toast = { 
					title = POD_shared_loca_failure
					POD_static_stress_gain = { COUNT = 50 }
				}
			}
		}
		POD_discipline_cost_effect = { COUNT = 1 }
	}
}
