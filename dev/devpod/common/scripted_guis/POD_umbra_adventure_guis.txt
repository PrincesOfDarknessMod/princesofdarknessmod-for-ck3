POD_reload_umbra_adventure = {
	scope = character
	effect = {
		var:umbra_adventure = {
			save_scope_as = umbra_adventure
		}
		POD_umbra_adventure_generate_map = {
			WIDTH = scope:umbra_adventure.var:width
			HEIGHT = scope:umbra_adventure.var:height
			ROWS = scope:umbra_adventure.var:rows
			COLUMNS = scope:umbra_adventure.var:columns
			PADDING = scope:umbra_adventure.var:padding
			SAVE_AS = node_map
		}

		var:umbra_adventure = {
			var:node_map = {
				end_story = yes
			}
			set_variable = {
				name = node_map
				value = scope:node_map
			}
		}
	}
}

POD_create_umbra_adventure = {
	scope = character
	effect = {
		if = {
			limit = {
				has_variable = umbra_adventure
			}
			var:umbra_adventure = {
				end_story = yes
			}
		}
		character:POD_wraith_01 = {
			create_story = {
				type = POD_story_umbra_adventure
				save_scope_as = umbra_adventure
			}
		}
		scope:umbra_adventure = {
			set_variable = {
				name = width
				value = 800
			}
			set_variable = {
				name = height
				value = 600
			}
			set_variable = {
				name = rows
				value = 6
			}
			set_variable = {
				name = columns
				value = 8
			}
			set_variable = {
				name = padding
				value = 25
			}
		}
		set_variable = {
			name = umbra_adventure
			value = scope:umbra_adventure
		}
		scope:umbra_adventure.var:padding = {
			debug_log_scopes = no
		}
		POD_umbra_adventure_generate_map = {
				WIDTH = scope:umbra_adventure.var:width
				HEIGHT = scope:umbra_adventure.var:height
				ROWS = scope:umbra_adventure.var:rows
				COLUMNS = scope:umbra_adventure.var:columns
				PADDING = scope:umbra_adventure.var:padding
				SAVE_AS = node_map
		}

		scope:umbra_adventure = {
			set_variable = {
				name = node_map
				value = scope:node_map
			}
		}

		character:POD_wraith_01 = {
			create_story = {
				type = POD_story_umbra_adventure_inventory_item
				save_scope_as = inventory_item
			}
		}
		scope:inventory_item = {
			set_variable = {
				name = type
				value = flag:umbra_inventory_item_essence
			}
			set_variable = {
				name = quantity
				value = 50
			}
		}
		POD_umbra_adventure_add_to_inventory = {
			ITEM = scope:inventory_item
		}

		character:POD_wraith_01 = {
			create_story = {
				type = POD_story_umbra_adventure_inventory_item
				save_scope_as = inventory_item
			}
		}

		# for reasons
		character:POD_wraith_01 = {
			create_artifact_soulfire_lantern_effect = {
				OWNER = character:POD_wraith_01
			}
		}
		scope:inventory_item = {
			set_variable = {
				name = type
				value = flag:umbra_inventory_item_artifact
			}
			set_variable = {
				name = artifact
				value = scope:newly_created_artifact
			}
		}
		POD_umbra_adventure_add_to_inventory = {
			ITEM = scope:inventory_item
		}
	}
}

POD_umbra_adventure_drop_item = {
	scope = character
	saved_scopes = { story }
	confirm_title = "THROW IT ON THE GROUND!"
	confirm_text = "Drop this item. FOREVER"
	effect = {
		var:umbra_adventure = {
			save_scope_as = umbra_adventure
		}
		scope:umbra_adventure = {
			remove_list_variable = {
				name = inventory
				target = scope:story
			}
		}
		scope:story = {
			end_story = yes
		}
	}
}

POD_umbra_adventure_inventory_placeholder = {
	scope = character
	saved_scopes = { story }
	is_shown = {
		scope:story.var:index = {
			save_temporary_scope_as = index
		}
		var:umbra_adventure = {
			OR = {
				NOT = { has_variable_list = inventory }
				variable_list_size = {
					name = inventory
					value <= scope:index
				}
			}
		}
	}
}

POD_umbra_adventure_is_essence = {
	scope = character
	saved_scopes = { story }
	is_shown = {
		scope:story.var:type ?= flag:umbra_inventory_item_essence
	}
}

POD_umbra_adventure_is_artifact = {
	scope = character
	saved_scopes = { story }
	is_shown = {
		scope:story.var:type ?= flag:umbra_inventory_item_artifact
	}
}
