# One stepper SGUI per knob (the GUI arrows pass the signed step as
# scope:delta via AddScope('delta', MakeScopeValue(...)) - same pattern
# as the umbra-adventure debug editors), plus the window close / reset /
# mode-clear buttons. Knob plumbing: pod_combat_tuning_effects.txt.

pod_combat_tuning_close = {
	scope = character
	effect = { remove_global_variable = pod_combat_tuning_open }
}

pod_combat_tuning_reset_all = {
	scope = character
	effect = { pod_combat_tuning_reset_all_effect = yes }
}

# Initiative no-reroll toggle: round 1's roll holds for the whole fight
# (fresh mid-fight seats still get their first roll). Note an
# initiative_debuff folded under no-reroll lasts the rest of the fight.
pod_combat_tune_toggle_no_reroll = {
	scope = character
	effect = {
		if = {
			limit = { exists = global_var:pod_combat_tune_init_no_reroll }
			remove_global_variable = pod_combat_tune_init_no_reroll
		}
		else = {
			set_global_variable = { name = pod_combat_tune_init_no_reroll value = yes }
		}
	}
}

# Initiative decouple-prowess toggle: use a flat baseline (pod_combat_tune_init_base,
# default 20) instead of prowess as the initiative starting line.
# AI decision trace: when set, the AI selector emits one machine-parseable
# debug_log line per pick and per turn (see pod_combat_ai_trace_* effects).
# OFF by default and every trace cost sits inside the gate, including the
# eligible-count scan, which doubles the per-action selector work.
pod_combat_tune_toggle_ai_trace = {
	scope = character
	effect = {
		if = {
			limit = { exists = global_var:pod_combat_tune_ai_trace }
			remove_global_variable = pod_combat_tune_ai_trace
		}
		else = {
			set_global_variable = { name = pod_combat_tune_ai_trace value = yes }
		}
	}
}
