spell_summon_demon_script = {
	scope = character
	
	is_valid = {
		POD_has_enough_energy_for_sorcery = { AMOUNT = sorcery_summon_demon_cost }
		has_perk = numina_conjuration_6_perk
		trigger_if = {
			limit = {
				scope:target != root
				scope:target = { POD_is_demon_trigger = yes }
			}
			scope:target = { has_character_modifier = trapped_in_abyss_modifier }
		}
		#scope:target = ROOT
	}
	effect = {
		POD_sorcery_cost_effect = { COUNT = sorcery_summon_demon_cost }
		random_list = {
			0 = {
				modifier = { add = POD_sorcery_success_chance }
				
				if = {
					limit = {
						scope:target = {
							this != root
							POD_is_demon_trigger = yes
							has_character_modifier = trapped_in_abyss_modifier
						}
					}
					scope:target = { save_scope_as = summoned_demon }
				}
				else = {
					random_living_character = {
						limit = { 
							POD_is_demon_trigger = yes
							has_character_modifier = trapped_in_abyss_modifier
						}
						save_scope_as = summoned_demon
					}
				}
				if = {
					limit = { NOT = { exists = scope:summoned_demon } }
					create_character = {
						age = { 16 1000 }
						culture = root.culture
						location = root.location
						gender_female_chance = 50
						dynasty = none
						trait = demon

						diplomacy = { 0 10 }
						martial = { 0 10 }
						stewardship = { 0 10 }
						intrigue = { 0 10 }
						learning = { 0 10 }
						prowess = { 5 15 }

						random_traits = yes
						
						after_creation = {
							set_immortal_age = { 16 90 }
							POD_demon_initialization_effect = yes
						}
						save_scope_as = summoned_demon
					}
				}
				custom_tooltip = {
					text = POD_summon_demon_demonology_interaction_success
					send_interface_toast = {
						title = POD_shared_loca_success
						right_icon = scope:actor
						left_icon = scope:summoned_demon
						add_courtier = scope:summoned_demon
						scope:summoned_demon ?= {
							POD_free_from_abyss_effect = yes
							hidden_effect = { return_to_court = yes }
							add_character_flag = { flag = blocked_from_leaving years = 5 }
						}
					}
				}
			}
			100 = {
				modifier = { add = POD_sorcery_success_chance_negated }
				POD_sorcery_default_failure_with_toast = yes
			}
		}
	}
}
