# For debugging, reroll support
POD_umbra_expedition_reload = {
	scope = character
	effect = {
		POD_create_umbra_expedition = yes
	}
}

POD_umbra_expedition_on_start = {
	scope = character
	effect = {
		if = {
			limit = { NOT = { exists = var:umbra_map } }
			POD_create_umbra_expedition = yes
		}
	}
}

POD_umbra_node_button_sgui = {
	scope = character
	saved_scopes = { story }
	is_shown = {
		scope:story.var:include ?= yes
	}
}

POD_umbra_node_start_sgui = {
	scope = character
	saved_scopes = { story }
	is_shown = {
		scope:story.var:type ?= flag:start
	}
}

POD_umbra_node_end_sgui = {
	scope = character
	saved_scopes = { story }
	is_shown = {
		scope:story.var:type ?= flag:end
	}
}

POD_umbra_node_unknown_sgui = {
	scope = character
	saved_scopes = { story }
	is_shown = {
		scope:story.var:type ?= flag:unknown
	}
}
