# For debugging, reroll support
POD_umbra_expedition_reload = {
	scope = character
	effect = {
		POD_create_umbra_expedition = yes
	}
}

POD_umbra_expedition_on_start = {
	scope = character
	effect = {
		if = {
			limit = { NOT = { exists = var:umbra_map } }
			POD_create_umbra_expedition = yes
		}
	}
}

POD_umbra_expedition_is_current_location = {
	scope = character
	saved_scopes = { story }
	is_shown = {
		scope:story = var:umbra_map.var:current_location
	}
}

POD_umbra_expedition_line = {
	scope = character
	saved_scopes = { story }

	is_shown = {
		scope:story.var:start_node = var:umbra_map.var:current_location
	}
}

POD_umbra_expedition_node = {
	scope = character
	saved_scopes = { story }

	is_shown = {
		scope:story.var:include = yes
	}

	is_valid = {
		var:umbra_map.var:current_location = {
			any_in_list = {
				variable = lines
				OR = {
					var:end_node = scope:story
					var:start_node = scope:story
				}
			}
		}
	}

	effect = {
		var:umbra_map = {
			set_variable = {
				name = current_node
				target = scope:story
			}
		}
	}
}
