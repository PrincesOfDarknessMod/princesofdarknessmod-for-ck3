
ahroun_snarl_sgui = {
    scope = story
    is_valid = {
        story_owner ?= { has_perk = snarl_of_challenge_perk }
    }
    is_shown = {
        story_owner ?= { has_variable = battle_combat_encounters_enabled }
    }
    
	effect = {
		story_owner ?= {
			if = {
				limit = { has_variable = battle_combat_encounters_enabled }
				remove_variable = battle_combat_encounters_enabled
			}
			else = {
				custom_tooltip = battle_combat_encounters_enabled_tooltip
				set_variable = battle_combat_encounters_enabled
			}
		}
    }
}
