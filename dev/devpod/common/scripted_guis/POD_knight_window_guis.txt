populate_and_sort_knights = {
	scope = character
	effect = {
		save_scope_as = root_char
		if = {
			limit = { has_variable_list = sorted_knights }
			clear_variable_list = sorted_knights
		}
		every_knight = {
			add_to_temporary_list = all_knights
		}

		ordered_in_list = {
			list = all_knights
			order_by = prowess
			max = list_size:all_knights

			save_scope_as = sorted_knight
			scope:root_char = {
				add_to_variable_list = {
					name = sorted_knights
					target = scope:sorted_knight
				}
			}
		}
	}
}
