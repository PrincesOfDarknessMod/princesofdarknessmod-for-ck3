sol_custom_religion_creator_select_detail_script = {
	saved_scopes = { selected_tenet }
	is_shown = {
		OR = {
			var:selected_institution  ?= { var:id = scope:selected_tenet.var:id }
			var:selected_devotional   ?= { var:id = scope:selected_tenet.var:id }
			var:selected_violence     ?= { var:id = scope:selected_tenet.var:id }
			var:selected_flexibility  ?= { var:id = scope:selected_tenet.var:id }
			var:selected_balance      ?= { var:id = scope:selected_tenet.var:id }
			var:selected_homeland     ?= { var:id = scope:selected_tenet.var:id }
		}
	}
	is_valid = {
		POD_can_select_custom_religion_tenet = {
			TENET = scope:selected_tenet
		}
	}
	effect = { 
		POD_sol_custom_religion_gui_select_tenet = { TENET = scope:selected_tenet }
	}
}

sol_custom_religion_creator_preview_script = {
	is_valid = {
		POD_can_finalize_sol_custom_religion_in_gui = yes
	}
	effect = {
		if = {
			limit = { POD_can_finalize_sol_custom_religion_in_gui = yes }
			
			POD_sol_create_custom_religion = {
				INSTITUTION  = var:selected_institution.var:id
				DEVOTIONAL   = var:selected_devotional.var:id
				VIOLENCE     = var:selected_violence.var:id
				FLEXIBILITY  = var:selected_flexibility.var:id
				BALANCE      = var:selected_balance.var:id
				HOMELAND     = var:selected_homeland.var:id
			}
		}
	}
}

sol_custom_religion_creator_finish_script = {
	is_valid = {
		POD_can_finalize_sol_custom_religion_in_gui = yes
	}
	effect = {
		if = {
			limit = { POD_can_finalize_sol_custom_religion_in_gui = yes }

			POD_sol_create_custom_religion = {
				INSTITUTION  = var:selected_institution.var:id
				DEVOTIONAL   = var:selected_devotional.var:id
				VIOLENCE     = var:selected_violence.var:id
				FLEXIBILITY  = var:selected_flexibility.var:id
				BALANCE      = var:selected_balance.var:id
				HOMELAND     = var:selected_homeland.var:id
			}
		}
	}
}
