tremere_doctrine_elector_initiate = {
	scope = character
	is_valid = {
		faith = faith:tremerecode

			variable = tremere_council_list

				}
			}
		}

			value = faith:tremerecode
		}
	}
}

doctrine_choose = {
	saved_scopes = { faith }
	is_shown = {
	}
	is_valid = {
		NOT = { root = doctrine:tenet_helios_truth }
		NOT = { root = doctrine:tenet_ivory_priesthood }
		NOT = { root = doctrine:tenet_hexcraft }
		NOT = { root = doctrine:tenet_vengeance_of_nut }
		NOT = { root = doctrine:tenet_innovative }
		NOT = { root = doctrine:tenet_paranoia }
		NOT = { root = doctrine:tenet_advocatus_diaboli }
		NOT = { root = doctrine:tenet_constantinian }
		NOT = { root = doctrine:tenet_dharmic_pacifism }
	}
	effect = {
		scope:faith = {
			if = {
				limit = { 
					is_target_in_variable_list = { 
						name = doctrine_to_change_list 
						target = root 
					} 
				}
				remove_list_variable = { 
					name = doctrine_to_change_list 
					target = root
				}
			}
			else = {
				add_to_variable_list = {
					name = doctrine_to_change_list
					target = root
				}
			}
		}
	}
}

# Enlightenment Virtues
doctrine_enlightenment_virtues_conscience_and_selfcontrol_select = {
	scope = character 
	is_shown = {
		NOT = { faith = { has_doctrine = doctrine_enlightenment_virtues_conscience_and_selfcontrol } }
	}
	is_valid = {
		NOT = { faith = { var:doctrine_enlightenment_virtues ?= flag:doctrine_enlightenment_virtues_conscience_and_selfcontrol } }

		faith = {

				set_variable = { 
					name = doctrine_enlightenment_virtues
					value = flag:doctrine_enlightenment_virtues_conscience_and_selfcontrol
				}
			}
		}
	}
}
