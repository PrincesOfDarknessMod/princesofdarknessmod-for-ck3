	scope = character
	is_valid = {
		faith = faith:tremerecode
		has_variable = influence_inner_council
	}

		every_in_global_list = {
			variable = tremere_council_list

				add_to_variable_list = {
					name = doctrine_elector_list
					target = prev
				}
			}
		}
		set_variable = {
			name = faith_to_change_doctrine
			value = faith:tremerecode
		}
	}
}

doctrine_choose = {
	saved_scopes = { faith }
	is_shown = {
	}
	is_valid = {
	}
	effect = {
		scope:faith = {
			if = {
				limit = { 
					is_target_in_variable_list = { 
						name = doctrine_to_change_list 
						target = root 
					} 
				}
				remove_list_variable = { 
					name = doctrine_to_change_list 
					target = root
				}
			}
			else = {
				add_to_variable_list = {
					name = doctrine_to_change_list
					target = root
				}
			}
		}
	}
}

apply_doctrine_change = {
	scope = character 
	saved_scopes = { faith }
	is_shown = {}
	is_valid = {
		custom_tooltip = {
			text = FAITH_DOCTRINE_CHANGE_WINDOW_POWERFUL_VASSALS_DESC
			NOT = {
				any_in_list = {
					variable = doctrine_elector_list
					POD_opposing_doctrine_change_trigger = { CHARACTER = root }
				}
			}
		}
		custom_tooltip = {
			text = FAITH_DOCTRINE_CHANGE_WINDOW_CURRENT_DOCTRINE
			scope:faith.POD_doctrine_change_value > 500
		}
		custom_tooltip = {
			text = FAITH_DOCTRINE_CHANGE_WINDOW_THREE_TENET
			scope:faith.POD_current_number_of_tenets = 3
		}
		prestige >= scope:faith.POD_doctrine_change_value
	}
	effect = {
		add_prestige = {
			value = scope:faith.POD_doctrine_change_value
			multiply = -1
		}
		scope:faith = {  
			POD_change_enlightenment_virtues_doctrine_effect = yes
			POD_change_vampirethemasquerade_doctrine_effect = yes
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_vampirediablerie }
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_vampirefrenzy }
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_vampirethinblood }
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_vampirebloodoath }
			POD_change_vampiretheghouling_doctrine_effect = yes
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_bloodsorcery }
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_infernalism }
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_lycanthropy }
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_magicraft }
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_haunting }
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_glamour }
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_undying }
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_kueijin }
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_ventrue }
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_toreador }
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_tremere }
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_brujah }
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_nosferatu }
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_gangrel }
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_malkavian }
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_lasombra }
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_tzimisce }
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_ravnos }
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_cappadocian }
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_setite }
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_assamite }
			every_in_list = {
				variable = doctrine_to_change_list
				if = {
					limit = { scope:faith = { has_doctrine = prev } }
					scope:faith = { remove_doctrine = prev }
				}
				else = {
					scope:faith = { add_doctrine = prev }
				}
			}
			clear_variable_list = doctrine_to_change_list
			clear_variable_list = doctrine_elector_list
		}
		remove_variable = faith_to_change_doctrine
	}
}
