	scope = character
	effect = {
		clear_variable_list = doctrine_elector_list

			root = { 
				add_to_variable_list = {
					name = doctrine_elector_list
					target = prev
				}
			}
		}
	}
}

opposing_doctrine_change = {
	scope = character 
	saved_scopes = { player }
	is_shown = {}
	is_valid = {
		POD_opposing_doctrine_change_trigger = { CHARACTER = scope:player }
	}
	effect = {
	}
}

apply_doctrine_change = {
	scope = character 
	is_shown = {}
	is_valid = {
		custom_tooltip = {
			text = FAITH_DOCTRINE_CHANGE_WINDOW_POWERFUL_VASSALS_DESC
			NOT = {
				any_in_list = {
					variable = doctrine_elector_list
					POD_opposing_doctrine_change_trigger = { CHARACTER = root }
				}
			}
		}
		custom_tooltip = {
			text = FAITH_DOCTRINE_CHANGE_WINDOW_CURRENT_DOCTRINE
			faith = {  
				OR = {
					has_variable = doctrine_enlightenment_virtues
					has_variable = doctrine_vampirethemasquerade
					has_variable = doctrine_vampirediablerie
					has_variable = doctrine_vampirefrenzy
					has_variable = doctrine_vampirethinblood
					has_variable = doctrine_vampirebloodoath
					has_variable = doctrine_vampireghouling
					has_variable = doctrine_bloodsorcery
					has_variable = doctrine_infernalism
					has_variable = doctrine_lycanthropy
					has_variable = doctrine_magicraft
					has_variable = doctrine_haunting
					has_variable = doctrine_glamour
					has_variable = doctrine_undying
					has_variable = doctrine_kueijin
				}
			}
		}
		prestige >= POD_doctrine_change_value
	}
	effect = {
		add_prestige = POD_doctrine_change_loss
		faith = {  
			POD_change_enlightenment_virtues_doctrine_effect = yes
			POD_change_vampirethemasquerade_doctrine_effect = yes
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_vampirediablerie }
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_vampirefrenzy }
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_vampirethinblood }
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_vampirebloodoath }
			POD_change_vampiretheghouling_doctrine_effect = yes
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_bloodsorcery }
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_infernalism }
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_lycanthropy }
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_magicraft }
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_haunting }
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_glamour }
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_undying }
			POD_change_crime_doctrine_effect = { DOCTRINE = doctrine_kueijin }
		}
	}
}

doctrine_enlightenment_virtues_conscience_and_selfcontrol_select = {
	scope = character 
	is_shown = {
		NOT = { faith = { has_doctrine = doctrine_enlightenment_virtues_conscience_and_selfcontrol } }
	}
	is_valid = {
		NOT = { faith = { var:doctrine_enlightenment_virtues ?= flag:doctrine_enlightenment_virtues_conscience_and_selfcontrol } }

		faith = {
			if = {

				set_variable = { 
					name = doctrine_enlightenment_virtues
