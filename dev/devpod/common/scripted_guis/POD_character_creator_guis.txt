character_creator_window_open = {
	scope = character
	is_shown = {}
	effect = {
		remove_variable = character_creator_points
		remove_variable = character_creator_sex
		remove_variable = character_creator_age
		remove_variable = character_creator_diplomacy
		remove_variable = character_creator_martial
		remove_variable = character_creator_stewardship
		remove_variable = character_creator_intrigue
		remove_variable = character_creator_learning
		remove_variable = character_creator_prowess
	}
}

character_creator_window_start = {
	scope = character
	is_shown = {}
	effect = {
		if = {
			limit = { NOT = { has_variable = character_creator_sex } }
			set_variable = { name = character_creator_sex value = 50 }
		}
		if = {
			limit = { NOT = { has_variable = character_creator_age } }
			set_variable = { name = character_creator_age value = 21 }
		}
		if = {
			limit = { NOT = { has_variable = character_creator_diplomacy } }
			set_variable = { name = character_creator_diplomacy value = 0 }
		}
		if = {
			limit = { NOT = { has_variable = character_creator_martial } }
			set_variable = { name = character_creator_martial value = 0 }
		}
		if = {
			limit = { NOT = { has_variable = character_creator_stewardship } }
			set_variable = { name = character_creator_stewardship value = 0 }
		}
		if = {
			limit = { NOT = { has_variable = character_creator_intrigue } }
			set_variable = { name = character_creator_intrigue value = 0 }
		}
		if = {
			limit = { NOT = { has_variable = character_creator_learning } }
			set_variable = { name = character_creator_learning value = 0 }
		}
		if = {
			limit = { NOT = { has_variable = character_creator_prowess } }
			set_variable = { name = character_creator_prowess value = 0 }
		}
	}
}

character_creator_window_enact = {
	scope = character
	is_shown = {}
	effect = {
		# Redirect
		switch = {
			trigger = var:character_creator_type
			flag:find_mortal = {
				trigger_event = POD_find_mortals.101
			}
			flag:hin = {
				trigger_event = podkueijin.300
			}
			flag:animantis = {
				trigger_event = POD_fae_interactions.400
			}
			flag:sprite = {
				trigger_event = POD_fae_interactions.600
			}
			flag:perfect_servant = {
				trigger_event = POD_bloodmagic_events.2
			}
			flag:manifestation = {
				trigger_event = POD_wraith_maintenance.4
			}
		}
		# Clean
		remove_variable = character_creator_type
		remove_variable = character_creator_points
	}
}
