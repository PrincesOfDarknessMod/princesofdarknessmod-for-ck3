
POD_fleshcrafting_button = {
	scope = character
	is_valid = {
		POD_can_fleshcraft = yes
		POD_fleshcrafting_target_has_all_selected_perks = no
	}
	effect = {
		if = {
			limit = {
				POD_can_fleshcraft = yes
				POD_fleshcrafting_target_has_all_selected_perks = no
			}
			POD_fleshcrafting_effect = {
				SHOW_SUCCESS_CHANCE = yes
			}
		}
	}
}

POD_fleshcrafting_preview = {
	scope = character
	is_shown = {
		POD_fleshcrafting_target_has_all_selected_perks = no
	}
	is_valid = {
		POD_can_fleshcraft = yes
	}
	effect = {
		POD_fleshcrafting_gui_preview = yes
	}
}

POD_fleshcrafting_cancel = {
	scope = character
	effect = {
		POD_close_fleshcrafting_gui = yes
	}
}

POD_fleshcrafting_clear = {
	scope = character
	effect = {
		POD_fleshcrafting_clear_all_layer_perks = yes
	}
}

POD_fleshcrafting_reset = {
	scope = character
	effect = {
		var:fleshcraft_target = { save_scope_as = fleshcraft_target }
		POD_fleshcrafting_init_all_layers = yes
	}
}

POD_fleshcrafting_modify_flesh = {
	scope = character
	effect = {
		custom_description_no_bullet = {
			text = POD_fleshcraft_modify_layer
			if = {
				limit = { exists = scope:layer }
				POD_fleshcrafting_select_layer_for_modification = {
					LAYER = scope:layer
				}
			}
		}
	}
}

POD_fleshcrafting_modify_bone = {
	scope = character
	is_valid = {
		has_perk = rend_the_osseous_frame_perk
	}
	effect = {
		if = {
			limit = { has_perk = rend_the_osseous_frame_perk }
			custom_description_no_bullet = {
				text = POD_fleshcraft_modify_layer
				if = {
					limit = { exists = scope:layer }
					POD_fleshcrafting_select_layer_for_modification = {
						LAYER = scope:layer
					}
				}
			}
		}
	}
}

POD_fleshcrafting_select_perk = {
	scope = character
	saved_scopes = { flag }
	is_shown = {
		always = yes
	}
	is_valid = {
		always = yes
	}
	effect = {
		POD_fleshcrafting_choose_perk = yes
	}
}

# needs scope:layer on the TopScope
POD_fleshcrafting_clear_layer_perk = {
	scope = character
	saved_scopes = { layer }
	effect = {
		POD_fleshcrafting_clear_layer_perk_from_specific_layer = yes
	}
}

POD_fleshcrafting_trait_tooltip = {
	# global scope, since this would also display the character's name if it were scoped to them
	effect = {
		POD_fleshcrafting_list_all_perk_effects_on_character = yes
	}
}
