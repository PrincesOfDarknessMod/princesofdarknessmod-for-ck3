pod_special_slots_map = {
	effect = {
		every_barony = {
			if = {
				limit = { title_province = { has_special_building_slot = yes } }
				if = {
					limit = { title_province = { has_special_building = yes } }
					set_color_from_title = title:k_kham
				}
				else = {
					set_color_from_title = title:k_pagan
				}
			}
			else = {
				set_color_from_title = title:k_angara
			}
		}
	}
	is_shown = {
		always = yes
	}
	is_valid = {
		always = yes
	}
}

pod_fetter_map = {
	effect = {
		random_player = {
			limit = { has_trait = wraith }
			save_scope_as = observer
			every_barony = {
				if = {
					limit = {
						barony_controller = { 
							OR = {
								is_fetter_of_trigger = { OWNER = scope:observer }
								any_courtier_or_guest = {
									is_fetter_of_trigger = { OWNER = scope:observer }
								}
							}
						}
					}
					set_color_from_title = title:k_kham
				}
				else_if = {
					limit = {
						barony_controller = { 
							OR = { 
								any_character_artifact = {
									is_fetter_of_trigger = { OWNER = scope:observer }
								}
								any_courtier_or_guest = {
									any_character_artifact = {
										is_fetter_of_trigger = { OWNER = scope:observer }
									}
								}
							}
						}
					}
					set_color_from_title = title:k_tsang
				}
				else_if = {
					limit = {
						any_this_title_or_de_jure_above = {
							scope:observer = { is_my_fetter_trigger = { FETTER = prev } }
						}
					}
					set_color_from_title = title:k_pagan
				}
				else = {
					set_color_from_title = title:k_angara
				}
			}
		}
	}
	saved_scopes = { player }
	is_shown = {
		exists = scope:player
		scope:player = { has_perk = sense_strand_perk }
	}
	is_valid = {
		always = yes
	}
}

pod_reset_title_name = {
	effect = {
		if = {
			limit = {
				has_global_variable = had_title_name_changed_to_map_label
			}
			every_independent_ruler = {
				primary_title = { 
					reset_title_name = yes
					save_scope_as = title
				}
				trigger_event = POD_title_flavor.0002
			}
			remove_global_variable = had_title_name_changed_to_map_label
		}
	}
	is_shown = {
		always = yes
	}
	is_valid = {
		always = yes
	}
}
