pod_special_slots_map = {
	effect = {
		every_barony = {
			if = {
				limit = { title_province = { has_special_building_slot = yes } }
				if = {
					limit = { title_province = { has_special_building = yes } }
					set_color_from_title = title:k_kham
				}
				else = {
					set_color_from_title = title:k_pagan
				}
			}
			else = {
				set_color_from_title = title:k_angara
			}
		}
	}
	is_shown = {
		always = yes
	}
	is_valid = {
		always = yes
	}
}

pod_fetter_map = {
	effect = {
		random_player = {
			limit = { has_trait = wraith }
			save_scope_as = observer
			every_barony = {
				if = {
					limit = {
						barony_controller = { 
							OR = {
								is_fetter_of_trigger = { OWNER = scope:observer }
								any_courtier_or_guest = {
									is_fetter_of_trigger = { OWNER = scope:observer }
								}
							}
						}
					}
					set_color_from_title = title:k_kham
				}
				else_if = {
					limit = {
						barony_controller = { 
							OR = { 
								any_character_artifact = {
									is_fetter_of_trigger = { OWNER = scope:observer }
								}
								any_courtier_or_guest = {
									any_character_artifact = {
										is_fetter_of_trigger = { OWNER = scope:observer }
									}
								}
							}
						}
					}
					set_color_from_title = title:k_tsang
				}
				else_if = {
					limit = {
						any_this_title_or_de_jure_above = {
							is_fetter_of_trigger = { OWNER = scope:observer }
						}
					}
					set_color_from_title = title:k_pagan
				}
				else = {
					set_color_from_title = title:k_angara
				}
			}
		}
	}
	is_shown = {

	is_valid = {
		always = yes
	}
}
