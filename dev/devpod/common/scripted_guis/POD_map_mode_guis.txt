pod_ruler_artifact_map = {
	effect = {
		every_barony = {
			if = {
				limit = {
					county.holder = {  
						any_character_artifact = {
							rarity = illustrious
						}
					}
				}
				set_color_from_title = title:d_tuyuhun
			}
			else_if = {
				limit = {
					county.holder = {
						any_character_artifact = {
							rarity = famed
						}
					}
				}
				set_color_from_title = title:c_amiens
			}
			else_if = {
				limit = {
					county.holder = {
						any_character_artifact = {
							rarity = masterwork
						}
					}
				}
				set_color_from_title = title:d_baghdad
			}
			else_if = {
				limit = {
					county.holder = {
						any_character_artifact = {
							rarity = common
						}
					}
				}
				set_color_from_title = title:d_alwa
			}
			else = {
				set_color_from_title = title:k_angara
			}
		}
	}
}

pod_fetter_map = {
	effect = {
		random_player = {
			limit = { has_trait = wraith }
			save_scope_as = observer
			every_barony = {
				if = {
					limit = {
						barony_controller = { 
							OR = {
								AND = {
									has_variable = fetter_owner
									var:fetter_owner = scope:observer
								}
								any_courtier_or_guest = {
									AND = {
										has_variable = fetter_owner
										var:fetter_owner = scope:observer
									}
								}
							}
						}
					}
					set_color_from_title = title:e_color_vampire
				}
				else_if = {
					limit = {
						barony_controller = { 
							OR = { 
								any_character_artifact = {
									AND = {
										has_variable = fetter_owner
										var:fetter_owner = scope:observer
									}
								}
								any_courtier_or_guest = {
									any_character_artifact = {
										AND = {
											has_variable = fetter_owner
											var:fetter_owner = scope:observer
										}
									}
								}
							}
						}
					}
					set_color_from_title = title:c_gyangze
				}
				else_if = {
					limit = {
						any_this_title_or_de_jure_above = {
							scope:observer = { 
								AND = {
									has_variable = fetter_owner
									var:fetter_owner = PREV
								}
							}
						}
					}
					set_color_from_title = title:k_pagan
				}
				else = {
					set_color_from_title = title:k_angara
				}
			}
		}
	}
	saved_scopes = { player }
	is_shown = {
		exists = scope:player
		scope:player = { has_perk = sense_strand_perk }
	}
}
