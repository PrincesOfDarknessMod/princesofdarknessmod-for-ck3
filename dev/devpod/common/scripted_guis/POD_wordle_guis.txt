POD_wordle_color_gui_1 = {
	scope = character
	is_valid = {
		var:wordle_minigame = {
			POD_wordle_green_check = { N = 1 L = scope:flag }
		}
	}
	is_shown = {
		var:wordle_minigame = {
			POD_wordle_yellow_check = { N = 1 L = scope:flag }
		}
	}
}

POD_wordle_color_gui_2 = {
	scope = character
	is_valid = {
		var:wordle_minigame = {
			POD_wordle_green_check = { N = 2 L = scope:flag }
		}
	}
	is_shown = {
		var:wordle_minigame = {
			NOT = { POD_wordle_green_check = { N = 2 L = scope:flag } }
			is_target_in_variable_list = {
				name = wordle_solution_letters
				target = scope:flag
			}
		}
	}
}
