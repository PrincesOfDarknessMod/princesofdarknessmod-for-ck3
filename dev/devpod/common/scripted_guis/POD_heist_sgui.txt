POD_heist_is_current_location_sgui = {
	scope = character
	saved_scopes = { story } # Node story_cycle
	is_shown = {
		var:current_heist ?= {
			var:current_location ?= scope:story
		}
	}
}

POD_heist_node_button_sgui = {
	scope = character
	saved_scopes = { story } # Node story_cycle
	is_shown = {
		scope:story = { var:explored = yes }
	}
	is_valid = {
		var:current_heist ?= {
			var:current_location ?= {
				POD_heist_is_node_connected_trigger = { TARGET = scope:story }
			}
		}
	}
	effect = {
		switch = {
			trigger = scope:story.var:type
			flag:node_exit      = {}
			flag:node_artifact  = {}
			flag:node_event     = {}
			flag:node_diplomacy = {}
			flag:node_intrigue  = {}
			flag:node_learning  = {}
			flag:node_fight     = {}
			flag:node_money     = {}
			flag:node_bribe     = {}
		}
		var:current_heist ?= {
			set_variable = {
				name  = current_location
				value = scope:story
			}
			POD_heist_refresh = yes
		}
	}
}

POD_heist_line_sgui = {
	scope = character
	saved_scopes = { story } # Line story_cycle
	is_valid = {
		scope:story.var:status ?= flag:available
	}
}
