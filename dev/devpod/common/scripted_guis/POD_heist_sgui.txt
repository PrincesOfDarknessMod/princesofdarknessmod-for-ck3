			text = POD_heist_node_isvalid_tt

				custom_tooltip = POD_heist_effect_node_event
			}

				add_prestige = miniscule_prestige_loss
				scope:diplomacy_duelist = {
					duel = {
						skill = diplomacy
						value = @skill_duel_target

							compare_modifier = {
								value = scope:duel_value
								multiplier = 0.5
							}
							desc = POD_heist_effect_node_diplomacy_success
						}
						13 = {
							desc = POD_heist_effect_node_diplomacy_fail
							root = { POD_heist_raise_suspicion_effect = { VALUE = minor_suspicion_value } }
						}
						13 = {
							compare_modifier = {
								value = scope:duel_value
								multiplier = -0.5
							}
							desc = POD_heist_effect_node_diplomacy_critfail
							root = { POD_heist_raise_suspicion_effect = { VALUE = major_suspicion_value } }
						}
					}
				}
			}
			flag:node_intrigue = {
				add_prestige = miniscule_prestige_loss
				scope:intrigue_duelist = {
					duel = {
						skill = intrigue

								value = scope:duel_value
								multiplier = -0.5
							}
							desc = POD_heist_effect_node_intrigue_critfail
							root = { POD_heist_raise_suspicion_effect = { VALUE = major_suspicion_value } }
						}
					}
				}
			}
			flag:node_learning = {
				add_prestige = miniscule_prestige_loss
				scope:learning_duelist = {
					duel = {
						skill = learning
						value = @skill_duel_target
						13 = {
							compare_modifier = {
								value = scope:duel_value
								multiplier = 0.5
							}
							desc = POD_heist_effect_node_learning_success

							desc = POD_heist_effect_node_learning_fail
