POD_heist_node_button_sgui = {
	scope = character
	saved_scopes = { story } # Node story_cycle
	is_shown = {
		scope:story = { var:explored = yes }
	}
	is_valid = {
		custom_tooltip = {
			text = POD_heist_heist_paused_tt
			var:current_heist ?= { POD_heist_is_paused_trigger = no }
		}
		custom_tooltip = {
			text = POD_heist_node_isvalid_tt
			var:current_heist ?= {
				var:current_location ?= {
					POD_heist_is_node_connected_trigger = { TARGET = scope:story }
				}
			}
		}
	}
	effect = {
		POD_save_coterie_duelists = yes

		switch = {
			trigger = scope:story.var:type
			flag:node_exit = {
				POD_heist_raise_suspicion_effect = { VALUE = minor_suspicion_value }
			}
			flag:node_empty = {
				custom_tooltip = NO_EFFECT
			}
			flag:node_artifact = {
				POD_heist_add_artifact_to_loot = yes
			}
			flag:node_vault = {
				POD_heist_raise_suspicion_effect = { VALUE = minor_suspicion_value }
			}
			flag:node_event = {
				custom_tooltip = POD_heist_effect_node_event
			}
			flag:node_diplomacy = {
				POD_heist_add_and_track_prestige = { AMOUNT = heist_node_prestige_loss }
				POD_heist_stat_duel_effect = {
					STAT   = diplomacy
					VALUE  = root.heist_stat_duel_difficulty
					WEIGHT = 13
					ONFAIL = ""
				}
			}
			flag:node_stewardship = {
				POD_heist_add_and_track_gold = { AMOUNT = heist_node_gold_loss }
				POD_heist_stat_duel_effect = {
					STAT   = stewardship
					VALUE  = root.heist_stat_duel_difficulty
					WEIGHT = 13
					ONFAIL = ""
				}
			}
			flag:node_intrigue = {
				POD_heist_add_and_track_prestige = { AMOUNT = heist_node_prestige_loss }
				POD_heist_stat_duel_effect = {
					STAT   = intrigue
					VALUE  = root.heist_stat_duel_difficulty
					WEIGHT = 13
					ONFAIL = ""
				}
			}
			flag:node_learning = {
				POD_heist_add_and_track_prestige = { AMOUNT = heist_node_prestige_loss }
				POD_heist_stat_duel_effect = {
					STAT   = learning
					VALUE  = root.heist_stat_duel_difficulty
					WEIGHT = 13
					ONFAIL = "root = { increase_wounds_effect = { REASON = wounds } }"
				}
			}
			flag:node_prowess = {
				POD_heist_add_and_track_piety = { AMOUNT = heist_node_piety_loss }
				POD_heist_stat_duel_effect = {
					STAT   = prowess
					VALUE  = root.heist_prowess_duel_difficulty
					WEIGHT = 13
					ONFAIL = "root = { increase_wounds_effect = { REASON = wounds } }"
				}
			}
			flag:node_fight_elite = {}
			flag:node_money = {
				POD_heist_raise_suspicion_effect = { VALUE = minor_suspicion_value }
			}
			flag:node_fame = {
				POD_heist_raise_suspicion_effect = { VALUE = minor_suspicion_value }
			}
			flag:node_gossip = {
				POD_heist_raise_suspicion_effect = { VALUE = minor_suspicion_value }
			}
			flag:node_destress = {
				POD_heist_raise_suspicion_effect = { VALUE = minor_suspicion_value }
			}
		}

		var:current_heist ?= {
			set_variable = {
				name  = current_location

			}

	}
}

POD_heist_option_gossip_sgui = {
	scope = character
	is_shown = {
		var:current_heist ?= {
			var:current_location ?= {
				var:type = flag:node_gossip
			}
		}
	}
	effect = {
		POD_heist_raise_suspicion_effect = { VALUE = major_suspicion_value }
		custom_tooltip = {
			text = POD_heist_effect_node_gossip
			var:current_heist.var:court_owner = {
				add_to_list = hook_targets
				if = {
					limit = { any_secret = { NOT = { is_known_by = root } } }
					add_to_list = secret_holders
				}
				every_courtier = {
					add_to_list = hook_targets
					if = {
						limit = { any_secret = { NOT = { is_known_by = root } } }
						add_to_list = secret_holders
					}
				}
				every_vassal = {
					add_to_list = hook_targets
					if = {
						limit = { any_secret = { NOT = { is_known_by = root } } }
						add_to_list = secret_holders
					}
				}
			}
			random_list = {
				50 = {
					trigger = {
						list_size = {
							name = secret_holders
							value > 0
						}
					}
					random_in_list = {
						list = secret_holders
						random_secret = {
							limit = { NOT = { is_known_by = root } }
							reveal_to = root
						}
					}
				}
				50 = {
					random_in_list = {
						list = hook_targets
						limit = {
							root = {
								can_add_hook = {
									type   = manipulation_hook
									target = prev
								}
							}
						}
						root = {
							add_hook = {
								type   = manipulation_hook
								target = prev
							}
						}
					}
				}
			}
		}
	}
}

POD_heist_option_destress_sgui = {
	scope = character
	is_shown = {
		var:current_heist ?= {
			var:current_location ?= {
				var:type = flag:node_destress
			}
		}
	}
	effect = {
		POD_heist_raise_suspicion_effect = { VALUE = major_suspicion_value }
		if = {
			limit = { POD_has_coterie_members_trigger = yes }
			var:coterie = {
				every_in_list = {
					variable = members
					limit = {
						POD_is_in_same_journey_trigger = { TARGET = root }
					}
					add_stress = medium_stress_loss
				}
			}
		}
		else = {
			add_stress = medium_stress_loss
		}
	}
