		}

		}
	}
	effect = {
		if = {
			limit = { POD_heist_found_all_keys_trigger = no }

		}
		else_if = {

				OR = {
					POD_can_cause_heist_bankruptcy_trigger = yes

				}

				1 = { POD_heist_raise_suspicion_effect = { VALUE = minor_suspicion_value } }

			}
			if = {
				limit = { POD_can_cause_heist_bankruptcy_trigger = yes }
				random_list = {
					5 = {}
					5 = { POD_heist_steal_and_track_gold = { AMOUNT = heist_main_objective_minor_gold_value } }
					5 = { POD_heist_steal_and_track_gold = { AMOUNT = heist_main_objective_major_gold_value } }
					3 = { POD_heist_steal_and_track_gold = { AMOUNT = heist_main_objective_massive_gold_value } }

			}
			else = {
				random_list = {
					5 = {}
					5 = { POD_heist_steal_and_track_gold = { AMOUNT = heist_main_objective_minor_gold_capped_value } }
					5 = { POD_heist_steal_and_track_gold = { AMOUNT = heist_main_objective_major_gold_capped_value } }

					text = POD_heist_treasury_amount_desc

POD_heist_option_gossip = {
	scope = character
	# is_shown = {
	# 	var:current_heist ?= {
	# 		var:current_location ?= {
	# 			var:type = flag:node_gossip
	# 		}
	# 	}
	# }
	effect = {
		#POD_heist_raise_suspicion_effect = { VALUE = major_suspicion_value }
		random_list = {
			1 = { POD_heist_raise_suspicion_effect = { VALUE = minor_suspicion_value } }
			1 = { POD_heist_raise_suspicion_effect = { VALUE = major_suspicion_value } }
		}
		custom_tooltip = {
			text = POD_heist_effect_node_gossip
			var:current_heist.var:court_owner = {
				add_to_list = hook_targets
				every_secret = {
					limit = { NOT = { is_known_by = root } }
					add_to_list = secrets
				}
				every_courtier = {
					limit = { POD_is_available_for_events = yes }
					add_to_list = hook_targets
					every_secret = {
						limit = { NOT = { is_known_by = root } }
						add_to_list = secrets
					}
				}
				every_vassal_or_below = {
					limit = { POD_is_available_for_events = yes }
					add_to_list = hook_targets
					every_secret = {
						limit = { NOT = { is_known_by = root } }
						add_to_list = secrets
					}
				}
			}
			random_list = {
				50 = {
					trigger = {
						list_size = {
							name = secrets
							value > 0
						}
					}
					random_in_list = {
						list = secrets
						save_scope_as = secret
						root = { trigger_event = POD_heists.6000 }
					}
				}
				50 = {
					random_in_list = {
						list = hook_targets
						limit = {
							root = {
								can_add_hook = {
									type   = manipulation_hook
									target = prev
								}
							}
						}
						root = {
							add_hook = {
								type   = manipulation_hook
								target = prev
							}
						}
					}
				}
			}
		}
	}
}

POD_heist_option_prisoners = {
	scope = character
	# is_shown = {
	# 	var:current_heist ?= {
	# 		var:current_location ?= {
	# 			var:type = flag:node_prisoners
	# 		}
	# 	}
	# }
	effect = {
		if = {
			limit = { var:current_heist.var:court_owner ?= { has_prisoners = yes } }
			POD_heist_raise_suspicion_effect = { VALUE = minor_suspicion_value }

			##### Prisoners are selected by order of importance

			var:current_heist.var:court_owner ?= {
				ordered_prisoner = {
					order_by = "this.opinion(root)" # Recalculating opinions is slow in the engine, maybe? idk
					limit = {
						OR = {
							is_spouse_of = root
							is_close_family_of = root
							has_relation_friend = root
							has_relation_lover = root
							has_relation_coterie_member = root
						}
					}
					save_scope_as = prisoner_to_release
					save_scope_value_as = {
						name  = prisoner_type
						value = flag:relation
					}
				}

				if = {
					limit = { NOT = { exists = scope:prisoner_to_release } }
					ordered_prisoner = {
						order_by = "this.opinion(root)"
						limit = {
							OR = {
								is_courtier_of = root
								is_vassal_or_below_of = root
							}
						}
						save_scope_as = prisoner_to_release
						save_scope_value_as = {
							name  = prisoner_type
							value = flag:subject
						}
					}
				}

				if = {
					limit = { NOT = { exists = scope:prisoner_to_release } }
					ordered_prisoner = {
						order_by = "this.opinion(root)"
						limit = {
							save_temporary_scope_as = potential_prisoner
							NOT = {
								POD_custom_incompatible_splats_trigger = {
									ACTOR = root
									RECIPIENT = scope:potential_prisoner
								}
							}
						}
						save_scope_as = prisoner_to_release
						save_scope_value_as = {
							name  = prisoner_type
							value = flag:friendly_splat
						}
					}
				}

				if = {
					limit = { NOT = { exists = scope:prisoner_to_release } }
					ordered_prisoner = {
						order_by = "this.opinion(root)"
						save_scope_as = prisoner_to_release
						save_scope_value_as = {
							name  = prisoner_type
							value = flag:hostile_splat
						}
					}
				}
			}

			custom_description = {
				text = POD_heist_release_prisoner_desc
				subject = scope:prisoner_to_release
				trigger_event = POD_heists.200
			}
		}
		else = {
			custom_tooltip = POD_heist_option_prisoners_empty
		}
	}
}
