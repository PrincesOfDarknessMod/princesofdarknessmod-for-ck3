POD_portrait_diffuse_r_toggle = {
	scope = character
	effect = {
		if = {
			limit = { has_variable = POD_portrait_diffuse_r }
			remove_variable = POD_portrait_diffuse_r
		}
		else = {
			set_variable = {
				name  = POD_portrait_diffuse_r
				value = 1.0
			}
		}
	}
}
POD_portrait_diffuse_r_decrease = {
	scope = character
	is_valid = {
		has_variable = POD_portrait_diffuse_r
		var:POD_portrait_diffuse_r > 0.05
	}
	effect = {
		change_variable = {
			name = POD_portrait_diffuse_r
			add  = -0.05
		}
	}
}
POD_portrait_diffuse_r_increase = {
	scope = character
	is_valid = {
		has_variable = POD_portrait_diffuse_r
		var:POD_portrait_diffuse_r < 1.0
	}
	effect = {
		change_variable = {
			name = POD_portrait_diffuse_r
			add  = 0.05
		}
	}
}
