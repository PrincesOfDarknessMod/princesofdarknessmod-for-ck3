# need to pass scope:index
POD_show_wound_box = {
	scope = character
	effect = {
		if = {
			limit = { scope:index <= POD_health_track_length }
			if = {
				limit = { scope:index <= POD_aggravated_damage_boxes }
				custom_description_no_bullet = { text = POD_aggravated_damage_box_tt }
			}
			else_if = {
				limit = { scope:index <= POD_total_damage_boxes }
				custom_description_no_bullet = { text = POD_damage_box_tt }
			}
			else = {
				custom_description_no_bullet = { text = POD_undamaged_box_tt }
			}
		}
	}
}
