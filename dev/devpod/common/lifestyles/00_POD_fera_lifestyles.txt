
commongifts_lifestyle = {
	is_valid = {
		trigger_if = {
			limit = {
				OR = {
					has_trait = kinfolk
					has_trait = abomination
				}
			}
			exists = var:fera_type
			#OR = {
				var:fera_type = trait:bastet
			#}
		}
		trigger_else = {
			#OR = {
				has_trait = bastet
			#}
		}
	}
	
	is_highlighted = {
		POD_has_gnosis_modifier_trigger = yes
	}

	xp_per_level = 1000
	base_xp_gain = 10
}

	}

	}
	xp_per_level = 1000
	base_xp_gain = 10
