breedgifts_lifestyle = {
	is_valid = {
		trigger_if = {
			limit = {
				OR = {
					has_trait = kinfolk
					has_trait = abomination
				}
			}
			exists = var:fera_type
			OR = {
				var:fera_type = trait:werewolf
				var:fera_type = trait:bastet
			}
		}
		trigger_else = {
			OR = {
				has_trait = werewolf
				has_trait = bastet
			}
		}
	}
	
	is_highlighted = {
		OR = {
			has_character_modifier = POD_gnosis_1_modifier
			has_character_modifier = POD_gnosis_2_modifier
			has_character_modifier = POD_gnosis_3_modifier
			has_character_modifier = POD_gnosis_4_modifier
			has_character_modifier = POD_gnosis_5_modifier
		}
	}

	xp_per_level = 1000
	base_xp_gain = 10
}
