animal_lifestyle = {
	
	is_valid = {
		OR = {
			has_trait = vampire
			has_trait = abomination
			has_trait = revenant
			has_trait = ghoul
			has_trait = dhampir
			has_character_flag = vamp_discipline_access

			AND = {
				has_perk = renaming_the_hekau_perk
				has_trait_with_flag = animal_discipline
			}
		}
	}
	
	is_highlighted = { }
		
	xp_per_level = 2000
	base_xp_gain = 10
}
