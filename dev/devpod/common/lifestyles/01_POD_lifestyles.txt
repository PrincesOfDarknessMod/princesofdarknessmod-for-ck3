	is_valid = {
		OR = {
			has_trait = vampire

				has_perk = renaming_the_hekau_perk
				has_trait_with_flag = animal_discipline
			}
		}
	}
	
	is_highlighted = { }
		
	xp_per_level = 2000
	base_xp_gain = 10
}
