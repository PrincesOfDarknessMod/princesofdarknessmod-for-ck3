spirit_wilds_influence_01 = {
	construction_time = quick_construction_time
	
	can_construct_potential = {
		scope:holder = {
			has_variable = time_in_spirit_wilds
		}
	}

		trigger_if = {

		POD_spirit_maa_pursuit_mult = 0.05
		POD_animals_maa_damage_mult = 0.05
		POD_animals_maa_pursuit_mult = 0.05

	}
}
