influence_animals_01 = {
	construction_time = quick_construction_time

	flag = vampire
	
	can_construct_potential = {
		scope:holder = {
			has_trait = vampire
			OR = {
				has_perk = call_of_the_wild_perk
				has_trait = animalismadvanced
				has_perk = master_of_snakes_perk
				has_perk = mother_of_monsters_perk
				has_trait = serpentisadvanced
				AND = {
					exists = dynasty
					dynasty = {
						OR = {
							has_dynasty_perk = gangrel_legacy_1
							has_dynasty_perk = nosferatu_legacy_1
							has_dynasty_perk = ravnos_legacy_1
						}
					}
				}
			}
		}
	}

	can_construct = {
	}
	
	is_enabled = {
		scope:holder = {
			has_trait = vampire
			OR = {
				has_perk = call_of_the_wild_perk
				has_trait = animalismadvanced
				has_perk = master_of_snakes_perk
				has_perk = mother_of_monsters_perk
				has_trait = serpentisadvanced
				AND = {
					exists = dynasty
					dynasty = {
						OR = {
							has_dynasty_perk = gangrel_legacy_1
							has_dynasty_perk = nosferatu_legacy_1
							has_dynasty_perk = ravnos_legacy_1
						}
					}
				}
			}
		}	
	}

	cost_prestige = cheap_building_tier_1_cost
	cost_piety = normal_building_tier_1_cost
	
	character_modifier = {
		stress_loss_mult = 0.02
		POD_animals_maa_damage_add = 1
		POD_animals_maa_toughness_add = 1
	}
	county_modifier = {
		development_growth_factor = -0.15
	}

	type_icon = "icon_building_animal.dds"
	
	ai_value = {
		base = 0
		modifier = {
			add = 55
			scope:holder = {
				has_trait = animalismadvanced
			}
		}
		modifier = {
			add = 35
			scope:holder = {
				OR = {
					is_gangrel_trigger = yes
					is_nosferatu_trigger = yes
				}
			}
		}
		modifier = {
			add = -200
			scope:holder = {
				highest_held_title_tier = tier_barony
			}
		}
		modifier = {
			add = 45
			scope:county = {
				development_level < 5
			}
		}
	}
	
	next_building = influence_animals_02
}
