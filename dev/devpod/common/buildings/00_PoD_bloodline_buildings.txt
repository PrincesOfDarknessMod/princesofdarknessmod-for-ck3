rookery_01 = {
	construction_time = slow_construction_time

	flag = vampireonly
	
	can_construct_potential = {
		scope:holder = {
			custom_tooltip = {
				text = POD_requirements_rookery
				has_trait = vampire
				OR = {
					has_trait = gargoyle
					this = character:790870
				}
			}
		}
	}
	
	is_enabled = {
		scope:holder = {
			custom_tooltip = {
				text = POD_requirements_rookery
				has_trait = vampire
				OR = {
					has_trait = gargoyle
					this = character:790870
				}
			}
		}
	}

	cost_gold = cheap_building_tier_1_cost
	cost_piety = cheap_building_tier_1_cost
	
	character_modifier = {
		POD_gargoyle_maa_recruitment_cost_mult = -0.05
		dynasty_opinion = 2
	}


	county_modifier = {
		hostile_raid_time = 0.075
		monthly_county_control_growth_factor = 0.075
	}
	
	province_modifier = {
		defender_holding_advantage = normal_building_advantage_tier_1
		fort_level = good_building_fort_level_tier_1
		stationed_POD_gargoyle_maa_damage_mult = 0.01
		stationed_POD_gargoyle_maa_toughness_mult = 0.05
		stationed_POD_gargoyle_maa_screen_mult = 0.05
		stationed_POD_gargoyle_maa_pursuit_mult = 0.025
	}

#	province_terrain_modifier = {
#		parameter = castle_holding_fort_level_in_mountain_bonus
#		terrain = mountain
#		additional_fort_level = castle_holding_fort_level_in_mountain_bonus_value
#	}


	type_icon = "rookery.dds"
	
	ai_value = {
		base = 10
		ai_tier_1_building_modifier = yes
		ai_general_building_modifier = yes
		modifier = {
			add = stacking_bonus_value
			scope:holder = {
				max_number_maa_soldiers_of_base_type = {
					type = POD_gargoyle_maa
					value >= 200
				}
			}
		}
		modifier = {
			add = non_stationed_maa_bonus
			has_stationed_regiment = yes
			NOT = { has_stationed_regiment_of_base_type = POD_gargoyle_maa }
		}
	}
	next_building = rookery_02
}
