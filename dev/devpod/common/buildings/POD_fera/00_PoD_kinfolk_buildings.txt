kinfolk_meeting_grounds_01 = {
	construction_time = quick_construction_time

	flag = fera
	flag = kinfolk

	type_icon = "icon_building_longhouses.dds"
	
	can_construct_potential = {
		scope:holder = { building_requirement_kinfolk = yes }
	}
	
	is_enabled = {
		scope:holder = { building_requirement_kinfolk = yes }
	}
	
	cost_prestige = kinfolk_building_prestige_cost_t1
	cost_gold = kinfolk_building_gold_cost_t1

	province_modifier = {
		supply_limit = poor_building_supply_limit_tier_1
		stationed_skirmishers_damage_mult = 0.1
		stationed_skirmishers_toughness_mult = 0.1
		stationed_archers_damage_mult = 0.1
		stationed_archers_toughness_mult = 0.1
		stationed_light_cavalry_damage_mult = 0.1
		stationed_light_cavalry_toughness_mult = 0.1
	}
	
	county_modifier = {
		county_opinion_add = 5
	}
	
	ai_value = {
		base = 20
		modifier = {
			add = stacking_bonus_value
			scope:holder = {
				OR = {
					max_number_maa_soldiers_of_base_type = { type = skirmishers value >= 200 }
					max_number_maa_soldiers_of_base_type = { type = archers value >= 200 }
					max_number_maa_soldiers_of_base_type = { type = light_cavalry value >= 200 }
				}
			}
		}
		modifier = {
			add = non_stationed_maa_bonus
			has_stationed_regiment = yes
			NOR = {  
				has_stationed_regiment_of_base_type = skirmishers
				has_stationed_regiment_of_base_type = archers
				has_stationed_regiment_of_base_type = light_cavalry
			}
		}
	}
	
	show_disabled = yes
	
	next_building = kinfolk_meeting_grounds_02
}
