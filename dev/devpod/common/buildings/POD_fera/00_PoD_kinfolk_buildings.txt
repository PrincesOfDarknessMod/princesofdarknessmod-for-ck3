	construction_time = quick_construction_time

	flag = fera
	flag = kinfolk
	
	can_construct_potential = {
		scope:holder = { building_requirement_kinfolk = yes }
	}
	
	is_enabled = {
		scope:holder = { building_requirement_kinfolk = yes }
	}
	
	cost_prestige = kinfolk_building_prestige_cost_t1
	cost_gold = kinfolk_building_gold_cost_t1
		
	province_modifier = {
		supply_limit = 200
		stationed_POD_fera_maa_damage_mult = 0.05
		stationed_POD_fera_maa_toughness_mult = 0.05
	}
	
	
	county_modifier = {
		county_opinion_add = 5
	}

	type_icon = "icon_building_guild_halls.dds"
	
	ai_value = {
		base = 20
		modifier = {
			add = stacking_bonus_value
			scope:holder = {
				max_number_maa_soldiers_of_base_type = {
					type = POD_fera_maa
					value >= 200
				}
			}
		}
		modifier = {
			add = non_stationed_maa_bonus
			has_stationed_regiment = yes
			NOT = {  has_stationed_regiment_of_base_type = POD_fera_maa }
		}
	}
	
	show_disabled = yes
	
	next_building = kinfolk_meeting_grounds_02
}
