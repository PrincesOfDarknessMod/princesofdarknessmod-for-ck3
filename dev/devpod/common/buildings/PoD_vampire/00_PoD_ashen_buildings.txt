### Ashen Nobles ###

ashen_court_01 = {
	construction_time = standard_construction_time
	
	flag = vampireonly
		
	is_enabled = {
		#building_disable_if_vastly_over_domain_limit_trigger = yes
		scope:holder = {
			has_trait = vampire
			}
		}
	
	can_construct_potential = {
		scope:holder = {
			has_trait = vampire
			dynasty ?= { has_dynasty_perk = ashen_nobles_legacy_1 }
		}
	}
		
	cost_gold = 400
	
	max_garrison = 100
	county_modifier = {
		monthly_county_control_growth_add = 0.1
	}
	character_modifier = {
		courtier_and_guest_opinion = 1
		monthly_prestige = 0.1
		monthly_diplomacy_lifestyle_xp_gain_mult = 0.01
	}
	
	type_icon = "icon_building_leisure_palace.dds"
	
	ai_value = {
		base = 0
		modifier = {
			add = 80
			AND = {
			scope:holder.dynasty = {
				has_dynasty_perk = ashen_nobles_legacy_5
				}
				is_county_capital = yes
				num_buildings > 2
			}
		}
		modifier = {
			add = stacking_bonus_value
			scope:holder = {
				max_number_maa_soldiers_of_base_type = {
					type = heavy_infantry
					value >= 200
				}
			}
		}
		modifier = {
			add = non_stationed_maa_bonus
			has_stationed_regiment = yes
			NOT = {  has_stationed_regiment_of_base_type = heavy_infantry }
		}
	}
	next_building = ashen_court_02
}
