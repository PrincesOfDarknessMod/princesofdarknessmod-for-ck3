influence_animals_01 = {
	construction_time = quick_construction_time

	flag = vampireonly
	
	can_construct_potential = {
		scope:holder = {
			has_trait = vampire
			building_requirement_influence_animals = yes
		}
	}

	can_construct = {
	}
	
	is_enabled = {
		scope:holder = {
			has_trait = vampire
			building_requirement_influence_animals = yes
		}	
	}

	cost_prestige = cheap_building_tier_1_cost
	cost_piety = normal_building_tier_1_cost
	
	province_modifier = {
		stationed_POD_animals_maa_damage_mult = 0.025
		stationed_POD_animals_maa_toughness_mult = 0.025		
	}
	
	character_modifier = {
		stress_loss_mult = 0.02
	}
	county_modifier = {
		development_growth_factor = -0.15
	}

	type_icon = "icon_building_animal.dds"
	
	ai_value = {
		base = 0
		modifier = {
			add = 55
			scope:holder = {
				has_trait = animalismadvanced
			}
		}
		modifier = {
			add = 35
			scope:holder = {
				OR = {
					is_gangrel_clan_or_bloodline_trigger = yes
					is_nosferatu_clan_or_bloodline_trigger = yes
				}
			}
		}
		modifier = {
			factor = 0
			scope:holder = {
				highest_held_title_tier = tier_barony
			}
		}
		modifier = {
			add = 45
			scope:county = {
				development_level < 5
			}
		}
		modifier = {
			add = stacking_bonus_value
			scope:holder = {
				max_number_maa_soldiers_of_base_type = {
					type = POD_animals_maa
					value >= 200
				}
			}
		}
		modifier = {
			add = non_stationed_maa_bonus
			has_stationed_regiment = yes
			NOT = { has_stationed_regiment_of_base_type = POD_animals_maa }
		}
	}
	
	next_building = influence_animals_02
}
