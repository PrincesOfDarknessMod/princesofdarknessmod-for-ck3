tower_of_pisa = {
	construction_time = very_slow_construction_time

	type_icon = "special_buildings/tower_of_pisa.dds"

	asset = {
		type = entity
		name = "gdp_building_special_pisa_straight_entity"
	}

	can_construct_potential = {
		building_requirement_tribal = no
	}

	cost_gold = 1000

	character_modifier = {
		monthly_prestige_gain_mult = 0.05
		monthly_dynasty_prestige_mult = 0.05
	}
	county_modifier = {
		prosperity_growth_factor = 0.1
		popular_opinion_add = building_miniscule_popular_opinion_gain
	}
	#market_modifier = {
	#	trade_good_stone_sell_price_mult = special_building_need_sell_price
	#}

	on_complete = {
		save_scope_as = tower_of_pisa_province
		if = {
			limit = {
				scope:character ?= { is_ai = no }
			}
			set_variable = tower_of_pisa_to_inspect
			scope:character = {
				if = {
					limit = {
						cp:councillor_steward ?= { is_available_ai_adult = yes }
					}
					cp:councillor_steward = { save_scope_as = tower_of_pisa_messenger }
				}
				else_if = {
					limit = {
						any_councillor = { is_available_ai_adult = yes }
					}
					random_councillor = {
						limit = { is_available_ai_adult = yes }
						save_scope_as = tower_of_pisa_messenger
					}
				}
				else_if = {
					limit = {
						any_courtier = { is_available_ai_adult = yes }
					}
					random_courtier = {
						limit = { is_available_ai_adult = yes }
						save_scope_as = tower_of_pisa_messenger
					}
				}
				if = {
					limit = { exists = scope:tower_of_pisa_messenger }
					trigger_event = gdp_travel_events.2002
				}
				else = {
					trigger_event = gdp_travel_events.2003
				}
			}
		}
		else = {
			county.holder = {
				trigger_event = {
					id = gdp_travel_events.2001
					days = 1
				}
			}
		}
	}

	ai_value = {
		base = 10
		ai_opinion_baseline_building_modifier = { VALUE = 5 }
		modifier = {
			factor = 0
			free_building_slots > 0
		}
		ai_popular_opinion_need_modifier = yes
		ai_special_district_modifier = { LEVEL = 1 }
	}

	#
	#	modifiers = {

	#			monthly_commerce_tax = special_building_need_tax
	#		}
	#	}

	flag = travel_point_of_interest_wonder
}
