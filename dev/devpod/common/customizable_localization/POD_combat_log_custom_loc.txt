# Customizable_localization for combat log entries.
#
# Used by the datamodel-driven log section in pod_combat_window.gui:
# the GUI invokes Custom('PODCombatLogResultText') with two saved
# scopes —
#   `entry` (the log entry story scope, for var:outcome /
#            var:attack_result / var:retal_fired axis reads AND for
#            var:damage / var:heal_amount / var:morale_damage_amount /
#            var:morale_heal_amount / var:retal_damage amount reads)
#   `move`  (the executing move story, for damage_type / damage_source /
#            targeting / buff-tier checks the rider customizable_locs
#            need)
# Each trigger branch composes against scope:entry's axis vars;
# matching branch resolves to the corresponding loc key, which
# composes its rendered text from the entry / move scopes plus the
# various rider customizable_locs invoked separately as siblings in
# the GUI hbox.
#
# Three-axis result model: canonical reference in COMBAT_REFERENCE.info §4.
# Branches gate on entry.var:outcome (top-level outcome) first, then
# on entry.var:attack_result (only meaningful when outcome=flag:attack).
# entry.var:retal_fired is an independent overlay — read by the
# separate PODCombatLogRetalRider customizable_loc invoked alongside
# this dispatcher in the GUI hbox.
#
# HP-track outcomes (outcome=attack + attack_result=hp_damage, and
# outcome=heal) split into _aggravated variants in the dispatch below
# — the wound-box icon swaps from superficial to aggravated based on
# scope:move.var:damage_type / heal_type.

PODCombatLogResultText = {
	type = character

	# Defensive short-circuit for system-message rows. The GUI
	# evaluates raw_text regardless of `visible`, so this dispatcher
	# fires even on rows where entry.var:move was never captured.
	# Without this branch the inner scope:move = { ... } context
	# switches in the aggravated-variant triggers would fail and
	# spam "scope:move trigger [Failed context switch]" errors on
	# every system row. The visible gate in the GUI keeps the result
	# column from DISPLAYING this text on system rows; this branch
	# keeps the evaluator from ERRORING.
	text = {
		trigger = { NOT = { exists = scope:move } }
		localization_key = pod_combat_log_result_text_unknown
	}

	# ── outcome=attack — inner attack_result dispatch ────────────
	# Each amount-bearing branch (hp_damage) gates on damage > 0
	# before showing the amount-styled loc; the 0-damage fallback
	# threads down to the unknown line so we don't render "0 HP"
	# rows that mean "the damage pipeline didn't fire properly".
	#
	# HP damage splits on damage_type so the line ends in the
	# matching wound-box icon (aggravated box vs superficial box)
	# rather than the literal word "HP". Aggravated is checked
	# first; bare superficial is the fallback. Soaked / missed /
	# dodged / stealthed / guard_blocked render as plain status
	# labels with no amount.

	# hp_damage — attack landed wounds
	text = {
		trigger = {
			scope:entry = {
				var:outcome = flag:attack
				has_variable = attack_result
				var:attack_result = flag:hp_damage
				var:damage > 0
			}
			scope:move = { has_variable = damage_type var:damage_type = flag:aggravated }
		}
		localization_key = pod_combat_log_result_text_hp_damage_aggravated
	}
	text = {
		trigger = {
			scope:entry = {
				var:outcome = flag:attack
				has_variable = attack_result
				var:attack_result = flag:hp_damage
				var:damage > 0
			}
		}
		localization_key = pod_combat_log_result_text_hp_damage
	}

	text = {
		trigger = {
			scope:entry = {
				var:outcome = flag:attack
				has_variable = attack_result
				var:attack_result = flag:guard_blocked
			}
		}
		localization_key = pod_combat_log_result_text_guard_blocked
	}
	text = {
		trigger = {
			scope:entry = {
				var:outcome = flag:attack
				has_variable = attack_result
				var:attack_result = flag:soaked
			}
		}
		localization_key = pod_combat_log_result_text_soaked
	}
	text = {
		trigger = {
			scope:entry = {
				var:outcome = flag:attack
				has_variable = attack_result
				var:attack_result = flag:missed
			}
		}
		localization_key = pod_combat_log_result_text_missed
	}
	text = {
		trigger = {
			scope:entry = {
				var:outcome = flag:attack
				has_variable = attack_result
				var:attack_result = flag:dodged
			}
		}
		localization_key = pod_combat_log_result_text_dodged
	}
	text = {
		trigger = {
			scope:entry = {
				var:outcome = flag:attack
				has_variable = attack_result
				var:attack_result = flag:stealthed
			}
		}
		localization_key = pod_combat_log_result_text_stealthed
	}

	# ── outcome=heal — HP heal ───────────────────────────────────
	text = {
		trigger = {
			scope:entry = {
				var:outcome = flag:heal
				has_variable = heal_amount
				var:heal_amount > 0
			}
			scope:move = { has_variable = heal_type var:heal_type = flag:aggravated }
		}
		localization_key = pod_combat_log_result_text_hp_healed_aggravated
	}
	text = {
		trigger = {
			scope:entry = {
				var:outcome = flag:heal
				has_variable = heal_amount
				var:heal_amount > 0
			}
		}
		localization_key = pod_combat_log_result_text_hp_healed
	}

	# ── outcome=morale_damage / morale_healed ────────────────────
	text = {
		trigger = {
			scope:entry = {
				var:outcome = flag:morale_damage
				has_variable = morale_damage_amount
				var:morale_damage_amount > 0
			}
		}
		localization_key = pod_combat_log_result_text_morale_damage
	}
	text = {
		trigger = {
			scope:entry = {
				var:outcome = flag:morale_healed
				has_variable = morale_heal_amount
				var:morale_heal_amount > 0
			}
		}
		localization_key = pod_combat_log_result_text_morale_healed
	}

	# ── outcome=utility (and explicit utility fallback) ──────────
	text = {
		trigger = { scope:entry = { var:outcome = flag:utility } }
		localization_key = pod_combat_log_result_text_utility
	}

	# Fallback — catches 0-amount rows that fell through the
	# "amount > 0" guards above, plus any unexpected combination
	# (outcome=attack with no attack_result, missing fields, etc.).
	# Renders the — placeholder rather than "0 HP" / "+0 Morale".
	text = {
		localization_key = pod_combat_log_result_text_unknown
	}
}

# ────────────────────────────────────────────────────────────────────
# Result-line riders — one customizable_loc per optional bit of info
# that gets appended to the result column. Each rider returns
# either a styled snippet or the empty string (`pod_combat_log_blank`)
# so the row stays clean when the rider doesn't apply.
#
# Riders are invoked as their OWN text_singles in the result-column
# hbox in pod_combat_window.gui, NOT nested inside the main result
# text's loc value. CK3 does seem to not propagate saved scopes
# through nested [GetPlayer.Custom('SubName')] calls in loc strings 
# the sub-custom's scope:X reads fail with "Undefined event target X".
# So every customizable_loc that needs a saved scope must be invoked
# at the GUI level via its own GuiScope.SetRoot(...).AddScope(...)
# chain.
#
# Each rider expects the scopes it dispatches on to be saved by its
# own GUI invocation:
#   Heal / Morale-damage / Morale-heal / Retal riders — `entry`, `move`
#       (entry for amount + outcome-axis reads, move for heal_type /
#       damage_type splits; outcome-based primary-line suppression
#       reads entry.var:outcome)
#   Buff rider                                — `move`
#       (priority-order has_variable checks on the move's stance / tier
#       buff / utility flag vars; no entry needed)

# Heal rider: appended to damage / utility lines when the move ALSO
# restores HP (life-drain pattern, "deal 5 damage and heal 3"). Gated
# on the entry's heal_amount field (set by push_log from
# pod_combat_last_heal_amount), so the rider only fires if a heal
# actually applied, capped heals (target near full HP, heal_amount=0)
# stay quiet.
#
# Suppressed only when outcome=flag:heal (pure-heal primary line,
# where the amount is already shown). HP heals fire regardless of
# attack hit status (the heal section in POD_execute_combat_move
# runs on has_variable = hp_heal alone, no hit gate), so Soaked! /
# Missed! / Dodged! rows still get the rider — those represent moves
# that attempted damage AND healed the actor on use.
#
# Splits on heal_type so the rider's wound-box icon matches what the
# heal actually restored (aggravated heals close agg wounds; default
# is superficial).
PODCombatLogHealRider = {
	type = character
	# Defensive short-circuit for system-message rows — see
	# PODCombatLogBuffRider for the rationale. The aggravated-variant
	# trigger below context-switches into scope:move and would error
	# on system rows where no move was captured.
	text = {
		trigger = { NOT = { exists = scope:move } }
		localization_key = pod_combat_log_blank
	}
	text = {
		trigger = {
			scope:entry = { has_variable = heal_amount var:heal_amount > 0 }
			NOT = { scope:entry = { var:outcome = flag:heal } }
			scope:move = { has_variable = heal_type var:heal_type = flag:aggravated }
		}
		localization_key = pod_combat_log_heal_rider_aggravated
	}
	text = {
		trigger = {
			scope:entry = { has_variable = heal_amount var:heal_amount > 0 }
			NOT = { scope:entry = { var:outcome = flag:heal } }
		}
		localization_key = pod_combat_log_heal_rider
	}
	text = {
		localization_key = pod_combat_log_blank
	}
}

# Morale-damage rider: appended when the move deals morale damage
# alongside HP damage. Gated on the entry's morale_damage_amount
# field. Suppressed on outcome=flag:morale_damage (the morale-only
# primary line already shows the amount).
PODCombatLogMoraleDamageRider = {
	type = character
	text = {
		trigger = {
			scope:entry = { has_variable = morale_damage_amount var:morale_damage_amount > 0 }
			NOT = { scope:entry = { var:outcome = flag:morale_damage } }
		}
		localization_key = pod_combat_log_morale_damage_rider
	}
	text = {
		localization_key = pod_combat_log_blank
	}
}

# Morale-heal rider — appended when a damage / utility move also
# heals morale. Gated on the entry's morale_heal_amount field.
# Suppressed on outcome=flag:morale_healed (the morale-heal primary
# line already shows the amount).
PODCombatLogMoraleHealRider = {
	type = character
	text = {
		trigger = {
			scope:entry = { has_variable = morale_heal_amount var:morale_heal_amount > 0 }
			NOT = { scope:entry = { var:outcome = flag:morale_healed } }
		}
		localization_key = pod_combat_log_morale_heal_rider
	}
	text = {
		localization_key = pod_combat_log_blank
	}
}

# Retal rider — appended whenever the row's retaliation overlay
# fired. Gated on entry.var:retal_fired (set by push_log from
# pod_combat_last_retal_fired) so utility / no-retal rows stay quiet.
# Two branches on the actual retal damage:
#   - retal_damage > 0  → "+ Counter -X HP" (retaliation landed)
#   - retal_damage = 0  → "+ Counter Soaked" (retaliation was absorbed)
# Independent of the main outcome dispatcher — every attack outcome
# (hp_damage / soaked / guard_blocked / missed / dodged / stealthed)
# can carry this overlay alongside the main result line.
PODCombatLogRetalRider = {
	type = character
	# Aggravated retal landed — show the aggravated wound-box icon
	# instead of the default superficial one. retal_agg is captured
	# by push_log from pod_combat_last_retal_agg, which is set by
	# pod_combat_apply_retaliation_helper when it routes the
	# counter-strike via REASON=duel_aggravated (i.e. the move that
	# armed the retal had retaliate_agg set).
	text = {
		trigger = {
			scope:entry = {
				has_variable = retal_fired
				has_variable = retal_damage
				var:retal_damage > 0
				has_variable = retal_agg
			}
		}
		localization_key = pod_combat_log_retal_rider_aggravated
	}
	# Superficial retal landed — default wound-box icon.
	text = {
		trigger = {
			scope:entry = {
				has_variable = retal_fired
				has_variable = retal_damage
				var:retal_damage > 0
			}
		}
		localization_key = pod_combat_log_retal_rider
	}
	# Retal fired but landed 0 damage (soaked / absorbed). No
	# aggravated/superficial distinction matters — both look like
	# "soaked" to the player.
	text = {
		trigger = {
			scope:entry = { has_variable = retal_fired }
		}
		localization_key = pod_combat_log_retal_rider_soaked
	}
	text = {
		localization_key = pod_combat_log_blank
	}
}

# Buff / debuff / status rider — emits a single label naming the most
# prominent stance / tier-buff / utility flag the move applied.
# Branch order is the priority — first match wins, so e.g. a move
# that applies BOTH +Soak and +Hit reports +Soak. Stance flags rank
# above tier buffs (their semantic impact is bigger), tier
# buffs/debuffs sit in the middle, and the utility-modifier flags
# (AP gain/drain, retaliate) sit at the bottom.
PODCombatLogBuffRider = {
	type = character

	# Defensive short-circuit for system-message rows. CK3's GUI
	# evaluates raw_text bindings regardless of the widget's `visible`
	# state, so the rider's AddScope('move', Scope.Var('move').Story.MakeScope)
	# chain still fires on rows where entry.var:move was never
	# captured (system-message rows). Without this branch the inner
	# scope:move = { ... } context switches below would fail and
	# spam "scope:move trigger [Failed context switch]" errors per
	# system row. The visible gate in the GUI keeps these rows from
	# DISPLAYING the rider; this branch keeps the evaluator from
	# ERRORING on them.
	text = {
		trigger = { NOT = { exists = scope:move } }
		localization_key = pod_combat_log_blank
	}

	# ── Stance flags ──
	text = {
		trigger = { scope:move = { has_variable = set_guard } }
		localization_key = pod_combat_log_buff_set_guard
	}
	text = {
		trigger = { scope:move = { has_variable = dodge } }
		localization_key = pod_combat_log_buff_dodge
	}
	text = {
		trigger = { scope:move = { has_variable = stealth } }
		localization_key = pod_combat_log_buff_stealth
	}
	text = {
		trigger = { scope:move = { has_variable = remove_guard } }
		localization_key = pod_combat_log_buff_remove_guard
	}

	# ── Tier buffs / debuffs ──
	text = {
		trigger = {
			scope:move = {
				OR = {
					has_variable = soak_buff_round_tier
					has_variable = soak_buff_combat_tier
				}
			}
		}
		localization_key = pod_combat_log_buff_soak_buff
	}
	text = {
		trigger = {
			scope:move = {
				OR = {
					has_variable = soak_debuff_round_tier
					has_variable = soak_debuff_combat_tier
				}
			}
		}
		localization_key = pod_combat_log_buff_soak_debuff
	}
	text = {
		trigger = {
			scope:move = {
				OR = {
					has_variable = hit_buff_round_tier
					has_variable = hit_buff_combat_tier
				}
			}
		}
		localization_key = pod_combat_log_buff_hit_buff
	}
	text = {
		trigger = {
			scope:move = {
				OR = {
					has_variable = hit_debuff_round_tier
					has_variable = hit_debuff_combat_tier
				}
			}
		}
		localization_key = pod_combat_log_buff_hit_debuff
	}

	# ── Utility modifiers ──
	text = {
		trigger = { scope:move = { has_variable = ap_gain } }
		localization_key = pod_combat_log_buff_ap_gain
	}
	text = {
		trigger = { scope:move = { has_variable = ap_drain } }
		localization_key = pod_combat_log_buff_ap_drain
	}
	text = {
		trigger = {
			scope:move = {
				OR = {
					has_variable = retaliate_if_hit
					has_variable = retaliate_if_missed
				}
			}
		}
		localization_key = pod_combat_log_buff_retaliate
	}
	text = {
		trigger = { scope:move = { has_variable = initiative_debuff } }
		localization_key = pod_combat_log_buff_initiative_debuff
	}

	# ── Blank fallback ──
	text = {
		localization_key = pod_combat_log_blank
	}
}

# ────────────────────────────────────────────────────────────────────
# Damage-source marker.
#
# Renders a small icon next to the result column in the combat log,
# and at the head of the move-tooltip damage line, picking the glyph
# from the move story's var:damage_source (fire / cold / lightning /
# sun / silver / gold / holy / balefire). Moves without a damage_source
# fall through to the blank loc and emit nothing — physical /
# supernatural / mental pipelines all telegraph their flavor through
# the move name itself, and a generic "this is an attack" icon next to
# every damage row would just add noise without telling the player
# anything new. 
#
# Customizable_loc types are limited to character / activity /
# landed_title / artifact / province (see vanilla CK3 examples). So
# rather than typing the loc against story_cycle directly, use the
# same pattern as PODCombatLogResultText above: type = character (root
# is the player), pass the move as a saved scope `move` from the GUI,
# and dispatch on `scope:move = { var:X = flag:Y }`.
#
# Called via:
#   [GuiScope.SetRoot(GetPlayer.MakeScope)
#       .AddScope('move', Scope.Var('move').Story.MakeScope)
#       .Custom('PODCombatDamageSourceMarker')]
# (Scope.Var('move') in the combat-log datamodel returns the stored
# story reference — .Story.MakeScope unwraps it to a real scope.)
#
# The blank-fallback branch at the end keeps the surrounding text-line
# from collapsing when no damage_source matches.

PODCombatDamageSourceMarker = {
	type = character

	# Defensive short-circuit for system-message rows — see
	# PODCombatLogBuffRider for the rationale. Every damage-source
	# branch below context-switches into scope:move and would error
	# on system rows where no move was captured.
	text = {
		trigger = { NOT = { exists = scope:move } }
		localization_key = pod_combat_damage_source_blank
	}

	# Each trigger guards `var:X = flag:Y` reads behind a
	# `has_variable` check so moves without the field don't spam
	# "var returned unset scope" errors during loc evaluation.

	# ── damage_source ────────────────────────────
	text = {
		trigger = { scope:move = { has_variable = damage_source var:damage_source = flag:fire } }
		localization_key = pod_combat_damage_source_fire
	}
	text = {
		trigger = { scope:move = { has_variable = damage_source var:damage_source = flag:cold } }
		localization_key = pod_combat_damage_source_cold
	}
	text = {
		trigger = { scope:move = { has_variable = damage_source var:damage_source = flag:lightning } }
		localization_key = pod_combat_damage_source_lightning
	}
	text = {
		trigger = { scope:move = { has_variable = damage_source var:damage_source = flag:sun } }
		localization_key = pod_combat_damage_source_sun
	}
	text = {
		trigger = { scope:move = { has_variable = damage_source var:damage_source = flag:silver } }
		localization_key = pod_combat_damage_source_silver
	}
	text = {
		trigger = { scope:move = { has_variable = damage_source var:damage_source = flag:gold } }
		localization_key = pod_combat_damage_source_gold
	}
	text = {
		trigger = { scope:move = { has_variable = damage_source var:damage_source = flag:holy } }
		localization_key = pod_combat_damage_source_holy
	}
	text = {
		trigger = { scope:move = { has_variable = damage_source var:damage_source = flag:balefire } }
		localization_key = pod_combat_damage_source_balefire
	}

	# Blank fallback — covers utility moves and damage moves with
	# no damage_source set. Keeps the marker slot from emitting an
	# icon when there's nothing distinctive to show.
	text = {
		localization_key = pod_combat_damage_source_blank
	}
}

# ────────────────────────────────────────────────────────────────────
# Combat-log target column dispatcher.
#
# Renders the Target cell of a log row. Replaces a 3-text_single
# stack (target-name OR dash, plus an "AoE" suffix marker) with a
# single dynamic line that swaps based on the executing move's
# var:targeting:
#   * team_enemy → "AoE Enemies"
#   * team_ally  → "AoE Allies"
#   * self       → "Self"
#   * otherwise  → target's first name (if entry has a target var)
#   * fallback   → — placeholder
#
# Only one saved scope is needed: `entry` (the log entry story).
# chain into the move via `var:move = { var:targeting = ... }`.
# Each chain hop is guarded by `has_variable` so move-less log
# entries (system messages with no executing move) fall through
# safely to the target-name or dash branch.
#
# Called via:
#   [GuiScope.SetRoot(GetPlayer.MakeScope)
#       .AddScope('entry', Scope.Story.MakeScope)
#       .Custom('PODCombatLogTargetText')]

## ────────────────────────────────────────────────────────────────────
# System-message dispatcher.
#
# Move-less log rows pushed by pod_combat_push_system_log_effect
# carry a flag-valued entry.var:system_message (combat_started /
# combat_ended_victory / combat_ended_defeat). The GUI's move-less
# branch of the result column invokes this custom_loc to render the
# matching text instead of falling through to the "—" placeholder.
#
# Branch order doesn't carry semantic priority here (the flag values
# are mutually exclusive) — keep additions grouped by lifecycle
# phase: combat_started at top, combat_ended_* together below.
#
# Adding a new system message: emit a new flag from the call site,
# add a branch here, add the loc key in pod_combat_static_l_english.yml.

PODCombatLogSystemMessageText = {
	type = character

	text = {
		trigger = { scope:entry = { var:system_message = flag:combat_started } }
		localization_key = pod_combat_log_system_combat_started
	}
	text = {
		trigger = { scope:entry = { var:system_message = flag:combat_ended_victory } }
		localization_key = pod_combat_log_system_combat_ended_victory
	}
	text = {
		trigger = { scope:entry = { var:system_message = flag:combat_ended_defeat } }
		localization_key = pod_combat_log_system_combat_ended_defeat
	}
	text = {
		trigger = { scope:entry = { var:system_message = flag:round_begins } }
		localization_key = pod_combat_log_system_round_begins
	}
	text = {
		trigger = { scope:entry = { var:system_message = flag:wraith_summoned } }
		localization_key = pod_combat_log_system_wraith_summoned
	}
	# Unknown / future-flag fallback — surfaces as a muted dash so an
	# unhandled flag value fails closed visually instead of silently
	# rendering blank.
	text = {
		localization_key = pod_combat_log_system_unknown
	}
}

PODCombatLogTargetText = {
	type = character

	# Per-target AoE branches — these fire on the per-target rows
	# pushed inside each AoE iteration, where entry.var:target
	# carries the specific iterated target's character ref. The loc
	# key renders "All enemies (Steve)" / "All allies (Jane)" — the
	# AoE marker keeps the burst context visible, the per-target name
	# disambiguates which row hit which character.
	#
	# Branch order matters here: these per-target branches MUST come
	# BEFORE the AoE-only branches below. Customizable_loc evaluates
	# top-down and the first matching branch wins, so a per-target
	# row with entry.var:target set would otherwise fall into the
	# generic "All enemies" branch and lose the name.
	text = {
		trigger = {
			scope:entry = {
				has_variable = move
				var:move = { has_variable = targeting var:targeting = flag:team_enemy }
				has_variable = target
			}
		}
		localization_key = pod_combat_log_target_aoe_enemies_per_target
	}
	text = {
		trigger = {
			scope:entry = {
				has_variable = move
				var:move = { has_variable = targeting var:targeting = flag:team_ally }
				has_variable = target
			}
		}
		localization_key = pod_combat_log_target_aoe_allies_per_target
	}

	# AoE-only fallback branches — fire on the §16 wrap-up row (when
	# AoE pushed actor-side state like a self-heal that the per-target
	# rows didn't cover) where entry.var:target is unset. Render the
	# generic "All enemies" / "All allies" marker without a name.
	text = {
		trigger = {
			scope:entry = {
				has_variable = move
				var:move = { has_variable = targeting var:targeting = flag:team_enemy }
			}
		}
		localization_key = pod_combat_log_target_aoe_enemies
	}
	text = {
		trigger = {
			scope:entry = {
				has_variable = move
				var:move = { has_variable = targeting var:targeting = flag:team_ally }
			}
		}
		localization_key = pod_combat_log_target_aoe_allies
	}
	text = {
		trigger = {
			scope:entry = {
				has_variable = move
				var:move = { has_variable = targeting var:targeting = flag:self }
			}
		}
		localization_key = pod_combat_log_target_self
	}

	# Single-target — render the target's first name.
	text = {
		trigger = { scope:entry = { has_variable = target } }
		localization_key = pod_combat_log_target_single
	}

	# Fallback — system entries with no captured target.
	text = {
		localization_key = pod_combat_log_target_none
	}
}

# Compact-card basic-effect line.
#
# Renders a short one-line summary of a move's PRIMARY effect for the
# skills window's compact pool list — the middle column of each row
# between the move name (left) and the AP cost / equipped checkbox
# (right).
#
# Scope of "basic effect": exactly the dominant action the move
# performs. Damage / heal / morale damage / morale heal each get a
# one-line summary. Stance moves (set_guard / dodge / stealth)
# resolve to a single "Enter X" line. Tier buffs / debuffs surface
# as "Raise Soak" / "Lower Hit". AP gain shows "+1 AP".
#
# Explicitly NOT included (the user asked these to stay out of the
# compact summary): bypass_dodge / bypass_stealth / bypass_guard
# (auto-hit / ignore-dodge / ignore-guard capability tags),
# requires_hit_roll flag, retaliate_if_hit / retaliate_if_miss
# (counter-attack riders), drain riders, damage source / type
# qualifiers, and any tier-magnitude detail. The full hover
# tooltip still surfaces every one of those — this is a glanceable
# summary, not a complete spec.
#
# Branch order matters: HP damage takes priority over heal (a
# combined move shows damage as primary), which takes priority over
# morale / stance / buff branches. Only the FIRST matching branch
# fires (customizable_loc semantics), so a move with hp_damage AND
# set_guard renders the damage summary, not the stance one.
#
# Called via:
#   [GuiScope.SetRoot(GetPlayer.MakeScope)
#       .AddScope('story', Scope.Story.MakeScope)
#       .Custom('PODCombatMoveBasicEffectInline')]
PODCombatMoveBasicEffectInline = {
	type = character

	# ── DAMAGE branches ──
	text = {
		trigger = { scope:story = { has_variable = hp_damage } }
		localization_key = pod_combat_basic_effect_hp_damage
	}

	# ── HEAL branches ──
	text = {
		trigger = { scope:story = { has_variable = hp_heal } }
		localization_key = pod_combat_basic_effect_hp_heal
	}

	# ── MORALE branches ──
	text = {
		trigger = { scope:story = { has_variable = morale_damage } }
		localization_key = pod_combat_basic_effect_morale_damage
	}
	text = {
		trigger = { scope:story = { has_variable = morale_heal } }
		localization_key = pod_combat_basic_effect_morale_heal
	}

	# ── STANCE / FLAG branches ──
	text = {
		trigger = { scope:story = { has_variable = set_guard } }
		localization_key = pod_combat_basic_effect_set_guard
	}
	text = {
		trigger = { scope:story = { has_variable = dodge } }
		localization_key = pod_combat_basic_effect_dodge
	}
	text = {
		trigger = { scope:story = { has_variable = stealth } }
		localization_key = pod_combat_basic_effect_stealth
	}
	text = {
		trigger = { scope:story = { has_variable = remove_guard } }
		localization_key = pod_combat_basic_effect_remove_guard
	}

	# ── AP branches ──
	text = {
		trigger = { scope:story = { has_variable = ap_gain } }
		localization_key = pod_combat_basic_effect_ap_gain
	}
	text = {
		trigger = { scope:story = { has_variable = ap_drain } }
		localization_key = pod_combat_basic_effect_ap_drain
	}

	# ── TIER BUFF / DEBUFF branches ──
	text = {
		trigger = {
			scope:story = {
				OR = {
					has_variable = hit_buff_round_tier
					has_variable = hit_buff_combat_tier
				}
			}
		}
		localization_key = pod_combat_basic_effect_hit_buff
	}
	text = {
		trigger = {
			scope:story = {
				OR = {
					has_variable = hit_debuff_round_tier
					has_variable = hit_debuff_combat_tier
				}
			}
		}
		localization_key = pod_combat_basic_effect_hit_debuff
	}
	text = {
		trigger = {
			scope:story = {
				OR = {
					has_variable = soak_buff_round_tier
					has_variable = soak_buff_combat_tier
				}
			}
		}
		localization_key = pod_combat_basic_effect_soak_buff
	}
	text = {
		trigger = {
			scope:story = {
				OR = {
					has_variable = soak_debuff_round_tier
					has_variable = soak_debuff_combat_tier
				}
			}
		}
		localization_key = pod_combat_basic_effect_soak_debuff
	}

	# ── INIT DEBUFF ──
	text = {
		trigger = { scope:story = { has_variable = initiative_debuff } }
		localization_key = pod_combat_basic_effect_init_debuff
	}

	# ── Blank fallback ── covers pure-utility moves with none of
	# the above (rare — there's almost always at least one of these
	# vars set if the move does anything mechanical).
	text = {
		localization_key = pod_combat_basic_effect_blank
	}
}

# Each entry returns either its component substring (when the
# move's matching cost variable is set on the story_cycle) or the
# vanilla blank_line key (resolves to ""). The dispatcher loc
# `pod_combat_desc_costs` concatenates all five so a move with

# 00_POD_journey_custom_loc.txt — the dispatcher calls
# `[story.Custom('PODCombatXxxCostInline')]` and the saved-scope
# story_cycle resolves to the customizable_localization's input
# scope, accessed inside the localization_key as ROOT.
# ---------------------------------------------------------------

# Per-slot Hit display dispatcher. Called from the combat window's
# slot card stat line AND the slot stats tooltip's Hit headline.
# Surfaces the SIGNED net tier modifier (buffs − debuffs) for EVERY
# slot — PC and NPC alike. The absolute slot_hit_chance percentage is
# now reserved for the combat window's top-row "Target / Hit"
# readout (PODCombatPCTargetDisplay below) which carries it as
# "PC vs the currently selected target" context. Per-slot cards stay
# focused on "this combatant's buff/debuff state at a glance" — a
# consistent semantic across all 10 slots — instead of mixing two
# meanings ("hit chance against current target" on slot 0, "stale
# rotating value" on slots 1-9).
#
# Branches (evaluated top-down):
#   1. net > 0   → "#P +N%#!" green
#   2. net < 0   → "#N N%#!"  red (sign embedded in SV output)
#   3. net = 0   → "—" muted dash (no active modifier)
# ---------------------------------------------------------------
PODCombatSlotHitDisplay = {
	type = story
	text = {
		trigger = { pod_combat_slot_hit_net_mod_value > 0 }
		localization_key = pod_combat_slot_hit_npc_buff
	}
	text = {
		trigger = { pod_combat_slot_hit_net_mod_value < 0 }
		localization_key = pod_combat_slot_hit_npc_debuff
	}
	text = {
		localization_key = pod_combat_slot_hit_npc_neutral
	}
}
