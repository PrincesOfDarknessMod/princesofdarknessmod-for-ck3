# Customizable_localization dispatchers for the combat-log columns.
# Invoked from the datamodel log section in pod_combat_window.gui with
# saved scopes `entry` (log-entry story: outcome/attack_result/retal axis
# vars + amounts) and `move` (executing move story: damage class /
# targeting / buff-tier).
# Three-axis result model + HP/heal _aggravated split: COMBAT_REFERENCE.info §4.

PODCombatLogResultText = {
	type = character

	# Defensive short-circuit: raw_text is evaluated even on system rows
	# (GUI `visible` gates display, not evaluation), so guard the scope:move
	# context switches below from erroring when no move was captured.
	text = {
		trigger = { NOT = { exists = scope:move } }
		localization_key = pod_combat_log_result_text_unknown
	}

	# ── outcome=attack - inner attack_result dispatch ────────────
	# Amount branches gate on damage > 0 (a 0 means the pipeline
	# misfired) and fall through to the unknown line, not "0 HP".
	# hp_damage splits aggravated-first on the entry's actual-result
	# marker (result_agg, stamped by push_log from PoD's per-victim
	# classification) so the wound-box icon always matches the health
	# track - a holy hit shows aggravated on a vampire, superficial
	# on a werewolf.

	# hp_damage - attack landed wounds.
	# A second split on soaked_count: PoD rolls soak per point, so a packet can
	# land partially. The partial variants append the absorbed count, which is
	# why the number is lower than the move advertises. A FULL soak lands 0 and
	# arrives here as attack_result = soaked instead, so the sets stay disjoint.
	# Order is most-specific-first: agg+soak, agg, soak, plain.
	text = {
		trigger = {
			scope:entry = {
				var:outcome = flag:attack
				has_variable = attack_result
				var:attack_result = flag:hp_damage
				var:damage > 0
				has_variable = result_agg
				has_variable = soaked_count
				var:soaked_count > 0
			}
		}
		localization_key = pod_combat_log_result_text_hp_damage_aggravated_partial_soak
	}
	text = {
		trigger = {
			scope:entry = {
				var:outcome = flag:attack
				has_variable = attack_result
				var:attack_result = flag:hp_damage
				var:damage > 0
				has_variable = result_agg
			}
		}
		localization_key = pod_combat_log_result_text_hp_damage_aggravated
	}
	text = {
		trigger = {
			scope:entry = {
				var:outcome = flag:attack
				has_variable = attack_result
				var:attack_result = flag:hp_damage
				var:damage > 0
				has_variable = soaked_count
				var:soaked_count > 0
			}
		}
		localization_key = pod_combat_log_result_text_hp_damage_partial_soak
	}
	text = {
		trigger = {
			scope:entry = {
				var:outcome = flag:attack
				has_variable = attack_result
				var:attack_result = flag:hp_damage
				var:damage > 0
			}
		}
		localization_key = pod_combat_log_result_text_hp_damage
	}

	text = {
		trigger = {
			scope:entry = {
				var:outcome = flag:attack
				has_variable = attack_result
				var:attack_result = flag:guard_blocked
			}
		}
		localization_key = pod_combat_log_result_text_guard_blocked
	}
	text = {
		trigger = {
			scope:entry = {
				var:outcome = flag:attack
				has_variable = attack_result
				var:attack_result = flag:soaked
			}
		}
		localization_key = pod_combat_log_result_text_soaked
	}
	text = {
		trigger = {
			scope:entry = {
				var:outcome = flag:attack
				has_variable = attack_result
				var:attack_result = flag:immune
			}
		}
		localization_key = pod_combat_log_result_text_immune
	}
	text = {
		trigger = {
			scope:entry = {
				var:outcome = flag:attack
				has_variable = attack_result
				var:attack_result = flag:missed
			}
		}
		localization_key = pod_combat_log_result_text_missed
	}
	text = {
		trigger = {
			scope:entry = {
				var:outcome = flag:attack
				has_variable = attack_result
				var:attack_result = flag:dodged
			}
		}
		localization_key = pod_combat_log_result_text_dodged
	}
	text = {
		trigger = {
			scope:entry = {
				var:outcome = flag:attack
				has_variable = attack_result
				var:attack_result = flag:stealthed
			}
		}
		localization_key = pod_combat_log_result_text_stealthed
	}

	# ── outcome=heal - HP heal ───────────────────────────────────
	text = {
		trigger = {
			scope:entry = {
				var:outcome = flag:heal
				has_variable = heal_amount
				var:heal_amount > 0
			}
			scope:move = { has_variable = heal_type var:heal_type = flag:aggravated }
		}
		localization_key = pod_combat_log_result_text_hp_healed_aggravated
	}
	text = {
		trigger = {
			scope:entry = {
				var:outcome = flag:heal
				has_variable = heal_amount
				var:heal_amount > 0
			}
		}
		localization_key = pod_combat_log_result_text_hp_healed
	}

	# ── outcome=morale_damage / morale_healed ────────────────────
	text = {
		trigger = {
			scope:entry = {
				var:outcome = flag:morale_damage
				has_variable = morale_damage_amount
				var:morale_damage_amount > 0
			}
		}
		localization_key = pod_combat_log_result_text_morale_damage
	}
	text = {
		trigger = {
			scope:entry = {
				var:outcome = flag:morale_healed
				has_variable = morale_heal_amount
				var:morale_heal_amount > 0
			}
		}
		localization_key = pod_combat_log_result_text_morale_healed
	}

	# ── outcome=utility (and explicit utility fallback) ──────────
	text = {
		trigger = { scope:entry = { var:outcome = flag:utility } }
		localization_key = pod_combat_log_result_text_utility
	}

	# Fallback: catches 0-amount rows that fell through the amount>0
	# guards, plus unexpected combinations. Renders the dash placeholder
	# rather than "0 HP" / "+0 Morale".
	text = {
		localization_key = pod_combat_log_result_text_unknown
	}
}

# Result-line riders - one customizable_loc per optional bit of info
# that gets appended to the result column. Each rider returns
# either a styled snippet or the empty string (`pod_combat_log_blank`)
# so the row stays clean when the rider doesn't apply.
#
# Riders are invoked as their OWN text_singles in the result-column
# hbox in pod_combat_window.gui, NOT nested inside the main result
# text's loc value. Nested [GetPlayer.Custom('SubName')] calls in loc
# strings do not see the outer saved scopes (observed in testing: the
# sub-custom's scope:X reads fail with "Undefined event target X").
# So every customizable_loc that needs a saved scope must be invoked
# at the GUI level via its own GuiScope.SetRoot(...).AddScope(...)
# chain.
#
# Each rider expects the scopes it dispatches on to be saved by its
# own GUI invocation:
#   Heal / Morale-damage / Morale-heal / Retal riders - `entry`, `move`
#       (entry for amount + outcome-axis reads, move for heal_type /
#       damage_type splits; outcome-based primary-line suppression
#       reads entry.var:outcome)
#   Buff rider - `move`
#       (priority-order has_variable checks on the move's stance / tier
#       buff / utility flag vars; no entry needed)

# Heal rider: appended to damage / utility lines when a move ALSO heals
# HP (life-drain). Gated on entry.heal_amount > 0, so capped heals stay
# quiet. Suppressed on outcome=flag:heal (the pure-heal line already
# shows the amount). Fires on miss/soak rows too: the heal runs with no
# hit gate. Splits on heal_type for the aggravated-vs-superficial icon.
PODCombatLogHealRider = {
	type = character
	# Defensive short-circuit for system-message rows: the GUI evaluates
	# raw_text even on invisible widgets, so without this no-move first
	# branch (same guard as PODCombatLogResultText's first branch, top
	# of file) the scope:move context switches below would error on
	# rows where no move was captured.
	text = {
		trigger = { NOT = { exists = scope:move } }
		localization_key = pod_combat_log_blank
	}
	text = {
		trigger = {
			scope:entry = { has_variable = heal_amount var:heal_amount > 0 }
			NOT = { scope:entry = { var:outcome = flag:heal } }
			scope:move = { has_variable = heal_type var:heal_type = flag:aggravated }
		}
		localization_key = pod_combat_log_heal_rider_aggravated
	}
	text = {
		trigger = {
			scope:entry = { has_variable = heal_amount var:heal_amount > 0 }
			NOT = { scope:entry = { var:outcome = flag:heal } }
		}
		localization_key = pod_combat_log_heal_rider
	}
	text = {
		localization_key = pod_combat_log_blank
	}
}

# Morale-damage rider: appended when the move deals morale damage
# alongside HP damage. Gated on the entry's morale_damage_amount
# field. Suppressed on outcome=flag:morale_damage (the morale-only
# primary line already shows the amount).
PODCombatLogMoraleDamageRider = {

			scope:entry = { has_variable = morale_damage_amount var:morale_damage_amount > 0 }
			NOT = { scope:entry = { var:outcome = flag:morale_damage } }
		}
		localization_key = pod_combat_log_morale_damage_rider
	}
	text = {
		localization_key = pod_combat_log_blank
	}
}

# Morale-heal rider - appended when a damage / utility move also
# heals morale. Gated on the entry's morale_heal_amount field.
# Suppressed on outcome=flag:morale_healed (the morale-heal primary

PODCombatLogMoraleHealRider = {

	text = {
		trigger = {
			scope:entry = { has_variable = morale_heal_amount var:morale_heal_amount > 0 }
			NOT = { scope:entry = { var:outcome = flag:morale_healed } }
		}
		localization_key = pod_combat_log_morale_heal_rider
	}
	text = {
		localization_key = pod_combat_log_blank
	}
}

# Retal rider - appended whenever the row's retaliation overlay
# fired. Gated on entry.var:retal_fired (set by push_log from
# pod_combat_last_retal_fired) so utility / no-retal rows stay quiet.
# Two branches on the actual retal damage:
# - retal_damage > 0  -> "+ Counter -X HP" (retaliation landed)
# - retal_damage = 0  -> "+ Counter Soaked" (retaliation was absorbed)
# Independent of the main outcome dispatcher, every attack outcome
# (hp_damage / soaked / guard_blocked / missed / dodged / stealthed)
# can carry this overlay alongside the main result line.
PODCombatLogRetalRider = {
	type = character
	# Morale counter - eroded will, not flesh. retal_morale is captured by push_log
	# from pod_combat_last_retal_morale (set when the counter routed via deal_morale).
	# First branch so it wins over the HP-damage branches below.
	text = {
		trigger = {
			scope:entry = {
				has_variable = retal_fired
				has_variable = retal_damage
				var:retal_damage > 0
				has_variable = retal_morale
			}
		}
		localization_key = pod_combat_log_retal_rider_morale
	}
	# Aggravated retal landed - show the aggravated wound-box icon
	# instead of the default superficial one. retal_agg is captured
	# by push_log from pod_combat_last_retal_agg, which
	# pod_combat_apply_retaliation_effect stamps from PoD's ACTUAL
	# per-victim classification of the typed counter (route's
	# scope:result_agg) - a holy counter reads aggravated only when
	# it really landed aggravated on this victim.
	text = {
		trigger = {
			scope:entry = {
				has_variable = retal_fired
				has_variable = retal_damage
				var:retal_damage > 0
				has_variable = retal_agg
			}
		}
		localization_key = pod_combat_log_retal_rider_aggravated
	}
	# Superficial retal landed - default wound-box icon.
	text = {
		trigger = {
			scope:entry = {
				has_variable = retal_fired
				has_variable = retal_damage
				var:retal_damage > 0
			}
		}
		localization_key = pod_combat_log_retal_rider
	}
	# Retal fired but landed 0 damage (soaked / absorbed). No
	# aggravated/superficial distinction matters - both look like
	# "soaked" to the player.
	text = {
		trigger = {
			scope:entry = { has_variable = retal_fired }
		}
		localization_key = pod_combat_log_retal_rider_soaked
	}
	text = {
		localization_key = pod_combat_log_blank
	}
}

# ────────────────────────────────────────────────────────────────────
# Combat-log target column dispatcher. Renders the Target cell, swapping
# on the move's var:targeting (team_enemy/team_ally/team_all -> AoE label; self /
# no-target collapse to empty, else target's first name). Saved scope `entry`; chains
# into the move via var:move. has_variable guards each hop so move-less
# system rows fall through safely.
#   [GuiScope.SetRoot(GetPlayer.MakeScope)
#       .AddScope('entry', Scope.Story.MakeScope)
#       .Custom('PODCombatLogTargetText')]
PODCombatLogTargetText = {
	type = character

	# Per-target AoE branches - these fire on the per-target rows
	# pushed inside each AoE iteration, where entry.var:target
	# carries the specific iterated target's character ref. The loc
	# key renders "All enemies (Steve)" / "All allies (Jane)" - the
	# AoE marker keeps the burst context visible, the per-target name
	# disambiguates which row hit which character.
	#
	# Branch order matters here: these per-target branches MUST come
	# BEFORE the AoE-only branches below. Customizable_loc evaluates
	# top-down and the first matching branch wins, so a per-target
	# row with entry.var:target set would otherwise fall into the
	# generic "All enemies" branch and lose the name.
	text = {
		trigger = {
			scope:entry = {
				has_variable = move
				var:move = { has_variable = targeting var:targeting = flag:team_enemy }
				has_variable = target
			}
		}
		localization_key = pod_combat_log_target_aoe_enemies_per_target
	}
	text = {
		trigger = {
			scope:entry = {
				has_variable = move
				var:move = { has_variable = targeting var:targeting = flag:team_ally }
				has_variable = target
			}
		}
		localization_key = pod_combat_log_target_aoe_allies_per_target
	}
	text = {
		trigger = {
			scope:entry = {
				has_variable = move
				var:move = { has_variable = targeting var:targeting = flag:team_all }
				has_variable = target
			}
		}
		localization_key = pod_combat_log_target_aoe_all_per_target
	}

	# AoE-only fallback branches - fire on the §16 wrap-up row (when
	# AoE pushed actor-side state like a self-heal that the per-target
	# rows didn't cover) where entry.var:target is unset. Render the
	# generic "All enemies" / "All allies" marker without a name.
	text = {
		trigger = {
			scope:entry = {
				has_variable = move
				var:move = { has_variable = targeting var:targeting = flag:team_enemy }
			}
		}
		localization_key = pod_combat_log_target_aoe_enemies
	}
	text = {
		trigger = {
			scope:entry = {
				has_variable = move
				var:move = { has_variable = targeting var:targeting = flag:team_ally }
			}
		}
		localization_key = pod_combat_log_target_aoe_allies
	}
	text = {
		trigger = {
			scope:entry = {
				has_variable = move
				var:move = { has_variable = targeting var:targeting = flag:team_all }
			}
		}
		localization_key = pod_combat_log_target_aoe_all
	}
	text = {
		trigger = {
			scope:entry = {
				has_variable = move
				var:move = { has_variable = targeting var:targeting = flag:self }
			}
		}
		localization_key = pod_combat_log_target_self
	}

	# Single-target - render the target's first name.
	text = {
		trigger = { scope:entry = { has_variable = target } }
		localization_key = pod_combat_log_target_single
	}

	# Fallback - system entries with no captured target.
	text = {
		localization_key = pod_combat_log_target_none
	}
}

# ---------------------------------------------------------------
# Per-cost line components for the combined Costs: dispatcher.
# Each entry returns either its component substring (when the
# move's matching cost variable is set on the story_cycle) or the
# vanilla blank_line key (resolves to ""). The dispatcher loc
# `pod_combat_desc_costs` concatenates all five so a move with
# two costs renders "Costs: 2 vitae 5 gold" on a single bullet.
#
# Why customizable_localization and not inline SelectLocalization:
# CK3's data-binding parser treats `#` as a comment marker and
# rejects nested `[X]` substitutions when they appear inside a
# SelectLocalization function-argument string. Wrapping each
# branch in its own loc-file entry sidesteps the limit because
# the `#bold`/`[Var]` tokens live in a normal loc string, not as
# a function argument.
#
# `type = story` matches PoD's PODShowJourneyName pattern in
# 00_POD_journey_custom_loc.txt - the dispatcher calls
# `[story.Custom('PODCombatXxxCostInline')]` and the saved-scope
# story_cycle resolves to the customizable_localization's input
# scope, accessed inside the localization_key as ROOT.
# ---------------------------------------------------------------
PODCombatEnergyCostInline = {
	type = story
	text = {
		trigger = { has_variable = energy_cost }
		localization_key = pod_combat_costline_energy
	}
	text = {
		localization_key = blank_line
	}
}

# Per-slot Hit display dispatcher. Called from the combat window's

# plus its hover breakdown, in POD_combat_static_l_english.yml), where

# Branches (evaluated top-down):
#   1. net > 0   -> "#P +N%#!" green
#   2. net < 0   -> "#N N%#!"  red (sign embedded in SV output)
#   3. net = 0   -> " - " muted dash (no active modifier)

PODCombatSlotHitDisplay = {

		localization_key = pod_combat_slot_hit_npc_buff
	}
	text = {
		trigger = { pod_combat_slot_hit_net_mod_value < 0 }
		localization_key = pod_combat_slot_hit_npc_debuff
	}

		localization_key = pod_combat_slot_hit_npc_neutral

# ---------------------------------------------------------------
# Per-slot focus-target readout. Names whoever this combatant went
# after on its last turn, from the AI's persistent cross-turn focus
# mark (pod_combat_ai_focus_char on the combatant). Invoked with the
# slot story as root, like PODCombatSlotHitDisplay:
#   [GuiScope.SetRoot(Scope.Story.MakeScope).Custom('PODCombatSlotFocusTarget')]
# Defensive first branch: the readout's raw_text evaluates on every
# slot card even when the row is hidden, so a slot with no mark (the
# PC, an AI before its first turn, a mind-controlled AI whose mark was
# cleared) resolves to blank instead of reading an unset variable.
# ---------------------------------------------------------------
PODCombatSlotFocusTarget = {
	type = story
	text = {
		trigger = {
			OR = {
				NOT = { exists = var:slot_char }
				var:slot_char = { NOT = { has_variable = pod_combat_ai_focus_char } }
			}
		}
		localization_key = pod_combat_slot_focus_none
	}
	text = {
		localization_key = pod_combat_slot_focus_target
	}
}
