# Customizable_localization for combat log entries.
#
# Used by the datamodel-driven log section in pod_combat_window.gui:
# the GUI invokes Custom('PODCombatLogResultText') with three saved
# scopes —
#   `entry` (the log entry story scope, for var:damage / var:heal_amount
#            / var:morale_damage_amount / var:morale_heal_amount reads)
#   `move`  (the executing move story, for damage_type / damage_source /
#            targeting / buff-tier checks the AoE intercept and rider
#            customizable_locs need)
#   `rcode` (a value-scope wrapping result_code, for the dispatch
#            triggers below)
# Each trigger branch checks scope:rcode against an integer; matching
# branch resolves to the corresponding loc key, which composes its
# rendered text from the entry / move scopes plus the various rider
# customizable_locs invoked separately as siblings in the GUI hbox.
#
# Result code map (must match pod_combat_push_log_effect):
#   0  — HP damage (-X [wound-box])
#   1  — guard-blocked
#   2  — soaked
#   3  — missed
#   4  — countered (currently no flag-source — placeholder rcode)
#   5  — dodged
#   6  — HP damage + retaliation triggered (-X [box] retaliated)
#   7  — HP damage + retaliation soaked (-X [box] counter soaked)
#   8  — utility move applied (no damage)
#   9  — morale damage only (-X Morale)
#  10  — HP healed (+X [box])
#  11  — morale healed (+X Morale)
#  12  — stealth-triggered auto-miss (distinct from rcode 5 dodged
#         so the loc line links to the Stealth concept, not Dodge)
#
# rcodes 0/6/7/10 split into _aggravated variants in the dispatch
# below — the wound-box icon swaps from neutral (superficial) to red
# (aggravated) based on scope:move.var:damage_type / heal_type.

PODCombatLogResultText = {
	type = character

	# ── AoE intercept ────────────────────────────────────────────
	# For team_enemy / team_ally moves whose primary target ended
	# in a per-target fail state (blocked / soaked / missed /
	# dodged), skip the per-target outcome label and surface the
	# move's intent damage. Per-target outcomes are
	# misleading on AoE rows because:
	#   * the move likely landed on OTHER members of the side
	#   * pod_combat_last_damage gets overwritten per iteration,
	#     so entry.var:damage is just whatever the LAST target
	#     resolved to (often 0 if that target soaked/dodged) —
	#     not a useful number to display on an aggregate row.
	#
	# Read move.var:hp_damage instead. That's the design value
	# of the strike, written once at move-init time, so the row
	# shows what the AoE was meant to do regardless of how the
	# per-target rolls shook out. Two damage-typed branches +
	# one no-hp-damage fallback (pure morale / utility AoE).
	#
	# rcode 4 (countered) and 7 (counter soaked) intentionally
	# aren't intercepted — those signal a counter-attack that
	# fired against the actor regardless of AoE shape, so the
	# per-target label is still meaningful.
	text = {
		trigger = {
			scope:move = {
				OR = {
					var:targeting = flag:team_enemy
					var:targeting = flag:team_ally
				}
				has_variable = hp_damage
				var:hp_damage > 0
				has_variable = damage_type
				var:damage_type = flag:aggravated
			}
			OR = {
				scope:rcode = 1
				scope:rcode = 2
				scope:rcode = 3
				scope:rcode = 5
			}
		}
		localization_key = pod_combat_log_result_text_aoe_aggravated
	}
	text = {
		trigger = {
			scope:move = {
				OR = {
					var:targeting = flag:team_enemy
					var:targeting = flag:team_ally
				}
				has_variable = hp_damage
				var:hp_damage > 0
			}
			OR = {
				scope:rcode = 1
				scope:rcode = 2
				scope:rcode = 3
				scope:rcode = 5
			}
		}
		localization_key = pod_combat_log_result_text_aoe
	}
	text = {
		trigger = {
			scope:move = {
				OR = {
					var:targeting = flag:team_enemy
					var:targeting = flag:team_ally
				}
			}
			OR = {
				scope:rcode = 1
				scope:rcode = 2
				scope:rcode = 3
				scope:rcode = 5
			}
		}
		localization_key = pod_combat_log_result_text_aoe_zero
	}

	# ── Standard rcode dispatch ─────────────────────────────────
	# Each amount-bearing rcode (0/6/7/9/10/11) checks damage > 0
	# before showing the amount-styled loc; the 0-damage variant
	# falls through to the "—" unknown line so  don't render
	# rows like "+0 HP" or "0 Morale" that mean "the heal/damage
	# pipeline didn't fire properly."
	#
	# HP-track rcodes (0 / 6 / 7 / 10) split on damage_type or
	# heal_type so the result line ends in the matching wound-box
	# icon (aggravated red box vs superficial neutral box) instead
	# of the literal word "HP". Aggravated is checked first; bare
	# superficial is the fallback. Morale rcodes (9 / 11) keep the
	# "Morale" word since morale lives off the HP track.

	# rcode 0 — HP damage
	text = {
		trigger = {
			scope:rcode = 0
			scope:entry = { var:damage > 0 }
			scope:move = { has_variable = damage_type var:damage_type = flag:aggravated }
		}
		localization_key = pod_combat_log_result_text_0_aggravated
	}
	text = {
		trigger = {
			scope:rcode = 0
			scope:entry = { var:damage > 0 }
		}
		localization_key = pod_combat_log_result_text_0
	}

	text = {
		trigger = { scope:rcode = 1 }
		localization_key = pod_combat_log_result_text_1
	}
	text = {
		trigger = { scope:rcode = 2 }
		localization_key = pod_combat_log_result_text_2
	}
	text = {
		trigger = { scope:rcode = 3 }
		localization_key = pod_combat_log_result_text_3
	}
	text = {
		trigger = { scope:rcode = 4 }
		localization_key = pod_combat_log_result_text_4
	}
	text = {
		trigger = { scope:rcode = 5 }
		localization_key = pod_combat_log_result_text_5
	}
	text = {
		trigger = { scope:rcode = 12 }
		localization_key = pod_combat_log_result_text_12
	}

	# rcode 6 — HP damage + retaliation triggered
	text = {
		trigger = {
			scope:rcode = 6
			scope:entry = { var:damage > 0 }
			scope:move = { has_variable = damage_type var:damage_type = flag:aggravated }
		}
		localization_key = pod_combat_log_result_text_6_aggravated
	}
	text = {
		trigger = {
			scope:rcode = 6
			scope:entry = { var:damage > 0 }
		}
		localization_key = pod_combat_log_result_text_6
	}

	# rcode 7 — HP damage + retaliation soaked
	text = {
		trigger = {
			scope:rcode = 7
			scope:move = { has_variable = damage_type var:damage_type = flag:aggravated }
		}
		localization_key = pod_combat_log_result_text_7_aggravated
	}
	text = {
		trigger = { scope:rcode = 7 }
		localization_key = pod_combat_log_result_text_7
	}

	text = {
		trigger = { scope:rcode = 8 }
		localization_key = pod_combat_log_result_text_8
	}
	text = {
		trigger = {
			scope:rcode = 9
			scope:entry = { has_variable = morale_damage_amount var:morale_damage_amount > 0 }
		}
		localization_key = pod_combat_log_result_text_9
	}

	# rcode 10 — pure HP heal
	text = {
		trigger = {
			scope:rcode = 10
			scope:entry = { has_variable = heal_amount var:heal_amount > 0 }
			scope:move = { has_variable = heal_type var:heal_type = flag:aggravated }
		}
		localization_key = pod_combat_log_result_text_10_aggravated
	}
	text = {
		trigger = {
			scope:rcode = 10
			scope:entry = { has_variable = heal_amount var:heal_amount > 0 }
		}
		localization_key = pod_combat_log_result_text_10
	}

	text = {
		trigger = {
			scope:rcode = 11
			scope:entry = { has_variable = morale_heal_amount var:morale_heal_amount > 0 }
		}
		localization_key = pod_combat_log_result_text_11
	}
	# Fallback — also catches all 0-amount rows that fall through
	# the "damage > 0" guards above (rcode 0/6/9/10/11 with no
	# captured amount). Renders the em-dash placeholder rather
	# than "0 HP" / "+0 HP" / "+0 Morale".
	text = {
		localization_key = pod_combat_log_result_text_unknown
	}
}

# ────────────────────────────────────────────────────────────────────
# Result-line riders — one customizable_loc per optional bit of info
# that gets appended to the result column. Each rider returns
# either a styled snippet or the empty string (`pod_combat_log_blank`)
# so the row stays clean when the rider doesn't apply.
#
# Riders are invoked as their OWN text_singles in the result-column
# hbox in pod_combat_window.gui, NOT nested inside the main result
# text's loc value. CK3 does seemt to not propagate saved scopes
# through nested [GetPlayer.Custom('SubName')] calls in loc strings —
# the sub-custom's scope:X reads fail with "Undefined event target X".
# So every customizable_loc that needs a saved scope must be invoked
# at the GUI level via its own GuiScope.SetRoot(...).AddScope(...)
# chain.
#
# Each rider expects the scopes it dispatches on to be saved by its
# own GUI invocation:
#   Heal / Morale-damage / Morale-heal riders — `entry`, `move`, `rcode`
#       (entry for amount-var reads, move for heal_type / damage_type
#       splits, rcode for primary-line suppression)
#   Buff rider                                — `move`
#       (priority-order has_variable checks on the move's stance / tier
#       buff / utility flag vars; no entry or rcode needed)

# Heal rider — appended to damage / utility lines when the move ALSO
# restores HP (life-drain pattern, "deal 5 damage and heal 3"). Gated
# on the entry's heal_amount field (set by push_log from
# pod_combat_last_heal_amount), so the rider only fires if a heal
# actually applied — capped heals (target near full HP, heal_amount=0)
# stay quiet.
#
# Suppressed only on rcode = 10 (pure-heal primary line, where the
# amount is already shown). HP heals fire regardless of attack hit
# status (the heal section in POD_execute_combat_move runs on
# has_variable = hp_heal alone, no hit gate), so Soaked! / Missed! /
# Dodged! rows still get the rider — those represent moves that
# attempted damage AND healed the actor on use.
#
# Splits on heal_type so the rider's wound-box icon matches what the
# heal actually restored (aggravated heals close agg wounds; default
# is superficial).
PODCombatLogHealRider = {
	type = character
	text = {
		trigger = {
			scope:entry = { has_variable = heal_amount var:heal_amount > 0 }
			NOT = { scope:rcode = 10 }
			scope:move = { has_variable = heal_type var:heal_type = flag:aggravated }
		}
		localization_key = pod_combat_log_heal_rider_aggravated
	}
	text = {
		trigger = {
			scope:entry = { has_variable = heal_amount var:heal_amount > 0 }
			NOT = { scope:rcode = 10 }
		}
		localization_key = pod_combat_log_heal_rider
	}
	text = {
		localization_key = pod_combat_log_blank
	}
}

# Morale-damage rider — appended when the move deals morale damage
# alongside HP damage. Gated on the entry's morale_damage_amount

PODCombatLogMoraleDamageRider = {

		trigger = {

		}

}

# Buff / debuff / status rider — emits a single label naming the most
# prominent stance / tier-buff / utility flag the move applied.
# Branch order is the priority — first match wins, so e.g. a move
# that applies BOTH +Soak and +Hit reports +Soak. Stance flags rank
# above tier buffs (their semantic impact is bigger), tier
# buffs/debuffs sit in the middle, and the utility-modifier flags
# (AP gain/drain, retaliate) sit at the bottom.
PODCombatLogBuffRider = {
	type = character

	# ── Stance flags ──
	text = {
		trigger = { scope:move = { has_variable = set_guard } }
		localization_key = pod_combat_log_buff_set_guard
	}
	text = {
		trigger = { scope:move = { has_variable = dodge } }
		localization_key = pod_combat_log_buff_dodge
	}
	text = {
		trigger = { scope:move = { has_variable = stealth } }
		localization_key = pod_combat_log_buff_stealth
	}
	text = {
		trigger = { scope:move = { has_variable = remove_guard } }
		localization_key = pod_combat_log_buff_remove_guard
	}

	# ── Tier buffs / debuffs ──
	text = {
		trigger = {
			scope:move = {
				OR = {
					has_variable = soak_buff_round_tier
					has_variable = soak_buff_combat_tier
				}
			}
		}
		localization_key = pod_combat_log_buff_soak_buff
	}
	text = {
		trigger = {
			scope:move = {
				OR = {
					has_variable = soak_debuff_round_tier
					has_variable = soak_debuff_combat_tier
				}
			}
		}
		localization_key = pod_combat_log_buff_soak_debuff
	}
	text = {
		trigger = {
			scope:move = {
				OR = {
					has_variable = hit_buff_round_tier
					has_variable = hit_buff_combat_tier
				}
			}
		}
		localization_key = pod_combat_log_buff_hit_buff
	}
	text = {
		trigger = {
			scope:move = {
				OR = {
					has_variable = hit_debuff_round_tier
					has_variable = hit_debuff_combat_tier
				}
			}
		}
		localization_key = pod_combat_log_buff_hit_debuff
	}

	# ── Utility modifiers ──
	text = {
		trigger = { scope:move = { has_variable = ap_gain } }
		localization_key = pod_combat_log_buff_ap_gain
	}
	text = {
		trigger = { scope:move = { has_variable = ap_drain } }
		localization_key = pod_combat_log_buff_ap_drain
	}
	text = {
		trigger = {
			scope:move = {
				OR = {
					has_variable = retaliate_if_hit
					has_variable = retaliate_if_missed
				}
			}
		}
		localization_key = pod_combat_log_buff_retaliate
	}
	text = {
		trigger = { scope:move = { has_variable = initiative_debuff } }
		localization_key = pod_combat_log_buff_initiative_debuff
	}

	# ── Blank fallback ──
	text = {
		localization_key = pod_combat_log_blank
	}
}

# ────────────────────────────────────────────────────────────────────
# Damage-source marker.
#
# Renders a small icon next to the result column in the combat log,
# and at the head of the move-tooltip damage line, picking the glyph
# from the move story's var:damage_source (fire / cold / lightning /
# sun / silver / gold / holy / balefire). Moves without a damage_source
# fall through to the blank loc and emit nothing — physical /
# supernatural / mental pipelines all telegraph their flavor through
# the move name itself, and a generic "this is an attack" icon next to
# every damage row would just add noise without telling the player
# anything new. (Mental had its own icon at one point; it was dropped
# alongside the bulk-Mental CSV migration since every morale-damage row
# in the log gained a redundant Mental marker.)
#
# Customizable_loc types are limited to character / activity /
# landed_title / artifact / province (see vanilla CK3 examples). So
# rather than typing the loc against story_cycle directly, use the
# same pattern as PODCombatLogResultText above: type = character (root
# is the player), pass the move as a saved scope `move` from the GUI,
# and dispatch on `scope:move = { var:X = flag:Y }`.
#
# Called via:
#   [GuiScope.SetRoot(GetPlayer.MakeScope)
#       .AddScope('move', Scope.Var('move').Story.MakeScope)
#       .Custom('PODCombatDamageSourceMarker')]
# (Scope.Var('move') in the combat-log datamodel returns the stored
# story reference — .Story.MakeScope unwraps it to a real scope.)
#
# The blank-fallback branch at the end keeps the surrounding text-line
# from collapsing when no damage_source matches.

PODCombatDamageSourceMarker = {
	type = character

	# Each trigger guards `var:X = flag:Y` reads behind a
	# `has_variable` check so moves without the field don't spam
	# "var returned unset scope" errors during loc evaluation.

	# ── damage_source ────────────────────────────
	text = {
		trigger = { scope:move = { has_variable = damage_source var:damage_source = flag:fire } }
		localization_key = pod_combat_damage_source_fire
	}
	text = {
		trigger = { scope:move = { has_variable = damage_source var:damage_source = flag:cold } }
		localization_key = pod_combat_damage_source_cold
	}
	text = {
		trigger = { scope:move = { has_variable = damage_source var:damage_source = flag:lightning } }
		localization_key = pod_combat_damage_source_lightning
	}
	text = {
		trigger = { scope:move = { has_variable = damage_source var:damage_source = flag:sun } }
		localization_key = pod_combat_damage_source_sun
	}
	text = {
		trigger = { scope:move = { has_variable = damage_source var:damage_source = flag:silver } }
		localization_key = pod_combat_damage_source_silver
	}
	text = {
		trigger = { scope:move = { has_variable = damage_source var:damage_source = flag:gold } }
		localization_key = pod_combat_damage_source_gold
	}
	text = {
		trigger = { scope:move = { has_variable = damage_source var:damage_source = flag:holy } }
		localization_key = pod_combat_damage_source_holy
	}
	text = {
		trigger = { scope:move = { has_variable = damage_source var:damage_source = flag:balefire } }
		localization_key = pod_combat_damage_source_balefire
	}

	# Blank fallback — covers utility moves and damage moves with
	# no damage_source set. Keeps the marker slot from emitting an
	# icon when there's nothing distinctive to show.
	text = {
		localization_key = pod_combat_damage_source_blank
	}
}

# ────────────────────────────────────────────────────────────────────
# Combat-log target column dispatcher.
#
# Renders the Target cell of a log row. Replaces a 3-text_single
# stack (target-name OR dash, plus an "AoE" suffix marker) with a
# single dynamic line that swaps based on the executing move's
# var:targeting:
#   * team_enemy → "AoE Enemies"
#   * team_ally  → "AoE Allies"
#   * self       → "Self"
#   * otherwise  → target's first name (if entry has a target var)
#   * fallback   → — placeholder
#
# Only one saved scope is needed: `entry` (the log entry story).
# chain into the move via `var:move = { var:targeting = ... }`.
# Each chain hop is guarded by `has_variable` so move-less log
# entries (system messages with no executing move) fall through
# safely to the target-name or dash branch.
#
# Called via:
#   [GuiScope.SetRoot(GetPlayer.MakeScope)
#       .AddScope('entry', Scope.Story.MakeScope)
#       .Custom('PODCombatLogTargetText')]

PODCombatLogTargetText = {
	type = character

	# AoE branches first — they override target rendering even if
	# the entry has a single-target captured (AoE moves still mark a
	# primary target during execution; don't want the column to
	# show a single name when the move actually hit the whole side).
	text = {
		trigger = {
			scope:entry = {
				has_variable = move
				var:move = { has_variable = targeting var:targeting = flag:team_enemy }
			}
		}
		localization_key = pod_combat_log_target_aoe_enemies
	}
	text = {
		trigger = {
			scope:entry = {
				has_variable = move
				var:move = { has_variable = targeting var:targeting = flag:team_ally }
			}
		}
		localization_key = pod_combat_log_target_aoe_allies
	}
	text = {
		trigger = {
			scope:entry = {
				has_variable = move
				var:move = { has_variable = targeting var:targeting = flag:self }
			}
		}
		localization_key = pod_combat_log_target_self
	}

	# Single-target — render the target's first name.
	text = {
		trigger = { scope:entry = { has_variable = target } }
		localization_key = pod_combat_log_target_single
	}

	# Fallback — system entries with no captured target.
	text = {
		localization_key = pod_combat_log_target_none
	}
}
