PODMasqueradeHint = {
	type = character

	text = {
		trigger = {
			has_character_modifier = masquerade0_modifier
		}
		localization_key = POD_hint_masquerade_maximum
	}
	text = {
		trigger = {
			is_character_interaction_valid = {
				recipient = this
				interaction = POD_cover_masquerade_interaction
			}
		}
		localization_key = POD_hint_masquerade_valid
	}
	text = {
		trigger = {
			AND = {
				faith = { has_doctrine_parameter = vampirethemasquerade_none }
				masquerade_exposure_level < 5
				NOT = { has_trait = paranoid }
			}
		}
		localization_key = POD_hint_masquerade_invalid_no_masquerade
	}
	text = {
		trigger = { masquerade_exposure_level <= minimum_masquerade_exposure }
		localization_key = POD_hint_masquerade_invalid_cap
	}
	text = {
		localization_key = POD_hint_masquerade_invalid_cooldown
	}
}
