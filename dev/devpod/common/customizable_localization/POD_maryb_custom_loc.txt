######################################################
# Mary the Black — Journey story panel custom loc
######################################################
#
# Drives the dynamic header text of Mary's POD_maryb_story panel. The story's
# custom_string_key resolves to a loc value that calls
# [Story.GetOwner.Custom('POD_maryb_story_phase')], so the panel prose tracks
# Mary's progress through the arc and, after the Deal with Michael, the ending
# she reached. First matching text wins; the unconditional entry is the
# pre-arc fallback.
#
# type = character: invoked on the story owner (Mary), so the triggers read her
# flags / variables / modifiers directly.
######################################################

POD_maryb_story_phase = {
	type = character

	# Resolved endings first (the three are mutually exclusive flags).
	text = {
		trigger = { has_character_flag = mary_resolved_black_dream }
		localization_key = POD_maryb_story_phase_blackdream
	}
	text = {
		trigger = { has_character_flag = mary_resolved_second_dream }
		localization_key = POD_maryb_story_phase_seconddream
	}

	# Ash and Gold: the synthesis oscillates, so name whichever half currently
	# looks out. Exactly one ascendancy modifier is present at a time; the
	# unmarked entry covers the instant before the first roll lands.
	text = {
		trigger = {
			has_character_flag = mary_resolved_ash_and_gold
			has_character_modifier = mary_ash_and_gold_mary_modifier
		}
		localization_key = POD_maryb_story_phase_ashgold_mary
	}
	text = {
		trigger = {
			has_character_flag = mary_resolved_ash_and_gold
			has_character_modifier = mary_ash_and_gold_michael_modifier
		}
		localization_key = POD_maryb_story_phase_ashgold_michael
	}
	text = {
		trigger = { has_character_flag = mary_resolved_ash_and_gold }
		localization_key = POD_maryb_story_phase_ashgold_neutral
	}

	# Pre-resolution: all three tablets read, the fourth waiting to be written.
	# Reads through the zero-defaulting script value, so no exists guard is needed.
	text = {
		trigger = {
			POD_maryb_tablets_recovered_value >= 3
		}
		localization_key = POD_maryb_story_phase_stylus
	}

	# Fallback: still gathering the tablets (or the arc has not started yet).
	text = {
		localization_key = POD_maryb_story_phase_seeking
	}
}
