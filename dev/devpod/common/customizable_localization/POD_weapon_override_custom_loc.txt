# Renders what an equipped weapon does to a `uses_weapon` combat move, in one
# standardised voice, so every surface says the same thing: the move card, the
# artifact's own description, and anything a designer writes later.
#
# ARTIFACT DESIGNERS: you do not write this text, and you do not touch the
# artifact's description loc either. Stamp the override var on the artifact in
# its creation effect and stop: the artifact tooltip and details window append
# the generated block automatically (their [Artifact.GetDesc] widgets render
# pod_artifact_desc_and_weapon_override instead of the bare desc). The sentence
# below is generated from the vars, so it can never drift from what the code
# actually does.
#
# Do NOT put [PODWeaponOverrideSelfText] inside a description loc key. GetDesc
# localizes the key in a fresh environment that has no Artifact data context
# (its ROOT is a character), so the reference errors out at runtime and the
# whole block renders blank. The macro only resolves at widget level, where the
# GUI's Artifact data context exists.
#
# SCOPE CONVENTION: the composed line reads its weapon from a scope named
# `pod_weapon`. Every caller binds it (see the PODWeaponOverrideText /
# PODWeaponOverrideSelfText macros in data_binding/POD_combat_macros.txt).
# Re-rooting a nested dispatcher on a named scope is the standard engine idiom.
#
# Each segment emits or goes blank, so a weapon carrying several overrides
# reads as one list and a weapon carrying none renders nothing at all.

PODWeaponOvrRetype = {
	type = artifact
	text = {
		trigger = { has_variable = artifact_weapon_damage_class  var:artifact_weapon_damage_class = flag:fire }
		localization_key = pod_weapon_ovr_retype_fire
	}
	text = {
		trigger = { has_variable = artifact_weapon_damage_class  var:artifact_weapon_damage_class = flag:cold }
		localization_key = pod_weapon_ovr_retype_cold
	}
	text = {
		trigger = { has_variable = artifact_weapon_damage_class  var:artifact_weapon_damage_class = flag:lightning }
		localization_key = pod_weapon_ovr_retype_lightning
	}
	text = {
		trigger = { has_variable = artifact_weapon_damage_class  var:artifact_weapon_damage_class = flag:sun }
		localization_key = pod_weapon_ovr_retype_sun
	}
	text = {
		trigger = { has_variable = artifact_weapon_damage_class  var:artifact_weapon_damage_class = flag:silver }
		localization_key = pod_weapon_ovr_retype_silver
	}
	text = {
		trigger = { has_variable = artifact_weapon_damage_class  var:artifact_weapon_damage_class = flag:gold }
		localization_key = pod_weapon_ovr_retype_gold
	}
	text = {
		trigger = { has_variable = artifact_weapon_damage_class  var:artifact_weapon_damage_class = flag:holy }
		localization_key = pod_weapon_ovr_retype_holy
	}
	text = {
		trigger = { has_variable = artifact_weapon_damage_class  var:artifact_weapon_damage_class = flag:balefire }
		localization_key = pod_weapon_ovr_retype_balefire
	}
	text = {
		trigger = { has_variable = artifact_weapon_damage_class  var:artifact_weapon_damage_class = flag:poison }

		trigger = { has_variable = artifact_weapon_damage_class  var:artifact_weapon_damage_class = flag:cold_iron }
		localization_key = pod_weapon_ovr_retype_cold_iron
	}
	text = {
		localization_key = blank_line
	}
}

# Emits the "In a duel:" heading only when the weapon carries any override, so
# the artifact surfaces can append the block to every artifact unconditionally.
PODWeaponOvrHeader = {
	type = artifact
	text = {
		trigger = {
			OR = {
				has_variable = artifact_weapon_damage_class
				has_variable = artifact_weapon_bonus_damage
				has_variable = artifact_weapon_undodgeable
				has_variable = artifact_deals_aggravated
			}
		}
		localization_key = pod_weapon_ovr_header
	}
	text = {
		localization_key = blank_line
	}
}

# The composed line. type = character because the callers root on a character
# (the viewer); the weapon arrives as the added scope `pod_weapon`.
PODWeaponOverrideLine = {
	type = character
	text = {
		localization_key = pod_weapon_ovr_line
	}
}
