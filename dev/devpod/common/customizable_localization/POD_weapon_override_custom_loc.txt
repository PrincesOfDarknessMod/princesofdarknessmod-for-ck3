# Renders what an equipped weapon does to a `uses_weapon` combat move, in one
# standardised voice, so every surface says the same thing: the move card, the
# artifact's own description, and anything a designer writes later.
#
# ARTIFACT DESIGNERS: you do not write this text, and you do not touch the
# artifact's description loc either. Stamp the override var on the artifact in
# its creation effect and stop: the artifact tooltip and details window append
# the generated block automatically (their [Artifact.GetDesc] widgets render
# pod_artifact_desc_and_weapon_override instead of the bare desc). The sentence
# below is generated from the vars, so it can never drift from what the code
# actually does.
#
# Do NOT put [PODWeaponOverrideSelfText] inside a description loc key. GetDesc
# localizes the key in a fresh environment that has no Artifact data context
# (its ROOT is a character), so the reference errors out at runtime and the
# whole block renders blank. The macro only resolves at widget level, where the
# GUI's Artifact data context exists.
#
# SCOPE CONVENTION: the composed line reads its weapon from a scope named
# `pod_weapon`. Both callers bind it: the PODWeaponOverrideSelfText macro
# (data_binding/POD_combat_macros.txt) on the artifact surfaces, and the move
# card's own `save_scope_as = pod_weapon` in pod_combat_skill_guis.txt.
# Re-rooting a nested dispatcher on a named scope is the standard engine idiom.
#
# Each segment emits or goes blank and opens with its own line break, so a
# weapon carrying several overrides stacks one line per override (the same
# demarcation move effects use) and one carrying none renders nothing at all.

# Each arm asks the shared class trigger rather than reading the var inline.
# This is a first-match dispatcher, so an artifact with no class at all walks
# every arm before reaching the blank - and neither AND nor OR short-circuits,
# so a flat sibling read would touch the value on all ten of them. The block
# renders for every artifact, not only weapons.
PODWeaponOvrRetype = {
	type = artifact
	text = {
		trigger = { POD_weapon_artifact_is_class_trigger = { CLASS = fire } }
		localization_key = pod_weapon_ovr_retype_fire
	}
	text = {
		trigger = { POD_weapon_artifact_is_class_trigger = { CLASS = cold } }
		localization_key = pod_weapon_ovr_retype_cold
	}
	text = {
		trigger = { POD_weapon_artifact_is_class_trigger = { CLASS = lightning } }
		localization_key = pod_weapon_ovr_retype_lightning
	}
	text = {
		trigger = { POD_weapon_artifact_is_class_trigger = { CLASS = sun } }
		localization_key = pod_weapon_ovr_retype_sun
	}
	text = {
		trigger = { POD_weapon_artifact_is_class_trigger = { CLASS = silver } }
		localization_key = pod_weapon_ovr_retype_silver
	}
	text = {
		trigger = { POD_weapon_artifact_is_class_trigger = { CLASS = gold } }
		localization_key = pod_weapon_ovr_retype_gold
	}
	text = {
		trigger = { POD_weapon_artifact_is_class_trigger = { CLASS = holy } }
		localization_key = pod_weapon_ovr_retype_holy
	}
	text = {
		trigger = { POD_weapon_artifact_is_class_trigger = { CLASS = balefire } }
		localization_key = pod_weapon_ovr_retype_balefire
	}
	text = {
		trigger = { POD_weapon_artifact_is_class_trigger = { CLASS = poison } }
		localization_key = pod_weapon_ovr_retype_poison
	}
	text = {
		trigger = { POD_weapon_artifact_is_class_trigger = { CLASS = cold_iron } }
		localization_key = pod_weapon_ovr_retype_cold_iron
	}
	text = {
		localization_key = blank_line
	}
}

# Blade of the Kings: the one override still keyed by artifact name, because
# its condition is a power-vs-prowess comparison rather than a tag. The victim
# is not in scope here, so the segment states the rule in words and the trigger
# only asks whether the blade is the weapon. artifact_power is required too:
# the line reads its value, and a blade forged before that var existed has none.
PODWeaponOvrBladeKings = {
	type = artifact
	text = {
		trigger = {
			has_variable = blade_of_the_kings
			has_variable = artifact_power
		}
		localization_key = pod_weapon_ovr_blade_kings
	}
	text = {
		localization_key = blank_line
	}
}

# Emits the "When wielded:" heading only when the weapon carries any override, so
# the artifact surfaces can prepend the block to every artifact unconditionally.
# The condition is POD_weapon_artifact_has_override_trigger rather than a local
# copy of the var list: a weapon whose only override was missing from that copy
# used to render its line with no heading at all.
PODWeaponOvrHeader = {
	type = artifact
	text = {
		trigger = { POD_weapon_artifact_has_override_trigger = yes }
		localization_key = pod_weapon_ovr_header
	}
	text = {
		localization_key = blank_line
	}
}

# The break between the override block and the artifact's own description.
# Gated on the same trigger as the heading, so an artifact with no overrides
# renders neither and its description starts where it always did.
PODWeaponOvrFooter = {
	type = artifact
	text = {
		trigger = { POD_weapon_artifact_has_override_trigger = yes }
		localization_key = pod_weapon_ovr_footer
	}
	text = {
		localization_key = blank_line
	}
}
