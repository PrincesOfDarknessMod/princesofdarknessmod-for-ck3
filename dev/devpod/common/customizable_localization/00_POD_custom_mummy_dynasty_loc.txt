PODGetCustomMummyDynastyName = {
	type = character
	
	text = {
		trigger = {
			exists = global_var:is_custom_mummy_dynasty_details_set
			character:POD_wraith_01 = {
				has_variable = custom_mummy_dynasty_details
				var:custom_mummy_dynasty_details = { 
					has_variable = custom_name
				}
			}
		}
		localization_key = POD_custom_mummy_dynasty_name_custom
	}
	
	text = {
		localization_key = POD_custom_mummy_dynasty_name_default
	}
}
