
@random_outcome_low_ceil = 33
@random_outcome_medium_ceil = 66


PODRandomOutcome = {
	type = all

	text = {
		trigger = { POD_show_random_outcomes_as_percentage = yes }
		localization_key = POD_random_outcome_percentage
		fallback = yes
	}

	text = {
		trigger = { POD_show_random_outcomes_as_uncertain = yes }
		localization_key = POD_random_outcome_uncertain
	}

	text = {
		trigger = {
			scope:value <= @random_outcome_low_ceil
		}
		localization_key = POD_random_outcome_low
	}

	text = {
		trigger = {
			scope:value  > @random_outcome_low_ceil
			scope:value <= @random_outcome_medium_ceil
		}
		localization_key = POD_random_outcome_medium
	}

	text = {
		trigger = {
			scope:value  > @random_outcome_medium_ceil
		}
		localization_key = POD_random_outcome_high
	}
}
