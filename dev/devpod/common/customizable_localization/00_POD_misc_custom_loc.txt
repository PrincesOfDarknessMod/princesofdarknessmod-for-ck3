# Customizable localization entries used by the combat move macros.
# Each entry maps a named "Custom" datafunction to a loc key — the
# macro chains a GuiScope with named sub-scopes, ends with
# .Custom('X'), and CK3 resolves 'X' here to pick the loc key that
# actually renders.
#
# Splitting rendering from data this way lets the macro feed visual
# treatments (currently just PODCombatMoveDesc's rich inline card)
# without touching the underlying scope plumbing.

PODCombatMoveInlineTooltip = {
	type = character

	text = {
