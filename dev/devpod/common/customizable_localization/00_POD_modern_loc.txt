FaeEra = {
	type = character

	text = {
		trigger = {
			has_global_variable = mythic_age
		}
		localization_key = custom_era_mythic_age
	}

	text = {
		trigger = {
			NOT = { has_global_variable = shattering }
		}
		localization_key = custom_era_sundering
	}

	text = {
		trigger = {
			has_global_variable = shattering
			current_date < 1400.1.1
		}
		localization_key = custom_era_shattering
	}

	text = {
		trigger = {
			has_global_variable = shattering

		}
		localization_key = custom_era_interregnum
	}

	text = {
		fallback = yes
		localization_key = custom_era_resurgence
	}
}
