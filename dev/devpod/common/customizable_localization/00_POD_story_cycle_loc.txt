PODPactFrequencyAndStatus = {
	type = character
	
	text = {
		trigger = { POD_pact_obligation_is_prohibitory_trigger = { OBLIGATION = scope:story.var:obligation } }
		localization_key = POD_story_pact_prohibitory
	}
	
	text = {
		trigger = { scope:story = { var:completed ?= yes } }
		localization_key = POD_story_pact_frequency_completed
	}
	
	text = {
		localization_key = POD_story_pact_frequency_notcompleted
	}
}

PODPactDeadline = {
	type = character

	# Prohibitions never come due on a timer -- they bind the thrall at all times.
	text = {
		trigger = { POD_pact_obligation_is_prohibitory_trigger = { OBLIGATION = scope:story.var:obligation } }
		localization_key = POD_story_pact_deadline_prohibitory
	}

	# Already fulfilled this cycle: NOT due. Show when the pact renews.
	text = {
		trigger = {
			scope:story = {
				var:completed ?= yes
				has_variable = years_until_obligation
			}
		}
		localization_key = POD_story_pact_deadline_completed
	}

	# Fulfilled, but the countdown has not been written yet (freshly sealed pact,
	# or a save from before this submod was added). Still not due; omit the timer.
	text = {
		trigger = { scope:story = { var:completed ?= yes } }
		localization_key = POD_story_pact_deadline_completed_nodate
	}

	# Pending and imminent: one year or less remaining and not yet satisfied.
	text = {
		trigger = {
			scope:story = {
				has_variable = years_until_obligation
				var:years_until_obligation <= 1
			}
		}
		localization_key = POD_story_pact_deadline_urgent
	}

	# Pending: countdown present, show years remaining.
	text = {
		trigger = { scope:story = { has_variable = years_until_obligation } }
		localization_key = POD_story_pact_deadline_normal
	}

	# Pending fallback: freshly-signed pact whose countdown has not been written
	# yet (the yearly effect_group has not ticked). The full period still remains.
	text = {
		localization_key = POD_story_pact_deadline_fresh
	}
}
