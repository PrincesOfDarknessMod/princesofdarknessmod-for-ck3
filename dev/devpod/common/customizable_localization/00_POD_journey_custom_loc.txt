PODShowJourneyName = {
	type = all
	
	text = {
		trigger = {
			exists = var:target
		}
		localization_key = POD_journey_name_targeted
	}
	
	text = {
		localization_key = POD_journey_name
	}
}

PODShowJourneyPhase = {
	type = character

	text = {
		trigger = {
			var:current_journey ?= {
				var:phase = flag:planning_phase
			}
		}
		localization_key = POD_journey_phase_planning
	}

	text = {
		trigger = {
			var:current_journey ?= {
				var:phase = flag:travel_phase
			}
		}
		localization_key = POD_journey_phase_travel
	}
	
	text = {
		localization_key = POD_journey_phase_error
	}
}

PODShowJourneyProgress = {
	type = character

	text = {
		trigger = {
			var:journey_detail_window = {
				var:progress_value = flag:instant
			}
		}
		localization_key = POD_journey_progress_instant
	}
	
	text = {
		localization_key = POD_journey_progress
	}
}

PODHasPausableJourney = {
	type = character

	text = {
		trigger = {
			has_variable = current_journey
		}
		localization_key = POD_journey_pausable
	}
	
	text = {
		localization_key = POD_journey_not_pausable
	}
}

PODShowJourneyCost = {
	type = character

	text = {
		trigger = {
			var:journey_detail_window = {
				var:gold_cost     != 0
				var:prestige_cost != 0
				var:piety_cost    != 0
			}
		}
		localization_key = POD_journey_cost_goldprestigepiety
	}

	text = {
		trigger = {
			var:journey_detail_window = {
				var:gold_cost     != 0
				var:prestige_cost != 0
				var:piety_cost     = 0
			}
		}
		localization_key = POD_journey_cost_goldprestige
	}

	text = {
		trigger = {
			var:journey_detail_window = {
				var:gold_cost     != 0
				var:prestige_cost  = 0
				var:piety_cost    != 0
			}
		}
		localization_key = POD_journey_cost_goldpiety
	}

	text = {
		trigger = {
			var:journey_detail_window = {
				var:gold_cost      = 0
				var:prestige_cost != 0
				var:piety_cost    != 0
			}
		}
		localization_key = POD_journey_cost_prestigepiety
	}

	text = {
		trigger = {
			var:journey_detail_window = {
				var:gold_cost     != 0
				var:prestige_cost  = 0
				var:piety_cost     = 0
			}
		}
		localization_key = POD_journey_cost_gold
	}

	text = {
		trigger = {
			var:journey_detail_window = {
				var:gold_cost      = 0
				var:prestige_cost != 0
				var:piety_cost     = 0
			}
		}
		localization_key = POD_journey_cost_prestige
	}

	text = {
		trigger = {
			var:journey_detail_window = {
				var:gold_cost      = 0
				var:prestige_cost  = 0
				var:piety_cost    != 0
			}
		}
		localization_key = POD_journey_cost_piety
	}
	
	text = {
		localization_key = POD_journey_cost_free
	}
}
