PODShowJourneyName = {
	type = story
	
	text = {
		trigger = {
			exists = var:ritual
		}
		localization_key = POD_journey_name_ritual
	}
	
	text = {
		trigger = {
			exists = var:target_trait
		}
		localization_key = POD_journey_name_targeted_trait
	}
	
	text = {
		trigger = {
			exists = var:target_artifact
		}
		localization_key = POD_journey_name_targeted_artifact
	}
	
	text = {
		trigger = {
			exists = var:target
		}
		localization_key = POD_journey_name_targeted
	}
	
	
	text = {
		localization_key = POD_journey_name
	}
}

	}
}

PODShowJourneyCost = {
	type = character

	text = {
		trigger = {
			journey_window_gold_cost     != 0
			journey_window_prestige_cost != 0
			journey_window_piety_cost    != 0
		}
		localization_key = POD_journey_cost_goldprestigepiety
	}

	text = {
		trigger = {
			journey_window_gold_cost     != 0
			journey_window_prestige_cost != 0
			journey_window_piety_cost     = 0
		}
		localization_key = POD_journey_cost_goldprestige
	}

	text = {
		trigger = {
			journey_window_gold_cost     != 0
			journey_window_prestige_cost  = 0
			journey_window_piety_cost    != 0
		}
		localization_key = POD_journey_cost_goldpiety
	}

	text = {
		trigger = {
			journey_window_gold_cost      = 0
			journey_window_prestige_cost != 0
			journey_window_piety_cost    != 0
		}
		localization_key = POD_journey_cost_prestigepiety
	}

	text = {
		trigger = {
			journey_window_gold_cost     != 0
			journey_window_prestige_cost  = 0
			journey_window_piety_cost     = 0
		}
		localization_key = POD_journey_cost_gold
	}

	text = {
		trigger = {
			journey_window_gold_cost      = 0
			journey_window_prestige_cost != 0
			journey_window_piety_cost     = 0
		}
		localization_key = POD_journey_cost_prestige
	}

	text = {
		trigger = {
			journey_window_gold_cost      = 0
			journey_window_prestige_cost  = 0
			journey_window_piety_cost    != 0
		}
		localization_key = POD_journey_cost_piety
	}
	
	text = {
		localization_key = POD_journey_cost_free
	}
}
