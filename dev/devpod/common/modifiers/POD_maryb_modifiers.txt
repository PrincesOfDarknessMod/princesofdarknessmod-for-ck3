# Ash and Gold — permanent identity anchor. Flavor only: the synthesis itself,
# the fact that neither Mary nor Michael has won. The mechanical weight lives
# on the two oscillating ascendancy modifiers below, one of which is always
# present and is re-rolled every few years (POD_maryb_ash_gold_reroll_effect,
# driven by POD_maryb.50's yearly pulse). This modifier never leaves.
mary_ash_and_gold_modifier = {
	icon = diplomacy_positive
}

# Ascendancy — Michael surfaces. The radiant, diplomatic, faith-bright half of
# the synthesis. Mirrors The Second Dream's kit (diplomacy per piety, same-faith
# opinion) at a lighter weight, since it isn't permanent. Loc is gold-italic.
mary_ash_and_gold_michael_modifier = {
	icon = diplomacy_positive
	diplomacy_per_piety_level = 1
	same_faith_opinion = 15
}

# Ascendancy — Mary surfaces. The intriguing, dread-wreathed, predatory half of
# the synthesis. Mirrors The Black Dream's kit (intrigue per piety, knight
# effectiveness from dread) at a lighter weight.
mary_ash_and_gold_mary_modifier = {
	icon = intrigue_positive
	intrigue_per_piety_level = 1
	knight_effectiveness_per_dread = 0.1
}

###############
# Dream House — cadet house modifier. Founded by POD_maryb_found_dream_house_decision
# on the Ash and Gold path once Mary-Michael hold Constantinople. Composes the
# Nephilim/Michael house effects (stress mitigation, prestige from happy
# powerful vassals, legitimacy gain) with a Baali/Moloch opinion shape
# (rf_kindred hatred, same-faith devotion, dynasty/house pull). The synthesis
# is a being of two natures; the house it founds reads as both at once —
# neither purely Michael's Nephilim line nor purely Mary's Baali one, but a
# new thing carved from both.
###############

house_maryb_dream_modifier = {
	icon = prestige_positive
	# Nephilim/Michael block — Toreador founder's house effects.
	stress_gain_mult = 0.5
	monthly_prestige_gain_per_happy_powerful_vassal_mult = 0.5
	legitimacy_gain_mult = 0.2
	# Baali/Moloch opinion shape — most of the Kindred world still sees Mary's
	# body and remembers the diablerist, even with Michael riding alongside.
	rf_kindred_opinion = -50
	same_faith_opinion = 50
	dynasty_opinion = -100
	dynasty_house_opinion = 100
}
