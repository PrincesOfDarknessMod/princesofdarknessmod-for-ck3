court_seer_court_position = {
	skill = learning
	max_available_positions = 1
	court_position_asset = { animation = throne_room_one_handed_passive_3 background = "gfx/interface/illustrations/event_scenes/armory.dds" }

	minimum_rank = duchy

	opinion = {
		value = regular_court_position_opinion
	}

	# How high the aptitude has to be to reach the 5 different modifier levels
	aptitude_level_breakpoints = { 20 40 60 80 }

	# How good is this character in this position? root scope is the holder of the court position
	aptitude = {
		value = 1
		add = {
			value = intrigue
			multiply = 1
			max = 50
			desc = court_position_skill_intrigue
		}
		add = {
			value = learning
			multiply = 1
			max = 50
			desc = court_position_skill_learning
		}

		# possessed and lunatic are both worth a lot, but tone down the
		# combined bonus for a character who has both.
		if = {
			limit = {
				has_trait = possessed
				NOT = { has_trait = lunatic }
			}
			add = {
				value = 20
				desc = court_position_possessed_trait
			}
		}
		if = {
			limit = {
				NOT = { has_trait = possessed }
				has_trait = lunatic
			}
			add = {
				value = 20
				desc = court_position_lunatic_trait
			}
		}
		if = {
			limit = {
				has_trait = possessed
				has_trait = lunatic
			}
			add = {
				value = 30
				desc = court_position_possessed_lunatic_trait
			}
		}

		if = {
			limit = {
				has_trait = blind
			}
			add = {
				value = 10
				desc = court_position_blind_trait
			}
		}

		if = {
			limit = {
				is_malkavian_trigger = yes
			}
			add = {
				# Only 10 because malkavians tend to have auspex as well
				value = 10
				desc = court_position_malkavian_trait
			}
		}
		if = {
			limit = {
				has_trait = auspexadvanced
			}
			add = {
				value = 20
				desc = court_position_auspexadvanced_trait
			}
		}
		else_if = {
			limit = {
				has_trait = auspex
			}
			add = {
				value = 10
				desc = court_position_auspex_trait
			}
		}

		if = { #Cultural Tradition that improves the aptitude of characters of that culture
			limit = {
				culture = { has_cultural_parameter = characters_are_better_court_poets }
			}
			add = {
				value = 5
				desc = court_position_tradition_poetry
			}
		}
		if = { #Being a poet helps a little bit
			limit = {
				has_trait = lifestyle_poet
			}
			add = {
				value = 5
				desc = court_position_poet_trait
			}
		}
		if = {
			limit = { has_trait = lifestyle_mystic }
			add = {
				value = 5
				if = {
					limit = {
						has_trait_xp = { trait = lifestyle_mystic value >= 50 }
					}
					add = 5
				}
				if = {
					limit = {
						has_trait_xp = { trait = lifestyle_mystic value >= 100 }
					}
					add = 5
				}
				desc = court_position_mystic_1_trait
			}
		}

		add = general_aptitude_bonus
	}

	is_shown = {
		is_vampire_trigger = yes
	}

	# Is this an available position for this court? (root is court owner)
	valid_position = {
		highest_held_title_tier >= tier_duchy
		is_vampire_trigger = yes
	}
	
	is_shown_character = {
		scope:employee = {
			OR = {
				is_vampire_trigger = yes
				is_ghoul_trigger = yes
			}
			base_court_position_validity_trigger = { EMPLOYER = scope:liege }
		}
	}
	
	valid_character = {
		scope:employee = {
			base_court_position_validity_trigger = { EMPLOYER = scope:liege }
			POD_has_only_one_full_time_court_position = {
				EXCEPTION = court_seer_court_position
			}
		}
	}

	revoke_cost = {
		prestige =  {
			value = major_court_position_prestige_revoke_cost
			# Set to 0 with Dynasty Perk Inner Circle
			if = {
				limit = {
					exists = dynasty
					dynasty = { has_dynasty_perk = fp2_coterie_legacy_1 }

					employs_court_position = court_seer_court_position
					any_court_position_holder = {
				        type = court_seer_court_position
						exists = dynasty
						house = prev.house
					}
				}
				multiply = 0
			}
		}
	}

	salary = {
		round = no
		gold = {
			value = medium_court_position_salary
			# Set to 0 with Dynasty Perk Inner Circle
			if = {
				limit = { court_position_inner_circle_salary_trigger = { POS = court_seer } }
				multiply = {
					value = 0.0
					desc = inner_circle_salary_mod
					format = "BASE_VALUE_FORMAT_DECIMALS_MINUS_POSITIVE"
				}
			}
			# Halved with obligation hook
			else_if = {
				limit = { 
					court_position_obligation_hook_salary_trigger = {
						POS = court_seer
						LIEGE = scope:liege
					}
				}
				multiply = {
					value = 0.5
					desc = obligation_hook_salary
					format = "BASE_VALUE_FORMAT_DECIMALS_MINUS_POSITIVE"
				}
			}
			# Halved salary
			if = {
				limit = {
					court_position_halved_salary_trigger = { POS = court_seer }
				}
				multiply = {
					value = 0.5
					desc = court_position_halved_salary
					format = "BASE_VALUE_FORMAT_DECIMALS_MINUS_POSITIVE"
				}
			}
			# Increased salary
			if = {
				limit = {
					court_position_increased_salary_trigger = { POS = court_seer }
				}
				multiply = {
					value = 1.5
					desc = court_position_increased_salary
					format = "BASE_VALUE_FORMAT_DECIMALS_MINUS_POSITIVE"
				}
			}
			#You're working with LAAMPS now
			if = {
				limit = { 
					scope:liege ?= { government_has_flag = government_is_landless_adventurer }
				}
				multiply = {
					value = 0
					desc = court_position_laamps_free
					format = "BASE_VALUE_FORMAT_DECIMALS_MINUS_POSITIVE"
				}
			}
		}		
		prestige = {
			value = 0.25
			if = {
				limit = { 
					scope:liege ?= { government_has_flag = government_is_landless_adventurer }
				}
				multiply = 0
			}
		}
	}

	# These modifiers are employed based on what level the aptitude reaches (levels are defined as breakpoints)
	scaling_employer_modifiers = {
		# The first two levels give penalties, but leaving the position empty
		# also gives penalties in the form of occasional bad events
		terrible = {
			monthly_piety = -0.5
		}
		poor = {
			monthly_piety = -0.3
		}
		average = {
			monthly_piety = 0.3
		}
		# Level 4 is also the minimum for getting gehenna prophecies, which give bonuses
		good = {
			monthly_piety = 0.5
			cowed_vassal_tax_contribution_mult = 0.05
		}
		excellent = {
			monthly_piety = 0.5
			intimidated_vassal_tax_contribution_mult = 0.05
			cowed_vassal_tax_contribution_mult = 0.1
		}
	}

	modifier = {
		monthly_prestige = 1
	}

	on_court_position_received = {
		basic_gained_court_position_effect = yes
	}
	on_court_position_revoked = {
		basic_revoked_court_position_effect = yes
	}
	on_court_position_invalidated = {
		basic_invalidated_court_position_effect = yes
	}

	# Scripted value to determine whether an AI will hire/fire a position or not. Hires over 0, fires under -50. Look in _court_positions.info for more extensive documentation!
	ai_position_score = {
		value = 10
		
		add = court_position_candidate_score_base_value
		add = court_position_candidate_aptitude_value
		add = court_position_debt_considerations_value
	}
}

# mediocre courtier reads "Terrible" and gives almost nothing. A polymath
# advisor reads "Excellent" and grants a substantial council-tier package.
#
# Limited to liege-tier characters (duchy and above) to keep them as a

@triumvir_skill_max = 25
@triumvir_breakpoint_poor = 25
@triumvir_breakpoint_average = 50

		animation = throne_room_one_handed_passive_3

		if = { limit = { has_trait = diligent }      add = { value = 10 desc = court_position_diligent_trait } }
		if = { limit = { has_trait = patient }       add = { value = 10 desc = court_position_patient_trait } }
		if = { limit = { has_trait = ambitious }     add = { value = 10 desc = court_position_ambitious_trait } }

		if = { limit = { has_trait = lazy }      add = { value = -25 desc = court_position_lazy_trait } }
		if = { limit = { has_trait = content }   add = { value = -15 desc = court_position_content_trait } }
		if = { limit = { has_trait = arrogant }  add = { value = -10 desc = court_position_arrogant_trait } }
		if = { limit = { has_trait = craven }    add = { value = -10 desc = court_position_craven_trait } }

		is_landless_adventurer = no

	# AND the employer is an independent ruler of at least king tier.

		is_independent_ruler = yes
		highest_held_title_tier >= tier_kingdom
		faith = { has_doctrine_parameter = legacy_of_trinity_unlock }

			base_court_position_validity_trigger = { EMPLOYER = scope:liege }
		}
	}

	valid_character = {

			is_imprisoned = no

# Mary the Black content — Pillar of Gold court position
######################################################
#
# The light of the Synthesis given a face: the seduced convert — radiant,
# faithful, or grieving — pulled down into Mary-Michael's orbit. The twin of
# the Pillar of Ash. Three meaningful differences from Ash:
#
#   1. Aptitude rewards the fall-from-grace profile (devout believer, bound
#      by love, radiant, charismatic, grief-hooked) rather than overt damnation.
#   2. valid_character deliberately does NOT require the candidate to already
#      be infernal — the radiant and faithful are the PREMIER fills.
#   3. on_court_position_received fires THE FALL: the holder takes a permanent
#      maryb_trinity_the_fall_modifier (corruption rides in on the seating).
#      Dismissing them later does NOT un-corrupt — the soul doesn't come back.
######################################################

pillar_of_gold_court_position = {
	skill = diplomacy
	max_available_positions = 1

	court_position_asset = {
		animation = throne_room_one_handed_passive_3
		background = "gfx/interface/illustrations/event_scenes/throne_room.dds"
	}

	opinion = { value = 10 }

	aptitude_level_breakpoints = { 5 9 13 18 }

	aptitude = {
		value = 0

		# The fall-from-grace core.
		if = {
			limit = {
				OR = {
					faith = faith:viacaeli
					faith = faith:codexoflegacies
					faith = faith:viadivinitatis
				}
			}
			add = { value = 3 desc = pillar_of_gold_apt_faith }
		}
		if = { limit = { has_trait = zealous } add = { value = 3 desc = pillar_of_gold_apt_believer } }
		# Bound to Mary by love (employer is always character:790149 here).
		if = { limit = { has_relation_lover = character:790149 } add = { value = 3 desc = pillar_of_gold_apt_lover } }

		# Radiance.
		if = { limit = { is_toreador_trigger = yes } add = { value = 3 desc = pillar_of_gold_apt_toreador } }
		if = { limit = { has_presence_trigger = yes } add = { value = 3 desc = pillar_of_gold_apt_presence } }

		# Soft charisma.
		if = { limit = { diplomacy >= 20 } add = { value = 3 desc = pillar_of_gold_apt_diplomat } }
		else_if = { limit = { diplomacy >= 16 } add = { value = 2 desc = pillar_of_gold_apt_diplomat } }
		else_if = { limit = { diplomacy >= 12 } add = { value = 1 desc = pillar_of_gold_apt_diplomat } }
		
		if = {
			limit = {
				OR = {
					has_trait = gregarious
					has_trait = beauty_good_1
					has_trait = beauty_good_2
					has_trait = beauty_good_3
				}
			}
			add = { value = 2 desc = pillar_of_gold_apt_charisma }
		}
		if = { limit = { learning >= 20 } add = { value = 3 desc = pillar_of_gold_apt_learned } }
		else_if = { limit = { learning >= 16 } add = { value = 2 desc = pillar_of_gold_apt_learned } }
		else_if = { limit = { learning >= 12 } add = { value = 1 desc = pillar_of_gold_apt_learned } }

		# Grief / grievance hook — the wound she can work with.
		if = { limit = { any_relation = { type = rival count >= 1 } } add = { value = 1 desc = pillar_of_gold_apt_grief } }
	}

	is_shown = {
		is_landless_adventurer = no
		has_character_flag = mary_resolved_ash_and_gold
	}
	valid_position = {
		is_alive = yes
	}

	is_shown_character = {
		scope:employee = {
			POD_is_vampire_trigger = yes
			base_court_position_validity_trigger = { EMPLOYER = scope:liege }
		}
	}
	valid_character = {
		scope:employee = {
			POD_is_vampire_trigger = yes
			is_adult = yes
			is_imprisoned = no
			# Deliberately NO "must be infernal" gate — the radiant and faithful
			# are the premier fills; the fall happens on receipt.
			NOT = { has_court_position = pillar_of_ash_court_position }
			base_court_position_validity_trigger = { EMPLOYER = scope:liege }
		}
	}

		piety = { value = 1 }
	}

		terrible  = { same_faith_opinion = 3 }
		poor      = { same_faith_opinion = 6 }
		average   = { same_faith_opinion = 10 monthly_piety_gain_mult = 0.1}
		good      = { same_faith_opinion = 15 monthly_piety_gain_mult = 0.2 }
		excellent = { same_faith_opinion = 25 monthly_piety_gain_mult = 0.3 domain_tax_same_faith_mult = 0.05 }
	}

	on_court_position_received = {
		basic_gained_court_position_effect = yes
		# THE FALL rides in on the seating. A permanent corruption modifier on
		# the holder (deception model — visible faith left intact). Dismissal
		# does NOT remove it; on_court_position_revoked deliberately omits it.
		scope:employee = {
			add_character_modifier = maryb_trinity_the_fall_modifier
			save_scope_as = pillar_holder

		add = court_position_candidate_aptitude_value
		add = court_position_debt_considerations_value
	}
