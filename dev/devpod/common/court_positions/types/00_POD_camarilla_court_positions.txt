praetor_court_position = {
	skill = stewardship
	max_available_positions = 1
	court_position_asset = { animation = throne_room_one_handed_passive_3 background = "gfx/interface/illustrations/event_scenes/armory.dds" }


	opinion = {
		value = ceremonial_court_position_opinion
	}
	
	# How high the aptitude has to be to reach the 5 different modifier levels
	aptitude_level_breakpoints = { 20 40 60 80 }

	# How good is this character in this position? root scope is the holder of the court position
	aptitude = {
		value = 1
		add = {
			value = stewardship
			multiply = 1
			max = 60
			desc = court_position_skill_stewardship
		}
		add = {
			value = diplomacy
			multiply = 1
			max = 30
			desc = court_position_skill_diplomacy
		}
		if = {
			limit = {
				has_trait = just
			}
			add = {
				value = 15
				desc = court_position_just_trait
			}
		}
		else = {
			add = 0.001
			subtract = {
				value = 0.001
				desc = court_position_just_trait_not
			}
		}
		if = {
			limit = {
				has_trait = gregarious
			}
			add = {
				value = 15
				desc = court_position_gregarious_trait
			}
		}
		else = {
			add = 0.001
			subtract = {
				value = 0.001
				desc = court_position_gregarious_trait_not
			}
		}
		if = {
			limit = {
				has_trait = diplomat
			}
			add = {
				value = 30
				desc = court_position_diplomat_trait
			}
		}
		else = {
			add = 0.001
			subtract = {
				value = 0.001
				desc = court_position_diplomat_trait_not
			}
		}
		if = {
			limit = {
				has_trait = dominateadvanced
			}
			add = {
				value = 20
				desc = court_position_dominateadvanced_trait
			}
		}
		else_if = {
			limit = {
				has_trait = dominate
			}
			add = {
				value = 10
				desc = court_position_dominate_trait
			}
		}
		else = {
			add = 0.001
			subtract = {
				value = 0.001
				desc = court_position_dominate_trait_not
			}
		}
		
		# NEGATIVES
		if = {
			limit = {
				has_trait = shy
			}
			add = {
				value = -10
				desc = court_position_shy_trait
			}
		}
		
		add = court_position_aptitude_low_penalty_value
		add = general_aptitude_bonus
	}

	is_shown = {
		faith = faith:camahumanity
		camarilla_status >= 7 # Justicar and Inner Circle
		is_landless_adventurer = no
	}

	# Is this an available position for this court? (root is court owner)
	valid_position = {
	}
	
	is_shown_character = {
		scope:employee = {
			faith = faith:camahumanity
			base_court_position_validity_trigger = { EMPLOYER = scope:liege }
		}
	}
	
	valid_character = {
		scope:employee = {
			faith = faith:camahumanity
			base_court_position_validity_trigger = { EMPLOYER = scope:liege }
		}
	}

	revoke_cost = {
		prestige =  {
			value = major_court_position_prestige_revoke_cost
			# Set to 0 with Dynasty Perk Inner Circle
			if = {
				limit = {
					exists = dynasty
					dynasty = { has_dynasty_perk = fp2_coterie_legacy_1 }

					employs_court_position = praetor_court_position
					any_court_position_holder = {
				        type = praetor_court_position
						exists = dynasty
						house = prev.house
					}
				}
				multiply = 0
			}
		}
	}

	salary = {
		round = no
		gold = {
			value = medium_court_position_salary
			# Set to 0 with Dynasty Perk Inner Circle
			if = {
				limit = { court_position_inner_circle_salary_trigger = { POS = praetor } }
				multiply = {
					value = 0.0
					desc = inner_circle_salary_mod
					format = "BASE_VALUE_FORMAT_DECIMALS_MINUS_POSITIVE"
				}
			}
			# Halved with obligation hook
			else_if = {
				limit = { 
					court_position_obligation_hook_salary_trigger = {
						POS = praetor
						LIEGE = scope:liege
					}
				}
				multiply = {
					value = 0.5
					desc = obligation_hook_salary
					format = "BASE_VALUE_FORMAT_DECIMALS_MINUS_POSITIVE"
				}
			}
			# Halved salary
			if = {
				limit = {
					court_position_halved_salary_trigger = { POS = praetor }
				}
				multiply = {
					value = 0.5
					desc = court_position_halved_salary
					format = "BASE_VALUE_FORMAT_DECIMALS_MINUS_POSITIVE"
				}
			}
			# Increased salary
			if = {
				limit = {
					court_position_increased_salary_trigger = { POS = praetor }
				}
				multiply = {
					value = 1.5
					desc = court_position_increased_salary
					format = "BASE_VALUE_FORMAT_DECIMALS_MINUS_POSITIVE"
				}
			}
			#You're working with LAAMPS now
			if = {
				limit = { 
					scope:liege ?= { government_has_flag = government_is_landless_adventurer }
				}
				multiply = {
					value = 0
					desc = court_position_laamps_free
					format = "BASE_VALUE_FORMAT_DECIMALS_MINUS_POSITIVE"
				}
			}
		}		
		prestige = {
			value = 0.25
			if = {
				limit = { 
					scope:liege ?= { government_has_flag = government_is_landless_adventurer }
				}
				multiply = 0
			}
		}
	}

	# These modifiers are employed based on what level the aptitude reaches (levels are defined as breakpoints) 
	scaling_employer_modifiers = {
		terrible = {
			vassal_limit = -5
		}
		poor = {
			vassal_limit = -2
		}
		average = {
			vassal_limit = 5
		}
		good = {
			vassal_limit = 10
			vassal_tax_contribution_add = 0.1
			vassal_levy_contribution_add = 0.1
		}
		excellent = {
			vassal_limit = 20
			vassal_tax_contribution_add = 0.2
			vassal_levy_contribution_add = 0.2
		}
	}

	modifier = {
		monthly_prestige = 0.5
	}

	on_court_position_received = {
		basic_gained_court_position_effect = yes
	}
	on_court_position_revoked = { 
		basic_revoked_court_position_effect = yes
	}
	on_court_position_invalidated = {
		basic_invalidated_court_position_effect = yes
	}

	# Scripted value to determine whether an AI will hire/fire a position or not. Hires over 0, fires under -50. Look in _court_positions.info for more extensive documentation!
	ai_position_score = {
		value = 10
		
		add = court_position_candidate_score_base_value
		add = court_position_candidate_aptitude_value
		add = court_position_debt_considerations_value
	}
}
