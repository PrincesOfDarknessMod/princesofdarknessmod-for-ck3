		add = {
			value = learning

			desc = court_position_skill_learning
		}
		
		if = {
			limit = { has_trait = eclipsed_sun }
			add = { value = 20 desc = court_position_auspice_trait }
		}

			add = 0.001
			subtract = { value = 0.001 desc = court_position_auspice_trait_not }
		}


		if = {
			limit = { has_trait = calm }

			add = 0.001
			subtract = { value = 0.001 desc = court_position_calm_trait_not }
		}
		if = {
			limit = { has_trait = patient }
			add = { value = 15 desc = court_position_patient_trait }
		}
		else = {
			add = 0.001
			subtract = { value = 0.001 desc = court_position_patient_trait_not }

		if = {
			limit = { has_trait = lifestyle_mystic }

		else = {
			add = 0.001

		if = {
			limit = { has_trait = lazy }
			add = { value = -15 desc = court_position_lazy_trait }
		}
		else = {
			# positive value since the trait would add a penalty
			add = { value = 0.001 desc = court_position_lazy_trait_not }
			subtract = 0.001
		}
		if = {
			limit = { has_trait = craven }
			add = { value = -10 desc = court_position_craven_trait }
		}
		else = {
			# positive value since the trait would add a penalty
			add = { value = 0.001 desc = court_position_craven_trait_not }
			subtract = 0.001
		}
		if = {
			limit = { has_trait = cynical }
			add = { value = -10 desc = court_position_cynical_trait }
		}
		else = {
			# positive value since the trait would add a penalty
			add = { value = 0.001 desc = court_position_cynical_trait_not }

		add = court_position_aptitude_low_penalty_value

	}

	is_shown = { POD_is_mokole_trigger = yes }

	# Is this an available position for this court? (root is court owner)
	valid_position = { POD_is_mokole_trigger = yes }
	
	is_shown_character = {
		scope:employee = {

			value = major_court_position_prestige_revoke_cost
			# Set to 0 with Dynasty Perk Inner Circle
			if = {
				limit = {
					exists = dynasty
					dynasty = { has_dynasty_perk = fp2_coterie_legacy_1 }

					employs_court_position = hibernation_attendant_court_position

						exists = dynasty
						house = prev.house
					}
				}
				multiply = 0
			}
			# Set to 0 via obligation hook
			if = {
				limit = { court_position_obligation_hook_salary_trigger = { POS = hibernation_attendant LIEGE = root } }
				multiply = 0
			}

				limit = {
					exists = this
					is_ai = yes
				}
				multiply = 0
			}
			# Special case for events
			if = {
				limit = {
					exists = this
					has_character_flag = free_revoke_cost
				}
				multiply = 0
			}
			#Lowering for LAAMPS, to match Camp Officer revoke cost
			if = {
				limit = { 
					exists = this
					is_landless_adventurer = yes

			if = {
				limit = { court_position_inner_circle_salary_trigger = { POS = hibernation_attendant } }

					value = 0.0
					desc = inner_circle_salary_mod
					format = "BASE_VALUE_FORMAT_DECIMALS_MINUS_POSITIVE"

						LIEGE = scope:liege
					}
				}
				multiply = {
					value = 0.5
					desc = obligation_hook_salary
					format = "BASE_VALUE_FORMAT_DECIMALS_MINUS_POSITIVE"
				}
			}
			# Halved salary
			if = {
				limit = {
					court_position_halved_salary_trigger = { POS = hibernation_attendant }
				}
				multiply = {
					value = 0.5
					desc = court_position_halved_salary
					format = "BASE_VALUE_FORMAT_DECIMALS_MINUS_POSITIVE"
				}
			}
			# Increased salary
			if = {
				limit = {
					court_position_increased_salary_trigger = { POS = hibernation_attendant }
				}
				multiply = {
					value = 1.5
					desc = court_position_increased_salary
					format = "BASE_VALUE_FORMAT_DECIMALS_MINUS_POSITIVE"
				}
			}
			#You're working with LAAMPS now
			if = {
				limit = { 
					scope:liege ?= { government_has_flag = government_is_landless_adventurer }
				}
				multiply = {
					value = 0
					desc = court_position_laamps_free
					format = "BASE_VALUE_FORMAT_DECIMALS_MINUS_POSITIVE"
				}
			}
		}
		prestige = {
			value = 0.25
			if = {
				limit = { 
					scope:liege ?= { government_has_flag = government_is_landless_adventurer }
				}
				multiply = 0
			}
		}
	}

	scaling_employer_modifiers = {
		terrible = {
			POD_terrain_mnesis_journey_speed_advantage = 0.2

		poor = {
			POD_terrain_mnesis_journey_speed_advantage = 0.4

		}
		excellent = {
			POD_terrain_mnesis_journey_speed_advantage = 1
		}
	}

	modifier = {
		monthly_lifestyle_xp_gain_mult = 0.25
		monthly_prestige = 0.5
	}

	on_court_position_received = { basic_gained_court_position_effect = yes }
	on_court_position_revoked = { basic_revoked_court_position_effect = yes }
	on_court_position_invalidated = { basic_invalidated_court_position_effect = yes }

		value = -100 # AI doesn't use journeys
	}
