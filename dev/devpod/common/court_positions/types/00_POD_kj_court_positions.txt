hidden_minister_of_jade_court_position = {
	skill = stewardship
	max_available_positions = 1
	minimum_rank = empire 
	category = court_position_category_hidden_minister

	opinion = {
		value = regular_court_position_opinion
	}
	
	# How high the aptitude has to be to reach the 5 different modifier levels
	aptitude_level_breakpoints = { 20 40 60 90 }

	# How good is this character in this position? root scope is the holder of the court position
	aptitude = {
		value = 25
		add = {
			value = stewardship
			multiply = 2.5
			max = 50
			desc = court_position_skill_stewardship
		}
		if = {
			limit = {
				has_trait = administrator
			}
			add = {
				value = 20
				desc = court_position_administrator_trait
			}
		}
		if = {
			limit = {
				has_trait = overseer
			}
			add = {
				value = 20
				desc = court_position_overseer_trait
			}
		}
		if = {
			limit = {
				has_trait = diligent
			}
			add = {
				value = 10
				desc = court_position_diligent_trait
			}
		}
		add = court_position_aptitude_family_business_value
		add = court_position_aptitude_low_penalty_value
	}

	is_shown = {
		has_POD_kj_blood_trigger = yes
		has_government = august_court_government
		highest_held_title_tier = tier_empire
	}
	
	valid_position = {
		has_government = august_court_government
		highest_held_title_tier = tier_empire
	}
	
	is_shown_character = {
		scope:employee = {
			# is_courtier_of = scope:liege
			has_POD_kj_blood_trigger = yes
		}
	}
	
	valid_character = {
		scope:employee = {
			is_imprisoned = no
			trigger_if = {
				limit = { NOT = { has_court_position = hidden_minister_of_jade_court_position } }
				court_position_does_not_already_have_a_job_trigger = yes # This positions is considered a job, and cannot be held at the same tims as another similar position
			}			
		}
	}

	revoke_cost = {
		prestige =  {
			value = major_court_position_prestige_revoke_cost
			# Set to 0 with Dynasty Perk Inner Circle
			if = {
				limit = {
					exists = dynasty
					dynasty = { has_dynasty_perk = fp2_coterie_legacy_1 }

					employs_court_position = hidden_minister_of_jade_court_position
					any_court_position_holder = {
				        type = hidden_minister_of_jade_court_position
						exists = dynasty
						house = prev.house
					}
				}
				multiply = 0
			}
		}
	}
	
	salary = {
		round = no
		gold = {
			value = major_court_position_salary
			# Set to 0 with Dynasty Perk Inner Circle
			if = {
				limit = {
					scope:liege = {
						exists = dynasty
						dynasty = { has_dynasty_perk = fp2_coterie_legacy_1 }

						employs_court_position = hidden_minister_of_jade_court_position
						any_court_position_holder = {
					        type = hidden_minister_of_jade_court_position
							exists = dynasty
							house = prev.house
						}
					}
				}
				multiply = {
					value = 0.0
					desc = inner_circle_salary_mod
					format = "BASE_VALUE_FORMAT_DECIMALS_MINUS_POSITIVE"
				}
			}
		}	
	}

	# These modifiers are employed based on what level the aptitude reaches (levels are defined as breakpoints) 
	scaling_employer_modifiers = {
		terrible = {
			vassal_tax_contribution_add = 0.05
			vassal_tax_contribution_mult = 0.1
			monthly_income_mult = 0.025
			domain_tax_mult = 0.025
		}
		poor = {
			vassal_tax_contribution_add = 0.1
			vassal_tax_contribution_mult = 0.2
			monthly_income_mult = 0.05
			domain_tax_mult = 0.05
		}
		average = {
			vassal_tax_contribution_add = 0.2
			vassal_tax_contribution_mult = 0.4
			monthly_income_mult = 0.1
			domain_tax_mult = 0.1
		}
		good = {
			vassal_tax_contribution_add = 0.3
			vassal_tax_contribution_mult = 0.6
			monthly_income_mult = 0.2
			domain_tax_mult = 0.2
		}
		excellent = {
			vassal_tax_contribution_add = 0.4
			vassal_tax_contribution_mult = 0.8
			monthly_income_mult = 0.3
			domain_tax_mult = 0.3
		}
	}

	modifier = {
		monthly_prestige = major_court_position_prestige_salary
	}

	on_court_position_received = {
		basic_gained_court_position_effect = yes
	}
	on_court_position_revoked = { 
		basic_revoked_court_position_effect = yes
	}
	on_court_position_invalidated = {
		basic_invalidated_court_position_effect = yes
	}

	# Scripted value to determine whether an AI will hire/fire a position or not. Hires over 0, fires under -50. Look in _court_positions.info for more extensive documentation!
	candidate_score = {
		value = 10
		add = scope:base_value
		add = court_position_candidate_score_base_value
		add = court_position_candidate_aptitude_value
		add = court_position_debt_considerations_value
	}
}
