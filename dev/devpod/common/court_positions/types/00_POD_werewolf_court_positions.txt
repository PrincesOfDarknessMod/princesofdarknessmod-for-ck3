
master_rite_court_position = {
	skill = learning
	max_available_positions = 1
	category = court_position_category_common
	minimum_rank = duchy 

	opinion = {
		value = regular_court_position_opinion
	}
	
	# How high the aptitude has to be to reach the 5 different modifier levels
	aptitude_level_breakpoints = { 20 40 60 80 }

	# How good is this character in this position? root scope is the holder of the court position
	aptitude = {
		value = 1
		add = {
			value = learning
			multiply = 3
			max = 60
			desc = court_position_skill_learning
		}
		add = {
			value = stewardship
			max = 50
			desc = court_position_skill_stewardship
		}		
		if = {
			limit = {
				has_trait = scholar
			}
			add = {
				value = 10
				desc = court_position_scholar_trait
			}
		}
		if = {
			limit = {
				has_trait = patient
			}
			add = {
				value = 10
				desc = court_position_patient_trait
			}
		}
		if = {
			limit = { has_trait = lifestyle_mystic }
			add = {
				value = 5
				if = {
					limit = {
						has_trait_xp = { trait = lifestyle_mystic value >= 50 }
					}
					add = 5
				}
				if = {
					limit = {
						has_trait_xp = { trait = lifestyle_mystic value >= 100 }
					}
					add = 5
				}
				desc = court_position_mystic_1_trait
			}
		}
		if = {
			limit = {
				has_trait = lifestyle_herbalist
			}
			add = {
				value = 10
				desc = court_position_herbalist_trait
			}
		}	
		add = court_position_aptitude_family_business_value
		
	}

	is_shown = {
		is_werewolf_trigger = yes
	}

	# Is this an available position for this court? (root is court owner)
	valid_position = {
		highest_held_title_tier >= tier_duchy
		is_werewolf_trigger = yes
	}
	
	is_shown_character = {
		scope:employee = {
			OR = {
				is_werewolf_trigger = yes
				has_trait = kinfolk
			}
		}
	}
	
	valid_character = {
		scope:employee = {
			is_imprisoned = no
			trigger_if = {
				limit = { NOT = { has_court_position = master_rite_court_position } }
				#court_position_does_not_already_have_a_job_trigger = yes # This positions is considered a job, and cannot be held at the same tims as another similar position
			}			
		}
	}

	revoke_cost = {
		prestige =  {
			value = major_court_position_prestige_revoke_cost
			# Set to 0 with Dynasty Perk Inner Circle
			if = {
				limit = {
					exists = dynasty
					dynasty = { has_dynasty_perk = fp2_coterie_legacy_1 }

					employs_court_position = master_rite_court_position
					any_court_position_holder = {
				        type = master_rite_court_position
						exists = dynasty
						house = prev.house
					}
				}
				multiply = 0
			}
		}
	}

	salary = {
		round = no
		gold = {
			value = medium_court_position_salary
			# Set to 0 with Dynasty Perk Inner Circle
			if = {
				limit = {
					scope:liege = {
						exists = dynasty
						dynasty = { has_dynasty_perk = fp2_coterie_legacy_1 }

						employs_court_position = master_rite_court_position
						any_court_position_holder = {
					        type = master_rite_court_position
							exists = dynasty
							house = prev.house
						}
					}
				}
				multiply = {
					value = 0.0
					desc = inner_circle_salary_mod
					format = "BASE_VALUE_FORMAT_DECIMALS_MINUS_POSITIVE"
				}
			}
			if = {
				limit = { scope:liege ?= { is_landless_adventurer = yes } }
				multiply = 0
			}
		}		
		prestige = {
			value = 0.25
			if = {
				limit = { scope:liege ?= { is_landless_adventurer = yes } }
				multiply = 0
			}
		}
	}

	# These modifiers are employed based on what level the aptitude reaches (levels are defined as breakpoints) 
	scaling_employer_modifiers = {
		terrible = {
			monthly_lifestyle_xp_gain_mult = 0.1
		}
		poor = {
			monthly_lifestyle_xp_gain_mult = 0.2
		}
		average = {
			monthly_lifestyle_xp_gain_mult = 0.3
		}
		good = {
			monthly_lifestyle_xp_gain_mult = 0.4
		}
		excellent = {
			monthly_lifestyle_xp_gain_mult = 0.5
		}
	}

	modifier = {
		monthly_prestige = minor_court_position_prestige_salary
	}

	on_court_position_received = {
		basic_gained_court_position_effect = yes
	}
	on_court_position_revoked = { 
		basic_revoked_court_position_effect = yes
	}
	on_court_position_invalidated = {
		basic_invalidated_court_position_effect = yes
	}

	# Scripted value to determine whether an AI will hire/fire a position or not. Hires over 0, fires under -50. Look in _court_positions.info for more extensive documentation!
	candidate_score = {
		value = 10
		add = scope:base_value
		add = court_position_candidate_score_base_value
		add = court_position_candidate_aptitude_value
		add = court_position_debt_considerations_value
	}
}

court_seer_court_position = {
	skill = learning
	max_available_positions = 1
	category = court_position_category_common
	minimum_rank = duchy

	opinion = {
		value = regular_court_position_opinion
	}

	# How high the aptitude has to be to reach the 5 different modifier levels
	aptitude_level_breakpoints = { 20 40 60 80 }

	# How good is this character in this position? root scope is the holder of the court position
	aptitude = {
		value = 1
		add = {
			value = intrigue
			multiply = 1
			max = 50
			desc = court_position_skill_intrigue
		}
		add = {
			value = learning
			multiply = 1
			max = 50
			desc = court_position_skill_learning
		}

		# possessed and lunatic are both worth a lot, but tone down the
		# combined bonus for a character who has both.
		if = {
			limit = {
				has_trait = possessed
				NOT = { has_trait = lunatic }
			}
			add = {
				value = 20
				desc = court_position_possessed_trait
			}
		}
		if = {
			limit = {
				NOT = { has_trait = possessed }
				has_trait = lunatic
			}
			add = {
				value = 20
				desc = court_position_lunatic_trait
			}
		}
		if = {
			limit = {
				has_trait = possessed
				has_trait = lunatic
			}
			add = {
				value = 30
				desc = court_position_possessed_lunatic_trait
			}
		}

		if = {
			limit = {
				has_trait = blind
			}
			add = {
				value = 10
				desc = court_position_blind_trait
			}
		}

		if = {
			limit = {
				is_malkavian_trigger = yes
			}
			add = {
				# Only 10 because malkavians tend to have auspex as well
				value = 10
				desc = court_position_malkavian_trait
			}
		}
		if = {
			limit = {
				has_trait = auspexadvanced
			}
			add = {
				value = 20
				desc = court_position_auspexadvanced_trait
			}
		}
		else_if = {
			limit = {
				has_trait = auspex
			}
			add = {
				value = 10
				desc = court_position_auspex_trait
			}
		}

		if = { #Cultural Tradition that improves the aptitude of characters of that culture
			limit = {
				culture = { has_cultural_parameter = characters_are_better_court_poets }
			}
			add = {
				value = 5
				desc = court_position_tradition_poetry
			}
		}
		if = { #Being a poet helps a little bit
			limit = {
				has_trait = poet
			}
			add = {
				value = 5
				desc = court_position_poet_trait
			}
		}
		if = {
			limit = { has_trait = lifestyle_mystic }
			add = {
				value = 5
				if = {
					limit = {
						has_trait_xp = { trait = lifestyle_mystic value >= 50 }
					}
					add = 5
				}
				if = {
					limit = {
						has_trait_xp = { trait = lifestyle_mystic value >= 100 }
					}
					add = 5
				}
				desc = court_position_mystic_1_trait
			}
		}

		add = court_position_aptitude_family_business_value

	}

		is_vampire_trigger = yes
	}

	# Is this an available position for this court? (root is court owner)
	valid_position = {
		highest_held_title_tier >= tier_duchy
		is_vampire_trigger = yes
	}

	is_shown_character = {
		scope:employee = {

	}

			}
