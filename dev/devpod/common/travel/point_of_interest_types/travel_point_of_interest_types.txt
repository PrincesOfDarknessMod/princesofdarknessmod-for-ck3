poi_capitals = {
	build_province_list = {
		# every_kingdom = {
		# 	limit = {
		# 		this = this.holder.primary_title
		# 	}
		# 	holder.capital_province ?= {
		# 		add_to_list = provinces
		# 	}
		# }
		# every_empire = {
		# 	limit = {
		# 		this = this.holder.primary_title
		# 	}
		# 	holder.capital_province ?= {
		# 		add_to_list = provinces
		# 	}
		# }
	}

	on_visit = {
		send_interface_toast = {
			title = poi_capitals.visit
			left_icon = root

			scope:province.barony.holder = {
				if = {
					limit = {
						primary_title = {
							tier >= tier_empire
						}
					}
					save_temporary_scope_value_as = {
						name = xp_gain_amount
						value = root.travel_major_xp
					}
				}
				else = {
					save_temporary_scope_value_as = {
						name = xp_gain_amount
						value = root.travel_medium_xp
					}
				}


				if = { # Merit-government Capitals that have royal courts give merit and the regular reward
					limit = {
						has_royal_court = yes
						government_allows = merit
						root = {
							government_allows = merit
						}
					}
					root = {
						change_merit = medium_merit_gain
					}
				}

				if = { # Merit-government Capitals without royal courts give just merit (no diplo)
					limit = {
						has_royal_court = no
						government_allows = merit
						root = {
							government_allows = merit
						}
					}
					root = {
						change_merit = medium_merit_gain
					}
				}
				else_if = {
					limit = {
						OR = {
							has_royal_court = no
							has_court_type = court_diplomatic
						}
					}
					root = {
						poi_lifestyle_experience_effect = {
							LIFESTYLE = diplomacy
							VALUE = scope:xp_gain_amount
						}
					}
				}
				else_if = {
					limit = {
						OR = {
							has_court_type = court_warlike
							has_court_type = court_tribal
							has_court_type = court_nomadic
						}
					}
					root = {
						poi_lifestyle_experience_effect = {
							LIFESTYLE = martial
							VALUE = scope:xp_gain_amount
						}
					}
				}
				else_if = {
					limit = {
						has_court_type = court_administrative
					}
					root = {
						poi_lifestyle_experience_effect = {
							LIFESTYLE = stewardship
							VALUE = scope:xp_gain_amount
						}
					}
				}
				else_if = {
					limit = {
						has_court_type = court_intrigue
					}
					root = {
						poi_lifestyle_experience_effect = {
							LIFESTYLE = intrigue
							VALUE = scope:xp_gain_amount
						}
					}
				}
				else_if = {
					limit = {
						has_court_type = court_scholarly
					}
					root = {
						poi_lifestyle_experience_effect = {
							LIFESTYLE = learning
							VALUE = scope:xp_gain_amount
						}
					}
				}
				else = {
					root = {
						poi_lifestyle_experience_effect = {
							LIFESTYLE = diplomacy
							VALUE = scope:xp_gain_amount
						}
					}
				}
				# Prestige for Wanderers
				wanderer_lifestyle_destination_effect = yes
			}

			traveler_travel_xp_effect = {
				MIN = 3
				MAX = 5
			}
			if = {
				limit = {
					has_government = landless_adventurer_government
					has_perk = organized_muster_rolls_perk
				}
				add_gold = see_how_its_done_gold_reward_value
			}
			if = {
				limit = {
					has_government = landless_adventurer_government
					has_perk = centralization_perk
				}
				custom_tooltip = {
					text = adventurers_centralization_perk.effect
				}
				every_courtier = {
					hidden_effect = {
						local_court_type_skill_point_effect = { PROVINCE = scope:province }
					}
				}
			}
		}

		visiting_poi_effect = yes
	}
}

# Walls and Forts
poi_special_buildings_martial = {
	build_province_list = {
		every_special_building_province = {
			limit = {
				#OR = {
					has_building_with_flag = { flag = travel_point_of_interest_martial }
					# POD CfV: No walls
					#has_building_with_flag = { flag = travel_point_of_interest_martial_minor }
				#}
			}
			add_to_list = provinces
		}
	}

	on_visit = {
		send_interface_toast = {
			title = poi_special_buildings.visit
			left_icon = root
			if = {
				# Walls and Forts
				limit = {
					scope:province = {
						has_building_with_flag = { flag = travel_point_of_interest_martial }
					}
				}
				add_poi_martial_experience_effect = yes
				traveler_danger_xp_effect = {
					MIN = 3
					MAX = 5
				}
			}
			else_if = {
				# Minor POI (mostly for multi-province defenses like Hadrian's wall)
				limit = {
					scope:province = {
						has_building_with_flag = { flag = travel_point_of_interest_martial_minor }
					}
				}
				poi_lifestyle_experience_effect = {
					LIFESTYLE = martial
					VALUE = travel_miniscule_lifestyle_xp
				}
				traveler_travel_xp_effect = {
					MIN = 1
					MAX = 1
				}
			}
			else = {
				# Rest / Fallback
				add_poi_learning_experience_effect = yes
				traveler_travel_xp_effect = {
					MIN = 3
					MAX = 5
				}
			}
			if = {
				limit = {
					has_government = landless_adventurer_government
					has_perk = organized_muster_rolls_perk
				}
				add_gold = see_how_its_done_gold_reward_value
			}
			
			# Prestige for Wanderers
			wanderer_lifestyle_destination_effect = yes
		}
		
		#For Cilician Gates and other EP3 monuments
		trigger_event = {
			on_action = on_visited_ep3_monument
		}

		visiting_poi_effect = yes
	}
}

# Holy sites with an enshrined artifact but no religious POI building
poi_standalone_holy_site = {
	build_province_list = {
		every_religion_global = {
			every_faith = {
				limit = {
					num_county_followers > 0
				}
				every_holy_site = {
					limit = {
						holy_site_has_artifact = yes
						holy_site_barony.title_province ?= {
							NOT = { has_building_with_flag = { flag = travel_point_of_interest_religious } }
						}
					}
					holy_site_barony.title_province ?= { add_to_list = provinces }
				}
			}
		}
	}

	# Remember which holy site caused this POI to be generated, so localization does not break if that site later goes inactive
	on_added = {
		barony ?= {
			random_barony_holy_site = {
				limit = {
					holy_site_has_artifact = yes
					any_holy_site_faith = { num_county_followers > 0 }
				}
				save_scope_as = poi_holy_site

				random_holy_site_faith = {
					limit = {
						num_county_followers > 0
					}
					save_scope_as = poi_holy_site_faith
				}
			}
		}
		if = {
			limit = {
				exists = scope:poi_holy_site
				exists = scope:poi_holy_site_faith
			}
			set_variable = {
				name = poi_standalone_holy_site
				value = scope:poi_holy_site
			}
			set_variable = {
				name = poi_standalone_holy_site_faith
				value = scope:poi_holy_site_faith
			}
		}
	}

	on_removed = {
		if = {
			limit = {
				has_variable = poi_standalone_holy_site
				has_variable = poi_standalone_holy_site_faith
			}
			remove_variable = poi_standalone_holy_site
			remove_variable = poi_standalone_holy_site_faith
		}
	}

	on_visit = {
		send_interface_toast = {
			title = poi_standalone_holy_site.visit
			left_icon = root

			add_poi_learning_experience_effect_minor = yes
			if = {
				limit = {
					scope:province.barony ?= {
						is_holy_site_of = root.faith
					}
				}
				scope:province.barony = {
					random_barony_holy_site = {
						limit = { any_holy_site_faith = { this = root.faith } }
						save_scope_as = holy_site
					}
				}
				if = {	# Regular Holy Site
					limit = {
						exists = scope:holy_site
						root.faith = {
							is_faith_holy_site = scope:holy_site
							NOT = { is_faith_eminent_holy_site = scope:holy_site }
						}
					}
					add_piety = point_of_interest_minor_piety_value
					if = { # Bonuses from Zeal Phases
						limit = {
							top_participant_group:the_christian_church ?=  {
								has_participant_group_parameter = the_christian_church_holy_site_poi_bonus
							}
						}
						add_poi_learning_experience_effect_minor = yes
						add_scholar_trait_or_xp_effect = { VALUE = 5 }
					}
				}
				if = { # Eminent Holy Site
					limit = {
						exists = scope:holy_site
						root.faith = {
							is_faith_eminent_holy_site = scope:holy_site
						}
					}
					add_piety = point_of_interest_piety_value
					if = { # Bonuses from Zeal Phases
						limit = {
							top_participant_group:the_christian_church ?= {
								has_participant_group_parameter = the_christian_church_holy_site_poi_bonus
							}
						}
						add_poi_learning_experience_effect = yes
						add_scholar_trait_or_xp_effect = { VALUE = 10 }
					}
				}
			}

			# Prestige for Wanderers
			wanderer_lifestyle_destination_effect = yes

			if = {
				limit = {
					has_trait = lifestyle_mystic
				}
				add_trait_xp = {
					trait = lifestyle_mystic
					value = lifestyle_mystic_xp_gain_minor_value
				}
			}

			traveler_travel_xp_effect = {
				MIN = 1
				MAX = 2
			}
			if = {
				limit = {
					has_government = landless_adventurer_government
					has_perk = organized_muster_rolls_perk
				}
				add_gold = see_how_its_done_gold_reward_value
			}
			if = {
				limit = {
					has_perk = church_and_state_perk
					has_government = landless_adventurer_government
				}
				add_piety = adventurer_church_and_state_perk_value
			}
		}

		visiting_poi_effect = yes
	}
}

poi_mandala_capital = {
	build_province_list = {
		every_in_global_list = {
			variable = mandala_poi_list
			limit = { has_building_with_flag = { flag = mandala_capital_building } }
			add_to_list = provinces
		}
	}

	on_visit = {
		send_interface_toast = {
			title = poi_gp_buildings.visit
			left_icon = root

			add_learning_lifestyle_xp = minor_lifestyle_experience

			#Visiting Mandala Capital
			scope:province = {
				if = {
					limit = {  #Visiting Mandala Overlord
						root = {
							OR = {
								is_tributary = yes
								is_independent_ruler = no
							}
						}
						root.overlord ?= { government_has_flag = government_is_mandala }
						this = root.overlord.capital_province
						has_ruined_great_building = no
					}
					save_temporary_scope_value_as = {
						name = piety_gain_amount
						value = {
							value = {
								if = {
									limit = {
										root = { government_has_flag = government_is_mandala }
									}
									value = medium_piety_value
								}
								else = {
									value = minor_piety_value
								}
							}
							multiply = {
								switch = {
									trigger = has_building 
									mandala_capital_01 = { value = 1 }
									mandala_capital_02 = { value = 2 }
									mandala_capital_03 = { value = 3 }
									mandala_capital_04 = { value = 4 }
									mandala_capital_05 = { value = 5 }
								}
							}
						}
					}
				}
				else_if = { #Visiting Mandala Capital - Unaffiliated
					limit = {
						this.county.holder.capital_province = this
						NOT = { root = this.county.holder }
						has_ruined_great_building = no
						root = { #Only Tribals, Mandalas, or those of the same religion care
							OR = {
								government_has_flag = government_is_mandala
								government_has_flag = government_is_tribal
								faith.religion = prev.county.holder.faith.religion
							}
						}
					}
					save_temporary_scope_value_as = {
						name = piety_gain_amount
						value = {
							value = {
								if = {
									limit = {
										root = { government_has_flag = government_is_mandala }
									}
									value = minor_piety_value
								}
								else = {
									value = miniscule_piety_value
								}
							}
							multiply = {
								switch = {
									trigger = has_building 
									mandala_capital_01 = { value = 1 }
									mandala_capital_02 = { value = 2 }
									mandala_capital_03 = { value = 3 }
									mandala_capital_04 = { value = 4 }
									mandala_capital_05 = { value = 5 }
								}
							}
						}
					}
				}
				else_if = { #Visiting Mandala Capital - Ruined
					limit = {
						NOT = { root = this.county.holder }
						has_ruined_great_building = yes
						root = { #Only Tribals, Mandalas, or those of the same religion care
							OR = {
								government_has_flag = government_is_mandala
								government_has_flag = government_is_tribal
								faith.religion = prev.county.holder.faith.religion
							}
						}
					}
					save_temporary_scope_value_as = {
						name = piety_gain_amount
						value = miniscule_piety_value
					}
				}
				#Otherwise we do not care at all

				#Now you get your piety:
				if = {
					limit = { exists = scope:piety_gain_amount }
					root = { add_piety = scope:piety_gain_amount }
				}
			}
			
			# Prestige for Wanderers
			wanderer_lifestyle_destination_effect = yes

			if = { #Small XP Boost based on Mandala holders aspect
				limit = {
					government_has_flag = government_is_mandala
					scope:province.county.holder.capital_province = scope:province
					scope:province.county.holder = { has_mandala_aspect_trigger = yes }
				}
				scope:province.county.holder.house ?= {
					switch = {
						trigger = has_house_aspiration_parameter
						aspect_of_creation = {
							root = {
								poi_lifestyle_experience_effect = {
									LIFESTYLE = stewardship
									VALUE = minor_lifestyle_xp
								}
							}
						}
						aspect_of_serenity = {
							root = {
								poi_lifestyle_experience_effect = {
									LIFESTYLE = diplomacy
									VALUE = minor_lifestyle_xp
								}
							}
						}
						aspect_of_destruction = {
							root = {
								poi_lifestyle_experience_effect = {
									LIFESTYLE = martial
									VALUE = minor_lifestyle_xp
								}
							}
						}
						aspect_of_trickery = {
							root = {
								poi_lifestyle_experience_effect = {
									LIFESTYLE = intrigue
									VALUE = minor_lifestyle_xp
								}
							}
						}
					}
				}
			}

			if = {
				limit = {
					has_trait = lifestyle_mystic
				}
				add_trait_xp = {
					trait = lifestyle_mystic
					value = lifestyle_mystic_xp_gain_minor_value
				}
			}

			if = {
				limit = {
					dynasty ?= {
						has_dynasty_perk = tgp_sea_legacy_1
					}
				}
				add_legitimacy_effect = { LEGITIMACY = minor_legitimacy_gain }
			}

			traveler_travel_xp_effect = {
				MIN = 1
				MAX = 2
			}

poi_grand_cathedral = {
	build_province_list = {
		every_in_global_list = {
			variable = grand_cathedral_poi_list
			limit = { has_building_with_flag = { flag = grand_cathedral_flag } }
			add_to_list = provinces
		}
	}

	on_visit = {
		send_interface_toast = {
			title = poi_grand_cathedral_visit
			left_icon = root

			if = {
				limit = {
					scope:province = {
						exists = var:grand_cathedral_tier
						var:grand_cathedral_tier >= 2
					}
				}
				if = {
					limit = { is_playable_character = yes }	# Only playable characters have spiritual fulfillment
					change_spiritual_fulfillment = minor_spiritual_fulfillment_value
				}
				add_piety = minor_piety_gain
			}

			#Christians get spiritual fulfillment and learning lifestyle xp from frescoes
			if = {
				limit = {
					has_spiritual_fulfillment_type = christian_fulfillment
				}
				if = {
					limit = {
						scope:province = {
							has_variable = cathedral_learning_xp
						}
					}
					add_poi_learning_experience_effect_minor = yes
					if = {
						limit = {
							is_playable_character = yes	# Only playable characters have spiritual fulfillment
							scope:province = {
								exists = var:grand_cathedral_tier
								var:grand_cathedral_tier >= 2
							}
						}
						change_spiritual_fulfillment = major_spiritual_fulfillment_value
					}
					else_if = {
						limit = { is_playable_character = yes }
						change_spiritual_fulfillment = medium_spiritual_fulfillment_value
					}
				}
			}
			else = {
				if = {
					limit = {
						scope:province = {
							has_variable = cathedral_learning_xp
						}
					}
					add_poi_learning_experience_effect = yes
				}
				else = {
					add_poi_learning_experience_effect_minor = yes
				}
			}

			#Variables set by frescoing the cathedral nicely
			#Stress loss
			if = {
				limit = {
					scope:province = {
						has_variable = cathedral_stress_loss
					}
				}
				stress_and_fulfillment_impact = { base = minor_stress_impact_loss }
			}
			if = {
				limit = {
					scope:province = {
						has_variable = cathedral_glass_marian_light
					}
				}
				stress_and_fulfillment_impact = { base = minor_stress_impact_loss }
				if = {
					limit = {
						has_spiritual_fulfillment_type = christian_fulfillment
					}
					if = {
						limit = {
							is_playable_character = yes	# Only playable characters have spiritual fulfillment
							has_personal_tenet = tenet_dulia
						}
						change_spiritual_fulfillment = major_spiritual_fulfillment_value
					}
					else_if = {
						limit = { is_playable_character = yes }
						change_spiritual_fulfillment = medium_spiritual_fulfillment_value
					}
				}
			}

			#Variables set by stained glass program
			#Tree of Jesse dynasty prestige is reserved for the travel plan leader so it doesn't spam for every entourage member
			if = {
				limit = {
					is_travel_leader = yes
					scope:province = {
						has_variable = cathedral_glass_tree_of_jesse
					}
					exists = dynasty
				}
				dynasty = {
					add_dynasty_prestige = minor_dynasty_prestige_gain
				}
			}
			if = {
				limit = {
					scope:province = {
						has_variable = cathedral_glass_dynasty
					}
				}
				add_prestige = minor_prestige_gain
			}
			if = {
				limit = {
					has_spiritual_fulfillment_type = christian_fulfillment
					scope:province = {
						has_variable = {
							name = cathedral_glass_martyrdom
							name = cathedral_glass_tree_of_jesse
							name = cathedral_glass_dynasty
							name = cathedral_glass_pilgrim
						}
					}
				}
				if = {
					limit = { is_playable_character = yes }	# Only playable characters have spiritual fulfillment
					change_spiritual_fulfillment = minor_spiritual_fulfillment_value
				}
			}

			#Big piety from combined piety cathedral and holy site
			if = {
				limit = {
					OR = {
						scope:province = {
							has_variable = cathedral_piety
						}
						scope:province = {
							has_variable = cathedral_glass_martyrdom
						}
						scope:province.barony ?= {
							is_holy_site_of = root.faith
						}
					}
				}
				add_piety = {
					value = point_of_interest_piety_value
					multiply = 1
				}
			}
			#Regular piety
			else_if = {
				limit = {
					scope:province = {
						has_variable = cathedral_piety
						barony ?= {
							is_holy_site_of = root.faith
						}
					}
				}
				add_piety = point_of_interest_piety_value
			}

			#Pilgrim XP
			if = {
				limit = {
					has_trait = pilgrim
					scope:province = {
						has_variable = cathedral_extended_nave
					}
				}
				add_trait_xp = {
					trait = pilgrim
					value = {
						value = pilgrim_xp_gain_minor_value
						multiply = 1
					}
				}
				add_piety = minor_piety_gain
			}
			else_if = {
				limit = {
					has_trait = pilgrim
					scope:province = {
						has_variable = cathedral_pilgrim_xp
					}
				}
				add_trait_xp = {
					trait = pilgrim
					value = {
						value = pilgrim_xp_gain_minor_value
						multiply = 1
					}
				}
			}
			else_if = {
				limit = {
					has_trait = pilgrim
					scope:province = {
						has_variable = cathedral_glass_pilgrim
					}
				}
				add_trait_xp = {
					trait = pilgrim
					value = {
						value = pilgrim_xp_gain_minor_value
						multiply = 1
					}
				}
				add_piety = minor_piety_gain
			}

			#Scholar XP
			if = {
				limit = {
					has_trait = lifestyle_scholar
					scope:province = {
						has_variable = cathedral_scholar_xp
					}
				}
				add_trait_xp = {
					trait = lifestyle_scholar
					value = {
						value = scholar_xp_gain_minor_value
						multiply = 1
					}
				}
			}
			if = {
				limit = {
					has_trait = zealous
					scope:province = {
						has_variable = cathedral_glass_martyrdom
					}
				}
				add_piety = major_piety_gain
				stress_and_fulfillment_impact = { base = minor_stress_impact_loss }
			}

			#Mystic
			if = {
				limit = {
					has_trait = lifestyle_mystic
				}
				if = {
					limit = {
						scope:province = {
							has_variable = cathedral_stars_vault
						}
					}
					add_trait_xp = {
						trait = lifestyle_mystic
						value = {
							value = lifestyle_mystic_xp_gain_medium_value
							multiply = 1
						}
					}
				}
				else_if = {
					limit = {
						scope:province = {
							has_variable = cathedral_mystic_xp
						}
					}
					add_trait_xp = {
						trait = lifestyle_mystic
						value = {
							value = lifestyle_mystic_xp_gain_medium_value
							multiply = 1
						}
					}
				}
				else = {
					add_trait_xp = {
						trait = lifestyle_mystic
						value = lifestyle_mystic_xp_gain_minor_value
					}
				}
			}

			# Prestige for Wanderers
			wanderer_lifestyle_destination_effect = yes

			traveler_travel_xp_effect = {
				MIN = 1
				MAX = 2
			}

			if = {
				limit = {
					has_government = landless_adventurer_government
					has_perk = organized_muster_rolls_perk
				}
				add_gold = see_how_its_done_gold_reward_value
			}
			if = {
				limit = {
					has_perk = church_and_state_perk
					has_government = landless_adventurer_government
				}
				add_piety = adventurer_church_and_state_perk_value
			}
		}

		visiting_poi_effect = yes
	}
