werewolf_moon_bridge_option = {
	is_shown = {
		is_ruler = yes
		OR = {
			is_werewolf_trigger = yes
			AND = {
				has_faith = faith:qorjin_baghatur
				is_target_in_global_variable_list = {
					name = unavailable_unique_decisions
					target = flag:qorjin_following_last_footsteps_objective_decision
				}
			}
		}
	}

	is_valid = {
		trigger_if = {
			limit = {
				is_target_in_global_variable_list = {
					name = unavailable_unique_decisions
					target = flag:qorjin_following_last_footsteps_objective_decision
				}
			}
			OR = {
				has_trait_xp = { trait = werewolf value >= 40 }
				has_faith = faith:qorjin_baghatur
			}
		}
		trigger_else = {
			has_trait_xp = { trait = werewolf value >= 40 }
		}
	}

	cost = {
		piety = 100
	}

	travel_modifier = {
		travel_speed = 30
		travel_safety = 10
	}

	ai_will_do = { value = 0 }
}

				count >= 3
			}
		}
	}

	}

			add = { value = -20 }
		}
	}
