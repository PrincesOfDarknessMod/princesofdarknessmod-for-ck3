	lifestyle = melancholy_lifestyle
	tree = obtenebration
	position = { 0 0 }
	icon = node_melancholy
	
	can_be_picked = {
		has_obtenebration_trigger = yes
	}		
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_melancholy_interest_trigger = yes
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = melancholy_obtenebration_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = shadow_play_perk }	
			}
			multiply = 0
		}
	}
	
	effect = {
		custom_description_no_bullet = { 
			text = eyes_of_the_abyss_perk_effect
		}
		if = {
			limit = {
				has_obtenebration_trigger = no
			}
			custom_description_no_bullet = { 
				text = NEEDS_DISCIPLINE_TEXT
			}
		}
	}
}
