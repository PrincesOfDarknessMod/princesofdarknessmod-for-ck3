seduction_perk = {
	lifestyle = breedgifts_lifestyle
	tree = kojingifts
	position = { 0 0 }
	icon = node_animal
	
	can_be_picked = {
		OR = {
			has_trait = kinfolk
			has_trait = homid_born
		}
		POD_has_access_to_fera_gifts = { TRAIT = trait:kitsune }
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				OR = {
					has_trait = kinfolk
					has_trait = homid_born
				}
				POD_has_access_to_fera_gifts = { TRAIT = trait:kitsune }
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = breedgifts_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = smell_of_man_perk }
			}
			multiply = 0
		}
	}

	character_modifier = {
		seduce_scheme_phase_duration_add = -37
	}
	
	effect = {
		if = {
			limit = {
				NOT = {
					has_trait = homid_born
				}
			}
			custom_description_no_bullet = { 
				text = NEEDS_HOMIDBORN_TEXT
			}
		}
	}
}
