tears_of_the_martyr_perk = {
	lifestyle = faith_lifestyle
	tree = ha_spiritus
	position = { 0 0 }
	icon = node_learning

	can_be_picked = {
		has_trait = supehunter
		OR = {
			faith = faith:shadowinquisition
			faith = { has_doctrine_parameter = access_to_holy_arts }
		}
		OR = {
			has_contemplative_order_dynasty_trigger = yes
			has_mendicant_order_dynasty_trigger = yes
			POD_has_true_faith_trigger = yes
		}
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}
	
	character_modifier = {
	}

	effect = {
		custom_description_no_bullet = {
			text = tears_of_the_martyr_perk_effect
		}
		if = {
			limit = {
				has_contemplative_order_dynasty_trigger = no
				has_mendicant_order_dynasty_trigger = no
				POD_has_true_faith_trigger = no
			}
			custom_description_no_bullet = { 
				text = SPIRITUS_NOT_PRIMARY_TEXT
			}
		}
		if = {
			limit = {
				NAND = {
					OR = {
						faith = { has_doctrine_parameter = access_to_holy_arts }
						faith = faith:shadowinquisition
					}
					has_trait = supehunter
				}
			}
			custom_description_no_bullet = { 
				text = SHADOW_INQUISITION_ONLY_TEXT
			}
		}
	}
}
