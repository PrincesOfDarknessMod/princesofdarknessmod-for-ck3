helter_skelter_perk = {
	lifestyle = horror_lifestyle
	tree = poltergeist
	position = { 0 0 }
	icon = node_soul_arts

	can_be_picked = {
		has_trait = lament
		NOT = { has_focus = horror_banshee_focus }
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = horror_poltergeist_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}
	
	character_modifier = {
		prowess_no_portrait = 1
	}
	
	effect = {
		if = {
			limit = { has_focus = horror_banshee_focus }
			custom_description_no_bullet = { text = has_horror_banshee_focus }
		}
		custom_description_no_bullet = { 
			text = helter_skelter_perk_effect
		}
	}
}
