inhabit_perk = {
	lifestyle = horror_lifestyle
	tree = haunter
	position = { 0 0 }
	icon = node_soul_arts

	can_be_picked = {
		has_trait = lament
		NOT = { has_focus = horror_phantasm_focus }
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = horror_haunter_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}
	
	effect = {
		if = {
			limit = { has_focus = horror_phantasm_focus }
			custom_description_no_bullet = { text = has_horror_phantasm_focus }
		}
		custom_description_no_bullet = { 
			text = inhabit_perk_effect
		}
	}
}
