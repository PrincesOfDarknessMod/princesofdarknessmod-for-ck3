# Meritocracy
meritocracy_perk = {
	lifestyle = stewardship_lifestyle
	tree = duty
	position = { 1 0 }
	icon = node_stewardship
	name = {
		first_valid = {
			triggered_desc = {
				trigger = { 
					exists = this
					government_has_flag = government_is_merchant 
				}
				desc = meritocracy_perk_merchant_name
			}
			desc = meritocracy_perk_name
		}
	}

	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_education_stewardship_trigger = yes
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = stewardship_duty_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
		courtier_and_guest_opinion = 5
	}
	government_character_modifier = {
		flag = government_has_merit
		monthly_merit_mult = 0.05
	}
	government_character_modifier = {
		flag = government_is_republic
		monthly_income_from_trade_mult = 0.05
	}

	effect = {
		if = {
			limit = {
				NOT = { government_has_flag = government_uses_landless_lifestyle_perks }
			}
			custom_description_no_bullet = {
				text = meritocracy_perk_aptitude_effect
			}
		}
		if = {
			limit = {
				NOR = {
					government_has_flag = government_uses_landless_lifestyle_perks
					government_has_flag = government_is_republic
				}
				OR = {
					NOT = { government_allows = merit }
					government_has_flag = government_is_meritocratic
				}
			}
			custom_description_no_bullet = {
				text = meritocracy_perk_effect
			}
		}
		if = {
			limit = { government_has_flag = government_is_merchant }
			custom_description_no_bullet = { text = meritocracy_perk_merchant_effect }
		}
	}
}

	name = {
		first_valid = {
			triggered_desc = {

				desc = toe_the_line_perk_merchant_name
			}
			triggered_desc = {
				trigger = { has_government = landless_adventurer_government }
				desc = toe_the_line_perk_adventurer_name
			}
			desc = toe_the_line_perk_name
		}
	}

	government_character_modifier = {
		flag = government_uses_landless_lifestyle_perks
		invert_check = yes
		civic_tax_mult = lifestyle_perk_civic_tax_mult
	}

	effect = {
		if = {
			limit = {
				NOR = {
					government_has_flag = government_is_landless_adventurer
					government_has_flag = government_is_merchant
				}
			}
			custom_description_no_bullet = {
				text = toe_the_line_perk_effect
			}
		}
		if = {
			limit = { government_has_flag = government_is_merchant }
			custom_description_no_bullet = { text = toe_the_line_perk_merchant_effect }
			hidden_effect = {
				if = {
					limit = {
						house ?= {
							OR = {
								has_house_unity_stage = rapacious_cs
								has_house_unity_stage = opportunistic_cs
							}
						}
					}
					add_character_modifier = {
						modifier = merchant_steady_hand_low_unity_modifier
					}
				}
			}
		}
		if = {
			limit = {
				government_has_flag = government_is_landless_adventurer
			}
			custom_tooltip = {
				text = adventurer_toe_the_line_perk_effect
			}
		}
		if = {
			limit = {
				government_has_flag = government_is_landless_adventurer
			}
			custom_tooltip = {
				text = adventurer_toe_the_line_perk_2_effect
			}
		}
	}
}
