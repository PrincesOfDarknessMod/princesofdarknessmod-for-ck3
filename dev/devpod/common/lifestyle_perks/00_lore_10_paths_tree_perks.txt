find_path_perk = {
	lifestyle = lore_lifestyle
	tree = paths
	position = { 0 0 }
	icon = node_demon_arts

	can_be_picked = {
		dynasty = dynasty:14000006
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = lore_house_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
		pursue_efficiency = 0.25
		retreat_losses = -0.1
	}
	
	effect = {
		if = {
			limit = { not = { dynasty = dynasty:14000006 } }
			custom_description_no_bullet = { text = NOT_has_annunaki_dynasty }
		}
		custom_description_no_bullet = { 
			text = find_path_perk_effect
		}
	}
}
