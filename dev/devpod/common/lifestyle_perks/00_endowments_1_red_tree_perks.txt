red_vigil_perk = {
	lifestyle = endowments_lifestyle
	tree = red
	position = { 0 0 }
	icon = red_vigil
	
	can_be_picked = {
		exists = dynasty
		OR = {
			dynasty = dynasty:13000001
			dynasty = { has_dynasty_modifier = custom_contemplative_order_modifier }
		}
		has_trait = supehunter
		faith = faith:shadowinquisition
	}		
	
	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = endowments_all_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = red_sunstone_perk }
			}
			multiply = 0
		}
	}
	effect = {
		add_trait_force_tooltip = red_vigil
		if = {
			limit = {
				NOR = {
					dynasty = dynasty:13000001
					AND = {
					exists = dynasty
					dynasty = { has_dynasty_modifier = custom_contemplative_order_modifier }
					}
				}
			}
			custom_description_no_bullet = { 
				text = red_only_text
			}
		}
	}
}
