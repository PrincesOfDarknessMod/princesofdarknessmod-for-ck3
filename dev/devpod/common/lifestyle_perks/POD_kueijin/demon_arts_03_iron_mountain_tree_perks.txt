demon_arts_infernal_fortitude_perk = {
	lifestyle = demon_arts_lifestyle
	tree = iron_mountain
	position = { 0 1 }
	icon = year_of_lotus

	can_be_picked = {
		demon_arts_lifestyle_perk_points >= demon_arts_infernal_fortitude_perk_cost
	}

	effect = {
		custom_description_no_bullet = { text = demon_arts_infernal_fortitude_perk_effect }
		if = { 
			limit = { 
				NOT = { has_perk = demon_arts_infernal_fortitude_perk } 
			} 
			custom_description_no_bullet = { text = demon_arts_infernal_fortitude_perk_cost_effect }
		}
		hidden_effect = {
			add_demon_arts_lifestyle_perk_points = -9
		}
	}
}


demon_arts_iron_mountain_perk = {
	lifestyle = demon_arts_lifestyle
	tree = iron_mountain
	position = { 0 3 }
	icon = year_of_lotus

	can_be_picked = {
		demon_arts_iron_mountain_modifier_stacks < 9
	}
	
	effect = {
		POD_add_modifier_and_track_stacks = {
			TYPE = character
			MODIFIER = demon_arts_iron_mountain_modifier
		}
		custom_description_no_bullet = { text = demon_arts_iron_mountain_perk_effect }
		custom_description_no_bullet = { text = demon_arts_stacking_effect }
		hidden_effect = {
			if = { limit = { demon_arts_iron_mountain_modifier_stacks < 9 } remove_perk = demon_arts_iron_mountain_perk }
		}
	}
}
