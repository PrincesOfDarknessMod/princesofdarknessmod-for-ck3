summon_wind_perk = {
	lifestyle = lore_lifestyle
	tree = winds
	position = { 0 0 }
	icon = node_demon_arts

	can_be_picked = {
		dynasty = dynasty:14000001
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = lore_house_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
		epidemic_resistance = 2
		epidemic_travel_danger = -5
		character_travel_speed_mult = 0.1
		naval_movement_speed_mult = 0.1
		movement_speed = 0.05
	}
	
	effect = {
		if = {
			limit = { not = { dynasty = dynasty:14000001 } }
			custom_description_no_bullet = { text = NOT_has_asharu_dynasty }
		}
		custom_description_no_bullet = { 
			text = summon_wind_perk_effect
		}
	}
}
