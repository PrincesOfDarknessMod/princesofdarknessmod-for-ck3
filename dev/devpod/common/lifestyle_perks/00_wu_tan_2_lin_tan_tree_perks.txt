read_wood_soul_perk = {
	lifestyle = wu_tan_lifestyle
	tree = lin_tan
	position = { 0 0 }
	icon = node_choleric
	
	can_be_picked = { has_trait = lin_tan }

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = wu_tan_lin_tan_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = commune_with_wood_perk }
				NOT = { has_perk = lin_tan_speak_with_jimu_tenno_perk }
			}
			multiply = 0
		}
	}

	character_modifier = {
	}
	
	effect = {
		if = {
			limit = { NOT = { has_trait = lin_tan } }
			custom_description_no_bullet = { 
				text = NEEDS_DISCIPLINE_TEXT
			}
		}
		custom_description_no_bullet = { 
			text = read_wood_soul_perk_effect
		}
	}
}
