kismet_perk = {
	lifestyle = arcanoi_lifestyle
	tree = fatalism
	position = { 0 0 }
	icon = node_soul_arts

	can_be_picked = {
		has_trait = wraith
		has_trait = arcanos_fatalism
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = arcanos_fatalism_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = distant_visions_perk }
			}
			multiply = 0
		}
	}

	character_modifier = {
		hostile_scheme_phase_duration_add = -10
	}
	
	effect = {
		if = {
			limit = {
				NOT = { has_trait = arcanos_fatalism }
			}
			custom_description_no_bullet = { 
				text = NOT_has_arcanos_fatalism_trait
			}
		}
		custom_description_no_bullet = { 
			text = kismet_perk_effect
		}
	}
}
