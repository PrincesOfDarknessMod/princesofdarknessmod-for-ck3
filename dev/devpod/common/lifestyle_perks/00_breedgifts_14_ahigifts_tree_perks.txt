bones_as_coils_perk = {
	lifestyle = breedgifts_lifestyle
	tree = ahigifts
	position = { 1 0 }
	icon = node_animal
	
	can_be_picked = {
		has_trait = metis
		trigger_if = {
			limit = { has_trait = abomination }
			has_character_flag = nagah_shape
		}
		trigger_else = {
			has_trait = nagah
		}
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_trait = nagah
				has_trait = metis
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = breedgifts_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
		hostile_scheme_phase_duration_add = -10
	}
	
	effect = {
		if = {
			limit = {
				NOT = {
					has_trait = metis
				}
			}
			custom_description_no_bullet = { 
				text = NEEDS_METIS_TEXT
			}
		}
	}
}
