	lifestyle = choleric_lifestyle
	tree = potence
	position = { 0 0 }
	icon = node_choleric
	
	can_be_picked = {
		has_potence_trigger = yes
	}		
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_choleric_interest_trigger = yes
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = choleric_potence_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = soaring_leap_perk }	
			}
			multiply = 0
		}
	}

	character_modifier = {
	}
	
	effect = {
		custom_description_no_bullet = {
			text = lethal_body_perk_stats_effect
		}
		if = {
			limit = {
				has_potence_trigger = no
			}
			custom_description_no_bullet = { 
				text = NEEDS_DISCIPLINE_TEXT
			}
		}
	}
}
