

via_geniorum_perk = {
	lifestyle = sorcery_lifestyle
	tree = theurgy
	position = { 0 0 }
	icon = trait_theurgist
	
	can_be_picked = {
		has_trait = theurgist
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = theurgy_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	effect = {
		if = {
			limit = {
				NOT = { has_trait = theurgist }
			}
			custom_description_no_bullet = { 
				text = NOT_has_theurgist_trait
			}
		}
		custom_description_no_bullet = {
			text = via_geniorum_perk_effect
		}
	}
}
