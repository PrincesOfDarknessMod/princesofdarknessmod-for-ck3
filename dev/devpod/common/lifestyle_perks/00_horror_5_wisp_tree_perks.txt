unearthly_repose_perk = {
	lifestyle = horror_lifestyle
	tree = wisp
	position = { 0 0 }
	icon = node_soul_arts

	can_be_picked = {
		has_trait = lament
		NOT = { has_focus = horror_poltergeist_focus }
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = horror_wisp_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}
	
	character_modifier = {
		happy_powerful_vassal_levy_contribution_mult = 0.15
	}
	
	effect = {
		if = {
			limit = { has_focus = horror_poltergeist_focus }
			custom_description_no_bullet = { text = has_horror_poltergeist_focus }
		}
		custom_description_no_bullet = { 
			text = unearthly_repose_perk_effect
		}
	}
}
