read_emotion_perk = {
	lifestyle = lore_lifestyle
	tree = longing
	position = { 0 0 }
	icon = node_demon_arts

	can_be_picked = {
		dynasty = dynasty:14000005
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = lore_house_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
		owned_personal_scheme_success_chance_add = 15
		owned_hostile_scheme_success_chance_add = 15
		enemy_hostile_scheme_success_chance_add = -15
	}
	
	effect = {
		if = {
			limit = { not = { dynasty = dynasty:14000005 } }
			custom_description_no_bullet = { text = NOT_has_lammasu_dynasty }
		}
		custom_description_no_bullet = { 
			text = read_emotion_perk_effect
		}
	}
}
