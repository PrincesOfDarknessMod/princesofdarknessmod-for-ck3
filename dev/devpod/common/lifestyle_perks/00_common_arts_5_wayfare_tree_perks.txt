hopscotch_perk = {
	lifestyle = common_arts_lifestyle
	tree = wayfare
	position = { 0 0 }
	icon = node_stewardship
	
	can_be_picked = { has_trait = wayfare }

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = common_arts_wayfare_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
		no_water_crossing_penalty = yes
		embarkation_cost_mult = -0.1
	}
	
	effect = {
		if = {
			limit = { NOT = { has_trait = wayfare } }
			custom_description_no_bullet = { 
				text = NEEDS_ART_TEXT
			}
		}
		custom_description_no_bullet = { 
			text = hopscotch_perk_effect
		}
	}
}
