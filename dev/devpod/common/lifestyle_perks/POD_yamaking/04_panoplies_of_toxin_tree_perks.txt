		custom_description_no_bullet = { text = panoplies_perk_requirement }
		custom_description_no_bullet = { text = panoplies_earth_apprentice_perk_effect }

			if = { limit = { has_character_modifier = POD_panoplies_taming_the_earth_modifier }

			}
		}
	}
