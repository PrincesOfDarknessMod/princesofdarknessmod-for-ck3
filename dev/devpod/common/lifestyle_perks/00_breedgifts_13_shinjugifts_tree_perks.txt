flow_of_aura_perk = {
	lifestyle = breedgifts_lifestyle
	tree = shinjugifts
	position = { 1 0 }
	icon = node_animal
	
	can_be_picked = {
		has_trait = metis
		trigger_if = {
			limit = { has_trait = abomination }
			has_character_flag = kitsune_shape
		}
		trigger_else = {
			has_trait = kitsune
		}
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_trait = kitsune
				has_trait = metis
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = breedgifts_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
		defender_holding_advantage = 5
	}
	
	effect = {
		if = {
			limit = {
				NOT = {
					has_trait = metis
				}
			}
			custom_description_no_bullet = { 
				text = NEEDS_METIS_TEXT
			}
		}
	}
}
