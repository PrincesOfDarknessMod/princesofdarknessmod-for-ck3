wilderness_sense_perk = {
	lifestyle = lore_lifestyle
	tree = wild
	position = { 0 0 }
	icon = node_demon_arts

	can_be_picked = {
		dynasty = dynasty:14000002
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = lore_house_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
		lifestyle_herbalist_xp_gain_mult = 1
		lifestyle_gardener_xp_gain_mult = 1
		lifestyle_hunter_xp_gain_mult = 1
	}
	
	effect = {
		add_trait = lifestyle_herbalist
		add_trait = lifestyle_gardener
		add_trait = lifestyle_hunter
		if = {
			limit = { NOT = { dynasty = dynasty:14000002 } }
			custom_description_no_bullet = { text = NOT_has_rabisu_dynasty }
		}
		custom_description_no_bullet = { 
			text = wilderness_sense_perk_effect
		}
	}
}
