	lifestyle = lore_lifestyle
	tree = wild
	position = { 0 0 }
	icon = node_demon_arts

	can_be_picked = {
		dynasty = dynasty:14000002
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = lore_house_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
	}
	
	effect = {
		if = {

			custom_description_no_bullet = { text = NOT_has_rabisu_dynasty }
		}
		custom_description_no_bullet = { 
			text = wilderness_sense_perk_effect
		}
	}
}
