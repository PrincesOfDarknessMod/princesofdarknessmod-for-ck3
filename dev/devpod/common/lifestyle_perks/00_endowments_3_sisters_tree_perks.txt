sisters_gospel_perk = {
	lifestyle = endowments_lifestyle
	tree = sisters
	position = { 0 0 }
	icon = sisters_gospel
	
	can_be_picked = {
		has_trait = supehunter
		faith = faith:shadowinquisition
		NOT = { dynasty ?= dynasty:13000011 }
		has_mendicant_order_dynasty_trigger = yes
	}		
	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = endowments_all_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = sisters_lux_perk }
			}
			multiply = 0
		}
	}
	effect = {
		add_trait_force_tooltip = sisters_gospel
		if = {
			limit = { has_mendicant_order_dynasty_trigger = no }
			custom_description_no_bullet = { 
				text = sisters_only_text
			}
		}
	}
}

sisters_lux_perk = {
	lifestyle = endowments_lifestyle
	tree = sisters
	position = { 1.5 0 }
	icon = sisters_lux

	can_be_picked = {
		exists = dynasty
		has_trait = supehunter
		faith = faith:shadowinquisition
		NOT = { dynasty ?= dynasty:13000011 }
		OR = {
			dynasty = dynasty:13000003
			dynasty = { has_dynasty_modifier = custom_mendicant_order_modifier }
		}
	}		
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_focus = endowments_all_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = sisters_gospel_perk }
			}
			multiply = 0
		}
	}
	character_modifier = {
	}
	
	effect = {
		add_trait_force_tooltip = sisters_lux
		if = {
			limit = {
				NOR = {
					dynasty = dynasty:13000003
					AND = {
						exists = dynasty
						dynasty = { has_dynasty_modifier = custom_mendicant_order_modifier }
					}
				}
			}
			custom_description_no_bullet = { 
				text = sisters_only_text
			}
		}
	}
}
