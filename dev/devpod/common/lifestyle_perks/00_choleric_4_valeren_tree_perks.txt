obeah_panacea_perk = {
	lifestyle = choleric_lifestyle
	tree = valeren
	position = { 1 0 }
	icon = node_choleric
	
	can_be_picked = {
		has_valeren_trigger = yes
		trigger_if = {
			limit = { 
				has_character_flag = triplicate_healer_valeren_cost_flag
			}
			choleric_lifestyle_perk_points > 1
		}
		trigger_if = {
			limit = { 
				has_character_flag = duplicate_healer_valeren_cost_flag
			}
			OR = {
				choleric_lifestyle_xp > 999
				choleric_lifestyle_perk_points > 1
			}
		}
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_choleric_interest_trigger = yes
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = choleric_valeren_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}
	
	character_modifier = {
		epidemic_travel_danger = -2
		epidemic_resistance = 2
	}
	
	effect = {
		if = {
			limit = { 
				has_character_flag = triplicate_healer_valeren_cost_flag
			}
			custom_description_no_bullet = { text = has_2_extra_perk_points_valeren }
			hidden_effect = { add_choleric_lifestyle_perk_points = -1 }
		}
		else_if = {
			limit = { 
				has_character_flag = duplicate_healer_valeren_cost_flag
			}
			custom_description_no_bullet = { text = has_1_extra_perk_points_valeren }
			hidden_effect = { 
				if = {
					limit = { choleric_lifestyle_xp < 1000 }
					add_choleric_lifestyle_perk_points = -1
					add_choleric_lifestyle_xp = 1000 
				}
				else = {
					add_choleric_lifestyle_xp = -1000 
				}
			}
			if = {
				limit = { has_character_flag = duplicate_warrior_valeren_cost_flag }
				remove_character_flag = duplicate_warrior_valeren_cost_flag
				add_character_flag = triplicate_warrior_valeren_cost_flag
			}
			if = {
				limit = { has_character_flag = duplicate_watcher_valeren_cost_flag }
				remove_character_flag = duplicate_watcher_valeren_cost_flag
				add_character_flag = triplicate_watcher_valeren_cost_flag
			}
		}
		else = {
			add_character_flag = duplicate_warrior_valeren_cost_flag
			add_character_flag = duplicate_watcher_valeren_cost_flag
		}
		if = {
			limit = {
				has_valeren_trigger = no
			}
			custom_description_no_bullet = { 
				text = NEEDS_DISCIPLINE_TEXT
			}
		}
		custom_description_no_bullet = {
			text = obeah_panacea_perk_effect
		}
	}
}
