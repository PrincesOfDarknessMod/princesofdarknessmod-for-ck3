sweet_hunters_smile_perk = {
	lifestyle = breedgifts_lifestyle
	tree = homidgifts_bastet
	position = { 0 0 }
	icon = node_animal
	
	can_be_picked = {
		OR = {
			has_trait = kinfolk
			has_trait = homid_born
			has_perk = royal_privilege_perk
		}
		POD_has_access_to_fera_gifts = { TRAIT = trait:bastet }
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				OR = {
					has_trait = kinfolk
					has_trait = homid_born
				}
				POD_has_access_to_fera_gifts = { TRAIT = trait:bastet }
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = breedgifts_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = cat_claws_perk }	
			}
			multiply = 0
		}
	}

	character_modifier = {
		befriend_scheme_phase_duration_add = -13
		seduce_scheme_phase_duration_add = -13
	}
	
	effect = {
		if = {
			limit = {
				NOR = {
					has_perk = royal_privilege_perk
					has_trait = homid_born
				}
			}
			custom_description_no_bullet = { 
				text = NEEDS_HOMIDBORN_TEXT
			}
		}
		custom_description_no_bullet = {
			text = sweet_hunters_smile_perk_effect
		}		
	}
}
