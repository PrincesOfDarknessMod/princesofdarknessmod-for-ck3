fast_perk = {
	lifestyle = commongifts_lifestyle
	tree = rokeagifts
	position = { 0 0 }
	icon = node_animal
	
	can_be_picked = {
		OR = {
			POD_has_access_to_fera_gifts = { TRAIT = trait:rokea }
			
			has_trait = kitsune
		}
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_trait = rokea
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = commongifts_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = sense_threat_perk }
			}
			multiply = 0
		}
	}

	character_modifier = {
		supply_limit = 500
		supply_capacity_add = 500
	}
	
	effect = {
		POD_gain_innovation_xp_effect = yes
	}
}
