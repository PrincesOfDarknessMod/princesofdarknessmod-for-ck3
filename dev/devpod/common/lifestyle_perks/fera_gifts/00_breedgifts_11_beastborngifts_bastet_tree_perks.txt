kittens_cry_perk = {
	lifestyle = breedgifts_lifestyle
	tree = beastborngifts_bastet
	position = { 0 0 }
	icon = node_animal
	
	can_be_picked = {
		OR = {
			has_perk = royal_privilege_perk
			has_trait = beast_born
		}
		POD_has_access_to_fera_gifts = { TRAIT = trait:bastet }
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_trait = beast_born
				POD_has_access_to_fera_gifts = { TRAIT = trait:bastet }
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = breedgifts_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = mark_as_mine_perk }	
			}
			multiply = 0
		}
	}

	character_modifier = {
		hard_casualty_modifier = -0.1
	}
	
	effect = {
		if = {
			limit = {
				NOR = {
					has_perk = royal_privilege_perk
					has_trait = beast_born
				}
			}
			custom_description_no_bullet = { 
				text = NEEDS_BEASTBORN_TEXT
			}
		}
		custom_description_no_bullet = {
			text = kittens_cry_perk_effect
		}		
	}
}
