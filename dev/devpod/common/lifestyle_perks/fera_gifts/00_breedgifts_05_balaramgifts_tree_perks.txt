cold_blood_perk = {
	lifestyle = breedgifts_lifestyle
	tree = balaramgifts
	position = { 0 0 }
	icon = node_animal
	
	can_be_picked = {
		OR = {
			has_trait = kinfolk
			has_trait = homid_born
		}
		POD_has_access_to_fera_gifts = { TRAIT = trait:nagah }
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				OR = {
					has_trait = kinfolk
					has_trait = homid_born
				}
				POD_has_access_to_fera_gifts = { TRAIT = trait:nagah }
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = breedgifts_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = prehensile_body_perk }
			}
			multiply = 0
		}
	}

	character_modifier = {
		desert_advantage = 5
		desert_mountains_advantage = 5
	}
	
	effect = {
		if = {
			limit = {
				NOT = {
					has_trait = homid_born
				}
			}
			custom_description_no_bullet = { 
				text = NEEDS_HOMIDBORN_TEXT
			}
		}
	}
}
