primal_song_perk = {
	lifestyle = lunar_auspicegifts_lifestyle
	tree = lunar_auspicegifts_galliard
	position = { 0 0 }
	icon = node_animal
	
	can_be_picked = {
		has_trait = galliard
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_trait = galliard
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = lunar_auspicegifts_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = howls_night_perk }
			}
			multiply = 0
		}
	}

	character_modifier = {
	}
	
	effect = {
		if = {
			limit = {
				NOT = { has_trait = galliard }
			}
			custom_description_no_bullet = { 
				text = NEEDS_GALLIARD_TEXT
			}
		}
		custom_description_no_bullet = {
			text = primal_song_perk_stats_effect
		}			
	}
}
