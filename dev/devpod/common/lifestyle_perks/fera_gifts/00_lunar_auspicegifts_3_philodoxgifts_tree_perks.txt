truth_gaia_perk = {
	lifestyle = lunar_auspicegifts_lifestyle
	tree = lunar_auspicegifts_philodox
	position = { 0 0 }
	icon = node_animal
	
	can_be_picked = {
		OR = {
			AND = {
				has_trait = philodox
				NOT = { has_trait = gurahl }
				NOT = { has_trait = ratkin }
			}
			has_character_modifier = sentai_role_lantern_modifier
		}
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				OR = {
					AND = {
						has_trait = philodox
						NOT = { has_trait = gurahl }
						NOT = { has_trait = ratkin }
					}
					has_character_modifier = sentai_role_lantern_modifier
				}
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = lunar_auspicegifts_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = aura_leadership_perk }
			}
			multiply = 0
		}
	}

	character_modifier = {
	}
	
	effect = {
		if = {
			limit = {
				NOR = {
					AND = {
						has_trait = philodox
						NOT = { has_trait = gurahl }
						NOT = { has_trait = ratkin }
					}
					has_character_modifier = sentai_role_lantern_modifier
				}
			}
			custom_description_no_bullet = { 
				text = NEEDS_PHILODOX_TEXT
			}
		}
