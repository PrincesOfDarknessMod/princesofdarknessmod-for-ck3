find_silver_path_perk = {
	lifestyle = kith_arts_lifestyle
	tree = dream_craft
	position = { 0 0 }
	icon = node_choleric
	
	can_be_picked = { has_trait = dream_craft }

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = kith_arts_dream_craft_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
		build_gold_cost = -0.1
		build_piety_cost = -0.1
		build_prestige_cost = -0.1
		POD_terrain_umbra_aptitude_advantage = 0.5
		POD_terrain_umbra_sense_advantage = 0.05
	}
	
	effect = {
		if = {
			limit = { NOT = { has_trait = dream_craft } }
			custom_description_no_bullet = { 
				text = NEEDS_ART_TEXT
			}
		}
		custom_description_no_bullet = { 
			text = find_silver_path_perk_effect
		}
	}
}
