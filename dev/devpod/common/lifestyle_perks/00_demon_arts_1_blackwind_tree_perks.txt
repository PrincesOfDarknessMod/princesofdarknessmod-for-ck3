	lifestyle = demon_arts_lifestyle
	tree = blackwind
	position = { 0 0 }
	icon = node_demon_arts
	
	can_be_picked = {
		OR = {
			has_trait = kueijin
			has_trait = dhampyr
		}
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_trait = dharma
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = demon_arts_blackwind_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				has_demon_arts_focus_trigger = no
			}
			multiply = 0
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}
	
	character_modifier = {
		movement_speed = 0.05
		stress_gain_mult = 0.05
	}
	
	effect = {
		custom_description_no_bullet = {
			text = black_wind_perk_effect
		}
	}
}
