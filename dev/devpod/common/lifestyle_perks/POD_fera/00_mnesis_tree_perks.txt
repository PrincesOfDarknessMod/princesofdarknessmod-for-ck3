	parent = age_of_man_perk
	
	can_be_picked = {
		POD_has_access_to_fera_gifts = { TRAIT = trait:mokole }
		POD_mokole_has_all_man_perks = yes
		mnesis_lifestyle_perk_points >= 2
	}
	
	character_modifier = {
		development_decline = -0.1
		POD_terrain_mokole_archid_option_advantage = 1
		POD_terrain_mnesis_journey_speed_advantage = -0.25
	}

	effect = {
		custom_description_no_bullet = { text = deeper_mnesis_effect }
		custom_description_no_bullet = { text = age_of_beasts_perk_effect }
		if = {
			limit = { POD_mokole_has_all_man_perks = no }
			custom_description_no_bullet = { text = mnesis_previous_age_req }
		}
		if = {
			limit = { is_ai = no }
			trigger_event = POD_mokole.100
		}
		else = { POD_mokole_assign_archid_options_effect = yes }
		POD_mokole_mnesis_perk_cost_effect = { COUNT = 2 }
	}
}
