bellow_perk = {
	lifestyle = solar_auspicegifts_lifestyle
	tree = solar_auspicegifts_rising_sun
	position = { 0 0 }
	icon = node_animal
	
	can_be_picked = {
		POD_has_access_to_fera_gifts = { TRAIT = trait:mokole }
		OR = {
			has_trait = rising_sun
			has_trait = vasanta
		}
	}
	
	auto_selection_weight = {
		value = 10
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOR = { 
					has_perk = master_of_fire_perk
					has_perk = eye_of_the_raptor_perk
				}
			}
			multiply = 0
		}
	}

	character_modifier = {
		prowess_no_portrait = 2
	}

	effect = { 
		custom_description_no_bullet = { text = bellow_perk_effect } 
		if = {
			limit = {
				NOR = {
					has_trait = rising_sun
					has_trait = vasanta
				}
			}
			custom_description_no_bullet = { text = rising_sun_perk_req }
		}
	}
}
