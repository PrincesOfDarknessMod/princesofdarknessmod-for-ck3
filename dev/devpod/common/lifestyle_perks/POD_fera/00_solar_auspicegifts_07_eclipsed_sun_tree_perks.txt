mokole_aura_of_confidence_perk = {
	lifestyle = solar_auspicegifts_lifestyle
	tree = solar_auspicegifts_eclipsed_sun
	position = { 1 0 }
	icon = node_animal
	
	can_be_picked = {
		POD_has_access_to_fera_gifts = { TRAIT = trait:mokole }
		OR = {
			has_trait = eclipsed_sun
			has_trait = sai_chau
		}
	}

	character_modifier = {
		enemy_hostile_scheme_phase_duration_add = 10
		knight_effectiveness_mult = 0.1
	}

	effect = { 
		custom_description_no_bullet = { text = mokole_aura_of_confidence_perk_effect } 
		if = {
			limit = {
				NOR = {
					has_trait = eclipsed_sun
					has_trait = sai_chau
				}
			}
			custom_description_no_bullet = { text = eclipsed_sun_perk_req }
		}
	}
}
