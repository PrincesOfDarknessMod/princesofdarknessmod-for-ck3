mothers_blessing_curse_perk = {
	lifestyle = tribalgifts_lifestyle
	tree = tribalgifts_bastet_celican
	position = { 0 0 }
	icon = node_animal
	
	can_be_picked = {
		OR = {
			has_perk = royal_privilege_perk
			faith = faith:hellcats
			faith = faith:ceilican
		}
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				OR = {
				has_faith = faith:hellcats
				has_faith = faith:ceilican
				}
			}
			add = 1989
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = satyrs_wisdom_perk }
			}
			multiply = 0
		}
	}

	character_modifier = {
	}
	
	effect = {
		custom_description_no_bullet = {
			text = mothers_blessing_curse_perk_effect
		}
	}
}
