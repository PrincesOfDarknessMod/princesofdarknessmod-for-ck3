bonds_that_tie_us_perk = {
	lifestyle = lunar_auspicegifts_lifestyle
	tree = lunar_auspicegifts_philodox
	position = { 2 2 }
	icon = node_animal

	parent = aura_of_the_just_perk
	parent = building_a_legend_perk
	parent = truth_of_gaia_perk

	character_modifier = {
		POD_terrain_pack_training_journey_speed_supply_limit_mult = 0.1
		vassal_opinion = 5
	}
	
	effect = {
		custom_description_no_bullet = { text = bonds_that_tie_us_perk_effect }
		if = {
			limit = { POD_has_coterie_trigger = yes }
			var:coterie ?= {
				every_in_list = {
					variable = members
					save_scope_as = pack_member
					every_relation = {
						type = rival
						limit = { POD_is_in_same_coterie_trigger = { TARGET = scope:pack_member } }
						remove_relation_rival = scope:pack_member
					}
				}
			}
		}
	}
}

break_the_bonds_perk = {
	lifestyle = lunar_auspicegifts_lifestyle
	tree = lunar_auspicegifts_philodox
	position = { 2 5 }
	icon = philodox
	
	can_be_picked = { always = no }

	character_modifier = {
		negate_diplomacy_penalty_add = 2
		negate_martial_penalty_add = 2
		negate_stewardship_penalty_add = 2
		negate_intrigue_penalty_add = 2
		negate_learning_penalty_add = 2
		negate_prowess_penalty_add = 2
	}
	
	effect = {
		custom_description_no_bullet = { text = break_the_bonds_perk_effect }

		#Intensive effect but should only be needed this once
		every_living_character = {
			limit = { has_hook = prev }
			remove_hook = { target = prev }
		}

		if = {
			limit = { NOT = { has_perk = break_the_bonds_perk } }
			custom_description_no_bullet = { text = auspice_endperk_requirement }
		}
	}
}
