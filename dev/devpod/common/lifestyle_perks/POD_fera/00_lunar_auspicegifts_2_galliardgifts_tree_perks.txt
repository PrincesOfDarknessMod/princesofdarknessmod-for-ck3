primal_song_perk = {
	lifestyle = lunar_auspicegifts_lifestyle
	tree = lunar_auspicegifts_galliard
	position = { 2 0 }
	icon = node_animal
	
	can_be_picked = {
		OR = {
			has_trait = galliard
			faith = { has_doctrine_parameter = can_learn_galliard_gifts }
		}
	}

	character_modifier = {
		POD_terrain_gnosis_regen_supply_limit_mult = 0.05
	}
	
	effect = {
		custom_description_no_bullet = { text = primal_song_perk_effect }
		if = {
			limit = {
				NOR = { 
					has_trait = galliard
					faith = { has_doctrine_parameter = can_learn_galliard_gifts }
				}
			}
			custom_description_no_bullet = { text = NEEDS_GALLIARD_TEXT }
		}
	}
}
