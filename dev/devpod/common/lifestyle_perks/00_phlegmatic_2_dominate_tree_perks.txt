cloud_memory_perk = {
	lifestyle = phlegmatic_lifestyle
	tree = dominate
	position = { 0.5 0 }
	icon = node_phlegmatic

	can_be_picked = {
		has_dominate_trigger = yes
	}	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_phlegmatic_interest_trigger = yes
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = phlegmatic_dominate_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = compel_perk }	
			}
			multiply = 0
		}
	}

	character_modifier = {
		#intrigue = 1 #reduce masquerade exposure, done by sparc
	}
	
	effect = {
		custom_description_no_bullet = { 
			text = CLOUD_MEMORY_PERK
		}
		if = {
			limit = {
				has_dominate_trigger = no
			}
			custom_description_no_bullet = { 
				text = NEEDS_DISCIPLINE_TEXT
			}
		}
	}
}
