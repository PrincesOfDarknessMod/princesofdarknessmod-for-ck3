# Skin of the Chameleon, like a weakened form of Protean's Mist Form

skinofchameleon_perk = {
	lifestyle = melancholy_lifestyle
	tree = visceratika
	position = { 1 0 }
	icon = node_melancholy
	
	can_be_picked = {
		has_visceratika_trigger = yes
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_melancholy_interest_trigger = yes
			}
			add = 1989
		}
		# if = {
			# limit = {
				# has_focus = melancholy_visceratika_focus
			# }
			# multiply = 5
		# }
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
		diablerize_enemy_scheme_phase_duration_add = 5
		murder_enemy_scheme_phase_duration_add = 5
	}
	
	effect = {
		custom_description_no_bullet = {
			text = skinofchameleon_perk_effect
		}
		if = {
			limit = {
				has_visceratika_trigger = no
			}
			custom_description_no_bullet = { 
				text = NEEDS_DISCIPLINE_TEXT
			}
		}
	}
}
