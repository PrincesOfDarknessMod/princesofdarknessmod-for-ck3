	lifestyle = phlegmatic_lifestyle
	tree = mytherceria
	position = { 0 0 }
	icon = node_phlegmatic

	can_be_picked = {
		POD_can_obtain_discipline_of_level_trigger = { DISCIPLINE = mytherceria LEVEL = 1 }
		POD_can_afford_discipline_perk_trigger = { LEVEL = 1 DISCIPLINE = mytherceria }
	}

	auto_selection_weight = {
		value = 10
		if = {
			limit = { has_phlegmatic_mytherceria_interest_trigger = yes }
			multiply = vampire_ai_interest_lifestyle_weight
		}
		if = {
			limit = {
				POD_vampire_ai_can_start_new_phlegmatic = no
				NOT = { has_perk = folderol_perk }
			}
			multiply = 0
		}
	}

	character_modifier = {
	}

	effect = {
		POD_deduct_extra_perk_cost_effect = { LEVEL = 1 DISCIPLINE = mytherceria }
		custom_description_no_bullet = {
			text = fey_sight_perk_effect
		}
		POD_can_obtain_discipline_of_level_effect = { DISCIPLINE = mytherceria LEVEL = 1 }
	}
}
