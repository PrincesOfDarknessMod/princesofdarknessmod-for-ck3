incubus_passion_perk = {
	lifestyle = phlegmatic_lifestyle
	tree = dementation
	position = { 0 0 }
	icon = node_phlegmatic

	can_be_picked = {
		POD_can_obtain_discipline_of_level_trigger = { DISCIPLINE = dementation LEVEL = 1 }
	}
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_phlegmatic_interest_trigger = yes
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = phlegmatic_dementation_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = passion_perk }
			}
			multiply = 0
		}
	}

	effect = {
		custom_description_no_bullet = {
			text = incubus_passion_perk_effect
		}
		POD_can_obtain_discipline_of_level_effect = { DISCIPLINE = dementation LEVEL = 1 }
	}
}
