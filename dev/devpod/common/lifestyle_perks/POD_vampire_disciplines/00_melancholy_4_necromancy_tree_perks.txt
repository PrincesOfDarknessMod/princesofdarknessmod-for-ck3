one_with_dead_1_perk = {
	lifestyle = melancholy_lifestyle
	tree = melancholy_necromancy
	position = { 0 0 }
	icon = node_melancholy

	can_be_picked = {
		POD_can_obtain_discipline_of_level_trigger = { DISCIPLINE = necromancy LEVEL = 1 }
	}

	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_melancholy_interest_trigger = yes
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = melancholy_necromancy_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = death_mastery_1_perk }
			}
			multiply = 0
		}
	}

	effect = {
		custom_description_no_bullet = {
			text = one_with_dead_1_perk_effect
		}
		POD_can_obtain_discipline_of_level_effect = { DISCIPLINE = necromancy LEVEL = 1 }
	}
}
