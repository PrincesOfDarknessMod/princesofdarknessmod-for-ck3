one_with_dead_1_perk = {
	lifestyle = melancholy_lifestyle
	tree = melancholy_necromancy
	position = { 0 0 }
	icon = node_melancholy

	can_be_picked = {
		POD_can_obtain_discipline_of_level_trigger = { DISCIPLINE = necromancy LEVEL = 1 }
		POD_can_afford_discipline_perk_trigger = { LEVEL = 1 DISCIPLINE = necromancy }
	}

	auto_selection_weight = {
		value = 10
		if = {
			limit = { has_melancholy_necromancy_interest_trigger = yes }
			multiply = vampire_ai_interest_lifestyle_weight
		}
		if = {
			limit = {
				POD_vampire_ai_can_start_new_melancholy = no
				NOT = { has_perk = death_mastery_1_perk }
			}
			multiply = 0
		}
	}

	effect = {
		POD_deduct_extra_perk_cost_effect = { LEVEL = 1 DISCIPLINE = necromancy }
		custom_description_no_bullet = {
			text = one_with_dead_1_perk_effect
		}
		POD_can_obtain_discipline_of_level_effect = { DISCIPLINE = necromancy LEVEL = 1 }
	}
}

	lifestyle = melancholy_lifestyle
	tree = melancholy_necromancy
	position = { 1 4 }
	icon = path_common

	can_be_picked = {
		POD_can_obtain_discipline_of_level_trigger = { DISCIPLINE = necromancy LEVEL = 6 }
		POD_can_afford_discipline_perk_trigger = { LEVEL = 6 DISCIPLINE = necromancy }
	}

	parent = one_with_dead_4_perk
	parent = death_mastery_4_perk

	effect = {
		POD_deduct_extra_perk_cost_effect = { LEVEL = 6 DISCIPLINE = necromancy }
		remove_trait = necromancy
		add_trait_force_tooltip = necromancyadvanced
		POD_can_obtain_discipline_of_level_effect = { DISCIPLINE = necromancy LEVEL = 6 }
		if = {
			limit = { is_ai = yes }
			set_focus = no
		}
	}

	character_modifier = {
		POD_wraith_maa_recruitment_cost_mult = -0.1
		POD_wraith_maa_maintenance_mult = -0.1
	}
}
