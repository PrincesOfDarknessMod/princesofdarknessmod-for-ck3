speak_with_spirits_perk = {
	lifestyle = animal_lifestyle
	tree = animal_spiritus
	position = { 0 0 }
	icon = node_animal

	can_be_picked = {
		POD_can_obtain_discipline_of_level_trigger = { DISCIPLINE = spiritus LEVEL = 1 }
	}

	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_animal_interest_trigger = yes
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = animal_spiritus_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = aid_from_spirits_perk }
			}
			multiply = 0
		}
	}

	effect = {
		custom_description_no_bullet = {
			text = speak_with_spirits_perk_effect
		}
		POD_can_obtain_discipline_of_level_effect = { DISCIPLINE = spiritus LEVEL = 1 }
	}
}
