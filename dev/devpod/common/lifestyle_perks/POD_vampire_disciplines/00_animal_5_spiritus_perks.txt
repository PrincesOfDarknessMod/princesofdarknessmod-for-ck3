speak_with_spirits_perk = {
	lifestyle = animal_lifestyle
	tree = spiritus
	position = { 0 0 }
	icon = node_animal

	can_be_picked = {
		POD_can_obtain_discipline_of_level_trigger = { DISCIPLINE = spiritus LEVEL = 1 }
		POD_can_afford_discipline_perk_trigger = { LEVEL = 1 DISCIPLINE = spiritus }
	}

	auto_selection_weight = {
		value = 10
		if = {
			limit = { has_animal_spiritus_interest_trigger = yes }
			multiply = vampire_ai_interest_lifestyle_weight
		}
		if = {
			limit = {
				POD_vampire_ai_can_start_new_animalism = no
				NOT = { has_perk = aid_from_spirits_perk }
			}
			multiply = 0
		}
	}

	effect = {
		POD_deduct_extra_perk_cost_effect = { LEVEL = 1 DISCIPLINE = spiritus }
		custom_description_no_bullet = {
			text = speak_with_spirits_perk_effect
		}
		POD_can_obtain_discipline_of_level_effect = { DISCIPLINE = spiritus LEVEL = 1 }
	}
}
