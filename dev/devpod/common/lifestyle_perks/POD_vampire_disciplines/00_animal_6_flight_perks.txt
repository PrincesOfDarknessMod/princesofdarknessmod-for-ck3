flight_1_perk = {
	lifestyle = animal_lifestyle
	tree = animal_flight
	position = { 1.5 0 }
	icon = node_animal

	can_be_picked = {
		POD_can_obtain_discipline_of_level_trigger = { DISCIPLINE = flight LEVEL = 1 }
		POD_can_afford_discipline_perk_trigger = { LEVEL = 1 DISCIPLINE = flight }
	}

	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_animal_interest_trigger = yes
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = animal_flight_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
		max_combat_roll = 2 #Simulating potential of ambush on the unprepared from the sky batman style
	}

	effect = {
		POD_deduct_extra_perk_cost_effect = { LEVEL = 1 DISCIPLINE = flight }
		custom_description_no_bullet = {
			text = flight_1_perk_effect
		}
		POD_can_obtain_discipline_of_level_effect = { DISCIPLINE = flight LEVEL = 1 }
	}
}
