awe_perk = {
	lifestyle = sanguine_lifestyle
	tree = presence
	position = { 0 0 }
	icon = node_sanguine

	can_be_picked = {
		POD_can_obtain_discipline_of_level_trigger = { DISCIPLINE = presence LEVEL = 1 }
	}
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_sanguine_interest_trigger = yes
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = sanguine_presence_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = daunt_perk }
			}
			multiply = 0
		}
	}
	### gives access to Befriend scheme
	character_modifier = {
		personal_scheme_phase_duration_add = -5
		contract_scheme_phase_duration_add = -5
		direct_vassal_opinion = 5
		legitimacy_gain_mult = 0.1
	}

	effect = {
		custom_description_no_bullet = {
			text = awe_perk_stats_effect
		}
		POD_can_obtain_discipline_of_level_effect = { DISCIPLINE = presence LEVEL = 1 }
	}
}
