arcane_knowledge_1_perk = {
	lifestyle = sanguine_lifestyle
	tree = bloodsorcery
	position = { 0 0 }
	icon = node_sanguine

	can_be_picked = {
		POD_can_obtain_discipline_of_level_trigger = { DISCIPLINE = bloodsorcery LEVEL = 1 }
		POD_can_afford_discipline_perk_trigger = { LEVEL = 1 DISCIPLINE = bloodsorcery }
	}

	auto_selection_weight = {
		value = 10
		if = {
			limit = { has_sanguine_bloodsorcery_interest_trigger = yes }
			multiply = vampire_ai_interest_lifestyle_weight
		}
		if = {
			limit = { POD_vampire_ai_can_start_new_sanguine = no }
			multiply = 0
		}
	}

	effect = {
		POD_deduct_extra_perk_cost_effect = { LEVEL = 1 DISCIPLINE = bloodsorcery }
		custom_description_no_bullet = {
			text = arcane_knowledge_1_perk_effect
		}
		POD_can_obtain_discipline_of_level_effect = { DISCIPLINE = bloodsorcery LEVEL = 1 }
	}
}
