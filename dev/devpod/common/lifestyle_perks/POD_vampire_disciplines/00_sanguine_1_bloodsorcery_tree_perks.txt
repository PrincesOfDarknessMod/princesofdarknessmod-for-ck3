arcane_knowledge_1_perk = {
	lifestyle = sanguine_lifestyle
	tree = bloodsorcery
	position = { 0 0 }
	icon = node_sanguine

	can_be_picked = {
		POD_can_obtain_discipline_of_level_trigger = { DISCIPLINE = bloodsorcery LEVEL = 1 }
	}

	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_sanguine_interest_trigger = yes
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = sanguine_bloodsorcery_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = arcane_mastery_1_perk }
			}
			multiply = 0
		}
	}

	effect = {
		custom_description_no_bullet = {
			text = arcane_knowledge_1_perk_effect
		}
		POD_can_obtain_discipline_of_level_effect = { DISCIPLINE = bloodsorcery LEVEL = 1 }
	}
}
