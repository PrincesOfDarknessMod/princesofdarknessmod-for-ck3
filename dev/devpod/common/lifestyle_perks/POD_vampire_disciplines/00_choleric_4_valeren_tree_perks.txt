obeah_panacea_perk = {
	lifestyle = choleric_lifestyle
	tree = valeren
	position = { 1 0 }
	icon = node_choleric

	can_be_picked = {
		POD_can_obtain_discipline_of_level_trigger = { DISCIPLINE = valeren LEVEL = 1 }
		POD_can_afford_valeren_perk_trigger = {
			LEVEL = 1
			VAR   = additional_healer_valeren_cost
		}
	}

	auto_selection_weight = {
		value = 10
		if = {
			limit = { has_choleric_valeren_interest_trigger = yes }
			multiply = vampire_ai_interest_lifestyle_weight
		}
		if = {
			limit = { POD_vampire_ai_can_start_new_choleric = no }
			multiply = 0
		}
	}

	character_modifier = {
		epidemic_travel_danger = -2
		epidemic_resistance = 2
	}

	effect = {
		POD_deduct_valeren_perk_cost_effect = {
			LEVEL = 1
			VAR   = additional_healer_valeren_cost
		}
		# if this is the first tree you pick, increase the cost of the others
		if = {
			limit = { NOT = { has_variable = additional_healer_valeren_cost } }
			set_variable = { name = additional_warrior_valeren_cost   value = 1 }
			set_variable = { name = additional_watcher_valeren_cost   value = 1 }
		}
		# if this is the second tree you pick, increase the cost of the one you haven't picked yet
		else_if = {
			limit = { var:additional_healer_valeren_cost = 1 }
			if = {
				limit = { has_variable = additional_warrior_valeren_cost }
				change_variable = { name = additional_warrior_valeren_cost   add = 1 }
			}
			if = {
				limit = { has_variable = additional_watcher_valeren_cost }
				change_variable = { name = additional_watcher_valeren_cost   add = 1 }
			}
		}
		custom_description_no_bullet = {
			text = obeah_panacea_perk_effect
		}
		POD_can_obtain_discipline_of_level_effect = { DISCIPLINE = valeren LEVEL = 1 }
	}
}
