resilence_perk = {
	lifestyle = melancholy_lifestyle
	tree = fortitude
	position = { 0 0 }
	icon = node_melancholy

	can_be_picked = {
		POD_can_obtain_discipline_of_level_trigger = { DISCIPLINE = fortitude LEVEL = 1 }
		pod_can_afford_discipline_perk_trigger = { LEVEL = 1 DISCIPLINE = fortitude LIFESTYLE = melancholy }
	}

	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_melancholy_interest_trigger = yes
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = melancholy_fortitude_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = unswayable_mind_perk }
			}
			multiply = 0
		}
	}

	character_modifier = {
		prowess_no_portrait = 5
		diablerize_enemy_scheme_phase_duration_add = 5
		murder_enemy_scheme_phase_duration_add = 5
	}

	effect = {
		pod_deduct_extra_perk_cost_effect = { LEVEL = 1 DISCIPLINE = fortitude LIFESTYLE = melancholy }
		custom_description_no_bullet = {
			text = resilience_perk_stats_effect
		}
		POD_can_obtain_discipline_of_level_effect = { DISCIPLINE = fortitude LEVEL = 1 }
	}
}
