cats_grace_perk = {
	lifestyle = choleric_lifestyle
	tree = celerity
	position = { 0 0 }
	icon = node_choleric

	can_be_picked = {
		POD_can_obtain_discipline_of_level_trigger = { DISCIPLINE = celerity LEVEL = 1 }
		POD_can_afford_discipline_perk_trigger = { LEVEL = 1 DISCIPLINE = celerity }
	}
	auto_selection_weight = {
		value = 10
		if = {
			limit = { has_choleric_celerity_interest_trigger = yes }
			multiply = vampire_ai_interest_lifestyle_weight
		}
		if = {
			limit = {
				POD_vampire_ai_can_start_new_choleric = no
				NOT = { has_perk = rapid_reflexes_perk }
			}
			multiply = 0
		}
	}

	character_modifier = {
		murder_enemy_scheme_phase_duration_add = 5
		abduct_enemy_scheme_phase_duration_add = 5
		diablerize_enemy_scheme_phase_duration_add = 5
	}

	effect = {
		POD_deduct_extra_perk_cost_effect = { LEVEL = 1 DISCIPLINE = celerity }
		custom_description_no_bullet = {
			text = cats_grace_perk_stats_effect
		}
		POD_can_obtain_discipline_of_level_effect = { DISCIPLINE = celerity LEVEL = 1 }
	}
}
