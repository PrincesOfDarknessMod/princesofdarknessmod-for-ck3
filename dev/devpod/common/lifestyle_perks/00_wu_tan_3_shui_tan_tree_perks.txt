read_water_soul_perk = {
	lifestyle = wu_tan_lifestyle
	tree = shui_tan
	position = { 0 0 }
	icon = node_choleric
	
	can_be_picked = { has_trait = shui_tan }

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = wu_tan_shui_tan_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = commune_with_water_perk }
				NOT = { has_perk = shui_tan_speak_with_jimu_tenno_perk }
			}
			multiply = 0
		}
	}

	character_modifier = {
	}
	
	effect = {
		if = {
			limit = { NOT = { has_trait = shui_tan } }
			custom_description_no_bullet = { 
				text = NEEDS_DISCIPLINE_TEXT
			}
		}
		custom_description_no_bullet = { 
			text = read_water_soul_perk_effect
		}
	}
}
