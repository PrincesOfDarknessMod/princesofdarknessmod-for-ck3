oculi_shepherd_perk = {
	lifestyle = endowments_lifestyle
	tree = oculi
	position = { 0 0 }
	icon = oculi_shepherd
	
	can_be_picked = {
		exists = dynasty
		OR = {
			dynasty = dynasty:13000005
			AND = {
				exists = dynasty
				dynasty = { has_dynasty_modifier = custom_lay_order_modifier }
			}
		}
		has_trait = supehunter
		faith = faith:shadowinquisition
	}		
	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = endowments_all_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = oculi_sight_perk }
			}
			multiply = 0
		}
	}
	effect = {
		add_trait_force_tooltip = oculi_shepherd
		if = {
			limit = {
				NOR = {
					dynasty = dynasty:13000005
					AND = {
						exists = dynasty
						dynasty = { has_dynasty_modifier = custom_lay_order_modifier }
					}
				}
			}
			custom_description_no_bullet = { 
				text = oculi_only_text
			}
		}
	}
}
