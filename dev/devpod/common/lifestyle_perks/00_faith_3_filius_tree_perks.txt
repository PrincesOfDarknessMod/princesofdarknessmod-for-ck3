sunder_perk = {
	lifestyle = faith_lifestyle
	tree = filius
	position = { 0 0 }
	icon = node_learning

	can_be_picked = {
		has_trait = supehunter
		OR = {
			faith = faith:shadowinquisition
			faith = { has_doctrine_parameter = access_to_holy_arts }
		}
		OR = {
			has_knightly_order_dynasty_trigger = yes
			POD_has_true_faith_trigger = yes
		}
	}

	character_modifier = {
	}

	effect = {
		custom_description_no_bullet = {
			text = sunder_perk_effect
		}
		if = {
			limit = {
				has_knightly_order_dynasty_trigger = no
				POD_has_true_faith_trigger = no
			}
			custom_description_no_bullet = { 
				text = FILIUS_NOT_PRIMARY_TEXT
			}
		}
		if = {
			limit = {
				NAND = {
					OR = {
						faith = { has_doctrine_parameter = access_to_holy_arts }
						faith = faith:shadowinquisition
					}
					has_trait = supehunter
				}
			}
			custom_description_no_bullet = { 
				text = SHADOW_INQUISITION_ONLY_TEXT
			}
		}
	}
}
