sunder_perk = {
	lifestyle = faith_lifestyle
	tree = filius
	position = { 0 0 }
	icon = node_learning

	can_be_picked = {
		has_trait = supehunter
		faith = faith:shadowinquisition
		OR = {
			AND = {
				exists = dynasty
				OR = {
					dynasty = dynasty:13000002
					dynasty = dynasty:13000006
					dynasty = dynasty:13000008
					dynasty = { has_dynasty_modifier = custom_knightly_order_modifier }
				}
			}
			has_trait = truefaith
			has_trait = beatific
		}
	}

	character_modifier = {
	}

	effect = {
		custom_description_no_bullet = {
			text = sunder_perk_effect
		}
		if = {
			limit = {
				NOR = {
					AND = {
						exists = dynasty
						OR = {
							dynasty = dynasty:13000002
							dynasty = dynasty:13000006
							dynasty = dynasty:13000008
							dynasty = { has_dynasty_modifier = custom_knightly_order_modifier }
						}
					}
					has_trait = truefaith
					has_trait = beatific
				}
			}
			custom_description_no_bullet = { 
				text = FILIUS_NOT_PRIMARY_TEXT
			}
		}
		if = {
			limit = {
				NAND = { 
					faith = faith:shadowinquisition
					has_trait = supehunter
				}
			}
			custom_description_no_bullet = { 
				text = SHADOW_INQUISITION_ONLY_TEXT
			}
		}
	}
}
