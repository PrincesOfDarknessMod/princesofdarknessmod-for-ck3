hearing_voices_perk = {
	lifestyle = arcanoi_lifestyle
	tree = keening
	position = { 1 0 }
	icon = node_soul_arts

	can_be_picked = {
		has_trait = wraith
		has_trait = arcanos_keening
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = arcanos_keening_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}
	
	character_modifier = {
		hostile_scheme_phase_duration_add = -5
	}
	
	effect = {
		if = {
			limit = {
				NOT = { has_trait = arcanos_keening }
			}
			custom_description_no_bullet = { 
				text = NOT_has_arcanos_keening_trait
			}
		}
		custom_description_no_bullet = { 
			text = hearing_voices_perk_effect
		}
	}
}
