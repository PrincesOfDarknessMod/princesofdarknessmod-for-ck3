the_gleaming_perk = {
	lifestyle = arcanoi_lifestyle
	tree = intimation
	position = { 0 0 }
	icon = node_soul_arts

	can_be_picked = {
		has_trait = wraith
		has_trait = arcanos_intimation
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = arcanos_intimation_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = arcanoi_know_thyself_perk }
			}
			multiply = 0
		}
	}
	
	effect = {
		if = {
			limit = {
				NOT = { has_trait = arcanos_intimation }
			}
			custom_description_no_bullet = { 
				text = NOT_has_arcanos_intimation_trait
			}
		}
		custom_description_no_bullet = { 
			text = the_gleaming_perk_effect
		}
	}
}
