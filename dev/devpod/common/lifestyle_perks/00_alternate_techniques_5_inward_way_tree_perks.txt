inner_balance_perk = {
	lifestyle = alternate_techniques_lifestyle
	tree = inward_way
	position = { 0 0 }
	icon = node_animal
	
	can_be_picked = {
		has_POD_kj_blood_trigger = yes
		faith = faith:inwardway
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_trait = dharma
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = alternate_techniques_inward_way_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				has_alternate_techniques_focus_trigger = no
			}
			multiply = 0
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}
	
	character_modifier = {
		negate_diplomacy_penalty_add = 2
		negate_martial_penalty_add = 2
		negate_stewardship_penalty_add = 2
		negate_intrigue_penalty_add = 2
		negate_learning_penalty_add = 2
	}
	
	effect = {
		if = {
			limit = { NOT = { faith = faith:inwardway } }
			custom_description_no_bullet = { text = NEEDS_BE_INWARDWAY_TEXT }
		}
		custom_description_no_bullet = {
			text = inner_balance_perk_effect
		}
	}
}
