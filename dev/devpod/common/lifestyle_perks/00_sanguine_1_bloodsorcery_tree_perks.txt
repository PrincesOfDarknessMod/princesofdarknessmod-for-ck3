corrosive_vitae_perk = {
	lifestyle = sanguine_lifestyle
	tree = bloodsorcery
	position = { 0 0 }
	icon = node_sanguine
	
	can_be_picked = {
		has_bloodsorcery_trigger = yes
	}	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_sanguine_interest_trigger = yes
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = sanguine_bloodsorcery_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = a_taste_of_blood_perk }				
			}
			multiply = 0
		}
	}

	character_modifier = {
		siege_phase_time = -0.15
	}
	
	effect = {
		custom_description_no_bullet = { 
			text = CORROSIVE_VITAE_PERK
		}
		if = {
			limit = {
				has_bloodsorcery_trigger = no
			}
			custom_description_no_bullet = { 
				text = NEEDS_DISCIPLINE_TEXT
			}
		}
	}
}
