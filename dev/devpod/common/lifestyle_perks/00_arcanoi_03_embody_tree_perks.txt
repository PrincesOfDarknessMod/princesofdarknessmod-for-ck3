	tree = embody

	icon = node_soul_arts

		has_trait = arcanos_embody
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = arcanos_embody_focus
			}
			multiply = 5
		}

			limit = {

	effect = {
		if = {

			}
		}

			text = whispers_perk_effect
		}
	}
}
