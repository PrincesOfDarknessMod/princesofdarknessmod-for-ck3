primal_anger_perk = {
	lifestyle = breedgifts_lifestyle
	tree = metisgifts
	position = { 0 0 }
	icon = node_animal
	
	can_be_picked = {
		has_trait = metis
		trigger_if = {
			limit = { has_trait = abomination }
			has_character_flag = werewolf_shape
		}
		trigger_else = {
			has_trait = werewolf
		}
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_trait = werewolf
				has_trait = metis
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = breedgifts_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = shed_perk }
			}
			multiply = 0
		}
	}

	character_modifier = {
		prowess_no_portrait = 1
	}
	
	effect = {
		if = {
			limit = {
				NOT = { has_trait = metis }
			}
			custom_description_no_bullet = { 
				text = NEEDS_METIS_TEXT
			}
		}
		custom_description_no_bullet = { 
			text = primal_anger_perk_effect
		}
	}
}
