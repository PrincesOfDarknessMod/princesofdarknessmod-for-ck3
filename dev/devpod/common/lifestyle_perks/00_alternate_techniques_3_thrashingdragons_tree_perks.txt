gender_change_perk = {
	lifestyle = alternate_techniques_lifestyle
	tree = yang_techniques
	position = { 1 2.5 }
	icon = node_animal
	
	can_be_picked = {
		OR = {
			has_trait = kueijin
			has_trait = dhampyr
		}
		has_perk = lotus_breath_perk
		religion = religion:thrashingdragons_religion
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_trait = dharma
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = alternate_techniques_yang_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				NOT = { has_perk = ten_thousand_forms_perk }
			}
			multiply = 0
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
	}
	
	effect = {
		if = {
			limit = { NOT = { religion = religion:thrashingdragons_religion } }
			custom_description_no_bullet = { text = NEEDS_BE_THRASHINGDRAGON_TEXT }
		}
		if = {
			limit = { NOT = { has_perk = lotus_breath_perk } }
			custom_description_no_bullet = { text = has_lotus_breath_perk }
		}
		custom_description_no_bullet = {
			text = gender_change_perk_effect
		}
	}
}
