heightened_senses_perk = {
	lifestyle = phlegmatic_lifestyle
	tree = auspex
	position = { 0.5 0 }
	icon = node_phlegmatic
	
	can_be_picked = {
		has_auspex_trigger = yes
	}		
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_phlegmatic_interest_trigger = yes
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = phlegmatic_auspex_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = sense_the_unseen_perk }	
			}
			multiply = 0
		}
	}

	character_modifier = {
		murder_scheme_resistance_add = 10
		diablerize_scheme_resistance_add = 5
	}
	
	effect = {
		custom_description_no_bullet = { 
			text = heightened_senses_perk_stats_effect
		}
		if = {
			limit = {
				has_auspex_trigger = no
			}
			custom_description_no_bullet = { 
				text = NEEDS_DISCIPLINE_TEXT
			}
		}
	}
}
