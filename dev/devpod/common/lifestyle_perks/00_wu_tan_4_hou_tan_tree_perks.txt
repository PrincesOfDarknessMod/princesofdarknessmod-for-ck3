read_fire_soul_perk = {
	lifestyle = wu_tan_lifestyle
	tree = hou_tan
	position = { 0 0 }
	icon = node_choleric
	
	can_be_picked = { has_trait = hou_tan }

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = wu_tan_hou_tan_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = commune_with_fire_perk }
				NOT = { has_perk = hou_tan_speak_with_jimu_tenno_perk }
			}
			multiply = 0
		}
	}

	character_modifier = {
		monthly_prestige = 0.5
		monthly_court_grandeur_change_add = 0.05
	}
	
	effect = {
		if = {
			limit = { NOT = { has_trait = hou_tan } }
			custom_description_no_bullet = { 
				text = NEEDS_ART_TEXT
			}
		}
		custom_description_no_bullet = { 
			text = read_fire_soul_perk_effect
		}
	}
}
