divine_potential_perk = {
	lifestyle = arcana_lifestyle
	tree = glory
	position = { 0 0 }
	icon = arcana

	can_be_picked = {
		has_trait = demon
	}
	
	character_modifier = {
		diplomacy_per_stress_level = 1
		martial_per_stress_level = 1
		stewardship_per_stress_level = 1
		intrigue_per_stress_level = 1
		learning_per_stress_level = 1
	}
	
	effect = {
		if = {
			limit = { NOT = { has_trait = demon } }
			custom_description_no_bullet = { text = NOT_has_demon_trait }
		}
		custom_description_no_bullet = { 
			text = divine_potential_perk_effect
		}
	}
}

arcana_longevity_perk = {
	lifestyle = arcana_lifestyle
	tree = glory
	position = { 0 1 }
	icon = arcana

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
		if = {
			limit = {
				is_immortal = no
			}
			multiply = 5
		}
	}

	character_modifier = {
		long_reign_bonus_mult = 0.1
	}
	
	effect = {
		custom_description_no_bullet = { 
			text = arcana_longevity_perk_effect
		}
	}
}

trait_aptitude_perk = {
	lifestyle = arcana_lifestyle
	tree = glory
	position = { 0 3 }
	icon = arcana

	can_be_picked = {
		arcana_lifestyle_perk_points > 2
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
		diplomacy = 2
		martial = 2
		stewardship = 2
		intrigue = 2
		learning = 2
	}
	
	effect = {
		if = {
			limit = { 
				OR = {
					arcana_lifestyle_perk_points <= 2 
					NOT = { has_perk = trait_aptitude_perk }
				}
			}
			custom_description_no_bullet = { text = has_2_extra_perk_points }
		}
		hidden_effect = {
			add_arcana_lifestyle_perk_points = -2
		}
		custom_description_no_bullet = { 
			text = trait_aptitude_perk_effect
		}
	}
}
