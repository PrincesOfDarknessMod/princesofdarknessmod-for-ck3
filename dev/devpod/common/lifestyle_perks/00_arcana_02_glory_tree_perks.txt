divine_potential_perk = {
	lifestyle = arcana_lifestyle
	tree = glory
	position = { 0 0 }
	icon = arcana

	can_be_picked = {
		has_trait = demon
	}
	
	character_modifier = {
		diplomacy_per_stress_level = 2
		martial_per_stress_level = 2
		stewardship_per_stress_level = 2
		intrigue_per_stress_level = 2
		learning_per_stress_level = 2
	}
	
	effect = {
		if = {
			limit = { NOT = { has_trait = demon } }
			custom_description_no_bullet = { text = NOT_has_demon_trait }
		}
		custom_description_no_bullet = { 
			text = divine_potential_perk_effect
		}
	}
}

arcana_longevity_perk = {
	lifestyle = arcana_lifestyle
	tree = glory
	position = { 0 1 }
	icon = arcana

	can_be_picked = {
		OR = {
			has_trait = demon
			is_a_thrall = yes
		}
	}
	
	can_be_auto_selected = { 
		
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
		life_expectancy = 25
		years_of_fertility = 20
		health = 2.5
	}
	
	effect = {
		custom_description_no_bullet = { 
			text = arcana_longevity_perk_effect
		}
	}
}

trait_aptitude_perk = {
	lifestyle = arcana_lifestyle
	tree = glory
	position = { 0 3 }
	icon = arcana

	can_be_picked = {
		OR = {
			has_trait = demon
			is_a_thrall = yes
		}
		arcana_lifestyle_perk_points > 3
	}
	
	can_be_auto_selected = { 
		
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
		diplomacy = 2
		martial = 2
		stewardship = 2
		intrigue = 2
		learning = 2
	}
	
	effect = {
		if = {
			limit = { arcana_lifestyle_perk_points <= 3 }
			custom_description_no_bullet = { text = has_3_extra_perk_points }
		}
		hidden_effect = {
			add_arcana_lifestyle_perk_points = -3
		}
		custom_description_no_bullet = { 
			text = trait_aptitude_perk_effect
		}
	}
}
