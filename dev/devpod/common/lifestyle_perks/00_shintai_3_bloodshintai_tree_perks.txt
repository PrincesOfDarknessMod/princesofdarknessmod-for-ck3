permeate_perk = {
	lifestyle = shintai_lifestyle
	tree = bloodshintai
	position = { 0 0 }
	icon = node_shintai
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_trait = dharma
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = shintai_blood_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}
	
	character_modifier = {
		enemy_hostile_scheme_phase_duration_add = 10
	}
	
	effect = {
		if = {
			limit = { faith = faith:inwardway }
			POD_kj_advance_dharma_effect = yes
		}
		custom_description_no_bullet = {
			text = permeate_perk_effect
		}
	}
}
