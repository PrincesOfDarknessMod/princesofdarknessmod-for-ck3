	tree = patterns

	icon = node_demon_arts

	can_be_picked = {
		dynasty = dynasty:14000007
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = lore_house_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
		personal_scheme_power_add = 15
		personal_scheme_power_mult = 0.15
		hostile_scheme_power_add = 15
		hostile_scheme_power_mult = 0.15
	}
	
	effect = {
		if = {
			limit = { not = { dynasty = dynasty:14000007 } }
			custom_description_no_bullet = { text = NOT_has_neberu_dynasty }
		}
		custom_description_no_bullet = { 
			text = sense_congruence_perk_effect
		}
	}
}
