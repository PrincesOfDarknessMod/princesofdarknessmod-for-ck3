	lifestyle = lore_lifestyle
	tree = patterns
	position = { 0 0 }
	icon = node_demon_arts

	can_be_picked = {
		OR = { dynasty = dynasty:14000007 has_character_flag = learnt_patterns_lore }
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = lore_house_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
		personal_scheme_phase_duration_add = -15
		hostile_scheme_phase_duration_add = -15
		contract_scheme_phase_duration_add = -5
	}
	
	effect = {
		
		custom_description_no_bullet = { 
			text = sense_congruence_perk_effect
		}
	}
}
