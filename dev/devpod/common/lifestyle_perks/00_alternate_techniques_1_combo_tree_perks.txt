taking_the_face_perk = {
	lifestyle = alternate_techniques_lifestyle
	tree = combination_techniques
	position = { 0 0 }
	icon = year_of_lotus
	
	can_be_picked = {
		has_POD_kj_blood_trigger = yes
		has_perk = long_neck_perk
		has_perk = bone_obedience_perk
		religion = religion:boneflowers_religion
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_trait = dharma
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = alternate_techniques_combination_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
		enemy_hostile_scheme_phase_duration_add = 5
	}
	
	effect = {
		if = {
			limit = { faith = faith:inwardway }
			POD_kj_advance_dharma_effect = yes
		}
		if = {
			limit = { NOT = { religion = religion:boneflowers_religion } }
			custom_description_no_bullet = { text = NEEDS_BE_SONGOFTHESHADOW_TEXT }
		}
		if = {
			limit = { NOT = { has_perk = long_neck_perk } }
			custom_description_no_bullet = { text = has_long_neck_perk }
		}
		if = {
			limit = { NOT = { has_perk = bone_obedience_perk } }
			custom_description_no_bullet = { text = has_bone_obedience_perk }
		}
		custom_description_no_bullet = {
			text = taking_the_face_perk_effect
		}
	}
}
