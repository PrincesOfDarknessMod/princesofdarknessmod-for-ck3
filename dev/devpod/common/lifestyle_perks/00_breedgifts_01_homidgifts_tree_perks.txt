persuasion_perk = {
	lifestyle = breedgifts_lifestyle
	tree = homidgifts
	position = { 0 0 }
	icon = node_animal
	
	can_be_picked = {
		OR = {
			AND = {
				has_trait = werewolf
				has_trait = homid_born
			}
			has_trait = kinfolk 
			has_trait = abomination
		}
	}
	
	can_be_auto_selected = { 
			
	}	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				OR = {
					AND = {
						has_trait = werewolf
						has_trait = homid_born
					}
					has_trait = kinfolk 
				}
			}
			add = 1000
		}
		if = {
			limit = {
				has_focus = breedgifts_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = apecraftsblessings_perk }	
		}
			multiply = 0
		}
	}

	character_modifier = {
		befriend_scheme_power_mult = 0.25
		seduce_scheme_power_mult = 0.25
	}
	
	effect = {
		if = {
			limit = {
				NOT = {
					has_trait = homid_born
					}
			}
			custom_description_no_bullet = { 
				text = NEEDS_HOMIDBORN_TEXT
			}
		}
	}
}
