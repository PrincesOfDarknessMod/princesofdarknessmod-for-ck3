persuasion_perk = {
	lifestyle = breedgifts_lifestyle
	tree = homidgifts
	position = { 0 0 }
	icon = node_animal
	
	can_be_picked = {
		trigger_if = {
			limit = { has_trait = kinfolk }
			OR = {
				has_character_flag = werewolf_breed
				has_character_flag = ajaba_breed
				has_character_flag = nuwisha_breed
			}
		}
		trigger_else = {
			has_trait = homid_born
			trigger_if = {
				limit = { has_trait = abomination }
				OR = {
					has_character_flag = werewolf_shape
					has_character_flag = ajaba_shape
					has_character_flag = nuwisha_shape
				}
			}
			trigger_else = {
				OR = {
					has_trait = werewolf
					has_trait = ajaba
					has_trait = nuwisha
				}
			}
		}
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				trigger_if = {
					limit = { has_trait = kinfolk }
					OR = {
						has_character_flag = werewolf_breed
						has_character_flag = ajaba_breed
						has_character_flag = nuwisha_breed
					}
				}
				trigger_else = {
					has_trait = homid_born
					OR = {
						has_trait = werewolf
						has_trait = ajaba
						has_trait = nuwisha
					}
				}
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = breedgifts_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = apecraftsblessings_perk }	
		}
			multiply = 0
		}
	}

	character_modifier = {
		befriend_scheme_phase_duration_add = -13
		seduce_scheme_phase_duration_add = -13
	}
	
	effect = {
		if = {
			limit = {
				NOT = {
					has_trait = homid_born
					}
			}
			custom_description_no_bullet = { 
				text = NEEDS_HOMIDBORN_TEXT
			}
		}
	}
}
