acre_catechumen_perk = {
	lifestyle = endowments_lifestyle
	tree = acre
	position = { 0 0 }
	icon = acre_catechumen
	
	can_be_picked = {
		OR = {
			has_knightly_order_dynasty_trigger = yes
			dynasty ?= { has_dynasty_modifier = support_from_knight_of_acre_modifier } 
		}
		has_trait = supehunter
		faith = faith:shadowinquisition
	}		
	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = endowments_all_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = acre_flesh_perk }
			}
			multiply = 0
		}
	}
	character_modifier = {
	}
	
	effect = {
		custom_description_no_bullet = { 
			text = acre_catechumen_perk_text
		}		
		if = {
			limit = { 
				has_knightly_order_dynasty_trigger = no
				dynasty ?= { NOT = { has_dynasty_modifier = support_from_knight_of_acre_modifier } }
			}
			custom_description_no_bullet = { 
				text = acre_only_text
			}
		}
	}
}
