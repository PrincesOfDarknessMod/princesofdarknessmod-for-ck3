rasa_perk = {
	lifestyle = soul_arts_lifestyle
	tree = chiiumuh
	position = { 0 0 }
	icon = node_soul_arts
	
	can_be_picked = {
		OR = {
			has_character_modifier = wisdom_stealer_modifier
			has_trait = kueijin
			has_trait = dhampyr
		}
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_focus = shintai_bone_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				has_focus = soul_arts_chiiumuh_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}
	
	character_modifier = {
		scheme_discovery_chance_mult = 0.10
	}
	
	effect = {
		if = {
			limit = { faith = faith:inwardway }
			POD_kj_advance_dharma_effect = yes
		}
		custom_description_no_bullet = {
			text = rasa_perk_effect
		}
	}
}
