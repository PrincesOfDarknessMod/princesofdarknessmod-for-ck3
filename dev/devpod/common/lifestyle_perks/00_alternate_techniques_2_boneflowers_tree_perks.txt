gentle_yielding_perk = {
	lifestyle = alternate_techniques_lifestyle
	tree = yin_techniques
	position = { 2 1.75 }
	icon = node_animal
	
	can_be_picked = {
		OR = {
			has_trait = kueijin
			has_trait = dhampyr
		}
		has_perk = lotus_breath_perk
		faith = faith:songoftheshadow
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_trait = dharma
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = alternate_techniques_yin_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				NOT = { has_perk = pelesit_perk }
			}
			multiply = 0
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}
	
	character_modifier = {
	}
	
	effect = {
		custom_description_no_bullet = {
			text = gentle_yielding_perk_effect
		}
	}
}
