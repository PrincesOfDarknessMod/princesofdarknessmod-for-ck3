king_fish_perk = {
	lifestyle = rokeaauspicegifts_lifestyle
	tree = dimwatergifts
	position = { 0 0 }
	icon = node_animal
	
	can_be_picked = {
		has_trait = dimwater
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_trait = dimwater
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = aspectgifts_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = strange_blood_perk }
				NOT = { has_perk = true_intentions_perk }
			}
			multiply = 0
		}
	}

	character_modifier = {
	}
	
	effect = {
		if = {
			limit = {
				NOT = { has_trait = dimwater }
			}
			custom_description_no_bullet = { 
				text = NEEDS_DIMWATER_TEXT
			}
		}
	}
}
