remote_viewing_perk = {
	lifestyle = lore_lifestyle
	tree = firmament
	position = { 0 0 }
	icon = node_demon_arts

	can_be_picked = {
		OR = { dynasty = dynasty:14000001 has_character_flag = learnt_firmament_lore }
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = lore_house_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
	}
	
	effect = {
		
		custom_description_no_bullet = { 
			text = remote_viewing_perk_effect
		}
		hidden_effect = {
			every_relation = {
				type = demonic_thrall
				root = { add_character_modifier = remote_viewing_perk_modifier }
			}	
		}
	}
}
