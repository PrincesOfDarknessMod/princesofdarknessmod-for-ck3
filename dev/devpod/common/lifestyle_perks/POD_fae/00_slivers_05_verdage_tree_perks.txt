slivers_verdage_uncontrolled_perk = {
	lifestyle = slivers_lifestyle
	tree = verdage
	position = { 0 0 }
	icon = node_stewardship
	
	can_be_picked = { has_trait = verdage }

	auto_selection_weight = {
		value = 2000
		if = { limit = { has_focus = slivers_verdage_focus } multiply = 5 }
		if = { limit = { can_start_new_lifestyle_tree_trigger = no } multiply = 0 }
	}

	character_modifier = {
		enemy_hostile_scheme_success_chance_add = -5
		enemy_hostile_scheme_phase_duration_add = 10
		POD_animals_maa_damage_add = 2
		POD_animals_maa_toughness_add = 2
		POD_animals_maa_damage_mult = 0.1
		POD_animals_maa_toughness_mult = 0.1
	}
	
	effect = {
		if = { limit = { NOT = { has_trait = verdage } } custom_description_no_bullet = { text = NEEDS_ART_TEXT } }
		custom_description_no_bullet = { text = slivers_verdage_uncontrolled_perk_effect }
	}
}
