slivers_stratus_uncontrolled_perk = {
	lifestyle = slivers_lifestyle
	tree = stratus
	position = { 0 0 }
	icon = node_stewardship
	
	can_be_picked = { has_trait = stratus }

	auto_selection_weight = {
		value = 2000
		if = { limit = { has_focus = slivers_stratus_focus } multiply = 5 }
		if = { limit = { can_start_new_lifestyle_tree_trigger = no } multiply = 0 }
	}

	character_modifier = {
		movement_speed = 0.03
		character_travel_speed = 2
		owned_hostile_scheme_success_chance_add = 5
		hostile_scheme_phase_duration_add = -10
	}
	
	effect = {
		custom_description_no_bullet = { text = slivers_stratus_uncontrolled_perk_effect }
		if = { limit = { NOT = { has_trait = stratus } } custom_description_no_bullet = { text = NEEDS_ART_TEXT } }
	}
}
