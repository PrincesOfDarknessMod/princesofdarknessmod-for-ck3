living_rock_perk = {
	lifestyle = slivers_lifestyle
	tree = petros
	position = { 0 0 }
	icon = node_choleric
	
	can_be_picked = { has_trait = petros }

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = slivers_petros_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
		build_speed = -0.1
		holding_build_speed = -0.1
	}
	
	effect = {
		if = {
			limit = { NOT = { has_trait = petros } }
			custom_description_no_bullet = { 
				text = NEEDS_ART_TEXT
			}
		}
		custom_description_no_bullet = { 
			text = living_rock_perk_effect
		}
	}
}
