	lifestyle = kith_arts_lifestyle
	tree = naming
	position = { 0 0 }
	icon = node_choleric
	
	can_be_picked = { has_trait = naming }

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = kith_arts_naming_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
		learn_language_scheme_phase_duration_add = -37
		max_learn_language_schemes_add = 1	
	}
	
	effect = {
		if = {
			limit = { NOT = { has_trait = naming } }
			custom_description_no_bullet = { 
				text = NEEDS_ART_TEXT
			}
		}
		custom_description_no_bullet = { 
			text = seek_n_spell_perk_effect
		}
	}
}
