	tree = stratus

	icon = node_choleric
	
	can_be_picked = { has_trait = stratus }

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = slivers_stratus_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
		character_travel_speed_mult = 0.3
		movement_speed = 0.1
	}
	
	effect = {
		if = {
			limit = { NOT = { has_trait = stratus } }
			custom_description_no_bullet = { 
				text = NEEDS_ART_TEXT
			}
		}
		custom_description_no_bullet = { 
			text = marathon_perk_effect
		}
	}
}
