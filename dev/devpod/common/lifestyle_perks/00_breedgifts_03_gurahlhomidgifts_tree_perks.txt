	lifestyle = breedgifts_lifestyle
	tree = gurahlhomidgifts
	position = { 1 0 }
	icon = node_animal
	
	can_be_picked = {
		OR = {
			has_trait = kinfolk
			has_trait = homid_born
		}
		POD_has_access_to_fera_gifts = { TRAIT = trait:gurahl }
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				OR = {
					has_trait = kinfolk
					has_trait = homid_born
				}
				POD_has_access_to_fera_gifts = { TRAIT = trait:gurahl }
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = breedgifts_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
		pursue_efficiency = 0.1
	}
	
	effect = {
		if = {
			limit = {
				NOT = {
					has_trait = homid_born
				}
			}
			custom_description_no_bullet = { 
				text = NEEDS_HOMIDBORN_TEXT
			}
		}
	}
}
