arcana_revelation_perk = {
	lifestyle = arcana_lifestyle
	tree = fate_portals
	position = { 0 0 }
	icon = arcana

	can_be_picked = {
		has_trait = demon
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = arcana_fate_portals_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}
	
	effect = {
		if = {
			limit = { NOT = { has_trait = demon } }
			custom_description_no_bullet = { text = NOT_has_demon_trait }
		}
		custom_description_no_bullet = { 
			text = arcana_revelation_perk_effect
		}
	}
}
