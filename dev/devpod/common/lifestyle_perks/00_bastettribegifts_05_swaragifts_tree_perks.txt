diamond_claws_perk = {
	lifestyle = bastettribegifts_lifestyle
	tree = swaragifts
	position = { 0 0 }
	icon = node_animal
	
	can_be_picked = {
		OR = {
			has_perk = royal_privilege_perk
			faith = faith:swara
		}
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_faith = faith:swara
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = bastettribegifts_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = impalas_flight_perk }
			}
			multiply = 0
		}
	}

	character_modifier = {
		prowess_no_portrait = 5
	}
	
	effect = {
		custom_description_no_bullet = {
			text = diamond_claws_perk_effect
		}		
	}
}
