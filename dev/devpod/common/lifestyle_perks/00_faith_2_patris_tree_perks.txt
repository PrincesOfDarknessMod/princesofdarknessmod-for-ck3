pry_open_perk = {
	lifestyle = faith_lifestyle
	tree = patris
	position = { 0 0 }
	icon = node_learning

	can_be_picked = {
		has_trait = supehunter
		OR = {
			faith = faith:shadowinquisition
			faith = { has_doctrine_parameter = access_to_holy_arts }
		}
		OR = {
			AND = {
				exists = dynasty
				OR = {
					dynasty = dynasty:13000004
					dynasty = dynasty:13000005
					dynasty = { has_dynasty_modifier = custom_lay_order_modifier }
				}
			}
			has_trait = truefaith
			has_trait = beatific
		}
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}
	
	character_modifier = {
		personal_scheme_power_mult = 0.5
	}

	effect = {
		custom_description_no_bullet = {
			text = pry_open_perk_effect
		}
		if = {
			limit = {
				NOR = {
					AND = {
						exists = dynasty
						OR = {
							dynasty = dynasty:13000004
							dynasty = dynasty:13000005
							dynasty = { has_dynasty_modifier = custom_lay_order_modifier }
						}
					}
					has_trait = truefaith
					has_trait = beatific
				}
			}
			custom_description_no_bullet = { 
				text = PATRIS_NOT_PRIMARY_TEXT
			}
		}
		if = {
			limit = {
				NAND = {
					OR = {
						faith = { has_doctrine_parameter = access_to_holy_arts }
						faith = faith:shadowinquisition
					}
					has_trait = supehunter
				}
			}
			custom_description_no_bullet = { 
				text = SHADOW_INQUISITION_ONLY_TEXT
			}
		}
	}
}
