wind_water_eye_perk = {
	lifestyle = chi_arts_lifestyle
	tree = fengshui
	position = { 0 0 }
	icon = node_chi_arts
	
	can_be_picked = {
		OR = {
			has_trait = kueijin
			has_trait = dhampyr
		}
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_trait = dharma
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = chi_arts_fengshui_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
	}
	
	effect = {
		custom_description_no_bullet = {
			text = wind_water_eye_perk_effect
		}
	}
}
