sleepsense_perk = {
	lifestyle = arcanoi_lifestyle
	tree = phantasm
	position = { 0 0 }
	icon = node_soul_arts

	can_be_picked = {
		has_trait = wraith
		has_trait = arcanos_phantasm
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = arcanos_phantasm_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = lingua_perk }
			}
			multiply = 0
		}
	}

	character_modifier = {
		hostile_scheme_phase_duration_add = -10
	}
	
	effect = {
		if = {
			limit = {
				NOT = { has_trait = arcanos_phantasm }
			}
			custom_description_no_bullet = { 
				text = NOT_has_arcanos_phantasm_trait
			}
		}
		custom_description_no_bullet = { 
			text = sleepsense_perk_effect
		}
	}
}
