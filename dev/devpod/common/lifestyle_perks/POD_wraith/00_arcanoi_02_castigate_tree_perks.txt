coax_perk = {
	lifestyle = arcanoi_lifestyle
	tree = castigate
	position = { 1 0 }
	icon = node_soul_arts

	can_be_picked = {
		has_trait = arcanos_castigate
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = arcanos_castigate_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}
	
	effect = {
		if = {
			limit = {
				NOT = { has_trait = arcanos_castigate }
			}
			custom_description_no_bullet = { 
				text = NOT_has_arcanos_castigate_trait
			}
		}
		custom_description_no_bullet = { 
			text = coax_perk_effect
		}
	}
}
