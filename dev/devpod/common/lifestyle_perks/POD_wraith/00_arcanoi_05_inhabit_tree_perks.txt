	tree = inhabit

		has_trait = arcanos_inhabit
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = arcanos_inhabit_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}

		}
	}

	character_modifier = {
		prowess_no_portrait = 2
	}
	
	effect = {
		if = {
			limit = {

				text = NOT_has_arcanos_inhabit_trait
			}
		}
		custom_description_no_bullet = { 
			text = lightnings_bite_perk_effect
		}
	}
}
