erasure_perk = {
	lifestyle = arcanoi_lifestyle
	tree = mnemosynis
	position = { 1 0 }
	icon = node_soul_arts

	can_be_picked = {
		has_trait = wraith
		has_trait = arcanos_mnemosynis
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = arcanos_mnemosynis_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}
	
	effect = {
		if = {
			limit = {
				NOT = { has_trait = arcanos_mnemosynis }
			}
			custom_description_no_bullet = { 
				text = NOT_has_arcanos_mnemosynis_trait
			}
		}
		custom_description_no_bullet = { 
			text = erasure_perk_effect
		}
	}
}
