weirdness_perk = {
	lifestyle = arcanoi_lifestyle
	tree = pandemonium
	position = { 1 0 }
	icon = node_soul_arts

	can_be_picked = {
		has_trait = arcanos_pandemonium
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = arcanos_pandemonium_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}
	
	effect = {
		if = {
			limit = {
				NOT = { has_trait = arcanos_pandemonium }
			}
			custom_description_no_bullet = { 
				text = NOT_has_arcanos_pandemonium_trait
			}
		}
		custom_description_no_bullet = { 
			text = weirdness_perk_effect
		}
	}
}
