wraithgrasp_perk = {
	lifestyle = arcanoi_lifestyle
	tree = outrage
	position = { 0 0 }
	icon = node_soul_arts

	can_be_picked = {
		has_trait = arcanos_outrage
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = arcanos_outrage_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = breath_of_the_grave_perk }
			}
			multiply = 0
		}
	}
	
	effect = {
		if = {
			limit = {
				NOT = { has_trait = arcanos_outrage }
			}
			custom_description_no_bullet = { 
				text = NOT_has_arcanos_outrage_trait
			}
		}
		custom_description_no_bullet = { 
			text = wraithgrasp_perk_effect
		}
	}
}
