enshroud_perk = {
	lifestyle = arcanoi_lifestyle
	tree = argos
	position = { 0 0 }
	icon = node_soul_arts

	can_be_picked = {
		has_trait = arcanos_argos
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = arcanos_argos_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = weather_eye_perk }
			}
			multiply = 0
		}
	}
	
	character_modifier = {
		POD_terrain_umbra_sense_advantage = 0.05
	}

	effect = {
		if = {
			limit = {
				NOT = { has_trait = arcanos_argos }
			}
			custom_description_no_bullet = { 
				text = NOT_has_arcanos_argos_trait
			}
		}
		custom_description_no_bullet = { 
			text = enshroud_perk_effect
		}
	}
}
