rot_perk = {
	lifestyle = arcanoi_lifestyle
	tree = flux
	position = { 1 0 }
	icon = node_soul_arts

	can_be_picked = {
		has_trait = wraith
		has_trait = arcanos_flux
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = arcanos_flux_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
		siege_phase_time = -0.2
	}
	
	effect = {
		if = {
			limit = {
				NOT = { has_trait = arcanos_flux }
			}
			custom_description_no_bullet = { 
				text = NOT_has_arcanos_flux_trait
			}
		}
		custom_description_no_bullet = { 
			text = rot_perk_effect
		}
	}
}
