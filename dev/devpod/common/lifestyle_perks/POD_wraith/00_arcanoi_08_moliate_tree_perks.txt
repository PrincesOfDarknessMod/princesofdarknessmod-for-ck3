imitate_perk = {
	lifestyle = arcanoi_lifestyle
	tree = moliate
	position = { 0 0 }
	icon = node_soul_arts

	can_be_picked = {
		has_trait = arcanos_moliate
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = arcanos_moliate_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = lizard_tail_perk }
			}
			multiply = 0
		}
	}
	
	effect = {
		if = {
			limit = {
				NOT = { has_trait = arcanos_moliate }
			}
			custom_description_no_bullet = { 
				text = NOT_has_arcanos_moliate_trait
			}
		}
		custom_description_no_bullet = { 
			text = imitate_perk_effect
		}
	}
}
