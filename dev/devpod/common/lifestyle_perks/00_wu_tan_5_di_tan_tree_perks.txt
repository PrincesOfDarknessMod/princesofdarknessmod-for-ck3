read_earth_soul_perk = {
	lifestyle = wu_tan_lifestyle
	tree = di_tan
	position = { 0 0 }
	icon = node_choleric
	
	can_be_picked = { has_trait = di_tan }

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = wu_tan_di_tan_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = commune_with_earth_perk }
				NOT = { has_perk = di_tan_speak_with_jimu_tenno_perk }
			}
			multiply = 0
		}
	}

	character_modifier = {
		happy_powerful_vassal_tax_contribution_add = 0.1
	}
	
	effect = {
		if = {
			limit = { NOT = { has_trait = di_tan } }
			custom_description_no_bullet = { 
				text = NEEDS_ART_TEXT
			}
		}
		custom_description_no_bullet = { 
			text = read_earth_soul_perk_effect
		}
	}
}
