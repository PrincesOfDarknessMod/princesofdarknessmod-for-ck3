sweet_hunters_smile_perk = {
	lifestyle = breedgifts_lifestyle
	tree = bastethomidgifts
	position = { 0 0 }
	icon = node_animal
	
	can_be_picked = {
		OR = {
			AND = {
				has_trait = bastet
				has_trait = homid_born
			}
			has_trait = kinfolk 
		}
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				OR = {
					AND = {
						has_trait = bastet
						has_trait = homid_born
					}
					has_trait = kinfolk 
				}
			}
			add = 1000
		}
		if = {
			limit = {
				has_focus = breedgifts_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = cat_claws_perk }	
			}
			multiply = 0
		}
	}

	character_modifier = {
		befriend_scheme_power_mult = 0.25
		seduce_scheme_power_mult = 0.25
	}
	
	effect = {
		if = {
			limit = {
				NOT = {
					has_trait = homid_born
				}
			}
			custom_description_no_bullet = { 
				text = NEEDS_HOMIDBORN_TEXT
			}
		}
	}
}
