lamp_of_faith_perk = {
	lifestyle = lore_lifestyle
	tree = celestials
	position = { 0 0 }
	icon = node_demon_arts

	can_be_picked = {
		piety_level >= low_piety_level
		dynasty = dynasty:14000004
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = lore_common_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
		scheme_discovery_chance_mult = 0.25
		pursue_efficiency = 0.25
		army_pursuit_mult = 0.25
		maa_pursuit_mult = 0.25
	}
	
	effect = {
		if = {
			limit = { piety_level < low_piety_level }
			custom_description_no_bullet = { text = lacks_low_piety_level }
		}
		custom_description_no_bullet = { 
			text = lamp_of_faith_perk_effect
		}
	}
}
