pedagogy_perk = {
	lifestyle = learning_lifestyle
	tree = scholarship
	position = { 1 0 }
	icon = node_learning
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_education_learning_trigger = yes
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = learning_scholarship_focus
			}
			multiply = 5
		}

		flag = government_has_merit

	}

	government_character_modifier = {

	}
	
	effect = {
		custom_description_no_bullet = {
			text = pedagogy_perk_effect
		}
	}
}

# Sanctioned Loopholes
sanctioned_loopholes_perk = {
	lifestyle = learning_lifestyle
	tree = scholarship
	position = { 0 4 }
	icon = node_learning

	parent = scholarly_circles_perk
	
	name = {
		first_valid = {
			triggered_desc = {
				trigger = { has_government = landless_adventurer_government }
				desc = sanctioned_loopholes_perk_adventurer_name
			}
			triggered_desc = {
				trigger = { government_has_flag = government_is_merchant }
				desc = sanctioned_loopholes_perk_merchant_name
			}
			triggered_desc = {
				trigger = {
					OR = {
						government_has_flag = government_has_merit
						government_has_flag = government_is_japan_administrative
						government_has_flag = government_is_japan_feudal
					}
				}
				desc = scholar_gentleman
			}
			desc = sanctioned_loopholes_perk_name
		}
	}
	
	government_character_modifier = {
		flag = government_is_landless_adventurer
		artifact_decay_reduction_mult = -0.75
	}
	government_character_modifier = {
		flag = government_has_merit
		confucian_education_xp_gain_mult = 0.15
		monthly_merit = 2
	}
	government_character_modifier = {
		flag = government_is_japan_administrative
		confucian_education_xp_gain_mult = 0.15
		domicile_monthly_piety_add = 0.5
	}
	government_character_modifier = {
		flag = government_is_japan_feudal
		confucian_education_xp_gain_mult = 0.15
		domicile_monthly_piety_add = 0.5
	}
	government_character_modifier = {
		flag = government_is_republic
		trade_scheme_phase_duration_add = -50
		owned_trade_scheme_success_chance_growth_add = 2
		trade_capacity_mult = 0.1
	}
	
	
	effect = {
		if = {
			limit = {
				NOR = {
					government_has_flag = government_is_landless_adventurer
					government_has_flag = government_is_merchant
					government_has_flag = government_has_merit
					government_has_flag = government_is_japan_administrative
					government_has_flag = government_is_japan_feudal
					government_has_flag = government_is_republic
					pam_beseech_claims_access_trigger = yes
				}
			}
			custom_description_no_bullet = {
				text = sanctioned_loopholes_perk_effect
			}
		}
		if = {
			limit = {
				pam_beseech_claims_access_trigger = yes
			}
			custom_description_no_bullet = {
				text = sanctioned_loopholes_perk_effect_beseech_claim
			}
		}
		if = {
			limit = {
				has_pam_dlc_trigger = yes
				government_has_flag = government_is_theocracy
				pam_is_archbishop_or_higher_trigger = yes
			}
			custom_description_no_bullet = {
				text = sanctioned_loopholes_perk_effect_sell_claim
			}
		}
		if = {
			limit = {
				government_has_flag = government_is_landless_adventurer
			}	
			custom_tooltip = {
				text = adventurer_sanctioned_loopholes_perk_effect
			}
		}
		if = {
			limit = { government_has_flag = government_is_merchant }
			switch = {
				trigger = has_realm_law
				company_purpose_family = { custom_description_no_bullet = { text = company_law_family_purpose } }
				company_purpose_artisan = { custom_description_no_bullet = { text = company_law_artisan_purpose } }
				company_purpose_creditor = { custom_description_no_bullet = { text = company_law_creditor_purpose } }
				company_purpose_caravaneer = { custom_description_no_bullet = { text = company_law_caravaneer_purpose } }
			}
			hidden_effect = {
				merchant_company_law_purpose_effect = yes
			}
		}
	}
}
