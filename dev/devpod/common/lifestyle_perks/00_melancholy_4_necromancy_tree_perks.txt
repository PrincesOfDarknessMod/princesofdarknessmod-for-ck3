impundulunecromancyadvanced_perk = {
	lifestyle = melancholy_lifestyle
	tree = melancholy_necromancy
	position = { 13.6 5 }
	icon = path_impundulu
	
	can_be_picked = {
		has_impundulu_necromancy_trigger = yes
	}
	
	parent = spectral_menacing_perk

	character_modifier = {
		spouse_opinion = 10
	}

	effect = {
		if = {
			limit = {
				NOT = { has_trait = necromancyadvanced }
			}
			remove_trait = necromancy
			add_trait_force_tooltip = necromancyadvanced
		}
		if = {
			limit = { has_impundulu_necromancy_trigger = no }
			custom_description_no_bullet = { text = needs_impundulu_necromancy_path_training }
		}
	}
}
