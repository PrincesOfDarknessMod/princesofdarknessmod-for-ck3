	lifestyle = commongifts_lifestyle
	tree = ajabagifts
	position = { 0 0 }
	icon = node_animal
	
	can_be_picked = {
		OR = {
			has_trait = ajaba
			has_character_flag = ajaba_shape
			has_character_flag = ajaba_breed

		}
	}

		value = 11
		if = {

			}

		}

			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = gift_of_terror_perk }
			}
			multiply = 0
		}
	}
