sisters_gospel_perk = {
	lifestyle = endowments_lifestyle
	tree = sisters
	position = { 0 0 }
	icon = sisters_gospel
	
	can_be_picked = {
		OR = {
			has_mendicant_order_dynasty_trigger = yes
			dynasty ?= { has_dynasty_modifier = support_from_sister_of_stjohn_modifier }
		}
		has_trait = supehunter
		OR = {
			faith = faith:shadowinquisition
			faith = { has_doctrine_parameter = access_to_inquisition_objectives }
		}
	}		
	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = endowments_all_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = sisters_lux_perk }
			}
			multiply = 0
		}
	}
	effect = {
		custom_description_no_bullet = { 
			text = sisters_gospel_perk_text
		}
		if = {
			limit = { 
				has_mendicant_order_dynasty_trigger = no 
				dynasty ?= { NOT = { has_dynasty_modifier = support_from_sister_of_stjohn_modifier } }
			}
			custom_description_no_bullet = { 
				text = sisters_only_text
			}
		}
	}
}
