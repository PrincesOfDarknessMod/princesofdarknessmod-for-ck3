red_vigil_perk = {
	lifestyle = endowments_lifestyle
	tree = red
	position = { 0 0 }
	icon = red_vigil
	
	can_be_picked = {
		OR = {
			has_contemplative_order_dynasty_trigger = yes
			dynasty ?= { has_dynasty_modifier = support_from_red_order_modifier }
			any_directly_owned_province = { has_building_with_flag = building_allow_ritae_perks }
		}
		has_trait = supehunter
		OR = {
			faith = faith:shadowinquisition
			faith = { has_doctrine_parameter = access_to_inquisition_objectives }
		}
	}		
	
	character_modifier = {
		monthly_lifestyle_xp_gain_mult = 0.2
		movement_speed = 0.2
	}
	
	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = endowments_all_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = red_sunstone_perk }
			}
			multiply = 0
		}
	}
	effect = {
		custom_description_no_bullet = { 
			text = red_vigil_perk_text
		}
		if = {
			limit = { 
				has_contemplative_order_dynasty_trigger = no
				dynasty ?= { NOT = { has_dynasty_modifier = support_from_red_order_modifier } }
				NOT = { any_directly_owned_province = { has_building_with_flag = building_allow_ritae_perks } }
			}
			custom_description_no_bullet = { 
				text = red_only_text
			}
		}
	}
}
