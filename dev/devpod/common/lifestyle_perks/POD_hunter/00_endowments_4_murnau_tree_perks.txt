murnau_knowledge_perk = {
	lifestyle = endowments_lifestyle
	tree = murnau
	position = { 0 0 }
	icon = murnau_knowledge
	
	can_be_picked = {
		has_trait = supehunter
		OR = {
			faith = faith:shadowinquisition
			faith = { has_doctrine_parameter = access_to_inquisition_objectives }
		}
		has_endowments_murnau_interest_trigger = yes
	}		
	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = endowments_all_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = murnau_oblige_perk }
			}
			multiply = 0
		}
	}
	
	character_modifier = {
	}
	
	effect = {
		custom_description_no_bullet = { 
			text = murnau_knowledge_perk_text
		}	
		if = {
			limit = {
				NOT = { dynasty ?= dynasty:13000004 }
				NOT = { dynasty ?= { has_dynasty_modifier = support_from_von_murnau_modifier } }
			}
			custom_description_no_bullet = { 
				text = murnau_only_text
			}
		}
	}
}
