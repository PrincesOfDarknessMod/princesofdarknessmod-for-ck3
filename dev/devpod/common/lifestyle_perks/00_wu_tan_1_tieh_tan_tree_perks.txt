read_metal_soul_perk = {
	lifestyle = wu_tan_lifestyle
	tree = tieh_tan
	position = { 0 0 }
	icon = node_choleric
	
	can_be_picked = { has_trait = tieh_tan }

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = wu_tan_tieh_tan_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = commune_with_metal_perk }
				NOT = { has_perk = tieh_tan_speak_with_jimu_tenno_perk }
			}
			multiply = 0
		}
	}

	character_modifier = {
		scheme_discovery_chance_mult = 0.10
	}
	
	effect = {
		if = {
			limit = { NOT = { has_trait = tieh_tan } }
			custom_description_no_bullet = { 
				text = NEEDS_ART_TEXT
			}
		}
		custom_description_no_bullet = { 
			text = read_metal_soul_perk_effect
		}
	}
}
