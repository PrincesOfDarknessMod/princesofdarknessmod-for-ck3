investiture_armor_perk = {
	lifestyle = investiture_lifestyle
	tree = demonic_investment
	position = { 0 0 }
	icon = trait_infernalist
	can_be_picked = {
		investiture_lifestyle_perk_points >= 1
		POD_investiture_armor_perk_modifier_stacks < 10 # soak caps at 50% = 10 stacks
	}
	effect = {
		custom_description_no_bullet = { text = investiture_stacking_effect }
		POD_investiture_perk_cost_effect = { VALUE = 1 }
		POD_add_modifier_and_track_stacks = {
			TYPE = character
			MODIFIER = POD_investiture_armor_perk_modifier
		}
		custom_description_no_bullet = { text = investiture_armor_perk_effect }
		# Re-offer the perk only while below the cap; the check runs AFTER the increment
		# above, so at the 18th stack it is not removed and stays taken.
		hidden_effect = {
			if = {
				limit = { POD_investiture_armor_perk_modifier_stacks < 10 }
				remove_perk = investiture_armor_perk
			}
		}
	}
}
