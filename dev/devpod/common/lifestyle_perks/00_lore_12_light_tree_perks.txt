light_perk = {
	lifestyle = lore_lifestyle
	tree = light
	position = { 0 0 }
	icon = node_demon_arts

	can_be_picked = {
		OR = { dynasty = dynasty:14000007 has_character_flag = learnt_light_lore }
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = lore_house_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
		pursue_efficiency = 0.1
		army_pursuit_mult = 0.1
	}
	
	effect = {
		
		custom_description_no_bullet = { 
			text = light_perk_effect
		}
	}
}
