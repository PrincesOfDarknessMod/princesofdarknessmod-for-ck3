	lifestyle = realm_lifestyle
	tree = realm_time
	position = { 0 5 }
	icon = realm_time
	
	parent = realm_time_4_perk

	can_be_picked = {
		OR = {
			has_trait = realm_time
			game_start_date != 1230.1.7
		}
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = realm_time_focus
			}
			multiply = 5
		}
	}

	character_modifier = {
	}
	
	effect = {
		if = {
			limit = { 
				NOT = { has_trait = realm_time } 
				game_start_date = 1230.1.7
			}
			custom_description_no_bullet = { text = NOT_has_realm_time_trait }
		}
		else = {
			add_trait = realm_time
		}
		custom_description_no_bullet = { 
			text = realm_time_5_perk_effect
		}
	}
}
