sense_the_sin_perk = {
	lifestyle = sanguine_lifestyle
	tree = daimoinon
	position = { 0 0 }
	icon = node_sanguine
	
	can_be_picked = {
		OR = {
			has_trait = daimoinon
			has_trait = daimoinonadvanced
		}
	}		
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_sanguine_interest_trigger = yes
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = sanguine_daimoinon_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = fear_of_the_void_perk }				
			}
			multiply = 0
		}
	}
	
	effect = {
		custom_description_no_bullet = { text = sense_the_sin_perk_effect }
		if = {
			limit = {
				has_daimoinon_trigger = no
			}
			custom_description_no_bullet = { 
				text = NEEDS_DISCIPLINE_TEXT
			}
		}
	}
}
