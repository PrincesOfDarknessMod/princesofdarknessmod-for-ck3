open_close_portals_perk = {
	lifestyle = lore_lifestyle
	tree = portals
	position = { 0 0 }
	icon = node_demon_arts

	can_be_picked = {
		dynasty = dynasty:14000007
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = lore_house_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
		siege_phase_time = -0.25
	}
	
	effect = {
		if = {
			limit = { not = { dynasty = dynasty:14000007 } }
			custom_description_no_bullet = { text = NOT_has_neberu_dynasty }
		}
		custom_description_no_bullet = { 
			text = open_close_portals_perk_effect
		}
	}
}
