speak_with_the_dead_perk = {
	lifestyle = lore_lifestyle
	tree = spirit
	position = { 0 0 }
	icon = node_demon_arts

	can_be_picked = {
		dynasty = dynasty:14000003
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = lore_house_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
		POD_wraith_maa_maintenance_mult = -0.05
		POD_wraith_maa_recruitment_cost_mult = -0.05
		POD_plasmic_maa_maintenance_mult = -0.05
		POD_plasmic_maa_recruitment_cost_mult = -0.05
		POD_plasmic_maa_max_size_add = 1
		POD_wraith_maa_max_size_add = 1
	}
	
	effect = {
		if = {
			limit = { NOT = { dynasty = dynasty:14000003 } }
			custom_description_no_bullet = { text = NOT_has_halaku_dynasty }
		}
		custom_description_no_bullet = { 
			text = speak_with_the_dead_perk_effect
		}
	}
}
