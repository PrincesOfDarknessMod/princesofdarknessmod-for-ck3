eyes_of_the_wound_perk = {
	lifestyle = rokeaauspicegifts_lifestyle
	tree = brightwatergifts
	position = { 0 0 }
	icon = node_animal
	
	can_be_picked = {
		has_trait = brightwater
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_trait = brightwater
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = aspectgifts_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = rokea_restraint_perk }
				NOT = { has_perk = unseen_attack_perk }
			}
			multiply = 0
		}
	}

	character_modifier = {
	}
	
	effect = {
		if = {
			limit = {
				NOT = { has_trait = brightwater }
			}
			custom_description_no_bullet = { 
				text = NEEDS_BRIGHTWATER_TEXT
			}
		}
	}
}
