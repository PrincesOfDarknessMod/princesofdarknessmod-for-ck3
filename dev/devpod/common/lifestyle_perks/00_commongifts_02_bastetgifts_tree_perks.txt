pathfinders_pride_perk = {
	lifestyle = commongifts_lifestyle
	tree = bastetgifts
	position = { 0 0 }
	icon = node_animal
	
	can_be_picked = {
		OR = {
			has_trait = bastet
			has_character_flag = bastet_shape
			has_character_flag = bastet_breed
			
			has_trait = kitsune
		}
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_trait = bastet
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = commongifts_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = silent_stalking_perk }
			}
			multiply = 0
		}
	}

	character_modifier = {
		character_travel_speed_mult = 0.3
		travel_danger = -5
	}

	effect = {
		custom_description_no_bullet = {
			text = pathfinders_pride_perk_effect
		}
	}
	
}

	}
	
	effect = {
		if = {
			limit = { 
				is_ai = yes
				NOT = { has_perk = withering_stare_perk }
			}
			set_focus = no
		}
		custom_description_no_bullet = {
			text = jump_to_the_moon_perk_effect
		}
		if = {
			limit = { is_ai = no }
			if = {
				limit = { NOT = { has_variable_list = known_umbra_realms_list } }
				POD_create_umbra_location = { NAME = flag:earth }
			}
			if = {
				limit = { NOT = { POD_umbra_realm_known_trigger = { NAME = flag:court_of_sokhta } } }
				POD_create_umbra_location = { NAME = flag:court_of_sokhta }
			}
		}
	}
