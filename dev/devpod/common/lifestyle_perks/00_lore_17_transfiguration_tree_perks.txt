lore_mimic_perk = {
	lifestyle = lore_lifestyle
	tree = transfiguration
	position = { 0 0 }
	icon = node_demon_arts

	can_be_picked = {
		OR = { dynasty = dynasty:14000005 has_character_flag = learnt_transfiguration_lore }
	}

	auto_selection_weight = {
		value = 2000
		if = {
			limit = {
				has_focus = lore_house_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}

	character_modifier = {
		personal_scheme_phase_duration_add = -13
	}
	
	effect = {
		
		custom_description_no_bullet = { 
			text = lore_mimic_perk_effect
		}
	}
}
