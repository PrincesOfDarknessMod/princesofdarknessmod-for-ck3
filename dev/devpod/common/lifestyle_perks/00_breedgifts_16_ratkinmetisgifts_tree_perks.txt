rat_mothers_touch_perk = {
	lifestyle = breedgifts_lifestyle
	tree = ratkinmetisgifts
	position = { 0 0 }
	icon = node_animal
	
	can_be_picked = {
		has_trait = metis
		trigger_if = {
			limit = { has_trait = abomination }
			has_character_flag = ratkin_shape
		}
		trigger_else = {
			has_trait = ratkin
		}
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_trait = ratkin
				has_trait = metis
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = breedgifts_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = ratkin_sense_wyrm_perk }
			}
			multiply = 0
		}
	}

	character_modifier = {
		hard_casualty_modifier = -0.01
	}
	
	effect = {
		if = {
			limit = {
				NOT = { has_trait = metis }
			}
			custom_description_no_bullet = { 
				text = NEEDS_METIS_TEXT
			}
		}
		custom_description_no_bullet = { 
			text = dragons_milk_perk_effect
		}
	}
}
