blood_of_the_deeps_perk = {
	lifestyle = rokeaauspicegifts_lifestyle
	tree = darkwatergifts
	position = { 0 0 }
	icon = node_animal
	
	can_be_picked = {
		has_trait = darkwater
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_trait = darkwater
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = aspectgifts_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
				NOT = { has_perk = silent_sending_perk }
				NOT = { has_perk = chill_perk }
			}
			multiply = 0
		}
	}

	character_modifier = {
	}
	
	effect = {
		if = {
			limit = {
				NOT = { has_trait = darkwater }
			}
			custom_description_no_bullet = { 
				text = NEEDS_DARKWATER_TEXT
			}
		}
	}
}
