	lifestyle = demon_arts_lifestyle
	tree = hellweaving
	position = { 0 0 }
	icon = node_demon_arts

	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_trait = dharma
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = demon_arts_hellweaving_focus
			}
			multiply = 5
		}

			limit = {
				has_demon_arts_focus_trigger = no
			}
			multiply = 0
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}

		}

		custom_description_no_bullet = {
			text = glimpse_of_yomi_perk_effect
		}
	}
}
