glimpse_of_yomi_perk = {
	lifestyle = demon_arts_lifestyle
	tree = hellweaving
	position = { 0 0 }
	icon = node_demon_arts
	
	can_be_picked = {
		has_POD_kj_blood_trigger = yes
		POD_is_akuma_trigger = yes
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_trait = dharma
			}
			add = 1989
		}
		if = {
			limit = {
				has_focus = demon_arts_hellweaving_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				has_demon_arts_focus_trigger = no
			}
			multiply = 0
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}
	
	character_modifier = {
		stress_gain_mult = 0.10
	}
	
	effect = {
		if = {
			limit = { faith = faith:inwardway }
			POD_kj_advance_dharma_effect = yes
		}
		if = {
			limit = { POD_is_akuma_trigger = no }
			custom_description_no_bullet = { text = NEEDS_BE_AKUMA_TEXT }
		}
		custom_description_no_bullet = {
			text = glimpse_of_yomi_perk_effect
		}
	}
}
