fate_of_the_one_perk = {
	lifestyle = soul_arts_lifestyle
	tree = tzuwei
	position = { 0 0 }
	icon = node_soul_arts
	
	can_be_picked = {
		has_POD_kj_blood_trigger = yes
	}
	
	auto_selection_weight = {
		value = 11
		if = {
			limit = {
				has_focus = shintai_bone_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				has_focus = soul_arts_tzuwei_focus
			}
			multiply = 5
		}
		if = {
			limit = {
				can_start_new_lifestyle_tree_trigger = no
			}
			multiply = 0
		}
	}
	
	character_modifier = {
		enemy_hostile_scheme_phase_duration_add = 5
	}
	
	effect = {
		if = {
			limit = { faith = faith:inwardway }
			POD_kj_advance_dharma_effect = yes
		}
		custom_description_no_bullet = {
			text = fate_of_the_one_perk_effect
		}
	}
}
