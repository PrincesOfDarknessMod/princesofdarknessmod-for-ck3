######################################
# Gratiano de Veronese - Shadow War
# Player-only vassalization war against Italian Ventrue princes.
######################################

POD_gratiano_shadow_war = {
	group = invasion
	interface_priority = 50

	war_name = "POD_GRATIANO_SHADOW_WAR_WAR_NAME"
	war_name_base = "POD_GRATIANO_SHADOW_WAR_WAR_NAME"
	cb_name = "POD_GRATIANO_SHADOW_WAR_CB_NAME"

	combine_into_one = yes
	should_show_war_goal_subview = yes

	target_titles = none
	ai_only_against_neighbors = yes

	attacker_score_from_occupation_scale = 50
	defender_score_from_occupation_scale = 50
	attacker_score_from_battles_scale = 200
	defender_score_from_battles_scale = 200

	on_primary_attacker_death = invalidate
	on_primary_defender_death = invalidate
	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	allow_hostages = no

	allowed_for_character = {
		this = character:505144
		is_ai = no
		has_character_flag = POD_gratiano_shadow_war_unlocked
		custom_description = {
			text = POD_gratiano_shadow_war_cooldown_block
			NOT = { has_character_flag = POD_gratiano_shadow_war_cooldown }
		}
	}

	# Keep the cooldown reason visible in the declaration list.
	allowed_for_character_display_regardless = {
		this = character:505144
		has_character_flag = POD_gratiano_shadow_war_unlocked
	}

	allowed_against_character = {
		scope:defender = {
			is_ventrue_trigger = yes
			is_independent_ruler = yes
			any_realm_county = { POD_gratiano_italian_sphere_trigger = yes }
			NOT = { highest_held_title_tier >= tier_kingdom }
		}
	}

	allowed_against_character_display_regardless = {
	}

	cost = {
		prestige = {
			add = {
				value = 300
				desc = CB_BASE_COST
			}
		}
	}

	valid_to_start = {
		always = yes
	}

	should_invalidate = {
		scope:defender = {
			OR = {
				is_ventrue_trigger = no
				is_independent_ruler = no
				NOT = { any_realm_county = { POD_gratiano_italian_sphere_trigger = yes } }
				# A prince who wins a crown mid-war is beyond the swear-fealty war's reach.
				highest_held_title_tier >= tier_kingdom
			}
		}
	}

	on_invalidated_desc = msg_POD_gratiano_shadow_war_invalidated

	on_declaration = {
		on_declared_war = yes
		scope:attacker = {
			add_character_flag = {
				flag = POD_gratiano_shadow_war_cooldown
				years = 5		# [TEAM-TUNABLE]
			}
		}
	}

	on_victory_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = POD_gratiano_shadow_war_victory_desc_defender
			}
			desc = POD_gratiano_shadow_war_victory_desc
		}
	}

	on_victory = {
		scope:attacker = { show_pow_release_message_effect = yes }
		scope:attacker = { accolade_attacker_war_end_glory_gain_med_effect = yes }

		create_title_and_vassal_change = {
			type = swear_fealty
			save_scope_as = change
			add_claim_on_loss = no
		}
		scope:defender = {
			change_liege = {
				liege = scope:attacker
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change

		add_truce_attacker_victory_effect = yes
		scope:attacker = { trigger_event = POD_gratiano.0095 }
	}

	on_white_peace_desc = {
		desc = POD_gratiano_shadow_war_white_peace_desc
	}

	on_white_peace = {
		scope:attacker = { show_pow_release_message_effect = yes }
		scope:defender = { accolade_defender_war_end_glory_gain_low_effect = yes }
		add_truce_white_peace_effect = yes
	}

	on_defeat_desc = {
		desc = POD_gratiano_shadow_war_defeat_desc
	}

	on_defeat = {
		scope:attacker = { show_pow_release_message_effect = yes }
		scope:defender = { accolade_defender_war_end_glory_gain_med_effect = yes }
		add_truce_attacker_defeat_effect = yes
	}

	ai_score = { value = 0 }
	ai_score_mult = { value = 0 }
}
