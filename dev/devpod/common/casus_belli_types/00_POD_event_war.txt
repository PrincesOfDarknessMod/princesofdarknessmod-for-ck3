POD_event_war = {
	group = event
	icon = invasion
	interface_priority = 1000
	
	war_name = "POD_event_war"
	war_name_base = "POD_event_war"
	cb_name = "POD_event_war"

	combine_into_one = yes
	should_show_war_goal_subview = yes
	
	target_titles = none
	target_de_jure_regions_above = yes
	use_de_jure_wargoal_only = yes
	defender_ticking_warscore = 0.01
	attacker_ticking_warscore = 0
	defender_ticking_warscore_delay = { months = 60 }
	attacker_score_from_occupation_scale = 20 # multiply 1/5
	defender_score_from_occupation_scale = 20 # multiply 1/5 
	max_defender_score_from_occupation = 50
	max_attacker_score_from_occupation = 150
	imprisonment_by_attacker_give_war_score = no
	imprisonment_by_defender_give_war_score = no
	landless_attacker_needs_armies = no

	occupation_participation_mult = 0.5
	siege_participation_mult = 1
	battle_participation_mult = 7.5
	
	ignore_effect = change_title_holder
	white_peace_possible = no
	
	transfer_behavior = transfer
	on_primary_attacker_death = invalidate
	on_primary_defender_death = inherit
	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	allow_hostages = no
	allowed_for_character = {
		has_character_modifier = final_war_cb_modifier
	}

	allowed_for_character_display_regardless = {
	}
	
	allowed_against_character = {
	}

	ai_can_target_all_titles = {
	}
	
	ai_score_mult = {
	}

	valid_to_start = {
	}

	should_invalidate = {
	}

	on_invalidated_desc = msg_invasion_war_invalidated_message
	
	on_invalidated = {
		if = {
			limit = { scope:attacker = character:POD_wraith_01 }
			scope:defender.house = { 
				every_house_member = { 
					death = {
						death_reason = death_soulforging
						killer = scope:attacker
					}
				}
			}
		}
		
		if = {
			limit = { scope:attacker = { has_character_flag = POD_quincunx_warrior } }
			scope:attacker = { remove_character_flag = POD_quincunx_warrior }
			title:e_anda = {
				every_de_jure_county = {
					limit = { faith = faith:viayasaq }
					set_county_faith = faith:liberators_of_the_wheel
				}
				every_in_de_jure_hierarchy = {
					limit = { tier = tier_kingdom }
					set_de_jure_liege_title = title:e_quincunx_vanguard
				}
			}
			scope:attacker = {
				set_primary_title_to = title:e_quincunx_vanguard
				destroy_title = title:e_anda
			}
		}
	}

	cost = {
	}
	
	on_declaration = {
		on_declared_war = yes
		
		every_player = { #Fluff for attackers/defenders
			limit = {
				scope:attacker = { has_character_flag = POD_quincunx_warrior }
				faith = { 
					OR = {
						has_doctrine = special_doctrine_against_to_yomi_wan_hostility
						has_doctrine = special_doctrine_allied_to_yomi_wan_hostility
					}
				}
			}
			trigger_event = PODquincunx.251
		}
	}	

	on_victory_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:attacker = { is_local_player = yes } }
				desc = POD_event_war_victory_desc_attacker
			}
			desc = POD_event_war_victory_desc
		}
	}
	
	on_victory = {
		scope:attacker = { show_pow_release_message_effect = yes }

		#EP2 Accolade glory gain from winning against higher ranked enemy
		scope:attacker = { accolade_attacker_war_end_glory_gain_med_effect = yes }
		
		if = {
			limit = { scope:attacker = { has_character_modifier = final_war_cb_modifier } }
			scope:defender = { 
				every_realm_county = { set_county_faith = scope:attacker.faith }
				every_vassal_or_below = {
					every_courtier_or_guest = {
						limit = { has_POD_resurrection_trigger = no }
						death = {
							death_reason = death_sacrificed_to_gods
							killer = scope:attacker
						}
					}
					if = {
						limit = { has_POD_resurrection_trigger = no }
						if = {
							limit = { primary_title.tier > tier_barony } # Less spam
							death = {
								death_reason = death_sacrificed_to_gods
								killer = scope:attacker
							}
						}
						else = {
							set_character_faith = scope:attacker.faith
						}
					}
				}
				every_courtier_or_guest = {
					limit = { has_POD_resurrection_trigger = no }
					death = {
						death_reason = death_sacrificed_to_gods
						killer = scope:attacker
					}
				}
			}
		}
		
		if = {
			limit = { scope:attacker = character:POD_wraith_01 }
			scope:defender.house = { 
				every_house_member = { 
					death = {
						death_reason = death_soulforging
						killer = scope:attacker
					}
				}
			}
		}
		else = {
			create_title_and_vassal_change = {
				type = conquest
				save_scope_as = change
				add_claim_on_loss = yes
			}
			scope:defender = {
				every_held_title = {
					limit = { NOT = { tier = tier_barony } }
					change_title_holder = {
						holder = scope:attacker
						change = scope:change
						take_baronies = yes
					}
				}
				every_vassal = {
					limit = { NOT = { primary_title.tier = tier_barony } }
					every_held_title = {
						limit = { NOT = { tier = tier_barony } }
						change_title_holder = {
							holder = scope:attacker
							change = scope:change
							take_baronies = yes
						}
					}
					every_vassal = {
						limit = { NOT = { primary_title.tier = tier_barony } }
						every_held_title = {
							limit = { NOT = { tier = tier_barony } }
							change_title_holder = {
								holder = scope:attacker
								change = scope:change
								take_baronies = yes
							}
						}
						every_vassal = {
							limit = { NOT = { primary_title.tier = tier_barony } }
							every_held_title = {
								limit = { NOT = { tier = tier_barony } }
								change_title_holder = {
									holder = scope:attacker
									change = scope:change
									take_baronies = yes
								}
							}
						}
					}
				}
			}
			resolve_title_and_vassal_change = scope:change
		}
		
		if = {
			limit = { scope:attacker = { has_character_flag = POD_quincunx_warrior } }
			scope:attacker = { 
				remove_character_flag = POD_quincunx_warrior 
				add_character_flag = POD_quincunx_avenger
			}
			title:e_anda = {
				every_de_jure_county = {
					limit = { faith = faith:viayasaq }
					set_county_faith = faith:liberators_of_the_wheel
				}
				every_in_de_jure_hierarchy = {
					limit = { tier = tier_kingdom }
					set_de_jure_liege_title = title:e_quincunx_vanguard
				}
			}
			scope:attacker = {
				set_primary_title_to = title:e_quincunx_vanguard
				destroy_title = title:e_anda
			}
		}
		if = {
			limit = { scope:attacker = { has_character_flag = POD_quincunx_avenger } }
			scope:attacker = { trigger_event = PODquincunx.300 }
		}
		
		if = {
			limit = { scope:attacker = { has_character_modifier = POD_war_fervor_modifier } }
			scope:attacker = { 
				trigger_event = { 
					id = PODinquisition.1000 
					days = { 3 10 }
				}
			}
		}

		# FP1: note the victory for future memorialisation via stele (if applicable).
		scope:attacker = { fp1_remember_recent_conquest_victory_effect = yes }
	}

	on_white_peace_desc = {
		desc = WHITE_PEACE_DISABLED
	}

	on_defeat_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = invasion_war_white_peace_desc_defender
			}
			desc = invasion_war_white_peace_desc
		}
	}
	
	on_defeat = {
		scope:attacker = { show_pow_release_message_effect = yes }

		#EP2 accolade glory gain for winning against higher ranked enemy
		scope:defender = { accolade_defender_war_end_glory_gain_med_effect = yes }
		
		scope:attacker = {
			if = {
				limit = { has_character_modifier = final_war_cb_modifier }
				depose_effect = { DEPOSER = scope:defender }
			}
		}
		scope:defender = { 
			if = {
				limit = { scope:attacker = title:e_stygia.holder }
				trigger_event = pod1230objectives.117
			}
		}
	}
}
