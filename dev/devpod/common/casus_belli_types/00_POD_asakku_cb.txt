######################################################
# Tier 1 - The Phalanx (Depth 25-74)
# Elioud vs infected characters
######################################################

POD_asakku_shadow_crusade_phalanx_cb = {
	group = religious
	icon = excommunication
	interface_priority = 80

	war_name = "POD_asakku_PHALANX_WAR_NAME"
	war_name_base = "POD_asakku_PHALANX_WAR_NAME"
	cb_name = "POD_asakku_PHALANX_CB_NAME"

	combine_into_one = no
	should_show_war_goal_subview = yes
	target_titles = none
	ai_only_against_neighbors = no

	attacker_score_from_occupation_scale = 50
	defender_score_from_occupation_scale = 50
	attacker_score_from_battles_scale = 200
	defender_score_from_battles_scale = 200

	on_primary_attacker_death = invalidate
	on_primary_defender_death = invalidate
	attacker_allies_inherit = yes

	allow_hostages = no

	cost = {
		piety = {
			add = {
				value = 200
				desc = CB_BASE_COST
			}
		}
	}

	allowed_for_character = {
		scope:attacker = {
			has_trait = POD_asakku_elioud
		}
	}

	allowed_against_character = {
		scope:defender = {
			POD_asakku_has_asakku_trigger = yes
		}
	}

	valid_to_start = {
		always = yes
	}

	should_invalidate = {
		OR = {
			scope:attacker = { is_alive = no }
			scope:defender = { is_alive = no }
			scope:defender = {
				NOT = { POD_asakku_has_asakku_trigger = yes }
			}
		}
	}

	on_victory = {
		# Imprison defender
		scope:attacker = {
			if = {
				limit = { is_imprisoned = no }
				imprison = {
					target = scope:defender
					type = house_arrest
				}
			}
		}
		# Remove Vicissitude Taint from defender's counties
		scope:defender = {
			every_held_title = {
				limit = { tier = tier_county }
				title_province = {
					if = {
						limit = { has_province_modifier = POD_asakku_vicissitude_taint }
						remove_province_modifier = POD_asakku_vicissitude_taint
					}
				}
			}
		}
		# Asakku Confirmed modifier for attacker
		scope:attacker = {
			add_character_modifier = {
				modifier = POD_asakku_asakku_confirmed
				years = 5
			}
		}
		# Hunted flag on defender
		scope:defender = {
			add_character_modifier = {
				modifier = POD_asakku_hunted
				years = 10
			}
		}
		add_truce_attacker_victory_effect = yes
	}

	on_defeat = {
		scope:defender = {
			add_character_modifier = {
				modifier = POD_asakku_shadow_crusade_repelled
				years = 5
			}
		}
		add_truce_attacker_defeat_effect = yes
	}

	on_white_peace = {
		add_truce_white_peace_effect = yes
	}

	on_victory_desc = {
		desc = POD_asakku_PHALANX_VICTORY_DESC
	}

	on_defeat_desc = {
		desc = POD_asakku_PHALANX_DEFEAT_DESC
	}

	on_white_peace_desc = {
		desc = POD_asakku_PHALANX_WHITE_PEACE_DESC
	}

	on_invalidated_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_alive = no } }
				desc = POD_asakku_invalidate_dead_message
			}
			desc = POD_asakku_invalidate_war_title
		}
	}

	ai_score = {
		value = 200
	}
		
	ai_score_mult = {
		value = 1
	}
}

# Tier 2 - Full Shadow Crusade (Depth 75+)
# Enhanced version with realm purge

	icon = excommunication
	interface_priority = 90

	war_name = "POD_asakku_FULL_CRUSADE_WAR_NAME"
	war_name_base = "POD_asakku_FULL_CRUSADE_WAR_NAME"
	cb_name = "POD_asakku_FULL_CRUSADE_CB_NAME"

	combine_into_one = no
	should_show_war_goal_subview = yes
	target_titles = none
	ai_only_against_neighbors = no

	attacker_score_from_occupation_scale = 75
	defender_score_from_occupation_scale = 50
	attacker_score_from_battles_scale = 250
	defender_score_from_battles_scale = 200

	on_primary_attacker_death = invalidate
	on_primary_defender_death = invalidate

	allow_hostages = no

		piety = {
			add = {

		scope:attacker = {
			has_trait = POD_asakku_elioud
		}
	}

	allowed_against_character = {
		scope:defender = {
			POD_asakku_depth_at_least_75_trigger = yes
		}
	}

	valid_to_start = {
		always = yes
	}

	should_invalidate = {
		OR = {
			scope:attacker = { is_alive = no }
			scope:defender = { is_alive = no }
		}
	}

	on_victory = {
		# Imprison defender
		scope:defender = {
			if = {
				limit = { is_imprisoned = no }

				}

						limit = { has_province_modifier = POD_asakku_vicissitude_taint }
						remove_province_modifier = POD_asakku_vicissitude_taint
					}
				}
			}
		}
		# Imprison all infected in defender's realm
		scope:defender = {
			every_vassal = {
				limit = { POD_asakku_has_any_asakku_trait_trigger = yes }
				imprison = {
					target = this
					type = house_arrest
				}
			}
		}
		# Attacker gets confirmed modifier
		scope:attacker = {
			add_character_modifier = {
				modifier = POD_asakku_asakku_confirmed
				years = 5
			}
		}
		# Hunted flag on defender
		scope:defender = {
			add_character_modifier = {
				modifier = POD_asakku_hunted
				years = 10
			}
		}
		add_truce_attacker_victory_effect = yes
	}

	on_defeat = {
		scope:defender = {
			add_character_modifier = {
				modifier = POD_asakku_defied_the_hand
				years = 5
			}
		}
		add_truce_attacker_defeat_effect = yes
	}

	on_white_peace = {
		add_truce_white_peace_effect = yes
	}

	on_victory_desc = {
		desc = POD_asakku_FULL_CRUSADE_VICTORY_DESC
	}

	on_defeat_desc = {
		desc = POD_asakku_FULL_CRUSADE_DEFEAT_DESC
	}

	on_white_peace_desc = {
		desc = POD_asakku_FULL_CRUSADE_WHITE_PEACE_DESC
	}

	on_invalidated_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_alive = no } }

			}
			desc = POD_asakku_invalidate_war_title
		}
	}
	ai_score = {
		value = 300
	}
		
	ai_score_mult = {
		value = 1
	}
}
