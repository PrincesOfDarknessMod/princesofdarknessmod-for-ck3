POD_masquerade_war = {
	group = invasion
	interface_priority = 50
	
	war_name = "POD_MASQUERADE_WAR_WAR_NAME"
	war_name_base = "POD_MASQUERADE_WAR_WAR_NAME"
	cb_name = "POD_MASQUERADE_WAR_CB_NAME"

	combine_into_one = yes
	should_show_war_goal_subview = yes
	
	target_titles = none
	ai_only_against_neighbors = yes
	
	attacker_score_from_occupation_scale = 50
	defender_score_from_occupation_scale = 50
	attacker_score_from_battles_scale = 200
	defender_score_from_battles_scale = 200
	
	on_primary_attacker_death = invalidate
	on_primary_defender_death = invalidate
	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	allow_hostages = no
	allowed_for_character = {
		OR = {
			has_trait = supehunter
			has_trait = vampire
		}
		trigger_if = {
			limit = { has_trait = vampire }
			faith = { has_doctrine_parameter = masquerade_war_cb_enabled }
		}	
	}
	
	allowed_for_character_display_regardless = {
		#trigger_if = {
		#	limit = { has_trait = vampire }
		#	faith = { has_doctrine_parameter = masquerade_war_cb_enabled }
		#}
	}
	
	allowed_against_character = {
		scope:defender = {
			has_trait = vampire
		}
		
		scope:defender = {
			trigger_if = {
				limit = { NOT = { has_character_modifier = masquerade5_modifier } }
				has_character_modifier = masquerade4_modifier
			}
			trigger_if = {
				limit = { NOT = { has_character_modifier = masquerade4_modifier } }
				has_character_modifier = masquerade5_modifier
			}
			trigger_if = {
				limit = { 
					exists = liege
					exists = scope:attacker.liege
					liege = scope:attacker.liege
				}
				NOT = { liege.faith = { has_doctrine_parameter = vampirethemasquerade_none } }
			}
		}		
	}
	
	allowed_against_character_display_regardless = {
	}
	
	cost = {
		prestige = {
			add = {
				value = 100
				desc = CB_BASE_COST
			}
			multiply = common_cb_piety_cost_multiplier
		}
	}

	valid_to_start = {
		always = yes
	}

	should_invalidate = {
		scope:defender = {
			OR = {
				has_character_modifier = masquerade0_modifier
				has_character_modifier = masquerade1_modifier
			}
		}
	}

	on_invalidated_desc = msg_POD_masquerade_war_invalidated_message
	
	on_invalidated = {
		scope:attacker = {
			#Used for the piety_change scripted effect.
			save_scope_as = winner
		}

		#Handle all piety gains/losses in scripted effects.
		excommunication_cb_piety_change = {
			WINNER = scope:attacker
			LOSER = scope:defender
		}

		scope:defender = {
			#Used for the piety_change scripted effect.
			save_scope_as = loser
		}
	}
	
	on_declaration = {
		on_declared_war = yes
	}	

	on_victory_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = POD_masquerade_war_victory_desc_defender
			}
			desc = POD_masquerade_war_victory_desc
		}
	}
	
	on_victory = {
		scope:attacker = { show_pow_release_message_effect = yes }

		#EP2 Accolade glory gain from winning against higher ranked enemy
		scope:attacker = { accolade_attacker_war_end_glory_gain_med_effect = yes }
		
		excommunication_cb_piety_change = {
			WINNER = scope:attacker
			LOSER = scope:defender
		}
		
		scope:defender = {
			every_vassal_or_below = {
				limit = {
					OR = {
						has_character_modifier = masquerade4_modifier
						has_character_modifier = masquerade5_modifier
					}
				}
				depose_effect = { DEPOSER = scope:attacker }
			}
			depose_effect = { DEPOSER = scope:attacker }
		}

		# Truce
		add_truce_attacker_victory_effect = yes
	}

	on_white_peace_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:defender = { is_local_player = yes }
				}
				desc = excommunication_war_white_peace_desc_defender
			}
			desc = excommunication_war_white_peace_desc
		}
	}

	on_white_peace = {
		scope:attacker = { show_pow_release_message_effect = yes }

		#EP2 Accolade glory gain for defender for being ok
		scope:defender = { accolade_defender_war_end_glory_gain_low_effect = yes }
		
		scope:attacker = {
			add_piety = excommunication_cb_piety_white_peace
			stress_impact = {
				ambitious = medium_stress_impact_gain
				arrogant = medium_stress_impact_gain
			}
		}

		scope:defender = {
			stress_impact = {
				arrogant = medium_stress_impact_gain
			}
		}

		# Truce
		add_truce_white_peace_effect = yes
	}

	on_defeat_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:attacker = { is_local_player = yes }
				}
				desc = POD_masquerade_war_defeat_desc_attacker
			}
			triggered_desc = {
				trigger = {
					scope:defender = { is_local_player = yes }
				}
				desc = POD_masquerade_war_defeat_desc_defender
			}
			desc = POD_masquerade_war_defeat_desc
		}
	}
	
	on_defeat = {
		scope:attacker = { show_pow_release_message_effect = yes }

		#EP2 Accolade glory gain for winning against higher ranked enemy
		scope:defender = { accolade_defender_war_end_glory_gain_med_effect = yes }
		
		#Handle all piety gains/losses in scripted effects.
		excommunication_cb_piety_change = {
			LOSER = scope:attacker
			WINNER = scope:defender
		}

		# Truce
		add_truce_attacker_defeat_effect = yes
	}
	
	ai_score = {
		value = raid_for_captives_ai_score_value
	}
		
	ai_score_mult = {
		value = 1
	}
}
