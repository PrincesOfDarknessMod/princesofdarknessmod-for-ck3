	icon = peasant_war
	
	ai_only_against_liege = yes
	target_titles = none
	allowed_for_character =	{
		scope:attacker = {
			is_leading_faction_type = hunter_faction
		}
	}

	target_de_jure_regions_above = yes
	target_top_liege_if_outside_realm = no

	valid_to_start = {
	}

	should_invalidate = {
		NOT = {
			scope:attacker = {
				has_variable = rebel_leader_hunters
				exists = joined_faction
				joined_faction = var:rebel_leader_hunters
			}
		}
	}

	on_declaration = {
		#on_declared_war = yes

		scope:defender = {
			add_opinion = {
				modifier = peasant_revolt_opinion
				target = scope:attacker
			}
		}
	}

	on_victory_desc = {
		desc = hunter_war_victory_desc
	}

	on_victory = {
	
		scope:defender = {
			add_prestige_level = -1
			trigger_event = pod_factions.1103
			
			
			if = {
				limit = { has_realm_law = crown_authority_1 }
				add_realm_law = crown_authority_0
			}
			if = {
				limit = { has_realm_law = crown_authority_2 }
				add_realm_law = crown_authority_1
			}
			if = {
				limit = { has_realm_law = crown_authority_3 }
				add_realm_law = crown_authority_2
			}			
		}	
		
		scope:attacker = {
		
		trigger_event = pod_factions.1102
		
		}

				# TARGET_TITLE = scope:target_title				
				# }
			# }
		#}

	}

	on_white_peace_desc = {
		desc = hunter_war_white_peace_desc
	}

	on_white_peace = {
		scope:attacker = {
			joined_faction = {
				destroy_faction = yes
			}
		}
	}

	on_defeat_desc = {
		desc = hunter_war_defeat_desc
	}

	on_defeat = {
		scope:attacker = {
			add_character_flag = hunter_revolt_do_not_kill
			remove_variable = rebel_leader_hunters
			joined_faction = {
				every_faction_county_member = {
					custom = hunter_faction_every_county
					add_county_modifier = {
						modifier = county_increased_opinion_modifier
						years = 10
					}
				}
				destroy_faction = yes
			}
		}
		scope:defender = {
			if = {
				limit = {
					NOT = { #In case defender has changed
						has_opinion_modifier = {
							modifier = peasant_revolt_opinion
							target = scope:attacker
						}
					}
				}
				add_opinion = {
					modifier = peasant_revolt_opinion
					target = scope:attacker
				}
			}
			if = {
				limit = {
					scope:attacker = { is_imprisoned = no }
				}
				imprison = {
					target = scope:attacker
					type = dungeon
				}
			}
		}
	}

	on_invalidated_desc = msg_invalidate_war_title

	on_primary_attacker_death = invalidate
	on_primary_defender_death = inherit

	transfer_behavior = transfer

	attacker_allies_inherit = no
	defender_allies_inherit = yes

	war_name = "HUNTER_WAR_NAME"

	interface_priority = 80

	use_de_jure_wargoal_only = yes

	attacker_wargoal_percentage = 0.8

	max_attacker_score_from_battles = 100
	max_defender_score_from_battles = 50

	max_ai_diplo_distance_to_title = 500
}
