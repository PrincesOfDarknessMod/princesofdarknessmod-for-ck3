POD_blood_hunt_war = {
	group = invasion
	icon = excommunication
	interface_priority = 50
	
	war_name = "POD_BLOOD_HUNT_WAR_WAR_NAME"
	war_name_base = "POD_BLOOD_HUNT_WAR_WAR_NAME"
	cb_name = "POD_BLOOD_HUNT_WAR_CB_NAME"

	combine_into_one = yes
	should_show_war_goal_subview = yes
	
	target_titles = none
	ai_only_against_neighbors = yes
	
	attacker_score_from_occupation_scale = 50
	defender_score_from_occupation_scale = 50
	attacker_score_from_battles_scale = 200
	defender_score_from_battles_scale = 200
	
	on_primary_attacker_death = invalidate
	on_primary_defender_death = invalidate
	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	allow_hostages = no
	allowed_for_character = {
		scope:attacker = { religion = religion:camarilla_religion }
	}
	
	allowed_for_character_display_regardless = {
		scope:attacker = { NOT = { has_trait = anathema } }
	}
	
	allowed_against_character = {
		scope:defender.camarilla_status < 7
	}
	
	allowed_against_character_display_regardless = {
		scope:defender = { is_imprisoned = no }
		scope:attacker = {
			trigger_if = {
				limit = { camarilla_status >= 7 }
				POD_has_higher_status_trigger = { CHARACTER = scope:defender }
				trigger_if = {
					limit = { NOT = { scope:defender = { has_trait = anathema } } }
					NOT = { scope:defender = { has_hook = scope:attacker } }
				}
			}
			trigger_else = {
				scope:defender = { has_trait = anathema }
			}
		}
	}
	
	cost = {
		prestige = {
			add = {
				value = 100
				desc = CB_BASE_COST
			}
			multiply = common_cb_piety_cost_multiplier
			if = {
				limit = { scope:defender = { has_trait = anathema } }
				multiply = {
					value = 0
					desc = trait_anathema
				}
			}
			if = {
				limit = { scope:defender = { religion = religion:camarilla_religion } }
				multiply = {
					value = scope:defender.camarilla_status
					desc = camarilla_religion
				}
			}
		}
	}

	valid_to_start = {
		always = yes
	}

	should_invalidate = {
		scope:defender.camarilla_status >= 7
	}

	on_invalidated_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_alive = no } }
				desc = msg_POD_blood_hunt_war_invalidated_hunted_unavailable_dead_message
			}
			triggered_desc = {
				trigger = { scope:defender.camarilla_status >= 7 }
				desc = msg_POD_blood_hunt_war_invalidated_hunted_justicar_message
			}
			desc = msg_POD_blood_hunt_war_invalidated_message
		}
	}
	
	on_invalidated = {
		scope:attacker = {
			#Used for the piety_change scripted effect.
			save_scope_as = winner
		}

		#Handle all piety gains/losses in scripted effects.
		excommunication_cb_piety_change = {
			WINNER = scope:attacker
			LOSER = scope:defender
		}

		scope:defender = {
			#Used for the piety_change scripted effect.
			save_scope_as = loser
		}
	}
	
	on_declaration = {
		on_declared_war = yes
		scope:attacker = { trigger_event = POD_camarilla.110 }
	}	

	on_victory_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = POD_blood_hunt_war_victory_desc_defender
			}
			desc = POD_blood_hunt_war_victory_desc
		}
	}
	
	on_victory = {
		scope:attacker = { show_pow_release_message_effect = yes }
		
		excommunication_cb_piety_change = {
			WINNER = scope:attacker
			LOSER = scope:defender
		}
		
		scope:defender = {
			if = { # Imprison them if they aren't imprisoned.
				limit = { is_imprisoned = no }
				add_character_flag = { flag = lost_duel days = 3 }
				hard_imprison_character_effect = {
					TARGET = this
					IMPRISONER = scope:attacker
				}
			}
			
			every_vassal_or_below = {
				limit = { is_imprisoned = no }
				add_character_flag = { flag = lost_duel days = 3 }
				hard_imprison_character_effect = {
					TARGET = this
					IMPRISONER = scope:attacker
				}
			}
		}

		# Truce
		add_truce_attacker_victory_effect = yes
	}

	on_white_peace_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:defender = { is_local_player = yes }
				}
				desc = POD_blood_hunt_war_defeat_desc_defender
			}
			desc = POD_blood_hunt_war_defeat_desc
		}
	}

	on_white_peace = {
		scope:attacker = { show_pow_release_message_effect = yes }
		scope:attacker = {
			add_piety = excommunication_cb_piety_white_peace
			stress_impact = {
				ambitious = medium_stress_impact_gain
				arrogant = medium_stress_impact_gain
			}
		}

		scope:defender = {
			stress_impact = {
				arrogant = medium_stress_impact_gain
			}
		}

		# Truce
		add_truce_white_peace_effect = yes
	}

	on_defeat_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:attacker = { is_local_player = yes }
				}
				desc = POD_blood_hunt_war_defeat_desc_attacker
			}
			triggered_desc = {
				trigger = {
					scope:defender = { is_local_player = yes }
				}
				desc = POD_blood_hunt_war_defeat_desc_defender
			}
			desc = POD_blood_hunt_war_defeat_desc
		}
	}
	
	on_defeat = {
		scope:attacker = { show_pow_release_message_effect = yes }
		#Handle all piety gains/losses in scripted effects.
		excommunication_cb_piety_change = {
			LOSER = scope:attacker
			WINNER = scope:defender
		}

		# Truce
		add_truce_attacker_defeat_effect = yes
	}
	
	ai_score = {
		value = raid_for_captives_ai_score_value
	}
		
	ai_score_mult = {
		value = 1
	}
}
