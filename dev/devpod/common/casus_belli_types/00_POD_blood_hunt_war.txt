POD_blood_hunt_war = {
	group = invasion
	icon = excommunication
	interface_priority = 50
	
	war_name = "POD_BLOOD_HUNT_WAR_WAR_NAME"
	war_name_base = "POD_BLOOD_HUNT_WAR_WAR_NAME"
	cb_name = "POD_BLOOD_HUNT_WAR_CB_NAME"

	combine_into_one = yes
	should_show_war_goal_subview = yes
	
	target_titles = none
	ai_only_against_neighbors = yes
	
	attacker_score_from_occupation_scale = 50
	defender_score_from_occupation_scale = 50
	attacker_score_from_battles_scale = 200
	defender_score_from_battles_scale = 200
	
	on_primary_attacker_death = invalidate
	on_primary_defender_death = invalidate
	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	allow_hostages = no
	allowed_for_character = {
		scope:attacker = { religion = religion:camarilla_religion }
	}
	
	allowed_for_character_display_regardless = {
		scope:attacker = { POD_is_anathema_trigger = no }
	}
	
	allowed_against_character = {
		scope:defender.camarilla_status < 7
	}
	
	allowed_against_character_display_regardless = {
		scope:defender = { is_imprisoned = no }
		scope:attacker = { scope:defender = { POD_is_anathema_trigger = yes } }
	}
	
	cost = {
		prestige = {
			add = {
				value = 100
				desc = CB_BASE_COST
			}
			multiply = common_cb_piety_cost_multiplier
		}
	}

	valid_to_start = {
		always = yes
	}

	should_invalidate = {
		scope:defender = { POD_is_anathema_trigger = no }
	}

	on_invalidated_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_alive = no } }
				desc = msg_POD_blood_hunt_war_invalidated_hunted_unavailable_dead_message
			}
			triggered_desc = {
				trigger = { scope:defender = { POD_is_anathema_trigger = no } }
				desc = msg_POD_blood_hunt_war_invalidated_hunted_no_anathema_message
			}
			desc = msg_invalidate_war_title
		}
	}
	
	on_invalidated = {
		scope:attacker = {
			#Used for the piety_change scripted effect.
			save_scope_as = winner
		}

		#Handle all piety gains/losses in scripted effects.
		excommunication_cb_piety_change = {
			WINNER = scope:attacker
			LOSER = scope:defender
		}

		scope:defender = {
			#Used for the piety_change scripted effect.
			save_scope_as = loser
		}
	}
	
	on_declaration = {
		on_declared_war = yes
		scope:attacker = { trigger_event = POD_camarilla.110 }
	}	

	on_victory_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = POD_blood_hunt_war_victory_desc_defender
			}
			desc = POD_blood_hunt_war_victory_desc
		}
	}
	
	on_victory = {
		scope:attacker = { show_pow_release_message_effect = yes }

		#EP2 Accolade glory gain from winning against higher ranked enemy
		scope:attacker = { accolade_attacker_war_end_glory_gain_med_effect = yes }
		
		excommunication_cb_piety_change = {
			WINNER = scope:attacker
			LOSER = scope:defender
		}
		
		scope:defender = {
			if = { # Imprison them if they aren't imprisoned.
				limit = { is_imprisoned = no }
				add_character_flag = { flag = lost_duel days = 3 }
				hard_imprison_character_effect = {
					TARGET = this
					IMPRISONER = scope:attacker
				}
			}
			
			every_vassal_or_below = {
				limit = { is_imprisoned = no }
				add_character_flag = { flag = lost_duel days = 3 }
				hard_imprison_character_effect = {
					TARGET = this
					IMPRISONER = scope:attacker
				}
			}
		}

		# Truce
		add_truce_attacker_victory_effect = yes
	}

	on_white_peace_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:defender = { is_local_player = yes }
				}
				desc = POD_blood_hunt_war_defeat_desc_defender
			}
			desc = POD_blood_hunt_war_defeat_desc
		}
	}

	on_white_peace = {
		scope:attacker = { show_pow_release_message_effect = yes }

		#EP2 Accolade glory gain for defender for being ok
		scope:defender = { accolade_defender_war_end_glory_gain_low_effect = yes }
		
		scope:attacker = {
			add_piety = excommunication_cb_piety_white_peace
			stress_impact = {
				ambitious = medium_stress_impact_gain
				arrogant = medium_stress_impact_gain
			}
		}

		scope:defender = {
			stress_impact = {
				arrogant = medium_stress_impact_gain
			}
		}

		# Truce
		add_truce_white_peace_effect = yes
	}

	on_defeat_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:attacker = { is_local_player = yes }
				}
				desc = POD_blood_hunt_war_defeat_desc_attacker
			}
			triggered_desc = {
				trigger = {
					scope:defender = { is_local_player = yes }
				}
				desc = POD_blood_hunt_war_defeat_desc_defender
			}
			desc = POD_blood_hunt_war_defeat_desc
		}
	}
	
	on_defeat = {
		scope:attacker = { show_pow_release_message_effect = yes }

		#EP2 Accolade glory gain for winning against higher ranked enemy
		scope:defender = { accolade_defender_war_end_glory_gain_med_effect = yes }
		
		#Handle all piety gains/losses in scripted effects.
		excommunication_cb_piety_change = {
			LOSER = scope:attacker
			WINNER = scope:defender
		}

		# Truce
		add_truce_attacker_defeat_effect = yes
	}
	
	ai_score = {
		value = raid_for_captives_ai_score_value
	}
		
	ai_score_mult = {
		value = 1
	}
}

	icon = ducal_conquest_cb
	group = religious_disorganised
	ai_only_against_neighbors = yes
	allow_hostages = no

	# scope:attacker is the attacker
	# scope:defender is the defender
	allowed_for_character_display_regardless = {
		custom_tooltip = {
			text = roadofmetamorphosis_war_cooldown
			NOT = {	scope:attacker = { has_character_flag = roadofmetamorphosis_war_cooldown } }
		}
	}
	
	allowed_against_character = {
		scope:attacker = {
			faith = { has_doctrine_parameter = roadofmetamorphosis_war_cb_active }
		}

		scope:defender = {
			is_independent_ruler = yes
			trigger_if = {
				limit = { scope:attacker = { is_ai = no } } # Already pre-filtered by the AI due to ai_only_against_neighbors = yes. any_neighboring_top_liege_realm_owner is expensive
				scope:attacker = {
					any_neighboring_top_liege_realm_owner = {
						this = scope:defender
					}
				}
			}
		}
	}

	cost = {
		piety = {
			add = {
				value = 100
				desc = CB_BASE_COST
			}
			multiply = common_cb_piety_cost_multiplier
	
			#Dynasty Salubri Perk
			if = {
				limit = {
					scope:attacker = {
						has_dynasty = yes
						dynasty = {
							has_dynasty_perk = salubri_legacy_5
						}
					}
				}
				multiply = {
					value = salubri_legacy_5_discount
					desc = CB_ATTACKER_DYNASTY_SALUBRI
				}
			}
			
			#Dynasty Banu Haqim Perk
			if = {
				limit = {
					scope:attacker = {
						has_dynasty = yes
						dynasty = {
							has_dynasty_perk = banuhaqim_legacy_5
						}
					}
				}
				multiply = {
					value = banuhaqim_legacy_5_discount
					desc = CB_ATTACKER_DYNASTY_BANUHAQIM
				}
			}
		}	
	}

	should_invalidate = {
		OR = {
			AND = {	#A formal flowery war only invalidates if neither side is an interested participant.
				scope:attacker.faith = {
					NOT = { has_doctrine_parameter = roadofmetamorphosis_war_cb_active }
				}
				scope:defender.faith = {
					NOT = { has_doctrine_parameter = roadofmetamorphosis_war_cb_active }
				}
			}
		}
	}

	on_invalidated_desc = {
		desc = flowery_war_cb_ended_invalid.desc_no_sacrificers
