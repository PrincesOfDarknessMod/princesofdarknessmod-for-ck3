POD_bridge_war = {
	group = invasion
	interface_priority = 50
	
	war_name = "POD_BRIDGE_WAR_WAR_NAME"
	war_name_base = "POD_BRIDGE_WAR_WAR_NAME"
	cb_name = "POD_BRIDGE_WAR_CB_NAME"

	combine_into_one = yes
	should_show_war_goal_subview = yes
	
	target_titles = none
	ai_only_against_neighbors = yes
	
	attacker_score_from_occupation_scale = 50
	defender_score_from_occupation_scale = 50
	attacker_score_from_battles_scale = 200
	defender_score_from_battles_scale = 200
	
	on_primary_attacker_death = invalidate
	on_primary_defender_death = invalidate
	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	allow_hostages = no
	allowed_for_character = {
		OR = {
			has_title = title:k_underbridge
			has_title = title:e_underbridge
		}
	}
	
	allowed_for_character_display_regardless = {
	}
	
	allowed_against_character = {
		scope:defender = {
			any_sub_realm_county = {
				is_riverside_county = yes 
			}
		}
	}
	
	allowed_against_character_display_regardless = {
	}
	
	cost = {
		prestige = {
			add = {
				value = 100
				desc = CB_BASE_COST
			}
			multiply = common_cb_piety_cost_multiplier
		}
	}

	valid_to_start = {
		always = yes
	}

	should_invalidate = {
		scope:defender = {
			NOT = {
				any_sub_realm_county = {
					is_riverside_county = yes 
				}
			}
		}
	}

	on_invalidated_desc = msg_POD_bridge_war_invalidated_message
	
	on_invalidated = {
		scope:attacker = {
			#Used for the piety_change scripted effect.
			save_scope_as = winner
		}

		scope:defender = {
			#Used for the piety_change scripted effect.
			save_scope_as = loser
		}
	}
	
	on_declaration = {
		on_declared_war = yes
	}	

	on_victory_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = POD_bridge_war_victory_desc_defender
			}
			desc = POD_bridge_war_victory_desc
		}
	}
	
	on_victory = {
		scope:attacker = { show_pow_release_message_effect = yes }
		
		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = change
			add_claim_on_loss = yes
		}
		
		scope:defender = {
			every_sub_realm_county = {
				limit = { 
					is_riverside_county = yes 
				}
				if = {
					limit = {
						tier = scope:defender.primary_title.tier
					}
					change_title_holder = {
						holder = scope:attacker
						change = scope:change
					}
				}
				else = {
					add_to_list = target_title_counties
				}
				if = {
					limit = {
						NOT = { duchy = title:d_underbridge }
						scope:attacker = { has_title = title:e_underbridge }
					}
					set_de_jure_liege_title = title:d_underbridge
				}
			}
			
			every_in_list = {
				list = target_title_counties
				change_title_holder = {
					holder = scope:attacker
					change = scope:change
				}
			}
		}
		
		resolve_title_and_vassal_change = scope:change

		# Truce
		add_truce_attacker_victory_effect = yes
	}

	on_white_peace_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = invasion_war_white_peace_desc_defender
			}
			desc = invasion_war_white_peace_desc
		}
	}

	on_white_peace = {
		scope:attacker = { show_pow_release_message_effect = yes }
		scope:attacker = {
			stress_impact = {
				ambitious = medium_stress_impact_gain
				arrogant = medium_stress_impact_gain
			}
		}

		scope:defender = {
			stress_impact = {
				arrogant = medium_stress_impact_gain
			}
		}

		# Truce
		add_truce_white_peace_effect = yes
	}

	on_defeat_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:attacker = { is_local_player = yes }
				}
				desc = POD_bridge_war_defeat_desc_attacker
			}
			triggered_desc = {
				trigger = {
					scope:defender = { is_local_player = yes }
				}
				desc = POD_bridge_war_defeat_desc_defender
			}
			desc = POD_bridge_war_defeat_desc
		}
	}
	
	on_defeat = {
		scope:attacker = { show_pow_release_message_effect = yes }

		# Truce
		add_truce_attacker_defeat_effect = yes
	}
	
	ai_score = {
		value = raid_for_captives_ai_score_value
	}
		
	ai_score_mult = {
		value = 1
	}
}
