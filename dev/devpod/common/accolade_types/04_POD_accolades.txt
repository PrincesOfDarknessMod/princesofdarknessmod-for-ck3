vampire_lackey_attribute = {
	adjective = accolade_vampire_lackey
	noun = vampire_lackey_attribute
	#grouping accolades
	accolade_categories = { common personality villainous martial dreadful disliked amoral forceful aggressive }

	#founding and successor requirements
	potential = {
		prowess >= accolade_prowess_requirement_value
		OR = {
			has_trait = sadistic
			has_trait = callous
			has_trait = vengeful
		}
		is_vampire_trigger = yes
	}
	weight = {
		value = accolade_uninteresting_base_value

		if = {
			limit = {
				scope:owner ?= {
					OR = {
						has_trait = sadistic
						has_trait = callous
						has_trait = vengeful
						dread > 40
					}
				}
			}
			add = 50
		}
		if = {
			limit = {
				OR = {
					has_trait = wrathful
					has_trait = aggressive_attacker
				}
			}
			add = 25
		}
		if = {
			limit = {
				dread > 20
			}
			add = 50
		}
		#weight for opinion
		if = {
			limit = {
				exists = liege
			}
			multiply = accolade_opinion_multiplier_value
		}

		if = {
			limit = {
				scope:owner ?= {
					OR = {
						has_trait = compassionate
						has_trait = forgiving
						has_trait = generous
						culture = {
							has_cultural_tradition = tradition_charitable
						}
						faith = {
							OR = {
								has_doctrine = tenet_pacifism
								has_doctrine = tenet_dharmic_pacifism
							}
						}
					}
				}
			}
			multiply = 0
		}
	}

	#accolade ranks from Glory XP
	ranks = {
		# glory_requirement =  { modifiers and unlocks }
		200 = { # rank 1
			accolade_parameters = {
				accolade_imprisonment_chance
			}
		}
		600 = { # rank 2
			accolade_parameters = {
				accolade_imprisonment_chance
			}
		}
		1200 = { # rank 3
			accolade_parameters = {
				accolade_imprisonment_chance
			}
			liege_modifier = {
				monthly_prestige_gain_per_dread_mult = 0.001
			}			
		}
		2000 = { # rank 4
			accolade_parameters = {
				accolade_imprisonment_chance
			}		
			liege_modifier = {
				monthly_prestige_gain_per_dread_mult = 0.002
			}			
		}
			
		3000 = { # rank 5
			accolade_parameters = {
				accolade_imprisonment_chance
			}
			liege_modifier = {
				monthly_prestige_gain_per_dread_mult = 0.005
			}
		}
		4200 = { # rank 6
			accolade_parameters = {

blood_hound_attribute = {

	noun = blood_hound_attribute
	accolade_categories = { skilled skill courtly aggressive amoral intrigue }

	potential = {
		prowess >= accolade_prowess_requirement_value
		OR = {
			intrigue >= @skill_accolade_qualifying_value
			has_trait = schemer
		}
		is_vampire_trigger = yes
	}
	
	weight = {
		# These are less interesting, so we weight them down for the AI.
		value = accolade_standard_base_value
		# Scope over to the liege to see if they even want this accolade.
		scope:owner ?= {
			# Skills.
			add = {
				value = intrigue
				multiply = 10
			}
			# Culture.
			## Courtly ethos.
			if = {
				limit = {
					culture ?= {
						OR = {
							has_cultural_pillar = ethos_courtly
							has_cultural_pillar = ethos_communal
						}
					}
				}
				add = 200
			}
			## Era.
			add = accolade_weighted_up_in_later_eras_value
			# AI values.
			add = {
				value = ai_vengefulness
				multiply = 0.75
			}
			add = {
				value = ai_honor
				multiply = -1
			}
			# Traits.
			## Additives.
			### +++ Deceitful.
			if = {
				limit = { has_trait = deceitful }
				add = @accolade_trait_additive_plus_3_value
			}
			### +++ Paranoid
			if = {
				limit = { has_trait = paranoid }
				add = @accolade_trait_additive_plus_3_value

				limit = { has_trait = education_intrigue }

				limit = { has_trait = sadistic }

			### - Just.
			if = {
				limit = { has_trait = just }
				add = @accolade_trait_additive_minus_1_value
			}
			## Factor 0s.
			### Honest.
			if = {
				limit = { has_trait = honest }
				multiply = 0
			}

			limit = {
				exists = liege
