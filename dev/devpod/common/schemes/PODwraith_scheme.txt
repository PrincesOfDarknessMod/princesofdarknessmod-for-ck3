PODenthrall = {
	skill = diplomacy
	uses_agents = no
	is_secret = yes
	power_per_skill_point = 20
	spymaster_power_per_skill_point = 0	
	resistance_per_skill_point = 0
	spymaster_resistance_per_skill_point = 0
	uses_resistance = no
	minimum_progress_chance = 95
	maximum_progress_chance = 99
	icon = icon_scheme_personal
	desc = PODenthrall_desc_general
	
	allow = {
	}
	
	valid = {
		
	}

	base_success_chance = {
		base = 50
		#SCHEME OWNER#
		compare_modifier = {
			desc = PODentrance_my_diplomacy
			target = scope:owner
			value = diplomacy
			multiplier = 2
		}
	}


	on_ready = {
		scope:owner = {
			send_interface_toast = {
				title = PODenthrall_success_toast
				left_icon = scope:target
				right_icon = scope:owner
					
				add_hook = {
					type = enthrall_hook
					target = scope:target
				}
				scope:target = {
					add_opinion = {
						modifier = love_opinion
						target = scope:owner
						opinion = 25
					}
				}
			}
		}
	}


	on_invalidated = {
	}

	on_monthly = {
		save_scope_as = scheme
		scheme_owner = { save_scope_as = owner }
		scheme_target = { save_scope_as = target }
	}
	
	success_desc = "PODenthrall_SUCCESS_DESC"
}
