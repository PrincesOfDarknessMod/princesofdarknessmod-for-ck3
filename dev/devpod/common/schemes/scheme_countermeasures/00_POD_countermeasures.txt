all_seeing_eye_t1 = {
	frame = "gfx/interface/icons/scheme_countermeasure_types/frame_bronze.dds"
	
	is_shown = {
		scheme_countermeasure_access_select_best_tier_trigger = {
			COUNTERMEASURE = all_seeing_eye
			TIER = 1
		}
		OR = {
			has_perk = eyes_of_the_beast_perk
		}
	}

	is_valid_showing_failures_only = {
		# You can only set them for your home court.
		can_set_own_countermeasures_trigger = yes
	}

	on_activate = {
		inform_player_of_relevant_countermeasure_activated_effect = { COUNTERMEASURE = all_seeing_eye }
	}

	owner_modifier = {
	}

	ai_will_do = { 
		add = countermeasure_ai_will_do_prio2
	}

	parameters = {
		# Check State parameters
		has_any_countermeasure_active = yes
		has_preemptive_countermeasure_active = yes
		# Effect parameters
		secrecy_vs_all_schemes_bonus_minor = yes
		#characters_more_likely_to_accept_agenthood_against_if_honest = yes
	}
}

#Physical active countermeasure, against "direct" schemes
supernatural_high_alert_t1 = {
	frame = "gfx/interface/icons/scheme_countermeasure_types/frame_bronze.dds"
	
	is_shown = {
		scheme_countermeasure_access_select_best_tier_trigger = {
			COUNTERMEASURE = supernatural_high_alert
			TIER = 1
		}
		OR = {
			has_perk = sense_the_beast_perk
			has_perk = taming_the_beast_perk
		}
	}

	is_valid_showing_failures_only = {
		# You can only set them for your home court.
		can_set_own_countermeasures_trigger = yes
	}

	on_activate = {
		inform_player_of_relevant_countermeasure_activated_effect = { COUNTERMEASURE = supernatural_high_alert }
	}

	owner_modifier = {
		stress_gain_mult = 0.25
	}

	ai_will_do = {
	if = {
		limit = {
			any_targeting_scheme = {
				is_scheme_exposed = yes
				var:apply_countermeasures ?= flag:calculating
			}
		}
		add = 100
	}
	if = {
		limit = {
			any_in_list = {
				variable = host_should_prevent_murder_schemes_list
				host ?= prev
			}
		}
		add = countermeasure_ai_will_do_prio4
	}	
	}

	parameters = {
		# Check State parameters
		has_any_countermeasure_active = yes
		has_aggressive_countermeasure_active = yes
		# Effect parameters
		success_chance_vs_calculated_schemes_bonus_medium = yes
	}
}
