infernalize_scheme = {
	skill = intrigue
	uses_agents = yes
	hostile = yes
	
	power_per_skill_point = 5.0
	resistance_per_skill_point = 5.0
	power_per_agent_skill_point = 0.5
	tier_resistance = 1
	
	minimum_progress_chance = 5
	maximum_progress_chance = 95
	minimum_success = 0
	maximum_secrecy = 95
	maximum_success = 95
	
	success_desc = "INFERNALIZE_SCHEME_SUCCESS_DESC"
	discovery_desc = "INFERNALIZE_DISCOVERY_DESC"
	agent_leave_threshold = -25
	is_secret = yes
	base_secrecy = 10
	icon = icon_scheme_hostile
	desc = infernalize_scheme_desc_general
	
	
	allow = { 
		OR = {
			POD_is_infernalist_trigger = yes
			has_perk = kneel_before_my_throne_perk
		}
		is_ai = no
	}
	valid = {
		
		
		scope:target = {
			NOR = {
				any_secret = { secret_type = secret_infernalist }
				has_trait = infernalist
				has_trait = demon
			}
			OR = {
				exists = location
				in_diplomatic_range = scope:owner
			}
		}
	}
	
	agent_join_chance = {
		base = 0
		
		# Base Reluctance, including negative AI Energy
		modifier = {
			add = {
				value = -20
				if = {
					limit = {
						ai_energy < 0
					}
					add = {
						value = ai_energy
						multiply = 0.5
					}
				}
			}
			desc = BASE_RELUCTANCE
		}

		#Is Owner sending a gift in the invite interaction?
		modifier = {
			add = bribe_acceptance_value
			exists = scope:gift
			always = scope:gift
			NOT = {
				OR = {
					scope:target = { is_parent_of = root } 
					root = { is_parent_of = scope:target } 
				}
			}
			desc = "SCHEME_WITH_GIFT"
		}


		modifier = {
			add = 30
			POD_is_infernalist_trigger = yes
		}
	}
	
	base_success_chance = {
		base = 10
		
		modifier = {
			add = -10
			scope:owner = { has_trait = infernalist }
			
		}
		compatibility_modifier = {
			who = scope:target
			compatibility_target = scope:owner
			min = -50
			max = 50
			multiplier = 2
		}
		
	}
	
	agent_success_chance = {
		base = 0
		
		hostile_scheme_agent_success_chance_modifier = yes #Adds a value depending on their "closeness" to target
	}
	
	on_monthly = {
		hostile_scheme_discovery_chance_effect = yes

		if = {
			limit = {
				NOT = { exists = scope:discovery_event_happening }
			}			
			scheme_owner = {
				trigger_event = {
					on_action = PODgeneric_intrigue_ongoing
					days = { 1 30 }
				}
			}
		}
	}
	
	on_start = {
		save_scope_as = scheme
		scheme_owner = { save_scope_as = owner }
		scheme_target = { save_scope_as = target }
		
	}
	
	
	on_ready = {
		scheme_owner = { 
			save_scope_as = actor 
			scope:target = { save_scope_as = recipient }
			trigger_event = POD_daimonion_events.6000
		}
		scope:scheme = {
			end_scheme = yes
		}

	}

}
