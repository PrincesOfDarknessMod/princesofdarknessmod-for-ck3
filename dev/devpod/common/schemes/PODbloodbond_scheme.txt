PODbloodbond = {
	
	skill = intrigue
	hostile = yes
	
	cooldown = { months = 6 }
	uses_agents = yes
	is_secret = yes
	icon = icon_scheme_hostile
	desc = PODbloodbond_desc_general
	
	power_per_skill_point = 2.0
	resistance_per_skill_point = 3
	power_per_agent_skill_point = 1
	
	minimum_progress_chance = 5
	maximum_progress_chance = 90
	minimum_success = 5
	maximum_secrecy = 95
	maximum_success = 95
	
	allow = {
		is_vampire_trigger = yes
		is_adult = yes
		scope:target = { 
			is_adult = yes 
			 has_POD_vamp_blood_trigger = yes
		}
	}
	
	valid = {
		
	}
	
	valid_agent = {
		age >= 14
	}
	
	base_success_chance = {
		base = 0
		
		hostile_scheme_base_chance_modifier = yes
	}
	
	agent_success_chance = {
		base = 0
		
		hostile_scheme_agent_success_chance_modifier = yes #Adds a value depending on their "closeness" to target
	}
	
	agent_leave_threshold = -25
	base_success_chance = {
		base = 5
		#SCHEME OWNER#
		
		compare_modifier = {
			desc = PODbloodbond_my_intrigue
			target = scope:owner
			value = intrigue
			multiplier = 3
		}
	}

	on_ready = {
		scheme_target = {
			trigger_event = POD_bloodbond.100
		}
	}

	on_invalidated = {
	}

	on_monthly = {
		save_scope_as = scheme
		scheme_owner = { save_scope_as = owner }
		scheme_target = { save_scope_as = target }

		hostile_scheme_discovery_chance_effect = yes
		if = {
			limit = {
				NOT = { exists = scope:discovery_event_happening }
			}			
			scheme_owner = {
				trigger_event = {
					on_action = PODgeneric_intrigue_ongoing
					days = { 1 30 }
				}
			}
		}
		#TODO_CD: Lifestyle gain?
	}
	success_desc = "PODBLOODBOND_SUCCESS_DESC"
	discovery_desc = "GENERIC_DISCOVERY_DESC"
	
}
