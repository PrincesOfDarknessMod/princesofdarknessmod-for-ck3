POD_ghoulify_mortal = {
	
	skill = diplomacy
	desc = POD_ghoulify_mortal_desc_general
	success_desc = "POD_GHOULIFY_MORTAL_SUCCESS_DESC"	
	discovery_desc = "MURDER_DISCOVERY_DESC"
	icon = icon_scheme_hostile
	illustration = "gfx/interface/illustrations/event_scenes/pod_abstract1.dds"
	category = hostile
	target_type = character
	is_secret = yes

	# Parameters
	speed_per_skill_point = t3_spsp_owner_value
	speed_per_target_skill_point = t1_spsp_target_value
	base_progress_goal = t3_base_phase_length_value
	maximum_secrecy = 95
	base_maximum_success = t3_base_max_success_value
	phases_per_agent_charge = 1
	success_chance_growth_per_skill_point = t3_scgpsp_value
	
	allow = {
		is_vampire_trigger = yes
		is_adult = yes
		scope:target = {
			is_adult = yes 
			POD_can_be_ghouled_scheme_trigger = yes
			NOT = {
				has_relation_ghoul_domitor = scope:owner
				# has_opinion_modifier ={
					# target = scope:owner
					# modifier = POD_ghoulthrall_opinion
				# }
			}
		}
	}
	
	valid = {
		custom_description = {
			object = scope:target
			text = POD_ghoulify_mortal_interaction_limit
			scope:owner = {
				num_of_relation_ghoul_thrall < max_ghoul_cap
			}	
		}
	}

	
	# Agents
	agent_leave_threshold = -25	
	
	agent_join_chance = {
		base = 0
		
		ai_agent_join_chance_basic_suite_modifier = yes
		ai_agent_join_chance_hostile_grievous_modifier = yes
	}
	agent_groups_owner_perspective = { courtiers guests scripted_relations }
	agent_groups_target_character_perspective = { courtiers vassals }
	valid_agent = { is_valid_agent_standard_trigger = yes }

	odds_prediction = {
		add = hostile_scheme_base_odds_prediction_target_is_char_value
		add = odds_skill_contribution_diplomacy_value
		add = agent_groups_owner_perspective_value
		add = agent_groups_target_character_perspective_value
		min = 0
	}
	
	base_success_chance = {
		base = 50
		scheme_type_skill_success_chance_modifier = { SKILL = DIPLOMACY }
		
		hostile_scheme_base_chance_modifier = yes

		# Countermeasures.
		apply_calculated_scheme_success_chance_adjustments_modifier = yes
	}
	
	base_secrecy = {
		add = secrecy_base_value
		add = countermeasure_apply_secrecy_maluses_value
	}

	on_start = {
		if = {
			limit = {
				scheme_owner = {
					NOT = { has_variable = agents_added }
				}
			}
			add_agent_slot = agent_intermediary
			add_agent_slot = agent_proponent
			add_agent_slot = agent_courtesan
			add_agent_slot = agent_infiltrator
			add_agent_slot = agent_muscle
		}
		else = {
			scheme_owner = { remove_variable = agents_added }
		}
	}

	on_phase_completed = {
		scheme_owner = { save_scope_as = owner }
		scope:target = { save_scope_as = target }
		scheme_owner = {
			trigger_event = POD_ghoulify.100
		}
	}


	on_invalidated = {
	}

	on_monthly = {
		save_scope_as = scheme
		scheme_owner = { save_scope_as = owner }
		scope:target = { save_scope_as = target }

		#TODO_CD: Lifestyle gain?
	}
}

	icon = icon_scheme_hostile
	illustration = "gfx/interface/illustrations/event_scenes/pod_abstract1.dds"
	category = hostile
	target_type = character
	is_secret = yes

	# Parameters
	speed_per_skill_point = t3_spsp_owner_value
	speed_per_target_skill_point = t1_spsp_target_value
	base_progress_goal = t3_base_phase_length_value
	maximum_secrecy = 95
	base_maximum_success = t3_base_max_success_value
	phases_per_agent_charge = 1
	success_chance_growth_per_skill_point = t3_scgpsp_value
	
	allow = {
		POD_is_any_kind_of_ghoul_trigger = yes
		is_adult = yes
		scope:target = {
			is_adult = yes
			POD_can_be_ghouled_scheme_trigger = yes
		}
	}
	
	valid = {
		trigger_if = {
			limit = { POD_is_supehunter_trigger = no }
			scope:target = { POD_is_supehunter_trigger = no }
		}
		custom_description = {
			object = scope:target
			text = I_have_no_bloodbag
			scope:owner = {
			any_prisoner = {
				POD_is_vampire_trigger = yes
				}
			}
		}
	}
	# Agents
	agent_leave_threshold = -25	
	
	agent_join_chance = {
		base = 0
		
		ai_agent_join_chance_basic_suite_modifier = yes
		ai_agent_join_chance_hostile_grievous_modifier = yes
	}
	agent_groups_owner_perspective = { courtiers guests scripted_relations }
	agent_groups_target_character_perspective = { courtiers vassals }
	valid_agent = { is_valid_agent_standard_trigger = yes }

	odds_prediction = {
		add = hostile_scheme_base_odds_prediction_target_is_char_value
		add = odds_skill_contribution_diplomacy_value
		add = agent_groups_owner_perspective_value
		add = agent_groups_target_character_perspective_value
		min = 0
	}
	
	base_success_chance = {
		base = 50
		scheme_type_skill_success_chance_modifier = { SKILL = DIPLOMACY }
		
		hostile_scheme_base_chance_modifier = yes

		# Countermeasures.
		apply_calculated_scheme_success_chance_adjustments_modifier = yes
	}
	
	base_secrecy = {
		add = secrecy_base_value
		add = countermeasure_apply_secrecy_maluses_value
	}

	on_start = {
		if = {
			limit = {
				scheme_owner = {
					NOT = { has_variable = agents_added }
				}
			}
			add_agent_slot = agent_intermediary
			add_agent_slot = agent_proponent
			add_agent_slot = agent_courtesan
			add_agent_slot = agent_infiltrator
			add_agent_slot = agent_muscle
		}
		else = {
			scheme_owner = { remove_variable = agents_added }
		}
	}

	on_phase_completed = {
		scheme_owner = { 
			save_scope_as = owner 
			var:bloodbag = { 
				if = { 
					limit = { is_alive = yes }
					save_scope_as = bloodbag 
				}
			}
			remove_variable = bloodbag
		}
		scope:target = { save_scope_as = target }
		scheme_owner = {
			trigger_event = POD_ghoulify.200
		}
	}


	on_invalidated = {
	}

	on_monthly = {
		save_scope_as = scheme
		scheme_owner = { save_scope_as = owner }
		scope:target = { save_scope_as = target }
		#TODO_CD: Lifestyle gain?
	}
}
