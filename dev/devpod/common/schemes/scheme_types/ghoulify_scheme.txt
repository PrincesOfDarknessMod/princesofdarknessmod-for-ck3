POD_ghoulify_mortal = {
	skill = diplomacy
	desc = POD_ghoulify_mortal_desc_general
	success_desc = "POD_GHOULIFY_MORTAL_SUCCESS_DESC"	
	discovery_desc = "MURDER_DISCOVERY_DESC"
	icon = icon_scheme_vampire
	illustration = "gfx/interface/illustrations/event_scenes/pod_abstract1.dds"
	category = personal
	target_type = character
	is_secret = no
	is_basic = yes
	
	# Parameters
	success_chance_growth_per_skill_point = 0
	speed_per_skill_point = 0
	spymaster_speed_per_skill_point = 0
	target_spymaster_speed_per_skill_point = 0
	tier_speed = 0
	uses_resistance = no
	base_progress_goal = 365
	base_maximum_success = 100
	minimum_success = 100
	#minimum_secrecy = 100
	
	allow = {
		POD_splat_can_create_ghouls_naturally = yes
		is_adult = yes
		scope:target = {
			is_adult = yes 
			trigger_if = {
				limit = { scope:owner = { has_trait = azhi_dahaka } }
				POD_can_be_ghouled_azhi_dahaka_scheme_trigger = yes
				trigger_if = {
					limit = { has_trait = dharma }
					custom_tooltip = {
						text = POD_ghoulify_mortal_interaction_kj_dharma
						has_trait_rank = { trait = dharma rank <= 5 }
					}
				}
			}
			trigger_else = { POD_can_be_ghouled_scheme_trigger = yes }
			NOT = {
				has_relation_ghoul_domitor = scope:owner
				# has_opinion_modifier ={
					# target = scope:owner
					# modifier = POD_ghoulthrall_opinion
				# }
			}
		}
	}
	
	valid = {
		custom_description = {
			object = scope:target
			text = POD_ghoulify_mortal_interaction_limit
			scope:owner = {
				num_of_relation_ghoul_thrall < max_ghoul_cap
			}	
		}
	}

	odds_prediction = {
		add = 100
	}

	base_success_chance = {
		base = 100
	}

	# base_secrecy = {
	# 	add = 100
	# }

	on_phase_completed = {
		scheme_owner = { save_scope_as = owner }
		scope:target = { save_scope_as = target }
		scheme_owner = {
			trigger_event = POD_ghoulify.100
		}
	}


	on_invalidated = {
	}

	on_monthly = {
		save_scope_as = scheme
		scheme_owner = { save_scope_as = owner }
		scope:target = { save_scope_as = target }

		#TODO_CD: Lifestyle gain?
	}
}

POD_ghoul_ghoulify = {
	skill = diplomacy
	desc = POD_ghoul_ghoulify_desc_general
	success_desc = "POD_GHOULIFY_SUCCESS_DESC"	
	discovery_desc = "MURDER_DISCOVERY_DESC"
	icon = icon_scheme_vampire
	illustration = "gfx/interface/illustrations/event_scenes/pod_abstract1.dds"
	category = personal
	target_type = character
	is_secret = no
	is_basic = yes
	
	# Parameters
	success_chance_growth_per_skill_point = 0
	speed_per_skill_point = 0
	spymaster_speed_per_skill_point = 0
	target_spymaster_speed_per_skill_point = 0
	tier_speed = 0
	uses_resistance = no
	base_progress_goal = 365
	base_maximum_success = 100
	minimum_success = 100
	#minimum_secrecy = 100
	
	allow = {
		POD_is_any_kind_of_ghoul_trigger = yes
		is_adult = yes
		scope:target = {
			is_adult = yes
			POD_can_be_ghouled_scheme_trigger = yes
		}
	}
	
	valid = {
		trigger_if = {
			limit = { POD_is_supehunter_trigger = no }
			scope:target = { POD_is_supehunter_trigger = no }
		}
		custom_description = {
			object = scope:target
			text = I_have_no_bloodbag
			scope:owner = {
				any_prisoner = {
					POD_is_vampire_or_abomination_trigger = yes
				}
			}
		}
	}

	odds_prediction = {
		add = 100
	}

	base_success_chance = {
		base = 100
	}

	# base_secrecy = {
	# 	add = 100
	# }

	on_phase_completed = {
		scheme_owner = { 
			save_scope_as = owner 
			var:bloodbag = { 
				if = { 
					limit = { is_alive = yes }
					save_scope_as = bloodbag 
				}
			}
			remove_variable = bloodbag
		}
		scope:target = { save_scope_as = target }
		scheme_owner = {
			trigger_event = POD_ghoulify.200
		}
	}


	on_invalidated = {
	}

	on_monthly = {
		save_scope_as = scheme
		scheme_owner = { save_scope_as = owner }
		scope:target = { save_scope_as = target }
		#TODO_CD: Lifestyle gain?
	}
}
