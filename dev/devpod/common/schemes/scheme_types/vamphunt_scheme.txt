vamphunt = {

	skill = intrigue
	#uses_agents = yes
	icon = icon_scheme_hostile
	desc = vamphunt_desc_general
	success_desc = "VAMPHUNT_SUCCESS_DESC"
	discovery_desc = "VAMPHUNT_DISCOVERY_DESC"
	illustration = "gfx/interface/illustrations/event_scenes/corridor.dds"
	category = hostile
	target_type = character
	is_secret = yes

	# Parameters
	speed_per_skill_point = t1_spsp_owner_value
	speed_per_target_skill_point = t3_spsp_target_value
	base_progress_goal = t1_base_phase_length_value
	maximum_secrecy = 95
	base_maximum_success = t1_base_max_success_value
	phases_per_agent_charge = 1
	success_chance_growth_per_skill_point = t1_scgpsp_value
	
	allow = {
		is_vampire_trigger = yes
		is_adult = yes
		is_imprisoned = no
		scope:target = { 
			is_adult = yes 
			POD_can_be_vamphunted_scheme_trigger = yes
			is_imprisoned = no
			is_landed = no
			NOT = { has_character_modifier = slightbloodloss_modifier }
			NOT = { has_character_modifier = smallbloodloss_modifier }
			NOT = { has_character_modifier = majorbloodloss_modifier }
		}
		#scope:target.location = scope.owner.location
	}
	
	valid = {
	}

	on_start = {
		if = {
			limit = {
				scheme_owner = {
					NOT = { has_variable = agents_added }
				}
			}
			add_agent_slot = agent_assassin
			add_agent_slot = agent_muscle
			add_agent_slot = agent_thug
			add_agent_slot = agent_infiltrator
			add_agent_slot = agent_ambusher
		}
		else = {
			scheme_owner = { remove_variable = agents_added }
		}
	}	
	# Agents
	agent_leave_threshold = -25	
	
	agent_join_chance = {
		base = 0
		
		ai_agent_join_chance_basic_suite_modifier = yes
		ai_agent_join_chance_hostile_grievous_modifier = yes
	}
	agent_groups_owner_perspective = { courtiers guests scripted_relations }
	agent_groups_target_character_perspective = { courtiers vassals }
	valid_agent = { is_valid_agent_standard_trigger = yes }

	odds_prediction = {
		add = hostile_scheme_base_odds_prediction_target_is_char_value
		add = odds_skill_contribution_intrigue_value
		add = agent_groups_owner_perspective_value
		add = agent_groups_target_character_perspective_value
		min = 0
	}
	
	base_success_chance = {
		base = 0
		scheme_type_skill_success_chance_modifier = { SKILL = INTRIGUE }
		
		hostile_scheme_base_chance_modifier = yes

		# Countermeasures.
		apply_calculated_scheme_success_chance_adjustments_modifier = yes
	}
	
	base_secrecy = {
		add = secrecy_base_value
		add = countermeasure_apply_secrecy_maluses_value
	}


	on_phase_completed = {	
	
		scheme_owner = {
			trigger_event = POD_vamphunt.1  
		}
	}


	on_invalidated = {
		if = { #Did the character die in another way?
			limit = {
				scope:target = {
					is_alive = no
					NOT = { killer = prev.scheme_owner } 
				}
			}
			scheme_owner = {
				#trigger_event = POD_vamphunt.2
			}
		}
		#TODO_CD: Do we need additional messages here?
	}

	on_monthly = {
		#hostile_scheme_discovery_effect = yes
		if = {
			limit = {
				NOT = { exists = scope:discovery_event_happening }
			}			
			scheme_owner = {
				#trigger_event = {
					#on_action = POD_huntmortal_ongoing
				#}
			}
		}
	}
	
}
