		add = hostile_scheme_base_odds_prediction_target_is_char_value

		add = agent_groups_target_character_perspective_value

	}
	
	base_success_chance = {
		base = 20
		modifier = {
			desc = "SCHEME_LEARNING_MODIFIER"
			add = {
				value = scope:owner.learning
				subtract = scope:target.learning
			}
		}
		modifier = {
			desc = POD_scheme_advanced_creature_effigy
			scope:owner = {
				any_character_artifact = {
					has_variable = creature
					var:active_effigy = yes	
					OR = {
						var:creature = flag:artifact_princely_creature
						var:creature = flag:artifact_divine_beast
					}			
				}
			}
			add = 35
		}
		# At War
		first_valid = {
			modifier = {
				add = -50
				desc = "SCHEME_AT_WAR"
				scope:target = {
					is_at_war_with = scope:owner
				}
			}
			modifier = {
				add = -25
				scope:target = {
					is_landed = no

			}
			modifier = {
				add = -25
				scope:target = {
					is_landed = yes
					exists = liege
					liege = {
						is_at_war_with = scope:owner
					}
				}
				desc = "SCHEME_AT_WAR_WITH_LIEGE"
			}
		}
	}
	on_start = {
		scope:owner = { add_character_flag = { flag = effigy_on_task }}
	}

	on_phase_completed = {
		random = {
			chance = scope:scheme.scheme_success_chance
			save_scope_value_as = {
				name = scheme_successful
				value = yes
			}
		}
		save_scope_as = scheme
		scheme_owner = { save_scope_as = owner }
		scope:target = { save_scope_as = target }
		scheme_owner = {
			if = {
				limit = { exists = scope:scheme_successful }
				trigger_event = POD_mummy_command_effigy.500 #lifestyle success
			}
			else = {
				trigger_event = POD_mummy_command_effigy.510 #lifestyle failure
			}

	}
