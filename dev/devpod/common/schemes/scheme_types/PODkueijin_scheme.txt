
PODsoul_bridge = {
	skill = intrigue
	desc = PODsoul_bridge_desc_general
	success_desc = "PODsoul_bridge_SUCCESS_DESC"	
	discovery_desc = "GENERIC_DISCOVERY_DESC"
	icon = icon_scheme_hostile
	illustration = "gfx/interface/illustrations/event_scenes/corridor.dds"
	category = hostile
	target_type = character
	is_secret = no
	cooldown = { years = 1 }
	is_basic = yes
	
	# Parameters
	speed_per_skill_point = t2_spsp_owner_value
	speed_per_target_skill_point = t3_spsp_target_value
	success_chance_growth_per_skill_point = 0
	tier_speed = 0
	#uses_resistance = no
	base_progress_goal = 365
	base_maximum_success = 100
	minimum_success = 100
	#minimum_secrecy = 100
	
	allow = {
		is_imprisoned = no
	}
	
	valid = {
		custom_description = {
			object = scope:target
			text = "I_have_imprisoned_target"
			scope:target = {
				NOT = { imprisoner = scope:owner }
			}
		}
		scope:target = {
			OR = {
				exists = location
				in_diplomatic_range = scope:owner
			}
		}		
	}
	
	odds_prediction = {
		add = 100
	}

	base_success_chance = {
		base = 100
	}

	# base_secrecy = {
	# 	add = 100
	# }

	on_phase_completed = {
		scheme_owner = {
			trigger_event = kueijin_scheme.200
		}
	}

	on_monthly = {
		save_scope_as = scheme
		scheme_owner = { save_scope_as = owner }
		scope:target = { save_scope_as = target }

	}
	
	on_invalidated = {

	}
}
