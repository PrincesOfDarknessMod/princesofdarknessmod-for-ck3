PODmesmerize = {
	
	skill = stewardship
	# cooldown = { months = 2 }
	uses_agents = no
	is_secret = yes
	power_per_skill_point = 20
	spymaster_power_per_skill_point = 0	
	resistance_per_skill_point = 0
	spymaster_resistance_per_skill_point = 0
	uses_resistance = no
	minimum_progress_chance = 95
	maximum_progress_chance = 99
	icon = icon_scheme_personal
	desc = PODmesmerize_desc_general
	
	allow = {
		scope:owner = {
			trigger_if = {
				limit = {
					has_hook_of_type = {
						target = scope:target
						type = rationalize_hook
					}
				}
				has_perk = conditioning_perk
			}
			trigger_else_if = {
				limit = {
					has_hook_of_type = {
						target = scope:target
						type = mesmerize_hook
					}
				}
				has_perk = rationalize_perk
			}
			trigger_else = {
				has_perk = mesmerize_perk
			}
		}
	}
	
	valid = {
		
	}

	base_success_chance = {
		base = 50
		#SCHEME OWNER#
		compare_modifier = {
			desc = PODmesmerize_my_stewardship
			target = scope:owner
			value = stewardship
			multiplier = 2
		}

	}


	on_ready = {
		scheme_target = {
			trigger_event = POD_mesmerize.100 
		}
	}


	on_invalidated = {
	}

	on_monthly = {
		save_scope_as = scheme
		scheme_owner = { save_scope_as = owner }
		scheme_target = { save_scope_as = target }

		#TODO_CD: Lifestyle gain?
	}
	success_desc = "PODmesmerize_SUCCESS_DESC"
}
