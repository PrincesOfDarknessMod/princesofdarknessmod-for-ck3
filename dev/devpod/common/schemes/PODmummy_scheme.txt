POD_discover_true_name = {
	skill = learning
	hostile = yes
	
	cooldown = { months = 12 }
	uses_agents = no
	is_secret = no
	#base_secrecy = 10
	icon = icon_scheme_hostile
	desc = POD_discover_true_name_desc_general
	
	power_per_skill_point = 1.5
	resistance_per_skill_point = 1.75
	
	minimum_progress_chance = 10
	maximum_progress_chance = 90
	minimum_success = 5
	#maximum_secrecy = 95
	maximum_success = 95
	icon = icon_scheme_hostile

	success_desc = "POD_discover_true_name_SUCCESS_DESC"
	
	allow = {
		is_imprisoned = no
		OR = {
		is_amenti_trigger = yes
		has_trait = demon
		}
	}
	
	valid = {
	}

	base_success_chance = {
		base = 0
		
		#SCHEME OWNER#
		compare_modifier = {
			desc = POD_discover_true_name_my_learning
			target = scope:owner
			value = learning
			multiplier = 3
		}
		
		# Various penalties
		first_valid = {
			modifier = { # Foreign Rulers are very hard to kidnap
				add = -75
				desc = "sway_foreign_target"
				scope:target = {
					is_ruler = yes
					NOT = {
						scope:target.top_liege = scope:owner.top_liege
					}
				}
			}
			modifier = { # Foreign subjects less so
				add = -40
				desc = "sway_foreign_target"
				scope:target = {
					is_foreign_court_or_pool_guest = no
					NOT = {
						scope:target.top_liege = scope:owner.top_liege
					}
				}
			}
			modifier = { # Rulers are harder to kidnap
				add = -50
				desc = "FABRICATE_HOOK_RULER_TARGET"
				scope:target = {
					is_ruler = yes
					scope:target.top_liege = scope:owner.top_liege
				}
			}
		}

		modifier = {
			add = -50
			scope:target = {				
				has_opinion_modifier = {
					target = scope:owner
					modifier = attempted_murder_me_crime
				}
			}
			desc = HAS_ATTEMPTED_TO_MURDER_TARGET_BEFORE
		}
		
		# Spiritual Head of Faith
		modifier = {
			add = -50
			desc = "SCHEME_VS_SPIRITUAL_HOF"
			scope:target = {
				faith = scope:owner.faith
				faith = {
					has_doctrine = doctrine_spiritual_head
				}
				faith = {
					exists = religious_head
					religious_head = {
			 			this = scope:target
			 		}
				}
			}
		}

		# Sponsored Inspiration
		modifier = {
			add = -25
			desc = target_has_sponsored_inspiration
			scope:target = {
				exists = inspiration
				inspiration = { exists = inspiration_sponsor }
			}
		}
		
		# At War
		first_valid = {
			modifier = {
				add = -50
				desc = "SCHEME_AT_WAR"
				scope:target = {
					is_at_war_with = scope:owner
				}
			}
			modifier = {
				add = -25
				scope:target = {
					is_landed = no
					exists = liege
					liege = {
						is_at_war_with = scope:owner
					}
				}
				desc = "SCHEME_AT_WAR_WITH_LIEGE"
			}
			modifier = {
				add = -25
				scope:target = {
					is_landed = yes
					exists = liege
					liege = {
						is_at_war_with = scope:owner
					}
				}
				desc = "SCHEME_AT_WAR_WITH_LIEGE"
			}
		}
		
		hostile_scheme_base_chance_modifier = yes
	}

	on_ready = {
		random = {
			chance = scope:scheme.scheme_success_chance

			save_scope_value_as = {
				name = scheme_successful
				value = yes
			}
		}
		save_scope_as = scheme
		scheme_owner = { save_scope_as = owner }
		scheme_target = { save_scope_as = target }
		scope:owner = {
			if = {
				limit = {
					exists = scope:scheme_successful
				}
		
				if = {
					limit = { scope:owner = { NOT = { has_hook_of_type = { target = scope:target type = true_name_hook } } } }
					scope:owner = {
						send_interface_toast = {
							title = POD_discover_true_name_success_toast
							left_icon = scope:target
							right_icon = scope:owner
					
							POD_discover_true_name_effect = {
								MASTER = scope:owner
								VICTIM = scope:target
							}
						}
					}
				}
				if = {
					limit = { 
						scope:target = { is_ruler = yes }
						scope:owner = { has_perk = echo_the_lesser_name_perk } 
					}
					scope:target = { 
						every_courtier_or_guest = { 
							limit = { 
								NOT = { this = scope:owner }
								scope:owner = { 
									NOR = { 
										has_hook_of_type = { target = prev type = true_name_hook } 
										has_hook_of_type = { target = prev type = half_true_name_hook }
										has_hook_of_type = { target = prev type = partial_true_name_hook } 
									}
								} 
							}
							scope:owner = { 
								add_hook = {
									target = prev
									type = partial_true_name_hook
								}
							}
						}
					}
				}
				if = {
					limit = { 
						scope:target = { is_ruler = yes }
						scope:owner = { has_perk = echo_the_greater_name_perk } 
					}
					scope:target = { 
						every_vassal_or_below = { 
							limit = { 
								NOT = { this = scope:owner }
								scope:owner = { 
									NOR = { 
										has_hook_of_type = { target = prev type = true_name_hook } 
										has_hook_of_type = { target = prev type = half_true_name_hook } 
										has_hook_of_type = { target = prev type = partial_true_name_hook } 
									}
								} 
							}
							scope:owner = { 
								add_hook = {
									target = prev
									type = partial_true_name_hook
								}
							}
						}
					}
				}
				scope:scheme = {
					end_scheme = yes
				}
			}
			else = {
				scope:owner = {
					trigger_event = POD_discover_true_name_outcome.2000  #scheme failure
				}	 
			}
		}
	}
	
	on_monthly = {
		save_scope_as = scheme
		scheme_target = { save_scope_as = target }
		scheme_owner = {
			save_scope_as = owner
			trigger_event = { on_action = POD_discover_true_name_ongoing }
		}
	}
	
	on_invalidated = {

	}
}

		is_mummy_trigger = yes
	}

	valid = {
		scope:owner = {
			#exists = {
				any_character_artifact = {
						has_variable = active_effigy
						var:active_effigy = yes	
					OR = {
						has_variable = simple_creature_effigy
						has_variable = lesser_creature_effigy
						has_variable = major_creature_effigy
						has_variable = superior_creature_effigy

					}			
				}		
			#}		
		}
	}

	base_success_chance = {
		base = 75
	}

	on_start = {
		scope:scheme = { add_scheme_progress = 4 } #only want scheme to last ~6 months
		scope:owner = { add_character_flag = { flag = effigy_on_task }}
	}

	on_ready = {
		scheme_owner = {
			trigger_event = POD_mummy_command_effigy.100 #gold success
		}
	}

	on_invalidated ={
		scope:owner = {

		}
	}

		save_scope_as = scheme
		scheme_owner = { 
			save_scope_as = owner
			trigger_event = { on_action = POD_mummy_command_effigy_ongoing }
