POD_discover_true_name = {
	skill = learning
	hostile = yes
	
	cooldown = { months = 12 }
	uses_agents = no
	is_secret = no
	#base_secrecy = 10
	icon = icon_scheme_hostile
	desc = POD_discover_true_name_desc_general
	
	power_per_skill_point = 1.5
	resistance_per_skill_point = 1.75
	
	minimum_progress_chance = 10
	maximum_progress_chance = 90
	minimum_success = 5
	#maximum_secrecy = 95
	maximum_success = 95

	success_desc = "POD_discover_true_name_SUCCESS_DESC"
	
	allow = {
		is_imprisoned = no
	}
	
	valid = {
		scope:target = {
			POD_is_not_clone_trigger = yes
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
			NOT = { this = character:790320 }
		}
		POD_cross_umbra_scheme_valid_trigger = yes
	}

	base_success_chance = {
		base = 0
		
		#SCHEME OWNER#
		compare_modifier = {
			desc = POD_discover_true_name_my_learning
			target = scope:owner
			value = learning
			multiplier = 2
		}
		
		# Various penalties
		first_valid = {
			modifier = { # Foreign Rulers are very hard to kidnap
				add = -40
				desc = "sway_foreign_target"
				scope:target = {
					is_ruler = yes
					NOT = {
						scope:target.top_liege = scope:owner.top_liege
					}
				}
			}
			modifier = { # Foreign subjects less so
				add = -20
				desc = "sway_foreign_target"
				scope:target = {
					is_foreign_court_or_pool_guest = no
					NOT = {
						scope:target.top_liege = scope:owner.top_liege
					}
				}
			}
			modifier = { # Rulers are harder to kidnap
				add = -15
				desc = "FABRICATE_HOOK_RULER_TARGET"
				scope:target = {
					is_ruler = yes
					scope:target.top_liege = scope:owner.top_liege
				}
			}
		}

		modifier = {
			add = -25
			scope:target = {				
				has_opinion_modifier = {
					target = scope:owner
					modifier = attempted_murder_me_crime
				}
			}
			desc = HAS_ATTEMPTED_TO_MURDER_TARGET_BEFORE
		}
		
		modifier = {
			add = 15
			scope:target = {				
				has_trait = confider
			}
		}
		
		modifier = {
			add = 10
			scope:target = {				
				has_trait = trusting
			}
		}
		
		modifier = {
			add = 10
			scope:target = {				
				has_trait = honest
			}
		}
		
		modifier = {
			add = 10
			scope:target = {				
				has_trait = arrogant
			}
		}
		
		modifier = {
			add = 5
			scope:target = {				
				has_trait = gregarious
			}
		}
		
		modifier = {
			add = -20
			scope:target = {				
				has_trait = paranoid
			}
		}
		
		modifier = {
			add = -15
			scope:target = {				
				has_trait = reclusive
			}
		}
		
		modifier = {
			add = -15
			scope:target = {				
				has_trait = deceitful
			}
		}
		
		modifier = {
			add = -10
			scope:target = {				
				has_trait = shy
			}
		}
		
		modifier = {
			add = -10
			scope:target = {				
				has_trait = humble
			}
		}

		modifier = {
			add = -10
			scope:target = {				
				age >= 100
			}
		}

		modifier = {
			add = -10
			scope:target = {				
				age >= 250
			}
		}	

		modifier = {
			add = -10
			scope:target = {				
				age >= 500
			}
		}

		modifier = {
			add = -10
			scope:target = {				
				age >= 1000
			}
		}
	
		modifier = {
			add = -15
			desc = strong_mental_defenses_trigger
			scope:target = {				
				strong_mental_defenses_trigger = yes
			}
		}

		# At War
		first_valid = {
			modifier = {
				add = -25
				desc = "SCHEME_AT_WAR"
				scope:target = {
					is_at_war_with = scope:owner
				}
			}
			modifier = {
				add = -10
				scope:target = {
					is_landed = no
					exists = liege
					liege = {
						is_at_war_with = scope:owner
					}
				}
				desc = "SCHEME_AT_WAR_WITH_LIEGE"
			}
			modifier = {
				add = -10
				scope:target = {
					is_landed = yes
					exists = liege
					liege = {
						is_at_war_with = scope:owner
					}
				}
				desc = "SCHEME_AT_WAR_WITH_LIEGE"
			}
		}
		
		modifier = {
			add = 50
			scope:owner = {				
				has_character_modifier = umbra_demon_binding_modifier
			}
			desc = "umbra_demon_binding_modifier_scheme"
		}
		
		hostile_scheme_base_chance_modifier = yes
	}

	on_ready = {
		random = {
			chance = scope:scheme.scheme_success_chance

			save_scope_value_as = {
				name = scheme_successful
				value = yes
			}
		}
		save_scope_as = scheme
		scheme_owner = { save_scope_as = owner }
		scheme_target = { save_scope_as = target }
		scope:owner = {
			if = {
				limit = {
					exists = scope:scheme_successful
				}
		
				if = {
					limit = { scope:owner = { NOT = { has_hook_of_type = { target = scope:target type = true_name_hook } } } }
					scope:owner = {
						send_interface_toast = {
							title = POD_discover_true_name_success_toast
							left_icon = scope:target
							right_icon = scope:owner
					
							POD_discover_true_name_effect = {
								MASTER = scope:owner
								VICTIM = scope:target
							}
						}
					}
				}
				if = {
					limit = { 
						scope:target = { exists = house }
						scope:owner = { has_character_modifier = nian_shun_modifier } 
					}
					scope:target = { 
						every_close_family_member = {
							limit = { 
								NOT = { this = scope:owner }
								scope:owner = { 
									NOR = { 
										has_hook_of_type = { target = prev type = true_name_hook } 
										has_hook_of_type = { target = prev type = half_true_name_hook }
										has_hook_of_type = { target = prev type = partial_true_name_hook } 
									}
								} 
							}
							scope:owner = { 
								add_hook = {
									target = prev
									type = partial_true_name_hook
								}
							}
						}
					}
				}
				if = {
					limit = { 
						scope:target = { is_ruler = yes }
						scope:owner = { has_perk = echo_the_lesser_name_perk } 
					}
					set_while_counter_variable_effect = yes
					while = {
						limit = { var:while_counter < echo_the_lesser_name_perk_max }
						scope:target = { 
							random_courtier_or_guest = { 
								limit = {
									NOT = { scope:owner = { has_hook = prev } } 
								}
								scope:owner = { 
									add_hook = {
										target = prev
										type = partial_true_name_hook
									}
								}
							}
						}
						increase_while_counter_variable_effect = yes
					}
					remove_while_counter_variable_effect = yes
				}
				if = {
					limit = { 
						scope:target = { is_ruler = yes }
						scope:owner = { has_perk = echo_the_greater_name_perk } 
					}
					set_while_counter_variable_effect = yes
					while = {
						limit = { var:while_counter < echo_the_greater_name_perk_max }
						scope:target = { 
							random_vassal_or_below = { 
								limit = {
									NOT = { scope:owner = { has_hook = prev } } 
								}
								scope:owner = { 
									add_hook = {
										target = prev
										type = partial_true_name_hook
									}
								}
							}
						}
						increase_while_counter_variable_effect = yes
					}
					remove_while_counter_variable_effect = yes
				}
				trigger_event = POD_mummy_scheme.102
			}
			else = {
				scope:owner = {
					trigger_event = POD_discover_true_name_outcome.2000  #scheme failure
				}	 
			}
		}
	}
	
	on_monthly = {
		save_scope_as = scheme
		scheme_target = { save_scope_as = target }
		scheme_owner = {
			save_scope_as = owner
			trigger_event = { on_action = POD_discover_true_name_ongoing }
		}
	}
	
	on_invalidated = {

	}
}

POD_command_effigy_lifestyle_scheme = {
	skill = learning
	hostile = no
	uses_agents = no
	is_secret = no
	icon = icon_scheme_culture
	desc = POD_command_effigy_lifestyle_scheme_desc_general
	
	power_per_skill_point = 2
	resistance_per_skill_point = 1.25
	
	minimum_progress_chance = 50
	maximum_progress_chance = 90
	minimum_success = 20
	maximum_success = 75

	cooldown = { years = 5 }
	success_desc = POD_command_effigy_lifestyle_scheme_SUCCESS_DESC
	
	allow = {
		is_imprisoned = no
		is_mummy_trigger = yes
	}
	
	valid = {
		any_character_artifact = {
			has_variable = creature
			var:active_effigy = yes	
			OR = {
				var:creature = flag:artifact_superior_creature
				var:creature = flag:artifact_princely_creature
				var:creature = flag:artifact_kingly_beast
				var:creature = flag:artifact_divine_beast
			}			
		}
	}
	base_success_chance = {
		base = 20
		modifier = {
			desc = "SCHEME_LEARNING_MODIFIER"
			add = {
				value = scope:owner.learning
				subtract = scope:target.learning
			}
		}
		modifier = {
			desc = POD_scheme_advanced_creature_effigy
			scope:owner = {
				any_character_artifact = {
					has_variable = creature
					var:active_effigy = yes	
					OR = {
						var:creature = flag:artifact_princely_creature
						var:creature = flag:artifact_divine_beast
					}			
				}
			}
			add = 35
		}
		# At War
		first_valid = {
			modifier = {
				add = -50
				desc = "SCHEME_AT_WAR"
				scope:target = {
					is_at_war_with = scope:owner
				}
			}
			modifier = {
				add = -25
				scope:target = {
					is_landed = no
					exists = liege
					liege = {
						is_at_war_with = scope:owner
					}
				}
				desc = "SCHEME_AT_WAR_WITH_LIEGE"
			}
			modifier = {
				add = -25
				scope:target = {
					is_landed = yes
					exists = liege
					liege = {
						is_at_war_with = scope:owner
					}
				}
				desc = "SCHEME_AT_WAR_WITH_LIEGE"
			}
		}
	}
	on_start = {
		scope:owner = { add_character_flag = { flag = effigy_on_task }}
	}

	on_ready = {
		random = {
			chance = scope:scheme.scheme_success_chance
			save_scope_value_as = {
				name = scheme_successful
				value = yes
			}
		}
		save_scope_as = scheme
		scheme_owner = { save_scope_as = owner }
		scheme_target = { save_scope_as = target }
		scheme_owner = {
			if = {
				limit = { exists = scope:scheme_successful }
				trigger_event = POD_mummy_command_effigy.500 #lifestyle success
			}
			else = {
				trigger_event = POD_mummy_command_effigy.510 #lifestyle failure
			}
		}
	}

	on_invalidated = {
		scope:owner = {
			remove_character_flag = effigy_on_task
		}
	}

	on_monthly = {
		save_scope_as = scheme
		scheme_owner = { 
			save_scope_as = owner
			trigger_event = { on_action = POD_mummy_command_effigy_ongoing } 
		}
	}
