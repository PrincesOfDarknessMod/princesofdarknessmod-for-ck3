chase_bounty = {
	
	skill = prowess
	hostile = yes
	# cooldown = { months = 2 }
	uses_agents = no
	is_secret = yes
	base_secrecy = 10
	icon = icon_scheme_hostile
	desc = chase_bounty_desc_general
	
	allow = {
		is_adult = yes
		is_imprisoned = no
		web_of_knives_member_trigger = yes
		scope:target = { 
			any_in_global_list = {
				variable = global_web_of_knives_targets
				this = prev
			}
		}
		
	}
	
	valid = {
		is_imprisoned = no
		web_of_knives_member_trigger = yes
		scope:target = { 
			any_in_global_list = {
				variable = global_web_of_knives_targets
				this = prev
			}
			NOT = { has_character_flag = won_bounty_duel }
		}
	}

	base_success_chance = {
		base = 5
		#SCHEME OWNER#
		compare_modifier = {
			desc = chase_bounty_my_prowess
			target = scope:owner
			value = prowess
			multiplier = 1
		}
		
		

	}


	on_ready = {	
		scheme_target = { add_character_flag = { flag = won_bounty_duel years = 10 } }
	
		scheme_owner = {
			trigger_event = POD_web_of_knives.6
		}
	}


	on_invalidated = {
		if = { #Did the character die in another way?
			limit = {
				scheme_target = {
					is_alive = no
					NOT = { killer = prev.scheme_owner } 
				}
			}
			scheme_owner = {
				#trigger_event = POD_vamphunt.2
			}
		}
		#TODO_CD: Do we need additional messages here?
	}

	on_monthly = {
		hostile_scheme_discovery_chance_effect = yes
		if = {
			limit = {
				NOT = { exists = scope:discovery_event_happening }
			}			
			scheme_owner = {
				trigger_event = {
					on_action = PODgeneric_intrigue_ongoing
					days = { 1 30 }
				}
			}
		}
		
	}

	power_per_skill_point = 1.5
	resistance_per_skill_point = 0.5
	power_per_agent_skill_point = 0
	spymaster_power_per_skill_point = 0	
	spymaster_resistance_per_skill_point = 0
	uses_resistance = yes
	minimum_progress_chance = 5
	maximum_progress_chance = 100
	minimum_success = 5
	maximum_secrecy = 95
	maximum_success = 95
	
	success_desc = "CHASE_BOUNTY_SUCCESS_DESC"
	discovery_desc = "CHASE_BOUNTY_DISCOVERY_DESC"
	
}
