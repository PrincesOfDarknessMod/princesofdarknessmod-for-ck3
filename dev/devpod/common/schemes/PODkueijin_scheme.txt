PODsoulbond = {
	skill = intrigue
	hostile = yes
	
	cooldown = { months = 6 }
	uses_agents = no
	is_secret = yes
	base_secrecy = 10
	
	power_per_skill_point = 2.0
	resistance_per_skill_point = 1.75
	
	minimum_progress_chance = 10
	maximum_progress_chance = 90
	minimum_success = 5
	maximum_secrecy = 95
	maximum_success = 95
	icon = icon_scheme_hostile
	desc = PODsoulbond_desc_general

	success_desc = "PODsoulbond_SUCCESS_DESC"
	
	allow = {
		is_imprisoned = no
		scope:owner = { has_perk = soul_shackles_perk }
		scope:owner = { 
			NOT = { 
				has_hook_of_type = { 
					target = scope:target 
					type = soul_bond_hook 
				}
			}
		}
	}
	
	valid = {
		custom_description = {
			object = scope:target
			text = "I_have_imprisoned_target"
			scope:target = {
				NOT = { imprisoner = scope:owner }
			}
		}
		scope:target = {
			OR = {
				exists = location
				in_diplomatic_range = scope:owner
			}
		}
	}

	base_success_chance = {
		base = 50
		#SCHEME OWNER#
		
		compare_modifier = {
			desc = PODsoulbond_my_intrigue
			target = scope:owner
			value = intrigue
			multiplier = 3
		}
	}

	on_ready = {
		save_scope_as = scheme
		scheme_owner = { save_scope_as = owner }
		scheme_target = { save_scope_as = target }
		scope:target = {
			POD_soulbond_effect = {
				MASTER = scope:owner
				VICTIM = scope:target
			}
			trigger_event = POD_mummy_scheme.102
		}
		end_scheme = yes
	}

	on_monthly = {
		save_scope_as = scheme
		scheme_owner = { save_scope_as = owner }
		scheme_target = { save_scope_as = target }
	}
	
	on_invalidated = {

	}
}
