embrace = {
	
	skill = intrigue
	hostile = yes
	# cooldown = { months = 2 }
	uses_agents = no
	is_secret = yes
	power_per_skill_point = 20
	spymaster_power_per_skill_point = 0	
	resistance_per_skill_point = 0
	spymaster_resistance_per_skill_point = 0
	uses_resistance = no
	minimum_progress_chance = 95
	maximum_progress_chance = 99
	icon = icon_scheme_hostile
	desc = embrace_desc_general
	
	allow = {
		is_vampire_trigger = yes
		#is_adult = yes
		scope:target = { 
			is_adult = yes 
			POD_can_be_embraced_scheme_trigger = yes
		}
		scope:target = { is_imprisoned = no }
	}
	
	valid = {
		scope:target = { NOT = { has_character_modifier = boon_of_anubis_perk_modifier } }
	}

	base_success_chance = {
		base = 95
		#SCHEME OWNER#
		compare_modifier = {
			desc = embrace_my_intrigue
			target = scope:owner
			value = intrigue
			multiplier = 2
		}

	}


	on_ready = {
		scheme_owner = { save_scope_as = owner }
		scheme_target = { save_scope_as = target }
		if = {
			limit = {
				scheme_owner = { faith = { has_doctrine_parameter = willing_embrace } }
			}
			scheme_target = { trigger_event = POD_embrace.300 }
		}	
		else_if = {
			limit = {
				scheme_target = { POD_is_supehunter_trigger = yes }
			}
			scheme_owner = { trigger_event = POD_embrace.200 }
		}
		else_if = {
			limit = {
				scheme_target = { POD_is_werewolf_trigger = yes }
			}
			scheme_owner = { trigger_event = POD_embrace.400 }
		}
		else = {
			scheme_owner = {
				trigger_event = POD_embrace.100 
			}
		}
	}


	on_invalidated = {
	}

	on_monthly = {
		save_scope_as = scheme
		scheme_owner = { save_scope_as = owner }
		scheme_target = { save_scope_as = target }

		#TODO_CD: Lifestyle gain?
	}
	success_desc = "EMBRACE_SUCCESS_DESC"
}
