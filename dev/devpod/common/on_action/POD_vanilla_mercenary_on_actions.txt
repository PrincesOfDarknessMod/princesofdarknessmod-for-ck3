# Appending our additions to vanilla on_actions in a separate file, to keep vanilla diffs clean


# Fires when a mercenary company is hired by a character
#
# root - the mercenary company
# scope:employer - the character who has hired the mercenary company 
#
on_mercenary_company_hired = {
	on_actions = { POD_on_mercenary_company_hired }
}

POD_on_mercenary_company_hired = {
	effect = {
		if = {
			limit = { scope:employer.domicile ?= { has_domicile_building = mokole_sunforged_outfitter_06 } }
			scope:employer = { add_character_modifier = { modifier = sunforged_outfitters_modifier years = 2 } }
		}
	}
}
