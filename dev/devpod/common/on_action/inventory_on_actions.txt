# An artifact has changed ownership
# root = the artifact
# scope:owner = the person who now holds the artifact
# scope:old_owner = the person who used to hold the artifact
on_artifact_changed_owner = {
	events = {
		artifact.0011
		artifact.0101
	}

	effect = {
		save_scope_as = artifact
		grant_banner_house_claim_effect = yes
		every_artifact_claimant = { #This on_action is not triggered on succession for the ones getting the claim, so we force it here
			trigger_event = {
				on_action = on_artifact_claim_gained
			}
		}
		# if = {
		# 	limit = { 
		# 		has_variable = fetter_owner
		# 		NOT = { var:fetter_owner = scope:owner }
		# 	}
		# 	var:fetter_owner = {
		# 		if = {
		# 			limit = { 
		# 				root = { var:fetter_value > 1 }
		# 			}
		# 			send_interface_toast = {
		# 				title = fetter_neglected_notification
		# 				left_icon = this
							
		# 				POD_lose_fetter_effect = { FETTER = root }
		# 			}
		# 		}
		# 		else = {
		# 			send_interface_toast = {
		# 				title = fetter_destroyed_notification
		# 				left_icon = this
							
		# 				POD_destroy_fetter_effect = { FETTER = root }
		# 			}
		# 		}
		# 	}
		# }
		if = {
			limit = { 
				has_variable = reliquarie
				NOT = { var:reliquarie = scope:owner }
			}
			var:reliquarie = {
				save_scope_as = killed
				POD_remove_stone_prison_effect = yes
				send_interface_toast = {
					title = reliquarie_destroyed_notification
					left_icon = this
						
					remove_trait = earthbound
					save_scope_as = reliquarie_destroyed
				}
				trigger_event = POD_demon_maintenance.20
			}
		}
		if = {
			limit = { POD_artifact_is_unavailable_trigger = yes }
			destroy_artifact = this
		}
		if = {
			limit = { scope:owner = { has_trait = wraith } }
			scope:owner = { POD_pulse_chance_passion = { TYPE = greed } }
		}
		# abandon heist journeys if the player gains the artifact through other means
		scope:owner = {
			if = {
				limit = {
					var:current_journey ?= {
						var:target_artifact ?= scope:artifact
						var:journey_type = flag:artifact_heist
					}
					# hacky way to not trigger this message on a successful heist
					NOT = { has_variable = hide_heist_window }
					NOT = { has_variable = current_heist }
				}
				var:current_journey.var:travel_plan ?= {
					resume_travel_plan = yes
				}
				return_home = yes
				send_interface_message = {
					type  = POD_journey_abandoned
					title = POD_journey_abandoned_title
					desc  = POD_journey_abandoned_desc_artifact
					right_icon = scope:artifact
				}
				POD_journey_cleanup_effect = yes
			}
			every_in_list = {
				variable = paused_journeys
				limit = {
					var:target_artifact ?= scope:artifact
					var:journey_type = flag:artifact_heist
				}
				prev = {
					remove_list_variable = {
						name = paused_journeys
						target = prev
					}
					send_interface_message = {
						type  = POD_journey_abandoned
						title = POD_journey_abandoned_title
						desc  = POD_journey_abandoned_desc_artifact
						right_icon = scope:artifact
					}
				}
				end_story = yes
			}
		}
	}
}

# An artifact has changed ownership due to succession
# This is called just before the artifact changes hands.
# If you destroy it or change its owner, it won't change hands
# on_artifact_changed_owner will be called as normal when it changes hands
# root = the artifact
# scope:owner = the person who now holds the artifact
# scope:old_owner = the person who used to hold the artifact
# scope:old_primary = the heir's primary before this succession happened
on_artifact_succession = {
	events = {
		artifact.0031
	}
}

# An artifact has reached 0 durability due to decay
# root = the artifact
# scope:owner = the person who owns the artifact
on_artifact_broken_through_decay = {
	events = {
		artifact.1000
	}
}

# An artifact has reached 0 durability due to an effect ( ex. damage )
# root = the artifact
# scope:owner = the person who owns the artifact
on_artifact_broken_through_effect = {
	events = {
		artifact.0021
	}
}

# Character gained a claim on an artifact
# root = character
# scope:owner = current artifact owner
# scope:artifact = claimed artifact
on_artifact_claim_gained = {
	effect = {
		send_interface_message = {
			type = msg_artifact_claim_gained
			title = artifact_claim_gained_notification
			right_icon = scope:artifact
			left_icon = scope:owner
			
			show_as_tooltip = {
				add_personal_artifact_claim = scope:artifact
			}
		}
	}
}

# Character lost a claim on an artifact
# root = character
# scope:owner = current artifact owner
# scope:artifact = artifact
on_artifact_claim_lost = {
	effect = {
		if = {
			limit = {
				NOT = { scope:owner = root }
			}
			send_interface_message = {
				type = msg_artifact_claim_lost
				title = artifact_claim_lost_notification
				right_icon = scope:artifact
				left_icon = scope:owner
				
				show_as_tooltip = {
					remove_personal_artifact_claim = scope:artifact
				}
			}
		}
	}
}
