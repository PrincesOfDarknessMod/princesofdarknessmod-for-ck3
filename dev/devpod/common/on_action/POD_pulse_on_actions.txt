POD_player_yearly_pulse = {
	trigger = {
		is_ai = no
	}
	events = {
		POD_maintenance.100		# MAIN MAINTENANCE EVENT (effects that need RNG go here)
		POD_predatorstories.1	# initialize predator stories for vampires. only triggers once to set up a recurring event
		POD_maintenance.3		# remove battle modifiers that got stuck
	}
	effect = {
		# let player select predator type if it isn't set yet
		if = {
			limit = {
				has_trait = vampire
				has_predator_type_trigger = no 
				years_from_game_start >= 1
			}
			trigger_event = { 
				id = POD_chargen.1000
				days = 5
			}
		}

		# Vamp and Ghoul blood sorcery exp (old bloodsorcery)
		# if = {
		# 	limit = { has_bloodsorcery_trigger = yes }
		# 	POD_initiate_blood_sorcery_exp_effect = yes
		# 	trigger_event = POD_blood_sorcery_maintenance.100
		# }

		# Necromancer stress/xp
		if = {
			limit = { num_of_relation_bound_wraith >= 1 }
			if = {
				limit = { wraith_maintenance_stress_impact_gain > 0 }
				send_interface_message = {
					type = event_generic_bad
					title = POD_wraith_maintenance
					custom_tooltip = POD_wraith_maintenance.tt
					add_stress = wraith_maintenance_stress_impact_gain
				}
			}
			every_courtier = {
				trigger_event = POD_maintenance.11 
			}
		}

		# XP gain for infernalists
		POD_yearly_pulse_infernalist_xp_effect = yes

		# clone maintenance
		POD_yearly_pulse_clone_maintenance_effect = yes

		# Knight Loyalty check
		if = {
			limit = {
				years_from_game_start >= 1
				POD_is_loyalty_disabled = no
			}

			trigger_event = {
				id = pod_knights.99
				# Random delay to make the event less exploitable
				days = { 1 90 }
			}
		}
	}
}

POD_coterie_maintenance_quarterly_pulse = {
	trigger = {
		is_ai = no
		has_variable = coterie
	}
	effect = {
		save_scope_as = coterie_leader

		every_in_list = {
			variable = potential_coterie_rivals
			if = {
				limit = {
					is_alive = yes
					var:coterie ?= { this = scope:coterie_leader.var:coterie }
					opinion = {
						target = scope:coterie_leader
						value <= coterie_rivalry_opinion
					}
					is_ai = yes
				}
				POD_leave_coterie_with_rivalry_effect = { RIVAL = scope:coterie_leader }
			}
		}

		clear_variable_list = potential_coterie_rivals

		var:coterie = {
			every_in_list = {
				variable = members
				limit = {
					NOT = { this = scope:coterie_leader }
					opinion = {
						target = scope:coterie_leader
						value <= coterie_rivalry_opinion
					}
					is_ai = yes
				}
				save_scope_as = potential_rival
				scope:coterie_leader = {
					add_to_variable_list = {
						name = potential_coterie_rivals
						target = scope:potential_rival
					}
					trigger_event = POD_coterie_maintenance.30
				}
			}
		}
	}
}
