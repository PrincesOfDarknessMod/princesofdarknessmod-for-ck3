POD_player_yearly_pulse = {
	trigger = {
		is_ai = no
	}
	events = {
		POD_predatorstories.1
		POD_maintenance.3
		POD_nushi_duties.1
		POD_demon_demand.1
	}
	on_actions = {
		#Set up quarterly pulses
		delay = { days = 45 }
		POD_player_quarterly_pulse
		delay = { days = 135 }
		POD_player_quarterly_pulse
		delay = { days = 225 }
		POD_player_quarterly_pulse
		delay = { days = 315 }
		POD_player_quarterly_pulse
	}
	effect = {
		if = {
			limit = {
				has_trait = vampire
				has_predator_type_trigger = no 
				years_from_game_start >= 1
			}
			trigger_event = { 
				id = POD_chargen.1000
				days = 5
			}
		}
		if = { # Vamp and Ghoul blood sorcery exp
			limit = { has_bloodsorcery_trigger = yes }
			POD_initiate_blood_sorcery_exp_effect = yes
			trigger_event = POD_blood_sorcery_maintenance.100
		}
		if = {
			limit = { num_of_relation_bound_wraith >= 1 }
			if = {
				limit = { wraith_maintenance_stress_impact_gain > 0 }
				send_interface_message = {
					type = event_generic_bad
					title = POD_wraith_maintenance
					custom_tooltip = POD_wraith_maintenance.tt
					add_stress = wraith_maintenance_stress_impact_gain
				}
			}
			if = {
				limit = { 
					has_trait = mummy 
					hekau_necromancy_lifestyle_unlockable_perks > 0
				}
				send_interface_message = {
					type = event_generic_good
					title = POD_hekau_necromancy_maintenance
					custom_tooltip = POD_hekau_necromancy_maintenance.tt
					add_hekau_necromancy_lifestyle_xp = wraith_maintenance_hekau_necromancy_gain
				}
			}
			every_relation = {
				type = bound_wraith
				if = {
					limit = { has_trait = spectre }
					POD_remove_angst_modifiers_effect = yes
					POD_add_angst_modifiers_effect = yes
				}
				else = {
					POD_remove_pathos_modifiers_effect = yes
					POD_add_pathos_modifiers_effect = yes
				}
			}
			every_relation = {
				type = spirit_companion
				POD_remove_rage_modifiers_effect = yes
				add_character_modifier = POD_rage_5_modifier
				POD_remove_gnosis_modifiers_effect = yes
				add_character_modifier = POD_gnosis_5_modifier
			}
		}
		if = {
			limit = {
				NOT = { has_trait = demon }
				OR = {
					is_a_cultist = yes
					is_a_diabolist = yes
				}
			}
			set_local_variable = { name = demonic_investiture_gain value = 0 }
			set_local_variable = { name = diabolist_xp value = 0 }
			set_local_variable = { name = thrall_xp value = 0 }
			every_relation = {
				type = bound_demon
				save_temporary_opinion_value_as = { name = target_opinion target = root }
				change_local_variable = { name = demonic_investiture_gain add = scope:target_opinion }
				change_local_variable = { name = diabolist_xp add = 1 }
			}
			every_relation = {
				type = worshipped_demon
				save_temporary_opinion_value_as = { name = target_opinion target = root }
				change_local_variable = { name = demonic_investiture_gain add = scope:target_opinion }
				change_local_variable = { name = thrall_xp add = 1 }
			}
			send_interface_message = {
				type = event_generic_good
				title = POD_investiture_maintenance
				custom_tooltip = POD_investiture_maintenance.tt
				add_investiture_lifestyle_xp = local_var:demonic_investiture_gain
				add_diabolist_xp = { VALUE = local_var:diabolist_xp }
				add_thrall_xp = { VALUE = local_var:thrall_xp }
			}
		}
		save_scope_as = cloned_character
		every_relation = {
			type = shadow_twin_servant
			clear_traits = yes
	        copy_traits = scope:cloned_character
			POD_clone_maintenance_effect = yes
		}
		every_relation = {
			type = mirror_duplicate_servant
			clear_traits = yes
	        copy_traits = scope:cloned_character
			POD_clone_maintenance_effect = yes
		}
		every_relation = {
			type = darkling_mirror_servant
			clear_traits = yes
	        copy_traits = scope:cloned_character
			POD_clone_maintenance_effect = yes
		}
		every_relation = {
			type = shadowy_duplicate_servant
			clear_traits = yes
	        copy_traits = scope:cloned_character
			POD_clone_maintenance_effect = yes
		}
		every_relation = {
			type = phantom_shadow_servant
			clear_traits = yes
			add_trait = wraith
			add_trait = shadow
	        copy_traits = scope:cloned_character
			POD_remove_wrong_supernatural_traits_effect = yes
			POD_clone_maintenance_effect = yes
		}
		every_relation = {
			type = separated_khaibit_servant
			clear_traits = yes
			add_trait = wraith
			add_trait = shadow
	        copy_traits = scope:cloned_character
			POD_remove_mummy_traits_effect = yes
			invert_personality_effect = yes
			POD_clone_maintenance_effect = yes
		}
		every_relation = {
			type = separated_shadow_servant
			clear_traits = yes
			add_trait = wraith
			add_trait = shadow
	        copy_traits = scope:cloned_character
			invert_personality_effect = yes
			POD_clone_maintenance_effect = yes
			
			POD_set_catharsis_modifier_effect = yes
		}
		every_relation = {
			type = unspeakable_servitor_servant
			clear_traits = yes
			add_trait = automaton
	        copy_traits = scope:cloned_character
			remove_trait = demon
			POD_remove_demon_traits_effect = yes
			POD_clone_maintenance_effect = yes
		}
		# Knight Loyalty check
		if = {
			limit = {
				years_from_game_start >= 1
				POD_is_samuel_haight_mode_enabled = no
			}

			trigger_event = {
				id = pod_knights.99
				# Random delay to make the event less exploitable
				days = { 1 90 }
			}
		}
	}
}

POD_coterie_maintenance_quarterly_pulse = {
	trigger = {
		is_ai = no
		has_variable = coterie
	}
	effect = {
		save_scope_as = coterie_leader

		every_in_list = {
			variable = potential_coterie_rivals
			if = {
				limit = {
					is_alive = yes
					var:coterie ?= { this = scope:coterie_leader.var:coterie }
					opinion = {
						target = scope:coterie_leader
						value <= coterie_rivalry_opinion
					}
				}
				POD_leave_coterie_with_rivalry_effect = { RIVAL = scope:coterie_leader }
			}
		}

		clear_variable_list = potential_coterie_rivals

		var:coterie = {
			every_in_list = {
				variable = members
				limit = {
					NOT = { this = scope:coterie_leader }
					opinion = {
						target = scope:coterie_leader
						value <= coterie_rivalry_opinion
					}
				}
				save_scope_as = potential_rival
				scope:coterie_leader = {
					add_to_variable_list = {
						name = potential_coterie_rivals
						target = scope:potential_rival
					}
					trigger_event = POD_coterie_maintenance.30
				}
			}
		}
	}
}
