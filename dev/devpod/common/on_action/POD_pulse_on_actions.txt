POD_player_power_maintenance_quarterly_pulse = {
	trigger = {
		is_ai = no
	}
	effect = {
		if = {
			limit = {
				has_perk = regeneration_perk
				NOT = { has_trait_with_flag = can_not_cure }
				NOT = { has_character_modifier = eye_of_sakhmet_modifier }
			}
			remove_wound_trait_effect = yes
		}
	}
	on_actions = {
		POD_player_power_maintenance_monthly_pulse
		POD_werewolf_wound_recovery_pulse
		POD_werewolf_disease_recovery_pulse
		POD_dark_passion_maintenance_monthly_pulse

POD_player_power_maintenance_monthly_pulse = {
	trigger = {
		is_ai = no
		is_imprisoned = no
	}
	effect = {
		if = {
			limit = {
				NOT = { has_trait = kueijin  }
				has_chi_trigger = yes
				NOT = { has_character_modifier = celestial_barrier_modifier }
				NOT = { has_character_modifier = traveling_spirit_world_modifier }
			}
			if = {
				limit = {
					is_east_mummy_trigger = no
					has_yinchi_5_or_greater_trigger = no
				}
				POD_yinchi_increase_1_effect = yes
			}
			if = {
				limit = {
					NOT = { has_character_modifier = yangchi10_modifier }
					OR = {
						is_east_mummy_trigger = yes
						has_yangchi_5_or_greater_trigger = no
					}
				}
				POD_yangchi_increase_1_effect = yes
			}
		}
		if = {
			limit = {
				has_trait = shemsuheru
				NOT = { has_character_modifier = sekhem10_modifier }
				NOT = { has_character_modifier = traveling_spirit_world_modifier }
			}
			if = {
				limit = { 
					exists = location 
					location = { in_land_of_faith_trigger = yes }
				}
				POD_sekhem_increase_effect = yes
			}
			else = {
				send_interface_message = {
					type = event_generic_bad
					title = POD_shemsuheru_maintenance
					left_icon = root
					right_icon = prev
					custom_tooltip = POD_shemsuheru_maintenance.tt
					POD_willpower_rouse_check_effect = yes
					POD_sekhem_increase_effect = yes
				}
			}
		}
		if = {
			limit = {
				has_trait = cabiri
				NOT = { has_character_modifier = sekhem10_modifier }
				NOT = { has_character_modifier = traveling_spirit_world_modifier }
				has_character_flag = restore_pneuma_by_magic_flag
			}
			if = {
				limit = { capital_province = { has_building_or_higher = occult_affiliation_04 } }
			}
			else_if = {
				limit = { capital_province = { has_building = occult_affiliation_03 } }
				remove_short_term_gold = 1
			}
			else_if = {
				limit = { capital_province = { has_building = occult_affiliation_02 } }
				remove_short_term_gold = 2
			}
			else_if = {
				limit = { capital_province = { has_building = occult_affiliation_01 } }
				remove_short_term_gold = 3
			}
			else = {
				remove_short_term_gold = 4
			}
			POD_sekhem_increase_effect = yes
		}
		if = {
			limit = {
				OR = {
					has_trait = laham
					has_trait = nephilim
					is_a_thrall = yes
				}
				NOT = { has_character_modifier = resolve10_modifier }
			}
			POD_resolve_increase_effect = yes
		}
	}
}

POD_dark_passion_maintenance_monthly_pulse = {
	trigger = {
		has_trait = wraith
		NOT = { has_trait = spectre }
		NOT = { has_trait = ferryman }
		NOT = { has_trait = shadow }
	}
	effect = {
		if = {
			limit = { can_execute_decision = POD_wraith_dark_passion_start_a_hostile_scheme_decision }
			execute_decision = POD_wraith_dark_passion_start_a_hostile_scheme_decision
		}
		if = {
			limit = { can_execute_decision = POD_wraith_dark_passion_start_a_seduction_decision }
			execute_decision = POD_wraith_dark_passion_start_a_seduction_decision
		}
		if = {
			limit = { can_execute_decision = POD_wraith_dark_passion_start_a_war_decision }
			execute_decision = POD_wraith_dark_passion_start_a_war_decision
		}
		if = {
			limit = { can_execute_decision = POD_wraith_dark_passion_raise_tyranny_decision }
			execute_decision = POD_wraith_dark_passion_raise_tyranny_decision
		}
		if = {
			limit = { can_execute_decision = POD_wraith_dark_passion_exceeding_the_domain_limit_decision }
			execute_decision = POD_wraith_dark_passion_exceeding_the_domain_limit_decision
		}
		if = {
			limit = { can_execute_decision = POD_wraith_dark_passion_exceeding_the_vassal_limit_decision }
			execute_decision = POD_wraith_dark_passion_exceeding_the_vassal_limit_decision
		}
		if = {
			limit = { can_execute_decision = POD_wraith_dark_passion_blackmail_decision }
			execute_decision = POD_wraith_dark_passion_blackmail_decision
		}
		if = {
			limit = { can_execute_decision = POD_wraith_dark_passion_revoke_title_decision }
			execute_decision = POD_wraith_dark_passion_revoke_title_decision
		}

			limit = { can_execute_decision = POD_wraith_dark_passion_execute_prisoner_decision }
			execute_decision = POD_wraith_dark_passion_execute_prisoner_decision
		}
		if = {
			limit = { can_execute_decision = POD_wraith_dark_passion_banish_prisoner_decision }
			execute_decision = POD_wraith_dark_passion_banish_prisoner_decision
		}
	}
}
