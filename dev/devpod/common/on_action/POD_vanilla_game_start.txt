# Appending our additions to vanilla on_actions in a separate file, to keep vanilla diffs clean

on_game_start = {
	on_actions = { POD_on_game_start }
}

POD_on_game_start = {
	events = {
		pod_artifacts.0001			# POD Artifacts Generator	
		POD_title_flavor.0001
		POD_fae_maintenance.10
		POD_world_initialization.3
		POD_world_initialization.10
	}
	effect = {
		# Global variable for mod compatibility
		set_global_variable = {
			name = POD_loaded
			value = yes
		}
		
		if = {
			limit = { POD_barons_are_disabled = yes }
			every_county = {
				every_county_province = {
					limit = {
						has_holding = yes
						is_county_capital = no
					}
					remove_holding = yes
				}
			}
		}
		
		# Domicile setup
		every_ruler = {
			POD_create_haven_landed_domicile = yes
		}
		#Bagdhad setup
		if = {
			limit = {
				game_start_date >= 1258.1.1
			}
		title:b_baghdad.title_province = {
			add_province_modifier = POD_baghdad_sacked_rebuilt_modifier
		}
		}
		else = {
		title:b_baghdad.title_province = {
			add_province_modifier = POD_baghdad_complacency_modifier
		}
		}		
		
		if = {
			limit = {
				POD_bookmark_has_fae = yes
			}
		title:c_somerset = { add_county_modifier = red_and_white_springs }
		title:c_cardiganshire = { #Brianne
			random_list = {
				33 = { 
					add_county_modifier = albion_pool_of_youth_modifier 
					random_list = { 
						50 = { 
							title:k_brittany = { random_de_jure_county = { add_county_modifier = albion_pool_of_heal_modifier } } 
							title:k_scotland = { random_de_jure_county = { add_county_modifier = albion_pool_of_control_modifier } }
						}
						50 = { 
							title:k_brittany = { random_de_jure_county = { add_county_modifier = albion_pool_of_control_modifier } } 
							title:k_scotland = { random_de_jure_county = { add_county_modifier = albion_pool_of_heal_modifier } }
						}
					}
				}
				33 = { 
					add_county_modifier = albion_pool_of_heal_modifier
					random_list = { 
						50 = { 
							title:k_brittany = { random_de_jure_county = { add_county_modifier = albion_pool_of_youth_modifier } } 
							title:k_scotland = { random_de_jure_county = { add_county_modifier = albion_pool_of_control_modifier } }
						}
						50 = { 
							title:k_brittany = { random_de_jure_county = { add_county_modifier = albion_pool_of_control_modifier } } 
							title:k_scotland = { random_de_jure_county = { add_county_modifier = albion_pool_of_youth_modifier } }
						}
					}
				}
				33 = { 
					add_county_modifier = albion_pool_of_control_modifier 
					random_list = { 
						50 = { 
							title:k_brittany = { random_de_jure_county = { add_county_modifier = albion_pool_of_youth_modifier } } 
							title:k_scotland = { random_de_jure_county = { add_county_modifier = albion_pool_of_heal_modifier } }
						}
						50 = { 
							title:k_brittany = { random_de_jure_county = { add_county_modifier = albion_pool_of_heal_modifier } } 
							title:k_scotland = { random_de_jure_county = { add_county_modifier = albion_pool_of_youth_modifier } }
						}
					}
				}
			}
		}
		}
		
		#Flaying plague
		if = {
			limit = {
				POD_bookmark_has_flaying_plague = yes
			}
			#Happens in 1229 lore wise, but leave a little breathing room for intro stuff to WW players
			character:POD_wraith_01 = { 
				trigger_event = {
					id = POD_blackdeath.0003
					months = 6
				}
			}
		}

		#Bandaid fix for Fera not getting starting buildings if they own any land added by AuH. generate_building doesn't work either.
		if = {
			limit = { POD_bookmark_has_fera = yes }
			every_ruler = {
				limit = {
					has_government = fera_tribe_government
					OR = {
						POD_eastern_fera_splat_trigger = yes #Should grab most of them
						has_title = title:d_mukdahan #Chalawan
						has_title = title:c_aru #Wolaru
						has_title = title:c_gomoco #Yusuf
					}
				}
				if = {
					limit = { highest_held_title_tier = tier_kingdom }
					add_gold = 500
					if = {
						limit = { has_trait = mokole }
						create_maa_or_upgrade_regiment_effect = { TYPE = mokole_hatchlings SIZE = 3 }
					}
					else_if = {
						limit = { has_trait = werewolf }
						create_maa_or_upgrade_regiment_effect = { TYPE = cliath SIZE = 3 }
					}
					else_if = {
						limit = { has_trait = bastet }
						create_maa_or_upgrade_regiment_effect = { TYPE = tekhmet SIZE = 3 }
					}
					create_maa_or_upgrade_regiment_effect = { TYPE = kinfolk_militia SIZE = 2 }
					create_maa_or_upgrade_regiment_effect = { TYPE = mangonel SIZE = 1 }
				}
				else_if = {
					limit = { highest_held_title_tier = tier_duchy }
					add_gold = 300
					if = {
						limit = { has_trait = mokole }
						create_maa_or_upgrade_regiment_effect = { TYPE = mokole_hatchlings SIZE = 2 }
					}
					else_if = {
						limit = { has_faith = faith:hakken }
						create_maa_or_upgrade_regiment_effect = { TYPE = hakken_samurai_maa SIZE = 2 }
					}
					else_if = {
						limit = { religion = { is_in_family = rf_wyrm } }
						create_maa_or_upgrade_regiment_effect = { TYPE = skull_pigs SIZE = 2 }
					}
					else_if = {
						limit = { has_trait = werewolf }
						create_maa_or_upgrade_regiment_effect = { TYPE = cliath SIZE = 2 }
					}
					else_if = {
						limit = { has_trait = bastet }
						create_maa_or_upgrade_regiment_effect = { TYPE = tekhmet SIZE = 2 }
					}
					create_maa_or_upgrade_regiment_effect = { TYPE = kinfolk_militia SIZE = 1 }
					create_maa_or_upgrade_regiment_effect = { TYPE = onager SIZE = 1 }
				}
				else = {
					add_gold = 100
					if = {
						limit = { has_trait = mokole }
						create_maa_or_upgrade_regiment_effect = { TYPE = mokole_hatchlings SIZE = 1 }
						create_maa_or_upgrade_regiment_effect = { TYPE = kinfolk_militia SIZE = 1 }
					}
					else_if = {
						limit = { has_trait = werewolf }
						create_maa_or_upgrade_regiment_effect = { TYPE = cliath SIZE = 1 }
					}
					else_if = {
						limit = { has_trait = bastet }
						create_maa_or_upgrade_regiment_effect = { TYPE = tekhmet SIZE = 1 }
					}
					create_maa_or_upgrade_regiment_effect = { TYPE = kinfolk_militia SIZE = 1 }
					create_maa_or_upgrade_regiment_effect = { TYPE = onager SIZE = 1 }
				}
			}
		}
		
		POD_setup_legitimacy = yes
		POD_initialize_journey_GUI = yes
		POD_initialize_canon_coteries = yes
		POD_initialize_house_unities = yes
		POD_initialize_house_relations = yes
		POD_initialize_diarchies = yes
		POD_initialize_colorschemes = yes
		
		POD_build_plundered_constantinople_effect = yes
		trigger_event = { id = POD_maintenance.22 years = 30 }
	}
}

on_game_start_after_lobby = {
	on_actions = { POD_on_game_start_after_lobby }
}

POD_on_game_start_after_lobby = {
	effect = {
		every_ruler = {
			trigger_event = POD_maintenance.40	
		}
		
		every_player = {
			trigger_event = {
				id = POD_error_suppression.0002
				days = 1
			}
		}
		every_player = {
			trigger_event = {
				id = POD_error_suppression.0004
				days = 1
			}
		}
		every_player = {
			trigger_event = {
				id = POD_error_suppression.0005
				days = 1
			}
		}
		#every_player = {
		#	if = {
		#		limit = {
		#			highest_held_title_tier >= tier_kingdom
		#			NOT = { has_character_flag = warning_primogen }
		#			is_vampire_trigger = yes			
		#		}
		#		trigger_event =	{
		#			id = POD_primogens.0001
		#			days = 6
		#		}
		#	}
		#}
		
		#Pope setup
		if = {
			limit = {
				any_player = { faith = faith:shadowinquisition }
			}
			trigger_event = {
				id = PODask_pope.2
				years = { 10 25 } 
			}
		}		
		
		#WoK setup
		if = {
			limit = {
				any_player = { is_assamite_trigger = yes } #Technically should also check for faith but that leaves open conversions to WoK friendly faiths post start
			}
			trigger_event = {
				id = POD_web_of_knives.12
				years = { 10 25 } 
			}
		}
		
		every_player = { 
			trigger_event = { 
				on_action = POD_player_yearly_pulse # yearly on_action don't seems fire in first year
			}
		}
		
		POD_setup_bad_legitimacy = yes
		POD_setup_tributaries_effect = yes
		
		if = {
			limit = {
				NOT = { has_game_rule = pod_endgame_delay_canon }
				current_date < 1430.1.1
			}
			set_global_variable = {
				name = premature_inquisition_year
				value = {
					value = current_year
					# values are off-by-one
					# since the actual player-facing events start 1 year later
					if = {
						limit = { has_game_rule = pod_endgame_delay_2 }
						add = 1
					}
					else_if = {
						limit = { has_game_rule = pod_endgame_delay_20 }
						add = 19
					}
					else_if = {
						limit = { has_game_rule = pod_endgame_delay_50 }
						add = 49
					}
					else_if = {
						limit = { has_game_rule = pod_endgame_delay_100 }
						add = 99
					}
					else_if = {
						limit = { has_game_rule = pod_endgame_delay_random }
						add = { 1 499 }
					}
				}
			}
		}

		set_global_variable = POD_game_after_lobby
	}
	events = {
		POD_world_initialization.1
		POD_world_initialization.2
		POD_character_initialization.1001
		POD_character_initialization.10	
		POD_character_canon_modifier_initalization.1
		podintroductions.2001	
		delay = { days = 4 }
		podintroductions.10
	}
}
