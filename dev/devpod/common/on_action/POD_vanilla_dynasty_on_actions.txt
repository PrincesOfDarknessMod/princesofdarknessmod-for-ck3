


on_dynasty_created = {
	on_actions = { POD_on_dynasty_created }
}

POD_on_dynasty_created = {
	effect = {
		if = {
			limit = { dynast = { POD_mokole_has_stream_trigger = yes } }
			dynast.var:mokole_stream ?= { save_scope_as = dynasty_stream }
			dynast.dynasty = { set_variable = { name = mokole_stream value = scope:dynasty_stream } }
		}
	}
}
