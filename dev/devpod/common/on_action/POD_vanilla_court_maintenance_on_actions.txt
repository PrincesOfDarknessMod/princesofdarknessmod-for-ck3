# Appending our additions to vanilla on_actions in a separate file, to keep vanilla diffs clean

on_join_court = {
	on_actions = { POD_on_join_court }
}

POD_on_join_court = {
	effect = {
		if = {
			limit = {
				is_landed = no
				NOT = { has_character_flag = had_POD_character_initialization }
			}
			trigger_event = POD_character_initialization.1
		}
		
		if = {
			limit = {
				POD_custom_incompatible_splats_trigger = { ACTOR = root RECIPIENT = scope:new_employer }
			}
			trigger_event = { on_action = pod_leave_court days = 1 } # Delay for those apply blocked_from_leaving after
		}
	}
}

pod_leave_court = {
	trigger = {
		NOT = { has_character_flag = blocked_from_leaving }
		is_pool_character = no
		OR = {
			is_courtier = yes
			is_foreign_court_or_pool_guest = yes
		}
	}
	effect = {
		move_to_pool = yes
	}
}


on_leave_court = {
	on_actions = { POD_on_leave_court }
}

POD_on_leave_court = {
	effect = {
		# Bandaid fix for characters leaving for seemingly no reason
		if = {
			limit = {
				# DEPRECATED for now since it comes with unintended side effects
				always = no
				scope:old_employer = {
					is_ai = no
					NOT = { var:player_court_owner ?= scope:old_employer } # prevent infinite loop
					save_temporary_scope_as = liege
					prev = { courtier_allowed_to_leave_trigger = no }
				}
				is_landed = no
				NOT = {
					POD_custom_incompatible_splats_trigger = {
						ACTOR = root
						RECIPIENT = scope:old_employer
					}
				}
			}
			set_variable = {
				name = player_court_owner
				value = scope:old_employer
				days = 1
			}
			liege_or_court_owner ?= { remove_courtier_or_guest = prev }
			var:player_court_owner = { add_courtier = prev }
			return_to_court = yes
			remove_variable = player_court_owner
		}
		else = {
			if = {
				limit = { has_variable = umbra_stationed_realm }
				scope:old_employer = { POD_dismiss_umbra_explorer_effect = { TARGET = root } }
			}
		}

		if = {
			limit = { 
				scope:old_employer = { has_perk = dependence_perk } 
				POD_strong_mental_defenses_custom_trigger = { ACTOR = scope:old_employer RECIPIENT = THIS }
				is_landed = no
				any_held_title = { count = 0 }
			}
			scope:old_employer = { add_courtier = prev }
			return_to_court = yes
		}
	}
}
