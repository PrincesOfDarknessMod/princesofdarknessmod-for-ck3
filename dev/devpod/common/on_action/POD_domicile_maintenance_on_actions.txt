on_title_gain = {
	on_actions = {
		POD_on_title_gain_domicile_maintenance
	}
}

POD_on_title_gain_domicile_maintenance = {
	effect = {
		# A character becomes landed? Generate new domicile if none already exists
		if = {
			limit = {
				scope:title = { is_nomad_title = no }
				NOT = { any_held_title = { is_nomad_title = yes } }
			}
			POD_create_haven_landed_domicile = yes
		}
		# Inheriting a haven_landed domicile
		else_if = {
			limit = {
				scope:title = {
					POD_title_is_haven_landed_domicile = yes
				}
			}
			# If you're not the actual player heir of the predecessor, "redirect" the domicile
			if = {
				limit = {
					scope:title = { exists = var:proper_heir }
					NOT = { this = scope:title.var:proper_heir }
				}
				create_title_and_vassal_change = {
					type = returned
					save_scope_as = change
					add_claim_on_loss = no
				}
				scope:title = {
					change_title_holder = {
						holder = var:proper_heir
						change = scope:change
					}
				}
				resolve_title_and_vassal_change = scope:change
			}
			# Inherit the player's domicile
			else_if = {
				limit = {
					scope:title = {
						exists = var:inherited_from_player
					}
					POD_government_is_valid_for_character = {
						GOVERNMENT = scope:title.var:haven_government
					}
				}
				every_held_title = {
					limit = {
						is_nomad_title = yes
						NOT = { this = scope:title }
					}
					prev = { destroy_title = prev }
				}
				#save_scope_as = holder
				scope:title = {
					#set_title_name = POD_domicile_title
					remove_variable = proper_heir
					remove_variable = inherited_from_player
				}
				POD_save_haven_buildings = yes
			}
			# Otherwise, destroy the domicile
			else = {
				destroy_title = scope:title
			}
		}
	}
}


on_title_lost = {
	on_actions = { POD_on_title_lost_domicile_maintenance }
}

POD_on_title_lost_domicile_maintenance = {
	effect = {
		# Become courtier if losing all non-domicile titles
		if = {
			limit = {
				any_held_title = {
					count = 0
					POD_title_is_haven_landed_domicile = no
					tier = tier_county
				}
			}
			every_held_title = {
				limit = { POD_title_is_haven_landed_domicile = yes }
				prev = { destroy_title = prev }
			}
		}
	}
}


on_government_change = {
	on_actions = { POD_on_government_change }
}

POD_on_government_change = {
	effect = {
		if = {
			limit = {
				NOT = {
					any_held_title = {
						POD_title_is_haven_landed_domicile = yes
						var:haven_government ?= prev.government_type
					}
				}
				is_landless_adventurer = no
			}
			POD_refresh_domicile_on_government_change = yes
		}
		if = {
			limit = { is_landless_adventurer = no }
			POD_save_haven_buildings = yes
		}
	}
}
