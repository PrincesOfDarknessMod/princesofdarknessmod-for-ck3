# Appending our additions to vanilla on_actions in a separate file, to keep vanilla diffs clean

on_faith_created = {
	on_actions = { POD_on_faith_created }
}

POD_on_faith_created = {
	effect = {
		# Set original faith this one was based on
		if = {
			limit = {
				exists = scope:old_faith.founder
				scope:old_faith.founder = { has_variable = foundational_faith }
			}
			set_variable = {
				name = foundational_faith
				value = scope:old_faith.founder.var:foundational_faith
			}
		}
		else = {
			set_variable = {
				name = foundational_faith
				value = scope:old_faith
			}
		}
		
		POD_adopt_faith_side_effect = yes
	}
}

on_character_faith_change = {
	on_actions = { POD_on_character_faith_change }
}

POD_on_character_faith_change = {
	effect = {
		POD_adopt_faith_side_effect = yes

		#A tremere councilor converts
		if = {
			limit = {
				has_variable = influence_inner_council
			}
		trigger_event = pod_tremere_council.5
		}
	
		#An ephor converts
		if = {
			limit = {
			 is_target_in_global_variable_list = {
					 name = ventrue_ephorate_list
					 target = root
				}
			is_ai = yes		
			}
		trigger_event = pod_ventrue_ephorate.12
		}
	
		#An Camarilla member converts
		if = {
			limit = { 
				NOT = { religion = religion:camarilla_religion }
				POD_is_inner_circle_trigger = yes
			}
			initiate_inner_circle_effect = yes
			initiate_justicar_effect = yes
			if = {
				limit = { religion = religion:sabbat_religion }
				set_anathema_effect = yes
			}
			faith:camahumanity = { 
				remove_list_variable = {
					name = inner_circle
					target = prev
				}
			}
		}
		if = {
			limit = { 
				NOT = { religion = religion:camarilla_religion }
				POD_is_justicar_trigger = yes 
			}
			initiate_justicar_effect = yes
			if = {
				limit = { religion = religion:sabbat_religion }
				set_anathema_effect = yes
			}
		}
	
		#An Sabbat member converts
		if = {
			limit = { NOT = { religion = religion:sabbat_religion } }
			if = {
				limit = { POD_is_sabbat_regent_trigger = yes }
				remove_sabbat_regent_effect = yes
			}
			if = {
				limit = { POD_is_cardinal_trigger = yes }
				remove_cardinal_effect = yes
			}
			if = {
				limit = { POD_is_black_hand_trigger = yes }
				remove_black_hand_effect = yes
			}
		}
	
		#An Talmahera member converts
		if = {
			limit = { NOT = { religion = religion:talmahera_religion } }
			if = {
				limit = { POD_is_wazir_trigger = yes }
				initiate_wazir_effect = yes
				remove_wazir_effect = yes
			}
		}

		POD_fake_faith_modifier_pulse_effect = yes
	}
}
