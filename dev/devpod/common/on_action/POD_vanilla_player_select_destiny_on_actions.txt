# Appending our additions to vanilla on_actions in a separate file, to keep vanilla diffs clean



### Called when player has confirmed their selection of a destiny character
# Scopes:
# root: destiny character (is already current player)
# scope:destiny_type_flag - flag of selected destiny type
# scope:previous_player_character - previous player character
# scope:previous_player_heir - previous player heir
on_player_select_destiny_confirmed = {
	on_actions = { POD_on_player_select_destiny_confirmed }
}


POD_on_player_select_destiny_confirmed = {
	effect = {

		#Give player Mokole the Suchid choice event since AI don't use them
		if = {
			limit = { POD_is_mokole_trigger = yes }
			trigger_event = { id = POD_chargen.2633 days = 1 }
		}
	}
}
