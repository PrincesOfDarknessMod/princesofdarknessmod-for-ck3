on_game_start_after_lobby = {
	on_actions = {
		POD_wetrius_setup
	}
}

POD_wetrius_setup = {
	effect = {
		trigger_event = {
			id = POD_april_fools.1
			days = 1
		}
		every_player = {
			trigger_event = {
				id = POD_april_fools.2
				days = 1
			}
		}
	}
}


on_death = {
	on_actions = {
		POD_wetrius_death
	}
}

POD_wetrius_death = {
	effect = {
		if = {
			limit = {
				scope:killer ?= { is_ai = no }
				has_character_flag = is_wetrius
			}
			scope:killer = {
				set_variable = {
					name  = POD_wetrius_killcount_var
					value = {
						value = POD_wetrius_killcount
						add = 1
					}
				}
			}
		}
	}
}
