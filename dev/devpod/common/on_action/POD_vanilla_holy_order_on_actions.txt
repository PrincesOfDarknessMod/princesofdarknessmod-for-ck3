# Appending our additions to vanilla on_actions in a separate file, to keep vanilla diffs clean


# A holy order is hired by a ruler.
# root = the holy order
# scope:patron = the current patron of the holy order, if any
# scope:actor = the character hiring the holy order
on_holy_order_hired = {
	on_actions = { POD_on_holy_order_hired }
}

POD_on_holy_order_hired = {
	effect = {
		if = {
			limit = { scope:actor.domicile ?= { has_domicile_building = mokole_sunforged_outfitter_06 } }
			scope:actor = { add_character_modifier = { modifier = sunforged_outfitters_modifier years = 2 } }
		}
	}
}
