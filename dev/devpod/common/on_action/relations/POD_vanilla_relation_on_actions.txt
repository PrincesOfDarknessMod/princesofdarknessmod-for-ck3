# Appending our additions to vanilla on_actions in a separate file, to keep vanilla diffs clean

on_set_relation_lover = {
	on_actions = { POD_on_set_relation_lover }
}

POD_on_set_relation_lover = {
	effect = {
		if = {
			limit = { POD_is_blocked_from_gaining_lovers = yes }
			# TODO POSSESSED
		}
		else = {
			if = {
				limit = { has_trait = wraith }
				POD_pulse_passion = { TYPE = lust }
			}
			POD_achiev_helena2_check = yes
			add_to_variable_list = {
				name = new_relations
				target = scope:target
				years = 10
			}
			if = {
				limit = {
					OR = {
						POD_gains_thrashingdragons_dharma_xp = yes
						POD_gains_risingphoenix_dharma_xp = yes
						POD_gains_greatclans_dharma_xp = yes
					}
				}
				POD_give_dharma_xp_effect = { VALUE = POD_kueijin_dharma_xp_value_very_high }
			}
			if = {
				limit = {
					POD_gains_emma_o_dharma_xp = yes
					scope:target = {
						culture = culture:japanese
						is_female = yes
						is_kueijin_trigger = yes
					}
				}
				POD_give_dharma_xp_effect = { VALUE = POD_kueijin_dharma_xp_value_huge }
			}

			if = {
				limit = {
					scope:target ?= {
						PoD_is_member_of_seasonal_courts = yes
						religion != ROOT.religion
						situation:PoD_war_of_seasons ?= { situation_top_has_catalyst = PoD_war_of_seasons_seduce_court }
					}
				}
				situation:PoD_war_of_seasons ?= {
					trigger_situation_catalyst = {
						catalyst = PoD_war_of_seasons_seduce_court
						character = scope:target
					}
				}
			}
		}
	}
}

on_set_relation_friend = {
	on_actions = { POD_on_set_relation_friend }
}

POD_on_set_relation_friend = {
	effect = {
		if = {
			limit = { POD_is_blocked_from_gaining_friends = yes }
			# TODO POSSESSED
		}
		else = {
			if = {
				limit = { has_trait = wraith }
				POD_pulse_passion = { TYPE = compassion }
			}
			add_to_variable_list = {
				name = new_relations
				target = scope:target
				years = 10
			}
			if = {
				limit = {
					OR = {
						POD_gains_thrashingdragons_dharma_xp = yes
						POD_gains_risingphoenix_dharma_xp = yes
						POD_gains_greatclans_dharma_xp = yes
					}
				}
				POD_give_dharma_xp_effect = { VALUE = POD_kueijin_dharma_xp_value_very_high }
			}

			if = {
				limit = {
					scope:target ?= {
						PoD_is_member_of_seasonal_courts = yes
						religion != ROOT.religion
						situation:PoD_war_of_seasons ?= { situation_top_has_catalyst = PoD_war_of_seasons_befriend_court }
					}
				}
				situation:PoD_war_of_seasons ?= {
					trigger_situation_catalyst = {
						catalyst = PoD_war_of_seasons_befriend_court
						character = scope:target
					}
				}
			}

			if = {
				limit = { scope:target.dynasty ?= { has_dynasty_perk = leanhaun_legacy_3 } }
				add_character_modifier = POD_leanhaun_legacy_3_modifier
			}
			else_if = {
				limit = { scope:target.dynasty ?= { has_dynasty_perk = leanhaun_legacy_2 } }
				add_character_modifier = POD_leanhaun_legacy_2_modifier
			}

			POD_judges_words_of_healing_effect = yes
		}
	}
}

}

POD_on_remove_relation_friend = {

		if = {

					}
				}
			}
			POD_remove_character_modifier = { MODIFIER = POD_leanhaun_legacy_2_modifier }
		}
		if = {
			limit = {
				NOT = {
					any_relation = {
						type = friend
						type = best_friend
						dynasty ?= { has_dynasty_perk = leanhaun_legacy_3 }
					}
				}
			}
			POD_remove_character_modifier = { MODIFIER = POD_leanhaun_legacy_3_modifier }
		}

		POD_judges_words_of_healing_effect = yes
	}
}
