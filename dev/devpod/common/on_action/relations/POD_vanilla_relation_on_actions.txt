# Appending our additions to vanilla on_actions in a separate file, to keep vanilla diffs clean

on_set_relation_lover = {
	on_actions = { POD_on_set_relation_lover }
}

POD_on_set_relation_lover = {
	effect = {	
		if = {
			limit = { has_perk = runic_circle_perk }
			scope:target = { add_character_modifier = runic_circle_perk_modifier }
		}
		if = {
			limit = { 
				exists = dynasty 
				dynasty = { has_dynasty_perk = unseelie_romantic_legacy_1 }
			}
			lover_competition_effect = yes
		}
		if = {
			limit = { 
				exists = dynasty 
				dynasty = { has_dynasty_perk = unseelie_romantic_legacy_4 }
			}
			add_character_modifier = POD_gamester_modifier
		}
		
		if = {
			limit = { has_trait = wraith }
			POD_pulse_passion = { TYPE = lust }
		}
		POD_achiev_helena2_check = yes
		add_to_variable_list = {
			name = new_relations
			target = scope:target
			years = 10
		}
		if = {
			limit = {
				OR = {
					POD_gains_thrashingdragons_dharma_xp = yes
					POD_gains_risingphoenix_dharma_xp = yes
					POD_gains_greatclans_dharma_xp = yes
				}
			}
			POD_give_dharma_xp_effect = { VALUE = POD_kueijin_dharma_xp_value_very_high }
		}
	}
}


on_remove_relation_lover = {
	on_actions = { POD_on_remove_relation_lover }
}

POD_on_remove_relation_lover = {
	effect = {
		if = {
			limit = { 
				has_character_modifier = runic_circle_perk_modifier
				can_keep_runic_circle_trigger = no
			}
			remove_character_modifier = runic_circle_perk_modifier
		}
		if = {
			limit = { 
				exists = dynasty 
				dynasty = { has_dynasty_perk = unseelie_romantic_legacy_1 }
			}
			lover_competition_effect = yes
		}
		if = {
			limit = { 
				exists = dynasty 
				dynasty = { has_dynasty_perk = unseelie_romantic_legacy_4 }
			}
			remove_character_modifier = POD_gamester_modifier
		}
		if = {
			limit = { POD_gains_thousandwhispers_dharma_xp = yes }
			POD_give_dharma_xp_effect = { VALUE = POD_kueijin_dharma_xp_value_huge } 
		}
	}
}

		add_to_variable_list = {
			name = new_relations
			target = scope:target
			years = 10
		}
		if = {
			limit = {
				OR = {
					POD_gains_thrashingdragons_dharma_xp = yes
					POD_gains_risingphoenix_dharma_xp = yes
					POD_gains_greatclans_dharma_xp = yes
				}
			}
			POD_give_dharma_xp_effect = { VALUE = POD_kueijin_dharma_xp_value_very_high }
		}
		if = {
			limit = { uses_cairo_djedi_main_reward = yes }
			add_character_modifier = cairo_canton_djedi_modifier
		}
		scope:target = {
			if = {
				limit = { uses_cairo_djedi_main_reward = yes }
				add_character_modifier = cairo_canton_djedi_modifier
			}
		}
	}
}
