on_set_relation_demonic_master = {
	effect = {
		scope:target = {
			add_hook_no_toast = {
				type = thrall_hook
				target = prev
			}
		}
		if = {
			limit = { POD_splat_can_be_infernalist = yes }
			give_infernalist_secret_or_trait_effect = yes
			give_cultist_secret_or_trait_effect = yes
			add_thrall_xp = { VALUE = 5 }
		}
		if = {
			limit = {
				scope:target = {
					is_ai = no
					POD_is_demon_trigger = yes
				}
				highest_held_title_tier >= tier_kingdom
			}
			set_global_variable = pod_achiev_demontutorial # Achievement: Master of Puppets
		}
	}
}

on_remove_relation_demonic_master = {
	effect = {
		scope:target = {
			if = {
				limit = { has_hook = prev }
				remove_hook = { target = prev }
			}
		}
	}
}

on_set_relation_worshipped_demon = {
	effect = {
		if = {
			limit = {
				faith = {
					OR = {
						religion = religion:demons_religion # worshippers of other sins will switch allegiances
						is_demon_worshipper_religion_trigger = no # other infernalists don't switch
					}
				}
			}
			set_character_faith = scope:target.faith
		}
		if = {
			limit = { POD_splat_can_be_infernalist = yes }
			give_infernalist_secret_or_trait_effect = yes
			give_cultist_secret_or_trait_effect = yes
			add_thrall_xp = { VALUE = 5 }
			if = {
				limit = { num_of_relation_worshipped_demon > 1 }
				add_secret = { type = secret_masspact }
				add_character_modifier = bound_demon_assist_investiture_modifier
			}
		}
		POD_refresh_pact_modifiers_effect = yes
	}
}

on_remove_relation_worshipped_demon = {
	effect = {
		POD_adopt_correct_religion_effect = yes
		remove_character_modifier = bound_demon_assist_investiture_modifier
		POD_refresh_pact_modifiers_effect = yes
		if = {
			limit = {
				has_trait = mortal
				can_hold_title_as_mortal_trigger = no
			}
			depose_effect = { DEPOSER = root }
		}
	}
}

on_death_relation_worshipped_demon = {
	effect = {
		POD_adopt_correct_religion_effect = yes
		remove_character_modifier = bound_demon_assist_investiture_modifier
		POD_refresh_pact_modifiers_effect = yes
		if = {

				has_trait = mortal
				can_hold_title_as_mortal_trigger = no
			}
			depose_effect = { DEPOSER = root }
		}
	}
}

on_set_relation_worshipped_reborn = {
	effect = {
		if = {
			limit = { exists = scope:target.dynasty }
			if = {
				limit = { scope:target.dynasty = { has_dynasty_perk = mortuary_cult_mummy_legacy_5 } }
				remove_character_modifier = POD_neteru_legacy_1_modifier
				remove_character_modifier = POD_neteru_legacy_2_modifier
				remove_character_modifier = POD_neteru_legacy_3_modifier
				remove_character_modifier = POD_neteru_legacy_4_modifier
				add_character_modifier = POD_neteru_legacy_5_modifier
			}
			else_if = {
				limit = { scope:target.dynasty = { has_dynasty_perk = mortuary_cult_mummy_legacy_4 } }
				remove_character_modifier = POD_neteru_legacy_1_modifier
				remove_character_modifier = POD_neteru_legacy_2_modifier
				remove_character_modifier = POD_neteru_legacy_3_modifier
				add_character_modifier = POD_neteru_legacy_4_modifier
			}
			else_if = {
				limit = { scope:target.dynasty = { has_dynasty_perk = mortuary_cult_mummy_legacy_3 } }
				remove_character_modifier = POD_neteru_legacy_1_modifier
				remove_character_modifier = POD_neteru_legacy_2_modifier
				add_character_modifier = POD_neteru_legacy_3_modifier
			}
			else_if = {
				limit = { scope:target.dynasty = { has_dynasty_perk = mortuary_cult_mummy_legacy_2 } }
				remove_character_modifier = POD_neteru_legacy_1_modifier
				add_character_modifier = POD_neteru_legacy_2_modifier
			}
			else_if = {
				limit = { scope:target.dynasty = { has_dynasty_perk = mortuary_cult_mummy_legacy_1 } }
				add_character_modifier = POD_neteru_legacy_1_modifier
			}
		}
	}
}
