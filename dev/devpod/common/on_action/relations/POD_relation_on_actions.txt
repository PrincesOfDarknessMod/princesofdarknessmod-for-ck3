on_set_relation_demonic_master = {
	effect = {
		scope:target = {
			if = {
				limit = { has_perk = usurpation_of_flesh_perk }
				add_hook_no_toast = {
					type = usurpation_of_flesh_perk_hook
					target = prev
				}
			}
			else = {
				add_hook_no_toast = {
					type = thrall_hook
					target = prev
				}
			}
		}
		POD_character_resolve_initiation_effect = yes
		if = {
			limit = { exists = scope:target.dynasty }
			if = {
				limit = { scope:target.dynasty = { has_dynasty_perk = pacts_legacy_1 } }
				add_learning_skill = 5
			}
			if = {
				limit = { scope:target.dynasty = { has_dynasty_perk = pacts_legacy_2 } }
				add_diplomacy_skill = 5
			}
			if = {
				limit = { scope:target.dynasty = { has_dynasty_perk = pacts_legacy_3 } }
				add_intrigue_skill = 5
			}
			if = {
				limit = { scope:target.dynasty = { has_dynasty_perk = pacts_legacy_4 } }
				add_stewardship_skill = 5
			}
			if = {
				limit = { scope:target.dynasty = { has_dynasty_perk = pacts_legacy_5 } }
				add_martial_skill = 5
			}
		}
	}
}

on_remove_demonic_master = {
	effect = {
		scope:target = {
			if = {
				limit = {
					has_hook_of_type = {
						target = prev
						type = usurpation_of_flesh_perk_hook
					}
				}
				remove_hook = {
					target = prev
					type = usurpation_of_flesh_perk_hook
				}
			}
			if = {
				limit = {
					has_hook_of_type = {
						target = prev
						type = thrall_hook
					}
				}
				remove_hook = {
					target = prev
					type = thrall_hook
				}
			}
		}
		POD_remove_resolve_modifiers_effect = yes
		if = {
			limit = { exists = scope:target.dynasty }
			if = {
				limit = { scope:target.dynasty = { has_dynasty_perk = pacts_legacy_1 } }
				add_learning_skill = -5
			}
			if = {
				limit = { scope:target.dynasty = { has_dynasty_perk = pacts_legacy_2 } }
				add_diplomacy_skill = -5
			}
			if = {
				limit = { scope:target.dynasty = { has_dynasty_perk = pacts_legacy_3 } }
				add_intrigue_skill = -5
			}
			if = {
				limit = { scope:target.dynasty = { has_dynasty_perk = pacts_legacy_4 } }
				add_stewardship_skill = -5
			}
			if = {
				limit = { scope:target.dynasty = { has_dynasty_perk = pacts_legacy_5 } }
				add_martial_skill = -5
			}
		}
	}
}
