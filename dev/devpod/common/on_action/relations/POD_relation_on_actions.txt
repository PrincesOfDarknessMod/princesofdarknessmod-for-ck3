on_set_relation_demonic_master = {
	effect = {
		scope:target = {
			if = {
				limit = { has_perk = usurpation_of_flesh_perk }
				add_hook_no_toast = {
					type = usurpation_of_flesh_perk_hook
					target = prev
				}
			}
			else = {
				add_hook_no_toast = {
					type = thrall_hook
					target = prev
				}
			}
		}
		POD_character_resolve_initiation_effect = yes
	}
}

on_remove_demonic_master = {
	effect = {
		scope:target = {
			if = {
				limit = {
					has_hook_of_type = {
						target = prev
						type = usurpation_of_flesh_perk_hook
					}
				}
				remove_hook = {
					target = prev
					type = usurpation_of_flesh_perk_hook
				}
			}
			if = {
				limit = {
					has_hook_of_type = {
						target = prev
						type = thrall_hook
					}
				}
				remove_hook = {
					target = prev
					type = thrall_hook
				}
			}
		}
		POD_remove_resolve_modifiers_effect = yes
	}
}
