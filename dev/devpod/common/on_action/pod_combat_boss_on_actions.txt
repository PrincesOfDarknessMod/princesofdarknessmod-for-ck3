######################################################
###### BOSS CHARACTER CREATION HOOK
######################################################

on_character_created = {
	on_actions = {
		pod_combat_boss_on_character_created
	}
}

pod_combat_boss_on_character_created = {
	effect = {
		# The boss carries pod_combat_boss flags once its template's after_creation has run, which is before this hook fires. 
		if = {
			limit = { has_character_flag = pod_combat_boss }
			debug_log = pod_combat_boss_created_hook_log
			debug_log_scopes = no
			POD_combat_boss_finish_creation_effect = yes
		}
	}
}
