POD_psi_regenerate_cycle = {
    effect = {
        add_character_flag = POD_psi_is_on_biocontrol_cycle
        if = { limit = { has_trait = wounded } 
            random = {
                chance = 33
                modifier = {
                    add = 33
                    has_perk = biocontrol_2_perk
                }
                modifier = {
                    add = 34
                    has_perk = biocontrol_3_perk
                }
                send_interface_toast = {
                    title = POD_psi_regenerate_cycle
                    if = { limit = { has_trait = wounded_1 } remove_trait = wounded_1 }
                    else_if = { limit = { has_trait = wounded } change_trait_rank = { trait = wounded rank = -1 } }
                }
            }
        }
        if = { limit = { has_perk = biocontrol_1_perk } trigger_event = { on_action = POD_psi_regenerate_cycle months = 4 } }
        else = { remove_character_flag = POD_psi_is_on_biocontrol_cycle }
    }
}
POD_psi_coterie_buff = {
    effect = {
        add_character_flag = POD_psi_is_on_coterie_buff_cycle
        ROOT = {
            remove_all_character_modifier_instances = psi_synergy_coterie_stress_buff
        }
        if = { limit = { has_variable = coterie } 
            var:coterie = {
                every_in_list = {
                    variable = members
                    limit = {
                        has_trait = psychic
                    }
                    ROOT = {
                        add_character_modifier = psi_synergy_coterie_stress_buff
                    }
                }
            }
        }
        if = { limit = { has_perk = synergy_2_perk } trigger_event = { on_action = POD_psi_coterie_buff months = 4 } }
        else = { remove_character_flag = POD_psi_is_on_coterie_buff_cycle }
    }
}
