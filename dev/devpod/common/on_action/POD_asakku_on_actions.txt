######################################################
# Court Join - Detect Vicissitude carriers entering court
# Root = joining character, scope:new_employer = court owner
######################################################

on_join_court = {
	on_actions = { POD_asakku_on_join_court }
}

POD_asakku_on_join_court = {
	effect = {
		if = {
			limit = {
				# Root is the joining character
				has_vicissitude_trigger = yes
				is_vampire_trigger = yes
				NOT = { POD_asakku_has_any_asakku_trait_trigger = yes }
				NOT = { has_character_flag = POD_asakku_new_eye_fired }
				exists = scope:new_employer
				scope:new_employer = {
					POD_asakku_has_any_asakku_trait_trigger = yes
				}
			}
			save_scope_as = new_vicissitude_carrier
			add_character_flag = {
				flag = POD_asakku_new_eye_fired
			}
			scope:new_employer = {
				save_scope_as = court_owner
				trigger_event = {
					id = POD_asakku.0101
					days = { 1 3 }
				}
			}
		}
	}
}

######################################################
# Yearly Elioud Pulse - Rivalry and AI recruitment
# Runs for Elioud characters to maintain rivalries
# and trigger AI scheme usage
######################################################

yearly_playable_pulse = {
	on_actions = { POD_asakku_elioud_yearly_pulse }
}

POD_asakku_elioud_yearly_pulse = {
	trigger = {
		has_trait = POD_asakku_elioud
	}
	effect = {
		# Create rivalries with any Asakku characters we don't already rival
		every_living_character = {
			limit = {
				OR = {
					has_trait = POD_asakku_asakku
					has_trait = POD_asakku_iryi_ascendant
				}
				NOT = { has_relation_rival = root }
				NOT = { this = root }
				in_diplomatic_range = root
			}
			root = {
				set_relation_rival = {
					reason = rival_asakku_threat
					target = prev
				}
			}
		}
		# AI Elioud: attempt Kamut Recruitment
		if = {
			limit = {
				is_ai = yes
				NOT = { has_character_flag = POD_asakku_kamut_recruiting }
				NOT = { has_character_flag = POD_asakku_kamut_recruit_cooldown }
				# Only recruit if there are at least 3 infected characters
				any_living_character = {
					count >= 3
					POD_asakku_has_any_asakku_trait_trigger = yes
				}
				# Cap total Elioud at 8
				NOT = {
					any_living_character = {
						count >= 8
						has_trait = POD_asakku_elioud
					}
				}
			}
			# 10% chance per year to attempt recruitment
			random = {
				chance = 10
				# Find a valid TMR character to recruit
				random_living_character = {
					limit = {
						POD_asakku_is_valid_kamut_recruitment_target_trigger = yes
						in_diplomatic_range = root
					}
					weight = {
						base = 1
						modifier = {
							add = 5
							martial >= 8
						}
						modifier = {
							add = 5
							martial >= 12
						}
						modifier = {
							add = 5
							martial >= 16
						}
						modifier = {
							add = 5
							intrigue >= 8
						}
						modifier = {
							add = 5
							intrigue >= 12
						}
						modifier = {
							add = 5
							intrigue >= 16
						}
					}
					save_scope_as = kamut_target
				}
				if = {
					limit = { exists = scope:kamut_target }
					start_scheme = {
						type = POD_asakku_kamut_recruitment_scheme
						target_character = scope:kamut_target
					}
					# Long cooldown - 10 years per Elioud between recruitment attempts
					add_character_flag = {
						flag = POD_asakku_kamut_recruit_cooldown
						years = 10
					}
				}
			}
		}
	}
}
