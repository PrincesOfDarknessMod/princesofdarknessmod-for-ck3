POD_yearly_health_pulse = {
	trigger = {
		has_trait = wounded
	}
	effect = {
		random = {
			chance = 50
			POD_remove_character_modifier = { MODIFIER = infected_wound_modifier }
		}
		if = {
			limit = {
				POD_health_has_superficial_wound_trigger = yes
				OR = {
					is_ai = yes
					# from POD_can_naturally_heal_trigger
					NOT = { has_trait = vampire }
				}
				POD_can_be_healed_trigger = yes
				save_temporary_scope_value_as = {
					name = biyearly_countdown
					value = {
						value = current_year
						modulo = POD_years_to_heal_superficial_damage_naturally
					}
				}
				scope:biyearly_countdown = 0
			}
			send_interface_message = {
				type = send_interface_message_good
				title = POD_yearly_health_pulse_normal.title
				left_icon = ROOT
				POD_heal_superficial_wound_effect = {
					COUNT = POD_amount_when_healing_superficial_damage_naturally
				}
			}
		}
		else_if = {
			limit = {
				has_trait = torpor
				save_temporary_scope_value_as = {
					name = biyearly_countdown
					value = {
						value = current_year
						modulo = 2
					}
				}
				scope:biyearly_countdown = 0
			}
			send_interface_message = {
				type = send_interface_message_good
				title = POD_yearly_health_pulse_normal.title
				left_icon = ROOT
				if = {
					limit = { POD_health_has_aggravated_wound_trigger = yes }
					POD_convert_aggravated_to_superficial_wound = {
						COUNT = POD_amount_when_converting_aggravated_to_superficial_damage_in_torpor
					}
				}
				else = {
					POD_heal_superficial_wound_effect = {
						COUNT = POD_amount_when_healing_superficial_damage_in_torpor
					}
				}
			}
		}
		else_if = {
			limit = {
				POD_health_has_aggravated_wound_trigger = yes
				save_temporary_scope_value_as = {
					name = biyearly_countdown
					value = {
						value = current_year
						modulo = POD_years_to_convert_aggravated_to_superficial_damage
					}
				}
				scope:biyearly_countdown = 0
			}
			send_interface_message = {
				type = send_interface_message_good
				title = POD_yearly_health_pulse_aggravated.title
				left_icon = ROOT
				POD_convert_aggravated_to_superficial_wound = {
					COUNT = POD_amount_when_converting_aggravated_to_superficial_damage
				}
				add_trait = scarred
				add_trait_xp = {
					trait = scarred
					value = 5
				}
			}
		}
	}
}
