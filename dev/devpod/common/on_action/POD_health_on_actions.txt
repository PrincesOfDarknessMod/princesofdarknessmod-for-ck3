POD_yearly_health_pulse = {
	trigger = {
		has_trait = wounded
	}
	effect = {
		random = {
			chance = 50
			if = {
				limit = { has_character_modifier = infected_wound_modifier }
				remove_character_modifier = infected_wound_modifier
			}
		}
		if = {
			limit = {
				POD_can_naturaly_heal_trigger = yes
				POD_health_has_normal_wound_trigger = yes
				save_temporary_scope_value_as = {
					name = biyearly_countdown
					value = { value = current_year modulo = POD_health_years_to_heal }
				}
				scope:biyearly_countdown = 0
			}
			send_interface_message = {
				type = send_interface_message_good
				title = POD_yearly_health_pulse_normal.title
				left_icon = ROOT
				POD_heal_normal_wound_effect = { COUNT = POD_health_yearly_pulse_heal }
			}
		}
		else_if = {
			limit = {
				has_trait = torpor
				save_temporary_scope_value_as = {
					name = biyearly_countdown
					value = { value = current_year modulo = 2 }
				}
				scope:biyearly_countdown = 0
			}
			send_interface_message = {
				type = send_interface_message_good
				title = POD_yearly_health_pulse_normal.title
				left_icon = ROOT
				if = { limit = { POD_health_has_aggravated_wound_trigger = yes } 
					POD_heal_aggravated_wound_effect = { COUNT = 1 }
					POD_apply_normal_damage_effect = { COUNT = 1 }
				}
				else = {
					POD_heal_normal_wound_effect = { COUNT = 1 }
				}
			}
		}
		else_if = {
			limit = {
				POD_health_has_aggravated_wound_trigger = yes
				save_temporary_scope_value_as = {
					name = biyearly_countdown
					value = { value = current_year modulo = POD_health_years_to_convert }
				}
				scope:biyearly_countdown = 0
			}
			send_interface_message = {
				type = send_interface_message_good
				title = POD_yearly_health_pulse_aggravated.title
				left_icon = ROOT
				POD_heal_aggravated_wound_effect = { COUNT = POD_health_yearly_pulse_convert }
				POD_apply_normal_damage_effect = { COUNT = POD_health_yearly_pulse_convert }
				add_trait = scarred
				add_trait_xp = {
					trait = scarred
					value = 5
				}
			}
		}
	}
}
