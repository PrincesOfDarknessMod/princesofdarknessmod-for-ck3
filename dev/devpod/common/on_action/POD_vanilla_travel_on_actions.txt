# Appending our additions to vanilla on_actions in a separate file, to keep vanilla diffs clean


on_travel_plan_movement = {
	on_actions = { POD_on_travel_plan_movement }
}

POD_on_travel_plan_movement = {
	effect = {
		if = {
			limit = {
				has_perk = madness_season_perk #to quickly skip the rest
				current_travel_plan = { has_travel_option = werewolf_ragabash_option }
				POD_has_enough_gnosis_for_gift = { COUNT = 1 }
				current_travel_plan.current_location.duchy.holder = {
					NOR = {
						top_liege ?= root
						top_liege ?= root.top_liege
						has_character_modifier = madness_season_modifier
					}
				}
			}
			current_travel_plan.current_location.duchy.holder = { save_scope_as = madness_target }
			send_interface_message = {
				type = POD_werewolf_message_good
				title = madness_season_perk_name
				right_icon = scope:madness_target
				scope:madness_target = { add_character_modifier = { modifier = madness_season_modifier years = 5 } }
				POD_gift_cost_effect = { COUNT = 1 }
			}
		}
	}
}

on_travel_plan_complete = {
	on_actions = { POD_on_travel_plan_complete }
}

POD_on_travel_plan_complete = {
	effect = {
		# Clean up the current Journey (provoke_vatican is handled differently)
		if = {
			limit = { NOT = { has_character_flag = spawn_the_lamb } }

			POD_journey_post_travel_cleanup = yes

			# Suggest a Journey to continue
			if = {
				limit = {
					any_in_list = {
						variable = paused_journeys
						count > 0
						var:total_progress >= 100
					}
				}

				trigger_event = {
					id = POD_journey_maintenance.110
					days = 1
				}
			}
		}
	}
}


on_travel_plan_abort = {
	on_actions = { POD_on_travel_plan_abort }
}

POD_on_travel_plan_abort = {
	effect = {
		# Clean up the current Journey (provoke_vatican is handled differently)
		if = {
			limit = { NOT = { has_character_flag = spawn_the_lamb } }

			POD_journey_post_travel_cleanup = yes

			if = {
				limit = {
					any_in_list = {
						variable = paused_journeys
						count > 0
						var:total_progress >= 100
					}
				}

				trigger_event = {
					id = POD_journey_maintenance.110
					days = 1
				}
			}
		}
		# Abort contracts that start a travel plan AFTER you "accept" them
		current_travel_plan = {
			if = {
				limit = {
					var:task_contract ?= {
						has_variable = recreate_on_travel_abort
					}
				}
				var:task_contract ?= { invalidate_contract = yes }
			}
		}
	}
}

				limit = {
					var:task_contract ?= {
						has_variable = recreate_on_travel_abort
					}
				}
				var:task_contract ?= { invalidate_contract = yes }
			}
		}
	}
}
