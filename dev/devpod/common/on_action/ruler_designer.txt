# Root = New character after being made
on_ruler_designer_finished = {
	effect = {
		# Don't fire marriage notification events.
		if = {
			limit = { faith = { has_doctrine_parameter = marriage_event } }
			add_character_flag = ignore_marriage_event
		}
		# Do we need to fire a regency for this character?
		if = {
			limit = {
				OR = {
					is_adult = no
					is_incapable = yes
				}
			}
			add_to_global_variable_list = {
				name = rd_chars_needing_regencies
				target = root
			}
		}
		# If your capital on game start is Mecca, you get the Hajjaj trait (since you can't Hajj to your capital)
		if = {
			limit = {
				this = title:b_makka.county.holder
				faith = { has_doctrine_parameter = mandatory_hajj }
				NOT = { has_trait = hajjaj }
			}
			# Have some Hajjaj
			add_trait = hajjaj
		}

		## Landless adventurer
		if = {
			limit = {
				NOT = { is_landed = yes }
			}
			create_landless_adventurer_title_effect = {
				REASON = flag:voluntary
				FLAVOR_CHAR = root
			}
		}		
		# PoD
		set_global_variable = used_ruler_designer # disables some achievements

		every_spouse = { POD_character_initialization_for_commoners_effect = yes }
		every_child = { POD_character_initialization_for_generated_child_effect = yes }
		
		if = {
			limit = { has_trait = vampire }
			if = {
				limit = { has_POD_vampire_clan_trigger = no }
				POD_character_vampire_clan_initialization_effect = yes
			}
			if = {
				limit = { has_POD_vampire_generation_trigger = no }
				add_trait = vampgeneration8
			}
			if = {
				limit = { NOT = { any_character_trait = { has_trait_flag = discipline } } }
				POD_character_vampire_discipline_initialization_effect = yes
			}
		}
		if = {
			limit = { #Ruler Designer Fallback for those who don't qualify for any
				has_trait = necromancy
				has_any_necromancy_path = no
			}
			add_character_flag = has_cappadocian_necromancy_flag
		}
		if = {
			limit = {
				has_trait = bloodsorcerydiscipline
				has_any_bloodmagic_path = no
			}
			add_character_flag = has_thaumaturgy_bloodmagic_flag
		}
		if = {
			limit = { 
				POD_is_infernalist_trigger = yes 
				POD_has_true_faith_trigger = yes
				POD_splat_mixing_haight_mode = no
			}
			POD_lose_true_faith_effect = yes
		}
		if = {
			limit = { # Prevent Tzimisce face randomization for ruler designed characters
				is_tzimisce_trigger = yes
			}
			add_character_flag = had_POD_character_tzimisce_looks
		}
		if = {
			limit = { # Allow the player to choose whether their custom Nosferatu gets further nosferization
				is_nosferatu_trigger = yes
			}
			add_character_flag = had_POD_character_nosferatu_looks
			trigger_event = { id = POD_maintenance.28 days = 1 }
		}
		if = {
			limit = { POD_has_illegal_religion_trigger = yes }
			POD_adopt_correct_religion_effect = yes
		}
		if = {
			limit = { is_landed = yes }
			POD_adopt_correct_government_effect = yes # No need check illegal... you can't choose gov in RD
		}
		POD_remove_wrong_supernatural_traits_effect = yes
		trigger_event = { id = POD_maintenance.16 days = 1 } # For player who make broken character then play other character
	}
}
