# Appending our additions to vanilla on_actions in a separate file, to keep vanilla diffs clean

### NOTE:
# Additions to on_natural_death_second_chance have to be in the vanilla file
# so that our mechanics can act as exceptions to the vanilla case

on_death = {
	on_actions = { POD_on_death }
}

POD_on_death = {
	events = {
		pod_tremere_council.5 #A tremere councilor dies
		pod_ventrue_ephorate.5 #A ventrue ephor dies
	}
	effect = {
		scope:killer ?= { POD_commit_murder_effect = yes }
		
		# Domicile inheritance
		if = {
			limit = {
				exists = domicile
				any_held_title = {
					POD_title_is_haven_landed_domicile = yes
					save_temporary_scope_as = haven_landed
				}
			}
			if = {
				limit = {
					exists = player_heir
				}
				# "redirect" the domicile to the player heir
				# since otherwise it gets inherited separately from elective successions
				scope:haven_landed = {
					set_variable = {
						name  = proper_heir
						value = prev.player_heir
						days  = 1
					}
					if = {
						limit = { prev = { is_ai = no } }
						set_variable = {
							name  = inherited_from_player
							value = yes
							days  = 1
						}
					}
				}
			}
			else = {
				destroy_title = scope:haven_landed # TODO: keep main building instead?
			}
		}
		
		if = {
			limit = {
				exists = var:curia_support_var

					name  = curia_support_var

				}

			limit = { exists = var:military_order_precept }

					name  = military_order_precept

				}

					name  = military_order_rank
					value = prev.POD_military_order_rank
				}
			}
		}

		if = {
			limit = { POD_has_roguelike_curse = yes }
			if = {
				limit = { has_game_rule = POD_roguelike_mode_enabled_no_permadeath }
				POD_select_roguelike_character_effect = yes
			}
			else = {
				POD_roguelike_ensure_gameover_on_death_effect = yes
			}
		}
		
		if = {
			limit = {
				POD_has_coterie_trigger = yes
			}
			
			if = {
				limit = { POD_has_coterie_members_trigger = no }
				POD_disband_coterie_effect = yes
			}
			else = {
				if = {
					limit = { is_ai = no }
					
					save_scope_as = dead_member
					var:coterie ?= { save_scope_as = coterie }
					
					player_heir ?= {
						if = {
							limit = {
								trigger_if = {
									limit = { exists = var:coterie }
									NOT = { scope:coterie ?= var:coterie }
								}
								trigger_else = { always = yes }
							}
							trigger_event = {
								id = POD_coterie_maintenance.20
								days = 1
							}
						}
					}
				}
				POD_leave_coterie_effect = yes
			}
		}
				
		if = {
			limit = {
				exists = player_heir
				player_heir = {
					is_ventrue_trigger = yes 
					has_prey_exclusion_trigger = no
				}
			}
			player_heir = {
				trigger_event = { 
					id = POD_chargen.1100 
					days = 1
				}
			}
		}
		
		if = {
			limit = {
				has_variable = golconda_mentee
				var:golconda_mentee ?= { is_alive = yes }
			}
			var:golconda_mentee = { remove_variable = golconda_mentor }
		}
			
		#Achiev stuff
		if = {
			limit = {
				OR = {
					AND = {
						this = character:790309 # Menele
						scope:killer = {
							this = character:790179 # Helena
							is_ai = no
						}
					}
					AND = {
						this = character:790179 # Helena
						scope:killer = {
							this = character:790309 # Menele
							is_ai = no
						}
					}
				}
			}
			set_global_variable = pod_achiev_helena1 # Achievement: A Fitting End for this Story
		}
		
		#Moncada special content
		if = {
			limit = {
				this = character:790178
				character:505154 = { 
					is_alive = yes
					is_ai = no	
					}
			}
			character:505154 = {
				trigger_event = {
					id =  PODcanon_chars.20
					days = 1
					}
				}
		}		

		#Hardestadt's takeover
		if = {
			limit = {
				this = character:505050
				any_child = { 
					is_alive = yes
					is_female = no	
					}
				}
			character:505050 = {
				random_child = {
					limit = {
						is_female = no
						}
				trigger_event = {
					id =  PODcanon_chars.58
					days = 1
					}
				}
			}
		}
		
		#Twins special content
		if = {
			limit = {
				this = character:792537
				character:792536 = { 
					is_alive = yes
					is_ai = yes	
					}
			}
			character:792536 = {
				if = {
					limit = {
						has_character_modifier = tvarivich_twins_modifier
						}
					remove_character_modifier = tvarivich_twins_modifier
					add_character_modifier = bad_moncada_modifier
				}
			}
		}
		if = {
			limit = {
				this = character:792536
				character:792537 = { 
					is_alive = yes
					is_ai = no	
				}
			}
			character:792537 = {
				trigger_event = {
					id =  PODcanon_chars.61
					days = 1
				}
			}
		}		
		
		### Kill Reward ###
		if = {
			limit = {  scope:killer ?= { has_perk = arcana_reapers_hunger_perk } }
			scope:killer = { add_character_modifier = { modifier = POD_reapers_hunger_modifier years = 5 } }
		}

		if = {
			limit = { 
				scope:killer ?= { has_perk = firebringer_perk }
				has_supernatural_race_trigger = yes
			}
			scope:killer = { 
				random = {
					chance = 5
					add_diplomacy_lifestyle_perk_points = 1
				}
				random = {
					chance = 5
					add_martial_lifestyle_perk_points = 1
				}
				random = {
					chance = 5
					add_stewardship_lifestyle_perk_points = 1
				}
				random = {
					chance = 5
					add_intrigue_lifestyle_perk_points = 1
				}
				random = {
					chance = 5
					add_learning_lifestyle_perk_points = 1
				}
			}
		}
		
		if = {
			limit = { 
				scope:killer ?= { has_perk = ending_the_watch_perk }
				has_main_supernatural_race_trigger = no
			}
			scope:killer = { 
				random_list = {
					20 = { add_diplomacy_lifestyle_xp = 50 }
					20 = { add_martial_lifestyle_xp = 50 }
					20 = { add_stewardship_lifestyle_xp = 50 }
					20 = { add_intrigue_lifestyle_xp = 50 }
					20 = { add_learning_lifestyle_xp = 50 }
				}
			}
		}
			
		if = {
			limit = {
				has_trait = shapeshifter
				OR = {
					AND = {
						scope:killer ?= { POD_is_wyrm_servant_trigger = no }
						POD_is_wyrm_servant_trigger = yes
					}
					AND = {
						scope:killer ?= { POD_is_wyrm_servant_trigger = yes }
						POD_is_wyrm_servant_trigger = no
					}
				}
			}
			scope:killer = { POD_gain_glory_xp_effect = yes }
		}

		if = { 
			limit = { scope:killer ?= { POD_has_integrity = yes } }
			if = { 
				limit = { scope:killer = { has_execute_reason = prev } }
				scope:killer = { POD_gain_integrity = { VALUE = 1 } }
			}
			else = {
				scope:killer = { POD_gain_integrity = { VALUE = -3 } }
			}
		}
		if = { 
			limit = {
				scope:killer ?= {
					OR = {
						POD_gains_thrashingdragons_dharma_xp = yes
						POD_gains_risingphoenix_dharma_xp = yes
						POD_gains_greatclans_dharma_xp = yes
					}
					has_execute_reason = PREV
				}
			}
			scope:killer = { POD_give_dharma_xp_effect = { VALUE = POD_kueijin_dharma_xp_value_medium } }
		}
		if = { 
			limit = { 
				scope:killer ?= {
					POD_gains_screamer_dharma_xp = yes
				}
			}
			scope:killer = { POD_give_dharma_xp_effect = { VALUE = POD_kueijin_dharma_xp_value_medium } }
		}
		if = { 
			limit = { 
				scope:killer ?= {
					POD_gains_thousandwhispers_dharma_xp = yes
					has_relation_to = PREV
				}
			}
			scope:killer = { POD_give_dharma_xp_effect = { VALUE = POD_kueijin_dharma_xp_value_huge } }
		}
		
		### Sect ###
		
		# Camarilla
		if = {
			limit = { POD_is_inner_circle_trigger = yes }
			initiate_inner_circle_effect = yes
			initiate_justicar_effect = yes
			faith:camahumanity = { 
				remove_list_variable = {
					name = inner_circle
					target = prev
				}
			}
		}
		if = {
			limit = { POD_is_justicar_trigger = yes }
			initiate_justicar_effect = yes
		}
		if = {
			limit = { POD_is_anathema_trigger = yes }
			if = {
				limit = { scope:killer ?= { faith = faith:camahumanity } }
				scope:killer = {
					send_interface_toast = {
						right_icon = scope:killer
						left_icon = root
						title = kill_reward
						if = {
							limit = { has_court_position = alastor_court_position }
							become_red_alastor_effect = yes
							liege_or_court_owner = {
								send_interface_toast = {
									right_icon = scope:killer
									left_icon = root
									title = kill_reward
									add_gold = root.anathema_bounty_value 
								}
							}
						}
						else = {
							add_gold = root.anathema_bounty_value 
						}
						if = {
							limit = { POD_is_red_list_trigger = yes }
							red_list_tropy_effect = yes
						}
						else = {
							anathema_tropy_effect = yes
						}
					}
				}
			}
			if = {
				limit = { POD_is_red_list_trigger = yes }
				faith:camahumanity = {
					remove_list_variable = {
						name = red_list
						target = root
					}
				}
			}
			else = {
				faith:camahumanity = {
					remove_list_variable = {
						name = anathema
						target = root
					}
				}
			}	
		}
		# Sabbat
		if = {
			limit = { POD_is_sabbat_regent_trigger = yes }
			remove_sabbat_regent_effect = yes
		}
		if = {
			limit = { POD_is_cardinal_trigger = yes }
			remove_cardinal_effect = yes
		}
		if = {
			limit = { POD_is_black_hand_trigger = yes }
			remove_black_hand_effect = yes
		}
		# Patron Council
		if = {
			limit = { POD_is_patron_trigger = yes }
			remove_patron_council_status_effect = yes
		}
		# Talmahera
		if = {
			limit = { POD_is_wazir_trigger = yes }
			initiate_wazir_effect = yes
			remove_wazir_effect = yes
		}
		
		### Wraith ###
		if = {
			limit = { has_variable = fetter_owner }
			var:fetter_owner = {
				send_interface_toast = {
					title = fetter_destroyed_notification
					left_icon = this
					right_icon = root
						
					PREV = { 
						POD_destroy_fetter = yes 
					}
					POD_static_stress_gain = { COUNT = 150 }
				}
			}
		}

		# if = {
		# 	limit = { exists = house }
		# 	every_player = {
		# 		if = {
		# 			limit = { 
		# 				has_character_flag = passion_protect_family_flag
		# 				NOT = { has_character_flag = finish_protect_family_flag }
		# 				root.house = this.house
		# 			}
		# 			send_interface_toast = {
		# 				title = passion_fail_notification

		# 				left_icon = this
		# 				right_icon = root

		# 				custom_tooltip = passion_fail_protect_family_tooltip
		# 				remove_character_flag = passion_protect_family_flag
		# 				change_variable = { name = passion_count add = 2 }
		# 			}
		# 		}
		# 	}
		# }
		
		every_relation = {
			type = possessesing_wraith
			limit = { is_alive = yes }

			POD_wraith_end_possession = {
				WRAITH = this
				PUPPET = root
			}
		
			if = {
				limit = { 
					root = { 
						has_variable = master_on_control 
						var:master_on_control = prev
					}
				}
				death = { death_reason = death_souleaten }
			}
		}
		
		every_relation = {
			type = possessed_target
			limit = { 
				has_variable = master_on_control 
				var:master_on_control = prev
			}
			trigger_event = POD_wraith_possession.1000
		}
		
		POD_clear_wraith_support_modifier = yes
		
		### Misc ###
		
		if = {
			limit = { has_variable = oath_guardian }
			var:oath_guardian = {
				var:oath_guardian_caster = {
					POD_oathbreaking_effect = {
						TARGET = prev
						THIRD = root
						OATH = oath_of_guardianship
					}
				}
				remove_variable = ward_to_oath
				remove_variable = oath_guardian_caster
			}
		}
		
		# Prevent artifact dissapear due to no owner
		if = {
			limit = {
				OR = {
					is_ruler = no 
					NOT = { any_heir = { exists = this } }
				}
				any_character_artifact = { exists = this }
			}
			random_independent_ruler = {
				limit = { is_ai = yes }
				save_scope_as = new_owner 
			}
			every_character_artifact = {
				set_owner = {
					target = scope:new_owner
					history = {
						recipient = scope:new_owner
						type = inherited
					}
				}
			}
		}

		# Adapted from AGOT/Dylan's Personal Coat of Arms
		
		if = {
			limit = {
				has_variable = my_personal_coa
			}
			trigger_event = POD_personal_coat_of_arms.200
		}
		else_if = {
			limit = {
				has_variable = my_coa_belongs_to
			}
			trigger_event = POD_personal_coat_of_arms.300
		}
		
		if = {
			limit = { 
				has_trait = chulorviosis 
				NOT = { exists = scope:killer }
			}
			if = {
				limit = { is_imprisoned = yes }
				release_from_prison = yes # no death_apoplexy
			}
			POD_remove_all_supernatural_atribute_effect = yes
			add_trait = alien
			give_infernalist_secret_or_trait_effect = yes
		}
		

		if = {
			limit = {
				is_ai = no				
				exists = player_heir 
			}
			player_heir = { POD_inherit_umbra_level_effect = { TARGET = root } }
		}
		var:patron_house ?= {
			remove_house_modifier = nushi_patron_5_modifer 
			remove_house_modifier = nushi_patron_4_modifer 
			remove_house_modifier = nushi_patron_3_modifer 
			remove_house_modifier = nushi_patron_2_modifer 
			remove_house_modifier = nushi_patron_1_modifer
			remove_variable = nushi
		}
		
		if = {
			limit = { scope:killer ?= { var:impious ?= root } }
			scope:killer = {
				add_character_flag = finish_punish_impious_flag
			}
		}
		if = {
			limit = {
				scope:killer ?= { has_character_flag = demand_kill_pure_flag }
				OR = {
					POD_is_supehunter_trigger = yes

					has_dharma_greater_than_trigger = { LEVEL = 8 }

			}
			scope:killer = { add_character_flag = finish_kill_pure_flag }
		}
		if = {
			limit = { has_variable = umbra_stationed_realm }
			var:umbra_stationed_realm = { remove_variable = explorer }
		}
		if = {
			limit = {
				has_trait = hsien
				has_any_nickname = yes
				is_ai = yes
			}
			POD_create_next_hotei_effect = yes
		}

		if = {
			limit = { has_perk = master_of_samsara_perk }
			scope:killer = {
				add_character_modifier = {
					modifier = master_of_samsara_modifier
					years = 25
				}
			}
		}

		POD_spew_wide_the_pit_effect = yes

		if = {
			limit = { dynasty ?= { has_dynasty_perk = kj_shennong_legacy_5 } }
			dynasty ?= { add_dynasty_prestige = 100 }
		}
		save_scope_as = wraith
	}
