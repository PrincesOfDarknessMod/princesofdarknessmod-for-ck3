# Appending our additions to vanilla on_actions in a separate file, to keep vanilla diffs clean

on_birth_mother = {
	on_actions = { POD_on_birth_mother }
}

POD_on_birth_mother = {
	effect = {
		#BF deal with baby boys
		if = {
			limit = {
				faith = { religion_tag = blackfury_religion }
				has_trait = werewolf
				scope:child = { is_female = no }
				OR = {
					matrilinear_marriage = yes
					is_married = no
				}
			}
			trigger_event = POD_werewolves.7
		}
	}
}


on_birth_child = {
	on_actions = { POD_on_birth_child }
}

POD_on_birth_child = {
	effect = {
		# Neteru/Cultists on birth
		if = {
			limit = {
				is_alive = yes
				exists = scope:child.house
				scope:child.house.house_head = { is_alive = yes }
				NOT = { scope:child = scope:child.house.house_head }
			}
			
			if = {
				limit = { scope:child.house.house_head = { is_amenti_trigger = yes } }
				set_relation_worshipped_reborn = scope:child.house.house_head
			}
			if = {
				limit = { scope:child.house.house_head = { has_trait = demon } }
				set_relation_worshipped_demon = scope:child.house.house_head
			}
		}
		
		# Neteru legacy perk 2
		hidden_effect = {
			if = {
				limit = {
					OR = {
						real_father ?= {
							OR = {
								has_character_modifier = POD_neteru_legacy_2_modifier
								has_character_modifier = POD_neteru_legacy_3_modifier
								has_character_modifier = POD_neteru_legacy_4_modifier
								has_character_modifier = POD_neteru_legacy_5_modifier
							}
						}
						mother ?= {
							OR = {
								has_character_modifier = POD_neteru_legacy_2_modifier
								has_character_modifier = POD_neteru_legacy_3_modifier
								has_character_modifier = POD_neteru_legacy_4_modifier
								has_character_modifier = POD_neteru_legacy_5_modifier
							}
						}
					}
				}
				POD_coinflip_to_remove_trait = { TRAIT = beauty_bad_1 }
				POD_coinflip_to_remove_trait = { TRAIT = beauty_bad_2 }
				POD_coinflip_to_remove_trait = { TRAIT = beauty_bad_3 }
				POD_coinflip_to_remove_trait = { TRAIT = intellect_bad_1 }
				POD_coinflip_to_remove_trait = { TRAIT = intellect_bad_2 }
				POD_coinflip_to_remove_trait = { TRAIT = intellect_bad_3 }
				POD_coinflip_to_remove_trait = { TRAIT = physique_bad_1 }
				POD_coinflip_to_remove_trait = { TRAIT = physique_bad_2 }
				POD_coinflip_to_remove_trait = { TRAIT = physique_bad_3 }
				POD_coinflip_to_remove_trait = { TRAIT = clubfooted }
				POD_coinflip_to_remove_trait = { TRAIT = hunchbacked }
				POD_coinflip_to_remove_trait = { TRAIT = lisping }
				POD_coinflip_to_remove_trait = { TRAIT = stuttering }
				POD_coinflip_to_remove_trait = { TRAIT = dwarf }
				POD_coinflip_to_remove_trait = { TRAIT = spindly }
				POD_coinflip_to_remove_trait = { TRAIT = scaly }
				POD_coinflip_to_remove_trait = { TRAIT = wheezing }
				POD_coinflip_to_remove_trait = { TRAIT = bleeder }
				POD_coinflip_to_remove_trait = { TRAIT = infertile }
				POD_coinflip_to_remove_trait = { TRAIT = possessed_genetic }
				POD_coinflip_to_remove_trait = { TRAIT = depressed_genetic }
				POD_coinflip_to_remove_trait = { TRAIT = lunatic_genetic }
				POD_coinflip_to_remove_trait = { TRAIT = inbred }
			}
		}
		
		if = { 
			limit = { 
				OR = {
					scope:mother ?= { has_perk = ras_awakening_perk }
					scope:real_father ?= { has_perk = ras_awakening_perk }
				}
			}
			add_character_modifier = POD_hekau_ras_awakening_modifier
		}
		
		####### Adding supernatural traits #######
		POD_supernatural_trait_assignment_on_birth = {
			MOTHER = scope:mother
			#FATHER = scope:father
			REAL_FATHER = scope:real_father
		}
		
		if = {
			limit = { has_POD_race_trigger = no }
			add_trait = mortal
		}
	}
}
