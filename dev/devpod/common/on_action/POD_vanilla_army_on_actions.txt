# Appending our additions to vanilla on_actions in a separate file, to keep vanilla diffs clean


# called for armies whenever they enter a province
# root is the owner of the army
# scope:army is the army
on_army_enter_province = {
	on_actions = { POD_on_army_enter_province }
}


POD_on_army_enter_province = {
	effect = {


		#Mokole Rulers of the Sea perk
		if = {
			limit = {
				scope:army ?= {
					location ?= { is_sea_province = yes }
					army_owner = { 
						any_war_enemy = { 
							has_perk = rulers_of_the_sea_perk 
							save_temporary_scope_as = mokole_enemy
						}
					}
				}
			}
			scope:army ?= {
				random = { chance = scope:mokole_enemy.mokole_battle_effect_chance subtract_supply = 0.1 }
				army_commander ?= {
					random = {
						chance = scope:mokole_enemy.mokole_sea_effect_chance
						POD_increase_wounds_hostile_effect = { ATTACKER = scope:mokole_enemy COUNT = POD_base_wound_damage REASON = sea_creatures }
					}
				}
				army_owner = {
					every_knight = {
						limit = { location = scope:army.location }
						random = {
							chance = scope:mokole_enemy.mokole_sea_effect_chance
							POD_increase_wounds_hostile_effect = { ATTACKER = scope:mokole_enemy COUNT = POD_base_wound_damage REASON = sea_creatures }
						}
					}
				}
			}
		}



		#Koldunic Trap
		if = {
			limit = {
				scope:army ?= {
					location ?= {
						OR = {
							has_province_modifier = kupalas_exhalation_modifier
							has_province_modifier = kupalas_exhalation_modifier2
						}
						save_temporary_scope_as = trap_location
					}
					army_owner = {
						OR = {
							is_at_war_with = scope:trap_location.var:trap_owner
							is_at_war_with = scope:trap_location.var:trap_owner.liege
						}
					}
				}
			}
			if = {
				limit = { location = { has_province_modifier = kupalas_exhalation_modifier2 } }
				scope:army ?= {
					army_commander ?= { POD_increase_wounds_effect = { COUNT = POD_double_cold_damage REASON = cold } }
					army_owner = {
						every_knight = {
							limit = { location = scope:trap_location }
							POD_increase_wounds_effect = { COUNT = POD_double_cold_damage REASON = cold }
						}
					}
				}
			}
			else = {
				scope:army ?= {
					army_commander ?= { POD_increase_wounds_effect = { COUNT = POD_base_cold_damage REASON = cold } }
					army_owner = {
						every_knight = {
							limit = { location = scope:trap_location }
							POD_increase_wounds_effect = { COUNT = POD_base_cold_damage REASON = cold }
						}
					}
				}
			}

			location = {
				remove_province_modifier = kupalas_exhalation_modifier
				remove_province_modifier = kupalas_exhalation_modifier2
				remove_variable = trap_owner
			}
		}
	}
}
