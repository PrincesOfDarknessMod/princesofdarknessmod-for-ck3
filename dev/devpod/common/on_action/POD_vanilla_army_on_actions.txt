# Appending our additions to vanilla on_actions in a separate file, to keep vanilla diffs clean


# called for armies whenever they enter a province
# root is the owner of the army
# scope:army is the army
on_army_enter_province = {
	on_actions = { POD_on_army_enter_province }
}


POD_on_army_enter_province = {
	effect = {


		#Mokole Rulers of the Sea perk
		if = {
			limit = {
				scope:army = {
					location ?= { is_sea_province = yes }
					army_owner = { 
						any_war_enemy = { 
							has_perk = rulers_of_the_sea_perk 
							save_temporary_scope_as = mokole_enemy
						}
					}
				}
			}
			scope:army = {
				random = { chance = scope:mokole_enemy.mokole_battle_effect_chance add_supply = -10 }
				army_commander ?= {
					random = {
						chance = scope:mokole_enemy.mokole_sea_effect_chance
						POD_increase_wounds_hostile_effect = { ATTACKER = scope:mokole_enemy COUNT = 1 REASON = sea_creatures }
					}
				}
				army_owner = {
					every_knight = {
						limit = { location = scope:army }
						random = {
							chance = scope:mokole_enemy.mokole_sea_effect_chance
							POD_increase_wounds_hostile_effect = { ATTACKER = scope:mokole_enemy COUNT = 1 REASON = sea_creatures }
						}
					}
				}
			}
		}
	}
}
