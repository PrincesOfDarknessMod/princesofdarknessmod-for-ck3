# Appending our additions to vanilla on_actions in a separate file, to keep vanilla diffs clean

on_combat_start = {
	on_actions = { POD_on_combat_start }
}

POD_on_combat_start = {
	effect = {
		# Track characters who had any armies raised (for Enticer objective)
		combat_war ?= {
			every_war_participant = {
				limit = { any_army = {} }
				prev = {
					add_to_variable_list = {
						name = combat_participants
						target = prev
					}
				}
			}
		}
	}
}

on_combat_unit_join_side = {
	on_actions = { POD_on_combat_unit_join_side }
}

POD_on_combat_unit_join_side = {
	effect = {
		# Track characters who had any armies raised (for Enticer objective)
		combat.combat_war ?= {
			every_war_participant = {
				limit = { any_army = {} }
				prev = {
					add_to_variable_list = {
						name = combat_participants
						target = prev
					}
				}
			}
		}
	}
}

on_combat_end_winner = {
	on_actions = { POD_on_combat_end_winner }
}

POD_on_combat_end_winner = {
	effect = {
		# Remove discipline effects that are used in battles
		if = {
			limit = {
				exists = side_commander
			}
			side_commander = {
				remove_character_flag = POD_used_disciplines_in_battle_flag
				if = {
					limit = {
						is_at_war = no
						liege_or_court_owner ?= { is_at_war = no }
					}
					POD_commander_clear_discipline_battle_effect = yes
				}
			}
		}
		
		every_side_knight = {
			remove_character_flag = POD_used_disciplines_in_battle_flag
			POD_knight_clear_discipline_battle_effect = yes
			
			POD_endowments_reward_small_effect = yes
		}
		
		combat = {
			random = {
				chance = 2
				location = {
					add_province_modifier = {
						modifier = nihil_modifier
						years = 5
					}
				}
			}
			# Track characters who had any armies raised (for Enticer objective)
			combat_war ?= {
				every_war_participant = {
					limit = { any_army = {} }
					prev = {
						add_to_variable_list = {
							name = combat_participants
							target = prev
						}
					}
				}
			}
		}
	}
}
