on_building_completed = {
	on_actions = { POD_on_building_completed }
}

POD_on_building_completed = {
	effect = {
		if = {
			limit = {
				NOT = { exists = scope:character }
			}
			root.county.holder = {
				save_scope_as = character
			}
		}
		scope:character = {
			if = {
				limit = { POD_should_keep_track_of_built_any_buildings = yes }
				POD_set_or_change_variable = { VAR = built_any_buildings VAL = 1 }
			}
		}
	}
}
