POD_random_yearly_journey_pulse = {
	trigger = {
		exists = var:current_journey
	}
	on_actions = {
        POD_journey_siochain_1_pulse
        POD_journey_siochain_2_pulse
	}
}


POD_journey_siochain_1_pulse = {
    trigger = {
        var:current_journey = { var:journey_type = flag:siochain_1  }
    }
    random_events = {
        chance_of_no_event = { value = 80 }
        100 = POD_siochain.6
    }
}
POD_journey_siochain_2_pulse = {
    trigger = {
        var:current_journey = { var:journey_type = flag:siochain_2  }
    }
    random_events = {
        chance_of_no_event = { value = 80 }
        100 = POD_siochain.7
    }
}
