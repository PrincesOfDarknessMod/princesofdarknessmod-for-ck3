# root = character
# scope:created_journey = the journey
# triggered after the journey story_cycle is set up
# doesn't necessarily mean the journey becomes your active journey. if it does, POD_on_journey_activated runs right after
POD_on_journey_created = {
	effect = {
		if = {
			limit = { scope:created_journey = { POD_is_seek_knowledge_journey_trigger = yes } }
			POD_increment_seek_knowledge_cost_effect = yes
		}
		if = {
			limit = { scope:created_journey.var:journey_type = flag:mokole_mnesis_quest }
			scope:created_journey = {
				set_variable = {
					name = dreamhunter_counter
					value = 0
				}
			}
		}
	}
}

# root = character
# scope:activated_journey = the journey
# triggered after the journey becomes your active journey (the one gaining monthly progress)
POD_on_journey_activated = {
	effect = {
		if = {
			limit = { scope:activated_journey.var:journey_type = flag:mokole_mnesis_quest }
			POD_mokole_mnesis_add_sleep_effect = yes
		}
		if = {
			limit = { scope:activated_journey.var:journey_type = flag:breaking_the_blond_bond }
			add_character_modifier = POD_breaking_the_blond_bond_modifier
		}
	}
}

# root = character
# scope:paused_journey = the journey
# triggered after the journey gets paused
POD_on_journey_paused = {
	effect = {
		if = {
			limit = { scope:paused_journey.var:journey_type = flag:mokole_mnesis_quest }
			POD_mokole_mnesis_clear_sleep_effect = yes
		}
	}
}

# root = character
# scope:abandoned_journey = the journey
# triggered before the journey story_cycle gets deleted
POD_on_journey_abandoned = {
	effect = {
		if = {
			limit = { scope:abandoned_journey = { POD_is_seek_knowledge_journey_trigger = yes } }
			POD_decrement_seek_knowledge_cost_effect = yes
		}
		if = {
			limit = {
				POD_is_active_journey = { JOURNEY = scope:abandoned_journey }
				scope:abandoned_journey.var:journey_type = flag:mokole_mnesis_quest
			}
			POD_mokole_mnesis_clear_sleep_effect = yes
		}
		if = {
			limit = { scope:abandoned_journey.var:journey_type = flag:breaking_the_blond_bond }
			remove_character_modifier = POD_breaking_the_blond_bond_modifier
		}
	}
}
