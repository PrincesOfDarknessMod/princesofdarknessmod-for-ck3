######################################
# Mary the Black — On-Action Hooks
######################################

# Hook into PoD's existing yearly playable pulse to fire Mary's yearly maintenance check.
# Multiple on_action declarations stack additively, so this does not override PoD.
POD_yearly_playable_pulse = {
	events = {
		POD_maryb.50	# Mary the Black yearly maintenance (commissions, A Lingering Debt)
	}
}

# Hook into vanilla on_title_gain via the events list. PoD and vanilla both
# define an 'effect' block on on_title_gain; only one such block may exist
# across mods, so commission completion is handled inside an event (POD_maryb.51)
# that receives scope:title from the on-action.
on_title_gain = {
	events = {
		POD_maryb.51
	}
}

# Hook into vanilla on_character_faith_change for Michael's permanent faith
# lock. POD_maryb.221's own trigger filters to characters carrying the lock
# flag with a non-codex faith; everyone else hits the trigger and the event
# silently no-ops, so the global hook is cheap.
on_character_faith_change = {
	events = {
		POD_maryb.221
	}
}
