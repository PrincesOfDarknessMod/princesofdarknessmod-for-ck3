on_title_gain = {
	on_actions = { POD_on_title_gain }
}

POD_on_title_gain = {
	events = {
		POD_maintenance.2
		POD_character_initialization.100
	}
	effect = {
		trigger_event = POD_maintenance.9
		
		if = {
			limit = {
				scope:title = {
					tier = tier_county
					is_coastal_county = yes
				}
				OR = {
					AND = {
						scope:previous_holder ?= { POD_is_wyrm_servant_trigger = no }
						POD_is_wyrm_servant_trigger = yes
					}
					AND = {
						scope:previous_holder ?= { POD_is_wyrm_servant_trigger = yes }
						POD_is_wyrm_servant_trigger = no
					}
				}
			}
			POD_fera_add_renown_effect = { TYPE = harmony }
		}
		
		scope:title = { update_dynamic_coa = yes }
		
		if = {
			limit = { scope:title = { tier >= tier_kingdom } }
			trigger_event =	POD_title_flavor.0002
		}
		
		#if = {
		#	limit = {
		#		scope:title = { tier >= tier_kingdom }
		#		is_vampire_trigger = yes
		#		is_ai = no
		#		NOT = { has_character_flag = warning_primogen }
		#	}
		#	trigger_event =	POD_primogens.0001
		#}
		
		if = {
			limit = {
				is_anh_trigger = yes
				# disabled for the AI until we can make them smarter about choosing their domain
				is_ai = no
				POD_suppress_clan_curse_trigger = no
			}
			scope:title = { POD_apply_anh_clan_curse = yes }
		}
		else = {
			scope:title = { POD_remove_anh_clan_curse = yes }
		}
		
		if = {
			limit = {
				this = character:790315
				scope:title = title:c_guangzhou_1
				is_ai = no
			}
			set_global_variable = pod_achiev_macao # Achievement: Am I Early?
		}
	}
}


on_title_gain_inheritance = {
	on_actions = { POD_on_title_gain_inheritance }
}

POD_on_title_gain_inheritance = {
	events = {
		delay = { days = 1 }
		POD_maintenance.9
		POD_character_initialization.100
	}
	effect = {
		if = {
			limit = { scope:title = { tier >= tier_kingdom } }
			trigger_event =	POD_title_flavor.0002
		}
		if = {
			limit = {
				scope:previous_holder = { is_a_cultist = yes }
				religion = religion:demons_religion
				NOT = { has_trait = demon }
				is_a_cultist = no
			}
			scope:previous_holder = {
				random_relation = {
					type = worshipped_demon
					save_scope_as = demon
				}
			}
			set_relation_worshipped_demon = scope:demon
			set_character_faith = scope:demon.faith
			if = {
				limit = { has_trait = supehunter }
				scope:demon = { save_scope_as = vampiresire }
				POD_hunter_enthrall_effect = yes
			}
		}
	}
}


on_title_gain_usurpation = {
	on_actions = { POD_on_title_gain_usurpation }
}

POD_on_title_gain_usurpation = {
	events = {
		delay = { days = 1 }
		POD_maintenance.9
	}
	effect = {
		if = {
			limit = { scope:title = { tier >= tier_kingdom } }
			POD_fera_add_renown_effect = { TYPE = infamy }
		}

		if = {
			limit = {
				scope:new_holder ?= {
					has_trait = kithain
					religion != ROOT.religion
					situation:PoD_war_of_seasons ?= { situation_top_has_catalyst = PoD_war_of_seasons_usurp_title_court }
				}
			}
			situation:PoD_war_of_seasons ?= {
				trigger_situation_catalyst = {
					catalyst = PoD_war_of_seasons_usurp_title_court
					character = scope:new_holder
				}
			}
		}
	}
}


on_title_lost = {
	on_actions = { POD_on_title_lost }
}

POD_on_title_lost = {
	effect = {
		if = {
			limit = { is_playable_character = no }
			# Properly lose haven buildings
			trigger_event = {
				id = POD_maintenance.27
				days = 1
			}
			if = {
				limit = { POD_has_roguelike_curse = yes }
				POD_roguelike_ensure_gameover_effect = yes
			}
		}
		#To remove flag if a character becomes unlanded
		if = {
			limit = {
				is_landed = no
				is_immortal = yes
				has_character_flag = had_POD_checkpoint
			}
			remove_character_flag = had_POD_checkpoint
		}

		#Reset Cairo's Heliopolis faith
		if = {
			limit = {
				scope:title = { this = title:c_cairo }
				is_setite_trigger = yes
				NOT = { has_global_variable = pathofheliopolis_active }
			}

			remove_character_flag = had_dreamcourt_event
			remove_character_flag = had_dustandbones_event
			remove_character_flag = had_crocodiletemple_event
			faith:pathofheliopolis = {
				remove_doctrine = tenet_carnal_exaltation
				remove_doctrine = tenet_hedonistic
				remove_doctrine = tenet_ritual_celebrations
				remove_doctrine = tenet_ashen_faith
				remove_doctrine = tenet_sacred_shadows
				remove_doctrine = tenet_pursuit_of_power
				remove_doctrine = tenet_struggle_submission
				remove_doctrine = tenet_gruesome_festivals
				remove_doctrine = tenet_warmonger

				if = {
					limit = { NOT = { has_doctrine = tenet_adaptive } }
					add_doctrine = tenet_adaptive
				}
				if = {
					limit = { NOT = { has_doctrine = tenet_legalism } }
					add_doctrine = tenet_legalism
				}
				if = {
					limit = { NOT = { has_doctrine = tenet_esotericism } }
					add_doctrine = tenet_esotericism
				}
			}
		}
	}
}


on_rank_up = {
	on_actions = { POD_on_rank_up }
}

POD_on_rank_up = {
	effect = {
		if = {
			limit = { NOT = { exists = scope:title } } # Were they un-titled?
			trigger_event = POD_maintenance.10
			if = {
				limit = { is_landed = yes } # No merc
				POD_adopt_correct_government_effect = yes
			}
		}
		POD_endowments_reward_huge_effect = yes
	}
}


on_vassal_gained = {
	on_actions = { POD_on_vassal_gained }
}

POD_on_vassal_gained = {
	events = {
		POD_maintenance.6
	}
	effect = {
		POD_endowments_reward_small_effect = yes
		if = {
			limit = {
				has_trait = demon
				is_ai = yes
				scope:vassal = {
					is_a_cultist = no
				}
			}
			scope:vassal = { set_relation_worshipped_demon = root }
		}

		if = {
			limit = {
				has_trait = kithain
				faith = { religion = religion:summer_court_religion }
				
				situation:PoD_war_of_seasons ?= {
					var:summer_leader ?= {
						save_temporary_scope_value_as = {
							name = vassal_amount_other
							value = {
								value = 0
								every_vassal = { add = 1 }
							}
						}
						ROOT != THIS
					}
				}
				save_temporary_scope_value_as = {
					name = vassal_amount_my
					value = {
						value = 0
						every_vassal = { add = 1 }
					}
				}
				
				scope:vassal_amount_other > scope:vassal_amount_my
			}
			POD_pulse_season_leader = { SEASON = religion:summer_court_religion }
		}
	}
}


on_baron_found_or_created_for_title = {
	on_actions = { POD_on_baron_found_or_created_for_title }
}

POD_on_baron_found_or_created_for_title = {
	effect = {
		if = {
			limit = {
				scope:liege = { POD_has_matriarchal_faith = yes }
				is_male = yes
			}
			POD_spawn_new_baron_for_title = {
				GENDER = female
				LIEGE = scope:liege
				TITLE = scope:title
			}
			death = { death_reason = death_disappearance }
		}
	}
	events = {
		POD_character_initialization.1
	}
}
