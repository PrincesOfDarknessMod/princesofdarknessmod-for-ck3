	effect = {

		# 		NOT = { var:fetter_owner = scope:owner }
		# 	}
		# 	var:fetter_owner = {
		# 		if = {
		# 			limit = { 
		# 				root = { var:fetter_value > 1 }
		# 			}
		# 			send_interface_toast = {
		# 				title = fetter_neglected_notification
		# 				left_icon = this
							
		# 				POD_lose_fetter_effect = { FETTER = root }
		# 			}
		# 		}
		# 		else = {
		# 			send_interface_toast = {

		# 				POD_destroy_fetter_effect = { FETTER = root }
		# 			}
		# 		}

		if = {
			limit = { scope:owner = { has_trait = wraith } }
			scope:owner = { POD_pulse_chance_passion = { TYPE = greed } }
		}
		
		scope:owner = {
			POD_greed_defilement_concession_effect = { VALUE = 1 }
		}

		# abandon heist journeys if the player gains the artifact through other means
		scope:owner = {
			if = {
				limit = {
					var:current_journey ?= {
						var:target_artifact ?= scope:artifact
						var:journey_type = flag:artifact_heist
					}
					# hacky way to not trigger this message on a successful heist
					NOT = { has_variable = hide_heist_window }
					NOT = { has_variable = current_heist }
				}
				var:current_journey.var:travel_plan ?= {
					resume_travel_plan = yes
					prev = { return_home = yes }
				}
				send_interface_message = {
					type  = POD_journey_abandoned
					title = POD_journey_abandoned_title
					desc  = POD_journey_abandoned_desc_artifact
					right_icon = scope:artifact
				}
				POD_journey_cleanup_effect = yes
			}
			every_in_list = {
				variable = paused_journeys
				limit = {
					var:target_artifact ?= scope:artifact
					var:journey_type = flag:artifact_heist
				}
				prev = {
					remove_list_variable = {
						name = paused_journeys
						target = prev
					}
					send_interface_message = {
						type  = POD_journey_abandoned
						title = POD_journey_abandoned_title
						desc  = POD_journey_abandoned_desc_artifact
						right_icon = scope:artifact
					}
				}
				end_story = yes
			}
		}
		
		# no charon artifacts for you
		if = {
			limit = { POD_artifact_is_unavailable_trigger = yes }
			destroy_artifact = this
		}
	}
}
