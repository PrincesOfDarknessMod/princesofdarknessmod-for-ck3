POD_umbra_maintenance_monthly_playable_pulse = {
	effect = {
		save_scope_as = player
		if = {
			limit = { has_variable = umbra_stationed_realm } 
			POD_change_umbra_progress_effect = { VICTIM = root }
		}
		every_relation = {
			type = umbra_explorer
			root = { POD_change_umbra_progress_effect = { VICTIM = prev } }
		}
		if = {
			limit = { has_variable_list = umbra_action_cooldown } 
			clear_variable_list = umbra_action_cooldown
		}
	}
}
