# Appending our additions to vanilla on_actions in a separate file, to keep vanilla diffs clean

on_character_created = {
	on_actions = { POD_on_character_created }
}

# Root is the new character; scope:creation_reason is the engine's reason flag 
POD_on_character_created = {
	effect = {
		# Temporary bosses: the template's after_creation has set pod_combat_boss by then. The
		# launcher still finishes a boss itself if that order ever changes.
		if = {
			limit = { has_character_flag = pod_combat_boss }
			debug_log = pod_combat_boss_created_hook_log
			debug_log_scopes = no
			POD_combat_boss_finish_creation_effect = yes
		}
	}
}
