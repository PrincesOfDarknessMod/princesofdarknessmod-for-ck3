# Appending our additions to vanilla on_actions in a separate file, to keep vanilla diffs clean


seduce_success = {
	on_actions = { POD_seduce_success }
}


POD_seduce_success = {
	effect = {
		scope:owner ?= {
			if = {
				limit = {
					faith = { has_doctrine_parameter = learn_secrets_from_seduction }
					scope:target ?= {
						OR = {
							AND = {
								has_any_secrets = yes
								any_secret = { NOT = { is_known_by = scope:owner } }
							}
							any_known_secret = { NOT = { is_known_by = scope:owner } }
						}
					}
				}
				scope:target = {
					every_secret = { reveal_to = scope:owner }
					every_known_secret = { reveal_to = scope:owner }
				}
			}
		}
	}
}
