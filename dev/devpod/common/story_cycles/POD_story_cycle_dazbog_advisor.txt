
POD_story_dazbog_advisor = {
	visible = yes
	
	icon = {
		reference = "gfx/interface/icons/faith/shadowinquisition.dds"
	}
	
	background = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}
	
	visualization = {
		custom_string_key = "POD_story_dazbog_advisor_custom_string"

		basic_counter = {
			variable_name = "years_left"
			min = 0
			max = 10
			label = "POD_story_dazbog_advisor_years_left_label"
			min_label = "POD_story_dazbog_advisor_years_left_min_label"
			max_label = "POD_story_dazbog_advisor_years_left_max_label"
		}

		decisions = {
			dazbog_brothers_guidance_decision
		}
	}
	
	on_setup = {
		POD_set_story_cycle_icon_bg_colors = {
			R = 0
			G = 0
			B = 0
			A = 0.5
		}
		POD_hide_story_cycle_header = yes
		POD_display_story_cycle_icon_right = yes

		set_variable = { name = years_left value = 10 }
	}
	
	effect_group = {
		years = 1
		chance = 100

		triggered_effect = {
			trigger = {
				NOT = { exists = var:requirement_is_satisfied }
				story_owner ?= { is_ai = no }
			}
			effect = {
				POD_set_or_change_variable = {
					VAR = years_left
					VAL = -1
				}

				if = {
					limit = { var:years_left <= 0 }
					set_variable = {
						name = years_left
						value = 0
					}
					set_variable = { name = requirement_is_satisfied value = yes }
				}

				story_owner ?= {
					if = {
						limit = {
							any_councillor = {
								count < 1
								has_faith = faith:shadowinquisition
							}
						}
						set_variable = { name = years_left value = 10 }
					}
				}
			}
		}
	}

	on_owner_death = {
		# Ensures continuity if your character dies before you can take the next decision
		if = {
			limit = {
				NOT = {
					is_target_in_global_variable_list = {
						name = unavailable_unique_decisions
						target = flag:dazbog_brothers_guidance_decision
					}
				}
			}
			story_owner = {
				primary_heir ?= {
					if = {
						limit = { has_faith = faith:warriors_dazbog }
						create_story = POD_story_dazbog_advisor  # will reset the timer
					}
				}
			}
		}
		end_story = yes
	}
}
