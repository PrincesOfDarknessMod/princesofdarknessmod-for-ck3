POD_story_umbra_adventure_inventory_item = {
	on_end = {
		if = {
			limit = { has_variable = artifact }
			destroy_artifact = var:artifact

		}
	}
}

POD_story_umbra_adventure = {
	on_setup = {
		save_scope_value_as = {
			name = index
			value = 0
		}
		while = {
			limit = { scope:index < 8 }

			character:POD_wraith_01 = {
				create_story = {
					type = POD_story_umbra_adventure_inventory_item
					save_scope_as = inventory_item
				}
			}

			scope:inventory_item = {
				set_variable = {
					name = index
					value = scope:index
				}
			}

			add_to_variable_list = {
				name = dummy_inventory
				target = scope:inventory_item
			}

			POD_increment_scope_value = { SCOPE = index }
		}
	}

	on_end = {
		every_in_list = {
			variable = inventory
			end_story = yes
		}

		every_in_list = {
			variable = dummy_inventory
			end_story = yes
		}

		if = {
			limit = { has_variable = node_map }
			var:node_map = {
				end_story = yes
			}
		}
	}
}
