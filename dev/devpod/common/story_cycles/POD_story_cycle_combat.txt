# them; each player's equipped list is pod_combat_loadout (character
# variable list of story references).

# POD_combat_log_entry — empty story_cycle type used as a per-turn log
# record in the combat window. One entry created per executed move by
# pod_combat_push_log_effect. Stories are OWNED by character:POD_wraith_01
# (Charon — same off-map anchor as the move catalogue), but REFERENCED
# from the combat host (PC) via the `pod_combat_log` variable list of
# story refs. The GUI iterates the host's list via datamodel and reads
# each entry's vars (move, actor, target, damage, result_code) directly
# from the Charon-owned story scope. New entries are inserted at the
# FRONT of the list (newest first) so the GUI shows recent actions at
# the top without ordering. All entries are end_story-d in
# pod_combat_cleanup_effect at combat end.
#
# result_code currently ranges 0..12 — see PODCombatLogResultText
# custom_loc in pod_combat_static_l_english.yml for the symbol-to-
# message mapping. Code 12 (stealth-triggered auto-miss) is the
# newest addition; pre-stealth result codes were 0..11.
#
# Storing the move story scope (rather than a numeric msg_id) lets the
# GUI read move-static properties like attack_type, damage_source,
# targeting, and stance flags directly from the move's own vars —
# no per-move duplication needed in the log.
POD_combat_log_entry = {
	on_setup = {}
	on_end = {}
	on_owner_death = {}
}


# POD_combat_sc_context — empty story_cycle type used as a lightweight scope

# scope chain at the moment configure_start_single_combat_effect was
# intercepted, plus the rule flags). One context story is created per
# duel-driven minigame run by pod_combat_sc_launch_minigame_effect on
# character:POD_wraith_01 (Charon — same anchor as the move catalogue).
# A pointer to the story is stashed on the PC as the variable
# pod_combat_sc_context. At combat end pod_combat.9999 reads through
# the pointer, restores all aliases / rule scopes, fires the saved
# follow-up event, then pod_combat_cleanup_effect ends the story and
# removes the pointer — atomic teardown of the entire context bag in
# two lines.
POD_combat_sc_context = {}
