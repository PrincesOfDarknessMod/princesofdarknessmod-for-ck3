######################################################
######   POD COMBAT - STORY_CYCLE TYPES
######################################################
# Five data-container story types for the combat mini-game, all declared
# empty: no story-cycle machinery runs on them, they exist only as scopes
# that carry variables. All five are owned by character:POD_wraith_01
# (Charon, immortal and off-map), so a story survives the PC dying
# mid-fight instead of tripping an owner-death cleanup race. Users hold
# references only, in variable lists or pointer vars.


# One story per catalogue move. Built at game start by
# POD_initialize_combat_moves (generated body in
# pod_combat_move_list_effects.txt; the POD_init_combat_move constructor
# and the setter effects it calls live in pod_combat_move_init_effects.txt),
# reached from common/on_action/POD_vanilla_game_start.txt. The global
# list POD_combat_moves is the whole catalogue; a character's equipped
# subset is the character variable list pod_combat_loadout. Only
# POD_reload_combat_moves rebuilds it, and never mid-combat.
POD_combat_move = {}


# One stackable conditional bonus on a move: its own condition (cond_target /
# cond_predicate / cond_threshold / cond_param, the same var names the
# move-level gate uses, so pod_combat_move_condition_met_trigger evaluates a
# clause verbatim) plus its own payload (bonus_effect / bonus_value /
# bonus_param, and the presence var hide_if_unmet for the stat card).
# Built by POD_combat_move_add_bonus_clause, hung off its move in the move's
# bonus_clauses variable list, and registered in the global list
# POD_combat_move_clauses so POD_reload_combat_moves can end them: the move
# stories that point at them are ended in the same pass, so the global list is
# the only handle that survives long enough to tear a clause down.
POD_combat_move_clause = {}


# One row of the rolling combat log. pod_combat_push_log_effect pushes one
# per resolved outcome (so an AoE phase pushes one per target); the move-less
# rows come from pod_combat_push_system_log_effect (lifecycle bookends), its
# _named sibling (rows naming a combatant, e.g. a mid-fight arrival), and the
# pod_combat_push_{mind,dot_tick,dot_applied,ap_drain}_log_effect family. All
# of them append through pod_combat_append_capped_log_entry_effect. Rows are
# appended oldest-first to the host's pod_combat_log list with a per-fight
# log_seq stamp, trimmed at pod_combat_log_max_entries by dropping the
# lowest seq, and end_story-d at combat cleanup. The log vbox renders
# bottomtotop, so oldest-first storage still displays newest-at-top with
# no per-push list rebuild.
#
# var:move is a live reference to the executed POD_combat_move, not a
# per-row copy, so the log dispatchers read the move's static attributes
# off it instead of duplicating them on every row; var:actor and
# var:target carry the character refs, target absent on self / team moves.
# Result is the outcome / attack_result / retal_fired flag model, where
# attack_result includes immune as its own resolved result. Per-primitive
# segment visibility (seg_<x>_on / seg_<x>_sep) is baked once at push
# time, so the GUI reads those flags rather than re-deriving them.
#
# Field catalogue: the COMBAT LOG section header in pod_combat_effects.txt.
# Result model: COMBAT_REFERENCE.info §4.
POD_combat_log_entry = {
	on_setup = {}
	on_end = {}
	on_owner_death = {}
}


# Per-combatant state for one fight. Identity is the pair (combat_team =
# flag:player | flag:enemy, team_position = int 0+, 0 = leader): there is
# no flat slot index, and team size is bounded only by the soft cap
# pod_combat_max_team_size. pod_combat_set_slot_char_effect is the single
# creation point and always creates, so a story exists for exactly the
# positions a fight seats; mid-fight arrivals seat through
# pod_combat_add_ally_midfight_effect / pod_combat_add_enemy_midfight_effect,
# which delegate to the same duel-builder add wrappers and then init the new
# slot with pod_combat_init_one_midfight_slot_effect. Slots are addressed by
# iterating the host's pod_combat_active_slots and matching the identity
# pair. The pointer vars on the fight are the two leaders, pod_combat_pc_slot
# (player/0) and pod_combat_enemy_leader_slot (enemy/0), plus
# pod_combat_active_control_slot, which follows whichever slot the human is
# currently driving rather than naming a fixed board position.
#
# The story holds per-fight gameplay state (slot_alive, slot_morale,
# hit / initiative, stance flags, tier mods, retaliate, the standing
# slot_target_* choice, and slot_human_controller when a human drives it). HP, soak and
# prowess are NOT slot storage: they are read live off var:slot_char, so
# wound-track penalties keep feeding back as a fight wears on.
# slot_mirror_total / _agg / _maxhp are a display-only wound-bar mirror.
#
# Ended by the fight story's on_end cascade, which walks the roster it holds;
# every slot var goes with the story, so no per-field cleanup pass is needed.
# Field catalogue and lifecycle: COMBAT_REFERENCE.info §3.
POD_combat_slot = {}


# The fight object: one per fight, created by POD_combat_create_duel_effect
# and ended by pod_combat_cleanup_effect (or by
# pod_combat_host_died_release_effect when the host dies mid-fight).
# Participants point at it through the character var pod_combat_session.
#
# Being Charon-owned rather than host-owned is what makes it the state
# owner: story vars stay writable after a participant dies, where a dead
# character's vars can only be read (see the dead-mutation rule in
# COMBAT_REFERENCE.info §7). Fight state is migrating here from the host
# character in stages; ctx_host names the character that anchors the
# fight's events (the pod_combat.9995 shell recipient), which is a
# separate question from where state lives.
#
# It carries the launcher's own handoff data too: the principals
# (initiator, attacker, defender), the fatality / fixed / locale rule
# flags and the follow-up event id - but never the caller's scope table.
# On the live path the shell event pod_combat.9995 inherits the launching
# event's whole scope table and holds it open for the fight, so
# pod_combat.9999 takes only the follow-up id from here.
# POD_combat_restore_sc_context_effect and
# POD_combat_restore_sc_rule_flags_effect run on the raw-GUI fallback
# route instead, where the parent's own aliases are unrecoverable and the
# follow-up runs degraded. Full model: the file header of
# pod_combat_sc_context_effects.txt.
POD_combat_fight = {
	# Child-story cascade. Both lists live on this story, so the iterators are
	# already in the right scope, and ending the fight tears down every slot and
	# every log row with it. That is why the teardown paths no longer carry
	# their own end_story loops: leaving one in place would end a story here and
	# then end it again from this block.
	#
	# pod_combat_turn_order_list is deliberately NOT cascaded: it is a reordered
	# VIEW of the same slot stories, so ending it would end each alive slot twice.
	# Nor are the move refs a slot or log row holds (slot_moves_used_*, entry
	# var:move) - those point at the permanent catalogue.
	on_setup = {}
	on_end = {
		every_in_list = {
			variable = pod_combat_active_slots
			end_story = yes
		}
		every_in_list = {
			variable = pod_combat_log
			end_story = yes
		}
	}
	# Charon owns this and is immortal, so this is unreachable - declared empty
	# to say so, and to keep owner death from ever firing the cascade.
	on_owner_death = {}
}
