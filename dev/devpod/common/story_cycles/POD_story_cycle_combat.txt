# off-map and immortal). The global list POD_combat_moves holds all of
# them; each player's equipped list is pod_combat_loadout (character
# variable list of story references).
POD_combat_move = {}


# POD_combat_log_entry: empty story_cycle type used as a per-turn log
# record in the combat window. One entry created per executed move by
# pod_combat_push_log_effect. Stories are OWNED by character:POD_wraith_01
# (Charon — same off-map anchor as the move catalogue), but REFERENCED
# from the combat host (PC) via the `pod_combat_log` variable list of
# story refs. The GUI iterates the host's list via datamodel and reads
# each entry's vars (move, actor, target, damage, result) directly
# from the Charon-owned story scope. New entries are inserted at the
# FRONT of the list (newest first) so the GUI shows recent actions at
# the top without ordering. All entries are end_story-d in
# pod_combat_cleanup_effect at combat end.
#
# result is a flag-valued var (hp_damage / guard_blocked / soaked /
# missed / dodged / hp_damage_retal / soaked_retal / utility /
# morale_damage / hp_healed / morale_healed / stealthed). See
# PODCombatLogResultText custom_loc in pod_combat_log_custom_loc.txt
# for the dispatch and COMBAT_REFERENCE.info §4 for the full map.
#
# Storing the move story scope (rather than a numeric msg_id) lets the
# GUI read move-static properties — emitted capability flags
# (requires_hit_roll, checks_guard, checks_stance,
# checks_spirit_gate), damage_source, damage_type, targeting,
# combat_category — directly from the move's own vars without
# per-move duplication in the log.
POD_combat_log_entry = {
	on_setup = {}
	on_end = {}
	on_owner_death = {}
}


# POD_combat_slot: empty story_cycle type used as a per-slot data
# container during combat. Ten of these stories are created on Charon
# at combat init (one per slot, indices 0-9) and end_story-d at combat
# cleanup. Each slot story holds the combatant's per-slot state
# (slot_index, slot_char, slot_alive, soak, prowess, stance flags,
# tier mods, retaliate, etc.). This is the sole storage for per-slot
# combat state.
#
# Stored on character:POD_wraith_01 (Charon, same off-map anchor used
# by the move catalogue, log entries, and sc_context). The combat
# host references the slot stories two ways:
#   1. pod_combat_slots / pod_combat_active_slots — variable lists of
#      story refs. pod_combat_slots is all 10 stories (filled or not);
#      pod_combat_active_slots is the subset with slot_char set, which
#      every dispatch loop iterates.
#   2. pod_combat_slot_{0..9} — ten named pointer vars, one per slot,
#      used for parametric direct access (`var:pod_combat_slot_$SLOT$`
#      resolves to the slot story at compile time when $SLOT$ is a
#      parametric integer).
# end_story-d in pod_combat_clear_slot_stories_effect, which is called
# from pod_combat_group_cleanup_effect at combat end. All slot vars go
# with the story, so no per-field cleanup pass is needed.
POD_combat_slot = {}


# POD_combat_sc_context: empty story_cycle type used as a lightweight scope
# carrying the entire single-combat / group-combat caller context (the
# scope chain at the moment configure_start_single_combat_effect was
# intercepted, plus the rule flags). One context story is created per
# duel-driven minigame run by pod_combat_sc_launch_minigame_effect on
# character:POD_wraith_01 (Charon — same anchor as the move catalogue).
# A pointer to the story is stashed on the PC as the variable
# pod_combat_sc_context. At combat end pod_combat.9999 reads through
# the pointer, restores all aliases / rule scopes, fires the saved
# follow-up event, then pod_combat_cleanup_effect ends the story and
# removes the pointer. Atomic teardown of the entire context bag in
# two lines.
POD_combat_sc_context = {}
