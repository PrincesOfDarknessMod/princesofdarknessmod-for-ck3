######################################################
# Mary the Black — Story Cycle
######################################################
#
# Surfaces the entire Mary the Black campaign in a single in-game story panel:
#
#   - tug_of_war_counter for the Mary/Michael balance (−100 to +100, mirroring
#     char var:mary_michael_balance)
#   - basic_counter for tablets recovered (0 to 4, mirroring
#     char var:mary_tablets_recovered)
#   - custom_string_key for narrative flavour and per-tablet status
#   - modifiers list surfacing whichever of Mary's path/status modifiers are
#     currently active (Fragmented Self tier, Most Hated, Angel's Soul, Tanith Protection /
#     Legacy / Commission, the eventual Black Dream / Second Dream / Ash and
#     Gold resolution modifier)
#
# Created in podintroductions.380's immediate block (both options) so the
# panel appears as soon as Mary makes her opening choice. Lives until Mary
# dies. Story variables are synced from character variables via
# POD_maryb_sync_story_effect, which is called from every event that 
# mutates the underlying character variables (tablet recovery, 
# commission completion, deal resolution).
#
######################################################

POD_maryb_story = {
	visible = yes

	# Mary/Michael journey icon. Placeholder is a copy of PoD's
	# core_tenet_michael.dds (the gold-italic Michael, matching the panel's
	# "The Black Dream" title). Drop custom art at this path to replace it.
	icon = {
		reference = "gfx/interface/icons/story_cycle/mary_michael.dds"
	}

	background = {
		reference = "gfx/interface/illustrations/decisions/decision_POD_demonology.dds"
	}

	visualization = {
		custom_string_key = "POD_maryb_story_custom_string"

		tug_of_war_counter = {
			variable_name = "michael_balance"
			min = -100
			max = 100
			label = "POD_maryb_story_balance_label"
			min_label = "POD_maryb_story_balance_min_label"
			max_label = "POD_maryb_story_balance_max_label"
		}

		basic_counter = {
			variable_name = "tablets_recovered"
			min = 0
			max = 3
			label = "POD_maryb_story_tablets_label"
			min_label = "POD_maryb_story_tablets_min_label"
			max_label = "POD_maryb_story_tablets_max_label"
		}

		modifiers = {
			mary_fragmented_self_modifier
			mary_fragmented_self_modifier_t1
			mary_fragmented_self_modifier_t2
			mary_most_hated_modifier
			angel_soul_modifier
			mary_tanith_protection_modifier
			mary_tanith_legacy_modifier
			mary_tanith_commission_diyarbakr_modifier
			mary_tanith_commission_cyprus_modifier
			mary_tanith_commission_cyprus_brujah_modifier
			mary_tanith_commission_olives_modifier
			mary_black_dream_modifier
			mary_second_dream_modifier
			mary_ash_and_gold_modifier
			mary_ash_and_gold_michael_modifier
			mary_ash_and_gold_mary_modifier
		}
	}

	on_setup = {
		# Mirror character variables into story-scope variables on creation.
		# story_owner is Mary; the variables may or may not already exist
		# depending on whether podintroductions.380's immediate block runs
		# the mirror before or after this on_setup.
		set_variable = { name = michael_balance value = 0 }
		set_variable = { name = tablets_recovered value = 0 }
		if = {
			limit = { story_owner ?= { has_variable = mary_michael_balance } }
			set_variable = {
				name = michael_balance
				value = story_owner.var:mary_michael_balance
			}
		}
		if = {
			limit = { story_owner ?= { has_variable = mary_tablets_recovered } }
			set_variable = {
				name = tablets_recovered
				value = story_owner.var:mary_tablets_recovered
			}
		}
	}

	# Yearly defensive re-sync. Most updates already push from the firing event
	# via POD_maryb_sync_story_effect, but if something modifies the underlying
	# character variables outside our event chain, the panel still catches up.
	#
	# The `exists = var:...` checks in the trigger serve double duty: they
	# satisfy the engine's static-analysis "variable set but never read"
	# pass (the visualization block's `variable_name = ...` does not count
	# as a read), and they short-circuit the sync if the story panel hasn't
	# been initialized yet.
	effect_group = {
		years = 1
		chance = 100

		triggered_effect = {
			trigger = {
				story_owner ?= { is_alive = yes }
				exists = var:michael_balance
				exists = var:tablets_recovered
			}
			effect = {
				story_owner ?= {
					POD_maryb_sync_story_effect = yes
				}
			}
		}
	}

	on_owner_death = {
		end_story = yes
	}
}
