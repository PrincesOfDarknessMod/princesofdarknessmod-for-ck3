######################################################
# Depth Accumulation Story Cycle
# Created when a character receives Soul Infection
# Fires every 3 years to increment depth and check thresholds
# Only accumulates for Soul Infection holders (not Asakku)
######################################################

POD_asakku_depth_accumulation = {
	effect_group = {
		years = 3
		chance = 100

		triggered_effect = {
			trigger = {
				story_owner ?= {
					is_alive = yes
					POD_asakku_has_any_asakku_trait_trigger = yes
				}
			}
			effect = {
				story_owner ?= {
					# Soul Infection carriers get full passive accumulation
					if = {
						limit = { has_trait = POD_asakku_soul_infection }
						POD_asakku_increment_depth_effect = { AMOUNT = POD_asakku_depth_accumulation_amount }
						POD_asakku_check_depth_thresholds_effect = yes
					}
					# Asakku characters accumulate at a slower rate (+1 per cycle)
					else_if = {
						limit = { has_trait = POD_asakku_asakku }
						POD_asakku_increment_depth_effect = { AMOUNT = 1 }
						POD_asakku_check_depth_thresholds_effect = yes
					}
				}
			}
		}
	}

	on_owner_death = {
		end_story = yes
	}
}
