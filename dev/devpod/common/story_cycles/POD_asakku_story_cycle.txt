######################################################
# Depth Accumulation Story Cycle
# Created when a character receives Soul Infection
# Fires every 3 years to increment depth and check thresholds
# Surfaces asakku_depth via the in-game story cycle visualization
# (basic_counter), driven by a story-scope `depth` variable that
# is mirrored from character var:asakku_depth on every change.
######################################################

POD_asakku_depth_accumulation = {
	visible = yes

	icon = {
		reference = "gfx/interface/icons/traits/POD_asakku_asakku.dds"
	}

	background = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}

	visualization = {

		basic_counter = {
			variable_name = "depth"
			min = 0
			max = 100
			label = "POD_asakku_story_depth_label"
			min_label = "POD_asakku_story_depth_min_label"
			max_label = "POD_asakku_story_depth_max_label"
		}

		decisions = {
			POD_asakku_phase1_objective_decision
			POD_asakku_phase2_objective_decision
			POD_asakku_phase3_objective_decision
			POD_asakku_phase4_objective_decision
		}

		custom_string_key = "POD_asakku_story_custom_string"

	}

	on_setup = {
		# Initialise the story-scope mirror from the owner's current depth
		set_variable = {
			name = depth
			value = story_owner.POD_asakku_depth
		}
	}

	effect_group = {
		years = 3
		chance = 100

		triggered_effect = {
			trigger = {
				story_owner ?= {
					is_alive = yes
					POD_asakku_has_any_asakku_trait_trigger = yes
					POD_is_not_torpid = yes
				}
			}
			effect = {
				story_owner ?= {
					# Soul Infection carriers get full passive accumulation,
					# scaled by Vicissitude trait. Inline-selected to avoid
					# passing a script value through $AMOUNT$ substitution.
					if = {
						limit = { has_trait = POD_asakku_soul_infection }
						if = {
							limit = { has_trait = vicissitudeadvanced }
							POD_asakku_change_depth_effect = { AMOUNT = 3 }
						}
						else_if = {
							limit = { has_trait = vicissitude }
							POD_asakku_change_depth_effect = { AMOUNT = 2 }
						}
						else = {
							POD_asakku_change_depth_effect = { AMOUNT = 1 }
						}
						POD_asakku_check_depth_thresholds_effect = yes
					}
					# Asakku characters accumulate at a slower rate (+1 per cycle)
					else_if = {
						limit = { has_trait = POD_asakku_asakku }
						POD_asakku_change_depth_effect = { AMOUNT = 1 }
						POD_asakku_check_depth_thresholds_effect = yes
					}
				}
			}
		}
	}


	on_owner_death = {
		end_story = yes
	}
}
