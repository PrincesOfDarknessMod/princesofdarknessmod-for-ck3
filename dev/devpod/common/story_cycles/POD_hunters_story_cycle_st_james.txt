#######################################################
# Paranoia/Trust Accumulation Story Cycle
# Created at the start of the game.
# Have a value named paranoia_or_trust, that mesures if
# the Sword of Saint James is going full paranoia or if
# they are open to work with the other orders.
#######################################################

POD_hunters_story_cycle_st_james = {
    visible = yes

	icon = {
		reference = "gfx/interface/icons/story_cycles/hunters_st_james.dds"
	}

	background = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}

	visualization = {

        custom_string_key = "POD_hunters_story_st_james_paranoia_or_trust_custom_string"
		tug_of_war_counter = {
			variable_name = "paranoia_or_trust"
			min = -100
			max = 100
			label = "POD_hunters_story_st_james_paranoia_or_trust_label"
			min_label = "POD_hunters_story_st_james_paranoia_or_trust_min_label"
			max_label = "POD_hunters_story_st_james_paranoia_or_trust_max_label"
		}

		decisions = {
		    st_james_support_objective_decision
			hunters_st_james_compostela_decision
		#	TODO: Decision to increase paranoia
        #   TODO: Decision to decrease paranoia
		}
        
        modifiers = {
            rodrigue_paranoia_modifier
        }

	}

	on_setup = {
        POD_hide_story_cycle_header = yes
        POD_display_story_cycle_icon_right = yes
		# Mirror character variables into story-scope variables on creation.
		# story_owner is Mary; the variables may or may not already exist
		# depending on whether podintroductions.380's immediate block runs
		# the mirror before or after this on_setup.
		set_variable = { name = paranoia_or_trust value = 0 }
		if = {
			limit = { story_owner ?= { has_variable = paranoia_or_trust } }
			set_variable = {
				name = paranoia_or_trust
				value = story_owner.var:paranoia_or_trust
			}
		}
    }

 	# Yearly defensive re-sync. Most updates already push from the firing event
	# via POD_maryb_sync_story_effect, but if something modifies the underlying
	# character variables outside our event chain, the panel still catches up.
	#
	# The `exists = var:...` checks in the trigger serve double duty: they
	# satisfy the engine's static-analysis "variable set but never read"
	# pass (the visualization block's `variable_name = ...` does not count
	# as a read), and they short-circuit the sync if the story panel hasn't
	# been initialized yet.   
	effect_group = {
		years = 1
		chance = 100
    
		triggered_effect = {
			trigger = {
				story_owner ?= { is_alive = yes }
				exists = var:paranoia_or_trust
			}
			effect = {
				story_owner ?= {
					POD_hunters_st_james_story_effect = yes
				}
			}
		}
	}

	effect_group = {}
	effect_group = {}

	on_end = {}
	on_owner_death = {
        make_story_owner = story_owner.primary_heir
    }
}
