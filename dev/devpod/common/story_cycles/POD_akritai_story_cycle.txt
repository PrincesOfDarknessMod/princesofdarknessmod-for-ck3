#create_story = POD_story_akritai_pulled_in_two_directions
#random_owned_story = { type = POD_story_akritai_pulled_in_two_directions end_story = yes }

POD_story_akritai_pulled_in_two_directions = {
	visible = yes

	icon = {
		reference = "gfx/interface/icons/faith/christianity_orthodox_cross_01.dds"
	}

	background = {
		reference = "gfx/interface/illustrations/decisions/decision_byz_throne.dds"
	}

	visualization = {

		tug_of_war_counter = {
			variable_name = "akritai_influence_balance"
			min = -100
			max = 100
			label = "POD_story_akritai_pulled_in_two_directions_influence_label"
			min_label = "POD_story_akritai_pulled_in_two_directions_influence_min_label"
			max_label = "POD_story_akritai_pulled_in_two_directions_influence_max_label"
		}

		basic_counter = {
			variable_name = "akritai_patience"
			min = 0
			max = 10
			label = "POD_story_akritai_pulled_in_two_directions_patience_label"
			min_label = "POD_story_akritai_pulled_in_two_directions_patience_min_label"
			max_label = "POD_story_akritai_pulled_in_two_directions_patience_max_label"
		}

		decisions = {
			akritai_hunters_duty_objective_decision
			akritai_recruitment_drives_objective_decision
		}

		custom_string_key = "POD_story_akritai_pulled_in_two_directions_custom_string"

	}

	on_setup = {
		POD_set_story_cycle_icon_bg_colors = {
			R = 0.5
			G = 0
			B = 0
			A = 0.5
		}

		#set_variable = { name = akritai_influence_balance value = 0 }
		story_owner ?= { set_variable = { name = akritai_influence_balance value = 0 } }
		if = {
			limit = { story_owner ?= { has_variable = akritai_influence_balance } }
			set_variable = {
				name = akritai_influence_balance
				value = story_owner.var:akritai_influence_balance
			}
		}

		#set_variable = { name = akritai_patience value = 10 }
		story_owner ?= { set_variable = { name = akritai_patience value = 10 } }
		if = {
			limit = { story_owner ?= { has_variable = akritai_patience } }
			set_variable = {
				name = akritai_patience
				value = story_owner.var:akritai_patience
			}
		}
	}

	# Defensive resync
	effect_group = {
		years = 1
		chance = 100

		triggered_effect = {
			trigger = {
				story_owner ?= {
					is_alive = yes
					has_faith = faith:akritai
					POD_story_akritai_pulled_in_two_directions_resolved_trigger = no
				}
				exists = var:akritai_influence_balance
				exists = var:akritai_patience
			}
			effect = {
				story_owner ?= {
					POD_story_akritai_pulled_in_two_directions_resync_effect = yes
				}
			}
		}
	}

	# Sometimes pull a random event to help shift the balance of power
	effect_group = {
		months = 6
		chance = 25

		triggered_effect = {
			trigger = {
				story_owner ?= {
					is_alive = yes
					has_faith = faith:akritai
					POD_story_akritai_pulled_in_two_directions_resolved_trigger = no
				}
			}
			effect = {
				story_owner ?= {
					# TODO AKRITAI: create event pool to shift things
				}
			}
		}
	}


	on_owner_death = {
		end_story = yes
	}
}
