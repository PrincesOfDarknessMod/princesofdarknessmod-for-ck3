POD_story_cycle_bone_lion = {
	effect_group = {
		years = 2
		chance = 100

		triggered_effect = {
			trigger = { always = yes }
			effect = {
				POD_set_or_change_variable = {
					VAR = damage_given
					VAL = 1
				}
				POD_increase_wounds_hostile_effect = { ATTACKER = var:caster COUNT = 1 REASON = duel_aggravated }
				if = {
					limit = { var:damage_given ?= 5 }
					end_story = yes
				}
			}
		}
	}

	on_owner_death = {
		end_story = yes
	}
}
