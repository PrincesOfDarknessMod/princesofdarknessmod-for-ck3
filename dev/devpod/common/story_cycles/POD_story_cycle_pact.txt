
POD_story_pact = {
	visible = yes
	
	icon = {
		reference = "gfx/interface/icons/court_position_types/cult_leader_court_position.dds"
	}
	
	background = {
		reference = "gfx/interface/illustrations/decisions/decision_POD_demonology.dds"
	}
	
	visualization = {
		# takes up too much space
		# character = {
		# 	variable_name = "demon"
		# 	label = "POD_story_pact_demon"
		# }
		
		custom_string_key = "POD_story_pact_custom_string"
	}
	
	effect_group = {
		years = 1
		chance = 100

		triggered_effect = {
			trigger = { story_owner ?= { is_ai = no } }
			effect = {
				POD_set_or_change_variable = {
					VAR = years_passed_frequency
					VAL = 1
				}
				if = {
					limit = { 
						var:years_passed_frequency >= var:frequency
						NOT = { POD_pact_obligation_is_prohibitory_trigger = { OBLIGATION = var:obligation } }
					}
					if = {
						limit = { var:completed = no }
						POD_pact_consequence_from_fine_print = { 
							THRALL = story_owner
							CONSEQUENCE = var:consequence
						}
					}
					else = {
						set_variable = { name = completed value = no }
					}
					set_variable = {
						name = years_passed_frequency
						value = 0
					}
				}
				story_owner ?= {
					if = {
						limit = { is_at_war = no }
						POD_pact_break_prohibition = { OBLIGATION = power_of_war }
					}

					if = {
						limit = { POD_health_has_any_wound_trigger = no }
						POD_pact_break_prohibition = { OBLIGATION = power_of_pain }
					}
				}

				if = {
					limit = { NOT = { POD_pact_obligation_is_prohibitory_trigger = { OBLIGATION = var:obligation } } }
					set_variable = {
						name = years_until_obligation
						value = {
							add = var:frequency
							subtract = var:years_passed_frequency
							min = 0
						}
					}
				}
			}
		}
	}

	on_owner_death = {
		end_story = yes
	}
}

POD_story_pact_stipulation = {
	on_owner_death = {
		end_story = yes
	}
}
