
POD_story_pact = {

	on_setup = {
	}

	on_end = {
	}

	effect_group = {
		years = 1
		chance = 100

		triggered_effect = {
			trigger = { always = yes }
			effect = {
				POD_set_or_change_variable = {
					VAR = years_passed_frequency
					VAL = 1
				}
				if = {
					limit = { 
						var:years_passed_frequency >= var:frequency
						NOT = { POD_pact_obligation_is_prohibitory_trigger = { OBLIGATION = var:obligation } }
					}
					if = {
						limit = { var:completed = no }
						POD_pact_consequence_effect = { 
							THRALL = story_owner
							CONSEQUENCE = var:consequence
						}
					}
					else = {
						set_variable = { name = completed value = no }
					}
					set_variable = {
						name = years_passed_frequency
						value = 0
					}
				}
				story_owner ?= {
					if = {
						limit = { is_at_war = no }
						POD_pact_complete_obligation_effect = { OBLIGATION = power_of_war }
					}

					if = {
						limit = { 
							NOR = {
								POD_health_has_superficial_wound_trigger = yes
								POD_health_has_aggravated_wound_trigger = yes
							}
						}
						POD_pact_complete_obligation_effect = { OBLIGATION = power_of_pain }
					}
				}
			}
		}
	}

	on_owner_death = {
		end_story = yes
	}
}
