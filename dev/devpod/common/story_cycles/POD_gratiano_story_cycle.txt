# Gratiano de Veronese - conspiracy story and dashboard
# Created by podintroductions.128; Born to Rule and POD_gratiano.0010 provide fallbacks.
# phase: 1 Proclamation | 2 Courtier (Plan I) | 3 Liberator (Plan II) | 4 Kingslayer (Plan III)
######################################

POD_gratiano_story = {
	visible = yes

	icon = {
		reference = "gfx/interface/icons/traits/lasombra.dds"
	}

	background = {
		reference = "gfx/interface/illustrations/decisions/decision_faith_in_shadows.dds"
	}

	visualization = {
		custom_string_key = "POD_gratiano_story_custom_string"

		basic_counter = {
			variable_name = "trust_mirror"
			min = 0
			max = 100
			label = "POD_gratiano_story_trust_label"
			min_label = "POD_gratiano_story_trust_min_label"
			max_label = "POD_gratiano_story_trust_max_label"
		}

		basic_counter = {
			variable_name = "conspiracy_mirror"
			min = 0
			max = 100
			label = "POD_gratiano_story_conspiracy_label"
			min_label = "POD_gratiano_story_conspiracy_min_label"
			max_label = "POD_gratiano_story_conspiracy_max_label"
		}

		decisions = {
			gratiano_decision
			POD_gratiano_academy_decision
			POD_gratiano_curry_favor_decision
			POD_gratiano_pursue_giovanni_decision
			POD_gratiano_pursue_ventrue_decision
			POD_gratiano_pursue_moncada_decision
			POD_gratiano_pursue_nahir_decision
			POD_gratiano_strike_the_castle
		}

		modifiers = {
			POD_gratiano_dominated_mind
			POD_gratiano_spirit_displeasure
			POD_gratiano_assamite_vengeance
		}
	}

	on_setup = {
		set_variable = { name = phase value = 1 }
		story_owner ?= {
			# Preserve existing counters when restoring the story in a save.
			if = {
				limit = { NOT = { has_variable = POD_gratiano_trust } }
				set_variable = { name = POD_gratiano_trust value = 100 }
			}
			if = {
				limit = { NOT = { has_variable = POD_gratiano_conspiracy } }
				set_variable = { name = POD_gratiano_conspiracy value = 0 }
			}
			add_character_flag = POD_gratiano_started
			# Set the Proclamation gate before the intro is answered, or Courtier content briefly appears.
			if = {
				limit = { NOT = { has_character_modifier = bad_gratiano_modifier } }
				add_character_modifier = bad_gratiano_modifier
			}
		}
		# Leave mirror variables unset until Courtier to hide the Proclamation counters.

		POD_add_journey_to_story_cycle = { TYPE = POD_gratiano_pact_journey }

		story_owner ?= {
			if = {
				limit = { is_ai = no }
				send_interface_toast = {
					type = event_generic_neutral
					title = POD_gratiano_story_begins.toast
					left_icon = this
				}
			}
		}
	}

	# Quarterly phase, milestone and counter sync.
	effect_group = {
		months = 3
		chance = 100

		triggered_effect = {
			trigger = {
				# Must also run in Proclamation, before mirror variables exist.
				story_owner ?= { is_alive = yes }
			}
			effect = {
				story_owner ?= {
					POD_gratiano_story_sync_effect = yes
				}
			}
		}
	}

	# Yearly Trust loss during the active conspiracy.
	effect_group = {
		years = 1
		chance = 100

		triggered_effect = {
			trigger = {
				var:phase >= 2
				var:phase <= 4
				story_owner ?= {
					is_alive = yes
					is_ai = no
				}
			}
			effect = {
				story_owner ?= {
					POD_gratiano_change_trust_effect = { DELTA = POD_gratiano_trust_bleed_value }
				}
			}
		}
	}

	on_owner_death = {
		end_story = yes
	}
}
