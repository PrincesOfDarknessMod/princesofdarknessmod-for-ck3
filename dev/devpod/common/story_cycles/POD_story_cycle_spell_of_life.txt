
POD_story_custom_mummy_dynasty = {

	on_setup = {
	}

	on_end = {
		if = {
			limit = { has_variable = custom_name }
			
			var:root_char = {
				remove_localized_text = POD_custom_mummy_dynasty_name_text
				remove_variable = POD_custom_mummy_dynasty_name_text
			}
		}
		if = {
			limit = { has_variable = custom_shadow_realm }
			
			var:root_char = {
				remove_localized_text = POD_custom_mummy_dynasty_shadow_realm_text
				remove_variable = POD_custom_mummy_dynasty_shadow_realm_text
			}
		}
	}

	on_owner_death = {}
}
