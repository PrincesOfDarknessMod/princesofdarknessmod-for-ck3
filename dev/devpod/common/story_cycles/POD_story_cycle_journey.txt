# use POD_start_journey_effect so that variables can get set up properly
POD_story_journey = {

	on_setup = {
		set_variable = {
			name  = total_progress
			value = 0
		}
		set_variable = {
			name  = phase
			value = flag:planning_phase
		}
	}

	on_end = {
	}

	on_owner_death = {
		end_story = yes
	}

}

POD_story_coterie = {

	on_setup = {
	}

	on_end = {
		if = {
			limit = { has_variable = custom_name }
			
			var:root_char = {
				remove_localized_text = POD_coterie_name_text
				remove_variable = POD_coterie_name_text
			}
		}
		if = {
			limit = { has_variable_list = members } # error suppression
			# Because it seems like a bad idea to remove entries from a list you're iterating on
			every_in_list = {
				variable = members
				prev = {
					add_to_variable_list = {
						name = temp_members
						target = prev
					}
				}
				send_interface_toast = {
					title = POD_coterie_disbanded_title
					left_icon = this
					custom_tooltip = POD_coterie_disbanded_tt
				}
			}
			every_in_list = {
				variable = temp_members
				POD_leave_coterie_effect = yes
			}
		}
	}

	on_owner_death = {
	}

}
