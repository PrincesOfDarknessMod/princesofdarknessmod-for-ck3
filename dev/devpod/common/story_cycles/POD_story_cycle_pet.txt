POD_story_pet_dragon = {
	on_setup = {
		assign_dragon_gender_effect = { GENDER = random }
		set_variable = {
			name = dragon_age_variable
			value = 0
		}
		story_owner = {
			add_character_modifier = {
				modifier = dragon_story_modifier
			}
			add_character_flag = had_dragon_story
		}
	}
	on_end = {
		story_owner = {
			remove_dragon_story_modifiers_effect = yes
			remove_dragon_name_effect = yes
		}
	}
	on_owner_death = {
		scope:story = { end_story = yes }
	}

	# Give the dragon a name!
	effect_group = {
		days = { 2 4 }
		chance = 100
	
		triggered_effect = {
			trigger = {
				story_owner = {
					NOT = { has_character_flag = is_naming_dragon }
				}
				NOT = { exists = story_owner.var:story_cycle_dragon_name }
			}
			effect = {
				story_owner = {
					add_character_flag = is_naming_dragon
					trigger_event = POD_umbra_pet.101
				}
			}
		}
	}

	#To save the name in the story's scope (can for some reason not be done in the naming event itself)
	effect_group = {
		days = 5
		chance = 100
	
		triggered_effect = {
			trigger = {
				exists = story_owner.var:story_cycle_dragon_name
			}
			effect = {
				set_variable = {
					name = story_cycle_dragon_name
					value = scope:story.story_owner.var:story_cycle_dragon_name
				}
			}
		}
	}

	#Age the dragon
	effect_group = {
		days = 365

		trigger = {
			exists = var:dragon_age_variable
		}

		triggered_effect = {
			trigger = { always = yes }
			effect = {
				change_variable = {
					name = dragon_age_variable
					add = 1
				}
			}
		}
	}

	# The dragon dies :(
	effect_group = {
		days = { 5000 10000 }
		chance = 100
	
		triggered_effect = {
			trigger = {
				exists = story_owner.var:story_cycle_dragon_name
				fae_era >= 2
			}
			effect = {
				story_owner = {
					add_character_flag = {
						flag = dragon_is_dying
					}
					trigger_event = POD_umbra_pet.199
				}
			}
		}
	}

	# Something happens - random events
	effect_group = {
		days = { 365 600 }
		chance = 50

		trigger = {
			exists = story_owner.var:story_cycle_dragon_name
			story_owner = {
				NOT = {
					has_character_flag = dragon_is_dying
				}
			}
		}

		first_valid = {
			triggered_effect = {
				trigger = { always = yes }
				effect = {
					story_owner = {
						trigger_event = {
							on_action = ongoing_dragon_events
						}
					}
				}
			}
		}
	}
}
