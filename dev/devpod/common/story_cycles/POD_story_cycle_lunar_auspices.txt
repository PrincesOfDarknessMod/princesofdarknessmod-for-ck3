POD_story_lunar_auspice_ahroun = {
	visible = yes

	icon = { reference = "gfx/interface/icons/traits/ahroun.dds" }
	background = { reference = "gfx/interface/illustrations/decisions/decision_get_of_fenris.dds" }

	on_setup = {
		POD_hide_story_cycle_header = yes
		POD_add_sgui_checkbox_to_story_cycle = { SGUI = ahroun_snarl_sgui }
		set_variable = auspice_story
	}

	visualization = {
		traits = { ahroun }
		character_list = {
			variable_name = "ahroun_powerful_targets"
			label = "POD_story_lunar_auspice_ahroun_powerful_target_list"
		}
		character_list = {
			variable_name = "ahroun_targets"
			label = "POD_story_lunar_auspice_ahroun_target_list"
		}

		custom_string_key = "POD_story_lunar_auspice_ahroun_extra_text"
	}


	effect_group = {
		years = 2

		trigger = { story_owner = { is_ai = no } }

		triggered_effect = {
			trigger = { story_owner = { has_trait_xp = { trait = ahroun value < 100 } } }
			effect = {
				story_owner = { save_scope_as = owner }
				clear_variable_list = ahroun_powerful_targets
				clear_variable_list = ahroun_targets

				#The list picks relevant and flavorful targets. players are always free to fight others
				every_ruler = {
					limit = {
						in_diplomatic_range = scope:owner
						POD_is_available_for_events = yes
						has_main_supernatural_race_trigger = yes
						trigger_if = {
							limit = { scope:owner = { religion = { is_in_family = rf_gaia } } }
							NOT = { has_trait = werewolf }
						}
						NOR = {
							has_trait = wraith
							has_trait = mummy
						}
					}
					if = {
						limit = {
							prowess >= scope:owner.POD_ahroun_target_prowess_min
							prowess <= scope:owner.POD_ahroun_target_prowess_upper
						}
						add_to_list = potential_targets
					}
					every_courtier_or_guest = {
						limit = {
							prowess >= scope:owner.POD_ahroun_target_prowess_min
							prowess <= scope:owner.POD_ahroun_target_prowess_upper
							POD_is_available_for_events = yes
							has_main_supernatural_race_trigger = yes
							trigger_if = {
								limit = { scope:owner = { religion = { is_in_family = rf_gaia } } }
								NOT = { has_trait = werewolf }
							}
							NOR = {
								has_trait = wraith
								has_trait = mummy
							}
						}
						add_to_list = potential_targets
					}
				}
				ordered_in_list = {
					list = potential_targets
					order_by = prowess
					max = 6
					check_range_bounds = no
					root = {
						add_to_variable_list = { name = ahroun_powerful_targets target = prev }
					}
				}
				ordered_in_list = {
					list = potential_targets
					limit = { NOT = { root = { is_target_in_variable_list = { name = ahroun_powerful_targets target = prev } } } }
					max = 6
					check_range_bounds = no
					root = {
						add_to_variable_list = { name = ahroun_targets target = prev }
					}
				}
			}
		}
	}


	on_owner_death = { end_story = yes }

}

									modifier = { add = scope:ragabash.POD_fera_rank_xp }

								}

									modifier = { add = scope:ragabash.POD_fera_rank_xp }

								}

												top_liege ?= scope:ragabash
												top_liege ?= scope:ragabash.top_liege
											}
										}
										weight = {
											base = 1
											modifier = {
												add = 70

											}

												add = 10

											}
											modifier = {
												add = 10
												POD_is_vampire_trigger = yes
											}

												add = 30

											}
										}
										save_scope_as = mark_owner
									}
								}

									random_landed_ruler = {
										limit = {
											in_diplomatic_range = scope:ragabash

		POD_hide_story_cycle_header = yes

			limit = { story_owner = { num_of_relation_spirit_companion > 0 } }

			}
		}
	}

		traits = { theurge }

		}
