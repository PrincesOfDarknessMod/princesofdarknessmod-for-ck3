vampire_tax_slot = {
	default_obligation = default_tax_collector

	obligations = {
		#Old system obligation that never worked : 
		#destruction rights
		#progeny rights
		#blood oath
		#masquerade protocol
		
		POD_strict_taxation_tax_collector
		POD_hard_control_tax_collector
		legacy_rights_protected_tax_collector
		domain_revocation_rights_protected_tax_collector
		guaranteed_council_tax_collector
		POD_tributary_tax_collector
		show_of_force_tax_collector
		blood_tithe_required_tax_collector
		rose_treaty_tax_collector
		roaming_court_tax_collector
		ravnos_war_machine_tax_collector
		free_state_tax_collector
		cult_tax_collector
		the_pyramid_tax_collector
		tzimisce_hospitality_tax_collector
		family_business_tax_collector
		eternal_senate_tax_collector
	}

	tax_slot_vassal_limit = {
		add = {
			value = 5
			desc = tax_slot_vassal_level_base_desc
		}
	}

	is_valid_tax_collector = {
		is_courtier_of = scope:liege
		is_landed = no
		is_imprisoned = no
		is_adult = yes
		NOT = { has_trait = incapable }
		NOT = { has_trait = torpor }
		NOT = { has_character_flag = POD_do_not_disturb_flag }
		NOT = { has_character_modifier = fp3_incapable_tax_collector_modifier }

		# Gender has to be allowed to be granted titles to be eligible
		#trigger_if = {
		#	limit = {
		#		scope:liege.faith = {
		#			has_doctrine = doctrine_gender_male_dominated
		#		}
		#	}
		#	is_male = yes
		#}
		#trigger_if = {
		#	limit = {
		#		scope:liege.faith = {
		#			has_doctrine = doctrine_gender_female_dominated
		#		}
		#	}
		#	is_female = yes
		#}
		trigger_if = {
			limit = {
				is_tax_collector = no
			}
			court_position_does_not_already_have_a_job_trigger = yes
		}
	}

	aptitude_level_breakpoints = { 20 45 75 110	}

	tax_collector_aptitude = tax_collector_aptitude_score

	on_tax_collector_hired = {
		scope:tax_collector = {
			add_character_flag = is_tax_collector
			if = {
				limit = {
					has_opinion_modifier = {
						target = scope:liege
						modifier = fired_from_tax_collector_position_opinion
					}
				}
				remove_opinion = {
					target = scope:liege
					modifier = fired_from_tax_collector_position_opinion
				}
			}
		}
	}

	on_tax_collector_fired = {
		scope:tax_collector = {
			remove_character_flag = is_tax_collector
			if = {
				limit = {
					NOT = { this = scope:liege }
				}
				add_opinion = {
					target = scope:liege
					modifier = fired_from_tax_collector_position_opinion
				}
			}
		}
	}
}
