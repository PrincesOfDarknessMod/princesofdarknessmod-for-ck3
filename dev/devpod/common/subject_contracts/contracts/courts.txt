@ai_standard_liege_desire = 2
@ai_standard_vassal_desire = 1

type_of_courts = {
	display_mode = radiobutton
	is_shown = {
		scope:subject.primary_title.tier >= tier_duchy
		scope:liege = {
			has_trait = kueijin
			has_government = august_court_government
		}
		scope:subject = {
			has_trait = kueijin
			has_government = august_court_government
		}						
	}
	obligation_levels = {
		type_of_courts_political = {

			# is_valid = {
				# scope:subject.primary_title.tier >= tier_duchy
			# }
			position = { 0 1 }
			default = yes
			
			subject_modifier = {
				levy_size = 0.2
				men_at_arms_maintenance = -0.1
				army_maintenance_mult = -0.1
			}
			
			ai_liege_desire =  {
				value = 0
				if = {
					limit = {
						scope:subject = {
							vassal_contract_obligation_level:feudal_government_taxes > feudal_tax_normal_level
							vassal_contract_obligation_level:feudal_government_levies > feudal_levies_normal_level
						}
					}
					add = @ai_standard_liege_desire
				}
			}
		}
		type_of_courts_social = {

			is_valid = {
				scope:subject.primary_title.tier >= tier_duchy
			}
			parent = type_of_courts_political
			position = { 1 0 }
			
			tax_factor = palatinate_contract_taxes
			levies_factor = palatinate_contract_levies
			
			liege_modifier = {
				monthly_prestige_gain_mult = 0.05
			}
			
			subject_modifier = {
				monthly_county_control_decline_factor = -0.2
				monthly_prestige_gain_mult = 0.2
				cultural_head_fascination_mult = 0.2
			}
			
			flag = has_social_courts_contract
			
			ai_liege_desire =  {
				value = 0
				if = {
					limit = {
						scope:subject = {
							vassal_contract_obligation_level:feudal_government_taxes < feudal_tax_low_level
							vassal_contract_obligation_level:feudal_government_levies < feudal_levies_low_level
						}
					}
					add = @ai_standard_liege_desire
				}
			}
		}
	}
}
