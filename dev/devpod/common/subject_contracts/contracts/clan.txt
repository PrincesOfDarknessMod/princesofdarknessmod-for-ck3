clan_government_obligations = {
	uses_opinion_of_liege = yes
	obligation_levels = {
		default = {
			levies = {
				scope:subject = {
					if = {
						limit = {
							is_ai = yes
						}
						value = clan_government_levies_default
						if = {
							limit = {
								scope:opinion_of_liege > 0
							}
							add =
							{
								value = scope:opinion_of_liege
								multiply = clan_government_levies_opinion_scale_positive
							}
						}
						else = {
							add =
							{
								value = scope:opinion_of_liege
								multiply = clan_government_levies_opinion_scale_negative
							}
						}
					}
					else = { # is a player
						value = clan_government_levies_player
					}
				}
			}
			min_levies = {
				scope:liege = {
					if = {
						limit = {
							has_realm_law = crown_authority_1
						}
						value = clan_government_levies_min_ca_1
					}
					else_if = {
						limit = {
							has_realm_law = crown_authority_2
						}
						value = clan_government_levies_min_ca_2
					}
					else_if = {
						limit = {
							has_realm_law = crown_authority_3
						}
						value = clan_government_levies_min_ca_3
					}
					else = {
						value = 0.0
					}
				}
			}
			tax = {
				scope:subject = {
					if = {
						limit = {
							is_ai = yes
						}
						value = clan_government_tax_default
						if = { 
							limit = {
								scope:opinion_of_liege > 0
							}
							add =
							{
								value = scope:opinion_of_liege
								multiply = clan_government_tax_opinion_scale_positive
							}
						}
						else = {
							add =
							{
								value = scope:opinion_of_liege
								multiply = clan_government_tax_opinion_scale_negative
							}
						}
					}
					else = { # is a player
						value = clan_government_tax_player
					}
				}
			}
			min_tax = {
				scope:liege = {
					if = {
						limit = {
							has_realm_law = crown_authority_1
						}
						value = clan_government_tax_min_ca_1
					}
					else_if = {
						limit = {
							has_realm_law = crown_authority_2
						}
						value = clan_government_tax_min_ca_2
					}
					else_if = {
						limit = {
							has_realm_law = crown_authority_3
						}
						value = clan_government_tax_min_ca_3
					}
					else = {
						value = 0.0
					}
				}
			}
			tax_contribution_postfix = "clan_government_tax_contribution_postfix"
			levies_contribution_postfix = "clan_government_levies_contribution_postfix"
			
			unclamped_contribution_label = "clan_government_unclamped_contribution_label"
			min_contribution_label = "clan_government_min_contribution_label"
		}
	}
}

clan_tax_collector_obligations = {
	obligation_levels = {
		default = {
			levies = {
				value = 0
				if = {
					limit = { exists = scope:tax_collector }
					add = clan_government_levies_default
					multiply = {
						value =	{
							add = scope:tax_collector.tax_collector_aptitude:clan_tax_slot
							save_temporary_value_as = tax_collector_aptitude_value
							if = {
								limit = {
									scope:tax_collector_aptitude_value = 4
								}
								add = 8.5
							}
							else_if = {
								limit = {
									scope:tax_collector_aptitude_value = 3
								}
								add = 5.75
							}
							else_if = {
								limit = {
									scope:tax_collector_aptitude_value = 2
								}
								add = 3
							}
							else_if = {
								limit = {
									scope:tax_collector_aptitude_value = 1
								}
								add = 1.5
							}
							else = {
								add = 1
							}
						}
					}
				}
			}

			tax = {
				value = 0
				if = {
					limit = { exists = scope:tax_collector }
					add = {
						value = clan_government_tax_default
						multiply = {
							value =	{

								save_temporary_value_as = tax_collector_aptitude_value
								if = {
									limit = {
										scope:tax_collector_aptitude_value = 4
									}
									add = 11
								}
								else_if = {
									limit = {
										scope:tax_collector_aptitude_value = 3
									}
									add = 7
								}
								else_if = {
									limit = {
										scope:tax_collector_aptitude_value = 2
									}
									add = 4
								}
								else_if = {
									limit = {
										scope:tax_collector_aptitude_value = 1
									}
									add = 2
								}
								else = {
									add = 1
								}
							}
						}
					}
				}
			}
		}
	}
}
