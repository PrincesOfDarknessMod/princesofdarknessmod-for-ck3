				LEVEL_3 = yes
			}
		}
		else_if = {
			limit = { years_from_game_start > 50 }
			add_prowess_skill = { -5 5 }

				TRAIT = lifestyle_blademaster

			}

				TRAIT = lifestyle_blademaster
				LEVEL_1 = yes

			}
		}
	}
