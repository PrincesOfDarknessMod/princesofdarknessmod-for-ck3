	after_creation = {
		save_scope_as = spawned_character

		#Random education boost
		random = {
			chance = {
				value = 50
				if = {
					limit = { scope:spawning_character = { has_character_flag = beastborn_no_interference } }
					add = 50
				}
			}
			rank_up_education_effect = yes
		}
		random = {
			chance = 50
			rank_up_education_effect = yes
		}

		#Stat boosts from events
		if = {
			limit = { scope:spawning_character = { has_character_flag = beastborn_more_diplomacy } }
			random_list = {
				1 = { add_diplomacy_skill = 3 }
				1 = { add_diplomacy_skill = 5 }
				1 = { add_diplomacy_skill = 8 }
			}
		}
		if = {
			limit = { scope:spawning_character = { has_character_flag = beastborn_more_martial } }
			random_list = {
				1 = { add_martial_skill = 3 }
				1 = { add_martial_skill = 5 }
				1 = { add_martial_skill = 8 }
			}
		}
		if = {
			limit = { scope:spawning_character = { has_character_flag = beastborn_more_stewardship } }
			random_list = {
				1 = { add_stewardship_skill = 3 }
				1 = { add_stewardship_skill = 5 }
				1 = { add_stewardship_skill = 8 }
			}
		}
		if = {
			limit = { scope:spawning_character = { has_character_flag = beastborn_more_intrigue } }
			random_list = {
				1 = { add_intrigue_skill = 3 }
				1 = { add_intrigue_skill = 5 }
				1 = { add_intrigue_skill = 8 }
			}
		}
		if = {
			limit = { scope:spawning_character = { has_character_flag = beastborn_more_learning } }
			random_list = {
				1 = { add_learning_skill = 3 }
				1 = { add_learning_skill = 5 }
				1 = { add_learning_skill = 8 }
			}
		}

		##Domicile bonuses
		#Mokole
		if = {
			limit = { scope:spawning_character.domicile ?= { has_domicile_building_or_higher = mokole_suchid_hatchery_01 } }
			if = {
				limit = { scope:spawning_character.domicile ?= { has_domicile_parameter = fera_beast_born_prowess_t3 } }
				add_prowess_skill = 6
			}
			else_if = {
				limit = { scope:spawning_character.domicile ?= { has_domicile_parameter = fera_beast_born_prowess_t2 } }
				add_prowess_skill = 4
			}
			else = { add_prowess_skill = 2 }

			if = {
				limit = { scope:spawning_character.domicile ?= { has_domicile_parameter = fera_beast_born_all_stats_t2 } }
				add_diplomacy_skill = 2
				add_martial_skill = 2
				add_stewardship_skill = 2
				add_intrigue_skill = 2
				add_learning_skill = 2
			}
			else_if = {
				limit = { scope:spawning_character.domicile ?= { has_domicile_parameter = fera_beast_born_all_stats_t1 } }
				add_diplomacy_skill = 1
				add_martial_skill = 1
				add_stewardship_skill = 1
				add_intrigue_skill = 1
				add_learning_skill = 1
			}
		}

		#Fera type
		if = {
			limit = { scope:spawning_character = { uses_cairo_bast_main_reward = yes } }
			add_trait = bastet
			set_character_faith = faith:bubasti
		}
		else = {
			scope:spawning_character = {
				random_character_trait = {
					limit = { has_trait_flag = fera_type }
					scope:spawned_character = { add_trait = prev }
				}
			}
		}

		#Mokole Stream
		if = {
			limit = { scope:spawning_character = { POD_is_mokole_trigger = yes } }
			scope:spawning_character.dynasty.var:mokole_stream = { save_scope_as = dynasty_stream }
			POD_assign_mokole_stream_from_dynasty_effect = { TARGET = scope:dynasty_stream }
		}

		POD_fera_initialization_effect = yes
	}
}
