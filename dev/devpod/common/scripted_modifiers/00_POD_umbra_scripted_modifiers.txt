wraith_ai_accept_modifier = {
	opinion_modifier = { # A lot more likely if Recipient likes Actor
		who = scope:recipient
		opinion_target = scope:actor
		multiplier = 0.75
		desc = AI_OPINION_REASON
	}
	
	modifier = {
		add = 25
		scope:actor = {
			has_trait = wraith
		}
		desc = ACTOR_WRAITH_REASON
	}
	
	modifier = {
		add = 50
		scope:recipient = {
			has_dread_level_towards = {
				target = scope:actor
				level = 1
			}
		}
		desc = INTIMIDATED_REASON
	}
	
	modifier = {
		add = 100
		scope:recipient = {
			has_dread_level_towards = {
				target = scope:actor
				level = 2
			}
		}
		desc = COWED_REASON
	}
	
	modifier = {
		scope:recipient.house = scope:actor.house
		add = 50
		desc = IS_SAME_HOUSE_REASON
	}
	
	modifier = { # Recipient wants to marry their lovers
		add = 25
		scope:recipient = {
			has_relation_lover = scope:actor
			NOT = { has_relation_soulmate = scope:actor }
		}
		desc = ARE_LOVERS_REASON
	}
	
	modifier = { # Recipient wants to marry their soulmates
		add = 100
		scope:recipient = {
			has_relation_soulmate = scope:actor
		}
		desc = ARE_SOULMATES_REASON
	}

	modifier = { 
		add = 10
		desc = WE_ARE_FRIENDS
		scope:recipient = {
			has_relation_friend = scope:actor
		}
	}
	
	modifier = { 
		add = 30
		desc = WE_ARE_BEST_FRIENDS
		scope:recipient = {
			has_relation_best_friend = scope:actor
		}
	}

	modifier = {
		add = -25
		desc = WE_ARE_RIVALS
		scope:recipient = {
			has_relation_rival = scope:actor
		}
	}
	
	modifier = {
		add = -100
		desc = WE_ARE_NEMESES
		scope:recipient = {
			has_relation_nemesis = scope:actor
		}
	}
}
