POD_caitiff_discipline_initialization_modifier = {
	modifier = {
		add = {
			value = 20
			subtract = $BASE_CHANCE$
		}
		scope:vampiresire ?= {
			has_trait = $TRAIT$
		}
	}
	modifier = {
		add = {
			value = 30
			subtract = $BASE_CHANCE$
		}
		scope:vampiresire ?= {
			has_trait = $TRAIT$advanced
		}
	}
	modifier = {
		factor = 0
		has_trait = $TRAIT$discipline
	}
}

POD_revenant_discipline_initialization_modifier = {
	modifier = {
		add = $CHANCE_PER_PARENT$
		mother = { has_trait = $TRAIT$discipline }
	}
	modifier = {
		add = $CHANCE_PER_PARENT$
		real_father = { has_trait = $TRAIT$discipline }
	}
	modifier = {
		factor = 2
		dynasty ?= { has_dynasty_perk = aspiration_legacy_5 }	
	}
}
