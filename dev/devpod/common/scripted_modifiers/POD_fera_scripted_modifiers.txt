POD_fera_spirit_quest_weight_easy_modifier = {
	modifier = {
		fera_spirit_quest_difficulty >= 1
		fera_spirit_quest_difficulty < 3
		add = 1
	}
	modifier = {
		fera_spirit_quest_difficulty = 3
		add = -0.25
	}
}

	}
}
