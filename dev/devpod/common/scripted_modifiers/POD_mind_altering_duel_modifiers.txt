POD_entrance_attacker_modifier = {
	modifier = {
		add = {
			value = scope:duel_value
			multiply = tiny_positive_duel_skill_multiplier
		}
	}
	modifier = {
		add = 2
		$ATTACKER$ = { has_perk = majesty_perk }
	}
	modifier = {
		add = 5
		$ATTACKER$ = { has_trait = presenceadvanced }
	}
	modifier = {
		add = 10
		$ATTACKER$ = { has_perk = ironclad_command_perk }
	}
	modifier = {
		add = {
			value = $ATTACKER$.diplomacy
			multiply = 0.5
		}
		$ATTACKER$ = { has_perk = as_below_so_above_perk }
	}
	#if player chracter add the +5 for willpower spent by attacker in duel chance as this will usually not be intialized before being called
	modifier = {
		add = 5
		$ATTACKER$ = { has_character_flag = willpower_spent }
	}
}
