# equivalent to increasing the victim's stats by 5/10/15
inquihunt_contract_tier_success_modifier = {
	modifier = {
		add = {
			value = 0
			if = {
				limit = { exists = scope:task_contract }
				
				if = {
					limit = { scope:task_contract.task_contract_tier = tier_duchy }
					add = -2.5
				}
				else_if = {
					limit = { scope:task_contract.task_contract_tier = tier_kingdom }
					add = -5
				}
				else_if = {
					limit = { scope:task_contract.task_contract_tier >= tier_empire }
					add = -7.5
				}
			}
		}
	}
}
