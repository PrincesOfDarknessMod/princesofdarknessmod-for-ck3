doctrine_nomarriage = {
	is_shown = {
		OR = {
			religion = { is_in_family = rf_kindred }	
			religion = { is_in_family = rf_gaia }
			religion = { is_in_family = rf_wyrm }		
		}
	}
	piety_cost = {
		value = faith_doctrine_cost_low
		if = {
			limit = { has_doctrine = doctrine_nomarriage }
			multiply = 0.75
		}
	}
	parameters = {
		number_of_spouses = 0
		number_of_consorts = 0
	}
}
