				rite = root.faction_target.rite
			}
			desc = "FACTION_DISCONTENT_BURGHER_EXCOMMUNICATED"
		}

		modifier = {
			desc = "FACTION_REASON_GAME_RULE_REALM_STABILITY_LESSER"
			add = 0.75
			has_game_rule = lesser_realm_stability
		}
		modifier = {
			desc = "FACTION_REASON_GAME_RULE_REALM_STABILITY_HIGHER"

		}
		modifier = {
			desc = "FACTION_REASON_GAME_RULE_REALM_STABILITY_EXTREME"
			add = -1.5

		}
		modifier = {
			desc = "FACTION_REASON_DIFFICULTY_EASY"
			add = -0.5
			has_game_rule = easy_difficulty
			faction_target = { is_ai = no }
		}
		modifier = {
			desc = "FACTION_REASON_DIFFICULTY_HARD"
			add = 0.5
			has_game_rule = hard_difficulty
			faction_target = { is_ai = no }
		}
	}

	is_valid = {
		always = no
		trigger_if = {
			limit = { exists = special_character }
			special_character = { is_alive = yes }
		}

		trigger_if = {
			limit = { faction_is_at_war = no }
			faction_target = { burgher_liege_valid_trigger = yes }
		}
	}

	is_county_valid = {
		always = no
		holder = scope:faction.faction_target
	}

	is_character_valid = {
		has_valid_faction_members_trigger = yes
	}

	can_character_join = {
		joined_faction = scope:faction
	}

	can_county_join = {
		trigger_if = {
			limit = { scope:faction = { faction_is_at_war = no } }
			burgher_county_can_join_trigger = yes
		}
		holder = scope:faction.faction_target
		NOT = { has_variable = won_self_rule }
		burgher_county_faction_on_cooldown_trigger = no
		burgher_county_founder_faith_compatible_trigger = yes
	}

	can_county_create = {
		burgher_county_is_urban_trigger = yes
		scope:target = {
			burgher_liege_valid_trigger = yes
		}
		holder = scope:target
		NOT = { has_variable = won_self_rule }
		burgher_county_faction_on_cooldown_trigger = no
		custom_description = {
			text = character_is_immune_to_factions
			subject = scope:target
			NOT = { scope:target = { immune_to_factions_trigger = yes } }
		}
	}

	can_character_become_leader = {

	}

		}

		compare_modifier = {

				burgher_liege_can_be_crowned_trigger = yes

			}
		}

		modifier = {
			add = 25
			scope:target = { has_trait = excommunicated }

		}

		modifier = {
			add = 15

			add = 15

		modifier = {
			add = -50
			has_game_rule = easy_difficulty
			scope:target = { is_ai = no }
		}
		modifier = {
			add = -150
			has_game_rule = very_easy_difficulty
			scope:target = { is_ai = no }
		}
		modifier = {
			add = 25
			has_game_rule = lesser_realm_stability

			add = -50
			has_game_rule = higher_realm_stability

			trigger = { prosperity_level > 0 }
			value = prosperity_level
			multiplier = 1.0
		}

		compare_modifier = {
			desc = "FACTION_REASON_POPULAR_OPINION"
			trigger = { NOT = { popular_opinion = 0 } }
			value = popular_opinion
			multiplier = -0.5
		}

		compare_modifier = {
			desc = "FACTION_REASON_HIGH_COUNTY_CONTROL"
			trigger = { faction_county_high_control_suppression > 0 }
			value = faction_county_high_control_suppression
			multiplier = -1.0
		}

		compare_modifier = {
			desc = "FACTION_REASON_STATIONED_GARRISON"
			trigger = { faction_county_garrison_suppression > 0 }
			value = faction_county_garrison_suppression
			multiplier = -2.0
		}

		modifier = {
			desc = "FACTION_REASON_CIVIC_AGITATION"
			add = 25
			has_county_modifier = gdp_civic_agitation_county_modifier
		}

		modifier = {
			desc = "FACTION_REASON_BROKEN_CHARTER"
			add = 25
			has_county_modifier = burgher_broken_charter_county_modifier
		}

		modifier = {
			desc = "FACTION_REASON_SHARED_CULTURE_FAITH"
			add = 40
			scope:faction = { exists = var:burgher_county }
			title_province.culture = scope:faction.var:burgher_county.title_province.culture
			title_province.faith = scope:faction.var:burgher_county.title_province.faith
		}

		modifier = {
			desc = "FACTION_REASON_COUNTY_CORRUPTION"

		}

			desc = "FACTION_REASON_DEBASED_COINAGE"

		}

		modifier = {
			desc = "FACTION_REASON_AT_WAR_WITH_TARGET"
			add = 1000

			}
		}

	county_power = {
		value = burgher_county_militia_value
	}

	demand = {
		save_scope_as = faction

		burgher_resolve_county_effect = { FACTION = this }

		faction_target = {
			trigger_event = {
				on_action = burgher_autonomy_demand
			}
		}
	}

	ai_demand_chance = {
		base = 0

		modifier = {
			add = 100
			any_faction_county_member = {}
		}

		modifier = {
			add = -1000
			faction_target = {
				involved_activity ?= {
					has_activity_type = activity_coronation
					activity_host = prev
				}
			}
		}
	}

	on_creation = {
		ordered_faction_county_member = {
			order_by = title_province.province_size_level
			save_scope_as = founding_county
		}

		set_variable = {
			name = burgher_county
			value = scope:founding_county
		}
		set_variable = {
			name = burgher_grievance
			value = 0
		}
	}

	on_destroy = {
		set_variable = {
			name = burgher_destroying
			value = yes
		}

		if = {
			limit = {
				NOT = { has_variable = burgher_outcome_reported }
			}
			burgher_notify_sponsor_effect = {
				FACTION = this
				TOAST_TYPE = event_toast_effect_neutral
				TITLE = burgher_sponsor_dispersed_toast
				EFFECT = "custom_tooltip = burgher_sponsor_dispersed_toast_desc"
			}
		}

		burgher_faction_clear_sponsor_effect = yes

		if = {
			limit = { exists = special_character }
			special_character = { burgher_dissolve_leader_effect = yes }
		}
	}

	on_war_start = {
		every_faction_county_member = {
			hidden_effect = {
				save_scope_as = mustering_county
				title_province = { save_scope_as = mustering_location }
			}

			change_county_control = burgher_war_starts_county_control_loss

			root.faction_leader = {
				spawn_army = {
					name = burgher_militia_army_name
					levies = scope:mustering_county.burgher_county_militia_value
					location = scope:mustering_location
					war = root.faction_war
					inheritable = no
					war_keep_on_attacker_victory = yes
				}
			}
		}

		root.faction_leader = {
			add_gold = burgher_faction_war_chest_value

			spawn_army = {
				name = burgher_militia_army_name
				men_at_arms = {
					type = pikemen_militia
					stacks = burgher_leader_pikemen_stacks_value
				}
				men_at_arms = {
					type = bowmen
					stacks = burgher_leader_bowmen_stacks_value
				}
				war = root.faction_war
				location = root.faction_leader.location
				origin = root.faction_leader.location
				inheritable = no
				war_keep_on_attacker_victory = yes
			}
		}

		burgher_sponsor_war_support_effect = { FACTION = root }
	}

	leader_leaves = {
		if = {
			limit = {
				NOT = { has_variable = burgher_destroying }
				exists = faction_war
			}
			faction_war = {
				end_war = defender
			}
		}
	}

	special_character_title = "FACTIONS_WINDOW_LEADER"

	inherit_membership = no

	county_can_switch_to_other_faction = yes
}
