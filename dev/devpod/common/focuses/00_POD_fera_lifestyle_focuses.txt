
breedgifts_focus = {
	lifestyle = breedgifts_lifestyle

	is_valid = {
		OR = {
			POD_has_access_to_fera_gifts = { TRAIT = trait:werewolf }
			POD_has_access_to_fera_gifts = { TRAIT = trait:bastet }
		}
	}
	
	modifier = {
		stress_gain_mult = -0.05
	}

	desc = {
		desc = breedgifts_focus_desc
		desc = line_break
	}

	auto_selection_weight = {
		value = 1500
		
		if = {
			limit = {
				has_trait = homid_born
				OR = {
					AND = {
						has_perk = assimilation_perk
						has_perk = beyond_human_perk
					}
					AND = {
						has_perk = deny_the_hungry_perk
						has_perk = madness_perk
					}
				}
			}
			multiply = 0
		}

		if = {
			limit = {
				has_trait = beast_born
				OR = {
					AND = {
						has_perk = boon_fathers_perk
						has_perk = elemental_gift_perk
					}
					AND = {
						has_perk = judgement_of_pestilence_perk
						has_perk = revolt_of_the_land_perk
					}
				}
			}
			multiply = 0
		}
	}
}
