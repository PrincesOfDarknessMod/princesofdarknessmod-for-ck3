action_can_weaken_or_bind_wraith_alert = {
	type = alert
	icon = action_can_add_herd_alert

	check_create_action = {
		if = {
			limit = {
				any_relation = {
					type = bound_wraith
					save_temporary_scope_as = bound_wraith
					ROOT = {
						OR = {
							is_character_interaction_valid = {
								recipient = scope:bound_wraith
								interaction = POD_weaken_fetter_interaction
							}
							is_character_interaction_valid = {
								recipient = scope:bound_wraith
								interaction = POD_bind_interaction
							}
						}
					}
				}
			}
			try_create_important_action = {
				important_action_type = action_can_weaken_or_bind_wraith_alert
				actor = root
				recipient = scope:bound_wraith
			}
		}
	}

	effect = {
		if = {
			limit = {
				is_character_interaction_valid = {
					recipient = scope:recipient
					interaction = POD_weaken_fetter_interaction
				}
			}
			open_interaction_window = {
				interaction = POD_weaken_fetter_interaction
				actor = scope:actor
				recipient = scope:recipient
			}
		}
		else = {
			open_interaction_window = {
				interaction = POD_bind_interaction
				actor = scope:actor
				recipient = scope:recipient
			}

		}
	}
}
