action_can_add_herd_alert = {
    type = alert
	icon = action_can_add_herd_alert

	check_create_action = {
		if = {
			limit = {
				root = {
					has_trait = vampire
					num_of_relation_herd_member < max_herd_cap
					NOT = { any_scheme = { scheme_type = herdify_mortal } }
					any_courtier_or_guest = {
						POD_can_be_herd_trigger = yes
						NOT = { has_relation_herd_owner = root }
						is_adult = yes
						trigger_if = {
							limit = { root = { is_ventrue_trigger = yes } }
							ventrue_weakness_compatible_trigger = {
								ACTOR = root
								RECIPIENT = this
							}
						}
					}
				}
			}
			ordered_courtier_or_guest = {
				limit = { 
					POD_can_be_herd_trigger = yes 
					NOT = { has_relation_herd_owner = root }
					is_adult = yes
					trigger_if = {
						limit = { root = { is_ventrue_trigger = yes } }
						ventrue_weakness_compatible_trigger = {
							ACTOR = root
							RECIPIENT = this
						}
					}
				}
				order_by = resonance_rating_value
				position = 0
	    		save_temporary_scope_as = potential_herd
			}
			try_create_important_action = {
				important_action_type = action_can_add_herd_alert
				actor = root
				recipient = scope:potential_herd
			}
		}
	}

	effect = {
		open_interaction_window = {
			interaction = POD_herdify_mortal_interaction
			actor = scope:actor
			recipient = scope:recipient
		}
	}
}
