action_can_set_fetter_alert = {
    type = alert
	icon = action_dying

	check_create_action = {
		if = {
			limit = {
				root = {
					has_trait = wraith
					unset_fetter_point > 0
				}
			}
			try_create_important_action = {
				important_action_type = action_can_set_fetter_alert
				actor = root
				recipient = root
			}
		}
	}

	effect = {
		open_interaction_window = {
			interaction = POD_set_fetter_title_interaction
			actor = scope:actor
			recipient = scope:recipient
		}
	}
}

				important_action_type = action_can_fallen_possess_alert

	}

			recipient = scope:recipient
		}
	}
}

action_host_not_married = {
	type = alert
	icon = action_adult_ruler_not_married

	check_create_action = {
		if = {
			limit = {
				NOT = { has_pending_interaction_of_type = marry_off_interaction }
				NOT = { has_pending_interaction_of_type = arrange_marriage_interaction }
				has_trait = demon
				has_variable = host
				AND = {
					var:host = {
						can_marry_trigger = yes
						is_concubine = no
						is_married = no
						is_betrothed = no
					}
					is_character_interaction_valid = {
						recipient = var:host
						interaction = arrange_marriage_interaction
					}
				}
			}
			try_create_important_action = {
				important_action_type = action_host_not_married
				actor = root
				recipient = var:host
			}
		}
	}

	effect = {
		open_interaction_window = {
			interaction = arrange_marriage_interaction
			actor = scope:actor
			recipient = scope:recipient
		}
	}
}

	#priority = 600
	combine_into_one = yes
	check_create_action = {
		if = {
			limit = {
				gold >= 0
			}
			every_in_global_list = {
				# Mummies get put in a global variable list to save performance

		}
	}

	effect = {
