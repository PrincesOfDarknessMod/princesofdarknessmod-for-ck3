action_can_set_fetter_alert = {
    type = alert
	icon = action_dying

	check_create_action = {
		if = {
			limit = {
				root = {
					has_trait = wraith
					has_variable = fetter_count
					var:fetter_count > 0
				}
			}
			try_create_important_action = {
				important_action_type = action_can_set_fetter_alert
				actor = root
				recipient = root
			}
		}
	}

	effect = {
		open_interaction_window = {
			interaction = POD_set_fetter_title_interaction
			actor = scope:actor
			recipient = scope:recipient
		}
	}
}

action_can_demonic_possess_alert = {
    type = alert
	icon = action_dying

	check_create_action = {
		if = {
			limit = {
				NOT = {
					is_target_in_global_variable_list = {
						name = unavailable_unique_decisions
						target = flag:breakdown_the_abyss_objective_decision
					}
				}
				root = {
					has_trait = demon
					NOR = {
						has_trait = truefallen
						has_trait = earthbound
					}
				}
			}
			try_create_important_action = {
				important_action_type = action_can_demonic_possess_alert
				actor = root
				recipient = root
			}
		}
	}

	effect = {
		open_interaction_window = {
			interaction = POD_demonic_possess_title_interaction
			actor = scope:actor
			recipient = scope:recipient
		}
	}
}

action_can_fallen_possess_alert = {
    type = alert
	icon = action_dying

	check_create_action = {
		if = {
			limit = {
				NOT = {
					is_target_in_global_variable_list = {
						name = unavailable_unique_decisions
						target = flag:breakdown_the_abyss_objective_decision
					}
				}
				root = {
					has_trait = demon
					has_trait = truefallen
					num_of_relation_possessed_target = 0
					any_relation = {
						type = demonic_thrall
						can_be_demonic_possesed = yes
					}
				}
			}
			random_relation = {
				type = demonic_thrall
				limit = { can_be_demonic_possesed = yes }
				save_temporary_scope_as = thrall
			}
			try_create_important_action = {
				important_action_type = action_can_fallen_possess_alert
				actor = root
				recipient = scope:thrall
			}
		}
	}

	effect = {
		open_interaction_window = {
			interaction = POD_demonic_possess_character_interaction
			actor = scope:actor
			recipient = scope:recipient
		}
	}
}
