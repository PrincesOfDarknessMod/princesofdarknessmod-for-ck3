action_can_set_fetter_alert = {
    type = alert
	icon = action_dying

	check_create_action = {
		if = {
			limit = {
				root = {
					has_trait = wraith
					has_variable = unset_fetter_point
					var:unset_fetter_point > 0
				}
			}
			try_create_important_action = {
				important_action_type = action_can_set_fetter_alert
				actor = root
				recipient = root
			}
		}
	}

	effect = {
		open_interaction_window = {
			interaction = POD_set_fetter_title_interaction
			actor = scope:actor
			recipient = scope:recipient
		}
	}
}

action_can_fallen_possess_alert = {
    type = alert
	icon = action_dying

	check_create_action = {
		if = {
			limit = {
				NOT = {
					is_target_in_global_variable_list = {
						name = unavailable_unique_decisions
						target = flag:breakdown_the_abyss_objective_decision
					}
				}
				root = {
					has_trait = demon
					has_trait = truefallen
					num_of_relation_possessed_target = 0
					any_relation = {
						type = demonic_thrall
						can_be_demonic_possesed = yes
					}
				}
			}
			random_relation = {
				type = demonic_thrall
				limit = { can_be_demonic_possesed = yes }

			}
			try_create_important_action = {
				important_action_type = action_can_fallen_possess_alert
				actor = root
				recipient = scope:thrall
			}
		}
	}

	effect = {
		open_interaction_window = {
			interaction = POD_demonic_possess_character_interaction
			actor = scope:actor
			recipient = scope:recipient
		}
	}
}

action_can_add_herd_alert = {
    type = alert
	icon = action_can_add_herd_alert

	check_create_action = {
		if = {
			limit = {
				root = {
					has_trait = vampire
					num_of_relation_herd_member < max_herd_cap
					NOT = { any_scheme = { scheme_type = herdify_mortal } }
					any_courtier_or_guest = {
						POD_can_be_herd_trigger = yes
						NOT = { has_relation_herd_owner = root }
						is_adult = yes
					}
				}
			}
			ordered_courtier_or_guest = {
				limit = { 
					POD_can_be_herd_trigger = yes 
					NOT = { has_relation_herd_owner = root }
					is_adult = yes
				}
				order_by = resonance_rating_value
				position = 0
	    		save_temporary_scope_as = potential_herd
			}
			try_create_important_action = {
				important_action_type = action_can_add_herd_alert
				actor = root
				recipient = scope:potential_herd
			}
		}
	}

	effect = {
		open_interaction_window = {
			interaction = POD_herdify_mortal_interaction
			actor = scope:actor
			recipient = scope:recipient
		}
	}
}
