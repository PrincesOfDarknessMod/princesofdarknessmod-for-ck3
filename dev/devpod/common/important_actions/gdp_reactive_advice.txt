action_reactive_advice_construction_queue = {
	type = tutorial
	icon = generic_reactive_advice
	soundeffect = "event:/SFX/UI/Notifications/TopBar/sfx_ui_notifications_topbar_reactive_advice"
	check_create_action = {
		if = {
			limit = {
				should_show_tutorial_alert = { LESSON = reactive_advice_construction_queue }
				has_queued_construction = yes
			}
			try_create_important_action = {
				important_action_type = action_reactive_advice_construction_queue
				actor = root
			}
		}
	}
	effect = {
		start_tutorial_lesson = reactive_advice_construction_queue
	}
}


action_reactive_advice_merchant_introduction = {
	type = tutorial
	icon = generic_reactive_advice
	soundeffect = "event:/SFX/UI/Notifications/TopBar/sfx_ui_notifications_topbar_reactive_advice"
	check_create_action = {
		if = {
			limit = {
				government_has_flag = government_is_merchant

				should_show_tutorial_alert = { LESSON = reactive_advice_merchant_introduction }
			}
			try_create_important_action = {
				important_action_type = action_reactive_advice_merchant_introduction
				actor = root
			}
		}
	}
	effect = {
		start_tutorial_lesson = reactive_advice_merchant_introduction
	}
}


action_reactive_advice_company_direction = {
	type = tutorial
	icon = generic_reactive_advice
	soundeffect = "event:/SFX/UI/Notifications/TopBar/sfx_ui_notifications_topbar_reactive_advice"
	check_create_action = {
		if = {
			limit = {
				government_has_flag = government_is_merchant
				house ?= { has_house_unity = yes }
				should_show_tutorial_alert = { LESSON = reactive_advice_company_direction }
				OR = {
					is_widgetid_open = dynasty_house_view
					months_from_game_start >= 32 # Reactive Advice Queue, Position 8
				}
			}
			try_create_important_action = {
				important_action_type = action_reactive_advice_company_direction
				actor = root
			}
		}
	}
	effect = {
		start_tutorial_lesson = reactive_advice_company_direction
	}
}
