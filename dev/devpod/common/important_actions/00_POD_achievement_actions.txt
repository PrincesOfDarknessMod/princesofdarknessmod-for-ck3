action_initiate_achievement_alert = {
    type = alert
	icon = action_inspiration

	check_create_action = {
		if = {
			limit = {
				OR = {
					AND = {
						is_tutorial_lesson_completed = reactive_advice_achievement_01
						NOT = { has_global_variable = pod_achievement_01 }
					}
					AND = {
						is_tutorial_lesson_completed = reactive_advice_achievement_02
						NOT = { has_global_variable = pod_achievement_02 }
					}
				}
			}
			try_create_important_action = {
				important_action_type = action_initiate_achievement_alert
				actor = root
				recipient = root
			}
		}
	}

	effect = {
		if = {
			limit = { 
				is_tutorial_lesson_completed = reactive_advice_achievement_01
				NOT = { has_global_variable = pod_achievement_01 }
			}
			open_interaction_window = {
				interaction = POD_achievement_01_interaction
				actor = scope:actor
				recipient = scope:recipient
			}
		}
		if = {
			limit = { 
				is_tutorial_lesson_completed = reactive_advice_achievement_02
				NOT = { has_global_variable = pod_achievement_02 }
			}
			open_interaction_window = {
				interaction = POD_achievement_02_interaction
				actor = scope:actor
				recipient = scope:recipient
			}
		}
	}
}
