POD_soul_bridge_interaction = {
	category = interaction_category_vampire
	desc = POD_soul_bridge_interaction_desc
	icon = POD_art_interaction
	
	cooldown = { months = 3 }	

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { 
			OR = {
				has_trait = kueijin 
				has_trait = dhampyr
			}
		}
		scope:actor = { has_perk = soul_bridge_perk }
		scope:recipient = { POD_is_clone_trigger = no }
		POD_is_in_same_plane_trigger = yes
	}

	is_valid_showing_failures_only = {
		scope:recipient = { 
			NOT = { has_trait = torpor }
			NOT = { has_trait = effigy }
		}
		custom_description = {
            object = scope:actor
            text = "soul_demon_incompatible"
            scope:actor = { using_demon_arts_trigger = no }
        }
	}
	
	send_option = {
		is_shown = {
			NOR = {
				scope:actor = { 
					has_hook_of_type = {
						target = scope:recipient
						type = strong_soul_shackles_hook
					}
				}
				scope:recipient = { is_imprisoned_by = scope:actor }
				scope:recipient = { has_relation_necromancer = scope:actor }
			}
		}
		is_valid = {
			scope:actor = {
				can_start_scheme = {
					type = PODsoul_bridge
					target = scope:recipient
				}
			}
		}
		flag = scheme_attack
		localization = POD_interaction_scheme_attack
	}
	
	send_option = {
		is_shown = {
			NOR = {
				scope:actor = { 
					has_hook_of_type = {
						target = scope:recipient
						type = strong_soul_shackles_hook
					}
				}
				scope:recipient = { is_imprisoned_by = scope:actor }
				scope:recipient = { has_relation_necromancer = scope:actor }
			}
		}
		is_valid = {
			custom_description = {
				object = scope:actor
				text = "I_must_be_at_the_same_location"
				scope:actor = { exists = location }
				scope:recipient = { exists = location }
				scope:actor = { location = scope:recipient.location }
			}
		}
		flag = duel_attack
		localization = POD_interaction_duel_attack
	}
	
	send_option = {
		is_valid = {
			OR = {
				scope:actor = { 
					has_hook_of_type = {
						target = scope:recipient
						type = strong_soul_shackles_hook
					}
				}
				scope:recipient = { is_imprisoned_by = scope:actor }
				scope:recipient = { has_relation_necromancer = scope:actor }
			}
		}
		flag = automatic
		localization = POD_interaction_automatic
	}

	on_accept = {
		scope:actor = { POD_wraith_willpower_check_effect = yes }
		scope:actor = { POD_chi_decrease_effect = { COUNT = 1 } }	
		if = {
			limit = {
				scope:scheme_attack = yes
			}
			scope:actor = {
				start_scheme = {
					type = PODsoul_bridge
					target = scope:recipient
				}
			}
		}
	
		if = {
			limit = {
				scope:duel_attack = yes
			}
			scope:actor = {
				configure_start_single_combat_effect = {
					SC_INITIATOR = scope:actor
					SC_ATTACKER = scope:actor
					SC_DEFENDER = scope:recipient
					FATALITY = no
					FIXED = no
					LOCALE = throne_room
					OUTPUT_EVENT = kueijin_scheme.200
					INVALIDATION_EVENT = tribal.1022
				}
			}
		}
		
		if = {
			limit = {
				scope:automatic = yes
			}
			scope:actor = { trigger_event = kueijin_scheme.200 }
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
