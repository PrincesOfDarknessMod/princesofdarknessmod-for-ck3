POD_set_thrall_interaction = {
	category = interaction_category_demon
	common_interaction = yes
	interface_priority = 100
	desc = POD_set_thrall_interaction_desc
	icon = POD_arcana_interaction

	ai_min_reply_days = 4
	ai_max_reply_days = 9
	
	cost = { prestige = minor_prestige_value }
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			is_playable_character = yes
			has_perk = enthrall_perk
		}
		scope:recipient = { 
			is_a_thrall = no
		}
	}

	is_valid = {
		scope:recipient = {
			faith = scope:actor.faith
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			NOT =  { is_at_war_with = scope:recipient }
		}
		custom_tooltip = {
			text = POD_enthrall_interaction_limit
			scope:actor = {
				num_of_relation_demonic_thrall < max_thrall_cap
			}
		}
		scope:recipient = {
			can_be_thrall = yes
			is_busy_in_events_localised = yes
			NOT = { has_strong_hook = scope:actor }
			NOT = { has_trait = automaton }
			POD_is_not_clone_trigger = yes
		}
	}

	#Use hook
	send_option = {
		is_valid = {
			scope:actor = {
				has_usable_hook = scope:recipient
			}
		}
		flag = hook
		localization = SCHEME_HOOK
	}
	should_use_extra_icon = {
		scope:actor = { has_usable_hook = scope:recipient }
	}
	extra_icon = "gfx/interface/icons/character_interactions/hook_icon.dds"

	send_options_exclusive = no

	on_accept = {
		scope:actor = {
			POD_resolve_cost_effect = { COUNT = 1 }
		}
		scope:recipient = {
			if = {
				limit = { NOT = { has_relation_worshipped_demon = scope:actor } }
				set_relation_worshipped_demon = scope:actor
			}
			set_relation_demonic_master = scope:actor
			give_infernalist_secret_or_trait_effect = yes
			give_cultist_secret_or_trait_effect = yes
			if = {
				limit = { has_trait = supehunter }
				scope:actor = { save_scope_as = vampiresire }
				POD_hunter_enthrall_effect = yes
			}
			
			show_as_tooltip = {
				POD_character_resolve_initiation_effect = yes
			}			
		}
	}
	
	ai_targets = {
		ai_recipients = vassals
	}
	
	ai_frequency = 60
	
	ai_potential = {
		primary_title.tier > tier_county
		prestige > minor_prestige_value
	}
	
	ai_accept = {
		base = 0 # Try to make it 0 for most interactions

		modifier = {
			scope:hook = yes
			add = 50
			desc = SCHEME_WEAK_HOOK_USED
		}
		
		opinion_modifier = { # Opinion Factor
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.5
			desc = AI_SIMPLE_OPINION_REASON
		}
		
		modifier = {
			add = -50
			NOT = { scope:actor.faith = scope:recipient.faith }
			desc = "ZEAL_AGAINST_INFIDELS"	
		}

		modifier = {
			add = 25
			scope:actor.faith = scope:recipient.faith 
			desc = "SAME_FAITH"
		}

		modifier = {
			add = -50
			has_relation_rival = scope:recipient
		}
		modifier = {
			add = -80
			has_relation_nemesis = scope:recipient
		}
		
		modifier = { # Safegurad for failed infernalist baron initiation
			add = 1000
			scope:actor = { is_ai = yes }
		}
	}
	
	ai_will_do = {
		base = 100
		
		modifier = { # The AI will only use a Hook if they couldn't otherwise do this
			scope:hook = yes
			add = -1
		}
	}
}

POD_revelation_interaction = {
	category = interaction_category_demon
	common_interaction = yes
	interface_priority = 120
	desc = POD_revelation_interaction_desc
	icon = POD_arcana_interaction
	
	cooldown = { months = 1 }
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_trait = demon }
		scope:actor = { has_perk = arcana_revelation_perk }
	}

	is_valid = {
		scope:actor = { 
			OR = {
				has_character_modifier = POD_shapeshift_form_apocalyptic_angelic_modifier
				has_character_modifier = POD_shapeshift_form_apocalyptic_damned_modifier
				has_POD_visage_form_trigger = yes
			}
		}
		scope:recipient = { 
			OR = {
				POD_can_be_vamphunted_scheme_trigger = yes
				has_relation_worshipped_demon = scope:actor
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = { 
			piety >= 0
		}
		scope:actor = { dread >= scope:recipient.ai_boldness }
	}

	on_accept = {
		scope:recipient = {
			while = {
				count = 5
				random = {
					chance = 5
					scope:actor = { POD_resolve_income_effect = { COUNT = 1 } }
					add_stress = minor_stress_impact_gain 
					add_piety = minor_piety_loss
					modifier = { 
						add = {
							value = scope:actor.dread
							multiply = 1
						}
						NOT = { scope:actor.dread = 0 }
					}
					modifier = {
						add = {
							value = ai_boldness
							multiply = -1
						}
						NOT = { ai_boldness = 0 }
					}
				}
			}
			if = { 
				limit = { scope:actor = { has_character_modifier = POD_shapeshift_form_apocalyptic_damned_modifier } }
				add_opinion = {
					target = scope:actor
					modifier = hurt_opinion
					opinion = -25
				}
			}
			else = {
				add_opinion = {
					target = scope:actor
					modifier = pleased_opinion
					opinion = 15
				}
			}
			if = { 
				limit = { has_trait = mortal }
				add_trait = POD_knows_too_much 
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

					has_trait = earthbound

			scope:recipient = { can_be_demonic_possesed = yes }
		}
		scope:recipient = {
			trigger_if = {
				limit = { has_trait = automaton }
				has_relation_unspeakable_servitor_master = scope:actor
			}
			NOT = { has_character_modifier = excluded_from_possession_modifier }
		}
	}
	
	on_accept = {
		scope:actor = {
			POD_free_from_abyss_effect = yes

		}
		scope:recipient = {
			if = {
				limit = { has_trait = automaton }
				custom_tooltip = POD_demonic_possess_servitor.tt
			}
			else = {
				custom_tooltip = POD_demonic_possess_host_leech.tt
				custom_tooltip = POD_demonic_possess_maintenance.tt
			}
		}
	}
	
	auto_accept = yes
	
	ai_frequency = 120

	ai_potential = {
		# The basic requirements
		is_imprisoned = no 
		is_landed = yes
		primary_title.tier > tier_barony
	}
	ai_targets = {
		ai_recipients = courtiers
		ai_recipients = peer_vassals
		max = 10
	}
	
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 100
			num_of_relation_possessed_target = 0
		}
	}
}
