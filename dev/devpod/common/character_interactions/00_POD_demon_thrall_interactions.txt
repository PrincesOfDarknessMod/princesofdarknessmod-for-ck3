POD_set_thrall_interaction = {
	category = interaction_category_demon
	common_interaction = yes
	interface_priority = 100
	desc = POD_set_thrall_interaction_desc
	icon = POD_arcana_interaction

	ai_min_reply_days = 4
	ai_max_reply_days = 9
	
	cost = { prestige = minor_prestige_value }
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			is_landed = yes
			has_perk = enthrall_perk
		}
		scope:recipient = { 
			NOT = { has_relation_demonic_master = scope:actor }
		}
	}

	is_valid = {
		scope:recipient = {
			faith = scope:actor.faith
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			NOT =  { is_at_war_with = scope:recipient }
		}
		scope:recipient = {
			can_be_thrall = yes
			is_busy_in_events_localised = yes
			NOT = { has_strong_hook = scope:actor }
			NOT = { has_trait = statue }
			POD_is_not_clone_trigger = yes
		}
	}

	#Use hook
	send_option = {
		is_valid = {
			scope:actor = {
				has_usable_hook = scope:recipient
			}
		}
		flag = hook
		localization = SCHEME_HOOK
	}
	should_use_extra_icon = {
		scope:actor = { has_usable_hook = scope:recipient }
	}
	extra_icon = "gfx/interface/icons/character_interactions/hook_icon.dds"

	send_options_exclusive = no

	on_accept = {
		scope:actor = {
			POD_resolve_decrease_effect = yes
		}
		scope:recipient = {
			add_trait = infernalist
			set_relation_demonic_master = scope:actor
			if = {
				limit = { has_trait = supehunter }
				scope:actor = { save_scope_as = vampiresire }
				POD_hunter_enthrall_effect = yes
			}
			
			show_as_tooltip = {
				POD_character_resolve_initiation_effect = yes
			}			
		}
	}
	
	ai_targets = {
		ai_recipients = vassals
	}
	
	ai_frequency = 60
	
	ai_potential = {
		primary_title.tier > tier_county
		prestige > minor_prestige_value
	}
	
	ai_accept = {
		base = 0 # Try to make it 0 for most interactions

		modifier = {
			scope:hook = yes
			add = 50
			desc = SCHEME_WEAK_HOOK_USED
		}
		
		opinion_modifier = { # Opinion Factor
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.5
			desc = AI_SIMPLE_OPINION_REASON
		}
		
		modifier = {
			add = -50
			NOT = { scope:actor.faith = scope:recipient.faith }
			desc = "ZEAL_AGAINST_INFIDELS"	
		}

		modifier = {
			add = 25
			scope:actor.faith = scope:recipient.faith 
			desc = "SAME_FAITH"
		}

		modifier = {
			add = -50
			has_relation_rival = scope:recipient
		}
		modifier = {
			add = -80
			has_relation_nemesis = scope:recipient
		}
		
		modifier = { # Safegurad for failed infernalist baron initiation
			add = 1000
			scope:actor = { is_ai = yes }
		}
	}
	
	ai_will_do = {
		base = 100
		
		modifier = { # The AI will only use a Hook if they couldn't otherwise do this
			scope:hook = yes
			add = -1
		}
	}
}

POD_revelation_interaction = {
	category = interaction_category_demon
	common_interaction = yes
	interface_priority = 120
	desc = POD_revelation_interaction_desc
	icon = POD_arcana_interaction
	
	cooldown = { months = 1 }
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_trait = demon }
		scope:actor = { has_perk = arcana_revelation_perk }
	}

	is_valid = {
		scope:actor = { 
			OR = {
				has_character_modifier = POD_shapeshift_form_apocalyptic_angelic_modifier
				has_character_modifier = POD_shapeshift_form_apocalyptic_damned_modifier
			}
		}
		scope:recipient = { 
			OR = {
				POD_can_be_vamphunted_scheme_trigger = yes
				has_relation_demonic_master = scope:actor
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = { 
			piety >= 0
		}
		scope:actor = { dread >= scope:recipient.ai_boldness }
	}

	on_accept = {
		scope:recipient = {
			while = {
				count = 5
				random = {
					chance = 5
					scope:actor = { POD_resolve_increase_effect = yes }
					add_stress = minor_stress_impact_gain 
					add_piety = minor_piety_loss
					modifier = { 
						add = {
							value = scope:actor.dread
							multiply = 1
						}
						NOT = { scope:actor.dread = 0 }
					}
					modifier = {
						add = {
							value = ai_boldness
							multiply = -1
						}
						NOT = { ai_boldness = 0 }
					}
				}
			}
			if = { 
				limit = { scope:actor = { has_character_modifier = POD_shapeshift_form_apocalyptic_damned_modifier } }
				add_opinion = {
					target = scope:actor
					modifier = hurt_opinion
					opinion = -25
				}
			}
			else = {
				add_opinion = {
					target = scope:actor
					modifier = pleased_opinion
					opinion = 15
				}
			}
			if = { 
				limit = { has_trait = mortal }
				add_trait = POD_knows_too_much 
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

	is_shown = {
		NOT = { scope:actor = scope:recipient }

			trigger_if = {
				limit = { 
					has_trait = truefallen 
					has_variable = host
				}
				NOT = { var:host = scope:recipient }
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = { 
			trigger_if = { 
				limit = { 
					NOT = { has_perk = soul_burn_perk } 
					NOT = { has_character_modifier = trapped_in_abyss_modifier }
				}
				NOT = { has_character_modifier = resolve0_modifier } 
			}
			trigger_if = {
				limit = {
					OR = {
						has_trait = earthbound
						AND = {
							has_trait = truefallen
							has_variable = host
						}
					}
				}
				has_perk = transfer_vessel_perk
			}
		}
		scope:recipient = {
			has_relation_demonic_master = scope:actor
			can_be_demonic_possesed = yes
		}
	}
	
	on_accept = {
		scope:actor = {
			if = { 
				limit = { 
					has_character_modifier = trapped_in_abyss_modifier
				}
				add_piety = -10000
				remove_variable = spirit_location
				remove_character_modifier = trapped_in_abyss_modifier
			}
			if = { 
				limit = { has_trait = earthbound }
				if = { 
					limit = { has_variable = stone_prison }
					save_scope_as = owner
					var:stone_prison = {
						save_scope_as = this_artifact
						set_artifact_description = artifact_old_reliquarie_desc 
						remove_variable = reliquarie
						set_variable = possessed_artifact
					}
					remove_variable = stone_prison
				}
				if = { 
					limit = { has_variable = earth_bound }
					var:earth_bound = {
						remove_variable = reliquarie
						every_de_jure_county = { 
							remove_county_modifier = earthbound_modifier 
							remove_county_modifier = earthbound_negative_modifier 
						}
					}
					remove_variable = earth_bound
				}
				remove_trait = earthbound
			}
			if = { 
				limit = { NOT = { has_trait = truefallen } }
				add_trait_force_tooltip = truefallen
			}
			set_relation_possessed_target = scope:recipient
			custom_tooltip = POD_demonic_possess_interaction.tt
			POD_remove_demon_vessel_effect = yes
			POD_first_demon_vessel_effect = { HOST = scope:recipient }
		}
		scope:recipient = {
			custom_tooltip = POD_demonic_possess_host_leech.tt
			custom_tooltip = POD_demonic_possess_maintenance.tt
		}
	}
	
	auto_accept = yes
	
	ai_frequency = 120

	ai_potential = {
		# The basic requirements
		is_imprisoned = no 
		is_landed = yes
		primary_title.tier > tier_barony
		
		has_trait = truefallen
	}
	ai_targets = {
		ai_recipients = courtiers
		ai_recipients = peer_vassals
		max = 10
	}
	
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 100
			num_of_relation_possessed_target = 0
		}
	}
}
