POD_artifact_heal_interaction = {
	category = interaction_category_friendly
	icon = artifact
	desc = POD_artifact_heal_interaction_desc
	options_heading = POD_artifact_heal_interaction_options_heading
	
	target_type = artifact
	target_filter = actor_artifacts
	
	can_be_picked_artifact = {
		scope:target = {
			has_variable = healing_artifact
		}
	}
	
	is_shown = {
		scope:actor = { any_equipped_character_artifact = { has_variable = healing_artifact } }
		POD_interaction_is_shown = yes
	}

	is_valid = {
		scope:recipient = {
			POD_has_injury_or_main_trait_trigger = yes
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			POD_can_be_healed_trigger = yes
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}
	
	send_option = {
		is_shown = {
			scope:recipient = {
				has_trait = scarred
			}
		}
		is_valid = {
			scope:recipient = { POD_is_trait_not_permanent_trigger = { TRAIT = scarred } }
		}
		flag = scarred
		localization = POD_heal_scarred
	}
	
	send_option = {
		is_shown = {
			scope:recipient = {
				has_trait = disfigured
			}
		}
		is_valid = {
			scope:recipient = { POD_is_trait_not_permanent_trigger = { TRAIT = disfigured } }
		}
		flag = disfigured
		localization = POD_heal_disfigured
	}
	
	send_option = {
		is_shown = {
			scope:recipient = {
				POD_health_has_superficial_wound_trigger = yes
			}
		}
		flag = wounded
		localization = POD_heal_wounded
	}

	send_option = {
		is_shown = {
			scope:recipient = {
				has_trait = blind
			}
		}
		is_valid = {
			scope:recipient = { POD_is_trait_not_permanent_trigger = { TRAIT = blind } }
		}
		flag = blind
		localization = POD_heal_blind
	}
	
	send_option = {
		is_shown = {
			scope:recipient = {
				has_trait = maimed
			}
		}
		is_valid = {
			scope:recipient = { POD_is_trait_not_permanent_trigger = { TRAIT = maimed } }
		}
		flag = maimed
		localization = POD_heal_maimed
	}
	
	send_option = {
		is_shown = {
			scope:recipient = {
				has_trait = one_legged
			}
		}
		is_valid = {
			scope:recipient = { POD_is_trait_not_permanent_trigger = { TRAIT = one_legged } }
		}
		flag = one_legged
		localization = POD_heal_one_legged
	}

	send_option = {
		is_shown = {
			scope:recipient = {
				has_trait = one_eyed
			}
		}
		is_valid = {
			scope:recipient = { POD_is_trait_not_permanent_trigger = { TRAIT = one_eyed } }
		}
		flag = one_eyed
		localization = POD_heal_one_eyed
	}
	
	send_option = {
		is_shown = {
			scope:recipient = {
				has_trait = eunuch_1
			}
		}
		is_valid = {
			scope:recipient = { POD_is_trait_not_permanent_trigger = { TRAIT = eunuch_1 } }
		}
		flag = eunuch
		localization = POD_heal_eunuch
	}
	
	on_accept = {
		if = {
			limit = { scope:actor != scope:recipient }
			scope:actor = { POD_commit_heal_effect = yes }
			scope:recipient = {
				add_opinion = {
					target = scope:actor
					modifier = pleased_opinion
					opinion = 5
				}
			}
		}
		scope:recipient = {
			play_music_cue = "mx_cue_peace_ensues"
			switch = {
				trigger = yes
				scope:scarred = {
					remove_trait = scarred
					scope:target = { add_durability = -5 }
				}
				scope:disfigured = {
					remove_trait = disfigured
					scope:target = { add_durability = -10 }
				}
				scope:wounded = {
					POD_heal_superficial_wound_effect = { COUNT = POD_base_heal_value }
					scope:target = { add_durability = -5 }
				}
				scope:one_eyed = {
					remove_trait = one_eyed
					scope:target = { add_durability = -10 }
				}
				scope:blind = {
					remove_trait = blind
					add_trait = one_eyed
					scope:target = { add_durability = -10 }
				}
				scope:one_legged = {
					remove_trait = one_legged
					scope:target = { add_durability = -15 }
				}
				scope:maimed = {
					remove_trait = maimed
					scope:target = { add_durability = -15 }
				}
				scope:eunuch = {
					remove_trait = eunuch_1
					scope:target = { add_durability = -15 }
				}
			}
		}
	}
	
	auto_accept = yes
}

POD_caladbolg_interaction = {
	category = interaction_category_hostile
	icon = artifact
	desc = POD_caladbolg_interaction_desc
	
	cooldown = { years = 1 }

	is_shown = {
		scope:actor != scope:recipient
		scope:actor = { any_equipped_character_artifact = { has_variable = caladbolg } }
		scope:recipient = { is_playable_character = yes }
		scope:recipient.highest_held_title_tier > tier_barony
		scope:actor = { NOT = { target_is_liege_or_above = scope:recipient } }
		POD_hostile_interaction_is_shown = yes
	}
	
	is_valid_showing_failures_only = {
		POD_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:recipient = {
			every_knight = {
				add_opinion = {
					target = scope:actor
					modifier = hurt_opinion
					opinion = -25
				}
				if = {
					limit = {
						scope:actor = {
							prowess_diff = {
								target = scope:recipient
								value >= 50
							}
						}
					}
					increase_wounds_multiple_hostile_effect = { REASON = lightning AMOUNT = 4 ATTACKER = scope:actor }
				}
				else_if = {
					limit = {
						scope:actor = {
							prowess_diff = {
								target = scope:recipient
								value >= 25
							}
						}
					}
					increase_wounds_multiple_hostile_effect = { REASON = lightning AMOUNT = 3 ATTACKER = scope:actor }
				}
				else_if = {
					limit = {
						scope:actor = {
							prowess_diff = {
								target = scope:recipient
								value >= 0
							}
						}
					}
					increase_wounds_multiple_hostile_effect = { REASON = lightning AMOUNT = 2 ATTACKER = scope:actor }
				}
				else = {
					increase_wounds_multiple_hostile_effect = { REASON = lightning AMOUNT = 1 ATTACKER = scope:actor }
				}
			}
		}
		scope:actor = {
			random_equipped_character_artifact = {
				limit = { has_variable = caladbolg }
				add_durability = -25
			}
		}
	}
	
	auto_accept = yes
}

			limit = {
				exists = scope:target

			}
			scope:actor = {
				trigger_if = {
					limit = { POD_splat_has_access_to_hekau = yes }
					has_perk = greater_relic_perk #Mummy
				}
				trigger_else = {
					has_perk = dream_the_form_perk #Mokole
				}
			}
		}
		trigger_else = { always = yes }
		
		trigger_if = {
			limit = { scope:recipient = { POD_deconstruct_artifact_cost > 0 } }
			scope:recipient = { POD_has_enough_splat_agnostic_energy_reserves = { AMOUNT = POD_deconstruct_artifact_cost } }
		}
		POD_self_targeted_interaction_is_valid_showing_failures_only = yes
	}

	can_be_picked_artifact = {
		scope:target = {
			POD_artifact_is_destructible_by_events_trigger = yes
		}
	}
	

	on_accept = {
		scope:actor = {
			if = {
				limit = { POD_deconstruct_artifact_cost > 0 }
				POD_splat_agnostic_energy_cost = { COUNT = POD_deconstruct_artifact_cost }
			}
			if = {
				limit = { has_perk = reshape_object_perk }
				add_short_term_gold = tiny_gold_value
			}
			if = {
				limit = { domicile ?= { has_domicile_parameter = deconstruction_lower_gnosis_t1 } }

			}
		}
		scope:target = {
			switch = {
				trigger = rarity

				common = { scope:recipient = { gain_common_artifact_pieces_effect = { VALUE = POD_deconstruct_artifact_amount } } }
				masterwork = { scope:recipient = { gain_masterwork_artifact_pieces_effect = { VALUE = POD_deconstruct_artifact_amount } } }

				illustrious = { scope:recipient = { gain_illustrious_artifact_pieces_effect = { VALUE = POD_deconstruct_artifact_amount } } }
			}

			POD_destroy_artifact_side_effect = yes
			destroy_artifact = this
		}
	}

	auto_accept = yes
	
	ai_frequency = 0
	ai_will_do = { base = 0 }
