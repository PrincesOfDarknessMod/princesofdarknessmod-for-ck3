POD_absorb_chi_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	interface_priority = 100
	desc = POD_absorb_chi_interaction_desc
	icon = POD_art_interaction
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { 
			OR = {
				has_trait = kueijin 
				has_trait = hun_and_po 
			}
		}
		NOT = { has_global_variable = wan_xian_wan_kuei }
	}
	
	is_valid = {
		scope:recipient = {
			is_imprisoned_by = scope:actor
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = { NOT = { has_character_modifier = celestial_barrier_modifier } }
		scope:recipient = {
			NOT = { has_perk = frozen_river_perk }
			NOT = { has_trait = torpor }
			NOT = { has_trait = automaton }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
			POD_is_not_clone_trigger = yes
		}
		POD_cross_umbra_interaction_valid_trigger = yes
		custom_description = {
		    object = scope:actor
		    text = "I_am_full_of_chi"
			scope:actor = {
				NAND = { 
					has_character_modifier = yinchi10_modifier
					has_character_modifier = yangchi10_modifier
				}
			}
		}
	}
	
	send_options_exclusive = no
	
	send_option = {
		is_shown = {
			scope:actor = {
				has_midhigh_dharma_trigger = yes
			}
			scope:recipient = {
				yin_energy_points_value >= 1
			}
		}
		is_valid = {
			scope:yangchi = no
			scope:flesh = no
			trigger_if = {
				limit = { scope:devore_chi_even_overdose = no }
				custom_description = {
					object = scope:actor
					text = "I_am_full_of_chi"
					scope:actor = { yin_difference >= 1 }
				}
			}
		}
		flag = yinchi
		localization = POD_absorb_chi_interaction_breath_yin
	}
	
	send_option = {
		is_shown = {
			scope:actor = {
				has_midhigh_dharma_trigger = yes
			}
			scope:recipient = {
				yang_energy_points_value >= 1
			}
		}
		is_valid = {
			scope:yinchi = no
			scope:flesh = no
			trigger_if = {
				limit = { scope:devore_chi_even_overdose = no }
				custom_description = {
					object = scope:actor
					text = "I_am_full_of_chi"
					scope:actor = { yang_difference >= 1 }
				}
			}
		}
		flag = yangchi
		localization = POD_absorb_chi_interaction_breath_yang
	}
				
	# Devore flesh
	send_option = {
		is_shown = {
			scope:recipient = {
				POD_is_not_spirit_trigger = yes
			}
		}
		is_valid = {
			scope:yinchi = no
			scope:yangchi = no
			trigger_if = {
				limit = { scope:devore_chi_even_overdose = no }
				custom_description = {
					object = scope:actor
					text = "I_am_full_of_chi"
					scope:actor = {
						OR = {
							yin_difference >= 1
							yang_difference >= 1
						}
					}
				}
			}
		}
		flag = flesh
		localization = POD_absorb_chi_interaction_flesh
	}
	
	send_option = {
		flag = devore_chi_even_overdose
		localization = POD_absorb_chi_interaction_devore_chi_even_overdose
	}

	on_accept = {
		switch = {
			trigger = yes
			scope:flesh = {
				scope:recipient = {
					add_opinion = {
						target = scope:actor
						modifier = hurt_opinion
						opinion = -20
					}
				}
				POD_absorb_chi_from_flesh_or_blood_effect = { MURDERER = scope:actor VICTIM = scope:recipient }
			}
			scope:yinchi = {
				scope:recipient = {
					add_opinion = {
						target = scope:actor
						modifier = hurt_opinion
						opinion = -10
					}
				}
				POD_absorb_yinchi_from_breath_effect = { MURDERER = scope:actor VICTIM = scope:recipient }
			}
			scope:yangchi = {
				scope:recipient = {
					add_opinion = {
						target = scope:actor
						modifier = hurt_opinion
						opinion = -10
					}
				}
				POD_absorb_yangchi_from_breath_effect = { MURDERER = scope:actor VICTIM = scope:recipient }
			}
		}
	}
	
	auto_accept = yes
	
	
	ai_will_do = {
		base = 0
	}
}

	interface_priority = 100

			NOT = { has_trait = torpor }
			NOT = { has_trait = automaton }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
			POD_is_not_clone_trigger = yes
		}
		POD_cross_umbra_interaction_valid_trigger = yes
		scope:recipient = {
			is_imprisoned_by = scope:actor
		}
	}
	
	send_option = {
		flag = weapon
		localization = POD_steal_chi_interaction_weapon
	}

	send_option = {
		is_shown = {
			scope:actor = {
				is_ai = no
				has_perk = bite_of_the_mu_perk
			}
		}
		is_valid = {
			custom_tooltip = {
				text = "i_am_full_of_yangchi"
				scope:actor = {
					yang_difference >= 1
				}
			}
		}
		flag = yangchi
		localization = POD_steal_chi_interaction_yangchi
	}

	send_option = {
		is_shown = {
			scope:actor = {
				is_ai = no
				has_perk = bite_of_the_mu_perk
			}
		}
		is_valid = {
			custom_tooltip = {
				text = "i_am_full_of_yinchi"
				scope:actor = {
					yin_difference >= 1
				}
			}
		}
		flag = yinchi
		localization = POD_steal_chi_interaction_yinchi
	}

	on_accept = {
		switch = {
			trigger = yes
			scope:yangchi = { 
				scope:actor = {
					POD_absorb_yangchi_from_breath_effect = { MURDERER = scope:actor VICTIM = scope:recipient }
				}
			}
			scope:yinchi = { 
				scope:actor = {
					POD_absorb_yinchi_from_breath_effect = { MURDERER = scope:actor VICTIM = scope:recipient }
				}
			}
			scope:weapon = {
				scope:recipient = { POD_universal_energy_decrease_multiple_hostile_effect = { COUNT = 10 ATTACKER = scope:actor } }
				scope:actor = {
					add_character_modifier = {
						modifier = chi_enhanced_weapon_modifier
						years = 1
					}
				}
			}
		}
	}
	
	auto_accept = yes

	ai_potential = {
		# The basic requirements
		is_imprisoned = no 
		is_landed = yes 
	}
	ai_targets = {
		ai_recipients = prisoners
	}
	ai_frequency = 36
	ai_will_do = {
		base = 0
		
		opinion_modifier = {
			opinion_target = scope:recipient
			multiplier = -0.25
		}
		modifier = {
		    add = 50
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
		modifier = {
			factor = 0
			scope:recipient = {
				is_ai = no
			}
		}
	}
}
