POD_absorb_chi_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	interface_priority = 100
	desc = POD_absorb_chi_interaction_desc
	icon = POD_art_interaction
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { is_kueijin_trigger = yes }
	}

	is_valid_showing_failures_only = {
		scope:actor = { NOT = { has_character_modifier = celestial_barrier_modifier } }
		scope:recipient = {
			is_imprisoned_by = scope:actor
			NOT = { has_perk = frozen_river_perk }
			NOT = { has_trait = torpor }
			NOT = { has_trait = statue }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
			POD_is_not_clone_trigger = yes
			POD_is_not_trapped_in_umbra_trigger = { ACTOR = scope:actor }
		}
		custom_description = {
            object = scope:actor
            text = "I_am_full_of_chi"
			scope:actor = {
				OR = {
					yin_difference >= 1
					yang_difference >= 1
				}
			}
        }
	}
	
	send_option = {
		is_shown = {
			scope:actor = {
				has_midhigh_dharma_trigger = yes
			}
			scope:recipient = {
				yin_energy_points_value >= 1
			}
		}
		flag = yinchi
		localization = POD_absorb_chi_interaction_breath
	}
	
	send_option = {
		is_shown = {
			scope:actor = {
				has_midhigh_dharma_trigger = yes
			}
			scope:recipient = {
				yang_energy_points_value >= 1
			}
		}
		flag = yangchi
		localization = POD_absorb_chi_interaction_breath
	}
				
	# Devore flesh
	send_option = {
		is_shown = {
			scope:recipient = {
				NOT = { has_trait = wraith }
				NOT = { has_trait = demon }
			}
		}
		flag = flesh
		localization = POD_absorb_chi_interaction_flesh
	}

	on_accept = {
		if = {
			limit = {
				scope:flesh = yes
			}
			POD_absorb_chi_from_flesh_or_blood_effect = yes
			scope:recipient = {
				add_opinion = {
					target = scope:actor
					modifier = hurt_opinion
					opinion = -20
				}
			}
		}
		if = {
			limit = {
				scope:yinchi = yes
			}
			POD_absorb_yinchi_from_breath_effect = yes
			scope:recipient = {
				add_opinion = {
					target = scope:actor
					modifier = hurt_opinion
					opinion = -10
				}
			}
		}
		if = {
			limit = {
				scope:yangchi = yes
			}
			POD_absorb_yangchi_from_breath_effect = yes
			scope:recipient = {
				add_opinion = {
					target = scope:actor
					modifier = hurt_opinion
					opinion = -10
				}
			}
		}
	}
	
	auto_accept = yes
	
	
	ai_will_do = {
		base = 0
	}
}

POD_steal_chi_interaction = {
	category = interaction_category_hostile
	common_interaction = yes
	interface_priority = 100
	desc = POD_steal_chi_interaction_desc
	icon = POD_art_interaction
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { 
			OR = {
				has_trait = shih
				has_trait = dhampyr
			}
			has_perk = clutch_of_the_wan_kuei_perk
		}
		scope:recipient = {
			energy_points_value >= 1
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = { NOT = { has_character_modifier = celestial_barrier_modifier } }
		scope:recipient = {
			NOT = { has_perk = frozen_river_perk }
			NOT = { has_trait = torpor }
			NOT = { has_trait = statue }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
			POD_is_not_clone_trigger = yes
			POD_is_not_trapped_in_umbra_trigger = { ACTOR = scope:actor }
		}
		scope:recipient = {
			is_imprisoned_by = scope:actor
		}
	}
	
	send_option = {
		flag = weapon
		localization = POD_steal_chi_interaction_weapon
	}

	send_option = {
		is_shown = {
			scope:actor = {
				is_ai = no
				has_perk = bite_of_the_mu_perk
			}
		}
		is_valid = {
			custom_tooltip = {
				text = "i_am_full_of_yangchi"
				scope:actor = {
					yang_difference >= 1
				}
			}
		}
		flag = yangchi
		localization = POD_steal_chi_interaction_yangchi
	}

	send_option = {
		is_shown = {
			scope:actor = {
				is_ai = no
				has_perk = bite_of_the_mu_perk
			}
		}
		is_valid = {
			custom_tooltip = {
				text = "i_am_full_of_yinchi"
				scope:actor = {
					yin_difference >= 1
				}
			}
		}
		flag = yinchi
		localization = POD_steal_chi_interaction_yinchi
	}

	on_accept = {
		if = {
			limit = { scope:yangchi = yes }
			scope:actor = {
				POD_absorb_yangchi_from_breath_effect = yes
			}
		}
		else_if = {
			limit = { scope:yinchi = yes }
			scope:actor = {
				POD_absorb_yinchi_from_breath_effect = yes
			}
		}
		else = {
			scope:recipient = { POD_universal_energy_decrease_multiple_effect = { COUNT = 10 } }
			if = {
				limit = { scope:weapon = yes }
				scope:actor = {
					add_character_modifier = {
						modifier = chi_enhanced_weapon_modifier
						years = 1
					}
				}
			}
		}
	}
	
	auto_accept = yes

	ai_potential = {
		# The basic requirements
		is_imprisoned = no 
		is_landed = yes 
	}
	ai_targets = {
		ai_recipients = prisoners
	}
	ai_frequency = 36
	ai_will_do = {
		base = 0
		
		opinion_modifier = {
			opinion_target = scope:recipient
			multiplier = -0.25
		}
		modifier = {
		    add = 50
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
		modifier = {
			factor = 0
			scope:recipient = {
				is_ai = no
			}
		}
	}
}
