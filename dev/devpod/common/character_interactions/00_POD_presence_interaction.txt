POD_entrance_interaction = {
	category = interaction_category_vampire
	#common_interaction = yes
	#interface_priority = 89
	scheme = PODentrance
	icon = POD_presence_interaction
	
	#ai_frequency = 0

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		scope:actor = { 
			OR = {
				has_perk = entrancement_perk 
				has_trait = presenceadvanced
			}
		}	
		scope:actor = { has_POD_vamp_blood_trigger = yes }
		scope:actor = { is_ai = no }
		scope:recipient = {
			NOT = {
				has_opinion_modifier = {
					target = scope:actor
					modifier = entrancement_opinion
				}
			}
		}
		POD_is_in_same_plane_trigger = yes
	}

	is_valid_showing_failures_only = {
		scope:recipient = { 
			NOT = { has_trait = torpor }
			NOT = { has_trait = statue }
			NOT = { has_trait = wight }
			NOT = { has_trait = barghest }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }	
			POD_is_not_clone_trigger = yes
		}
		custom_description = {
            object = scope:actor
            text = "I_must_be_at_the_same_location"
			scope:actor = { exists = location }
			scope:recipient = { exists = location }
            scope:actor = { location = scope:recipient.location }
        }
		custom_description = {
            object = scope:recipient
            text = "strong_mental_defenses_trigger"
            scope:recipient = { strong_mental_defenses_trigger = no }
        }				
		scope:actor = {
			can_start_scheme = {
				type = PODentrance
				target = scope:recipient
			}
		NOT = { has_character_modifier = bloodhunger5_modifier }	
		}
	}

	desc = {
		triggered_desc = {
			trigger = {
				scope:actor = {
					NOT = { is_at_war_with = scope:recipient }
					can_start_scheme = {
						type = PODentrance
						target = scope:recipient
					}

				}
			}
			desc = scheme_interaction_tt_PODentrance_approved
		}

	}
	
	on_accept = {
		hidden_effect = {
			scope:actor = {
				start_scheme = {
					type = PODentrance
					target = scope:recipient
				}
			}
		}	
	}

	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_mask_of_empathy_interaction = {
    category = interaction_category_vampire
	#common_interaction = yes
	#interface_priority = 96
    desc = POD_mask_of_empathy_interaction_desc
	icon = POD_presence_interaction
	
	ai_frequency = 0
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_adult = yes }
		scope:actor = { has_POD_vamp_blood_trigger = yes }
		scope:actor = { has_perk = mask_of_empathy_perk }
		scope:actor = { is_ai = no }		
		POD_is_in_same_plane_trigger = yes
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
			NOT = { has_character_modifier = bloodhunger4_modifier }
			NOT = { has_character_modifier = bloodhunger5_modifier }
		}
		custom_description = {
			object = scope:actor
			text = no_valid_relations
			scope:recipient = {
				OR = {
					any_relation = {
						type = friend
						NOT = { this = scope:actor }
						count >= 1
					}
					any_relation = {
						type = rival
						NOT = { this = scope:actor }
						count >= 1
					}
					any_relation = {
						type = lover
						NOT = { this = scope:actor }
						count >= 1
					}
				}
			}
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			NOT = { has_trait = statue }
			NOT = { has_trait = wight }
			NOT = { has_trait = barghest }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }		
			POD_is_not_clone_trigger = yes
		}									
	}
	
	populate_recipient_list = {
		scope:recipient = {
			every_relation = {
				type = friend
				limit = {
				    is_alive = yes
					NOT = { this = scope:actor }
				}
				
				add_to_list = characters
			
			}
			every_relation = {
				type = rival
				limit = {
				    is_alive = yes
					NOT = { this = scope:actor }
				}
				
				add_to_list = characters
			
			}
			every_relation = {
				type = lover
				limit = {
				    is_alive = yes
					NOT = { this = scope:actor }
				}
				
				add_to_list = characters
			
			}
		}
	}
	
	send_option = {
		flag = friend
		is_shown = {
			scope:secondary_recipient = {
				has_relation_friend = scope:recipient
			}
		}
		is_valid = {
			scope:secondary_recipient = {
				NOT = { has_relation_best_friend = scope:recipient }
			}
		}
		localization = POD_mask_of_empathy_friend
	}
	
	send_option = {
		flag = rival
		is_shown = {
			scope:secondary_recipient = {
				has_relation_rival = scope:recipient
			}
		}
		is_valid = {
			scope:secondary_recipient = {
				NOT = { has_relation_nemesis = scope:recipient }
			}
		}
		localization = POD_mask_of_empathy_rival
	}
	
	send_option = {
		flag = lover
		is_shown = {
			scope:secondary_recipient = {
				has_relation_lover = scope:recipient
			}
		}
		is_valid = {
			scope:secondary_recipient = {
				NOT = { has_relation_soulmate = scope:recipient }
			}
		}
		localization = POD_mask_of_empathy_lover
	}
	#This option exists because my horrible spaghetti code won't work without it
	send_option = {
		flag = nothing
		localization = POD_mask_of_empathy_nothing
	}
	
	on_accept = {
		custom_tooltip = POD_mask_of_empathy_interaction.tt
		hidden_effect = {
			if = {
				limit = { scope:nothing = no }
				scope:actor = {
					trigger_event = {
						id = POD_mask_of_empathy.001
						days = 10
					}
				}
			}
			if = {
				limit = {
					scope:friend = yes
				}
				scope:recipient = {
					add_character_flag = losing_friend
				}
			}
			if = {
				limit = {
					scope:rival = yes
				}
				scope:recipient = {
					add_character_flag = losing_rival
				}
			}
			if = {
				limit = {
					scope:lover = yes
				}
				scope:recipient = {
					add_character_flag = losing_lover
				}
			}
		}
	}
	
	auto_accept = yes
}

POD_perfect_world_interaction = {
	category = interaction_category_vampire
    desc = POD_perfect_world_interaction_desc
	icon = POD_presence_interaction
	
	ai_frequency = 360

	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { is_adult = yes }
		scope:actor = { has_POD_vamp_blood_trigger = yes }
		scope:actor = { has_perk = perfect_world_perk }
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
		    NOT = { has_character_modifier = bloodhunger4_modifier }
			NOT = { has_character_modifier = bloodhunger5_modifier }
		}
	}

	on_accept = {
	   custom_tooltip = POD_perfect_world_interaction.tt
	   scope:actor = {
	       trigger_event = POD_perfect_world.001
	   }
    }
	
	ai_targets = {
		ai_recipients = self
	}
	
	auto_accept = yes
	
	cooldown = { years = 15 }
	
	ai_will_do = {
		base = 0
	}  	
}
