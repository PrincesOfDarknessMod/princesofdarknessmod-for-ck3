knighting_interaction = {
	category = interaction_category_friendly
	common_interaction = yes
	#interface_priority = 83
	scheme = knighting
	icon = POD_hunter_interaction
	
	ai_frequency = 0

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		scope:actor = { has_trait = supehunter  }
		scope:recipient = { NOT = { has_trait = supehunter } }
		scope:actor = {
			OR = {	
				dynasty = dynasty:13000002
				dynasty = dynasty:13000006 
				dynasty = dynasty:13000008
				AND = {
					exists = dynasty
					dynasty = {
						has_dynasty_modifier = custom_knightly_order_modifier
					}
				}
			}
		}
	}
	
	is_valid = {
		scope:recipient = { 
			has_trait = mortal
		}
	}

	is_valid_showing_failures_only = {
		custom_description = {
			subject = scope:actor
			text =  "i_can_only_hunterize_one_at_a_time"
			scope:actor = {
				NOT = {
					any_scheme = {
						scheme_type = knighting
					}
				}
			}
		}
		custom_description = {
			object = scope:actor
			text = "I_must_be_at_the_same_location"
			scope:actor = { exists = location }
			scope:recipient = { exists = location }
			scope:actor = { location = scope:recipient.location }
		}
		scope:recipient = { is_married = no }
		custom_tooltip = {
			text = "I_must_be_older"
			scope:actor = {
				age > scope:recipient.age
			}
		}			
		
		
		scope:actor = {
			can_start_scheme = {
				type = knighting
				target = scope:recipient
			}
		}
	}

	desc = {
		triggered_desc = {
			trigger = {
				scope:actor = {
					NOT = { is_at_war_with = scope:recipient }
					can_start_scheme = {
						type = knighting
						target = scope:recipient
					}

				}
			}
			desc = scheme_interaction_tt_knighting_approved
		}

	}

	on_accept = {
		hidden_effect = {
			scope:actor = {
				start_scheme = {
					type = knighting
					target = scope:recipient
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

ordinationrecommendation_interaction = {
	category = interaction_category_friendly
    desc = ordinationrecommendation_interaction_desc
	ai_frequency = 0
	icon = POD_hunter_interaction
	
	is_shown = {
		scope:actor = { is_ai = no }
		NOT = { scope:actor = scope:recipient }
		scope:actor = { is_adult = yes }
		scope:recipient = { is_adult = yes }
		scope:actor = { 
			OR = { 
				has_trait = supehunter 
				has_trait = wutian
				has_trait = shemsuheru
			}
		}
		scope:recipient = { 
			has_trait = supehunter 
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = { is_landed = yes }
		scope:recipient = { is_available = yes }
		NOT = { scope:actor = { is_at_war_with = scope:recipient } }
		custom_description = {
			subject = scope:recipient
			text =  target_can_only_hunterize_one_at_a_time
			scope:recipient = {
				NOR = {
					any_scheme = { scheme_type = knighting }
					any_scheme = { scheme_type = ordinationsisters }
					any_scheme = { scheme_type = ordinationoculi }
					any_scheme = { scheme_type = ordinationmurnau }
					any_scheme = { scheme_type = ordinationshih }
					any_scheme = { scheme_type = ordinationindep }
				}
			}
		}
		trigger_if = {
			limit = { scope:actor = { has_trait = wutian } }
			scope:recipient = { has_trait = shih }
		}
		trigger_if = {
			limit = { scope:actor = { has_trait = shemsuheru } }
			scope:recipient = { faith = faith:ikhwan_al_safa }
		}
		trigger_if = {
			limit = { scope:recipient = { POD_is_gantenbein_trigger = yes } }
			scope:actor = { POD_is_gantenbein_trigger = yes }
		}
		scope:recipient = {
			NOT = { dynasty ?= dynasty:13000004 }
		}
	}
	
	populate_recipient_list = {
		scope:actor = {
			every_courtier_or_guest = {
				limit = {
					trigger_if = {
						limit = { scope:recipient.dynasty = dynasty:13000007 }
						POD_is_gantenbein_trigger = no
						OR = {
							has_trait = mortal
							has_trait = ghoul
						}
					}
					trigger_else = {
						POD_is_mortal_trigger = yes
					}
					trigger_if = {
						limit = { scope:recipient.dynasty = dynasty:13000003 }
						is_female = yes
					}
					trigger_if = {
						limit = { scope:recipient.dynasty = dynasty:13000004 }
						dynasty = dynasty:13000004
					}
				}
				add_to_list = characters
			}
		}
		scope:recipient = {
			every_courtier_or_guest = {
				limit = {
					trigger_if = {
						limit = { scope:recipient.dynasty = dynasty:13000007 }
						POD_is_gantenbein_trigger = no
						OR = {
							has_trait = mortal
							has_trait = ghoul
						}
					}
					trigger_else = {
						POD_is_mortal_trigger = yes
					}
					trigger_if = {
						limit = { scope:recipient.dynasty = dynasty:13000003 }
						is_female = yes
					}
					trigger_if = {
						limit = { scope:recipient.dynasty = dynasty:13000004 }
						dynasty = dynasty:13000004
					}
				}
				add_to_list = characters
			}
		}
	}
	
	cost = {
		prestige = major_prestige_value
	}

	on_accept = {

		scope:recipient = {
