POD_learn_truefaith_interaction = {
	category = interaction_category_friendly
	#common_interaction = yes
	#interface_priority = 89
    desc = POD_learn_truefaith_interaction_desc
	
	ai_frequency = 0

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		scope:actor = { POD_is_supehunter_trigger = yes }
		scope:recipient = { POD_is_supehunter_trigger = yes }
		scope:recipient = { is_imprisoned = no }
		scope:actor = { is_ai = no }
		scope:actor = {
		    NOT = { has_trait = beatific }
			NOT = { has_trait = truefaith }
		}
		scope:recipient = { 
		    OR = {
			    has_trait = beatific 
				has_trait = truefaith
			}
		}
	}
	
	cost = { piety = 500 }
	cooldown = { years = 5 }
	
	is_valid_showing_failures_only = {
	custom_tooltip = {
			text = "He_must_be_my_friend"
		scope:recipient = { 
			OR = {
		has_relation_lover = scope:actor
		has_relation_friend = scope:actor
		has_relation_best_friend = scope:actor
		has_relation_soulmate = scope:actor
			}
		}
		}
		scope:actor = {
			NOT = { is_at_war_with = scope:recipient }
			}		
	}
	
	on_accept = {
	    scope:recipient = {
		    save_scope_as = teacher
		}
		scope:actor = {
				save_scope_as = student
			}
	    scope:actor = {
		    trigger_event = {
			    id = POD_learntruefaith.001
				days = 2
			}
        }	
	}   
}
