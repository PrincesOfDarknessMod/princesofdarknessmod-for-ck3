
POD_aura_absorption_perk_interaction = {
	category = interaction_category_vampire
	icon = POD_mytherceria_interaction
    desc = POD_aura_absorption_perk_interaction_desc
	
	cooldown = { months = 6 }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_perk = cleansed_in_blood_perk }
		scope:recipient = { has_POD_shapeshifted_form_trigger = yes }
	}

	is_valid_showing_failures_only = {
		scope:actor = {
		   NOT = { has_character_modifier = bloodhunger5_modifier }
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }	
		}
		POD_cross_umbra_interaction_valid_trigger = yes			
	}
	
	on_accept = {
		scope:actor = { POD_willpower_rouse_check_effect = yes }
		scope:actor = { POD_hunger_increase_effect = yes }
		scope:recipient = {
			custom_tooltip = POD_aura_absorption_perk_interaction_desc
			hidden_effect = {
				POD_clear_resonance_effect = yes
				POD_set_resonance_effect = yes
			}
		}
	}

	ai_frequency = 0
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_steal_the_mind_perk_interaction = {
	category = interaction_category_vampire
	icon = POD_mytherceria_interaction
    desc = POD_steal_the_mind_perk_interaction_desc
	
	cooldown = { years = 3 }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_perk = steal_the_mind_perk }
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			NOT = { has_character_modifier = bloodhunger1_modifier }
			NOT = { has_character_modifier = bloodhunger2_modifier }
			NOT = { has_character_modifier = bloodhunger3_modifier }
			NOT = { has_character_modifier = bloodhunger4_modifier }
			NOT = { has_character_modifier = bloodhunger5_modifier }
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			NOT = { has_trait = statue }
			NOT = { has_trait = wight }
			NOT = { has_trait = barghest }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
			POD_is_not_clone_trigger = yes
		}
		POD_cross_umbra_interaction_valid_trigger = yes			
	}
	
	send_option = {
		flag = diplomacy
		localization = diplomacy
	}
	
	send_option = {
		flag = martial
		localization = martial
	}
	
	send_option = {
		flag = stewardship
		localization = stewardship
	}
	
	send_option = {
		flag = intrigue
		localization = intrigue
	}
	
	send_option = {
		flag = learning
		localization = learning
	}
	
	on_accept = {
		scope:actor = { POD_willpower_rouse_check_effect = yes }
		scope:actor = {
			if = {
				limit = {
					scope:diplomacy = yes
					scope:martial = yes
					scope:stewardship = yes
					scope:intrigue = yes
					scope:learning = yes
				}
				POD_hunger_increase_multiple_effect = { COUNT = 5 }
			}
			else_if = {
				limit = {
					calc_true_if = {
						amount = 4
						scope:diplomacy = yes
						scope:martial = yes
						scope:stewardship = yes
						scope:intrigue = yes
						scope:learning = yes
					}
				}
				POD_hunger_increase_multiple_effect = { COUNT = 4 }
			}
			else_if = {
				limit = {
					calc_true_if = {
						amount = 3
						scope:diplomacy = yes
						scope:martial = yes
						scope:stewardship = yes
						scope:intrigue = yes
						scope:learning = yes
					}
				}
				POD_hunger_increase_multiple_effect = { COUNT = 3 }
			}
			else_if = {
				limit = {
					calc_true_if = {
						amount = 2
						scope:diplomacy = yes
						scope:martial = yes
						scope:stewardship = yes
						scope:intrigue = yes
						scope:learning = yes
					}
				}
				POD_hunger_increase_multiple_effect = { COUNT = 2 }
			}
			else = {
				POD_hunger_increase_effect = yes
			}
		}
		scope:actor = {
			duel = {
				skill = intrigue
				target = scope:recipient
				30 = {
					desc = POD_cleansed_in_blood_perk_interaction_sucess
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					scope:recipient = {
						if = {
							limit = { scope:diplomacy = yes }
							while = {
								count = diplomacy
								add_character_modifier = {
									modifier = steal_the_mind_diplomacy_loss_modifier
									years = 1
								}
							}
						}
						if = {
							limit = { scope:martial = yes }
							while = {
								count = martial
								add_character_modifier = {
									modifier = steal_the_mind_martial_loss_modifier
									years = 1
								}
							}
						}
						if = {
							limit = { scope:stewardship = yes }
							while = {
								count = stewardship
								add_character_modifier = {
									modifier = steal_the_mind_stewardship_loss_modifier
									years = 1
								}
							}
						}
						if = {
							limit = { scope:intrigue = yes }
							while = {
								count = intrigue
								add_character_modifier = {
									modifier = steal_the_mind_intrigue_loss_modifier
									years = 1
								}
							}
						}
						if = {
							limit = { scope:learning = yes }
							while = {
								count = learning
								add_character_modifier = {
									modifier = steal_the_mind_learning_loss_modifier
									years = 1
								}
							}
						}
					}
				}
				70 = {
					desc = POD_cleansed_in_blood_perk_interaction_fail
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier 
					}
					scope:actor = {
					add_prestige = -100
					}
				}
			}
		}
		scope:recipient = { 
			add_opinion = {
				target = scope:actor
				modifier = hurt_opinion
				opinion = -50
			}
			progress_towards_rival_effect = {
				CHARACTER = scope:actor
				REASON = rival_cast_spell
				OPINION = 0
			}
		}
	}
	
	auto_accept = yes

	ai_potential = {
		# The basic requirements
		is_imprisoned = no 
		is_landed = yes 
	}
	ai_targets = {
		ai_recipients = primary_war_enemies
	}
	ai_frequency = 120
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 45
			scope:actor = {
			    is_at_war_with = scope:recipient
			}
		}
		modifier = {
			factor = 2
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
		modifier = {
			factor = 0
			scope:recipient = {
				is_ai = no
			}
		}
	}
}

POD_stone_travel_perk_interaction = {
	category = interaction_category_vampire
	icon = POD_mytherceria_interaction
	desc = POD_stone_travel_perk_interaction_desc
	
	use_diplomatic_range = no
	cooldown = { months = 1 }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_perk = stone_travel_perk }
		scope:actor = { is_commanding_army = yes }
		scope:recipient = { exists = location }
	}

	is_valid_showing_failures_only = {
		scope:actor = {
		   NOT = { has_character_modifier = bloodhunger5_modifier }
		}
		scope:actor.location = { 
			NOT = { terrain = oasis }
			NOT = { terrain = desert }
			NOT = { terrain = desert_mountains }
			NOT = { terrain = wetlands }
			NOT = { terrain = floodplains }
		}
		scope:recipient.location = { 
			NOT = { terrain = oasis }
			NOT = { terrain = desert }
			NOT = { terrain = desert_mountains }
			NOT = { terrain = wetlands }
			NOT = { terrain = floodplains }
		}
	}

	on_accept = {
		scope:actor = { POD_willpower_rouse_check_effect = yes }
		scope:actor = { POD_hunger_increase_effect = yes }
		scope:actor = {
			duel = {
				skill = learning
				value = 10
			
				50 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					
					commanding_army = {
						set_army_location = scope:recipient.location
					}
				}
				50 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier
					}
					
					add_stress = minor_stress_impact_gain
				}
			}
		}
	}

	ai_frequency = 0
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
