# to be used with POD_clear_topscope_and_trigger_event
POD_trigger_delayed_event_interaction = {
	hidden = yes
	
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	
	is_shown = {
	}
	
	on_accept = {
		scope:actor = {
			if = {
				limit = { exists = var:jank_event_id }
				trigger_event = {
					saved_event_id = var:jank_event_id
					days = POD_jank_event_delay
				}
				remove_variable = jank_event_id
				remove_variable = jank_event_delay
			}
		}
	}
	
	auto_accept = yes
}
