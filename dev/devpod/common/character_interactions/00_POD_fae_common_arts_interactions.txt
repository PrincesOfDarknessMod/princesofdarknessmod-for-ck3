POD_fuddle_perk_interaction = {
	category = interaction_category_fae
	icon = POD_chicanery_interaction
	desc = POD_fuddle_perk_interaction_desc
	use_diplomatic_range = no
	
	# cooldown = { months = realm_time_monthly_cooldown }
	# cost = { piety = miniscule_piety_value }
	
	is_shown = {
		scope:actor = { has_POD_fae_blood_trigger = yes }
		scope:actor = { has_perk = fuddle_perk }
	}
	
	is_valid_showing_failures_only = {
		scope:actor = { 
			trigger_if = { 
				limit = { has_trait = hsien }
				custom_tooltip = {
					text = "i_am_empty_of_chi"
					NAND = { 
						has_character_modifier = yinchi0_modifier
						has_character_modifier = yangchi0_modifier
					}
				}
			}
			trigger_else = { NOT = { has_character_modifier = glamour0_modifier } }
		}
		realm_actor = yes
		realm_fae = yes
		realm_scene = yes
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
			custom_tooltip = { text = has_artifact_fans_of_ma_yi_yuan NOT = { any_equipped_character_artifact = { has_variable = fans_of_ma_yi_yuan } } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}
	
	send_option = {
		is_shown = {
			scope:recipient = {
				NOT = { has_character_modifier = fuddle_perk_positive_modifier }
			}
		}
		flag = positive
		localization = fuddle_perk_positive_modifier	
	}
	
	send_option = {
		is_shown = {
			scope:recipient = {
				NOT = { has_character_modifier = fuddle_perk_negative_modifier }
			}
		}
		flag = negative
		localization = fuddle_perk_negative_modifier	
	}
	
	send_option = {
		is_shown = {
			scope:recipient = {
				OR = { 
					has_character_modifier = fuddle_perk_positive_modifier 
					has_character_modifier = fuddle_perk_negative_modifier 
				}
			}
		}
		flag = remove
		localization = TITLE_VIEW_CLEAR_LAWS_ACCEPT	
	}
	
	on_accept = {
		scope:actor = { POD_willpower_rouse_check_effect = yes }
		scope:actor = { POD_cantrip_cost_effect = { COUNT = 1 } }
		scope:recipient = {
			if = {
				limit = { scope:positive = yes }
				add_fae_character_modifier = {
					MODIFIER = fuddle_perk_positive_modifier
				}
				if = {
					limit = { NOT = { scope:actor = scope:recipient } }
					scope:recipient = { 
						add_opinion = {
							target = scope:actor
							modifier = gift_opinion
							opinion = 5
						}
					}
				}
			}
			if = {
				limit = { scope:negative = yes }
				add_fae_character_modifier = {
					MODIFIER = fuddle_perk_negative_modifier
				}
				if = {
					limit = { NOT = { scope:actor = scope:recipient } }
					scope:recipient = { 
						add_opinion = {
							target = scope:actor
							modifier = hurt_opinion
							opinion = -5
						}
					}
				}
			}
			if = {
				limit = { scope:remove = yes }
				if = {
					limit = { has_character_modifier = fuddle_perk_positive_modifier }
					if = {
						limit = { NOT = { scope:actor = scope:recipient } }
						scope:recipient = { 
							add_opinion = {
								target = scope:actor
								modifier = gift_opinion
								opinion = -5
							}
						}
					}
					remove_character_modifier = fuddle_perk_positive_modifier
				}
				if = {
					limit = { has_character_modifier = fuddle_perk_negative_modifier }
					remove_character_modifier = fuddle_perk_negative_modifier
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no
		is_landed = yes
	}
	ai_targets = {
		ai_recipients = self
	}
	ai_targets = {
		ai_recipients = primary_war_enemies
	}
	ai_frequency = 36
	ai_will_do = {
		base = 0
		modifier = { 
			add = 100
			scope:actor = scope:recipient
			scope:positive = yes
		}
		modifier = {
			add = 45
			NOT = { scope:actor = scope:recipient }
			scope:negative = yes
			scope:actor = {
			    is_at_war_with = scope:recipient
			}
		}
		modifier = {
			factor = 2
			NOT = { scope:actor = scope:recipient }
			scope:negative = yes
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
		modifier = {
			factor = 0
			scope:recipient = {
				is_ai = no
			}
		}
	}
}
