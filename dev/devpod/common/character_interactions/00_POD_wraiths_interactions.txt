POD_control_puppet_interaction = {
	category = interaction_category_wraith
	icon = POD_puppetry_interaction
	desc = POD_control_puppet_interaction_desc
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { 
			has_opinion_modifier = {
				target = scope:actor
				modifier = POD_complete_possession_opinion
			}
		}
	}
	
	is_valid = {
		scope:recipient = { 
			is_ai = yes
			is_landed = yes
		}
	}
	
	is_valid_showing_failures_only = {
		POD_cross_umbra_interaction_valid_trigger = yes
		scope:recipient = { primary_title.tier >= tier_county }
	}
	
	on_accept = {
		scope:recipient = { 
			set_variable = {
				name = master_on_control
				value = scope:actor
			}
		}
		scope:actor = { 
			POD_wraith_willpower_check_effect = yes 
			add_character_modifier = control_puppet_modifier
			set_player_character = scope:recipient
		}
		custom_tooltip = POD_control_puppet_interaction_tt
	}
	
	auto_accept = yes
}

POD_possess_interaction = {
	category = interaction_category_wraith
	icon = POD_puppetry_interaction
	desc = POD_possess_interaction_desc
	
	cooldown = { years = 1 }
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			OR = { 
				has_perk = twitch_perk
				has_perk = puppetry_perk
			}
		}
		scope:recipient = { 
			trigger_if = {
				limit = { num_of_relation_possessesing_wraith > 0 }
				has_relation_possessesing_wraith = scope:actor
			}
			NOT = {
				has_opinion_modifier = {
					target = scope:actor
					modifier = POD_complete_possession_opinion
				}
			}
		}
	}
	
	is_valid_showing_failures_only = {
		scope:actor = { 
			trigger_if = {
				limit = { has_trait = spectre }
				NOT = { has_character_modifier = angst0_modifier } 
			}
			trigger_else = {
				NOT = { has_character_modifier = pathos0_modifier } 
			}
		}
		scope:actor = {
			custom_description = {
				text = POD_not_in_possession
				OR = { 
					has_perk = legion_perk
					num_of_relation_possessed_target = 0 
					has_relation_possessed_target = scope:recipient
				}
			}
			NOT = { has_trait = risen }
			trigger_if = {
				limit = { 
					has_hook_of_type  = {
						target = scope:recipient
						type = possessed_hook
					}
				}
				has_perk = rein_in_the_mind_perk
			}
			trigger_if = {
				limit = { 
					has_hook_of_type  = {
						target = scope:recipient
						type = possessed_weak_hook
					}
				}
				has_perk = masters_voice_perk
			}
			trigger_if = {
				limit = {
					NOR = {
						has_perk = vector_perk
						has_perk = wraithride_perk
					}
				}
				scope:recipient = {
					POD_can_be_possessed_trigger = yes
				}
			}
		}
		scope:recipient = {
			POD_not_has_possess_ward_trigger = yes
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}
	
	on_accept = {
		scope:actor = { POD_wraith_willpower_check_effect = yes }
		scope:actor = { POD_arcanoi_cost_effect = { COUNT = 1 } }
		scope:recipient = {
			trigger_event = POD_wraith_possession.100
		}
		custom_tooltip = POD_possess_interaction_desc
	}
	
	auto_accept = yes
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no 
		is_landed = yes 
	}
	ai_targets = {
		ai_recipients = liege
	}
	ai_targets = {
		ai_recipients = neighboring_rulers
		ai_recipients = peer_vassals
		max = 10
	}
	ai_frequency = 72
	ai_will_do = {
		base = 0
		
		opinion_modifier = {
			opinion_target = scope:recipient
			multiplier = -0.25
		}
		modifier = {
		    add = 50
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
		modifier = {
			factor = 0
			scope:recipient = {
				is_ai = no
			}
		}
	}
}
