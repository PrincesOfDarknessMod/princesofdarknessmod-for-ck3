POD_infiltrate_interaction = {
	category = interaction_category_hostile
	#common_interaction = yes
	#interface_priority = 90

	ai_frequency = 0

	is_shown = {
		scope:actor != scope:recipient
		scope:actor = { this = character:790179 }
		scope:recipient = { is_adult = yes }
		scope:recipient = { is_playable_character = yes }
		scope:recipient = { is_vampire_trigger = yes }
		is_target_in_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:helena_decision
		}
		POD_hostile_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:recipient = { is_imprisoned = no }
		POD_interaction_is_valid_showing_failures_only = yes
	}

	desc = POD_infiltrate_interaction_desc
	
	cost = { gold = 150 }
	
	cooldown = { months = 6 }

	on_accept = {
		scope:recipient = {
			add_character_modifier = {
				modifier = infiltrate_modifier
				years = 5
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_asakku_blood_inception_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	interface_priority = 75
	icon = POD_asakku_asakku

	cooldown = { years = 1 }
	ai_frequency = 24

	is_shown = {
		scope:actor = {
			POD_asakku_has_asakku_trigger = yes
		}
		scope:recipient = {
			POD_is_vampire_trigger = yes
			POD_asakku_has_asakku_trigger = no
			NOT = { has_trait = true_faith }
			is_imprisoned_by = scope:actor
		}
		scope:actor != scope:recipient
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			POD_has_enough_blood_for_discipline = { AMOUNT = 1 }
		}
		scope:recipient = {
			is_adult = yes
		}
	}

	desc = POD_asakku_blood_inception_interaction_desc

	on_accept = {
		scope:recipient = {
			save_scope_as = target
		}
		scope:recipient = {
			# Grant Asakku
			POD_asakku_grant_asakku_effect = yes
			# Grant Vicissitude if target lacks it
			if = {
				limit = {
					has_vicissitude_trigger = no
				}
				add_trait = vicissitude
			}
		}
		# Applies a strong bloodbond hook
		scope:actor = {
			add_hook = {
				target = scope:recipient
				type = strong_bloodoath_hook
			}
		}
		# Increment Blood Inception counter on the actor
		scope:actor = {
			if = {
				limit = { NOT = { has_variable = POD_asakku_blood_inception_count } }
				set_variable = {
					name = POD_asakku_blood_inception_count
					value = 0
				}
			}
			change_variable = {
				name = POD_asakku_blood_inception_count
				add = 1
			}
			# Fire Blood Inception event
			POD_hunger_rouse_check_effect = yes
			trigger_event = POD_asakku.0301
		}
	}

	auto_accept = yes

	ai_will_do = {
		base = 75
	}
	ai_targets = {
		ai_recipients = prisoners
	}
}
