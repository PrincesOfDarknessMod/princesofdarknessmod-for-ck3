knighting_interaction = {
	category = interaction_category_friendly
	common_interaction = yes
	#interface_priority = 83
	scheme = knighting
	icon = POD_hunter_interaction
	
	ai_frequency = 0

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_adult = yes }
		scope:recipient = { is_a_cultist = no }
		scope:actor = { is_adult = yes }
		scope:actor = { has_trait = supehunter  }
		scope:recipient = { NOT = { has_trait = supehunter } }
		scope:actor = { has_knightly_order_dynasty_trigger = yes }
	}
	
	is_valid = { scope:recipient = { has_trait = mortal } }

	is_valid_showing_failures_only = {
		custom_description = {
			subject = scope:actor
			text =  "i_can_only_hunterize_one_at_a_time"
			scope:actor = {
				NOT = {
					any_scheme = {
						scheme_type = knighting
					}
				}
			}
		}
		scope:recipient = { is_married = no }
		custom_tooltip = {
			text = "I_must_be_older"
			scope:actor = {
				age > scope:recipient.age
			}
		}			
		
		
		scope:actor = {
			can_start_scheme = {
				type = knighting
				target_character = scope:recipient
			}
		}
	}

	desc = {
		triggered_desc = {
			trigger = {
				scope:actor = {
					NOT = { is_at_war_with = scope:recipient }
					can_start_scheme = {
						type = knighting
						target_character = scope:recipient
					}

				}
			}
			desc = scheme_interaction_tt_knighting_approved
		}

	}

	on_accept = {
		hidden_effect = {
			scope:actor = {
				start_scheme = {
					type = knighting
					target_character = scope:recipient
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

ordinationrecommendation_interaction = {
	category = interaction_category_friendly
    desc = ordinationrecommendation_interaction_desc
	ai_frequency = 0
	icon = POD_hunter_interaction
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { is_adult = yes }
		scope:recipient = { is_adult = yes }
		scope:actor = { 
			OR = { 
				has_trait = supehunter 
				has_trait = wutian
				has_trait = shemsuheru
			}
		}
		scope:recipient = { 
			has_trait = supehunter 
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = { is_playable_character = yes }
		scope:recipient = { is_available = yes }
		NOT = { scope:actor = { is_at_war_with = scope:recipient } }
		custom_description = {
			subject = scope:recipient
			text =  target_can_only_hunterize_one_at_a_time
			scope:recipient = {
				NOR = {
					any_scheme = { scheme_type = knighting }
					any_scheme = { scheme_type = ordinationsisters }
					any_scheme = { scheme_type = ordinationred }
					any_scheme = { scheme_type = ordinationoculi }
					any_scheme = { scheme_type = ordinationmurnau }
					any_scheme = { scheme_type = ordinationshih }
					any_scheme = { scheme_type = ordinationindep }
					any_scheme = { scheme_type = ordinationsanbenito }
				}
			}
		}
		trigger_if = {
			limit = { scope:actor = { has_trait = wutian } }
			scope:recipient = { has_trait = shih }
		}
		trigger_if = {
			limit = { scope:actor = { has_trait = shemsuheru } }
			scope:recipient = { faith = faith:ikhwan_al_safa }
		}
		trigger_if = {
			limit = { scope:recipient = { POD_is_gantenbein_trigger = yes } }
			scope:actor = { POD_is_gantenbein_trigger = yes }
		}
		trigger_if = {
			limit = { scope:recipient = { faith = faith:sanbenito } }
			scope:actor = { faith = faith:sanbenito }
		}
		trigger_if = {
			limit = { exists = scope:secondary_recipient }
			scope:secondary_recipient = {
				trigger_if = {
					limit = { 
						scope:recipient.dynasty = { 
							OR = {
								this = dynasty:13000003 
								this = dynasty:13000011
							} 
						}
					}
					is_female = yes
				}
				trigger_if = {
					limit = { scope:recipient.dynasty = dynasty:13000004 }
					dynasty ?= dynasty:13000004
				}
				trigger_if = {
					limit = { scope:recipient = { has_trait = order_member } }
					is_married = no
				}
			}
		}
	}
	
	populate_recipient_list = {
		scope:actor = {
			every_courtier_or_guest = {
				limit = {
					trigger_if = {
						limit = { scope:recipient.faith = faith:sanbenito }
						NOT = { has_trait = supehunter }
						POD_is_infernalist_trigger = no
					}
					trigger_else_if = {
						limit = { scope:recipient.dynasty = dynasty:13000007 }
						POD_is_gantenbein_trigger = no
						OR = {
							has_trait = mortal
							has_trait = ghoul
						}
					}
					trigger_else = {
						has_trait = mortal
					}
				}
				add_to_list = characters
			}
		}
		scope:recipient = {
			every_courtier_or_guest = {
				limit = {
					trigger_if = {
						limit = { scope:recipient.faith = faith:sanbenito }
						NOT = { has_trait = supehunter }
						POD_is_infernalist_trigger = no
					}
					trigger_else_if = {
						limit = { scope:recipient.dynasty = dynasty:13000007 }
						POD_is_gantenbein_trigger = no
						OR = {
							has_trait = mortal
							has_trait = ghoul
						}
					}
					trigger_else = {
						has_trait = mortal
					}
				}
				add_to_list = characters
			}
		}
	}
	
	cost = {
		prestige = major_prestige_value
	}

	on_accept = {

		scope:recipient = {
			if = {
				limit = {
					faith = faith:sanbenito
				}
				start_scheme = {
					type = ordinationsanbenito
					target_character = scope:secondary_recipient
				}
			}
			else_if = {
				limit = {
					has_trait = shih
				}
				start_scheme = {
					type = ordinationshih
					target_character = scope:secondary_recipient
				}
			}
			else_if = {
				limit = {
					dynasty = dynasty:13000007
				}
				start_scheme = {
					type = ordinationgantenbein
					target_character = scope:secondary_recipient
				}
			}
			else_if = {
				limit = { has_knightly_order_dynasty_trigger = yes }
				start_scheme = {
					type = knighting
					target_character = scope:secondary_recipient
				}
			}
			else_if = {
				limit = { has_mendicant_order_dynasty_trigger = yes }
				start_scheme = {
					type = ordinationsisters
					target_character = scope:secondary_recipient
				}
			}
			else_if = {
				limit = { has_contemplative_order_dynasty_trigger = yes }
				start_scheme = {
					type = ordinationred
					target_character = scope:secondary_recipient
				}
			}
			else_if = {
				limit = {
					dynasty = dynasty:13000004
				}
				start_scheme = {
					type = ordinationmurnau
					target_character = scope:secondary_recipient
				}
			}
			else_if = {
				limit = { has_lay_order_dynasty_trigger = yes }
				start_scheme = {
					type = ordinationoculi
					target_character = scope:secondary_recipient
				}
			}
			else = {
				start_scheme = {
					type = ordinationindep
					target_character = scope:secondary_recipient
				}
			}
		}
		
		scope:secondary_recipient = {
			add_opinion = {
				modifier = grateful_opinion
				target = scope:actor
				opinion = 20
			}
		}
	}
	
	ai_accept = {
		base = 0 # Try to make it 0 for most interactions
		
		opinion_modifier = { # Opinion Factor
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 1.0
			desc = AI_OPINION_REASON
		}
		
		opinion_modifier = { # Opinion Factor
			who = scope:recipient
			opinion_target = scope:secondary_recipient
			multiplier = 0.25
			desc = AI_OPINION_REASON
		}
	}
}
