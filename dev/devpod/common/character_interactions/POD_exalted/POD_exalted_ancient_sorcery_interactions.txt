POD_awakened_eye_of_the_dragon_interaction = {
	category = interaction_category_exalted
	desc = POD_awakened_eye_of_the_dragon_interaction_desc
	icon = ancient_sorcery

	target_type = title
	target_filter = actor_domain_titles

	cooldown = { years = 6 }

	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { has_perk = ancient_sorcery_awakened_eye_of_the_dragon_perk }
		scope:actor = {
			any_sub_realm_county = { title_province = { has_building_or_higher = dragon_nest_01 } }
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = { POD_has_enough_essence_for_charm = { AMOUNT = 10 } }
		POD_self_targeted_interaction_is_valid_showing_failures_only = yes
	}

	can_be_picked = {
		scope:target = {
			tier = tier_county
			title_province = { has_building_or_higher = dragon_nest_01 }
			NOT = { title_province = { has_building_or_higher = dragon_nest_07 } }
		}
	}

	on_accept = {
		scope:actor = {
			POD_ancient_sorcery_ceremony_effect = { 
				SORCERY = ancient_sorcery_awakened_eye_of_the_dragon 
				SORCERY_SPEED = 1
				SORCERY_COST = 10
				EFFECT = POD_awakened_eye_of_the_dragon_effect 
				TARGET = scope:target.title_province
			}
		}
	}

	auto_accept = yes

	ai_will_do = {
		base = 0
	}
}
