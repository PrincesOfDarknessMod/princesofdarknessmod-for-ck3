POD_discard_dragon_touched_interaction = {
	category = interaction_category_exalted
	desc = POD_discard_dragon_touched_interaction_desc
	icon = terrestrial_exalted
	
	target_type = title
	target_filter = actor_domain_titles

	common_interaction = yes

	is_shown = {
		scope:actor = { has_trait = terrestrial_exalted }
		scope:recipient = {
			has_trait = dragon_touched
			is_child_of = scope:actor
		}
		POD_hostile_interaction_is_shown = yes
	}

	can_be_picked_title = {
		scope:target = {
			title_province = {
				has_building_or_higher = dragon_nest_01
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			custom_tooltip = {
				text = POD_discard_dragon_touched_interaction_req
				any_directly_owned_province = {
					has_building_or_higher = dragon_nest_01
				}
			}
		}
		scope:recipient = {
			is_courtier_of = scope:actor
			age >= 18
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	send_option = {
		flag = stimulate_dragon_nest
		localization = POD_discard_dragon_touched_interaction_stimulate_dragon_nest
	}
	send_option = {
		flag = expand_dragon_nest
		localization = POD_discard_dragon_touched_interaction_expand_dragon_nest
	}
	send_option = {
		flag = stabilize_dragon_nest
		localization = POD_discard_dragon_touched_interaction_stabilize_dragon_nest
	}

	on_accept = {
		create_character = {
			save_scope_as = new_parent
			location = scope:actor.location
			culture = scope:actor.culture
			faith = scope:actor.faith
			gender = male
			random_traits = yes
		}
		create_character = {
			save_scope_as = new_parent2
			location = scope:actor.location
			culture = scope:actor.culture
			faith = scope:actor.faith
			gender = female
			random_traits = yes
		}
		scope:recipient = { 
			custom_tooltip = {
				text = POD_discard_dragon_touched_interaction_tt
				hidden_effect_new_object = {
					POD_set_parent_effect = { PARENT = scope:new_parent }
					POD_set_parent_effect = { PARENT = scope:new_parent2 }
				}
				set_to_lowborn = yes # To remove from dynasty
				move_to_pool = yes
				every_claim = { prev = { remove_claim = prev } }
				every_spouse = { divorce = prev }
				POD_remove_positive_vanilla_relations = yes
				POD_remove_negative_vanilla_relations = yes
				death = { death_reason = death_disappearance }
			}
		}
		scope:new_parent ?= {
			death = { death_reason = death_disappearance }
		}
		scope:new_parent2 ?= {
			death = { death_reason = death_disappearance }
		}
		scope:target.title_province = {
			switch = {
				trigger = yes
				scope:stimulate_dragon_nest = {
					save_scope_value_as = {
						name = new_level
						value = {
							value = 0
							if = { limit = { has_variable = stimulate_dragon_nest_var } value = var:stimulate_dragon_nest_var }
							add = 1
							if = {
								limit = { scope:actor.faith ?= { has_doctrine = special_restored_mothers_favor_doctrine } }
								add = 2
							}
						}
					}
					hidden_effect = {
						remove_province_modifier = POD_discard_dragon_touched_interaction_stimulate_dragon_nest_modifier
					}
					add_province_modifier = POD_discard_dragon_touched_interaction_stimulate_dragon_nest_modifier
					set_variable = {
						name = stimulate_dragon_nest_var
						value = scope:new_level
					}
				}
				scope:expand_dragon_nest = {
					save_scope_value_as = {
						name = new_level
						value = {
							value = 0
							if = { limit = { has_variable = expand_dragon_nest_var } value = var:expand_dragon_nest_var }
							add = 1
						}
					}
					hidden_effect = {
						remove_province_modifier = POD_discard_dragon_touched_interaction_expand_dragon_nest_modifier
					}
					add_province_modifier = POD_discard_dragon_touched_interaction_expand_dragon_nest_modifier
					set_variable = {
						name = expand_dragon_nest_var
						value = scope:new_level
					}
				}
				scope:stabilize_dragon_nest = {
					save_scope_value_as = {
						name = new_level
						value = {
							value = 0
							if = { limit = { has_variable = stabilize_dragon_nest_var } value = var:stabilize_dragon_nest_var }
							add = 1
						}
					}
					hidden_effect = {
						remove_province_modifier = POD_discard_dragon_touched_interaction_stabilize_dragon_nest_modifier
					}
					add_province_modifier = POD_discard_dragon_touched_interaction_stabilize_dragon_nest_modifier
					set_variable = {
						name = stabilize_dragon_nest_var
						value = scope:new_level
					}
				}
			}
		}
	}

	auto_accept = yes

	ai_will_do = { base = 0 }
}

POD_exalted_essence_training_interaction = {
	category = interaction_category_friendly
	icon = essence_icon
	desc = POD_exalted_essence_training_interaction_desc

	cooldown = { years = 5 }

	common_interaction = yes

	is_shown = {
		scope:actor = { has_trait = exalted }
		scope:recipient = { has_trait = exalted }
		POD_interaction_is_shown = yes
	}
	
	is_valid_showing_failures_only = {
		scope:recipient = { 
			NOT = { has_relation_rival = scope:actor }
			NOT = { has_relation_nemesis = scope:actor }
			is_imprisoned = no 
			custom_tooltip = {
				text = POD_exalted_essence_training_interaction_rating
				POD_exalted_rating < scope:actor.POD_exalted_rating
			}
			custom_tooltip = {
				text = POD_exalted_essence_training_interaction_close
				OR = {
					is_courtier_of = scope:actor
					is_vassal_of = scope:actor
					is_child_of = scope:actor
				}
			}

		}
		scope:actor = {
			POD_can_start_planning_phase_trigger = yes
			POD_not_already_on_journey_trigger = { TYPE = exalted_essence_training TARGET = scope:recipient }
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		POD_journey_interaction_pause_hint = yes
		scope:actor = {
			custom_tooltip = POD_exalted_essence_training_interaction_effect
			POD_start_targeted_journey_effect = {
				TYPE    = exalted_essence_training
				MONTHLY = exalted_essence_training
				TARGET  = scope:recipient
			}
		}
	}

	auto_accept = yes
}
