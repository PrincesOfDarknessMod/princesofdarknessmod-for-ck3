POD_manipulate_passive_charms_interaction = {
	category = interaction_category_exalted
	desc = POD_manipulate_passive_charms_interaction_desc
	icon = pod_icon #Temp

	needs_confirmation = { always = no }
	common_interaction = yes

	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { POD_is_exalted_trigger = yes }
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			POD_has_enough_essence_for_charm = { AMOUNT = 1 }
		}
		POD_self_targeted_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:actor = { set_variable = POD_open_passive_charms_gui }
	}

	auto_accept = yes

	ai_will_do = { base = 0 }
}

POD_discard_dragon_touched_interaction = {
	category = interaction_category_exalted
	desc = POD_discard_dragon_touched_interaction_desc
	icon = pod_icon #Temp

	common_interaction = yes

	is_shown = {
		scope:actor = { has_trait = exalted_terrestrial }
		scope:recipient = {
			has_trait = dragon_touched
			is_child_of = scope:actor
		}
		POD_hostile_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			any_directly_owned_province = {
				has_building_or_higher = dragon_nest_01
			}
		}
		scope:recipient = {
			is_courtier_of = scope:actor
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:recipient = { 
			hidden_effect = {
				set_to_lowborn = yes # To remove from dynasty
				move_to_pool = yes
				every_claim = { prev = { remove_claim = prev } }
				every_spouse = { divorce = prev }
				POD_remove_positive_vanilla_relations = yes
				POD_remove_negative_vanilla_relations = yes
			}
			death = { death_reason = death_disappearance }
			POD_add_modifier_and_track_stacks = {
				TYPE = character
				MODIFIER = POD_discard_dragon_touched_interaction_modifier
			}
		}
	}

	auto_accept = yes

	ai_will_do = { base = 0 }
}
