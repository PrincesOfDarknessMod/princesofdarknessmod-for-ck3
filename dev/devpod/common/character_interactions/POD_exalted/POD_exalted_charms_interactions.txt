POD_gale_force_research_technique_interaction = {
	category = interaction_category_exalted
	desc = POD_gale_force_research_technique_interaction_desc
	icon = terrestrial_aspect_air
	target_type = artifact
	target_filter = actor_artifacts

	cooldown = { years = 5 }

	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { has_perk = exalted_gale_force_research_technique_perk }
		scope:actor = {
			any_character_artifact = {
				OR = {
					artifact_slot_type = book
					artifact_slot_type = journal
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			POD_has_enough_essence_for_charm = { AMOUNT = 1 }
		}
		POD_self_targeted_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:target = {
			switch = {
				trigger = rarity
				common = { save_scope_value_as = { name = scaling value = 1 } }
				masterwork = { save_scope_value_as = { name = scaling value = 2 } }
				famed = { save_scope_value_as = { name = scaling value = 3 } }
				illustrious = { save_scope_value_as = { name = scaling value = 4 } }
			}
		}
		scope:actor = {
			POD_essence_cost_effect = { COUNT = 1 }
			add_character_modifier = { 
				modifier = POD_gale_force_research_technique_modifier
				years = 5
			}
		}
	}

	auto_accept = yes

	ai_will_do = {
		base = 0
	}
}
