make_prayer_interaction = {
	category = interaction_category_specialweapons
	desc = make_prayer_interaction_desc
	icon = POD_art_interaction
	
	cooldown = { months = 1 }

	cost = { gold = 20 }

	is_shown = {
		scope:actor = { has_trait = shih }
		scope:actor = { has_perk = celestial_punishment_perk }
	}

	is_valid_showing_failures_only = {
		scope:recipient = { 
			NOT = { has_character_modifier = warning_prayer_modifier }
			NOT = { has_character_modifier = glowing_prayer_modifier }
			NOT = { has_character_modifier = bursing_prayer_modifier }
			NOT = { has_character_modifier = indenfensible_prayer_modifier }
			NOT = { has_character_modifier = explosive_prayer_modifier }
			NOT = { has_character_modifier = burn_out_prayer_modifier }
			NOT = { has_character_modifier = cut_off_prayer_modifier }
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
			POD_is_not_trapped_in_umbra_trigger = { ACTOR = scope:actor }
		}
	}

	on_accept = {
		if = {
			limit = {
				scope:actor = {
					has_perk = divine_prohibition_perk
				}
			}
			scope:recipient = {
				send_interface_toast = {
					title = gotprayer_notification
					add_character_modifier = {
						modifier = cut_off_prayer_modifier
						years = 5
					}
				}
			}
			scope:actor = { POD_chi_decrease_multiple_effect = { COUNT = 5 } }
		}
		else_if = {
			limit = {
				scope:actor = {
					has_perk = heavens_judgment_perk
				}
			}
			scope:recipient = {
				send_interface_toast = {
					title = gotprayer_notification
					add_character_modifier = {
						modifier = burn_out_prayer_modifier
						years = 5
					}
				}
			}
			scope:actor = { POD_chi_decrease_multiple_effect = { COUNT = 4 } }
		}
		else_if = {
			limit = {
				scope:actor = {
					has_perk = heavens_thunder_perk
				}
			}
			scope:recipient = {
				send_interface_toast = {
					title = gotprayer_notification
					add_character_modifier = {
						modifier = explosive_prayer_modifier
						years = 5
					}
				}
			}
			scope:actor = { POD_chi_decrease_multiple_effect = { COUNT = 3 } }
		}
		else_if = {
			limit = {
				scope:actor = {
					has_perk = celestial_fires_perk
				}
			}
			scope:recipient = {
				send_interface_toast = {
					title = gotprayer_notification
					add_character_modifier = {
						modifier = indenfensible_prayer_modifier
						years = 5
					}
				}
			}
			scope:actor = { POD_chi_decrease_multiple_effect = { COUNT = 3 } }
		}
		else_if = {
			limit = {
				scope:actor = {
					has_perk = heavenly_flames_perk
				}
			}
			scope:recipient = {
				send_interface_toast = {
					title = gotprayer_notification
					add_character_modifier = {
						modifier = bursing_prayer_modifier
						years = 5
					}
				}
			}
			scope:actor = { POD_chi_decrease_multiple_effect = { COUNT = 2 } }
		}
		else_if = {
			limit = {
				scope:actor = {
					has_perk = divine_bindings_perk
				}
			}
			scope:recipient = {
				send_interface_toast = {
					title = gotprayer_notification
					add_character_modifier = {
						modifier = glowing_prayer_modifier
						years = 5
					}
				}
			}
			scope:actor = { POD_chi_decrease_multiple_effect = { COUNT = 2 } }
		}
		else_if = {
			limit = {
				scope:actor = {
					has_perk = celestial_punishment_perk
				}
			}
			scope:recipient = {
				send_interface_toast = {
					title = gotprayer_notification
					add_character_modifier = {
						modifier = warning_prayer_modifier
						years = 5
					}
				}
			}
			scope:actor = { POD_chi_decrease_effect = yes }
		}
		
		if = {
			limit = { NOT = { scope:actor = scope:recipient } }
			scope:recipient = {
				add_opinion = {
					target = scope:actor
					modifier = gift_opinion
					opinion = 20
				}
			}
		}
	}

	auto_accept = yes

	ai_potential = {
		# The basic requirements
		is_imprisoned = no 
		is_landed = yes 
		
		number_of_knights > 0
		primary_title.tier > tier_barony
		
		short_term_gold >= 50
	}
	ai_targets = {
		ai_recipients = peer_vassals
		ai_recipients = courtiers
		max = 10
	}
	ai_frequency = 36
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 100
			scope:recipient = {
				is_knight_of = scope:actor
			}
		}
	}
}
