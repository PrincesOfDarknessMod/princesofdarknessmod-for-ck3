POD_choose_sire_interaction = {
	category = interaction_category_friendly
	common_interaction = yes
	use_diplomatic_range = no
	icon = icon_dynasty

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			POD_is_vampire_trigger = yes
			has_character_flag = chargen_choosing_sire_flag
		}
		scope:recipient = {
			POD_is_vampire_trigger = yes
			NOT = {
				is_vampgeneration16_trigger = yes
			}
		}
	}
	auto_accept = yes

	desc = POD_choose_sire_interaction_desc

	on_accept = {
		custom_tooltip = POD_choose_parent_interaction.tt
		scope:recipient = { save_scope_as = vampiresire }	
		scope:actor = { save_scope_as = created_vampirechilde }
		scope:actor = { trigger_event = POD_chargen.600 }
	}
}

POD_choose_parent_interaction = {
	category = interaction_category_friendly
	common_interaction = yes
	use_diplomatic_range = no
	icon = icon_dynasty

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			OR = {
				has_character_flag = chargen_choosing_sire_flag
				AND = {
					has_perk = heart_of_reincarnation_perk
					has_variable = spirit_location
					OR = {
						var:spirit_location = flag:hades
						var:spirit_location = flag:gates_of_heaven
						var:spirit_location = flag:shadow_realm
					}
				}
			}
			NOR = {
				has_trait = vampire
				has_trait = kueijin
			}
		}
	}
	
	is_valid_showing_failures_only = {
		trigger_if = {
			limit = {
				scope:actor = { has_trait = mummy }
			}
			scope:recipient = { 
				is_female = yes
				NOT = { has_trait = demon }
				NOT = { has_trait = wraith }
				NOT = { has_trait = statue }
			}
		}
		trigger_else = {
			scope:recipient = { 
				POD_race_pregnancy_trigger = yes 
			}
		}
	}
	
	auto_accept = yes

	desc = POD_choose_parent_interaction_desc

	on_accept = {
		custom_tooltip = POD_choose_parent_interaction.tt
		scope:actor = { remove_character_flag = chargen_choosing_sire_flag }
		
		scope:recipient = { save_scope_as = vampiresire }	
		scope:actor = { save_scope_as = created_vampirechilde }
		scope:actor = { trigger_event = POD_chargen.1610 }
	}
}

POD_charjoin_choose_sire_interaction = {
	category = interaction_category_friendly
	common_interaction = yes
	icon = icon_dynasty

	is_shown = {
		scope:actor = {
			has_character_flag = independent_char_join_clan_flag
		}
		scope:recipient = {
			POD_is_vampire_trigger = yes
			#Catubodua can't Embrace
			NOT = { this = character:791232 }
		}
	}
	
	is_valid_showing_failures_only = {
		custom_description = {
			object = scope:actor
			text = "He_must_be_my_friend"
			scope:recipient = { 
				OR = {
				has_relation_lover = scope:actor
				has_relation_friend = scope:actor
				has_relation_best_friend = scope:actor
				has_relation_soulmate = scope:actor
				}
			}
		}
		scope:actor = {
			NOT = { is_at_war_with = scope:recipient }
		}	
		scope:recipient = {
			is_caitiff_trigger = no
			is_vampgeneration16_trigger = no
		}			
	}
	
	auto_accept = yes

	desc = POD_charjoin_choose_sire_interaction_desc

	on_accept = {
		custom_tooltip = POD_choose_parent_interaction.tt
		scope:recipient = { save_scope_as = vampiresire }
		scope:actor = {
			save_scope_as = created_vampirechilde 
			remove_character_flag = independent_char_join_clan_flag
			if = {
				limit = {
					OR = {
						POD_is_mortal_trigger = yes
						POD_is_ghoul_trigger = yes
						POD_is_revenant_trigger = yes
					}
				}
				trigger_event = POD_charjoin.100
			}
			else_if = {
				limit = {
					POD_is_supehunter_trigger = yes
				}
				trigger_event = POD_charjoin.200
			}
		}
	}
}

POD_charjoin_choose_mentor_interaction = {
	category = interaction_category_friendly
	common_interaction = yes
	icon = icon_dynasty

	is_shown = {
		scope:actor = {
			has_character_flag = independent_char_join_order_flag
		}
		scope:recipient = {
			POD_is_supehunter_trigger = yes
		}
	}
	
	is_valid_showing_failures_only = {
		custom_description = {
			object = scope:actor
			text = "He_must_be_my_friend"
			scope:recipient = { 
				OR = {
				has_relation_lover = scope:actor
				has_relation_friend = scope:actor
				has_relation_best_friend = scope:actor
				has_relation_soulmate = scope:actor
				}
			}
		}
		scope:actor = {
			NOT = { is_at_war_with = scope:recipient }
		}	
		scope:recipient = {
			OR = {
				dynasty = dynasty:13000001
				dynasty = dynasty:13000005
				dynasty = dynasty:13000002
				dynasty = dynasty:13000006
				dynasty = dynasty:13000008
				AND = { 
					dynasty = dynasty:13000003
					scope:actor = { is_female = yes }
				}
			}
		}	
	}
	
	auto_accept = yes

	desc = POD_charjoin_choose_mentor_interaction_desc

	on_accept = {
		custom_tooltip = POD_choose_parent_interaction.tt
		scope:recipient = { save_scope_as = huntermentor }
		scope:actor = {
			save_scope_as = hunterpupil
			remove_character_flag = independent_char_join_order_flag
		}
	
		if = {
			limit = {
				scope:actor = { is_mortal_trigger = yes }
			}
			if = { 
				limit = {
					scope:recipient = { dynasty = dynasty:13000005 }
				}
				scope:actor = { trigger_event = POD_charjoin.300 }
			}
			else = {
				scope:actor = {
					# Divorce branch
					trigger_event = POD_charjoin.310
				}
			}
		}		
		else_if = {
			limit = {
				scope:actor = { POD_is_supehunter_trigger = yes }
				}
				if = { 
					limit = {
						scope:recipient = { dynasty = dynasty:13000005 }
					}
					scope:actor = { trigger_event = POD_charjoin.400 }
				}
				else = {
					scope:actor = {
						# Divorce branch
						trigger_event = POD_charjoin.410
					}
				}
			}
		}
	
}
