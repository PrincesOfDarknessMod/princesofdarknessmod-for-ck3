POD_train_disciplines_interaction = {
	category = interaction_category_vampire
	#common_interaction = yes
	#interface_priority = 89
    desc = POD_train_disciplines_interaction_desc
	icon = icon_culture
	
	ai_frequency = 0
	
	ai_potential = { always = no }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_POD_vamp_blood_trigger = yes  }
		scope:recipient = { has_POD_vamp_blood_trigger = yes }
	    scope:actor = {
				OR = {
					choleric_lifestyle_xp > 999
					choleric_lifestyle_perk_points > 0
					phlegmatic_lifestyle_xp > 999
					phlegmatic_lifestyle_perk_points > 0
					sanguine_lifestyle_xp > 999
					sanguine_lifestyle_perk_points > 0
					melancholy_lifestyle_xp > 999
					melancholy_lifestyle_perk_points > 0
					animal_lifestyle_xp > 999
					animal_lifestyle_perk_points > 0
				}
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = { is_imprisoned = no }
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }	
		}
		POD_cross_umbra_interaction_valid_trigger = yes		
	}
	
	on_accept = {
	    scope:recipient = {
		    save_scope_as = disciplinetrainee
		}
		scope:actor = {
				save_scope_as = disciplinetrainer
			}
		scope:actor = {
				POD_hunger_rouse_check_effect = yes
			}	
	    scope:actor = {
		    trigger_event = POD_disciplinetrain.100
        }	
	}   
}

POD_train_perk_interaction = {
	category = interaction_category_friendly
	desc = POD_train_perk_interaction_desc
	icon = icon_culture

	ai_frequency = 0
	
	ai_potential = { always = no }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { 
			OR = { 
				diplomacy_lifestyle_perk_points >= 1
				martial_lifestyle_perk_points >= 1
				stewardship_lifestyle_perk_points >= 1
				intrigue_lifestyle_perk_points >= 1
				learning_lifestyle_perk_points >= 1
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = { is_imprisoned = no }
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }	
		}
		POD_cross_umbra_interaction_valid_trigger = yes		
	}
	
	send_options_exclusive = no
	
	send_option = {
		is_shown = {
			scope:actor = { diplomacy_lifestyle_perk_points >= 1 }
		}
		flag = diplomacy_lifestyle
		localization = diplomacy_lifestyle_name
	}
	
	send_option = {
		is_shown = {
			scope:actor = { martial_lifestyle_perk_points >= 1 }
		}
		flag = martial_lifestyle
		localization = martial_lifestyle_name
	}
	
	send_option = {
		is_shown = {
			scope:actor = { stewardship_lifestyle_perk_points >= 1 }
		}
		flag = stewardship_lifestyle
		localization = stewardship_lifestyle_name
	}
	
	send_option = {
		is_shown = {
			scope:actor = { intrigue_lifestyle_perk_points >= 1 }
		}
		flag = intrigue_lifestyle
		localization = intrigue_lifestyle_name
	}
	
	send_option = {
		is_shown = {
			scope:actor = { learning_lifestyle_perk_points >= 1 }
		}
		flag = learning_lifestyle
		localization = learning_lifestyle_name
	}
	
	on_accept = {
		scope:actor = {
			stress_impact = {
				base = medium_stress_gain
			}
			if = {
				limit = { has_perk = firebringer_perk }
				add_prestige = major_prestige_gain
				dynasty = { add_dynasty_prestige = major_dynasty_prestige_gain }
			}
		}
		if = {
			limit = { scope:diplomacy_lifestyle = yes }
			scope:actor = { add_diplomacy_lifestyle_perk_points = -1 }
			scope:recipient = { add_diplomacy_lifestyle_perk_points = 1 }
		}
		if = {
			limit = { scope:martial_lifestyle = yes }
			scope:actor = { add_martial_lifestyle_perk_points = -1 }
			scope:recipient = { add_martial_lifestyle_perk_points = 1 }
		}
		if = {
			limit = { scope:stewardship_lifestyle = yes }
			scope:actor = { add_stewardship_lifestyle_perk_points = -1 }
			scope:recipient = { add_stewardship_lifestyle_perk_points = 1 }
		}
		if = {
			limit = { scope:intrigue_lifestyle = yes }
			scope:actor = { add_intrigue_lifestyle_perk_points = -1 }
			scope:recipient = { add_intrigue_lifestyle_perk_points = 1 }
		}
		if = {
			limit = { scope:learning_lifestyle = yes }
			scope:actor = { add_learning_lifestyle_perk_points = -1 }
			scope:recipient = { add_learning_lifestyle_perk_points = 1 }
		}
	    scope:recipient = { 
			add_opinion = {
				target = scope:actor
				modifier = mentored_me_opinion
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
