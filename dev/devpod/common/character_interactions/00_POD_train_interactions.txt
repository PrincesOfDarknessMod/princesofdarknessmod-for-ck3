POD_train_disciplines_interaction = {
	category = interaction_category_vampire
	#common_interaction = yes
	#interface_priority = 89
	desc = POD_train_disciplines_interaction_desc
	icon = icon_culture
	
	ai_frequency = 0

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_POD_vamp_blood_trigger = yes  }
		scope:recipient = { has_POD_vamp_blood_trigger = yes }
		scope:actor = {
			OR = {
				choleric_lifestyle_perk_half_points > 0
				phlegmatic_lifestyle_perk_half_points > 0
				sanguine_lifestyle_perk_half_points > 0
				melancholy_lifestyle_perk_half_points > 0
				animal_lifestyle_perk_half_points > 0
				has_perk = cycle_of_the_midgard_serpent_perk
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = { is_imprisoned = no }
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}
	
	on_accept = {
		scope:recipient = {
			save_scope_as = disciplinetrainee
		}
		scope:actor = {
				save_scope_as = disciplinetrainer
			}
		scope:actor = {
				POD_hunger_rouse_check_effect = yes
			}
		scope:actor = {
			trigger_event = POD_disciplinetrain.100
		}
	}
}
