
POD_set_fetter_character_interaction = {
	category = interaction_category_wraith
	desc = POD_set_fetter_interaction_desc
	options_heading = POD_set_fetter_interaction_options_heading
	icon = prison
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_trait = wraith } 
		scope:actor = { 
			has_variable = unset_fetter_point
			var:unset_fetter_point > 0
		} 
		scope:recipient = { NOT = { has_variable = fetter_value } }
	}
	
	is_valid_showing_failures_only = {
		scope:actor = { POD_can_interact_with_skinlands_trigger = { RECIPIENT = scope:recipient } }
		scope:recipient = {
			trigger_if = {
				limit = { NOT = { is_any_family_relation_or_spouse_trigger = { CHARACTER = scope:actor } } }
				scope:actor = { has_perk = souls_interwoven_perk }
			}
			trigger_if = {
				limit = { has_trait = wraith }
				scope:actor = { has_perk = lifeline_perk }
			}
		}
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				has_variable = unset_fetter_point
				var:unset_fetter_point >= 1 
			}
		}
		flag = POD_num1
		localization = POD_num1
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				has_variable = unset_fetter_point
				var:unset_fetter_point >= 2 
			}
		}
		flag = POD_num2
		localization = POD_num2
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				has_variable = unset_fetter_point
				var:unset_fetter_point >= 3 
			}
		}
		flag = POD_num3
		localization = POD_num3
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				has_variable = unset_fetter_point
				var:unset_fetter_point >= 4 
			}
		}
		flag = POD_num4
		localization = POD_num4
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				has_variable = unset_fetter_point
				var:unset_fetter_point >= 5 
			}
		}
		flag = POD_num5
		localization = POD_num5
	}
	
	on_accept = {
		scope:actor = {
			custom_tooltip = POD_set_fetter_interaction_desc
			custom_tooltip = POD_set_fetter_character_interaction_tooltip
			add_to_variable_list = { name = owned_fetter_list target = scope:recipient }
			if = {
				limit = { scope:POD_num1 = yes }
				change_variable = { name = unset_fetter_point subtract = 1 }
				scope:recipient = { set_variable = { name = fetter_value value = 1 } }
			}
			else_if = {
				limit = { scope:POD_num2 = yes }
				change_variable = { name = unset_fetter_point subtract = 2 }
				scope:recipient = { set_variable = { name = fetter_value value = 2 } }
			}
			else_if = {
				limit = { scope:POD_num3 = yes }
				change_variable = { name = unset_fetter_point subtract = 3 }
				scope:recipient = { set_variable = { name = fetter_value value = 3 } }
			}
			else_if = {
				limit = { scope:POD_num4 = yes }
				change_variable = { name = unset_fetter_point subtract = 4 }
				scope:recipient = { set_variable = { name = fetter_value value = 4 } }
			}
			else_if = {
				limit = { scope:POD_num5 = yes }
				change_variable = { name = unset_fetter_point subtract = 5 }
				scope:recipient = { set_variable = { name = fetter_value value = 5 } }
			}
		}
		scope:recipient = {
			set_variable = {
				name = fetter_owner
				value = scope:actor
			}
		}
	}
	
	ai_accept = {
		base = -50
		
		opinion_modifier = { # Opinion Factor
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 1
			desc = AI_OPINION_REASON
		}
	}
	
	ai_frequency = 0
	
	ai_will_do = {
		base = 0
	}  
}

POD_resolve_fetter_character_interaction = {
	category = interaction_category_wraith
	desc = POD_resolve_fetter_interaction_desc
	icon = prison
	
	cost = { piety = 1000 }
	cooldown = { years = 3 }
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_trait = wraith }
		scope:recipient = { is_fetter_of_actor_trigger = yes }
	}
	
	on_accept = {
		scope:actor = {
			duel = {
				skill = diplomacy
				value = 30
			
				50 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					
					custom_tooltip = POD_resolve_fetter_interaction_tooltip
					POD_resolve_fetter_effect = { FETTER = scope:recipient }
				}
				45 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier
					}
					
					add_stress = massive_stress_impact_gain
				}
				5 = {
					custom_tooltip = fetter_destroyed_tooltip
					send_interface_toast = {
						title = fetter_destroyed_notification

						left_icon = this
						right_icon = scope:recipient
							
						POD_destroy_fetter_effect = { FETTER = scope:recipient }
					}
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_frequency = 0
	
	ai_will_do = {
		base = 0
	}  
}
