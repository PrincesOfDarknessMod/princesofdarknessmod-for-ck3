	category = interaction_category_vampire

	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { has_perk = mirros_visage_perk }
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
		    NOT = { has_character_modifier = bloodhunger4_modifier }
			NOT = { has_character_modifier = bloodhunger5_modifier }
		    num_of_relation_mirror_duplicate_servant = 0
		}
	}

	on_accept = {
	   custom_tooltip = POD_mirror_duplicate_interaction_desc
	   scope:actor = {
	       trigger_event = POD_mirror_duplicate.001
	   }
    }
	
	auto_accept = yes
	
	cooldown = { years = 20 }
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no
		is_landed = yes
		
		piety > 800
	}
	ai_targets = {
		ai_recipients = self
	}
	ai_frequency = 360
	ai_will_do = {
		base = 5
		
		modifier = {
		    add = 10
			is_at_war = yes
		}
	} 
}
