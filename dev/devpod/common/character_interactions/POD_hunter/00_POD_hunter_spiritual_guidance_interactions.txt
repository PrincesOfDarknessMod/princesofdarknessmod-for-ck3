POD_spiritual_guidance_interaction = {
	category = interaction_category_friendly
	#common_interaction = yes
	#interface_priority = 89
	desc = POD_spiritual_guidance_interaction_desc
	
	ai_frequency = 0

	is_shown = {
		NOT = { scope:actor = scope:recipient }

			POD_is_supehunter_trigger = yes
			has_faith = faith:shadowinquisition
			has_perk = oculi_trust_perk
		}
		scope:recipient = {
			overlord = scope:actor
			OR = {
				has_trait = mortal
				has_trait = supehunter
			}
			has_faith = faith:shadowinquisition
			is_imprisoned = no
		}
		POD_interaction_is_shown = yes
	}
	
	is_valid_showing_failures_only = {
		custom_description = {
			object = scope:actor
			text = "I_have_virtues_i_can_teach"
			scope:actor = {
				OR = {
					AND = {
						scope:actor = { has_trait = chaste }
						NOT = { scope:recipient = { has_trait = chaste } }
					}
			 		AND = {
						scope:actor = { has_trait = temperate }
						NOT = { scope:recipient = { has_trait = temperate } }
					}
					AND = {
						scope:actor = { has_trait = generous }
			 			NOT = { scope:recipient = { has_trait = generous } }
					}
					AND = {
						scope:actor = { has_trait = forgiving }
			 			NOT = { scope:recipient = { has_trait = forgiving } }
					}
			 		AND = {
						scope:actor = { has_trait = calm }
			 			NOT = { scope:recipient = { has_trait = calm } }
			 		}
			 		AND = {
						scope:actor = { has_trait = humble }
						NOT = { scope:recipient = { has_trait = humble } }
			 		}
					AND = {
			 			scope:actor = { has_trait = diligent }
						NOT = { scope:recipient = { has_trait = diligent } }
					}
				}
			}
		}
		custom_description = {
			object = scope:recipient
			text = "I_don't_have_the_same_sins"
			scope:recipient = {
		 		OR = {
					AND = {
		 				scope:recipient = { has_trait = lustful }
						NOT = { scope:actor = { has_trait = lustful } }
					}
					AND = {
						scope:recipient = { has_trait = vengeful }
		 				NOT = { scope:actor = { has_trait = vengeful } }
					}
		 			AND = {
						scope:recipient = { has_trait = greedy }
			 			NOT = { scope:actor = { has_trait = greedy } }
					}
					AND = {
						scope:recipient = { has_trait = gluttonous }
			 			NOT = { scope:actor = { has_trait = gluttonous } }
			 		}
					AND = {
						scope:recipient = { has_trait = arrogant }
						NOT = { scope:actor = { has_trait = arrogant } }
					}
					AND = {
						scope:recipient = { has_trait = lazy }
						NOT = { scope:actor = { has_trait = lazy } }
					}
					AND = {
						scope:recipient = { has_trait = wrathful }
						NOT = { scope:actor = { has_trait = wrathful } }
					}
		 		}
		 	}
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}
	
	cooldown = { years = 5 }
	cost = { piety = 150 }
	
	on_accept = {
		scope:recipient = {
			save_scope_as = student
		}

		scope:actor = {
				save_scope_as = teacher
			}
		scope:actor = {
			trigger_event = {
				id = POD_spiritual_guidance.001
				days = 1
			}
		}
	}
	
	ai_will_do = {
		base = 0
	}
}
