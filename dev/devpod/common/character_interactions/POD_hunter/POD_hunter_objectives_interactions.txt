sultan_ask_for_gold_interaction = {
	icon = icon_gold
	category = interaction_category_religion

	desc = sultan_ask_for_gold_interaction_desc

	cooldown_against_recipient = { years = 5 }

	is_shown = {
		scope:actor = {
			is_playable_character = yes
			faith = faith:ikhwan_al_safa
			has_trait = supehunter
		}
		scope:recipient = scope:actor
		is_target_in_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:ikhwan_sultans_patronage_objective_decision
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			prestige >= major_prestige_value
		}
	}

	cost = {
		prestige = major_prestige_value
	}

	on_accept = {
		add_short_term_gold = { 50 500 }
	}

	ai_targets = {
		ai_recipients = self
	}
	ai_target_quick_trigger = {
		adult = yes
	}
	ai_frequency_by_tier = {
		barony = 0
		county = 36
		duchy = 24
		kingdom = 6
		empire = 6
		hegemony = 6
	}
	ai_will_do = {
		base = 0
		modifier = {
			highest_held_title_tier < tier_county
			scope:recipient.treasury_or_gold < 3000
			add = -50
		}
		modifier = {
			highest_held_title_tier = tier_duchy
			top_liege != this
			add = -25
		}
		modifier = {
			add = 50
			scope:actor = {
				is_at_war = yes
				gold < medium_gold_value
			}
		}
		modifier = {
			add = 50
			scope:actor = {
				gold < 0
			}
		}
		modifier = {
			add = 50
			scope:actor = {
				gold < massive_gold_value_negative
			}
		}
		modifier = {
			add = 25
			scope:actor = {
				piety > massive_piety_value
				NOR = {
					has_character_modifier = vow_of_poverty_modifier
					has_trait = generous
				}
			}
		}
		modifier = { # Or when greedy
			add = scope:actor.ai_greed
			scope:actor = {
				ai_greed > 0
			}
		}
	}
}

POD_seeking_out_evil_interaction = {
	category = interaction_category_hostile
	desc = POD_seeking_out_evil_interaction_desc
	icon = POD_spiritual_guidance_interaction

	cooldown = { months = 3 }
	
	is_shown = {
		scope:actor = {
			faith = faith:ikhwan_al_safa
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:ikhwan_infernal_complications_objective_decision
			}
		}
		scope:recipient = { POD_splat_has_masquerade_trigger = yes }
	}

	is_valid_showing_failures_only = {
		scope:recipient = { POD_is_forever_infamous_trigger = no }
		custom_tooltip = {
			text = POD_cover_masquerade_interaction_cap_not
			scope:recipient = { masquerade_exposure_level < 5 }
		}
		POD_hostile_interaction_is_shown = yes 
	}

	cost = { piety = 200 }
	
	on_accept = {
		scope:recipient = {
			POD_masquerade_exposure_increase_multiple_effect = { AMOUNT = 1 }

			# If secret infernalist, reveal it to all
			if = {
				limit = { any_secret = { type = secret_witch } }
				POD_give_and_expose_witch_trait_effect = yes
			}
		}
	}
	
	auto_accept = yes

	ai_will_do = {
		base = 0
	}
}

POD_khans_servant_interaction = {
	category = interaction_category_holyarts
	icon = POD_spiritual_guidance_interaction
	interface_priority = 108
	
	ai_frequency = 62
	ai_targets = {
		ai_recipients = war_enemies
	}
	
	is_shown = {
		scope:actor != scope:recipient
		scope:actor = {
			has_perk = endowments_baghatur_5_perk
		}
		POD_hostile_interaction_is_shown = yes
	}
	
	is_valid = {
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			NOT = { has_character_modifier = POD_baghatur_harassed_by_raiders_modifier }
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}
	
	cost = { prestige = 350 }

	desc = POD_khans_servant_interaction_desc
	
	on_accept = {
		scope:actor = {
			send_interface_toast = {
				title = POD_khans_servant_interaction_toast
				left_icon = scope:actor
				right_icon = scope:recipient
				scope:recipient = {
					send_interface_toast = {
						title = POD_khans_servant_interaction_toast
						left_icon = scope:actor
						right_icon = scope:recipient
						add_character_modifier = {
							modifier = POD_baghatur_harassed_by_raiders_modifier
							years = 5
						}
					}
				}
			}
		}
		scope:recipient = {
			add_opinion = {
				target = scope:actor
				modifier = hurt_opinion
				opinion = -15
			}
			progress_towards_rival_effect = {
				CHARACTER = scope:actor
				REASON = rival_raid_estate
				OPINION = 0
			}
		}
	}

	auto_accept = yes
	
	ai_will_do = {
		base = 0
		modifier = {
			add = 25
			scope:actor = { is_at_war_with = scope:recipient }
		}
		modifier = {
			add = 50
			scope:actor = { is_at_war_with = scope:recipient }
			any_character_war = {
				is_war_leader = scope:recipient
			}
			scope:recipient = { martial >= 15 }
		}
	}
}
