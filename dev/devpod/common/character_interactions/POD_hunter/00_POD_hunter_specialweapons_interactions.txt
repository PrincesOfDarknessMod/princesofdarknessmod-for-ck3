holyweapon_interaction = {
	category = interaction_category_friendly
	#interface_priority = 87
	desc = holyweapon_interaction_desc
	
	ai_frequency = 0

	cost = {
		gold = 100
		piety = 100
	}

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			has_trait = red_forge
		}
		scope:recipient = {
			has_trait = supehunter
		}
		POD_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:actor.gold >= 100
		custom_tooltip = {
			text = has_holyweapon
			scope:recipient = {
				NOT = {	has_character_modifier = endo_holyweapon_modifier }
			}
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}
	
	cooldown = { months = 1 }

	on_accept = {
		scope:actor = {
			send_interface_toast = {
				title = holyweapon_interaction_toast
				left_icon = scope:actor
				right_icon = scope:recipient
				scope:recipient = {
					add_character_modifier = endo_holyweapon_modifier
				}
			}
		}
	}

	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

firebombs_interaction = {
	category = interaction_category_friendly
	#interface_priority = 60
	desc = firebombs_interaction_desc

	greeting = positive
	notification_text = FIREBOMBS_PROPOSAL

	answer_accept_key = FIREBOMBS_ACCEPT
	answer_reject_key = FIREBOMBS_REJECT

	is_shown = {
		NOT = {
			scope:recipient = scope:actor
		}
		scope:actor = { has_trait = supehunter }
		scope:recipient = { has_trait = supehunter }
		scope:actor.dynasty = {
			OR = {
			 	has_dynasty_modifier = meng_huo_you_supply_modifier
			 	has_dynasty_modifier = masters_of_fire
			}
		}
		POD_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:actor.gold >= 25
		custom_description = {
			text = has_fireweapons
			scope:recipient = {
				NOT = { has_character_modifier = firebombs_modifier }
				NOT = { has_character_modifier = firespewer_modifier }
			}
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}
	
	cost = { gold = 25 }

	on_accept = {
		scope:actor = {
			send_interface_message = {
				type = event_generic_neutral
				title = gavefirebombs_notification
				right_icon = scope:recipient
				scope:recipient = {
					send_interface_message = {
						type = event_generic_neutral
						title = gotfirebombs_notification
						right_icon = scope:actor
						add_character_modifier = firebombs_modifier
					}
				}
			}
		}
	}
	
	ai_accept = {
		base = 100
	}
	
	ai_min_reply_days = 1
	ai_max_reply_days = 1
	
	ai_will_do = {
		base = 0
	}
}
