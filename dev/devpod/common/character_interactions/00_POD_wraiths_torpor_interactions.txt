POD_awake_spirit_interaction = {
	category = interaction_category_wraith
	icon = POD_wraith_interaction
	desc = POD_awake_spirit_interaction_desc
	
	cooldown = { months = 10 }	
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_POD_kj_blood_trigger = yes }
		scope:actor = { has_perk = blood_awakening_perk }
		scope:recipient = { NOT = { has_character_modifier = caul_modifier } }
		scope:recipient = { has_trait = torpor }
		scope:recipient = { POD_is_wraith_trigger = yes }
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
			custom_tooltip = {
				text = "i_am_empty_of_chi"
				NAND = { 
					has_character_modifier = yinchi0_modifier
					has_character_modifier = yangchi0_modifier
				}
			}
		}
		scope:recipient = { has_relation_necromancer = scope:actor }
		scope:recipient = {
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}
	
	on_accept = {
		scope:actor = { POD_yangchi_cost_effect = { COUNT = 1 }	 }
		scope:recipient = {
			duel = {
				skill = intrigue
				value = scope:recipient.intrigue
			
				50 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					scope:recipient = { remove_trait = torpor }
				}
				50 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier
					}
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_remove_caul_interaction = {
	category = interaction_category_wraith
	icon = POD_wraith_interaction
	desc = POD_remove_caul_interaction_desc
	
	cost = { gold = 50 }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { POD_is_wraith_trigger = yes }
		scope:recipient = { has_character_modifier = caul_modifier }
		scope:recipient = { has_trait = torpor }
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
			trigger_if = {
				limit = {
					NOT = { has_trait = wraith }
					NOT = { has_trait = demon }
				}
				has_access_to_underworld_trigger = yes
			}
		}
		scope:recipient = {
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		# POD_cross_umbra_interaction_valid_trigger = yes
	}
	
	on_accept = {
		scope:recipient = {
			remove_character_modifier = caul_modifier
			remove_trait = torpor	
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
