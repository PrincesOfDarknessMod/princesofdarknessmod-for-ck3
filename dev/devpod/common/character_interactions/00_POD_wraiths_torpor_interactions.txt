POD_awake_spirit_interaction = {
	category = interaction_category_wraith
	icon = POD_wraith_interaction
	desc = POD_awake_spirit_interaction_desc
	icon = POD_art_interaction
	
	cooldown = { months = 10 }	
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { 
			OR = {
				has_trait = kueijin 
				has_trait = dhampyr
			}
		}
		scope:actor = { has_perk = blood_awakening_perk }
		scope:recipient = { NOT = { has_character_modifier = enfant_modifier } }
		scope:recipient = { has_trait = torpor }
		scope:recipient = { POD_is_wraith_trigger = yes }
	}
	
	is_valid_showing_failures_only = {
		scope:recipient = { has_relation_necromancer = scope:actor }
	}
	
	on_accept = {
		scope:actor = { POD_yangchi_decrease_effect = { COUNT = 1 }	 }
		scope:recipient = {
			duel = {
				skill = intrigue
				value = scope:recipient.intrigue
			
				50 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					scope:recipient = { remove_trait = torpor }
				}
				50 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier
					}
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
