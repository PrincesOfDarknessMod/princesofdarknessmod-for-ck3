
POD_awaken_true_jade_interaction = {
	category = interaction_category_fae
	icon = POD_art_interaction
	desc = POD_awaken_true_jade_interaction_desc
	options_heading = POD_awaken_true_jade_interaction_options_heading
	
	cost = { gold = 50 }
	
	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { has_trait = hsien }
		scope:actor = { 
			OR = { 
				has_perk = tieh_tan_awaken_true_jade_perk 
				has_perk = lin_tan_awaken_true_jade_perk 
				has_perk = shui_tan_awaken_true_jade_perk 
				has_perk = hou_tan_awaken_true_jade_perk 
				has_perk = di_tan_awaken_true_jade_perk 
			}
		}
	}

	is_valid_showing_failures_only= {
		scope:actor = { 
			custom_tooltip = {
				text = "i_am_empty_of_chi"
				NAND = { 
					has_character_modifier = yinchi0_modifier
					has_character_modifier = yangchi0_modifier
				}
			}
		}
	}
	
	send_option = {
		is_valid = { scope:actor = { has_perk = lin_tan_awaken_true_jade_perk } }
		flag = dragon_jade
		localization = POD_awaken_true_jade_interaction_option_dragon_jade
	}
	
	send_option = {
		is_valid = { scope:actor = { has_perk = hou_tan_awaken_true_jade_perk } }
		flag = feng_jade
		localization = POD_awaken_true_jade_interaction_option_feng_jade
	}
	
	send_option = {
		is_valid = { scope:actor = { has_perk = tieh_tan_awaken_true_jade_perk } }
		flag = tiger_jade
		localization = POD_awaken_true_jade_interaction_option_tiger_jade
	}
	
	send_option = {
		is_valid = { scope:actor = { has_perk = shui_tan_awaken_true_jade_perk } }
		flag = tortoise_jade
		localization = POD_awaken_true_jade_interaction_option_tortoise_jade
	}
	
	send_option = {
		is_valid = { scope:actor = { has_perk = di_tan_awaken_true_jade_perk } }
		flag = mandarin_jade
		localization = POD_awaken_true_jade_interaction_option_mandarin_jade
	}

	on_accept = {
		scope:actor = { 
			POD_willpower_rouse_check_effect = yes 
			POD_chi_cost_effect = { COUNT = 1 }
			duel = {
				desc = bunk.duel.desc
				skill = stewardship
				value = low_skill_rating
				50 = {
					desc = board_games.0001.tt.success
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					scope:recipient = {
						if = {
							limit = { scope:dragon_jade = yes }
							create_artifact_dragon_jade_effect = { OWNER = scope:actor }
						}
						if = {
							limit = { scope:feng_jade = yes }
							create_artifact_feng_jade_effect = { OWNER = scope:actor }
						}
						if = {
							limit = { scope:tiger_jade = yes }
							create_artifact_tiger_jade_effect = { OWNER = scope:actor }
						}
						if = {
							limit = { scope:tortoise_jade = yes }
							create_artifact_tortoise_jade_effect = { OWNER = scope:actor }
						}
						if = {
							limit = { scope:mandarin_jade = yes }
							create_artifact_mandarin_jade_effect = { OWNER = scope:actor }
						}
					}
				}	
				50 = {
					desc = board_games.0001.tt.failure
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier 
					}
					modifier = { OR = { NOT = { has_trait = siochain } global_banality = 0 } factor = global_banality }
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
