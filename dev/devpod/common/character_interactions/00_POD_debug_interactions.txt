POD_debug_force_into_prison_interaction = {
	category = interaction_debug_main
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	common_interaction = yes
	interface_priority = 1500
	
	icon = pod_icon
	desc = POD_debug_force_into_prison_interaction_desc
	
	is_shown = {
		debug_only = yes
		NOT = { scope:actor = scope:recipient }
		OR = {
			scope:recipient = { is_imprisoned = no }
			AND = {
				scope:actor = { is_imprisoned = no }
				scope:recipient = { is_ruler = yes }
			}
		}
	}
	
	send_option = {
		is_valid = { scope:recipient = { is_imprisoned = no } }
		flag = imprison_recipient
		localization = POD_debug_become_laamp_interaction_imprison_recipient
	}

	send_option = {
		is_valid = {
			scope:actor = { is_imprisoned = no }
			scope:recipient = { is_ruler = yes }
		}
		flag = imprison_actor
		localization = POD_debug_become_laamp_interaction_imprison_actor
	}
	
	on_accept = {
		switch = {
			trigger = yes
			scope:imprison_recipient = {
				scope:recipient = { add_character_flag = { flag = lost_duel days = 1 } }
				scope:actor = {
					imprison = {
						target = scope:recipient
						type = house_arrest
					}
					custom_tooltip = POD_debug_force_into_prison_interaction_desc
				}
			}
			scope:imprison_actor = {
				scope:actor = { add_character_flag = { flag = lost_duel days = 1 } }
				scope:recipient = {
					imprison = {
						target = scope:actor
						type = house_arrest
					}
					custom_tooltip = POD_debug_force_into_prison_interaction_desc
				}
			}
		}
	}
	
	auto_accept = yes
}

POD_debug_edit_clothing_colors_interaction = {
	category = interaction_debug_main
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	
	icon = pod_icon
	desc = POD_debug_edit_clothing_colors_interaction_desc
	
	is_shown = {
		debug_only = yes
	}

	send_option = {
		flag = POD_colors_personal
		localization = POD_debug_edit_clothing_colors_personal
	}

	send_option = {
		is_shown = { scope:recipient = { POD_has_coterie_trigger = yes } }
		flag = POD_colors_coterie
		localization = POD_debug_edit_clothing_colors_coterie
	}

	send_option = {
		is_shown = { scope:recipient = { exists = house } }
		flag = POD_colors_house
		localization = POD_debug_edit_clothing_colors_house
	}

	on_accept = {
		scope:recipient = { save_scope_as = target }
		scope:target = {
			switch = {
				trigger = yes
				scope:POD_colors_personal = {
					# Variables are for the event widget, scopes are for the color display
					set_variable = {
						name  = color_root
						value = this
					}
					save_scope_as = color_root
				}
				scope:POD_colors_coterie = {
					set_variable = {
						name  = color_root
						value = var:coterie
					}
					var:coterie = { save_scope_as = color_root }
					random_relation = {
						type = coterie_member
						save_scope_as = secondary_target
					}
				}
				scope:POD_colors_house = {
					set_variable = {
						name  = color_root
						value = house
					}
					house = { save_scope_as = color_root }
					house = {
						random_house_member = {
							limit = { NOT = { this = scope:target } }
							save_scope_as = secondary_target
						}
					}
				}
			}
		}
		custom_tooltip = POD_current_clothing_colors
		scope:actor = { trigger_event = POD_style.1 }
	}
	
	auto_accept = yes
}
