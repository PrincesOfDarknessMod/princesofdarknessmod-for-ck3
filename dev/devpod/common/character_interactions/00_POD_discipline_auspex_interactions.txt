POD_discern_the_aura_interaction = {
	category = interaction_category_vampire
	icon = POD_auspex_interaction
	desc = POD_discern_the_aura_interaction_desc
	
	cooldown = { months = 3 }

	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { has_perk = discern_the_aura_perk }
	}

	on_accept = {
		scope:actor = {
			custom_tooltip = POD_discern_the_aura_interaction_effect
			remove_short_term_gold = medium_gold_value
			add_prestige = medium_prestige_loss
			set_variable = { name = character_creator_type value = flag:find_mortal }
			set_variable = { name = character_creator_points value = 30 }
		}
	}
	
	auto_accept = yes
	ai_frequency = 0
	use_diplomatic_range = no
	
	ai_will_do = {
		base = 0
	}
}

POD_the_mind_revealed_interaction = {
	category = interaction_category_vampire
	icon = POD_auspex_interaction
	desc = POD_the_mind_revealed_interaction_desc
	
	cooldown_against_recipient = { years = 1 } 
	cost = { piety = minor_piety_value }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_perk = the_mind_revealed_perk }
	}

	is_valid_showing_failures_only = {
		scope:actor = {
		   NOT = { has_character_modifier = bloodhunger5_modifier }
		}
		scope:recipient = { 
			NOT = { has_trait = statue }
			NOT = { has_trait = torpor }
			NOT = { has_trait = wight }
			NOT = { has_trait = barghest }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }	
			POD_is_not_clone_trigger = yes
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}
	
	on_accept = {
		scope:actor = { 
			POD_willpower_rouse_check_effect = yes
			POD_hunger_increase_effect = yes
			duel = {
				skill = learning
				target = scope:recipient
				75 = {
					desc = POD_kueijin_spy.duel.success
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					send_interface_toast = {
						title = POD_kueijin_spy.duel.success
						left_icon = scope:actor
						right_icon = scope:recipient
						
						custom_tooltip = POD_kueijin_spy.duel.tt.evaluate
						custom_tooltip = POD_kueijin_spy.duel.tt.scrutinize
					}
					hidden_effect = {
						scope:recipient = {
							every_scheme = {
								expose_scheme = yes
							}
						}
					}
				}
				25 = {
					desc = POD_kueijin_spy.duel.failure
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier 
					}
					send_interface_toast = {
						title = POD_kueijin_spy.duel.failure
						left_icon = scope:actor
						right_icon = scope:recipient
						
						custom_tooltip = POD_kueijin_spy.duel.tt.attacker_lost
						add_stress = medium_stress_gain
					}
				}
			}
		}
		scope:recipient = { 
			add_opinion = {
				target = scope:actor
				modifier = hurt_opinion
				opinion = -30
			}
			progress_towards_rival_effect = {
				CHARACTER = scope:actor
				REASON = rival_cast_spell
				OPINION = 0
			}
		}
	}

	auto_accept = yes

	ai_potential = {
		# The basic requirements
		is_imprisoned = no 
		is_landed = yes 
	}
	ai_targets = {
		ai_recipients = liege
	}
	ai_targets = {
		ai_recipients = neighboring_rulers
		ai_recipients = peer_vassals
		max = 10
	}
	ai_frequency = 48
	ai_will_do = {
		base = 0
		
		opinion_modifier = {
			opinion_target = scope:recipient
			multiplier = -0.25
		}
		modifier = {
		    add = 50
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
		modifier = {
			factor = 0
			scope:recipient = {
				is_ai = no
			}
		}
	}
}

POD_sexuality_metamorph_interaction = {
	category = interaction_category_vampire
	icon = POD_auspex_interaction
	desc = POD_sexuality_metamorph_interaction_desc

	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { has_perk = personality_metamorph_perk }
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			NOT = { has_character_modifier = bloodhunger5_modifier }
		}	
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { NOT = { has_sexuality = heterosexual } }
		}
		flag = heterosexual
		localization = sexuality_he
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { NOT = { has_sexuality = homosexual } }
		}
		flag = homosexual
		localization = sexuality_ho
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { NOT = { has_sexuality = bisexual } }
		}
		flag = bisexual
		localization = sexuality_bi
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { NOT = { has_sexuality = asexual } }
		}
		flag = asexual
		localization = sexuality_as
	}

	on_accept = {
		scope:actor = {
			POD_willpower_rouse_check_effect = yes
			POD_hunger_rouse_check_effect = yes
		}
		if = {
			limit = { scope:heterosexual = yes }
			scope:recipient = { set_sexuality = heterosexual }
		}
		if = {
			limit = { scope:homosexual = yes }
			scope:recipient = { set_sexuality = homosexual }
		}
		if = {
			limit = { scope:bisexual = yes }
			scope:recipient = { set_sexuality = bisexual }
		}
		if = {
			limit = { scope:asexual = yes }
			scope:recipient = { set_sexuality = asexual }
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
