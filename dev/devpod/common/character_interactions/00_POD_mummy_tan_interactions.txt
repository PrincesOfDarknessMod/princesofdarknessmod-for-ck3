POD_tread_the_dragon_tracks_interaction = {
	category = interaction_category_mummy
	icon = POD_art_interaction
	desc = POD_dragon_chariot_interaction_desc
	
	cooldown = { months = 1 }

	is_shown = {
		scope:actor = { has_trait = mummy }
		scope:actor = { has_perk = tread_the_dragon_tracks_perk }
		POD_teleport_interaction_army_is_shown_trigger = yes
	}
	
	is_valid = {
		scope:actor.location = {
			OR = {
				custom_tooltip = {
					text = has_dragon_nest
					has_building_with_flag = dragon_nest
				}
				has_province_modifier = temporary_dragon_nest_modifier
				has_special_building_slot = yes # Natural Dragon Nest?
			}
		}
		scope:recipient.location = {
			OR = {
				custom_tooltip = {
					text = has_dragon_nest
					has_building_with_flag = dragon_nest
				}
				has_province_modifier = temporary_dragon_nest_modifier
				has_special_building_slot = yes # Natural Dragon Nest?
			}
		}
		POD_teleport_interaction_army_is_valid_trigger = yes
	}

	is_valid_showing_failures_only = {
	}

	on_accept = {
		scope:actor = { 		
			POD_chi_cost_effect = { COUNT = 3 }
			POD_willpower_rouse_check_effect = yes
			POD_teleport_army_interaction_effect = yes
		}
	}

	ai_frequency = 0
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
