POD_spawn_craft_tutor_interaction = {
	category = interaction_category_uncategorized
	hidden = yes
	icon = POD_wraith_interaction
	desc = POD_spawn_craft_tutor_interaction_desc
	
	is_shown = {
		scope:actor = {
			NOT = { 
				any_in_list = {
					variable = craft_creator
					has_character_flag = craft_creator_tutor
				}
			}
		}
	}
	
	is_valid = {
		
	}

	is_valid_showing_failures_only = {
	}

	on_accept = {
		scope:recipient = {
			add_gold = {
				value = 0
				add = water_cost
				add = fuel_cost
				add = herb_cost
			}
			create_character = {
				employer = scope:actor
				template = celebration_artisan_character
				after_creation = {
					POD_become_wraith_effect = yes
					add_trait = sorcerer
					add_perk = herbalism_1_perk
					add_perk = herbalism_2_perk
					add_perk = herbalism_3_perk
					add_perk = herbalism_4_perk
					add_perk = herbalism_5_perk
					add_character_flag = {
						flag = craft_creator_tutor
						days = 50
					}
					scope:actor = {
						add_to_variable_list = {
							name = craft_creator
							target = prev
						}
					}
					trigger_event = {
						id = POD_disappearance.1
						days = 200
					}
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
