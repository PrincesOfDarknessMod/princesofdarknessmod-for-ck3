POD_activate_effigy_interaction = {
	category = interaction_category_mummy
	desc = POD_activate_effigy_interaction_desc
	icon = POD_hekau_interaction
	
	target_type = artifact
	target_filter = actor_artifacts
	
	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { has_trait = mummy }
		scope:actor = { 
			any_character_artifact = {
				NOR = {
					has_variable = lesser_chattel_effigy
					has_variable = complex_chattel_effigy
					has_variable = superior_chattel_effigy
					has_variable = princely_chattel_effigy
				}
				var:active_effigy ?= no
			}
		}
	}
	
	is_valid_showing_failures_only = {
		scope:actor = { 
			trigger_if = {
				limit = { is_east_mummy_trigger = yes }
				NOT = { has_character_modifier = yangchi0_modifier }
			}
			trigger_if = {
				limit = { is_amenti_trigger = yes }
				NOT = { has_character_modifier = sekhem0_modifier }
			}
		}
	}
	
	can_be_picked = {
		scope:target = {
			NOR = {
				has_variable = lesser_chattel_effigy
				has_variable = complex_chattel_effigy
				has_variable = superior_chattel_effigy
				has_variable = princely_chattel_effigy
			}
			var:active_effigy ?= no
		}
	}

	on_accept = {
		scope:actor = { POD_hekau_cost_effect = { COUNT = 1 } }
		scope:target = {
			set_variable = {
				name = active_effigy
				value = yes
			}
			set_active_effigy_name_effect = yes
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_deactivate_effigy_interaction = {
	category = interaction_category_mummy
	desc = POD_deactivate_effigy_interaction_desc
	icon = POD_hekau_interaction
	
	target_type = artifact
	target_filter = actor_artifacts
	
	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { has_trait = mummy }
		scope:actor = {
			any_character_artifact = { 
				has_variable = active_effigy
				var:active_effigy = yes
			} 
		}
	}
	
	can_be_picked = {
		scope:target = {
			has_variable = active_effigy
			var:active_effigy = yes
		}
	}

	on_accept = {
		scope:recipient = {
			custom_tooltip = POD_deactivate_effigy_interaction_desc
			scope:target = {
				set_variable = {
					name = active_effigy
					value = no
				}
				if = {
					limit = { 
						OR = {
							has_variable = lesser_chattel_effigy
							has_variable = complex_chattel_effigy
							has_variable = superior_chattel_effigy
							has_variable = princely_chattel_effigy
						}
					}
					clear_artifact_modifiers = yes
				}
				else = {
					set_original_effigy_name_effect = yes
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
