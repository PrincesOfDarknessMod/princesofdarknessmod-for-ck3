POD_command_effigy_interaction = {
	category = interaction_category_mummy
	desc = POD_command_effigy_interaction_desc
	options_heading = POD_command_effigy_interaction_options_heading
	icon = POD_hekau_interaction
	
	is_shown = {
		scope:actor = { has_trait = mummy }
		scope:actor = { has_perk = simple_servitor_perk }
		scope:actor = scope:recipient
	}

	cooldown = { years = 1 }
	
	send_option = {
		flag = POD_command_effigy_gold
		localization = POD_command_effigy_gold
	}
	
	send_option = {
		flag = POD_command_effigy_prestige
		localization = POD_command_effigy_prestige
	}
	
	send_option = {
		flag = POD_command_effigy_piety
		localization = POD_command_effigy_piety
	}
	
	send_option = {
		flag = POD_command_effigy_renown
		localization = POD_command_effigy_renown
	}

	on_accept = {
		scope:actor = {
			remove_variable = POD_command_effigy_gold
			remove_variable = POD_command_effigy_prestige
			remove_variable = POD_command_effigy_piety
			remove_variable = POD_command_effigy_renown
			if = {
				limit = { scope:POD_command_effigy_gold = yes }
				custom_tooltip = { text = "POD_command_effigy_interaction_gold" set_variable = POD_command_effigy_gold }
			}
			if = {
				limit = { scope:POD_command_effigy_prestige = yes }
				custom_tooltip = { text = "POD_command_effigy_interaction_prestige" set_variable = POD_command_effigy_prestige }
			}
			if = {
				limit = { scope:POD_command_effigy_piety = yes }
				custom_tooltip = { text = "POD_command_effigy_interaction_piety" set_variable = POD_command_effigy_piety }
			}
			if = {
				limit = { scope:POD_command_effigy_renown = yes }
				custom_tooltip = { text = "POD_command_effigy_interaction_renown" set_variable = POD_command_effigy_renown }
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_copy_artifact_interaction = {
	category = interaction_category_mummy
	desc = POD_copy_artifact_interaction_desc
	icon = POD_hekau_interaction
	
	target_type = artifact
	target_filter = actor_artifacts
	
	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { has_perk = lesser_chattel_perk }
		scope:actor = {
			any_character_artifact = { 
				count >= 1
			} 
		}
	}
	
	is_valid_showing_failures_only = {
		scope:actor = { 
			NOT = { has_character_modifier = sekhem0_modifier } 
			any_character_artifact = {
				trigger_if = { limit = { PREV = { has_perk = princely_chattel_perk } } }
				trigger_else_if = { limit = { PREV = { has_perk = superior_chattel_perk } } NOT = { rarity = illustrious } }
				trigger_else_if = { limit = { PREV = { has_perk = complex_chattel_perk } } NOT = { rarity = famed rarity = illustrious } }
				trigger_else = { NOT = { rarity = masterwork rarity = famed rarity = illustrious } }
                                is_unique = no
			}
		}
	}

	on_accept = {
		scope:actor = {
			POD_copy_artifact_effect = yes
			hidden_effect_new_object = {
				scope:newly_created_artifact = {
					set_max_durability = 25
					set_artifact_name = POD_copy_artifact_name
					set_artifact_description = POD_copy_artifact_desc
					reforge_artifact = { template = mummy_copy_template }
				}
			} 
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
