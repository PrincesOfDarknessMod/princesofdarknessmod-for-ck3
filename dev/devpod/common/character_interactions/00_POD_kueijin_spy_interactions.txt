POD_kueijin_spy_interaction = {
	category = interaction_category_vampire
	desc = POD_kueijin_spy_interaction_desc
	icon = POD_art_interaction
	options_heading = POD_kueijin_spy_interaction_options_heading
	use_diplomatic_range = no
	
	cooldown = { months = 6 }
	
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { 
			OR = {
				has_trait = kueijin 
				has_trait = dhampyr
			}
		}
		scope:actor = {
			OR = {
				has_perk = evaluate_perk
				has_perk = scrutinize_perk
				has_perk = permeate_perk
				has_perk = detach_limb_perk
				# has_perk = placate_the_earth_dragon_perk
				has_perk = pelesit_perk
			}
		}
		
		scope:recipient = {
			OR = {
				any_secret = {
					NOT = {is_known_by = scope:actor}
				}
				has_owned_scheme = yes
			}
		}
		POD_is_in_same_plane_trigger = yes
	}

	is_valid_showing_failures_only = {
		scope:recipient = { 
			NOT = {
				has_trait = torpor
			}
		}
	}

	# Level 1
	send_option = {
		is_shown = {
			scope:actor = {
				has_perk = evaluate_perk
			}
			scope:recipient = {
				any_secret = {
					NOT = {is_known_by = scope:actor}
				}
			}
		}
		is_valid = {
			custom_description = {
				object = scope:actor
				text = "soul_demon_incompatible"
				scope:actor = { using_demon_arts_trigger = no }
			}
		}
		flag = evaluate
		localization = POD_kueijin_spy_interaction_evaluate
	}

	# Level 1
	send_option = {
		is_shown = {
			scope:actor = {
				has_perk = scrutinize_perk
			}
			scope:recipient = {
				has_owned_scheme = yes
			}
		}
		is_valid = {
			custom_description = {
				object = scope:actor
				text = "soul_demon_incompatible"
				scope:actor = { using_demon_arts_trigger = no }
			}
		}
		flag = scrutinize
		localization = POD_kueijin_spy_interaction_scrutinize
	}

	# Level 1
	send_option = {
		is_shown = {
			scope:actor = {
				has_perk = permeate_perk
				NOT = { has_perk = placate_the_earth_dragon_perk }
			}
			scope:recipient = {
				any_secret = {
					NOT = {is_known_by = scope:actor}
				}
			}
		}	
		flag = permeate
		localization = POD_kueijin_spy_interaction_permeate
	}

	# Level 2
	send_option = {
		is_shown = {
			scope:actor = {
				has_perk = detach_limb_perk
			}
			scope:recipient = {
				has_owned_scheme = yes
			}
		}	
		flag = detach_limb
		localization = POD_kueijin_spy_interaction_detach_limb
	}

	# Level 1+3
	send_option = {
		is_shown = {
			scope:actor = {
				has_perk = permeate_perk
				has_perk = placate_the_earth_dragon_perk
			}
			scope:recipient = {
				any_secret = {
					NOT = {is_known_by = scope:actor}
				}
			}
		}	
		flag = permeate_plus
		localization = POD_kueijin_spy_interaction_permeate_plus
	}

	# Level 4
	send_option = {
		is_shown = {
			scope:actor = {
				has_perk = pelesit_perk
			}
			scope:recipient = {
				has_owned_scheme = yes
			}
		}	
		flag = pelesit
		localization = POD_kueijin_spy_interaction_pelesit
	}
	
	send_options_exclusive = yes

	on_accept = {
		scope:actor = { 
			POD_willpower_rouse_check_effect = yes
			if = {
				limit = {
					scope:evaluate = yes
				}
				duel = {
					skill = learning
					target = scope:recipient
					70 = {
						desc = POD_kueijin_spy.duel.success
						compare_modifier = {
							value = scope:duel_value
							multiplier = high_positive_duel_skill_multiplier
						}
						send_interface_toast = {
							title = POD_kueijin_spy.duel.success
							left_icon = scope:actor
							right_icon = scope:recipient
							
							custom_tooltip = POD_kueijin_spy.duel.tt.evaluate
						}
						hidden_effect = {
							scope:recipient = {
								random_secret = {
									reveal_to = scope:actor
								}
							}
						}
					}
					30 = {
						desc = POD_kueijin_spy.duel.failure
						compare_modifier = {
							value = scope:duel_value
							multiplier = low_negative_duel_skill_multiplier 
						}
						send_interface_toast = {
							title = POD_kueijin_spy.duel.failure
							left_icon = scope:actor
							right_icon = scope:recipient
							
							custom_tooltip = POD_kueijin_spy.duel.tt.attacker_lost
						}
					}
				}
				scope:actor = { POD_chi_decrease_effect = { COUNT = 1 } }
			}
			if = {
				limit = {
					scope:scrutinize = yes
				}
				duel = {
					skill = learning
					target = scope:recipient
					70 = {
						desc = POD_kueijin_spy.duel.success
						compare_modifier = {
							value = scope:duel_value
							multiplier = high_positive_duel_skill_multiplier
						}
						send_interface_toast = {
							title = POD_kueijin_spy.duel.success
							left_icon = scope:actor
							right_icon = scope:recipient
							
							custom_tooltip = POD_kueijin_spy.duel.tt.scrutinize
						}
						hidden_effect = {
							scope:recipient = {
								every_scheme = {
									expose_scheme = yes
								}
							}
						}
					}
					30 = {
						desc = POD_kueijin_spy.duel.failure
						compare_modifier = {
							value = scope:duel_value
							multiplier = low_negative_duel_skill_multiplier 
						}
						send_interface_toast = {
							title = POD_kueijin_spy.duel.failure
							left_icon = scope:actor
							right_icon = scope:recipient
							
							custom_tooltip = POD_kueijin_spy.duel.tt.attacker_lost
						}
					}
				}
				scope:actor = { POD_chi_decrease_effect = { COUNT = 1 } }
			}
			if = {
				limit = {
					scope:permeate = yes
				}
				duel = {
					skill = intrigue
					target = scope:recipient
					70 = {
						desc = POD_kueijin_spy.duel.success
						compare_modifier = {
							value = scope:duel_value
							multiplier = high_positive_duel_skill_multiplier
						}
						send_interface_toast = {
							title = POD_kueijin_spy.duel.success
							left_icon = scope:actor
							right_icon = scope:recipient
							
							custom_tooltip = POD_kueijin_spy.duel.tt.permeate
						}
						hidden_effect = {
							scope:recipient = {
								random_secret = {
									reveal_to = scope:actor
								}
							}
						}
					}
					30 = {
						desc = POD_kueijin_spy.duel.failure
						compare_modifier = {
							value = scope:duel_value
							multiplier = low_negative_duel_skill_multiplier 
						}
						send_interface_toast = {
							title = POD_kueijin_spy.duel.failure
							left_icon = scope:actor
							right_icon = scope:recipient
							
							custom_tooltip = POD_kueijin_spy.duel.tt.attacker_lost
						}
					}
				}
				scope:actor = { POD_chi_decrease_effect = { COUNT = 1 } }
			}	
			if = {
				limit = {
					scope:detach_limb = yes
				}
				duel = {
					skill = intrigue
					target = scope:recipient
					60 = {
						desc = POD_kueijin_spy.duel.success
						compare_modifier = {
							value = scope:duel_value
							multiplier = high_positive_duel_skill_multiplier
						}
						send_interface_toast = {
							title = POD_kueijin_spy.duel.success
							left_icon = scope:actor
							right_icon = scope:recipient
							
							custom_tooltip = POD_kueijin_spy.duel.tt.detach_limb
						}
						hidden_effect = {
							scope:recipient = {
								every_scheme = {
									add_scheme_progress = scheme_progress_loss
								}
							}
						}
					}
					40 = {
						desc = POD_kueijin_spy.duel.failure
						compare_modifier = {
							value = scope:duel_value
							multiplier = low_negative_duel_skill_multiplier 
						}
						send_interface_toast = {
							title = POD_kueijin_spy.duel.failure
							left_icon = scope:actor
							right_icon = scope:recipient
							
							custom_tooltip = POD_kueijin_spy.duel.tt.attacker_lost
						}
					}
				}
				scope:actor = { POD_chi_decrease_effect = { COUNT = 2 } }
			}	
			if = {
				limit = {
					scope:permeate_plus = yes
				}
				duel = {
					skill = intrigue
					target = scope:recipient
					50 = {
						desc = POD_kueijin_spy.duel.success
						compare_modifier = {
							value = scope:duel_value
							multiplier = high_positive_duel_skill_multiplier
						}
						send_interface_toast = {
							title = POD_kueijin_spy.duel.success
							left_icon = scope:actor
							right_icon = scope:recipient
							
							custom_tooltip = POD_kueijin_spy.duel.tt.permeate_plus
						}
						hidden_effect = {
							scope:recipient = {
								every_secret = {
									reveal_to = scope:actor
								}
							}
						}
					}
					50 = {
						desc = POD_kueijin_spy.duel.failure
						compare_modifier = {
							value = scope:duel_value
							multiplier = low_negative_duel_skill_multiplier 
						}
						send_interface_toast = {
							title = POD_kueijin_spy.duel.failure
							left_icon = scope:actor
							right_icon = scope:recipient
							
							custom_tooltip = POD_kueijin_spy.duel.tt.attacker_lost
						}
					}
				}
				scope:actor = {
					add_piety = minor_piety_loss
					POD_chi_decrease_effect = { COUNT = 4 } 
				}
			}	
			if = {
				limit = {
					scope:pelesit = yes
				}
				duel = {
					skill = intrigue
					target = scope:recipient
					50 = {
						desc = POD_kueijin_spy.duel.success
						compare_modifier = {
							value = scope:duel_value
							multiplier = high_positive_duel_skill_multiplier
						}
						send_interface_toast = {
							title = POD_kueijin_spy.duel.success
							left_icon = scope:actor
							right_icon = scope:recipient
							
							custom_tooltip = POD_kueijin_spy.duel.tt.pelesit
						}
						hidden_effect = {
							scope:recipient = {
								every_scheme = {
									end_scheme = yes
								}
							}
						}
					}
					50 = {
						desc = POD_kueijin_spy.duel.failure
						compare_modifier = {
							value = scope:duel_value
							multiplier = low_negative_duel_skill_multiplier 
						}
						send_interface_toast = {
							title = POD_kueijin_spy.duel.failure
							left_icon = scope:actor
							right_icon = scope:recipient
							
							custom_tooltip = POD_kueijin_spy.duel.tt.attacker_lost
						}
					}
				}
				scope:actor = {
					POD_losing_health_level_effect = { COUNT = 1 }
					POD_chi_decrease_effect = { COUNT = 4 } 
				}
			}
		}
	}
	
	auto_accept = yes

	ai_potential = {
		# The basic requirements
		is_imprisoned = no 
		is_landed = yes 
	}
	ai_targets = {
		ai_recipients = liege
	}
	ai_targets = {
		ai_recipients = neighboring_rulers
		ai_recipients = peer_vassals
		max = 10
	}
	ai_frequency = 72
	ai_will_do = {
		base = 0
		
		opinion_modifier = {
			opinion_target = scope:recipient
			multiplier = -0.25
		}
		modifier = {
		    add = 50
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
		modifier = {
			factor = 0
			scope:recipient = {
				is_ai = no
			}
		}
	}
}
