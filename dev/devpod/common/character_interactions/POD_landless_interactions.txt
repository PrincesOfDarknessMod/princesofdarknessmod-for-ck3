	category = interaction_category_diplomacy
	#common_interaction = yes
	#interface_priority = 90
	desc = POD_seek_ambassadorial_patron_interaction_desc
	
	#ai_frequency = 0

	is_shown = {
		scope:actor = { has_character_flag = ambassador_searching_for_patron_flag }
		NOT = { scope:recipient = scope:actor }
		POD_interaction_is_shown = yes
	}
	
	is_valid_showing_failures_only = {
		scope:recipient = {
			trigger_if = {

			}

		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:actor = {
			trigger_event = pod1230objectives.169
		}
	}


	ai_will_do = {
		base = 0
	}
}

			is_imprisoned = no
		}

	}

POD_request_dragon_nest_access_interaction = {
	category = interaction_category_diplomacy
	icon = icon_transfer_county
	desc = POD_request_dragon_nest_access_interaction_desc

	common_interaction = yes

	target_type = title
	target_filter = recipient_domain_titles

	cooldown_against_recipient = {
		years = 5
	}
	
	cost = { prestige = 1000 }

	is_shown = {
		scope:actor = {
			POD_is_terrestrial_exalted_trigger = yes
			government_has_flag = government_is_landless_adventurer
		}
		scope:recipient = { 
			is_landed = yes
			any_sub_realm_county = {
				title_province = { has_building_or_higher = dragon_nest_01 }
			} 
		}
		scope:recipient.highest_held_title_tier > tier_barony
		POD_interaction_is_shown = yes
	}
	
	can_be_picked_title = {
		scope:target = {
			tier = tier_county
			title_province = { has_building_or_higher = dragon_nest_01 }
		}
	}
	
	is_valid_showing_failures_only = {
		trigger_if = {
			limit = { exists = scope:target }
			custom_tooltip = {
				text = POD_request_dragon_nest_access_interaction_same_location
				scope:actor = { exists = location }
				scope:target ?= { title_province = scope:actor.location }
			}
		}
		custom_tooltip = {
			text = POD_request_dragon_nest_access_interaction_already_have
			scope:actor = {
				NOT = { has_variable = POD_landless_dragon_nest }
			}
		}
		scope:actor = { is_imprisoned = no }
		scope:recipient = { 
			is_imprisoned = no 
			is_contact_of = scope:actor
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}
	should_use_extra_icon = {
		scope:actor = { has_usable_hook = scope:recipient }
	}
	extra_icon = "gfx/interface/icons/character_interactions/hook_icon.dds"

	on_accept = {
		scope:actor = {
			send_interface_message = {
				type = event_generic_good
				title = POD_landless_dragon_nest_gained
				custom_tooltip = POD_request_dragon_nest_access_interaction_effect
				custom_tooltip = POD_request_dragon_nest_access_interaction_warning
				set_variable = {
					name = POD_landless_dragon_nest
					value = scope:target.title_province
				}
				if = {
					limit = {
						has_usable_hook = scope:recipient
						scope:recipient = { is_ai = yes }
					}
					use_hook = scope:recipient
					custom_tooltip = contact_list_request_interaction.tt.hook_auto_used
				}
			}
		}
	}

	ai_accept = {
		base = -25 # Extra hard

		modifier = {
			add = 50
			scope:actor = { has_usable_hook = scope:recipient }
			desc = SCHEME_WEAK_HOOK_USED
		}

		modifier = {
			add = -50
			scope:recipient = {
				NOR = {
					is_contact_of = scope:actor
					is_close_or_extended_family_of = scope:actor
					is_consort_of = scope:actor
				}
			}
			desc = contact_list_request_interaction.no_contact
		}

		opinion_modifier = { # Opinion Factor
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.5
			desc = AI_OPINION_REASON
		}

		modifier = {
			add = -50
			scope:recipient = {
				has_opinion_modifier = {
					modifier = laamp_used_contact_opinion
					target = scope:actor
				}
			}
			desc = contact_list_request_interaction.repeated_request
		}

		ai_value_modifier = {
			ai_compassion = -0.5
			ai_greed = -0.5
			min = -50
			max = 50
		}

		modifier = {
			add = 25
			desc = contact_list_request_interaction.adventurer
			scope:recipient = {
				has_trait = adventurer
			}
		}

		modifier = {
			add = 25
			desc = contact_list_request_interaction.adventurer_follower
			scope:recipient = {
				has_trait = adventurer_follower
			}
		}

		modifier = {
			add = 20
			desc = contact_list_request_interaction.flexible_truces_perk
			scope:actor = {
				has_perk = flexible_truces_perk
			}
		}
