POD_rasail_al_shifa_healing_interaction = {
	category = interaction_category_holyarts
	desc = POD_rasail_al_shifa_healing_interaction_desc
	icon = POD_spiritual_guidance_interaction
	
	cooldown = { years = 2 }
	cooldown_against_recipient = { years = 6 }
	
	is_shown = {
		scope:actor = { has_perk = rasail_al_shifa_perk }
		scope:actor != scope:recipient
		POD_interaction_is_shown = yes
	}
	
	is_valid_showing_failures_only = {
		scope:recipient = { POD_health_has_superficial_wound_trigger = yes }
		scope:actor = { 
			custom_tooltip = {
				text = POD_rasail_al_shifa_healing_interaction_requirement
				any_equipped_character_artifact = {
					artifact_type = journal
					prev = { can_benefit_from_artifact = prev }
				}
			}
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:actor = {
			scope:recipient = { POD_heal_superficial_wound_effect = { COUNT = POD_base_heal_value } }

			duel = {
				skill = learning
				target = scope:recipient
			
				10 = {
					compare_modifier = { value = scope:duel_value multiplier = high_positive_duel_skill_multiplier }
					trigger = { any_equipped_character_artifact = { artifact_type = journal prev = { can_benefit_from_artifact = prev } rarity = illustrious } }

					scope:recipient = { POD_heal_superficial_wound_effect = { COUNT = POD_triple_heal_value } }
				}
				20 = {
					compare_modifier = { value = scope:duel_value multiplier = low_positive_duel_skill_multiplier }
					trigger = { any_equipped_character_artifact = { artifact_type = journal prev = { can_benefit_from_artifact = prev } OR = { rarity = illustrious rarity = famed } } }
					
					scope:recipient = { POD_heal_superficial_wound_effect = { COUNT = POD_double_heal_value } }
				}
				40 = {
					trigger = { any_equipped_character_artifact = { artifact_type = journal prev = { can_benefit_from_artifact = prev } OR = { rarity = illustrious rarity = famed rarity = masterwork } } }
					
					scope:recipient = { POD_heal_superficial_wound_effect = { COUNT = POD_base_heal_value } }
				}
				60 = {
					compare_modifier = { value = scope:duel_value multiplier = low_negative_duel_skill_multiplier }
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_frequency = 0
	ai_will_do = { base = 0 }
}

POD_plasmic_forging_interaction = {
	category = interaction_category_hostile
	common_interaction = no
	interface_priority = 30
	desc = POD_plasmic_forging_interaction_desc
	icon = artisan_inspiration

	cooldown = { years = 3 }

	target_type = artifact
	target_filter = actor_artifacts

	can_be_picked_artifact = {
		scope:actor = {
			any_character_artifact = {
				this = scope:target
			}
		}
		#Artifacts locked from being giftable by a variable
		scope:target = {
			NOR = {
				has_variable = 1025_treasure_map
				has_variable = ungiftable
				has_variable = is_tribute_mission_artifact
			}
		}
	}

	is_shown = {
		scope:recipient != scope:actor
		scope:actor = { 
			has_any_artifact = yes 
			has_perk = endowments_judge_magi_3_perk
		}
		scope:recipient = {
			OR = {
				POD_is_wraith_trigger = yes
				POD_is_spirit_trigger = yes
			}
		}
		POD_hostile_interaction_is_shown = yes
	}
	
	is_valid = {
		scope:recipient = {
			OR = {
				scope:actor = { has_strong_hook = scope:recipient }
				AND = {
					is_imprisoned = yes
					is_imprisoned_by = scope:actor
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			is_imprisoned = no
		}
	}

	on_accept = {
		scope:target = { 
			scope:recipient ?= { POD_crafting_give_modifier_based_on_character = { ARTIFACT = PREV } }
		}
		scope:recipient = {
			death = {
				death_reason = death_souleaten
				killer = scope:actor
			}
		}
	}

	ai_frequency = 0

	auto_accept = yes

	ai_will_do = {
		base = 0
	}
}
