POD_numina_sorcery_open_gui_interaction = {
	category = interaction_category_mage
	icon = POD_sorcery_interaction
	desc = POD_numina_sorcery_open_gui_interaction_desc
	common_interaction = yes
	is_shown = {
		scope:actor = { 
			is_sorcerer_trigger = yes
		}
		OR = {
			POD_hostile_interaction_is_shown = yes
			# ugly bandaid to make this usable on demons in the abyss
			AND = {
				scope:actor = { has_perk = numina_conjuration_6_perk }
				scope:recipient = { POD_is_demon_trigger = yes }
				scope:recipient = { has_character_modifier = trapped_in_abyss_modifier }
				POD_basic_interaction_is_shown = yes
			}
		}
	}
	on_accept = {
		custom_tooltip = open_s_menu
		scope:actor = {
			POD_open_sorcery_gui = { TARGET = scope:recipient }
		}
	}
	auto_accept = yes
}

POD_train_sorcery_interaction = {
	category = interaction_category_mage
	icon = POD_sorcery_interaction
	desc = POD_train_sorcery_interaction_desc
	common_interaction = yes

	cooldown = { months = 12 }
	cost = { piety = medium_piety_value }

	is_shown = {
		scope:actor != scope:recipient
		scope:actor = { is_sorcerer_trigger = yes }
		scope:recipient = { is_sorcerer_trigger = yes }
		POD_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			sorcery_lifestyle_perk_points >= 1
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:actor = {
			add_sorcery_lifestyle_perk_points = -1
			stress_impact = {
				base = medium_stress_gain
			}
		}
		scope:recipient = {
			add_sorcery_lifestyle_perk_points = 1
			add_opinion = {
				target = scope:actor
				modifier = mentored_me_opinion
			}
		}
	}

	#ai_frequency = 0

	auto_accept = yes

	ai_will_do = {
		base = 0
	}
}

POD_initiate_sorcery_interaction = {
	category = interaction_category_mage
	icon = POD_sorcery_interaction
	desc = POD_initiate_sorcery_interaction_desc

	cooldown = { months = 12 }
	cost = { piety = major_piety_value }

	is_shown = {
		scope:actor != scope:recipient
		scope:actor = { is_sorcerer_trigger = yes }
		scope:recipient = { can_learn_sorcery_trigger = yes }
		POD_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			sorcery_lifestyle_perk_points >= 1
		}
		scope:recipient = {
			is_imprisoned = no
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:actor = {
			add_sorcery_lifestyle_perk_points = -1
			stress_impact = {
				base = major_stress_gain
			}
			duel = {
				skill = learning
				value = 15
				25 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					scope:recipient = {
						give_sorcerer_secret_or_trait_effect = yes
						add_opinion = {
							target = scope:actor
							modifier = mentored_me_opinion
						}
					}
				}
				50 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier
					}
				}
			}
		}
	}

	#ai_frequency = 0

	auto_accept = yes

	ai_will_do = {
		base = 0
	}
}
