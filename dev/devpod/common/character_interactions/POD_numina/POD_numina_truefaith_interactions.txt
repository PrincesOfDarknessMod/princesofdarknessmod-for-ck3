POD_ha_bless_interaction = {
	category = interaction_category_holyarts
	common_interaction = yes
	interface_priority = 111
	
	ai_frequency = 62
	
	ai_targets = {
		ai_recipients = vassals
		ai_recipients = courtiers
		ai_recipients = children
	}
	
	is_shown = {
		scope:actor != scope:recipient
		scope:actor = {
			OR = {
				has_perk = dominus_vobiscum_perk
				has_trait = beatific
				any_equipped_character_artifact = { has_variable = ring_chrysostom }
				any_equipped_character_artifact = { has_variable = fingerbone_hannah }
			}
		}
		POD_interaction_is_shown = yes
	}
	
	is_valid = {
		scope:actor = {
			trigger_if = {
				limit = { POD_is_unholy_trigger = no }
				scope:recipient = { POD_is_unholy_trigger = no }
			}
		}
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
			NOT = { has_character_modifier = crisis_of_faith_modifier }
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}
	
	cooldown = { months = 6 }
	
	cost = { piety = 100 }

	desc = POD_ha_bless_interaction_desc
	
	on_accept = {
		scope:actor = {
			play_music_cue = "mx_cue_sacredrite"
			if = {
				limit = {
					NAND = {
						has_trait = ha_patris_mastery
						has_trait = ha_filius_mastery
						has_trait = ha_spiritus_mastery
					}
				}
				add_faith_lifestyle_xp = 50
			}
			send_interface_toast = {
				title = POD_ha_bless_interaction_toast
				left_icon = scope:actor
				right_icon = scope:recipient
				scope:recipient = {
					add_character_modifier = {
						modifier = blessed_by_truefaith_modifier
						years = 5
					}
				}
			}
			POD_side_effects_from_using_true_faith = yes
		}
		scope:recipient = {
			add_opinion = {
				target = scope:actor
				modifier = gift_opinion
				opinion = 15
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
		modifier = {
			add = 100
			scope:actor = {
				piety > 200
				is_at_war = yes
			}
			scope:recipient = {
				is_knight_of = scope:actor
				prowess >= 20
			}
		}
	}
}

POD_ha_intercession_interaction = {
	category = interaction_category_holyarts

	ai_frequency = 62
	
	ai_targets = {

	}
	
	is_shown = {

			OR = {
				has_perk = intercession_perk
				has_trait = beatific
				any_equipped_character_artifact = { has_variable = eunuch_scapular }
			}
		}

	}
	
	is_valid = {
		scope:actor = {
			trigger_if = {
				limit = { POD_is_unholy_trigger = no }
				scope:recipient = { POD_is_unholy_trigger = no }
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			trigger_if = {
				limit = { POD_splat_mixing_haight_mode = no }
				POD_is_infernalist_trigger = no
			}
		}
		scope:actor = {
			 NOT = { has_character_modifier = crisis_of_faith_modifier }
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}
	
	cooldown = { months = 6 }
	
	cost = { piety = 100 }

	desc = POD_ha_intercession_interaction_desc
	
	on_accept = {
		scope:actor = {
			play_music_cue = "mx_cue_sacredrite"
			if = {
				limit = {
					NAND = {
						has_trait = ha_patris_mastery
						has_trait = ha_filius_mastery
						has_trait = ha_spiritus_mastery
					}
				}
				add_faith_lifestyle_xp = 100
			}
			random = {
				chance = 25
				modifier = {
					add = 75
					scope:actor = {
						any_directly_owned_province = {
							has_building_with_flag = building_improves_intercession
						}
					}
				}
				if = {
					limit = {
						scope:recipient = {
							POD_has_true_faith_trigger = no
						}
					}
					add_character_flag = {
						flag = intercession_success
						days = 1
					}
					show_as_tooltip = {
						scope:recipient = { add_trait = truefaith }
					}
				}
			}
			send_interface_toast = {
				title = POD_ha_intercession_interaction_toast
				left_icon = scope:actor
				right_icon = scope:recipient
				scope:recipient = {
					add_piety = 200
				}
				if = {
					limit = {
						has_character_flag = intercession_success
					}
					scope:recipient = { add_trait = truefaith }
				}
			}
			POD_side_effects_from_using_true_faith = yes
		}
		scope:recipient = {
			add_opinion = {
				target = scope:actor
				modifier = gift_opinion
				opinion = 15
			}
		}
	}

	auto_accept = yes
	
	ai_will_do = {
		base = 0
		modifier = {
			add = 100
			scope:actor = { piety > 2000 }
		}
	}
}
