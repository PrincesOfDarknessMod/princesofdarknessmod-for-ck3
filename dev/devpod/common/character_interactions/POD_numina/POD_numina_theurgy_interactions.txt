POD_numina_theurgy_medicamenti_interaction = {
	category = interaction_category_holyarts
	icon = POD_theurgy_interaction
	desc = POD_numina_theurgy_medicamenti_interaction_desc
	common_interaction = yes

	is_shown = {
		scope:actor = { has_perk = numina_via_medicamenti_1_perk }
		POD_interaction_is_shown = yes
	}
	is_valid_showing_failures_only = {
		POD_interaction_is_valid_showing_failures_only = yes
	}

	send_options_exclusive = no

	send_option = {
		is_shown = {
			scope:actor = {
				has_perk = numina_via_medicamenti_5_perk
			}
		}
		is_valid = {
			scope:recipient = {
				POD_health_has_aggravated_wound_trigger = yes
			}
		}
		flag = aggravated
		localization = POD_heal_aggravated_wound
	}

	on_accept = {
		scope:actor = {
			POD_theurgy_cost_effect = { COUNT = 2 }
			if = {
				limit = { NOT = { scope:actor = scope:recipient } }
				POD_commit_heal_effect = yes
			}
		}
		scope:recipient = {
			if = {
				limit = { scope:aggravated = yes }
				POD_convert_aggravated_to_superficial_wound = { COUNT = POD_numina_via_medicamenti_5_convert }
			}
			else = {
				if = {
					limit = {
						scope:actor = { has_perk = numina_via_medicamenti_5_perk }
					}
					recover_from_disease_effect = { DISEASE = cancer }
					recover_from_disease_effect = { DISEASE = great_pox }
					recover_from_disease_effect = { DISEASE = gout_ridden }
					recover_from_disease_effect = { DISEASE = leper }
					recover_from_disease_effect = { DISEASE = lovers_pox }
					recover_from_disease_effect = { DISEASE = smallpox }
					recover_from_disease_effect = { DISEASE = typhus }
					recover_from_disease_effect = { DISEASE = bubonic_plague }
					recover_from_disease_effect = { DISEASE = measles }
				}
				if = {
					limit = {
						scope:actor = { has_perk = numina_via_medicamenti_4_perk }
					}
					recover_from_disease_effect = { DISEASE = ill }
					recover_from_disease_effect = { DISEASE = consumption }
					recover_from_disease_effect = { DISEASE = pneumonic }
					recover_from_disease_effect = { DISEASE = dysentery }
					recover_from_disease_effect = { DISEASE = ergotism }
					recover_from_disease_effect = { DISEASE = early_great_pox }
				}
				if = {
					limit = {
						scope:actor = { has_perk = numina_via_medicamenti_3_perk }
					}
					POD_heal_superficial_wound_effect = {
						COUNT = scope:actor.POD_numina_via_medicamenti_heal
					}
				}
			}
			add_character_modifier = {
				modifier = numina_via_medicamenti_1_modifier
				years = 5
			}
			if = {
				limit = {
					scope:actor = { has_perk = numina_via_medicamenti_2_perk }
				}
				add_character_modifier = {
					modifier = numina_via_medicamenti_2_modifier
					years = 5
				}
			}
		}
	}
	auto_accept = yes
}

POD_train_theurgy_interaction = {
	category = interaction_category_holyarts
	icon = POD_theurgy_interaction
	desc = POD_train_theurgy_interaction_desc
	common_interaction = yes

	cooldown = { months = 12 }
	cost = { piety = medium_piety_value }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_trait = theurgist }
		scope:recipient = { has_trait = theurgist }
		POD_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			theurgy_lifestyle_perk_points >= 1
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:actor = {
			add_theurgy_lifestyle_perk_points = -1
			stress_impact = {
				base = medium_stress_gain
			}
		}
		scope:recipient = {
			add_theurgy_lifestyle_perk_points = 1
			add_opinion = {
				target = scope:actor
				modifier = mentored_me_opinion
			}
		}
	}

	ai_frequency = 0

	auto_accept = yes

	ai_will_do = {
		base = 0
	}
}

POD_initiate_theurgy_interaction = {
	category = interaction_category_holyarts
	icon = POD_theurgy_interaction
	desc = POD_initiate_theurgy_interaction_desc

	cooldown = { months = 12 }
	cost = { piety = major_piety_value }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_trait = theurgist }
		scope:recipient = { can_learn_theurgy_trigger = yes }
		POD_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			theurgy_lifestyle_perk_points >= 1
		}
		scope:recipient = {
			is_imprisoned = no
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:actor = {
			add_theurgy_lifestyle_perk_points = -1
			stress_impact = {
				base = major_stress_gain
			}
			duel = {
				skill = learning
				value = 15
				25 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					scope:recipient = {
						add_trait = theurgist
						add_opinion = {
							target = scope:actor
							modifier = mentored_me_opinion
						}
					}
				}
				50 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier
					}
				}
			}
		}
	}

	ai_frequency = 0

	auto_accept = yes

	ai_will_do = {
		base = 0
	}
}
