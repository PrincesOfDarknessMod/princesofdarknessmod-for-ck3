		scope:actor = scope:recipient

	}

		POD_self_targeted_interaction_is_valid_showing_failures_only = yes
	}

		scope:actor = {
			POD_mummy_memory_check_effect = { LEVEL = 1 }

				create_character = {
					save_scope_as = summoned_wraith
					dynasty = none
					template = POD_wraith_character
					location = this.location
				}
			}
			hidden_effect = {
				send_interface_toast = {
					title = POD_theurgy_summon_wraith_from_shadowlands_interaction

					right_icon = scope:actor
					left_icon = scope:summoned_wraith
				}

				add_courtier = scope:summoned_wraith
				scope:summoned_wraith = {
					hidden_effect = {
						return_to_court = yes
					}
					add_character_flag = {
						flag = blocked_from_leaving
						years = 5
					}
				}
			}
		}
	}

	auto_accept = yes

	ai_will_do = {
		base = 0
	}
