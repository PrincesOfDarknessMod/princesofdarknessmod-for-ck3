		is_shown = {
			NOR = {
				scope:greater_flood = yes
				scope:meteor_shower = yes
				scope:sand_storm = yes
				scope:flood = yes
				scope:early_apocalypse = yes
			}
			scope:actor = { uses_cairo_sobek_main_reward = no }
		}
		is_valid = { scope:actor = { has_perk = cycle_of_ra_perk } }
		flag = cycle_of_ra
		localization = pod_weather_cycle_of_ra
	}

	on_accept = {
		save_scope_value_as = {
			name = integrity_impact
			value = {
				switch = {
					trigger = yes
					scope:county	= { value = -2 }
					scope:duchy		= { value = -4 }
					scope:kingdom	= { value = -6 }
					scope:empire	= { value = -8 }
					scope:continent	= { value = -10 }
				}
				if = {
					limit = {
						OR = {
							scope:meteor_shower = yes
							scope:early_apocalypse = yes
						}
					}
					add = -2
				}
				if = {
					limit = { scope:cycle_of_ra = yes }
					multiply = -1
				}
			}
		}
		
		switch = {
			trigger = yes
			scope:flood				= { save_scope_value_as = { name = weather_type   value = flag:pod_weather_flood } }
			scope:greater_flood		= { save_scope_value_as = { name = weather_type   value = flag:pod_weather_greater_flood } }
			scope:meteor_shower		= { save_scope_value_as = { name = weather_type   value = flag:pod_weather_meteor_shower } }
			scope:early_apocalypse	= { save_scope_value_as = { name = weather_type   value = flag:pod_weather_early_apocalypse } }
			scope:sand_storm		= { save_scope_value_as = { name = weather_type   value = flag:pod_weather_sand_storm } }
			scope:cycle_of_ra		= { save_scope_value_as = { name = weather_type   value = flag:pod_weather_cycle_of_ra } }
		}
		
		
		scope:actor = {
			POD_mummy_memory_check_effect = { LEVEL = 1 }
			if = {
				limit = { POD_hekau_weather_magic_interaction_cost > 0 }
				POD_hekau_cost_effect = { COUNT = POD_hekau_weather_magic_interaction_cost }
			}
			POD_gain_integrity = { VALUE = scope:integrity_impact }
			
			switch = {
				trigger = yes
				scope:county	= { POD_setup_weather_magic_counties_from_title = { TITLE = scope:recipient.location.county } }
				scope:duchy		= { POD_setup_weather_magic_counties_from_title = { TITLE = scope:recipient.location.duchy } }
				scope:kingdom	= { POD_setup_weather_magic_counties_from_title = { TITLE = scope:recipient.location.kingdom } }
				scope:empire	= { POD_setup_weather_magic_counties_from_title = { TITLE = scope:recipient.location.empire } }
				scope:continent	= { POD_setup_weather_magic_counties_from_province_region = { PROVINCE = scope:recipient.location } }
			}
			
			POD_weather_magic_effect = {
				MONTHS = POD_weather_mummy_duration
			}
		}
	}

	can_send = {
		custom_description = {
			text = send_option_selected
			OR = {
				scope:county ?= yes
				scope:duchy ?= yes
				scope:kingdom ?= yes
				scope:empire ?= yes
				scope:continent ?= yes
			}
			OR = {
				scope:flood ?= yes
				scope:greater_flood ?= yes
				scope:meteor_shower ?= yes
				scope:early_apocalypse ?= yes
				scope:sand_storm ?= yes
				scope:cycle_of_ra ?= yes
			}
		}
	}

	auto_accept = yes

	ai_frequency = 0
}
