POD_command_effigy_interaction = {
	category = interaction_category_mummy
	desc = POD_command_effigy_interaction_desc
	options_heading = POD_command_effigy_interaction_options_heading
	icon = POD_hekau_interaction
	
	is_shown = {
		scope:actor = {
			has_perk = simple_servitor_perk
		}
		scope:actor = scope:recipient
	}

	is_valid_showing_failures_only = {
		POD_self_targeted_interaction_is_valid_showing_failures_only = yes
	}

	cooldown = { years = 1 }
	
	send_option = {
		flag = POD_command_effigy_gold
		localization = POD_command_effigy_gold
	}
	
	send_option = {
		flag = POD_command_effigy_prestige
		localization = POD_command_effigy_prestige
	}
	
	send_option = {
		flag = POD_command_effigy_piety
		localization = POD_command_effigy_piety
	}
	
	send_option = {
		flag = POD_command_effigy_renown
		localization = POD_command_effigy_renown
	}
	
	send_option = {
		flag = none
		localization = POD_command_effigy_none
	}

	on_accept = {
		scope:actor = {
			POD_mummy_memory_check_effect = { LEVEL = 1 }
			POD_remove_character_modifier = { MODIFIER = POD_command_effigy_interaction_gold }
			POD_remove_character_modifier = { MODIFIER = POD_command_effigy_interaction_prestige }
			POD_remove_character_modifier = { MODIFIER = POD_command_effigy_interaction_piety }
			POD_remove_character_modifier = { MODIFIER = POD_command_effigy_interaction_renown }
			switch = {
				trigger = yes
				scope:POD_command_effigy_gold = { add_character_modifier = POD_command_effigy_interaction_gold }
				scope:POD_command_effigy_prestige = { add_character_modifier = POD_command_effigy_interaction_prestige }
				scope:POD_command_effigy_piety = { add_character_modifier = POD_command_effigy_interaction_piety }
				scope:POD_command_effigy_renown = { add_character_modifier = POD_command_effigy_interaction_renown }
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_copy_artifact_interaction = {
	category = interaction_category_mummy
	desc = POD_copy_artifact_interaction_desc
	icon = POD_hekau_interaction
	
	target_type = artifact
	target_filter = actor_artifacts
	
	is_shown = {
		scope:actor = scope:recipient
		scope:actor = {
			has_perk = lesser_chattel_perk
		}
		scope:actor = {
			any_character_artifact = {
				count >= 1
				trigger_if = { limit = { scope:actor = { has_perk = princely_chattel_perk } } }
				trigger_else_if = { limit = { scope:actor = { has_perk = superior_chattel_perk } } NOT = { rarity = illustrious } }
				trigger_else_if = { limit = { scope:actor = { has_perk = complex_chattel_perk } } NOT = { rarity = famed rarity = illustrious } }
				trigger_else = { NOT = { rarity = masterwork rarity = famed rarity = illustrious } }
				is_unique = no
				POD_artifact_can_be_copied_trigger = yes
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			POD_has_enough_sekhem_for_hekau = { AMOUNT = POD_copy_artifact_interaction_cost }
		}
		POD_self_targeted_interaction_is_valid_showing_failures_only = yes
	}

	can_be_picked_artifact = {
		scope:target = {
			trigger_if = { limit = { scope:actor = { has_perk = princely_chattel_perk } } }
			trigger_else_if = { limit = { scope:actor = { has_perk = superior_chattel_perk } } NOT = { rarity = illustrious } }
			trigger_else_if = { limit = { scope:actor = { has_perk = complex_chattel_perk } } NOT = { rarity = famed rarity = illustrious } }
			trigger_else = { NOT = { rarity = masterwork rarity = famed rarity = illustrious } }
			is_unique = no
			POD_artifact_can_be_copied_trigger = yes
		}
	}

	on_accept = {
		scope:actor = {
			POD_copy_artifact_effect = yes
			POD_hekau_cost_effect = { COUNT = POD_copy_artifact_interaction_cost }
			POD_mummy_memory_check_effect = { LEVEL = 2 }
			hidden_effect_new_object = {
				scope:newly_created_artifact = {
					set_max_durability = 25
					set_artifact_name = POD_copy_artifact_name
					set_artifact_description = POD_copy_artifact_desc
					POD_artifact_type_injector_artifact_scope = {
						EFFECT = POD_add_copied_mummy_artifact_requirements
					}
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
