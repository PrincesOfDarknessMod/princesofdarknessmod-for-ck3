
POD_aid_from_spirits_perk_interaction = {
	category = interaction_category_vampire
	icon = POD_spiritus_interaction
	desc = POD_aid_from_spirits_perk_interaction_desc

	target_type = artifact
	target_filter = actor_artifacts
	
	cost = { piety = 500 }
	
	cooldown = { years = 3 }
	
	is_shown = {
		scope:actor = { has_perk = aid_from_spirits_perk }
		scope:actor = scope:recipient
		scope:recipient = { has_any_artifact = yes }
	}

	can_be_picked = {
		scope:target = {
			NOT = { has_variable = artifact_pieces } 
			NOT = { has_variable = possessed_artifact }
			NOT = { has_variable = summoned_artifact }
			NOT = { has_variable = reliquarie } 
			NOT = { has_variable = wraith_jade }
			NOT = { artifact_type = animal }
		}
	}

	is_valid_showing_failures_only= {
		scope:actor = {
		   NOT = { has_character_modifier = bloodhunger5_modifier }
		}	
	}
	
	send_option = {
		flag = prowess
		localization = POD_interaction_artifact_prowess_modifier
	}
	
	send_option = {
		flag = diplomacy
		localization = POD_interaction_artifact_diplomacy_modifier
	}
	
	send_option = {
		flag = martial
		localization = POD_interaction_artifact_martial_modifier
	}
	
	send_option = {
		flag = stewardship
		localization = POD_interaction_artifact_stewardship_modifier
	}
	
	send_option = {
		flag = intrigue
		localization = POD_interaction_artifact_intrigue_modifier
	}
	
	send_option = {
		flag = learning
		localization = POD_interaction_artifact_learning_modifier
	}

	on_accept = {
		scope:actor = { POD_willpower_rouse_check_effect = yes }
		scope:actor = { POD_discipline_cost_effect = { COUNT = 1 } }
		scope:target = {
			if = {
				limit = { scope:prowess = yes }
				add_artifact_modifier = artifact_prowess_1_modifier
			}
			if = {
				limit = { scope:diplomacy = yes }
				add_artifact_modifier = artifact_diplomacy_1_modifier
			}
			if = {
				limit = { scope:martial = yes }
				add_artifact_modifier = artifact_martial_1_modifier
			}
			if = {
				limit = { scope:stewardship = yes }
				add_artifact_modifier = artifact_stewardship_1_modifier
			}
			if = {
				limit = { scope:intrigue = yes }
				add_artifact_modifier = artifact_intrigue_1_modifier
			}
			if = {
				limit = { scope:learning = yes }
				add_artifact_modifier = artifact_learning_1_modifier
			}
			set_variable = possessed_artifact
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_hunt_spirit_interaction = {
	category = interaction_category_vampire
	icon = POD_spiritus_interaction
	desc = POD_hunt_spirit_interaction_desc
	
	auto_accept = yes

	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { 
			OR = { 
				has_perk = engling_fury_perk
				has_perk = beastly_hunger_perk 
			}
		}
	}
	
	is_valid_showing_failures_only = {
		scope:recipient = {
			NOT = { has_character_modifier = spirit_opinion0_modifier }
		}
	}

	on_accept = {
	   scope:actor = {
			POD_willpower_regain_check_effect = yes
			POD_anger_the_local_spirit_effect = yes
			if = {
				limit = { has_perk = engling_fury_perk }
				add_prestige = minor_prestige_value
			}
			if = {
				limit = { has_perk = beastly_hunger_perk }
				add_piety = minor_piety_value
			}
	   }
    }
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no 
		is_landed = yes 
		
		OR = {
			prestige < medium_prestige_value
			piety < medium_piety_value
		}
	}
	ai_targets = {
		ai_recipients = self
	}
	ai_frequency = 120
	ai_will_do = {
		base = 100
	}  
}
