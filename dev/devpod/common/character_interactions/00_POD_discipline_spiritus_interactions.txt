POD_hunt_spirit_interaction = {
	category = interaction_category_vampire
	icon = POD_spiritus_interaction
	desc = POD_hunt_spirit_interaction_desc
	
	auto_accept = yes

	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { has_spiritus_trigger = yes }
		scope:actor = { 
			OR = { 
				has_perk = engling_fury_perk
				has_perk = beastly_hunger_perk 
			}
		}
	}
	
	is_valid_showing_failures_only = {
		scope:recipient = {
			NOT = { has_character_modifier = spirit_opinion0_modifier }
		}
	}

	on_accept = {
	   scope:actor = {
			POD_willpower_regain_check_effect = yes
			POD_anger_the_local_spirit_effect = yes
			if = {
				limit = { has_perk = engling_fury_perk }
				add_prestige = minor_prestige_value
			}
			if = {
				limit = { has_perk = beastly_hunger_perk }
				add_piety = minor_piety_value
			}
	   }
    }
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no 
		is_landed = yes 
		
		OR = {
			prestige < medium_prestige_value
			piety < medium_piety_value
		}
	}
	ai_targets = {
		ai_recipients = self
	}
	ai_frequency = 120
	ai_will_do = {
		base = 100
	}  
}
