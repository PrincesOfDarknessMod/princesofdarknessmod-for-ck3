


POD_learn_spirit_language_interaction = {
	category = interaction_category_spirit
	icon = icon_scheme_learn_language
	desc = POD_learn_spirit_language_interaction_desc
	common_interaction = yes

	send_name = START_SCHEME

	scheme = learn_language
	ignores_pending_interaction_block = yes


	is_shown = {
		scope:recipient = { has_trait = spirit }
		POD_knows_spirit_language_trigger = no
		POD_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			can_start_scheme = {
				type = learn_language
				target_character = scope:recipient
			}
		}
		scope:recipient = {
			custom_tooltip = {
				text = POD_learn_spirit_language_interaction_req
				POD_has_spirit_ancestor_language_trigger = no
			}
		}
		trigger_if = {
			limit = { scope:recipient = { var:spirit_type = flag:spirit_innocent } }
			scope:actor = { POD_is_mokole_trigger = no }
		}
		POD_personal_scheme_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:actor = {
			send_interface_toast = {
				type = event_toast_effect_neutral
				title = learn_language_interaction_notification

				left_icon = scope:actor
				right_icon = scope:recipient

				begin_scheme_basic_effect = {
					SCHEME_TYPE = learn_language
					TARGET_TYPE = target_character
					TARGET_SCOPE = scope:recipient
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_frequency = 0
	ai_will_do = { base = 0 }
}

POD_summon_spirit_interaction = {
	category = interaction_category_spirit
	icon = POD_spirit_interaction
	desc = POD_summon_spirit_interaction_desc
	common_interaction = yes
	
	cooldown = { years = 5 }
	
	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { POD_can_summon_spirit_trigger = yes }
	}
	
	is_valid_showing_failures_only = {
		POD_self_targeted_interaction_is_valid_showing_failures_only = yes
	}

	#Random
	send_option = {
		is_shown = { scope:actor = { POD_can_summon_random_spirit_trigger = yes } }
		is_valid = { scope:actor = { POD_has_enough_splat_agnostic_energy_reserves = { AMOUNT = 2 } } }
		flag = random
		localization = POD_summon_spirit_interaction_random
	}

	#Gaian
	send_option = {
		is_shown = { scope:actor = { POD_can_summon_gaian_spirit_trigger = yes } }
		is_valid = { scope:actor = { POD_has_enough_splat_agnostic_energy_reserves = { AMOUNT = 3 } } }
		flag = gaian
		localization = POD_summon_spirit_interaction_gaian
	}

	#Bane
	send_option = {
		is_shown = { scope:actor = { POD_can_summon_bane_spirit_trigger = yes } }
		is_valid = { scope:actor = { POD_has_enough_splat_agnostic_energy_reserves = { AMOUNT = 3 } } }
		flag = bane
		localization = POD_summon_spirit_interaction_bane
	}

	#Elemental Bane
	send_option = {
		is_shown = { scope:actor = { POD_can_summon_elemental_bane_spirit_trigger = yes } }
		is_valid = { scope:actor = { POD_has_enough_splat_agnostic_energy_reserves = { AMOUNT = 5 } } }
		flag = elemental_bane
		localization = POD_summon_spirit_interaction_elemental_bane
	}


	send_options_exclusive = yes
	can_send = {
		custom_description = {
			text = send_option_selected
			OR = {
				scope:random ?= yes
				scope:gaian ?= yes
				scope:bane ?= yes
				scope:elemental_bane ?= yes
			}
		}
	}
	
	on_accept = {
		scope:actor = {
			save_scope_value_as = { name = is_summoned_spirit value = yes }
			if = {
				limit = { scope:random = yes }
				POD_splat_agnostic_energy_cost = { COUNT = 2 }
			}
			else_if = {
				limit = { scope:elemental_bane = yes }
				POD_splat_agnostic_energy_cost = { COUNT = 5 }
			}
			else = { POD_splat_agnostic_energy_cost = { COUNT = 3 } }


			switch = {
			trigger = yes

				scope:random = { POD_spawn_random_spirit_effect = yes }
				scope:gaian = { POD_spawn_gaian_spirit_effect = yes }
				scope:bane = { POD_spawn_bane_spirit_effect = yes }
				scope:elemental_bane = { POD_spawn_elemental_bane_spirit_effect = yes }
			}
		}
	}
	auto_accept = yes

	ai_frequency = 0
	ai_will_do = { base = 0 }
}




POD_spirit_pact_interaction = {
	category = interaction_category_spirit
	icon = POD_spirit_interaction
	desc = POD_spirit_pact_interaction_desc
	common_interaction = yes
	
	cooldown = { years = 5 }
	cost = { piety = 250 }
	
	is_shown = {
		scope:recipient = { has_trait = spirit }
		scope:actor = { NOT = { has_relation_spirit_companion = scope:recipient } }
		POD_interaction_is_shown = yes
	}
	
	is_valid_showing_failures_only = {
		POD_knows_spirit_language_trigger = yes
		scope:actor = { POD_is_demon_trigger = no }
		scope:actor = { POD_is_wraith_trigger = no }
		custom_tooltip = {
			text = POD_spirit_pact_interaction_tooltip
			scope:recipient = { num_of_relation_human_companion = 0 }
		}
		
		custom_tooltip = {
			text = POD_spirit_pact_interaction_tooltip2
			scope:recipient = { NOT = { has_character_flag = spirit_pact_block } }
		}
		trigger_if = {
			limit = { scope:recipient = { var:spirit_type = flag:spirit_innocent } }
			scope:actor = { POD_is_mokole_trigger = no }
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}
	
	on_accept = {
		scope:actor = {
			POD_form_spirit_pact_effect = { TARGET = scope:recipient }
			if = {
				limit = {
					OR = {
						POD_gains_faceofthegods_dharma_xp = yes
						POD_gains_faceofthegoddess_dharma_xp = yes
					}
				}
				POD_give_dharma_xp_effect = { VALUE = POD_kueijin_dharma_xp_value_very_high }
			}
		}
	}

	ai_accept = {
		base = -10
		
		#Opinion
		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.75
			desc = AI_OPINION_REASON
		}
		
		#Fera
		modifier = {
			add = 10
			scope:actor = { has_trait = fera }
			desc = ACTOR_FERA_REASON
		}
		
		#Vampire
		modifier = {
			add = -25
			scope:actor = { has_trait = vampire }
			scope:recipient = { has_religion = religion:gaia_religion }
			desc = ACTOR_VAMPIRE_REASON
		}
		
		#Same House
		modifier = {
			scope:recipient.house = scope:actor.house
			add = 10
			desc = IS_SAME_HOUSE_REASON
		}

		#Close Family
		modifier = {
			scope:actor = { is_close_family_of = scope:recipient }
			add = 25
			desc = CLOSE_FAMILY_REASON
		}

		#Rival
		modifier = {
			add = -50
			scope:recipient = { has_relation_rival = scope:actor }
			desc = WE_ARE_RIVALS
		}
		
		#Nemesis
		modifier = {
			add = -100
			scope:recipient = { has_relation_nemesis = scope:actor }
			desc = WE_ARE_NEMESES
		}

		#Soft cap
		modifier = {
			add = {
				value = scope:actor.num_of_relation_spirit_companion
				multiply = 15
				multiply = -1
			}
			scope:actor = { num_of_relation_spirit_companion >= 1 }
			desc = NUM_OF_SPIRIT_PACTS
		}


		#Put all perk or similar effects in this scripted_modifier
		POD_spirit_pact_acceptance_modifier = yes
	}

	ai_frequency = 0
	ai_will_do = { base = 0 }
}

POD_spirit_chiminage_interaction = {
	category = interaction_category_spirit
	icon = POD_spirit_interaction
	desc = POD_spirit_chiminage_interaction_desc
	common_interaction = yes
	
	cooldown = { years = 5 }
	
	is_shown = {
		scope:actor = { has_relation_spirit_companion = scope:recipient }
		scope:recipient = {
			has_trait = spirit
			POD_is_gaian_spirit_trigger = yes
			NOT = { POD_spirit_rank = 5 }
		}
		POD_interaction_is_shown = yes
	}
	
	is_valid_showing_failures_only = {
		scope:actor = { has_trait = fera }
		scope:actor = { POD_has_spendable_fera_renown_trigger = { COUNT = 1 } }
		POD_knows_spirit_language_trigger = yes
		POD_interaction_is_valid_showing_failures_only = yes
	}

	send_option = {
		is_valid = {
			scope:actor = {
				POD_has_spendable_fera_renown_trigger = { COUNT = 1 }
				piety >= 100
			}
		}
		flag = short
		localization = POD_spirit_chiminage_interaction_short
	}

	send_option = {
		is_valid = {
			scope:actor = {
				POD_has_spendable_fera_renown_trigger = { COUNT = 2 }
				piety >= 200
			}
		}
		flag = medium
		localization = POD_spirit_chiminage_interaction_medium
	}

	send_option = {
		is_valid = {
			scope:actor = {
				POD_has_spendable_fera_renown_trigger = { COUNT = 3 }
				piety >= 300
			}
		}
		flag = long
		localization = POD_spirit_chiminage_interaction_long
	}

	send_options_exclusive = yes
	can_send = {
		custom_description = {
			text = send_option_selected
			OR = {
				scope:short ?= yes
				scope:medium ?= yes
				scope:long ?= yes
			}
		}
	}
	
	on_accept = {
		switch = {
			trigger = yes

			scope:short = {
				save_temporary_scope_value_as = { name = chiminage_mult value = 1 }
				scope:actor = { add_character_modifier = { modifier = spirit_short_chiminage_modifier years = 5 } }
			}
			scope:medium = {
				save_temporary_scope_value_as = { name = chiminage_mult value = 2 }
				scope:actor = { add_character_modifier = { modifier = spirit_medium_chiminage_modifier years = 5 } }
			}
			scope:long = {
				save_temporary_scope_value_as = { name = chiminage_mult value = 3 }
				scope:actor = { add_character_modifier = { modifier = spirit_long_chiminage_modifier years = 5 } }
			}
		}
		scope:actor = {
			POD_fera_spend_renown_effect = { COUNT = scope:chiminage_mult }
			add_piety = { value = -100 multiply = scope:chiminage_mult }
		}
		scope:recipient = {
			save_temporary_scope_value_as = {
				name = old_spirit_rank
				value = POD_spirit_rank
			}
			add_trait_xp = { trait = spirit value = POD_spirit_chiminage_gain }
			hidden_effect = {
				if = {
					limit = { POD_spirit_rank > scope:old_spirit_rank }
					scope:actor = { POD_update_spirit_pact_modifiers = yes }
				}
			}
		}
	}

	auto_accept = yes

	ai_frequency = 0
	ai_will_do = { base = 0 }
}

POD_spirit_bane_empowerment_interaction = {
	category = interaction_category_spirit
	icon = POD_spirit_interaction
	desc = POD_spirit_bane_empowerment_interaction_desc
	common_interaction = yes
	
	cooldown_against_recipient = { years = 5 }
	
	is_shown = {
		scope:actor = { has_relation_spirit_companion = scope:recipient }
		scope:recipient = {
			has_trait = spirit
			POD_is_wyrm_spirit_trigger = yes
			NOT = { POD_spirit_rank = 5 }
		}
		POD_interaction_is_shown = yes
	}
	
	is_valid_showing_failures_only = {
		scope:actor = { POD_has_enough_splat_agnostic_energy_reserves = { AMOUNT = 1 } }
		POD_knows_spirit_language_trigger = yes
		POD_interaction_is_valid_showing_failures_only = yes
	}
	
	on_accept = {
		scope:actor = {
			POD_splat_agnostic_energy_cost = { COUNT = 1 }
			
			if = {
				limit = { scope:actor = { has_perk = sanguine_spirit_perk } }
				POD_increase_wounds_effect = { REASON = self_mutilation COUNT = POD_base_wound_damage }
				add_stress = miniscule_stress_gain
			}

			else = {
				random_list = {
					33 = {
						POD_increase_wounds_effect = { REASON = self_mutilation COUNT = POD_base_wound_damage }
						add_stress = miniscule_stress_gain
					}
					33 = {
						POD_increase_wounds_effect = { REASON = self_mutilation COUNT = POD_double_wound_damage }
						add_stress = minor_stress_gain
					}
					33 = {
						POD_increase_wounds_effect = { REASON = self_mutilation COUNT = POD_triple_wound_damage }
						add_stress = medium_stress_gain
					}
				}
			}
		}
		scope:recipient = {
			save_temporary_scope_value_as = {
				name = old_spirit_rank
				value = POD_spirit_rank
			}
			add_trait_xp = { trait = spirit value = 10 }
			hidden_effect = {
				if = {
					limit = { POD_spirit_rank > scope:old_spirit_rank }
					scope:actor = { POD_update_spirit_pact_modifiers = yes }
				}
			}
		}
	}

	auto_accept = yes

	ai_frequency = 0
	ai_will_do = { base = 0 }
}
