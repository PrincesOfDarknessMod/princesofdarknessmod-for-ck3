	}
	
	is_valid_showing_failures_only = {
		scope:actor = { 
			trigger_if = { 
				limit = { has_trait = hsien }
				custom_tooltip = {
					text = "i_am_empty_of_chi"
					NAND = { 
						has_character_modifier = yinchi0_modifier
						has_character_modifier = yangchi0_modifier
					}
				}
			}
			trigger_else = { NOT = { has_character_modifier = glamour0_modifier } }
		}
		scope:actor = { has_perk = realm_scene_1_perk }
		realm_scene = yes
		realm_scene_area = yes
	}
	
	send_option = {
		flag = positive
		localization = dream_time_perk_county_positive_modifier	
	}
	
	send_option = {
		flag = negative
		localization = dream_time_perk_county_negative_modifier	
	}
	
	send_option = {
		flag = remove
		localization = TITLE_VIEW_CLEAR_LAWS_ACCEPT	
	}

	on_accept = {
	   scope:actor = {
			POD_willpower_rouse_check_effect = yes
			if = {
				limit = { 
					scope:target = { tier = tier_empire }
				}
				add_piety = -1000
				POD_cantrip_cost_effect = { COUNT = 4 }
			}
			else_if = {
				limit = { 
					scope:target = { tier = tier_kingdom }
				}
				add_piety = -500
				POD_cantrip_cost_effect = { COUNT = 3 }
			}
			else_if = {
				limit = { 
					scope:target = { tier = tier_duchy }
				}
				add_piety = -250
				POD_cantrip_cost_effect = { COUNT = 2 }
			}
			else = {
				POD_cantrip_cost_effect = { COUNT = 1 }
			}
			duel = {
				desc = bunk.duel.desc
				skill = learning
				value = low_skill_rating
				50 = {
					desc = board_games.0001.tt.success
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					scope:target = {
						if = {
							limit = { scope:positive = yes }
							custom_tooltip = POD_dream_time_interaction_desc.positive
						}
						if = {
							limit = { scope:negative = yes }
							custom_tooltip = POD_dream_time_interaction_desc.negative
						}
						hidden_effect = { 
							every_de_jure_county = {
								POD_dream_time_effect = yes
							}
						}
					}
				}
				50 = {
					desc = board_games.0001.tt.failure
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier 
					}
					modifier = { OR = { NOT = { has_trait = siochain } global_banality = 0 } factor = global_banality }
				}
			}
		}
    }
	
	auto_accept = yes
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no 
		is_landed = yes 
	}
	ai_targets = {
		ai_recipients = self
	}
	ai_targets = {
		ai_recipients = primary_war_enemies
	}
	ai_frequency = 120
	ai_will_do = {
		base = 0
		
		modifier = { 
			add = 100
			scope:actor = scope:recipient
			scope:positive = yes
		}
		modifier = {
			add = 45
			NOT = { scope:actor = scope:recipient }
			scope:negative = yes
			scope:actor = {
			    is_at_war_with = scope:recipient
			}
		}
		modifier = {
			factor = 2
			NOT = { scope:actor = scope:recipient }
			scope:negative = yes
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
		modifier = {
			factor = 0
			scope:recipient = {
				is_ai = no
			}
		}
	}  
}
