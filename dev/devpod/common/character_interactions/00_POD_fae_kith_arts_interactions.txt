		localization = dream_time_perk_county_positive_modifier

		is_shown = {

			}
		}
		flag = negative
		localization = dream_time_perk_county_negative_modifier	
	}
	
	send_option = {
		is_shown = {
			scope:recipient = {
				OR = { 
					has_character_modifier = dream_time_perk_county_positive_modifier 
					has_character_modifier = dream_time_perk_county_negative_modifier 
				}
			}
		}
		flag = remove
		localization = TITLE_VIEW_CLEAR_LAWS_ACCEPT	
	}

	   scope:actor = {
			POD_willpower_rouse_check_effect = yes

				limit = {

			if = {
				limit = { scope:positive = yes }

						modifier = hurt_opinion
						opinion = -5

				}
				if = {
					limit = { has_character_modifier = dream_time_perk_character_negative_modifier }
