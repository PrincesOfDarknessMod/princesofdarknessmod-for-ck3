# POD Schemes
# Vamphunt mortals
# Sparc

vamphunt_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	#interface_priority = 120
	scheme = vamphunt
	
	ai_frequency = 0
	
	#icon = "gfx/interface/icons/character_interactions/vamphunt.dds"

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		scope:recipient = { is_vampire_trigger = no }
		scope:actor = { is_vampire_trigger = yes  }
		scope:actor = { is_ai = no }
	}

	is_valid_showing_failures_only = {
		custom_description = {
            object = scope:actor
            text = "I_must_be_at_the_same_location"
			scope:actor = { exists = location }
			scope:recipient = { exists = location }
            scope:actor = { location = scope:recipient.location }
        }
		custom_description = {
			object = scope:actor
			text = "I_can_only_plot_one_vamphunt_at_time"
			scope:actor = {
				NOT = {
					any_scheme = {
						scheme_type = vamphunt
					}
				}
			}
		}
		custom_description = {
			object = scope:recipient
			text = "blood_is_holy"
			scope:recipient = {
				NOR = {
					has_trait = truefaith
					has_trait = beatific
					has_trait = orison_sanguine
				}
			}
		}
		scope:actor = {
			can_start_scheme = {
				type = vamphunt
				target = scope:recipient
			}
		}
		scope:recipient = {
			is_landed = no
		}
	}

	desc = {
		triggered_desc = {
			trigger = {
				scope:actor = {
					#NOT = { is_at_war_with = scope:recipient }
					can_start_scheme = {
						type = vamphunt
						target = scope:recipient
					}

				}
			}
			desc = scheme_interaction_tt_vamphunt_approved
		}

	}

	on_accept = {
		hidden_effect = {
			scope:actor = {
				start_scheme = {
					type = vamphunt
					target = scope:recipient
				}
			}
		}
	}

	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
