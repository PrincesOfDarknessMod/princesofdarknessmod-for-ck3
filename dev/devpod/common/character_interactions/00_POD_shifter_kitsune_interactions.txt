
POD_possession_interaction = {
	category = interaction_category_werewolf
	icon = POD_kitsune_interaction
	desc = POD_possession_interaction_desc
	
	cooldown = { years = 1 }
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { POD_splat_has_gnosis_trigger = yes }
		scope:actor = { has_perk = possession_perk }
		scope:recipient = { 
			trigger_if = {
				limit = { num_of_relation_possessesing_wraith > 0 }
				has_relation_possessesing_wraith = scope:actor
			}
			NOT = {
				has_opinion_modifier = {
					target = scope:actor
					modifier = POD_complete_possession_opinion
				}
			}
		}
	}
	
	is_valid_showing_failures_only = {
		scope:actor = { NOT = { has_character_modifier = POD_gnosis_0_modifier } }
		scope:actor = {
			custom_description = {
				text = POD_not_in_possession
				num_of_relation_possessed_target = 0 
			}
			trigger_if = {
				limit = {  
					scope:recipient = {
						POD_can_be_possessed_trigger = no
					}
				}
				has_perk = vector_perk
			}
		}
		scope:recipient = {
			NOT = { has_perk = banish_the_dead_perk }
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}
	
	on_accept = {
		scope:actor = { POD_wraith_willpower_check_effect = yes }
		scope:actor = { POD_gnosis_loss_check_effect = yes }
		scope:recipient = {
			trigger_event = POD_wraith_possession.100
		}
		custom_tooltip = POD_possess_interaction_desc
	}
	
	auto_accept = yes
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no 
		is_landed = yes 
	}
	ai_targets = {
		ai_recipients = liege
	}
	ai_targets = {
		ai_recipients = neighboring_rulers
		ai_recipients = peer_vassals
		max = 10
	}
	ai_frequency = 72
	ai_will_do = {
		base = 0
		
		opinion_modifier = {
			opinion_target = scope:recipient
			multiplier = -0.25
		}
		modifier = {
		    add = 50
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
		modifier = {
			factor = 0
			scope:recipient = {
				is_ai = no
			}
		}
	}
}

POD_blessing_the_blade_interaction = {
	category = interaction_category_werewolf
	icon = POD_kitsune_interaction
	desc = POD_blessing_the_blade_interaction_desc

	target_type = artifact
	target_filter = recipient_artifacts
	
	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { has_perk = blessing_the_blade_perk }
		scope:actor = { POD_splat_has_gnosis_trigger = yes }
	}

	can_be_picked = {
		scope:target = {
			artifact_slot_type = primary_armament
			NOT = { has_variable = enchanted_artifact }
			NOT = { has_variable = summoned_artifact }
			NOT = { has_variable = temporary_vessel }
			NOT = { has_variable = wraith_jade }
			NOT = { has_variable = alchemy }
		}
	}

	is_valid_showing_failures_only= {
		scope:actor = { NOT = { has_character_modifier = POD_gnosis_0_modifier } }
		scope:recipient = { 
			custom_description = {
				text = "demand_artifact_interaction_valid_tt_1"
				has_any_artifact = yes
			}
		}
	}
	
	send_option = {
		flag = prowess
		localization = POD_interaction_artifact_prowess_modifier
	}
	
	send_option = {
		flag = diplomacy
		localization = POD_interaction_artifact_diplomacy_modifier
	}
	
	send_option = {
		flag = martial
		localization = POD_interaction_artifact_martial_modifier
	}
	
	send_option = {
		flag = stewardship
		localization = POD_interaction_artifact_stewardship_modifier
	}
	
	send_option = {
		flag = intrigue
		localization = POD_interaction_artifact_intrigue_modifier
	}
	
	send_option = {
		flag = learning
		localization = POD_interaction_artifact_learning_modifier
	}

	on_accept = {
		scope:actor = { POD_willpower_rouse_check_effect = yes }
		scope:actor = { POD_gnosis_loss_check_effect = yes }
		scope:target = {
			if = {
				limit = { scope:prowess = yes }
				add_artifact_modifier = artifact_prowess_1_modifier
			}
			if = {
				limit = { scope:diplomacy = yes }
				add_artifact_modifier = artifact_diplomacy_1_modifier
			}
			if = {
				limit = { scope:martial = yes }
				add_artifact_modifier = artifact_martial_1_modifier
			}
			if = {
				limit = { scope:stewardship = yes }
				add_artifact_modifier = artifact_stewardship_1_modifier
			}
			if = {
				limit = { scope:intrigue = yes }
				add_artifact_modifier = artifact_intrigue_1_modifier
			}
			if = {
				limit = { scope:learning = yes }
				add_artifact_modifier = artifact_learning_1_modifier
			}
			set_variable = enchanted_artifact
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_kitsune_madness_perk_interaction = {
	category = interaction_category_werewolf
	icon = POD_kitsune_interaction
	desc = POD_madness_perk_interaction_desc
	options_heading = POD_madness_perk_interaction_options_heading

	cooldown = { months = 3 }
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_perk = kitsune_madness_perk }
		scope:actor = { POD_splat_has_gnosis_trigger = yes }
	}

	is_valid_showing_failures_only = {
		custom_description = {
            object = scope:recipient
            text = "strong_mental_defenses_trigger"
            strong_mental_defenses_interaction_trigger = yes
        }
		scope:actor = { NOT = { has_character_modifier = POD_gnosis_0_modifier } }
		scope:recipient = {
			NOT = { has_trait = torpor }
			NOT = { has_trait = automaton }
			NOT = { has_trait = wight }
			NOT = { has_trait = barghest }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }		
			POD_is_not_clone_trigger = yes
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				NOT = { has_trait = depressed }
			}
		}	
		flag = depressed
		localization = POD_exorcism_depressed
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				NOT = { has_trait = lunatic }
			}
		}	
		flag = lunatic
		localization = POD_exorcism_lunatic
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				NOT = { has_trait = possessed }
			}
		}	
		flag = possessed
		localization = POD_exorcism_possessed
	}
	
	on_accept = {
		scope:actor = { 
			POD_willpower_rouse_check_effect = yes
			POD_gnosis_loss_check_effect = yes	
			duel = {
				skill = learning
				target = scope:recipient
				70 = {
					desc = POD_kueijin_spy.duel.success
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					scope:recipient = {
						add_stress = medium_stress_gain

						if = {
							limit = {
								scope:depressed = yes
							}
							scope:recipient = {
								add_trait = depressed_1
							}
						}	

						if = {
							limit = {
								scope:lunatic = yes
							}
							scope:recipient = {
								add_trait = lunatic_1
							}
						}	

						if = {
							limit = {
								scope:possessed = yes
							}
							scope:recipient = {
								add_trait = possessed_1
							}
						}
					}
				}
				30 = {
					desc = POD_kueijin_spy.duel.failure
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier 
					}
					send_interface_toast = {
						title = POD_kueijin_spy.duel.failure
						left_icon = scope:actor
						right_icon = scope:recipient
						
						custom_tooltip = POD_kueijin_spy.duel.tt.attacker_lost
						add_stress = medium_stress_gain
					}
				}
			}
		}
		scope:recipient = { 
			add_opinion = {
				target = scope:actor
				modifier = hurt_opinion
				opinion = -30
			}
		}
	}
	
	auto_accept = yes
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no 
		is_landed = yes 
	}
	ai_targets = {
		ai_recipients = primary_war_enemies
	}
	ai_frequency = 48
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 45
			scope:actor = {
			    is_at_war_with = scope:recipient
			}
		}
		modifier = {
			factor = 2
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
		modifier = {
			factor = 0
			scope:recipient = {
				is_ai = no
			}
		}
	}
}
