POD_share_taint_interaction = {
	category = interaction_category_demon
	common_interaction = yes
	interface_priority = 100
	desc = POD_share_taint_interaction_desc
	icon = POD_defiled_interaction

	cooldown = { months = 6 }

	ai_min_reply_days = 4
	ai_max_reply_days = 9

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			has_perk = defilements_share_taint_perk
		}
		scope:recipient = {
			POD_is_yamaking_trigger = no
		}
		POD_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			NOT =  { is_at_war_with = scope:recipient }
			trigger_if = {
				limit = {
					has_perk = defilements_impose_taint_perk
					scope:recipient = { is_imprisoned_by = scope:actor }
				}
				scope:recipient = { is_imprisoned_by = scope:actor }
			}
			trigger_else_if = {
				limit = { has_perk = defilements_your_fellow_akuma_perk }
				scope:recipient = {
					opinion = {
						target = scope:actor
						value >= 10
					}
				}
			}
			trigger_else_if = {
				limit = { has_perk = defilements_harmless_taint_perk }
				scope:recipient = {
					opinion = {
						target = scope:actor
						value >= 25
					}
				}
			}
			trigger_else_if = {
				limit = { has_perk = defilements_hide_taint_perk }
				scope:recipient = {
					opinion = {
						target = scope:actor
						value >= 40
					}
				}
			}
			trigger_else = {
				scope:recipient = {
					opinion = {
						target = scope:actor
						value >= 50
					}
				}
			}
		}
		scope:recipient = {
			NOT = { has_strong_hook = scope:actor }
			POD_is_spirit_trigger = no
			POD_is_possessed_trigger = no
			POD_is_akuma_trigger = no
			NOT = { has_trait = cultist } #Cultists already have taint
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:actor = {
			POD_chance_concession_effect = { VALUE = 1 }
		}
		scope:recipient = {
			set_character_faith = scope:actor.faith
		}
	}


	auto_accept = yes

	ai_frequency = 0

	ai_will_do = {
		base = 0
	}
}
