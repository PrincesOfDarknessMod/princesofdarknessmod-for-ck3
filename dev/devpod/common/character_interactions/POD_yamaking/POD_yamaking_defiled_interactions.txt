POD_suggest_defilement_interaction = {
	category = interaction_category_demon
	common_interaction = yes
	interface_priority = 100
	desc = POD_suggest_defilement_interaction_desc
	icon = POD_yamaking_interaction

	ai_min_reply_days = 4
	ai_max_reply_days = 9

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			has_trait = yamaking
		}
		scope:recipient = {
			NOT = { has_trait = yamaking }
			NOT = { has_relation_worshipped_yamaking = scope:actor }
		}
		POD_hostile_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			NOT =  { is_at_war_with = scope:recipient }
		}
		scope:recipient = {
			NOT = { has_strong_hook = scope:actor }
			POD_is_spirit_trigger = no
		}
		custom_tooltip = {
			text = POD_suggest_defilement_interaction_limit
			scope:actor = { num_of_relation_worshipping_defiled < max_defiled_cap }
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	send_options_exclusive = no

	send_option = {
		is_valid = {
			scope:actor = { gold >= scope:recipient.major_gold_value }
			scope:recipient = { has_trait = greedy }
		}
		flag = defilement_gold
		localization = POD_suggest_defilement_interaction_gold
	}
	send_option = {
		is_valid = {
			scope:actor = { panoplies_lifestyle_perk_points >= 2 }
			scope:recipient = { has_trait = ambitious }
		}
		flag = defilement_power
		localization = POD_suggest_defilement_interaction_power
	}
	send_option = {
		is_valid = {
			scope:actor = { prestige >= 1000 }
			scope:recipient = { has_trait = arrogant }
		}
		flag = defilement_glory
		localization = POD_suggest_defilement_interaction_glory
	}

	on_accept = {
		scope:actor = {
			save_scope_as = yamaking
			switch = {
				trigger = yes
				defilement_gold = { pay_short_term_gold = { target = scope:recipient gold = scope:recipient.major_gold_value } }
				power = { add_panoplies_lifestyle_perk_points = -2 }
				defilement_glory = { add_prestige = -1000 }
			}
			add_hook = {
				target = scope:recipient
				type = defiled_hook
			}
		}
		scope:recipient = {
			add_trait_force_tooltip = defiled
			set_character_faith = scope:actor.faith
			every_relation = {
				type = worshipped_yamaking
				save_scope_as = yamaking
				scope:recipient = { remove_relation_worshipped_yamaking = scope:yamaking }
			}
			set_relation_worshipped_yamaking = scope:actor

			switch = {
				trigger = yes
				defilement_glory = { add_prestige = 1000 }
				power = { add_defilements_lifestyle_perk_points = 2 }
				fallback = {}
			}
		}
	}

	ai_accept = {
		base = -50

		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.5
			desc = AI_SIMPLE_OPINION_REASON
		}

		modifier = {
			add = -75
			scope:recipient ?= {
				OR = {
					faith = { has_doctrine = special_doctrine_against_to_yomi_wan_hostility }
				}
			}
			desc = DEFILEMENT_YOU_ARE_YAMA_KING
		}

		modifier = {
			add = -50
			scope:recipient ?= {
				faith = { has_doctrine = special_doctrine_allied_to_yomi_wan_hostility }
				NOT = { faith = scope:actor.faith }
			}
			desc = DEFILEMENT_I_ALREADY_HAVE_A_MASTER
		}

		modifier = {
			add = -40
			scope:recipient ?= {
				NOR = { # Sounds too good to be true, but I don't know Yama Kings exist
					faith = { has_doctrine = special_doctrine_against_to_yomi_wan_hostility }
					faith = { has_doctrine = special_doctrine_allied_to_yomi_wan_hostility }
				}
			}
			desc = DEFILEMENT_YOU_ARE_UNKNOWN
		}

		modifier = {
			add = 50
			scope:recipient ?= {
				faith = { has_doctrine = special_doctrine_allied_to_yomi_wan_hostility }
				faith = scope:actor.faith
			}
			desc = DEFILEMENT_MY_MASTER
		}

		modifier = {
			add = scope:actor.dread
			scope:recipient ?= {
				ai_boldness <= 0
			}
			desc = DEFILEMENT_I_AM_SCARED
		}

		modifier = {
			add = 25
			scope:defilement_gold ?= yes
			desc = DEFILEMENT_GREEDY
		}
		modifier = {
			add = 25
			scope:defilement_power ?= yes
			desc = DEFILEMENT_POWER
		}
		modifier = {
			add = 25
			scope:defilement_glory ?= yes
			desc = DEFILEMENT_GLORY
		}

		modifier = {
			scope:actor = { has_character_modifier = POD_dealing_with_competition_modifier }
			add = 35
			desc = POD_DEALING_WITH_COMPETITION_MODIFIER_DESC
		}

		modifier = {
			scope:actor = { has_character_modifier = POD_claiming_rightful_throne_modifier }
			add = 100
			desc = POD_CLAIMING_RIGHTFUL_THRONE_MODIFIER_DESC
		}
	}

	ai_frequency = 0

	ai_will_do = {
		base = 0
	}
}

POD_break_prisoner_interaction = {
	category = interaction_category_demon
	icon = POD_yamaking_interaction
	desc = POD_break_prisoner_interaction_desc

	cooldown_against_recipient = { years = 1 }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_trait = yamaking }
		scope:recipient = { is_imprisoned_by = scope:actor }
		POD_hostile_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		POD_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		custom_tooltip = POD_break_prisoner_interaction_effect
		random = {
			chance = {
				value = scope:actor.dread
				max = 25
				add = scope:recipient.stress
				add = 1
				max = 100
			}
			scope:recipient = {
				set_character_faith = scope:actor.faith
				every_relation = {
					type = worshipped_yamaking
					save_scope_as = yamaking
					scope:recipient = { remove_relation_worshipped_yamaking = scope:yamaking }
				}
			}
		}
		scope:actor = { POD_secondary_effects_from_torturing = yes }
		scope:recipient = {
			add_character_flag = {
				flag = is_being_tortured
				months = 6 # Players in MP can keep the event window open...
			}
		}
		scope:actor = { save_scope_as = bg_override_char }
		torture_interaction_actor_effect = yes
		show_as_tooltip = {
			scope:recipient = {
				add_character_modifier = {
					modifier = recently_tortured
					years = 5
				}
			}
		}
		if = {
			limit = {
				scope:actor = {
					domicile ?= { has_domicile_parameter = estate_torture_grants_influence }
				}
			}
			scope:actor = { custom_tooltip = estate_torture_grants_influence_desc }
		}
	}

	auto_accept = yes

	ai_will_do = {
		base = 0
	}
}
