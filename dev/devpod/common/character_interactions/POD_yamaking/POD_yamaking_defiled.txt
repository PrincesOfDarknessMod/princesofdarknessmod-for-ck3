POD_suggest_defilement_interaction = {
	category = interaction_category_demon
	common_interaction = yes
	interface_priority = 100
	desc = POD_suggest_defilement_interaction_desc
	icon = POD_demon_interaction

	ai_min_reply_days = 4
	ai_max_reply_days = 9

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			has_trait = yamaking
		}
		scope:recipient = {
			NOT = { has_trait = yamaking }
			NOT = { has_relation_worshipped_yamaking = scope:actor }
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			NOT =  { is_at_war_with = scope:recipient }
		}
		scope:recipient = {
			NOT = { has_strong_hook = scope:actor }
			POD_is_spirit_trigger = no
		}
		custom_tooltip = {
			text = POD_suggest_defilement_interaction_limit
			scope:actor = { num_of_relation_worshipping_defiled < max_defiled_cap }
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	send_options_exclusive = no

	send_option = {
		is_valid = { 
			scope:actor = { gold >= scope:recipient.major_gold_value } 
			scope:recipient = { has_trait = greedy }
		}
		flag = defilement_gold
		localization = POD_suggest_defilement_interaction_gold
	}
	send_option = {
		is_valid = { 
			# scope:actor = { panoply_lifestyle_perk_points >= 2 } 
			scope:recipient = { has_trait = ambitious }
		}
		flag = defilement_power
		localization = POD_suggest_defilement_interaction_power
	}
	send_option = {
		is_valid = { 
			scope:actor = { prestige >= 1000 } 
			scope:recipient = { has_trait = arrogant }
		}
		flag = defilement_glory
		localization = POD_suggest_defilement_interaction_glory
	}

	on_accept = {
		scope:actor = {
			save_scope_as = yamaking
			switch = {
				trigger = yes
				defilement_gold = { pay_short_term_gold = { target = scope:recipient gold = scope:recipient.major_gold_value } }
				# power = { add_panoply_lifestyle_perk_points = -2 }
				defilement_glory = { add_prestige = -1000 }
			}
			add_hook = {
				target = scope:recipient
				type = defiled_hook
			}
		}
		scope:recipient = {
			add_trait_force_tooltip = defiled
			set_character_faith = scope:actor.faith 
			every_relation = {
				type = worshipped_yamaking
				save_scope_as = yamaking
				scope:recipient = { remove_relation_worshipped_yamaking = scope:yamaking }
			}
			set_relation_worshipped_yamaking = scope:actor
			
			switch = {
				trigger = yes
				defilement_glory = { add_prestige = 1000 }
				# power = { add_defilements_lifestyle_perk_points = 2 }
				fallback = {}
			}
		}
	}

	ai_accept = {
		base = -50

		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.5
			desc = AI_SIMPLE_OPINION_REASON
		}

		modifier = {
			add = -75
			scope:recipient ?= { 
				OR = {
					faith = { has_doctrine = special_doctrine_against_to_yomi_wan_hostility } 
				}
			}
			desc = DEFILEMENT_YOU_ARE_YAMA_KING
		}

		modifier = {
			add = -50
			scope:recipient ?= { 
				faith = { has_doctrine = special_doctrine_allied_to_yomi_wan_hostility } 
				NOT = { faith = scope:actor.faith }
			}
			desc = DEFILEMENT_I_ALREADY_HAVE_A_MASTER
		}

		modifier = {
			add = -40
			scope:recipient ?= { 
				NOR = { # Sounds too good to be true, but I don't know Yama Kings exist
					faith = { has_doctrine = special_doctrine_against_to_yomi_wan_hostility } 
					faith = { has_doctrine = special_doctrine_allied_to_yomi_wan_hostility }
				}
			}
			desc = DEFILEMENT_YOU_ARE_UNKNOWN
		}

		modifier = {
			add = 50
			scope:recipient ?= { 
				faith = { has_doctrine = special_doctrine_allied_to_yomi_wan_hostility } 
				faith = scope:actor.faith
			}
			desc = DEFILEMENT_MY_MASTER
		}

		modifier = {
			add = scope:actor.dread
			scope:recipient ?= { 
				ai_boldness <= 0
			}
			desc = DEFILEMENT_I_AM_SCARED
		}

		modifier = {
			add = 25
			scope:defilement_gold ?= yes
			desc = DEFILEMENT_GREEDY
		}
		modifier = {
			add = 25
			scope:defilement_power ?= yes
			desc = DEFILEMENT_POWER
		}
		modifier = {
			add = 25
			scope:defilement_glory ?= yes
			desc = DEFILEMENT_GLORY
		}
	}

	ai_frequency = 0

	ai_will_do = {
		base = 0
	}
}
