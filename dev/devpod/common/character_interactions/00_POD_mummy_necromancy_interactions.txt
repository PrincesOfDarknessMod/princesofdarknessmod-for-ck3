POD_fertility_of_osiris_interaction = {
	category = interaction_category_mummy
	desc = POD_fertility_of_osiris_interaction_desc
	options_heading = POD_fertility_of_osiris_interaction_options_heading
	icon = POD_hekau_interaction
	
	is_shown = {
		scope:actor = { 
			OR = {
				has_trait = mummy
				has_character_modifier = wepauwet_modifier
				uses_cairo_osiris_main_reward = yes
			}
		}
		scope:actor = { has_perk = fertility_of_osiris_perk }
	}
	
	is_valid = {	
		scope:recipient = {
			is_female = yes
			is_pregnant = no
			is_visibly_fertile = yes
			custom_tooltip = {
				text = has_any_dead_husband
				any_spouse = {
					even_if_dead = yes
					is_alive = no
					is_female = no
				}
			}
		}
	}
	
	is_valid_showing_failures_only = {	
		scope:actor = { 
			trigger_if = {
				limit = { is_east_mummy_trigger = yes }
				NOT = { has_character_modifier = yangchi0_modifier }
			}
			trigger_if = {
				limit = { is_amenti_trigger = yes }
				NOT = { has_character_modifier = sekhem0_modifier }
			}
			trigger_if = {
				limit = { has_character_modifier = wepauwet_modifier }
				NOT = { has_character_modifier = POD_gnosis_0_modifier }
			}
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			NOT = { has_trait = automaton }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }	
			POD_is_not_clone_trigger = yes	
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}
	
	send_option = {
		flag = spouse_1
		localization = POD_fertility_of_osiris_spouse_1
	}
	
	send_option = {
		is_shown = {
			scope:recipient = {
				any_spouse = {
					even_if_dead = yes
					is_alive = no
					is_female = no
					count >= 2
				}
			}
		}
		flag = spouse_2
		localization = POD_fertility_of_osiris_spouse_2
	}
	
	send_option = {
		is_shown = {
			scope:recipient = {
				any_spouse = {
					even_if_dead = yes
					is_alive = no
					is_female = no
					count >= 3
				}
			}
		}
		flag = spouse_3
		localization = POD_fertility_of_osiris_spouse_3
	}
	
	send_option = {
		is_shown = {
			scope:recipient = {
				any_spouse = {
					even_if_dead = yes
					is_alive = no
					is_female = no
					count >= 4
				}
			}
		}
		flag = spouse_4
		localization = POD_fertility_of_osiris_spouse_4
	}
	
	send_option = {
		is_shown = {
			scope:recipient = {
				any_spouse = {
					even_if_dead = yes
					is_alive = no
					is_female = no
					count >= 5
				}
			}
		}
		flag = spouse_5
		localization = POD_fertility_of_osiris_spouse_5
	}

	on_accept = {
		scope:actor = { 
			if = {
				limit = { has_trait = mummy }
				POD_hekau_cost_effect = { COUNT = 1 }
			}
			else_if = {
				limit = { has_character_modifier = wepauwet_modifier }
				POD_gnosis_loss_check_effect = yes
			}
			else = { POD_willpower_cost_effect = { COUNT = 2 } }
		}
		scope:recipient = {
			every_spouse = {
				even_if_dead = yes
				limit = {
					is_alive = no
					is_female = no
				}
				add_to_list = dead_spouse
			}
		}

		if = {
			limit = { scope:spouse_1 = yes }
			ordered_in_list = {
				list = dead_spouse
				position = 0
				order_by = age
				save_scope_as = target_spouse
			}
		}
		if = {
			limit = { scope:spouse_2 = yes }
			ordered_in_list = {
				list = dead_spouse
				position = 1
				order_by = age
				save_scope_as = target_spouse
			}
		}
		if = {
			limit = { scope:spouse_3 = yes }
			ordered_in_list = {
				list = dead_spouse
				position = 2
				order_by = age
				save_scope_as = target_spouse
			}
		}
		if = {
			limit = { scope:spouse_4 = yes }
			ordered_in_list = {
				list = dead_spouse
				position = 3
				order_by = age
				save_scope_as = target_spouse
			}
		}
		if = {
			limit = { scope:spouse_5 = yes }
			ordered_in_list = {
				list = dead_spouse
				position = 4
				order_by = age
				save_scope_as = target_spouse
			}
		}
		show_as_tooltip = {
			scope:recipient = {
				make_pregnant = { father = scope:target_spouse }
			}
		}
		hidden_effect = {
			scope:target_spouse =  {
				save_scope_as = cloned_character
				
				create_character = {
					save_scope_as = clone
					age = scope:cloned_character.age
					faith = scope:cloned_character.faith
					gender = scope:cloned_character
					random_traits = no
					dynasty = none
					employer = scope:actor
				}
				
				scope:clone = {
					copy_traits = scope:cloned_character
					copy_inheritable_appearance_from = scope:cloned_character
					POD_copy_base_skills_effect = {
						CLONED_CHAR = scope:cloned_character
						CLONE = scope:clone
					}
				}
			
				scope:recipient = {
					make_pregnant = { father = scope:clone }
					scope:clone = { silent_disappearance_effect = yes }
					set_variable = {
						name = fertility_of_osiris
						value = scope:cloned_character
						days = 730
					}
				}
			}
		}
	
		if = {
			limit = { NOT = { scope:actor = scope:recipient } }
			scope:recipient = { 
				add_opinion = {
					target = scope:actor
					modifier = pleased_opinion
					opinion = 5
				}
			}
		}
	}
	
	auto_accept = yes
	
	
	ai_will_do = {
		base = 0
	}
}
