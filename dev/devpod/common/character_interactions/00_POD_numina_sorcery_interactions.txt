POD_sorcery_steel_artifact_interaction = {
    category = interaction_category_mage
	icon = POD_sorcery_interaction
    desc = POD_sorcery_steel_artifact_interaction_desc
    
    target_type = artifact
	target_filter = recipient_artifacts
	is_shown = {
        scope:actor = { has_perk = conjuration_5_perk }
    }
	
	can_be_picked = {
        scope:target = { 
			NOT = { has_variable = summoned_artifact }
			NOT = { has_variable = temporary_vessel }
			NOT = { has_variable = wraith_jade }
			NOT = { has_variable = alchemy }
			NOT = { artifact_type = animal }
		}
    }

    is_valid_showing_failures_only = {
		trigger_if = { 
            limit = { scope:actor = scope:recipient }
        }
        trigger_else_if = { 
            limit = { scope:actor.location = { squared_distance = { target = scope:recipient.location value > squared_distance_huge } } }
            scope:actor = { has_perk = conjuration_6_perk }
        }
        trigger_else_if = { 
            limit = { scope:actor.location = { squared_distance = { target = scope:recipient.location value > squared_distance_medium } } }
            scope:actor = { has_perk = conjuration_5_perk }
        }
		trigger_else = {}
		scope:recipient = { 
			custom_description = {
				text = "demand_artifact_interaction_valid_tt_1"
				has_any_artifact = yes
			}
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}

	cooldown = { months = 12 }
	cost = { piety = sorcery_massive_piety_cost }

	auto_accept = yes

    on_accept = {
	    scope:actor = { 
			stress_impact = { base = sorcery_extreme_stress_cost }
            duel = {
				skill = learning
				target = scope:recipient
				25 = {
					desc = board_games.0001.tt.success
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
                    modifier = {
                        has_perk = conjuration_6_perk
                        add = 50
                    }
                    POD_commit_theft_effect = yes 
					scope:target = {
						set_owner = {
							target = scope:actor
							history = {
								location = scope:recipient.location
								actor = scope:recipient
								recipient = scope:actor
								type = stolen
							}
						}
					}
					if = {
						limit = {
							NOT = { has_variable = stolen_artifact }
						}
						set_variable = {
							name = stolen_artifact
							value = 1
						}
					}
					else = {
						change_variable = {
							name = stolen_artifact
							add = 1
						}
					}
				}
				50 = {
					desc = board_games.0001.tt.failure
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier 
					}
				}
			}
		}			
	    scope:recipient = { 
			add_opinion = {
				target = scope:actor
				modifier = claimed_my_artifact_opinion
				opinion = -75
			}
			# Recipient grows closer to forming a rivalry with the actor
			if = {
				limit = {
					can_set_relation_potential_rival_trigger = { 
						CHARACTER = scope:actor
					}
				}
				set_relation_potential_rival = scope:actor
			}
			add_personal_artifact_claim = scope:target
		}	
	}
}
