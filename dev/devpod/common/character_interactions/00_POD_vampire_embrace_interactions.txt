embrace_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	#interface_priority = 83
	scheme = embrace
	
	ai_frequency = 0

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { is_vampire_trigger = yes }
		scope:recipient = { POD_can_be_embraced_scheme_trigger = yes }
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		#Catubodua can't Embrace
		scope:actor = { NOT = { this = character:791232 } }
		#Adisa can only embrace females
		OR = {
			scope:actor = { NOT = { this = character:791706 } }
			scope:recipient = { is_female = yes }
		}
		scope:recipient = { POD_is_clone_trigger = no }
		scope:recipient = { is_imprisoned = no }
	}

	is_valid_showing_failures_only = {
		scope:actor = { 
			NOT = { has_trait = abomination }
			is_vampgeneration16_trigger = no
			trigger_if = { 
				limit = { POD_is_thinblooded_trigger = yes }
				NOT = { has_trait = thinbloodalchemydiscipline }
			}
		}
		custom_description = {
			object = scope:actor
			text = "I_can_only_plot_one_embrace_at_time"
			scope:actor = {
				NOT = {
					any_scheme = {
						scheme_type = embrace
					}
				}
			}
		}
		custom_description = {
			object = scope:recipient
			text = "blood_is_holy"
			scope:recipient = {
				NOR = {
					POD_has_true_faith_trigger = yes
					has_trait = orison_sanguine
				}
			}
		}
		scope:actor = {
			can_start_scheme = {
				type = embrace
				target_character = scope:recipient
			}
		}
		trigger_if = { # Tooltip Only
			limit = { scope:recipient = { POD_is_supehunter_trigger = yes } }
			scope:recipient = { is_imprisoned_by = scope:actor } 
		}
		scope:recipient = {
			is_landed = no
			NOT  = { has_character_modifier = boon_of_anubis_perk_modifier }
		}
	}

	desc = {
		# triggered_desc = {
			# trigger = { scope:recipient.age >= scope:actor.age }
			# desc = embrace_childe_older_than_you_tt
		# }
		triggered_desc = {
			trigger = {
				scope:actor = {
					NOT = { is_at_war_with = scope:recipient }
					can_start_scheme = {
						type = embrace
						target_character = scope:recipient
					}

				}
			}
			desc = scheme_interaction_tt_embrace_approved
		}

	}

	on_accept = {
		hidden_effect = {
			scope:actor = {
				start_scheme = {
					type = embrace
					target_character = scope:recipient
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

embracerecommendation_interaction = {
	category = interaction_category_vampire
	desc = embracerecommendation_interaction_desc
	ai_frequency = 0
	icon = embrace_interaction
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { 
			OR = { 
				has_trait = vampire 
				has_trait = demon
			}
		}
		scope:recipient = { is_vampire_trigger = yes }
		scope:recipient = { NOT = { this = character:791232 } } #Catubodua can't Embrace
		scope:recipient = { POD_is_clone_trigger = no }
		scope:recipient = { is_ai = yes }
	}

	is_valid = {
		scope:recipient = { 
			OR = {
				trigger_if = {
					limit = { scope:actor = { has_trait = demon } }
					has_relation_worshipped_demon = scope:actor
				}
				trigger_if = {
					limit = {
						NOT = { is_great_grandchild_of = scope:actor }
						NOT = { is_grandchild_of = scope:actor }
					}
					is_child_of = scope:actor
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = { 
			NOT = { has_trait = abomination }
			is_vampgeneration16_trigger = no
			trigger_if = { 
				limit = { POD_is_thinblooded_trigger = yes }
				NOT = { has_trait = thinbloodalchemydiscipline }
			}
			
			is_landed = yes
			is_available = yes
			is_imprisoned = no
			is_busy_in_events_localised = yes
			
			is_vassal_or_below_of = scope:actor
			
			custom_tooltip = {
				text = liege_vassal_contract_forbids_progeny
				NOT = { vassal_contract_has_flag = progeny_rights_none }
				NOT = { vassal_contract_has_flag = progeny_rights_ghouls }
			}
		}
		NOT = { scope:actor = { is_at_war_with = scope:recipient } }
		trigger_if = {
			limit = { exists = scope:secondary_recipient }
			scope:secondary_recipient = {
				is_available = yes
				trigger_if = {
					limit = { scope:actor = { this = character:791706 } }
					is_female = yes
				}
				custom_description = {
					object = scope:secondary_recipient
					text = "blood_is_holy"
					NOR = {
						POD_has_true_faith_trigger = yes
						has_trait = orison_sanguine
					}
				}
				NOT = { has_trait = shapeshifter }
				NOT = { has_character_modifier = boon_of_anubis_perk_modifier }
			}
		}
	}
	
	populate_recipient_list = {
		scope:actor = {
			every_courtier_or_guest = {
				limit = {
					POD_can_be_embraced_scheme_trigger = yes
					is_adult = yes
				}
				add_to_list = characters
			}
		}
		scope:recipient = {
			every_courtier_or_guest = {
				limit = {
					POD_can_be_embraced_scheme_trigger = yes
					is_adult = yes
				}
				add_to_list = characters
			}
		}
	}
	
	cost = {
		prestige = major_prestige_value
	}

	on_accept = {
		scope:recipient  = { save_scope_as = vampiresire }
		scope:secondary_recipient = {
			POD_remove_supernatural_relation_effect = yes
			
			save_scope_as = created_vampirechilde
			
			POD_become_vampire_effect = yes
			POD_set_parent_effect = { PARENT = scope:vampiresire }
			POD_character_vampire_inherit_generation_effect = yes
			POD_character_vampire_inherit_clan_effect = yes
			if = { 
				limit = { scope:vampiresire = { religion = { is_in_family = rf_kindred } } }
				set_character_faith = scope:vampiresire.faith
			}
			else = { 
				POD_character_vampire_faith_initialization_effect = yes
			}
			POD_character_vampire_bloodpotency_initialization_effect = yes
			POD_character_vampire_bloodhunger_initialization_effect = yes
			POD_character_masquerade_initialization_effect = yes
			POD_character_vampire_discipline_initialization_effect = yes
			
			POD_character_vampire_clan_special_effect = yes
			POD_salubri_caste_new_vampire_initialization_effect = yes
		
			if = {
				limit = { is_landed = yes }
				change_government = traditions_government
			}
			remove_disease_trait_effect = yes
			add_character_flag = immune_to_disease
		}
	}
	
	ai_accept = {
		base = 0 # Try to make it 0 for most interactions
		
		opinion_modifier = { # Opinion Factor
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 1.0
			desc = AI_OPINION_REASON
		}
		
		opinion_modifier = { # Opinion Factor
			who = scope:recipient
			opinion_target = scope:secondary_recipient
			multiplier = 0.25
			desc = AI_OPINION_REASON
		}
	}
}
