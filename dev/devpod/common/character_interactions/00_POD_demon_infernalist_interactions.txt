POD_bind_demon_interaction = {
	category = interaction_category_demon
	icon = POD_infernalist_interaction
	desc = POD_bind_demon_interaction_desc

	cooldown = { years = 5 }
	
	is_shown = {
		scope:actor = { OR = { is_demonologist_trigger = yes has_character_modifier = umbra_demon_binding_modifier has_perk = reap_the_well_perk } }
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { has_trait = demon }
	}

	is_valid_showing_failures_only = {
		scope:actor = { 
			custom_description = {
				object = scope:recipient
				text = "has_full_true_name_hook"
				has_hook_of_type = {
					target = scope:recipient
					type = true_name_hook
				}
			}
		}
		custom_description = {
			text = POD_bind_interaction_limit
			scope:actor = {
				num_of_relation_bound_demon < max_wraith_cap
			}
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }	
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}

	on_accept = {
		custom_tooltip = POD_bind_interaction_desc
		scope:actor = { 
			POD_willpower_rouse_check_effect = yes
			give_infernalist_secret_or_trait_effect = yes
			duel = {
				skill = learning
				value = scope:recipient.stewardship
				40 = {
					desc = POD_wraith_binding.200.wp.success.desc
					modifier = {
						has_character_modifier = umbra_demon_binding_modifier
						add = 50
					}
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					# Success 
					custom_tooltip = POD_wraith_binding.200.wp.tt_success
					
					if = {
						limit = { has_relation_worshipped_demon = scope:recipient }
						remove_relation_worshipped_demon = scope:recipient
					}
					if = {
						limit = { has_relation_demonic_master = scope:recipient }
						remove_relation_demonic_master = scope:recipient
					}
					set_relation_bound_demon = scope:recipient
					remove_hook = {
						target = scope:recipient
						type = true_name_hook
					}
					add_hook = {
						target = scope:recipient
						type = necromancy_bind_permanent_hook
					}
					reverse_add_opinion = {
						modifier = POD_binding_opinion
						target = scope:recipient
					}
					add_diabolist_xp = { VALUE = 5 }
				}
				60 = {
					desc = POD_wraith_binding.200.wp.failure.desc
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier
						multiplier = scope:recipient.resolve_level
					}
					# Failure
					custom_tooltip = POD_wraith_binding.200.wp.tt_failure
				}				
			}
		}
		
		scope:recipient = { 
			add_opinion = {
				target = scope:actor
				modifier = hurt_opinion
				opinion = -100
			}
			progress_towards_rival_effect = {
				CHARACTER = scope:actor
				REASON = rival_cast_spell
				OPINION = 0
			}
		}
	}
	
	auto_accept = yes
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no 
		is_landed = yes 
	}
	ai_targets = {
		ai_recipients = liege
	}
	ai_targets = {
		ai_recipients = neighboring_rulers
		ai_recipients = peer_vassals
		max = 10
	}
	ai_frequency = 72
	ai_will_do = {
		base = 0
		
		opinion_modifier = {
			opinion_target = scope:recipient
			multiplier = -0.25
		}
		modifier = {
		    add = 50
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
		modifier = {
			factor = 0
			scope:recipient = {
				is_ai = no
			}
		}
	}
