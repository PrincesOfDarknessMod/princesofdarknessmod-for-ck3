POD_ghoulify_mortal_interaction = {
	category = interaction_category_vampire
	#common_interaction = yes
	#interface_priority = 30
	scheme = POD_ghoulify_mortal
	icon = POD_ghoulify_interaction
	
	ai_frequency = 0

	is_shown = {
		scope:actor != scope:recipient
		scope:actor = {
			POD_splat_can_create_ghouls_naturally = yes
			is_adult = yes
		}
		scope:recipient = {
			trigger_if = {
				limit = { scope:actor = { has_trait = azhi_dahaka } }
				POD_can_be_ghouled_azhi_dahaka_scheme_trigger = yes
			}
			trigger_else = { POD_can_be_ghouled_scheme_trigger = yes }
			is_ai = yes
			NOT = { has_relation_ghoul_domitor = scope:actor }
			is_imprisoned = no
		}
		POD_hostile_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			is_adult = yes
		}
		scope:actor = {
			can_start_scheme = {
				type = POD_ghoulify_mortal
				target_character = scope:recipient
			}
		}
		custom_description = {
			object = scope:recipient
			text = "blood_is_holy"
			scope:recipient = {
				NOR = {
					POD_has_true_faith_trigger = yes
					has_trait = orison_sanguine
				}
			}
		}
		
		custom_description = {
			text = POD_ghoulify_mortal_interaction_bp_one
			scope:actor = { blood_potency_value > 0 }
		}
		
		scope:actor = { POD_has_enough_blood_for_discipline = { AMOUNT = 1 } }
		scope:recipient = {
			is_ruler = no
		}

		
		POD_interaction_is_valid_showing_failures_only = yes
	}
	
	is_valid = {
	}

	desc = POD_ghoulify_mortal_interaction_desc

	on_accept = {
	
		hidden_effect = {
			scope:actor = {
				start_scheme = {
					type = POD_ghoulify_mortal
					target_character = scope:recipient
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

				}

				name = bloodbag

			}
		}

			scope:actor = {
				start_scheme = {
					type = POD_ghoul_ghoulify
					target_character = scope:recipient
				}
			}
		}
	}
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_manipulate_bound_ghouls = {
	category = interaction_category_vampire
	icon = POD_ghoulify_interaction
	desc = POD_manipulate_bound_ghouls_desc
	common_interaction = yes
	
	is_shown = {
		scope:actor = { has_POD_vamp_blood_trigger = yes }
		scope:actor = scope:recipient
		scope:actor = {
			num_of_relation_ghoul_thrall > 0
		}
	}
	
	populate_recipient_list = {
		scope:actor = {
			every_relation = {
				type = ghoul_thrall
				limit = { is_alive = yes }
				
				add_to_list = characters
			}
		}
	}
	
	is_valid_showing_failures_only = {
		POD_interaction_is_valid_showing_failures_only = yes
	}
	
	send_option = {
		is_shown = {
			exists = scope:secondary_recipient
			scope:actor = { POD_has_enough_blood_for_discipline = { AMOUNT = 1 } }
			scope:secondary_recipient = {
				OR = {
					POD_health_has_superficial_wound_trigger = yes
					has_trait = maimed
				}
			}
		}
		flag = heal
		localization = POD_manipulate_bound_ghouls_heal_option
	}
	
	send_option = {
		is_shown = {
			exists = scope:secondary_recipient
			scope:actor = {
				OR = {
					choleric_lifestyle_xp > 999
					choleric_lifestyle_perk_points > 0
					phlegmatic_lifestyle_xp > 999
					phlegmatic_lifestyle_perk_points > 0
					sanguine_lifestyle_xp > 999
					sanguine_lifestyle_perk_points > 0
					melancholy_lifestyle_xp > 999
					melancholy_lifestyle_perk_points > 0
					animal_lifestyle_xp > 999
					animal_lifestyle_perk_points > 0
					has_perk = cycle_of_the_midgard_serpent_perk
				}
			}
		}
		flag = traindiscipline
		localization = POD_manipulate_bound_ghouls_traindiscipline_option
	}
	
	send_option = {
		is_shown = { exists = scope:secondary_recipient }
		flag = giveblood
		localization = POD_manipulate_bound_ghouls_giveblood_option
	}

	send_option = {
		is_shown = {
			exists = scope:secondary_recipient
			scope:actor = { gold >= medium_gold_value }
		}
		flag = rewardgold
		localization = POD_manipulate_bound_ghouls_goldreward_option
	}
	
	send_option = {
		is_shown = {
			exists = scope:secondary_recipient
			scope:secondary_recipient = { gold >= medium_gold_value }
			NOT = {
				scope:secondary_recipient = {
					has_opinion_modifier = {
						target = scope:actor
						modifier = POD_ghoulthrallgolddemand_opinion
					}
				}
			}
		}
		flag = demandgold
		localization = POD_manipulate_bound_ghouls_demandgold_option
	}
	
	send_option = {
		is_shown = {
			scope:secondary_recipient ?= { is_married = yes }
		}
		flag = divorce
		localization = POD_manipulate_bound_ghouls_divorce_option
	}
	
	send_option = {
		is_shown = { exists = scope:secondary_recipient }
		flag = release
		localization = POD_manipulate_bound_ghouls_release_option
	}
	
	on_accept = {
		scope:secondary_recipient = { save_scope_as = ghoulthrall }
		scope:actor = {
			switch = {
				trigger = yes
				scope:heal = {
					if = {
						limit = { scope:ghoulthrall = { has_trait = maimed } }
						POD_small_risk_of_increasing_masquerade_exposure = yes
					}
					else = {
						POD_minimal_risk_of_increasing_masquerade_exposure = yes
					}

					POD_hunger_rouse_check_effect = yes
						
					scope:ghoulthrall = {
						remove_trait = maimed
						POD_blood_heal_wound_effect = yes
						hidden_effect = {
							remove_opinion = {
								target = scope:actor
								modifier = POD_ghoulthrall_opinion
							}
						}
						add_opinion = {
							target = scope:actor
							modifier = POD_ghoulthrall_opinion
						}
					}
				}
				scope:traindiscipline = {
					POD_hunger_rouse_check_effect = yes
					save_scope_as = disciplinetrainer
					scope:ghoulthrall = { save_scope_as = disciplinetrainee }
					trigger_event = {
						id = POD_disciplinetrain.100
						days = 7
					}
				}
				scope:giveblood = {
					POD_hunger_rouse_check_effect = yes
					hidden_effect = {
						remove_hook = {
							target = scope:ghoulthrall
							type = strong_bloodoath_hook
						}
					}
					add_hook = {
						target = scope:ghoulthrall
						type = strong_bloodoath_hook
					}
					scope:ghoulthrall = {
						hidden_effect = {
							remove_opinion = {
								target = scope:actor
								modifier = POD_ghoulthrall_opinion
							}
						}
						add_opinion = {
							target = scope:actor
							modifier = POD_ghoulthrall_opinion
						}
					}
				}
				scope:rewardgold = {
					pay_short_term_gold = {
						target = scope:ghoulthrall
						gold = medium_gold_value
					}
					scope:ghoulthrall = {
						add_opinion = {
							target = scope:actor
							modifier = POD_ghoulthrallgoldreward_opinion
						}
					}
				}
				scope:demandgold = {
					scope:ghoulthrall = {
						pay_short_term_gold = {
							target = scope:actor
							gold = medium_gold_value
						}
						add_opinion = {
							target = scope:actor
							modifier = POD_ghoulthrallgolddemand_opinion
						}
					}
				}
				scope:divorce = {
					scope:ghoulthrall = { divorce = primary_spouse }
				}
				scope:release = {
					custom_tooltip = POD_ghoulthrall_release_option.tt
					scope:ghoulthrall = {
						POD_clear_bound_ghoul_variables_and_relations = yes
						trigger_event = {
							id = POD_ghoul_maintenance.100
							days = 365
						}
					}
				}
			}
		}
	}

	ai_frequency = 0

	auto_accept = yes

	ai_will_do = {
		base = 0
	}
}

POD_beg_for_vitae_interaction = {
	category = interaction_category_ghoul
	icon = POD_ghoulify_interaction
	desc = POD_beg_for_vitae_interaction_desc

	is_shown = {
		scope:actor != scope:recipient
		scope:recipient = {
			POD_splat_can_create_ghouls_naturally = yes
			is_adult = yes
		}
		scope:actor = { 
			num_of_relation_ghoul_domitor = 0
			trigger_if = {
				limit = { NOT = { has_trait = ghoul } }
				POD_can_be_ghouled_scheme_trigger = yes 
			}
			dynasty ?= { NOT = { has_variable = POD_kill_upon_being_bound } }
		}
		POD_hostile_interaction_is_shown = yes # So hunters can still do this
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			is_unbondable_trigger = no
			
			trigger_if = {
				limit = { NOT = { has_character_flag = charjoin_become_ghoul_flag } }
				
				POD_can_beg_for_ghouldom_from_generation = {
					GENERATION = scope:recipient.POD_vampgeneration
				}
			}
		}
		scope:recipient = {
			is_imprisoned = no
			NOT = { has_faith = faith:viaservilis }
		}
	}

	cost = {
		prestige = {
			switch = {
				trigger = scope:recipient.POD_vampgeneration
				3 = { value = 5000 }
				4 = { value = 2000 }
				5 = { value = 1750 }
				6 = { value = 1500 }
				7 = { value = 1250 }
				8 = { value = 1000 }
				9 = { value = 750 }
				10 = { value = 500 }
				11 = { value = 250 }
				fallback = { value = 0 }
			}
			if = {
				limit = { scope:actor = { has_character_flag = charjoin_become_ghoul_flag } }
				multiply = 0
			}
		}
	}

	on_accept = {
		scope:actor = {
			if = {
				limit = { NOT = { has_character_flag = charjoin_become_ghoul_flag } }
				
				POD_fame_loss_from_begging_for_ghouldom = yes
			}
			else = {
				remove_character_flag = charjoin_become_ghoul_flag
			}
			
			POD_become_ghoul_effect = yes
			POD_become_bound_ghoul_effect = { DOMITOR = scope:recipient }
		}
	}

	ai_frequency = 0

	auto_accept = yes

	ai_will_do = {
		base = 0
	}
}
