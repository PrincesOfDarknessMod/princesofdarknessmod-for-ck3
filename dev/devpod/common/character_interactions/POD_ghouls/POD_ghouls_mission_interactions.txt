					task_contract_employer = scope:recipient
					POD_ghoul_mission_can_be_finished = yes
				}
				complete_task_contract = success_standard
			}
		}
	}

	#ai_frequency = 0

	auto_accept = yes

		base = 0
	}
}

POD_ghoul_mission_indoctrinate_plant_interaction = {
	category = interaction_category_ghoul
	icon = POD_ghoulify_interaction
	desc = POD_ghoul_mission_indoctrinate_plant_interaction_desc

	is_shown = {
		scope:actor != scope:recipient
		scope:recipient = {
			has_trait = mortal
		}
		scope:actor = {
			any_character_active_contract = {
				task_contract_type = POD_ghoul_mission_indoctrinate_plant
				POD_ghoul_mission_can_be_finished = yes
			}
		}
		POD_hostile_interaction_is_shown = yes
	}
	
	is_valid_showing_failures_only = {
		scope:recipient = {
			is_a_thrall = no
			# TODO YK: no defiled
			NOT = { has_trait = truepossessed }
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:actor = {
			random_character_active_contract = {
				task_contract_type = POD_ghoul_mission_indoctrinate_plant
				limit = { POD_ghoul_mission_can_be_finished = yes }
				save_scope_as = task_contract
				task_contract_employer = { save_scope_as = domitor }
			}
			duel = {
				skills = { diplomacy intrigue }
				target = scope:recipient
				50 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					scope:task_contract = { complete_task_contract = success_standard }
					
					if = {
						limit = { scope:domitor = { is_ruler = yes } }
						scope:domitor = { add_courtier = scope:recipient }
						scope:recipient = {
							add_opinion = {
								modifier = love_opinion
								opinion = 75
								target = scope:domitor
							}
							set_character_faith = scope:domitor.faith
							hidden_effect = { return_to_court = yes }
						}
					}
					else = {
						hidden_effect = {
							scope:recipient = { death = { death_reason = death_disappearance } }
						}
					}
				}
				50 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier
					}
					scope:task_contract = { complete_task_contract = failure_standard }
				}
			}
		}
	}

	#ai_frequency = 0

	auto_accept = yes

	ai_will_do = {
		base = 0
	}
}
