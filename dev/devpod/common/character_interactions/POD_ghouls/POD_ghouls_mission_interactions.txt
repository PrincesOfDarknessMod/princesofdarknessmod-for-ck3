POD_ghoul_mission_cover_masquerade_interaction = {
	category = interaction_category_vampire
	icon = POD_cover_masquerade_interaction
	desc = POD_ghoul_mission_cover_masquerade_interaction_desc
	options_heading = POD_cover_masquerade_interaction_options_heading

	use_diplomatic_range = no

	cooldown = { months = 12 }
	
	is_shown = {
		scope:recipient = {
			has_relation_ghoul_thrall = scope:actor
			POD_splat_has_masquerade_trigger = yes
		}
		scope:actor = {
			var:POD_ghoul_mission ?= flag:cover_up_the_masquerade
		}
	}

	is_valid_showing_failures_only = {
		custom_tooltip = {
			text = "no_masquerade"
			scope:recipient = {
				NAND = {
					faith = { has_doctrine_parameter = vampirethemasquerade_none }
					masquerade_exposure_level < 5
					NOT = { has_trait = paranoid }
				}
			}
		}
		scope:recipient = { POD_is_forever_infamous_trigger = no }
		custom_tooltip = {
			text = POD_ghoul_mission_cover_masquerade_interaction_cap
			scope:recipient = { masquerade_exposure_level > minimum_masquerade_exposure }
		}
	}

	send_option = {
		is_shown = {
			scope:actor = {
				OR = {
					has_trait = dominateadvanced
					has_perk = the_forgetful_mind_perk
				}
			}
		}
		is_valid = {
			scope:actor = { POD_has_enough_blood_for_discipline = { AMOUNT = 1 } }
		}
		flag = dominate
		localization = POD_cover_masquerade_dominate
	}
	
	send_option = {
		is_valid = {
			scope:actor = { gold >= major_gold_value }
		}
		flag = bribe
		localization = POD_cover_masquerade_bribe
	}

	send_option = {
		is_shown = {
			scope:actor = {
				OR = {
					has_trait = presenceadvanced
					has_perk = majesty_perk
				}
			}
		}
		is_valid = {
			scope:actor = { POD_has_enough_blood_for_discipline = { AMOUNT = 1 } }
		}
		flag = presence
		localization = POD_cover_masquerade_presence
	}
	
	send_option = {
		is_shown = {
			scope:actor = {
				has_trait = bloodsorceryadvanced
			}
		}
		is_valid = {
			scope:actor = { POD_has_enough_blood_for_discipline = { AMOUNT = 1 } }
		}
		flag = bloodsorcery
		localization = POD_cover_masquerade_bloodsorcery
	}
	
	on_accept = {
		POD_cover_up_masquerade_effect = {
			USER = scope:actor
			BENEFACTOR = scope:recipient
		}
		scope:actor = {
			POD_complete_ghoul_mission_effect = { MISSION = flag:cover_up_the_masquerade }
		}
	}
	
	auto_accept = yes

	ai_will_do = {
		base = 0
	}
}

POD_ghoul_mission_indoctrinate_plant_interaction = {
	category = interaction_category_vampire
	icon = POD_ghoulify_interaction
	desc = POD_ghoul_mission_indoctrinate_plant_interaction_desc

	is_shown = {
		scope:actor != scope:recipient
		scope:recipient = {
			has_trait = mortal
		}
		scope:actor = { var:POD_ghoul_mission ?= flag:indoctrinate_plant }
	}

	on_accept = {
		scope:actor = {
			duel = {
				skills = { diplomacy intrigue }
				target = scope:recipient
				50 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					POD_complete_ghoul_mission_effect = { MISSION = flag:indoctrinate_plant }
					random_relation = {
						type = ghoul_domitor
						add_courtier = scope:recipient
						scope:recipient = {
							add_opinion = {
								modifier = love_opinion
								opinion = 75
								target = PREV
							}
							set_character_faith = PREV.faith
							hidden_effect = { return_to_court = yes }
						}
					}
				}
				50 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier
					}
					POD_fail_ghoul_mission_effect = { MISSION = flag:indoctrinate_plant }
				}
			}
		}
	}

	ai_frequency = 0

	auto_accept = yes

	ai_will_do = {
		base = 0
	}
}
