POD_healing_waters_character_interaction = {
	category = interaction_category_fae
	icon = POD_aquis_interaction
	desc = POD_healing_waters_character_interaction_desc
	options_heading = POD_healing_waters_character_interaction_options_heading
	use_diplomatic_range = no

	cooldown = { months = realm_time_monthly_cooldown }	
	# cost = { piety = miniscule_piety_value }
	
	is_shown = {
		scope:actor = { has_POD_fae_blood_trigger = yes }
		scope:actor = { has_perk = healing_waters_perk }
	}

	is_valid = {
		scope:recipient = {
			custom_description = {
				text = POD_has_main_trait
				object = scope:recipient
				OR = {
					has_trait = scarred
					has_trait = disfigured
					has_trait = blind
					has_trait = one_eyed
					has_trait = maimed
					has_trait = one_legged
					has_trait = eunuch
				
					has_trait = wounded
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = { 
			trigger_if = { 
				limit = { has_trait = hsien }
				custom_tooltip = {
					text = "i_am_empty_of_chi"
					NAND = { 
						has_character_modifier = yinchi0_modifier
						has_character_modifier = yangchi0_modifier
					}
				}
			}
			trigger_else = { NOT = { has_character_modifier = glamour0_modifier } }
		}
		realm_actor = yes
		realm_fae = yes
		realm_scene = yes
		scope:recipient = {
			NOT = { has_trait_with_flag = can_not_cure }
			NOT = { has_character_modifier = eye_of_sakhmet_modifier }
			NOT = { has_character_modifier = thin_blood_perk_modifier }
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
			POD_is_not_trapped_in_umbra_trigger = { ACTOR = scope:actor }
			custom_tooltip = { text = has_artifact_fans_of_ma_yi_yuan NOT = { any_equipped_character_artifact = { has_variable = fans_of_ma_yi_yuan } } }
		}
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = scarred
			}
		}
		is_valid = {
			scope:recipient = { 
				custom_description = {
					text = POD_permanent_scarred
					object = scope:recipient
					NOT = { has_character_flag = POD_permanent_scarring_flag }
				}
			}
		}
		flag = scarred
		localization = POD_heal_scarred
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = disfigured
			}
		}
		is_valid = {
			scope:recipient = { 
				custom_description = {
					text = POD_permanent_disfigured
					object = scope:recipient
					NOT = { has_character_flag = POD_permanent_disfigured_flag }
				}
			}
		}
		flag = disfigured
		localization = POD_heal_disfigured
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = wounded
			}
		}	
		flag = wounded
		localization = POD_heal_wounded
	}

	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = blind
			}
		}
		is_valid = {
			scope:recipient = { 
				custom_description = {
					text = POD_permanent_blind
					object = scope:recipient
					NOT = { has_character_flag = POD_permanent_blind }
				}
			}
		}
		flag = blind
		localization = POD_heal_blind
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = maimed
			}
		}
		is_valid = {
			scope:recipient = { 
				custom_description = {
					text = POD_permanent_maimed
					object = scope:recipient
					NOT = { has_character_flag = POD_permanent_maimed }
				}
			}
		}
		flag = maimed
		localization = POD_heal_maimed
	}	
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = one_legged
			}
		}
		is_valid = {
			scope:recipient = { 
				custom_description = {
					text = POD_permanent_one_legged
					object = scope:recipient
					NOT = { has_character_flag = POD_permanent_one_legged }
				}
			}
		}
		flag = one_legged
		localization = POD_heal_one_legged
	}

	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = one_eyed
			}
		}
		is_valid = {
			scope:recipient = { 
				custom_description = {
					text = POD_permanent_one_eyed
					object = scope:recipient
					NOT = { has_character_flag = POD_permanent_one_eyed }
				}
			}
		}
		flag = one_eyed
		localization = POD_heal_one_eyed
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = eunuch
			}
		}	
		is_valid = {
			scope:recipient = { 
				custom_description = {
					text = POD_permanent_eunuch
					object = scope:recipient
					NOT = { has_character_flag = POD_permanent_eunuch }
				}
			}
		}
		flag = eunuch
		localization = POD_heal_eunuch
	}
	
	on_accept = {
		scope:actor = { POD_willpower_rouse_check_effect = yes }
		scope:recipient = { play_music_cue = "mx_cue_peace_ensues" }
		if = {
			limit = {
				scope:scarred = yes
			}
			scope:recipient = {
				remove_trait = scarred
			}
			scope:actor = { POD_cantrip_cost_effect = { COUNT = 1 } }
		}	

		if = {
			limit = {
				scope:disfigured = yes
			}
			scope:recipient = {
				remove_trait = disfigured
			}
			scope:actor = { POD_cantrip_cost_effect = { COUNT = 2 } }
		}
		
		if = {
			limit = {
				scope:wounded = yes
			}
			scope:recipient = {
				POD_remove_wound_trait_effect = yes
			}
			scope:actor = { POD_cantrip_cost_effect = { COUNT = 1 } }
		}		

		if = {
			limit = {
				scope:one_eyed = yes
			}
			scope:recipient = {
				remove_trait = one_eyed
			}
			scope:actor = { POD_cantrip_cost_effect = { COUNT = 2 } }
		}	

		if = {
			limit = {
				scope:blind = yes
			}
			scope:recipient = {
				remove_trait = blind
				add_trait = one_eyed
			}
			scope:actor = { POD_cantrip_cost_effect = { COUNT = 2 } }
		}	

		if = {
			limit = {
				scope:one_legged = yes
			}
			scope:recipient = {
				remove_trait = one_legged
			}
			scope:actor = { POD_cantrip_cost_effect = { COUNT = 3 } }
		}	

		if = {
			limit = {
				scope:maimed = yes
			}
			scope:recipient = {
				remove_trait = maimed
			}
			scope:actor = { POD_cantrip_cost_effect = { COUNT = 3 } }
		}	

		if = {
			limit = {
				scope:eunuch = yes
			}
			scope:recipient = {
				remove_trait = eunuch
			}
			scope:actor = { POD_cantrip_cost_effect = { COUNT = 3 } }
		}
		
		if = {
			limit = { NOT = { scope:actor = scope:recipient } }
			scope:recipient = { 
				add_opinion = {
					target = scope:actor
					modifier = pleased_opinion
					opinion = 15
				}
			}
		}
	}
	
	auto_accept = yes

	ai_potential = {
		# The basic requirements
		is_imprisoned = no
		is_landed = yes
	}
	ai_targets = {
		ai_recipients = self
	}
	ai_targets = {
		ai_recipients = courtiers
		ai_recipients = peer_vassals
		max = 10
	}
	ai_frequency = 12
	ai_will_do = {
		base = 0
		modifier = {
			add = 100
			scope:recipient = {
				OR = {
					this = scope:actor
					is_knight_of = scope:actor
				}
			}
		}
	}
}

	category = interaction_category_fae
	icon = POD_pyros_interaction
	desc = POD_strings_of_the_soul_interaction_desc
	use_diplomatic_range = no
	
	# cooldown = { months = realm_time_monthly_cooldown }
	# cost = { piety = miniscule_piety_value }
	
	scheme = PODsoul_bridge

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_POD_fae_blood_trigger = yes }
		scope:actor = { has_perk = strings_of_the_soul_perk }
	}

	is_valid_showing_failures_only = {
		scope:actor = { 
			trigger_if = { 
				limit = { has_trait = hsien }
				custom_tooltip = {
					text = "i_am_empty_of_chi"
					NAND = { 
						has_character_modifier = yinchi0_modifier
						has_character_modifier = yangchi0_modifier
					}
				}
			}
			trigger_else = { NOT = { has_character_modifier = glamour0_modifier } }
		}
		realm_actor = yes
		realm_fae = yes
		realm_scene = yes
		scope:recipient = {
			NOT = { has_trait = torpor }
			NOT = { has_trait = statue }
			NOT = { has_trait = wight }
			NOT = { has_trait = barghest }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }	
			POD_is_not_clone_trigger = yes	
			POD_is_not_trapped_in_umbra_trigger = { ACTOR = scope:actor }
			custom_tooltip = { text = has_artifact_fans_of_ma_yi_yuan NOT = { any_equipped_character_artifact = { has_variable = fans_of_ma_yi_yuan } } }
		}
	}
	
	send_option = {
		is_shown = {
			NOR = {
				scope:recipient = { is_imprisoned_by = scope:actor }
				scope:recipient = { has_relation_necromancer = scope:actor }
				scope:actor = { 
					has_hook_of_type = {
						target = scope:recipient
						type = strong_bloodoath_hook
					}
				}
			}
		}
		is_valid = {
			scope:actor = {
				can_start_scheme = {
					type = PODsoul_bridge
					target = scope:recipient
				}
			}
		}
		flag = scheme_attack
		localization = POD_interaction_scheme_attack
	}
	
	send_option = {
		is_shown = {
			NOR = {
				scope:recipient = { is_imprisoned_by = scope:actor }
				scope:recipient = { has_relation_necromancer = scope:actor }
				scope:actor = { 
					has_hook_of_type = {
						target = scope:recipient
						type = strong_bloodoath_hook
					}
				}
			}
		}
		is_valid = {
			scope:recipient = { can_start_single_combat_trigger = yes }
			custom_description = {
				object = scope:actor
				text = "I_must_be_at_the_same_location"
				scope:actor = { exists = location }
				scope:recipient = { exists = location }
				scope:actor = { location = scope:recipient.location }
			}
		}
		flag = duel_attack
		localization = POD_interaction_duel_attack
	}
	
	send_option = {
		is_shown = {
			OR = {
				scope:recipient = { is_imprisoned_by = scope:actor }
				scope:recipient = { has_relation_necromancer = scope:actor }
				scope:actor = { has_strong_usable_hook = scope:recipient }
			}
		}
		flag = automatic
		localization = POD_interaction_automatic
	}

	on_accept = {
		scope:actor = { POD_willpower_rouse_check_effect = yes }
		scope:actor = { POD_cantrip_cost_effect = { COUNT = 1 } }
		if = {
			limit = {
				scope:scheme_attack = yes
			}
			scope:actor = {
				start_scheme = {
					type = PODsoul_bridge
					target = scope:recipient
				}
			}
		}
	
		if = {
			limit = {
				scope:duel_attack = yes
			}
			scope:actor = {
				configure_start_single_combat_effect = {
					SC_INITIATOR = scope:actor
					SC_ATTACKER = scope:actor
					SC_DEFENDER = scope:recipient
					FATALITY = no
					FIXED = no
					LOCALE = throne_room
					OUTPUT_EVENT = kueijin_scheme.201
					INVALIDATION_EVENT = tribal.1022
				}
			}
		}
		
		if = {
			limit = {
				scope:automatic = yes
			}
			scope:actor = { trigger_event = kueijin_scheme.200 }
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
