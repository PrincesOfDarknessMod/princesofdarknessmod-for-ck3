POD_taming_the_beast_interaction = {
	category = interaction_category_vampire
	icon = POD_abombwe_interaction
	desc = POD_taming_the_beast_interaction_desc
	options_heading = POD_taming_the_beast_interaction_options_heading

	send_option = {
		flag = prowess
		localization = POD_taming_the_beast_prowess
	}

	send_option = {
		flag = health
		localization = POD_taming_the_beast_health
	}

	send_option = {
		flag = martial
		localization = POD_taming_the_beast_martial
	}

	cooldown = { months = 36 }

	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { has_abombwe_trigger = yes }

		#Don't show this interaction until the player has the right Abombwe perk
		scope:actor = { has_perk = taming_the_beast_perk }
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			POD_has_enough_blood_for_discipline = { AMOUNT = 1 }
			NOT = { has_character_modifier = bloodpotency0_modifier }
		}
		POD_self_targeted_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:actor = {
			POD_hunger_rouse_check_effect = yes
			add_stress = medium_stress_gain

		}
		if = {
			limit = {
				scope:prowess = yes
			}
			scope:actor = {
				add_character_modifier = {
					modifier = taming_the_beast_prowess_modifier
					days = 90
				}
			}
		}
		if = {
			limit = {
				scope:health = yes
			}
			scope:actor = {
				add_character_modifier = {
					modifier = taming_the_beast_health_modifier
					days = 90
				}
			}
		}
		if = {
			limit = {
				scope:martial = yes
			}
			scope:actor = {
				add_character_modifier = {
					modifier = taming_the_beast_martial_modifier
					days = 90
				}
			}
		}
	}

	auto_accept = yes

	ai_potential = {
		# The basic requirements
		is_imprisoned = no
		is_landed = yes
	}
	ai_targets = {
		ai_recipients = self
	}
	ai_frequency = 36
	ai_will_do = {
		base = 100
	}
}

POD_predators_mastery_interaction = {
	category = interaction_category_vampire
	icon = POD_abombwe_interaction
	desc = POD_predators_mastery_interaction_desc
	
	is_shown = {
		scope:actor != scope:recipient
		scope:actor = { has_abombwe_trigger = yes }

		#Don't show this interaction until the player has the right perk perk
		scope:actor = { has_perk = predators_mastery_perk }

		OR = {
			scope:recipient = { is_foreign_court_or_pool_guest_of = scope:actor }
			scope:recipient = { is_courtier_of = scope:actor }
		}
		
		POD_hostile_mental_interaction_is_shown = yes
	}
	
	is_valid_showing_failures_only = {
		POD_interaction_is_valid_showing_failures_only = yes
	}

	cooldown = { years = 1 }

	on_accept = {
		scope:actor = {
			custom_tooltip = POD_predators_mastery_interaction_accept
		}
		scope:recipient = {
			add_opinion = {
				target = scope:actor
				modifier = hurt_opinion
				opinion = -15
			}
		}
		hidden_effect = {
			scope:actor = {
				trigger_event = POD_predators_mastery.100
			}
		}
	}

	auto_accept = yes
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no
		is_landed = yes
	}
	ai_targets = {
		ai_recipients = courtiers
		max = 10
	}
	ai_frequency = 72
	ai_will_do = {
		base = 0
		
		opinion_modifier = {
			opinion_target = scope:recipient
			multiplier = -0.25
		}
		modifier = {
			add = 50
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
		modifier = {
			factor = 0
			scope:recipient = {
				is_ai = no
			}
		}
	}
}
