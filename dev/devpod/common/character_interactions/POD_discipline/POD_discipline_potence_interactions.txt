
POD_fist_of_the_titans_interaction = {
	category = interaction_category_vampire
	icon = POD_potence_interaction
	desc = POD_fist_of_the_titans_interaction_desc
	
	cooldown = { years = 5 }
	
	cost = { prestige = 200 }

	is_shown = {
		scope:actor != scope:recipient
		scope:actor = { has_potence_trigger = yes }
		scope:actor = { has_perk = fist_of_the_titans_perk }
		scope:recipient = { is_playable_character = yes }
		scope:recipient.highest_held_title_tier > tier_barony
		scope:actor = { NOT = { target_is_liege_or_above = scope:recipient } }
		POD_hostile_interaction_is_shown = yes
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
			POD_has_enough_energy_for_risen_discipline = { AMOUNT = 3 }
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:actor = {
			POD_willpower_rouse_check_effect = yes
			POD_risen_discipline_cost_effect = { COUNT = 3 }
		}
		scope:recipient = {
			every_knight = {
				add_opinion = {
					target = scope:actor
					modifier = hurt_opinion
					opinion = -25
				}
				if = {
					limit = {
						scope:actor = {
							prowess_diff = {
								target = scope:recipient
								value >= 75
							}
						}
					}
					increase_wounds_multiple_hostile_effect = { REASON = fight AMOUNT = 3 ATTACKER = scope:actor }
				}
				else_if = {
					limit = {
						scope:actor = {
							prowess_diff = {
								target = scope:recipient
								value >= 50
							}
						}
					}
					increase_wounds_multiple_hostile_effect = { REASON = fight AMOUNT = 2 ATTACKER = scope:actor }
				}
				#else_if = {
				#	limit = {
				#		scope:actor = {
				#			prowess_diff = {
				#				target = scope:recipient
				#				value >= 0
				#			}
				#		}
				#	}
				#	increase_wounds_multiple_hostile_effect = { REASON = fight AMOUNT = 1 ATTACKER = scope:actor }
				#}
				else = {
					increase_wounds_multiple_hostile_effect = { REASON = fight AMOUNT = 1 ATTACKER = scope:actor }
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no
		is_landed = yes
		
		piety > 1000
	}
	ai_targets = {
		ai_recipients = primary_war_enemies
	}
	ai_frequency = 48
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 45
			scope:actor = {
				is_at_war_with = scope:recipient
			}
		}
		modifier = {
			factor = 2
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
		modifier = {
			factor = 0
			scope:recipient = {
				is_ai = no
			}
		}
	}
}

POD_lend_the_supernatural_vigor_interaction = {
	category = interaction_category_vampire
	icon = POD_potence_interaction
	desc = POD_lend_the_supernatural_vigor_interaction_desc

	cost = { piety = minor_piety_value }

	is_shown = {
		scope:actor != scope:recipient
		scope:actor = { has_potence_trigger = yes }
		scope:actor = { has_perk = lend_the_supernatural_vigor_perk }
		POD_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			POD_has_enough_energy_for_risen_discipline = { AMOUNT = 5 }
		}
		scope:recipient = {
			NOT = { has_character_modifier = shared_strength_prowess_gain_modifier }
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:actor = {
			POD_risen_discipline_cost_effect = { COUNT = 5 }
		}
		scope:recipient = {
			add_character_modifier = {
				modifier = shared_strength_prowess_gain_modifier
				years = 5
			}
			add_opinion = {
				target = scope:actor
				modifier = pleased_opinion
				opinion = 25
			}
		}
	}

	auto_accept = yes

	ai_potential = {
		# The basic requirements
		is_imprisoned = no
		is_landed = yes
		
		is_knight = no
		number_of_knights > 0
		primary_title.tier > tier_barony
	}
	ai_targets = {
		ai_recipients = peer_vassals
		ai_recipients = courtiers
		max = 10
	}
	ai_frequency = 72
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 100
			scope:recipient = {
				is_knight_of = scope:actor
			}
		}
	}
}
