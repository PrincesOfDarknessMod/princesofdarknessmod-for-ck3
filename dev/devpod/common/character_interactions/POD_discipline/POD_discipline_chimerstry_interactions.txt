
POD_mirror_duplicate_interaction = {
	category = interaction_category_vampire
	icon = POD_chimerstry_interaction
	desc = POD_mirror_duplicate_interaction_desc

	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { has_chimerstry_trigger = yes }
		scope:actor = { has_perk = mirros_visage_perk }
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
			POD_has_enough_blood_for_discipline = { AMOUNT = 2 }
			num_of_relation_mirror_duplicate_servant = 0
		}
		POD_self_targeted_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
	   custom_tooltip = POD_mirror_duplicate_interaction_desc
	   scope:actor = {
		   trigger_event = POD_mirror_duplicate.001
	   }
	}
	
	auto_accept = yes
	
	cooldown = { years = 20 }
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no
		is_landed = yes
		
		piety > 800
	}
	ai_targets = {
		ai_recipients = self
	}
	ai_frequency = 360
	ai_will_do = {
		base = 5
		
		modifier = {
			add = 10
			is_at_war = yes
		}
	}
}

POD_sensory_deprivation_interaction = {
	category = interaction_category_vampire
	icon = POD_chimerstry_interaction
	desc = POD_sensory_deprivation_interaction_desc
	
	cooldown = { days = 1000 }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_chimerstry_trigger = yes }
		scope:actor = { has_perk = sensory_deprivation_perk }
		POD_hostile_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			POD_has_enough_blood_for_discipline = { AMOUNT = 5 }
		}
		scope:recipient = {
			POD_is_not_incorporeal_spirit = yes
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	send_options_exclusive = no
	
	send_option = {
		is_shown = {
			scope:recipient = { NOT = { has_character_flag = POD_permanent_blind } }
		}
		is_valid = {
			scope:recipient = { NOT = { has_trait = one_eyed } }
		}
		flag = sight
		localization = trait_blind
	}
	send_option = {
		is_shown = {
			scope:recipient = { NOT = { has_character_modifier = sensory_deprivation_deaf_modifier } }
		}
		flag = hearing
		localization = sensory_deprivation_deaf_modifier
	}
	send_option = {
		is_shown = {
			scope:recipient = { NOT = { has_character_modifier = sensory_deprivation_disequilibrium_modifier } }
		}
		flag = balance
		localization = sensory_deprivation_disequilibrium_modifier
	}
	send_option = {
		is_shown = {
			scope:recipient = { NOT = { has_character_modifier = sensory_deprivation_anosmia_modifier } }
		}
		flag = smell
		localization = sensory_deprivation_anosmia_modifier
	}
	send_option = {
		is_shown = {
			scope:recipient = { NOT = { has_character_modifier = sensory_deprivation_ageusia_modifier } }
		}
		flag = taste
		localization = sensory_deprivation_ageusia_modifier
	}
	send_option = {
		is_shown = {
			scope:recipient = { NOT = { has_character_modifier = sensory_deprivation_anaphia_modifier } }
		}
		flag = touch
		localization = sensory_deprivation_anaphia_modifier
	}

	on_accept = {
		scope:actor = {
			POD_discipline_cost_effect = { COUNT = 5 }
		}
		scope:recipient = {
			add_opinion = {
				target = scope:actor
				modifier = hurt_opinion
				opinion = -50
			}
			progress_towards_rival_effect = {
				CHARACTER = scope:actor
				REASON = rival_cast_spell
				OPINION = 0
			}
		}
		scope:actor = {
			duel = {
				skills = { diplomacy intrigue }
				target = scope:recipient
				50 = {
					desc = POD_shared_loca_success
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					if = {
						limit = { scope:sight = yes }
						scope:recipient = {
							add_trait = blind
							add_character_flag = {
								flag = POD_permanent_blind
								years = 10
							}
						}
					}
					if = {
						limit = { scope:hearing = yes }
						scope:recipient = {
							add_character_modifier = {
								modifier = sensory_deprivation_deaf_modifier
								years = 10
							}
						}
					}
					if = {
						limit = { scope:balance = yes }
						scope:recipient = {
							add_character_modifier = {
								modifier = sensory_deprivation_disequilibrium_modifier
								years = 10
							}
						}
					}
					if = {
						limit = { scope:smell = yes }
						scope:recipient = {
							add_character_modifier = {
								modifier = sensory_deprivation_anosmia_modifier
								years = 10
							}
							remove_trait = comfort_eater
							if = {
								limit = { can_be_inappetetic = yes }
								add_trait = inappetetic
							}
						}
					}
					if = {
						limit = { scope:taste = yes }
						scope:recipient = {
							add_character_modifier = {
								modifier = sensory_deprivation_ageusia_modifier
								years = 10
							}
							remove_trait = comfort_eater
							if = {
								limit = { can_be_inappetetic = yes }
								add_trait = inappetetic
							}
						}
					}
					if = {
						limit = { scope:touch = yes }
						scope:recipient = {
							add_character_modifier = {
								modifier = sensory_deprivation_anaphia_modifier
								years = 10
							}
						}
					}
					min = 1
				}
				50 = {
					desc = POD_shared_loca_failure
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier
					}
					scope:actor = {
						add_stress = medium_stress_impact_gain
						add_piety = medium_piety_loss
					}
					modifier = {
						add = 20
						scope:sight = yes
					}
					modifier = {
						add = 10
						scope:hearing = yes
					}
					modifier = {
						add = 10
						scope:balance = yes
					}
					modifier = {
						add = 10
						scope:smell = yes
					}
					modifier = {
						add = 10
						scope:taste = yes
					}
					modifier = {
						add = 10
						scope:touch = yes
					}
					min = 1
				}
			}
		}
	}
	
	auto_accept = yes
	
	#ai_potential = {
	#	# The basic requirements
	#	is_imprisoned = no
	#	is_landed = yes
	#}
	#ai_targets = {
	#	ai_recipients = primary_war_enemies
	#}
	
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 45
			scope:actor = {
				is_at_war_with = scope:recipient
			}
		}
		modifier = {
			factor = 2
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
		modifier = {
			factor = 0
			scope:recipient = {
				is_ai = no
			}
		}
	}
}
