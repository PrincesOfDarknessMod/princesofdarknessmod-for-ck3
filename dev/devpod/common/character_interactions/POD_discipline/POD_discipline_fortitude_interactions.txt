POD_fortitude_interaction = {
	category = interaction_category_vampire
	desc = POD_fortitude_interaction_desc
	icon = POD_fortitude_interaction
	auto_accept = yes

	is_shown = {
		scope:actor = scope:recipient
		scope:actor = {
			OR = {
				has_perk = prowess_from_pain_perk
				has_trait = fortitudeadvanced
			}
		}
	}

	send_option = {
		is_shown = {
			scope:actor = {
				OR = {
					has_perk = prowess_from_pain_perk
					has_trait = fortitudeadvanced
				}
			}
		}

		is_valid = {
			custom_tooltip = {
				text = prowess_from_pain_valid

				scope:actor = {
					has_trait = wounded
					NOT = { has_character_modifier = prowess_from_pain_1_modifier }
					NOT = { has_character_modifier = prowess_from_pain_2_modifier }
					NOT = { has_character_modifier = prowess_from_pain_3_modifier }
					NOT = { has_character_modifier = prowess_from_pain_1_enhanced_modifier }
					NOT = { has_character_modifier = prowess_from_pain_2_enhanced_modifier }
					NOT = { has_character_modifier = prowess_from_pain_3_enhanced_modifier }
				}
			}
		}
		flag = activate_prowess_from_pain
		localization = activate_prowess_from_pain_loc
	}

	send_option = {
		is_shown = {
			scope:actor = {
				has_perk = resilient_mind_perk
			}
		}
		is_valid = {
			scope:actor = {
				custom_tooltip = {
					text = resilient_mind_valid
					OR = {
						has_trait = depressed
						has_trait = lunatic
						has_trait = possessed
					}
				}
				trigger_if = {
					limit = {
						OR = {
							has_trait = depressed_genetic
							has_trait = lunatic_genetic
							has_trait = possessed_genetic
						}
						POD_suppress_clan_curse_trigger = no
					}
					NOT = { has_trait = malkavian }
				}
			}
		}
		flag = resilient_mind
		localization = resilient_mind_loc
	}

	can_send = {
		custom_description = {
			text = send_option_selected
			OR = {
				scope:activate_prowess_from_pain ?= yes
				scope:resilient_mind ?= yes
			}
		}
	}

	on_accept = {
		scope:actor = {
			if = {
				limit = {
					scope:activate_prowess_from_pain = yes
				}
				POD_hunger_rouse_check_effect = yes
				custom_tooltip = activate_prowess_from_pain_ct
				if = {
					limit = { has_perk = wrath_of_the_martyr_perk }
					if = {
						limit = { has_trait = wounded_1 }
						add_character_modifier = { modifier = prowess_from_pain_1_enhanced_modifier months = 1 }
					}
					else_if = {
						limit = { has_trait = wounded_2 }
						add_character_modifier = { modifier = prowess_from_pain_1_enhanced_modifier months = 2 }
					}
					else_if = {
						limit = { has_trait = wounded_3 }
						add_character_modifier = { modifier = prowess_from_pain_1_enhanced_modifier months = 3 }
					}
				}
				else = {
					if = {
						limit = { has_trait = wounded_1 }
						add_character_modifier = { modifier = prowess_from_pain_1_modifier months = 1 }
					}
					else_if = {
						limit = { has_trait = wounded_2 }
						add_character_modifier = { modifier = prowess_from_pain_1_modifier months = 2 }
					}
					else_if = {
						limit = { has_trait = wounded_3 }
						add_character_modifier = { modifier = prowess_from_pain_1_modifier months = 3 }
					}
				}
			}
			else_if = {
				limit = { scope:resilient_mind = yes }
				custom_tooltip = resilient_mind_ct
				
				remove_trait = depressed_1
				remove_trait = lunatic_1
				remove_trait = possessed_1
				if = {
					limit = {
						OR = {
							NOT = { has_trait = malkavian }
							POD_suppress_clan_curse_trigger = yes
						}
					}
					remove_trait = depressed_genetic
					remove_trait = lunatic_genetic
					remove_trait = possessed_genetic
				}
			}
		}
	}
}
