# POD Personal Character Interactions
#  Bind famulus

POD_bind_famulus_interaction = {
	category = interaction_category_vampire
	#interface_priority= 91
	icon = POD_animalism_interaction
	desc = POD_bind_famulus_interaction_desc
	options_heading = POD_bind_famulus_interaction_options_heading

	send_option = {
		is_shown = { scope:actor = { POD_can_summon_horse = yes } }
		flag = POD_bind_famulus_interaction_horse
		localization = POD_animal_horse
	}

	send_option = {
		is_shown = { scope:actor = { POD_can_summon_dog = yes } }
		flag = POD_bind_famulus_interaction_dog
		localization = POD_animal_dog
	}
	
	send_option = {
		is_shown = { scope:actor = { POD_can_summon_cat = yes } }
		flag = POD_bind_famulus_interaction_cat
		localization = POD_animal_cat
	}
	
	send_option = {
		is_shown = { scope:actor = { POD_can_summon_wolf = yes } }
		flag = wolf
		localization = POD_animal_wolf
	}
	
	send_option = {
		is_shown = { scope:actor = { POD_can_summon_bear = yes } }
		flag = bear
		localization = POD_animal_bear
	}
	
	send_option = {
		is_shown = { scope:actor = { POD_can_summon_lion = yes } }
		flag = lion
		localization = POD_animal_lion
	}
	
	send_option = {
		is_shown = { scope:actor = { POD_can_summon_tiger = yes } }
		flag = tiger
		localization = POD_animal_tiger
	}

	send_options_exclusive = yes
	
	auto_accept = yes
	ai_frequency = 120
	use_diplomatic_range = no
	
	ai_targets = {
		ai_recipients = self
	}
	
	cooldown = { months = 60 }
	
	is_shown = {
		scope:actor = scope:recipient
		scope:actor = {
			OR = {
				AND = {
					has_animalism_trigger = yes
					has_perk = bond_famulus_perk
				}
			}
		}
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
			POD_has_enough_blood_for_discipline = { AMOUNT = 1 }
		}
		POD_self_targeted_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:actor = {
			POD_hunger_rouse_check_effect = yes
		
			if = {
				limit = {
					scope:POD_bind_famulus_interaction_cat = yes
				}
				add_character_flag = bound_famulus_cat
				start_cat_story_cycle_effect = yes
			}
			
			if = {
				limit = {
					scope:POD_bind_famulus_interaction_dog = yes
				}
				add_character_flag = bound_famulus_dog
				start_dog_story_cycle_effect = yes
			}
			
			if = {
				limit = {
					scope:POD_bind_famulus_interaction_horse = yes
				}
				scope:actor = {
					remove_short_term_gold = medium_gold_value
				}
				add_character_flag = bound_famulus_horse
				scope:actor = {
					hidden_effect = { start_warhorse_story_cycle_effect = yes }
					add_character_modifier = horse_story_modifier
					add_character_flag = had_warhorse_story
				}
				random_owned_story = {
					limit = { story_type = story_cycle_martial_lifestyle_warhorse }
					assign_horse_gender_effect = { GENDER = female }
				}
			}
			
			if = {
				limit = {
					scope:wolf = yes
				}
				add_character_modifier = exotic_pet_wolf
				create_artifact_wolf_effect = { OWNER = this }
				scope:newly_created_artifact = { set_should_decay = no }
			}
			
			if = {
				limit = {
					scope:bear = yes
				}
				add_character_modifier = exotic_pet_bear
				create_artifact_bear_effect = { OWNER = this }
				scope:newly_created_artifact = { set_should_decay = no }
			}
			
			if = {
				limit = {
					scope:lion = yes
				}
				add_character_modifier = exotic_pet_lion
				create_artifact_lion_effect = { OWNER = this }
				scope:newly_created_artifact = { set_should_decay = no }
			}
			
			if = {
				limit = {
					scope:tiger = yes
				}
				add_character_modifier = exotic_pet_tiger
				create_artifact_tiger_effect = { OWNER = this }
				scope:newly_created_artifact = { set_should_decay = no }
			}
		}
	}

	ai_will_do = {
		base = 0
		modifier = {
			add = 100
			NOT = {

				scope:actor = { has_character_modifier =  cat_story_modifier  }
				scope:actor = { has_character_modifier =  dog_story_modifier  }
			}
		}
	}
}

POD_send_the_eighth_plague_interaction = {
	category = interaction_category_vampire
	icon = POD_animalism_interaction
	desc = POD_send_the_eighth_plague_interaction_desc
	
	cooldown = { months = 120 }
	cost = { piety = massive_piety_value }
	
	auto_accept = yes

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_animalism_trigger = yes }
		scope:actor = { has_perk = send_the_eighth_plague_perk }
		scope:recipient = { is_landed = yes }
		scope:recipient.highest_held_title_tier > tier_barony
		POD_hostile_interaction_is_shown = yes
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
			POD_has_enough_blood_for_discipline = { AMOUNT = 3 }
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:recipient = {
			add_opinion = {
				target = scope:actor
				modifier = hurt_opinion
				opinion = -80
			}
		}
	   scope:actor = {
			POD_willpower_rouse_check_effect = yes
			POD_discipline_cost_effect = { COUNT = 3 }
			
			if = {
				limit = {
					scope:recipient.highest_held_title_tier = tier_empire
				}
				add_piety = -2000
			}
			else_if = {
				limit = {
					scope:recipient.highest_held_title_tier >= tier_kingdom
				}
				add_piety = -1000
			}
			else_if = {
				limit = {
					scope:recipient.highest_held_title_tier >= tier_duchy
				}
				add_piety = -500
			}
	
			duel = {
				skill = learning
				value = 10
			
				50 = {
					desc = POD_weather_magic.duel.success
					
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					
					send_interface_toast = {
						title = POD_weather_magic.duel.success
						left_icon = scope:actor
						right_icon = scope:recipient
						custom_tooltip = POD_weather_magic.duel.tt.attacker_wins
					}
					
					POD_cast_specific_weather_magic_on_ruler = {
						TYPE  = flag:pod_weather_plague
						RULER = scope:recipient
					}
				}
				
				50 = {
					desc = POD_weather_magic.duel.failure

					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier
					}
					
					scope:actor = {
						send_interface_toast = {
							title = POD_weather_magic.duel.failure
							left_icon = scope:actor
							right_icon = scope:recipient
							custom_tooltip = POD_weather_magic.duel.tt.attacker_lost
						}
					}
				}
			}
		}
		scope:recipient = {
			progress_towards_rival_effect = {
				CHARACTER = scope:actor
				REASON = rival_cast_spell
				OPINION = 0
			}
		}
	}
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no
		is_landed = yes
		
		piety > 1500
	}
	ai_targets = {
		ai_recipients = primary_war_enemies
	}
	ai_frequency = 360
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 45
			scope:actor = {
				is_at_war_with = scope:recipient
			}
		}
		modifier = {
			factor = 2
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
	}
}
