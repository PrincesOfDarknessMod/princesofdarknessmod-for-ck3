POD_blink_interaction = {
	category = interaction_category_vampire
	desc = POD_blink_interaction_desc
	icon = POD_celerity_interaction

	is_shown = {	
		scope:actor = scope:recipient 
		scope:actor = {
			OR = {
				AND = {
					has_celerity_trigger = yes 
					has_perk = blink_perk 
				}
			}
		}
	}
	
	cooldown = { years = 5 } 

	is_valid_showing_failures_only = {	
		scope:actor = {
		   POD_has_enough_energy_for_risen_discipline = { AMOUNT = 1 }
		}	
	}

	on_accept = {
		scope:actor = {
			add_character_modifier = {
				modifier = POD_blink_modifier
				years = 1
			}
			POD_risen_discipline_cost_effect = { COUNT = 1 }
			if = {
				limit = {
					NOT = { has_perk = lightning_strike_perk }
				}
				add_character_modifier = {
					modifier = POD_drained_modifier
					years = 5
				}
			}
			if = {
				limit = {
					NOT = { has_perk = split_second_perk }
				}
				add_stress = major_stress_gain	
			}
		}
	}
	
	auto_accept = yes
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no
		is_landed = yes
	}
	ai_targets = {
		ai_recipients = self
	}
	ai_frequency = 36
	ai_will_do = {
		base = 100
	}
}
