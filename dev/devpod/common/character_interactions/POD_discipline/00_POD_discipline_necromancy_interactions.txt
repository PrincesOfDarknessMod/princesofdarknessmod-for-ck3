POD_find_wraith_interaction = {
	category = interaction_category_vampire
	icon = POD_wraith_interaction
	desc = POD_find_wraith_interaction_desc

	ai_frequency = 0

	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { has_necromancy_trigger = yes }
		scope:actor = {
			OR = {
				has_perk = ns_shroudsight_perk
				has_perk = ns_caul_of_the_neverborn_perk
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			POD_has_enough_blood_for_discipline = { AMOUNT = 1 }
		}
	}

	#cooldown = { years = 1 }

	on_accept = {
		scope:actor = {
			POD_hunger_rouse_check_effect = yes
			trigger_event = {
				id = POD_find_wraith.100
				days = { 1 14 }
			}
		}
	}

	ai_will_do = {
		base = 0
	}
}
