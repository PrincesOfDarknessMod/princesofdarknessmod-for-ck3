
POD_melpominee_interaction = {
	category = interaction_category_vampire
	desc = POD_melpominee_interaction_desc
	icon = POD_melpominee_interaction
	options_heading = POD_dementation_interaction_options_heading
	
	cooldown_against_recipient = { years = 5 }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_melpominee_trigger = yes }
		POD_hostile_mental_interaction_is_shown = yes
	}

	send_options_exclusive = no
	
	send_option = {
		is_shown = {
			scope:actor = { has_perk = toreadors_bane_perk }
		}
		flag = toreadors_bane
		localization = toreadors_bane_perk_name
	}
	
	send_option = {
		is_shown = {
			scope:actor = { has_perk = sirens_beckoning_perk }
		}
		flag = sirens_beckoning
		localization = sirens_beckoning_perk_name
	}
	
	send_option = {
		is_shown = {
			scope:actor = { has_perk = mosh_pit_perk }
			scope:recipient = { cant_frenzy_trigger = no }
		}
		flag = mosh_pit
		localization = mosh_pit_perk_name
	}
	
	send_option = {
		is_shown = {
			scope:actor = { has_perk = haunting_melody_perk }
		}
		flag = haunting_melody
		localization = haunting_melody_perk_name
	}
	
	send_option = {
		is_shown = {
			scope:actor = { has_perk = primal_scream_perk }
			scope:recipient = { exists = location }
			scope:recipient = { cant_frenzy_trigger = no }
		}
		flag = primal_scream
		localization = primal_scream_perk_name
	}
	
	is_valid_showing_failures_only = {
		custom_description = {
			object = scope:recipient
			text = "strong_mental_defenses_trigger"
			OR = {
				scope:actor = { has_trait = melpomineeadvanced }
				strong_mental_defenses_interaction_trigger = yes
			}
		}
		scope:actor = {
			POD_has_enough_blood_for_discipline = { AMOUNT = POD_melpominee_interaction_cost }
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}
	
	on_accept = {
		scope:actor = { POD_willpower_rouse_check_effect = yes }
		scope:actor = {
			POD_discipline_cost_effect = { COUNT = POD_melpominee_interaction_cost }
		}
		scope:recipient = {
			add_opinion = {
				target = scope:actor
				modifier = hurt_opinion
				opinion = -40
			}
		}
		scope:actor = {
			duel = {
				skill = diplomacy
				target = scope:recipient
				50 = {
					desc = POD_melpominee_interaction_sucess
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					modifier = {
						factor = 2
						has_perk = persistent_echo_perk
					}
					scope:recipient = {
						add_stress = medium_stress_impact_gain
						add_piety = medium_piety_loss
					}
					if = {
						limit = { scope:toreadors_bane = yes }
						scope:recipient = {
							add_character_modifier = {
								modifier = POD_toreador_bane_modifier
								years = 5
							}
						}
					}
					if = {
						limit = { scope:sirens_beckoning = yes }
						scope:recipient = {
							POD_stress_break_effect = yes
						}
					}
					if = {
						limit = { scope:mosh_pit = yes }
						scope:recipient = {
							add_character_modifier = {
								modifier = murderous_frenzy_modifier
								years = 5
							}
						}
					}
					if = {
						limit = { scope:primal_scream = yes }
						scope:recipient = {
							location = {
								add_province_modifier = {
									modifier = POD_frenzy_terrified_populace_modifier
									years = 5
								}
							}
						}
					}
					if = {
						limit = { scope:haunting_melody = yes }
						scope:recipient = {
							add_character_modifier = {
								modifier = haunting_melody_perk_modifier
								years = 5
							}
						}
					}
					min = 1
				}
				50 = {
					desc = POD_melpominee_interaction_fail
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier
					}
					modifier = {
						add = 5
						scope:toreadors_bane = yes
					}
					modifier = {
						add = 10
						scope:sirens_beckoning = yes
					}
					modifier = {
						add = 15
						scope:mosh_pit = yes
					}
					modifier = {
						add = 15
						scope:primal_scream = yes
					}
					modifier = {
						add = 25
						scope:haunting_melody = yes
					}
					min = 1
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no
		is_landed = yes
	}
	ai_targets = {
		ai_recipients = primary_war_enemies
	}
	ai_frequency = 240
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 45
			scope:actor = {
				is_at_war_with = scope:recipient
			}
		}
		modifier = {
			factor = 2
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
		modifier = {
			factor = 0
			scope:recipient = {
				is_ai = no
			}
		}
	}
}

POD_calling_song_interaction = {
	category = interaction_category_vampire
	icon = POD_melpominee_interaction
	desc = POD_calling_song_interaction_desc
	
	cooldown = { months = 6 }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_melpominee_trigger = yes }
		scope:actor = { has_perk = calling_song_perk }
		NOT = { scope:actor = { location = scope:recipient.location } }
		scope:recipient = { is_playable_character = yes }
		POD_hostile_mental_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			POD_has_enough_blood_for_discipline = { AMOUNT = 1 }
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}
	
	on_accept = {
		scope:actor = { POD_willpower_rouse_check_effect = yes }
		scope:actor = { POD_discipline_cost_effect = { COUNT = 1 } }
		scope:actor = {
			duel = {
				skill = diplomacy
				target = scope:recipient
				30 = {
					desc = POD_calling_song_interaction_sucess
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					scope:recipient = {
						every_courtier_or_guest = {
							limit = {
								POD_is_available = yes
								strong_mental_defenses_trigger = no
							}
							scope:actor = { add_courtier = prev }
							hidden_effect = {
								return_to_court = yes
							}
							add_character_flag = {
								flag = blocked_from_leaving
								years = 5
							}
						}
					}
				}
				30 = {
					desc = POD_calling_song_interaction_fail
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier
					}
				}
			}
		}
	}

	ai_frequency = 0
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_madrigal_interaction = {
	category = interaction_category_vampire
	icon = POD_melpominee_interaction
	desc = POD_soul_bridge_interaction_desc
	
	cooldown = { months = 3 }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_melpominee_trigger = yes }
		scope:actor = { has_perk = madrigal_perk }
		POD_hostile_mental_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			POD_has_enough_blood_for_discipline = { AMOUNT = 1 }
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}
	
	send_option = {
		is_shown = {
			NOR = {
				scope:recipient = { is_imprisoned_by = scope:actor }
				scope:recipient = { has_relation_necromancer = scope:actor }
				scope:actor = {
					has_hook_of_type = {
						target = scope:recipient
						type = strong_bloodoath_hook
					}
				}
			}
		}
		is_valid = {
			scope:actor = {
				can_start_scheme = {
					type = PODsoul_bridge
					target_character = scope:recipient
				}
			}
		}
		flag = scheme_attack
		localization = POD_interaction_scheme_attack
	}
	
	send_option = {
		is_shown = {
			NOR = {
				scope:recipient = { is_imprisoned_by = scope:actor }
				scope:recipient = { has_relation_necromancer = scope:actor }
				scope:actor = {
					has_hook_of_type = {
						target = scope:recipient
						type = strong_bloodoath_hook
					}
				}
			}
		}
		is_valid = {
			scope:recipient = { can_start_single_combat_trigger = yes }
			custom_description = {
				text = same_location
				scope:actor = { exists = location }
				scope:recipient = { exists = location }
				scope:actor = { location = scope:recipient.location }
			}
		}
		flag = duel_attack
		localization = POD_interaction_duel_attack
	}
	
	send_option = {
		is_shown = {
			OR = {
				scope:recipient = { is_imprisoned_by = scope:actor }
				scope:recipient = { has_relation_necromancer = scope:actor }
				scope:actor = { has_strong_usable_hook = scope:recipient }
			}
		}
		flag = automatic
		localization = POD_interaction_automatic
	}

	on_accept = {
		scope:actor = { POD_willpower_rouse_check_effect = yes }
		scope:actor = { POD_discipline_cost_effect = { COUNT = 1 } }
		switch = {
			trigger = yes
			scope:scheme_attack = {
				scope:actor = {
					start_scheme = {
						type = PODsoul_bridge
						target_character = scope:recipient
					}
				}
			}
			scope:duel_attack = {
				scope:actor = {
					configure_start_single_combat_effect = {
						SC_INITIATOR = scope:actor
						SC_ATTACKER = scope:actor
						SC_DEFENDER = scope:recipient
						FATALITY = no
						FIXED = no
						LOCALE = throne_room
						OUTPUT_EVENT = kueijin_scheme.201
						INVALIDATION_EVENT = tribal.1022
					}
				}
			}
			scope:automatic = {
				scope:actor = { trigger_event = kueijin_scheme.200 }
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
