POD_kai_elder_4_perk_interaction = { #Raijinto
	category = interaction_category_vampire
	icon = POD_kai_interaction
	desc = POD_kai_elder_4_perk_interaction_desc

	cooldown = { years = 5 }
	ai_frequency = 0

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			has_perk = kai_elder_4_perk
			has_kai_trigger = yes
			POD_splat_has_access_to_disciplines = yes
		}
		POD_hostile_interaction_is_shown = yes
	}

	can_send = {
		exists = scope:secondary_recipient
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			POD_has_enough_blood_for_discipline = { AMOUNT = 5 }
			NOT = { has_trait = wounded }
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	populate_recipient_list = {
		scope:actor = {
			every_relation = {
				limit = {
					is_imprisoned_by = scope:actor
				}
				type = ghoul_thrall
				add_to_list = characters
			}
			dynasty = {
				every_dynasty_member = {
					limit = {
						is_imprisoned_by = scope:actor
					}
					add_to_list = characters
				}
			}
		}
	}

	on_accept = {
		scope:actor = { 
			POD_discipline_cost_effect = { COUNT = 5 }
			POD_commit_heinous_act_effect = yes
			increase_wounds_multiple_effect = { REASON = lightning AMOUNT = 3 }
			if = {
				limit = {
					faith = { has_doctrine_parameter = hurt_mortal_lose_piety }
				}
				add_piety = -200
			}
		}	
		scope:recipient = {
			if = {
				limit = {
					POD_has_advanced_defensive_capabilities_trigger = {
						AMOUNT = "amount >= 4"
					}
				}
				increase_wounds_multiple_hostile_effect = { REASON = lightning AMOUNT = 1 ATTACKER = scope:actor }
			}
			else_if = {
				limit = {
					POD_has_advanced_defensive_capabilities_trigger = {
						AMOUNT = "amount >= 2"
					}
				}
				increase_wounds_multiple_hostile_effect = { REASON = lightning AMOUNT = 2 ATTACKER = scope:actor }
			}
			else_if = {
				limit = {
					POD_has_advanced_defensive_capabilities_trigger = {
						AMOUNT = "amount = 1"
					}
				}
				increase_wounds_multiple_hostile_effect = { REASON = lightning AMOUNT = 3 ATTACKER = scope:actor }
			}
			else = {
				pod_death = {
					DEATH_REASON = death_lightning
					KILLER = scope:actor
				}
			}
		}
		scope:secondary_recipient = {
			pod_death = {
				DEATH_REASON = death_execution
				KILLER = scope:actor
			}
		}
	}

	auto_accept = yes

	ai_will_do = {
		base = 0
	}
}
