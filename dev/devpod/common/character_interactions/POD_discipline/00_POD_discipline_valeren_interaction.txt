POD_pariahs_caress_perk_interaction = {
	category = interaction_category_vampire
	icon = POD_valeren_interaction
	desc = POD_pariahs_caress_perk_interaction_desc
	
	auto_accept = yes

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_valeren_trigger = yes }
		scope:actor = { 
			has_perk = pariahs_caress_perk
		}
		scope:recipient = {
			NOT = { has_character_modifier = aversion_perk_modifier }
			NOT = { has_character_modifier = pariahs_caress_perk_modifier }
		}
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
			POD_has_enough_blood_for_discipline = { AMOUNT = 1 }
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			NOT = { has_trait = automaton }
			NOT = { has_trait = wight }
			NOT = { has_trait = barghest }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }	
			POD_is_not_clone_trigger = yes		
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}

	on_accept = {
		scope:actor = { POD_willpower_rouse_check_effect = yes }
		scope:actor = { POD_discipline_cost_effect = { COUNT = 1 } }
		if = {
			limit = {
				scope:actor = { has_perk = aversion_perk }
			}
			scope:recipient = {
				if = {
					limit = { prestige >= 0 }
					random = {
						chance = 5
						add_prestige = {
							value = -1
							subtract = scope:recipient.prestige
						}
					}
				}
				add_character_modifier = {
					modifier = aversion_perk_modifier
					years = 3
				}
				add_opinion = {
					target = scope:actor
					modifier = hurt_opinion
					opinion = -100
				}
			}
		}
		else = {
			scope:recipient = {
				add_character_modifier = {
					modifier = pariahs_caress_perk_modifier
					years = 3
				}
				add_opinion = {
					target = scope:actor
					modifier = hurt_opinion
					opinion = -50
				}
			}
		}
    }
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no 
		is_landed = yes 
	}
	ai_targets = {
		ai_recipients = primary_war_enemies
	}
	ai_frequency = 72
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 45
			scope:actor = {
			    is_at_war_with = scope:recipient
			}
		}
		modifier = {
			factor = 2
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
		modifier = {
			factor = 0
			scope:recipient = {
				is_ai = no
			}
		}
	}
}
