
POD_kiss_of_lachesis_perk_interaction = {
	category = interaction_category_vampire
	icon = POD_temporis_interaction
	desc = POD_kiss_of_lachesis_perk_interaction_desc
	
	cooldown = { years = 1 }
	
	cost = { prestige = 200 }

	is_shown = {
		scope:actor != scope:recipient
		scope:actor = { has_temporis_trigger = yes }
		scope:actor = { has_perk = kiss_of_lachesis_perk }
		POD_hostile_mental_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			POD_has_enough_blood_for_discipline = { AMOUNT = 2 }
		}
		scope:recipient = {
			is_ruler = no
			NOT = { has_character_modifier = baba_yagas_curse_modifier }
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:actor = { POD_willpower_rouse_check_effect = yes }
		scope:actor = { POD_discipline_cost_effect = { COUNT = 2 } }
		if = {
			limit = {
				OR = {
					scope:recipient = { is_imprisoned_by = scope:actor }
					scope:recipient = { can_start_single_combat_trigger = no }
					scope:actor = {
						has_hook_of_type = {
							target = scope:recipient
							type = strong_bloodoath_hook
						}
					}
				}
			}
			scope:actor = { trigger_event = POD_temporis.100 }
		}
		else = {
			scope:actor = {
				configure_start_single_combat_effect = {
					SC_INITIATOR = scope:actor
					SC_ATTACKER = scope:actor
					SC_DEFENDER = scope:recipient
					FATALITY = no
					FIXED = no
					LOCALE = throne_room
					OUTPUT_EVENT = POD_temporis.101
					INVALIDATION_EVENT = tribal.1022
				}
			}
		}
		scope:recipient = {
			add_opinion = {
				target = scope:actor
				modifier = hurt_opinion
				opinion = -200
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_clios_kiss_perk_interaction = {
	category = interaction_category_vampire
	icon = POD_temporis_interaction
	desc = POD_clios_kiss_perk_interaction_desc
	
	target_type = title
	target_filter = recipient_domain_titles
	
	cost = {
		piety = {
			value = {
				add = scope:recipient.max_military_strength
				multiply = scope:target.tier
				multiply = 0.05
			}
		}
	}
	
	is_shown = {
		NOT = { scope:recipient = scope:actor }
		scope:actor = { has_temporis_trigger = yes }
		scope:actor = { has_perk = clios_kiss_perk }
		POD_hostile_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:recipient = { is_ruler = yes }
		scope:actor = {
			POD_has_enough_blood_for_discipline = { AMOUNT = 5 }
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}
	
	can_be_picked_title = {
		custom_description = {
			text = POD_has_previous_holder
			object = scope:target
			scope:target = {
				exists = previous_holder
				previous_holder = { is_alive = yes }
			}
		}
	}
	
	on_accept = {
		scope:actor = { POD_willpower_rouse_check_effect = yes }
		scope:actor = { POD_discipline_cost_effect = { COUNT = 5 } }
		scope:recipient = {
			add_opinion = {
				target = scope:actor
				modifier = hurt_opinion
				opinion = -20
			}
		}
		scope:actor = {
			duel = {
				skill = learning
				target = scope:recipient
				30 = {
					desc = POD_clios_kiss_perk_interaction_sucess
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					create_title_and_vassal_change = {
						type = usurped
						save_scope_as = title_change
						add_claim_on_loss = yes
					}
					scope:target = {
						change_title_holder = {
							holder = scope:target.previous_holder
							change = scope:title_change
						}
					}
					resolve_title_and_vassal_change = scope:title_change
				}
				70 = {
					desc = POD_clios_kiss_perk_interaction_fail
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier
					}
				}
			}
		}
	}

	#ai_frequency = 0
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
