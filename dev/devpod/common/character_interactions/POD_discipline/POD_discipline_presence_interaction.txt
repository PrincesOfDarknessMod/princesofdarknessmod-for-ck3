POD_entrance_interaction = {
	category = interaction_category_vampire
	#common_interaction = yes
	#interface_priority = 89
	scheme = PODentrance
	icon = POD_presence_interaction
	
	#ai_frequency = 0

	is_shown = {
		scope:actor != scope:recipient
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		scope:actor = {
			OR = {
				AND = {
					has_presence_trigger = yes
					has_perk = entrancement_perk
				}
			}
		}
		scope:recipient = {
			NOT = {
				has_opinion_modifier = {
					target = scope:actor
					modifier = entrancement_opinion
				}
			}
		}
		POD_hostile_mental_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		POD_strong_mental_defenses_trigger = yes
		scope:actor = {
			can_start_scheme = {
				type = PODentrance
				target_character = scope:recipient
			}
			POD_has_enough_blood_for_discipline = { AMOUNT = 1 }
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	desc = {
		triggered_desc = {
			trigger = {
				scope:actor = {
					NOT = { is_at_war_with = scope:recipient }
					can_start_scheme = {
						type = PODentrance
						target_character = scope:recipient
					}

				}
			}
			desc = scheme_interaction_tt_PODentrance_approved
		}

	}
	
	on_accept = {
		hidden_effect = {
			scope:actor = {
				start_scheme = {
					type = PODentrance
					target_character = scope:recipient
				}
			}
		}
	}

	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_summon_interaction = {
	category = interaction_category_vampire
	#common_interaction = yes
	#interface_priority = 87
	scheme = PODsummon
	
	ai_frequency = 0

	is_shown = {
		scope:actor != scope:recipient
		scope:recipient = { exists = location }
		NOT = { scope:actor = { location = scope:recipient.location } }
		scope:actor = {
			OR = {
				AND = {
					has_presence_trigger = yes
					has_perk = summon_perk
				}
			}
		}
		POD_hostile_mental_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:recipient = { is_ruler = no }
		scope:recipient = { is_imprisoned = no }
		scope:actor = {
			can_start_scheme = {
				type = PODsummon
				target_character = scope:recipient
			}
			POD_has_enough_blood_for_discipline = { AMOUNT = 1 }
		}
		POD_strong_mental_defenses_trigger = yes
		POD_summon_valid_trigger = yes
		POD_interaction_is_valid_showing_failures_only = yes
	}

	desc = {
		triggered_desc = {
			trigger = {
				scope:actor = {
					NOT = { is_at_war_with = scope:recipient }
					can_start_scheme = {
						type = PODsummon
						target_character = scope:recipient
					}

				}
			}
			desc = scheme_interaction_tt_PODsummon_approved
		}

	}
	
	on_accept = {
		scope:actor = {
			POD_hunger_rouse_check_effect = yes
		}
		hidden_effect = {
			scope:actor = {
				start_scheme = {
					type = PODsummon
					target_character = scope:recipient
				}
				trigger_event = POD_summon.10
			}
		}
	}

	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_mask_of_empathy_interaction = {
    category = interaction_category_vampire
	#common_interaction = yes
	#interface_priority = 96
    desc = POD_mask_of_empathy_interaction_desc
	icon = POD_presence_interaction
	
	ai_frequency = 0
	
	is_shown = {
		scope:actor != scope:recipient
		scope:actor = { has_presence_trigger = yes }
		scope:actor = { has_perk = mask_of_empathy_perk }
		POD_hostile_mental_interaction_is_shown = yes
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
			POD_has_enough_blood_for_discipline = { AMOUNT = 2 }
		}
		custom_description = {
			object = scope:actor
			text = no_valid_relations
			scope:recipient = {
				OR = {
					any_relation = {
						type = friend
						NOT = { this = scope:actor }
						count >= 1
					}
					any_relation = {
						type = rival
						NOT = { this = scope:actor }
						count >= 1
					}
					any_relation = {
						type = lover
						NOT = { this = scope:actor }
						count >= 1
					}
				}
			}
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}
	
	populate_recipient_list = {
		scope:recipient = {
			every_relation = {
				type = friend
				limit = {
				    is_alive = yes
					NOT = { this = scope:actor }
				}
				
				add_to_list = characters
			
			}
			every_relation = {
				type = rival
				limit = {
				    is_alive = yes
					NOT = { this = scope:actor }
				}
				
				add_to_list = characters
			
			}
			every_relation = {
				type = lover
				limit = {
				    is_alive = yes
					NOT = { this = scope:actor }
				}
				
				add_to_list = characters
			
			}
		}
	}
	
	send_option = {
		is_shown = {
			exists = scope:secondary_recipient
			scope:secondary_recipient = {
				has_relation_friend = scope:recipient
			}
		}
		is_valid = {
			scope:secondary_recipient = {
				NOT = { has_relation_best_friend = scope:recipient }
			}
		}
		flag = friend
		localization = POD_mask_of_empathy_friend
	}
	
	send_option = {
		is_shown = {
			exists = scope:secondary_recipient
			scope:secondary_recipient = {
				has_relation_rival = scope:recipient
			}
		}
		is_valid = {
			scope:secondary_recipient = {
				NOT = { has_relation_nemesis = scope:recipient }
			}
		}
		flag = rival
		localization = POD_mask_of_empathy_rival
	}
	
	send_option = {
		is_shown = {
			exists = scope:secondary_recipient
			scope:secondary_recipient = {
				has_relation_lover = scope:recipient
			}
		}
		is_valid = {
			scope:secondary_recipient = {
				NOT = { has_relation_soulmate = scope:recipient }
			}
		}
		flag = lover
		localization = POD_mask_of_empathy_lover
	}
	#This option exists because my horrible spaghetti code won't work without it
	send_option = {
		flag = nothing
		localization = POD_mask_of_empathy_nothing
	}
	
	on_accept = {
		custom_tooltip = POD_mask_of_empathy_interaction.tt
		hidden_effect = {
			if = {
				limit = { scope:nothing = no }
				scope:actor = {
					trigger_event = {
						id = POD_mask_of_empathy.001
						days = 10
					}
				}
			}
			if = {
				limit = {
					scope:friend = yes
				}
				scope:recipient = {
					add_character_flag = losing_friend
				}
			}
			if = {
				limit = {
					scope:rival = yes
				}
				scope:recipient = {
					add_character_flag = losing_rival
				}
			}
			if = {
				limit = {
					scope:lover = yes
				}
				scope:recipient = {
					add_character_flag = losing_lover
				}
			}
		}
	}
	
	auto_accept = yes
}

		ai_recipients = self
	}

	}
