POD_doorway_to_hell_perk_interaction = {
	category = interaction_category_vampire
	icon = POD_rift_interaction
	desc = POD_doorway_to_hell_perk_perk_interaction_desc
	
	use_diplomatic_range = no
	cooldown = { months = 1 }

	is_shown = {
		scope:actor = { has_perk = doorway_to_hell_perk }
		scope:actor = { has_rift_trigger = yes }
		OR = {
			POD_teleport_interaction_army_is_shown_trigger = yes #see first 
			POD_teleport_interaction_travel_trigger = yes
		}
	}

	is_valid = {
		trigger_if = {
			limit = { POD_teleport_interaction_travel_trigger = no }
			POD_teleport_interaction_army_no_size_limit_is_valid_trigger = yes #Custom trigger removes army limit
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			POD_has_enough_blood_for_discipline = { AMOUNT = 1 }
		}
		#Some way to scope capital location here?
	}
	
	send_option = { #THIS OPTION FOR TELEPORTING ARMY
		is_shown = { POD_teleport_interaction_army_is_shown_trigger = yes }
		is_valid = {
			POD_teleport_interaction_army_no_size_limit_is_valid_trigger = yes #Custom trigger removes army limit
			custom_tooltip = {
				text = POD_doorway_to_hell_perk_perk_interaction_tt1
				scope:recipient.location = { 
					NOT = { terrain = oasis }
					NOT = { terrain = wetlands }
					NOT = { terrain = floodplains }
				}
			}
		}
		flag = teleport_army
		localization = POD_teleport_army
	}
	
	send_option = { #THIS OPTION FOR TRAVELING
		is_shown = { POD_teleport_interaction_travel_trigger = yes }
		is_valid = {
			custom_tooltip = {
				text = POD_doorway_to_hell_perk_interaction_tt2
				scope:actor.current_travel_plan.next_destination_province ?= { 
					#scope:capital #Some way to scope capital location here?
				}
			}
		}
		flag = teleport_travel
		localization = POD_teleport_travel	
	}

	on_accept = {
		scope:actor = { POD_willpower_rouse_check_effect = yes }
		scope:actor = { POD_discipline_cost_effect = { COUNT = 1 } }
		scope:teleport_travel = {
			POD_teleport_travel_interaction_effect = yes
		}
		#APPLY GLIMPSE OF PAIN HERE TO YOUR KNIGHTS
	}

	ai_frequency = 0
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
