POD_glimpse_of_the_pain_interaction = { #Short duration effect, could maybe do with a monthly auto renew event?
	category = interaction_category_vampire
	icon = POD_rift_interaction
	desc = POD_glimpse_of_the_pain_interaction_desc
	
	cooldown = { months = 1 }
	ai_frequency = 0
	auto_accept = yes

	is_shown = {
		scope:actor = { has_perk = glimpse_of_the_pain_perk }
		scope:actor = { has_rift_trigger = yes }
		POD_basic_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		POD_interaction_is_valid_showing_failures_only = yes
		scope:actor = {
			POD_has_enough_blood_for_discipline = { AMOUNT = 2 }
		}
	}

	on_accept = {
		scope:actor = { 
			POD_discipline_cost_effect = { COUNT = 2 }
			POD_commit_heinous_act_effect = yes
			POD_willpower_cost_effect = { COUNT = 7 } #35 flat stress
		}	
		scope:recipient = {
			POD_willpower_cost_effect = { COUNT = 7 } #Matches actor willpower cost
			add_character_modifier = {
				modifier = glimpse_of_the_pain_modifier
				months = 1
			}
		}
	}

	ai_will_do = {
		base = 0
		
		modifier = {
			add = 45
			scope:actor = {
			    is_at_war_with = scope:recipient
			}
		}
		modifier = {
			factor = 2
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
		modifier = {
			factor = 0
			scope:recipient = {
				is_ai = no
			}
		}
	}
}

POD_passing_the_inner_realm_interaction = {
	category = interaction_category_vampire
	desc = POD_passing_the_inner_realm_interaction_desc
	icon = POD_rift_interaction
	cooldown = { months = 3 } #Somewhat lengthy to prevent toggle off micromanagement to avoid the stress negatives
	ai_frequency = 0
	auto_accept = yes
	
	is_shown = {
		scope:actor = scope:recipient
		scope:actor = {
			OR = {
				AND = {
					has_rift_trigger = yes
					has_perk = passing_the_inner_realm_perk
				}
			}
		}
	}
	
	# send_option = {
	# 	is_shown = {
	# 		scope:actor = {
	# 			NOT = { has_character_flag = active_toughness}
	# 		}
	# 	}
	# 	flag = activate_toughness
	# 	localization = activate_toughness_loc
	# }
	
	# send_option = {
	# 	is_shown = {
	# 		scope:actor = {
	# 			has_character_flag = active_toughness
	# 		}
	# 	}
	# 	flag = deactivate_toughness
	# 	localization = deactivate_toughness_loc
	# }

	send_option = {
		is_shown = {
			scope:actor = {
				NOT = { has_character_flag = active_rift } 
			}
		}
		flag = activate_rift
		localization = activate_passing_the_inner_realm_loc
	}
	
	send_option = {
		is_shown = {
			scope:actor = {
				has_character_flag = active_rift
			}
		}
		flag = deactivate_rift
		localization = deactivate_passing_the_inner_realm_loc
	}
	
	on_accept = {
		scope:actor = {
			if = {
				limit = {
					scope:activate_rift = yes
				}
				if = {
					limit = {
						has_perk = passing_the_inner_realm_perk
						NOT = { has_perk = rift_gateway_perk }
					}
					add_character_modifier = passing_the_inner_realm_modifier
				} 
				else_if = {
					limit = {
						has_perk = passing_the_inner_realm_perk
						has_perk = rift_gateway_perk 
					}
					remove_character_modifier = passing_the_inner_realm_modifier
					add_character_modifier = rift_gateway_modifier
				}
				add_character_flag = active_rift
				custom_tooltip = activate_passing_the_inner_realm_ct
			}
			else_if = {
				limit = {
					scope:deactivate_rift = yes
				}
				remove_character_modifier = passing_the_inner_realm_modifier
				remove_character_modifier = rift_gateway_modifier
				remove_character_flag = active_rift
				custom_tooltip = deactivate_passing_the_inner_realm_ct
			}
		}
	}	
}

POD_doorway_to_hell_interaction = {
	category = interaction_category_vampire
	icon = POD_rift_interaction
	desc = POD_doorway_to_hell_interaction_desc
	
	use_diplomatic_range = no
	cooldown = { months = 1 } 

	is_shown = {
		scope:actor = { has_perk = doorway_to_hell_perk }
		scope:actor = { has_rift_trigger = yes }
		scope:actor = scope:recipient
		OR = {
			is_commanding_army = yes #see first
			POD_teleport_interaction_travel_trigger = yes
		}
	}

	is_valid = {
		trigger_if = {
			limit = { POD_teleport_interaction_travel_trigger = no }
			POD_teleport_interaction_army_no_size_limit_is_valid_trigger = yes #Custom trigger removes army limit
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			POD_has_enough_blood_for_discipline = { AMOUNT = 1 }
		}
	}
	
	send_option = { 
		is_shown = { is_commanding_army = yes }
		is_valid = {
			POD_teleport_interaction_army_no_size_limit_is_valid_trigger = yes #Custom trigger removes army limit
		}
		flag = teleport_army
		localization = POD_teleport_army
	}
	
	send_option = { 
		is_shown = { POD_teleport_interaction_travel_trigger = yes }
		is_valid = {
			custom_tooltip = {
				text = POD_teleport_travel_tt
				scope:actor.current_travel_plan.next_destination_province ?= scope:actor.capital_province
			}
		}
		flag = teleport_travel
		localization = POD_teleport_travel	
	}

	on_accept = {
		if = {
			limit = {
				is_commanding_army = no
			}
			current_travel_plan = {
				every_entourage_character = { #Skipped out on applying the mental strain to the entourage lest you make your entire travelling party insane/wights etc, easy to add back in
					add_character_modifier = {
						modifier = glimpse_of_the_pain_modifier
						months = 1
					}
				} 
			}
		}
		if = {
			limit = {
				is_commanding_army = yes
			}
			every_knight = { 
				limit = { 
					knight_army ?= prev.commanding_army 
				} 
				add_character_modifier = {
					modifier = glimpse_of_the_pain_modifier
					months = 1
				}
		    }
		}
		if = {
			limit = {
				scope:actor = { 
					has_character_modifier = glimpse_of_the_pain_modifier
				}
			}
			scope:actor = {
				remove_character_modifier = glimpse_of_the_pain_modifier	
			}
		}

		scope:actor = {			
			POD_willpower_rouse_check_effect = yes
			POD_discipline_cost_effect = { COUNT = 1 }			
			switch = {
				trigger = yes
				scope:teleport_army = {
					custom_description = {
						text = POD_teleport_army_tt
						object = scope:actor.capital_province.barony
						commanding_army = {
							set_army_location = scope:actor.capital_province
						}
					}
				}
				scope:teleport_travel = {
					POD_teleport_travel_interaction_effect = yes
				}
			}
			
		}
	}

	ai_frequency = 0
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
