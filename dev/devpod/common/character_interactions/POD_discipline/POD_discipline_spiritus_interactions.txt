POD_hunt_spirit_interaction = {
	category = interaction_category_vampire
	icon = POD_spiritus_interaction
	desc = POD_hunt_spirit_interaction_desc
	
	auto_accept = yes

	cooldown = { months = 1 } #1 month guarantees they take the cost/benefit 

	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { has_spiritus_trigger = yes }
		scope:actor = {
			OR = {
				has_perk = engling_fury_perk
				has_perk = beastly_hunger_perk
			}
		}
	}
	
	is_valid_showing_failures_only = {
		POD_self_targeted_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:actor = {
			add_character_modifier = spirit_hunt_5_modifier
			if = {
				limit = {
					OR = {
						has_character_modifier = spirit_hunt_modifier
						has_character_modifier = spirit_hunt_2_modifier
						has_character_modifier = spirit_hunt_3_modifier
						has_character_modifier = spirit_hunt_4_modifier
					}
				}
				# Toggle off
				POD_remove_character_modifier = { MODIFIER = spirit_hunt_modifier }
				POD_remove_character_modifier = { MODIFIER = spirit_hunt_2_modifier }
				POD_remove_character_modifier = { MODIFIER = spirit_hunt_3_modifier }
				POD_remove_character_modifier = { MODIFIER = spirit_hunt_4_modifier }
				custom_tooltip = deactivate_spirit_hunt_modifier_ct
			}
			else = {
				# Toggle on
				if = {
					limit = { has_perk = engling_fury_perk }
					limit = { has_perk = beastly_hunger_perk }	
					add_character_modifier = spirit_hunt_4_modifier
				}
				else_if = {
					limit = { has_perk = beastly_hunger_perk }
					add_character_modifier = spirit_hunt_3_modifier
				}
				else_if = {
					limit = { has_perk = engling_fury_perk }
					add_character_modifier = spirit_hunt_2_modifier
				}
				else = {
					add_character_modifier = spirit_hunt_modifier
				}
				
				custom_tooltip = activate_spirit_hunt_modifier_ct
			}
		}
	}
	
	ai_frequency = 0
	ai_will_do = {
		base = 0
	}
}
