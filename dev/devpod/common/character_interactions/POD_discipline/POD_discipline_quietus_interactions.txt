
POD_dagons_call_perk_interaction = {
	category = interaction_category_vampire
	desc = POD_dagons_call_perk_interaction_desc
	icon = POD_quietus_interaction
	
	cooldown = { years = 10 }	
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_quietus_trigger = yes }
		scope:actor = { 
			OR = { 
				has_perk = sorcerer_dagons_call_perk 
				has_perk = warrior_dagons_call_perk 
			}
		}
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
			POD_has_enough_blood_for_discipline = { AMOUNT = 1 }
		}	
		POD_damage_spirit_trigger = yes
		scope:actor = {
			prowess >= POD_prowess_low_skill_rating
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }	
		}
		POD_cross_umbra_interaction_valid_trigger = yes	
	}

	#Use Condemn the Sins of the Father
	send_option = {
		is_shown = {
			scope:actor = { has_perk = condemn_the_sins_of_the_father_perk }
			scope:recipient = { exists = house }
		}
		flag = condemn_the_sins_of_the_father
		localization = condemn_the_sins_of_the_father_perk_name
	}

	send_options_exclusive = no

	on_accept = {	
		scope:actor = { POD_willpower_rouse_check_effect = yes }
		scope:actor = { 
			duel = {
				skill = prowess
				target = scope:recipient
				10 = {
					trigger = {
						prowess >= POD_prowess_extremely_high_skill_rating
					}
					if = { 
						limit = { 
							NAND = { 
								has_perk = sorcerer_dagons_call_perk 
								has_perk = warrior_dagons_call_perk 
							} 
						}
						add_piety = -1000
					}
					POD_discipline_cost_effect = { COUNT = 4 }
					scope:recipient = { 
						if = {
							limit = { scope:condemn_the_sins_of_the_father = yes }
							house = {
								every_house_member = {
									limit = { age <= scope:recipient.age }
									increase_wounds_multiple_hostile_effect = { REASON = fight AMOUNT = 4 ATTACKER = scope:actor }
								}
							}
						}
						else = {
							increase_wounds_multiple_hostile_effect = { REASON = fight AMOUNT = 4 ATTACKER = scope:actor } 
						}
					}
				}
				20 = {
					trigger = {
						prowess >= POD_prowess_high_skill_rating
					}
					if = { 
						limit = { 
							NAND = { 
								has_perk = sorcerer_dagons_call_perk 
								has_perk = warrior_dagons_call_perk 
							} 
						}
						add_piety = -500
					}
					POD_discipline_cost_effect = { COUNT = 3 }
					scope:recipient = { 
						if = {
							limit = { scope:condemn_the_sins_of_the_father = yes }
							house = {
								every_house_member = {
									limit = { age <= scope:recipient.age }
									increase_wounds_multiple_hostile_effect = { REASON = fight AMOUNT = 3 ATTACKER = scope:actor }
								}
							}
						}
						else = {
							increase_wounds_multiple_hostile_effect = { REASON = fight AMOUNT = 3 ATTACKER = scope:actor } 
						}
					}
				}
				30 = {
					trigger = {
						prowess >= POD_prowess_medium_skill_rating
					}
					if = { 
						limit = { 
							NAND = { 
								has_perk = sorcerer_dagons_call_perk 
								has_perk = warrior_dagons_call_perk 
							} 
						}
						add_piety = -300
					}
					POD_discipline_cost_effect = { COUNT = 2 }
					scope:recipient = { 
						if = {
							limit = { scope:condemn_the_sins_of_the_father = yes }
							house = {
								every_house_member = {
									limit = { age <= scope:recipient.age }
									increase_wounds_multiple_hostile_effect = { REASON = fight AMOUNT = 2 ATTACKER = scope:actor }
								}
							}
						}
						else = {
							increase_wounds_multiple_hostile_effect = { REASON = fight AMOUNT = 2 ATTACKER = scope:actor } 
						}
					}
				}
				40 = {
					if = { 
						limit = { 
							NAND = { 
								has_perk = sorcerer_dagons_call_perk 
								has_perk = warrior_dagons_call_perk 
							} 
						}
						add_piety = -100
					}
					POD_discipline_cost_effect = { COUNT = 1 }
					scope:recipient = { 
						if = {
							limit = { scope:condemn_the_sins_of_the_father = yes }
							house = {
								every_house_member = {
									limit = { age <= scope:recipient.age }
									increase_wounds_multiple_hostile_effect = { REASON = fight AMOUNT = 1 ATTACKER = scope:actor }
								}
							}
						}
						else = {
							increase_wounds_multiple_hostile_effect = { REASON = fight AMOUNT = 1 ATTACKER = scope:actor } 
						}
					}
				}
			}
		}
		scope:recipient = { 
			if = {
				limit = { is_alive = yes }
				if = {
					limit = { scope:condemn_the_sins_of_the_father = yes }
					house = {
						every_house_member = {
							add_opinion = {
								target = scope:actor
								modifier = hurt_opinion
								opinion = -25
							}
						}
					}
				}
				else = {
					add_opinion = {
						target = scope:actor
						modifier = hurt_opinion
						opinion = -25
					}
				}
			}
		}
	}
	
	auto_accept = yes

	#	#is_imprisoned = no 
	#	#is_landed = yes
	#	always = no
	#}
	#ai_targets = {
	#	ai_recipients = primary_war_enemies
	#}
	ai_frequency = 0
	ai_will_do = {
		base = 0
		
		#modifier = {
		#	add = 45
		#	scope:actor = {
		#	    is_at_war_with = scope:recipient
		#	}
		#}
		#modifier = {
		#	factor = 2
		#	scope:actor = {
		#		has_any_bad_relationship_with_character_trigger = {
		#			CHARACTER = scope:recipient
		#		}
		#	}
		#}
		#modifier = {

POD_cleansed_in_blood_perk_interaction = {
	category = interaction_category_vampire
	icon = POD_quietus_interaction
    desc = POD_cleansed_in_blood_perk_interaction_desc
	
	cooldown = { months = 6 }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_quietus_trigger = yes }
		scope:actor = { has_perk = cleansed_in_blood_perk }
		scope:recipient = { has_POD_shapeshifted_form_trigger = yes }
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			POD_has_enough_blood_for_discipline = { AMOUNT = 1 }
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }	
		}
		POD_cross_umbra_interaction_valid_trigger = yes				
	}
	
	on_accept = {
		scope:actor = { POD_willpower_rouse_check_effect = yes }
		scope:actor = { POD_discipline_cost_effect = { COUNT = 1 } }
		scope:actor = {
			duel = {
				skill = intrigue
				target = scope:recipient
				50 = {
					desc = POD_shared_loca_success
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					scope:recipient = {
						POD_shapeshift_revert_all_effect = yes
					}
				}
				50 = {
					desc = POD_shared_loca_failure
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier 
					}
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no 
		is_landed = yes 
	}
	ai_targets = {
		ai_recipients = primary_war_enemies
	}
	ai_frequency = 72
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 45
			scope:actor = {
			    is_at_war_with = scope:recipient
			}
		}
		modifier = {
			factor = 2
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
		modifier = {
			factor = 0
			scope:recipient = {
				is_ai = no
			}
		}
	}
}
