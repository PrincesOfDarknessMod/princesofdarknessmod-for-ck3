
POD_dagons_call_perk_interaction = {
	category = interaction_category_vampire
	desc = POD_dagons_call_perk_interaction_desc
	icon = POD_quietus_interaction
	
	cooldown = { years = 10 }
	
	is_shown = {
		scope:actor != scope:recipient
		scope:actor = { has_quietus_trigger = yes }
		scope:actor = {
			OR = {
				has_perk = sorcerer_dagons_call_perk
				has_perk = warrior_dagons_call_perk
			}
		}
		POD_hostile_interaction_is_shown = yes
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
			trigger_if = {
				limit = { prowess >= POD_prowess_very_high_skill_rating }
				POD_has_enough_blood_for_discipline = { AMOUNT = 3 }
			}
			trigger_else_if = {
				limit = { prowess >= POD_prowess_medium_skill_rating }
				POD_has_enough_blood_for_discipline = { AMOUNT = 2 }
			}
			trigger_else = { POD_has_enough_blood_for_discipline = { AMOUNT = 1 } }
		}
		POD_damage_spirit_trigger = yes
		scope:actor = { prowess >= POD_prowess_low_skill_rating }
		POD_interaction_is_valid_showing_failures_only = yes
	}

	#Use Condemn the Sins of the Father
	send_option = {
		is_shown = {
			scope:actor = { has_perk = condemn_the_sins_of_the_father_perk }
			scope:recipient = { exists = house }
		}
		flag = condemn_the_sins_of_the_father
		localization = condemn_the_sins_of_the_father_perk_name
	}

	send_options_exclusive = no

	on_accept = {
		scope:actor = {
			POD_willpower_rouse_check_effect = yes

			duel = {
				skill = prowess
				target = scope:recipient
				20 = {
					compare_modifier = { value = scope:duel_value multiplier = high_positive_duel_skill_multiplier }
					trigger = { prowess >= POD_prowess_very_high_skill_rating }
					if = {
						limit = {
							NAND = {
								has_perk = sorcerer_dagons_call_perk
								has_perk = warrior_dagons_call_perk
							}
						}
						add_piety = -500
					}
					POD_discipline_cost_effect = { COUNT = 3 }
					scope:recipient = {
						if = {
							limit = { scope:condemn_the_sins_of_the_father = yes }
							house = {
								every_house_member = {
									limit = { age <= scope:recipient.age }
									increase_wounds_multiple_hostile_effect = { REASON = fight AMOUNT = 2 ATTACKER = scope:actor }
								}
							}
						}
						else = { increase_wounds_multiple_hostile_effect = { REASON = fight AMOUNT = 3 ATTACKER = scope:actor } }
					}
				}
				30 = {
					compare_modifier = { value = scope:duel_value multiplier = low_positive_duel_skill_multiplier }
					trigger = { prowess >= POD_prowess_medium_skill_rating }
					if = {
						limit = {
							NAND = {
								has_perk = sorcerer_dagons_call_perk
								has_perk = warrior_dagons_call_perk
							}
						}
						add_piety = -250
					}
					POD_discipline_cost_effect = { COUNT = 2 }
					scope:recipient = {
						if = {
							limit = { scope:condemn_the_sins_of_the_father = yes }
							house = {
								every_house_member = {
									limit = { age <= scope:recipient.age }
									increase_wounds_multiple_hostile_effect = { REASON = fight AMOUNT = 1 ATTACKER = scope:actor }
								}
							}
						}
						else = {
							increase_wounds_multiple_hostile_effect = { REASON = fight AMOUNT = 2 ATTACKER = scope:actor }
						}
					}
				}
				50 = {
					if = {
						limit = {
							NAND = {
								has_perk = sorcerer_dagons_call_perk
								has_perk = warrior_dagons_call_perk
							}
						}
						add_piety = -100
					}
					POD_discipline_cost_effect = { COUNT = 1 }
					scope:recipient = {
						if = {
							limit = { scope:condemn_the_sins_of_the_father = yes }
							house = {
								every_house_member = {
									limit = { age <= scope:recipient.age }
									increase_wounds_multiple_hostile_effect = { REASON = fight AMOUNT = 1 ATTACKER = scope:actor }
								}
							}
						}
						else = {
							increase_wounds_multiple_hostile_effect = { REASON = fight AMOUNT = 1 ATTACKER = scope:actor }
						}
					}
				}
			}
		}
		scope:recipient = {
			if = {
				limit = { is_alive = yes }
				if = {
					limit = { scope:condemn_the_sins_of_the_father = yes }
					house = {
						every_house_member = {
							add_opinion = {
								target = scope:actor
								modifier = hurt_opinion
								opinion = -25
							}
						}
					}
				}
				else = {
					add_opinion = {
						target = scope:actor
						modifier = hurt_opinion
						opinion = -25
					}
				}
			}
		}
	}
	
	auto_accept = yes
	
	#ai_potential = {
	#	# The basic requirements
	#	#is_imprisoned = no
	#	#is_landed = yes
	#	always = no
	#}
	#ai_targets = {
	#	ai_recipients = primary_war_enemies
	#}
	ai_frequency = 0
	ai_will_do = {
		base = 0
		
		#modifier = {
		#	add = 45
		#	scope:actor = {
		#	    is_at_war_with = scope:recipient
		#	}
		#}
		#modifier = {
		#	factor = 2
		#	scope:actor = {
		#		has_any_bad_relationship_with_character_trigger = {
		#			CHARACTER = scope:recipient
		#		}
		#	}
		#}
		#modifier = {
		#	factor = 0
		#	scope:recipient = {
		#		is_ai = no
		#	}
		#}
	}
}

POD_cleansed_in_blood_perk_interaction = {
	category = interaction_category_vampire
	icon = POD_quietus_interaction

	cooldown = { months = 6 }

	is_shown = {
		scope:actor != scope:recipient
		scope:actor = { has_quietus_trigger = yes }
		scope:actor = { has_perk = cleansed_in_blood_perk }
		scope:recipient = { has_POD_shapeshifted_form_trigger = yes }
		POD_hostile_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			POD_has_enough_blood_for_discipline = { AMOUNT = 1 }
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}
	
	on_accept = {
		scope:actor = { POD_willpower_rouse_check_effect = yes }
		scope:actor = { POD_discipline_cost_effect = { COUNT = 1 } }
		scope:actor = {
			duel = {
				skill = intrigue
				target = scope:recipient
				50 = {
					desc = POD_shared_loca_success
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					scope:recipient = {
						POD_shapeshift_revert_all_effect = yes
					}
				}
				50 = {
					desc = POD_shared_loca_failure
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier
					}
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no
		is_landed = yes
	}
	ai_targets = {
		ai_recipients = primary_war_enemies
	}
	ai_frequency = 72
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 45
			scope:actor = {
				is_at_war_with = scope:recipient
			}
		}
		modifier = {
			factor = 2
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
		modifier = {
			factor = 0
			scope:recipient = {
				is_ai = no
			}
		}
	}
}
