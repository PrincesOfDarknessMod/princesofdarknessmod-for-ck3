	cost = { piety = major_piety_value }

		scope:actor = { POD_hunger_rouse_check_effect = yes	}
		scope:actor = {
			random_targeting_faction = {		
				destroy_faction = yes			
			}
		}
	}
	
	auto_accept = yes
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no 
		is_landed = yes
	}
	ai_targets = {
		ai_recipients = self
	}
	ai_frequency = 72
	ai_will_do = {
		base = 5
	}
}

POD_boon_of_anubis_perk_interaction = {
	category = interaction_category_vampire
	icon = POD_bardo_interaction
	desc = POD_boon_of_anubis_perk_interaction_desc

	cost = { piety = minor_piety_value }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_bardo_trigger = yes }
		scope:actor = { has_perk = boon_of_anubis_perk }
	}

	is_valid = {
		scope:recipient = { 
			trigger_if = { 
				limit = { NOT = { has_character_modifier = boon_of_anubis_perk_modifier } }
				NOR = {
					has_trait = vampire
					has_trait = shapeshifter
					has_trait = wraith
					has_trait = fae
					has_trait = demon
					has_trait = mummy
					has_trait = kueijin
					has_trait = spirit
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			POD_has_enough_blood_for_discipline = { AMOUNT = 3 }
		}	
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}

	on_accept = {
		scope:actor = {
			scope:actor = { POD_discipline_cost_effect = { COUNT = 3 } }
		}
		scope:recipient = { 
			if = {
				limit = { has_character_modifier = boon_of_anubis_perk_modifier }
				remove_character_modifier = boon_of_anubis_perk_modifier
			}
			else = { add_character_modifier = boon_of_anubis_perk_modifier }
		}
	}

	auto_accept = yes

	ai_potential = {
		# The basic requirements
		is_imprisoned = no 
		is_landed = yes 
		
		is_knight = no
		number_of_knights > 0
		primary_title.tier > tier_barony
	}
	ai_targets = {
		ai_recipients = peer_vassals
		ai_recipients = courtiers
		max = 10
	}
	ai_frequency = 72
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 100
			scope:recipient = {
				is_knight_of = scope:actor
			}
		}
	}
}
