					modifier = aversion_perk_modifier

				}
				add_opinion = {
					target = scope:actor
					modifier = hurt_opinion
					opinion = -100
				}
			}
		}
		else = {
			scope:recipient = {

					years = 3
				}

					target = scope:actor

	}
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no
		is_landed = yes
	}
	ai_targets = {
		ai_recipients = primary_war_enemies
	}
	ai_frequency = 72
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 45
			scope:actor = {
				is_at_war_with = scope:recipient
			}
		}
		modifier = {
			factor = 2
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
		modifier = {

	}

POD_rayzeels_song_interaction = {
	category = interaction_category_vampire
	icon = POD_valeren_interaction
	desc = POD_rayzeels_song_interaction_desc
	
	cooldown = { months = 6 }

	is_shown = {
		scope:actor = { has_valeren_trigger = yes }
		scope:actor = { has_perk = rayzeels_song_perk }
		scope:recipient = { is_ruler = yes }
		POD_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			POD_has_enough_blood_for_discipline = { AMOUNT = 5 }
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}
	
	on_accept = {
		if = {
			limit = { scope:actor != scope:recipient }
			scope:actor = { POD_commit_heal_effect = yes }
			scope:recipient = {
				add_opinion = {
					target = scope:actor
					modifier = pleased_opinion
					opinion = 15
				}
			}
		}
		scope:actor = { POD_willpower_rouse_check_effect = yes }
		scope:actor = { POD_discipline_cost_effect = { COUNT = 5 } }
		scope:recipient = {
			if = {
				limit = {
					POD_is_available = yes
					POD_can_be_healed_trigger = yes
				}
				if = {
					limit = { POD_health_has_superficial_wound_trigger = yes }
					POD_heal_superficial_wound_effect = { COUNT = POD_base_heal_value }
				}
				POD_remove_disease_traits_effect = yes
				POD_remove_supernatural_disease_traits_effect = yes
			}
			every_courtier_or_guest = {
				limit = {
					POD_is_available = yes
					POD_can_be_healed_trigger = yes
				}
				if = {
					limit = { POD_health_has_superficial_wound_trigger = yes }
					POD_heal_superficial_wound_effect = { COUNT = POD_base_heal_value }
				}
				POD_remove_disease_traits_effect = yes
				POD_remove_supernatural_disease_traits_effect = yes
				add_opinion = {
					target = scope:actor
					modifier = pleased_opinion
					opinion = 15
				}
			}
		}
	}
	
	auto_accept = yes

	ai_potential = {
		# The basic requirements
		is_imprisoned = no
		is_landed = yes
	}
	ai_targets = {
		ai_recipients = self
	}
	ai_frequency = 120
	ai_will_do = {
		base = 100
	}
}

POD_soul_exchange_interaction = {
	category = interaction_category_vampire
	icon = POD_valeren_interaction
	desc = POD_soul_exchange_interaction_desc
	
	cooldown = { months = 12 }	

	is_shown = {
		scope:actor = { has_valeren_trigger = yes }
		scope:actor = { has_perk = watcher_soul_exchange_perk }
		POD_hostile_mental_interaction_is_shown = yes
	}

	is_valid = {
		scope:recipient = {
			trigger_if = { limit = { NOT = { scope:actor = scope:recipient } } is_ruler = no }
		}		
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			POD_has_enough_blood_for_discipline = { AMOUNT = 5 }
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	populate_recipient_list = {
		scope:actor = {
			every_courtier_or_guest = {
				limit = { 
					POD_hostile_mental_interaction_is_shown = yes
				}
				add_to_list = characters
			}
		}
	}
	
	cost = { piety = major_piety_value }

	on_accept = {
		scope:actor = { POD_willpower_rouse_check_effect = yes }
		scope:actor = { POD_discipline_cost_effect = { COUNT = 5 } }
		scope:recipient = { add_opinion = { target = scope:actor modifier = hurt_opinion opinion = -30 } }
		scope:secondary_recipient = { add_opinion = { target = scope:actor modifier = hurt_opinion opinion = -30 } }
		
		show_as_tooltip = { 
			scope:recipient = { POD_custom_copy_vanilla_traits_effect = { ORIGIN = scope:secondary_recipient } }
			scope:recipient = { POD_remove_vanilla_traits_effect = yes }
			scope:recipient = { POD_copy_sexuality_effect = { ORIGIN = scope:secondary_recipient } }
			scope:secondary_recipient = { POD_custom_copy_vanilla_traits_effect = { ORIGIN = scope:recipient } } 
			scope:secondary_recipient = { POD_remove_vanilla_traits_effect = yes }
			scope:secondary_recipient = { POD_copy_sexuality_effect = { ORIGIN = scope:recipient } }
		}
		
		hidden_effect = {
			POD_create_dummyclone_effect = { CHARACTER = scope:recipient }
			if = {
				limit = { exists = scope:dummyclone }
				scope:dummyclone = { POD_custom_copy_vanilla_traits_effect = { ORIGIN = scope:recipient } }
				scope:dummyclone = { POD_copy_sexuality_effect = { ORIGIN = scope:recipient } }
				scope:recipient = { POD_remove_vanilla_traits_effect = yes }
				scope:recipient = { POD_custom_copy_vanilla_traits_effect = { ORIGIN = scope:secondary_recipient } }
				scope:recipient = { POD_copy_sexuality_effect = { ORIGIN = scope:secondary_recipient } }
				scope:secondary_recipient = { POD_remove_vanilla_traits_effect = yes }
				scope:secondary_recipient = { POD_custom_copy_vanilla_traits_effect = { ORIGIN = scope:dummyclone } }
				scope:secondary_recipient = { POD_copy_sexuality_effect = { ORIGIN = scope:dummyclone } }
				scope:dummyclone = { POD_remove_dummyclone_effect = yes }
			}
			else = {
				send_interface_toast = {
					title = POD_kueijin_spy.duel.failure
					left_icon = scope:actor
					right_icon = scope:recipient
				}
			}
		}
	}
	
	auto_accept = yes

	ai_frequency = 0
	
	ai_will_do = {
		base = 0
	}
}
