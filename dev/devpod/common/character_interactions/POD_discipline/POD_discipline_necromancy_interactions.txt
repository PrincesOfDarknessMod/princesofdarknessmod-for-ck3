POD_necromancy_gui_interaction = {
	category = interaction_category_vampire
	icon = POD_necromancy_interaction
	common_interaction = yes
	#interface_priority = 84
	
	is_shown = {
		scope:actor = {
			has_necromancy_trigger = yes
		}
		POD_hostile_interaction_is_shown = yes
	}

	}
	on_accept = {
		custom_tooltip = open_ns_menu
		
		scope:actor = {
			POD_open_necromagic_gui = { TARGET = scope:recipient }
		}
	}
	
	auto_accept = yes
}
