		NOT = { scope:actor = scope:recipient }

            object = scope:recipient

		scope:actor = {
			duel = {
				skill = diplomacy

						value = scope:duel_value

					}
					scope:recipient = {
						set_employer = scope:actor
						return_to_court = yes
						add_character_flag = {
							flag = blocked_from_leaving
							years = 5
						}
					}
				}
				30 = {
					desc = POD_sorcery_summon_interaction_fail

						value = scope:duel_value

POD_sorcery_weather_control_interaction = {

	options_heading = POD_weather_magic_interaction_options_heading
	
	cooldown = { months = 120 }
	cost = { piety = massive_piety_value }
	
	auto_accept = yes
	send_options_exclusive = yes

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_trait = sorcerer }
		scope:actor = { has_perk = weather_control_perk }
		scope:recipient = { is_landed = yes }
		scope:recipient.highest_held_title_tier > tier_barony
	}
	
	is_valid_showing_failures_only = {
	}
	
	send_option = {
		flag = flow
		localization = POD_weather_magic_flow
	}
	
	send_option = {
		flag = storm
		localization = POD_weather_magic_storm
	}
	
	send_option = {
		flag = earthquake
		localization = POD_weather_magic_earthquake
	}
	
	send_option = {
		flag = flood
		localization = POD_weather_magic_flood
	}
	
	send_option = {
		flag = drought
		localization = POD_weather_magic_drought
	}

	on_accept = {
	   scope:actor = {
			POD_willpower_rouse_check_effect = yes	
			
			if = {
				limit = { 
					scope:recipient.highest_held_title_tier = tier_empire
				}
				add_piety = -2000
			}
			else_if = {
				limit = { 
					scope:recipient.highest_held_title_tier >= tier_kingdom 
				}
				add_piety = -1000
			}
			else_if = {
				limit = { 
					scope:recipient.highest_held_title_tier >= tier_duchy 
				}
				add_piety = -500
			}
				
			if = {
				limit = { 
					scope:flow = yes
				}
				custom_tooltip = POD_weather_magic_interaction.tt.flow
			}
			if = {
				limit = { 
					scope:storm = yes
				}
				custom_tooltip = POD_weather_magic_interaction.tt.storm
			}
			if = {
				limit = { 
					scope:earthquake = yes
				}
				custom_tooltip = POD_weather_magic_interaction.tt.earthquake
			}
			if = {
				limit = { 
					scope:flood = yes
				}
				custom_tooltip = POD_weather_magic_interaction.tt.flood
			}
			if = {
				limit = { 
					scope:drought = yes
				}
				custom_tooltip = POD_weather_magic_interaction.tt.drought
			}
	
			duel = {
				skill = learning
				value = 10
				desc = POD_weather_magic.duel.desc
			
				50 = {
					desc = POD_weather_magic.duel.success
					
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					
					scope:actor = {
						send_interface_toast = {
							title = POD_weather_magic.duel.success
							left_icon = scope:actor
							right_icon = scope:recipient	
							custom_tooltip = POD_weather_magic.duel.tt.attacker_wins
						}
					}
					
					hidden_effect = {
						scope:recipient = {
							POD_weather_magic_specific_effect = yes
							every_vassal_or_below = {
								limit = { NOT = { primary_title.tier = tier_barony } }
								POD_weather_magic_specific_effect = yes
							}
						}
					}
				}
				
				50 = {
					desc = POD_weather_magic.duel.failure

					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier
					}
					
					scope:actor = {
						send_interface_toast = {
							title = POD_weather_magic.duel.failure
							left_icon = scope:actor
							right_icon = scope:recipient
							custom_tooltip = POD_weather_magic.duel.tt.attacker_lost
						}
					}
				}
			}
	   }
    }
	
	ai_frequency = 3600
	
	ai_targets = {
		ai_recipients = primary_war_enemies
	}
	
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 45
			scope:actor = {
			    is_at_war_with = scope:recipient
			}
			piety > omega_piety_gain
		}
		modifier = {
		    add = 10
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
	}

	}

	is_valid_showing_failures_only = {
	}
	
	send_option = {
		flag = knight
		localization = POD_sorcery_auto_resurrection_interaction_knight
	}

		localization = POD_sorcery_auto_resurrection_interaction_family
	}
	
	send_option = {
		flag = friend
		localization = POD_sorcery_auto_resurrection_interaction_friend
	}
	
	send_option = {
		flag = lover
		localization = POD_sorcery_auto_resurrection_interaction_lover
	}

	on_accept = {
		scope:actor = { 
			POD_willpower_rouse_check_effect = yes

				limit = { scope:knight = yes }

					limit = { is_alive = yes }

			}
			
			if = {
				limit = { scope:family = yes }
				house = {
					every_house_member = {
						limit = { 
							is_alive = yes 
							NOT = { this = scope:actor }
						}
						scope:actor = { add_to_variable_list = { name = auto_resurrect_list target = prev } }
					}
				}
			}
			
			if = {
				limit = { scope:friend = yes }
				every_relation = {
					type = potential_friend
					limit = { is_alive = yes }
					
					scope:actor = { add_to_variable_list = { name = auto_resurrect_list target = prev } }
				}
				every_relation = {
					type = friend
					limit = { is_alive = yes }
					
					scope:actor = { add_to_variable_list = { name = auto_resurrect_list target = prev } }
				}
				every_relation = {
					type = best_friend
					limit = { is_alive = yes }

			}

					scope:actor = { add_to_variable_list = { name = auto_resurrect_list target = prev } }

					type = soulmate
					limit = { is_alive = yes }
					
					scope:actor = { add_to_variable_list = { name = auto_resurrect_list target = prev } }
