######################################################
######   POD COMBAT DEBUG INTERACTIONS   ######
######################################################
#
# Debug interactions that launch combat through the same API effects
# used by real duels, so they exercise the full override flow including
# shapeshifting, scope stashing, and follow-up dispatch.
# Only visible in debug mode (debug_only = yes inside is_shown).

# 1v1: PC vs target — uses configure_start_single_combat_effect
pod_combat_debug_1v1_interaction = {
	category = interaction_debug_main
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	common_interaction = yes
	interface_priority = 1000

	icon = icon_combat

	desc = pod_combat_debug_1v1_desc

	is_shown = {
		debug_only = yes
		scope:actor != scope:recipient
		NOT = { scope:actor = { has_variable = pod_combat_active } }
	}

	on_accept = {
		configure_start_single_combat_effect = {
			SC_INITIATOR = scope:actor
			SC_ATTACKER = scope:actor
			SC_DEFENDER = scope:recipient
			FATALITY = possible
			FIXED = no
			LOCALE = terrain_scope
			OUTPUT_EVENT = pod_combat_debug.0001
			INVALIDATION_EVENT = pod_combat_debug.0002
		}
	}

	auto_accept = yes
}
