# POD Schemes for torpor 
# Karde

POD_awaketorpor_interaction = {
	category = interaction_category_hostile
	#common_interaction = yes
	#interface_priority = 82
	scheme = awaketorpor

	ai_frequency = 0

	is_shown = {
		scope:actor = { is_ai = no }
		scope:recipient = { is_ai = yes }
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		scope:recipient = { has_trait = torpor  }
		scope:recipient = { is_imprisoned = no }
		scope:recipient = { NOT = { has_character_modifier = enfant_modifier } }
	}

	is_valid_showing_failures_only = {
		custom_description = {
			object = scope:actor
			text = "I_can_only_plot_one_awaketorpor_at_time"
			scope:actor = {
				NOT = {
					any_scheme = {
						scheme_type = awaketorpor
					}
				}
			}
		}
		scope:recipient = {
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
			
		scope:actor = {
			can_start_scheme = {
				type = awaketorpor
				target = scope:recipient
			}
		}
	}

	desc = {
		triggered_desc = {
			trigger = {
				scope:actor = {
					NOT = { is_at_war_with = scope:recipient }
					can_start_scheme = {
						type = awaketorpor
						target = scope:recipient
					}

				}
			}
			desc = scheme_interaction_tt_awaketorpor_approved
		}

	}

	on_accept = {
		if = {
			limit = {
				scope:recipient = { 
					is_vampire_trigger = yes 
				}
			}
			scope:actor = {
				start_scheme = {
					type = awaketorpor
					target = scope:recipient
				}
			}
		}
		else = {
			scope:actor = { add_prestige = medium_prestige_loss }
			scope:recipient = { remove_trait = torpor }
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_awaketorpor_prisoner_interaction = {
	category = interaction_category_hostile
	icon = POD_awaketorpor_interaction
	desc = POD_awaketorpor_prisoner_interaction_desc

	ai_frequency = 0
	
	cost = { prestige = medium_prestige_value }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		scope:recipient = { has_trait = torpor  }
		scope:actor = { is_ai = no }
		scope:recipient = { is_ai = yes }
		scope:recipient = { is_imprisoned_by = scope:actor }
		scope:recipient = { NOT = { has_character_modifier = enfant_modifier } }
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
	}

	on_accept = {
		scope:recipient = { remove_trait = torpor }
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
