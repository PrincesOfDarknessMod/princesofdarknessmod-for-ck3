POD_chi_heal_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	interface_priority = 110
	icon = POD_art_interaction
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:actor = scope:recipient  }
				desc = POD_chi_heal_interaction_desc
			}
			desc = POD_yangchi_heal_interaction_desc
		}
	}
	options_heading = POD_chi_heal_interaction_options_heading

	cooldown = { months = 3 }	
	
	is_shown = {
		scope:actor = { has_POD_kj_blood_trigger = yes }
	}

	is_valid = {	
		scope:recipient = {
			POD_has_injury_or_main_trait_trigger = yes
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			custom_tooltip = {
				text = "i_am_empty_of_chi"
				NAND = { 
					has_character_modifier = yinchi0_modifier
					has_character_modifier = yangchi0_modifier
				}
			}
		}
		scope:recipient = {
			POD_can_be_healed_trigger = yes
		}	
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = scarred
			}
		}	
		is_valid = {
			custom_description = {
				text = POD_permanent_scarred
				object = scope:recipient
				scope:recipient = { NOT = { has_character_flag = POD_permanent_scarring_flag } }
			}
		}

		localization = POD_heal_scarred
	}
	
	send_option = {

					remove_trait = eunuch_1
					scope:actor = { POD_chi_cost_effect = { COUNT = 3 } }
				}
			}
		}
	}

	ai_potential = {
		is_imprisoned = no 
	}
	ai_targets = {
		ai_recipients = self
	}
	ai_targets = {
		ai_recipients = courtiers

	}

	ai_will_do = {
		base = 0
		
		modifier = {
			add = 100
			scope:recipient = {

}
