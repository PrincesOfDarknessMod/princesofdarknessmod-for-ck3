	common_interaction = yes

		}

			POD_can_be_healed_trigger = yes
		}	
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_any_mortal_disease_trigger = yes
			}
		}
		is_valid = {
			scope:actor = { 
				has_perk = transcendant_living_flame_perk
			}
		}
		flag = disease
		localization = POD_heal_disease
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = scarred
			}
		}
		is_valid = {
			custom_description = {
				text = POD_permanent_scarred
				object = scope:recipient

			limit = { NOT = { scope:actor = scope:recipient } }
			scope:actor = { POD_commit_heal_effect = yes }
		}
		scope:actor = { 
			POD_willpower_rouse_check_effect = yes
		}
		scope:recipient = {
			play_music_cue = "mx_cue_peace_ensues"
			switch = {
				trigger = yes
				scope:scarred = {
					remove_trait = scarred
					scope:actor = { POD_chi_cost_effect = { COUNT = 1 } }

				scope:disfigured = {
					remove_trait = disfigured
					scope:actor = { POD_chi_cost_effect = { COUNT = 2 } }
				}
				scope:wounded = {

					scope:actor = { POD_chi_cost_effect = { COUNT = 1 } }
				}
				scope:one_eyed = {
					remove_trait = one_eyed
					scope:actor = { POD_chi_cost_effect = { COUNT = 2 } }
				}
				scope:blind = {
					remove_trait = blind
					add_trait = one_eyed
					scope:actor = { POD_chi_cost_effect = { COUNT = 2 } }
				}
				scope:one_legged = {
					remove_trait = one_legged
					scope:actor = { POD_chi_cost_effect = { COUNT = 3 } }
				}

			limit = { scope:actor = { has_perk = lingering_aid_of_the_caregiver_perk } }

				add_character_modifier = { 
					modifier = renewed_vitality_modifier 
					days = 400
				}
			}
		}

			limit = { scope:actor = { has_perk = sweet_dreamss_caress_perk } }

		}

			limit = { NOT = { scope:actor = scope:recipient } }
