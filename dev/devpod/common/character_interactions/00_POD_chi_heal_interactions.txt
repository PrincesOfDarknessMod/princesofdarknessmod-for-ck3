#  Chi Heal
POD_chi_heal_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	interface_priority = 110
	icon = POD_art_interaction
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:actor = scope:recipient  }
				desc = POD_chi_heal_interaction_desc
			}

		}
	}
	options_heading = POD_chi_heal_interaction_options_heading

	cooldown = { months = 3 }

		scope:actor = { 
			OR = {
				has_trait = kueijin

			}

	is_valid_showing_failures_only = {
		scope:actor = {
			NAND = { 
				has_character_modifier = yinchi0_modifier
				has_character_modifier = yangchi0_modifier
			}
		}
		scope:recipient = {
			NOT = { has_trait_with_flag = can_not_cure }
			NOT = { has_character_modifier = eye_of_sakhmet_modifier }
			custom_description = {
				text = POD_has_main_trait
				object = scope:recipient
				OR = {
					has_trait = scarred
					has_trait = disfigured
					has_trait = blind
					has_trait = one_eyed
					has_trait = maimed
					has_trait = one_legged
					has_trait = eunuch
				
					has_trait = wounded_1
					has_trait = wounded_2
					has_trait = wounded_3
				}
			}
		}	
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = scarred
			}
		}	
		is_valid = {
			custom_description = {
				text = POD_permanent_scarred
				object = scope:recipient
				scope:recipient = { NOT = { has_character_flag = POD_permanent_scarring_flag } }

	on_accept = {
		scope:actor = { 
			POD_willpower_rouse_check_effect = yes

			play_music_cue = "mx_cue_peace_ensues"

				limit = {
					scope:scarred = yes

					remove_trait = scarred
				}

					scope:disfigured = yes

			}

				limit = {
					scope:one_eyed = yes
				}

					remove_trait = one_eyed
				}
				scope:actor = { POD_chi_decrease_effect = { COUNT = 2 } }
			}	

			if = {
				limit = {
					scope:blind = yes
				}
				scope:recipient = {
					remove_trait = blind
					add_trait = one_eyed
				}
				scope:actor = { POD_chi_decrease_effect = { COUNT = 2 } }
			}	

			if = {
				limit = {
					scope:one_legged = yes
				}
				scope:recipient = {
					remove_trait = one_legged
				}
				scope:actor = { POD_chi_decrease_effect = { COUNT = 3 } }
			}	

			if = {
				limit = {
					scope:maimed = yes
				}
				scope:recipient = {
					remove_trait = maimed
				}
				scope:actor = { POD_chi_decrease_effect = { COUNT = 3 } }
			}

			if = {
				limit = {
					scope:eunuch = yes
				}
				scope:recipient = {
					remove_trait = eunuch
				}
				scope:actor = { POD_chi_decrease_effect = { COUNT = 3 } }
			}
		}
	}
	
	auto_accept = yes

	ai_potential = {
		is_imprisoned = no 
	}
	ai_targets = {
		ai_recipients = self
	}
	ai_targets = {
		ai_recipients = courtiers
		max = 10
	}
	ai_frequency = 12
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 100
			scope:recipient = {
				OR = {
					this = scope:actor
					is_knight_of = scope:actor
				}
			}
		}
	}
}
