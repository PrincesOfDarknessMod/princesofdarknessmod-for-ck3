POD_set_thrall_interaction = {
	category = interaction_category_demon
	common_interaction = yes
	interface_priority = 100
	desc = POD_set_thrall_interaction_desc
	icon = POD_arcana_interaction

	ai_min_reply_days = 4
	ai_max_reply_days = 9
	
	cost = { prestige = minor_prestige_value }
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			is_landed = yes
			has_perk = enthrall_perk
		}
		scope:recipient = { 
			NOT = { has_relation_demonic_master = scope:actor }
		}
		POD_is_in_same_plane_trigger = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			NOT =  { is_at_war_with = scope:recipient }
		}
		scope:recipient = {
			faith = scope:actor.faith
			is_busy_in_events_localised = yes
			NOT = { has_strong_hook = scope:actor }
			NOT = { has_trait = demon }
			NOT = { has_trait = effigy }
			NOT = { has_trait = fae } # TODO: allow changeling be thrall
			NOT = { has_trait = wraith }
		}
	}

	#Use hook
	send_option = {
		is_valid = {
			scope:actor = {
				has_usable_hook = scope:recipient
			}
		}
		flag = hook
		localization = SCHEME_HOOK
	}
	should_use_extra_icon = {
		scope:actor = { has_usable_hook = scope:recipient }
	}
	extra_icon = "gfx/interface/icons/character_interactions/hook_icon.dds"

	send_options_exclusive = no

	on_accept = {
		scope:actor = {
			POD_resolve_decrease_effect = yes
		}
		scope:recipient = {
			add_trait = infernalist
			set_relation_demonic_master = scope:actor
				
			if = {
				limit = {
					has_trait = supehunter 
				}
				scope:actor = { save_scope_as = vampiresire }
				remove_trait = supehunter
				add_trait_force_tooltip = mortal
				POD_hunterembrace_effect = yes
				POD_remove_shih_atribute_effect = yes
			}
			
			show_as_tooltip = {
			POD_character_resolve_initiation_effect = yes
			if = {
				limit = {
					exists = scope:actor.dynasty
					scope:actor.dynasty = { has_dynasty_perk = pacts_legacy_1 }
				}
			scope:recipient = { add_learning_skill = 5 }
			}
			if = {
				limit = {
					exists = scope:actor.dynasty
					scope:actor.dynasty = { has_dynasty_perk = pacts_legacy_2 }
				}
			scope:recipient = { add_diplomacy_skill = 5 }
			}			
			if = {
				limit = {
					exists = scope:actor.dynasty
					scope:actor.dynasty = { has_dynasty_perk = pacts_legacy_3 }
				}
			scope:recipient = { add_intrigue_skill = 5 }
			}			
			if = {
				limit = {
					exists = scope:actor.dynasty
					scope:actor.dynasty = { has_dynasty_perk = pacts_legacy_4 }
				}
			scope:recipient = { add_stewardship_skill = 5 }
			}
			if = {
				limit = {
					exists = scope:actor.dynasty
					scope:actor.dynasty = { has_dynasty_perk = pacts_legacy_5 }
				}
			scope:recipient = { add_martial_skill = 5 }
			}
			}			
		}
	}
	
	ai_targets = {
		ai_recipients = vassals
	}
	
	ai_frequency = 120
	
	ai_potential = {
		primary_title.tier > tier_county
		prestige > minor_prestige_value
	}
	
	ai_accept = {
		base = 0 # Try to make it 0 for most interactions

		modifier = {
			scope:hook = yes
			add = 50
			desc = SCHEME_WEAK_HOOK_USED
		}
		
		opinion_modifier = { # Opinion Factor
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.5
			desc = AI_SIMPLE_OPINION_REASON
		}
		
		modifier = {
			add = -50
			NOT = { scope:actor.faith = scope:recipient.faith }
			desc = "ZEAL_AGAINST_INFIDELS"	
		}

		modifier = {
			add = 25
			scope:actor.faith = scope:recipient.faith 
			desc = "SAME_FAITH"
		}

		modifier = {
			add = -50
			has_relation_rival = scope:recipient
		}
		modifier = {
			add = -80
			has_relation_nemesis = scope:recipient
		}
	}
	
	ai_will_do = {
		base = 100
		
		modifier = { # The AI will only use a Hook if they couldn't otherwise do this
			scope:hook = yes
			add = -1
		}
	}
}

			trigger_if = { 
				limit = { NOT = { has_perk = soul_burn_perk } }

			}
			trigger_if = {
				limit = { has_trait = earthbound }

			}
		}
		scope:recipient = {
			has_relation_demonic_master = scope:actor
		}
	}
	
	on_accept = {
		scope:actor = { POD_resolve_decrease_multiple_effect = { COUNT = 5 } }

				}
				add_piety = -10000
				remove_variable = spirit_location
				remove_character_modifier = trapped_in_abyss_modifier
			}
			if = { 
				limit = { has_trait = earthbound }
				if = { 
					limit = { has_variable = stone_prison }
					save_scope_as = owner
					var:stone_prison = {
						save_scope_as = this_artifact

						remove_variable = reliquarie
						set_variable = impressed_artifact
					}
					remove_variable = stone_prison
				}
				if = { 
					limit = { has_variable = earth_bound }
					remove_variable = earth_bound
				}
				remove_trait = earthbound
			}
			add_trait_force_tooltip = truefallen
			set_relation_possessed_target = scope:recipient
		}
		scope:recipient = { 
			add_trait = possessed_genetic
			copy_inheritable_appearance_from = scope:actor
		}
	}
	
	auto_accept = yes
	
	ai_frequency = 60

	ai_potential = {
		# The basic requirements
		is_imprisoned = no 
		is_landed = yes
		primary_title.tier > tier_barony
		
		has_trait = truefallen
	}
	ai_targets = {
		ai_recipients = vassals
		max = 10
	}
	
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 100
			num_of_relation_possessed_target = 0
		}
	}
}

POD_demonic_submission_interaction = {
	category = interaction_category_demon
	icon = grant_titles_interaction
	desc = POD_demonic_submission_interaction_desc
	
	cost = { prestige = 250 }
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_trait = demon }
		POD_incompatible_splats_trigger = no
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			is_landed = yes
		}
		scope:actor = {
			highest_held_title_tier >= tier_duchy
			NOT = { is_vassal_of = scope:recipient }
		}
	}

	on_accept = {
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = title_change
			add_claim_on_loss = yes
		}
		if = {
			limit = { scope:actor.primary_title.tier = tier_duchy }
			scope:actor = {
				every_held_title = {
					limit = { tier = tier_duchy }
					change_title_holder = {
						holder = scope:recipient
						change = scope:title_change
					}
					scope:recipient = {
						add_opinion = {
							target = scope:actor
							modifier = received_title_duchy
						}
					}
				}
			}
		}
		else_if = {
			limit = { scope:actor.primary_title.tier = tier_kingdom }
			scope:actor = {
				every_held_title = {
					limit = { tier = tier_kingdom }
					change_title_holder = {
						holder = scope:recipient
						change = scope:title_change
					}
					scope:recipient = {
						add_opinion = {
							target = scope:actor
							modifier = received_title_kingdom
						}
					}
				}
			}
		}
		else_if = {
			limit = { scope:actor.primary_title.tier = tier_empire }
			scope:actor = {
				every_held_title = {
					limit = { tier = tier_empire }
					change_title_holder = {
						holder = scope:recipient
						change = scope:title_change
					}
					scope:recipient = {
						add_opinion = {
							target = scope:actor
							modifier = received_title_empire
						}
					}
				}
			}
		}
		resolve_title_and_vassal_change = scope:title_change
		
		create_title_and_vassal_change = {
			type = swear_fealty
			save_scope_as = change
		}
		scope:actor = {
			change_liege = {
				liege = scope:recipient
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
