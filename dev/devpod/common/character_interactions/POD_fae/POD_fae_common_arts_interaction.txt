POD_soothsay_interaction = {
	category = interaction_category_fae
	icon = POD_soothsay_interaction
	desc = POD_soothsay_interaction_desc

	cooldown = { months = realm_time_monthly_cooldown }

	use_diplomatic_range = no
	
	is_shown = {
		scope:actor = { has_trait = soothsay }
		POD_hostile_interaction_is_shown = yes
	}
	
	is_valid_showing_failures_only = {
		scope:recipient = {
			POD_not_has_cantrip_defenses_trigger = yes
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}
	
	send_option = {
		is_shown = {
			NOT = { scope:actor = scope:recipient }
		}
		is_valid = {
			scope:actor = {
				has_perk = common_arts_omen_perk
				POD_has_enough_glamour_for_character_cantrip_interaction = { AMOUNT = 1 }
			}
			scope:recipient = {
				custom_description = {
					object = scope:recipient
					text = "strong_mental_defenses_trigger"
					strong_mental_defenses_art_interaction_trigger = yes
				}
			}
		}
		flag = omen
		localization = common_arts_omen_perk_name
	}
	
	send_option = {
		is_shown = {
			scope:actor = scope:recipient
		}
		is_valid = {
			scope:actor = {
				has_perk = common_arts_seers_wisp_perk
				has_perk = realm_prop_1_perk
				POD_has_enough_glamour_for_character_cantrip_interaction = { AMOUNT = 3 }
			}
		}
		flag = seers_wisp_artifact
		localization = POD_soothsay_interaction_seers_wisp_artifact
	}
	
	send_option = {
		is_shown = {
			scope:actor = scope:recipient
		}
		is_valid = {
			scope:actor = {
				has_perk = common_arts_seers_wisp_perk
				has_perk = realm_prop_1_perk
				POD_has_enough_glamour_for_character_cantrip_interaction = { AMOUNT = 2 }
			}
		}
		flag = seers_wisp_gold
		localization = POD_soothsay_interaction_seers_wisp_gold
	}
	
	send_option = {
		is_shown = {
			scope:actor = scope:recipient
		}
		is_valid = {
			scope:actor = {
				has_perk = common_arts_augury_perk
				POD_has_enough_glamour_for_character_cantrip_interaction = { AMOUNT = 4 }
			}
		}
		flag = augury
		localization = common_arts_augury_perk_name
	}
	
	send_option = {
		is_valid = {
			scope:actor = {
				has_perk = common_arts_face_fate_perk
				POD_has_enough_glamour_for_character_cantrip_interaction = { AMOUNT = 2 }
			}
		}
		flag = face_fate_positive
		localization = POD_soothsay_interaction_face_fate_positive
	}
	
	send_option = {
		is_valid = {
			scope:actor = {
				has_perk = common_arts_face_fate_perk
				POD_has_enough_glamour_for_character_cantrip_interaction = { AMOUNT = 2 }
			}
		}
		flag = face_fate_negative
		localization = POD_soothsay_interaction_face_fate_negative
	}
	
	on_accept = {
		scope:actor = {
			save_scope_value_as = {
				name = cantrip_difficulty_soothsay_reduction
				value = {
					value = 0
					if = { limit = { has_perk = common_arts_tattletale_perk } add = 1 }
					if = { limit = { has_perk = common_arts_augury_perk } add = 1 }
					if = { limit = { has_perk = common_arts_face_fate_perk } add = 2 }
				}
			}
			switch = {
				trigger = yes
				scope:omen = {
					POD_character_cantrip_interaction_cost = { COUNT = 1 }
					save_scope_value_as = {
						name = cantrip_difficulty
						value = {
							value = cantrip_difficulty_easy
							subtract = scope:cantrip_difficulty_soothsay_reduction
						}
					}
					POD_bunk_effect = {
						SKILL = learning
						DIFFICULTY = scope:cantrip_difficulty
						SUCCESS = "
							custom_tooltip = {
								text = POD_kueijin_spy.duel.tt.evaluate
								scope:recipient = {
									every_secret = {
										reveal_to = scope:actor
									}
								}
							}
						"
						FAILURE = ""
					}
				}
				scope:seers_wisp_artifact = {
					POD_character_cantrip_interaction_cost = { COUNT = 3 }
					save_scope_value_as = {
						name = cantrip_difficulty
						value = {
							value = cantrip_difficulty_standard
							subtract = scope:cantrip_difficulty_soothsay_reduction
						}
					}
					POD_bunk_effect = {
						SKILL = learning
						DIFFICULTY = scope:cantrip_difficulty
						SUCCESS = "
							save_scope_as = owner
							switch = {
								trigger = realm_prop_level
								1 = { set_artifact_rarity_common = yes }
								2 = { set_artifact_rarity_masterwork = yes }
								3 = { set_artifact_rarity_famed = yes }
								4 = { set_artifact_rarity_illustrious = yes }
								5 = { set_artifact_rarity_illustrious = yes }
							}
							create_random_artifact_effect = yes
							scope:newly_created_artifact ?= { set_variable = summoned_artifact }
						"
						FAILURE = ""
					}
				}
				scope:seers_wisp_gold = {
					POD_character_cantrip_interaction_cost = { COUNT = 2 }
					save_scope_value_as = {
						name = cantrip_difficulty
						value = {
							value = cantrip_difficulty_easy
							subtract = scope:cantrip_difficulty_soothsay_reduction
						}
					}
					POD_bunk_effect = {
						SKILL = learning
						DIFFICULTY = scope:cantrip_difficulty
						SUCCESS = "
							switch = {
								trigger = realm_prop_level
								1 = { add_gold = 20 }
								2 = { add_gold = 40 }
								3 = { add_gold = 60 }
								4 = { add_gold = 80 }
								5 = { add_gold = 100 }
							}
						"
						FAILURE = ""
					}
				}
				scope:augury = {
					POD_character_cantrip_interaction_cost = { COUNT = 4 }
					POD_bunk_effect = {
						SKILL = learning
						DIFFICULTY = cantrip_difficulty_standard
						SUCCESS = "
							custom_tooltip = POD_discern_the_aura_interaction_effect
							set_variable = { name = character_creator_type value = flag:find_mortal }
							set_variable = { name = character_creator_points value = 15 }

							if = {
								limit = {
									any_directly_owned_province = {
										has_building_with_flag = grand_city_commoners
									}
								}
								custom_tooltip = POD_find_mortals_interaction_grand_city
							}
						"
						FAILURE = ""
					}
				}
				scope:face_fate_positive = {
					POD_character_cantrip_interaction_cost = { COUNT = 2 }
					POD_bunk_effect = {
						SKILL = learning
						DIFFICULTY = cantrip_difficulty_standard
						SUCCESS = "
							scope:recipient = { 
								add_fae_character_modifier = { MODIFIER = face_fate_positive_modifier }
							}
						"
						FAILURE = ""
					}
				}
				scope:face_fate_negative = {
					POD_character_cantrip_interaction_cost = { COUNT = 2 }
					POD_bunk_effect = {
						SKILL = learning
						DIFFICULTY = cantrip_difficulty_standard
						SUCCESS = "
							scope:recipient = { 
								add_fae_character_modifier = { MODIFIER = face_fate_negative_modifier }
							}
						"
						FAILURE = ""
					}
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = { base = 0 }
}
