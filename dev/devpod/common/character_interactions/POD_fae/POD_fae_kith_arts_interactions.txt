	can_be_picked = {
		scope:target = {
			POD_title_can_receive_county_modifiers = yes

		scope:actor = { has_perk = dream_time_perk }
		scope:recipient = { is_landed = yes }
		scope:recipient.highest_held_title_tier > tier_barony
		POD_hostile_interaction_is_shown = yes
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
			POD_has_enough_glamour_for_area_cantrip_interaction = {
				AMOUNT = POD_area_cantrip_interaction_glamour_cost
			}
			trigger_if = {
				limit = { POD_area_cantrip_interaction_piety_cost > 0 }
				piety >= POD_area_cantrip_interaction_piety_cost
			}
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}
	
	send_option = {
		flag = positive
		localization = dream_time_perk_county_positive_modifier
	}
	
	send_option = {
		flag = negative
		localization = dream_time_perk_county_negative_modifier
	}
	
	send_option = {
		flag = remove
		localization = TITLE_VIEW_CLEAR_LAWS_ACCEPT
	}

	on_accept = {
	   scope:actor = {
			POD_willpower_rouse_check_effect = yes
			if = {
				limit = { POD_area_cantrip_interaction_piety_cost > 0 }
				add_piety = {
					value = POD_area_cantrip_interaction_piety_cost
					multiply = -1
				}
			}
			POD_area_cantrip_interaction_cost = {
				COUNT = POD_area_cantrip_interaction_glamour_cost
			}
			duel = {
				desc = bunk.duel.desc
				skill = learning
				value = low_skill_rating
				50 = {
					desc = POD_shared_loca_success
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					scope:target = {
						if = {
							limit = { scope:positive = yes }
							custom_tooltip = POD_dream_time_interaction_desc.positive
						}
						if = {
							limit = { scope:negative = yes }
							custom_tooltip = POD_dream_time_interaction_desc.negative
						}
						hidden_effect = {
							every_de_jure_county = {
								POD_dream_time_effect = yes
							}
						}
					}
				}
				50 = {
					desc = POD_shared_loca_failure
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier

				}
			}

		# The basic requirements
		is_imprisoned = no
		is_landed = yes
	}
	ai_targets = {
		ai_recipients = self
	}

		ai_recipients = primary_war_enemies
	}
	ai_frequency = 120
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 100
			scope:actor = scope:recipient
			scope:positive = yes
		}
		modifier = {
			add = 45
			scope:actor != scope:recipient
			scope:negative = yes
			scope:actor = {
				is_at_war_with = scope:recipient
			}
		}
		modifier = {
			factor = 2

POD_dream_time_character_interaction = {

			POD_not_has_cantrip_defenses_trigger = yes
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}
	
	send_option = {
		is_shown = {
			scope:recipient = {
				NOT = { has_character_modifier = dream_time_perk_character_positive_modifier }
			}
		}
		flag = positive
		localization = dream_time_perk_character_positive_modifier
	}
	
	send_option = {
		is_shown = {
			scope:actor != scope:recipient
			scope:recipient = {
				NOT = { has_character_modifier = dream_time_perk_character_negative_modifier }
			}
		}
		flag = negative
		localization = dream_time_perk_character_negative_modifier
	}
	
	send_option = {
		is_shown = {
			scope:recipient = {
				OR = {
					has_character_modifier = dream_time_perk_character_positive_modifier
					has_character_modifier = dream_time_perk_character_negative_modifier
				}
			}
		}
		flag = remove
		localization = TITLE_VIEW_CLEAR_LAWS_ACCEPT
	}
	
	on_accept = {
		scope:actor = {
			POD_willpower_rouse_check_effect = yes
			POD_character_cantrip_interaction_cost = { COUNT = 1 }
		}
		if = {
			limit = { scope:positive = yes }
			scope:actor = {
				duel = {
					desc = bunk.duel.desc
					skill = intrigue
					value = low_skill_rating
					50 = {
						desc = POD_shared_loca_success
						compare_modifier = {
							value = scope:duel_value
							multiplier = high_positive_duel_skill_multiplier
						}
						scope:recipient = {
							add_fae_character_modifier = {
								MODIFIER = dream_time_perk_character_positive_modifier
							}
						}
					}
					50 = {
						desc = POD_shared_loca_failure
						compare_modifier = {
							value = scope:duel_value
							multiplier = low_negative_duel_skill_multiplier
						}
						POD_bunk_failure_modifier = yes
					}
				}
			}
			if = {
				limit = { scope:actor != scope:recipient }
				scope:recipient = {
					add_opinion = {
						target = scope:actor
						modifier = gift_opinion
						opinion = 5
					}
				}
			}
		}
		else_if = {
			limit = { scope:negative = yes }
			scope:actor = {
				duel = {
					desc = bunk.duel.desc
					skill = intrigue
					target = scope:recipient
					50 = {
						desc = POD_shared_loca_success
						compare_modifier = {
							value = scope:duel_value
							multiplier = high_positive_duel_skill_multiplier
						}
						scope:recipient = {
							add_fae_character_modifier = {
								MODIFIER = dream_time_perk_character_negative_modifier
							}
						}
					}
					50 = {
						desc = POD_shared_loca_failure
						compare_modifier = {
							value = scope:duel_value
							multiplier = low_negative_duel_skill_multiplier
						}
						POD_bunk_failure_modifier = yes
					}
				}
			}
		}
		else = {
			scope:recipient = {
				if = {
					limit = { has_character_modifier = dream_time_perk_character_positive_modifier }
					if = {
						limit = { scope:actor != scope:recipient }
						scope:recipient = {
							add_opinion = {
								target = scope:actor
								modifier = gift_opinion
								opinion = -5
							}
						}
					}
					remove_character_modifier = dream_time_perk_character_positive_modifier
				}
				POD_remove_character_modifier = { MODIFIER = dream_time_perk_character_negative_modifier }
			}
		}
	}
	
	auto_accept = yes
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no
		is_landed = yes
	}
	ai_targets = {
		ai_recipients = self
	}
	ai_frequency = 36
	ai_will_do = {
		base = 0
		modifier = {
			add = 100
			scope:actor = scope:recipient
			scope:positive = yes
		}
	}
