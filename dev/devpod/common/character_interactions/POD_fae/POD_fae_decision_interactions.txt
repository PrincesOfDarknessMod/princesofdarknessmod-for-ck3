POD_season_kingmaker_interaction = {
	category = interaction_category_fae
	icon = hook_icon
	desc = POD_season_kingmaker_interaction_desc
	use_diplomatic_range = no
	
	cooldown_against_recipient = { years = 10 }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			has_character_modifier = season_kingmaker_modifier
			can_add_hook = {
				target = scope:recipient
				type = favor_hook
			}
		}
		POD_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		trigger_if = {
			limit = { faith:solstice_court = { has_doctrine = doctrine_winter_time } }
			scope:recipient = { religion = religion:winter_court_religion }
		}
		trigger_if = {
			limit = { faith:solstice_court = { has_doctrine = doctrine_spring_time } }
			scope:recipient = { religion = religion:spring_court_religion }
		}
		trigger_if = {
			limit = { faith:solstice_court = { has_doctrine = doctrine_summer_time } }
			scope:recipient = { religion = religion:summer_court_religion }
		}
		trigger_if = {
			limit = { faith:solstice_court = { has_doctrine = doctrine_autumn_time } }
			scope:recipient = { religion = religion:autumn_court_religion }
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:actor = {
			add_hook = {
				type = favor_hook
				target = scope:recipient
			}
		}
	}
	
	auto_accept = yes
}

POD_changeling_interaction = {
	category = interaction_category_fae
	icon = POD_fae_interaction
	desc = POD_changeling_interaction_desc
	
	cooldown = { years = 10 }
	
	is_shown = {
		NOT = { scope:recipient = scope:actor }
		scope:actor = { has_trait = fae }
		scope:recipient = { NOT = { has_trait = fae } }
		POD_hostile_mental_interaction_is_shown = yes
	}

	is_valid = {
		scope:actor = { any_child = { is_adult = no } }
		scope:recipient = { any_child = { is_adult = no } }
	}

	is_valid_showing_failures_only = {
		POD_interaction_is_valid_showing_failures_only = yes
	}

	populate_actor_list = {
		scope:actor = {
			every_child = {
				limit = { is_adult = no }
				add_to_list = characters
			}
		}
	}

	populate_recipient_list = {
		scope:recipient = {
			every_child = {
				limit = { is_adult = no }
				add_to_list = characters
			}
		}
	}
	
	on_accept = {
		scope:actor = {
			duel = {
				skill = intrigue
				target = scope:recipient
				50 = {
					desc = POD_shared_loca_success
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					POD_changeling_effect = {
						CHILD = scope:secondary_recipient
						ADOPTER = scope:actor
					}
					POD_changeling_effect = {
						CHILD = scope:secondary_actor
						ADOPTER = scope:recipient
					}
				}
				50 = {
					desc = POD_shared_loca_failure
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier
					}
				}
			}
		}
	}
}

POD_call_patron_interaction = {
	category = interaction_category_fera
	common_interaction = yes
	desc = POD_call_patron_interaction_desc
	interface_priority = 60
	interface = call_ally
	special_interaction = call_ally_interaction
	popup_on_receive = yes
	pause_on_receive = yes
	icon = alliance

	greeting = positive
	notification_text = CALL_ALLY_NOTIFICATION

	is_shown = {
		scope:actor = { is_at_war = yes }
		scope:actor = { has_trait = patron_council }
		POD_interaction_is_shown = yes
	}
	
	has_valid_target = {
		exists = scope:target
	}

	has_valid_target_showing_failures_only = {
		scope:target = {
			is_war_leader = scope:actor
		}

		joiner_not_already_in_another_war_with_any_target_war_participants_trigger = {
			WARRIOR = scope:actor
			JOINER = scope:recipient
		}
	}
	
	is_valid = {
		trigger_if = {
			limit = {
				scope:actor = {
					OR = {
						has_character_modifier = philospher_queen_modifier
						has_character_modifier = philospher_king_modifier
					}
				}
			}
			scope:recipient = {
				has_trait = patron_council
			}
		}
		trigger_else_if = {
			limit = { scope:actor = { has_trait = patron_council_austere_conclave } }
			scope:recipient = { has_trait = patron_council_austere_conclave }
		}
		trigger_else_if = {
			limit = { scope:actor = { has_trait = patron_council_seeker } }
			scope:recipient = { has_trait = patron_council_seeker }
		}
		trigger_else = {
			scope:recipient = { has_trait = patron_council_vanguard }
		}
		scope:target = {
			trigger_if = {
				limit = { casus_belli.primary_attacker = scope:actor }
				casus_belli.primary_defender = {
					NOT = {
						has_trait = patron_council
					}
				}
			}
			trigger_if = {
				limit = { casus_belli.primary_defender = scope:actor }
				casus_belli.primary_attacker = {
					NOT = {
						has_trait = patron_council
					}
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			is_ruler = yes
		}

		# Only War Leaders can call allies
		scope:actor = {
			custom_description = {
				text = must_be_war_leader
				any_character_war = {
					is_war_leader = scope:actor
				}
			}
		}

		# Cannot call the same character multiple times
		scope:actor = {
			custom_description = {
				text = cannot_call_same_character_multiple_times
				subject = scope:recipient
				any_character_war = {
					ALL_FALSE = {
						was_called = scope:recipient
						is_participant = scope:recipient
					}
				}
			}
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	can_be_picked = {
		can_join_war_liege_vassal_check_trigger = { WARRIOR = scope:actor JOINER = scope:recipient }

		scope:target = {
			is_war_leader = scope:actor

			using_cb = POD_blood_hunt_war
			
			# not already in target war
			custom_description = {
				text = join_war_interaction_already_in_target_war
				NOR = {
					any_war_attacker = { this = scope:recipient }
					any_war_defender = { this = scope:recipient }
				}
			}
		}
	}

	on_auto_accept = {
		scope:target = {
			if = {
				limit = {
					scope:recipient = {
						any_active_accolade = {
							has_accolade_parameter = acclaimed_knight_piety_from_battle
						}
					}
				}
				scope:recipient = {
					add_piety = miniscule_piety_gain
				}
			}
			if = {
				limit = {
					scope:recipient = {
						any_active_accolade = {
							has_accolade_parameter = acclaimed_knight_piety_from_battle_high
						}
					}
				}
				scope:recipient = {
					add_piety = minor_piety_gain
				}
			}
		}
	}
		
	on_accept = {
		blood_hunt_call_interaction_effect = yes
		scope:target = {
			if = {
				limit = {
					scope:recipient = {
						any_active_accolade = {
							has_accolade_parameter = acclaimed_knight_piety_from_battle
						}
					}
				}
				scope:recipient = {
					add_piety = minor_piety_gain
				}
			}
			if = {
				limit = {
					scope:recipient = {
						any_active_accolade = {
							has_accolade_parameter = acclaimed_knight_piety_from_battle_high
						}
					}
				}
				scope:recipient = {
					add_piety = medium_piety_gain
				}
			}
		}
	}
	
	auto_accept = yes
}

POD_become_enchanted_interaction = {
	category = interaction_category_fae
	common_interaction = yes
	icon = artisan_inspiration
	desc = POD_become_enchanted_interaction_desc
	
	target_type = artifact
	target_filter = actor_artifacts

	can_be_picked_artifact = {
		scope:target = {
			has_variable = enchantment
		}
	}

	is_shown = {
		scope:actor = scope:recipient
		scope:actor = {
			any_character_artifact = { has_variable = enchantment }
			POD_can_become_glamour_enchanted = yes
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = { POD_is_not_weaver_splat_trigger = yes }
		POD_self_targeted_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:recipient = { POD_become_enchanted_effect = yes }
		scope:target = {
			if = {
				limit = { rarity = illustrious }
				add_durability = {
					value = artifact_max_durability
					multiply = -0.25
				}
			}
			else_if = {
				limit = { rarity = famed }
				add_durability = {
					value = artifact_max_durability
					multiply = -0.50
				}
			}
			else_if = {
				limit = { rarity = masterwork }
				add_durability = {
					value = artifact_max_durability
					multiply = -0.75
				}
			}
			else = {
				add_durability = {
					value = artifact_max_durability
					multiply = -0.95
				}
			}
		}
	}
	
	auto_accept = yes
}

POD_ask_enchantment_interaction = {

	common_interaction = yes
	icon = artisan_inspiration
	desc = POD_ask_enchantment_interaction_desc
	
	cost = { prestige = minor_prestige_value }
	cooldown_against_recipient = { years = 1 }

	is_shown = {
		scope:actor = {
			POD_can_become_glamour_enchanted = yes
		}
		scope:recipient = {
			OR = {
				has_trait = fae
				has_trait = changeling
			}
		}
		POD_interaction_is_shown = yes
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
			POD_is_not_weaver_splat_trigger = yes
			NOT = { is_at_war_with = scope:recipient }
			custom_tooltip = {
				text = "He_must_be_my_friend"
				scope:recipient = {
					OR = {
						has_relation_lover = scope:actor
						has_relation_friend = scope:actor
						has_relation_best_friend = scope:actor
						has_relation_soulmate = scope:actor
						scope:actor = {
							AND = {
								has_strong_hook = scope:recipient
								has_usable_hook = scope:recipient
							}
						}
					}
				}
			}
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}
	
	auto_accept = yes

	on_accept = {
		scope:actor = { POD_become_enchanted_effect = yes }
		scope:recipient = {
			if = {
				limit = {
					can_add_hook = {
						target = scope:actor
						type = favor_hook
					}
				}
				add_hook = {
					type = favor_hook
					target = scope:actor
				}
			}
		}
	}
