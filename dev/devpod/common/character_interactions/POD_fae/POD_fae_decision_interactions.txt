	target_filter = actor_artifacts

		scope:target = {
			has_variable = enchantment
		}
	}

	is_shown = {
		scope:actor = scope:recipient
		scope:actor = {
			any_character_artifact = { has_variable = enchantment }
			POD_can_become_glamour_enchanted = yes
		}
	}

		scope:actor = { POD_is_not_weaver_splat_trigger = yes }

	}

	on_accept = {
		scope:recipient = { POD_become_enchanted_effect = yes }
		scope:target = {
			if = {
				limit = { rarity = illustrious }
				add_durability = {
					value = artifact_max_durability
					multiply = -0.25
				}
			}
			else_if = {
				limit = { rarity = famed }
				add_durability = {
					value = artifact_max_durability
					multiply = -0.50
				}
			}
			else_if = {
				limit = { rarity = masterwork }
				add_durability = {
					value = artifact_max_durability
					multiply = -0.75
				}
			}
			else = {
				add_durability = {
					value = artifact_max_durability
					multiply = -0.95
				}
			}
		}
	}
	
	auto_accept = yes
}
