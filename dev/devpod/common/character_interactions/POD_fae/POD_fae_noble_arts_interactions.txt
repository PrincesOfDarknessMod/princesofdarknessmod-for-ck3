POD_chronos_artifact_interaction = {
	category = interaction_category_fae
	icon = POD_chronos_interaction
	desc = POD_chronos_interaction_desc

	cooldown = { months = realm_time_monthly_cooldown }

	use_diplomatic_range = no

	target_type = artifact
	target_filter = recipient_artifacts

	is_shown = {
		scope:actor = { has_trait = chronos }
		POD_hostile_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			has_perk = noble_arts_set_in_stone_perk
			POD_has_enough_glamour_for_artifact_cantrip_interaction = { AMOUNT = 4 }
		}
		scope:recipient = {
			POD_not_has_cantrip_defenses_trigger = yes
			custom_description = {
				text = "demand_artifact_interaction_valid_tt_1"
				has_any_artifact = yes
			}
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:actor = {
			POD_artifact_cantrip_interaction_cost = { COUNT = 4 }
			POD_bunk_effect = {
				SKILL = intrigue
				DIFFICULTY = cantrip_difficulty_standard
				SUCCESS = "
					scope:target = {
						custom_tooltip = {
							text = POD_chronos_artifact_interaction_effect
							set_should_decay = no
						}
						add_artifact_modifier = cantrip_set_in_stone_modifier
					}
				"
				FAILURE = ""
			}
		}
	}

	auto_accept = yes

	ai_will_do = { base = 0 }
}

POD_chronos_interaction = {
	category = interaction_category_fae
	icon = POD_chronos_interaction
	desc = POD_chronos_interaction_desc

	cooldown = { months = realm_time_monthly_cooldown }

	use_diplomatic_range = no

	is_shown = {
		scope:actor = { has_trait = chronos }
		POD_hostile_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = { has_perk = noble_arts_deja_vu_perk }
		scope:recipient = { POD_not_has_cantrip_defenses_trigger = yes }
		POD_interaction_is_valid_showing_failures_only = yes
	}

	send_option = {
		is_valid = {
			scope:actor = {
				has_perk = noble_arts_deja_vu_perk
				POD_has_enough_glamour_for_character_cantrip_interaction = { AMOUNT = 3 }
			}
		}
		flag = deja_vu
		localization = noble_arts_deja_vu_perk_name
	}

	send_option = {
		is_valid = {
			scope:actor = {
				has_perk = noble_arts_time_dilatation_perk
				POD_has_enough_glamour_for_character_cantrip_interaction = { AMOUNT = 5 }
			}
			scope:recipient = {
				is_immortal = no
			}
		}
		flag = time_dilatation
		localization = noble_arts_time_dilatation_perk_name
	}

	on_accept = {
		scope:actor = {
			save_scope_value_as = {
				name = recipient_age_calc
				value = {
					value = scope:recipient.age
					add = 25
				}
			}
			switch = {
				trigger = yes
				scope:deja_vu = {
					POD_character_cantrip_interaction_cost = { COUNT = 3 }
					POD_bunk_effect = {
						SKILL = intrigue
						DIFFICULTY = cantrip_difficulty_standard
						SUCCESS = "
							scope:recipient = { add_fae_character_modifier = { MODIFIER = cantrip_deja_vu_modifier } }
						"
						FAILURE = ""
					}
				}
				scope:time_dilatation = {
					POD_character_cantrip_interaction_cost = { COUNT = 5 }
					POD_bunk_effect = {
						SKILL = intrigue
						DIFFICULTY = cantrip_difficulty_hard
						SUCCESS = "
							scope:recipient = { 
								set_age = scope:recipient_age_calc
							}
						"
						FAILURE = ""
					}
				}
			}
		}
	}

	auto_accept = yes

	ai_will_do = { base = 0 }
}
