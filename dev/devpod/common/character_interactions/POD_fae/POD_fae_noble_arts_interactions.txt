		scope:actor = { has_trait = contract }
		scope:actor != scope:recipient
		POD_hostile_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = { has_perk = noble_arts_done_deal_perk }
		scope:recipient = {
			POD_not_has_cantrip_defenses_trigger = yes
			trigger_if = {
				limit = {
					scope:actor = { has_perk = noble_arts_casual_contract_perk }
					opinion = {
						value < 50
						target = scope:actor
					}
				}
				custom_description = {
					text = same_location
					scope:actor = { exists = location }
					scope:recipient = { exists = location }
					scope:actor = { location = scope:recipient.location }
				}
			}
			trigger_else = {
				opinion = {
					value >= 50
					target = scope:actor
				}
			}
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	populate_recipient_list = {
		scope:actor = {
			add_to_list = characters
			if = {
				limit = { scope:recipient = { is_landed = yes } }
				every_neighboring_top_liege_realm_owner = {
					limit = { NOT = { this = scope:recipient } }
					add_to_list = characters
				}
			}
			every_relation = {
				type = rival
				limit = { NOT = { this = scope:recipient } }
				add_to_list = characters
			}
			every_relation = {
				type = nemesis
				limit = { NOT = { this = scope:recipient } }
				add_to_list = characters
			}
		}
		scope:recipient = {
			if = {
				limit = { is_landed = yes }
				every_neighboring_top_liege_realm_owner = {
					limit = { NOT = { this = scope:actor } }
					add_to_list = characters
				}
			}
		}
	}

	send_option = {
		is_shown = {
			scope:recipient = { is_landed = yes }
		}
		is_valid = {
			scope:actor = {
				has_perk = noble_arts_done_deal_perk
				POD_has_enough_glamour_for_character_cantrip_interaction = { AMOUNT = 3 }
			}

			scope:recipient = { is_landed = no }

				POD_has_enough_glamour_for_character_cantrip_interaction = { AMOUNT = 2 }

				has_character_modifier = cantrip_aphrodisia_modifier
			}
		}
		is_valid = {
			scope:actor = {
				has_perk = noble_arts_aphrodisia_perk

						}

								scope:actor = { add_courtier = prev }

					}
