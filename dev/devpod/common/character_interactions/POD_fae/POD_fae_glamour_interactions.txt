POD_ravaging_interaction = {
	category = interaction_category_fae
	common_interaction = yes
	interface_priority = 120
	icon = POD_fae_interaction
	desc = POD_ravaging_interaction_desc
	
	is_shown = {
		scope:actor != scope:recipient
		scope:actor = { POD_can_use_epiphanies = yes }
		POD_hostile_mental_interaction_is_shown = yes
	}

	is_valid = {
		scope:recipient = {
			is_imprisoned_by = scope:actor
			stress_level < 3
			trigger_if = {
				limit = { has_ep1_dlc_trigger = yes }
				custom_description = {
					text = POD_has_inspiration
					object = this
					exists = inspiration
				}
			}
			trigger_else = {
				NOT = { has_character_modifier = intrigue_broken_modifier }
			}
		}
	}


	is_valid_showing_failures_only = {
		scope:actor = {
			POD_is_not_full_on_energy_including_vessels = yes
		}
		scope:recipient = {
			POD_not_has_cantrip_defenses_trigger = yes
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		random_list = {
			10 = {
				scope:actor = { POD_glamour_income_effect = { COUNT = 5 } }
				scope:recipient = {
					add_stress = massive_stress_impact_gain
					add_piety = massive_piety_loss
				}
			}
			20 = {
				scope:actor = { POD_glamour_income_effect = { COUNT = 3 } }
				scope:recipient = {
					add_stress = major_stress_impact_gain
					add_piety = major_piety_loss
				}
			}
			30 = {
				scope:actor = { POD_glamour_income_effect = { COUNT = 2 } }
				scope:recipient = {
					add_stress = medium_stress_impact_gain
					add_piety = medium_piety_loss
				}
			}
			40 = {
				scope:actor = { POD_glamour_income_effect = { COUNT = 1 } }
				scope:recipient = {
					add_stress = minor_stress_impact_gain
					add_piety = minor_piety_loss
				}
			}
		}
		scope:actor = {
			if = {
				limit = { POD_splat_has_access_to_fae_arts = yes }
				add_common_arts_lifestyle_xp = 50
			}
			if = {
				limit = { POD_splat_has_access_to_kith_arts = yes }
				add_kith_arts_lifestyle_xp = 25
			}
			if = {
				limit = { POD_splat_has_access_to_slivers = yes }
				add_slivers_lifestyle_xp = 25
			}
			if = {
				limit = { POD_splat_has_access_to_wu_tan = yes }
				add_wu_tan_lifestyle_xp = 25
			}
		}
		scope:recipient = {
			if = {
				limit = { exists = inspiration }
				destroy_inspiration = inspiration
			}
			else = {
				if = {
					limit = { has_character_modifier = inspiration_modifier }
					remove_character_modifier = inspiration_modifier
				}
				else = {
					add_character_modifier = {
						modifier = intrigue_broken_modifier
						years = 15
					}
				}
			}
			if = {
				limit = { has_trait = mortal }
				add_trait = POD_knows_too_much
			}
			add_opinion = {
				target = scope:actor
				modifier = hurt_opinion
				opinion = -25
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_rhapsody_interaction = {
	category = interaction_category_fae
	common_interaction = yes
	interface_priority = 120
	icon = POD_fae_interaction
	desc = POD_rhapsody_interaction_desc
	
	cost = { piety = 250 }
	
	is_shown = {
		scope:actor != scope:recipient
		scope:actor = { POD_can_use_epiphanies = yes }
		POD_hostile_mental_interaction_is_shown = yes
	}

	is_valid = {
		trigger_if = {
			limit = { scope:actor = { NOT = { any_character_artifact = { has_variable = crown_of_dreams } } } }
			scope:recipient = { stress_level >= 3 }
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			POD_is_not_full_on_energy_including_vessels = yes
		}
		scope:recipient = {
			POD_not_has_cantrip_defenses_trigger = yes
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:actor = { POD_glamour_income_effect = { COUNT = 3 } }
		scope:recipient = {
			if = {
				limit = { exists = inspiration }
				destroy_inspiration = inspiration
				scope:actor = {
					if = {
						limit = { POD_splat_has_access_to_fae_arts = yes }
						add_common_arts_lifestyle_xp = 100
					}
					if = {
						limit = { POD_splat_has_access_to_kith_arts = yes }
						add_kith_arts_lifestyle_xp = 50
					}
					if = {
						limit = { POD_splat_has_access_to_slivers = yes }
						add_slivers_lifestyle_xp = 50
					}
					if = {
						limit = { POD_splat_has_access_to_wu_tan = yes }
						add_wu_tan_lifestyle_xp = 50
					}
				}
			}
			if = {
				limit = { has_trait = mortal }
				add_trait = POD_knows_too_much
			}
			add_character_modifier = {
				modifier = intrigue_broken_modifier
				years = 5
			}
			add_opinion = {
				target = scope:actor
				modifier = hurt_opinion
				opinion = -45
			}
			increase_wounds_no_death_effect = { REASON = treatment }
			custom_tooltip = stress_threshold_level_3_event_tooltip
			if = {
				limit = { is_imprisoned = yes }
				trigger_event = { on_action = stress_threshold_level_3_event_prison }
			}
			else = {
				trigger_event = { on_action = stress_threshold_level_3_event }
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

	common_interaction = yes
	interface_priority = 120
	icon = POD_fae_interaction

	cost = { piety = 300 }

		scope:target = {
			tier = tier_county
			NOT = { has_county_modifier = glamour_reaved_modifier }
		}
	}
	
	is_shown = {
		scope:actor = { POD_can_use_epiphanies = yes }
		POD_hostile_mental_interaction_is_shown = yes

	is_valid_showing_failures_only = {

			POD_is_not_full_on_energy_including_vessels = yes
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:actor = {
			POD_glamour_income_effect = { COUNT = 5 }
			add_character_modifier = {
				modifier = glamour_reaver_modifier
				years = 5
			}
		}
		scope:target = {
			add_county_modifier = {
				modifier = glamour_reaved_modifier
				years = 5
			}
		}
		scope:recipient = {
			if = {
				limit = { NOT = { this = scope:actor } }
				add_opinion = {
					target = scope:actor
					modifier = POD_casted_harmful_weather_magic
				}
				progress_towards_rival_effect = {
					CHARACTER = scope:actor
					REASON = rival_cast_spell
					OPINION = 0
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
