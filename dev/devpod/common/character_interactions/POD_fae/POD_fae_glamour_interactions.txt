POD_ravaging_interaction = {
	category = interaction_category_fae
	common_interaction = yes
	interface_priority = 120
	icon = POD_fae_interaction
	desc = POD_ravaging_interaction_desc
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { POD_can_use_ravaging = yes }
		scope:recipient = { 
			has_trait = mortal
			is_in_the_same_court_as_or_guest = scope:actor
		}
		POD_hostile_mental_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			POD_is_not_full_on_energy_including_vessels = yes
		}
		scope:recipient = {
			POD_not_has_cantrip_defenses_trigger = yes
			stress_level < 3
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	cooldown_against_recipient = { years = 2 }

	on_accept = {
		scope:actor = {
			custom_tooltip = POD_ravaging_interaction_tt
			if = {
				limit = { scope:recipient.epiphanies_quality >= 1 } 
				POD_glamour_income_effect = { COUNT = scope:recipient.epiphanies_quality }
				save_scope_value_as = {
					name = realm_xp
					value = {
						value = scope:recipient.epiphanies_quality
						multiply = 40
					}
				}
				save_scope_value_as = {
					name = art_xp
					value = {
						value = scope:recipient.epiphanies_quality
						multiply = 20
					}
				}
				add_realm_lifestyle_xp = scope:realm_xp
				add_common_arts_lifestyle_xp = scope:art_xp
			}
			if = {
				limit = { faith = { has_doctrine = doctrine_fae_morality_seelie } }
				custom_tooltip = POD_ravaging_interaction_seelie_tt
				add_piety = -250
			}
		}
		scope:recipient = {
			add_stress = 100
			POD_add_modifier_and_track_stacks = {
				TYPE = character
				MODIFIER = POD_drained_creativity
			}
		}
	}

	auto_accept = yes
}
