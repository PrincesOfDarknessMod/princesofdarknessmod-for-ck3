			scope:recipient = { religion = religion:spring_court_religion }

			limit = { faith:solstice_court = { has_doctrine = doctrine_summer_time } }
			scope:recipient = { religion = religion:summer_court_religion }
		}
		trigger_if = { 
			limit = { faith:solstice_court = { has_doctrine = doctrine_autumn_time } }
			scope:recipient = { religion = religion:autumn_court_religion }
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }

	}

	on_accept = {
		scope:actor = {

				type = favor_hook

			}
		}
	}

		}
	}

	populate_recipient_list = {
		scope:recipient = {
			every_child = {

						CHILD = scope:secondary_recipient

					}

						CHILD = scope:secondary_actor

POD_call_patron_interaction = {
	category = interaction_category_fera
	common_interaction = yes
	desc = POD_call_patron_interaction_desc
	interface_priority = 60
	interface = call_ally
	special_interaction = call_ally_interaction
	popup_on_receive = yes
	pause_on_receive = yes
	icon = alliance

	greeting = positive
	notification_text = CALL_ALLY_NOTIFICATION

	is_shown = {
		scope:actor = { is_at_war = yes }
		scope:actor = { has_trait = patron_council }
	}
	
	has_valid_target = {
		exists = scope:target
	}	

	has_valid_target_showing_failures_only = {
		scope:target = {
			is_war_leader = scope:actor
		}

		joiner_not_already_in_another_war_with_any_target_war_participants_trigger = {
			WARRIOR = scope:actor
			JOINER = scope:recipient
		}
	}
	
	is_valid = {
		trigger_if = {
			limit = { 
				scope:actor = {
					OR = {
						has_character_modifier = philospher_queen_modifier
						has_character_modifier = philospher_king_modifier
					}
				}
			}
			scope:recipient = {
				has_trait = patron_council
			}
		}
		trigger_else_if = {
			limit = { scope:actor = { has_trait = patron_council_austere_conclave } }
			scope:recipient = { has_trait = patron_council_austere_conclave }
		}
		trigger_else_if = {
			limit = { scope:actor = { has_trait = patron_council_seeker } }
			scope:recipient = { has_trait = patron_council_seeker }
		}
		trigger_else = {
			scope:recipient = { has_trait = patron_council_vanguard }
		}
		scope:target = {
			trigger_if = {
				limit = { casus_belli.primary_attacker = scope:actor }
				casus_belli.primary_defender = { 
					NOT = {
						has_trait = patron_council
					} 
				}
			}
			trigger_if = {
				limit = { casus_belli.primary_defender = scope:actor }
				casus_belli.primary_attacker = { 
					NOT = {
						has_trait = patron_council
					} 
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			is_ruler = yes
		}

		# Only War Leaders can call allies
		scope:actor = {
			custom_description = {
				text = must_be_war_leader
				any_character_war = {
					is_war_leader = scope:actor
				}
			}
		}

		# Cannot call the same character multiple times
		scope:actor = {
			custom_description = {
				text = cannot_call_same_character_multiple_times
				subject = scope:recipient
				any_character_war = {
					ALL_FALSE = {
						was_called = scope:recipient
						is_participant = scope:recipient
					}
				}
			}
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}

	can_be_picked = {
		can_join_war_liege_vassal_check_trigger = { WARRIOR = scope:actor JOINER = scope:recipient }

		scope:target = {
			is_war_leader = scope:actor

			using_cb = POD_blood_hunt_war
			
			# not already in target war
			custom_description = {
				text = join_war_interaction_already_in_target_war
				NOR = {
					any_war_attacker = { this = scope:recipient }
					any_war_defender = { this = scope:recipient }
				}
			}
		}
	}

	on_auto_accept = {
		scope:target = {
			if = {
				limit = {
					scope:recipient = {
						any_active_accolade = {
							has_accolade_parameter = acclaimed_knight_piety_from_battle
						}
					}
				}
				scope:recipient = {
					add_piety = miniscule_piety_gain
				}
			}
			if = {
				limit = {
					scope:recipient = {
						any_active_accolade = {
							has_accolade_parameter = acclaimed_knight_piety_from_battle_high
						}
					}
				}
				scope:recipient = {
					add_piety = minor_piety_gain
				}
			}
		}
	}
		
	on_accept = {
		blood_hunt_call_interaction_effect = yes
		scope:target = {
			if = {
				limit = {
					scope:recipient = {
						any_active_accolade = {
							has_accolade_parameter = acclaimed_knight_piety_from_battle
						}
					}
				}
				scope:recipient = {
					add_piety = minor_piety_gain
				}
			}
			if = {
				limit = {
					scope:recipient = {
						any_active_accolade = {
							has_accolade_parameter = acclaimed_knight_piety_from_battle_high
						}
					}
				}
				scope:recipient = {
					add_piety = medium_piety_gain
				}
			}
		}
	}
	
	auto_accept = yes
}

POD_become_enchanted_interaction = {
	category = interaction_category_fae
	common_interaction = yes
	icon = artisan_inspiration
	desc = POD_become_enchanted_interaction_desc
	
	target_type = artifact
	target_filter = actor_artifacts

	can_be_picked_artifact = {
		scope:target = {
			has_variable = enchantment
		}
	}

	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { any_character_artifact = { has_variable = enchantment } }
		scope:actor = { 
			 NOT = { has_trait = fae }
			 NOT = { has_trait = changeling }
			 has_trait = kinain
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = { NOT = { has_character_modifier = glamour5_modifier } }
		scope:actor = { POD_is_not_weaver_splat_trigger = yes }
	}

	on_accept = {
		if = {
			limit = {
				scope:actor = { is_malkavian_trigger = no } #Malkavians don't feel the glamour injection
			}
			scope:actor = { add_stress = massive_stress_gain }
		}
		scope:recipient = { POD_become_enchanted_effect = yes }
		scope:target = {
			if = {
				limit = { rarity = illustrious }
				add_durability = {
					value = artifact_max_durability
					multiply = -0.25
				}
			}
			else_if = {
				limit = { rarity = famed }
				add_durability = {
					value = artifact_max_durability
					multiply = -0.50
				}
			}
			else_if = {
				limit = { rarity = masterwork }
				add_durability = {
					value = artifact_max_durability
					multiply = -0.75
				}
			}
			else = {
				add_durability = {
					value = artifact_max_durability
					multiply = -0.95
				}
			}
		}
	}
	
	auto_accept = yes
}
