			save_scope_value_as = {
				name = cantrip_difficulty_petros_reduction
				value = {
					value = 0
					if = { limit = { has_perk = slivers_petros_uncontrolled_perk } add = 1 }
					if = { limit = { has_perk = slivers_petros_unstable_perk } add = 1 }

					if = { limit = { has_perk = slivers_petros_stable_perk } add = 1 }

				trigger = yes
