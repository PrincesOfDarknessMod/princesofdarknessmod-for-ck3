POD_shih_training_interaction = {
	category = interaction_category_friendly
	icon = POD_art_interaction
	desc = POD_shih_training_interaction_desc

	cooldown = { years = 3 }

        common_interaction = yes

	is_shown = {
		scope:actor = { has_trait = shih }
		scope:recipient = { has_trait = shih }
	}
	is_valid_showing_failures_only = {
		scope:actor = {
			is_imprisoned = no
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			NOT = { has_trait = automaton }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
			POD_is_not_clone_trigger = yes
			trigger_if = { limit = { POD_can_train_any_shih_trigger = no } is_child_of = scope:actor  }
			is_imprisoned = no
		}
		POD_cross_umbra_interaction_valid_trigger = yes
		scope:actor = {
			POD_can_start_planning_phase_trigger = yes
			OR = {
				POD_not_already_on_journey_trigger = { TYPE = shih_training_knowledge TARGET = scope:recipient }
				POD_not_already_on_journey_trigger = { TYPE = shih_training_technique TARGET = scope:recipient }
				POD_not_already_on_journey_trigger = { TYPE = shih_training_body TARGET = scope:recipient }
				POD_not_already_on_journey_trigger = { TYPE = shih_training_meditation TARGET = scope:recipient }
				POD_not_already_on_journey_trigger = { TYPE = shih_training_hunt TARGET = scope:recipient }
			}
		}
	}

	send_option = {
		flag = shih_training_knowledge
		localization = POD_shih_training_interaction_shih_training_knowledge
	}
	send_option = {
		flag = shih_training_technique
		localization = POD_shih_training_interaction_shih_training_technique
	}
	send_option = {
		flag = shih_training_body
		localization = POD_shih_training_interaction_shih_training_body
	}
	send_option = {
		flag = shih_training_meditation
		localization = POD_shih_training_interaction_shih_training_meditation
	}
	send_option = {
		flag = shih_training_hunt
		localization = POD_shih_training_interaction_shih_training_hunt
	}

	on_accept = {
		POD_journey_interaction_pause_hint = yes
		scope:actor = {
			custom_tooltip = POD_shih_training_interaction_tt
			switch = {
				trigger = yes
				scope:shih_training_knowledge = {  
					POD_start_targeted_journey_effect = {
						TYPE    = shih_training_knowledge
						MONTHLY = shih_training
						TARGET  = scope:recipient
					}
				}
				scope:shih_training_technique = {  
					POD_start_targeted_journey_effect = {
						TYPE    = shih_training_technique
						MONTHLY = two_years
						TARGET  = scope:recipient
					}
				}
				scope:shih_training_body = {  
					POD_start_targeted_journey_effect = {
						TYPE    = shih_training_body
						MONTHLY = two_years
						TARGET  = scope:recipient
					}
				}
				scope:shih_training_meditation = {  
					POD_start_targeted_journey_effect = {
						TYPE    = shih_training_meditation
						MONTHLY = two_years
						TARGET  = scope:recipient
					}
				}
				scope:shih_training_hunt = {  
					POD_start_targeted_journey_effect = {
						TYPE    = shih_training_hunt
						MONTHLY = two_years
						TARGET  = scope:recipient
					}
				}
			}

	}
