POD_create_coterie_interaction = {
	category = interaction_category_coterie
	common_interaction = yes
	desc = POD_create_coterie_interaction_desc
	icon = council
	
	is_shown = {
		scope:actor = {
			POD_has_coterie_trigger = no
			NOT = { has_character_flag = coterie_creation_in_progress }
		}
		scope:actor = scope:recipient
	}
	
	on_accept = {
		custom_tooltip = {
			text = POD_create_coterie_interaction_effect
			scope:actor = {
				POD_create_coterie_effect = yes
				trigger_event = POD_coterie_maintenance.2
			}
		}
		custom_tooltip = POD_create_coterie_interaction_effect2
	}

	auto_accept = yes
	
	ai_frequency = 0
	ai_will_do = {
		base = 0
	}
}

POD_invite_to_coterie_interaction = {
	category = interaction_category_coterie
	desc = POD_invite_to_coterie_interaction_desc
	icon = council

	is_shown = {
		scope:actor = {
			POD_has_coterie_trigger = yes
			NOT = { has_relation_coterie_member = scope:recipient }
			is_adult = yes
		}
		
		scope:actor != scope:recipient
		
		POD_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		POD_coterie_interaction_requirements_trigger = yes

		trigger_if = {
			limit = { scope:recipient = { POD_has_coterie_trigger = no } }
			scope:recipient = {
				opinion = {
					target = scope:actor
					value >= 50
				}
			}
		}
		trigger_else = {
			scope:recipient = {
				opinion = {
					target = scope:actor
					value >= 100
				}
				custom_tooltip = {
					text = "He_must_be_my_friend"
					scope:recipient = {
						OR = {
							has_relation_lover = scope:actor
							has_relation_friend = scope:actor
							has_relation_best_friend = scope:actor
							has_relation_soulmate = scope:actor
							scope:actor = {
								AND = {
									has_strong_hook = scope:recipient
									has_usable_hook = scope:recipient
								}
							}
						}
					}
				}
			}
		}

		custom_tooltip = {
			text = POD_at_coterie_size_cap
			scope:actor = {
				POD_coterie_at_max_size_trigger = no
			}
		}
		custom_tooltip = {
			text = POD_coterie_requirement_same_realm
			OR = {
				scope:actor.top_liege = scope:recipient.top_liege
				scope:recipient = { is_spouse_of = scope:actor }
			}
		}
		
		scope:actor = {
			custom_tooltip = {
				text = POD_invite_to_coterie_interaction_mortal
				trigger_if = {
					limit = {
						NOR = {
							has_main_supernatural_race_trigger = no
							POD_is_demon_trigger = yes
							POD_is_mummy_trigger = yes
						}
					}
					scope:recipient = {
						is_mortal_trigger = no
					}
				}
			}
			custom_tooltip = {
				text = POD_invite_to_coterie_interaction_mortal_cultist
				trigger_if = {
					limit = {
						POD_is_demon_trigger = yes
						NOT = { has_relation_worshipping_cultist = scope:recipient }
					}
					scope:recipient = {
						is_mortal_trigger = no
					}
				}
			}
			custom_tooltip = {
				text = POD_invite_to_coterie_interaction_mortal_neteru
				trigger_if = {
					limit = {
						POD_is_mummy_trigger = yes
						NOT = { has_relation_worshipping_neteru = scope:recipient }
					}
					scope:recipient = {
						is_mortal_trigger = no
					}
				}
			}
		}
		
		scope:recipient = {
			POD_is_valid_coterie_member_trigger = yes
		}
		
		POD_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:recipient = {
			if = {
				limit = { POD_has_coterie_trigger = yes }
				custom_tooltip = {
					text = POD_leave_coterie_recipient_tt
					POD_leave_coterie_effect = yes
				}
			}
		}
		custom_tooltip = {
			text = POD_invite_to_coterie_interaction_effect
			scope:actor.var:coterie = {
				save_scope_as = coterie
			}
			scope:recipient = {
				POD_join_coterie_effect = yes
			}
		}
		if = {
			limit = {
				scope:actor = {
					has_trait = fera
					POD_is_wyrm_servant_trigger = no
				}
				scope:recipient = {
					has_trait = fera
				}
			}
			custom_tooltip = POD_invite_to_coterie_interaction_packbond
		}
		else = {
			custom_tooltip = POD_invite_to_coterie_interaction_rivalry
		}
	}
	
	auto_accept = yes
	
	# ai_accept = {
		# base = -25
		
		# opinion_modifier = { # Opinion Factor
			# who = scope:recipient
			# opinion_target = scope:actor
			# desc = AI_SIMPLE_OPINION_REASON
		# }
	# }
	
	ai_frequency = 0
	ai_will_do = {
		base = 0
	}
}

			}

		}
		else = {

						has_trait = fera

			text = POD_kick_from_coterie_interaction_packbond
		}
	}
	
	on_accept = {
		scope:recipient = {
			POD_leave_coterie_with_rivalry_effect = { RIVAL = scope:actor }
		}
	}

	auto_accept = yes
	
	ai_frequency = 0
	ai_will_do = {
		base = 0
	}
}

POD_disband_coterie_interaction = {
	category = interaction_category_coterie
	common_interaction = yes
	desc = POD_disband_coterie_interaction_desc
	icon = council
	
	is_shown = {
		scope:actor = {
			OR = {
				POD_has_coterie_trigger = yes
				# safeguard so the player can clean out the variable if the story_cycle was deleted
				AND = {
					has_variable = coterie
					NOT = { exists = var:coterie }
				}
			}
			NOT = { has_character_flag = coterie_creation_in_progress }
		}
		scope:actor = scope:recipient
	}

	is_valid_showing_failures_only = {
		POD_coterie_interaction_requirements_trigger = yes
		custom_tooltip = {
			scope:actor = {
				trigger_if = {
					limit = {
						has_trait = fera
					}
					OR = {
						POD_is_wyrm_servant_trigger = yes
						POD_has_coterie_members_trigger = no
					}
				}
			}
			text = POD_disband_coterie_interaction_packbond
		}
	}
	
	on_accept = {
		scope:actor = {
			var:coterie ?= {
				every_in_list = {
					variable = members
					limit = {
						NOT = { this = scope:actor }
					}
					POD_leave_coterie_with_rivalry_effect = { RIVAL = scope:actor }
				}
			}
			custom_tooltip = {
				text = POD_disband_coterie_interaction_effect
				POD_disband_coterie_effect = yes
			}
		}
	}

	auto_accept = yes
	
	ai_frequency = 0
	ai_will_do = {
		base = 0
	}
}
