
POD_theurgy_heal_interaction = {
	category = interaction_category_holyarts
	icon = POD_theurgy_interaction
	desc = POD_theurgy_heal_interaction_desc
	options_heading = POD_theurgy_heal_interaction_options_heading
	
	cooldown = { months = 12 }
	cost = { piety = medium_piety_value }
	
	is_shown = {
		scope:actor = { has_trait = theurgist }
		scope:actor = { has_perk = via_medicamenti_perk }
	}
	
	is_valid = {
		scope:recipient = {
			OR = {
				has_trait = scarred
				has_trait = disfigured
				has_trait = wounded
				has_trait = ill
				has_trait = cancer
				has_trait = consumption
				has_trait = early_great_pox
				has_trait = great_pox
				has_trait = gout_ridden
				has_trait = leper
				has_trait = lovers_pox
				has_trait = pneumonic
				has_trait = smallpox
				has_trait = typhus
				has_trait = bubonic_plague
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			NOT = { has_trait_with_flag = can_not_cure }
			NOT = { has_character_modifier = eye_of_sakhmet_modifier }
			NOT = { has_character_modifier = thin_blood_perk_modifier }
		}
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = scarred
			}
		}	
		is_valid = {
			custom_description = {
				text = POD_permanent_scarred
				object = scope:recipient
				scope:recipient = { NOT = { has_character_flag = POD_permanent_scarring_flag } }
			}
		}
		flag = scarred
		localization = POD_heal_scarred
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = disfigured
			}
		}	
		is_valid = {
			custom_description = {
				text = POD_permanent_disfigured
				object = scope:recipient
				scope:recipient = { NOT = { has_character_flag = POD_permanent_disfigured_flag } }
			}
		}
		flag = disfigured
		localization = POD_heal_disfigured
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = wounded
			}
		}
		flag = wounded
		localization = POD_heal_wounded
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = ill
			}
		}
		flag = ill
		localization = POD_heal_ill
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				OR = {
					has_trait = cancer
					has_trait = consumption
					has_trait = early_great_pox
					has_trait = great_pox
					has_trait = gout_ridden
					has_trait = leper
					has_trait = lovers_pox
					has_trait = pneumonic
					has_trait = smallpox
					has_trait = typhus
					has_trait = bubonic_plague
				}
			}
		}
		is_valid = {
			trigger_if = {
				limit = { scope:actor = scope:recipient }
				scope:actor = { has_perk = biocontrol_perk }
			}
			trigger_else = {
				scope:actor = { has_perk = psychic_healing_perk }
			}
		}
		flag = disease
		localization = POD_heal_disease
	}
	
	on_accept = {
		scope:actor = { 
			POD_willpower_rouse_check_effect = yes
		}
		scope:recipient = {
			play_music_cue = "mx_cue_peace_ensues"

			if = {
				limit = {
					scope:scarred = yes
				}
				scope:recipient = {
					remove_trait = scarred
				}
			}	

			if = {
				limit = {
					scope:disfigured = yes
				}
				scope:recipient = {
					remove_trait = disfigured
				}
			}
			
			if = {
				limit = {
					scope:wounded = yes
				}
				scope:recipient = {
					POD_remove_wound_trait_effect = yes
				}
			}
			if = {
				limit = {
					scope:ill = yes
				}
				scope:recipient = {		
					recover_from_disease_effect = { DISEASE = ill }
					remove_disease_treatment_effect = yes
				}
			}	

			if = {
				limit = {
					scope:disease = yes
				}
				scope:recipient = {		
					remove_disease_trait_effect = yes
				}
			}
		}
		if = {
			limit = { NOT = { scope:actor = scope:recipient } }
			scope:recipient = { 
				add_opinion = {
					target = scope:actor
					modifier = pleased_opinion
					opinion = 15
				}
			}
		}
	}
	
	auto_accept = yes

	ai_potential = {
		# The basic requirements
		is_imprisoned = no
		is_landed = yes
		
		piety > major_piety_value
	}
	ai_targets = {
		ai_recipients = self
	}
	ai_targets = {
		ai_recipients = courtiers
		ai_recipients = peer_vassals
		max = 10
	}
	ai_frequency = 12
	ai_will_do = {
		base = 0
		modifier = {
			add = 100
			scope:recipient = {
				OR = {
					this = scope:actor
					is_knight_of = scope:actor
				}
			}
		}
	}
}


POD_train_theurgy_interaction = {
	category = interaction_category_holyarts
	icon = POD_theurgy_interaction
    desc = POD_train_theurgy_interaction_desc
	common_interaction = yes
	
	cooldown = { months = 12 }
	cost = { piety = medium_piety_value }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_trait = theurgist }
		scope:recipient = { has_trait = theurgist }
	}

	is_valid_showing_failures_only = {
	    scope:actor = {
			sorcery_lifestyle_perk_points >= 1
			NOT = { faith = { has_doctrine = doctrine_supehunting_crime } }
		}
		scope:recipient = { is_imprisoned = no }
	}
	
	on_accept = {
	    scope:actor = { 
			add_sorcery_lifestyle_perk_points = -1
			stress_impact = {
				base = medium_stress_gain
			}
		}			
	    scope:recipient = { 
			add_sorcery_lifestyle_perk_points = 1 
			add_opinion = {
				target = scope:actor
				modifier = mentored_me_opinion
			}
		}	
	}

	ai_frequency = 0
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_initiate_theurgy_interaction = {
	category = interaction_category_holyarts
	icon = POD_theurgy_interaction
    desc = POD_initiate_theurgy_interaction_desc
	
	cooldown = { months = 12 }
	cost = { piety = major_piety_value }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_trait = theurgist }
		scope:recipient = { NOT = { has_trait = theurgist } }
	}

	is_valid_showing_failures_only = {
	    scope:actor = {
			sorcery_lifestyle_perk_points >= 1
		}
		scope:recipient = { 
			is_imprisoned = no 
			NOT = { faith = { has_doctrine = doctrine_supehunting_crime } }
		}
	}
	
	on_accept = {
	    scope:actor = { 
			add_sorcery_lifestyle_perk_points = -1
			stress_impact = {
				base = major_stress_gain
			}
			duel = {
				skill = learning
				value = 10
				50 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					scope:recipient = {
						add_trait = theurgist
						add_opinion = {
							target = scope:actor
							modifier = mentored_me_opinion
						}
					}
				}
				50 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier 
					}
				}
			}
		}	
	}

	ai_frequency = 0
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

	is_shown = {
		always = no # deprecated in favor of the Journey
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { has_trait = theurgist }
	}

	is_valid_showing_failures_only = {
		custom_tooltip = {
			text = "He_must_be_my_friend"
			scope:recipient = { 
				OR = {
					has_relation_lover = scope:actor
					has_relation_friend = scope:actor
					has_relation_best_friend = scope:actor
					has_relation_soulmate = scope:actor
				}
			}
		}
		scope:actor = {
			NOT = { is_at_war_with = scope:recipient }
			NOT = { faith = { has_doctrine = doctrine_supehunting_crime } }
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
			POD_is_not_trapped_in_umbra_trigger = { ACTOR = scope:actor }
		}
	}
	
	on_accept = {
	    scope:actor = { 
			add_trait = theurgist
			add_opinion = {
				target = scope:recipient
				modifier = mentored_me_opinion
			}
		}
	}

	ai_frequency = 0
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
