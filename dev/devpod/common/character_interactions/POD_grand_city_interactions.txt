		}

		localization = crime_witch
	}
	
	send_option = {
		is_valid = {
			scope:recipient = {
				has_character_modifier = known_non_believer
			}
		}
		flag = infidel
		localization = crime_infidel
	}
	
	send_option = {
		is_valid = {
			scope:recipient = {
				has_trait = sorcerer
			}
		}
		flag = sorcerer
		localization = secret_sorcerer
	}
	
	on_accept = {
		if = {
			limit = { scope:deviant = yes }

			}
		}
		if = {
			limit = { scope:cannibal = yes }
			scope:recipient = {
				remove_trait = cannibal
				add_secret = { TYPE = secret_cannibal }
			}
		}
		if = {
			limit = { scope:sodomite = yes }
			scope:recipient = {
				remove_trait = sodomite
				add_secret = { TYPE = secret_homosexual }
			}
		}
		if = {
			limit = { scope:witch = yes }
			scope:recipient = {
				remove_trait = witch
				add_secret = { TYPE = secret_witch }
			}
		}
		if = {
			limit = { scope:infidel = yes }
			scope:recipient = {
				remove_character_modifier = known_non_believer
				add_secret = { TYPE = secret_non_believer }
			}
		}
		if = {
			limit = { scope:sorcerer = yes }
			scope:recipient = {
				remove_trait = sorcerer
				add_secret = { TYPE = secret_sorcerer }
			}
		}
	}
	
	auto_accept = yes

	ai_potential = {
		# The basic requirements
		is_imprisoned = no
		is_landed = yes
	}
	ai_targets = {
		ai_recipients = self
	}
	ai_frequency = 42
	ai_will_do = {
		base = 100
	}
}
