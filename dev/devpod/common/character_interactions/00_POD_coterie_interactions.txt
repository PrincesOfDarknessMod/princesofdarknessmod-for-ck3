POD_create_coterie_interaction = {
	category = interaction_category_coterie
	common_interaction = yes
	desc = POD_create_coterie_interaction_desc
	icon = council
	
	is_shown = {
		scope:actor = {
			POD_has_coterie_trigger = no
			NOT = { has_character_flag = coterie_creation_in_progress }
		}
		scope:actor = scope:recipient
	}
	
	on_accept = {
		custom_tooltip = {
			text = POD_create_coterie_interaction_effect
			scope:actor = {
				POD_create_coterie_effect = yes
				trigger_event = POD_coterie_maintenance.2
			}
		}
		custom_tooltip = POD_create_coterie_interaction_effect2
	}

	auto_accept = yes
	
	ai_frequency = 0
	ai_will_do = {
		base = 0
	}
}

POD_invite_to_coterie_interaction = {
	category = interaction_category_coterie
	desc = POD_invite_to_coterie_interaction_desc
	icon = council

	is_shown = {
		scope:actor = {
			POD_has_coterie_trigger = yes
			NOT = { has_relation_coterie_member = scope:recipient }
			is_adult = yes
		}
		POD_cross_umbra_interaction_valid_trigger = yes
		
		NOT = { scope:actor = scope:recipient }
		
		POD_incompatible_splats_trigger = no
		
		scope:recipient = {
			is_ai = yes
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
	}

	is_valid_showing_failures_only = {
		POD_coterie_interaction_requirements_trigger = yes
		custom_tooltip = {
			scope:recipient = {
				opinion = {
					target = scope:actor
					value >= 50
				}
			}
			text = POD_coterie_opinion_requirement_tt
		}
		custom_tooltip = {
			text = POD_at_coterie_size_cap
			scope:actor = {
				POD_coterie_at_max_size_trigger = no
			}
		}
		custom_tooltip = {
			text = POD_coterie_requirement_same_realm
			scope:actor.top_liege = scope:recipient.top_liege
		}
		
		scope:actor = {
			custom_tooltip = {
				text = POD_invite_to_coterie_interaction_mortal
				trigger_if = {
					limit = {
						NOR = {
							is_mortal_trigger = yes
							POD_is_supehunter_trigger = yes
							POD_is_demon_trigger = yes
							POD_is_mummy_trigger = yes
						}
					}
					scope:recipient = {
						is_mortal_trigger = no
					}
				}
			}
			custom_tooltip = {
				text = POD_invite_to_coterie_interaction_mortal_thrall
				trigger_if = {
					limit = {
						POD_is_demon_trigger = yes
						NOT = { has_relation_demonic_thrall = scope:recipient }
					}
					scope:recipient = {
						is_mortal_trigger = no
					}
				}
			}
			custom_tooltip = {
				text = POD_invite_to_coterie_interaction_mortal_neteru
				trigger_if = {
					limit = {
						POD_is_mummy_trigger = yes
						NOT = { has_relation_worshipping_neteru = scope:recipient }
					}
					scope:recipient = {
						is_mortal_trigger = no
					}
				}
			}
		}
		
		scope:recipient = {
			custom_tooltip = {
				text = POD_recipient_already_has_coterie
				POD_has_coterie_trigger = no
			}
			
			POD_is_valid_coterie_member_trigger = yes
		}
	}

	on_accept = {
		custom_tooltip = {
			text = POD_invite_to_coterie_interaction_effect
			scope:actor.var:coterie = {
				save_scope_as = coterie
			}
			scope:recipient = {
				POD_join_coterie_effect = yes
			}
		}
		if = {
			limit = {
				scope:actor = {
					has_trait = shapeshifter
					NOT = { religion = religion:blackspiraldancer_religion }
				}
			}
			custom_tooltip = POD_invite_to_coterie_interaction_packbond
		}
		else = {
			custom_tooltip = POD_invite_to_coterie_interaction_rivalry
		}
		if = {
			limit = {
				scope:actor = {
					has_trait = shapeshifter
					faith = { has_doctrine = special_doctrine_beast_courts }
					POD_coterie_at_max_size_trigger = yes
				}
			}
			scope:actor = { trigger_event = POD_werewolves_pack.10 }
		}
	}
	
	auto_accept = yes
	
	# ai_accept = {
		# base = -25
		
		# opinion_modifier = { # Opinion Factor
			# who = scope:recipient
			# opinion_target = scope:actor
			# desc = AI_SIMPLE_OPINION_REASON
		# }
	# }
	
	ai_frequency = 0
	ai_will_do = {
		base = 0
	}
}

		}
	}

	
	auto_accept = yes

POD_disband_coterie_interaction = {
	category = interaction_category_coterie
	common_interaction = yes
	desc = POD_disband_coterie_interaction_desc
	icon = council
	
	is_shown = {
		scope:actor = {
			POD_has_coterie_trigger = yes
			NOT = { has_character_flag = coterie_creation_in_progress }
		}
		scope:actor = scope:recipient
	}

	is_valid_showing_failures_only = {
		POD_coterie_interaction_requirements_trigger = yes
		custom_tooltip = {
			scope:actor = {
				trigger_if = {
					limit = {
						has_trait = shapeshifter
					}
					OR = {
						religion = religion:blackspiraldancer_religion
						POD_has_coterie_members_trigger = no
					}
				}
			}
			text = POD_disband_coterie_interaction_packbond
		}
	}
	
	on_accept = {
		scope:actor = {
			custom_tooltip = POD_disband_coterie_interaction_effect
			var:coterie = {
				every_in_list = {
					variable = members
					limit = {
						NOT = { this = scope:actor }
					}
					POD_leave_coterie_with_rivalry_effect = { RIVAL = scope:actor }
				}
			}
		}
	}

	auto_accept = yes
	
	ai_frequency = 0
	ai_will_do = {
		base = 0
	}
}
