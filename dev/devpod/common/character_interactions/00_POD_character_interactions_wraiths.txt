    category = interaction_category_wraith

		scope:recipient = {

				type = convert_to_wraith

POD_command_wraith_interaction = {
	desc = POD_command_wraith_interaction_desc
	common_interaction = yes
	interface_priority = 90
	category = interaction_category_wraith
	
	is_shown = {
		scope:actor = { is_ai = no }
		scope:actor = {
			OR = {
				has_trait = wraith
				has_trait = oblivionadvanced
				has_perk = oblivions_sight_perk
				has_trait = necromancyadvanced
				has_perk = lifeless_tongues_perk
				has_perk = bind_the_dead_perk
				has_perk = divine_fetters_perk
				has_perk = saturnal_manes_perk
				has_perk = via_geniorum_perk
				has_perk = flesh_flux_perk
				has_perk = command_the_dead_perk
				has_perk = voice_of_the_yama_kings_perk
				has_perk = durance_perk
			}
		}
		scope:actor = { num_of_relation_bound_wraith >= 1 }
	}
	
	is_valid_showing_failures_only = {
		scope:recipient = {
			NOT = { has_trait = wraith }
			NOT = { has_trait = torpor }
		}
	}
	
	populate_actor_list = {
		scope:actor = {
			every_relation = {
				type = bound_wraith
				limit = { 
					is_alive = yes 
					NOR = { 
						has_trait = frozen 
						has_character_modifier = traveling_spirit_world_modifier
					}
				}
				
				add_to_list = characters
			}
		}
	}
	
	send_option = {
		is_shown = {
			exists = scope:secondary_actor
		}
		is_valid = {
			scope:haunt = no
			scope:spy = no
			scope:depossess = no
				
			trigger_if = {
				limit = { scope:recipient = {has_character_flag = possessed } }
				scope:recipient = { var:wraith = scope:secondary_actor }
			}
			NOT = {
				scope:recipient = {
					has_opinion_modifier = {
						target = scope:secondary_actor
						modifier = POD_complete_possession_opinion
					}
				}
			}
			
			scope:recipient = {
				NOT = { has_perk = banish_the_dead_perk }
			}
			
			custom_description = {
				text = POD_not_in_possession
				scope:secondary_actor = { 
					OR = { 
						has_perk = legion_perk
						num_of_relation_possessed_target = 0 
					}
				}
				scope:recipient = { num_of_relation_possessesing_wraith = 0 }
			}
		
		}
		flag = possess
		localization = command_wraith_interaction.possess
	
	}
	
	send_option = {
		is_shown = {
			exists = scope:secondary_actor
			scope:recipient = { 
				has_variable = wraith
				var:wraith = scope:secondary_actor 
			}
		}
		is_valid = {
			scope:haunt = no
			scope:spy = no
			scope:possess = no
		
		}
		flag = depossess
		localization = command_wraith_interaction.depossess
	
	}
	send_option = {
		is_shown = { exists = scope:secondary_actor }
		is_valid = {
			scope:spy = no
			scope:possess = no
			scope:depossess = no
			scope:recipient = {
				NOT = { has_trait = effigy }
				NOT = { has_perk = sense_the_dead_perk }
			}
		}
		flag = haunt
		localization = command_wraith_interaction.haunt
	
	}
	send_option = {
		is_shown = { exists = scope:secondary_actor }
		is_valid = {
			scope:haunt = no
			scope:possess = no
			scope:depossess = no
			scope:recipient = {
				any_secret = {
					NOT = { is_known_by = scope:actor }
				}
				NOT = {
					has_perk = sense_the_dead_perk
				}
			}
		}
		flag = spy
		localization = command_wraith_interaction.spy
	}
	send_option = {
		is_shown = {
			exists = scope:secondary_actor
		}
		is_valid = {
			scope:actor = {
				has_usable_hook = scope:secondary_actor
			}
			OR = {
				scope:haunt = yes
				scope:possess = yes
				scope:spy = yes
			}
		}
		flag = hook
		localization = command_wraith_interaction.hook
	}

	
	send_options_exclusive = no
	
	on_accept = {
		scope:actor = { save_scope_as = necromancer }
		scope:secondary_actor = { save_scope_as = actor }
		if = {
			limit = { scope:possess = yes }
			scope:recipient = {

				trigger_event = POD_wraith_possession.100
				custom_tooltip = POD_command_wraith_possess_desc
			}
		}
		if = {
			limit = { scope:depossess = yes }
			 POD_wraith_end_possession = {
				 WRAITH = scope:actor
				 PUPPET = scope:recipient
			 }
		}
		if = {
			limit = { scope:spy = yes }
			scope:recipient = {
				
				trigger_event = POD_wraith_spying.100
				custom_tooltip = POD_command_wraith_spy_desc
			}
		}
		if = {
			limit = { scope:haunt = yes }
			scope:actor = {

				trigger_event = POD_wraith_haunting.100
				custom_tooltip = POD_command_wraith_haunt_desc
			}
		}
		if = {
			limit = { scope:hook = yes }
			scope:necromancer = {
				use_hook = scope:actor
			}
		}
		if = {
			limit = { scope:secondary_actor = { is_ruler = no } }
		
			hidden_effect = {
				scope:secondary_actor = {
					set_employer = scope:recipient
					return_to_court = yes
				}
			}
			custom_tooltip = POD_command_wraith_interaction.movecourt
		
		}
		
		
		
	}
	can_send = {
		OR = {
			scope:haunt = yes
			scope:spy = yes
			scope:possess = yes
			scope:depossess = yes
		}
	
	}
	auto_accept = {
		custom_description = {
			text = "spending_hook"
			subject = scope:actor
			object = scope:secondary_actor
			scope:hook = yes
			scope:actor = { has_strong_hook = scope:secondary_actor }
		}
	
	}
	
	ai_accept = {
		base = -50
		opinion_modifier = { # More likely to accept if Recipient likes the Actor
			who = scope:secondary_actor
			opinion_target = scope:actor
			multiplier = 1.0
			desc = AI_OPINION_REASON
		}
		modifier = {
			add = 50
			scope:secondary_actor = {
				has_dread_level_towards = {
					target = scope:actor
					level = 1
				}
			}
			desc = INTIMIDATED_REASON
		}
		modifier = {
			add = 100
			scope:secondary_actor = {
				has_dread_level_towards = {
					target = scope:actor
					level = 2
				}
			}
			desc = COWED_REASON
		}
		modifier = {
			scope:secondary_actor.house = scope:actor.house
			add = 50
			desc = IS_SAME_HOUSE_REASON
		}
	}

}

		ai_recipients = self
	}

	}

POD_set_fetter_character_interaction = {
	category = interaction_category_wraith
	desc = POD_set_fetter_interaction_desc
	options_heading = POD_set_fetter_interaction_options_heading
	icon = prison
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { is_ai = no }
		scope:actor = { has_trait = wraith } 
		scope:actor = { 
			has_variable = fetter_count
			var:fetter_count > 0
		} 
		scope:recipient = { NOT = { has_character_flag = is_a_fetter_flag } }
	}
	
	is_valid_showing_failures_only = {
		scope:recipient = {
			trigger_if = {
				limit = { NOT = { is_any_family_relation_or_spouse_trigger = { CHARACTER = scope:actor } } }
				scope:actor = { has_perk = souls_interwoven_perk }
			}
			trigger_if = {
				limit = { has_trait = wraith }
				scope:actor = { has_perk = lifeline_perk }
			}
		}
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				has_variable = fetter_count
				var:fetter_count >= 1 
			}
		}
		flag = POD_num1
		localization = POD_num1
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				has_variable = fetter_count
				var:fetter_count >= 2 
			}
		}
		flag = POD_num2
		localization = POD_num2
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				has_variable = fetter_count
				var:fetter_count >= 3 
			}
		}
		flag = POD_num3
		localization = POD_num3
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				has_variable = fetter_count
				var:fetter_count >= 4 
			}
		}
		flag = POD_num4
		localization = POD_num4
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				has_variable = fetter_count
				var:fetter_count >= 5 
			}
		}
		flag = POD_num5
		localization = POD_num5
	}
	
	on_accept = {
		scope:actor = {
			custom_tooltip = POD_set_fetter_character_interaction_tooltip
			if = {
				limit = { scope:POD_num1 = yes }
				add_to_variable_list = { name = fetter_1_list target = scope:recipient }
				change_variable = { name = fetter_count subtract = 1 }
			}
			if = {
				limit = { scope:POD_num2 = yes }
				add_to_variable_list = { name = fetter_2_list target = scope:recipient }
				change_variable = { name = fetter_count subtract = 2 }
			}
			if = {
				limit = { scope:POD_num3 = yes }
				add_to_variable_list = { name = fetter_3_list target = scope:recipient }
				change_variable = { name = fetter_count subtract = 3 }
			}
			if = {
				limit = { scope:POD_num4 = yes }
				add_to_variable_list = { name = fetter_4_list target = scope:recipient }
				change_variable = { name = fetter_count subtract = 4 }
			}
