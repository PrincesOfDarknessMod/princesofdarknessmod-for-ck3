
POD_use_alchemical_product_interaction  = {
	category = interaction_category_mummy
	desc = POD_use_alchemical_product_interaction_desc
	options_heading = POD_use_alchemical_product_interaction_options_heading
	icon = POD_hekau_interaction
	
	target_type = artifact
	target_filter = actor_artifacts
	
	can_be_picked_artifact = {
		scope:target = { has_variable = alchemy }
	}
	
	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { has_trait = mummy }
	}

	is_valid = {
		scope:recipient = { can_use_alchemical_product_trigger = yes }
	}

	is_valid_showing_failures_only = {
		scope:recipient = { 
			custom_description = {
				text = "demand_artifact_interaction_valid_tt_1"
				has_any_artifact = yes
				any_character_artifact = { has_variable = alchemy }
			}
		}
	}
			
	on_accept = {
		scope:target = {
			POD_alchemy_usage_effect = { CHARACTER = scope:recipient }
		}
	}
	
	auto_accept = yes
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no
		is_landed = yes
	}
	ai_targets = {
		ai_recipients = self
	}
	ai_frequency = 18
	ai_will_do = {
		base = 20
	}
}

			NOT = { has_trait = torpor }

					NOT = { has_variable = poison }
				}
			}
		}

	on_accept = {
		scope:target = { 
			POD_alchemy_opinion_effect = { CHARACTER1 = scope:actor CHARACTER2 = scope:recipient }
			POD_alchemy_usage_effect = { CHARACTER = scope:recipient }
		}
	}
	
	auto_accept = yes
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no
		is_landed = yes
		
		number_of_knights > 0
		primary_title.tier > tier_barony
	}

		ai_recipients = peer_vassals
		ai_recipients = courtiers
		max = 10
	}
	ai_frequency = 60
	ai_will_do = {
		base = 0
		modifier = {
			add = 50
			scope:recipient = {
				is_knight_of = scope:actor
			}
		}
	}
}

POD_sell_alchemical_product_interaction = {

	target_filter = actor_artifacts

		}

	}

		scope:recipient = { is_ruler = yes }
		scope:recipient = { is_available = yes }

			limit = { exists = scope:target }
			can_afford_alchemy_product_with_opinion_trigger = {
				BUYER   = scope:recipient
				SELLER  = scope:actor
				PRODUCT = scope:target
			}
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			NOT = { has_trait = automaton }
			NOT = { has_trait = wight }

				text = "demand_artifact_interaction_valid_tt_1"

					has_variable = alchemy 
					NOT = { has_variable = poison }
				}
			}
		}
	}
	
	on_accept = {
		scope:target = { 
			POD_alchemy_sell_effect = { CHARACTER1 = scope:actor CHARACTER2 = scope:recipient }
			POD_alchemy_usage_effect = { CHARACTER = scope:recipient }
		}
	}
	
	ai_frequency = 48
	
	ai_min_reply_days = 0
	ai_max_reply_days = 0

	ai_potential = {
		# The basic requirements
		is_imprisoned = no 
		is_landed = yes 
		
		is_at_war = no
	}
	
	ai_targets = {
		ai_recipients = neighboring_rulers
		max = 10
	}
	
	ai_accept = {
		base = -20 # Try to make it 0 for most interactions
		
		opinion_modifier = { # Opinion Factor
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.25
			desc = AI_OPINION_REASON
		}
		
		modifier = {
			add = 25
			desc = GOLD_REASON
		}
		
		modifier = {
			desc = VAMPIRE_IS_SHUNNED
			add = -1000
			scope:target ?= { var:alchemy = flag:vampire_concoction }
			scope:recipient.faith = { has_doctrine_parameter = vampirism_shunned }
		}
		
		modifier = {
			desc = VAMPIRE_IS_CRIME
			add = -1000
			scope:target ?= { var:alchemy = flag:vampire_concoction }
			scope:recipient.faith = { has_doctrine_parameter = vampirism_illegal }
		}
	}
	
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 20
			scope:actor = { gold < medium_gold_value }
		}
		
		modifier = {
			add = 40
			scope:actor = { gold < minor_gold_value }
		}
		
		modifier = {
			add = 80
			scope:actor = { gold < 0 }
		}
		
		modifier = {
			factor = 0
			scope:recipient = { is_ai = no }
		}
		
		modifier = {
			factor = 0
			scope:recipient = { has_trait = mummy }
		}
	}
}

POD_buy_alchemical_product_interaction = {
	category = interaction_category_diplomacy
	desc = POD_buy_alchemical_product_interaction_desc
	options_heading = POD_buy_alchemical_product_interaction_options_heading
	common_interaction = yes
	icon = POD_hekau_interaction
	
	target_type = artifact
	target_filter = recipient_artifacts
	
	can_be_picked_artifact = {
		scope:target = { has_variable = alchemy }
	}
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { is_ruler = yes }
		scope:recipient = { has_trait = mummy }
		POD_incompatible_splats_trigger = no
	}

	is_valid = {
		scope:actor = { can_use_alchemical_product_trigger = yes }
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			NOT = { is_at_war_with = scope:recipient }
			is_imprisoned = no
		}
		scope:recipient = { is_available = yes }
		POD_cross_umbra_interaction_valid_trigger = yes		
		scope:recipient = { 
			custom_description = {
				text = "demand_artifact_interaction_valid_tt_1"
				has_any_artifact = yes
				any_character_artifact = { has_variable = alchemy }
			}
		}
	}
	
	on_accept = {
		scope:target = {
			POD_alchemy_buy_effect = { CHARACTER1 = scope:recipient CHARACTER2 = scope:actor }
			if = {
				limit = { has_variable = poison }
				set_owner = {
					target = scope:actor
					history = {
						location = scope:actor.location
						actor = scope:recipient
						recipient = scope:actor
						type = given
					}
				}
			}
			else = {
				POD_alchemy_usage_effect = { CHARACTER = scope:actor }
			}
		}
	}
	
	ai_frequency = 48
	
	ai_min_reply_days = 0
	ai_max_reply_days = 0

	ai_potential = {
		# The basic requirements
		is_imprisoned = no 
		is_landed = yes 
		
		is_at_war = no
	}
	
	ai_targets = {
		ai_recipients = neighboring_rulers
		max = 10
	}
	
	ai_accept = {
		base = -10 # Try to make it 0 for most interactions
		
		opinion_modifier = { # Opinion Factor
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 1.0
			desc = AI_OPINION_REASON
		}
		
		modifier = {
			add = 35
			scope:actor = { is_ai = yes }
		}
	}
	
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 20
			scope:actor = { gold > medium_gold_value }
		}
		
		modifier = {
			factor = 0
			scope:recipient = { is_ai = no }
		}
		
		modifier = {
			factor = 0
			scope:actor = { has_trait = mummy }
		}
	}
}

POD_poison_alchemical_product_interaction = {
	category = interaction_category_mummy
	desc = POD_poison_alchemical_product_interaction_desc
	options_heading = POD_poison_alchemical_product_interaction_options_heading
	icon = POD_hekau_interaction
	
	target_type = artifact
	target_filter = actor_artifacts
	
	can_be_picked_artifact = {
		scope:target = { 
			has_variable = alchemy
			has_variable = poison
		}
	}
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_trait = mummy }
	}

	is_valid = {}

	is_valid_showing_failures_only = {
		scope:recipient = {
			NOT = { has_trait = torpor }
			NOT = { has_trait = automaton }
			NOT = { has_trait = wight }
			NOT = { has_trait = barghest }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }	
			POD_is_not_clone_trigger = yes	
		}
		POD_cross_umbra_interaction_valid_trigger = yes
		scope:actor = { 
			custom_description = {
				text = "demand_artifact_interaction_valid_tt_1"
				has_any_artifact = yes
				any_character_artifact = { 
					has_variable = alchemy 
					has_variable = poison
				}
			}
		}
	}
	
	on_accept = {
		scope:recipient = { 
			add_opinion = {
				target = scope:actor
				modifier = hurt_opinion
				opinion = -50
			}
			progress_towards_rival_effect = {
				CHARACTER = scope:actor
				REASON = rival_poison_vengeance
				OPINION = 0
			}
		}
		scope:actor = {
			duel = {
				skill = intrigue
				target = scope:recipient
				50 = {
					desc = board_games.0001.tt.success
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					scope:target = { POD_alchemy_usage_effect = { CHARACTER = scope:recipient } }
				}
				50 = {
					desc = board_games.0001.tt.failure
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier 
					}
				}
			}
		}
	}
	
	auto_accept = yes
}
