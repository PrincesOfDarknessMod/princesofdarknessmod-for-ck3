					var:alchemy_craft_slot_9 = 0
					var:alchemy_craft_slot_10 = 0
				}
			}
		}
	}
			
	on_accept = {
		scope:actor = {
			save_scope_as = drinker
			save_scope_as = offer
			trigger_event = POD_mummy_alchemy.1000
		}
	}
	
	auto_accept = yes
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no
		is_landed = yes

	}

POD_offer_alchemical_product_interaction = {
	category = interaction_category_mummy
	desc = POD_offer_alchemical_product_interaction_desc
	options_heading = POD_offer_alchemical_product_interaction_options_heading
	icon = POD_hekau_interaction
	
	is_shown = {
		NOT = {scope:actor = scope:recipient }
		scope:actor = { has_trait = mummy }
		scope:actor = { has_alchemy_craft_slot_trigger = yes }
	}

	is_valid_showing_failures_only = {
		scope:recipient = { has_relation_worshipped_reborn = scope:actor }
		scope:recipient = { is_available = yes }
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
			POD_is_not_trapped_in_umbra_trigger = { ACTOR = scope:actor }
		}
		custom_description = {
			text = alchemy_craft_slots_are_empty
			scope:actor = {
				NAND = { 
					var:alchemy_craft_slot_1 = 0
					var:alchemy_craft_slot_2 = 0
					var:alchemy_craft_slot_3 = 0
					var:alchemy_craft_slot_4 = 0
					var:alchemy_craft_slot_5 = 0
					var:alchemy_craft_slot_6 = 0
					var:alchemy_craft_slot_7 = 0
					var:alchemy_craft_slot_8 = 0
					var:alchemy_craft_slot_9 = 0
					var:alchemy_craft_slot_10 = 0
				}
			}
		}
	}
			
	on_accept = {
		scope:recipient = {
			add_opinion = {
				target = scope:actor
				modifier = gift_opinion
				opinion = advanced_alchemy_product_opinion_value
			}
			save_scope_as = drinker
		}
		scope:actor = {
			save_scope_as = offer
			trigger_event = POD_mummy_alchemy.1000
		}
	}
	
	auto_accept = yes
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no
		is_landed = yes
		
		number_of_knights > 0
		primary_title.tier > tier_barony
	}
	ai_targets = {
		ai_recipients = peer_vassals
		ai_recipients = courtiers
		max = 10
	}
	ai_frequency = 60
	ai_will_do = {
		base = 0
		modifier = {
			add = 50
			scope:recipient = {
				is_knight_of = scope:actor
			}
		}
		modifier = {
			factor = 2
			OR = {
				scope:princely_tonic = yes
				scope:kingly_tonic = yes
				scope:divine_tonic = yes
			}
		}
	}
}
