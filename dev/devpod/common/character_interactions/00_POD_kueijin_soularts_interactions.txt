		}

		localization = POD_interaction_scheme_attack

	send_option = {
		is_shown = {
			NOR = {
				scope:actor = { 
					has_hook_of_type = {
						target = scope:recipient
						type = strong_soul_shackles_hook
					}
				}
				scope:recipient = { is_imprisoned_by = scope:actor }
				scope:recipient = { has_relation_necromancer = scope:actor }
			}
		}
		is_valid = {
			scope:recipient = { can_start_single_combat_trigger = yes }

				object = scope:actor
				text = "I_must_be_at_the_same_location"

		}
		flag = duel_attack
		localization = POD_interaction_duel_attack
	}
	
	send_option = {
		is_valid = {
			OR = {
				scope:actor = { 
					has_hook_of_type = {
						target = scope:recipient
						type = strong_soul_shackles_hook
					}
				}
				scope:recipient = { is_imprisoned_by = scope:actor }
				scope:recipient = { has_relation_necromancer = scope:actor }
			}
		}
		flag = automatic
		localization = POD_interaction_automatic
	}
	
	on_accept = {
		scope:recipient = {
			custom_tooltip = POD_project_negative_psychic_energy.tt
		}
		
		if = {
			limit = {
				scope:scheme_attack = yes
			}
			scope:actor = {
				start_scheme = {
					type = POD_project_negative_psychic_energy

				}

		is_landed = yes

		ai_recipients = primary_war_enemies

		base = 0

		    factor = 2
