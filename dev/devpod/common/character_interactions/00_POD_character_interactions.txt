blood_sorcery_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	#interface_priority = 84
	
	is_shown = {
		scope:actor = { 
			OR = {
				has_trait = bloodsorcery
				has_trait = bloodsorceryadvanced
			}
		}
		
	}
	
	is_valid_showing_failures_only = {
		scope:actor = { 
			custom_tooltip = {
				text = song_of_jupiter_botch.ct
				NOT = { has_character_flag = song_of_jupiter_botch }
			} 
			custom_tooltip = {
				text = blood_sorcery_open.ct
				NOT = { has_variable = blood_sorcery_open }
			} 
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
			POD_is_not_trapped_in_umbra_trigger = { ACTOR = scope:actor }
		}	
	}
	on_accept = {
		custom_tooltip = open_bs_menu
		
		scope:actor = {
		if = {
			limit = {
				NOT = { has_character_flag = got_BS_exp }
			}
			if = {
				limit = { has_trait = bloodsorceryadvanced }
				set_variable = { name = blood_sorcery_points value = 600}
			}
			else = {
				set_variable = { name = blood_sorcery_points value = 100}
			}
			add_character_flag = got_BS_exp
			
			
		}
			#remove_variable = blood_sorcery_open
			add_to_variable_list = {
				name = path_of_mars
				target = flag:strike_true
			}
			add_to_variable_list = {
				name = path_of_mars
				target = flag:wind_dance
			}
			add_to_variable_list = {
				name = path_of_mars
				target = flag:comrades_in_arms
			}
			add_to_variable_list = {
				name = path_of_blood
				target = flag:a_taste_of_blood
			}
			add_to_variable_list = {
				name = path_of_blood
				target = flag:blood_of_potency
			}
			add_to_variable_list = {
				name = path_of_blood
				target = flag:theft_of_vitae
			}
			
			add_to_variable_list = {
				name = path_of_neptunes_might
				target = flag:eyes_of_the_sea
			}
			add_to_variable_list = {
				name = path_of_neptunes_might
				target = flag:prison_of_water
			}
			add_to_variable_list = {
				name = path_of_neptunes_might
				target = flag:dehydrate
			}
			add_to_variable_list = {
				name = path_of_curses
				target = flag:stigma
			}
			add_to_variable_list = {
				name = path_of_curses
				target = flag:malady
			}
			add_to_variable_list = {
				name = path_of_curses
				target = flag:acedia
			}
			add_to_variable_list = {
				name = path_of_focused_mind
				target = flag:readiness
			}
			add_to_variable_list = {
				name = path_of_focused_mind
				target = flag:centering
			}
			add_to_variable_list = {
				name = path_of_focused_mind
				target = flag:perfect_clarity
			}
			add_to_variable_list = {
				name = path_of_praapti
				target = flag:emergency_recall
			}
			add_to_variable_list = {
				name = path_of_praapti
				target = flag:bind_in_vine_and_sand
			}
			add_to_variable_list = {
				name = path_of_praapti
				target = flag:travel_on_mythic_strings
			}
			add_to_variable_list = {
				name = path_of_music_of_the_shperes
				target = flag:song_of_venus
			}
			add_to_variable_list = {
				name = path_of_music_of_the_shperes
				target = flag:song_of_saturn
			}
			add_to_variable_list = {
				name = path_of_music_of_the_shperes
				target = flag:song_of_jupiter
			}
			if = {
				limit = {NOT = {has_variable = blood_sorcery_points}}
				set_variable = { name = blood_sorcery_points value = 0}
			}
			
			set_variable = { name = blood_sorcery_cost value = 0}
			
			set_variable = { name = chosen_path value = flag:nothing}
			set_variable = { name = target value = scope:recipient}
			
			
			trigger_event = path_of_blood.3000
			
			
		}
	}
	
	auto_accept = yes
}

convert_to_government_interaction = { 
	category = interaction_category_vassal
	common_interaction = yes
	
	desc = convert_to_government_interaction_desc
	icon = icon_contract_modification_single
	
	is_shown = {
		scope:actor = { is_landed = yes }
		scope:recipient = { is_landed = yes }
		NOT = { scope:recipient = { has_same_government = scope:actor } }
		scope:actor.liege = scope:recipient
		OR = {
			AND = {
				scope:actor = {
					OR = {
						has_trait = demon 
						has_trait = laham
						has_trait = nephilim
						is_a_thrall = yes
					}
				}
				scope:recipient = {
					OR = {
						has_trait = demon 
						has_trait = laham
						has_trait = nephilim
						is_a_thrall = yes
					}
				}
			}
			AND = {
				scope:actor = {
					has_POD_vamp_blood_trigger = yes
					NOT = { religion = { is_in_family = rf_hunters } }
				}
				scope:recipient = {
					has_POD_vamp_blood_trigger = yes
					NOT = { religion = { is_in_family = rf_hunters } }
				}
			}
			AND = {
				scope:actor = {
					has_POD_kj_blood_trigger = yes
					NOT = { has_trait = hun_and_po }
				}
				scope:recipient = {
					has_POD_kj_blood_trigger = yes
					NOT = { has_trait = hun_and_po }
				}
			}
			AND = {
				scope:actor = {
					has_trait = wraith 
					NOR = { 
						has_trait = spectre
						has_trait = ferryman
					}
				}
				scope:recipient = {
					has_trait = wraith 
					NOR = { 
						has_trait = spectre
						has_trait = ferryman
					}
				}
			}
			AND = {
				has_global_variable = celestial_army_restored
				scope:actor = {
					has_trait = shih
				}
				scope:recipient = {
					has_trait = shih
				}
			}
			AND = {
				scope:actor = {
					has_POD_fae_blood_trigger = yes
				}
				scope:recipient = {
					has_POD_fae_blood_trigger = yes
				}
			}
			AND = {
				scope:actor = {
					OR = {
						has_trait = kinfolk
						has_trait = shapeshifter
					}
				}
				scope:recipient = {
					OR = {
						has_trait = kinfolk
						has_trait = shapeshifter
					}
				}
			}
		}
	}
	
	on_accept = {
		scope:actor = {
			if = {
				limit = { liege = { has_government = infernal_court_government } }
				change_government = infernal_court_government
			}
			if = {
				limit = { liege = { has_government = celestial_army_government } }
				change_government = celestial_army_government
			}
			if = {
				limit = { liege = { has_government = traditions_government } }
				change_government = traditions_government
			}
			if = {
				limit = { liege = { has_government = free_state_government } }
				change_government = free_state_government
			}
			if = {
				limit = { liege = { has_government = august_court_government } }
				change_government = august_court_government
			}
			if = {
				limit = { liege = { has_government = heretical_court_government } }
				change_government = heretical_court_government
			}
			if = {
				limit = { liege = { has_government = hierarchy_government } }
				change_government = hierarchy_government
			}
			if = {
				limit = { liege = { has_government = guild_government } }
				change_government = guild_government
			}
			if = {
				limit = { liege = { has_government = heretic_government } }
				change_government = heretic_government
			}
			if = {
				limit = { liege = { has_government = far_shore_government } }
				change_government = far_shore_government
			}
			if = {
				limit = { liege = { has_government = yellow_springs_government } }
				change_government = yellow_springs_government
			}
			if = {
				limit = { liege = { has_government = renegade_government } }
				change_government = renegade_government
			}
			if = {
				limit = { liege = { has_government = household_government } }
				change_government = household_government
			}
			if = {
				limit = { liege = { has_government = litany_government } }
				POD_set_litany_government_effect = yes
			}
			else = {	
				every_held_title = {
					limit = { tier <= 2 }
					title_province = {
						if = {
							limit = { has_holding_type = tribal_holding }
							POD_replace_tribal_holding_effect = yes
						}
					}
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_min_reply_days = 0
	ai_max_reply_days = 0
	
	ai_frequency = 60
	
	ai_targets = {
		ai_recipients = liege
	}
	
	ai_will_do = {
		base = 100
	}
}
