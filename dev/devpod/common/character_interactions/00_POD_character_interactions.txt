convert_to_government_interaction = { 
	category = interaction_category_vassal
	common_interaction = yes
	
	desc = convert_to_government_interaction_desc
	icon = icon_contract_modification_single
	
	is_shown = {
		scope:actor = { is_landed = yes }
		scope:recipient = { is_landed = yes }

			AND = {

						has_trait = demon

					}
				}

					OR = {
						has_trait = demon 
						has_trait = laham

					}
				}
			}
			AND = {
				scope:actor = {
					has_POD_vamp_blood_trigger = yes
					NOT = { religion = { is_in_family = rf_hunters } }
				}
				scope:recipient = {
					has_POD_vamp_blood_trigger = yes
					NOT = { religion = { is_in_family = rf_hunters } }

				scope:actor = {

					has_trait = wraith

						has_trait = spectre
						has_trait = ferryman
					}
				}
				scope:recipient = {
					has_trait = wraith 
					NOR = { 
						has_trait = spectre
						has_trait = ferryman
					}
				}
			}
			AND = {
				has_global_variable = celestial_army_restored
				scope:actor = {
					has_trait = shih
				}
				scope:recipient = {
					has_trait = shih
				}
			}
			AND = {
				scope:actor = {
					has_POD_fae_blood_trigger = yes
				}
				scope:recipient = {
					has_POD_fae_blood_trigger = yes
				}
			}
			AND = {
				scope:actor = {
					OR = {
						has_trait = kinfolk
						has_trait = shapeshifter
					}
				}
				scope:recipient = {
					OR = {
						has_trait = kinfolk
						has_trait = shapeshifter
					}
				}
			}
		}
	}
	
	on_accept = {
		scope:actor = {
			if = {
				limit = { liege = { has_government = infernal_court_government } }
				change_government = infernal_court_government
			}
			if = {
				limit = { liege = { has_government = celestial_army_government } }
				change_government = celestial_army_government
			}
			if = {
				limit = { liege = { has_government = traditions_government } }
				change_government = traditions_government
			}
			if = {
				limit = { liege = { has_government = free_state_government } }
				change_government = free_state_government
			}
			if = {
				limit = { liege = { has_government = august_court_government } }
				change_government = august_court_government
			}
			if = {
				limit = { liege = { has_government = heretical_court_government } }
				change_government = heretical_court_government
			}
			if = {
				limit = { liege = { has_government = hierarchy_government } }
				change_government = hierarchy_government
			}
			if = {
				limit = { liege = { has_government = guild_government } }
				change_government = guild_government
			}
			if = {
				limit = { liege = { has_government = heretic_government } }
				change_government = heretic_government
			}
			if = {
				limit = { liege = { has_government = far_shore_government } }
				change_government = far_shore_government
			}
			if = {
				limit = { liege = { has_government = yellow_springs_government } }
				change_government = yellow_springs_government
			}
			if = {
				limit = { liege = { has_government = renegade_government } }
				change_government = renegade_government
			}
			if = {
				limit = { liege = { has_government = household_government } }
				change_government = household_government
			}
			if = {
				limit = { liege = { has_government = litany_government } }
				POD_set_litany_government_effect = yes
			}
			else = {	
				every_held_title = {
					limit = { tier <= 2 }
					title_province = {
						if = {
							limit = { has_holding_type = tribal_holding }
							POD_replace_tribal_holding_effect = yes
						}
					}
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_min_reply_days = 0
	ai_max_reply_days = 0
	
	ai_frequency = 60
	
	ai_targets = {
		ai_recipients = liege
	}
	
	ai_will_do = {
		base = 100
	}
}

POD_thorns_caucus_interaction = {
	category = interaction_category_sect
	common_interaction = yes
	icon = alliance
	desc = POD_thorns_caucus_interaction_desc

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_variable = custom_convention_of_thorns }
		faith:camahumanity = {
			is_target_in_variable_list = { 
				name = thorn_elector_list 
				target = scope:recipient 
			}
		}
	}

	is_valid_showing_failures_only = {
		trigger_if = {
			limit = { NOT = { scope:hook ?= yes } }
			scope:actor.gold >= scope:actor.monumental_gold_value
		}
	}

	#Use hook
	send_option = {
		is_valid = {
			scope:actor = {
				has_usable_hook = scope:recipient
			}
		}
		flag = hook	
		localization = GENERIC_SPEND_A_HOOK
	}
	should_use_extra_icon = {
		scope:actor = { has_usable_hook = scope:recipient }
	}
	extra_icon = "gfx/interface/icons/character_interactions/hook_icon.dds"

	send_options_exclusive = no

	on_accept = {
		scope:actor = {
			if = {
				limit = {
					scope:hook = yes
					has_hook = scope:recipient
				}
				use_hook = scope:recipient
			}
			else = {
				pay_short_term_gold = {
					target = scope:recipient
					gold = monumental_gold_value
				}
			}
			every_in_list = {
				variable = voted_tenet
				save_scope_as = tenet
				scope:recipient = { vote_tenet_effect = { TENET = scope:tenet } }
			}
			faith:camahumanity = { calcule_voted_tenet_effect = yes }
		}
		scope:recipient = {
			custom_tooltip = POD_thorns_caucus_interaction.tt
		}
	}
	
	auto_accept = {
		custom_description = {
			text = "spending_hook"
			subject = scope:actor
			object = scope:recipient
			scope:hook = yes
		}
	}
	
	ai_min_reply_days = 0
	ai_max_reply_days = 0
	
	ai_accept = {
		base = -20 # Try to make it 0 for most interactions
		
		opinion_modifier = { # Opinion Factor
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.25
			desc = AI_OPINION_REASON
		}
		
		modifier = {
			add = 25
			scope:hook = no
		}
	}
}

blood_hunt_call_interaction = {
	category = interaction_category_sect
	common_interaction = yes
	desc = blood_hunt_call_interaction_desc
	interface_priority = 60
	interface = call_ally
	special_interaction = call_ally_interaction
	popup_on_receive = yes
	pause_on_receive = yes
	icon = alliance

	greeting = positive
	notification_text = CALL_ALLY_NOTIFICATION

	is_shown = {
		scope:actor = {
			is_at_war = yes
			any_character_war = { using_cb = POD_blood_hunt_war }
		}
		scope:actor.faith = faith:camahumanity
	}
	
	has_valid_target = {
		exists = scope:target
	}	

	has_valid_target_showing_failures_only = {
		scope:target = {
			is_war_leader = scope:actor
		}

		joiner_not_already_in_another_war_with_any_target_war_participants_trigger = {
			WARRIOR = scope:actor
			JOINER = scope:recipient
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient.faith = faith:camahumanity
		
		scope:recipient = {
			is_ruler = yes
		}

		# Only War Leaders can call allies
		scope:actor = {
			custom_description = {
				text = must_be_war_leader
				any_character_war = {
					is_war_leader = scope:actor
				}
			}
		}

		# Cannot call the same character multiple times
		scope:actor = {
			custom_description = {
				text = cannot_call_same_character_multiple_times
				subject = scope:recipient
				any_character_war = {
					ALL_FALSE = {
						was_called = scope:recipient
						is_participant = scope:recipient
					}
