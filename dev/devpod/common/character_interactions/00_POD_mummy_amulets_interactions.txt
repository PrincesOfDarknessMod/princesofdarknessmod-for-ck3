POD_healing_bandages_interaction = {
	category = interaction_category_mummy
	common_interaction = yes
	interface_priority = 110
	desc = POD_healing_bandages_interaction_desc
	options_heading = POD_healing_bandages_interaction_options_heading
	icon = POD_hekau_interaction	

	use_diplomatic_range = no

	cooldown = { months = 3 }	
	
	is_shown = {
		scope:actor = { 
			has_trait = mummy
			has_perk = wrapping_of_imhotep_perk
		}
		POD_is_in_same_plane_trigger = yes
	}


	is_valid_showing_failures_only = {
		scope:actor = { 
			trigger_if = {
				limit = { is_east_mummy_trigger = yes }
				NOT = { has_character_modifier = yangchi0_modifier }
			}
			trigger_if = {
				limit = { is_amenti_trigger = yes }
				NOT = { has_character_modifier = sekhem0_modifier }
			}
		}
		scope:recipient = {
			NOT = { has_trait_with_flag = can_not_cure }
			NOT = {	has_character_modifier = eye_of_sakhmet_modifier }
			custom_description = {
				text = POD_has_injury_trait
				object = scope:recipient
				OR = {
					has_trait = scarred
					has_trait = disfigured
				
					has_trait = wounded_1
					has_trait = wounded_2
					has_trait = wounded_3
				}
			}
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			NOT = { has_trait = statue }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }		
		}
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = scarred
			}
		}	
		is_valid = {
			custom_description = {
				text = POD_permanent_scarred
				object = scope:recipient
				scope:recipient = { NOT = { has_character_flag = POD_permanent_scarring_flag } }
			}
		}
		flag = scarred
		localization = POD_heal_scarred
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = disfigured
			}
		}	
		is_valid = {
			custom_description = {
				text = POD_permanent_disfigured
				object = scope:recipient
				scope:recipient = { NOT = { has_character_flag = POD_permanent_disfigured_flag } }
			}
			scope:actor = { has_perk = wrappings_of_osiris_perk }
		}
		flag = disfigured
		localization = POD_heal_disfigured
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				OR = {
					has_trait = wounded_1
					has_trait = wounded_2
				}
			}
		}	
		flag = wounded
		localization = POD_heal_wounded
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = wounded_3
			}
		}	
		is_valid = {
			scope:actor = { 
				has_perk = wrappings_of_osiris_perk
			}
		}	
		flag = very_wounded
		localization = POD_heal_wounded
	}
	
	on_accept = {
		scope:actor = { 
			POD_willpower_rouse_check_effect = yes
		}
		scope:recipient = {
			play_music_cue = "mx_cue_peace_ensues"

			if = {
				limit = {
					scope:scarred = yes
				}
				scope:recipient = {
					remove_trait = scarred
				}
				scope:actor = { POD_hekau_cost_effect = { COUNT = 1 } }
			}	
			
			if = {
				limit = {
					OR = {
						scope:wounded = yes
						scope:very_wounded = yes
					}
				}
				scope:recipient = {
					remove_trait = wounded_3
					remove_trait = wounded_2
					remove_trait = wounded_1
				}
				scope:actor = { POD_hekau_cost_effect = { COUNT = 2 } }
			}		

			if = {
				limit = {
					scope:disfigured = yes
				}
				scope:recipient = {
					remove_trait = disfigured
				}
				scope:actor = { POD_hekau_cost_effect = { COUNT = 3 } }
			}	
		}
	}
	
	auto_accept = yes

	ai_potential = {
		is_imprisoned = no 
	}
	ai_targets = {
		ai_recipients = self
	}
	ai_targets = {
		ai_recipients = courtiers
		max = 10
	}
	ai_frequency = 12
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 100
			scope:recipient = {
				OR = {
					this = scope:actor
					is_knight_of = scope:actor
				}
			}
		}
	}
}

POD_eye_of_sakhmet_interaction = {
	category = interaction_category_mummy
	desc = POD_eye_of_sakhmet_interaction_desc
	icon = POD_hekau_interaction

	cooldown = { months = 3 }	
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { 
			has_trait = mummy
			has_perk = eye_of_sakhmet_perk
		}
		scope:recipient = {
			NOT = { has_character_modifier = eye_of_sakhmet_modifier }
		}	
		POD_is_in_same_plane_trigger = yes
	}

	is_valid_showing_failures_only = {		
		hurt_wraith_trigger = yes
		scope:actor = { 
			trigger_if = {
				limit = { is_east_mummy_trigger = yes }
				NOT = { has_character_modifier = yangchi0_modifier }
			}
			trigger_if = {
				limit = { is_amenti_trigger = yes }
				NOT = { has_character_modifier = sekhem0_modifier }
			}
		}
		scope:recipient = { 
			NOT = { has_trait = torpor }
		}
		scope:recipient = {
			has_trait = wounded
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			NOT = { has_trait = statue }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }		
		}
	}
	
	send_option = {
		is_shown = {
			NOR = {
				scope:recipient = { is_imprisoned_by = scope:actor }
				scope:recipient = { has_relation_necromancer = scope:actor }
			}
		}
		is_valid = {
			scope:actor = {
				can_start_scheme = {
					type = POD_cast_hekau
					target = scope:recipient
				}
			}
		}
		flag = scheme_attack
		localization = POD_interaction_scheme_attack
	}
	
	send_option = {
		is_shown = {
			NOR = {
				scope:recipient = { is_imprisoned_by = scope:actor }
				scope:recipient = { has_relation_necromancer = scope:actor }
			}
		}
		is_valid = {
			custom_description = {
				object = scope:actor
				text = "I_must_be_at_the_same_location"
				scope:actor = { exists = location }
				scope:recipient = { exists = location }
				scope:actor = { location = scope:recipient.location }
			}
		}
		flag = duel_attack
		localization = POD_interaction_duel_attack
	}
	
	send_option = {
		is_valid = {
			OR = {
				scope:recipient = { is_imprisoned_by = scope:actor }
				scope:recipient = { has_relation_necromancer = scope:actor }
			}
		}
		flag = automatic
		localization = POD_interaction_automatic
	}
	
	on_accept = {
		POD_hekau_cost_effect = { COUNT = 2 }
		if = {
			limit = {
				scope:scheme_attack = yes
			}
			scope:actor = {
				start_scheme = {
					type = POD_cast_hekau
					target = scope:recipient
				}
			}
		}
	
		if = {
			limit = {
				scope:duel_attack = yes
			}
			scope:actor = {
				configure_start_single_combat_effect = {
					SC_INITIATOR = scope:actor
					SC_ATTACKER = scope:actor
					SC_DEFENDER = scope:recipient
					FATALITY = no
					FIXED = no
					LOCALE = throne_room
					OUTPUT_EVENT = POD_mummy_scheme.100
					INVALIDATION_EVENT = tribal.1022
				}
			}
		}
		
		if = {
			limit = {
				scope:automatic = yes
			}
			scope:actor = { trigger_event = POD_mummy_scheme.100 }
		}
	}
	
	auto_accept = yes
	
	ai_frequency = 360
	
	ai_targets = {
		ai_recipients = primary_war_enemies
	}
	
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 45
			scope:actor = {
			    is_at_war_with = scope:recipient
			}
		}
		modifier = {
		    add = 10
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
	}
