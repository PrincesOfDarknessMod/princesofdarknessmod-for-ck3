
POD_thinbloodalchemy_use_alchemical_product_interaction  = {
	category = interaction_category_vampire
	desc = POD_thinbloodalchemy_use_alchemical_product_interaction_desc
	options_heading = POD_thinbloodalchemy_use_alchemical_product_interaction_options_heading
	icon = POD_thinbloodalchemy_interaction
	
	target_type = artifact
	target_filter = actor_artifacts
	
	can_be_picked_artifact = {
		scope:target = { has_variable = alchemy }
	}
	
	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { has_thinbloodalchemy_trigger = yes }
	}

	is_valid = {
		scope:recipient = { can_use_alchemical_product_trigger = yes }
	}

	is_valid_showing_failures_only = {
		scope:recipient = { 
			custom_description = {
				text = "demand_artifact_interaction_valid_tt_1"
				has_any_artifact = yes
				any_character_artifact = { has_variable = alchemy }
			}
		}
	}
			
	on_accept = {
		scope:target = {
			POD_alchemy_usage_effect = { CHARACTER = scope:recipient }
		}
	}
	
	auto_accept = yes
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no
		is_landed = yes
	}
	ai_targets = {
		ai_recipients = self
	}
	ai_frequency = 18
	ai_will_do = {
		base = 20
	}
}

POD_thinbloodalchemy_offer_alchemical_product_interaction = {
	category = interaction_category_vampire
	desc = POD_thinbloodalchemy_offer_alchemical_product_interaction_desc
	options_heading = POD_thinbloodalchemy_offer_alchemical_product_interaction_options_heading
	icon = POD_thinbloodalchemy_interaction
	
	target_type = artifact
	target_filter = actor_artifacts
	
	can_be_picked_artifact = {
		scope:target = { has_variable = alchemy }
	}
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_thinbloodalchemy_trigger = yes }
	}

	is_valid = {
		scope:recipient = { can_use_alchemical_product_trigger = yes }
	}

	is_valid_showing_failures_only = {
		scope:recipient = { is_child_of = scope:actor }
		scope:recipient = { is_available = yes }
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes
		scope:recipient = { 
			custom_description = {
				text = "demand_artifact_interaction_valid_tt_1"
				has_any_artifact = yes
				any_character_artifact = { has_variable = alchemy }
			}
		}
	}
	
	on_accept = {
		scope:target = { 
			POD_alchemy_opinion_effect = { CHARACTER1 = scope:actor CHARACTER2 = scope:recipient }
			POD_alchemy_usage_effect = { CHARACTER = scope:recipient }
		}
	}
	
	auto_accept = yes
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no
		is_landed = yes
		
		number_of_knights > 0
		primary_title.tier > tier_barony
	}
	ai_targets = {
		ai_recipients = peer_vassals
		ai_recipients = courtiers
		max = 10
	}
	ai_frequency = 60
	ai_will_do = {
		base = 0
		modifier = {
			add = 50
			scope:recipient = {
				is_knight_of = scope:actor
			}
		}
	}
}

POD_thinbloodalchemy_sell_alchemical_product_interaction = {
	category = interaction_category_vampire

	target_type = artifact
	target_filter = actor_artifacts
	
	can_be_picked_artifact = {
		scope:target = { has_variable = alchemy }
	}
	
	is_shown = {
		NOT = {scope:actor = scope:recipient }
		scope:actor = { has_thinbloodalchemy_trigger = yes }
	}

	is_valid = {
		scope:recipient = { can_use_alchemical_product_trigger = yes }
	}

	is_valid_showing_failures_only = {
		scope:recipient = { is_ruler = yes }
		scope:recipient = { is_available = yes }
		scope:recipient = { debt_level < 0 }
		scope:recipient = {
			NOT = { has_trait = torpor }
			NOT = { has_trait = automaton }
			NOT = { has_trait = wight }
			NOT = { has_trait = barghest }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }	
			POD_is_not_clone_trigger = yes
		}
		POD_cross_umbra_interaction_valid_trigger = yes
		scope:recipient = { 
			custom_description = {
				text = "demand_artifact_interaction_valid_tt_1"
				has_any_artifact = yes
				any_character_artifact = { has_variable = alchemy }
			}
		}
	}
	
	on_accept = {
		scope:target = { 
			POD_alchemy_sell_effect = { CHARACTER1 = scope:actor CHARACTER2 = scope:recipient }
			POD_alchemy_usage_effect = { CHARACTER = scope:recipient }
		}
	}
	
	ai_frequency = 48
	
	ai_min_reply_days = 0
	ai_max_reply_days = 0

	ai_potential = {
		# The basic requirements
		is_imprisoned = no 
		is_landed = yes 
		
		is_at_war = no
	}
	
	ai_targets = {
		ai_recipients = neighboring_rulers
		max = 10
	}
	
	ai_accept = {
		base = -20 # Try to make it 0 for most interactions
		
		opinion_modifier = { # Opinion Factor
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.25
			desc = AI_OPINION_REASON
		}
		
		modifier = {
			add = 25
			desc = GOLD_REASON
		}
		
		modifier = {
			desc = VAMPIRE_IS_SHUNNED
			add = -1000
			scope:target ?= { var:alchemy = flag:vampire_concoction }
			scope:recipient.faith = { has_doctrine_parameter = vampirism_shunned }
		}
		
		modifier = {
			desc = VAMPIRE_IS_CRIME
			add = -1000
			scope:target ?= { var:alchemy = flag:vampire_concoction }
			scope:recipient.faith = { has_doctrine_parameter = vampirism_illegal }
		}
	}
	
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 20
			scope:actor = { gold < medium_gold_value }
		}
		
		modifier = {
			add = 40
			scope:actor = { gold < minor_gold_value }
		}
		
		modifier = {
			add = 80
			scope:actor = { gold < 0 }
		}
		
		modifier = {
			factor = 0
			scope:recipient = { is_ai = no }
		}
		
		modifier = {
			factor = 0
			scope:recipient = { has_trait = mummy }
		}
	}
}
