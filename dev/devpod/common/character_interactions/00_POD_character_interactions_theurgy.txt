
POD_theurgy_heal_interaction = {
	category = interaction_category_holyarts
	icon = POD_theurgy_interaction
	desc = POD_theurgy_heal_interaction_desc
	options_heading = POD_theurgy_heal_interaction_options_heading
	
	cooldown = { months = 12 }
	cost = { piety = medium_piety_value }
	
	is_shown = {
		scope:actor = { has_trait = theurgist }
		scope:actor = { has_perk = via_medicamenti_perk }
		scope:recipient = {
			OR = {
				AND = {
					has_trait = scarred
					NOT = { has_character_flag = POD_permanent_scarring }
				}
				has_trait = wounded_1
				has_trait = wounded_2
				has_trait = wounded_3
				has_trait = ill
				has_trait = great_pox
				has_trait = gout_ridden
				has_trait = consumption
				has_trait = cancer
				has_trait = pneumonic
				has_trait = smallpox
				has_trait = typhus
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			NOT = { has_trait_with_flag = can_not_cure }
			NOT = { has_character_modifier = eye_of_sakhmet_modifier }
		}
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = scarred
				NOT = { has_character_flag = POD_permanent_scarring }
			}
		}
		flag = scarred
		localization = POD_yangchi_heal_scarred
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				OR = {
					has_trait = wounded_1
					has_trait = wounded_2
					has_trait = wounded_3
				}
			}
		}
		flag = wounded
		localization = POD_yangchi_heal_wounded
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = ill
			}
		}
		flag = ill
		localization = POD_yangchi_heal_ill
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				OR = {
					has_trait = pneumonic
					has_trait = gout_ridden
					has_trait = typhus
					has_trait = consumption
					has_trait = cancer
					has_trait = great_pox
					has_trait = smallpox
				}
			}
		}
		is_valid = {
			trigger_if = {
				limit = { scope:actor = scope:recipient }
				scope:actor = { has_perk = biocontrol_perk }
			}
			trigger_else = {
				scope:actor = { has_perk = psychic_healing_perk }
			}
		}
		flag = disease
		localization = POD_yangchi_heal_disease
	}
	
	on_accept = {
		scope:actor = { 
			POD_willpower_rouse_check_effect = yes
		}
		scope:recipient = {
			play_music_cue = "mx_cue_peace_ensues"

			if = {
				limit = {
					scope:scarred = yes
				}
				scope:recipient = {
					remove_trait = scarred
				}
			}	
			
			if = {
				limit = {
					scope:wounded = yes
				}
				scope:recipient = {
					remove_trait = wounded_3
					remove_trait = wounded_2
					remove_trait = wounded_1
				}
			}
			if = {
				limit = {
					scope:ill = yes
				}
				scope:recipient = {		
					recover_from_disease_effect = { DISEASE = ill }
					remove_disease_treatment_effect = yes
				}
			}	

			if = {
				limit = {
					scope:disease = yes
				}
				scope:recipient = {		
					recover_from_disease_effect = { DISEASE = pneumonic }
					recover_from_disease_effect = { DISEASE = gout_ridden }
					recover_from_disease_effect = { DISEASE = typhus }
					recover_from_disease_effect = { DISEASE = consumption }
					recover_from_disease_effect = { DISEASE = cancer }
					recover_from_disease_effect = { DISEASE = great_pox }
					recover_from_disease_effect = { DISEASE = smallpox }
					remove_disease_treatment_effect = yes
				}
			}
		}
	}
	
	auto_accept = yes

	ai_potential = {
		is_imprisoned = no 
		piety > major_piety_value
	}
	ai_targets = {
		ai_recipients = self
	}
	ai_targets = {
		ai_recipients = courtiers
		max = 10
	}
	ai_frequency = 12
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 100
			scope:recipient = {
				OR = {
					this = scope:actor
					is_knight_of = scope:actor
				}
			}
		}
	}
}


POD_train_theurgy_interaction = {
	category = interaction_category_holyarts
	icon = POD_theurgy_interaction
    desc = POD_train_theurgy_interaction_desc
	common_interaction = yes
	
	cooldown = { months = 12 }
	cost = { piety = medium_piety_value }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_trait = theurgist }
		scope:recipient = { has_trait = theurgist }
	}

	is_valid_showing_failures_only = {
		custom_description = {
            object = scope:actor
            text = "I_must_be_at_the_same_location"
			scope:actor = { exists = location }
			scope:recipient = { exists = location }
            scope:actor = { location = scope:recipient.location }
        }
	    scope:actor = {
			sorcery_lifestyle_perk_points >= 1
		}
		scope:recipient = { is_imprisoned = no }
	}
	
	on_accept = {
	    scope:actor = { 
			add_sorcery_lifestyle_perk_points = -1
			stress_impact = {
				base = medium_stress_gain
			}
		}			
	    scope:recipient = { 
			add_sorcery_lifestyle_perk_points = 1 
			add_opinion = {
				target = scope:actor
				modifier = mentored_me_opinion
			}
		}	
	}

	ai_frequency = 0
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_initiate_theurgy_interaction = {
	category = interaction_category_holyarts
	icon = POD_theurgy_interaction
    desc = POD_initiate_theurgy_interaction_desc
	
	cooldown = { months = 12 }
	cost = { piety = major_piety_value }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_trait = theurgist }
		scope:recipient = { NOT = { has_trait = theurgist } }
		scope:recipient = { has_supernatural_race_trigger = no }
	}

	is_valid_showing_failures_only = {
		custom_description = {
            object = scope:actor
            text = "I_must_be_at_the_same_location"
			scope:actor = { exists = location }
			scope:recipient = { exists = location }
            scope:actor = { location = scope:recipient.location }
        }
	    scope:actor = {
			sorcery_lifestyle_perk_points >= 1
		}
		scope:recipient = { is_imprisoned = no }
	}
	
	on_accept = {
	    scope:actor = { 
			add_sorcery_lifestyle_perk_points = -1
			stress_impact = {
				base = major_stress_gain
			}
			duel = {
				skill = learning
				value = 10
				50 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					scope:recipient = {
						add_trait = theurgist
						add_opinion = {
							target = scope:actor
							modifier = mentored_me_opinion
						}
					}
				}
				50 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier 
					}
				}
			}
		}	
	}

	ai_frequency = 0
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
