POD_ghoul_choose = {
	category = interaction_category_vampire
	common_interaction = yes
	#interface_priority = 89
	desc = POD_ghoul_choose_desc
	ai_frequency = 0
	icon = POD_ghoulify_interaction
	
	is_shown = {
		scope:actor = { has_POD_vamp_blood_trigger = yes }
		scope:actor = scope:recipient
		scope:actor = {
			num_of_relation_ghoul_thrall > 0
		}
	}
	
	populate_recipient_list = {
	    scope:actor = {
			every_relation = {
				type = ghoul_thrall
				limit = { is_alive = yes }
				
				add_to_list = characters
			}
		}
	}
	
		
	is_valid_showing_failures_only = {
			
	}
	
	send_option = {
		is_shown = {
			exists = scope:secondary_recipient
			scope:actor = {
				POD_has_enough_blood_for_discipline = { AMOUNT = 1 }
			}
			scope:secondary_recipient = {
				OR = {
					POD_health_has_normal_wound_trigger = yes
					has_trait = maimed
				}
			}
		}	
		flag = heal
		localization = POD_ghoulthrall_heal_option
	}	
	
	send_option = {
		is_shown = {
			exists = scope:secondary_recipient
			scope:actor = {
				OR = {
					choleric_lifestyle_xp > 999
					choleric_lifestyle_perk_points > 0
					phlegmatic_lifestyle_xp > 999
					phlegmatic_lifestyle_perk_points > 0
					sanguine_lifestyle_xp > 999
					sanguine_lifestyle_perk_points > 0
					melancholy_lifestyle_xp > 999
					melancholy_lifestyle_perk_points > 0
					animal_lifestyle_xp > 999
					animal_lifestyle_perk_points > 0
					has_perk = cycle_of_the_midgard_serpent_perk
				}
			}
		}	
		flag = traindiscipline
		localization = POD_ghoulthrall_traindiscipline_option
	}	
	
	send_option = {
		is_shown = {
			exists = scope:secondary_recipient
		}	
		flag = giveblood
		localization = POD_ghoulthrall_giveblood_option
	}	

	send_option = {
		is_shown = {
			exists = scope:secondary_recipient
			scope:actor = {
				gold >= medium_gold_value
			}
		}	
		flag = rewardgold
		localization = POD_ghoulthrall_goldreward_option
	}	
	
	send_option = {
		is_shown = {
			exists = scope:secondary_recipient
			scope:secondary_recipient = {
				gold >= medium_gold_value
			}
			NOT = {
				scope:secondary_recipient = {
					has_opinion_modifier = {
						target = scope:actor
						modifier = POD_ghoulthrallgolddemand_opinion
					}		
				}
			}
		}	
		flag = demandgold
		localization = POD_ghoulthrall_demandgold_option
	}	
	
	send_option = {
		is_shown = {
			exists = scope:secondary_recipient
			scope:actor = {
				POD_has_enough_blood_for_discipline = { AMOUNT = 1 }
				OR = {
					has_perk = malleable_visage_perk
					has_perk = transmogrify_the_mortal_clay_perk
					has_perk = rend_the_osseous_frame_perk
					has_trait = vicissitudeadvanced
				}
			}
		}	
		flag = transmogrify
		localization = POD_ghoulthrall_transmogrify_option
	}	
	
	send_option = {
		is_shown = {
			scope:secondary_recipient ?= {
				is_married = yes
			}
		}	
		flag = divorce
		localization = POD_ghoulthrall_divorce_option
	}
	
	send_option = {
		is_shown = {
			exists = scope:secondary_recipient
		}	
		flag = release
		localization = POD_ghoulthrall_release_option
	}	
	
	on_accept = {
		
		scope:secondary_recipient = {save_scope_as = ghoulthrall }

		if = {
			limit = {
				scope:traindiscipline = yes
			}
			scope:actor = {
				POD_hunger_rouse_check_effect = yes
			}		
			scope:secondary_recipient = {
				save_scope_as = disciplinetrainee
			}
			
			scope:actor = {
				save_scope_as = disciplinetrainer
			}
			
			scope:actor = {
				trigger_event = {
					id = POD_disciplinetrain.100
					days = 7
				}		
			}	
		}


		if = {
			limit = {
				scope:giveblood = yes
			}
			scope:actor = {
				POD_hunger_rouse_check_effect = yes
			}
			scope:actor = {
				hidden_effect = {
					remove_hook = {
						target = scope:secondary_recipient
						type = strong_bloodoath_hook
						}
					}
				add_hook = {
				target = scope:secondary_recipient
				type = strong_bloodoath_hook
				}
			}
			scope:secondary_recipient = {
				hidden_effect = {
					remove_opinion = {
						target = scope:actor
						modifier = POD_ghoulthrall_opinion
					}
				}
				add_opinion= {
					target = scope:actor
					modifier = POD_ghoulthrall_opinion
				}
			}	
		}

		if = {
			limit = {
				scope:rewardgold = yes
			}
			scope:actor = {
				pay_short_term_gold = {
					target = scope:secondary_recipient
					gold = scope:actor.medium_gold_value
				}
			}
			scope:secondary_recipient = {
				add_opinion= {
					target = scope:actor
					modifier = POD_ghoulthrallgoldreward_opinion
				}
			}
		}

		if = {
			limit = {
				scope:demandgold = yes
			}
			scope:secondary_recipient = {
				pay_short_term_gold = {
					target = scope:actor
					gold = 15
				}
				add_opinion= {
					target = scope:actor
					modifier = POD_ghoulthrallgolddemand_opinion
				}
			}
		}

		if = {
			limit = {
				scope:transmogrify = yes
			}
			custom_tooltip = POD_ghoulthrall_transmogrify_option.tt
			
			scope:secondary_recipient = {
				save_scope_as = transmogrifiee
			}
			
			scope:actor = {
				save_scope_as = transmogrifier
			}
			
			scope:actor = {
				trigger_event = POD_transmogrify.100
			}
		}

		if = {
			limit = {
				scope:divorce = yes
			}
			scope:secondary_recipient = {
				divorce = primary_spouse
			}
		}

		if = {
			limit = {
				scope:heal = yes
			}
		
			if = {
				limit = {
					scope:ghoulthrall = {
						has_trait = maimed
					}
				}
				scope:actor = {
					POD_small_risk_of_increasing_masquerade_exposure = yes
					POD_hunger_rouse_check_effect = yes
				}
			}
				
			if = {
				limit = {
					scope:ghoulthrall = {
						NOT = { has_trait = maimed }
					}
				}
				scope:actor = {
					POD_minimal_risk_of_increasing_masquerade_exposure = yes
					POD_hunger_rouse_check_effect = yes
				}
			}				
				
			scope:ghoulthrall = {
				remove_trait = maimed
				POD_blood_heal_wound_effect = yes
				hidden_effect = {
					remove_opinion = {
						target = scope:actor
						modifier = POD_ghoulthrall_opinion
					}
				}
				add_opinion= {
					target = scope:actor
					modifier = POD_ghoulthrall_opinion
				}
			}
		}

		if = {
			limit = {
				scope:release = yes
			}
			
			scope:actor = {
				custom_tooltip = POD_ghoulthrall_release_option.tt
			}
			
			scope:secondary_recipient = {
				save_scope_as = releasedghoul
			}
			
			scope:actor = {
				save_scope_as = formerdomitor
			}
			
			scope:releasedghoul = {
				hidden_effect = {
				    remove_relation_ghoul_domitor = scope:formerdomitor
				}
				remove_character_flag = blocked_from_leaving	
				remove_character_flag = blood_bonded				
			}
			
			scope:formerdomitor = {
				remove_hook = {
					target = scope:releasedghoul
					type = strong_bloodoath_hook
				}
				#add_hook = {
				#	target = scope:releasedghoul
				#	type = weak_bloodoath_hook
				#}
			}		
			
			scope:releasedghoul = {
				trigger_event = {
					id = POD_ghoul_maintenance.50
					days = 7
				}
				trigger_event = {
					id = POD_ghoul_maintenance.100
					days = 365
				}		
			}
		}
	}
	
	auto_accept = yes
}

POD_ghoulify_mortal_interaction = {
	category = interaction_category_vampire
	#common_interaction = yes
	#interface_priority = 30
	scheme = POD_ghoulify_mortal
	icon = POD_ghoulify_interaction
	
	ai_frequency = 0

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_ai = yes }
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		scope:actor = { POD_is_vampire_trigger = yes }
		scope:actor = { POD_is_thinblooded_trigger = no }
		scope:recipient = { POD_can_be_ghouled_scheme_trigger = yes }
		scope:recipient = { NOT = { has_relation_ghoul_domitor = scope:actor } }
		scope:recipient = { is_imprisoned = no }
	}

	is_valid_showing_failures_only= {
		scope:actor = {
			can_start_scheme = {
				type = POD_ghoulify_mortal
				target_character = scope:recipient
			}
		}
		custom_description = {
			object = scope:recipient
			text = "blood_is_holy"
			scope:recipient = {
				NOR = {
					POD_has_true_faith_trigger = yes
					has_trait = orison_sanguine
				}
			}
		}
		
		custom_description = {
			text = POD_ghoulify_mortal_interaction_bp_one
			scope:actor = { has_blood_potency_greater_than_zero_trigger = yes }	
		}
		
		scope:actor = { POD_has_enough_blood_for_discipline = { AMOUNT = 1 } }
		scope:recipient = {
			is_ruler = no
		}
	}
	
	is_valid = {		
	}

	desc = POD_ghoulify_mortal_interaction_desc

	on_accept = {
	
		hidden_effect = {
			scope:actor = {
				start_scheme = {
					type = POD_ghoulify_mortal
					target_character = scope:recipient
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_ghoul_ghoulify_interaction = {
	category = interaction_category_ghoul
	common_interaction = yes
	#interface_priority = 30
	scheme = POD_ghoul_ghoulify
	icon = POD_ghoulify_interaction
	
	ai_frequency = 0

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		scope:actor = { POD_is_any_kind_of_ghoul_trigger = yes }
		scope:recipient = { POD_can_be_ghouled_scheme_trigger = yes }
		scope:recipient = { is_ai = yes }
		scope:recipient = { POD_is_clone_trigger = no }		
		scope:recipient = { is_imprisoned = no }
	}

	is_valid_showing_failures_only= {
		custom_description = {
			object = scope:recipient
			text = "blood_is_holy"
			scope:recipient = {
				NOR = {
					POD_has_true_faith_trigger = yes
					has_trait = orison_sanguine
				}
			}
		}
		custom_description = {
			text = I_have_no_bloodbag
			scope:actor = {
			any_prisoner = {
				POD_is_vampire_trigger = yes
				}
			}
		}
		scope:recipient = {
			is_ruler = no
		}
		scope:actor = {
			trigger_if = {
				limit = { POD_is_supehunter_trigger = no }
				scope:recipient = {
					POD_is_supehunter_trigger = no
				}
			}
		}
	}

	desc = POD_ghoul_ghoulify_interaction_desc
	
	populate_recipient_list = {
		scope:actor = {
			every_prisoner = {
				limit = {
					POD_is_vampire_trigger = yes
				}
				add_to_list = characters
			}
		}
	}
	
	on_accept = {
	
	scope:actor = {
		set_variable = {
			name = bloodbag 
			value = scope:secondary_recipient
			}
		}
	hidden_effect = {
			scope:actor = {
				start_scheme = {
					type = POD_ghoul_ghoulify
					target_character = scope:recipient
				}
			}
		}
	}
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

		scope:recipient = {is_imprisoned_by = scope:actor}
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_ai = yes }
		scope:actor = { POD_is_vampire_trigger = yes }
		scope:actor = { POD_is_thinblooded_trigger = no }
		scope:recipient = { POD_can_be_ghouled_scheme_trigger = yes }
		scope:recipient = { NOT = { has_relation_ghoul_domitor = scope:actor } }
	}

	is_valid_showing_failures_only = {
		custom_description = {
			object = scope:recipient
			text = "blood_is_holy"
			scope:recipient = {
				NOR = {
					POD_has_true_faith_trigger = yes
					has_trait = orison_sanguine
				}
			}
		}
		
		custom_description = {
			text = POD_ghoulify_mortal_interaction_bp_one
			scope:actor = { has_blood_potency_greater_than_zero_trigger = yes }	
		}
		
		scope:actor = { POD_has_enough_blood_for_discipline = { AMOUNT = 1 } }
		scope:recipient = {
			is_ruler = no
		}

		custom_description = {
			object = scope:recipient
			text = POD_ghoulify_mortal_interaction_limit
			scope:actor = {
				num_of_relation_ghoul_thrall < max_ghoul_cap
			}	
		}
	}


	on_accept = {
	
		custom_tooltip = POD_ghoulify_prisoner_interaction.tt
		
		scope:recipient = {
			save_scope_as = ghoulthrall
		}
		scope:actor = {
			save_scope_as = domitor
			
			POD_player_character_create_ghoul_effect = yes
			
			scope:recipient = {
				add_character_flag = blocked_from_leaving	
				add_character_flag = blood_bonded 
				set_relation_ghoul_domitor = scope:actor
			}
			
			add_hook = {
				target = scope:recipient
				type = strong_bloodoath_hook
			}
			add_courtier = scope:recipient
			POD_hunger_rouse_check_effect = yes
			POD_high_risk_of_increasing_masquerade_exposure = yes		
	
			POD_give_regnant_secret_effect = yes
			
			add_prestige = -100
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
