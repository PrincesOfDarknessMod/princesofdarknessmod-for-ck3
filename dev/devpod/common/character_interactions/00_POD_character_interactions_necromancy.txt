    		}

    }
}

POD_soul_feasting_character_interaction = {
	category = interaction_category_vampire
	icon = POD_wraith_interaction
	desc = POD_soul_feasting_character_interaction_desc
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { is_ai = no }
		scope:actor = { has_POD_vamp_blood_trigger = yes }
		scope:actor = { has_perk = soul_feasting_perk }
		scope:recipient = { has_trait = wraith }
		POD_is_in_same_plane_trigger = yes
	}

	is_valid_showing_failures_only = {
		custom_description = {
            object = scope:actor
            text = "I_must_be_at_the_same_location"
			scope:actor = { exists = location }
			scope:recipient = { exists = location }
            scope:actor = { location = scope:recipient.location }
        }
		custom_description = {
			text = POD_not_in_possession
			scope:recipient = { num_of_relation_possessed_target = 0 }
		}
		scope:recipient = { has_relation_necromancer = scope:actor }
		scope:actor = { NOT = { has_trait = methuselahs_thirst } }
	}
	
	send_option = {
		flag = sip
		localization = POD_drink_herd_sip_option
	}
	
	send_option = {
		flag = drink
		localization = POD_drink_herd_drink_option
	}
	
	send_option = {
		flag = alot
		localization = POD_drink_herd_alot_option
	}
	
	send_option = {
		flag = drain
		localization = POD_drink_herd_drain_option
	}

	on_accept = {
		if = {
			limit = {scope:sip = yes}
			scope:recipient = {
				POD_pathos_decrease_multiple_effect = { COUNT = 1 }
			}
			
			scope:actor = {POD_hunger_feeding_sip_effect = yes}
		}
		if = {
			limit = {scope:drink = yes}
			scope:recipient = {
				POD_pathos_decrease_multiple_effect = { COUNT = 2 }
			}
			
			scope:actor = {POD_hunger_feeding_drink_effect = yes}
		}
		if = {
			limit = {scope:alot = yes}
			scope:recipient = {
				POD_pathos_decrease_multiple_effect = { COUNT = 3 }
			}
			scope:actor = {POD_hunger_feeding_lot_effect = yes}
		}
		if = {
			limit = {scope:drain = yes}
			scope:recipient = {
				POD_pathos_decrease_multiple_effect = { COUNT = 4 }
			}
			scope:actor = {POD_hunger_feeding_drain_effect = yes}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
