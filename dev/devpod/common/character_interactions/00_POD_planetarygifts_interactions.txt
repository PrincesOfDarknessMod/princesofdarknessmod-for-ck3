
POD_ebon_binding_interaction = {
	category = interaction_category_werewolf
	desc = POD_ebon_binding_interaction_desc
	icon = POD_planetarygifts_interaction
	
	cooldown = { years = 1 }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { POD_splat_has_gnosis_trigger = yes }
		scope:actor = { has_perk = planetarygifts_ebon_binding_perk }
	}

	is_valid_showing_failures_only = {
		scope:actor = { 
			NOT = { has_character_modifier = POD_gnosis_0_modifier }
		}
		scope:recipient = { 
			any_secret = {
				is_known_by = scope:actor
			}
			NOT = { has_character_modifier = ebon_binding_modifier }
		}
	}
	on_accept = {
		scope:actor = {
			POD_gnosis_loss_check_effect = yes	
		}
		scope:recipient = {
			add_character_modifier  = { 
				modifier = ebon_binding_modifier
				years = 10
			}
		}
	}
	
	auto_accept = yes

	ai_frequency = 0
	
	ai_will_do = {
		base = 0
	}
}

	cooldown = { months = 3 }	

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { POD_splat_has_gnosis_trigger = yes }
		scope:actor = { has_perk = earth_heal_perk }
		POD_is_in_same_plane_trigger = yes

			exists = location 
			location = { has_building_or_higher = wyrm_taint_01 }

				}
			}
		}
		if = {
			limit = { scope:taint_land = { has_building = wyrm_taint_04 } }
			scope:actor = { 
				POD_gnosis_loss_check_effect = yes 
				POD_gnosis_loss_check_effect = yes 
				POD_gnosis_loss_check_effect = yes 
				POD_gnosis_loss_check_effect = yes

					value = extremely_high_skill_rating

						compare_modifier = {
							value = scope:duel_value
							multiplier = high_positive_duel_skill_multiplier
						}
						add_prestige = massive_prestige_gain
						scope:taint_land = { remove_building = wyrm_taint_04 }
					}
					50 = {
						compare_modifier = {
							value = scope:duel_value
							multiplier = low_negative_duel_skill_multiplier

					}
				}
			}
			
		}
		else_if = {
			limit = { scope:taint_land = { has_building = wyrm_taint_03 } }

							value = scope:duel_value

						}
					}
				}
			}
		}
		else_if = {
			limit = { scope:taint_land = { has_building = wyrm_taint_02 } }
			scope:actor = { 
				POD_gnosis_loss_check_effect = yes 
				POD_gnosis_loss_check_effect = yes

							value = scope:duel_value
