POD_unseas_blessing_interaction = {
	category = interaction_category_werewolf
	icon = POD_rokea_interaction
	desc = POD_unseas_blessing_interaction_desc
	
	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { POD_splat_has_rage_trigger = yes }
		scope:actor = { has_perk = unseas_blessing_perk }
	}
	
	is_valid = {
		scope:recipient = {
			OR = {
				has_trait = scarred
				has_trait = disfigured

			}
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			POD_can_be_healed_trigger = yes
		}
	}
	
	on_accept = {
		if = {
			limit = { NOT = { scope:actor = scope:recipient } }
			scope:actor = { POD_commit_heal_effect = yes }
		}
		scope:actor = { 
			POD_willpower_rouse_check_effect = yes
		}
		scope:recipient = {
			play_music_cue = "mx_cue_peace_ensues"
			
			remove_scar_trait_effect = yes
			POD_remove_wound_trait_effect = yes
		}
	}
	
	auto_accept = yes

	ai_potential = {
		# The basic requirements
		is_imprisoned = no
		is_landed = yes
	}
	ai_targets = {
		ai_recipients = self
	}
	ai_frequency = 12
	ai_will_do = {
		base = 100
	}
}

POD_common_foe_interaction = {
	category = interaction_category_werewolf
	common_interaction = yes
	desc = POD_common_foe_interaction_desc
	interface_priority = 60
	interface = call_ally
	special_interaction = call_ally_interaction
	popup_on_receive = yes
	pause_on_receive = yes
	icon = POD_rokea_interaction

	greeting = positive
	notification_text = CALL_ALLY_NOTIFICATION

	is_shown = {
		scope:actor = { is_at_war = yes }
		scope:actor = { has_perk = common_foe_perk }
		scope:actor = { POD_splat_has_gnosis_trigger = yes }
	}
	
	has_valid_target = {
		exists = scope:target
	}	

	has_valid_target_showing_failures_only = {
		scope:target = {
			is_war_leader = scope:actor
		}

		joiner_not_already_in_another_war_with_any_target_war_participants_trigger = {
			WARRIOR = scope:actor
			JOINER = scope:recipient
		}
	}
	
	is_valid = {
		scope:recipient = { POD_is_wyrm_servant_trigger = no }
		scope:target = {
			trigger_if = {
				limit = { casus_belli.primary_attacker = scope:actor }
				casus_belli.primary_defender = { POD_is_wyrm_servant_trigger = yes }
			}
			trigger_if = {
				limit = { casus_belli.primary_defender = scope:actor }
				casus_belli.primary_attacker = { POD_is_wyrm_servant_trigger = yes }
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = { NOT = { has_character_modifier = POD_gnosis_0_modifier } }
		
		scope:recipient = {
			is_ruler = yes
		}

		# Only War Leaders can call allies
		scope:actor = {
			custom_description = {
				text = must_be_war_leader
				any_character_war = {
					is_war_leader = scope:actor
				}
			}
		}

		# Cannot call the same character multiple times
		scope:actor = {
			custom_description = {
				text = cannot_call_same_character_multiple_times
				subject = scope:recipient
				any_character_war = {
					ALL_FALSE = {
						was_called = scope:recipient
						is_participant = scope:recipient
					}
				}
			}
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}

	can_be_picked = {
		can_join_war_liege_vassal_check_trigger = { WARRIOR = scope:actor JOINER = scope:recipient }

		scope:target = {
			is_war_leader = scope:actor

			using_cb = POD_blood_hunt_war
			
			# not already in target war
			custom_description = {
				text = join_war_interaction_already_in_target_war
				NOR = {
					any_war_attacker = { this = scope:recipient }
					any_war_defender = { this = scope:recipient }
				}
			}
		}
	}

	on_auto_accept = {
		scope:actor = {
			POD_gnosis_loss_check_effect = yes
		}
		scope:target = {
			if = {
				limit = {
					scope:recipient = {
						any_active_accolade = {
							has_accolade_parameter = acclaimed_knight_piety_from_battle
						}
					}
				}
				scope:recipient = {
					add_piety = miniscule_piety_gain
				}
			}
			if = {
				limit = {
					scope:recipient = {
						any_active_accolade = {
							has_accolade_parameter = acclaimed_knight_piety_from_battle_high
						}
					}
				}
				scope:recipient = {
					add_piety = minor_piety_gain
				}
			}
		}
	}
		
	on_accept = {
		blood_hunt_call_interaction_effect = yes

		scope:actor = {
			POD_gnosis_loss_check_effect = yes
		}
		scope:target = {
			if = {
				limit = {
					scope:recipient = {
						any_active_accolade = {
							has_accolade_parameter = acclaimed_knight_piety_from_battle
						}
					}
				}
				scope:recipient = {
					add_piety = minor_piety_gain
				}
			}
			if = {
				limit = {
					scope:recipient = {
						any_active_accolade = {
							has_accolade_parameter = acclaimed_knight_piety_from_battle_high
						}
					}
				}
				scope:recipient = {
					add_piety = medium_piety_gain
				}
			}
		}
	}
	
	auto_accept = yes
}
