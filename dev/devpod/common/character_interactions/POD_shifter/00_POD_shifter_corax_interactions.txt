POD_gift_of_eyes_interaction = {
	category = interaction_category_fera
	icon = POD_corax_interaction
	desc = POD_gift_of_eyes_interaction_desc
	options_heading = POD_gift_of_eyes_interaction_options_heading
	
	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { POD_splat_has_gnosis_trigger = yes }
		scope:actor = { has_perk = gift_of_eyes_perk }
	}

	is_valid_showing_failures_only = {
		scope:actor = { POD_has_enough_gnosis_for_gift = { COUNT = 1 } }
	}
	
	send_option = {
		flag = family
		localization = POD_sorcery_auto_resurrection_interaction_family
	}
	
	send_option = {
		flag = consort
		localization = POD_sorcery_auto_resurrection_interaction_consort
	}
	
	send_options_exclusive = no
	
	on_accept = {
		scope:actor = { 
			POD_willpower_rouse_check_effect = yes
			
			if = {
				limit = { scope:family = yes }
				house = {
					every_house_member = {
						even_if_dead = yes
						limit = { is_alive = no }
						set_killer_public = yes
					}
				}
			}
			if = {
				limit = { scope:consort = yes }
				every_consort = {
					even_if_dead = yes
					limit = { is_alive = no }
					set_killer_public = yes
				}
			}
		}
	}

	ai_frequency = 0
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
