POD_hathors_blessing_interaction = {
	category = interaction_category_fera
	icon = POD_apis_interaction
	desc = POD_hathors_blessing_interaction_desc
	
	is_shown = {
		scope:actor = { POD_splat_has_gnosis_trigger = yes }
		scope:actor = { has_perk = hathors_blessing_perk }
		scope:recipient = { NOT = { has_character_modifier = hathors_blessing_perk_modifier } }
	}
	
	is_valid_showing_failures_only = {	
		scope:actor = { 
			POD_has_enough_gnosis_for_gift = { COUNT = 3 }
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			NOT = { has_trait = automaton }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }	
			POD_is_not_clone_trigger = yes	
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}

	on_accept = {
		scope:actor = {
			POD_gift_cost_effect = { COUNT = 3 }
		}
		scope:recipient = {
			add_character_modifier = {
				modifier = hathors_blessing_perk_modifier
				years = 10
			}
		}
	}
	
	ai_min_reply_days = 0
	ai_max_reply_days = 0
	
	ai_accept = {
		base = -50
		
		opinion_modifier = { # Opinion Factor
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 1.0
			desc = AI_OPINION_REASON
		}
	}
}
