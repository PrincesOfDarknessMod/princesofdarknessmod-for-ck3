
POD_dance_of_abandon_interaction = {
	category = interaction_category_werewolf
	icon = POD_nuwisha_interaction
	desc = POD_dance_of_abandon_interaction_desc

	cooldown = { months = 6 }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { POD_splat_has_gnosis_trigger = yes }
		scope:actor = { has_perk = dance_of_abandon_perk }
	}	

	is_valid_showing_failures_only = {	
		scope:actor = { POD_has_enough_gnosis_for_gift = { COUNT = 1 } }
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}

	on_accept = {
		scope:actor = { 
			POD_willpower_rouse_check_effect = yes
			POD_gnosis_loss_check_effect = yes
		}
		scope:recipient = {
			every_scheme = {
				end_scheme = yes
			}
			random = {
				chance = 5
				every_character_war = {
					end_war = white_peace
				}
			}
		}
	}
	
	auto_accept = yes
}

POD_tricksters_skin_interaction = {
	category = interaction_category_werewolf
	desc = POD_tricksters_skin_interaction_desc
	icon = POD_nuwisha_interaction
	
	is_shown = {
		OR = {
			NOT = { scope:actor = scope:recipient }
			scope:actor = { has_variable = original_face }
		}
		scope:actor = { POD_splat_has_gnosis_trigger = yes }
		scope:actor = { has_perk = tricksters_skin_perk }
	}

	is_valid_showing_failures_only = {
		scope:actor = { POD_has_enough_gnosis_for_gift = { COUNT = 1 } }
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}
	
	send_option = {
		is_shown = {
			scope:actor = { has_variable = original_face }
		}
		is_valid = {
			scope:actor = scope:recipient
		}
		flag = restore_face
		localization = POD_vampshapeshift_interaction_restore_face
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { has_variable = original_face }
			NOT = { scope:actor = scope:recipient }
		}
		flag = restore_face_recipient
		localization = POD_vampshapeshift_interaction_restore_face_recipient
	}
	
	send_option = {
		is_shown = {
			NOT = { scope:actor = scope:recipient }
		}
		flag = copy_face
		localization = POD_vampshapeshift_interaction_copy_face
	}
	
	send_option = {
		is_shown = {
			NOT = { scope:actor = scope:recipient }
		}
		flag = paste_face
		localization = POD_vampshapeshift_interaction_paste_face
	}

	on_accept = {
		scope:actor = {
			POD_gnosis_loss_check_effect = yes
		}
		if = {
			limit = {
				scope:copy_face = yes
			}
			scope:actor = {
				hidden_effect = { 
					if = {
						limit = { NOT = { has_variable = original_face } }
						POD_create_dummyclone_effect = { CHARACTER = scope:actor }
						if = {
							limit = { exists = scope:dummyclone }
							scope:dummyclone = { copy_inheritable_appearance_from = scope:actor }
							set_variable = {
								name = original_face
								value = scope:dummyclone
							}
						}
						else = {
							send_interface_toast = {
								title = POD_kueijin_spy.duel.failure
								left_icon = scope:actor
								right_icon = scope:recipient
							}
						}
					}
				}
				copy_inheritable_appearance_from = scope:recipient 
			}
		}
		if = {
			limit = {
				scope:paste_face = yes
			}
			scope:recipient = { 
				copy_inheritable_appearance_from = scope:actor 
				hidden_effect = { 
					if = {
						limit = { NOT = { has_variable = original_face } }
						create_character = {
							age = scope:recipient.effective_age
							faith = scope:recipient.faith
							gender = scope:recipient
							random_traits = no
							dynasty = none
							employer = character:POD_wraith_01
							trait = wraith
							save_scope_as = new_created_character
						}
						set_variable = {
							name = original_face
							value = scope:new_created_character
						}
					}
				}
				add_opinion = {
					target = scope:actor
					modifier = hurt_opinion
					opinion = -10
				}
			}
		}
		if = {
			limit = {
				scope:restore_face = yes
			}
			scope:actor = { 
				copy_inheritable_appearance_from = var:original_face 
				remove_variable = original_face
			}
		}
		if = {
			limit = {
				scope:restore_face_recipient = yes
			}
			scope:recipient = { 
				copy_inheritable_appearance_from = var:original_face 
				remove_variable = original_face
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

								modifier = trapped_in_abyss_modifier

							limit = { scope:arcadia = yes }
							add_character_modifier= {
								modifier = trapped_in_arcadia_modifier
								years = 10
							}
							POD_umbra_send_effect = { LOCATION = flag:arcadia }
						}
						if = { 
							limit = { scope:yomi = yes }

								modifier = trapped_in_yomi_modifier
								years = 10
							}

							add_character_modifier= {
								modifier = trapped_in_paradox_realm_modifier
								years = 10
							}
							POD_umbra_send_effect = { LOCATION = flag:paradox_realms }
						}
					}
				}

					desc = board_games.0001.tt.failure
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier 
					}
				}
			}
		}

			add_opinion = {
				target = scope:actor
				modifier = hurt_opinion
				opinion = -50
			}

				CHARACTER = scope:actor
				REASON = rival_cast_spell
				OPINION = 0
			}
		}
	}
	
	auto_accept = yes
}

POD_coyotes_howl_interaction = {
	category = interaction_category_werewolf
	icon = POD_nuwisha_interaction
	desc = POD_coyotes_howl_interaction_desc
	
	cooldown = { years = 1 }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { POD_splat_has_gnosis_trigger = yes }
		scope:actor = { has_perk = coyotes_howl_perk }
		scope:recipient = { is_playable_character = yes }
		scope:recipient.highest_held_title_tier > tier_barony
		scope:actor = { NOT = { target_is_liege_or_above = scope:recipient } }
	}
	
	is_valid_showing_failures_only = {
		scope:actor = { POD_has_enough_gnosis_for_gift = { COUNT = 1 } }
	}

	on_accept = {
		scope:actor = { 
			POD_willpower_rouse_check_effect = yes
			POD_gift_cost_effect = { COUNT = 5 }
		}
		scope:recipient = {
			every_knight = {
				add_opinion = {
					target = scope:actor
					modifier = hurt_opinion
					opinion = -25
				}
				if = {
					limit = {
						scope:actor = { 
							prowess_diff = {
								target = scope:recipient
								value >= 50
							}
						}
					}
					increase_wounds_multiple_hostile_effect = { REASON = howl AMOUNT = 4 ATTACKER = scope:actor }
				}
				else_if = {
					limit = {
						scope:actor = { 
							prowess_diff = {
								target = scope:recipient
								value >= 25
							}
						}
					}
					increase_wounds_multiple_hostile_effect = { REASON = howl AMOUNT = 3 ATTACKER = scope:actor }
				}
				else_if = {
					limit = {
						scope:actor = { 
							prowess_diff = {
								target = scope:recipient
								value >= 0
							}
						}
					}
					increase_wounds_multiple_hostile_effect = { REASON = howl AMOUNT = 2 ATTACKER = scope:actor }
				}
				else = {
					increase_wounds_multiple_hostile_effect = { REASON = howl AMOUNT = 1 ATTACKER = scope:actor }
				}
			}					
		}
    }
	
	auto_accept = yes
}
