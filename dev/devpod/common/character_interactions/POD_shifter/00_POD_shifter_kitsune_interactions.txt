
POD_possession_interaction = {
	category = interaction_category_werewolf
	icon = POD_kitsune_interaction
	desc = POD_possession_interaction_desc
	
	cooldown = { years = 1 }
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { POD_splat_has_gnosis_trigger = yes }
		scope:actor = { has_perk = possession_perk }
		scope:recipient = { 
			trigger_if = {
				limit = { num_of_relation_possessesing_wraith > 0 }
				has_relation_possessesing_wraith = scope:actor
			}
			NOT = {
				has_opinion_modifier = {
					target = scope:actor
					modifier = POD_complete_possession_opinion
				}
			}
		}
	}
	
	is_valid_showing_failures_only = {
		scope:actor = { POD_has_enough_gnosis_for_gift = { COUNT = 1 } }
		scope:actor = {
			custom_description = {
				text = POD_not_in_possession
				num_of_relation_possessed_target = 0 
			}
			trigger_if = {
				limit = {  
					scope:recipient = {
						POD_can_be_possessed_trigger = no
					}
				}
				has_perk = vector_perk
			}
		}
		scope:recipient = {
			POD_not_has_possess_ward_trigger = yes
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}
	
	on_accept = {
		scope:actor = { POD_wraith_willpower_check_effect = yes }
		scope:actor = { POD_gnosis_loss_check_effect = yes }
		scope:recipient = {
			trigger_event = POD_wraith_possession.100
		}
		custom_tooltip = POD_possess_interaction_desc
	}
	
	auto_accept = yes
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no 
		is_landed = yes 
	}
	ai_targets = {
		ai_recipients = liege
	}
	ai_targets = {
		ai_recipients = neighboring_rulers
		ai_recipients = peer_vassals
		max = 10
	}
	ai_frequency = 72
	ai_will_do = {
		base = 0
		
		opinion_modifier = {
			opinion_target = scope:recipient
			multiplier = -0.25
		}
		modifier = {
		    add = 50
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
		modifier = {
			factor = 0
			scope:recipient = {
				is_ai = no
			}
		}
	}
}

POD_end_possession_interaction = {
	category = interaction_category_werewolf
	icon = POD_kitsune_interaction
	desc = POD_end_possession_interaction_desc
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { POD_splat_has_gnosis_trigger = yes }
		scope:recipient = { has_relation_possessesing_wraith = scope:actor }
	}
	
	on_accept = {
		 POD_wraith_end_possession = {
			 WRAITH = scope:actor
			 PUPPET = scope:recipient
		 }
	
	}
	auto_accept = yes
}

POD_kitsune_madness_perk_interaction = {
	category = interaction_category_werewolf
	icon = POD_kitsune_interaction
	desc = POD_madness_perk_interaction_desc
	options_heading = POD_madness_perk_interaction_options_heading

	cooldown = { months = 3 }
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_perk = kitsune_madness_perk }
		scope:actor = { POD_splat_has_gnosis_trigger = yes }
	}

	is_valid_showing_failures_only = {
		custom_description = {
            object = scope:recipient
            text = "strong_mental_defenses_trigger"
            strong_mental_defenses_interaction_trigger = yes
        }
		scope:actor = { POD_has_enough_gnosis_for_gift = { COUNT = 1 } }
		scope:recipient = {
			NOT = { has_trait = torpor }
			NOT = { has_trait = automaton }
			NOT = { has_trait = wight }
			NOT = { has_trait = barghest }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }		
			POD_is_not_clone_trigger = yes
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				NOT = { has_trait = depressed }
			}
		}	
		flag = depressed
		localization = POD_exorcism_depressed
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				NOT = { has_trait = lunatic }
			}
		}	
		flag = lunatic
		localization = POD_exorcism_lunatic
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				NOT = { has_trait = possessed }
			}
		}	
		flag = possessed
		localization = POD_exorcism_possessed
	}
	
	on_accept = {
		scope:actor = { 
			POD_willpower_rouse_check_effect = yes
			POD_gnosis_loss_check_effect = yes	
			duel = {
				skill = learning
				target = scope:recipient
				70 = {
					desc = POD_kueijin_spy.duel.success
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					scope:recipient = {
						add_stress = medium_stress_gain

						if = {
							limit = {
								scope:depressed = yes
							}
							scope:recipient = {
								add_trait = depressed_1
							}
						}	

						if = {
							limit = {
								scope:lunatic = yes
							}
							scope:recipient = {
								add_trait = lunatic_1
							}
						}	

						if = {
							limit = {
								scope:possessed = yes
							}
							scope:recipient = {
								add_trait = possessed_1
							}
						}
					}
				}
				30 = {
					desc = POD_kueijin_spy.duel.failure
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier 
					}
					send_interface_toast = {
						title = POD_kueijin_spy.duel.failure
						left_icon = scope:actor
						right_icon = scope:recipient
						
						custom_tooltip = POD_kueijin_spy.duel.tt.attacker_lost
						add_stress = medium_stress_gain
					}
				}
			}
		}
		scope:recipient = { 
			add_opinion = {
				target = scope:actor
				modifier = hurt_opinion
				opinion = -30
			}
		}
	}
	
	auto_accept = yes
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no 
		is_landed = yes 
	}
	ai_targets = {
		ai_recipients = primary_war_enemies
	}
	ai_frequency = 48
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 45
			scope:actor = {
			    is_at_war_with = scope:recipient
			}
		}
		modifier = {
			factor = 2
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
		modifier = {
			factor = 0
			scope:recipient = {
				is_ai = no
			}
		}
	}
}
