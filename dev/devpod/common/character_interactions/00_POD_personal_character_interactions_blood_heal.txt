# POD Personal Character Interactions
#  Blood Heal

POD_blood_heal_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	#interface_priority= 110
	desc = POD_blood_heal_interaction_desc
	options_heading = POD_blood_heal_interaction_options_heading	

	use_diplomatic_range = no

	cooldown = { months = 3 }	
	
	is_shown = {
		scope:actor = scope:recipient 
		#scope:recipient = { is_adult = yes }
		scope:actor = { has_POD_vamp_blood_trigger = yes  }
		scope:actor = { 
			# NOT = { 
				# has_trait = holy_seal 
					# }
			}
		# is wounded
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			NOT = { has_character_modifier = bloodhunger5_modifier }
			NOT = { has_trait_with_flag = can_not_cure }
			NOT = { has_character_modifier = eye_of_sakhmet_modifier }
		}
		scope:recipient = {
			OR = {
				AND = {
					has_trait = scarred
					NOT = { has_character_flag = POD_permanent_scarring_flag }
				}
				AND = {
					has_trait = disfigured
					NOT = { has_character_flag = POD_permanent_disfigured_flag }
				}
				
				has_trait = wounded_1
				has_trait = wounded_2
				has_trait = wounded_3
				
				AND = {
					has_trait = blind
					NOT = { has_character_flag = POD_permanent_blind }
				}
				AND = {
					has_trait = maimed
					NOT = { has_character_flag = POD_permanent_maimed }
				}
				AND = {
					has_trait = one_legged
					NOT = { has_character_flag = POD_permanent_one_legged }
				}
				AND = {
					has_trait = one_eyed
					NOT = { has_character_flag = POD_permanent_one_eyed }
				}
				AND = {
					has_trait = eunuch
					NOT = { has_character_flag = POD_permanent_eunuch }
				}
			}
		}
	}

	send_option = {
		is_shown = {
			scope:actor = { 
				has_trait = blind
				POD_is_vampire_trigger = yes
				NOT = { has_character_flag = POD_permanent_blind }
			}
		}	
		flag = blind
		localization = POD_blood_heal_blind
		#current_description = POD_blood_heal_blind
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				has_trait = maimed
				POD_is_vampire_trigger = yes
				NOT = { has_character_flag = POD_permanent_maimed }
			}
		}	
		flag = maimed
		localization = POD_blood_heal_maimed
		#current_description = POD_blood_heal_maimed
	}	
	
	send_option = {
		is_shown = {
			scope:actor = { 
				has_trait = one_legged
				POD_is_vampire_trigger = yes
				NOT = { has_character_flag = POD_permanent_one_legged }
			}
		}	
		flag = one_legged
		localization = POD_blood_heal_one_legged
		#current_description = POD_blood_heal_one_legged
	}

	send_option = {
		is_shown = {
			scope:actor = { 
				has_trait = one_eyed
				POD_is_vampire_trigger = yes
				NOT = { has_character_flag = POD_permanent_one_eyed }
			}
		}	
		flag = one_eyed
		localization = POD_blood_heal_one_eyed
		#current_description = POD_blood_heal_one_eyed
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				OR = {
					has_trait = wounded_1
					has_trait = wounded_2
					has_trait = wounded_3
				}
			}
		}	
		flag = wounded
		localization = POD_blood_heal_wounded
		#current_description = POD_blood_heal_wounded
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = eunuch
				POD_is_vampire_trigger = yes
				NOT = { has_character_flag = POD_permanent_eunuch }
			}
		}
		flag = eunuch
		localization = POD_yangchi_heal_eunuch
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				has_trait = scarred
				NOT = { has_character_flag = POD_permanent_scarring_flag }
			}
		}	
		flag = scarred
		localization = POD_blood_heal_scarred
		#current_description = POD_blood_heal_scarred
	}
	
	send_option = {
		is_shown = {
			scope:actor = { 
				has_trait = disfigured
				NOT = { has_character_flag = POD_permanent_disfigured_flag }
			}
		}	
		flag = disfigured
		localization = POD_blood_heal_disfigured
		#current_description = POD_blood_heal_disfigured
	}
	
	on_accept = {
		scope:actor = {
			POD_hunger_rouse_check_effect = yes
			
			if = {
				limit = {
					scope:blind = yes
				}
				scope:actor = {
					remove_trait = blind
					add_trait = one_eyed
					POD_minimal_risk_of_increasing_masquerade_exposure = yes
				}
			}	

			if = {
				limit = {
					scope:maimed = yes
				}
				scope:actor = {
					remove_trait = maimed
					POD_small_risk_of_increasing_masquerade_exposure = yes
				}
			}		

			if = {
				limit = {
					scope:one_legged = yes
				}
				scope:actor = {
					remove_trait = one_legged
					POD_small_risk_of_increasing_masquerade_exposure = yes
				}
			}				

			if = {
				limit = {
					scope:disfigured = yes
				}
				scope:actor = {
					remove_trait = disfigured
					POD_small_risk_of_increasing_masquerade_exposure = yes
				}
			}

			if = {
				limit = {
					scope:eunuch = yes
				}
				scope:recipient = {
					remove_trait = eunuch
				}
				scope:actor = { POD_minimal_risk_of_increasing_masquerade_exposure = yes }
			}

			if = {
				limit = {
					scope:one_eyed = yes
					NOT = { has_character_flag = POD_permanent_one_eyed }
				}
				scope:actor = {
					remove_trait = one_eyed
					POD_minimal_risk_of_increasing_masquerade_exposure = yes
				}
			}
			
			if = {
				limit = {
					scope:wounded = yes
				}
				scope:actor = {
					POD_blood_heal_wound_effect = yes
					POD_minimal_risk_of_increasing_masquerade_exposure = yes
				}
			}

			if = {
				limit = {
					scope:scarred = yes
				}
				scope:actor = {
					remove_trait = scarred
					POD_minimal_risk_of_increasing_masquerade_exposure = yes
				}
			}				
		}
	}
	
	auto_accept = yes
	ai_targets = {
		ai_recipients = self
	}
	ai_frequency = 12
	ai_will_do = {
		base = 100
		
	}
}
