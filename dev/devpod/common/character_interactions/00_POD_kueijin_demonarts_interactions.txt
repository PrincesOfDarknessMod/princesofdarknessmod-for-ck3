POD_manifest_demon_weapon_interaction = {
	category = interaction_category_vampire
	icon = POD_art_interaction
	desc = POD_manifest_demon_weapon_interaction_desc
	
	cooldown = { days = 1000 }	

	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { has_character_modifier = demonic_feature_demon_weapon_modifier }
		scope:actor = { NOT = { any_character_artifact = { has_variable = tainted_relic } } }
	}

	is_valid_showing_failures_only = {
	}
	
	send_option = {
		flag = sword
		localization = artifact_sword
	}
	send_option = {
		flag = axe
		localization = artifact_axe
	}
	send_option = {
		flag = spear
		localization = artifact_spear
	}
	send_option = {
		flag = hammer
		localization = artifact_hammer
	}
	send_option = {
		flag = dagger
		localization = artifact_dagger
	}
	send_option = {
		flag = mace
		localization = artifact_mace
	}

	on_accept = {
		scope:actor = {
			POD_willpower_rouse_check_effect = yes
			POD_chi_decrease_pre_effect = { COUNT = 5 }
			soulforged_artifact_quality_tooltip_effect = yes
			
			scope:actor = { save_scope_as = owner }
			scope:actor = { save_scope_as = creator }
			scope:recipient = { save_scope_as = killed }
			if = {
				limit = { scope:sword = yes }
				show_as_tooltip = {
					create_artifact = {
						name = artifact_demon_weapon_name
						description = artifact_demon_weapon_description
						type = sword
						visuals = sword
						modifier = artifact_placeholder_modifier
					}
				}
				hidden_effect = { create_soulforged_artifact_effect = { VISUALS = sword TYPE = sword } }
			}
			if = {
				limit = { scope:axe = yes }
				show_as_tooltip = {
					create_artifact = {
						name = artifact_demon_weapon_name
						description = artifact_demon_weapon_description
						type = axe
						visuals = axe
						modifier = artifact_placeholder_modifier
					}
				}
				hidden_effect = { create_soulforged_artifact_effect = { VISUALS = axe TYPE = axe } }
			}
			if = {
				limit = { scope:spear = yes }
				show_as_tooltip = {
					create_artifact = {
						name = artifact_demon_weapon_name
						description = artifact_demon_weapon_description
						type = spear
						visuals = spear
						modifier = artifact_placeholder_modifier
					}
				}
				hidden_effect = { create_soulforged_artifact_effect = { VISUALS = spear TYPE = spear } }
			}
			if = {
				limit = { scope:hammer = yes }
				show_as_tooltip = {
					create_artifact = {
						name = artifact_demon_weapon_name
						description = artifact_demon_weapon_description
						type = hammer
						visuals = hammer
						modifier = artifact_placeholder_modifier
					}
				}
				hidden_effect = { create_soulforged_artifact_effect = { VISUALS = hammer TYPE = hammer } }
			}
			if = {
				limit = { scope:dagger = yes }
				show_as_tooltip = {
					create_artifact = {
						name = artifact_demon_weapon_name
						description = artifact_demon_weapon_description
						type = dagger
						visuals = dagger
						modifier = artifact_placeholder_modifier
					}
				}
				hidden_effect = { create_soulforged_artifact_effect = { VISUALS = dagger TYPE = dagger } }
			}
			if = {
				limit = { scope:mace = yes }
				show_as_tooltip = {
					create_artifact = {
						name = artifact_demon_weapon_name
						description = artifact_demon_weapon_description
						type = mace
						visuals = mace
						modifier = artifact_placeholder_modifier
					}
				}
				hidden_effect = { create_soulforged_artifact_effect = { VISUALS = mace TYPE = mace } }
			}
			hidden_effect = {
				scope:newly_created_artifact = {
					set_artifact_name = artifact_demon_weapon_name
					set_artifact_description = artifact_demon_weapon_description
					set_variable = tainted_relic
					set_max_durability = 5
					flag_as_trash_artifact = yes
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
