POD_manifest_demon_weapon_interaction = {
	category = interaction_category_vampire
	icon = POD_art_interaction
	desc = POD_manifest_demon_weapon_interaction_desc
	
	cooldown = { days = 1000 }	

	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { has_character_modifier = demonic_feature_demon_weapon_modifier }
		scope:actor = { NOT = { any_character_artifact = { has_variable = demon_weapon } } }
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			custom_tooltip = {
				text = "i_am_empty_of_chi"
				NAND = { 
					has_character_modifier = yinchi0_modifier
					has_character_modifier = yangchi0_modifier
				}
			}
		}
	}
	
	send_option = {
		flag = sword
		localization = artifact_sword
	}
	send_option = {
		flag = axe
		localization = artifact_axe
	}
	send_option = {
		flag = spear
		localization = artifact_spear
	}
	send_option = {
		flag = hammer
		localization = artifact_hammer
	}
	send_option = {
		flag = dagger
		localization = artifact_dagger
	}
	send_option = {
		flag = mace
		localization = artifact_mace
	}

	on_accept = {
		scope:actor = {
			POD_willpower_rouse_check_effect = yes
			POD_chi_decrease_multiple_effect = { COUNT = 5 }
			custom_tooltip = POD_soulforging_weapon_tooltip
			
			scope:actor = { save_scope_as = owner }
			scope:actor = { save_scope_as = creator }
			scope:recipient = { save_scope_as = killed }
			if = {
				limit = { scope:sword = yes }
				show_as_tooltip = {
					create_artifact = {
						name = artifact_demon_weapon_name
						description = artifact_demon_weapon_description
						type = sword
						visuals = sword
						modifier = artifact_placeholder_modifier
					}
				}
				hidden_effect = { create_demon_weapon_artifact_effect = { VISUALS = sword TYPE = sword } }
			}
			if = {
				limit = { scope:axe = yes }
				show_as_tooltip = {
					create_artifact = {
						name = artifact_demon_weapon_name
						description = artifact_demon_weapon_description
						type = axe
						visuals = axe
						modifier = artifact_placeholder_modifier
					}
				}
				hidden_effect = { create_demon_weapon_artifact_effect = { VISUALS = axe TYPE = axe } }
			}
			if = {
				limit = { scope:spear = yes }
				show_as_tooltip = {
					create_artifact = {
						name = artifact_demon_weapon_name
						description = artifact_demon_weapon_description
						type = spear
						visuals = spear
						modifier = artifact_placeholder_modifier
					}
				}
				hidden_effect = { create_demon_weapon_artifact_effect = { VISUALS = spear TYPE = spear } }
			}
			if = {
				limit = { scope:hammer = yes }
				show_as_tooltip = {
					create_artifact = {
						name = artifact_demon_weapon_name
						description = artifact_demon_weapon_description
						type = hammer
						visuals = hammer
						modifier = artifact_placeholder_modifier
					}
				}
				hidden_effect = { create_demon_weapon_artifact_effect = { VISUALS = hammer TYPE = hammer } }
			}
			if = {
				limit = { scope:dagger = yes }
				show_as_tooltip = {
					create_artifact = {
						name = artifact_demon_weapon_name
						description = artifact_demon_weapon_description
						type = dagger
						visuals = dagger
						modifier = artifact_placeholder_modifier
					}
				}
				hidden_effect = { create_demon_weapon_artifact_effect = { VISUALS = dagger TYPE = dagger } }
			}
			if = {
				limit = { scope:mace = yes }
				show_as_tooltip = {
					create_artifact = {
						name = artifact_demon_weapon_name
						description = artifact_demon_weapon_description
						type = mace
						visuals = mace
						modifier = artifact_placeholder_modifier
					}
				}
				hidden_effect = { create_demon_weapon_artifact_effect = { VISUALS = mace TYPE = mace } }
			}
			hidden_effect = {
				scope:newly_created_artifact = {
					set_artifact_name = artifact_demon_weapon_name
					set_artifact_description = artifact_demon_weapon_description
					set_variable = summoned_artifact
					set_variable = demon_weapon
					set_max_durability = 5
					flag_as_trash_artifact = yes
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_voice_harm_interaction = {
	category = interaction_category_vampire
	icon = POD_art_interaction
	desc = POD_voice_harm_interaction_desc
	
	cooldown = { years = 1 }
	cost = { piety = medium_piety_value }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_POD_kj_blood_trigger = yes }
		scope:actor = { has_perk = cry_of_blood_perk }
		scope:recipient = { is_landed = yes }
		scope:recipient.highest_held_title_tier > tier_barony
		scope:actor = { NOT = { target_is_liege_or_above = scope:recipient } }
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
			custom_tooltip = {
				text = "i_am_empty_of_chi"
				NAND = { 
					has_character_modifier = yinchi0_modifier
					has_character_modifier = yangchi0_modifier
				}
			}
		}
	}

	on_accept = {
		scope:recipient = {
			if = {
				limit = { has_perk = tigers_roar_perk }
				remove_short_term_gold = medium_gold_value 
			}
			every_knight = {
				limit = {
					trigger_if = {
						limit = { has_trait = statue }
						scope:actor = { has_perk = tigers_roar_perk }
					}
					trigger_if = {
						limit = { POD_is_not_spirit_trigger = no }
						scope:actor = { has_perk = cry_of_the_phantasm_perk }
					}
				}
				add_opinion = {
					target = scope:actor
					modifier = hurt_opinion
					opinion = -25
				}
				scope:actor = { 
					duel = {
						skill = prowess
						target = prev
						desc = POD_distant_attack_interaction.duel.desc
						50 = {
							compare_modifier = {
								value = scope:duel_value
								multiplier = high_positive_duel_skill_multiplier 
							}
							prev = { increase_wounds_hostile_effect = { REASON = howl ATTACKER = scope:actor } }
						}
						50 = {
							compare_modifier = {
								value = scope:duel_value
								multiplier = low_negative_duel_skill_multiplier 
							}
							min = 1
						}
					}
				}
			}					
		}
    }
	
	auto_accept = yes
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no 
		is_landed = yes 
	}
	ai_targets = {
		ai_recipients = primary_war_enemies
	}
	ai_frequency = 120
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 45
			scope:actor = {
			    is_at_war_with = scope:recipient
			}
		}
		modifier = {
		    factor = 2
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
		modifier = {
			factor = 0
			scope:recipient = {
				is_ai = no
			}
		}
	}
}

	desc = POD_remembrance_of_yomi_interaction_desc
	
	cooldown = { years = 1 }	
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_POD_kj_blood_trigger = yes }
		scope:actor = { has_perk = remembrance_of_yomi_perk }
	}
	
	is_valid = {
		OR = {
			scope:recipient = { is_imprisoned_by = scope:actor }
			scope:actor = { has_strong_usable_hook = scope:recipient }
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = { 
			NOT = { has_trait = infernalist }
			NOT = { has_trait = torpor }
			NOT = { has_trait = statue }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }	
			POD_is_not_clone_trigger = yes
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}
	
	on_accept = {
		scope:recipient = {
			add_secret = { TYPE = secret_infernalist }
			random_secret = {
				limit = { secret_type = secret_infernalist }
				reveal_to = scope:actor
			}
		}
	}
	
	auto_accept = yes
}
