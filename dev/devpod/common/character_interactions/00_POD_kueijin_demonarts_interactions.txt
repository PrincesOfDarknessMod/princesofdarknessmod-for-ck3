
POD_voice_harm_interaction = {
	category = interaction_category_vampire
	icon = POD_art_interaction
	desc = POD_voice_harm_interaction_desc
	
	cooldown = { years = 1 }
	cost = { piety = medium_piety_value }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_POD_kj_blood_trigger = yes }
		scope:actor = { has_perk = cry_of_blood_perk }
		scope:recipient = { is_playable_character = yes }
		scope:recipient.highest_held_title_tier > tier_barony
		scope:actor = { NOT = { target_is_liege_or_above = scope:recipient } }
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
			custom_tooltip = {
				text = "i_am_empty_of_chi"
				NAND = { 
					has_character_modifier = yinchi0_modifier
					has_character_modifier = yangchi0_modifier
				}
			}
		}
	}

	on_accept = {
		scope:actor = {
			POD_willpower_rouse_check_effect = yes
			POD_yinchi_cost_effect = { COUNT = 1 }
			add_demon_arts_lifestyle_xp = minor_lifestyle_xp
		}
		scope:recipient = {
			if = {
				limit = { scope:actor = { has_perk = tigers_roar_perk } }
				remove_short_term_gold = medium_gold_value 
			}
			every_knight = {
				limit = {
					trigger_if = {
						limit = { has_trait = automaton }
						scope:actor = { has_perk = tigers_roar_perk }
					}
					trigger_if = {
						limit = { can_be_killed_physically_trigger = no }
						scope:actor = { has_perk = cry_of_the_phantasm_perk }
					}
				}
				add_opinion = {
					target = scope:actor
					modifier = hurt_opinion
					opinion = -25
				}
				scope:actor = { 
					duel = {
						skill = prowess
						target = prev
						50 = {
							compare_modifier = {
								value = scope:duel_value
								multiplier = high_positive_duel_skill_multiplier 
							}
							prev = { increase_wounds_hostile_effect = { REASON = howl ATTACKER = scope:actor } }
						}
						50 = {
							compare_modifier = {
								value = scope:duel_value
								multiplier = low_negative_duel_skill_multiplier 
							}
							min = 1
						}
					}
				}
			}					
		}
	}
	
	auto_accept = yes
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no 
		is_landed = yes 
	}
	ai_targets = {
		ai_recipients = primary_war_enemies
	}
	ai_frequency = 120
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 45
			scope:actor = {
				is_at_war_with = scope:recipient
			}
		}
		modifier = {
			factor = 2
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
		modifier = {
			factor = 0
			scope:recipient = {
				is_ai = no
			}
		}
	}
}

POD_remembrance_of_yomi_interaction = {
	category = interaction_category_vampire
	icon = POD_art_interaction
	desc = POD_remembrance_of_yomi_interaction_desc
	
	cooldown = { years = 1 }	
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_POD_kj_blood_trigger = yes }
		scope:actor = { has_perk = remembrance_of_yomi_perk }
	}
	
	is_valid = {
		OR = {
			scope:recipient = { is_imprisoned_by = scope:actor }
			scope:actor = { has_strong_usable_hook = scope:recipient }
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = { 
			NOT = { has_trait = infernalist }
			NOT = { has_trait = torpor }
			NOT = { has_trait = automaton }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }	
			POD_is_not_clone_trigger = yes
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}
	
	on_accept = {
		scope:actor = { 
			POD_yinchi_cost_effect = { COUNT = 1 }
			add_demon_arts_lifestyle_xp = minor_lifestyle_xp 
		}
		scope:recipient = {
			add_secret = { TYPE = secret_infernalist }
			random_secret = {
				limit = { secret_type = secret_infernalist }
				reveal_to = scope:actor
			}
		}
	}
	
	auto_accept = yes
}

		}

	}

		scope:actor = {

	on_accept = {
		scope:actor = { add_demon_arts_lifestyle_xp = minor_lifestyle_xp }
		scope:secondary_actor = {
			start_scheme = {
				type = murder
				target_character = scope:recipient
			}
		}
	}
