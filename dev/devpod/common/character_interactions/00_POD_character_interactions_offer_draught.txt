# POD Schemes
# Investigate
# Vampiric: Entrance, Rationalize, Summon, Hunt, Herdify, Ghoulify, Embrace, Diablerize, Draught
# To do: Bloodbond, Scry the Soul (see aura), Telepathy (get a secret)
# Blood Bond, for simplicity sake it is a major hook that lasts 6 months

POD_offer_draught_interaction  = {
	category = interaction_category_vampire
	common_interaction = yes
	desc = POD_offer_draught_interaction_desc

	is_shown = {
		scope:actor = {
			OR = {
				has_trait = potenceadvanced
				has_trait = celerityadvanced
				has_trait = fortitudeadvanced
				has_perk = draught_of_might_perk
				has_perk = draught_of_elegance_perk
				has_perk = draught_of_endurance_perk
			}	
		}
		POD_is_in_same_plane_trigger = yes
		scope:actor = { 
			OR = { 
				has_POD_vamp_blood_trigger = yes
				has_trait = risen
			}
		}
		scope:recipient = {
			OR = {
				is_vassal_of = scope:actor
				is_courtier_of = scope:actor
			}
		}
		NOT = { scope:recipient = scope:actor }
	}

	is_valid_showing_failures_only = {
		scope:recipient = { 
			NOT = { has_trait = torpor }
			NOT = { has_trait = frozen }
			NOT = { has_trait = statue }
			NOT = { has_trait = wight }
			NOT = { has_trait = barghest }
			trigger_if = {
				limit = { has_trait = wraith }
				has_trait = risen
			}
		}
		scope:recipient = { 
			is_busy_in_events_localised = yes
		}
		scope:actor = {
			NOT = { has_character_modifier = bloodhunger5_modifier }	
			NOT = { has_character_modifier = pathos_0 } 			
		}		
	}	
	
	options_heading = POD_offer_draught_interaction_options_heading

	send_option = {
		flag = POD_offer_draught_might
		localization = POD_offer_draught_interaction_might
		is_valid = {
			custom_description = {
				text = POD_offer_draught_might_requirements.tt
				OR = {
					AND = {
						scope:actor = {
							has_trait = potenceadvanced
						}
						scope:recipient = { NOT = { has_trait = potenceadvanced } }
					}
					AND = {
						scope:actor = {
							has_perk = draught_of_might_perk
						}
						scope:recipient = { 
							OR = {
								NOT = { has_trait = potenceadvanced }
								NOT = { has_trait = potence }
							}
						}
					}
				}
				
				scope:recipient = { 
					NOT = { has_trait = potenceadvanced } 
					NOT = { has_character_modifier = POD_greater_draught_of_might_modifier }
					NOT = { has_character_modifier = POD_lesser_draught_of_might_modifier }
				}
			}
		}		
	}

	send_option = {
		flag = POD_offer_draught_elegance
		localization = POD_offer_draught_interaction_elegance
		is_valid = {
			custom_description = {
				text = POD_offer_draught_elegance_requirements.tt
				OR = {
					AND = {
						scope:actor = {
							has_trait = celerityadvanced
						}
						scope:recipient = { NOT = { has_trait = celerityadvanced } }
					}
					AND = {
						scope:actor = {
							has_perk = draught_of_elegance_perk
						}
						scope:recipient = { 
							OR = {
								NOT = { has_trait = celerityadvanced }
								NOT = { has_trait = celerity }
							}
						}
					}
				}
				
				scope:recipient = { 
					NOT = { has_trait = celerityadvanced } 
					NOT = { has_character_modifier = POD_greater_draught_of_elegance_modifier }
					NOT = { has_character_modifier = POD_lesser_draught_of_elegance_modifier }
				}	
			}
		}		
	}

	send_option = {
		flag = POD_offer_draught_endurance
		localization = POD_offer_draught_interaction_endurance
		is_valid = {
			custom_description = {
				text = POD_offer_draught_endurance_requirements.tt
					OR = {
						AND = {
							scope:actor = {
								has_trait = fortitudeadvanced
							}
							scope:recipient = { NOT = { has_trait = fortitudeadvanced } }
						}
						AND = {
							scope:actor = {
								has_perk = draught_of_endurance_perk
							}
							scope:recipient = { 
								OR = {
									NOT = { has_trait = fortitudeadvanced }
									NOT = { has_trait = fortitude }
								}
							}
						}
					}
				
				scope:recipient = { 
					NOT = { has_trait = fortitudeadvanced } 
					NOT = { has_character_modifier = POD_greater_draught_of_endurance_modifier }
					NOT = { has_character_modifier = POD_lesser_draught_of_endurance_modifier }
				}
			}
		}
	}		


	send_option = {
		flag = POD_offer_draught_nothing
		localization = POD_offer_draught_interaction_nothing
	}

	cost = {

	}
	
	on_accept = {
		
		if = {
			limit = {
				scope:POD_offer_draught_might = yes
			}		
			scope:actor = {
				if = {
					limit = { has_trait = risen }
					POD_pathos_decrease_multiple_effect = { COUNT = 1 }
				}
				else = {
					POD_hunger_rouse_check_effect = yes
				}
			}
			scope:recipient = {
				POD_draught_of_might_effect = yes	
				add_opinion = {
					target = scope:actor
					modifier = gave_draught_opinion
				}
			}	
		}

		if = {
			limit = {
				scope:POD_offer_draught_elegance = yes
			}
			scope:actor = {
				if = {
					limit = { has_trait = risen }
					POD_pathos_decrease_multiple_effect = { COUNT = 1 }
				}
				else = {
					POD_hunger_rouse_check_effect = yes
				}
			}
			scope:recipient = {
				POD_draught_of_elegance_effect = yes					
				add_opinion = {
					target = scope:actor
					modifier = gave_draught_opinion
				}
			}
		}
		
		if = {
			limit = {
				scope:POD_offer_draught_endurance = yes
			}
			scope:actor = {
				if = {
					limit = { has_trait = risen }
					POD_pathos_decrease_multiple_effect = { COUNT = 1 }
				}
				else = {
					POD_hunger_rouse_check_effect = yes
				}
			}
			scope:recipient = {
				POD_draught_of_endurance_effect = yes			
				add_opinion = {
					target = scope:actor
					modifier = gave_draught_opinion
				}
				add_character_flag = { flag = granted_toughness months = 11 }
			}
		}		
		
		if = {
			limit = {
				scope:POD_offer_draught_nothing = yes
			}
		}
	}
	
	auto_accept = yes
	ai_potential = {
		is_at_war = yes
	}
	ai_targets = {
		ai_recipients = courtiers 
		ai_recipients = vassals
	}
	ai_will_do = {
		base = 0
		modifier = {
			add = 100
			scope:recipient = { is_knight_of = scope:actor }
		}
		modifier = {
			add = -50
			scope:actor = { has_character_modifier = bloodhunger4_modifier }
		}
		modifier = {
			add = -25
			scope:actor = { has_character_modifier = bloodhunger3_modifier }
		}
		modifier = {
			factor = 0.5
			scope:actor = {
				has_trait = paranoid
			}
		}
	}
	ai_frequency = 6
	use_diplomatic_range = yes
	cooldown = { months = 6 }	
}
