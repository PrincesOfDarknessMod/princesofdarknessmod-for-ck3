		scope:actor = {
			play_music_cue = "mx_cue_sacredrite"

						limit = {
							NAND = {
								has_trait = ha_patris_mastery
								has_trait = ha_filius_mastery
								has_trait = ha_spiritus_mastery
								}
							}
					add_faith_lifestyle_xp = 50
					}
				send_interface_toast = {
					title = POD_ha_bless_interaction_toast
					left_icon = scope:actor
					right_icon = scope:recipient
		scope:recipient = {
			add_character_modifier = {
			modifier = blessed_by_truefaith_modifier
			years = 5
			}
					}
				}	
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
		modifier = {
		add = 100
			scope:actor = {
				piety > 200
				is_at_war = yes
				any_knight = {
				this = scope:recipient
				prowess >= 20
				}
			}

	}
