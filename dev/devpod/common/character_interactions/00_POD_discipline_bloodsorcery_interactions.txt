# POD Schemes

blood_sorcery_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	#interface_priority = 84
	
	is_shown = {
		scope:actor = { 
			OR = {
				has_abyssal_mysticism_bloodmagic_trigger = yes
				has_bloodsorcery_trigger = yes
			}
		}
		
	}
	
	is_valid_showing_failures_only = {
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes	
	}
	on_accept = {
		custom_tooltip = open_bs_menu
		
		scope:actor = {
			POD_open_bloodmagic_gui = { TARGET = scope:recipient }
		}
	}
	
	auto_accept = yes
}
