blood_sorcery_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	#interface_priority = 84
	
	is_shown = {
		scope:actor = { 
			has_bloodsorcery_trigger = yes
		}
		
	}

	send_options_exclusive = no

	send_option = {
		is_shown = { scope:actor = { OR = { 
				has_koldunic_bloodmagic_trigger = yes
				has_sadhana_bloodmagic_trigger = yes
				has_akhu_bloodmagic_trigger = yes
				has_dur_an_ki_bloodmagic_trigger = yes
				has_thaumaturgy_bloodmagic_trigger = yes 
			}
		} }
		flag = old_bloodmagic
		localization = POD_old_bloodmagic
	}
	
	is_valid_showing_failures_only = {
		scope:actor = { 
			custom_tooltip = {
				text = song_of_jupiter_botch.ct
				NOT = { has_character_flag = song_of_jupiter_botch }
			} 
			custom_tooltip = {
				text = blood_sorcery_open.ct
				NOT = { has_variable = blood_sorcery_open }
			} 
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes	
	}
	on_accept = {
		custom_tooltip = open_bs_menu
		
		scope:actor = {
			if = { 
				limit = { 
					OR = { 
						scope:old_bloodmagic = yes 
						scope:actor = { 
							AND = { 
								has_koldunic_bloodmagic_trigger = no
								has_sadhana_bloodmagic_trigger = no
								has_akhu_bloodmagic_trigger = no
								has_dur_an_ki_bloodmagic_trigger = no
								has_thaumaturgy_bloodmagic_trigger = no 
							}
						} 
					} 
				} 
				POD_initiate_blood_sorcery_exp_effect = yes
				#remove_variable = blood_sorcery_open
				add_to_variable_list = {
					name = path_of_mars
					target = flag:strike_true
				}
				add_to_variable_list = {
					name = path_of_mars
					target = flag:wind_dance
				}
				add_to_variable_list = {
					name = path_of_mars
					target = flag:comrades_in_arms
				}
				add_to_variable_list = {
					name = path_of_blood
					target = flag:a_taste_of_blood
				}
				add_to_variable_list = {
					name = path_of_blood
					target = flag:blood_of_potency
				}
				add_to_variable_list = {
					name = path_of_blood
					target = flag:theft_of_vitae
				}
				
				add_to_variable_list = {
					name = path_of_neptunes_might
					target = flag:eyes_of_the_sea
				}
				add_to_variable_list = {
					name = path_of_neptunes_might
					target = flag:prison_of_water
				}
				add_to_variable_list = {
					name = path_of_neptunes_might
					target = flag:dehydrate
				}
				add_to_variable_list = {
					name = path_of_curses
					target = flag:stigma
				}
				add_to_variable_list = {
					name = path_of_curses
					target = flag:malady
				}
				add_to_variable_list = {
					name = path_of_curses
					target = flag:acedia
				}
				add_to_variable_list = {
					name = path_of_focused_mind
					target = flag:readiness
				}
				add_to_variable_list = {
					name = path_of_focused_mind
					target = flag:centering
				}
				add_to_variable_list = {
					name = path_of_focused_mind
					target = flag:perfect_clarity
				}
				add_to_variable_list = {
					name = path_of_praapti
					target = flag:emergency_recall
				}
				add_to_variable_list = {
					name = path_of_praapti
					target = flag:bind_in_vine_and_sand
				}
				add_to_variable_list = {
					name = path_of_praapti
					target = flag:travel_on_mythic_strings
				}
				add_to_variable_list = {
					name = path_of_music_of_the_shperes
					target = flag:song_of_venus
				}
				add_to_variable_list = {
					name = path_of_music_of_the_shperes
					target = flag:song_of_saturn
				}
				add_to_variable_list = {
					name = path_of_music_of_the_shperes
					target = flag:song_of_jupiter
				}
				
				set_variable = { name = blood_sorcery_cost value = 0}
				
				set_variable = { name = chosen_path value = flag:nothing}
				set_variable = { name = target value = scope:recipient}
				
				
				trigger_event = path_of_blood.3000
				
				
			}
			else = {
				scope:actor = { set_variable = { name = spell_target value = scope:recipient } POD_open_bloodmagic_gui = yes }
			}
		}
	}
	
	auto_accept = yes
}
