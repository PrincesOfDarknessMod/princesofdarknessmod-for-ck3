		NOT = { scope:actor = scope:recipient }

POD_psi_leech_interaction = {
	category = interaction_category_psychic
	icon = POD_psi_interaction
	desc = POD_psi_leech_interaction_desc
	options_heading = POD_psi_leech_interaction_options_heading
	
	cooldown = { months = 3 }
	cost = { piety = medium_piety_value }
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_trait = psychic }
		scope:actor = {
			OR = {
				has_perk = psychic_vampirism_perk
				has_perk = soulstealing_perk
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = { NOT = { has_character_modifier = celestial_barrier_modifier } }
		scope:actor = { POD_can_interact_with_skinlands_trigger = { RECIPIENT = scope:recipient } }
		scope:recipient = {
			is_imprisoned_by = scope:actor
			NOT = { has_perk = anti_psi_perk }
			NOT = { has_trait = torpor }
			NOT = { has_trait = statue }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }	
			POD_is_not_clone_trigger = yes
			POD_is_not_trapped_in_umbra_trigger = { ACTOR = scope:actor }
		}
	}
	
	send_option = {
		is_valid = {
			scope:actor = { has_perk = psychic_vampirism_perk }
		}
		flag = willpower
		localization = POD_psi_leech_interaction_willpower
	}
	
	send_option = {
		is_valid = {
			scope:actor = { has_perk = psychic_vampirism_perk }
		}
		flag = lifeforce
		localization = POD_psi_leech_interaction_lifeforce
	}
	
	send_option = {
		is_valid = {
			scope:actor = { has_perk = soulstealing_perk }
		}
		flag = gnosis
		localization = POD_psi_leech_interaction_gnosis
	}

	on_accept = {
		if = {
			limit = {
				scope:willpower = yes
			}
			scope:actor = {
				send_interface_toast = {
					title = court_physician_treatment_success_message_title
					add_stress = medium_stress_loss
				}
			}
			scope:recipient = {
				add_stress = major_stress_gain
			}
		}
		if = {
			limit = {
				scope:lifeforce = yes
			}
			scope:actor = {
				if = {
					limit = {
						POD_has_health_loss_modifier_trigger = yes
					}
					POD_recovering_health_level_effect = yes
				}
				else = {
					add_character_modifier = { 
						modifier = healing_energy_modifier 
						days = 400
					}
				}
			}
			scope:recipient = {
				POD_losing_health_level_effect = { COUNT = 2 }
			}
		}
		if = {
			limit = {
				scope:gnosis = yes
			}
			scope:actor = {
				add_psi_lifestyle_xp = 200
			}
			scope:recipient = { 
				POD_universal_energy_decrease_multiple_effect = { COUNT = 10 }
			}
		}
		
		scope:recipient = { 
			add_opinion = {
				target = scope:actor
				modifier = hurt_opinion
				opinion = -20
			}
		}
	}
	
	auto_accept = yes
	
	
	ai_will_do = {
		base = 0
	}
}
