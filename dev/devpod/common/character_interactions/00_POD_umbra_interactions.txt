POD_ask_reweave_interaction = {
	category = interaction_category_friendly
	desc = POD_ask_reweave_interaction_desc
	common_interaction = yes
	icon = POD_umbra_interaction
	
	ai_min_reply_days = 0
	ai_max_reply_days = 0

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			any_character_artifact = {
				has_variable = artifact_pieces
			}
		}
		POD_is_in_same_plane_trigger = yes
	}

	is_valid_showing_failures_only= {
		scope:actor = {
			custom_description = {
				text = POD_has_equipped_artifact_pieces
				any_equipped_character_artifact = {
					count >= 2
					artifact_slot_type = miscellaneous
					has_variable = artifact_pieces
				}
			}
		}
		scope:recipient = { 
			trigger_if = { 
				limit = { has_trait = wraith }
				has_trait = arcanos_inhabit 
			}
			trigger_if = { 
				limit = { has_trait = mummy }
				has_perk = major_restoration_perk
			}
		}
		scope:recipient = { 
			NOT = { has_trait = torpor } 
		}
	}
	
	send_option = {
		flag = weapon
		localization = add_artifact_interaction_weapon
	}
	send_option = {
		flag = armor
		localization = add_artifact_interaction_armor
	}
	send_option = {
		flag = shinies
		localization = add_artifact_interaction_shinies
	}
	send_option = {
		is_shown = { scope:actor = { has_royal_court = yes } }
		flag = book
		localization = add_artifact_interaction_book
	}
	send_option = {
		is_shown = { scope:actor = { has_royal_court = yes } }
		flag = furniture
		localization = add_artifact_interaction_furniture
	}
	send_option = {
		flag = trinket
		localization = add_artifact_interaction_trinket
	}

	on_accept = {
		create_artifact_reweaved_artifact_effect = {
			OWNER = scope:actor
			CREATOR = scope:recipient
		}
	}
	
	ai_accept = {
		base = 0 # Try to make it 0 for most interactions
		
		opinion_modifier = { # Opinion Factor
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 1.0
			desc = AI_OPINION_REASON
		}
	}
	
	ai_will_do = {
		base = 0
	}
}
