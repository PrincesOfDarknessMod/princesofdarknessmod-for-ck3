POD_ask_reweave_interaction = {
	category = interaction_category_wraith
	desc = POD_ask_reweave_interaction_desc
	common_interaction = yes
	icon = POD_umbra_interaction
	
	ai_min_reply_days = 0
	ai_max_reply_days = 0

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			any_character_artifact = {
				has_variable = artifact_pieces
			}
		}
	}

	is_valid_showing_failures_only= {
		scope:actor = {
			custom_description = {
				text = POD_has_equipped_artifact_pieces
				any_equipped_character_artifact = {
					count >= 2
					artifact_slot_type = miscellaneous
					has_variable = artifact_pieces
				}
			}
		}
		scope:recipient = { 
			trigger_if = {
				limit = { is_east_mummy_trigger = yes }
				NOT = { has_character_modifier = yangchi0_modifier }
			}
			trigger_else_if = {
				limit = { is_amenti_trigger = yes }
				NOT = { has_character_modifier = sekhem0_modifier }
			}
			trigger_else = { 
				NOT = { has_character_modifier = pathos0_modifier }
			}
			trigger_if = { 
				limit = { has_trait = mummy }
				has_perk = major_restoration_perk
			}
			trigger_else = { 
				has_trait = arcanos_inhabit 
			}
		}
		scope:recipient = { 
			NOT = { has_trait = torpor } 
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }	
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}
	
	send_option = {
		flag = weapon
		localization = add_artifact_interaction_weapon
	}
	send_option = {
		flag = armor
		localization = add_artifact_interaction_armor
	}
	send_option = {
		flag = shinies
		localization = add_artifact_interaction_shinies
	}
	send_option = {
		is_shown = { scope:actor = { has_royal_court = yes } }
		flag = book
		localization = add_artifact_interaction_book
	}
	send_option = {
		is_shown = { scope:actor = { has_royal_court = yes } }
		flag = furniture
		localization = add_artifact_interaction_furniture
	}
	send_option = {
		flag = trinket
		localization = add_artifact_interaction_trinket
	}

	on_accept = {
		scope:recipient = {
			if = {
				limit = { has_trait = mummy }
				POD_hekau_cost_effect = { COUNT = 4 }
			}
			else = {
				POD_arcanoi_cost_effect = { COUNT = 2 }
			}
			save_scope_as = creator
		}
		scope:actor = { 
			remove_short_term_gold = umbra_reweave_artifact_gold_value
			custom_tooltip = POD_recreate_artifact_interaction_desc
			hidden_effect = {
				if = {
					limit = { scope:weapon = yes }
					create_artifact_reweaved_weapon_effect = { OWNER = scope:actor }
				}
				else_if = {
					limit = { scope:armor = yes }
					create_artifact_reweaved_armor_effect = { OWNER = scope:actor }
				}
				else_if = {
					limit = { scope:shinies = yes }
					create_artifact_reweaved_shinies_effect = { OWNER = scope:actor }
				}
				else_if = {
					limit = { scope:book = yes }
					create_artifact_reweaved_book_effect = { OWNER = scope:actor }
				}
				else_if = {
					limit = { scope:furniture = yes }
					create_artifact_reweaved_furniture_effect = { OWNER = scope:actor }
				}
				else_if = {
					limit = { scope:trinket = yes }
					create_artifact_reweaved_trinket_effect = { OWNER = scope:actor }
				}
			}
		}
	}
	
	ai_accept = {
		base = 0 # Try to make it 0 for most interactions
		opinion_modifier = { # Opinion Factor
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 1.0
			desc = AI_OPINION_REASON
		}
		modifier = {
			add = 50
			scope:recipient = {
				has_dread_level_towards = {
					target = scope:actor
					level = 1
				}
			}
			desc = INTIMIDATED_REASON
		}
		modifier = {
			add = 100
			scope:recipient = {
				has_dread_level_towards = {
					target = scope:actor
					level = 2
				}
			}
			desc = COWED_REASON
		}
		modifier = {
			scope:recipient.house = scope:actor.house
			add = 50
			desc = IS_SAME_HOUSE_REASON
		}
	}
	
	ai_will_do = {
		base = 0
	}
}

POD_ask_empower_artifact_piece_interaction = {
	category = interaction_category_mummy
	desc = POD_ask_empower_artifact_piece_interaction_desc
	common_interaction = yes
	icon = POD_umbra_interaction
	
	ai_min_reply_days = 0
	ai_max_reply_days = 0

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			any_character_artifact = {
				has_variable = artifact_pieces
			}
		}
		scope:recipient = { has_trait = mummy }
	}

	is_valid_showing_failures_only = {
		scope:actor = { 
			custom_description = {
				text = POD_has_equipped_artifact_pieces
				any_equipped_character_artifact = {
					count >= 2
					has_variable = artifact_pieces
				}
			}
		}
		scope:recipient = { 
			trigger_if = {
				limit = { is_east_mummy_trigger = yes }
				NOT = { has_character_modifier = yangchi0_modifier }
			}
			trigger_if = {
				limit = { is_amenti_trigger = yes }
				NOT = { has_character_modifier = sekhem0_modifier }
			}
			trigger_if = {
				limit = { restore_artifact_check <= 1 }
				has_perk = major_restoration_perk
			}
			trigger_else_if = {
				limit = { restore_artifact_check = 2 }
				has_perk = greater_restoration_perk
			}
			trigger_else = {
				has_perk = princely_restoration_perk
			}
		}
	}

	on_accept = {
		scope:actor = { 
			remove_short_term_gold = umbra_reweave_artifact_gold_value
			custom_tooltip = POD_artifact_pieces.100.c.tt
		}
		scope:recipient = { 
			POD_hekau_cost_effect = { COUNT = 3 } 
			save_scope_as = creator
			duel = {
				skill = stewardship
				value = scope:actor.restore_artifact_check
				50 = {
					desc = board_games.0001.tt.success
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					scope:actor = {
						send_interface_toast = {
							title = board_games.0001.tt.success
							left_icon = scope:actor
							right_icon = scope:recipient
							
							custom_tooltip = {
								text = POD_artifact_pieces.100.c.success
								create_artifact_restored_artifact_pieces_effect = { OWNER = scope:actor }
							}
						}
					}
				}
				50 = {
					desc = board_games.0001.tt.failure
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier 
					}
					scope:actor = {
						every_equipped_character_artifact = {
							limit = {
								artifact_slot_type = miscellaneous
								has_variable = artifact_pieces
							}
							destroy_artifact = this
						}
					}
				}
			}
		}
	}
	
	ai_accept = {
		base = 0 # Try to make it 0 for most interactions
		opinion_modifier = { # Opinion Factor
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 1.0
			desc = AI_OPINION_REASON
		}
		modifier = {
			add = 50
			scope:recipient = {
				has_dread_level_towards = {
					target = scope:actor
					level = 1
				}
			}
			desc = INTIMIDATED_REASON
		}
		modifier = {
			add = 100
			scope:recipient = {
				has_dread_level_towards = {
					target = scope:actor
					level = 2
				}
			}
			desc = COWED_REASON
		}
		modifier = {
			scope:recipient.house = scope:actor.house
			add = 50
			desc = IS_SAME_HOUSE_REASON
		}
	}
	
	ai_will_do = {
		base = 0
	}
}
