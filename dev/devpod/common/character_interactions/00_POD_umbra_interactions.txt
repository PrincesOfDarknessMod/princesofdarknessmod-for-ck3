POD_ask_empower_artifact_piece_interaction = {
	category = interaction_category_mummy
	desc = POD_ask_empower_artifact_piece_interaction_desc
	common_interaction = yes
	icon = POD_umbra_interaction
	
	ai_min_reply_days = 0
	ai_max_reply_days = 0

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { has_trait = mummy }
	}

	is_valid_showing_failures_only = {
		scope:recipient = { 
			trigger_if = {
				limit = { is_east_mummy_trigger = yes }
				NOT = { has_character_modifier = yangchi0_modifier }
			}
			trigger_if = {
				limit = { is_amenti_trigger = yes }
				NOT = { has_character_modifier = sekhem0_modifier }
			}
			has_perk = major_restoration_perk
		}
	}
	
	send_option = {
		is_valid = {
			scope:actor = {
				custom_tooltip = {
					text = enough_common_artifact_piece
					common_artifact_piece_reserve >= 3
				}
			}
		}
		flag = common
		localization = common
	}
	
	send_option = {
		is_valid = {
			scope:actor = {
				custom_tooltip = {
					text = enough_masterwork_artifact_piece
					masterwork_artifact_piece_reserve >= 3
				}
			}
			scope:recipient = { 
				has_perk = greater_restoration_perk
			}
		}
		flag = masterwork	
		localization = masterwork
	}
	
	send_option = {
		is_valid = {
			scope:actor = {
				custom_tooltip = {
					text = enough_famed_artifact_piece
					famed_artifact_piece_reserve >= 3
				}
			}
			scope:recipient = { 
				has_perk = princely_restoration_perk
			}
		}
		flag = famed
		localization = famed
	}

	on_accept = {
		scope:actor = { 
			remove_short_term_gold = minor_gold_value
			custom_tooltip = POD_artifact_pieces.100.c.tt
		}
		scope:recipient = { 
			POD_hekau_cost_effect = { COUNT = 3 } 
			save_scope_as = creator
			scope:actor = {
				send_interface_toast = {
					title = board_games.0001.tt.success
					left_icon = scope:actor
					right_icon = scope:recipient
					
					switch = {
						trigger = yes
						scope:common = {
							lose_common_artifact_pieces_effect = { VALUE = 3 }
							gain_masterwork_artifact_pieces_effect = { VALUE = 1 }
						}
						scope:masterwork = {
							lose_masterwork_artifact_pieces_effect = { VALUE = 3 }
							gain_famed_artifact_pieces_effect = { VALUE = 1 }
						}
						scope:famed = {
							lose_famed_artifact_pieces_effect = { VALUE = 3 }
							gain_illustrious_artifact_pieces_effect = { VALUE = 1 }
						}
					}
				}
			}
		}
	}
	
	ai_accept = {
		base = 0 # Try to make it 0 for most interactions
		opinion_modifier = { # Opinion Factor
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 1.0
			desc = AI_OPINION_REASON
		}
		modifier = {
			add = 50
			scope:recipient = {
				has_dread_level_towards = {
					target = scope:actor
					level = 1
				}
			}
			desc = INTIMIDATED_REASON
		}
		modifier = {
			add = 100
			scope:recipient = {
				has_dread_level_towards = {
					target = scope:actor
					level = 2
				}
			}
			desc = COWED_REASON
		}
		modifier = {
			scope:recipient.house = scope:actor.house
			add = 50
			desc = IS_SAME_HOUSE_REASON
		}
	}
	
	ai_will_do = {
		base = 0
	}
}
