POD_bears_bounty_interaction = {
	category = interaction_category_vampire
	icon = POD_gurahl_interaction
	desc = POD_bears_bounty_interaction_desc
	
	cooldown = { months = 120 }
	cost = { piety = massive_piety_value }
	
	auto_accept = yes

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { POD_splat_has_gnosis_trigger = yes }
		scope:actor = { has_perk = bears_bounty_perk }
		scope:recipient = { is_landed = yes }
		scope:recipient.highest_held_title_tier > tier_barony
	}
	
	is_valid_showing_failures_only = {
		scope:actor = { NOT = { has_character_modifier = POD_gnosis_0_modifier } }
		POD_cross_umbra_interaction_valid_trigger = yes
	}

	on_accept = {
		scope:recipient = { 
			add_opinion = {
				target = scope:actor
				modifier = hurt_opinion
				opinion = -80
			}
		}
	   scope:actor = {
			POD_willpower_rouse_check_effect = yes	
			POD_gnosis_loss_check_effect = yes
			
			if = {
				limit = { 
					scope:recipient.highest_held_title_tier = tier_empire
				}
				add_piety = -2000
			}
			else_if = {
				limit = { 
					scope:recipient.highest_held_title_tier >= tier_kingdom 
				}
				add_piety = -1000
			}
			else_if = {
				limit = { 
					scope:recipient.highest_held_title_tier >= tier_duchy 
				}
				add_piety = -500
			}
				
			custom_tooltip = POD_weather_magic_interaction.tt.plague
	
			duel = {
				skill = learning
				value = 10
				desc = POD_weather_magic.duel.desc
			
				50 = {
					desc = POD_weather_magic.duel.success
					
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					
					scope:actor = {
						send_interface_toast = {
							title = POD_weather_magic.duel.success
							left_icon = scope:actor
							right_icon = scope:recipient	
							custom_tooltip = POD_weather_magic.duel.tt.attacker_wins
						}
					}
					
					hidden_effect = {
						scope:recipient = {
							send_interface_toast = {
								title = POD_weather_magic_from_enemy_toast
								left_icon = scope:recipient
								right_icon = scope:actor
								custom_tooltip = POD_weather_magic_from_enemy.tt
							}
							every_sub_realm_county = {
								add_county_modifier = {
									modifier = weather_magic_plague_modifier
									years = 5
								}
							}
							add_opinion = {
								target = scope:actor
								modifier = POD_cast_evil_weather_magic_to_my_realm
							}
						}
					}
				}
				
				50 = {
					desc = POD_weather_magic.duel.failure

					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier
					}
					
					scope:actor = {
						send_interface_toast = {
							title = POD_weather_magic.duel.failure
							left_icon = scope:actor
							right_icon = scope:recipient
							custom_tooltip = POD_weather_magic.duel.tt.attacker_lost
						}
					}
				}
			}
		}
		scope:recipient = { 
			progress_towards_rival_effect = {
				CHARACTER = scope:actor
				REASON = rival_cast_spell
				OPINION = 0
			}
		}
    }
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no 
		is_landed = yes 
		
		piety > 1500
	}
	ai_targets = {
		ai_recipients = primary_war_enemies
	}
	ai_frequency = 360
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 45
			scope:actor = {
			    is_at_war_with = scope:recipient
			}
		}
		modifier = {
		    factor = 2
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
	} 
}
