POD_piety_for_perk_interaction = {
	category = interaction_category_vampire
	icon = POD_golconda_interaction
	desc = POD_piety_for_perk_interaction_desc
	options_heading = POD_piety_for_perk_interaction_options_heading

	send_option = {
		flag = animal_perk
		localization = POD_piety_for_perk_interaction_animal	
	}

	send_option = {
		flag = choleric_perk
		localization = POD_piety_for_perk_interaction_choleric
	}

	send_option = {
		flag = melancholic_perk
		localization = POD_piety_for_perk_interaction_melancholic
	}

	send_option = {
		flag = phlegmatic_perk
		localization = POD_piety_for_perk_interaction_phlegmatic
	}

	send_option = {
		flag = sanguine_perk
		localization = POD_piety_for_perk_interaction_sanguine
	}
	
	send_options_exclusive = yes
	
	cost = { piety = 1000 }

	auto_accept = yes
	ai_frequency = 0
	use_diplomatic_range = no
	
	is_shown = {
		scope:actor = scope:recipient 
		scope:actor = {
			is_ai = no
			is_adult = yes
			has_trait_with_flag = perks_for_piety_flag

		scope:actor = {

				add_choleric_lifestyle_perk_points = 1
			}
			
			if = {
				limit = { scope:melancholic_perk = yes }
				add_melancholy_lifestyle_perk_points = 1
			}
			
			if = {
				limit = { scope:phlegmatic_perk = yes }
				add_phlegmatic_lifestyle_perk_points = 1
			}
			
			if = {
				limit = { scope:sanguine_perk = yes }
				add_sanguine_lifestyle_perk_points = 1
			}
		}	
	}
}

POD_golconda_mentoring_interaction = {
	category = interaction_category_vampire
	icon = POD_golconda_interaction
	desc = POD_golconda_mentoring_interaction_desc

	cooldown = { years = 1 }
	cost = { piety = 500 }

	ai_frequency = 0
	
	send_option = {
		flag = only_once
		localization = POD_golconda_mentoring_interaction_only_once
	}
	
	send_option = {
		flag = repeating
		localization = POD_golconda_mentoring_interaction_repeating
	}

	send_options_exclusive = yes

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			is_ai = no
			is_adult = yes
			OR = {
				has_trait = golconda
				has_trait = golconda_seeker
			}
		}
		scope:recipient = {
			POD_is_vampire_trigger = yes
			NOT = { has_trait = golconda }
		}
	}

	is_valid_showing_failures_only = {
		custom_tooltip = {
			text = POD_golconda_mentoring_interaction_requirements_mentee
			scope:actor = {
				NOT = { has_variable = golconda_mentee }
			}
		}
		custom_tooltip = {
			text = POD_golconda_mentoring_interaction_requirements
			scope:actor = {
				trigger_if = {
					limit = { has_trait = golconda_seeker }
					has_trait_xp = {
						trait = golconda_seeker
						value >= 20
					}
				}
			}
		}
		custom_tooltip = {
			text = POD_golconda_mentoring_interaction_requirements_2
			scope:recipient = {
				OR = {
					NOT = { has_trait = golconda_seeker }
					suspire_success_chance < scope:actor.suspire_success_chance
				}
			}
		}
	}

	on_accept = {
		scope:actor = {
			send_interface_toast = {
				title = POD_golconda_mentoring_interaction_toast
				left_icon = scope:actor
				right_icon = scope:recipient
				scope:recipient = {
					if = {
						limit = { NOT = { has_trait = golconda_seeker } }
						add_trait = golconda_seeker
					}
					random_list = {
						1 = { add_trait_xp = { trait = golconda_seeker value = 3 } }
						1 = { add_trait_xp = { trait = golconda_seeker value = 4 } }
						1 = { add_trait_xp = { trait = golconda_seeker value = 5 } }
						1 = { add_trait_xp = { trait = golconda_seeker value = 6 } }
						1 = { add_trait_xp = { trait = golconda_seeker value = 7 } }
					}
				}
			}
			if = {
				limit = { scope:repeating = yes }
				custom_tooltip = {
					text = POD_golconda_mentoring_interaction_repeating_effect
					set_variable = {
						name = golconda_mentee
						value = scope:recipient
					}
					trigger_event = {
						id = POD_golconda_journey.2000
						years = 1
					}
				}
			}
		}
	}
	
	ai_accept = {
		base = -50
		opinion_modifier = { # Opinion Factor
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 1.0
			desc = AI_OPINION_REASON
		}
	}
}

POD_guide_through_suspire_interaction = {
	category = interaction_category_vampire
	icon = POD_golconda_interaction
	desc = POD_guide_through_suspire_interaction_desc

	cost = { piety = 1000 }

	ai_frequency = 0
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			is_ai = no
			is_adult = yes
			has_trait = golconda
		}
		scope:recipient = {
			POD_is_vampire_trigger = yes
			NOT = { has_trait = golconda }
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			has_trait = golconda_seeker
		}
	}

	on_accept = {
		scope:actor = {
			send_interface_toast = {
				title = POD_guide_through_suspire_interaction_toast
				left_icon = scope:actor
				right_icon = scope:recipient
				scope:recipient = {
					random_list = {
						0 = {
							modifier = { add = { value = suspire_success_chance } }
							POD_achieve_golconda_effect = yes
						}
						100 = {
							modifier = { add = { value = 0 subtract = suspire_success_chance } }
							add_trait = wight
							depose = yes
						}
					}
				}
			}
		}
	}
	
	ai_accept = {
		base = -150
		opinion_modifier = { # Opinion Factor
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 1.0
			desc = AI_OPINION_REASON
		}
		modifier = {
			add = scope:recipient.suspire_success_chance
			desc = SUSPIRE_SUCCESS_REASON
		}
	}
}
