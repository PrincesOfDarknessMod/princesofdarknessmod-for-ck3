POD_piety_for_perk_interaction = {
	category = interaction_category_vampire
	#interface_priority= 91
	icon = POD_golconda_interaction
	desc = POD_piety_for_perk_interaction_desc
	options_heading = POD_piety_for_perk_interaction_options_heading

	send_option = {
		flag = animal_perk
		localization = POD_piety_for_perk_interaction_animal	
	}

	send_option = {
		flag = choleric_perk
		localization = POD_piety_for_perk_interaction_choleric
	}

	send_option = {
		flag = melancholic_perk
		localization = POD_piety_for_perk_interaction_melancholic
	}

	send_option = {
		flag = phlegmatic_perk
		localization = POD_piety_for_perk_interaction_phlegmatic
	}

	send_option = {
		flag = sanguine_perk
		localization = POD_piety_for_perk_interaction_sanguine
	}
	
	send_options_exclusive = yes
	
	cost = { piety = 1000 }

	auto_accept = yes
	ai_frequency = 0
	use_diplomatic_range = no
	
	is_shown = {
		scope:actor = scope:recipient 
		scope:actor = {
			is_ai = no
			is_adult = yes
			has_trait_with_flag = perks_for_piety_flag
		}
	}

	on_accept = {
		scope:actor = {
			if = {
				limit = { scope:animal_perk = yes }
				add_animal_lifestyle_perk_points = 1
			}
			
			if = {
				limit = { scope:choleric_perk = yes }
				add_choleric_lifestyle_perk_points = 1
			}
			
			if = {
				limit = { scope:melancholic_perk = yes }
				add_melancholy_lifestyle_perk_points = 1
			}
			
			if = {
				limit = { scope:phlegmatic_perk = yes }
				add_phlegmatic_lifestyle_perk_points = 1
			}
			
			if = {
				limit = { scope:sanguine_perk = yes }
				add_sanguine_lifestyle_perk_points = 1
			}
		}	
	}
}
