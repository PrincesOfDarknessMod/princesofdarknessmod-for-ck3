
POD_learn_arcanos_interaction = {
	category = interaction_category_wraith
	desc = POD_learn_arcanos_interaction_desc
	icon = icon_culture

	ai_frequency = 0
	
	ai_potential = { always = no }

	is_shown = {		
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		scope:actor = { has_trait = wraith }
		scope:recipient = { has_trait = wraith }
	}
	
	cooldown = { years = 5 } 

	is_valid_showing_failures_only = {
		custom_tooltip = {
			text = "He_must_be_my_friend"
			scope:recipient = { 
				OR = {
					has_relation_lover = scope:actor
					has_relation_friend = scope:actor
					has_relation_best_friend = scope:actor
					has_relation_soulmate = scope:actor
				}
			}
		}
		scope:actor = {
			NOT = { is_at_war_with = scope:recipient }
		}		
	}

	on_accept = {
		scope:actor = {
			custom_tooltip = POD_learn_arcanos_interaction_accept
		}
		hidden_effect = {
			scope:actor = {
				trigger_event = POD_mentoring_disciplines.0002
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_ask_learn_arcanos_interaction = {
	category = interaction_category_wraith
	desc = POD_ask_learn_arcanos_interaction_desc
	icon = icon_culture

	ai_frequency = 0
	
	ai_potential = { always = no }

	is_shown = {
		scope:actor = { num_of_relation_bound_wraith >= 1 }
		scope:recipient = { has_trait = wraith }
	}

	is_valid_showing_failures_only = {
		custom_tooltip = {
			text = "He_must_be_my_friend"
			scope:recipient = { 
				OR = {
					has_relation_lover = scope:actor
					has_relation_friend = scope:actor
					has_relation_best_friend = scope:actor
					has_relation_soulmate = scope:actor
				}
			}
		}
		scope:actor = {
			NOT = { is_at_war_with = scope:recipient }
		}	
		trigger_if = {
			limit = { exists = scope:secondary_actor }
			scope:secondary_actor = { NOT = { has_trait = torpor } }
			POD_cross_umbra_interaction_secondary_valid_trigger = yes
		}	
	}

	populate_actor_list = {
		scope:actor = {
			every_relation = {
				type = bound_wraith
				limit = { NOT = { this = scope:recipient } }
				add_to_list = characters
			}
		}
	}
	
	on_accept = {
		scope:actor = {
			custom_tooltip = POD_ask_learn_arcanos_interaction_accept
		}
		hidden_effect = {
			scope:actor = { save_scope_as = necromancer }
			scope:secondary_actor = { save_scope_as = actor }
			scope:necromancer = {
				trigger_event = POD_mentoring_disciplines.0002
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_mastering_realms_interaction = {
	category = interaction_category_fae
	desc = POD_mastering_realms_interaction_desc
	icon = icon_culture

	ai_frequency = 0
	
	ai_potential = { always = no }
	
	cost = { prestige = massive_prestige_value }

	is_shown = {		
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_POD_fae_blood_trigger = yes }
		scope:recipient = { has_POD_fae_blood_trigger = yes }
		scope:actor = { 
			NOR = { 
				has_trait = realm_actor
				has_trait = realm_fae
				has_trait = realm_prop
				has_trait = realm_nature
				has_trait = realm_scene
				has_trait = realm_time
			}
		}
	}

	is_valid = {
		scope:recipient = { 
			OR = { 
				has_trait = realm_actor
				has_trait = realm_fae
				has_trait = realm_prop
				has_trait = realm_nature
				has_trait = realm_scene
				has_trait = realm_time
			}
		}
	}

	is_valid_showing_failures_only = {
		custom_tooltip = {
			text = "He_must_be_my_friend"
			scope:recipient = { 
				OR = {
					has_relation_lover = scope:actor
					has_relation_friend = scope:actor
					has_relation_best_friend = scope:actor
					has_relation_soulmate = scope:actor
				}
			}
		}
		scope:actor = {
			NOT = { is_at_war_with = scope:recipient }
			NOT = { has_trait = kinain }
		}		
	}
	
	send_option = {
		is_shown = {
			scope:actor = { NOT = { has_trait = realm_actor } }
			scope:recipient = { has_trait = realm_actor }
		}
		flag = realm_actor
		localization = trait_realm_actor
	}
	
	send_option = {
		is_shown = {
			scope:actor = { NOT = { has_trait = realm_fae } }
			scope:recipient = { has_trait = realm_fae }
		}
		flag = realm_fae
		localization = trait_realm_fae
	}
	
	send_option = {
		is_shown = {
			scope:actor = { NOT = { has_trait = realm_prop } }
			scope:recipient = { has_trait = realm_prop }
		}
		flag = realm_prop
		localization = trait_realm_prop
	}
	
	send_option = {
		is_shown = {
			scope:actor = { NOT = { has_trait = realm_nature } }
			scope:recipient = { has_trait = realm_nature }
		}
		flag = realm_nature
		localization = trait_realm_nature
	}
	
	send_option = {
		is_shown = {
			scope:actor = { NOT = { has_trait = realm_scene } }
			scope:recipient = { has_trait = realm_scene }
		}
		flag = realm_scene
		localization = trait_realm_scene
	}
	
	send_option = {
		is_shown = {
			scope:actor = { NOT = { has_trait = realm_time } }
			scope:recipient = { has_trait = realm_time }
		}
		flag = realm_time
		localization = trait_realm_time
	}

	on_accept = {
		scope:actor = {
			add_stress = massive_stress_gain
		}
		if = {
			limit = { scope:realm_actor = yes }
			scope:actor = { add_trait = realm_actor }
		}
		if = {
			limit = { scope:realm_fae = yes }
			scope:actor = { add_trait = realm_fae }
		}
		if = {
			limit = { scope:realm_prop = yes }
			scope:actor = { add_trait = realm_prop }
		}
		if = {
			limit = { scope:realm_nature = yes }
			scope:actor = { add_trait = realm_nature }
		}
		if = {
			limit = { scope:realm_scene = yes }
			scope:actor = { add_trait = realm_scene }
		}
		if = {
			limit = { scope:realm_time = yes }
			scope:actor = { add_trait = realm_time }
		}
		scope:recipient = {
			add_hook = {
				target = scope:actor 
				type = favor_hook
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
