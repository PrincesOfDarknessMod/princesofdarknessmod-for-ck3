POD_season_kingmaker_interaction = {
	category = interaction_category_fae
	icon = POD_fae_interaction
	desc = POD_season_kingmaker_interaction_desc
	use_diplomatic_range = no
	
	cooldown_against_recipient = { years = 10 }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { 
			has_character_modifier = season_kingmaker_modifier 
			can_add_hook = {
				target = scope:recipient
				type = favor_hook
			}
		}
	}

	is_valid_showing_failures_only = {
		trigger_if = { 
			limit = { faith:solstice_court = { has_doctrine = doctrine_winter_time } }
			scope:recipient = { religion = religion:winter_court_religion }
		}
		trigger_if = {
			limit = { faith:solstice_court = { has_doctrine = doctrine_spring_time } }
			scope:recipient = { religion = religion:spring_court_religion }
		}
		trigger_if = { 
			limit = { faith:solstice_court = { has_doctrine = doctrine_summer_time } }
			scope:recipient = { religion = religion:summer_court_religion }
		}
		trigger_if = { 
			limit = { faith:solstice_court = { has_doctrine = doctrine_autumn_time } }
			scope:recipient = { religion = religion:autumn_court_religion }
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}

	on_accept = {
		scope:actor = {
			add_hook = {
				type = favor_hook
				target = scope:recipient
			}
		}
	}
	
	auto_accept = yes
}

POD_convert_to_catlover_interaction = {
	category = interaction_category_fae
	icon = POD_fae_interaction
	desc = POD_convert_to_catlover_interaction_desc
	scheme = convert_to_catlover

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_character_flag = prevent_black_death_route_cat }
	}

	is_valid_showing_failures_only= {
		scope:actor = {
			can_start_scheme = {
				type = convert_to_catlover
				target = scope:recipient
			}
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}

	on_accept = {
		hidden_effect = {
			scope:actor = {
				start_scheme = {
					type = convert_to_catlover
					target = scope:recipient
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_banish_fae_interaction = {
	category = interaction_category_fae
	icon = POD_fae_interaction
	desc = POD_banish_fae_interaction_desc
	
	cooldown = { years = 1 }
	cost = { gold = medium_gold_value }
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { has_trait = fae }
		scope:recipient = { is_in_umbra_trigger = no }
		scope:actor = {
			OR = {
				has_perk = nyarashahims_banishing_sign_perk
				has_perk = banishment_perk
			}
		}
	}
	
	is_valid = {
		scope:recipient = { 
			OR = {
				is_imprisoned_by = scope:actor
				custom_description = {
					object = scope:recipient
					text = "has_full_true_name_hook"
					has_hook_of_type = {
						target = scope:recipient
						type = true_name_hook
					}
				} 
			}
		}
		scope:recipient = { 
			NOT = { has_character_modifier = trapped_in_arcadia_modifier }
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }	
		}
		POD_cross_umbra_interaction_valid_trigger = yes	
	}

	on_accept = {
		scope:recipient = {
			POD_send_to_arcadia_effect = yes
			if = {
				limit = { 
					is_ruler = yes 
					is_ai = yes
				}
				depose = yes
				move_to_pool = yes
				add_trait = torpor
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_changeling_interaction = {
	icon = POD_fae_interaction
	category = interaction_category_fae
	interface_priority = 60

	desc = POD_changeling_interaction_desc
	use_diplomatic_range = yes
	
	is_shown = {
		NOT = { scope:recipient = scope:actor }
		scope:actor = { has_trait = fae }
		scope:recipient = { NOT = { has_trait = fae } }
	}

	is_valid = {
		scope:actor = { any_child = { is_adult = no } }
		scope:recipient = { any_child = { is_adult = no } }
	}

	is_valid_showing_failures_only = {
	}

	populate_actor_list = {
		scope:actor = {
			every_child = {
				limit = { is_adult = no }
				add_to_list = characters
			}
		}
	}

	populate_recipient_list = {
		scope:recipient = {
			every_child = {
				limit = { is_adult = no }
				add_to_list = characters
			}
		}
	}
	
	on_accept = {
		scope:actor = {
			duel = {
				skill = intrigue
				target = scope:recipient
				50 = {
					desc = board_games.0001.tt.success
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					POD_changeling_effect = {
						CHILD = scope:secondary_recipient
						ADOPTER = scope:actor
					}
					POD_changeling_effect = {
						CHILD = scope:secondary_actor
						ADOPTER = scope:recipient
					}
				}	
				50 = {
					desc = board_games.0001.tt.failure
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier 
					}
				}
			}
		}
	}
}
