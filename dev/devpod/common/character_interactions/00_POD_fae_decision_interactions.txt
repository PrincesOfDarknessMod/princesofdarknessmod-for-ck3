POD_season_kingmaker_interaction = {
	category = interaction_category_fae
	icon = POD_fae_interaction
	desc = POD_season_kingmaker_interaction_desc
	use_diplomatic_range = no
	
	cooldown_against_recipient = { years = 10 }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { 
			has_character_modifier = season_kingmaker_modifier 
			can_add_hook = {
				target = scope:recipient
				type = favor_hook
			}
		}
	}

	is_valid_showing_failures_only = {
		trigger_if = { 
			limit = { faith:solstice_court = { has_doctrine = doctrine_winter_time } }
			scope:recipient = { religion = religion:winter_court_religion }
		}
		trigger_if = {
			limit = { faith:solstice_court = { has_doctrine = doctrine_spring_time } }
			scope:recipient = { religion = religion:spring_court_religion }
		}
		trigger_if = { 
			limit = { faith:solstice_court = { has_doctrine = doctrine_summer_time } }
			scope:recipient = { religion = religion:summer_court_religion }
		}
		trigger_if = { 
			limit = { faith:solstice_court = { has_doctrine = doctrine_autumn_time } }
			scope:recipient = { religion = religion:autumn_court_religion }
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}

	on_accept = {
		scope:actor = {
			add_hook = {
				type = favor_hook
				target = scope:recipient
			}
		}
	}
	
	auto_accept = yes
}

	}

	is_valid_showing_failures_only = {

			NOT = { has_trait = torpor }

		}
		POD_cross_umbra_interaction_valid_trigger = yes	
	}

	on_accept = {

				limit = { 
					is_ruler = yes 
					is_ai = yes

				move_to_pool = yes
				add_trait = torpor
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_changeling_interaction = {
	category = interaction_category_fae
	icon = POD_fae_interaction
	desc = POD_changeling_interaction_desc
	
	cooldown = { years = 10 }
	
	is_shown = {
		NOT = { scope:recipient = scope:actor }
		scope:actor = { has_trait = fae }
		scope:recipient = { NOT = { has_trait = fae } }
	}

	is_valid = {
		scope:actor = { any_child = { is_adult = no } }
		scope:recipient = { any_child = { is_adult = no } }
	}

	is_valid_showing_failures_only = {
	}

	populate_actor_list = {
		scope:actor = {
			every_child = {
				limit = { is_adult = no }
				add_to_list = characters
			}
		}
	}

	populate_recipient_list = {
		scope:recipient = {
			every_child = {
				limit = { is_adult = no }
				add_to_list = characters
			}
		}
	}
	
	on_accept = {
		scope:actor = {
			duel = {
				skill = intrigue
				target = scope:recipient
				50 = {
					desc = board_games.0001.tt.success
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					POD_changeling_effect = {
						CHILD = scope:secondary_recipient
						ADOPTER = scope:actor
					}
					POD_changeling_effect = {
						CHILD = scope:secondary_actor
						ADOPTER = scope:recipient
					}
				}	
				50 = {
					desc = board_games.0001.tt.failure
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier 
					}
				}
			}
		}
	}
}

POD_create_changeling_interaction = {
	category = interaction_category_fae
	icon = POD_fae_interaction
	desc = POD_create_changeling_interaction_desc
	
	cooldown = { years = 10 }
	
	cost = { gold = scope:actor.medium_gold_value }
	
	is_shown = {
		NOT = { scope:recipient = scope:actor }
		scope:actor = { has_trait = fae }
	}

	is_valid = {
		scope:recipient = { has_relation_dreamer_owner = scope:actor }
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			is_imprisoned = no
			is_at_war = no
		}
		scope:recipient = {
			is_adult = yes
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
			NOT = { has_trait = changeling }
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}

	on_accept = {
		scope:recipient = {
			POD_become_changeling_effect = yes
		}
	}
	
	ai_min_reply_days = 0
	ai_max_reply_days = 0
	ai_accept = {
		base = -80
		
		opinion_modifier = { # Opinion Factor
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 1
			desc = AI_OPINION_REASON
		}
		
		modifier = {
			desc = CHANGELING_IS_SHUNNED
			add = { 
				value = -10
				if = {
					limit = { has_trait = zealous }
					multiply = 2
				}
				else_if = {
					limit = { has_trait = cynical }
					multiply = 0.5
				}
			}
			scope:recipient.faith = { has_doctrine = doctrine_glamour_shunned }
		}
		
		modifier = {
			desc = CHANGELING_IS_CRIME
			add = { 
				value = -20
				if = {
					limit = { has_trait = zealous }
					multiply = 2
				}
				else_if = {
					limit = { has_trait = cynical }
					multiply = 0.5
				}
			}
			scope:recipient.faith = { has_doctrine = doctrine_glamour_crime }
		}

		# Rivalry
		modifier = { # Say no to rivals
			add = -25
			scope:recipient = {
				has_relation_rival = scope:actor
			}
			desc = "ACTOR_RIVAL_TO_ME_REASON"
		}
		modifier = { # Say no to nemesis
			add = -50
			scope:recipient = {
				has_relation_nemesis = scope:actor
			}
			desc = "ACTOR_NEMESIS_TO_ME_REASON"
		}

		# Feud
		modifier = {
			add = -25
			scope:recipient = {
				existing_feud_against_target_trigger = { TARGET = scope:actor }
			}
			desc = MARRY_FAMILY_FEUD_REASON
		}
	}
}

POD_call_patron_interaction = {
	category = interaction_category_werewolf
	common_interaction = yes
	desc = POD_call_patron_interaction_desc
	interface_priority = 60
	interface = call_ally
	special_interaction = call_ally_interaction
	popup_on_receive = yes
	pause_on_receive = yes
	icon = POD_fae_interaction

	greeting = positive
	notification_text = CALL_ALLY_NOTIFICATION

	is_shown = {
		scope:actor = { is_at_war = yes }
		scope:actor = {
			OR = {
				has_character_modifier = seeker_membership_modifier
				has_character_modifier = vanguard_membership_modifier
				has_character_modifier = austere_conclave_membership_modifier
			}
		}
		scope:actor = { POD_splat_has_gnosis_trigger = yes }
	}
	
	has_valid_target = {
		exists = scope:target
	}	

	has_valid_target_showing_failures_only = {
		scope:target = {
			is_war_leader = scope:actor
		}

		joiner_not_already_in_another_war_with_any_target_war_participants_trigger = {
			WARRIOR = scope:actor
			JOINER = scope:recipient
		}
	}
	
	is_valid = {
		trigger_if = {
			limit = { 
				scope:actor = {
					OR = {
						has_character_modifier = philospher_queen_modifier
						has_character_modifier = philospher_king_modifier
					}
				}
			}
			scope:recipient = {
				OR = {
					has_character_modifier = seeker_membership_modifier
					has_character_modifier = vanguard_membership_modifier
					has_character_modifier = austere_conclave_membership_modifier
				}
			}
		}
		trigger_else_if = {
			limit = { scope:actor { has_character_modifier = austere_conclave_membership_modifier } }
			scope:recipient = { has_character_modifier = austere_conclave_membership_modifier }
		}
		trigger_else_if = {
			limit = { scope:actor { has_character_modifier = seeker_membership_modifier } }
			scope:recipient = { has_character_modifier = seeker_membership_modifier }
		}
		trigger_else = {
			scope:recipient = { has_character_modifier = vanguard_membership_modifier }
		}
		scope:target = {
			trigger_if = {
				limit = { casus_belli.primary_attacker = scope:actor }
				casus_belli.primary_defender = { 
					NOR = {
						has_character_modifier = seeker_membership_modifier
						has_character_modifier = vanguard_membership_modifier
						has_character_modifier = austere_conclave_membership_modifier
					} 
				}
			}
			trigger_if = {
				limit = { casus_belli.primary_defender = scope:actor }
				casus_belli.primary_attacker = { 
					NOR = {
						has_character_modifier = seeker_membership_modifier
						has_character_modifier = vanguard_membership_modifier
						has_character_modifier = austere_conclave_membership_modifier
					} 
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			is_ruler = yes
		}

		# Only War Leaders can call allies
		scope:actor = {
			custom_description = {
				text = must_be_war_leader
				any_character_war = {
					is_war_leader = scope:actor
				}
			}
		}

		# Cannot call the same character multiple times
		scope:actor = {
			custom_description = {
				text = cannot_call_same_character_multiple_times
				subject = scope:recipient
				any_character_war = {
					ALL_FALSE = {
						was_called = scope:recipient
						is_participant = scope:recipient
					}
				}
			}
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}

	can_be_picked = {
		can_join_war_liege_vassal_check_trigger = { WARRIOR = scope:actor JOINER = scope:recipient }

		scope:target = {
			is_war_leader = scope:actor

			using_cb = POD_blood_hunt_war
			
			# not already in target war
			custom_description = {
				text = join_war_interaction_already_in_target_war
				NOR = {
					any_war_attacker = { this = scope:recipient }
					any_war_defender = { this = scope:recipient }
				}
			}
		}
	}

	on_auto_accept = {
		scope:target = {
			if = {
				limit = {
					scope:recipient = {
						any_active_accolade = {
							has_accolade_parameter = acclaimed_knight_piety_from_battle
						}
					}
				}
				scope:recipient = {
					add_piety = miniscule_piety_gain
				}
			}
			if = {
				limit = {
					scope:recipient = {
						any_active_accolade = {
							has_accolade_parameter = acclaimed_knight_piety_from_battle_high
						}
					}
				}
				scope:recipient = {
					add_piety = minor_piety_gain
				}
			}
		}
	}
		
	on_accept = {
		blood_hunt_call_interaction_effect = yes
		scope:target = {
			if = {
				limit = {
					scope:recipient = {
						any_active_accolade = {
							has_accolade_parameter = acclaimed_knight_piety_from_battle
						}
					}
				}
				scope:recipient = {
					add_piety = minor_piety_gain
				}
			}
			if = {
				limit = {
					scope:recipient = {
						any_active_accolade = {
							has_accolade_parameter = acclaimed_knight_piety_from_battle_high
						}
					}
				}
				scope:recipient = {
					add_piety = medium_piety_gain
				}
			}
		}
	}
	
	auto_accept = yes
}
