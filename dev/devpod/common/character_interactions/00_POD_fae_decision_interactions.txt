POD_season_kingmaker_interaction = {
	category = interaction_category_fae
	icon = POD_fae_interaction
	desc = POD_season_kingmaker_interaction_desc
	use_diplomatic_range = no
	
	cooldown_against_recipient = { years = 10 }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { 
			has_character_modifier = season_kingmaker_modifier 
			can_add_hook = {
				target = scope:recipient
				type = favor_hook
			}
		}
	}

	is_valid_showing_failures_only = {
		trigger_if = { 
			limit = { faith:solstice_court = { has_doctrine = doctrine_winter_time } }
			scope:recipient = { religion = religion:winter_court_religion }
		}
		trigger_if = {
			limit = { faith:solstice_court = { has_doctrine = doctrine_spring_time } }
			scope:recipient = { religion = religion:spring_court_religion }
		}
		trigger_if = { 
			limit = { faith:solstice_court = { has_doctrine = doctrine_summer_time } }
			scope:recipient = { religion = religion:summer_court_religion }
		}
		trigger_if = { 
			limit = { faith:solstice_court = { has_doctrine = doctrine_autumn_time } }
			scope:recipient = { religion = religion:autumn_court_religion }
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}

	on_accept = {
		scope:actor = {
			add_hook = {
				type = favor_hook
				target = scope:recipient
			}
		}
	}
	
	auto_accept = yes
}

POD_convert_to_catlover_interaction = {
	category = interaction_category_fae
	icon = POD_fae_interaction
	desc = POD_convert_to_catlover_interaction_desc
	scheme = convert_to_catlover

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_character_flag = prevent_black_death_route_cat }
	}

	is_valid_showing_failures_only= {
		scope:actor = {
			can_start_scheme = {
				type = convert_to_catlover
				target = scope:recipient
			}
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}

	on_accept = {
		hidden_effect = {
			scope:actor = {
				start_scheme = {
					type = convert_to_catlover
					target = scope:recipient
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
