POD_learn_true_name_interaction = {
	category = interaction_category_hostile
	icon = POD_find_mortals_interaction
	desc = POD_learn_true_name_interaction_desc

	cooldown = { years = 5 }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { POD_can_learn_true_name_trigger = yes }
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			POD_not_already_on_journey_trigger = {
				TYPE   = learn_true_name
				TARGET = scope:recipient
			}
			POD_can_start_planning_phase_trigger = yes
		}
		scope:recipient = { POD_immune_against_learning_true_name_trigger = no }
	}
	
	on_accept = {
		scope:actor = { custom_tooltip = POD_learn_true_name_interaction_tt }
		
		POD_journey_interaction_pause_hint = yes
		
		hidden_effect = {
			scope:actor = {
				POD_start_targeted_journey_effect = {
					TYPE    = learn_true_name
					MONTHLY = learn_true_name
					TARGET  = scope:recipient
				}
			}
		}
	}

	auto_accept = yes
}
