POD_deny_the_hungry_perk_interaction = {
	category = interaction_category_werewolf
	icon = POD_bastet_interaction
	desc = POD_deny_the_hungry_perk_interaction_desc
	
	target_type = title
	target_filter = recipient_domain_titles
	
	cost = { piety = 500 }
	
	auto_accept = yes
	
	can_be_picked = {
		scope:target = { tier = tier_county }
	}

	is_shown = {
		scope:actor = { POD_splat_has_rage_trigger = yes }
		scope:actor = { has_perk = deny_the_hungry_perk }
		scope:recipient = { is_landed = yes }
		scope:recipient.highest_held_title_tier > tier_barony
	}

	is_valid_showing_failures_only = {
		scope:actor = { 
			NOT = { has_character_modifier = POD_rage_0_modifier }

	   scope:actor = {
			POD_willpower_rouse_check_effect = yes
			POD_rage_loss_check_effect = yes
	   }
	   scope:target = { 
			add_county_modifier = {
				modifier = deny_the_hungry_perk_modifier
				years = 5
			}
		}
    }
	
	ai_frequency = 360
	
	ai_targets = {
		ai_recipients = primary_war_enemies
	}
	
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 45
			scope:actor = {
			    is_at_war_with = scope:recipient
			}
			piety > 1500
		}
		modifier = {
		    add = 10
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
	}
