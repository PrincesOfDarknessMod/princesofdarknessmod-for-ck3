POD_unseas_blessing_interaction = {
	category = interaction_category_werewolf
	icon = POD_rokea_interaction
	desc = POD_unseas_blessing_interaction_desc
	
	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { POD_splat_has_rage_trigger = yes }
		scope:actor = { has_perk = unseas_blessing_perk }
	}
	
	is_valid = {
		scope:recipient = {
			OR = {
				has_trait = scarred
				has_trait = disfigured
				has_trait = wounded
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			POD_can_be_healed_trigger = yes
		}
	}
	
	on_accept = {
		scope:actor = { 
			POD_willpower_rouse_check_effect = yes
		}
		scope:recipient = {
			play_music_cue = "mx_cue_peace_ensues"
			
			remove_scar_trait_effect = yes
			POD_remove_wound_trait_effect = yes
		}
	}
	
	auto_accept = yes

	ai_potential = {
		# The basic requirements
		is_imprisoned = no
		is_landed = yes
	}
	ai_targets = {
		ai_recipients = self
	}
	ai_frequency = 12
	ai_will_do = {
		base = 100
	}
}

POD_channel_rage_interaction = {
	category = interaction_category_werewolf
	icon = POD_rokea_interaction
	desc = POD_channel_rage_interaction_desc
	options_heading = POD_channel_rage_interaction_options_heading
	
	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { POD_splat_has_rage_trigger = yes }
		scope:actor = { has_perk = mindless_fight_perk }
	}
	
	is_valid_showing_failures_only = {
		scope:actor = { NOT = { has_character_modifier = POD_rage_0_modifier } }
	}
	
	send_option = {
		is_shown = {
			scope:actor = { rage_level >= 1 }
		}
		flag = POD_num1
		localization = POD_num1
	}
	
	send_option = {
		is_shown = {
			scope:actor = { rage_level >= 2 }
		}
		flag = POD_num2
		localization = POD_num2
	}
	
	send_option = {
		is_shown = {
			scope:actor = { rage_level >= 3 }
		}
		flag = POD_num3
		localization = POD_num3
	}
	
	send_option = {
		is_shown = {
			scope:actor = { rage_level >= 4 }
		}
		flag = POD_num4
		localization = POD_num4
	}
	
	send_option = {
		is_shown = {
			scope:actor = { rage_level >= 5 }
		}
		flag = POD_num5
		localization = POD_num5
	}

	on_accept = {
		scope:actor = {
			if = {
				limit = { scope:POD_num5 = yes }
				POD_rage_decrease_multiple_effect = { COUNT = 5 }
				while = {
					count = 5
					add_character_modifier = {
						modifier = mindless_fight_perk_modifier
						years = 1
					}
				}
			}
			else_if = {
				limit = { scope:POD_num4 = yes }
				POD_rage_decrease_multiple_effect = { COUNT = 4 }
				while = {
					count = 4
					add_character_modifier = {
						modifier = mindless_fight_perk_modifier
						years = 1
					}
				}
			}
			else_if = {
				limit = { scope:POD_num3 = yes }
				POD_rage_decrease_multiple_effect = { COUNT = 3 }
				while = {
					count = 3
					add_character_modifier = {
						modifier = mindless_fight_perk_modifier
						years = 1
					}
				}
			}
			else_if = {
				limit = { scope:POD_num2 = yes }
				POD_rage_decrease_multiple_effect = { COUNT = 2 }
				while = {
					count = 2
					add_character_modifier = {
						modifier = mindless_fight_perk_modifier
						years = 1
					}
				}
			}
			else = {
				POD_rage_decrease_effect = yes
				while = {
					count = 1
					add_character_modifier = {
						modifier = mindless_fight_perk_modifier
						years = 1
					}
				}
			}
		}
	}
	
	auto_accept = yes
}
