POD_psychomachia_interaction = {
	category = interaction_category_vampire
	#common_interaction = yes
	#interface_priority = 88
	icon = POD_daimonion_interaction
	desc = POD_psychomachia_interaction_desc
	
	is_shown = {
		#NOT = { scope:actor = scope:recipient }
		scope:actor = { has_daimonion_trigger = yes }
		scope:actor = { has_perk = psychomachia_perk }
		scope:recipient = { 
			NOT = {
				has_character_flag = psychomachia_victim
			}
		}
	}
	
	is_valid = {
		scope:recipient = {
			OR = {
				has_character_modifier = amplified_lustful_negative_modifier     
						
				has_character_modifier = amplified_gluttonous_negative_modifier     
							
				has_character_modifier = amplified_greedy_negative_modifier     
							
				has_character_modifier = amplified_lazy_negative_modifier     
							
				has_character_modifier = amplified_wrathful_negative_modifier     
							
				has_character_modifier = amplified_arrogant_negative_modifier     
							
				has_character_modifier = amplified_deceitful_negative_modifier     
							
				has_character_modifier = amplified_craven_negative_modifier     
							
				has_character_modifier = amplified_arbitrary_negative_modifier 
			}
		}
	}
	
	is_valid_showing_failures_only = {
		scope:recipient = { 
			NOT = { has_trait = torpor }
			NOT = { has_trait = automaton }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }	
			POD_is_not_clone_trigger = yes
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}
	
	auto_accept = yes
	
	send_option = {
		flag = normal								
		localization = 	POD_psychomachia_interaction_normal						
		starts_enabled = { always = yes }															
	}
	send_option = {
		is_shown = { 
			scope:actor = { 
				OR = {
					has_trait = daimonionadvanced
					has_perk = face_hell_perk 
				} 
			}
		}
		flag = morepower							
		localization = 	POD_psychomachia_interaction_morepower																				
	}
	send_option = {
		is_shown = { 
			scope:actor = {
				OR = {
					has_trait = daimonionadvanced
					has_perk = face_hell_perk 
				}
			}
		}
		flag = lesspower								
		localization = 	POD_psychomachia_interaction_lesspower																			
	}
	
	on_accept = {
		scope:actor = {
			if = {
				limit = {
					is_infernalist_trigger = no
				}
				add_piety = -500
				custom_tooltip = POD_non_infernalists_pay
			}
			else = {
				custom_tooltip = POD_infernalists_dont_pay
			}
		}
		custom_tooltip = POD_psychomachia_interaction_ct
		scope:recipient = {
			POD_create_manifestation_effect = yes
		}
		if = {
			limit = {
				exists = scope:recipient
				exists = scope:manifestation
			}
			scope:manifestation = {
				POD_remove_vampire_discipline_traits_effect = yes
				if = {
					limit = { scope:recipient = { has_character_modifier =  amplified_lustful_negative_modifier } }
					if = {
						limit = { scope:lesspower = yes }
						add_trait = presence
					}
					else_if = {
						limit = { scope:morepower = yes }
						add_trait = presenceadvanced
						add_trait = potenceadvanced
					}
					else = {
						add_trait = presenceadvanced
					}
				}

				if = {
					limit = { scope:recipient = { has_character_modifier =  amplified_gluttonous_negative_modifier } }
					if = {
						limit = { scope:lesspower = yes }
						add_trait = fortitude
					}
					else_if = {
						limit = { scope:morepower = yes }
						add_trait = fortitudeadvanced
						add_trait = obtenebrationadvanced
					}
					else = {
						add_trait = fortitudeadvanced
					}
				}

				if = {
					limit = { scope:recipient = { has_character_modifier =  amplified_greedy_negative_modifier } }
					if = {
						limit = { scope:lesspower = yes }
						add_trait = daimonion
					}
					else_if = {
						limit = { scope:morepower = yes }
						add_trait = daimonionadvanced
						add_trait = auspexadvanced
					}
					else = {
						add_trait = daimonion
					}
				}

				if = {
					limit = { scope:recipient = { has_character_modifier =  amplified_lazy_negative_modifier } }
					if = {
						limit = { scope:lesspower = yes }
						add_trait = auspex
					}
					else_if = {
						limit = { scope:morepower = yes }
						add_trait = auspexadvanced
						add_trait = presenceadvanced
					}
					else = {
						add_trait = auspexadvanced
					}
				}

				if = {
					limit = { scope:recipient = { has_character_modifier =  amplified_wrathful_negative_modifier } }
					if = {
						limit = { scope:lesspower = yes }
						add_trait = potence
					}
					else_if = {
						limit = { scope:morepower = yes }
						add_trait = potenceadvanced
						add_trait = celerityadvanced
					}
					else = {
						add_trait = potenceadvanced
					}
				}

				if = {
					limit = { scope:recipient = { has_character_modifier =  amplified_arrogant_negative_modifier } }
					if = {
						limit = { scope:lesspower = yes }
						add_trait = celerity
					}
					else_if = {
						limit = { scope:morepower = yes }
						add_trait = celerityadvanced
						add_trait = dominateadvanced
					}
					else= {
						add_trait = celerityadvanced
					}
				}

				if = {
					limit = { scope:recipient = { has_character_modifier =  amplified_deceitful_negative_modifier } }
					if = {
						limit = { scope:lesspower = yes }
						add_trait = obtenebration
					}
					else_if = {
						limit = { scope:morepower = yes }
						add_trait = obtenebrationadvanced
						add_trait = obfuscateadvanced
					}
					else= {
						add_trait = obtenebrationadvanced
					}
				}

				if = {
					limit = { scope:recipient = { has_character_modifier =  amplified_craven_negative_modifier } }
					if = {
						limit = { scope:lesspower = yes }
						add_trait = obfuscate
					}
					else_if = {
						limit = { scope:morepower = yes }
						add_trait = obfuscateadvanced
						add_trait = fortitudeadvanced
					}
					else = {
						add_trait = obfuscateadvanced
					}
				}

				if = {
					limit = { scope:recipient = { has_character_modifier =  amplified_arbitrary_negative_modifier } }
					if = {
						limit = { scope:lesspower = yes }
						add_trait = protean
					}
					else_if = {
						limit = { scope:morepower = yes }
						add_trait = proteanadvanced
						add_trait = dominateadvanced
					}
					else = {
						add_trait = proteanadvanced
					}
				}
				
				add_prowess_skill = -300
				add_prowess_skill = 138
				
				POD_character_vampire_lifestyle_initialization_effect = yes
			}
			scope:actor = { trigger_event = { id = POD_daimonion_events.5001 days = 1 } }
		}
	}
}
