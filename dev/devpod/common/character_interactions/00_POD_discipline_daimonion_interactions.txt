POD_sense_the_sin_interaction = {
	category = interaction_category_vampire
	icon = POD_daimonion_interaction
	desc = POD_sense_the_sin_interaction_desc

	ai_frequency = 0

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_perk = sense_the_sin_perk }
	}

	is_valid_showing_failures_only = {
		scope:actor = { POD_has_enough_blood_for_discipline = { AMOUNT = 3 } }
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}

	cooldown = { years = 1 }

	on_accept = {
		scope:actor = {
			POD_discipline_cost_effect = { COUNT = 3 }
			custom_tooltip = {
				text = POD_sense_the_sin_interaction_effect
				scope:recipient = {
					if = { limit = { any_secret = { NOT = { is_known_by = scope:actor } } }
						random_secret = {
							limit = { NOT = { is_known_by = scope:actor } }
							scope:actor = {
								send_interface_message = {
									type = event_truce_neutral
									title = POD_shared_loca_success
									PREV = { reveal_to = scope:actor }
								}
							}
						}
					}
					else = {
						scope:actor = {
							send_interface_message = {
								type = event_truce_neutral
								title = POD_shared_loca_failure
							}
						}
					}
				}
			}
		}
	}

	auto_accept = yes

	ai_will_do = {
		base = 0
	}
}

POD_bring_the_plague_interaction = {
	category = interaction_category_vampire
	icon = POD_daimonion_interaction
	desc = POD_bring_the_plague_interaction_desc
	cooldown = { years = 20 }

	send_option = {
		flag = minor
		localization = spawn_epidemic_interaction
	}
	send_option = {
		flag = major
		localization = spawn_major_epidemic_interaction
	}
	send_option = {
		flag = apocalyptic
		localization = spawn_apocalyptic_epidemic_interaction
	}

	is_shown = {
		scope:actor = { has_perk = bring_the_plague_perk }
		scope:recipient = {
			exists = location
			location = { is_sea_province = no }
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = { POD_has_enough_blood_for_discipline = { AMOUNT = 5 } }
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}
	auto_accept = yes

	on_accept = {
		scope:actor = {
			POD_commit_heinous_act_effect = yes
			POD_discipline_cost_effect = { COUNT = 5 }
			if = { 
				limit = { is_infernalist_trigger = no }
				add_piety = -1000
				custom_tooltip = POD_non_infernalists_pay
			}
			duel = {
				skill = intrigue
				value = scope:recipient.location.epidemic_resistance
				40 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					custom_tooltip = POD_bring_the_plague_interaction_success
					scope:recipient.location = { 
						switch = {
							trigger = yes
							scope:minor = {
								create_epidemic_outbreak = { type = demonic_plague intensity = minor } 
							}
							scope:major = {
								create_epidemic_outbreak = { type = demonic_plague intensity = major }
								
							}
							scope:apocalyptic = {
								create_epidemic_outbreak = { type = demonic_plague intensity = apocalyptic }
							}
						}
					}
					hidden_effect = {
						send_interface_toast = {
							title = POD_bring_the_plague_interaction_success
							left_icon = scope:actor
							right_icon = scope:recipient
							custom_tooltip = POD_bring_the_plague_interaction_success.desc
						}
					}

				}
				60 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier
					}
					custom_tooltip = POD_bring_the_plague_interaction_failure
					hidden_effect = {
						send_interface_toast = {
							title = POD_bring_the_plague_interaction_failure
							left_icon = scope:actor
							right_icon = scope:recipient
							custom_tooltip = POD_bring_the_plague_interaction_failure.desc
						}
					}
				}
			}
		}
	}
}

					POD_character_vampire_clan_special_effect = yes
					POD_set_resonance_effect = yes	
				}
			}
