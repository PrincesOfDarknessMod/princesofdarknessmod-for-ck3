
POD_gangrel_challenge_interaction = {
	category = interaction_category_vampire
	#common_interaction = yes
	icon = icon_dynasty
	desc = POD_gangrel_challenge_interaction_desc
	
	cooldown_against_recipient = { years = 50 } 
	cooldown = { years = 5 }
	
	is_shown = {
		scope:actor = { 
			is_ai = no
			is_ruler = yes
		}		
		scope:actor = { has_trait = gangrel }
		scope:recipient = { has_trait = gangrel }
		scope:recipient = { NOT = { is_close_family_of = scope:actor } }
		NOT = { scope:actor = scope:recipient }
		scope:actor.dynasty = {
		    has_dynasty_perk = gangrel_legacy_4
		}
	
	}
	
	is_valid_showing_failures_only = {
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }	
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}
	
	on_accept = {
		
		custom_tooltip = gangrel_challenge_tt
		scope:actor = {
			trigger_event = podgangrel.1
		}
	}
	
	auto_accept = yes

}

POD_cappadocian_send_advisor_interaction = {
	category = interaction_category_vampire
	#common_interaction = yes
	#interface_priority = 88
	icon = icon_dynasty
	desc = POD_cappadocian_send_advisor_interaction_desc
	cooldown = { months = 24 }
	
	populate_recipient_list = {
		scope:actor = {
			every_child = {
				limit = {
					is_courtier_of = scope:actor
				}
				add_to_list = characters
			
			}
		}
	}
	
	is_shown = {
		scope:actor = { has_trait = cappadocian }
		NOT = { scope:recipient = { has_trait = cappadocian } }
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_ruler = yes }
		scope:actor = {
			any_child = {
				is_courtier_of = scope:actor
			}
		}
		scope:actor.dynasty = {
		    has_dynasty_perk = cappadocian_legacy_1
		}
		
	}
	
	send_option = {
		is_shown = { 
			exists = scope:secondary_recipient
		}
		is_valid = {
			always = yes
		}
		flag = diplomacy
		localization =  POD_cappadocian_send_advisor_interaction_chancellor_option
	
	}
	
	send_option = {
		is_shown = { 
			exists = scope:secondary_recipient
		}
		is_valid = {
			always = yes
		}
		flag = stewardship
		localization =  POD_cappadocian_send_advisor_interaction_steward_option
	
	}
	
	send_option = {
		is_shown = { 
			exists = scope:secondary_recipient
		}
		is_valid = {
			always = yes
		}
		flag = martial
		localization = POD_cappadocian_send_advisor_interaction_marshal_option
	
	}
	
	send_option = {
		is_shown = { 
			exists = scope:secondary_recipient
		}
		is_valid = {
			always = yes
		}
		flag = intrigue
		localization =  POD_cappadocian_send_advisor_interaction_spymaster_option
	
	}
	
	send_option = {
		is_shown = { 
			exists = scope:secondary_recipient
		}
		is_valid = {
			always = yes
		}
		flag = learning
		localization =  POD_cappadocian_send_advisor_interaction_court_chaplain_option
	
	}
	
	is_valid_showing_failures_only = {
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }	
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}
	
	ai_accept = {
		base = 5
		opinion_modifier = { # More likely to accept if Recipient likes the Actor
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.1
			desc = AI_OPINION_REASON
		}
		modifier = {
			add = 5
			scope:recipient = {
				has_dread_level_towards = {
					target = scope:actor
					level = 1
				}
			}
			desc = INTIMIDATED_REASON
		}
		modifier = {
			add = 10
			scope:recipient = {
				has_dread_level_towards = {
					target = scope:actor
					level = 2
				}
			}
			desc = COWED_REASON
		}
		modifier = {
			add = 10
			OR = {
				AND = {
					scope:diplomacy = yes
					scope:recipient = { 
						OR = { 
							NOT = { exists = cp:councillor_chancellor }
							scope:secondary_recipient.diplomacy > cp:councillor_chancellor.diplomacy 
						}
					}
				}
				AND = {
					scope:stewardship = yes
					scope:recipient = { 
						OR = { 
							NOT = { exists = cp:councillor_steward }
							scope:secondary_recipient.stewardship > cp:councillor_steward.stewardship 
						}
					}
				}
				AND = {
					scope:martial = yes
					scope:recipient = { 
						OR = { 
							NOT = { exists = cp:councillor_marshal }
							scope:secondary_recipient.martial > cp:councillor_marshal.martial 
						}
					}
				}
				AND = {
					scope:intrigue = yes
					scope:recipient = { 
						OR = { 
							NOT = { exists = cp:councillor_spymaster }
							scope:secondary_recipient.intrigue > cp:councillor_spymaster.intrigue 
						}
					}
				}
				AND = {
					scope:learning = yes
					scope:recipient = { 
						OR = { 
							NOT = { exists = cp:councillor_court_chaplain }
							scope:secondary_recipient.learning > cp:councillor_court_chaplain.learning 
						}
					}
				}
			}
			desc = BETTER_CHANC
		}
	}
	
	
	on_accept = {
		scope:recipient = {
			add_courtier = scope:secondary_recipient
			if = {
				limit = {
					scope:diplomacy = yes
				}
				if = {
					limit = {
						exists = cp:councillor_chancellor
					}
					fire_councillor = cp:councillor_chancellor
				}
				assign_councillor_type = {
					type = councillor_chancellor
					target = scope:secondary_recipient
				}
				scope:secondary_recipient = { add_character_flag = diplomacy_job_flag }
			}
			if = {
				limit = {
					scope:stewardship = yes
				}
				if = {
					limit = {
						exists = cp:councillor_steward
					}
					fire_councillor = cp:councillor_steward
				}
				assign_councillor_type = {
					type = councillor_steward
					target = scope:secondary_recipient
				}
				scope:secondary_recipient = { add_character_flag = steward_job_flag }
			}
			if = {
				limit = {
					scope:martial = yes
				}
				if = {
					limit = {
						exists = cp:councillor_marshal
					}
					fire_councillor = cp:councillor_marshal
				}
				assign_councillor_type = {
					type = councillor_marshal
					target = scope:secondary_recipient
				}
				scope:secondary_recipient = { add_character_flag = marshal_job_flag }
			}
			if = {
				limit = {
					scope:intrigue = yes
				}
				if = {
					limit = {
						exists = cp:councillor_spymaster
					}
					fire_councillor = cp:councillor_spymaster
				}
				assign_councillor_type = {
					type = councillor_spymaster
					target = scope:secondary_recipient
				}
				scope:secondary_recipient = { add_character_flag = spymaster_job_flag }
			}
			if = {
				limit = {
					scope:learning = yes
				}
				if = {
					limit = {
						exists = cp:councillor_court_chaplain
					}
					fire_councillor = cp:councillor_court_chaplain
				}
				assign_councillor_type = {
					type = councillor_court_chaplain
					target = scope:secondary_recipient
				}
				scope:secondary_recipient = { add_character_flag = chaplain_job_flag }
			}
	
		}
		custom_tooltip = POD_cappadocian_send_advisor_interaction_ct
		scope:secondary_recipient = { 
			trigger_event = { id = POD_cappadocian_power.1000  years = 1 } 
			set_variable = { name = block_fire_councillor value = scope:recipient }
		}
	}
	
	
	auto_accept = {
		OR = {
			AND = {
				scope:diplomacy = yes
				NOT = { scope:recipient = { exists = cp:councillor_chancellor } }
			}
			
			AND = {
				scope:stewardship = yes
				NOT = { scope:recipient = { exists = cp:councillor_steward } }
			}
			
			AND = {
				scope:martial = yes
				NOT = { scope:recipient = { exists = cp:councillor_marshal } }
			}
			
			AND = {
				scope:intrigue = yes
				NOT = { scope:recipient = { exists = cp:councillor_spymaster } }
			}
			
			AND = {
				scope:learning = yes
				NOT = { scope:recipient = { exists = cp:councillor_court_chaplain } }
			}
		
		}
	}
	
}

POD_ventrue_succor_interaction = {

	category = interaction_category_vampire
	#common_interaction = yes
	#interface_priority = 88
	icon = icon_dynasty
	desc = POD_ventrue_succor_interaction_desc
	
	cost = {
		prestige = 100
	}
	
	cooldown = { months = 12 }
	
	is_shown = {
		scope:actor = { has_trait = ventrue }
		scope:recipient = { has_trait = ventrue }
		NOT = {
			scope:actor  = {
				
				any_war_enemy = {
					scope:recipient = this
				}
			
			}
		}
		NOT = { scope:actor = scope:recipient }
		scope:actor.dynasty = {
		    has_dynasty_perk = ventrue_legacy_1
		}
		
	}
	
	is_valid_showing_failures_only = {	 
		NOT = {
			scope:actor = {
				has_hook = scope:recipient
			
			}
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }	
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}
	
	auto_accept = yes
	on_accept = {
		scope:actor = { add_hook = { target = scope:recipient type = minorboon_hook } }
		if = {
			limit = { 
				scope:recipient = { has_hook = scope:actor } 
			
			}
			scope:recipient = { add_hook = { target = scope:actor type = majorboon_hook } }
		}
		else = {
			scope:recipient = { add_hook = { target = scope:actor type = minorboon_hook } }
		}
		scope:actor = {
			stress_impact = {
				arrogant = major_stress_impact_gain
				paranoid = minor_stress_impact_gain
			}
		}
		custom_tooltip = POD_ventrue_succor_interaction_desc
		
	
	}
	
	

}
