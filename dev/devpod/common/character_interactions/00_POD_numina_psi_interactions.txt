				if = { limit = { scope:cockroach = yes } add_character_modifier = { modifier = prison_pet_cockroach_modifier years = 10 } }

			if = { limit = { scope:summon_rat_plague = yes }

				}

					skill = diplomacy
					target = scope:recipient

						POD_mesmerize_attacker_tt_effect = yes
						modifier = { add = psychic_add_success_chance }
						min = 0
						send_interface_toast = {
							title = POD_numina_psychic_animal_psi_interaction
							scope:recipient = {
								capital_county = {
									add_county_modifier = { modifier = psi_ratocalispe years = 10 }
									title_province = {
										random = {
											chance = 50
											create_epidemic_outbreak = { type = consumption intensity = major }
										}
									}
								}
							}
						}
					}
					50 = { POD_mesmerize_defender_modifier = yes 
						POD_mesmerize_defender_tt_effect = yes
						min = 0
						reverse_add_opinion = {
							modifier = attempted_trickery_opinion
							target = scope:recipient
							opinion = -30
						}
					}
				}
			}
		}
	}
	auto_accept = yes

POD_numina_psychic_biocontrol_interaction = {
	category = interaction_category_psychic
	common_interaction = yes
	icon = POD_psi_interaction
	desc = POD_numina_psychic_biocontrol_interaction_desc	
	cooldown = { months = 6 }
	
	is_shown = { 
		scope:actor = { 
			has_perk = biocontrol_2_perk
			OR = {
				has_trait = scarred
				has_trait = disfigured
				has_trait = wounded
				has_any_disease_trigger = yes
			} 
		} 
		scope:actor = scope:recipient 
	}

	is_valid_showing_failures_only = {
		scope:recipient = { POD_can_be_healed_trigger = yes }
	}
	
	send_option = {
		is_shown = { scope:recipient = { has_trait = scarred } }	
		is_valid = {
			custom_description = { text = POD_permanent_scarred object = scope:recipient scope:recipient = { NOT = { has_character_flag = POD_permanent_scarring_flag } } }
		}
		flag = scarred
		localization = POD_heal_scarred
	}
	send_option = {
		is_shown = { scope:recipient = { has_trait = disfigured } }	
		is_valid = {
			scope:actor = { has_perk = biocontrol_3_perk }
			custom_description = { text = POD_permanent_disfigured object = scope:recipient scope:recipient = { NOT = { has_character_flag = POD_permanent_disfigured_flag } } }
		}
		flag = disfigured
		localization = POD_heal_disfigured
	}
	send_option = {
		is_shown = { scope:recipient = { has_trait = ill } }	
		is_valid = {
			scope:actor = { has_perk = biocontrol_2_perk }
		}
		flag = ill
		localization = POD_heal_ill
	}
	send_option = {
		is_shown = { scope:recipient = { has_short_disease_trigger = yes has_any_mortal_disease_trigger = yes } }	
		is_valid = {
			scope:actor = { has_perk = biocontrol_3_perk }
		}
		flag = easy_disease
		localization = POD_heal_easy_disease
	}
	send_option = {
		is_shown = { scope:recipient = { has_any_mortal_disease_trigger = yes } }
		is_valid = {
			scope:actor = { has_perk = biocontrol_4_perk }
		}
		flag = disease
		localization = game_concept_diseases
	}
	send_option = {
		is_shown = { scope:recipient = { has_any_supernatural_disease_trigger = yes } }
		is_valid = {
			scope:actor = { has_perk = biocontrol_5_perk }
		}
		flag = super_disease
		localization = supernatural_diseases
	}
	
	on_accept = {
		scope:actor = {
			play_music_cue = "mx_cue_peace_ensues"
			add_trait_xp = { trait = psychic value = 1 }

			switch = {
				trigger = yes
				scope:scarred = { POD_static_stress_gain = { COUNT = psychic_willpower_cost_small } remove_trait = scarred }	
				scope:disfigured = { POD_static_stress_gain = { COUNT = psychic_willpower_cost_medium } remove_trait = disfigured }	
				scope:ill = { POD_static_stress_gain = { COUNT = psychic_willpower_cost_medium } recover_from_disease_effect = { DISEASE = ill } remove_disease_treatment_effect = yes }	
				scope:easy_disease = { POD_static_stress_gain = { COUNT = psychic_willpower_cost_medium } remove_disease_trait_effect = yes }		
				scope:disease = { POD_static_stress_gain = { COUNT = psychic_willpower_cost_huge } remove_disease_trait_effect = yes }		
				scope:super_disease = {
					POD_static_stress_gain = { COUNT = psychic_willpower_cost_massive } 
					remove_supernatural_disease_trait_effect = yes 
					change_trait_rank = { trait = wounded rank = 3 max = 3 }
				}		
			}		
		}
	}
	
	auto_accept = yes
}

POD_determine_psi_interaction = {
	category = interaction_category_psychic
	icon = POD_psi_interaction
    desc = POD_determine_psi_interaction_desc
	
	cooldown = { months = 12 }
	cost = { piety = major_piety_value }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_trait = psychic }
		scope:recipient = { NOT = { has_trait = psychic } }
		scope:recipient = { NOT = { has_character_flag = non_psychic } }
	}

	is_valid_showing_failures_only = {
	    scope:actor = {
			psi_lifestyle_perk_points >= 1
		}
		scope:recipient = { is_imprisoned = no }
	}
	
	on_accept = {
	    scope:actor = { 
			add_psi_lifestyle_perk_points = -1
			stress_impact = {
				base = major_stress_gain
			}
			duel = {
				skill = learning
				value = 10
				50 = {
					desc = POD_determine_psi_interaction_sucess
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					scope:recipient = {
						add_trait = psychic
						save_scope_as = victim
		POD_resonance_increase_intensity_effect = yes
						add_opinion = {
							target = scope:actor
							modifier = mentored_me_opinion
						}
					}
				}
				50 = {
					desc = POD_determine_psi_interaction_fail
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier 
					}
					scope:recipient = {
						add_character_flag = non_psychic
					}
				}
			}
		}	
	}

	ai_frequency = 0
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
