		flag = summon_rat_plague
		localization = POD_psi_summon_rat_plague
	}

		is_shown = { scope:summon_animals = yes scope:actor = { NOT = { any_owned_story = { story_type = story_cycle_martial_lifestyle_warhorse } } } }
		flag = horse 
		localization = POD_psychic_animal_psi_interaction_horse

		flag = dog
		localization = POD_psychic_animal_psi_interaction_dog	
	} 
	send_option = {
		is_shown = { scope:summon_animals = yes scope:actor = { NOT = { any_owned_story = { story_type = story_cycle_pet_cat } } } }
		flag = cat
		localization = POD_psychic_animal_psi_interaction_cat		
	}
	send_option = {
		is_shown = { scope:summon_animals = yes scope:actor = { NOT = { has_character_modifier = exotic_pet_wolf } } }
		flag = wolf
		localization = POD_psychic_animal_psi_interaction_wolf	
	}
	send_option = {
		is_shown = { scope:summon_animals = yes scope:actor = { NOT = { has_character_modifier = exotic_pet_bear } location = { OR = { geographical_region = world_europe geographical_region = world_tibet geographical_region = world_himalaya geographical_region = world_burma geographical_region = world_middle_east_persia } } } }

		localization = POD_psychic_animal_psi_interaction_lion

		localization = POD_psychic_animal_psi_interaction_tiger

		flag = rat
		localization = POD_psychic_animal_psi_interaction_rat
	}

		is_shown = { scope:summon_animals = yes scope:actor = { NOT = { has_character_modifier = prison_pet_cockroach_modifier } } }
		flag = cockroach
		localization = POD_psychic_animal_psi_interaction_cockroach

		scope:actor = {
			add_trait_xp = { trait = psychic value = 1 }
			if = { limit = { scope:contact_animals = yes } 
				POD_static_stress_gain = { COUNT = psychic_willpower_cost_small } 
				random = {
					chance = 25
					modifier = { has_perk = animal_psi_2_perk add = 25 }
					modifier = { has_perk = animal_psi_3_perk add = 25 }
					modifier = { has_perk = animal_psi_4_perk add = 25 }

				}
			}
			if = { limit = { scope:summon_animals = yes } 
				POD_static_stress_gain = { COUNT = psychic_willpower_cost_medium } 
				if = { limit = { scope:cat = yes } start_cat_story_cycle_effect = yes }	
				if = { limit = { scope:dog = yes } start_dog_story_cycle_effect = yes }

				if = { limit = { scope:lion = yes } add_character_modifier = { modifier = exotic_pet_lion years = 10 } create_artifact_lion_effect = { OWNER = this } }
				if = { limit = { scope:tiger = yes } add_character_modifier = { modifier = exotic_pet_tiger years = 10 } create_artifact_tiger_effect = { OWNER = this } }
				if = { limit = { scope:rat = yes } add_character_modifier = { modifier = prison_pet_rat_modifier years = 10 } }
				if = { limit = { scope:cockroach = yes } add_character_modifier = { modifier = prison_pet_cockroach_modifier years = 10 } }
			}
			if = { limit = { scope:summon_rat_plague = yes } 
				POD_static_stress_gain = { COUNT = psychic_willpower_cost_huge } 
				scope:recipient = {
					save_scope_as = defender
				}
				save_scope_as = attacker
				duel = {
					skill = diplomacy
					target = scope:recipient
					25 = { POD_mesmerize_attacker_modifier = yes 
						POD_mesmerize_attacker_tt_effect = yes
						modifier = { add = psychic_add_success_chance }
						min = 0
						send_interface_toast = {
							title = POD_psychic_animal_psi_interaction
							scope:recipient = {
								capital_county = {
									add_county_modifier = { modifier = psi_ratocalispe years = 10 }
									title_province = {
										random = {
											chance = 50
											create_epidemic_outbreak = { type = consumption intensity = major }
										}
									}
								}
							}
						}
					}
					50 = { POD_mesmerize_defender_modifier = yes 
						POD_mesmerize_defender_tt_effect = yes
						min = 0
						reverse_add_opinion = {
							modifier = attempted_trickery_opinion
							target = scope:recipient
							opinion = -30
						}
					}
				}
			}
		}
	}
	auto_accept = yes
}

POD_psychic_biocontrol_interaction = {
	category = interaction_category_psychic
	common_interaction = yes
	icon = POD_psi_interaction
	desc = POD_psychic_biocontrol_interaction_desc	
	cooldown = { years = 2 }
	
	is_shown = { 
		scope:actor = { 
			has_perk = biocontrol_2_perk
			OR = {
				has_trait = scarred
				has_trait = disfigured
				has_trait = wounded
				has_any_disease_trigger = yes
			} 
		} 
		scope:actor = scope:recipient 
	}

	is_valid_showing_failures_only = {
		scope:recipient = { POD_can_be_healed_trigger = yes }
	}
	
	send_option = {
		is_shown = { scope:recipient = { has_trait = scarred } }	
		is_valid = {
			custom_description = { text = POD_permanent_scarred object = scope:recipient scope:recipient = { NOT = { has_character_flag = POD_permanent_scarring_flag } } }
		}
		flag = scarred
		localization = POD_heal_scarred
	}
	send_option = {
		is_shown = { scope:recipient = { has_trait = disfigured } }	
		is_valid = {
			scope:actor = { has_perk = biocontrol_3_perk }
			custom_description = { text = POD_permanent_disfigured object = scope:recipient scope:recipient = { NOT = { has_character_flag = POD_permanent_disfigured_flag } } }
		}
		flag = disfigured
		localization = POD_heal_disfigured
	}
	send_option = {
		is_shown = { scope:recipient = { has_trait = ill } }	
		is_valid = {
			scope:actor = { has_perk = biocontrol_2_perk }
		}
		flag = ill
		localization = POD_heal_ill
	}
	send_option = {
		is_shown = { scope:recipient = { has_trait = wounded } }	
		is_valid = { scope:actor = { has_perk = biocontrol_3_perk } }
		flag = wounded
		localization = POD_heal_wounded
	}
	send_option = {
		is_shown = { scope:recipient = { has_any_mortal_disease_trigger = yes } }
		is_valid = {
			scope:actor = { has_perk = biocontrol_4_perk }
		}
		flag = disease
		localization = game_concept_diseases
	}
	send_option = {
		is_shown = { scope:recipient = { has_any_supernatural_disease_trigger = yes } }
		is_valid = {
			scope:actor = { has_perk = biocontrol_5_perk }
		}
		flag = super_disease
		localization = supernatural_diseases
	}
	
	on_accept = {
		if = {
			limit = { NOT = { scope:actor = scope:recipient } }
			scope:actor = { POD_commit_heal_effect = yes }
		}
		scope:actor = {
			play_music_cue = "mx_cue_peace_ensues"
			add_trait_xp = { trait = psychic value = 1 }

			switch = {
				trigger = yes
				scope:scarred = { POD_static_stress_gain = { COUNT = psychic_willpower_cost_small } remove_trait = scarred }	
				scope:disfigured = { POD_static_stress_gain = { COUNT = psychic_willpower_cost_medium } remove_trait = disfigured }	
				scope:ill = { POD_static_stress_gain = { COUNT = psychic_willpower_cost_medium } recover_from_disease_effect = { DISEASE = ill } remove_disease_treatment_effect = yes }	
				scope:wounded = {
					scope:recipient = { POD_remove_wound_trait_effect = yes }
					POD_static_stress_gain = { COUNT = psychic_willpower_cost_medium }
				}
				scope:disease = { POD_static_stress_gain = { COUNT = psychic_willpower_cost_huge } remove_disease_trait_effect = yes }		
				scope:super_disease = {
					POD_static_stress_gain = { COUNT = psychic_willpower_cost_massive } 
					remove_supernatural_disease_trait_effect = yes 
					change_trait_rank = { trait = wounded rank = 3 max = 3 }
				}		
			}		
		}
	}
	
	auto_accept = yes
}
