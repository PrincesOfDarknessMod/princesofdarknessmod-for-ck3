POD_ravaging_interaction = {
	category = interaction_category_fae
	common_interaction = yes
	interface_priority = 120
	icon = POD_fae_interaction
	desc = POD_ravaging_interaction_desc
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_POD_fae_blood_trigger = yes }
	}

	is_valid_showing_failures_only = {
		scope:recipient = { 
			is_imprisoned_by = scope:actor
			stress_level < 3
			custom_description = {
				text = POD_has_inspiration
				object = this
				exists = inspiration 
			}
		}
	}

	on_accept = {
		random_list = {
			10 = {
				scope:actor = { POD_glamour_increase_multiple_effect = { COUNT = 5 } }
				scope:recipient = { 
					add_stress = massive_stress_impact_gain 
					add_piety = massive_piety_loss
				}
			}
			20 = {
				scope:actor = { POD_glamour_increase_multiple_effect = { COUNT = 3 } }
				scope:recipient = { 
					add_stress = major_stress_impact_gain 
					add_piety = major_piety_loss
				}
			}
			30 = {
				scope:actor = { POD_glamour_increase_multiple_effect = { COUNT = 2 } }
				scope:recipient = { 
					add_stress = medium_stress_impact_gain 
					add_piety = medium_piety_loss
				}
			}
			40 = {
				scope:actor = { POD_glamour_increase_multiple_effect = { COUNT = 1 } }
				scope:recipient = { 
					add_stress = minor_stress_impact_gain 
					add_piety = minor_piety_loss
				}
			}
		}
		scope:actor = {
			add_common_arts_lifestyle_xp = 50
		}
		scope:recipient = { 
			destroy_inspiration = inspiration
			if = { 
				limit = { has_trait = mortal }
				add_trait = POD_knows_too_much 
			}
			add_opinion = {
				target = scope:actor
				modifier = hurt_opinion
				opinion = -25
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_rhapsody_interaction = {
	category = interaction_category_fae
	common_interaction = yes
	interface_priority = 120
	icon = POD_fae_interaction
	desc = POD_rhapsody_interaction_desc
	
	cost = { piety = 75 }
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_POD_fae_blood_trigger = yes }
	}

	is_valid_showing_failures_only = {
		scope:recipient = { stress_level >= 3 }
	}

	on_accept = {
		scope:actor = { POD_glamour_increase_multiple_effect = { COUNT = 3 } }
		scope:recipient = { 
			if = { 
				limit = { exists = inspiration }
				destroy_inspiration = inspiration
				scope:actor = {
					add_common_arts_lifestyle_xp = 100
				}
			}
			if = { 
				limit = { has_trait = mortal }
				add_trait = POD_knows_too_much 
			}
			add_opinion = {
				target = scope:actor
				modifier = hurt_opinion
				opinion = -45
			}
			increase_wounds_effect = { REASON = treatment }
			custom_tooltip = stress_threshold_level_3_event_tooltip
			if = { 
				limit = { is_imprisoned = yes }
				trigger_event = { on_action = stress_threshold_level_3_event_prison } 
			}
			else = { 
				trigger_event = { on_action = stress_threshold_level_3_event } 
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_reaving_interaction = {
	category = interaction_category_fae
	common_interaction = yes
	interface_priority = 120
	icon = POD_fae_interaction
	desc = POD_reaving_interaction_desc
	
	cost = { piety = 300 }
	
	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { has_POD_fae_blood_trigger = yes }
	}

	is_valid_showing_failures_only = {
		scope:recipient.location.county = { NOT = { has_county_modifier = glamour_reaved_modifier } }
	}

	on_accept = {
		scope:actor = { 
			POD_glamour_increase_multiple_effect = { COUNT = 5 } 
			add_character_modifier = {
				modifier = glamour_reaver_modifier 
				years = 5
			} 
		}
		scope:recipient = {
			location.county = {
				add_county_modifier = {
					modifier = glamour_reaved_modifier 
					years = 5
				} 
			}
		}
	   scope:recipient.location.county.holder = { 
			add_opinion = {
				target = scope:actor
				modifier = POD_cast_evil_weather_magic_to_my_realm
			}
			progress_towards_rival_effect = {
				CHARACTER = scope:actor
				REASON = rival_cast_spell
				OPINION = 0
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
