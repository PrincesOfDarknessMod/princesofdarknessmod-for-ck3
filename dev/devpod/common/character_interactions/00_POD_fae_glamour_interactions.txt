POD_ravaging_interaction = {
	category = interaction_category_fae
	common_interaction = yes
	interface_priority = 120
	icon = POD_fae_interaction
	desc = POD_ravaging_interaction_desc
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_POD_fae_blood_trigger = yes }
	}

	is_valid = {
		scope:recipient = { 
			is_imprisoned_by = scope:actor
			stress_level < 3
			trigger_if = {
				limit = { has_ep1_dlc_trigger = yes }
				custom_description = {
					text = POD_has_inspiration
					object = this
					exists = inspiration 
				}
			}
			trigger_else = {
				NOT = { has_character_modifier = intrigue_broken_modifier }
			}
		}
	}


	is_valid_showing_failures_only = {
		scope:actor = { 
			trigger_if = { 
				limit = { has_trait = hsien }
				custom_description = {
					object = scope:actor
					text = "I_am_full_of_chi"
					scope:actor = {
						OR = {
							yin_difference >= 1
							yang_difference >= 1
						}
					}
				}
			}
			trigger_else = { NOT = { has_character_modifier = glamour5_modifier } }
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
			POD_not_has_cantrip_defenses_trigger = yes
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}

	on_accept = {
		random_list = {
			10 = {
				scope:actor = { POD_glamour_or_chi_income_effect = { COUNT = 5 } }
				scope:recipient = { 
					add_stress = massive_stress_impact_gain 
					add_piety = massive_piety_loss
				}
			}
			20 = {
				scope:actor = { POD_glamour_or_chi_income_effect = { COUNT = 3 } }
				scope:recipient = { 
					add_stress = major_stress_impact_gain 
					add_piety = major_piety_loss
				}
			}
			30 = {
				scope:actor = { POD_glamour_or_chi_income_effect = { COUNT = 2 } }
				scope:recipient = { 
					add_stress = medium_stress_impact_gain 
					add_piety = medium_piety_loss
				}
			}
			40 = {
				scope:actor = { POD_glamour_or_chi_income_effect = { COUNT = 1 } }
				scope:recipient = { 
					add_stress = minor_stress_impact_gain 
					add_piety = minor_piety_loss
				}
			}
		}
		scope:actor = {
			add_common_arts_lifestyle_xp = 50
			if = {
				limit = { has_trait = kithain }
				add_kith_arts_lifestyle_xp = 25
			}
			if = {
				limit = { has_trait = phyla }
				add_slivers_lifestyle_xp = 25
			}
			if = {
				limit = { has_trait = hsien }
				add_wu_tan_lifestyle_xp = 25
			}
		}
		scope:recipient = { 
			if = { 
				limit = { exists = inspiration }
				destroy_inspiration = inspiration 
			}
			else = { 
				if = {
					limit = { has_character_modifier = inspiration_modifier }
					remove_character_modifier = inspiration_modifier
				}
				else = {
					add_character_modifier = {
						modifier = intrigue_broken_modifier
						years = 15
					}
				}
			}
			if = { 
				limit = { has_trait = mortal }
				add_trait = POD_knows_too_much 
			}
			add_opinion = {
				target = scope:actor
				modifier = hurt_opinion
				opinion = -25
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

					add_common_arts_lifestyle_xp = 100

				target = scope:actor

				opinion = -45
			}

			custom_tooltip = stress_threshold_level_3_event_tooltip
