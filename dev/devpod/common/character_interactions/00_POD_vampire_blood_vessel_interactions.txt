POD_store_blood_interaction = {
	category = interaction_category_vampire
	icon = POD_drink_herd
	desc = POD_store_blood_interaction_desc
	options_heading = POD_store_blood_interaction_options_heading
	
	target_type = artifact
	target_filter = actor_artifacts
	
	is_shown = {
		scope:actor = scope:recipient
		scope:actor = {
			has_POD_vamp_blood_trigger = yes
			any_character_artifact = {
				has_variable = stored_sekhem
				NOT = { has_variable = temporary_vessel }
			}
		}
	}
	
	is_valid_showing_failures_only = {
		scope:actor = { NOT = { has_character_modifier = bloodhunger5_modifier } }
	}
	
	can_be_picked_artifact = {
		scope:target = { vessel_difference >= 1 }
	}
	
	send_option = {
		is_shown = {
			scope:actor = { blood_level >= 1 }
			trigger_if = {
				limit = { exists = scope:target }
				scope:target = { vessel_difference >= 1 }		
			}
		}
		flag = POD_num1
		localization = POD_num1
	}
	
	send_option = {
		is_shown = {
			scope:actor = { blood_level >= 2 }
			trigger_if = {
				limit = { exists = scope:target }
				scope:target = { vessel_difference >= 2 }	
			}
		}
		flag = POD_num2
		localization = POD_num2
	}
	
	send_option = {
		is_shown = {
			scope:actor = { blood_level >= 3 }
			trigger_if = {
				limit = { exists = scope:target }
				scope:target = { vessel_difference >= 3 }		
			}
		}
		flag = POD_num3
		localization = POD_num3
	}
	
	send_option = {
		is_shown = {
			scope:actor = { blood_level >= 4 }
			trigger_if = {
				limit = { exists = scope:target }
				scope:target = { vessel_difference >= 4 }		
			}
		}
		flag = POD_num4
		localization = POD_num4
	}
	
	send_option = {
		is_shown = {
			scope:actor = { blood_level >= 5 }
			trigger_if = {
				limit = { exists = scope:target }
				scope:target = { vessel_difference >= 5 }	
			}
		}
		flag = POD_num5
		localization = POD_num5
	}
	
	on_accept = {
		scope:actor = {
			if = {
				limit = { scope:POD_num5 = yes }
				scope:target = { POD_blood_vessel_increase_multiple_effect = { COUNT = 5 OWNER = scope:recipient } }
				POD_hunger_increase_multiple_effect = { COUNT = 5 }
			}
			else_if = {
				limit = { scope:POD_num4 = yes }
				scope:target = { POD_blood_vessel_increase_multiple_effect = { COUNT = 4 OWNER = scope:recipient } }
				POD_hunger_increase_multiple_effect = { COUNT = 4 }
			}
			else_if = {
				limit = { scope:POD_num3 = yes }
				scope:target = { POD_blood_vessel_increase_multiple_effect = { COUNT = 3 OWNER = scope:recipient } }
				POD_hunger_increase_multiple_effect = { COUNT = 3 }
			}
			else_if = {
				limit = { scope:POD_num2 = yes }
				scope:target = { POD_blood_vessel_increase_multiple_effect = { COUNT = 2 OWNER = scope:recipient } }
				POD_hunger_increase_multiple_effect = { COUNT = 2 }
			}
			else = {
				scope:target = { POD_blood_vessel_increase_multiple_effect = { COUNT = 1 OWNER = scope:recipient } }
				POD_hunger_increase_effect = yes
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
