		flag = POD_colors_coterie

		flag = POD_colors_house

		scope:recipient = { save_scope_as = target }
		scope:target = {
			switch = {

				scope:POD_colors_personal = {
					# Variables are for the event widget, scopes are for the color display
					set_variable = {
						name  = color_root
						value = this
					}
					save_scope_as = color_root
				}
				scope:POD_colors_coterie = {
					set_variable = {
						name  = color_root
						value = var:coterie
					}
					var:coterie = { save_scope_as = color_root }
					random_relation = {
						type = coterie_member
						save_scope_as = secondary_target
					}
				}
				scope:POD_colors_house = {
					set_variable = {
						name  = color_root
						value = house
					}
					house = { save_scope_as = color_root }
					house = {
						random_house_member = {
							limit = { NOT = { this = scope:target } }
							save_scope_as = secondary_target
						}
					}
				}
			}
		}
		custom_tooltip = POD_current_clothing_colors
		scope:actor = { trigger_event = POD_style.1 }

# Deferred reinforcements: a 1v1 vs the target, then a PC vassal scheduled to enter on
# round 2 and a target vassal on round 3. Exercises the pending-reinforcement path
# (defer at round 1, seated by pod_combat_group_end_round_effect at the target round).
# Real vassals (landed rulers, hence PoD-initialized) so they can act on arrival.
pod_combat_debug_reinforce_interaction = {
	category = interaction_debug_combat
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	common_interaction = yes
	interface_priority = 993

	icon = icon_combat

	desc = pod_combat_debug_reinforce_desc

	is_shown = {
		debug_only = yes
		scope:actor != scope:recipient
		NOT = { scope:actor = { has_variable = pod_combat_active } }
	}

	on_accept = {
		scope:actor = {
			# Base 1v1 (PC vs recipient).
			POD_combat_create_duel_effect = {
				INITIATOR    = scope:actor
				ATTACKER     = scope:actor
				DEFENDER     = scope:recipient
				FATALITY     = possible
				FIXED        = no
				LOCALE       = terrain_scope
				OUTPUT_EVENT = pod_combat_debug.0001
			}
			POD_combat_duel_add_enemy_effect = { CHARACTER = scope:recipient }
			POD_combat_finalize_duel_effect = yes

			# Fight is live at round 1. Schedule a PC courtier as an ally (round 2) and a
			# target courtier as an enemy (round 3). Real existing characters, so there is no
			# create_character tooltip-build / root deref; they seat and appear on their round
			# regardless of whether they can act (an uninitialised courtier still shows up).
			# Needs the actor / target to have a court; a landed ruler always does.
			random_courtier = {
				limit = {
					is_alive = yes
					is_adult = yes
					NOT = { has_trait = incapable }
					NOT = { has_variable = pod_combat_active }
				}
				save_scope_as = reinf_ally
			}
			if = {
				limit = { exists = scope:reinf_ally }
				pod_combat_schedule_ally_reinforcement_effect = { CHARACTER = scope:reinf_ally  JOIN_ROUND = 2 }
			}
			scope:recipient = {
				random_courtier = {
					limit = {
						is_alive = yes
						is_adult = yes
						NOT = { has_trait = incapable }
						NOT = { has_variable = pod_combat_active }
					}
					save_scope_as = reinf_enemy
				}
			}
			if = {
				limit = { exists = scope:reinf_enemy }
				pod_combat_schedule_enemy_reinforcement_effect = { CHARACTER = scope:reinf_enemy  JOIN_ROUND = 3 }
			}
		}
	}

	auto_accept = yes
}

					# tooltip-build error (the var isn't set during description
					# preview).
					scope:actor = {
						POD_combat_duel_add_enemy_effect = { CHARACTER = scope:duel_enemy }
					}
				}
			}
			POD_combat_finalize_duel_effect = yes

# Big fight: PC + the PC's vassals vs the subject (recipient) + the subject's
# vassals (a ruler's "subjects"), each team filled in iteration order up to the
# chosen per-side size (default 8). This is the launcher that exercises team
# sizes past the old 5-per-side ceiling. The engine now addresses every
# combatant by (combat_team, team_position) with no positional cap. Vassals are
# landed rulers, hence always PoD-initialized with a splat trait + working
# loadout; general courtiers turned out to be a mix of initialized and raw pool
# characters, and the uninitialized ones can't be armed by the move system.
# The same failure mode a blank create_character dummy hits. Actual team size is
# bounded by how many valid vassals each side has, so pick high-tier rulers for a
# full-size fight.
# Combatants past the first panel rows are reached by scrolling.
pod_combat_debug_big_interaction = {
	category = interaction_debug_combat
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	common_interaction = yes
	interface_priority = 994

	icon = icon_combat

	desc = pod_combat_debug_big_desc

	is_shown = {
		debug_only = yes
		scope:actor != scope:recipient
		NOT = { scope:actor = { has_variable = pod_combat_active } }
	}

	# Per-side team size, leader included. send_options_exclusive makes these a
	# mutually-exclusive radio group, so no per-option is_valid cross-gating is
	# needed and `scope:<flag> = no` on an unselected option is malformed
	# anyway (the flag scope is unset, which surfaces in-game as the tooltip
	# error "none_scope_equal has no localization" and wedges the other
	# options). The on_accept switch maps the chosen flag to a size, falling
	# back to 8 when none is picked.
	send_options_exclusive = yes
	send_option = {
		flag = big_size_8
		localization = pod_combat_debug_big_size_8
	}
	send_option = {
		flag = big_size_12
		localization = pod_combat_debug_big_size_12
	}
	send_option = {
		flag = big_size_20
		localization = pod_combat_debug_big_size_20
	}

	on_accept = {
		scope:actor = {
			# Resolve the chosen per-side team size (leader included).
			switch = {
				trigger = yes
				scope:big_size_12 = { save_scope_value_as = { name = team_size value = 12 } }
				scope:big_size_20 = { save_scope_value_as = { name = team_size value = 20 } }
				fallback          = { save_scope_value_as = { name = team_size value = 8  } }
			}

			POD_combat_create_duel_effect = {
				INITIATOR    = scope:actor
				ATTACKER     = scope:actor
				DEFENDER     = scope:recipient
				FATALITY     = possible
				FIXED        = no
				LOCALE       = terrain_scope
				OUTPUT_EVENT = pod_combat_debug.0001
			}

			# Player team: the PC (already seated at player/0) plus the PC's
			# vassals (their "subjects"), in iteration order, until the team
			# reaches team_size. Vassals are landed rulers, hence always
			# PoD-initialized with a splat trait + working loadout, unlike
			# general courtiers, which are a mix of initialized and raw pool
			# characters; the uninitialized ones can't be armed by the move
			# system (same failure mode as a create_character dummy). The cap
			# is a trigger_if, not a bare limit compare: add_ally bumps
			# pod_combat_player_team_size, and reading that var directly would
			# throw during the interaction's tooltip-build (set_variable isn't
			# applied there). So has_variable gates the compare, and the read
			# is simply skipped during preview.
			every_vassal = {
				limit = {
					is_alive = yes
					is_adult = yes
					NOT = { has_trait = incapable }
					NOT = { this = scope:recipient }
					trigger_if = {
						limit = { scope:actor = { has_variable = pod_combat_player_team_size } }
						scope:actor = { var:pod_combat_player_team_size < scope:team_size }
					}
				}
				save_scope_as = duel_ally
				scope:actor = {
					POD_combat_duel_add_ally_effect = { CHARACTER = scope:duel_ally }
				}
			}

			# Enemy team: the subject (recipient) at enemy/0, then the subject's
			# vassals until that team reaches team_size (same tooltip-safe cap).
			POD_combat_duel_add_enemy_effect = { CHARACTER = scope:recipient }
			scope:recipient = {
				every_vassal = {
					limit = {
						is_alive = yes
						is_adult = yes
						NOT = { has_trait = incapable }
						NOT = { this = scope:actor }
						trigger_if = {
							limit = { scope:actor = { has_variable = pod_combat_enemy_team_size } }
							scope:actor = { var:pod_combat_enemy_team_size < scope:team_size }
						}
					}
					save_scope_as = duel_enemy
					scope:actor = {
						POD_combat_duel_add_enemy_effect = { CHARACTER = scope:duel_enemy }
					}
				}
			}

			POD_combat_finalize_duel_effect = yes
		}
	}

	auto_accept = yes
}

# NvN: PC + coterie vs target + their knights
pod_combat_debug_nvn_interaction = {
	category = interaction_debug_combat
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	common_interaction = yes
	interface_priority = 997

	icon = icon_combat

	desc = pod_combat_debug_nvn_desc

	is_shown = {
		debug_only = yes
		scope:actor != scope:recipient
		NOT = { scope:actor = { has_variable = pod_combat_active } }
	}

	on_accept = {
		# NvN via the duel builder: PC at player/0 plus coterie allies at
		# player/1+ vs the recipient at enemy/0 plus the recipient's knights
		# at enemy/1+.
		scope:actor = {
			POD_combat_create_duel_effect = {
				INITIATOR    = scope:actor
				ATTACKER     = scope:actor
				DEFENDER     = scope:recipient
				FATALITY     = possible
				FIXED        = no
				LOCALE       = terrain_scope
				OUTPUT_EVENT = pod_combat_debug.0001
			}
			every_relation = {
				type = coterie_member
				limit = {
					is_alive = yes
					is_adult = yes
					NOT = { has_trait = incapable }
				}
				save_scope_as = duel_ally
				scope:actor = {
					POD_combat_duel_add_ally_effect = { CHARACTER = scope:duel_ally }
				}
			}
			POD_combat_duel_add_enemy_effect = { CHARACTER = scope:recipient }
			scope:recipient = {
				every_knight = {
					limit = {
						is_alive = yes
						NOT = { has_trait = incapable }
						opinion = { target = scope:recipient value >= 0 }
					}
					save_scope_as = duel_enemy
					# No enemy_team_size gate: add_enemy self-caps at the soft
					# cap pod_combat_max_team_size (enemy/0 opponent + further
					# knights at enemy/1+). Reading the host var here also tripped a
					# tooltip-build error (the var isn't set during description
					# preview).
					scope:actor = {
						POD_combat_duel_add_enemy_effect = { CHARACTER = scope:duel_enemy }
					}
				}
			}
			POD_combat_finalize_duel_effect = yes
		}
	}

	auto_accept = yes
}
