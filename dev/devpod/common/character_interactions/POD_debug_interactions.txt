		flag = POD_colors_house

	on_accept = {
		scope:recipient = { save_scope_as = target }
		scope:target = {
			switch = {
				trigger = yes
				scope:POD_colors_personal = {
					# Variables are for the event widget, scopes are for the color display
					set_variable = {
						name  = color_root
						value = this
					}
					save_scope_as = color_root
				}
				scope:POD_colors_coterie = {
					set_variable = {
						name  = color_root
						value = var:coterie
					}
					var:coterie = { save_scope_as = color_root }

						type = coterie_member
						save_scope_as = secondary_target
					}
				}
				scope:POD_colors_house = {
					set_variable = {
						name  = color_root
						value = house
					}
					house = { save_scope_as = color_root }
					house = {
						random_house_member = {
							limit = { NOT = { this = scope:target } }
							save_scope_as = secondary_target
						}
					}
				}
			}
		}
		custom_tooltip = POD_current_clothing_colors
		scope:actor = { trigger_event = POD_style.1 }

				scope:trait_werewolf_kinfolk ?= yes
				scope:trait_kinain ?= yes
				scope:trait_laham ?= yes

				scope:trait_dhampyr ?= yes

			}
			
			switch = {
				trigger = yes
				
				scope:trait_ghoul = {
					POD_become_ghoul_effect = yes
				}
				
				scope:trait_werewolf_abomination = {
					# TODO: add arbitrary fera types
					add_trait = abomination
					POD_set_fera_type = { TRAIT = trait:werewolf }
					add_trait = homid_born

						1 = { add_trait = ragabash }
						1 = { add_trait = theurge }
						1 = { add_trait = philodox }
						1 = { add_trait = galliard }
						1 = { add_trait = ahroun }
						1 = { add_trait = ragabash }
					}
				}
				
				scope:trait_werewolf_kinfolk = {
					add_trait = kinfolk
					POD_set_fera_type = { TRAIT = trait:werewolf }
					POD_kinfolk_initialization_effect = yes
				}
				
				scope:trait_kinain = {
					add_trait = kinain
					POD_kinain_initialization_effect = yes
				}
				
				scope:trait_laham = {
					add_trait = laham
				}
				
				scope:trait_nephilim = {
					remove_trait = laham
					add_trait = nephilim
					POD_nephilim_initialization_effect = yes
				}
				
				scope:trait_dhampyr = {
					add_trait = dhampyr
					POD_dhampyr_initialization_effect = yes
				}
				
				scope:trait_dhampir = {
					add_trait = dhampir
					if = {
						limit = { POD_has_any_blood_hunger_modifier = no }
						add_character_modifier = bloodhunger0_modifier
					}
					if = {
						limit = { has_any_blood_potency_modifier = no }
						add_character_modifier = bloodpotency0_modifier
					}
				}
			}
			
			hidden_effect = { POD_change_to_appropriate_energy_type = yes }
			
			add_character_flag = had_POD_character_initialization
		}
	}
	
	auto_accept = yes
