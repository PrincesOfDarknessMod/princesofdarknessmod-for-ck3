	common_interaction = yes
	interface_priority = 90
    category = interaction_category_wraith
	ignores_pending_interaction_block = yes

	scheme = convert_to_wraith
	
	is_shown = {
		always = no
		#NOT = { scope:actor = scope:recipient }
		#scope:actor = { POD_can_covert_to_wraith_trigger = yes }
		#scope:recipient = { POD_is_clone_trigger = no }
		#scope:recipient = { can_convert_to_wraith_trigger = yes }
	}

	is_valid_showing_failures_only = {
		custom_description = {
			object = scope:recipient
			text = convert_to_wraith_truefaith # elderly kukul junk, can be delete or keeped
			scope:recipient = {
				POD_has_true_faith_trigger = no
			}
		}
		scope:actor = {
			can_start_scheme = {
				type = convert_to_wraith
				target_character = scope:recipient
			}
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			NOT = { has_trait = automaton }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }	
			POD_is_not_clone_trigger = yes	
		}
		POD_cross_umbra_interaction_valid_trigger = yes										
	}
	
	cost = { gold = scope:actor.medium_gold_value }
	
	on_accept = {
		scope:recipient = {
			add_character_flag = marked_for_return
		}
		scope:actor = {
					begin_scheme_basic_effect = {
						SCHEME_TYPE = convert_to_wraith
						TARGET_TYPE = target_character
						TARGET_SCOPE = scope:recipient
					}				
		}
		custom_tooltip = POD_convert_to_wraith_interaction_desc
	
	}
	
	auto_accept = yes

}

			}

POD_send_bound_wraith_interaction = {
	desc = POD_send_bound_wraith_interaction_desc
	common_interaction = yes
	interface_priority = 90
	category = interaction_category_wraith
	icon = POD_wraith_interaction
	
	is_shown = {
		scope:actor = { num_of_relation_bound_wraith >= 1 }
	}
	
	is_valid_showing_failures_only = {
		trigger_if = {
			limit = { exists = scope:secondary_actor }
			scope:secondary_actor = { 
				NOT = { has_trait = torpor } 
				NOT = { has_character_modifier = control_puppet_modifier }
			}
			POD_cross_umbra_interaction_secondary_valid_trigger = yes
		}
	}
	
	populate_recipient_list = {
		scope:actor = {
			every_relation = {
				type = bound_wraith
				add_to_list = characters
			}
		}
	}

	send_options_exclusive = yes

	# ACTIONS
	send_option = {
		flag = diplomacy_increase	
		localization = POD_send_bound_wraith_interaction_diplomacy_increase
	}
	send_option = {
		flag = martial_increase	
		localization = POD_send_bound_wraith_interaction_martial_increase
	}
	send_option = {
		flag = stewardship_increase	
		localization = POD_send_bound_wraith_interaction_stewardship_increase
	}
	send_option = {
		flag = intrigue_increase	
		localization = POD_send_bound_wraith_interaction_intrigue_increase
	}
	send_option = {
		flag = learning_increase	
		localization = POD_send_bound_wraith_interaction_learning_increase
	}
	
	on_accept = {
		scope:secondary_recipient = {
			if = { limit = { has_variable = current_haunting_level_diplomacy } 
				var:haunting_target ?= {
					while = {
						count = PREV.var:current_haunting_level_diplomacy
						remove_character_modifier = haunting_diplomacy_increase_modifier
					}
				}
				remove_variable = current_haunting_level_diplomacy
			}
			if = { limit = { has_variable = current_haunting_level_martial } 
				var:haunting_target ?= {
					while = {
						count = PREV.var:current_haunting_level_martial
						remove_character_modifier = haunting_martial_increase_modifier
					}
				}
				remove_variable = current_haunting_level_martial
			}
			if = { limit = { has_variable = current_haunting_level_stewardship } 
				var:haunting_target ?= {
					while = {
						count = PREV.var:current_haunting_level_stewardship
						remove_character_modifier = haunting_stewardship_increase_modifier
					}
				}
				remove_variable = current_haunting_level_stewardship
			}
			if = { limit = { has_variable = current_haunting_level_intrigue } 
				var:haunting_target ?= {
					while = {
						count = PREV.var:current_haunting_level_intrigue
						remove_character_modifier = haunting_intrigue_increase_modifier
					}
				}
				remove_variable = current_haunting_level_intrigue
			}
			if = { limit = { has_variable = current_haunting_level_learning } 
				var:haunting_target ?= {
					while = {
						count = PREV.var:current_haunting_level_learning
						remove_character_modifier = haunting_learning_increase_modifier
					}
				}
				remove_variable = current_haunting_level_learning
			}
			set_variable = { name = haunting_target value = scope:recipient }
		}
		scope:recipient = {
			if = { limit = { scope:diplomacy_increase = yes }
				while = {
					count = scope:secondary_recipient.necromancy_power_diplomacy
					add_character_modifier = haunting_diplomacy_increase_modifier
				}
				
				scope:secondary_recipient = { set_variable = { name = current_haunting_level_diplomacy value = necromancy_power_diplomacy } }
			}
			if = { limit = { scope:martial_increase = yes }
				while = {
					count = scope:secondary_recipient.necromancy_power_martial
					add_character_modifier = haunting_martial_increase_modifier
				}
				
				scope:secondary_recipient = { set_variable = { name = current_haunting_level_martial value = necromancy_power_martial } }
			}
			if = { limit = { scope:stewardship_increase = yes }
				while = {
					count = scope:secondary_recipient.necromancy_power_stewardship
					add_character_modifier = haunting_stewardship_increase_modifier
				}
				
				scope:secondary_recipient = { set_variable = { name = current_haunting_level_stewardship value = necromancy_power_stewardship } }
			}
			if = { limit = { scope:intrigue_increase = yes }

					count = scope:secondary_recipient.necromancy_power_intrigue
					add_character_modifier = haunting_intrigue_increase_modifier
