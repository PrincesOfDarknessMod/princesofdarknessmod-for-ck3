POD_convert_to_wraith_interaction = {
	icon = POD_wraith_interaction
	desc = POD_convert_to_wraith_interaction_desc
	common_interaction = yes
	interface_priority = 90
    category = interaction_category_wraith
	ignores_pending_interaction_block = yes

	scheme = convert_to_wraith
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		OR = {
			scope:actor ={ has_trait = oblivionadvanced }
			scope:actor = { has_perk = touch_of_oblivion_perk }
		}
		scope:recipient = { POD_is_clone_trigger = no }
		scope:recipient = { can_convert_to_wraith_trigger = yes }
		POD_is_in_same_plane_trigger = yes
	}
	is_valid_showing_failures_only = {
		custom_description = {
			object = scope:recipient
			text = convert_to_wraith_truefaith
			scope:recipient = {
				NOR = {
					has_trait = truefaith
					has_trait = beatific
					}
				}
		}
		scope:actor = {
			can_start_scheme = {
				type = convert_to_wraith
				target = scope:recipient
			}
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			NOT = { has_trait = statue }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }	
			POD_is_not_clone_trigger = yes	
		}										
	}
	
	cost = { gold = scope:actor.medium_gold_value }
	
	on_accept = {
		scope:recipient = {
			add_character_flag = marked_for_return
		}
		scope:actor = {
			start_scheme = {
				type = convert_to_wraith
				target = scope:recipient
			}
		}
		custom_tooltip = POD_convert_to_wraith_interaction_desc
	
	}
	
	auto_accept = yes

}

POD_command_wraith_interaction = {
	desc = POD_command_wraith_interaction_desc
	common_interaction = yes
	interface_priority = 90
	category = interaction_category_wraith
	icon = POD_wraith_interaction
	
	is_shown = {
		scope:actor = {
			OR = {
				has_trait = wraith
				has_trait = oblivionadvanced
				has_perk = oblivions_sight_perk
				has_trait = necromancyadvanced
				has_perk = lifeless_tongues_perk
				has_perk = bind_the_dead_perk
				has_perk = divine_fetters_perk
				has_perk = saturnal_manes_perk
				has_perk = via_geniorum_perk
				has_perk = flesh_flux_perk
				has_perk = command_the_dead_perk
				has_perk = voice_of_the_yama_kings_perk
				has_perk = durance_perk
			}
		}
		scope:actor = { num_of_relation_bound_wraith >= 1 }
	}
	
	is_valid_showing_failures_only = {
		scope:recipient = {
			NOT = { has_trait = wraith }
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			NOT = { has_trait = statue }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }	
			POD_is_not_clone_trigger = yes	
		}										
	}
	
	populate_actor_list = {
		scope:actor = {
			every_relation = {
				type = bound_wraith
				limit = { 
					is_alive = yes 
					NOR = { 
						has_trait = torpor 
						has_character_modifier = traveling_spirit_world_modifier
					}
				}
				
				add_to_list = characters
			}
		}
	}
	send_option = {
		is_shown = {
			exists = scope:secondary_actor
			scope:actor = scope:recipient
		}
		is_valid = {
			NOT = { scope:secondary_actor = { is_courtier_of = scope:actor }}
			scope:secondary_actor = { is_ruler = no }
			scope:haunt = no
			scope:spy = no
			scope:possess = no
			scope:depossess = no
			scope:hook = no
		}
		flag = recall
		localization = command_wraith_interaction.recall
	}
	send_option = {
		is_shown = {
			exists = scope:secondary_actor
		}
		is_valid = {
			scope:haunt = no
			scope:spy = no
			scope:depossess = no
				
			trigger_if = {
				limit = { scope:recipient = {has_character_flag = possessed } }
				scope:recipient = { var:wraith = scope:secondary_actor }
			}
			NOT = {
				scope:recipient = {
					has_opinion_modifier = {
						target = scope:secondary_actor
						modifier = POD_complete_possession_opinion
					}
				}
			}
			
			custom_description = {
				text = POD_not_in_possession
				scope:secondary_actor = { 
					OR = { 
						has_perk = legion_perk
						num_of_relation_possessed_target = 0 
					}
				}
				scope:recipient = { num_of_relation_possessesing_wraith = 0 }
			}
			
			scope:secondary_actor = { 
				NOT = { has_trait = risen }
				trigger_if = {
					limit = { 
						has_hook_of_type  = {
							target = scope:recipient
							type = possessed_hook
						}
					}
					has_perk = rein_in_the_mind_perk
				}
				trigger_if = {
					limit = { 
						has_hook_of_type  = {
							target = scope:recipient
							type = possessed_weak_hook
						}
					}
					has_perk = masters_voice_perk
				}
				trigger_if = {
					limit = {  
						scope:recipient = {
							POD_can_be_possessed_trigger = no
						}
					}
					has_perk = vector_perk
				}
			}
			scope:recipient = {
				NOT = { has_perk = banish_the_dead_perk }
			}
		}
		flag = possess
		localization = command_wraith_interaction.possess
	
	}
	
	send_option = {
		is_shown = {
			exists = scope:secondary_actor
			scope:recipient = { 
				has_variable = wraith
				var:wraith = scope:secondary_actor 
			}
		}
		is_valid = {
			scope:haunt = no
			scope:spy = no
			scope:possess = no
		
		}
		flag = depossess
		localization = command_wraith_interaction.depossess
	
	}
	send_option = {
		is_shown = { exists = scope:secondary_actor }
		is_valid = {
			scope:spy = no
			scope:possess = no
			scope:depossess = no
			scope:recipient = {
				NOT = { has_perk = sense_the_dead_perk }
				NOT = { has_trait = statue }
				POD_is_not_clone_trigger = yes
			}
		}
		flag = haunt
		localization = command_wraith_interaction.haunt
	
	}
	send_option = {
		is_shown = { exists = scope:secondary_actor }
		is_valid = {
			scope:haunt = no
			scope:possess = no
			scope:depossess = no
			scope:recipient = {
				any_secret = {
					NOT = { is_known_by = scope:actor }
				}
				NOT = {
					has_perk = sense_the_dead_perk
				}
			}
		}
		flag = spy
		localization = command_wraith_interaction.spy
	}
	send_option = {
		is_shown = {
			exists = scope:secondary_actor
		}
		is_valid = {
			scope:actor = {
				has_usable_hook = scope:secondary_actor
			}
			OR = {
				scope:haunt = yes
				scope:possess = yes
				scope:spy = yes
			}
		}
		flag = hook
		localization = command_wraith_interaction.hook
	}

	
	send_options_exclusive = no
	
	on_accept = {
		scope:actor = { save_scope_as = necromancer }
		scope:secondary_actor = { save_scope_as = actor }
		if = {
			limit = { scope:possess = yes }
			scope:recipient = {

				trigger_event = POD_wraith_possession.100
				custom_tooltip = POD_command_wraith_possess_desc
			}
		}
		if = {
			limit = { scope:depossess = yes }
			 POD_wraith_end_possession = {
				 WRAITH = scope:actor
				 PUPPET = scope:recipient
			 }
		}
		if = {
			limit = { scope:spy = yes }
			scope:recipient = {
				
				trigger_event = POD_wraith_spying.100
				custom_tooltip = POD_command_wraith_spy_desc
			}
		}
		if = {
			limit = { scope:haunt = yes }
			scope:actor = {

				trigger_event = POD_wraith_haunting.100
				custom_tooltip = POD_command_wraith_haunt_desc
			}
		}
		if = {
			limit = { scope:hook = yes }
			scope:necromancer = {
				use_hook = scope:actor
			}
		}
		if = {
			limit = { scope:secondary_actor = { is_ruler = no } }
		
			hidden_effect = {
				scope:secondary_actor = {
					set_employer = scope:recipient
					return_to_court = yes
				}
			}
			custom_tooltip = POD_command_wraith_interaction.movecourt
		
		}
		if = {
			limit = { scope:recall = yes }
			hidden_effect = {
				scope:secondary_actor = {
					set_employer = scope:recipient
					return_to_court = yes
					add_character_flag = {
						flag = blocked_from_leaving
						years = 5
					}
				}
			}
			#custom_tooltip = POD_command_wraith_interaction.movecourt
		}
		
		
	}
	can_send = {
		OR = {
			scope:haunt = yes
			scope:spy = yes
			scope:possess = yes
			scope:depossess = yes
			scope:recall = yes
		}
	
	}
	auto_accept = {
		custom_description = {
			text = "spending_hook"
			subject = scope:actor
			object = scope:secondary_actor
			scope:hook = yes
			scope:actor = { has_strong_hook = scope:secondary_actor }
		}
	
	}
	
	ai_accept = {
		base = -50
		opinion_modifier = { # More likely to accept if Recipient likes the Actor
			who = scope:secondary_actor
			opinion_target = scope:actor
			multiplier = 1.0
			desc = AI_OPINION_REASON
		}
		modifier = {
			add = 50
			scope:secondary_actor = {
				has_dread_level_towards = {
					target = scope:actor
					level = 1
				}
			}
			desc = INTIMIDATED_REASON
		}
		modifier = {
			add = 100
			scope:secondary_actor = {
				has_dread_level_towards = {
					target = scope:actor
					level = 2
				}
			}
			desc = COWED_REASON
		}
		modifier = {
			scope:secondary_actor.house = scope:actor.house
			add = 50
			desc = IS_SAME_HOUSE_REASON
		}
	}

}
