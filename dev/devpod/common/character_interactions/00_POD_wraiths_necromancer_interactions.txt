	icon = POD_wraith_interaction
	desc = POD_convert_to_wraith_interaction_desc
	common_interaction = yes
	interface_priority = 90
    category = interaction_category_wraith
	ignores_pending_interaction_block = yes

	scheme = convert_to_wraith
	
	is_shown = {
		always = no
		#NOT = { scope:actor = scope:recipient }
		#scope:actor = { POD_can_covert_to_wraith_trigger = yes }
		#scope:recipient = { POD_is_clone_trigger = no }
		#scope:recipient = { can_convert_to_wraith_trigger = yes }
	}

	is_valid_showing_failures_only = {
		custom_description = {
			object = scope:recipient
			text = convert_to_wraith_truefaith # elderly kukul junk, can be delete or keeped
			scope:recipient = {
				POD_has_true_faith_trigger = no
			}
		}
		scope:actor = {
			can_start_scheme = {
				type = convert_to_wraith
				target_character = scope:recipient
			}
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			NOT = { has_trait = automaton }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }	
			POD_is_not_clone_trigger = yes	
		}
		POD_cross_umbra_interaction_valid_trigger = yes										
	}
	
	cost = { gold = scope:actor.medium_gold_value }
	
	on_accept = {
		scope:recipient = {
			add_character_flag = marked_for_return
		}
		scope:actor = {
					begin_scheme_basic_effect = {
						SCHEME_TYPE = convert_to_wraith
						TARGET_TYPE = target_character
						TARGET_SCOPE = scope:recipient
					}				
		}
		custom_tooltip = POD_convert_to_wraith_interaction_desc
	
	}
	
	auto_accept = yes

}

		scope:recipient = {

					}

			}
		}
	}

				scope:actor = { num_of_relation_bound_wraith < max_wraith_cap }

			}
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
			POD_is_not_clone_trigger = yes
		}
		POD_cross_umbra_interaction_valid_trigger = yes								
	}

		scope:recipient = {
			if = { 
				limit = { has_character_modifier = caul_modifier }
				remove_character_modifier = caul_modifier
				remove_trait = torpor
			}
			trigger_event = POD_wraith_binding.100
		}
		custom_tooltip = POD_bind_interaction_desc
	
	}
	
	auto_accept = yes

			scope:actor = { has_perk = soul_feasting_perk }
		}
		flag = vamp_devour
		localization = "EXECUTION_DEVOUR_SOUL"
	}
	
	send_option = {
		is_shown = {
			exists = scope:secondary_recipient
			scope:actor = { has_POD_kj_blood_trigger = yes }
		}
		is_valid = {
			scope:actor = { has_perk = spirit_eating_perk }
		}
		flag = kj_devour
		localization = "EXECUTION_DEVOUR_SOUL"
	}
	
	send_option = {
		is_shown = {
			exists = scope:secondary_recipient
			scope:actor = { has_trait = demon }
		}
		is_valid = {
			trigger_if = {
				limit = { scope:secondary_recipient = { is_a_thrall = yes } }
				scope:actor = { has_perk = ravage_perk }
			}
			trigger_else = {
				scope:actor = { has_perk = theft_of_faith_perk }
			}
		}
		flag = demon_devour
		localization = "EXECUTION_DEVOUR_SOUL"

	send_option = {

		is_shown = {

			scope:actor = { has_trait = wraith }
		}
		is_valid = {
			scope:actor = { has_trait = spectre }
		}
		flag = wraith_devour
		localization = "EXECUTION_DEVOUR_SOUL"
	}
	
	send_option = {
		is_shown = {
			exists = scope:secondary_recipient
		}
		flag = release
		localization = command_wraith_interaction.release
	}
	
	on_accept = {
		if = {
			limit = { scope:vamp_devour = yes }
			scope:actor = {
				POD_drain_victim_effect = {
					VICTIM = scope:secondary_recipient
				}
			}
		}
		if = {
			limit = { scope:kj_devour = yes }
			POD_absorb_chi_from_flesh_or_blood_effect = { MURDERER = scope:actor VICTIM = scope:recipient }
			execute_opinion_effect = { VICTIM = scope:secondary_recipient EXECUTIONER = scope:actor }
			scope:secondary_recipient = { 
				death = { 
					death_reason = death_souleaten 
					killer = scope:actor
				}
			}
		}
		if = {
			limit = { scope:demon_devour = yes }
			scope:actor = {
				POD_resolve_income_effect = { COUNT = 5 }
				if = {
					limit = { has_perk = devour_the_soul_perk }
					if = {
						limit = { NOT = { any_memory = { has_memory_type = devore_mind_memory } } }
						add_stress = massive_stress_impact_gain
						trigger_event = kueijin_scheme.140
						create_character_memory = {
							type = devore_mind_memory
						}
						custom_tooltip = execution_devour_soul_tooltip
					}
					else = { 
						custom_tooltip = used_devour_soul
					}
				}
			}
			execute_opinion_effect = { VICTIM = scope:secondary_recipient EXECUTIONER = scope:actor }
			scope:secondary_recipient = { 
				death = { 
					death_reason = death_souleaten 
					killer = scope:actor
				}
			}
		}
		if = {
			limit = { scope:mummy_devour = yes }
			scope:actor = { POD_sekhem_income_effect = { COUNT = 10 } }
			execute_opinion_effect = { VICTIM = scope:secondary_recipient EXECUTIONER = scope:actor }
			scope:secondary_recipient = { 
				death = { 
					death_reason = death_souleaten 
					killer = scope:actor
				}
			}
		}
		if = {
			limit = { scope:wraith_devour = yes }
			scope:actor = { POD_pathos_or_angst_income_effect = { COUNT = 5 } }
			execute_opinion_effect = { VICTIM = scope:secondary_recipient EXECUTIONER = scope:actor }
			scope:secondary_recipient = { 
				death = { 
					death_reason = death_souleaten 
					killer = scope:actor
				}
			}
		}
		if = {
			limit = { scope:release = yes }
			POD_unbind_wraith_effect = {
				WRAITH = scope:secondary_recipient
				NECROMANCER = scope:actor
			}
		}
		if = {
			limit = { scope:recall = yes }
			scope:secondary_recipient = {
				scope:actor = { add_courtier = prev }
				hidden_effect = {
					return_to_court = yes
				}
				add_character_flag = {
					flag = blocked_from_leaving
					years = 5
				}
			}
		}
	}
	
	auto_accept = yes
}

POD_command_wraith_recall_interaction = {
	desc = POD_command_wraith_recall_interaction_desc
	common_interaction = yes
	interface_priority = 90
	category = interaction_category_wraith
	icon = POD_wraith_interaction
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { POD_can_command_wraith_trigger = yes }
		scope:recipient = { has_relation_necromancer = scope:actor }
		NOT = { scope:recipient = { is_courtier_of = scope:actor } }
	}
	
	is_valid_showing_failures_only = {
		scope:recipient = { is_ruler = no }
		scope:recipient = { is_imprisoned = no }
		scope:recipient = { NOT = { has_trait = torpor } }
		POD_cross_umbra_interaction_valid_trigger = yes
	}

	#Use hook
	send_option = {
		is_valid = {
			scope:actor = {
				has_usable_hook = scope:recipient
			}
		}
		flag = hook	
		localization = GENERIC_SPEND_A_HOOK
	}
	should_use_extra_icon = {
		scope:actor = { has_usable_hook = scope:recipient }
	}
	extra_icon = "gfx/interface/icons/character_interactions/hook_icon.dds"
	
	send_options_exclusive = no
	
	on_accept = {
		if = {
			limit = { scope:hook = yes }
			scope:actor = { use_hook = scope:recipient }
		}
		scope:recipient = {
			scope:actor = { add_courtier = prev }
			hidden_effect = {
				return_to_court = yes
			}
			add_character_flag = {
				flag = blocked_from_leaving
				years = 5
			}
		}
	}
	
	auto_accept = {
		custom_description = {
			text = "spending_hook"
			subject = scope:actor
			object = scope:recipient
			scope:hook = yes
		}
	}
	
	ai_min_reply_days = 0
	ai_max_reply_days = 0
	
	ai_accept = {
		base = -50
		
		wraith_ai_accept_modifier = yes
	}
}
