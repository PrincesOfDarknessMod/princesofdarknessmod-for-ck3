POD_hunt_target_interaction = {
	category = interaction_category_journeys
	common_interaction = yes
	icon = vamphunt_interaction
	
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:actor = { is_vampire_trigger = yes }
				}
				desc = POD_hunt_target_interaction_desc_vampire
			}
			triggered_desc = {
				trigger = { scope:actor = { has_trait = kueijin } }
				desc = POD_hunt_target_interaction_desc_chi
			}
			triggered_desc = {
				trigger = { scope:actor = { has_trait = banemummy } }
				desc = POD_hunt_target_interaction_desc_banemummy
			}
		}
	}
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		OR = {
			# Vampire hunt
			AND = {
				scope:actor = { is_vampire_trigger = yes }
				scope:recipient = { is_vampire_trigger = no }
			}
			# Bane hunt
			AND = {
				scope:actor = { has_trait = banemummy }
				scope:recipient = { NOT = { has_trait = banemummy } }
			}
			# Dharma Hunt
			AND = {
				scope:actor = { has_trait = dharma }
				scope:recipient = { NOT = { has_trait = dharma } }
			}
		}
		scope:recipient = {
			POD_can_be_vamphunted_scheme_trigger = yes
		}
		POD_hostile_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			POD_not_already_on_journey_trigger = {
				TYPE   = hunt_target
				TARGET = scope:recipient
			}
			POD_can_start_planning_phase_trigger = yes
		}
		POD_cant_target_coterie_trigger = yes
		
		custom_description = {
			object = scope:recipient
			text = "blood_is_holy"
			scope:recipient = {
				NOR = {
					POD_has_true_faith_trigger = yes
					has_trait = orison_sanguine
				}
			}
		}
		scope:actor = {
			is_adult = yes
			is_imprisoned = no
		}
		
		scope:recipient = {
			is_adult = yes
			is_ruler = no
			is_imprisoned = no
		}
		trigger_if = {
			limit = {
				scope:actor = {
					is_ventrue_trigger = yes
					is_ai = no
				}
			}
			custom_tooltip = {
				text = ventrue_bane_warning
				
				ventrue_weakness_compatible_trigger = {
					ACTOR = scope:actor
					RECIPIENT = scope:recipient
				}
			}
		}
		
		POD_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		if = {
			limit = {
				scope:actor = { is_vampire_trigger = yes }
			}
			custom_tooltip = POD_hunt_target_interaction_effect_vampire
		}
		else_if = {
			limit = { scope:actor = { has_trait = banemummy } }
			custom_tooltip = POD_hunt_target_interaction_effect_banemummy
		}
		
		custom_tooltip = POD_hunt_target_interaction_monthly
		
		POD_journey_interaction_pause_hint = yes
		
		hidden_effect = {
			scope:actor = {
				POD_start_targeted_journey_effect = {
					TYPE    = hunt_target
					MONTHLY = hunt_journey
					TARGET  = scope:recipient
				}
			}
		}
	}

	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

	target_filter = recipient_artifacts

		}

			}
		}

				exists = scope:target

		}
		trigger_else_if = {
			limit = {
				exists = scope:target
				scope:target = { rarity = masterwork }
			}

			limit = {
				exists = scope:target

		}

			limit = {

			limit = {
				exists = scope:target
