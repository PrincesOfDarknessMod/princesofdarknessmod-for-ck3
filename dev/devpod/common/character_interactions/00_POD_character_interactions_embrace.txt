embrace_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	#interface_priority = 83
	scheme = embrace
	
	ai_frequency = 0

	is_shown = {
		scope:actor = { is_ai = no }
		NOT = { scope:actor = scope:recipient }
		scope:actor = { 
			is_vampire_trigger = yes  
			NOT = { has_trait = abomination }
		}
		scope:recipient = { POD_can_be_embraced_scheme_trigger = yes }
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		#Catubodua can't Embrace
		scope:actor = { NOT = { this = character:791232 } }
		scope:actor = {is_vampgeneration16_trigger = no}
		#Adisa can only embrace females
		OR = {
			scope:actor = { NOT = { this = character:791706 } }
			scope:recipient = { is_female = yes }
		}
		scope:recipient = { POD_is_clone_trigger = no }
		scope:recipient = { is_imprisoned = no }
	}

	is_valid_showing_failures_only = {
		custom_description = {
			object = scope:actor
			text = "I_can_only_plot_one_embrace_at_time"
			scope:actor = {
				NOT = {
					any_scheme = {
						scheme_type = embrace
					}
				}
			}
		}
		custom_description = {
			object = scope:recipient
			text = "blood_is_holy"
			scope:recipient = {
				NOR = {
					has_trait = truefaith
					has_trait = beatific
					has_trait = orison_sanguine
					}
				}
		}
		custom_description = {
			object = scope:actor
			text = "I_must_be_at_the_same_location"
			scope:actor = { exists = location }
			scope:recipient = { exists = location }
			scope:actor = { location = scope:recipient.location }
		}	
		scope:actor = {
			can_start_scheme = {
				type = embrace
				target = scope:recipient
			}
		}
		trigger_if = {
			limit = {
				scope:recipient = { POD_is_supehunter_trigger = yes }
			}
			scope:recipient = { is_imprisoned_by = scope:actor }
		}
		scope:recipient = {
			is_landed = no
		}
	}

	desc = {
		# triggered_desc = {
			# trigger = { scope:recipient.age >= scope:actor.age }
			# desc = embrace_childe_older_than_you_tt
		# }
		triggered_desc = {
			trigger = {
				scope:actor = {
					NOT = { is_at_war_with = scope:recipient }
					can_start_scheme = {
						type = embrace
						target = scope:recipient
					}

				}
			}
			desc = scheme_interaction_tt_embrace_approved
		}

	}

	on_accept = {
		hidden_effect = {
			scope:actor = {
				start_scheme = {
					type = embrace
					target = scope:recipient
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_embrace_prisoner_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	icon = embrace_interaction
	desc = POD_embrace_prisoner_interaction_desc
	
	ai_frequency = 0

	is_shown = {
		scope:actor = { is_ai = no }
		NOT = { scope:actor = scope:recipient }
		scope:recipient = {is_imprisoned_by = scope:actor}
		scope:actor = { 
			is_vampire_trigger = yes  
			NOT = { has_trait = abomination }
		}
		scope:recipient = { POD_can_be_embraced_scheme_trigger = yes }
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		#Catubodua can't Embrace
		scope:actor = { NOT = { this = character:791232 } }
		scope:actor = {is_vampgeneration16_trigger = no}
		#Adisa can only embrace females
		OR = {
			scope:actor = { NOT = { this = character:791706 } }
			scope:recipient = { is_female = yes }
		}
		scope:recipient = { POD_is_clone_trigger = no }
	}

	is_valid_showing_failures_only = {
		custom_description = {
			object = scope:recipient
			text = "blood_is_holy"
			scope:recipient = {
				NOR = {
					has_trait = truefaith
					has_trait = beatific
					has_trait = orison_sanguine
				}
			}
		}
		scope:recipient = {
			is_landed = no
		}
		scope:actor = { faith = { NOT = { has_doctrine_parameter = willing_embrace } } }
	}

	on_accept = {
		scope:actor = { save_scope_as = owner }
		scope:recipient = { save_scope_as = target }
		scope:actor = { custom_tooltip = POD_embrace_prisoner_interaction.tt }
		
		if = {
			limit = { scope:recipient = { POD_is_supehunter_trigger = yes } }
			scope:actor = { trigger_event = POD_embrace.200 }
		}
		else_if = {
			limit = { scope:recipient = { POD_is_werewolf_trigger = yes } }
			scope:actor = { trigger_event = POD_embrace.400 }
		}
		else = {
			scope:actor = { trigger_event = POD_embrace.100 }
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
