POD_theurgy_heal_interaction = {
	category = interaction_category_holyarts
	icon = POD_theurgy_interaction
	desc = POD_theurgy_heal_interaction_desc
	options_heading = POD_theurgy_heal_interaction_options_heading
	
	cooldown = { months = 24 }
	cost = { piety = medium_piety_value }
	
	is_shown = {
		scope:actor = { has_perk = via_medicamenti_1_perk }
		POD_incompatible_splats_trigger = no
	}
	
	is_valid = {
		scope:recipient = {
			OR = {
				has_trait = scarred
				has_trait = disfigured
				has_trait = wounded
				has_any_mortal_disease_trigger = yes
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			POD_can_be_healed_trigger = yes
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = scarred
			}
		}	
		is_valid = {
			scope:actor = { 
				has_perk = via_medicamenti_2_perk
			}
			scope:recipient = { 
				custom_description = {
					text = POD_permanent_scarred
					object = scope:recipient
					NOT = { has_character_flag = POD_permanent_scarring_flag }
				}
			}
		}
		flag = scarred
		localization = POD_ha_holy_heal_scarred
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = disfigured
			}
		}
		is_valid = {
			scope:actor = { 
				has_perk = via_medicamenti_2_perk
			}
			scope:recipient = { 
				custom_description = {
					text = POD_permanent_disfigured
					object = scope:recipient
					NOT = { has_character_flag = POD_permanent_disfigured_flag }
				}
			}
		}
		flag = disfigured
		localization = POD_ha_holy_heal_disfigured
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = wounded
			}
		}	
		is_valid = {
			scope:actor = { 
				has_perk = via_medicamenti_3_perk
			}
		}
		flag = wounded
		localization = POD_heal_wounded
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = ill
			}
		}	
		is_valid = {
			scope:actor = { 
				has_perk = via_medicamenti_4_perk
			}
		}
		flag = ill
		localization = POD_heal_ill
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_any_mortal_disease_trigger = yes
			}
		}	
		is_valid = {
			scope:actor = { 
				has_perk = via_medicamenti_5_perk
			}
		}
		flag = disease
		localization = POD_heal_disease
	}
	
	on_accept = {
		if = {
			limit = { NOT = { scope:actor = scope:recipient } }
			scope:actor = { POD_commit_heal_effect = yes }
		}
		scope:actor = { 
			POD_willpower_rouse_check_effect = yes
		}
		scope:recipient = {
			play_music_cue = "mx_cue_peace_ensues"
			add_character_modifier = { modifier = theurgy_soothe_pain_modifier years = 4 }

			switch = {
				trigger = yes
				scope:scarred = {
					remove_trait = scarred
				}
				scope:disfigured = {
					remove_trait = disfigured
				}
				scope:wounded = {
					POD_remove_wound_trait_effect = yes
				}
				scope:ill = {
					recover_from_disease_effect = { DISEASE = ill }
					remove_disease_treatment_effect = yes
				}
				scope:disease = {
					remove_disease_trait_effect = yes
				}
			}
		}
		if = {
			limit = { NOT = { scope:actor = scope:recipient } }
			scope:recipient = { 
				add_opinion = {
					target = scope:actor
					modifier = pleased_opinion
					opinion = 15
				}
			}
		}
	}
	
	auto_accept = yes

	ai_potential = {
		# The basic requirements
		is_imprisoned = no
		is_landed = yes
		
		piety > major_piety_value
	}
	ai_targets = {
		ai_recipients = self
	}
	ai_targets = {
		ai_recipients = courtiers
		ai_recipients = peer_vassals
		max = 10
	}
	ai_frequency = 120
	ai_will_do = {
		base = 0
		modifier = {
			add = 100
			scope:recipient = {
				OR = {
					this = scope:actor
					is_knight_of = scope:actor
				}
			}
		}
	}
}

POD_train_theurgy_interaction = {
	category = interaction_category_holyarts
	icon = POD_theurgy_interaction
    desc = POD_train_theurgy_interaction_desc
	common_interaction = yes
	
	cooldown = { months = 12 }
	cost = { piety = medium_piety_value }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_trait = theurgist }
		scope:recipient = { has_trait = theurgist }
		POD_incompatible_splats_trigger = no
	}

	is_valid_showing_failures_only = {
	    scope:actor = {
			theurgy_lifestyle_perk_points >= 1
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}
	
	on_accept = {
	    scope:actor = { 
			add_theurgy_lifestyle_perk_points = -1
			stress_impact = {
				base = medium_stress_gain
			}
		}			
	    scope:recipient = { 
			add_theurgy_lifestyle_perk_points = 1 
			add_opinion = {
				target = scope:actor
				modifier = mentored_me_opinion
			}
		}	
	}

	ai_frequency = 0
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_initiate_theurgy_interaction = {
	category = interaction_category_holyarts
	icon = POD_theurgy_interaction
    desc = POD_initiate_theurgy_interaction_desc
	
	cooldown = { months = 12 }
	cost = { piety = major_piety_value }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_trait = theurgist }
		scope:recipient = { NOT = { has_trait = theurgist } }
		POD_incompatible_splats_trigger = no
	}

	is_valid_showing_failures_only = {
	    scope:actor = {
			theurgy_lifestyle_perk_points >= 1
		}
		scope:recipient = { 
			is_imprisoned = no 
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}
	
	on_accept = {
	    scope:actor = { 
			add_theurgy_lifestyle_perk_points = -1
			stress_impact = {
				base = major_stress_gain
			}
			duel = {
				skill = learning
				value = 15
				25 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					scope:recipient = {
						add_trait = theurgist
						add_opinion = {
							target = scope:actor
							modifier = mentored_me_opinion
						}
					}
				}
				50 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier 
					}
				}
			}
		}	
	}

	ai_frequency = 0
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
