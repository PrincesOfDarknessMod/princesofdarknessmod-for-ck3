POD_theurgy_heal_interaction = {
	category = interaction_category_holyarts
	icon = POD_theurgy_interaction
	desc = POD_theurgy_heal_interaction_desc
	options_heading = POD_theurgy_heal_interaction_options_heading
	
	cooldown = { months = 24 }
	cost = { piety = medium_piety_value }
	
	is_shown = {
		scope:actor = { has_perk = via_medicamenti_level_1_perk }
	}
	
	is_valid = {
		scope:recipient = {
			OR = {
				has_trait = scarred
				has_trait = disfigured
				has_trait = wounded
				has_any_mortal_disease_trigger = yes
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = { NOT = { has_trait = mage } }
		scope:recipient = {
			POD_can_be_healed_trigger = yes
		}
	}
	
	send_option = {
		is_valid = {
			scope:actor = { has_perk = via_medicamenti_level_1_perk }
		}
		flag = level_1
		localization = POD_theurgy_healing_level_1
	}
	send_option = {
		is_valid = {
			scope:actor = { 
				custom_tooltip = {
					text = "has_small_wounds_or_scars"
					OR = {
						has_trait = wounded_1
						AND = {
							custom_description = {
								text = POD_permanent_scarred
								object = scope:recipient
								scope:recipient = { NOT = { has_character_flag = POD_permanent_scarring_flag } }
							}
							has_trait = scarred
						}
					}
				}
				has_perk = via_medicamenti_level_2_perk 
			}
		}
		flag = level_2
		localization = POD_theurgy_healing_level_2
	}
	send_option = {
		is_valid = {
			scope:actor = { 
				custom_tooltip = {
					text = "has_wounds_or_disfigured"
					OR = {
						has_trait = wounded
						has_trait = disfigured
					}
				}
				has_perk = via_medicamenti_level_3_perk 
			}
		}
		flag = level_3
		localization = POD_theurgy_healing_level_3
	}
	send_option = {
		is_valid = {
			scope:actor = { 
				has_trait = ill
				has_perk = via_medicamenti_level_4_perk 
			}
		}
		flag = level_4
		localization = POD_theurgy_healing_level_3
	}
	send_option = {
		is_valid = {
			scope:actor = { 
				custom_tooltip = {
					text = "has_mortal_disease"
					has_any_mortal_disease_trigger = yes
				}
				has_perk = via_medicamenti_level_5_perk 
			}
		}
		flag = level_5
		localization = POD_theurgy_healing_level_5
	}
	
	on_accept = {
		if = {
			limit = { NOT = { scope:actor = scope:recipient } }
			scope:actor = { POD_commit_heal_effect = yes }
		}
		scope:actor = { 
			POD_willpower_rouse_check_effect = yes
		}
		scope:recipient = {
			play_music_cue = "mx_cue_peace_ensues"

			if = {
				limit = {
					scope:level_1 = yes
				}
				scope:recipient = {
					add_character_modifier = { modifier = theurgy_soothe_pain_modifier years = 4 }
				}
			}	

			if = {
				limit = {
					scope:level_2 = yes
				}
				scope:recipient = {
					if = { limit = { has_trait = scarred } remove_trait = scarred }
					if = { limit = { has_trait = wounded_1 } remove_trait = wounded_1 }
					add_character_modifier = { modifier = theurgy_easy_healing_process_modifier years = 1 }
				}
			}
			
			if = {
				limit = {
					scope:level_3 = yes
				}
				scope:recipient = {
					if = { limit = { has_trait = wounded } POD_remove_wound_trait_effect = yes }
					if = { limit = { has_trait = disfigured } remove_trait = disfigured }
					add_character_modifier = { modifier = theurgy_healing_process_modifier years = 2 }
				}
			}
			if = {
				limit = {
					scope:level_4 = yes
				}
				scope:recipient = {		
					recover_from_disease_effect = { DISEASE = ill }
					remove_disease_treatment_effect = yes
					add_character_modifier = { modifier = theurgy_hard_healing_process_modifier years = 3 }
				}
			}	

			if = {
				limit = {
					scope:level_5 = yes
				}
				scope:recipient = {		
					remove_disease_trait_effect = yes
					add_character_modifier = { modifier = theurgy_very_hard_healing_process_modifier years = 4 }
				}
			}
		}
		if = {
			limit = { NOT = { scope:actor = scope:recipient } }
			scope:recipient = { 
				add_opinion = {
					target = scope:actor
					modifier = pleased_opinion
					opinion = 15
				}
			}
		}
	}
	
	auto_accept = yes

	ai_potential = {
		# The basic requirements
		is_imprisoned = no
		is_landed = yes
		
		piety > major_piety_value
	}
	ai_targets = {
		ai_recipients = self
	}
	ai_targets = {
		ai_recipients = courtiers
		ai_recipients = peer_vassals
		max = 10
	}
	ai_frequency = 12
	ai_will_do = {
		base = 0
		modifier = {
			add = 100
			scope:recipient = {
				OR = {
					this = scope:actor
					is_knight_of = scope:actor
				}
			}
		}
	}
}

POD_theurgy_discover_true_name_interaction = {
	category = interaction_category_holyarts
	scheme = POD_discover_true_name
	use_diplomatic_range = no
	icon = POD_theurgy_interaction
	cooldown = { years = 5 }
	
	cost = { piety = 500 }
	
	desc = {
		triggered_desc = {
			trigger = {
				scope:actor = {
					NOT = { is_at_war_with = scope:recipient }
					can_start_scheme = {
						type = POD_discover_true_name
						target = scope:recipient
					}

				}
			}
			desc = scheme_interaction_tt_POD_discover_true_name_approved
		}
	}

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			has_perk = via_geniorum_level_3_perk
		}
		scope:recipient = { POD_is_clone_trigger = no }
		scope:actor = { 
			NOT = { 
				has_hook_of_type = {
					target = scope:recipient
					type = true_name_hook
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			can_start_scheme = {
				type = POD_discover_true_name
				target = scope:recipient
			}
		}
		POD_cross_umbra_interaction_valid_trigger = yes	
	}

	on_accept = {
		hidden_effect = {
			scope:actor = {
				start_scheme = {
					type = POD_discover_true_name
					target = scope:recipient
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no 
		is_landed = yes 
	}
	ai_targets = {
		ai_recipients = liege
	}
	ai_targets = {
		ai_recipients = neighboring_rulers
		ai_recipients = peer_vassals
		max = 10
	}
	ai_frequency = 72
	ai_will_do = {
		base = 0
		
		opinion_modifier = {
			opinion_target = scope:recipient
			multiplier = -0.25
		}
		modifier = {
		    add = 50
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
		modifier = {
			factor = 0
			scope:recipient = {
				is_ai = no
			}
		}
	}
}

	use_diplomatic_range = no

	is_shown = {
		scope:actor = scope:recipient

			has_perk = via_necromantiae_level_5_perk
		}
	}
	on_accept = {
		custom_tooltip = "summon_random_wraith"
		scope:actor = {
			POD_willpower_rouse_check_effect = yes

				create_character = {
					save_scope_as = summoned_wraith
					dynasty = none
					template = POD_wraith_character
					location = this.location
				}
				send_interface_toast = {
					title = POD_theurgy_summon_wraith_from_shadowlands_interaction
					
					right_icon = scope:actor
					left_icon = scope:summoned_wraith
				}
				
				add_courtier = scope:summoned_wraith
				scope:summoned_wraith = {
					hidden_effect = {
						return_to_court = yes
					}
					add_character_flag = {
						flag = blocked_from_leaving
						years = 5
					}
				}
			}
		}
	}
	ai_frequency = 0
	ai_will_do = {
		base = 0
	}
}

	cooldown = { months = 12 }
	cost = { piety = medium_piety_value }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_trait = theurgist }
		scope:recipient = { has_trait = theurgist }
	}

	is_valid_showing_failures_only = {
	    scope:actor = {
			theurgy_lifestyle_perk_points >= 1
			NOT = { faith = { has_doctrine = doctrine_supehunting_crime } }
		}
		scope:recipient = { is_imprisoned = no }
	}
	
	on_accept = {
	    scope:actor = { 
			add_theurgy_lifestyle_perk_points = -1
			stress_impact = {
				base = medium_stress_gain
			}
		}			
	    scope:recipient = { 
			add_theurgy_lifestyle_perk_points = 1 
			add_opinion = {
				target = scope:actor
				modifier = mentored_me_opinion
			}
		}	
	}

	ai_frequency = 0
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

	}

	is_valid_showing_failures_only = {
	    scope:actor = {
			theurgy_lifestyle_perk_points >= 1
		}
		scope:recipient = {

	on_accept = {
	    scope:actor = { 
			add_theurgy_lifestyle_perk_points = -1

				base = major_stress_gain
			}
			duel = {
				skill = learning
				value = 10

					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					scope:recipient = {
						add_trait = theurgist
						add_opinion = {
							target = scope:actor
							modifier = mentored_me_opinion
						}
					}
				}
				50 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier 
					}
				}
			}
		}	
	}

	ai_frequency = 0
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
