		scope:recipient.fetter_bind_level < 10
		POD_hostile_interaction_is_shown = yes
	}

	is_valid = {
		scope:recipient = {
			OR = {
				custom_description = {
					object = scope:actor
					text = is_courtier_or_vassal_of
					OR = {
						is_courtier_of = scope:actor
						is_vassal_of = scope:actor
					}
				}
				is_imprisoned_by = scope:actor
			}
			custom_tooltip = {
				text = POD_bind_interaction_free_fetter
				free_fetter_points >= 1
			}
		}
	}

	is_valid_showing_failures_only = {
		custom_description = {
			text = POD_bind_interaction_limit
			OR = {
				scope:actor = { num_of_relation_bound_wraith < max_wraith_cap }
				scope:recipient = { has_relation_necromancer = scope:actor }
			}
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:actor = {
			POD_willpower_rouse_check_effect = yes
			duel = {
				skills = { diplomacy martial stewardship intrigue learning prowess }
				target = scope:recipient
				50 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_positive_duel_skill_multiplier
					}
					modifier = { add = POD_bind_chance_buff }

					desc = POD_bind_interaction.success
					scope:recipient = {
						if = {
							limit = { has_character_modifier = caul_modifier }
							remove_character_modifier = caul_modifier
							remove_trait = torpor
						}
						if = { limit = { POD_has_fetter_of_type = { TYPE = bind } }
							POD_increase_fetter_type_level = { TYPE = bind VALUE = 1 ACTION = add }
						}
						else = {
							POD_create_full_fetter = { TYPE = bind LEVEL = 1 OBJECT = scope:recipient }
						}
						custom_tooltip = POD_bind_interaction_hook
						custom_tooltip = POD_bind_interaction_level
						POD_give_hook_based_on_bind_level_effect = { NECROMANCER = scope:actor }
					}
					if = {
						limit = { POD_gains_boneflowers_dharma_xp = yes }
						POD_give_dharma_xp_effect = { VALUE = POD_kueijin_dharma_xp_value_medium }
					}
					if = {
						limit = { NOT = { has_relation_bound_wraith = scope:recipient } }
						set_relation_bound_wraith = scope:recipient
					}
				}
				50 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier
					}
					modifier = { add = { value = POD_bind_chance_buff multiply = -1 } }
					desc = POD_bind_interaction.failure
					POD_static_stress_gain = { COUNT = 25 }
				}
			}
			custom_tooltip = POD_bind_interaction_desc
		}
	}

	auto_accept = yes
}
