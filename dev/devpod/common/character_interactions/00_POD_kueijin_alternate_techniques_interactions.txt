POD_taking_the_face_interaction = {
	category = interaction_category_vampire
	desc = POD_taking_the_face_interaction_desc
	icon = POD_art_interaction
	
	is_shown = {
		OR = {
			NOT = { scope:actor = scope:recipient }
			scope:actor = { has_variable = original_face }
		}
		scope:actor = { has_POD_kj_blood_trigger = yes }
		scope:actor = { has_perk = taking_the_face_perk }
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			custom_tooltip = {
				text = "i_am_empty_of_chi"
				NAND = { 
					has_character_modifier = yinchi0_modifier
					has_character_modifier = yangchi0_modifier
				}
			}
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}
	
	send_option = {
		is_shown = {
			scope:actor = { has_variable = original_face }
		}
		is_valid = {
			scope:actor = scope:recipient
		}
		flag = restore_face
		localization = POD_vampshapeshift_interaction_restore_face
	}
	
	send_option = {
		is_valid = {
			NOT = { scope:actor = scope:recipient }
		}
		flag = copy_face
		localization = POD_vampshapeshift_interaction_copy_face
	}

	on_accept = {
		if = {
			limit = {
				scope:copy_face = yes
			}
			scope:actor = { 
				hidden_effect = { 
					if = {
						limit = { NOT = { has_variable = original_face } }
						POD_create_dummyclone_effect = { CHARACTER = scope:actor }
						if = {
							limit = { exists = scope:dummyclone }
							scope:dummyclone = { copy_inheritable_appearance_from = scope:actor }
							set_variable = {
								name = original_face
								value = scope:dummyclone
							}
						}
						else = {
							send_interface_toast = {
								title = POD_kueijin_spy.duel.failure
								left_icon = scope:actor
								right_icon = scope:recipient
							}
						}
					}
				}
				copy_inheritable_appearance_from = scope:recipient 
			}
			scope:actor = {
				POD_chi_cost_effect = { COUNT = 1 }
				add_alternate_techniques_lifestyle_xp = minor_lifestyle_xp
			}
		}
		if = {
			limit = {
				scope:restore_face = yes
			}
			scope:actor = { 
				copy_inheritable_appearance_from = var:original_face 
				remove_variable = original_face
			}
			scope:actor = {
				POD_chi_cost_effect = { COUNT = 1 }
				add_alternate_techniques_lifestyle_xp = minor_lifestyle_xp
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

			}
		}
		scope:recipient = { is_in_umbra_trigger = no }

	}

	is_valid_showing_failures_only = {

			NOT = { has_trait = torpor }

		}
		POD_cross_umbra_interaction_valid_trigger = yes
		scope:recipient = { is_ruler = no }				
	}

	on_accept = {
		scope:actor = { POD_chi_cost_effect = { COUNT = 1 } }

			}
			add_alternate_techniques_lifestyle_xp = minor_lifestyle_xp
		}
	}
	
	auto_accept = yes

		base = 0
