######################################################
# NOSFERATU — Pull a Thread from the Warren
# Buy a hook on a target character
######################################################
nosferatu_warren_buy_hook = {
	category = interaction_category_hostile
	desc = nosferatu_warren_buy_hook_desc
	icon = icon_scheme_blackmail

	auto_accept = yes

	cooldown = { years = 3 }

	is_shown = {
		scope:actor != scope:recipient
		scope:actor = {
			is_playable_character = yes
			POD_is_vampire_trigger = yes
			is_nosferatu_clan_or_bloodline_trigger = yes
			domicile ?= {
				has_domicile_building_or_higher = vampire_nosferatu_warren_network_01
			}
		}
		POD_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			# Level 1-2: Court members only
			trigger_if = {
				limit = {
					scope:actor = {
						domicile ?= {
							NOT = { has_domicile_building_or_higher = vampire_nosferatu_warren_network_03 }
						}
					}
				}
				custom_tooltip = {
					text = nosferatu_warren_buy_hook_court_only_tt
					is_courtier_of = scope:actor
				}
			}
			# Level 3-4: Realm characters
			trigger_else_if = {
				limit = {
					scope:actor = {
						domicile ?= {
							NOT = { has_domicile_building_or_higher = vampire_nosferatu_warren_network_05 }
						}
					}
				}
				custom_tooltip = {
					text = nosferatu_warren_buy_hook_realm_only_tt
					OR = {
						is_courtier_of = scope:actor
						target_is_liege_or_above = scope:actor
						is_vassal_of = scope:actor
					}
				}
			}
			# Level 5: Diplomatic range
			trigger_else = { always = yes }
		}
		scope:actor = {
			can_add_hook = {
				target = scope:recipient
				type = favor_hook
			}
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	cost = {
		gold = {
			value = 50
			if = {
				limit = { scope:recipient = { highest_held_title_tier >= tier_county } }
				add = 50
			}
			if = {
				limit = { scope:recipient = { highest_held_title_tier >= tier_duchy } }
				add = 100
			}
			if = {
				limit = { scope:recipient = { highest_held_title_tier >= tier_kingdom } }
				add = 200
			}
			if = {
				limit = { scope:recipient = { highest_held_title_tier >= tier_empire } }
				add = 200
			}
		}
	}

	on_accept = {
		scope:actor = {
			add_hook = {
				type = favor_hook
				target = scope:recipient
			}
		}
		scope:recipient = {
			if = {
				limit = { is_landed = no }
				add_gold = 50
			}
			else = {
				if = {
					limit = { highest_held_title_tier >= tier_empire }
					add_gold = 600
				}
				else_if = {
					limit = { highest_held_title_tier >= tier_kingdom }
					add_gold = 400
				}
				else_if = {
					limit = { highest_held_title_tier >= tier_duchy }
					add_gold = 200
				}
				else_if = {
					limit = { highest_held_title_tier >= tier_county }
					add_gold = 100
				}
				else = {
					add_gold = 50
				}
			}
		}
	}

	# Needs an option to decline before AI can use this
	# ai_targets = {
	# 	ai_recipients = liege
	# 	ai_recipients = vassals
	# }
	# ai_frequency = 24
	# ai_potential = {
	# 	gold >= 100
	# }
	# ai_will_do = {
	# 	base = 10
	# }
	ai_frequency = 0
	ai_will_do = { base = 0 }
}

######################################################
# SALUBRI WATCHER — Share Knowledge
# Both characters receive lifestyle XP
######################################################
salubri_watcher_share_knowledge = {
	category = interaction_category_friendly
	desc = salubri_watcher_share_knowledge_desc
	icon = icon_diplomacy

	is_shown = {
		scope:actor != scope:recipient
		scope:actor = {
			has_trait = salubri_watcher
			is_playable_character = yes
			POD_is_vampire_trigger = yes
			is_salubri_trigger = yes
			domicile ?= {
				has_domicile_building_or_higher = vampire_salubri_chronicle_vault_01
			}
		}
		scope:recipient = {
			is_salubri_trigger = yes
		}
		POD_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			is_available = yes
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	cost = {
		piety = 50
	}

	cooldown = {
		years = 10
	}

	# High acceptance — Salubri mutual aid
	ai_accept = {
		base = 80
	}

	on_accept = {
		# Determine XP amount based on building level
		# Level 1-2: 100 XP, Level 3-4: 200 XP, Level 5: 300 XP

		# Grant XP based on TARGET's caste
		if = {
			limit = { scope:recipient = { has_trait = salubri_warrior } }
			# Both gain martial XP
			custom_tooltip = salubri_watcher_share_knowledge_martial_tt
			scope:actor = {
				if = {
					limit = { domicile ?= { has_domicile_building_or_higher = vampire_salubri_chronicle_vault_06 } }
					add_martial_lifestyle_xp = 400
					scope:recipient = { add_martial_lifestyle_xp = 400 }
				}
				else_if = {
					limit = { domicile ?= { has_domicile_building_or_higher = vampire_salubri_chronicle_vault_05 } }
					add_martial_lifestyle_xp = 300
					scope:recipient = { add_martial_lifestyle_xp = 300 }
				}
				else_if = {
					limit = { domicile ?= { has_domicile_building_or_higher = vampire_salubri_chronicle_vault_03 } }
					add_martial_lifestyle_xp = 200
					scope:recipient = { add_martial_lifestyle_xp = 200 }
				}
				else = {
					add_martial_lifestyle_xp = 100
					scope:recipient = { add_martial_lifestyle_xp = 100 }
				}
			}
		}
		else_if = {
			limit = { scope:recipient = { has_trait = salubri_watcher } }
			# Both gain intrigue XP
			custom_tooltip = salubri_watcher_share_knowledge_intrigue_tt
			scope:actor = {
				if = {
					limit = { domicile ?= { has_domicile_building_or_higher = vampire_salubri_chronicle_vault_06 } }
					add_intrigue_lifestyle_xp = 400
					scope:recipient = { add_intrigue_lifestyle_xp = 400 }
				}
				else_if = {
					limit = { domicile ?= { has_domicile_building_or_higher = vampire_salubri_chronicle_vault_05 } }
					add_intrigue_lifestyle_xp = 300
					scope:recipient = { add_intrigue_lifestyle_xp = 300 }
				}
				else_if = {
					limit = { domicile ?= { has_domicile_building_or_higher = vampire_salubri_chronicle_vault_03 } }
					add_intrigue_lifestyle_xp = 200
					scope:recipient = { add_intrigue_lifestyle_xp = 200 }
				}
				else = {
					add_intrigue_lifestyle_xp = 100
					scope:recipient = { add_intrigue_lifestyle_xp = 100 }
				}
			}
		}
		else = {
			# Default (healer or no caste): both gain learning XP
			custom_tooltip = salubri_watcher_share_knowledge_learning_tt
			scope:actor = {
				if = {
					limit = { domicile ?= { has_domicile_building_or_higher = vampire_salubri_chronicle_vault_06 } }
					add_learning_lifestyle_xp = 400
					scope:recipient = { add_learning_lifestyle_xp = 400 }
				}
				else_if = {
					limit = { domicile ?= { has_domicile_building_or_higher = vampire_salubri_chronicle_vault_05 } }
					add_learning_lifestyle_xp = 300
					scope:recipient = { add_learning_lifestyle_xp = 300 }
				}
				else_if = {
					limit = { domicile ?= { has_domicile_building_or_higher = vampire_salubri_chronicle_vault_03 } }
					add_learning_lifestyle_xp = 200
					scope:recipient = { add_learning_lifestyle_xp = 200 }
				}
				else = {
					add_learning_lifestyle_xp = 100
					scope:recipient = { add_learning_lifestyle_xp = 100 }
				}
			}
		}
	}

	ai_targets = {
		ai_recipients = realm_characters
		chance = 0.5
	}
	ai_frequency = 12
	ai_potential = {
		always = yes
	}
	ai_will_do = {
		base = 30
	}
}

######################################################
# SETITE — Serpent's Embrace
# Launches the setite_serpent_embrace scheme to
# convert target to the Setite's faith
######################################################
pod_setite_serpent_embrace_interaction = {
	category = interaction_category_friendly
	desc = pod_setite_serpent_embrace_interaction_desc
	icon = icon_scheme_seduce

	is_shown = {
		scope:actor != scope:recipient
		scope:actor = {
			is_playable_character = yes
			POD_is_vampire_trigger = yes
			domicile ?= {
				has_domicile_building_or_higher = vampire_setite_hidden_temple_01
			}
			is_setite_trigger = yes
		}
		scope:recipient = {
			POD_is_vampire_trigger = yes
			is_adult = yes
			NOT = { religion = religion:roadofset_religion } 
		}
		POD_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			trigger_if = {
				limit = { is_landless_adventurer = no }
				is_travelling = no
			}
			custom_tooltip = {
				text = pod_setite_serpent_embrace_already_scheming_tt
				NOT = {
					any_scheme = {
						scheme_type = pod_setite_serpent_embrace
					}
				}
			}
		}
		scope:recipient = {
			is_imprisoned = no
			trigger_if = {
				limit = {
					scope:actor = { is_landless_adventurer = no }
				}
				is_travelling = no
			}
			custom_tooltip = {
				text = pod_setite_serpent_embrace_interaction_blocked_tt
				NOT = { has_character_flag = block_pod_setite_serpent_embrace }
			}
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:actor = {
			start_scheme = {
				type = pod_setite_serpent_embrace
				target_character = scope:recipient
			}
		}
	}

	auto_accept = yes

	ai_targets = {
		ai_recipients = neighboring_rulers
		ai_recipients = peer_vassals
		chance = 0.25
	}
	ai_frequency = 36
	ai_potential = {
		always = yes
	}
	ai_will_do = {
		base = 20
		modifier = {
			add = 20
			scope:recipient = { has_trait = cynical }
		}
		modifier = {
			add = -30
			scope:recipient = { has_trait = zealous }
		}
	}
}
