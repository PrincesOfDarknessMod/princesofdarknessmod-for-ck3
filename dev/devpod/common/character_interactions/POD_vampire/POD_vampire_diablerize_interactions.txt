# Diablerize
# Sparc

diablerize_interaction = {
	category = interaction_category_vampire
	#common_interaction = yes
	#interface_priority = 82
	scheme = diablerize

	ai_frequency = 0

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		scope:actor = { is_vampire_trigger = yes  }
		scope:recipient = { is_vampire_trigger = yes }
	}

	is_valid_showing_failures_only = {
		scope:recipient = { 
			POD_is_not_clone_trigger = yes
			NOT = { has_trait = torpor }
			NOT = { has_character_modifier = control_puppet_modifier }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }	
		}
		POD_cross_umbra_interaction_valid_trigger = yes
		custom_description = {
			object = scope:actor
			text = "I_can_only_plot_one_diablerie_at_time"
			scope:actor = {
				NOT = {
					any_scheme = {
						scheme_type = diablerize
					}
				}
			}
		}
		
		scope:actor = {
			NOT = {
				has_opinion_modifier = {
					target = scope:recipient
					modifier = entrancement_opinion
				}
			}
		}		
		scope:actor = {
			can_start_scheme = {
				type = diablerize
				target_character = scope:recipient
			}
		}
	}

	desc = {
		triggered_desc = {
			trigger = {
				scope:actor = {
					NOT = { is_at_war_with = scope:recipient }
					can_start_scheme = {
						type = diablerize
						target_character = scope:recipient
					}

				}
			}
			desc = scheme_interaction_tt_diablerize_approved
		}

	}

	on_accept = {
		hidden_effect = {
			scope:actor = {
				start_scheme = {
					type = diablerize
					target_character = scope:recipient
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
