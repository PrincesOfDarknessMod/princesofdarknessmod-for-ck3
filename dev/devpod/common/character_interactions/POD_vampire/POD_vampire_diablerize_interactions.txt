# POD Schemes
# Diablerize
# Sparc

diablerize_interaction = {
	category = interaction_category_vampire
	#common_interaction = yes
	#interface_priority = 82
	scheme = diablerize

	ai_frequency = 0

	is_shown = {
		scope:actor = {
			is_vampire_trigger = yes
			is_adult = yes
		}
		scope:recipient = {
			is_vampire_trigger = yes
			is_adult = yes
		}
		scope:actor != scope:recipient
		POD_hostile_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			NOT = { has_character_modifier = control_puppet_modifier }
		}
		custom_description = {
			object = scope:actor
			text = "I_can_only_plot_one_diablerie_at_time"
			scope:actor = {
				NOT = {
					any_scheme = {
						scheme_type = diablerize
					}
				}
			}
		}
		
		scope:actor = {
			NOT = {
				has_opinion_modifier = {
					target = scope:recipient
					modifier = entrancement_opinion
				}
			}
		}
		scope:actor = {
			can_start_scheme = {
				type = diablerize
				target_character = scope:recipient
			}
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	desc = {
		triggered_desc = {
			trigger = {
				scope:actor = {
					NOT = { is_at_war_with = scope:recipient }
					can_start_scheme = {
						type = diablerize
						target_character = scope:recipient
					}

				}
			}
			desc = scheme_interaction_tt_diablerize_approved
		}

	}

	on_accept = {
		hidden_effect = {
			scope:actor = {
				start_scheme = {
					type = diablerize
					target_character = scope:recipient
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
