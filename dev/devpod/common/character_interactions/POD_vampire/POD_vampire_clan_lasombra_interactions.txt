#Used during split for non ashirra
POD_courts_blood_interaction = {
	category = interaction_category_hostile
	#common_interaction = yes
	#interface_priority = 89
	desc = POD_courts_blood_interaction_desc
	icon = icon_liege
	
	#ai_frequency = 0
	
	cooldown = { years = 5 }

	is_shown = {
	#	always = no
		scope:actor = { is_lasombra_trigger = yes }
		scope:recipient = {
			is_lasombra_trigger = yes
		}
		scope:actor = { NOT = { has_religion = religion:ashirra_religion } }
		scope:recipient = { NOT = { has_religion = religion:ashirra_religion } }
		scope:actor != scope:recipient
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:lasombra_victory_objective_decision
			}
		}
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:ashirra_victory_objective_decision
			}
		}
		POD_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		POD_interaction_is_valid_showing_failures_only = yes
	}
	
	on_accept = {
		scope:actor = {
			trigger_event = pod_lasombraevents.2
		}
		custom_tooltip = courts_blood_tooltip
	}

	auto_accept = yes
}
