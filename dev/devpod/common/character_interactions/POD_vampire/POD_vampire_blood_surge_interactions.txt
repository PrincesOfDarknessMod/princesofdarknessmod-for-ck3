# POD Personal Character Interactions
# Blood Surge

POD_blood_surge_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	desc = POD_blood_surge_interaction_desc
	options_heading = POD_blood_surge_interaction_options_heading
	
	send_option = {
		flag = prowess
		localization = POD_blood_surge_prowess
	}
	
	send_option = {
		flag = health
		localization = POD_blood_surge_health
	}
	
	send_option = {
		flag = diplomacy
		localization = POD_blood_surge_diplomacy
	}
	
	send_option = {
		flag = martial
		localization = POD_blood_surge_martial
	}

	send_option = {
		flag = stewardship
		localization = POD_blood_surge_stewardship
	}
	
	send_option = {
		flag = intrigue
		localization = POD_blood_surge_intrigue
	}

	send_option = {
		flag = learning
		localization = POD_blood_surge_learning
	}
	send_options_exclusive = yes
	
	auto_accept = yes
	ai_frequency = 0
	
	cooldown = { months = 36 }
	
	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { POD_is_vampire_trigger = yes }
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
			POD_has_enough_blood_for_discipline = { AMOUNT = 1 }
			NOT = { has_character_modifier = bloodpotency0_modifier }
			POD_is_unable_to_surge_trigger = no
		}
	}

	on_accept = {
		scope:actor = {
			POD_hunger_rouse_check_effect = yes
			
			switch = {
				trigger = yes

				scope:prowess		= { POD_blood_surge_effect = { STAT = prowess } }
				scope:health		= { POD_blood_surge_effect = { STAT = health } }
				scope:diplomacy		= { POD_blood_surge_effect = { STAT = diplomacy } }
				scope:martial		= { POD_blood_surge_effect = { STAT = martial } }
				scope:stewardship	= { POD_blood_surge_effect = { STAT = stewardship } }
				scope:intrigue		= { POD_blood_surge_effect = { STAT = intrigue } }
				scope:learning		= { POD_blood_surge_effect = { STAT = learning } }
			}
		}
	}

	ai_will_do = {
		base = 0
	}
}
