pod_angra_mainyu_conversion_interaction = {
	category = interaction_category_friendly
	desc = pod_angra_mainyu_conversion_interaction_desc
	icon = POD_vampire_interaction
	
	common_interaction = yes

	is_shown = {
		scope:actor = {
			has_personal_tenet_flag = unlocks_tempt_conversion_scheme
			POD_is_vampire_trigger = yes
			is_playable_character = yes
		}
		scope:recipient = {
			POD_is_vampire_trigger = yes
			is_adult = yes
			faith != faith:roadofagramainyu
		}
		scope:actor != scope:recipient
		POD_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			trigger_if = {
				limit = { is_landless_adventurer = no }
				is_travelling = no
			}
			custom_tooltip = {
				text = pod_angra_mainyu_conversion_already_scheming_tt
				NOT = {
					any_scheme = {
						scheme_type = pod_angra_mainyu_conversion
					}
				}
			}
		}
		scope:recipient = {
			is_imprisoned = no
			trigger_if = {
				limit = {
					scope:actor = { is_landless_adventurer = no }
				}
				is_travelling = no
			}
			custom_tooltip = {
				text = pod_setite_serpent_embrace_interaction_blocked_tt
				NOT = { has_variable = block_pod_setite_serpent_embrace }
			}
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:actor = {
			start_scheme = {
				type = pod_angra_mainyu_conversion
				target_character = scope:recipient
			}
		}
	}

	auto_accept = yes

	ai_targets = {
		ai_recipients = neighboring_rulers
		ai_recipients = peer_vassals
		chance = 0.25
	}
	ai_frequency_by_tier = {
		barony = 0
		county = 72
		duchy = 36
		kingdom = 36
		empire = 36
		hegemony = 36
	}
	ai_potential = {
		always = yes
	}
	ai_will_do = {
		base = 20
		modifier = {
			add = 20
			scope:recipient = { has_trait = cynical }
		}
		modifier = {
			add = -30
			scope:recipient = { has_trait = zealous }
		}
	}
}
