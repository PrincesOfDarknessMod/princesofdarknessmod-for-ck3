# POD Personal Character Interactions
# veil masquerade

POD_cover_masquerade_interaction = {
	category = interaction_category_vampire
	#interface_priority = 92
	desc = POD_cover_masquerade_interaction_desc
	options_heading = POD_cover_masquerade_interaction_options_heading

	use_diplomatic_range = no

	cooldown = { months = 12 }
	
	is_shown = {
		scope:actor = {
			OR = {
				this = scope:recipient
				any_character_active_contract = {
					task_contract_type = POD_ghoul_mission_cover_up_the_masquerade
					task_contract_employer = scope:recipient
					POD_ghoul_mission_can_be_finished = yes
				}
			}
		}
		scope:recipient = { POD_splat_has_masquerade_trigger = yes }
	}

	is_valid_showing_failures_only = {
		custom_tooltip = {
			text = "no_masquerade"
			scope:recipient = {
				NAND = {
					faith = { has_doctrine_parameter = vampirethemasquerade_none }
					masquerade_exposure_level < 5
					NOT = { has_trait = paranoid }
				}
			}
		}
		scope:recipient = { POD_is_forever_infamous_trigger = no }
		custom_tooltip = {
			text = POD_cover_masquerade_interaction_cap
			scope:recipient = { masquerade_exposure_level > minimum_masquerade_exposure }
		}
		POD_self_targeted_interaction_is_valid_showing_failures_only = yes
	}

	send_option = {
		is_shown = {
			scope:actor = {
				OR = {
					has_trait = dominateadvanced
					has_perk = the_forgetful_mind_perk
				}
			}
		}
		is_valid = {
			scope:actor = { POD_has_enough_blood_for_discipline = { AMOUNT = 1 } }
		}
		flag = dominate
		localization = POD_cover_masquerade_dominate
		#current_description = POD_cover_masquerade_dominate
	}
	
	send_option = {
		is_shown = {
			always = yes
		}
		is_valid = {
			scope:actor = { gold >= major_gold_value }
		}
		flag = bribe
		localization = POD_cover_masquerade_bribe
		#current_description = POD_cover_masquerade_bribe
	}
	
	send_option = {
		is_shown = {
			always = yes
		}
		flag = murderspree
		localization = POD_cover_masquerade_murderspree
		#current_description = POD_cover_masquerade_muderspree
	}
	
	send_option = {
		is_shown = {
			scope:actor = scope:recipient
		}
		flag = layinglow
		localization = POD_cover_masquerade_layinglow
		#current_description = POD_cover_masquerade_layinglow
	}

	send_option = {
		is_shown = {
			scope:actor = {
				OR = {
					has_trait = presenceadvanced
					has_perk = majesty_perk
				}
			}
		}
		is_valid = {
			scope:actor = { POD_has_enough_blood_for_discipline = { AMOUNT = 1 } }
		}
		flag = presence
		localization = POD_cover_masquerade_presence
		#current_description = POD_cover_masquerade_presence
	}
	
	send_option = {
		is_shown = {
			scope:actor = {
				has_trait = bloodsorceryadvanced
			}
		}
		is_valid = {
			scope:actor = { POD_has_enough_blood_for_discipline = { AMOUNT = 1 } }
		}
		flag = bloodsorcery
		localization = POD_cover_masquerade_bloodsorcery
		#current_description = POD_cover_masquerade_bloodsorcery
	}
	
	on_accept = {
		POD_cover_up_masquerade_effect = {
			USER = scope:actor
			BENEFACTOR = scope:recipient
		}
		scope:actor = {
			random_character_active_contract = {
				task_contract_type = POD_ghoul_mission_cover_up_the_masquerade
				limit = {
					task_contract_employer = scope:recipient
					POD_ghoul_mission_can_be_finished = yes
				}
				complete_task_contract = success_standard
			}
		}
	}
	
	auto_accept = yes

	ai_will_do = {
		base = 0
	}
}

POD_enforce_masquerade_interaction = {
	category = interaction_category_vampire
	#common_interaction = yes
	#interface_priority = 93
	desc = POD_enforce_masquerade_interaction_desc
	
	ai_frequency = 360

	is_shown = {
		scope:actor != scope:recipient
		scope:recipient = { liege ?= scope:actor }
		scope:actor = {
			is_vampire_trigger = yes
			is_adult = yes
			POD_is_hungerless_trigger = no
		}
		scope:recipient = {
			POD_is_vampire_trigger = yes
			NOT = { has_relation_herd_owner = scope:actor }
			is_imprisoned = no
		}
		scope:actor = {
			dynasty ?= { has_dynasty_perk = ventrue_legacy_3 }
		}
		POD_interaction_is_shown = yes
	}
	
	is_valid_showing_failures_only = {
		scope:recipient = {
			is_adult = yes
			is_imprisoned = no
		}
		custom_description = {
			object = scope:recipient
			text = "has_high_masquerade_exposure"
			scope:recipient = { masquerade_exposure_level > 0 }
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	cooldown = { years = 15 }

	on_accept = {
		scope:recipient = {
			save_scope_as = target
		}
		scope:actor = {
			save_scope_as = enforcer
		}
		scope:actor = {
			trigger_event = {
				id = POD_masquerade_enforcement.001
				days = 2
			}
		}
	}
	
	ai_targets = {
		ai_recipients = peer_vassals
		ai_recipients = vassals
	}
	
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 20
			scope:recipient = {
				has_character_modifier = masquerade2_modifier
			}
		}
		
		modifier = {
			add = 40
			scope:recipient = {
				has_character_modifier = masquerade3_modifier
			}
		}
		
		modifier = {
			add = 60
			scope:recipient = {
				has_character_modifier = masquerade4_modifier
			}
		}
		
		modifier = {
			add = 80
			scope:recipient = {
				has_character_modifier = masquerade5_modifier
			}
		}
		
		modifier = {
			factor = 0
			scope:recipient = { is_ai = no }
		}
	}
}
