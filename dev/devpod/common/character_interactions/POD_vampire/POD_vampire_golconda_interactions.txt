				scope:choleric_perk    = yes

			scope:perks_5  = no
			scope:perks_10 = no
		}
		is_shown = {
			scope:actor = { piety >= 20000 }
		}
	}
	
	send_options_exclusive = no

	auto_accept = yes
	ai_frequency = 0
	use_diplomatic_range = no
	
	is_shown = {
		scope:actor = scope:recipient
		scope:actor = {
			is_ai = no
			is_adult = yes
			has_trait_with_flag = perks_for_piety_flag
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = { piety >= 1000 }
	}

	on_accept = {
		scope:actor = {
			if = {
				limit = {
					OR = {
						scope:animal_perk      = yes
						scope:choleric_perk    = yes
						scope:melancholic_perk = yes
						scope:phlegmatic_perk  = yes
						scope:sanguine_perk    = yes
					}
				}
				switch = {
					trigger = yes
					scope:perks_5  = { save_scope_value_as = { name = perks_num value = 5  } }
					scope:perks_10 = { save_scope_value_as = { name = perks_num value = 10 } }
					scope:perks_20 = { save_scope_value_as = { name = perks_num value = 20 } }
					fallback       = { save_scope_value_as = { name = perks_num value = 1  } }
				}
				add_piety = {
					value = -1000
					multiply = scope:perks_num
				}
				switch = {
					trigger = yes
					scope:animal_perk      = { add_animal_lifestyle_perk_points     = scope:perks_num }
					scope:choleric_perk    = { add_choleric_lifestyle_perk_points   = scope:perks_num }
					scope:melancholic_perk = { add_melancholy_lifestyle_perk_points = scope:perks_num }
					scope:phlegmatic_perk  = { add_phlegmatic_lifestyle_perk_points = scope:perks_num }
					scope:sanguine_perk    = { add_sanguine_lifestyle_perk_points   = scope:perks_num }
					fallback               = {}
				}
			}
		}
	}
}

POD_golconda_mentoring_interaction = {
	category = interaction_category_vampire
	icon = POD_golconda_interaction
	desc = POD_golconda_mentoring_interaction_desc

	cooldown = { years = 1 }
	cost = { piety = 500 }

	ai_frequency = 0
	
	send_option = {
		flag = only_once
		localization = POD_golconda_mentoring_interaction_only_once
	}
	
	send_option = {
		flag = repeating
		localization = POD_golconda_mentoring_interaction_repeating
	}

	send_options_exclusive = yes

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			is_ai = no
			is_adult = yes
			OR = {
				has_trait = golconda
				has_trait = golconda_seeker
			}
		}
		scope:recipient = {
			POD_is_vampire_trigger = yes
			NOT = { has_trait = golconda }
		}
		POD_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			is_adult = yes
		}
		custom_tooltip = {
			text = POD_golconda_mentoring_interaction_requirements_mentee
			scope:actor = {
				NOT = { has_variable = golconda_mentee }
			}
		}
		custom_tooltip = {
			text = POD_golconda_mentoring_interaction_requirements
			scope:actor = {
				trigger_if = {
					limit = { has_trait = golconda_seeker }
					has_trait_xp = {
						trait = golconda_seeker
						value >= 20
					}
				}
			}
		}
		custom_tooltip = {
			text = POD_golconda_mentoring_interaction_requirements_2
			scope:recipient = {
				OR = {
					NOT = { has_trait = golconda_seeker }
					suspire_success_chance < scope:actor.suspire_success_chance
				}
			}
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:actor = {
			POD_golconda_mentoring_effect = {
				STUDENT = scope:recipient
			}
			if = {
				limit = { scope:repeating = yes }
				custom_tooltip = {
					text = POD_golconda_mentoring_interaction_repeating_effect
					set_variable = {
						name = golconda_mentee
						value = scope:recipient
					}
					scope:recipient = {
						save_scope_as = golconda_mentee
						set_variable = {
							name = golconda_mentor
							value = scope:actor
						}
					}
					trigger_event = {
						id = POD_golconda_journey.2000
						years = 1
					}
				}
			}
		}
	}
	
	ai_accept = {
		base = -50
		opinion_modifier = { # Opinion Factor
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 1.0
			desc = AI_OPINION_REASON
		}
	}
}

POD_guide_through_suspire_interaction = {
	category = interaction_category_vampire
	icon = POD_golconda_interaction
	desc = POD_guide_through_suspire_interaction_desc

	cost = { piety = 1000 }

	ai_frequency = 0
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			is_ai = no
			is_adult = yes
			has_trait = golconda
		}
		scope:recipient = {
			POD_is_vampire_trigger = yes
			NOT = { has_trait = golconda }
		}
		POD_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:recipient = { has_trait = golconda_seeker }
		scope:recipient = {
			is_adult = yes
			is_available_allow_travelling = yes
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:actor = {
			send_interface_toast = {
				title = POD_guide_through_suspire_interaction_toast
				left_icon = scope:actor
				right_icon = scope:recipient
				scope:recipient = {
					random_list = {
						0 = {
							modifier = { add = { value = suspire_success_chance } }
							POD_achieve_golconda_effect = yes
						}
						100 = {
							modifier = { add = { value = 0 subtract = suspire_success_chance } }
							POD_become_wight_effect = yes
						}
					}
				}
			}
			hidden_effect = {
				if = {
					limit = { var:golconda_mentee ?= scope:recipient }
					remove_variable = golconda_mentee
				}
			}
		}
	}
	
	ai_accept = {
		base = -150
		opinion_modifier = { # Opinion Factor
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 1.0
			desc = AI_OPINION_REASON
		}
		modifier = {
			add = scope:recipient.suspire_success_chance
			desc = SUSPIRE_SUCCESS_REASON
		}
	}
}
