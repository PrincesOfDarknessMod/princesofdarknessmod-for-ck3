	}

		}
		is_valid = {
			always = yes
		}

		}
		is_valid = {

		}

		}
		is_valid = {
			always = yes
		}

		is_valid = {

	}

		base = 5
		opinion_modifier = { # More likely to accept if Recipient likes the Actor
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.1
			desc = AI_OPINION_REASON
		}

			add = 5
			scope:recipient = {
				has_dread_level_towards = {
					target = scope:actor
					level = 1
				}

				}

					scope:stewardship = yes
				}

					}

					type = councillor_steward
					target = scope:secondary_recipient
				}
				scope:secondary_recipient = { add_character_flag = steward_job_flag }
			}
			if = {
				limit = {
					scope:martial = yes
				}
				if = {
					limit = {
						exists = cp:councillor_marshal
					}
					fire_councillor = cp:councillor_marshal
				}
				assign_councillor_type = {
					type = councillor_marshal
					target = scope:secondary_recipient
				}
				scope:secondary_recipient = { add_character_flag = marshal_job_flag }
			}
			if = {
				limit = {
					scope:intrigue = yes
				}
				if = {
					limit = {
						exists = cp:councillor_spymaster
					}
					fire_councillor = cp:councillor_spymaster
				}
				assign_councillor_type = {
					type = councillor_spymaster
					target = scope:secondary_recipient
				}
				scope:secondary_recipient = { add_character_flag = spymaster_job_flag }
			}
			if = {
				limit = {
					scope:learning = yes
				}
				if = {

						exists = cp:councillor_court_chaplain
					}
					fire_councillor = cp:councillor_court_chaplain
				}
				assign_councillor_type = {
					type = councillor_court_chaplain
					target = scope:secondary_recipient
				}
				scope:secondary_recipient = { add_character_flag = chaplain_job_flag }
			}
	
		}
		custom_tooltip = POD_cappadocian_send_advisor_interaction_ct
		scope:secondary_recipient = { 
			trigger_event = { id = POD_cappadocian_power.1000  years = 1 } 
			set_variable = { name = block_fire_councillor value = scope:recipient }
		}
	}
	
	
	auto_accept = {
		OR = {
			AND = {
				scope:diplomacy = yes
				NOT = { scope:recipient = { exists = cp:councillor_chancellor } }
			}
			
			AND = {
				scope:stewardship = yes
				NOT = { scope:recipient = { exists = cp:councillor_steward } }
			}
			
			AND = {
				scope:martial = yes
				NOT = { scope:recipient = { exists = cp:councillor_marshal } }
			}
			
			AND = {
				scope:intrigue = yes
				NOT = { scope:recipient = { exists = cp:councillor_spymaster } }
			}
			
			AND = {
				scope:learning = yes
				NOT = { scope:recipient = { exists = cp:councillor_court_chaplain } }
			}
		
		}
	}

		scope:actor = { has_trait = ventrue }

		NOT = {

					scope:recipient = this
				}
			
			}
		}

		scope:actor.dynasty = {
		    has_dynasty_perk = ventrue_legacy_1
		}
		
	}

			}

		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}
	
	auto_accept = yes
	on_accept = {
		scope:actor = { add_hook = { target = scope:recipient type = minorboon_hook } }
		if = {
			limit = { 
				scope:recipient = { has_hook = scope:actor }

		}

				arrogant = major_stress_impact_gain

	}

		scope:actor = scope:recipient

			is_target_in_variable_list = {
				name = unavailable_unique_decisions

				has_artifact_modifier = sagron_codex_shard_modifier
			}

		}
	}

		trigger_if = {

				scope:actor = {

							trigger_if = { limit = { has_variable = understood_by } 
								NOT = { var:understood_by = scope:actor } 
							}
						}
					
				}
			}
			trigger_if = {
				limit = { 
					any_living_character = {
						any_character_artifact = {
							has_artifact_modifier = sagron_codex_shard_modifier
						}
						NOT = { THIS = scope:actor }
						NOT = { THIS = character:POD_wraith_01 }
						save_temporary_scope_as = example
					}
				}
				custom_tooltip = {
					text = POD_understand_sargon_codex_shard_interaction_custom
				}
			}
		}
