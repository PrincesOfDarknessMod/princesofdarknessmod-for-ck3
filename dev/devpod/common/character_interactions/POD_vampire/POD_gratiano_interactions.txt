######################################
# RECRUIT A LASOMBRA
######################################

POD_gratiano_recruit_lasombra_interaction = {
	category = interaction_category_vampire
	icon = icon_dynasty
	desc = POD_gratiano_recruit_lasombra_interaction_desc

	ai_frequency = 0

	cooldown = { months = 12 }

	is_shown = {
		scope:actor = { POD_gratiano_lever_actor_trigger = yes }
		scope:actor != scope:recipient
		scope:recipient = {
			is_lasombra_trigger = yes
			in_diplomatic_range = scope:actor
			POD_gratiano_recruit_excluded_trigger = no
		}
		NOT = {
			scope:actor = {
				POD_gratiano_pool_member_trigger = { CHAR = scope:recipient }
			}
		}
		NOT = { scope:actor = { has_relation_coterie_member = scope:recipient } }
		POD_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:recipient = { NOT = { has_trait = torpor } }
		POD_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:actor = {
			custom_tooltip = POD_gratiano_recruit_accept_tt
			hidden_effect = {
				POD_gratiano_add_to_pool_effect = { TARGET = scope:recipient }
				# Evaluate the recruit, not Gratiano, before passing the gain to the conspiracy effect.
				scope:recipient = {
					save_scope_value_as = {
						name = pod_gratiano_recruit_gain
						value = POD_gratiano_recruit_conspiracy_gain_value
					}
				}
				POD_gratiano_change_conspiracy_effect = {
					DELTA = scope:pod_gratiano_recruit_gain
				}
				send_interface_toast = {
					type = event_generic_good
					title = POD_gratiano_recruit_joined_toast
					left_icon = scope:actor
					right_icon = scope:recipient
				}
			}
		}
	}

	on_decline = {
		scope:actor = {
			custom_tooltip = POD_gratiano_recruit_decline_tt
			hidden_effect = {
				random = {
					chance = 35
					modifier = {
						add = 25
						scope:recipient = { has_trait = zealous }
					}
					scope:recipient = { save_scope_as = gratiano_informant }
					POD_gratiano_schedule_report_effect = { SEVERITY = 2 }
				}
				send_interface_message = {
					type = event_generic_bad
					title = POD_gratiano_recruit_refused_message
					left_icon = scope:recipient
					custom_tooltip = POD_gratiano_recruit_refused_message_tt
				}
			}
		}
	}

	ai_min_reply_days = 5
	ai_max_reply_days = 12

	ai_accept = {
		base = -15
		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.25
			desc = AI_OPINION_REASON
		}
		modifier = {
			add = 20
			scope:recipient = { has_trait = ambitious }
			desc = POD_gratiano_recruit_accept_ambitious
		}
		modifier = {
			add = 15
			scope:recipient = { has_trait = deceitful }
			desc = POD_gratiano_recruit_accept_deceitful
		}
		modifier = {
			add = 25
			scope:recipient = { has_trait = disloyal }
			desc = POD_gratiano_recruit_accept_disloyal
		}
		modifier = {
			add = -20
			scope:recipient = { has_trait = content }
			desc = POD_gratiano_recruit_accept_content
		}
		modifier = {
			add = -15
			scope:recipient = { has_trait = zealous }
			NOT = { scope:actor = { POD_gratiano_academy_creed_value = 2 } }
			desc = POD_gratiano_recruit_accept_zealous
		}
		modifier = {
			add = 10
			scope:actor = {
				POD_gratiano_academy_creed_value = 2
				POD_gratiano_academy_tier_value >= 1
			}
			desc = POD_gratiano_recruit_accept_eldest
		}
		modifier = {
			add = 5
			scope:recipient = {
				has_dread_level_towards = {
					target = scope:actor
					level = 1
				}
			}
			desc = INTIMIDATED_REASON
		}
	}
}

######################################
# LEVERAGE A HOOK
######################################

POD_gratiano_leverage_hook_interaction = {
	category = interaction_category_vampire
	icon = icon_hostile
	desc = POD_gratiano_leverage_hook_interaction_desc

	ai_frequency = 0

	cooldown = { months = 6 }

	should_use_extra_icon = {
		scope:actor = { has_usable_hook = scope:recipient }
	}
	extra_icon = "gfx/interface/icons/character_interactions/hook_icon.dds"

	is_shown = {
		scope:actor = { POD_gratiano_lever_actor_trigger = yes }
		scope:actor != scope:recipient
		scope:recipient = {
			POD_is_vampire_trigger = yes
			POD_gratiano_recruit_excluded_trigger = no
		}
		scope:actor = { has_usable_hook = scope:recipient }
		NOT = {
			scope:actor = {
				POD_gratiano_pool_member_trigger = { CHAR = scope:recipient }
			}
		}
		NOT = { scope:actor = { has_relation_coterie_member = scope:recipient } }
		POD_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:recipient = { NOT = { has_trait = torpor } }
		POD_interaction_is_valid_showing_failures_only = yes
	}

	auto_accept = yes

	on_accept = {
		scope:actor = {
			custom_tooltip = POD_gratiano_leverage_hook_tt
			hidden_effect = {
				if = {
					limit = { scope:actor = { has_strong_hook = scope:recipient } }
					POD_gratiano_change_conspiracy_effect = {
						DELTA = POD_gratiano_hook_strong_gain_value
					}
				}
				else = {
					POD_gratiano_change_conspiracy_effect = {
						DELTA = POD_gratiano_hook_weak_gain_value
					}
				}
				POD_gratiano_conscript_to_pool_effect = { TARGET = scope:recipient }
				use_hook = scope:recipient
				trigger_event = {
					id = POD_gratiano.0041
					days = { 1 3 }
				}
			}
		}
	}
}

######################################
# SELL OUT A CONSPIRATOR
######################################

POD_gratiano_sell_out_interaction = {
	category = interaction_category_hostile
	icon = icon_hostile
	desc = POD_gratiano_sell_out_interaction_desc

	ai_frequency = 0

	cooldown = { years = 2 }

	is_shown = {
		scope:actor = { POD_gratiano_lever_actor_trigger = yes }
		scope:actor != scope:recipient
		OR = {
			scope:actor = {
				POD_gratiano_pool_member_trigger = { CHAR = scope:recipient }
			}
			scope:actor = { has_relation_coterie_member = scope:recipient }
		}
		POD_hostile_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		POD_interaction_is_valid_showing_failures_only = yes
	}

	# The victim is not consulted; the founder's court is.
	auto_accept = yes

	on_accept = {
		scope:actor = {
			custom_tooltip = POD_gratiano_sell_out_tt
			hidden_effect = {
				scope:recipient = { save_scope_as = gratiano_sellout_target }
				send_interface_message = {
					type = event_generic_bad
					title = POD_gratiano_sell_out_sent_message
					left_icon = scope:actor
					right_icon = scope:recipient
					custom_tooltip = POD_gratiano_sell_out_sent_message_tt
				}
				trigger_event = {
					id = POD_gratiano.0042
					days = { 5 10 }
				}
			}
		}
	}
}

	desc = POD_gratiano_frame_elder_interaction_desc

	scheme = POD_gratiano_frame_scheme

	ai_frequency = 0

	is_shown = {
		scope:actor = {
			this = character:505144
			is_ai = no
			has_character_flag = POD_gratiano_assamite_pact
		}
		scope:recipient = { POD_gratiano_frame_target_trigger = yes }
		scope:actor != scope:recipient
		NOT = { has_global_variable = variable_POD_lasombra_dead }
		POD_hostile_interaction_is_shown = yes

	is_valid_showing_failures_only = {
		scope:actor = {
			is_incapable = no
			NOT = { POD_vampire_is_defeated = { CHARACTER = scope:recipient } }
		}
		scope:recipient = { in_diplomatic_range = scope:actor }
		scope:actor = {
			can_start_scheme = {
				type = POD_gratiano_frame_scheme
				target_character = scope:recipient
			}
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	auto_accept = yes

	on_accept = {
		hidden_effect = {
			scope:actor = {
				start_scheme = {
					type = POD_gratiano_frame_scheme
					target_character = scope:recipient
				}
			}
		}
	}
}
