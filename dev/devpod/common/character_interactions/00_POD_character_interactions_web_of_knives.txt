POD_chase_bounty_interaction = {
	category = interaction_category_vampire
	#interface_priority= 91
	desc = POD_chase_bounty_interaction_desc
	scheme = chase_bounty
	ignores_pending_interaction_block = yes
	icon = icon_hostile

	cooldown = { months = 1 }
	
	is_shown = {
		scope:actor = {
			web_of_knives_member_trigger = yes
			can_start_scheme = {
				type = chase_bounty
				target = scope:recipient
			}	
		}
		scope:recipient = {
			NOT = {
				any_in_list = {
					variable = contractor
					this = scope:actor
				}
			}
		}
		
		
	}
	
	is_valid = {
		title:k_webofknives = { is_title_created = yes }
	}
	
	is_valid_showing_failures_only = {
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
			POD_is_not_trapped_in_umbra_trigger = { ACTOR = scope:actor }
		}
	}
	
	auto_accept = yes
	
	on_accept = {
		scope:actor = {
			start_scheme = {
				type = chase_bounty
				target = scope:recipient
			}
		}
		
	}
	
		
	ai_targets = {
		ai_recipients = realm_characters
		ai_recipients = neighboring_rulers
	}
	
	ai_frequency = 360
	
	
	
	ai_will_do = {
		base = -50
		
		
		modifier = {
			add = { value = prowess subtract = scope:recipient.prowess multiply = 2 }
			always = yes
		}
		
		opinion_modifier = { # Opinion Factor
			who = scope:actor
			opinion_target = scope:recipient
			multiplier = -0.4
		}
		
		modifier = { # Don't gang up on people 
			factor = 0
			scope:recipient = { 
				any_targeting_scheme = {
					count > 2
				}
			}
		}
		
		
		modifier = {
			add = {
				value = scope:actor.ai_greed
				divide = 4
			}
			scope:recipient = {
				has_variable = web_gold_reward
				var:web_gold_reward = 50
			}
			
		}
		
		modifier = {
			add = scope:actor.ai_greed
			scope:recipient = {
				has_variable = web_gold_reward
				var:web_gold_reward = 100
			}	
		}
		
		modifier = {
			add = {
				value = scope:actor.ai_greed
				multiply = 1.5
			}

			scope:recipient = {
				has_variable = web_gold_reward
				var:web_gold_reward = 500
			}	
		}
		
		modifier = {
			add = {
				value = scope:actor.ai_greed
				multiply = 2
			}
			scope:recipient = {
				has_variable = web_gold_reward
				var:web_gold_reward = 1000
			}	
		}
		
		modifier = {
			add = {
				value = scope:actor.ai_greed
				multiply = 3
			}
			scope:recipient = {
				has_variable = web_gold_reward
				var:web_gold_reward = 5000
			}
		}
		
		modifier = {
			factor = 0.5
			scope:actor = {
				has_dread_level_towards = {
					target = scope:recipient
					level >= 1
				}
			}
		}
		
		modifier = {
			add = -100
			scope:actor = {
				has_relation_friend = scope:recipient
			}
		}
		modifier = {
			add = -25
			scope:actor = {
				has_relation_lover = scope:recipient
			}
		}
		modifier = {
			add = -1000
			scope:actor = {
				OR = {
					has_relation_best_friend = scope:recipient
					has_relation_soulmate = scope:recipient
				}
			}
		}
		modifier = {
			factor = { value = 1 divide = scope:recipient.highest_held_title_tier }
			scope:recipient = { is_landed = yes }
		}
		
		modifier = {
			add = scope:actor.ai_vengefulness
			scope:actor.ai_vengefulness > 0
			scope:recipient = {
				OR = {
					has_relation_rival = scope:actor
					has_relation_nemesis = scope:actor
					exposed_cheating_on_spouse_trigger = { SPOUSE = scope:actor }
				}
			}
		
		}
		
		modifier = {
			factor = 0 
			scope:recipient = { has_character_flag = won_bounty_duel  }
		}
		
		modifier = {
			factor = 0
			title:k_webofknives.holder = {
				scope:recipient = {
					OR = {
						is_close_family_of = prev
						is_consort_of = prev
						is_courtier_of = prev
					}
				}
			}
		}
		
		modifier = {
			add = -25
			any_player = {
				scope:recipient = {
					OR = {
						is_close_family_of = prev
						is_consort_of = prev
						is_courtier_of = prev
					}
				}
			}
		}
	}
}

	#icon = "gfx/interface/icons/character_interactions/blood_surge_interaction.dds"
	desc = POD_create_bounty_interaction_desc
	icon = icon_hostile
	
	is_shown = {
		NOT = { 
			scope:actor = scope:recipient
		}
		scope:actor = { has_trait = vampire }
	}
	is_valid = {
		scope:actor = {
			gold > 50
		}
		custom_description = {
			text = fellow_web_of_knives_member
			scope:recipient = {
				NOT = { web_of_knives_member_trigger = yes }
			}
		}
		scope:recipient = {	is_ai = yes }		
		title:k_webofknives = { is_title_created = yes }
	}
	is_valid_showing_failures_only = {
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
			POD_is_not_trapped_in_umbra_trigger = { ACTOR = scope:actor }
		}
	}
	send_option = {
		starts_enabled = { always = yes	} # Or could accept without choose option
		flag = reward_gold_50
		localization = reward_gold_50_loc	
	}
	
	send_option = {
		flag = reward_gold_100
		is_shown = {
			scope:actor = { gold > 99 }
		}
		localization = reward_gold_100_loc	
	}
	
	send_option = {
		flag = reward_gold_250
		is_shown = {
			scope:actor = { gold > 250 }
		}
		localization = reward_gold_250_loc	
	}
	
	send_option = {

		}

	}

		}

	}
	
	send_option = {
		flag = reward_gold_5000

			scope:actor = { gold > 4999 }
		}
		localization = reward_gold_5000_loc	
	}
	
	auto_accept = yes
	on_accept = {
		scope:actor = {
			add_to_global_variable_list = {
				name = global_web_of_knives_targets
				target = scope:recipient
			}
		}
		scope:actor = {
			add_to_variable_list = {
				name = web_of_knives_targets
				target = scope:recipient
			}
			
			add_to_variable_list = {
				name = web_of_knives_contracts
				target = scope:recipient
			}
		}
		every_player = {
			add_to_variable_list = {
				name = web_of_knives_targets
				target = scope:recipient
			}
		}
		scope:recipient = {
			every_in_list = {
				variable = contractor
				scope:recipient = { remove_list_variable = { name = contractor target = prev } }
			}
			add_to_variable_list = { name = contractor target = scope:actor }
		
		}
		custom_tooltip = POD_create_bounty_interaction_ct
		if = {
			limit = {
				scope:reward_gold_50 = yes
			}
			scope:recipient = {
				set_variable = { name = web_gold_reward value = 50 }
			}
		}
		
		if = {
			limit = {
				scope:reward_gold_100 = yes
			}
			scope:recipient = {
				set_variable = { name = web_gold_reward value = 100 }
			}
		}
		
		if = {
			limit = {
				scope:reward_gold_250 = yes
			}
			scope:recipient = {
				set_variable = { name = web_gold_reward value = 250 }
			}
		}
		
		if = {
			limit = {
				scope:reward_gold_500 = yes
			}
			scope:recipient = {
				set_variable = { name = web_gold_reward value = 500 }
			}
		}
		
		if = {
			limit = {
				scope:reward_gold_1000 = yes
			}
			scope:recipient = {
				set_variable = { name = web_gold_reward value = 1000 }
			}
		}
		
		if = {
			limit = {
				scope:reward_gold_5000 = yes
			}
			scope:recipient = {
				set_variable = { name = web_gold_reward value = 5000 }
			}
		}
		scope:recipient = { 
			trigger_event = { id = POD_web_of_knives.1000 years = 20 }
			trigger_event = { id = POD_web_of_knives.10 years = 50 }
		}
	}
	
	ai_targets = {
