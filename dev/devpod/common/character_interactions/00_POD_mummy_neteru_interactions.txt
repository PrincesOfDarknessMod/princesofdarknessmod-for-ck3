POD_set_neteru_interaction = {
	category = interaction_category_mummy
	common_interaction = yes
	interface_priority = 100
	desc = POD_set_neteru_interaction_desc
	icon = POD_hekau_interaction

	ai_min_reply_days = 4
	ai_max_reply_days = 9
	
	cost = { prestige = minor_prestige_value }
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			is_landed = yes
			is_amenti_trigger = yes
		}
		scope:recipient = { 
			NOT = { has_relation_worshipped_reborn = scope:actor }
			NOT = { has_trait = mummy }
		}
		POD_is_in_same_plane_trigger = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			NOT =  { is_at_war_with = scope:recipient }
		}
		scope:recipient = {
			custom_description = {
				object = scope:actor
				text = is_courtier_or_vassal_of
				OR = {
					is_courtier_of = scope:actor
					is_vassal_of = scope:actor
				}
			}
			is_busy_in_events_localised = yes
			NOT = { has_strong_hook = scope:actor }
		}
		trigger_if = {
			limit = { scope:actor = { faith = { has_doctrine  = special_doctrine_against_to_apophis_hostility } } }
			NOT = { scope:recipient = { faith = { has_doctrine  = special_doctrine_allied_to_apophis_hostility } } }
		}
		trigger_if = {
			limit = { scope:actor = { faith = { has_doctrine  = special_doctrine_allied_to_apophis_hostility } } }
			NOT = { scope:recipient = { faith = { has_doctrine  = special_doctrine_against_to_apophis_hostility } } }
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			NOT = { has_trait = statue }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
			POD_is_not_clone_trigger = yes		
		}
	}

	#Use hook
	send_option = {
		is_valid = {
			scope:actor = {
				has_usable_hook = scope:recipient
			}
		}
		flag = hook
		localization = SCHEME_HOOK
	}
	should_use_extra_icon = {
		scope:actor = { has_usable_hook = scope:recipient }
	}
	extra_icon = "gfx/interface/icons/character_interactions/hook_icon.dds"

	send_options_exclusive = no

	on_accept = {
		scope:recipient = {
			set_relation_worshipped_reborn = scope:actor
			if = {
				limit = { 
					faith = { 
						NOR = { 
							has_doctrine = doctrine_undying_accepted
							has_doctrine = doctrine_undying_ignorant
						} 
					} 
				}
				neteru_faith_initiation_effect = { LIEGE = scope:actor }
			}
		}
	}
	
	ai_targets = {
		ai_recipients = vassals
	}
	
	ai_frequency = 72
	
	ai_potential = {
		primary_title.tier > tier_county
		prestige > minor_prestige_value
	}
	
	ai_accept = {
		base = 0 # Try to make it 0 for most interactions

		modifier = {
			scope:hook = yes
			add = 50
			desc = SCHEME_WEAK_HOOK_USED
		}
		
		opinion_modifier = { # Opinion Factor
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.5
			desc = AI_SIMPLE_OPINION_REASON
		}
		
		modifier = {
			add = {
				value = ai_zeal
				divide = 2
				multiply = -1
			}
			ai_zeal > 0
			desc = TAKE_THE_VOWS_ZEALOUS
		}

		modifier = {
			add = -50
			has_relation_rival = scope:recipient
		}
		modifier = {
			add = -80
			has_relation_nemesis = scope:recipient
		}
		
		modifier = {
			add = 35
			scope:actor = { is_ai = yes }
		}
	}
	
	ai_will_do = {
		base = 100
		
		modifier = { # The AI will only use a Hook if they couldn't otherwise do this
			scope:hook = yes
			add = -1
		}
	}
}

POD_remove_neteru_interaction = {
	category = interaction_category_mummy
	common_interaction = yes
	interface_priority = 100
	desc = POD_remove_neteru_interaction_desc
	icon = POD_hekau_interaction
	
	cost = { prestige = major_prestige_value }
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_trait = mummy }
		scope:recipient = { has_relation_worshipped_reborn = scope:actor }
	}

	on_accept = {
		scope:recipient = {
			remove_relation_worshipped_reborn = scope:actor
			if = {
				limit = {
					exists = house
					house = scope:actor.house
					can_execute_decision = create_cadet_branch_decision
				}
				create_cadet_branch = yes
			}
			if = {
				limit = {
					has_trait = mortal
					has_game_rule = world_of_monsters
					can_hold_title_as_mortal_trigger = no
				}
				depose = yes
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_spell_of_life_interaction = {
	category = interaction_category_mummy
	common_interaction = yes
	interface_priority = 90
	desc = POD_spell_of_life_interaction_desc
	icon = POD_hekau_interaction
	
	cooldown = { years = 10 }
	cost = { piety = 10000 }
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			OR = {
				this = character:792593
				AND = {
				this = character:792600
				NOT = { has_character_flag = cabirus_giveup_flag }
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = { 
			has_relation_worshipped_reborn = scope:actor
			is_available_adult = yes
		}
		custom_description = {
			object = scope:recipient
			text = POD_mortal_race_trigger
			scope:recipient = { has_supernatural_race_trigger = no }
		}	
	}

	on_accept = {
		scope:recipient = {
			remove_relation_worshipped_reborn = scope:actor
			POD_remove_supernatural_relation_effect = yes
			add_opinion = {
				modifier = POD_spell_of_life_opinion
				target = scope:actor
			}
			POD_remove_all_supernatural_atribute_effect = yes
			add_trait = mummy
			if = {
				limit = { scope:actor = { this = character:792593 } }
				add_trait = shemsuheru
			}
			if = {
				limit = { scope:actor = { this = character:792600 } }
				add_trait = cabiri
			}
			if = {
				limit = { 
					OR = { 
						NOT = { exists = dynasty }
						NOT = { dynasty = scope:actor.dynasty }
					}
				}
				set_house = scope:actor.house
				create_cadet_branch = yes
			}
			POD_mummy_initialization_effect = yes
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_restore_pneuma_by_magic_interaction = {
	category = interaction_category_mummy
	common_interaction = yes
	icon = POD_hekau_interaction

	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:actor = { NOT = { has_character_flag = restore_pneuma_by_magic_flag } } }
				desc = POD_restore_pneuma_by_magic_interaction_desc
			}
			desc = POD_not_restore_pneuma_by_magic_interaction_desc
		}
	}
	
	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { has_trait = cabiri }
		trigger_if = {
			limit = { scope:actor = { has_character_flag = restore_pneuma_by_magic_flag } }
			scope:actor = { capital_province = { NOT = { has_building_or_higher = occult_affiliation_04 } } }
		}
	}

	is_valid_showing_failures_only = {
	}
	
	on_accept = {
		scope:actor = { 
			custom_tooltip = POD_cabiri_maintenance.tt
			if = {
				limit = { scope:actor = { has_character_flag = restore_pneuma_by_magic_flag } }
				remove_character_flag = restore_pneuma_by_magic_flag
				custom_tooltip = POD_not_restore_pneuma_by_magic_interaction_desc
			}
			else = {
				add_character_flag = restore_pneuma_by_magic_flag
				custom_tooltip = POD_restore_pneuma_by_magic_interaction_desc
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
