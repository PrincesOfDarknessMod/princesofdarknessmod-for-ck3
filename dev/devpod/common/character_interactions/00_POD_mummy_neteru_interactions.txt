POD_set_neteru_interaction = {
	category = interaction_category_mummy
	common_interaction = yes
	interface_priority = 100
	desc = POD_set_neteru_interaction_desc
	icon = POD_hekau_interaction

	ai_min_reply_days = 4
	ai_max_reply_days = 9
	
	cost = { prestige = minor_prestige_value }
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			is_playable_character = yes
			OR = {
				is_amenti_trigger = yes
				AND = {
					scope:actor = character:790512
					has_character_flag = kemintiriisdoneforeal
				}
			}
		}
		scope:recipient = { 
			NOT = { has_relation_worshipped_reborn = scope:actor }
			NOT = { has_trait = mummy }
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			NOT =  { is_at_war_with = scope:recipient }
		}
		scope:recipient = {
			custom_description = {
				object = scope:actor
				text = is_courtier_or_vassal_of
				OR = {
					is_courtier_of = scope:actor
					is_vassal_of = scope:actor
				}
			}
			is_busy_in_events_localised = yes
			NOT = { has_strong_hook = scope:actor }
		}
		trigger_if = {
			limit = { scope:actor = { faith = { has_doctrine  = special_doctrine_against_to_apophis_hostility } } }
			NOT = { scope:recipient = { faith = { has_doctrine  = special_doctrine_allied_to_apophis_hostility } } }
		}
		trigger_if = {
			limit = { scope:actor = { faith = { has_doctrine  = special_doctrine_allied_to_apophis_hostility } } }
			NOT = { scope:recipient = { faith = { has_doctrine  = special_doctrine_against_to_apophis_hostility } } }
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			NOT = { has_trait = automaton }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
			POD_is_not_clone_trigger = yes		
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}

	#Use hook
	send_option = {
		is_valid = {
			scope:actor = {
				has_usable_hook = scope:recipient
			}
		}
		flag = hook
		localization = SCHEME_HOOK
	}
	should_use_extra_icon = {
		scope:actor = { has_usable_hook = scope:recipient }
	}
	extra_icon = "gfx/interface/icons/character_interactions/hook_icon.dds"

	send_options_exclusive = no

	on_accept = {
		scope:recipient = {
			set_relation_worshipped_reborn = scope:actor
			if = {
				limit = { 
					faith = { 
						NOR = { 
							has_doctrine = doctrine_undying_accepted
							has_doctrine = doctrine_undying_ignorant
						} 
					} 
				}
				neteru_faith_initiation_effect = { LIEGE = scope:actor }
			}
		}
	}
	
	ai_targets = {
		ai_recipients = vassals
	}
	
	ai_frequency = 72
	
	ai_potential = {
		primary_title.tier > tier_county
		prestige > minor_prestige_value
	}
	
	ai_accept = {
		base = 0 # Try to make it 0 for most interactions

		modifier = {
			scope:hook = yes
			add = 50
			desc = SCHEME_HOOK_USED
		}
		
		opinion_modifier = { # Opinion Factor
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.5
			desc = AI_SIMPLE_OPINION_REASON
		}
		
		modifier = {
			add = {
				value = ai_zeal
				divide = 2
				multiply = -1
			}
			ai_zeal > 0
			desc = TAKE_THE_VOWS_ZEALOUS
		}

		modifier = {
			add = -50
			has_relation_rival = scope:recipient
		}
		modifier = {
			add = -80
			has_relation_nemesis = scope:recipient
		}
		
		modifier = {
			add = 35
			scope:actor = { is_ai = yes }
		}
	}
	
	ai_will_do = {
		base = 100
		
		modifier = { # The AI will only use a Hook if they couldn't otherwise do this
			scope:hook = yes
			add = -1
		}
	}
}

POD_remove_neteru_interaction = {
	category = interaction_category_mummy
	common_interaction = yes
	interface_priority = 100
	desc = POD_remove_neteru_interaction_desc
	icon = POD_hekau_interaction
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		OR = {
			scope:actor = { has_trait = mummy }
			AND = {
				scope:actor = character:790512
				scope:actor = { has_character_flag = kemintiriisdoneforeal }
			}
		}		
		scope:recipient = { has_relation_worshipped_reborn = scope:actor }
	}

	on_accept = {
		scope:recipient = {
			remove_relation_worshipped_reborn = scope:actor
			if = {
				limit = {
					exists = house
					house = scope:actor.house
					can_execute_decision = create_cadet_branch_decision
				}
				create_cadet_branch = yes
			}
			if = {
				limit = {
					has_trait = mortal
					can_hold_title_as_mortal_trigger = no
				}
				POD_depose_effect = yes
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_spell_of_life_interaction = {
	category = interaction_category_mummy
	common_interaction = yes
	interface_priority = 90
	desc = POD_spell_of_life_interaction_desc
	icon = POD_hekau_interaction
	
	cooldown = { years = 10 }
	cost = {
		piety = {
			if = {
				limit = {
					scope:actor = {
						this = character:790512
						has_character_flag = kemintirinoredemption_flag
					}
				}
				value = 2500
			}
			else = {
				value = 10000
			}
		}
	}
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			NOT = { has_trait = banemummy }
			OR = {
				this = character:792593
				AND = {
					this = character:792600
					NOT = { has_character_flag = cabirus_giveup_flag }
				}
				AND = {
					this = character:790512
					has_character_flag = kemintiriisdoneforeal
				}
			}
		}
	}
	
	is_valid = {
		scope:recipient = { has_main_supernatural_race_trigger = no }
	}

	is_valid_showing_failures_only = {
		scope:recipient = { 
			has_relation_worshipped_reborn = scope:actor
			is_available_adult = yes
		}
	}

	on_accept = {
		scope:recipient = {
			save_temporary_scope_value_as = {
				name = saved_immortal_age
				value = POD_visual_age
			}
			POD_remove_splat_traits_effect = yes
			add_trait = mummy
			hidden_effect = { set_immortal_age = scope:saved_immortal_age }
			
			POD_remove_wrong_supernatural_traits_effect = yes
			POD_remove_wrong_supernatural_modifiers_effect = yes
			remove_relation_worshipped_reborn = scope:actor
			POD_remove_supernatural_relation_effect = yes
			
			add_opinion = {
				modifier = POD_spell_of_life_opinion
				target = scope:actor
			}
			if = {
				limit = { scope:actor = { this = character:792593 } }
				add_trait = shemsuheru
			}
			if = {
				limit = { scope:actor = { this = character:792600 } }
				add_trait = cabiri
			}
			POD_character_mummy_dynasty_initiation_effect = yes
			POD_mummy_random_perk_initialization_for_noble_effect = yes
			POD_mummy_initialization_effect = yes
			if = {
				limit = { scope:actor = { this = character:790512 } }
				set_character_faith = scope:actor.faith
			}			
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_ask_neteru_for_conversion_interaction = {
	category = interaction_category_religion
	common_interaction = yes
	icon = POD_hekau_interaction
	
	desc = {
		triggered_desc = {
			trigger = {
				scope:actor = { faith = { has_doctrine = special_doctrine_against_to_apophis_hostility } }
			}
			desc = POD_ask_neteru_for_osiris_conversion_interaction_desc
		}
		triggered_desc = {
			trigger = {
				scope:actor = { faith = { has_doctrine = special_doctrine_allied_to_apophis_hostility } }
			}
			desc = POD_ask_neteru_for_apophis_conversion_interaction_desc
		}
	}
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			has_trait = mummy
			OR = {
				has_character_modifier = horus_first_modifier
				has_character_modifier = horus_chosen_modifier
				has_character_modifier = pure_evil_modifier
			}
		}
	}

	is_valid = {
		scope:recipient = { 
			has_relation_worshipped_reborn = scope:actor 
			trigger_if = {
				limit = {
					scope:actor = { 
						OR = {
							has_character_modifier = horus_first_modifier
							has_character_modifier = horus_chosen_modifier
						}
					}
				}
				scope:recipient = { 
					NOT = { faith = { has_doctrine = special_doctrine_against_to_apophis_hostility } }
					OR = {
						has_POD_vamp_blood_trigger = yes
						has_trait = werewolf
						has_trait = kinfolk
						has_trait = supehunter
						has_trait = mortal
					}
				}
			}
			trigger_if = {
				limit = {
					scope:actor = { has_character_modifier = pure_evil_modifier }
				}
				scope:recipient = { 
					NOT = { faith = { has_doctrine  = special_doctrine_allied_to_apophis_hostility } }
					OR = {
						has_POD_vamp_blood_trigger = yes
						has_trait = kueijin
						has_trait = shapeshifter
						has_trait = kinfolk
						has_trait = supehunter
						has_trait = mortal
						has_trait = wraith
					}
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		valid_demand_conversion_conditions_trigger = yes
	}
	
	on_accept = {
		demand_conversion_interaction_effect = yes
		hidden_effect = {
			scope:actor = {
				send_interface_message = {	
					title = ask_for_conversion_courtier_interaction.tt
					left_icon = scope:recipient
					scope:recipient = {
						show_as_tooltip = { demand_conversion_interaction_effect = yes }
					}
				}
			}
		}
		scope:recipient = { 
			if = { 
				limit = { has_trait = mortal }
				set_character_faith = scope:actor.faith
			}
			else = { 
				neteru_faith_initiation_effect = { LIEGE = scope:actor }
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_restore_pneuma_by_magic_interaction = {
	category = interaction_category_mummy
	common_interaction = yes
	icon = POD_hekau_interaction

	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:actor = { NOT = { has_character_flag = restore_pneuma_by_magic_flag } } }
				desc = POD_restore_pneuma_by_magic_interaction_desc
			}
			desc = POD_not_restore_pneuma_by_magic_interaction_desc
		}
	}
	
	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { has_trait = cabiri }
		trigger_if = {
			limit = { scope:actor = { has_character_flag = restore_pneuma_by_magic_flag } }
			scope:actor = { capital_province = { NOT = { has_building_or_higher = occult_affiliation_04 } } }
		}
	}

	is_valid_showing_failures_only = {
	}
	
	on_accept = {
		scope:actor = { 
			custom_tooltip = POD_cabiri_maintenance.tt
			if = {
				limit = { scope:actor = { has_character_flag = restore_pneuma_by_magic_flag } }
				remove_character_flag = restore_pneuma_by_magic_flag
				custom_tooltip = POD_not_restore_pneuma_by_magic_interaction_desc
			}
			else = {
				add_character_flag = restore_pneuma_by_magic_flag
				custom_tooltip = POD_restore_pneuma_by_magic_interaction_desc
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
