POD_abandon_journey_interaction = {
	hidden = yes
	category = interaction_category_journeys
    desc = POD_abandon_journey_interaction_desc
	icon = council
	
	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { has_variable = current_journey }
	}
	
	on_accept = {
		custom_tooltip = {
			text = POD_abandon_journey_interaction_effect
			scope:actor = {
				POD_abandon_journey_effect = yes
			}
		}
		custom_tooltip = POD_abandon_journey_interaction_effect_2
	}

	auto_accept = yes
	
	ai_frequency = 0
	ai_will_do = {
		base = 0
	}
}

POD_hunt_target_interaction = {
	category = interaction_category_journeys
	common_interaction = yes
	icon = vamphunt_interaction
	
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:actor = { is_vampire_trigger = yes } }
				desc = POD_hunt_target_interaction_desc_vampire
			}
			triggered_desc = {
				trigger = { scope:actor = { has_trait = kueijin } }
				desc = POD_hunt_target_interaction_desc_chi
			}
			triggered_desc = {
				trigger = { scope:actor = { has_trait = banemummy } }
				desc = POD_hunt_target_interaction_desc_banemummy
			}
		}
	}
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { is_ai = no }
		scope:recipient = { is_adult = yes }
		OR = {
			# Vampire hunt
			AND = {
				scope:actor = { is_vampire_trigger = yes }
				scope:recipient = { is_vampire_trigger = no }
			}
			# Chi hunt
			AND = {
				scope:actor = { has_trait = kueijin }
				scope:recipient = { NOT = { has_trait = kueijin } }
			}
			# Bane hunt
			AND = {
				scope:actor = { has_trait = banemummy }
				scope:recipient = { NOT = { has_trait = banemummy } }
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = { POD_not_already_on_journey_trigger = { TYPE = hunt_target } }
		POD_cant_target_coterie_trigger = yes
		
		custom_description = {
			object = scope:recipient
			text = "blood_is_holy"
			scope:recipient = {
				NOR = {
					has_trait = truefaith
					has_trait = beatific
					has_trait = orison_sanguine
				}
			}
		}
		scope:actor = {
			is_adult = yes
			is_imprisoned = no
		}
		
		scope:recipient = {
			is_landed = no
			is_imprisoned = no
			POD_can_be_vamphunted_scheme_trigger = yes
			NOT = { has_character_modifier = slightbloodloss_modifier }
			NOT = { has_character_modifier = smallbloodloss_modifier }
			NOT = { has_character_modifier = majorbloodloss_modifier }
			
		}
		trigger_if = {
			limit = {
				scope:actor = {
					is_ventrue_trigger = yes
					is_ai = no
				}
			}
			custom_tooltip = {
				text = ventrue_bane_warning
				
				ventrue_weakness_compatible_trigger = yes
			}
		}
	}

	on_accept = {
		if = {
			limit = { scope:actor = { is_vampire_trigger = yes } }
			custom_tooltip = { text = POD_hunt_target_interaction_effect_vampire }
		}
		else_if = {
			limit = { scope:actor = { has_trait = kueijin } }
			custom_tooltip = { text = POD_hunt_target_interaction_effect_chi }
		}
		else_if = {
			limit = { scope:actor = { has_trait = banemummy } }
			custom_tooltip = { text = POD_hunt_target_interaction_effect_banemummy }
		}
		
		custom_tooltip = { text = POD_hunt_target_interaction_monthly }
		
		hidden_effect = {
			scope:actor = {
				POD_start_targeted_journey_effect = {
					TYPE    = hunt_target
					MONTHLY = ten_months
					TARGET  = scope:recipient
				}
			}
		}
	}

	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}


POD_wake_from_torpor_interaction = {
	category = interaction_category_journeys
    desc = POD_wake_from_torpor_interaction_desc
	icon = POD_awaketorpor_interaction

	ai_frequency = 0

	is_shown = {
		scope:actor = { is_ai = no }
		scope:recipient = { is_ai = yes }
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		scope:recipient = { has_trait = torpor  }
		scope:recipient = { is_imprisoned = no }
		scope:recipient = { NOT = { has_trait = statue } }
		scope:recipient = { NOT = { has_character_modifier = enfant_modifier } }
	}

	is_valid_showing_failures_only = {
		scope:actor = { POD_not_already_on_journey_trigger = { TYPE = wake_from_torpor } }
		
		scope:recipient = {
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
			POD_is_not_trapped_in_umbra_trigger = { ACTOR = scope:actor }
			OR = {
				exists = location
				in_diplomatic_range = scope:actor
			}
		}
	}

	on_accept = {
		if = {
			limit = {
				scope:recipient = { 
					is_vampire_trigger = yes 
				}
			}
			custom_tooltip = {
				text = POD_wake_from_torpor_interaction_effect
				scope:actor = {
					POD_start_targeted_journey_effect = {
						TYPE    = wake_from_torpor
						MONTHLY = five_years
						TARGET  = scope:recipient
					}
				}
			}
			custom_tooltip = { text = POD_wake_from_torpor_interaction_monthly }
		}
		else = {
			scope:actor = { add_prestige = medium_prestige_loss }
			scope:recipient = { remove_trait = torpor }
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
