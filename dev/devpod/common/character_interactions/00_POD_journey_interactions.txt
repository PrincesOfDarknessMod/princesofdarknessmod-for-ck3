POD_hunt_target_interaction = {
	category = interaction_category_journeys
	common_interaction = yes
	icon = vamphunt_interaction
	
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:actor = { is_vampire_trigger = yes } }
				desc = POD_hunt_target_interaction_desc_vampire
			}
			triggered_desc = {
				trigger = { scope:actor = { has_trait = kueijin } }
				desc = POD_hunt_target_interaction_desc_chi
			}
			triggered_desc = {
				trigger = { scope:actor = { has_trait = banemummy } }
				desc = POD_hunt_target_interaction_desc_banemummy
			}
		}
	}
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		OR = {
			# Vampire hunt
			AND = {
				scope:actor = { is_vampire_trigger = yes }
				scope:recipient = { is_vampire_trigger = no }
			}
			# Chi hunt
			AND = {
				scope:actor = { has_trait = kueijin }
				scope:recipient = { NOT = { has_trait = kueijin } }
			}
			# Bane hunt
			AND = {
				scope:actor = { has_trait = banemummy }
				scope:recipient = { NOT = { has_trait = banemummy } }
			}
		}
		scope:recipient = {
			NOT = { has_character_flag = POD_do_not_disturb_flag }
			POD_can_be_vamphunted_scheme_trigger = yes
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			POD_not_already_on_journey_trigger = {
				TYPE   = hunt_target
				TARGET = scope:recipient
			}
			POD_can_start_planning_phase_trigger = yes
		}
		POD_cant_target_coterie_trigger = yes
		
		custom_description = {
			object = scope:recipient
			text = "blood_is_holy"
			scope:recipient = {
				NOR = {
					POD_has_true_faith_trigger = yes
					has_trait = orison_sanguine
				}
			}
		}
		scope:actor = {
			is_adult = yes
			is_imprisoned = no
		}
		
		scope:recipient = {
			is_adult = yes
			is_landed = no
			is_imprisoned = no
		}
		trigger_if = {
			limit = {
				scope:actor = {
					is_ventrue_trigger = yes
					is_ai = no
				}
			}
			custom_tooltip = {
				text = ventrue_bane_warning
				
				ventrue_weakness_compatible_trigger = {
					ACTOR = scope:actor
					RECIPIENT = scope:recipient
				}
			}
		}
	}

	on_accept = {
		if = {
			limit = { scope:actor = { is_vampire_trigger = yes } }
			custom_tooltip = POD_hunt_target_interaction_effect_vampire
		}
		else_if = {
			limit = { scope:actor = { has_trait = kueijin } }
			custom_tooltip = POD_hunt_target_interaction_effect_chi
		}
		else_if = {
			limit = { scope:actor = { has_trait = banemummy } }
			custom_tooltip = POD_hunt_target_interaction_effect_banemummy
		}
		
		custom_tooltip = POD_hunt_target_interaction_monthly
		
		custom_tooltip = POD_journey_interaction_pause_hint
		
		hidden_effect = {
			scope:actor = {
				POD_start_targeted_journey_effect = {
					TYPE    = hunt_target
					MONTHLY = hunt_journey
					TARGET  = scope:recipient
				}
			}
		}
	}

	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}


POD_wake_from_torpor_interaction = {
	category = interaction_category_journeys
	desc = POD_wake_from_torpor_interaction_desc
	icon = POD_awaketorpor_interaction

	ai_frequency = 0

	use_diplomatic_range = no

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			is_ai = no
		}
		scope:recipient = {
			is_ai = yes
			is_adult = yes
			has_trait = torpor
			is_imprisoned = no
			NOT = { has_trait = statue }
			NOT = { has_character_modifier = caul_modifier }
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			POD_not_already_on_journey_trigger = {
				TYPE   = wake_from_torpor
				TARGET = scope:recipient
			}
			POD_can_start_planning_phase_trigger = yes
			is_adult = yes
			is_imprisoned = no
		}
		
		scope:recipient = {
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
			OR = {
				exists = location
				in_diplomatic_range = scope:actor
			}
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}

	on_accept = {
		if = {
			limit = {
				scope:recipient = { 
					is_vampire_trigger = yes 
				}
			}
			custom_tooltip = {
				text = POD_wake_from_torpor_interaction_effect
				scope:actor = {
					if = {
						limit = { has_perk = waken_perk }
						POD_start_targeted_journey_effect = {
							TYPE    = wake_from_torpor
							MONTHLY = instant
							TARGET  = scope:recipient
						}
					}
					else = {
						POD_start_targeted_journey_effect = {
							TYPE    = wake_from_torpor
							MONTHLY = wake_from_torpor
							TARGET  = scope:recipient
						}
					}
				}
			}
			custom_tooltip = POD_wake_from_torpor_interaction_monthly
		
			custom_tooltip = POD_journey_interaction_pause_hint
		}
		else = {
			scope:actor = {
				if = {
					limit = { NOT = { has_perk = waken_perk } }
					add_prestige = medium_prestige_loss 
				}
			}
			scope:recipient = { remove_trait = torpor }
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

			limit = { scope:recipient = { has_character_modifier = tracked_down_modifier } }

		}
