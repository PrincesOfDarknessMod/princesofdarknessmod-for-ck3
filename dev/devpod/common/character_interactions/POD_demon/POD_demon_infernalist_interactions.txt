POD_summon_demon_demonology_interaction = {
	category = interaction_category_demon
	icon = POD_infernalist_interaction
	desc = POD_summon_demon_demonology_interaction_desc

	is_shown = {
		OR = {
			scope:actor = scope:recipient
			AND = {
				scope:actor != scope:recipient
				scope:recipient = { POD_is_demon_trigger = yes }
				scope:recipient = { has_character_modifier = trapped_in_abyss_modifier }
			}
		}
		scope:actor = {
			has_trait_xp = {
				trait = demonology
				value >= 5
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			custom_tooltip = { 
				text = POD_summon_demon_demonology_interaction_valid 
				no_resurrecting_prisoner_count >= 1
			}
		}
		POD_self_targeted_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:actor = {
			save_scope_value_as = {
				name = summon_success_chance
				value = 0
			}
			custom_tooltip = {
				text = POD_summon_demon_demonology_interaction_effect1
				ordered_prisoner = {
					max = 10
					check_range_bounds = no
					limit = { POD_has_guaranteed_resurrection = no }
					order_by = {
						if = {
							limit = { is_lowborn = yes }
							add = 10
						}
						subtract = highest_held_title_tier
					}
					trigger_event = char_interaction.0171
					execute_prisoner_effect = {
						VICTIM = this
						EXECUTIONER = scope:actor
					}
					save_scope_value_as = {
						name = summon_success_chance
						value = {
							value = scope:summon_success_chance
							add = 10
						}
					}
				}
			}
			random = {
				chance = scope:summon_success_chance
				custom_tooltip = {
					text = POD_summon_demon_demonology_interaction_success
					if = {
						limit = { scope:recipient != scope:actor }
						scope:recipient = { save_scope_as = summoned_demon }
					}
					else = {
						random_living_character = {
							limit = { 
								POD_is_demon_trigger = yes
								has_character_modifier = trapped_in_abyss_modifier
							}
							save_scope_as = summoned_demon
						}
					}
					if = {
						limit = { NOT = { exists = scope:summoned_demon } }
						create_character = {
							location = scope:actor.location
							template = POD_summoned_demon_character
							save_scope_as = summoned_demon
						}
					}
					send_interface_toast = {
						type = event_generic_good
						title = POD_summon_demon_demonology_interaction_success
						right_icon = scope:actor
						left_icon = scope:summoned_demon
						add_courtier = scope:summoned_demon
						scope:summoned_demon ?= {
							POD_free_from_abyss_effect = yes
							hidden_effect = { return_to_court = yes }
							add_character_flag = { flag = blocked_from_leaving years = 5 }
						}
					}
				}
			}
			if = {
				limit = { NOT = { exists = scope:summoned_demon } }
				send_interface_toast = {
					type = event_generic_bad
					title = POD_shared_loca_failure
					left_icon = scope:actor
				}
			}
		}
	}
	
	auto_accept = yes
	
	
	
	ai_will_do = {
		base = 0
	}

	}

		scope:target = {
			POD_artifact_is_destructible_trigger = yes

			# but it's probably in-theme to allow any old trash/summoned artifact

		}
	}

	on_accept = {
		scope:actor = {
			random_in_list = {
				variable = demonic_pacts
				limit = { var:id = flag:offering_of_artifact }
