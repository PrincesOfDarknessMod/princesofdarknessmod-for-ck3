POD_reset_investiture_perks_interaction = {
	category = interaction_category_demon
	icon = POD_infernalist_interaction
	desc = POD_banish_demon_interaction_desc
	
	cooldown = { years = 5 }

	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { POD_character_can_use_investments = yes }
	}

	is_valid_showing_failures_only = {
		POD_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:actor = {
			refund_perks = investiture_lifestyle
			hidden_effect = {
				POD_remove_all_character_modifier_instances = { MODIFIER = POD_investiture_armor_perk_modifier }
				remove_variable = POD_investiture_armor_perk_modifier_stacks
				POD_remove_all_character_modifier_instances = { MODIFIER = POD_investiture_backgrounds_perk_modifier }
				remove_variable = POD_investiture_backgrounds_perk_modifier_stacks
				POD_remove_all_character_modifier_instances = { MODIFIER = POD_investiture_demonic_strength_perk_modifier }
				remove_variable = POD_investiture_demonic_strength_perk_modifier_stacks
				POD_remove_all_character_modifier_instances = { MODIFIER = POD_investiture_longevity_perk_modifier }
				remove_variable = POD_investiture_longevity_perk_modifier_stacks
				POD_remove_all_character_modifier_instances = { MODIFIER = POD_investiture_service_perk_modifier }
				remove_variable = POD_investiture_service_perk_modifier_stacks
			}
			add_investiture_lifestyle_perk_points = -1
		}
	}
