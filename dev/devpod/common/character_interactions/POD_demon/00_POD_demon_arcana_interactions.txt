			}
		}

	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_any_supernatural_disease_trigger = yes
			}
		}
		is_valid = {
			scope:actor = {
				has_trait = demon
				custom_tooltip = {
					text = POD_need_0_or_10_torment_text
					OR = {
						torment_value = 0
						torment_value = 10 
					}
				}
			}
		}
		flag = supernatural_disease

			limit = { NOT = { scope:actor = scope:recipient } }

				add_opinion = {
					target = scope:actor
					modifier = pleased_opinion
					opinion = 15
				}
			}
		}
		scope:recipient = {
			play_music_cue = "mx_cue_peace_ensues"
			remove_disease_trait_effect = yes
			switch = {
				trigger = yes
				scope:disease = { remove_disease_trait_effect = yes }
				scope:supernatural_disease = { remove_supernatural_disease_trait_effect = yes }
			}
			scope:actor = { POD_resolve_cost_effect = { COUNT = 3 } }
		}
	}
	
	auto_accept = yes

	ai_potential = {
		# The basic requirements
		is_imprisoned = no
		is_landed = yes
	}
	ai_targets = {
		ai_recipients = self
	}
	ai_targets = {
		ai_recipients = courtiers
		ai_recipients = peer_vassals
		max = 10
	}
	ai_frequency = 12
	ai_will_do = {
		base = 0
		modifier = {
			add = 100
			scope:recipient = {
				OR = {
					this = scope:actor
					is_knight_of = scope:actor
				}
			}
		}
	}
