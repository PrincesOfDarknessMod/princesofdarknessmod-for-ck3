
POD_tempt_interaction = {
	icon = POD_demon_interaction
	category = interaction_category_demon
	interface_priority = 90
	scheme = POD_tempt_scheme

	send_name = START_SCHEME
	ignores_pending_interaction_block = yes
	
	ai_frequency = 0
	
	cooldown_against_recipient = { years = 20 } 

	is_shown = {
		scope:actor = { has_trait = demon }
		scope:actor = { religion = religion:demons_religion }
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		can_be_tempted_trigger = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			NOT = { is_at_war_with = scope:recipient }
			can_start_scheme = {
				type = POD_tempt_scheme
				target_character = scope:recipient
			}
		}
		scope:recipient = { POD_has_true_faith_trigger = no }
		scope:recipient = {
			NOT = { has_trait = torpor }
			NOT = { has_trait = automaton }
			NOT = { has_trait = wight }
			NOT = { has_trait = barghest }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
			POD_is_not_clone_trigger = yes	
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}

	# Scheme Starter Packages
	options_heading = schemes.t.agent_packages
	send_options_exclusive = yes
	## Balanced agents.
	send_option = {
		flag = agent_focus_balance
		current_description = POD_tempt_interaction.tt.agent_focus_balance
	}
	## Focused on Success Chance.
	send_option = {
		flag = agent_focus_success
		current_description = POD_tempt_interaction.tt.agent_focus_success
	}
	## Focused on Speed.
	send_option = {
		flag = agent_focus_speed
		current_description = POD_tempt_interaction.tt.agent_focus_speed
	}

	desc = POD_tempt_interaction_desc

	on_accept = {
		hidden_effect = {
			scope:actor = {
				# Success.
				if = {
					limit = { scope:agent_focus_success ?= yes }
					begin_scheme_with_agents_effect = {
						SCHEME_TYPE = POD_tempt_scheme
						TARGET_TYPE = target_character
						TARGET_SCOPE = scope:recipient
						# Success.
						AGENT_1 = agent_shill
						AGENT_2 = agent_diplomat
						AGENT_3 = agent_comrade_in_arms
						AGENT_4 = agent_musician
						# Speed.
						AGENT_5 = agent_socialite
					}
				}
				# Speed.
				else_if = {
					limit = { scope:agent_focus_speed ?= yes }
					begin_scheme_with_agents_effect = {
						SCHEME_TYPE = POD_tempt_scheme
						TARGET_TYPE = target_character
						TARGET_SCOPE = scope:recipient
						# Success.
						AGENT_1 = agent_shill
						AGENT_2 = agent_diplomat
						# Speed.
						AGENT_3 = agent_socialite
						AGENT_4 = agent_infiltrator
						AGENT_5 = agent_gabbler
					}
				}
				# Balanced.
				else = {
					begin_scheme_with_agents_effect = {
						SCHEME_TYPE = POD_tempt_scheme
						TARGET_TYPE = target_character
						TARGET_SCOPE = scope:recipient
						# Success.
						AGENT_1 = agent_shill
						AGENT_2 = agent_diplomat
						AGENT_3 = agent_comrade_in_arms
						# Speed.
						AGENT_4 = agent_socialite
						AGENT_5 = agent_infiltrator
					}
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
	
}
