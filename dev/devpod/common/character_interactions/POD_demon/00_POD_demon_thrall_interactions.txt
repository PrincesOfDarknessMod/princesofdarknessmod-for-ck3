POD_demonic_submission_interaction = {
	category = interaction_category_demon
	icon = POD_demon_interaction
	desc = POD_demonic_submission_interaction_desc
	
	cost = { prestige = 250 }
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_trait = demon }
		POD_incompatible_splats_trigger = no
		scope:recipient = {
			is_landed = yes
		}
		scope:actor = {
			highest_held_title_tier >= tier_duchy
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = { NOT = { is_vassal_of = scope:recipient } }
	}

	on_accept = {
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = title_change
			add_claim_on_loss = yes
		}
		if = {
			limit = { scope:actor.primary_title.tier = tier_duchy }
			scope:actor = {
				every_held_title = {
					limit = { tier = tier_duchy }
					change_title_holder = {
						holder = scope:recipient
						change = scope:title_change
					}
					scope:recipient = {
						add_opinion = {
							target = scope:actor
							modifier = received_title_duchy
						}
					}
				}
			}
		}
		else_if = {
			limit = { scope:actor.primary_title.tier = tier_kingdom }
			scope:actor = {
				every_held_title = {
					limit = { tier = tier_kingdom }
					change_title_holder = {
						holder = scope:recipient
						change = scope:title_change
					}
					scope:recipient = {
						add_opinion = {
							target = scope:actor
							modifier = received_title_kingdom
						}
					}
				}
			}
		}
		else_if = {
			limit = { scope:actor.primary_title.tier = tier_empire }
			scope:actor = {
				every_held_title = {
					limit = { tier = tier_empire }
					change_title_holder = {
						holder = scope:recipient
						change = scope:title_change
					}
					scope:recipient = {
						add_opinion = {
							target = scope:actor
							modifier = received_title_empire
						}
					}
				}
			}
		}
		resolve_title_and_vassal_change = scope:title_change
		
		create_title_and_vassal_change = {
			type = swear_fealty
			save_scope_as = change
		}
		scope:actor = {
			change_liege = {
				liege = scope:recipient
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change
		scope:recipient = { 
			add_opinion = {
				target = scope:actor
				modifier = gift_opinion
				opinion = 20
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

			scope:hook = yes
			add = 50
			desc = SCHEME_WEAK_HOOK_USED
		}

			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.5
			desc = AI_SIMPLE_OPINION_REASON
		}

		modifier = {
			add = -50
			NOT = { scope:actor.faith = scope:recipient.faith }
			desc = "ZEAL_AGAINST_INFIDELS"
		}

		modifier = {
			add = 25
			scope:actor.faith = scope:recipient.faith
			desc = "SAME_FAITH"
		}

		modifier = {
			add = -50
			has_relation_rival = scope:recipient
		}
		modifier = {
			add = -80
			has_relation_nemesis = scope:recipient
		}

		modifier = { # Safegurad for failed infernalist baron initiation
			add = 1000
			scope:actor = { is_ai = yes }
		}
	}

	ai_will_do = {
		base = 100

		modifier = { # The AI will only use a Hook if they couldn't otherwise do this
			scope:hook = yes
			add = -1
		}
	}

			trigger_if = {
				limit = { scope:secondary_recipient = { primary_title.tier = tier_kingdom } }
				scope:actor = { arcana_lifestyle_perk_points >= 8 }
			}
			trigger_else_if = {
				limit = { scope:secondary_recipient = { primary_title.tier = tier_duchy } }
				scope:actor = { arcana_lifestyle_perk_points >= 4 }
			}
			trigger_else_if = {
				limit = { scope:secondary_recipient = { primary_title.tier = tier_county } }
				scope:actor = { arcana_lifestyle_perk_points >= 2 }
			}
			trigger_else = {

			}
		}

	}

		scope:recipient = {
			every_vassal = {
				limit = {
					NOT = { this = scope:actor }
					primary_title.tier > tier_barony
				}

			}
		}
	}

			scope:recipient = { add_investiture_lifestyle_perk_points = 8 }
		}
		else_if = {
			limit = { scope:secondary_recipient = { primary_title.tier = tier_duchy } }
			scope:actor = { add_arcana_lifestyle_perk_points = -4 }
			scope:recipient = { add_investiture_lifestyle_perk_points = 4 }
		}
		else_if = {
			limit = { scope:secondary_recipient = { primary_title.tier = tier_county } }
			scope:actor = { add_arcana_lifestyle_perk_points = -2 }
			scope:recipient = { add_investiture_lifestyle_perk_points = 2 }
		}
		else = {
			scope:actor = { add_arcana_lifestyle_perk_points = -1 }
			scope:recipient = { add_investiture_lifestyle_perk_points = 1 }
		}
		create_title_and_vassal_change = {
			type = swear_fealty
			save_scope_as = change
		}
		scope:secondary_recipient = {
			change_liege = {
				liege = scope:actor
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change
	}

	ai_will_do = {
		base = 0
	}
