POD_fuddle_perk_interaction = {
	category = interaction_category_fae
	icon = POD_chicanery_interaction
	desc = POD_fuddle_perk_interaction_desc
	use_diplomatic_range = no
	
	is_shown = {
		scope:actor = { has_POD_fae_blood_trigger = yes }
		scope:actor = { has_perk = fuddle_perk }
	}
	
	is_valid_showing_failures_only = {
		scope:actor = { NOT = { has_character_modifier = glamour0_modifier } }
		realm_actor = yes
		realm_fae = yes
		realm_scene = yes
	}
	
	send_option = {
		is_shown = {
			scope:recipient = {
				NOT = { has_character_modifier = fuddle_perk_positive_modifier }
			}
		}
		flag = positive
		localization = fuddle_perk_positive_modifier	
	}
	
	send_option = {
		is_shown = {
			scope:recipient = {
				NOT = { has_character_modifier = fuddle_perk_negative_modifier }
			}
		}
		flag = negative
		localization = fuddle_perk_negative_modifier	
	}
	
	send_option = {
		is_shown = {
			scope:recipient = {
				OR = { 
					has_character_modifier = fuddle_perk_positive_modifier 
					has_character_modifier = fuddle_perk_negative_modifier 
				}
			}
		}
		flag = remove
		localization = POD_remove	
	}
	
	on_accept = {
		scope:actor = { POD_willpower_rouse_check_effect = yes }
		scope:actor = { POD_glamour_decrease_effect = yes }
		scope:recipient = {
			if = {
				limit = { scope:positive = yes }
				add_character_modifier = {
					modifier = fuddle_perk_positive_modifier
					months = realm_time
				}
			}
			if = {
				limit = { scope:negative = yes }
				add_character_modifier = {
					modifier = fuddle_perk_negative_modifier
					months = realm_time
				}
			}
			if = {
				limit = { scope:remove = yes }
				remove_character_modifier = fuddle_perk_positive_modifier
				remove_character_modifier = fuddle_perk_negative_modifier
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_fugue_perk_interaction = {
	category = interaction_category_fae
	icon = POD_chicanery_interaction
	desc = POD_fugue_perk_interaction_desc
	use_diplomatic_range = no
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_POD_fae_blood_trigger = yes }
		scope:actor = { has_perk = fugue_perk }
	}
	
	is_valid_showing_failures_only = {
		scope:actor = { NOT = { has_character_modifier = glamour0_modifier } }
		scope:actor = { NOT = { has_character_modifier = glamour1_modifier } }
		scope:actor = { NOT = { has_character_modifier = glamour2_modifier } }
		realm_actor = yes
		realm_fae = yes
		realm_scene = yes
	}
	
	send_option = {
		is_valid = {
			scope:recipient = {
				any_unspent_known_secret = {
					secret_owner = scope:actor
				}
			}
		}
		flag = secret
		localization = game_concept_secrets	
	}
	
	send_option = {
		is_valid = {
			scope:recipient = {
				any_claim = {
					holder = scope:actor
				}
			}
		}
		flag = title_claim
		localization = game_concept_title_claims	
	}
	
	send_option = {
		is_valid = {
			scope:actor = {
				any_character_artifact = {
					scope:recipient = { has_artifact_claim = prev }
				}
			}
		}
		flag = artifact_claim
		localization = POD_artifact_claims	
	}
	
	send_option = {
		is_valid = {
			scope:recipient = {
				has_usable_hook = scope:actor
			}
		}
		flag = hook
		localization = game_concept_hooks	
	}
	
	on_accept = {
		scope:actor = { POD_willpower_rouse_check_effect = yes }
		scope:actor = { POD_glamour_decrease_multiple_effect = { COUNT = 3 } }
		if = {
			limit = { scope:secret = yes }
			scope:recipient = {
				random_unspent_known_secret = {
					limit = { secret_owner = scope:actor }
					disable_exposure_by = scope:recipient
				}
			}
		}
		if = {
			limit = { scope:title_claim = yes }
			scope:recipient = {
				random_claim = { 
					limit = { holder = scope:actor }
					scope:recipient = { 
						if = {
							limit = { has_strong_claim_on = prev }
							make_claim_weak = prev
						}
						else = {
							remove_claim = prev 
						}
					}
				}
			}
		}
		if = {
			limit = { scope:artifact_claim = yes }
			scope:actor = {
				random_character_artifact = { 
					limit = { scope:recipient = { has_artifact_claim = prev } }
					scope:recipient = { remove_personal_artifact_claim = prev }
				}
			}
		}
		if = {
			limit = { scope:hook = yes }
			scope:recipient = { use_hook = scope:actor }
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
