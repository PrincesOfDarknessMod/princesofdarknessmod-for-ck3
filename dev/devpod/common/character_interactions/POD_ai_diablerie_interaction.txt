POD_ai_diablerize_prisoner_interaction = {
	category = interaction_category_vampire
	desc = POD_malkavian_sacrifice_interaction_desc
	icon = POD_vampire_interaction
	is_shown = {
		has_game_rule = pod_heightened_fatality_enabled
		NOT = { scope:actor = scope:recipient }		
		scope:recipient = {
			is_imprisoned_by = scope:actor
		}
		scope:actor = { 
			has_trait = vampire
			is_ruler = yes
			highest_held_title_tier >= tier_county
			}
		scope:recipient = { has_trait = vampire }
		OR = {
			scope:actor = { is_ai = yes }
			debug_only = yes
		}
	}
	ai_frequency = 19
	
	ai_targets = {
		ai_recipients = prisoners
	}
	ai_potential = {
		POD_is_on_golconda_trigger = no
	}
	
	on_accept = {
		scope:actor = {
			save_scope_as = diablerist
			if = {
				limit = { scope:actor.faith = { has_doctrine = doctrine_vampirediablerie_accepted } }
				set_variable = {
					name = known_diablerie
					value = yes
				}
			}
			else = {				
				set_variable = {
					name = hidden_diablerie
					value = yes
				}
			}
			POD_queue_diablerie_suspicion_event = { VICTIM = scope:recipient }		 
			POD_diablerie_effect = { VICTIM = scope:recipient }
			debug_log = debug_log.character_diablerized
			debug_log_scopes = yes
			if = { limit = { exists = var:known_diablerie } #cleanup
				remove_variable = known_diablerie }
			if = { limit = { exists = var:hidden_diablerie }
				remove_variable = hidden_diablerie }
		}
	}
	ai_will_do = {  
		base = -30

		modifier = {  
			has_trait_rank = { trait = "vampgeneration" character > scope:recipient }  
			add = 60 
		}
		modifier = {
			has_execute_reason = scope:recipient 
			factor = 1.25
			}
		opinion_modifier = {  
			opinion_target = scope:recipient  
			multiplier = -0.3
		} 
		modifier = {  
			has_trait = sadistic  
			add = 10  
		}
		modifier = {
			has_trait = wrathful
			add = 5
			}
		modifier = {  
			is_diablerist_trigger = yes #addictive
			add = 10
		}
		modifier = {  
			faith = { has_doctrine_parameter = sanctioned_diableries }  
			add = 10  
		}  
		modifier = {
			has_character_modifier = POD_izhim_ur_baal_curse_modifier
			add = 50
			}
		modifier = {
			scope:recipient = { has_trait_rank = { trait = "vampgeneration" rank > 6 } } #Elder vitae is more tempting
			factor = 2
			}
		modifier = {  
			faith = { has_doctrine = doctrine_vampirediablerie_accepted }  
			factor = 4  
		} 		
		modifier = {
			has_trait = ambitious  
			scope:actor = { has_trait_rank = { trait = "vampgeneration" character > scope:recipient } }
			factor = 3
		}  
		modifier = {  
			AND = {
				OR = {
					has_trait = wrathful
					has_trait = sadistic  
				OR = {  
					has_trait = callous  
					has_trait = arbitrary  
					has_trait = deceitful  
					}
				}
			}  
			factor = 2.5
		}  
		modifier = {  
			faith = { has_doctrine = doctrine_vampirediablerie_crime }  
			factor = 0.05
		} 
		modifier = {  
			piety < 300
			add = -30 #vampires will diablerize themselves into piety bankruptcy otherwise
		}  
		modifier = {  
			OR = { 	has_trait = compassionate
					has_trait = forgiving
					AND = {
						has_trait = just
						faith = { has_doctrine = doctrine_vampirediablerie_crime }
						}
					}
			factor = 0 
		} 
	}

	auto_accept = yes
}
