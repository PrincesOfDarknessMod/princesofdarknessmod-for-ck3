    desc = POD_nocturne_on_personal_interaction_desc

		}

		       limit = {
			       has_perk = nocturne_perk
				   OR = {
                       has_perk = the_darkness_within_perk
					   has_trait = obtenebrationadvanced
				   }
				   NOT = { has_perk = devour_all_hope_perk }
			   }
			   POD_moderate_risk_of_increasing_masquerade_exposure = yes
		   }
		   else_if = {
		       limit = {
			       has_perk = devour_all_hope_perk
			   }
			   POD_hunger_rouse_check_effect = yes
			   POD_very_high_risk_of_increasing_masquerade_exposure = yes

	}
	
	ai_targets = {
		ai_recipients = self
	}
	
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 45
			is_at_war = yes
			piety > 450
		}
		
		modifier = {
			add = 100
			is_commanding_army = yes
			piety > 250
		}
	}

POD_feed_the_darkness_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	#interface_priority = 88
	desc = POD_feed_the_darkness_interaction_desc
	ai_frequency = 0
	
	is_shown = {
		scope:actor = { has_POD_vamp_blood_trigger = yes }
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		scope:actor = {
			has_character_modifier = ritual_feed_the_darkness_modifier
		}
		scope:recipient = { OR = { is_courtier_of = scope:actor }  }
	}
	is_valid_showing_failures_only = {
		scope:recipient = { strong_mental_defenses_trigger = no }
	}
	
	send_option = {
		flag = sip
		is_shown = {
			scope:recipient = { drinkable_blood_left >= 1 }
		}
		localization = POD_drink_herd_sip_option
	}
	
	send_option = {
		flag = drink
		is_shown = {
			scope:recipient = { drinkable_blood_left >= 2 }
		}
		localization = POD_drink_herd_drink_option
	}
	
	send_option = {
		flag = alot
		is_shown = {
			scope:recipient = { drinkable_blood_left >= 3 }
		}
		localization = POD_drink_herd_alot_option
	}
	
	send_option = {
		flag = drain
		localization = POD_drink_herd_drain_option
	}
	
	on_accept = {
		#Sets the proper scopes so that the effects work
		scope:actor = { save_scope_as = vampire }
		scope:recipient = { save_scope_as = victim }
		scope:actor = { save_scope_as = herd_owner }
		scope:recipient = { save_scope_as = herdmember }
		
		hidden_effect = {
			scope:victim = {
				POD_set_resonance_effect = yes
			}
		}

		if = {
			limit = { scope:sip = yes }

			scope:actor = {
				POD_drink_from_victim_no_secondary_effect = {
					VICTIM = scope:recipient
					AMOUNT = 1
				}
			}
		}

		if = {
			limit = { scope:drink = yes }

			scope:actor = {
				POD_drink_from_victim_no_secondary_effect = {
					VICTIM = scope:recipient
					AMOUNT = 2
				}
			}
		}

		if = {
			limit = { scope:alot = yes }

			scope:actor = {
				POD_drink_from_victim_no_secondary_effect = {
					VICTIM = scope:recipient
					AMOUNT = 3
				}
			}
		}

		if = {
			limit = { scope:drain = yes }

			scope:actor = {
				if = { limit = { scope:recipient = { has_main_supernatural_race_trigger = yes } } 
					duel = {
						skills = { prowess intrigue }
						target = scope:recipient
						50 = {
							desc = board_games.0001.tt.success
							compare_modifier = {
								value = scope:duel_value
							}
							POD_drain_victim_effect = { VICTIM = scope:recipient }
							remove_character_modifier = ritual_feed_the_darkness_modifier
							POD_static_stress_gain = { COUNT = 100 } 
						}
						50 = {
							desc = board_games.0001.tt.failure
							compare_modifier = {
								value = scope:duel_value
								multiplier = high_negative_duel_skill_multiplier 
							}
							scope:actor = { 
								add_stress = 125
								add_piety = massive_piety_loss
							}
						}
					}
				}
				else = {
					POD_drain_victim_effect = { VICTIM = scope:recipient }
				}
			}
		}
	}
	auto_accept = yes
}

POD_call_leviathan_to_war_interaction = {
	category = interaction_category_vampire
	icon = alliance
	desc = POD_call_leviathan_to_war_interaction_desc
	
	cost = { piety = 500 gold = 50 }

	is_shown = {
		scope:actor = scope:recipient 
		scope:actor = { has_character_flag = access_to_leviathan }
	}

	is_valid = {
		scope:actor = {
			is_at_war = yes
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			trigger_if = {
				limit = { has_variable = leviathan_summon }
				custom_tooltip = {
					text = POD_call_leviathan_to_war_interaction_tt
					NOT = { any_character_war = { this = scope:actor.var:leviathan_summon } }
				}
			}
		} 
	}

	on_accept = {
		scope:actor = {
			random_character_war = { save_temporary_scope_as = war }
			spawn_army = {
				men_at_arms = {
					type = the_leviathan
					stacks = 1
				}
				location = capital_province
				war = scope:war
				inheritable = no
				name = army_of_leviathans
			}
			set_variable = { name = leviathan_summon value = scope:war }
		}
	}

	auto_accept = yes
}
