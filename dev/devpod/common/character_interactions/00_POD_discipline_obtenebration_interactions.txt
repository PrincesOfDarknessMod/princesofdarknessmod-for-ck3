POD_drain_the_life_interaction = {
	category = interaction_category_vampire
	#common_interaction = yes
	#interface_priority = 90
	icon = POD_obtenebration_interaction
    desc = POD_drain_the_life_interaction_desc
	
	ai_frequency = 0

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_obtenebration_trigger = yes }
		scope:actor = { 
		   has_perk = drain_the_life_perk
		}
		scope:recipient = { is_landed = yes }
		scope:recipient.highest_held_title_tier > tier_barony
		scope:actor = { NOT = { target_is_liege_or_above = scope:recipient } }
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
			POD_has_enough_blood_for_discipline = { AMOUNT = 2 }
		}
	}

	on_accept = {
	   custom_tooltip = POD_drain_the_life_interaction.tt
	   scope:actor = {
	       trigger_event = POD_drain_the_life.001
	   }
    }
	
	#ai_targets = {
	#	ai_recipients = primary_war_enemies
	#}
	
	auto_accept = yes
	
	cooldown = { years = 5 }
	
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 45
			scope:actor = {
			    is_at_war_with = scope:recipient
			}
			piety > 300
		}
		modifier = {
		    add = 10
			scope:actor = {
			    has_relation_rival = scope:recipient
			}
		}
	}  	
}


POD_nocturne_on_personal_interaction = {
	category = interaction_category_vampire
	#common_interaction = yes
	#interface_priority = 90
	icon = POD_obtenebration_interaction
    desc = POD_nocturne_on_personal_interaction_desc
	
	ai_frequency = 20

	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { has_obtenebration_trigger = yes }
		scope:actor = { 
		   has_perk = nocturne_perk
		}
		scope:actor = { NOT = { has_character_modifier = nocturne_power_modifier } }
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
			POD_has_enough_blood_for_discipline = { AMOUNT = 1 }
		}
	}

	on_accept = {
	   custom_tooltip = POD_nocturne_on_personal_interaction.tt
	   scope:actor = {
	       add_character_modifier = nocturne_power_modifier
		   POD_hunger_rouse_check_effect = yes
		   if = {
		       limit = {
			       has_perk = nocturne_perk
				   NOT = { has_perk = the_darkness_within_perk }
				   NOT = { has_trait = obtenebrationadvanced }
			   }
			   POD_small_risk_of_increasing_masquerade_exposure = yes
		   }
		   else_if = {
		       limit = {
			       has_perk = nocturne_perk
				   OR = {
                       has_perk = the_darkness_within_perk
					   has_trait = obtenebrationadvanced
				   }
				   NOT = { has_perk = devour_all_hope_perk }
			   }
			   POD_moderate_risk_of_increasing_masquerade_exposure = yes
		   }
		   else_if = {
		       limit = {
			       has_perk = devour_all_hope_perk
			   }
			   POD_hunger_rouse_check_effect = yes
			   POD_very_high_risk_of_increasing_masquerade_exposure = yes
		   }
	   }
	}
	
	ai_targets = {
		ai_recipients = self
	}
	
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 45
			is_at_war = yes
			piety > 450
		}
		
		modifier = {
			add = 100
			is_commanding_army = yes
			piety > 250
		}
	}  	
}
