POD_torture_all_prisoners_interaction = {
	interface_priority = 201
	category = interaction_category_prison
	icon = torture_interaction

	desc = POD_torture_all_prisoners_interaction_desc
	use_diplomatic_range = no
	
	common_interaction = yes

	is_shown = {
		scope:actor = scope:recipient # self only
		scope:actor = {
			any_prisoner = {
				NOT = { has_character_modifier = recently_tortured }
				count > 1
			}
		}
	}

	is_valid = {
	}

	is_valid_showing_failures_only = {
	}

	on_accept = {
		scope:actor = {
			custom_tooltip = POD_torture_all_prisoners_interaction_effect

			if = {
				limit = { mass_torture_piety_gain != 0 }
				custom_tooltip = POD_torture_all_prisoners_interaction_effect_piety
			}

			if = {
				limit = { has_trait = sadistic }
				custom_tooltip = POD_torture_all_prisoners_interaction_effect_stress_sadistic
			}
			if = {
				limit = { has_trait = zealous }
				custom_tooltip = POD_torture_all_prisoners_interaction_effect_stress_zealous
			}
			if = {
				limit = {
					OR = {
						has_trait = compassionate
						has_trait = craven
						has_trait = just
						has_trait = forgiving
						has_trait = generous
						has_trait = trusting
						has_trait = content
					}
				}
				custom_tooltip = POD_torture_all_prisoners_interaction_effect_stress_gain
			}

			hidden_effect = {
				save_scope_value_as = { name = torture_method value = flag:whip }
				every_prisoner = {
					limit = {
						NOT = { has_character_modifier = recently_tortured }
						NOT = { has_character_flag = is_being_tortured }
					}
					save_scope_as = torture_recipient
	
					POD_torture_interaction_actor_effect = yes
					POD_torturer_interaction_recipient_effect = yes
					POD_house_feud_torture_event_effect = yes
	
					prev = {
						POD_secondary_effects_from_torturing = yes
						
						if = {
							limit = { has_perk = dark_insights_perk }
							dark_insights_skill_gain_roll_effect = yes
						}
					}
				}
			}
		}
	}

	ai_frequency = 0
	
	auto_accept = yes
}

POD_mass_demand_payment_interaction = {
	interface_priority = 201
	category = interaction_category_diplomacy
	icon = icon_gold

	desc = POD_mass_demand_payment_interaction_desc
	use_diplomatic_range = no
	
	common_interaction = yes

	is_shown = {
		scope:actor = scope:recipient # self only
		scope:actor = {
			OR = {
				has_perk = golden_obligations_perk
				AND = {
					any_character_struggle = {
						involvement = involved
						has_struggle_phase_parameter = unlocks_demand_payments_for_all
						# is_secondary_character_involvement_either_trigger = {
						# 	CHAR = scope:recipient
						# }
					}
				}
			}
		}
		scope:actor = {
			any_hooked_character = {
				count > 1

				gold >= 1

				is_ai = yes
				is_imprisoned = no
				NOT = { is_at_war_with = scope:actor }

				# If payment was unlocked via struggle, check if both characters are in the same struggle
				trigger_if = {
					limit = {
						scope:actor = {
							NOT = { has_perk = golden_obligations_perk }
							any_character_struggle = {
								involvement = involved
								has_struggle_phase_parameter = unlocks_demand_payments_for_all
							}
						}
					}
					any_character_struggle = {
						involvement = involved
						has_struggle_phase_parameter = unlocks_demand_payments_for_all
						is_secondary_character_involvement_either_trigger = {
							CHAR = scope:actor
						}
					}
				}
			}
		}
	}

	is_valid = {
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			is_physically_able_adult = yes
		}
	}

	send_options_exclusive = no

	send_option = {
		flag = exclude_partial_payments
		localization = POD_mass_demand_payment_interaction_exclude_partial_payments
	}

	send_option = {
		flag = strong_hooks_only
		localization = POD_mass_demand_payment_interaction_strong_hooks_only
	}

	on_accept = {
		scope:actor = {
			custom_description = {
				text = POD_mass_demand_payment_num
				subject = scope:actor
				value = mass_payment_num_of_characters
			}
			custom_description = {
				text = POD_mass_demand_payment_gold
				subject = scope:actor
				value = mass_payment_total_gold
			}

			hidden_effect = {
				every_hooked_character = {
					limit = {
						trigger_if = {
							limit = { scope:exclude_partial_payments = no }
							gold >= 1
						}
						trigger_else = {
							gold > golden_obligation_value
						}

						trigger_if = {
							limit = { scope:strong_hooks_only = yes }
							scope:actor = { has_strong_usable_hook = prev }
						}
						trigger_else = {
							scope:actor = { has_usable_hook = prev }
						}

						is_ai = yes
						is_imprisoned = no
						NOT = { is_at_war_with = scope:actor }

						# If payment was unlocked via struggle, check if both characters are in the same struggle
						trigger_if = {
							limit = {
								scope:actor = {
									NOT = { has_perk = golden_obligations_perk }
									any_character_struggle = {
										involvement = involved
										has_struggle_phase_parameter = unlocks_demand_payments_for_all
									}
								}
							}
							any_character_struggle = {
								involvement = involved
								has_struggle_phase_parameter = unlocks_demand_payments_for_all
								is_secondary_character_involvement_either_trigger = {
									CHAR = scope:actor
								}
							}
						}
					}

					save_scope_as = payment_recipient

					if = {
						limit = { gold > golden_obligation_value }
						pay_short_term_gold = {
							target = scope:actor
							gold = golden_obligation_value
						}
					}
					else = {
						pay_short_term_gold = {
							target = scope:actor
							gold = {
								value = scope:payment_recipient.gold
								floor = yes
							}
						}
					}

					prev = {
						use_hook = prev
						
						stress_impact = {
							generous = medium_stress_impact_gain
						}
					}

					# If we're a clan this interaction affects unity
					add_clan_unity_interaction_effect = {
						CHARACTER = scope:actor
						TARGET = scope:payment_recipient
						VALUE = minor_unity_loss
						DESC = clan_unity_demand_payment.desc
						REVERSE_NON_HOUSE_TARGET = no
					}
				}
			}
		}
	}

	ai_frequency = 0
	
	auto_accept = yes
}
