POD_torture_all_prisoners_interaction = {
	interface_priority = 201
	category = interaction_category_prison
	icon = torture_interaction

	desc = POD_torture_all_prisoners_interaction_desc
	use_diplomatic_range = no
	
	common_interaction = yes

	is_shown = {
		scope:actor = scope:recipient # self only
		scope:actor = {
			any_prisoner = {
				NOT = { has_character_modifier = recently_tortured }
				count > 1
			}
		}
	}

	is_valid = {
	}

	is_valid_showing_failures_only = {
	}

	on_accept = {
		scope:actor = {
			custom_tooltip = POD_torture_all_prisoners_interaction_effect

			if = {
				limit = { mass_torture_piety_gain != 0 }
				custom_tooltip = POD_torture_all_prisoners_interaction_effect_piety
			}

			if = {
				limit = { has_trait = sadistic }
				custom_tooltip = POD_torture_all_prisoners_interaction_effect_stress_sadistic
			}
			if = {
				limit = { has_trait = zealous }
				custom_tooltip = POD_torture_all_prisoners_interaction_effect_stress_zealous
			}
			if = {
				limit = {
					OR = {
						has_trait = compassionate
						has_trait = craven
						has_trait = just
						has_trait = forgiving
						has_trait = generous
						has_trait = trusting
						has_trait = content
					}
				}
				custom_tooltip = POD_torture_all_prisoners_interaction_effect_stress_gain
			}

			hidden_effect = {
				every_prisoner = {
					limit = {
						NOT = { has_character_modifier = recently_tortured }
						NOT = { has_character_flag = is_being_tortured }
					}
					save_scope_as = torture_recipient
	
					POD_torture_interaction_actor_effect = yes
					POD_torturer_interaction_recipient_effect = yes
					POD_house_feud_torture_event_effect = yes
	
					prev = {
						POD_secondary_effects_from_torturing = yes
						
						if = {
							limit = { has_perk = dark_insights_perk }
							dark_insights_skill_gain_roll_effect = yes
						}
					}
				}
			}
		}
	}

	ai_frequency = 0
	
	auto_accept = yes
}
