POD_chi_arts_soul_bridge_interaction = {
	category = interaction_category_vampire
	icon = POD_art_interaction
	desc = POD_chi_arts_soul_bridge_interaction_desc

	cooldown = { years = 5 }

	is_shown = {
		scope:actor = { has_perk = chi_arts_soul_bridge_perk }
		NOT = { scope:actor = scope:recipient }
		POD_hostile_interaction_is_shown = yes
	}
	is_valid_showing_failures_only = {
		scope:actor = {
			POD_has_enough_chi_for_kj_art = { AMOUNT = 2 }
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	send_options_exclusive = no

	send_option = {
		is_shown = {
			scope:actor = {
				has_perk = chi_arts_voice_of_emperor_perk
			}
		}
		is_valid = {
			custom_description = {
				text = same_location
				scope:actor = { exists = location }
				scope:recipient = { exists = location }
				scope:actor = { location = scope:recipient.location }
			}
		}
		flag = force
		localization = POD_chi_arts_soul_bridge_interaction_force
	}

	on_accept = {
		scope:actor = {
			POD_willpower_rouse_check_effect = yes
			POD_kj_art_cost_effect = { COUNT = 2 }
			if = {
				limit = { scope:force ?= yes }
				duel = {
					skill = diplomacy
					target = scope:recipient
					10 = {
						compare_modifier = {
							value = scope:duel_value
							multiplier = high_positive_duel_skill_multiplier
						}
						if = {
							limit = {
								OR = {
									AND = {
										has_perk = chi_arts_authority_over_yin_soul_perk
										scope:recipient = { is_male = no }
									}
									AND = {
										has_perk = chi_arts_authority_over_yang_soul_perk
										scope:recipient = { is_male = yes }
									}
								}
							}
							add_hook = {
								type = strong_soul_bridge_hook
								target = scope:recipient
								years = 10
							}
						}
						else = {
							add_hook = {
								type = weak_soul_bridge_hook
								target = scope:recipient
								years = 10
							}
						}
					}
					10 = {
						compare_modifier = {
							value = scope:duel_value
							multiplier = low_negative_duel_skill_multiplier
						}
					}
				}
			}
			else = {
				if = {
					limit = {
						OR = {
							AND = {
								has_perk = chi_arts_authority_over_yin_soul_perk
								scope:recipient = { is_male = no }
							}
							AND = {
								has_perk = chi_arts_authority_over_yang_soul_perk
								scope:recipient = { is_male = yes }
							}
						}
					}
					add_hook = {
						type = strong_soul_bridge_hook
						target = scope:recipient
						years = 10
					}
				}
				else = {
					add_hook = {
						type = weak_soul_bridge_hook
						target = scope:recipient
						years = 10
					}
				}
			}
		}
		scope:recipient = {
			if = {
				limit = {
					NOT = { scope:actor = { has_perk = chi_arts_soul_shackles_perk } }
				}
				add_hook = {
					type = weak_soul_bridge_hook
					target = scope:actor
					years = 10
				}
			}
		}
	}
	
	auto_accept = {
		scope:force ?= yes
	}

	ai_accept = {
		base = -75
		opinion_modifier = { # More likely to accept if Recipient likes the Actor
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.25
			min = 0
			desc = AI_OPINION_REASON
		}
		modifier = {
			add = 20
			scope:recipient = {
				has_trait = trusting
			}
			desc = INTERACTION_TRUSTING
		}
		# Rivalry
		modifier = { # Say no to rivals
			add = -100
			scope:recipient = {
				has_relation_rival = scope:actor
				NOT = { has_relation_nemesis = scope:actor }
			}
			desc = "ACTOR_RIVAL_TO_ME_REASON"
		}
		modifier = { # Say no to nemesis
			add = -300
			scope:recipient = {
				has_relation_nemesis = scope:actor
			}
			desc = "ACTOR_NEMESIS_TO_ME_REASON"
		}
	}
}
