			}
		}
		flag = drought
		localization = pod_weather_drought
	}
	send_option = {
		is_shown = {
			NOR = {
				scope:storm = yes
				scope:drought = yes
			}
		}
		flag = fertile_land
		localization = pod_weather_fertile_land
	}
	send_option = {
		is_shown = {
			NOR = {
				scope:fertile_land = yes
				scope:drought = yes
			}
		}
		is_valid = { scope:actor = { has_perk = shintai_chamber_bed_of_clouds_perk } }
		flag = storm
		localization = pod_weather_storm
	}

	on_accept = {
		switch = {
			trigger = yes
			scope:drought			= { save_scope_value_as = { name = weather_type   value = flag:pod_weather_drought } }
			scope:fertile_land		= { save_scope_value_as = { name = weather_type   value = flag:pod_weather_fertile_land } }
			scope:storm				= { save_scope_value_as = { name = weather_type   value = flag:pod_weather_storm } }
		}
		
		if = {
			limit = { POD_shintai_arms_of_the_sea_interaction_cost = 0 }
			custom_tooltip = POD_hekau_weather_magic_interaction_sellect_size # reused loc
		}
		else_if = {
			limit = { NOT = { exists = scope:weather_type } }
			custom_tooltip = POD_hekau_weather_magic_interaction_sellect_weather # reused loc
		}
		else = {
			scope:actor = {
				POD_kj_art_cost_effect = { COUNT = POD_shintai_arms_of_the_sea_interaction_cost }
				
				switch = {
					trigger = yes
					scope:county	?= { POD_setup_weather_magic_counties_from_title = { TITLE = scope:recipient.location.county } }
					scope:duchy		?= { POD_setup_weather_magic_counties_from_title = { TITLE = scope:recipient.location.duchy } }
					scope:kingdom	?= { POD_setup_weather_magic_counties_from_title = { TITLE = scope:recipient.location.kingdom } }
					scope:empire	?= { POD_setup_weather_magic_counties_from_title = { TITLE = scope:recipient.location.empire } }
				}
				
				POD_weather_magic_effect = {
					MONTHS = POD_weather_default_duration
				}
			}
		}
	}
