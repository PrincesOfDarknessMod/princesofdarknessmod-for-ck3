POD_demon_arts_infernal_might_interaction = {
	category = interaction_category_vampire
	icon = POD_art_interaction
	desc = POD_demon_arts_infernal_might_interaction_desc

	cooldown = { years = 10 }

	common_interaction = yes

	is_shown = {
		scope:actor = { has_perk = demon_arts_infernal_might_perk }
		scope:actor = scope:recipient
	}
	is_valid_showing_failures_only = {
		POD_self_targeted_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:actor = {
			add_character_modifier = {
				modifier = demon_arts_infernal_might_modifier
				years = 10
			}
			POD_change_dharma_xp_effect = { VALUE = -10 }
		}
	}

	auto_accept = yes
}
