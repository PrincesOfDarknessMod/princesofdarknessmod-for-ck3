POD_resist_shadow_interaction = {
    category = interaction_category_wraith
	icon = POD_wraith_interaction
	desc = POD_resist_shadow_interaction_desc

	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { has_trait = wraith }
		scope:actor = { has_trait = shadow NOT = { has_trait = spectre } }
	}

	on_accept = {
		scope:actor = { POD_static_stress_gain = { COUNT  = 50 } }
		scope:actor = {
            duel = {
			    skills = { diplomacy martial stewardship intrigue learning }
				value = 10
                25 = { # CRITICAL SUCCESS
					compare_modifier = {
						value = scope:duel_value
						multiplier = 10
					}
					modifier = {
						has_perk = coax_perk
						add = 25
					}
					modifier = {
						has_perk = defiance_perk
						add = 25
					}
					modifier = {
						has_perk = thorn_whispers_perk
						add = -10
					}
					modifier = {
						has_perk = shadow_familiar_perk
						add = -10
					}
					modifier = {
						has_perk = shadow_traits_perk
						add = -5
					}
                    desc = POD_shared_loca_critical_success
                    send_interface_toast = {
                        title = POD_shared_loca_critical_success
                        POD_end_shadow = yes
                    }
                }
                25 = { # SUCCESS
					compare_modifier = {
						value = scope:duel_value
						multiplier = 5
					}
					modifier = {
						has_perk = coax_perk
						add = 12.5
					}
					modifier = {
						has_perk = defiance_perk
						add = 12.5
					}
					modifier = {
						has_perk = thorn_whispers_perk
						add = -5
					}
					modifier = {
						has_perk = shadow_familiar_perk
						add = -5
					}
					modifier = {
						has_perk = shadow_traits_perk
						add = -2.5
					}
                    desc = POD_shared_loca_success
                    send_interface_toast = {
                        title = POD_shared_loca_success
                        POD_end_shadow = yes
                        POD_static_stress_gain = { COUNT  = 50 }
                    }
                }
                25 = { # FAILURE
					compare_modifier = {
						value = scope:duel_value
						multiplier = -5
					}
					modifier = {
						has_perk = coax_perk
						add = -12.5
					}
					modifier = {
						has_perk = defiance_perk
						add = -12.5
					}
					modifier = {
						has_perk = thorn_whispers_perk
						add = 5
					}
					modifier = {
						has_perk = shadow_familiar_perk
						add = 5
					}
					modifier = {
						has_perk = shadow_traits_perk
						add = 2.5
					}
                    desc = POD_shared_loca_failure
                    send_interface_toast = {
                        title = POD_shared_loca_failure
                    }
                }
                25 = { # SCRITICAL FAILURE
					compare_modifier = {
						value = scope:duel_value
						multiplier = -10
					}
					modifier = {
						has_perk = coax_perk
						add = -25
					}
					modifier = {
						has_perk = defiance_perk
						add = -25
					}
					modifier = {
						has_perk = thorn_whispers_perk
						add = 10
					}
					modifier = {
						has_perk = shadow_familiar_perk
						add = 10
					}
					modifier = {
						has_perk = shadow_traits_perk
						add = 5
					}
                    desc = POD_shared_loca_critical_failure
                    send_interface_toast = {
                        title = POD_shared_loca_critical_failure
                        POD_static_stress_gain = { COUNT  = 50 }
                    }
                }
            }
		}
    }
	
	auto_accept = yes
	
	ai_frequency = 24
	
	ai_will_do = {
		base = 100
	} 
	ai_potential = {
		is_imprisoned = no
		is_landed = yes
	}
	ai_targets = {
		ai_recipients = self
	}
}
POD_pact_of_doom_interaction = {
	category = interaction_category_wraith
	icon = POD_wraith_interaction
	desc = POD_pact_of_doom_interaction_desc
	
	cooldown = { years = 5 }	

	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { has_trait = wraith }
		scope:actor = { has_perk = pact_of_doom_perk }
	}

	is_valid_showing_failures_only = {
		POD_cross_umbra_interaction_valid_trigger = yes
		scope:recipient = { NOT = { has_trait = shadow } }
	}

	on_accept = {
		scope:actor = { 
			POD_wraith_willpower_check_effect = yes 
			POD_assign_random_thorn_perk = yes 
			add_arcanoi_lifestyle_perk_points = 2 
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
