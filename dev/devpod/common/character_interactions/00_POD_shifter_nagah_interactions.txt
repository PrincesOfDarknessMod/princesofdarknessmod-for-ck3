
POD_long_strike_interaction = {
	category = interaction_category_werewolf
	#interface_priority = 90
	desc = POD_long_strike_interaction_desc
	icon = POD_nagah_interaction
	
	cooldown = { months = 9 }	
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_perk = long_strike_perk }
		scope:actor = { POD_splat_has_rage_trigger = yes }
	}
	
	is_valid = {
		scope:actor = {
			OR = {
				has_character_modifier = POD_nagah_form_azhi_dahaka_modifier
				has_character_modifier = POD_nagah_form_silkaram_modifier
				has_character_modifier = POD_nagah_form_kali_dahaka_modifier
				has_character_modifier = POD_nagah_form_vasuki_modifier
			}
		}
	}
	
	is_valid_showing_failures_only = {
		scope:actor = { NOT = { has_character_modifier = POD_rage_0_modifier } }
		scope:actor = {
			prowess >= POD_prowess_low_skill_rating
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}

	on_accept = {	
		scope:actor = { POD_willpower_rouse_check_effect = yes }
		scope:recipient = { 
			add_opinion = {
				target = scope:actor
				modifier = hurt_opinion
				opinion = -25
			}
		}
		scope:actor = { 
			duel = {
				skill = prowess
				target = scope:recipient
				desc = POD_distant_attack_interaction.duel.desc
				10 = {
					trigger = {
						prowess >= POD_prowess_extremely_high_skill_rating
					}
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					add_piety = -1000
					POD_rage_loss_check_effect = yes
					POD_rage_loss_check_effect = yes
					POD_rage_loss_check_effect = yes
					POD_rage_loss_check_effect = yes
					POD_rage_loss_check_effect = yes
					scope:recipient = { increase_wounds_multiple_hostile_effect = { REASON = fight AMOUNT = 4 ATTACKER = scope:actor } }
				}
				20 = {
					trigger = {
						prowess >= POD_prowess_high_skill_rating
					}
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					add_piety = -500
					POD_rage_loss_check_effect = yes
					POD_rage_loss_check_effect = yes
					POD_rage_loss_check_effect = yes
					scope:recipient = { increase_wounds_multiple_hostile_effect = { REASON = fight AMOUNT = 3 ATTACKER = scope:actor } }
				}
				30 = {
					trigger = {
						prowess >= POD_prowess_medium_skill_rating
					}
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					add_piety = -300
					POD_rage_loss_check_effect = yes
					POD_rage_loss_check_effect = yes
					scope:recipient = { increase_wounds_multiple_hostile_effect = { REASON = fight AMOUNT = 2 ATTACKER = scope:actor } }
				}
				40 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier 
					}
					add_piety = -100
					POD_rage_loss_check_effect = yes
					scope:recipient = { increase_wounds_hostile_effect = { REASON = fight ATTACKER = scope:actor } }
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no 
		is_landed = yes 
		
		piety > 1000
	}
	ai_targets = {
		ai_recipients = primary_war_enemies
	}
	ai_frequency = 48
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 45
			scope:actor = {
			    is_at_war_with = scope:recipient
			}
		}
		modifier = {
		    factor = 2
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
		modifier = {
			factor = 0
			scope:recipient = {
				is_ai = no
			}
		}
	}
}

POD_dance_of_shiva_interaction = {
	category = interaction_category_werewolf
	icon = POD_nagah_interaction
	desc = POD_dance_of_shiva_interaction_desc
	
	target_type = title
	target_filter = recipient_domain_titles
	
	can_be_picked = {
		scope:target = { 
			tier >= tier_county
			is_titular = no
			any_de_jure_county = {  
				development_level >= 1
			}
		}
	}

	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { POD_splat_has_rage_trigger = yes }
		scope:actor = {
			has_perk = dance_of_shiva_perk
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = { NOT = { has_character_modifier = POD_gnosis_0_modifier } }
		scope:actor = {
			custom_description = { 
				text = POD_infernalist_sacrifice_region_interaction_valid 
				any_held_title = {
					tier >= tier_county
					is_titular = no
					any_de_jure_county = {  
						development_level >= 1
					}
				}
			}
		}
	}

	on_accept = {
		scope:actor = {
			POD_gnosis_loss_check_effect = yes
			POD_gnosis_loss_check_effect = yes
			POD_gnosis_loss_check_effect = yes
			POD_gnosis_loss_check_effect = yes
			POD_gnosis_loss_check_effect = yes
			add_prestige = POD_sacrifice_region_investiture_xp_gain
			custom_tooltip = POD_dance_of_shiva_interaction_desc
			scope:target = {
				change_development_level = -100
				hidden_effect = { 
					every_de_jure_county = {
						every_county_province = {
							if = {
								limit = { 
									exists = province_owner
									province_owner = { primary_title.tier = tier_barony }
								}
								province_owner = { 
									pod_death_no_killer = {
										DEATH_REASON = death_fire
									}
								}
							}
						}
					}
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_frequency = 0
	
	ai_will_do = {
		base = 0
	}
}
