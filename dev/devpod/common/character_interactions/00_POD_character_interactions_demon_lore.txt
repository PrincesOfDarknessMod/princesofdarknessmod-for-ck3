POD_train_lore_interaction = {
	category = interaction_category_demon
	icon = POD_lore_interaction
    desc = POD_train_lore_interaction_desc
	common_interaction = yes
	
	cooldown = { months = 12 }
	cost = { piety = medium_piety_value }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { 
			has_trait = demon
		}
		scope:recipient = { 
			has_trait = demon
		}
	}

	is_valid_showing_failures_only = {
	    scope:actor = {
			lore_lifestyle_perk_points >= 1
		}
		scope:recipient = { is_imprisoned = no }
		scope:recipient = {	is_available = yes }
	}
	
	on_accept = {
	    scope:actor = { 
			add_lore_lifestyle_perk_points = -1
			stress_impact = {
				base = medium_stress_gain
			}
		}			
	    scope:recipient = { 
			add_lore_lifestyle_perk_points = 1 
			add_opinion = {
				target = scope:actor
				modifier = mentored_me_opinion
			}
		}	
	}

	ai_frequency = 0
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_break_alliance_interaction = {
	category = interaction_category_demon
	icon = POD_lore_interaction
    desc = POD_break_alliance_interaction_desc
	
	cooldown = { months = 6 }
	cost = { piety = medium_piety_value }

	is_shown = {
		scope:actor = { 
			has_trait = demon 
			has_perk = fade_perk
		}
		scope:recipient = {
			is_ruler = yes
			any_ally = { count > 0 }
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = { 
			trigger_if = { 
				limit = { NOT = { has_perk = soul_burn_perk } }
				NOT = { has_character_modifier = resolve0_modifier } 
			}
		}
	}

	populate_actor_list = {
		scope:recipient = {
			every_ally = {
				add_to_list = characters
			}
		}
	}
	
	on_accept = {
		scope:actor = { 
			POD_willpower_rouse_check_effect = yes
			POD_resolve_decrease_multiple_effect = { COUNT = 5 } 
		}	
		scope:secondary_actor = { 
			break_alliance = scope:recipient
		}
	}

	ai_frequency = 0
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
