POD_create_gargoyle_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	##interface_priority = 49
    desc = POD_create_gargoyle_interaction_desc
	ai_frequency = 0
	
	ai_potential = { always = no }
	send_options_exclusive = no
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }	
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		scope:actor = { is_vampire_trigger = yes }
		OR = {
			AND = {
				scope:actor = { is_tremere_trigger = yes }
				scope:actor = { has_perk = bloodsorceryadvanced_perk }
				scope:actor = { dynasty = { has_dynasty_perk = tremere_legacy_1 } }
			}
			AND = {
				scope:actor = { dynasty = { has_dynasty_perk = gargoyle_legacy_4 } }	
				scope:actor = { is_gargoyle_trigger = yes }		
				scope:actor = { 
					OR = {
						has_trait = bloodsorcery
						has_perk = bloodsorceryadvanced_perk
					}
				}
			}
		}
		scope:recipient = { 
			OR = {
				POD_is_vampire_trigger = yes
				POD_is_werewolf_trigger = yes	
				is_kueijin_trigger = yes
			}
		}	
	}

	is_valid_showing_failures_only = {
		scope:recipient = { is_imprisoned_by = scope:actor }
	}
	
	populate_recipient_list = {
		scope:actor = {
			every_prisoner = {
				limit = {
					OR = {
						POD_is_vampire_trigger = yes
						POD_is_werewolf_trigger = yes
						is_kueijin_trigger = yes
					}
					is_imprisoned_by = scope:actor
					NOT = { this = scope:recipient }
				}
				add_to_list = characters
			}
			# every_in_list = {
				# limit = {
					# is_alive = yes
					# NOT = { this = scope:actor }
					# NOT = { this = scope:recipient }
				# }
				# variable = potential_experimental_subject
				# add_to_list = characters
			# }			
		}
	}	

	on_accept = {	
		
		scope:actor = {
			save_scope_as = gargoyle_experiment_thaumaturgist
		}
		
		scope:recipient = {
			save_scope_as = gargoyle_experiment_victim_one
		}
		
		scope:secondary_recipient = {
			save_scope_as = gargoyle_experiment_victim_two
		}
		
#Costs
		
		scope:actor = { 
			custom_tooltip = POD_create_gargoyle_interaction.gargoyle_experiment_chance.tt
			remove_short_term_gold = minor_gold_value 
			POD_willpower_rouse_check_effect = yes
			POD_moderate_risk_of_increasing_masquerade_exposure = yes
			
			if = {
				limit = {
					faith = {
						OR = {
							has_doctrine_parameter = vampirism_illegal
							has_doctrine_parameter = vampirism_shunned
							has_doctrine_parameter = hurt_mortal_lose_piety
						}
					}
				}
				add_piety = massive_piety_loss
			}

			if = {
				limit = {
					faith = { has_doctrine_parameter = hurt_mortal_lose_piety }
					scope:recipient = { has_trait = orison_innocent }
				}
				scope:actor = { add_piety = -500 }
			}

			if = {
				limit = {
					faith = { has_doctrine_parameter = hurt_mortal_lose_piety }
					scope:secondary_recipient = { has_trait = orison_innocent }
				}
				scope:actor = { add_piety = -500 }
			}

			stress_impact = { compassionate = major_stress_impact_gain }
		}
	
#Results	
		scope:actor = { 
			add_sanguine_lifestyle_xp = minor_lifestyle_xp
			
			POD_character_create_vampire_gargoyle_effect = yes
		}

		scope:recipient = { 
			pod_death = { 
				KILLER = scope:actor 
				DEATH_REASON = death_execution
			}
		}

		scope:secondary_recipient = { 
			pod_death = { 
				KILLER = scope:actor 
				DEATH_REASON = death_execution
			}
		}
		
	}
	
	auto_accept = yes
}
