# POD Personal Character Interactions
#  Blood Heal

POD_blood_heal_interaction = {
	category = interaction_category_vampire
	common_interaction = yes
	#interface_priority= 110
	desc = POD_blood_heal_interaction_desc
	options_heading = POD_blood_heal_interaction_options_heading	

	use_diplomatic_range = no

	cooldown = { months = 3 }	
	
	is_shown = {
		scope:actor = { has_POD_vamp_blood_trigger = yes }
		trigger_if = {
			limit = { NOT = { scope:actor = scope:recipient } }
			scope:actor = { has_perk = obeah_panacea_perk }
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			NOT = { has_character_modifier = bloodhunger5_modifier }
		}
		scope:recipient = {
			NOT = { has_trait_with_flag = can_not_cure }
			NOT = { has_character_modifier = eye_of_sakhmet_modifier }
			custom_description = {
				text = POD_has_main_trait
				object = scope:recipient
				OR = {
					has_trait = scarred
					has_trait = disfigured
					has_trait = blind
					has_trait = one_eyed
					has_trait = maimed
					has_trait = one_legged
					has_trait = eunuch
				
					has_trait = wounded_1
					has_trait = wounded_2
					has_trait = wounded_3
				}
			}
		}
	}

	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = blind
				POD_is_vampire_trigger = yes
				scope:actor = scope:recipient
			}
		}	
		is_valid = {
			custom_description = {
				text = POD_permanent_blind
				object = scope:recipient
				scope:recipient = { NOT = { has_character_flag = POD_permanent_blind } }
			}
		}
		flag = blind
		localization = POD_heal_blind
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = maimed
				POD_is_vampire_trigger = yes
				scope:actor = scope:recipient
			}
		}	
		is_valid = {
			custom_description = {
				text = POD_permanent_maimed
				object = scope:recipient
				scope:recipient = { NOT = { has_character_flag = POD_permanent_maimed } }
			}
		}
		flag = maimed
		localization = POD_heal_maimed
	}	
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = one_legged
				POD_is_vampire_trigger = yes
				scope:actor = scope:recipient
			}
		}	
		is_valid = {
			custom_description = {
				text = POD_permanent_one_legged
				object = scope:recipient
				scope:recipient = { NOT = { has_character_flag = POD_permanent_one_legged } }
			}
		}
		flag = one_legged
		localization = POD_heal_one_legged
	}

	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = one_eyed
				POD_is_vampire_trigger = yes
				scope:actor = scope:recipient
			}
		}	
		is_valid = {
			custom_description = {
				text = POD_permanent_one_eyed
				object = scope:recipient
				scope:recipient = { NOT = { has_character_flag = POD_permanent_one_eyed } }
			}
		}
		flag = one_eyed
		localization = POD_heal_one_eyed
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = eunuch
				POD_is_vampire_trigger = yes
				scope:actor = scope:recipient
			}
		}
		is_valid = {
			custom_description = {
				text = POD_permanent_eunuch
				object = scope:recipient
				scope:recipient = { NOT = { has_character_flag = POD_permanent_eunuch } }
			}
		}
		flag = eunuch
		localization = POD_heal_eunuch
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				OR = {
					has_trait = wounded_1
					has_trait = wounded_2
					has_trait = wounded_3
				}
			}
		}	
		is_valid = {
			trigger_if = {
				limit = { NOT = { scope:actor = scope:recipient } }
				scope:actor = { has_perk = healers_touch_perk }
			}
		}
		flag = wounded
		localization = POD_heal_wounded
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = scarred
			}
		}	
		is_valid = {
			scope:recipient = { 
				custom_description = {
					text = POD_permanent_scarred
					object = scope:recipient
					NOT = { has_character_flag = POD_permanent_scarring_flag }
				}
			}
		}
		flag = scarred
		localization = POD_heal_scarred
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = disfigured
			}
		}	
		is_valid = {
			scope:recipient = { 
				custom_description = {
					text = POD_permanent_disfigured
					object = scope:recipient
					NOT = { has_character_flag = POD_permanent_disfigured_flag }
				}
			}
		}
		flag = disfigured
		localization = POD_heal_disfigured
	}
	
	on_accept = {
		scope:actor = { POD_hunger_rouse_check_effect = yes }	
		if = {
			limit = {
				scope:blind = yes
			}
			scope:recipient = {
				remove_trait = blind
				add_trait = one_eyed
			}
			scope:actor = { POD_minimal_risk_of_increasing_masquerade_exposure = yes }
		}	

		if = {
			limit = {
				scope:maimed = yes
			}
			scope:recipient = {
				remove_trait = maimed
			}
			scope:actor = { POD_small_risk_of_increasing_masquerade_exposure = yes }
		}		

		if = {
			limit = {
				scope:one_legged = yes
			}
			scope:recipient = {
				remove_trait = one_legged
			}
			scope:actor = { POD_small_risk_of_increasing_masquerade_exposure = yes }
		}				

		if = {
			limit = {
				scope:disfigured = yes
			}
			scope:recipient = {
				remove_trait = disfigured
			}
			scope:actor = { POD_small_risk_of_increasing_masquerade_exposure = yes }
		}

		if = {
			limit = {
				scope:eunuch = yes
			}
			scope:recipient = {
				remove_trait = eunuch
			}
			scope:actor = { POD_minimal_risk_of_increasing_masquerade_exposure = yes }
		}

		if = {
			limit = {
				scope:one_eyed = yes
			}
			scope:recipient = {
				remove_trait = one_eyed
			}
			scope:actor = { POD_minimal_risk_of_increasing_masquerade_exposure = yes }
		}
		
		if = {
			limit = {
				scope:wounded = yes
			}
			if = {
				limit = {
					scope:actor = { has_perk = healers_touch_perk }
				}
				scope:recipient = {
					remove_trait = wounded_1
					remove_trait = wounded_2
					remove_trait = wounded_3
				}
			}
			else = {
				scope:recipient = {
					POD_blood_heal_wound_effect = yes
				}
			}
			scope:actor = { POD_minimal_risk_of_increasing_masquerade_exposure = yes }
		}

		if = {
			limit = {
				scope:scarred = yes
			}
			scope:recipient = {
				remove_trait = scarred
			}
			scope:actor = { POD_minimal_risk_of_increasing_masquerade_exposure = yes }
		}
	}
	
	auto_accept = yes
	ai_targets = {
		ai_recipients = self
	}
	ai_frequency = 12
	ai_will_do = {
		base = 100
		
	}
}
