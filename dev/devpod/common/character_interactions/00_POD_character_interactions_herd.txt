		hidden_effect = {
			scope:victim = {
				POD_set_resonance_effect = yes
			}
		}

		scope:actor = { save_scope_as = vampire }
		scope:recipient = { save_scope_as = victim }
		scope:actor = { save_scope_as = herd_owner }
		scope:recipient = { save_scope_as = herdmember }

			limit = {scope:sip = yes}
			
			#If the the player is a Ventrue then we need to make sure to check for their bane when feeding on mortals.

			#Run the effect that handles one step of manual hunger slaking
			POD_herd_feed_manual_slake1_effect = yes
		}

		if = {

			POD_herd_feed_manual_drain_effect = yes
		}

			limit = {
				scope:modify = yes 
			}

			scope:actor = { add_stress = minor_stress_impact_gain }	
			scope:victim = { add_stress = medium_stress_impact_gain }
		}
		
		if = {
			limit = {
				scope:choleric = yes

				chance = 60 
				scope:victim = { custom_tooltip = POD_drink_herd_change_to_choleric.tt }
				scope:victim = { POD_resonance_change_to_choleric_effect = yes }	
				modifier = {
					add = 35
					scope:actor = {
						stewardship >= 30
					}						
				}
				modifier = {
					add = 15
					AND = {
						scope:actor = {
							stewardship >= 15
						}	
						scope:actor = {
							stewardship < 30
						}	
					}
				}
			}
		}								

		if = {
			limit = {
				scope:melancholic = yes 
			}
			random = {
				chance = 60 
				scope:victim = { custom_tooltip = POD_drink_herd_change_to_melancholic.tt }
				scope:victim = { POD_resonance_change_to_melancholic_effect = yes }	
				modifier = {
					add = 35
					scope:actor = {
						stewardship >= 30
					}						
				}
				modifier = {
					add = 15
					AND = {
						scope:actor = {
							stewardship >= 15
						}	
						scope:actor = {
							stewardship < 30
						}	
					}
				}
			}
		}

		if = {
			limit = {
				scope:phlegmatic = yes 
			}
			random = {
				chance = 60 
				scope:victim = { custom_tooltip = POD_drink_herd_change_to_phlegmatic.tt }

						scope:actor = {
							stewardship >= 15
						}	
						scope:actor = {
							stewardship < 30
						}	
					}
				}
			}
		}			
		
		if = {
			limit = {
				scope:sanguine = yes 
			}
			random = {
				chance = 60 
				scope:victim = { custom_tooltip = POD_drink_herd_change_to_sanguine.tt }

					add = 35
					scope:actor = {
						stewardship >= 30
					}						
				}
				modifier = {
					add = 15
					AND = {
						scope:actor = {
							stewardship >= 15
						}	
						scope:actor = {
							stewardship < 30
						}	
					}
				}
			}	
		}		

		if = {
			limit = {
				scope:increase = yes 
			}
			random_list = {
				50 = {
					scope:victim = { POD_resonance_increase_intensity_effect = yes }
					modifier = {
						add = 55
						scope:actor = {
							stewardship >= 30
						}						
					}
					modifier = {
						add = 40
						AND = {
							scope:actor = {
								stewardship >= 15
							}	
							scope:actor = {
								stewardship < 30
							}	
						}
					}
				}
				50 = {
					scope:victim = { POD_resonance_decrease_effect = yes }
					modifier = {
						add = -20
						scope:actor = {
							stewardship >= 15
						}
					}
					
					modifier = {
						add = 20
						scope:actor = {
							stewardship <= 5
						}
					}
				}
			}
		}
		
		if = {
			limit = {
				scope:release = yes 
			}
			
			scope:actor = {	
				POD_minimal_risk_of_increasing_masquerade_exposure = yes
				remove_hook = {
					target = scope:victim
					type = herd_hook
				}
			}
			
			scope:victim = {
				remove_relation_herd_owner = scope:actor
				# remove_opinion = {
					# target = scope:actor
					# modifier = POD_herd_opinion
				# }
				remove_character_flag = blocked_from_leaving
			}
			
			scope:victim = { custom_tooltip = POD_drink_herd_release.tt }
		}
	}
	auto_accept = yes
}

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_adult = yes }

		scope:actor = { is_vampire_trigger = yes  }
		scope:recipient = { 
			POD_can_be_herd_trigger = yes
		}
		scope:actor = { is_ai = no }
		scope:recipient = {
			NOT = {
				has_relation_herd_owner = scope:actor
				# has_opinion_modifier ={
					# target = scope:actor
					# modifier = POD_herd_opinion
				# }
			}
		}
	}

	is_valid_showing_failures_only= {
		custom_description = {
            object = scope:actor
            text = "I_must_be_at_the_same_location"
			scope:actor = { exists = location }
			scope:recipient = { exists = location }
            scope:actor = { location = scope:recipient.location }
        }
		scope:actor = {
			can_start_scheme = {
				type = herdify_mortal
				target = scope:recipient
			}
		}
		custom_description = {
			text = POD_herdify_interaction_limit
			scope:actor = {
				num_of_relation_herd_member < max_herd_cap
			}
		}
	}

	desc = POD_herdify_mortal_interaction_desc

	on_accept = {
	
		hidden_effect = {
			scope:actor = {
				start_scheme = {
					type = herdify_mortal
					target = scope:recipient
				}
			}
		}
		
		if = {
			limit = {
				OR = {
					scope:actor = {has_relation_best_friend = scope:recipient}
					scope:actor = {has_relation_friend = scope:recipient}
					scope:actor = {has_relation_lover = scope:recipient}
				}
			}
			scope:actor = {
				stress_impact = {
					compassionate = medium_stress_impact_gain
				}
			}
		}
		
		if = {
			limit = {
					scope:actor = {
						faith = {has_doctrine_parameter = vampirism_illegal }
					}
			}
			scope:actor = {add_piety = major_piety_loss}
		}
		
		else_if = {
			limit = {
				scope:actor = {
					faith = {has_doctrine_parameter = vampirism_shunned} 
				}
			}
			scope:actor = {add_piety = medium_piety_loss}
		}
	}
	
	#ai_potential = {
	#	always = no
	#}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
