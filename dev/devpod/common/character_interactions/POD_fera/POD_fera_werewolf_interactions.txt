						title = POD_divide_interaction_success

					compare_modifier = { value = scope:duel_value multiplier = low_negative_duel_skill_multiplier }

						title = POD_shared_loca_failure

		}
	}
	
	auto_accept = yes
	
	ai_frequency = 0

POD_rhythm_of_war_interaction = {
	category = interaction_category_fera
	icon = POD_werewolf_interaction
	desc = POD_rhythm_of_war_interaction_desc
	common_interaction = yes
	interface_priority = 90
	options_heading = POD_rhythm_of_war_interaction_options_heading
	
	cooldown = { years = 2 }

	is_shown = {
		scope:actor = { has_perk = rhythm_of_war_perk }
		scope:actor != scope:recipient
		POD_hostile_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = { is_travelling = no }
		scope:recipient = { prowess >= scope:actor.POD_ahroun_target_prowess_min }
		scope:recipient = { is_imprisoned = no }
		POD_interaction_is_valid_showing_failures_only = yes
	}

	send_option = {
		is_valid = { always = yes }
		flag = lethal
		localization = POD_rhythm_of_war_interaction_lethal
	}
	send_option = {
		is_valid = { 
			scope:recipient = {
				has_trait = werewolf
				trigger_if = {
					limit = { scope:actor.religion = { is_in_family = rf_gaia } }
					religion = { is_in_family = rf_gaia }
				}
				trigger_else = {
					religion = { is_in_family = rf_wyrm }
				}
			}
		}
		flag = nonlethal
		localization = POD_rhythm_of_war_interaction_nonlethal
	}



	can_send = {
		custom_description = {
			text = send_option_selected
			OR = {
				scope:lethal ?= yes
				scope:nonlethal ?= yes
			}
		}
		scope:actor = { prestige >= scope:recipient.POD_ahroun_interaction_prestige_cost }
	}

	on_accept = {
		scope:actor = {
			save_scope_value_as = { name = prestige_cost value = scope:recipient.POD_ahroun_interaction_prestige_cost } #scope value so refund works in planner
			add_prestige = { value = scope:prestige_cost multiply = -1 }
			switch = {
				trigger = yes

				scope:lethal = { save_scope_value_as = { name = lethality value = yes } }
				scope:nonlethal = { save_scope_value_as = { name = lethality value = no } }
			}
			if = {
				limit = { scope:recipient.location = scope:actor.location }
				trigger_event = POD_werewolves.100
			}
			else = {
				start_travel_plan = {
					destination = scope:recipient.location
					on_arrival_event = POD_werewolves.100
					on_travel_planner_cancel_event = POD_werewolves.101
					return_trip = yes
					on_arrival_destinations = first
					players_use_planner = yes
				}
			}
		}
	}

	send_options_exclusive = yes

	auto_accept = yes
	ai_frequency = 0
	ai_will_do = { base = 0 }
}

	desc = POD_dark_of_night_interaction_desc
	icon = POD_werewolf_interaction
	
	cooldown = { years = 5 }
	cost = { prestige = 200 }
	
	is_shown = {
		scope:actor = { has_perk = dark_of_night_perk }
		scope:actor != scope:recipient
		POD_hostile_interaction_is_shown = yes
	}
	
	is_valid_showing_failures_only = {
		scope:actor = { POD_has_enough_gnosis_for_gift = { COUNT = 2 } }
		scope:recipient = { NOT = { is_at_war_with = scope:recipient } }
		trigger_if = {
			limit = { scope:recipient = { is_knight = yes } }
			scope:recipient = { liege_or_court_owner = { NOT = { is_at_war_with = scope:actor } } }
		}
		custom_tooltip = {
			text = target_in_duel_tooltip
			scope:recipient = { NOT = { has_variable = engaged_in_single_combat } }
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:actor = {
			POD_willpower_rouse_check_effect = yes
			POD_gift_cost_effect = { COUNT = 2 }
			
			duel = {
				skills = { martial prowess } 
				target = scope:recipient
			
				30 = {
					desc = POD_dark_of_night_interaction_success
					compare_modifier = { value = scope:duel_value multiplier = high_positive_duel_skill_multiplier }

					scope:recipient = { add_character_modifier = { modifier = dark_of_night_perk_modifier years = 5 } }
					
					hidden_effect = {
						send_interface_toast = {
							title = POD_dark_of_night_interaction_success
							left_icon = scope:actor
							right_icon = scope:recipient
						}
					}
				}
				
				70 = {
					compare_modifier = { value = scope:duel_value multiplier = low_negative_duel_skill_multiplier }
					hidden_effect = {
						send_interface_toast = {
							title = POD_shared_loca_failure
							left_icon = scope:actor
							right_icon = scope:recipient
						}
					}
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_frequency = 0
	ai_will_do = { base = 0 }
}
