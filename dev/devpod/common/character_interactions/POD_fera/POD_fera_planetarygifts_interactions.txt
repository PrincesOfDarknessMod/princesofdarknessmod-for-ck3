POD_ebon_binding_interaction = {
	category = interaction_category_fera
	desc = POD_ebon_binding_interaction_desc
	icon = POD_planetarygifts_interaction
	
	cooldown = { years = 1 }

	is_shown = {
		scope:actor = { has_perk = planetarygifts_ebon_binding_perk }
		NOT = { scope:actor = scope:recipient }
		POD_hostile_interaction_is_shown = yes
	}

	is_valid = {
		custom_tooltip = {
			text = any_secret_is_known_by_actor
			scope:recipient = {
				any_secret = {
					is_known_by = scope:actor
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = { POD_has_enough_gnosis_for_gift = { COUNT = 1 } }
		scope:recipient = {
			NOT = { has_character_modifier = ebon_binding_modifier }
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}
	on_accept = {
		scope:actor = {
			POD_gift_cost_effect = { COUNT = 1 }
		}
		scope:recipient = {
			add_character_modifier  = {
				modifier = ebon_binding_modifier
				years = 10
			}
		}
		scope:recipient = {
			add_opinion = {
				target = scope:actor
				modifier = hurt_opinion
				opinion = -20
			}
		}
	}
	
	auto_accept = yes
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no
		is_landed = yes
	}
	ai_targets = {
		ai_recipients = liege
	}
	ai_targets = {
		ai_recipients = neighboring_rulers
		ai_recipients = peer_vassals
		max = 10
	}
	ai_frequency = 60
	ai_will_do = {
		base = 0
		
		opinion_modifier = {
			opinion_target = scope:recipient
			multiplier = -0.25
		}
		modifier = {
		    add = 50
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
		modifier = {
			factor = 0
			scope:recipient = {
				is_ai = no
			}
		}
	}
}

POD_earth_heal_interaction = {
	category = interaction_category_fera
	desc = POD_earth_heal_interaction_desc
	icon = POD_planetarygifts_interaction
	
	cooldown = { months = 3 }

	is_shown = {
		scope:actor = { has_perk = earth_heal_perk }
		NOT = { scope:actor = scope:recipient }
		scope:recipient = {
			exists = location
			location = { has_building_or_higher = wyrm_taint_01 }
		}
		POD_hostile_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			POD_has_enough_gnosis_for_gift = {
				COUNT = POD_earth_heal_interaction_cost
			}
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:recipient.location = {
			save_scope_as = taint_land
			province_owner = {
				if = {
					limit = { faith = { has_doctrine = doctrine_fera_triat_alignment_wyrm } }
					add_opinion = {
						target = scope:actor
						modifier = POD_casted_harmful_weather_magic
					}
				}
			}
		}
		scope:actor = {
			POD_gift_cost_effect = { COUNT = POD_earth_heal_interaction_cost }
			
			
			if = {
				limit = { scope:taint_land = { has_building = wyrm_taint_03 } }
				duel = {
					skill = learning
					value = very_high_skill_rating
					50 = {
						compare_modifier = {
							value = scope:duel_value
							multiplier = high_positive_duel_skill_multiplier
						}
						add_prestige = major_prestige_gain
						scope:taint_land = { remove_building = wyrm_taint_03 }
					}
					50 = {
						compare_modifier = {
							value = scope:duel_value
							multiplier = low_negative_duel_skill_multiplier
						}
					}
				}
			}
			else_if = {
				limit = { scope:taint_land = { has_building = wyrm_taint_02 } }
				duel = {
					skill = learning
					value = high_skill_rating
					50 = {
						compare_modifier = {
							value = scope:duel_value
							multiplier = high_positive_duel_skill_multiplier
						}
						add_prestige = medium_prestige_gain
						scope:taint_land = { remove_building = wyrm_taint_02 }
					}
					50 = {
						compare_modifier = {
							value = scope:duel_value
							multiplier = low_negative_duel_skill_multiplier
						}
					}
				}
			}
			else = {
				duel = {
					skill = learning
					value = decent_skill_rating
					50 = {
						compare_modifier = {
							value = scope:duel_value
							multiplier = high_positive_duel_skill_multiplier
						}
						add_prestige = minor_prestige_gain
						scope:taint_land = { remove_building = wyrm_taint_01 }
					}
					50 = {
						compare_modifier = {
							value = scope:duel_value
							multiplier = low_negative_duel_skill_multiplier
						}
					}
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
