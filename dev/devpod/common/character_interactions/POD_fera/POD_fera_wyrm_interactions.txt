



POD_dancing_shadows_interaction = {
	category = interaction_category_fera
	icon = POD_wyrm_interaction
	desc = POD_dancing_shadows_interaction_desc

	cooldown = { years = 5 }
	
	is_shown = {
		scope:actor = { has_perk = dancing_shadows_perk }
		scope:actor != scope:recipient
		POD_hostile_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = { POD_has_enough_gnosis_for_gift = { COUNT = 1 } }
		POD_interaction_is_valid_showing_failures_only = yes
	}
	
	on_accept = {
		scope:actor = {
			POD_willpower_rouse_check_effect = yes
			POD_gift_cost_effect = { COUNT = 1 }

			duel = {
				skill = prowess
				target = scope:recipient
			
				10 = {
					compare_modifier = { value = scope:duel_value multiplier = high_positive_duel_skill_multiplier }

					scope:recipient = { POD_increase_wounds_hostile_effect = { ATTACKER = scope:actor COUNT = POD_wound_triple_damage REASON = fight } }
				}
				20 = {
					compare_modifier = { value = scope:duel_value multiplier = low_positive_duel_skill_multiplier }
					
					scope:recipient = { POD_increase_wounds_hostile_effect = { ATTACKER = scope:actor COUNT = POD_wound_double_damage REASON = fight } }
				}
				30 = {
					
					scope:recipient = { POD_increase_wounds_hostile_effect = { ATTACKER = scope:actor COUNT = POD_wound_base_damage REASON = fight } }
				}
				40 = {
					compare_modifier = { value = scope:duel_value multiplier = low_negative_duel_skill_multiplier }
					send_interface_toast = {
						title = POD_shared_loca_failure
						left_icon = scope:actor
						right_icon = scope:recipient
					}
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_frequency = 0
	ai_will_do = { base = 0 }
}

POD_splintered_soul_interaction = {
	category = interaction_category_fera
	icon = POD_wyrm_interaction
	desc = POD_splintered_soul_interaction_desc

	cooldown = { years = 5 }
	
	is_shown = {
		scope:actor = {
			has_perk = splintered_soul_perk
			NOT = {
				any_relation = {
					type = spirit_companion
					has_character_flag = splintered_soul_ancestor
				}
			}
		}
		scope:actor = scope:recipient
	}

	is_valid_showing_failures_only = {
		scope:actor = { POD_has_enough_gnosis_for_gift = { COUNT = 5 } }
		scope:actor = { has_trait = fera }
		POD_self_targeted_interaction_is_valid_showing_failures_only = yes
	}
	
	on_accept = {
		scope:actor = {
			POD_willpower_rouse_check_effect = yes
			POD_gift_cost_effect = { COUNT = 5 }
			add_character_modifier = { modifier = wyrm_splintered_soul_modifier years = 10 }
			custom_tooltip = POD_splintered_soul_interaction_effect

			hidden_effect = {
				save_scope_as = cloned_character
				create_character = {
					save_scope_as = soul_ancestor
					age = scope:cloned_character.age
					faith = scope:cloned_character.faith
					culture = scope:cloned_character.culture
					gender = scope:cloned_character
					random_traits = no
					dynasty = none
					employer = scope:cloned_character
				}
				if = {
					limit = {
						exists = scope:cloned_character
						exists = scope:soul_ancestor
					}
					scope:soul_ancestor = {
						POD_copy_base_skills_effect = {
							CLONED_CHAR = scope:cloned_character
							CLONE = scope:soul_ancestor
						}
						copy_traits = scope:cloned_character
						POD_clear_invalid_clone_traits = yes
						POD_copy_vampire_modifiers_effect = yes
						POD_copy_supernatural_secrets_effect = yes
						POD_copy_trait_xp_for_clone_effect = yes
						POD_copy_sexuality_effect = { ORIGIN = scope:cloned_character }
						change_first_name = { template_character = scope:cloned_character }
						POD_copy_physical_appearance_effect = yes
						POD_copy_vanilla_lifestyle_perks_effect = yes
						POD_copy_supernatural_lifestyle_perks_effect = yes
						add_character_flag = blocked_from_leaving
						POD_become_ancestor_spirit_effect = yes
						add_character_flag = splintered_soul_ancestor
						if = {
							limit = { scope:actor = { is_ai = yes } }
							add_character_flag = ai_spirit
						}
					}
					scope:cloned_character = {
						send_interface_message = {
							type = POD_fera_good
							title = POD_splintered_soul_interaction
							desc = POD_splintered_soul_interaction_message
							left_icon = root

							POD_form_spirit_pact_effect = { TARGET = scope:soul_ancestor }
						}
					}
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_frequency = 0
	ai_will_do = { base = 0 }
}
