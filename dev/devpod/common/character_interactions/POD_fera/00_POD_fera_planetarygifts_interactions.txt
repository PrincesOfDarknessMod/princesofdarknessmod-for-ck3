POD_ebon_binding_interaction = {
	category = interaction_category_fera
	desc = POD_ebon_binding_interaction_desc
	icon = POD_planetarygifts_interaction
	
	cooldown = { years = 1 }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { POD_splat_has_gnosis_trigger = yes }
		scope:actor = { has_perk = planetarygifts_ebon_binding_perk }
	}

	is_valid = {
		custom_tooltip = {
			text = any_secret_is_known_by_actor
			scope:recipient = { 
				any_secret = {
					is_known_by = scope:actor
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = { POD_has_enough_gnosis_for_gift = { COUNT = 1 } }
		scope:recipient = { 
			NOT = { has_character_modifier = ebon_binding_modifier }
		}
	}
	on_accept = {
		scope:actor = {
			POD_gnosis_decrease_effect = yes	
		}
		scope:recipient = {
			add_character_modifier  = { 
				modifier = ebon_binding_modifier
				years = 10
			}
		}
		scope:recipient = { 
			add_opinion = {
				target = scope:actor
				modifier = hurt_opinion
				opinion = -20
			}
		}
	}
	
	auto_accept = yes
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no 
		is_landed = yes 
	}
	ai_targets = {
		ai_recipients = liege
	}
	ai_targets = {
		ai_recipients = neighboring_rulers
		ai_recipients = peer_vassals
		max = 10
	}
	ai_frequency = 60
	ai_will_do = {
		base = 0
		
		opinion_modifier = {
			opinion_target = scope:recipient
			multiplier = -0.25
		}
		modifier = {
		    add = 50
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
		modifier = {
			factor = 0
			scope:recipient = {
				is_ai = no
			}
		}
	}
}
