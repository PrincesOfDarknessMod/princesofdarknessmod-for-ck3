
POD_silver_claws_interaction = {
	category = interaction_category_fera
	icon = POD_mokole_interaction
	desc = POD_silver_claws_interaction_desc

	cooldown = { years = 5 }
	
	is_shown = {
		scope:actor = { has_perk = silver_claws_perk }
		scope:actor = scope:recipient
	}

	is_valid_showing_failures_only = {
		scope:actor = { POD_has_enough_gnosis_for_gift = { COUNT = 1 } }
		POD_self_targeted_interaction_is_valid_showing_failures_only = yes
	}
	
	on_accept = {
		scope:actor = {
			POD_willpower_rouse_check_effect = yes
			POD_gift_cost_effect = { COUNT = 1 }
			POD_increase_wounds_effect = { REASON = self_mutilation COUNT = 1 }
			add_character_modifier = { modifier = POD_mokole_silver_claws_modifier years = 3 }
		}
	}
	
	auto_accept = yes
	
	ai_frequency = 0
	ai_will_do = { base = 0 }
}


POD_walk_into_my_eyes_interaction = {
	category = interaction_category_fera
	icon = POD_mokole_interaction
	desc = POD_walk_into_my_eyes_interaction_desc

	cooldown = { years = 5 }
	cooldown_against_recipient = { years = 10 }
	
	is_shown = {
		scope:actor = { has_perk = walk_into_my_eyes_perk }
		NOT = { scope:actor = scope:recipient }
		POD_hostile_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = { POD_has_enough_gnosis_for_gift = { COUNT = 3 } }
		custom_description = {
			object = scope:recipient
			text = "strong_mental_defenses_trigger"
			strong_mental_defenses_interaction_trigger = yes
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}
	
	on_accept = {
		scope:actor = {
			POD_willpower_rouse_check_effect = yes
			POD_gift_cost_effect = { COUNT = 3 }

			duel = {
				skill = learning
				target = scope:recipient
			
				40 = {
					desc = POD_walk_into_my_eyes_interaction_success
					compare_modifier = { value = scope:duel_value multiplier = high_positive_duel_skill_multiplier }

					add_truce_both_ways = { character = scope:recipient name = walk_into_my_eyes_perk_name years = 5 result = victory }
					
					hidden_effect = {
						scope:recipient = { trigger_event = { id = POD_mokole.500 years = 5 } }
						send_interface_toast = {
							title = POD_walk_into_my_eyes_interaction_success
							left_icon = scope:actor
							right_icon = scope:recipient
						}
					}
				}
				
				60 = {
					compare_modifier = { value = scope:duel_value multiplier = low_negative_duel_skill_multiplier }
					hidden_effect = {
						send_interface_toast = {
							title = POD_shared_loca_failure
							left_icon = scope:actor
							right_icon = scope:recipient
						}
					}
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_frequency = 0
	ai_will_do = { base = 0 }
}
