	desc = POD_dream_the_beast_interaction_desc

	interface_priority = 90
	options_heading = POD_dream_the_beast_interaction_options_heading

		scope:actor = { has_perk = dream_the_beast_perk }

	}

	is_valid_showing_failures_only = {

		is_valid = { scope:actor = { POD_has_enough_gnosis_for_gift = { COUNT = 1 } } }

		localization = archator_tooltip

	}

		is_shown = { scope:actor = { is_target_in_variable_list = { name = unlocked_dinosaur_maa target = flag:avithons } } }
		is_valid = { scope:actor = { POD_has_enough_gnosis_for_gift = { COUNT = 2 } } }
		flag = avithon
		localization = avithon_tooltip
	}
	send_option = {
		is_shown = { scope:actor = { is_target_in_variable_list = { name = unlocked_dinosaur_maa target = flag:dicravenators } } }
		is_valid = { scope:actor = { POD_has_enough_gnosis_for_gift = { COUNT = 2 } } }
		flag = dicravenator
		localization = dicravenator_tooltip
	}
	send_option = {
		is_shown = { scope:actor = { is_target_in_variable_list = { name = unlocked_dinosaur_maa target = flag:rhincanthuses } } }
		is_valid = { scope:actor = { POD_has_enough_gnosis_for_gift = { COUNT = 4 } } }
		flag = rhincanthus
		localization = rhincanthus_tooltip
	}
	send_option = {
		is_shown = { scope:actor = { is_target_in_variable_list = { name = unlocked_dinosaur_maa target = flag:baroceratops } } }
		is_valid = { scope:actor = { POD_has_enough_gnosis_for_gift = { COUNT = 4 } } }
		flag = baroceratops
		localization = baroceratops_tooltip
	}
	send_option = {
		is_shown = { scope:actor = { is_target_in_variable_list = { name = unlocked_dinosaur_maa target = flag:dracomuses } } }
		is_valid = { scope:actor = { POD_has_enough_gnosis_for_gift = { COUNT = 6 } } }
		flag = dracomus
		localization = dracomus_tooltip
	}
	send_option = {
		is_shown = { scope:actor = { is_target_in_variable_list = { name = unlocked_dinosaur_maa target = flag:archeorexes } } }
		is_valid = { scope:actor = { POD_has_enough_gnosis_for_gift = { COUNT = 6 } } }
		flag = archeorex
		localization = archeorex_tooltip
	}



	can_send = {
		custom_description = {
			text = send_option_selected
			OR = {
				scope:archator ?= yes
				scope:dipelta ?= yes
				scope:avithon ?= yes
				scope:dicravenator ?= yes
				scope:rhincanthus ?= yes
				scope:baroceratops ?= yes
				scope:dracomus ?= yes
				scope:archeorex ?= yes
			}
		}
	}

	on_accept = {
		switch = {
			trigger = yes

			scope:archator = {
				scope:actor = { POD_gift_cost_effect = { COUNT = 1 } }
				scope:recipient = { create_artifact_archator_effect = { OWNER = scope:recipient } }
			}

			scope:dipelta = {
				scope:actor = { POD_gift_cost_effect = { COUNT = 1 } }
				scope:recipient = { create_artifact_dipelta_effect = { OWNER = scope:recipient } }
			}

			scope:avithon = {
				scope:actor = { POD_gift_cost_effect = { COUNT = 2 } }
				scope:recipient = { create_artifact_avithon_effect = { OWNER = scope:recipient } }
			}

			scope:dicravenator = {
				scope:actor = { POD_gift_cost_effect = { COUNT = 2 } }
				scope:recipient = { create_artifact_dicravenator_effect = { OWNER = scope:recipient } }
			}

			scope:rhincanthus = {
				scope:actor = { POD_gift_cost_effect = { COUNT = 4 } }
				scope:recipient = { create_artifact_rhincanthus_effect = { OWNER = scope:recipient } }
			}

			scope:baroceratops = {
				scope:actor = { POD_gift_cost_effect = { COUNT = 4 } }
				scope:recipient = { create_artifact_baroceratops_effect = { OWNER = scope:recipient } }
			}

			scope:dracomus = {
				scope:actor = { POD_gift_cost_effect = { COUNT = 6 } }
				scope:recipient = { create_artifact_dracomus_effect = { OWNER = scope:recipient } }
			}

			scope:archeorex = {
				scope:actor = { POD_gift_cost_effect = { COUNT = 6 } }
				scope:recipient = { create_artifact_archeorex_effect = { OWNER = scope:recipient } }
			}
		}
		
		if = {
			limit = { scope:actor.domicile ?= { has_domicile_building_or_higher = mokole_main_summoning_array_01 } }
			hidden_effect_new_object = {
				scope:newly_created_artifact = {
					add_artifact_modifier = artifact_monthly_minor_prestige_5_modifier

					if = {
						limit = { scope:actor.domicile ?= { has_domicile_parameter = dreamed_dinosaurs_illustrious } }
						set_artifact_rarity = illustrious
					}
					else_if = {
						limit = { scope:actor.domicile ?= { has_domicile_parameter = dreamed_dinosaurs_famed } }
						set_artifact_rarity = famed
					}
					else_if = {
						limit = { scope:actor.domicile ?= { has_domicile_parameter = dreamed_dinosaurs_masterwork } }
						set_artifact_rarity = masterwork
					}

					if = {
						limit = { scope:actor.domicile ?= { has_domicile_parameter = dreamed_dinosaurs_no_decay } }
						set_should_decay = no
					}
				}
			}
		}
	}

	send_options_exclusive = yes

	auto_accept = yes
	ai_frequency = 0
	ai_will_do = { base = 0 }
}


POD_stampede_interaction = {
	category = interaction_category_fera
	icon = POD_mokole_interaction
	desc = POD_stampede_interaction_desc

	cooldown = { years = 10 }
	
	is_shown = {
		scope:actor = { has_perk = stampede_perk }
		scope:actor != scope:recipient
		scope:recipient = { is_landed = yes }
		POD_hostile_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = { POD_has_enough_gnosis_for_gift = { COUNT = 2 } }
		scope:recipient = { is_at_war_with = scope:actor }
		POD_interaction_is_valid_showing_failures_only = yes
	}

	#Capital
	send_option = {
		is_valid = { scope:actor = { POD_has_enough_gnosis_for_gift = { COUNT = 2 } } }
		flag = capital_target
		localization = capital
	}
	#Extra Province
	send_option = {
		is_shown = {
			scope:recipient = { 
				any_held_title = {
					count > 1
					tier < tier_duchy
				}
			}
		}
		is_valid = {
			scope:actor = { has_perk = kings_call_perk }
			scope:actor = { POD_has_enough_gnosis_for_gift = { COUNT = 4 } }
		}
		flag = one_extra
	}
	#Extra 2 Provinces
	send_option = {
		is_shown = {
			scope:recipient = { 
				any_held_title = {
					count > 2
					tier < tier_duchy
				}
			}
		}
		is_valid = {
			scope:actor = { has_perk = kings_call_perk }
			scope:actor = { POD_has_enough_gnosis_for_gift = { COUNT = 6 } }
		}
		flag = two_extra
	}

	can_send = {
		custom_description = {
			text = send_option_selected
			OR = {
				scope:capital_target ?= yes
				scope:one_extra ?= yes
				scope:two_extra ?= yes
			}
		}
	}
	
	on_accept = {
		scope:recipient = {
			random_character_war = {
				limit = { any_war_participant = { this = scope:actor } }
				save_scope_as = dinosaur_war
			}
			if = {
				limit = { scope:capital_target = no } # one or two extra
				random_held_title = {
					limit = {
						tier < tier_duchy
						NOT = { capital_province.county = this }
					}
					title_province = { save_scope_as = one_extra_location }
				}
			}
			if = {
				limit = { scope:two_extra = yes }
				random_held_title = {
					limit = {
						tier < tier_duchy
						NOR = { this = capital_province.county this = scope:one_extra_location }
					}
					title_province = { save_scope_as = two_extra_location }
				}
			}
		}
		scope:actor = {
			POD_willpower_rouse_check_effect = yes
			switch = {
				trigger = yes

				scope:capital_target = { POD_gift_cost_effect = { COUNT = 2 } }
				scope:one_extra = { POD_gift_cost_effect = { COUNT = 4 } }
				scope:two_extra = { POD_gift_cost_effect = { COUNT = 6 } }
			}


			duel = {
				skill = learning
				target = scope:recipient
			
				30 = {
					desc = POD_stampede_interaction_success
					compare_modifier = { value = scope:duel_value multiplier = high_positive_duel_skill_multiplier }
					modifier = { add = mokole_grondr_add_success_chance }
					
					send_interface_toast = {
						title = POD_stampede_interaction_success
						left_icon = scope:actor
						right_icon = scope:recipient
						
						POD_mokole_stampede_effect = {
							LOCATION = scope:recipient.capital_province
							WAR = scope:dinosaur_war
						}

						if = {
							limit = { scope:capital_target = no } # one or two extra
							POD_mokole_stampede_effect = {
								LOCATION = scope:one_extra_location
								WAR = scope:dinosaur_war
							}
						}
						if = {
							limit = { scope:two_extra = yes }
							POD_mokole_stampede_effect = {
								LOCATION = scope:two_extra_location
								WAR = scope:dinosaur_war
							}
						}
					}
				}
				
				70 = {
					compare_modifier = { value = scope:duel_value multiplier = low_negative_duel_skill_multiplier }
					
					send_interface_toast = {
						title = POD_shared_loca_failure
						left_icon = scope:actor
						right_icon = scope:recipient
					}
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_frequency = 0
	ai_will_do = { base = 0 }
}



POD_rulers_of_the_earth_interaction = {
	category = interaction_category_fera
	icon = POD_mokole_interaction
	desc = POD_rulers_of_the_earth_interaction_desc
	options_heading = POD_rulers_of_the_earth_interaction_options_heading

	target_type = title
	target_filter = actor_domain_titles

	cooldown = { years = 5 }

	is_shown = {
		scope:actor = { has_perk = rulers_of_the_earth_perk }
		scope:actor = scope:recipient
	}
	
	is_valid_showing_failures_only = {
		scope:actor = { POD_has_enough_gnosis_for_gift = { COUNT = 3 } }
		scope:actor = { primary_title = { tier >= tier_kingdom } } #In case you're crazy and still don't have a kingdom by this point
		POD_self_targeted_interaction_is_valid_showing_failures_only = yes
	}

	can_be_picked_title = {
		trigger_if = {
			limit = { scope:actor.primary_title = { tier = tier_hegemony } }
			scope:target = { tier = tier_empire }
		}
		trigger_else_if = {
			limit = { scope:actor.primary_title = { tier = tier_empire } }
			scope:target = { tier = tier_kingdom }
		}
		trigger_else = { scope:target = { tier = tier_duchy } }
		NOT = { scope:actor.primary_title = { is_de_jure_liege_or_above_target = scope:target } }
	}

	has_valid_target_showing_failures_only = {
		scope:target = {
			is_titular = no
			is_head_of_faith = no
			is_landless_type_title = no
		}
	}

	on_accept = {
		scope:actor = {
			POD_willpower_rouse_check_effect = yes
			POD_gift_cost_effect = { COUNT = 3 }
		}
		scope:target = { set_de_jure_liege_title = scope:actor.primary_title }
	}

	auto_accept = yes
	ai_frequency = 0
	ai_will_do = { base = 0 }
}






POD_banish_innocent_interaction = {
	category = interaction_category_fera
	icon = POD_mokole_interaction
	desc = POD_banish_innocent_interaction_desc
	common_interaction = yes
	interface_priority = 100
	options_heading = POD_banish_innocent_interaction_heading

	use_diplomatic_range = no
	
	cooldown_against_recipient = { years = 5 }

	is_shown = {
		scope:actor = {
			POD_is_mokole_trigger = yes
			has_relation_innocent_tormentor = scope:recipient
		}
		POD_hostile_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:recipient = { NOT = { has_character_modifier = innocent_recent_tormentor_modifier } }
		POD_interaction_is_valid_showing_failures_only = yes
	}


	#Fight it off
	send_option = {
		is_valid = { scope:actor = { POD_has_enough_gnosis_for_gift = { COUNT = 3 } } }
		flag = fight
		localization = innocent_fight
	}

	#Trap it in the Umbra
	send_option = {
		is_valid = {
			scope:actor = {
				POD_has_enough_gnosis_for_gift = { COUNT = 1 }
				piety >= 250
			}
		}
		flag = trap
		localization = innocent_trap
	}

	#Use a Rite
	send_option = {
		is_valid = {
			scope:actor = {
				POD_has_enough_gnosis_for_gift = { COUNT = 1 }
				piety >= 250
			}
		}
		flag = rite
		localization = innocent_rite
	}

	#Use Domicile Nets
	send_option = {
		is_valid = {
			scope:actor = {
				domicile ?= { has_domicile_building_or_higher = mokole_hallowed_net_01 }
				POD_has_enough_gnosis_for_gift = { COUNT = 1 }
				piety >= mokole_innocent_banishment_domicile_cost
				NOT = { has_character_modifier = innocent_hallowed_net_cd }
			}
		}
		flag = hallowed_net
		localization = innocent_hallowed_net
	}

	can_send = {
		custom_description = {
			text = send_option_selected
			OR = {
				scope:fight ?= yes
				scope:trap ?= yes
				scope:rite ?= yes
				scope:hallowed_net ?= yes
			}
		}
	}

	on_accept = {
		switch = {
			trigger = yes

			#Brute force is always harder
			scope:fight = {
				scope:actor = { 
					POD_gift_cost_effect = { COUNT = 3 }
				
					duel = {
						skills = { martial prowess } 
						target = scope:recipient
					
						30 = {
							desc = POD_banish_innocent_interaction_success
							compare_modifier = { value = scope:duel_value multiplier = high_positive_duel_skill_multiplier }
							modifier = { add = mokole_grondr_add_success_chance }
							modifier = { add = { value = mokole_innocent_banishment_success } }
							
							send_interface_toast = {
								title = POD_banish_innocent_interaction_success
								left_icon = scope:actor
								right_icon = scope:recipient

								POD_mokole_banish_innocent = { INNOCENT = scope:recipient }
							}
						}
						
						70 = {
							compare_modifier = { value = scope:duel_value multiplier = low_negative_duel_skill_multiplier }

							send_interface_toast = {
								title = POD_shared_loca_failure
								left_icon = scope:actor
								right_icon = scope:recipient

								POD_increase_wounds_hostile_effect = {
									ATTACKER = scope:recipient
									COUNT = scope:recipient.mokole_innocent_damage_value
									REASON = fight
								}
							}
						}
					}
				}
			}

			#Trapping is intrigue, something most mokole suck at
			scope:trap = {
				scope:actor = {
					POD_gift_cost_effect = { COUNT = 1 }
					add_piety = -250
				
					duel = {
						skills = { intrigue }
						target = scope:recipient
					
						40 = {
							desc = POD_banish_innocent_interaction_success
							compare_modifier = { value = scope:duel_value multiplier = high_positive_duel_skill_multiplier }
							modifier = { add = mokole_grondr_add_success_chance }
							modifier = { add = { value = mokole_innocent_banishment_success } }
							
							send_interface_toast = {
								title = POD_banish_innocent_interaction_success
								left_icon = scope:actor
								right_icon = scope:recipient

								POD_mokole_banish_innocent = { INNOCENT = scope:recipient }
							}
						}
						
						60 = {
							compare_modifier = { value = scope:duel_value multiplier = low_negative_duel_skill_multiplier }

							send_interface_toast = {
								title = POD_shared_loca_failure
								left_icon = scope:actor
								right_icon = scope:recipient

								add_stress = major_stress_gain
							}
						}
					}
				}
			}

			#Learning uses piety, but most mokole are good with learning so this has lower base chance
			scope:rite = {
				scope:actor = {
					POD_gift_cost_effect = { COUNT = 1 }
					add_piety = -250
				
					duel = {
						skills = { learning }
						target = scope:recipient
					
						40 = {
							desc = POD_banish_innocent_interaction_success
							compare_modifier = { value = scope:duel_value multiplier = high_positive_duel_skill_multiplier }
							modifier = { add = mokole_grondr_add_success_chance }
							modifier = { add = { value = mokole_innocent_banishment_success } }
							
							send_interface_toast = {
								title = POD_banish_innocent_interaction_success
								left_icon = scope:actor
								right_icon = scope:recipient

								POD_mokole_banish_innocent = { INNOCENT = scope:recipient }
							}
						}
						
						60 = {
							compare_modifier = { value = scope:duel_value multiplier = low_negative_duel_skill_multiplier }

							send_interface_toast = {
								title = POD_shared_loca_failure
								left_icon = scope:actor
								right_icon = scope:recipient

								add_stress = major_stress_gain
							}
						}
					}
				}
			}


			#Domicile is instant success but has a cooldown
			scope:hallowed_net = {
				scope:actor = {
					POD_gift_cost_effect = { COUNT = 1 }
					
					if = {
						limit = { mokole_innocent_banishment_domicile_cost > 0 }
						add_piety = {
							value = mokole_innocent_banishment_domicile_cost
							multiply = -1
						}
					}
					
					send_interface_toast = {
						title = POD_banish_innocent_interaction_success
						left_icon = scope:actor
						right_icon = scope:recipient

						POD_mokole_banish_innocent = { INNOCENT = scope:recipient }
						add_character_modifier = { modifier = innocent_hallowed_net_cd years = 15 }
					}

				}
			}

			fallback = {}
		}
	}

	send_options_exclusive = yes

	auto_accept = yes
	ai_frequency = 0
	ai_will_do = { base = 0 }
}

		scope:actor = { POD_has_enough_gnosis_for_gift = { COUNT = 3 } }
		POD_interaction_is_valid_showing_failures_only = yes
	}


	#Change Traits
	send_option = {
		is_valid = { scope:actor = scope:recipient }
		flag = traits
		localization = change_personality
	}

	#Copy Appearance
	send_option = {
		is_valid = { 
			scope:actor != scope:recipient
			scope:recipient = { is_adult = yes }
		}
		flag = copy_appearance
		localization = copy_appearance
	}

	#Restore Appearance
	send_option = {
		is_shown = { scope:actor = { has_variable = original_face } }
		is_valid = { scope:actor = scope:recipient }
		flag = restore_face
		localization = restore_appearance
	}

	can_send = {
		custom_description = {
			text = send_option_selected
			OR = {
				scope:traits ?= yes
				scope:copy_appearance ?= yes
				scope:restore_face ?= yes
			}
		}
	}

	on_accept = {
		scope:actor = {
			POD_willpower_rouse_check_effect = yes
			POD_gift_cost_effect = { COUNT = 1 }
			
			switch = {
				trigger = yes

				scope:traits = {
					custom_tooltip = POD_a_thousand_secret_faces_interaction_tooltip
					trigger_event = POD_change_personality.100
				}
				
				scope:copy_appearance = {
					hidden_effect_new_object = {
						if = {
							limit = { NOT = { has_variable = original_face } }
							POD_create_dummyclone_effect = { CHARACTER = scope:actor }
							scope:dummyclone = { copy_inheritable_appearance_from = scope:actor }
							set_variable = {
								name = original_face
								value = scope:dummyclone
							}
						}
					}
					copy_inheritable_appearance_from = scope:recipient
				}

				scope:restore_face = {
					copy_inheritable_appearance_from = var:original_face
					remove_variable = original_face
				}
			}
		}
	}

	auto_accept = yes
	ai_frequency = 0
	ai_will_do = { base = 0 }
}

	icon = POD_mokole_interaction
	desc = POD_take_the_true_form_interaction_desc

	cooldown = { years = 5 }
	cooldown_against_recipient = { years = 10 }
	
	is_shown = {
		scope:actor = { has_perk = take_the_true_form_perk }
		scope:actor != scope:recipient
		POD_hostile_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = { POD_has_enough_gnosis_for_gift = { COUNT = 2 } }
		POD_interaction_is_valid_showing_failures_only = yes
	}
	
	on_accept = {
		scope:actor = {
			POD_willpower_rouse_check_effect = yes
			POD_gift_cost_effect = { COUNT = 2 }

			duel = {
				skill = learning
				target = scope:recipient
			
				30 = {
					desc = POD_take_the_true_form_interaction_success
					compare_modifier = { value = scope:duel_value multiplier = high_positive_duel_skill_multiplier }
					modifier = { add = mokole_grondr_add_success_chance }

					send_interface_toast = {
						title = POD_take_the_true_form_interaction_success
						left_icon = scope:actor
						right_icon = scope:recipient

						scope:recipient = {
							POD_shapeshift_revert_all_effect = yes
							add_character_modifier = {
								modifier = POD_mokole_take_the_true_form_modifier
								years = 10
							}
						}
					}
				}
				
				70 = {
					compare_modifier = { value = scope:duel_value multiplier = low_negative_duel_skill_multiplier }

					send_interface_toast = {
						title = POD_shared_loca_failure
						left_icon = scope:actor
						right_icon = scope:recipient
					}
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_frequency = 0
	ai_will_do = { base = 0 }
}
