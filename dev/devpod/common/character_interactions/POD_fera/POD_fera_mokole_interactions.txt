POD_a_thousand_secret_faces_interaction = {
	category = interaction_category_fera
	icon = POD_mokole_interaction
	desc = POD_a_thousand_secret_faces_interaction_desc

	cooldown = { years = 10 }

	is_shown = {
		scope:actor = { has_perk = a_thousand_secret_faces_perk }
		POD_hostile_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = { POD_has_enough_gnosis_for_gift = { COUNT = 3 } }
		POD_interaction_is_valid_showing_failures_only = yes
	}


	#Change Traits
	send_option = {
		is_valid = { scope:actor = scope:recipient }
		flag = traits
		localization = change_personality
	}

	#Copy Appearance
	send_option = {
		is_valid = { 
			NOT = { scope:actor = scope:recipient }
			scope:recipient = { is_adult = yes }
		}
		flag = copy_appearance
		localization = copy_appearance
	}

	#Restore Appearance
	send_option = {
		is_shown = { scope:actor = { has_variable = original_face } }
		is_valid = { scope:actor = scope:recipient }
		flag = restore_face
		localization = restore_appearance
	}

	can_send = {
		custom_description = {
			text = send_option_selected
			OR = {
				scope:traits ?= yes
				scope:copy_appearance ?= yes
				scope:restore_face ?= yes
			}
		}
	}

	on_accept = {
		scope:actor = {
			POD_willpower_rouse_check_effect = yes
			POD_gift_cost_effect = { COUNT = 1 }
		}
		switch = {
			trigger = yes

			scope:traits = { scope:actor = { custom_tooltip = POD_a_thousand_secret_faces_interaction_tooltip trigger_event = POD_change_personality.100 } }
			
			scope:copy_appearance = {
				scope:actor = {
					hidden_effect = {
						if = {
							limit = { NOT = { has_variable = original_face } }
							POD_create_dummyclone_effect = { CHARACTER = scope:actor }
							if = {
								limit = { exists = scope:dummyclone }
								scope:dummyclone = { copy_inheritable_appearance_from = scope:actor }
								set_variable = {
									name = original_face
									value = scope:dummyclone
								}
							}
						}
					}
					copy_inheritable_appearance_from = scope:recipient 
				}
			}

			scope:restore_face = { scope:actor = { copy_inheritable_appearance_from = var:original_face remove_variable = original_face } }
		}
	}

	auto_accept = yes
	ai_frequency = 0
	ai_will_do = { base = 0 }
}
