

POD_mokole_weather_magic_interaction = {
	category = interaction_category_fera
	icon = POD_mokole_interaction
	desc = POD_mokole_weather_magic_interaction_desc
	common_interaction = yes
	interface_priority = 100
	options_heading = POD_mokole_weather_magic_interaction_options_heading
	

	is_shown = {
		scope:actor = {
			OR = {
				has_perk = flood_the_land_perk
				has_perk = call_the_tides_perk
				has_perk = anger_of_the_wani_perk
			}
		}
		POD_hostile_interaction_is_shown = yes
		scope:recipient = { exists = location }
	}

	is_valid_showing_failures_only = {
		scope:actor = { POD_has_enough_gnosis_for_gift = { COUNT = POD_mokole_weather_magic_interaction_cost } }
		POD_interaction_is_valid_showing_failures_only = yes
	}

	send_options_exclusive = no

	#County
	send_option = {
		is_shown = {
			NOR = {
				scope:duchy = yes
				scope:kingdom = yes
				scope:empire = yes
			}
		}
		is_valid = { always = yes }
		flag = county
		localization = county
	}
	#Duchy
	send_option = {
		is_shown = {
			NOR = {
				scope:county = yes
				scope:kingdom = yes
				scope:empire = yes
			}
		}
		is_valid = { has_perk = age_of_beasts_perk }
		flag = duchy
		localization = duchy
	}
	#Kingdom
	send_option = {
		is_shown = {
			NOR = {
				scope:county = yes
				scope:duchy = yes
				scope:empire = yes
			}
		}
		is_valid = { has_perk = age_of_sleep_perk }
		flag = kingdom
		localization = kingdom
	}
	#Empire
	send_option = {
		is_shown = {
			NOR = {
				scope:county = yes
				scope:duchy = yes
				scope:kingdom = yes
			}
		}
		is_valid = { has_perk = age_of_kings_perk }
		flag = empire
		localization = empire
	}



	#Flood
	send_option = {
		is_shown = {
			scope:actor = {
				OR = {
					has_perk = flood_the_land_perk
					has_perk = call_the_tides_perk
				}
			}
			NOR = {
				scope:greater_flood = yes
				scope:storm = yes
				scope:blizzard = yes
			}
		}
		flag = flood
		localization = pod_weather_flood
	}
	#Greater Flood
	send_option = {
		is_shown = {
			scope:actor = { uses_cairo_sobek_alternate_reward = yes }
			NOR = {
				scope:flood = yes
				scope:storm = yes
				scope:blizzard = yes
			}
		}
		flag = greater_flood
		localization = pod_weather_greater_flood
	}
	#Storm
	send_option = {
		is_shown = {
			scope:actor = { has_perk = anger_of_the_wani_perk }
			NOR = {
				scope:flood = yes
				scope:greater_flood = yes
				scope:blizzard = yes
			}
		}
		flag = storm
		localization = pod_weather_storm
	}
	#Blizzard
	send_option = {
		is_shown = {
			scope:actor = { has_perk = anger_of_the_wani_perk }
			NOR = {
				scope:flood = yes
				scope:greater_flood = yes
				scope:storm = yes
			}
		}
		flag = blizzard
		localization = pod_weather_blizzard
	}
	
	can_send = {
		custom_description = {
			text = send_option_selected
			OR = {
				scope:county ?= yes
				scope:duchy ?= yes
				scope:kingdom ?= yes
				scope:empire ?= yes
			}
			OR = {
				scope:flood ?= yes
				scope:greater_flood ?= yes
				scope:storm ?= yes
				scope:blizzard ?= yes
			}
		}
	}

	on_accept = {
		switch = {
			trigger = yes
			scope:flood				= { save_scope_value_as = { name = weather_type   value = flag:pod_weather_flood } }
			scope:greater_flood		= { save_scope_value_as = { name = weather_type   value = flag:pod_weather_greater_flood } }
			scope:storm				= { save_scope_value_as = { name = weather_type   value = flag:pod_weather_storm } }
			scope:blizzard			= { save_scope_value_as = { name = weather_type   value = flag:pod_weather_blizzard } }
		}
		

		scope:actor = {
			POD_gift_cost_effect = { COUNT = POD_mokole_weather_magic_interaction_cost }
			
			switch = {
				trigger = yes
				scope:county	= { POD_setup_weather_magic_counties_from_title = { TITLE = scope:recipient.location.county } }
				scope:duchy		= { POD_setup_weather_magic_counties_from_title = { TITLE = scope:recipient.location.duchy } }
				scope:kingdom	= { POD_setup_weather_magic_counties_from_title = { TITLE = scope:recipient.location.kingdom } }
				scope:empire	= { POD_setup_weather_magic_counties_from_title = { TITLE = scope:recipient.location.empire } }
			}
			
			if = {
				limit = {
					OR = {
						scope:flood = yes
						scope:greater_flood = yes
					}
					has_perk = flood_the_land_perk
					has_perk = call_the_tides_perk
				}
				POD_weather_magic_effect = { MONTHS = 96 } #8 years
			}
			else = { POD_weather_magic_effect = { MONTHS = POD_weather_default_duration } } #5 years
		}
	}

	auto_accept = yes
	ai_frequency = 0
	ai_will_do = { base = 0 }
}


POD_dream_the_beast_interaction = {
	category = interaction_category_fera
	icon = POD_mokole_interaction
	desc = POD_dream_the_beast_interaction_desc
	common_interaction = yes
	interface_priority = 90
	options_heading = POD_dream_the_beast_interaction_options_heading
	
	cooldown = { years = 10 }

	is_shown = {
		scope:actor = { has_perk = dream_the_beast_perk }
		scope:actor = scope:recipient
	}

	is_valid_showing_failures_only = {
		POD_self_targeted_interaction_is_valid_showing_failures_only = yes
	}

	

	#Archator
	send_option = {
		is_valid = { scope:actor = { POD_has_enough_gnosis_for_gift = { COUNT = 1 } } }
		flag = archator
		localization = archator
	}

	can_send = {
		custom_description = {
			text = send_option_selected
			OR = {
				scope:archator ?= yes
			}
		}
	}

	on_accept = {
		switch = {
			trigger = yes

			scope:archator = {
				scope:actor = { POD_gift_cost_effect = { COUNT = 1 } }
				scope:recipient = {
					set_artifact_rarity_common = yes
					create_artifact = {
						name = artifact_archator_name
						description = artifact_archator_description
						type = animal
						template = animal_template
						visuals = animal_trinket
						modifier = archator_modifier
						max_durability = 10
						wealth = scope:wealth
						quality = scope:quality
						history = {
							type = created
							actor = scope:actor
							recipient = scope:actor
							location = scope:actor.location
						}
						save_scope_as = newly_created_artifact
					}
					hidden_effect_new_object = { scope:newly_created_artifact = { set_variable = mokole_dinosaur set_variable = summoned_artifact } }
				}
			}




			fallback = {}
		}
		
		if = {
			limit = { scope:actor.domicile ?= { has_domicile_building_or_higher = mokole_main_summoning_array_01 } }
			hidden_effect_new_object = {
				scope:newly_created_artifact = {
					add_artifact_modifier = artifact_monthly_minor_prestige_5_modifier

					if = {
						limit = { scope:actor.domicile ?= { has_domicile_parameter = dreamed_dinosaurs_illustrious } }
						set_artifact_rarity = illustrious
					}
					else_if = {
						limit = { scope:actor.domicile ?= { has_domicile_parameter = dreamed_dinosaurs_famed } }
						set_artifact_rarity = famed
					}
					else_if = {
						limit = { scope:actor.domicile ?= { has_domicile_parameter = dreamed_dinosaurs_masterwork } }
						set_artifact_rarity = masterwork
					}

					if = {
						limit = { scope:actor.domicile ?= { has_domicile_parameter = dreamed_dinosaurs_no_decay } }
						set_should_decay = no
					}
				}
			}
		}
	}

	#send_options_exclusive = yes

	auto_accept = yes
	ai_frequency = 0
	ai_will_do = { base = 0 }
}


POD_stampede_interaction = {
	category = interaction_category_fera
	icon = POD_mokole_interaction
	desc = POD_stampede_interaction_desc

	cooldown = { years = 10 }
	
	is_shown = {
		scope:actor = { has_perk = stampede_perk }
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_landed = yes }
		POD_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:recipient = { is_at_war_with = scope:actor }
		POD_interaction_is_valid_showing_failures_only = yes
	}

	#Capital
	send_option = {
		is_valid = { scope:actor = { POD_has_enough_gnosis_for_gift = { COUNT = 2 } } }
		flag = capital_target
		localization = capital
	}
	#Extra Province
	send_option = {
		is_shown = {
			scope:recipient = { 
				any_held_title = {
					count > 1
					tier < duchy
				}
			}
		}
		is_valid = {
			scope:actor = { has_perk = kings_call_perk }
			scope:actor = { POD_has_enough_gnosis_for_gift = { COUNT = 4 } }
		}
		flag = one_extra
		localization = one_extra
	}
	#Extra 2 Provinces
	send_option = {
		is_shown = {
			scope:recipient = { 
				any_held_title = {
					count > 2
					tier < duchy
				}
			}
		}
		is_valid = {
			scope:actor = { has_perk = kings_call_perk }
			scope:actor = { POD_has_enough_gnosis_for_gift = { COUNT = 6 } }
		}
		flag = two_extra
		localization = one_extra
	}

	can_send = {
		custom_description = {
			text = send_option_selected
			OR = {
				scope:capital_target ?= yes
				scope:one_extra ?= yes
				scope:two_extra ?= yes
			}
		}
	}
	
	on_accept = {
		scope:recipient = {
			random_character_war = {
				limit = { any_war_participant = { this = scope:actor } }
				save_scope_as = dinosaur_war
			}
			if = {
				limit = { scope:capital_target = no }
				random_held_title = {
					limit = {
						tier < duchy
						NOT = { capital_province.county = this }
					}
					title_province = { save_scope_as = one_extra_location }
				}
			}
			if = {
				limit = { scope:two_extra = yes }
				random_held_title = {
					limit = {
						tier < duchy
						NOR = { this = capital_province.county this = scope:one_extra_location }
					}
					title_province = { save_scope_as = two_extra_location }
				}
			}
		}
		scope:actor = {
			POD_willpower_rouse_check_effect = yes
			switch = {
				trigger = yes

				scope:capital_target = { POD_gift_cost_effect = { COUNT = 2 } }
				scope:one_extra = { POD_gift_cost_effect = { COUNT = 4 } }
				scope:two_extra = { POD_gift_cost_effect = { COUNT = 6 } }
			}


			duel = {
				skill = learning
				target = scope:recipient
			
				30 = {
					desc = POD_stampede_interaction_success
					compare_modifier = { value = scope:duel_value multiplier = high_positive_duel_skill_multiplier }
					modifier = { add = mokole_grondr_add_success_chance }

					scope:actor = {
						if = {
							limit = { has_perk = kings_call_perk }
							spawn_army = {
								men_at_arms = { type = dryptoraptors stacks = 2  }
								men_at_arms = { type = archators stacks = 1 }
								men_at_arms = { type = baroceratops stacks = 1 }
								location = scope:recipient.capital_province
								inheritable = no
								uses_supply = no
								war = scope:dinosaur_war
								name = summoned_dinosaurs
							}
						}
						else = {
							spawn_army = {
								men_at_arms = { type = dryptoraptors stacks = 2 }
								location = scope:recipient.capital_province
								inheritable = no
								uses_supply = no
								war = scope:dinosaur_war
								name = summoned_dinosaurs
							}
						}
					}

					if = {
						limit = { scope:one_extra = yes }
						scope:actor = {
							if = {
								limit = { has_perk = kings_call_perk }
								spawn_army = {
									men_at_arms = { type = dryptoraptors stacks = 2  }
									men_at_arms = { type = archators stacks = 1 }
									men_at_arms = { type = baroceratops stacks = 1 }
									location = scope:one_extra_location
									inheritable = no
									uses_supply = no
									war = scope:dinosaur_war
									name = summoned_dinosaurs
								}
							}
							else = {
								spawn_army = {
									men_at_arms = { type = dryptoraptors stacks = 2 }
									location = scope:one_extra_location
									inheritable = no
									uses_supply = no
									war = scope:dinosaur_war
									name = summoned_dinosaurs
								}
							}
						}
					}
					if = {
						limit = { scope:two_extra = yes }
						scope:actor = {
							if = {
								limit = { has_perk = kings_call_perk }
								spawn_army = {
									men_at_arms = { type = dryptoraptors stacks = 2  }
									men_at_arms = { type = archators stacks = 1 }
									men_at_arms = { type = baroceratops stacks = 1 }
									location = scope:two_extra_location
									inheritable = no
									uses_supply = no
									war = scope:dinosaur_war
									name = summoned_dinosaurs
								}
							}
							else = {
								spawn_army = {
									men_at_arms = { type = dryptoraptors stacks = 2 }
									location = scope:two_extra_location
									inheritable = no
									uses_supply = no
									war = scope:dinosaur_war
									name = summoned_dinosaurs
								}
							}
						}
					}
					
					hidden_effect = {
						send_interface_toast = {
							title = POD_stampede_interaction_success
							left_icon = scope:actor
							right_icon = scope:recipient
						}
					}
				}
				
				70 = {
					compare_modifier = { value = scope:duel_value multiplier = low_negative_duel_skill_multiplier }
					hidden_effect = {
						send_interface_toast = {
							title = POD_shared_loca_failure
							left_icon = scope:actor
							right_icon = scope:recipient
						}
					}
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_frequency = 0
	ai_will_do = { base = 0 }
}

	}

	has_valid_target_showing_failures_only = {
		scope:target = { is_titular = no }

	}

	on_accept = {
		scope:actor = {
			POD_willpower_rouse_check_effect = yes
			POD_gift_cost_effect = { COUNT = 3 }
		}
		scope:target = { set_de_jure_liege_title = scope:actor.primary_title }
	}

	auto_accept = yes
	ai_frequency = 0
	ai_will_do = { base = 0 }
}

	}

	send_option = {
		is_valid = { scope:actor = scope:recipient }
		flag = traits
		localization = change_personality
	}

	#Copy Appearance
	send_option = {
		is_valid = { 
			NOT = { scope:actor = scope:recipient }
			scope:recipient = { is_adult = yes }
		}
		flag = copy_appearance
		localization = copy_appearance
	}

	#Restore Appearance
	send_option = {
		is_shown = { scope:actor = { has_variable = original_face } }
		is_valid = { scope:actor = scope:recipient }
		flag = restore_face
		localization = restore_appearance
	}

	can_send = {
		custom_description = {
			text = send_option_selected
			OR = {
				scope:traits ?= yes
				scope:copy_appearance ?= yes
				scope:restore_face ?= yes
			}
		}
	}

	on_accept = {
		scope:actor = {
			POD_willpower_rouse_check_effect = yes
			POD_gift_cost_effect = { COUNT = 1 }
		}
		switch = {
			trigger = yes

			scope:traits = { scope:actor = { custom_tooltip = POD_a_thousand_secret_faces_interaction_tooltip trigger_event = POD_change_personality.100 } }
			
			scope:copy_appearance = {
				scope:actor = {
					hidden_effect = {
						if = {
							limit = { NOT = { has_variable = original_face } }
							POD_create_dummyclone_effect = { CHARACTER = scope:actor }
							if = {
								limit = { exists = scope:dummyclone }
								scope:dummyclone = { copy_inheritable_appearance_from = scope:actor }
								set_variable = {
									name = original_face
									value = scope:dummyclone
								}
							}
						}
					}
					copy_inheritable_appearance_from = scope:recipient 
				}
			}

			scope:restore_face = { scope:actor = { copy_inheritable_appearance_from = var:original_face remove_variable = original_face } }
		}
	}

	auto_accept = yes

	ai_will_do = { base = 0 }
