POD_freyjas_blessing_interaction = {
	category = interaction_category_fera
	icon = POD_bastet_interaction
	desc = POD_freyjas_blessing_interaction_desc
	
	is_shown = {
		scope:actor = { has_perk = freyjas_blessing_perk }
		scope:actor != scope:recipient
		POD_interaction_is_shown = yes
	}
	
	is_valid = {
		scope:recipient = {
			is_female = yes
			is_pregnant = no
		}
		POD_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = {  POD_has_enough_gnosis_for_gift = { COUNT = 1 } }
		scope:actor = { NOT = { has_trait = abomination } }
		scope:recipient = {
			is_visibly_fertile = yes
			is_adult = yes
		}
		trigger_if = {
			limit = { scope:actor = { is_female = yes } }
			scope:recipient = { is_married = yes }
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:actor = { POD_gift_cost_effect = { COUNT = 1 } }

		if = {
			limit = { scope:actor = { is_female = no } }
			scope:recipient = { make_pregnant = { father = scope:actor } }
		}
		else = {
			scope:recipient = { make_pregnant = { father = primary_spouse } }
		}
	}
	
	ai_min_reply_days = 0
	ai_max_reply_days = 0
	
	ai_accept = {
		base = -50
		
		opinion_modifier = { # Opinion Factor
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 1.0
			desc = AI_OPINION_REASON
		}
	}
}
