POD_mask_of_a_thousand_faces_interaction = {
	category = interaction_category_vampire
	desc = POD_mask_of_a_thousand_faces_interaction_desc
	icon = POD_obfuscate_interaction
	
	is_shown = {
		OR = {
			NOT = { scope:actor = scope:recipient }
			scope:actor = { has_variable = original_face }
		}
		scope:actor = { has_perk = mask_of_a_thousand_faces_perk }
	}

	is_valid_showing_failures_only = {
		scope:actor = {
		   NOT = { has_character_modifier = bloodhunger5_modifier }
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
			POD_is_not_trapped_in_umbra_trigger = { ACTOR = scope:actor }
		}
	}
	
	send_option = {
		is_shown = {
			scope:actor = { has_variable = original_face }
		}
		is_valid = {
			scope:actor = scope:recipient
		}
		flag = restore_face
		localization = POD_vampshapeshift_interaction_restore_face
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { has_variable = original_face }
			NOT = { scope:actor = scope:recipient }
		}
		flag = restore_face_recipient
		localization = POD_vampshapeshift_interaction_restore_face_recipient
	}
	
	send_option = {
		is_shown = {
			NOT = { scope:actor = scope:recipient }
		}
		flag = copy_face
		localization = POD_vampshapeshift_interaction_copy_face
	}
	
	send_option = {
		is_shown = {
			NOT = { scope:actor = scope:recipient }
		}
		is_valid = {
			scope:actor = { has_perk = mask_of_janus_perk }
		}
		flag = paste_face
		localization = POD_vampshapeshift_interaction_paste_face
	}

	on_accept = {
		scope:actor = {
			if = {
				limit = { has_trait = risen }
				POD_arcanoi_cost_effect = { COUNT = 1 }
			}
			else = {
				POD_hunger_increase_effect = yes
			}
		}
		if = {
			limit = {
				scope:copy_face = yes
			}
			scope:actor = { 
				copy_inheritable_appearance_from = scope:recipient 
				hidden_effect = { 
					if = {
						limit = { NOT = { has_variable = original_face } }
						create_character = {
							age = scope:actor.effective_age
							faith = scope:actor.faith
							gender = scope:actor
							random_traits = no
							dynasty = none
							employer = character:POD_wraith_01
							save_scope_as = new_created_character
						}
						set_variable = {
							name = original_face
							value = scope:new_created_character
						}
					}
				}
			}
		}
		if = {
			limit = {
				scope:paste_face = yes
			}
			scope:recipient = { 
				copy_inheritable_appearance_from = scope:actor 
				hidden_effect = { 
					if = {
						limit = { NOT = { has_variable = original_face } }
						create_character = {
							age = scope:recipient.effective_age
							faith = scope:recipient.faith
							gender = scope:recipient
							random_traits = no
							dynasty = none
							employer = character:POD_wraith_01
							save_scope_as = new_created_character
						}
						set_variable = {
							name = original_face
							value = scope:new_created_character
						}
					}
				}
				add_opinion = {
					target = scope:actor
					modifier = hurt_opinion
					opinion = -10
				}
			}
		}
		if = {
			limit = {
				scope:restore_face = yes
			}
			scope:actor = { 
				copy_inheritable_appearance_from = var:original_face 
				remove_variable = original_face
			}
		}
		if = {
			limit = {
				scope:restore_face_recipient = yes
			}
			scope:recipient = { 
				copy_inheritable_appearance_from = var:original_face 
				remove_variable = original_face
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
