POD_unleash_maelstrom_interaction = {
	category = interaction_category_wraith
	icon = POD_horror_interaction
	desc = POD_unleash_maelstrom_interaction_desc
	
	cooldown = { years = 5 }
	cost = { piety = 1500 }
	
	auto_accept = yes

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_trait = lament }
		scope:actor = { has_perk = storm_gloam_perk }
		scope:recipient = { is_landed = yes }
		scope:recipient.highest_held_title_tier > tier_barony
		scope:actor = { NOT = { target_is_liege_or_above = scope:recipient } }
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
			trigger_if = {
				limit = { NOT = { has_trait = wraith } }
				NOT = { has_trait_with_flag = is_spirit }
			}
		}
		POD_cross_umbra_interaction_valid_trigger = yes
		custom_description = {
            object = scope:actor
            text = "has_five_laments_courtiers"
            scope:actor = {
				any_courtier = { 
					count >= 5
					has_trait = lament 
				}
			}
        }
	}

	on_accept = {
	   custom_tooltip = POD_unleash_maelstrom_interaction.tt
	   scope:actor = {
			POD_willpower_rouse_check_effect = yes
			duel = {
				skill = learning
				target = scope:recipient
			
				50 = {
					desc = POD_unleash_maelstrom_interaction.duel.success
					
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					
					
					custom_tooltip = POD_unleash_maelstrom_interaction.duel.tt.attacker_wins
					
					
					hidden_effect = {
						scope:recipient = {
							send_interface_toast = {
								title = POD_unleash_maelstrom_interaction_recipient_toast
								left_icon = scope:recipient
								right_icon = scope:actor
							}
							every_sub_realm_county = {
								add_county_modifier = {
									modifier = maelstrom_winds_modifier
									years = 5
								}
							}
							add_opinion = {
								target = scope:actor
								modifier = POD_unleash_maelstrom_in_my_realm
							}						
						}
						
						scope:actor = {
							send_interface_toast = {
								title = POD_unleash_maelstrom_interaction_actor_toast
								left_icon = scope:actor
								right_icon = scope:recipient
							}
						}
					}
				}
				
				50 = {
					desc = POD_unleash_maelstrom_interaction.duel.failure

					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier
					}
					
					modifier = {
						add = 10
						scope:recipient = { 
							has_trait = obtenebrationadvanced	
						}
					}
					modifier = {
						add = 5
						scope:recipient = { 
							has_trait = truefaith	
						}
					}
					modifier = {
						add = 10
						scope:recipient = { 
							has_trait = beatific	
						}
					}
					
					custom_tooltip = POD_unleash_maelstrom_interaction.duel.tt.attacker_lost
					hidden_effect = {
						scope:actor = {
							send_interface_toast = {
								title = POD_unleash_maelstrom_interaction_failed_toast
								left_icon = scope:actor
								right_icon = scope:recipient
							}
						}
					}
				}
			}
		}
		scope:recipient = { 
			progress_towards_rival_effect = {
				CHARACTER = scope:actor
				REASON = rival_cast_spell
				OPINION = 0
			}
		}
    }
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no 
		is_landed = yes 
		
		piety > 1500
	}
	ai_targets = {
		ai_recipients = primary_war_enemies
	}
	ai_frequency = 360
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 45
			scope:actor = {
			    is_at_war_with = scope:recipient
			}
		}
		modifier = {
		    factor = 2
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
	}  
}

POD_create_clone_interaction = {
	category = interaction_category_wraith
	icon = POD_horror_interaction
	desc = POD_create_clone_interaction_desc
	
	cost = { piety = 250 }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_trait = lament }
		scope:actor = { has_perk = doppelganger_perk }
	}
	
	is_valid = {
		scope:recipient = { is_imprisoned_by = scope:actor }
	}
	
	is_valid_showing_failures_only = {
		scope:actor = {
			trigger_if = {
				limit = { NOT = { has_trait = wraith } }
				NOT = { has_trait_with_flag = is_spirit }
			}
		}
		scope:recipient = { POD_can_be_killed_physically_trigger = yes }
		POD_cross_umbra_interaction_valid_trigger = yes
		scope:recipient = { 
			NOT = { has_trait = torpor }
			NOT = { has_trait = automaton }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }	
			POD_is_not_clone_trigger = yes
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}

	on_accept = {
		scope:actor = {			
			duel = {
				skill = learning
				value = scope:recipient.average_of_all_skills_and_prowess
			
				50 = {
					desc = POD_create_clone_interaction.duel.success
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					scope:recipient = { save_scope_as = cloned_character }
					hidden_effect = {
						create_character = {
							save_scope_as = effigy_character
							age = 1
							faith = scope:cloned_character.faith
							gender = scope:cloned_character
							random_traits = no
							dynasty = none
							employer = scope:actor
						}
						if = {
							limit = { exists = scope:effigy_character } # error suppression
							
							scope:effigy_character = {
								copy_inheritable_appearance_from = scope:cloned_character
								set_age = 1
								copy_traits = scope:cloned_character
								POD_remove_splat_traits_effect = yes
								add_trait = mortal
								POD_remove_wrong_supernatural_traits_on_new_character = yes
								POD_copy_base_skills_effect = {
									CLONED_CHAR = scope:cloned_character
									CLONE = scope:effigy_character
								}
								POD_copy_sexuality_effect = { ORIGIN = scope:cloned_character }
								add_character_flag = blocked_from_leaving
								add_character_modifier = doppelganger_modifier
							}
						}
						trigger_event = podmummy.3100
					}
				}    
				
				50 = {
					desc = POD_create_clone_interaction.duel.failure
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier
					}
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
