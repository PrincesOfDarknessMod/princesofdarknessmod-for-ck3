POD_command_the_earth_interaction = {
	category = interaction_category_possessed
	icon = POD_fomori_interaction
	desc = POD_weather_magic_specific_interaction_desc
	options_heading = POD_weather_magic_interaction_options_heading
	
	cooldown = { months = 120 }
	cost = { piety = massive_piety_value }
	
	auto_accept = yes
	send_options_exclusive = yes

	is_shown = {
		scope:actor != scope:recipient
		scope:actor = { POD_is_possessed_trigger = yes }
		scope:actor = { 
			OR = { 
				has_perk = gorgon_command_the_earth_perk
				has_perk = kami_command_the_earth_perk 
			}
		}
		scope:recipient = { is_landed = yes }
		scope:recipient.highest_held_title_tier > tier_barony
	}
	
	is_valid_showing_failures_only = {
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}
	
	send_option = {
		flag = flow
		localization = pod_weather_flow
	}
	
	send_option = {
		flag = storm
		localization = pod_weather_storm
	}
	
	send_option = {
		flag = earthquake
		localization = pod_weather_earthquake
	}
	
	send_option = {
		flag = flood
		localization = pod_weather_flood
	}
	
	send_option = {
		flag = drought
		localization = pod_weather_drought
	}

	on_accept = {
	   scope:actor = {
			POD_willpower_rouse_check_effect = yes	
			
			if = {
				limit = { scope:recipient.highest_held_title_tier = tier_empire }
				add_piety = -2000
			}
			else_if = {
				limit = { scope:recipient.highest_held_title_tier >= tier_kingdom }
				add_piety = -1000
			}
			else_if = {
				limit = { scope:recipient.highest_held_title_tier >= tier_duchy }
				add_piety = -500
			}
			
			switch = {
				trigger = yes
				scope:drought			= { save_scope_value_as = { name = weather_type   value = flag:pod_weather_drought } }
				scope:earthquake		= { save_scope_value_as = { name = weather_type   value = flag:pod_weather_earthquake } }
				scope:flow				= { save_scope_value_as = { name = weather_type   value = flag:pod_weather_flow } }
				scope:storm				= { save_scope_value_as = { name = weather_type   value = flag:pod_weather_storm } }
				scope:flood	= {
					if = {
						limit = { POD_can_use_greater_flood = yes }
						save_scope_value_as = { name = weather_type   value = flag:pod_weather_greater_flood }
					}
					else = {
						save_scope_value_as = { name = weather_type   value = flag:pod_weather_flood }
					}
				}
			}
	
			duel = {
				skill = learning
				value = 10
			
				50 = {
					desc = POD_weather_magic.duel.success
					
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					
					scope:actor = {
						send_interface_toast = {
							title = POD_weather_magic.duel.success
							left_icon = scope:actor
							right_icon = scope:recipient	
							custom_tooltip = POD_weather_magic.duel.tt.attacker_wins
							POD_setup_weather_magic_counties_from_ruler = {
								RULER = scope:recipient
							}
							POD_weather_magic_effect = {
								MONTHS = POD_weather_default_duration
							}
						}
					}
				}
				
				50 = {
					desc = POD_weather_magic.duel.failure

					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier
					}
					
					scope:actor = {
						send_interface_toast = {
							title = POD_weather_magic.duel.failure
							left_icon = scope:actor
							right_icon = scope:recipient
							custom_tooltip = POD_weather_magic.duel.tt.attacker_lost
						}
					}
				}
			}
		}
		scope:recipient = { 
			progress_towards_rival_effect = {
				CHARACTER = scope:actor
				REASON = rival_cast_spell
				OPINION = 0
			}
		}
    }
	
	ai_potential = {
		# The basic requirements
		is_imprisoned = no 
		is_landed = yes 
		
		piety > 1500
	}
	ai_targets = {
		ai_recipients = primary_war_enemies
	}
	ai_frequency = 360
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 45
			scope:actor = {
			    is_at_war_with = scope:recipient
			}
		}
		modifier = {
		    factor = 2
			scope:actor = {
				has_any_bad_relationship_with_character_trigger = {
					CHARACTER = scope:recipient
				}
			}
		}
	}  
}
