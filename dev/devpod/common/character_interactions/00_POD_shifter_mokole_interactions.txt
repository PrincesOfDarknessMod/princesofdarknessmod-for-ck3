
POD_dragons_milk_interaction = {

	icon = POD_mokole_interaction
	desc = POD_dragons_milk_interaction_desc
	options_heading = POD_dragons_milk_interaction_options_heading
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_perk = dragons_milk_perk }
		scope:actor = { POD_splat_has_gnosis_trigger = yes }
	}
	
	is_valid = {
		scope:recipient = {
			OR = {
				has_trait = scarred
				has_trait = disfigured
				has_trait = wounded
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = { NOT = { has_character_modifier = POD_gnosis_0_modifier } }
		scope:recipient = {
			POD_can_be_healed_trigger = yes
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = scarred
			}
		}	
		is_valid = {
			custom_description = {
				text = POD_permanent_scarred
				object = scope:recipient
				scope:recipient = { NOT = { has_character_flag = POD_permanent_scarring_flag } }
			}
		}
		flag = scarred
		localization = POD_heal_scarred
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = disfigured
			}
		}	
		is_valid = {
			custom_description = {
				text = POD_permanent_disfigured
				object = scope:recipient
				scope:recipient = { NOT = { has_character_flag = POD_permanent_disfigured_flag } }
			}
		}
		flag = disfigured
		localization = POD_heal_disfigured
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = wounded
			}
		}
		flag = wounded
		localization = POD_heal_wounded
	}
	
	on_accept = {
		if = {
			limit = { NOT = { scope:actor = scope:recipient } }
			scope:actor = { POD_commit_heal_effect = yes }
		}
		scope:actor = { 
			POD_willpower_rouse_check_effect = yes
			POD_gnosis_loss_check_effect = yes
		}
		scope:recipient = {
			play_music_cue = "mx_cue_peace_ensues"

			if = {
				limit = {
					scope:scarred = yes
				}
				scope:recipient = {
					remove_trait = scarred
				}
			}	

			if = {
				limit = {
					scope:disfigured = yes
				}
				scope:recipient = {
					remove_trait = disfigured
				}
			}
			
			if = {
				limit = {
					scope:wounded = yes
				}
				scope:recipient = {
					POD_remove_wound_trait_effect = yes
				}
			}
		}
		if = {
			limit = { NOT = { scope:actor = scope:recipient } }
			scope:recipient = { 
				add_opinion = {
					target = scope:actor
					modifier = pleased_opinion
					opinion = 15
				}
			}
		}
	}
	
	auto_accept = yes

	ai_potential = {
		# The basic requirements
		is_imprisoned = no
		is_landed = yes
		
		piety > major_piety_value
	}
	ai_targets = {
		ai_recipients = self
	}
	ai_targets = {
		ai_recipients = courtiers
		ai_recipients = peer_vassals
		max = 10
	}
	ai_frequency = 12
	ai_will_do = {
		base = 0
		modifier = {
			add = 100
			scope:recipient = {
				OR = {
					this = scope:actor
					is_knight_of = scope:actor
				}
			}
		}
	}
}

POD_a_thousand_secret_faces_interaction = {
	category = interaction_category_werewolf
	desc = POD_a_thousand_secret_faces_interaction_desc
	icon = POD_mokole_interaction
	
	is_shown = {
		OR = {
			NOT = { scope:actor = scope:recipient }
			scope:actor = { has_variable = original_face }
		}
		scope:actor = { POD_splat_has_gnosis_trigger = yes }
		scope:actor = { has_perk = a_thousand_secret_faces_perk }
	}

	is_valid_showing_failures_only = {
		scope:actor = { NOT = { has_character_modifier = POD_gnosis_0_modifier } }
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes
	}
	
	send_option = {
		is_shown = {
			scope:actor = { has_variable = original_face }
		}
		is_valid = {
			scope:actor = scope:recipient
		}
		flag = restore_face
		localization = POD_vampshapeshift_interaction_restore_face
	}
	
	send_option = {
		is_shown = {
			NOT = { scope:actor = scope:recipient }
		}
		flag = copy_face
		localization = POD_vampshapeshift_interaction_copy_face
	}

	on_accept = {
		scope:actor = {
			POD_gnosis_loss_check_effect = yes
		}
		if = {
			limit = {
				scope:copy_face = yes
			}
			scope:actor = {
				hidden_effect = { 
					if = {
						limit = { NOT = { has_variable = original_face } }
						POD_create_dummyclone_effect = { CHARACTER = scope:actor }
						if = {
							limit = { exists = scope:dummyclone }
							scope:dummyclone = { copy_inheritable_appearance_from = scope:actor }
							set_variable = {
								name = original_face
								value = scope:dummyclone
							}
						}
						else = {
							send_interface_toast = {
								title = POD_kueijin_spy.duel.failure
								left_icon = scope:actor
								right_icon = scope:recipient
							}
						}
					}
				}
				copy_inheritable_appearance_from = scope:recipient 
			}
		}
		if = {
			limit = {
				scope:restore_face = yes
			}
			scope:actor = { 
				copy_inheritable_appearance_from = var:original_face 
				remove_variable = original_face
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

			POD_is_not_spirit_trigger = yes

		POD_cross_umbra_interaction_valid_trigger = yes
	}

		scope:actor = { 
			POD_willpower_rouse_check_effect = yes 
			POD_gnosis_loss_check_effect = yes
		}
		scope:recipient = { 
			add_opinion = {
				target = scope:actor
				modifier = hurt_opinion
				opinion = -50
			}
			random = {
				chance = 5
				scope:recipient = {
					add_trait = incapable
					add_character_flag = POD_stoneskin_form_flag
					POD_depose_effect = yes
				}
			}
		}
		scope:actor = { 
			duel = {
				skill = prowess
				target = scope:recipient
				10 = {
					trigger = {
						prowess >= POD_prowess_extremely_high_skill_rating
					}
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier

					add_piety = -1000

				20 = {
					trigger = {
						prowess >= POD_prowess_high_skill_rating
					}
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					add_piety = -500
					scope:recipient = { increase_wounds_multiple_hostile_effect = { REASON = fight AMOUNT = 3 ATTACKER = scope:actor } }
