	   scope:actor = {

			if = {
				limit = { 
					scope:recipient.highest_held_title_tier = tier_empire
				}
				add_piety = -2000
			}
			else_if = {
				limit = { 
					scope:recipient.highest_held_title_tier >= tier_kingdom

			}
			else_if = {
				limit = { 
					scope:recipient.highest_held_title_tier >= tier_duchy 
				}
				add_piety = -500
			}
				
			if = {
				limit = { 
					scope:flow = yes
				}
				custom_tooltip = POD_weather_magic_interaction.tt.flow
			}
			if = {
				limit = { 
					scope:storm = yes
				}
				custom_tooltip = POD_weather_magic_interaction.tt.storm
			}
			if = {
				limit = { 
					scope:earthquake = yes
				}
				custom_tooltip = POD_weather_magic_interaction.tt.earthquake
			}
			if = {
				limit = { 
					scope:flood = yes
				}
				custom_tooltip = POD_weather_magic_interaction.tt.flood
			}
			if = {
				limit = { 
					scope:drought = yes
				}
				custom_tooltip = POD_weather_magic_interaction.tt.drought
			}
	
			duel = {
				skill = learning
				value = 10
				desc = POD_weather_magic.duel.desc
			
				50 = {
					desc = POD_weather_magic.duel.success
					
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					
					scope:actor = {
						send_interface_toast = {
							title = POD_weather_magic.duel.success
							left_icon = scope:actor
							right_icon = scope:recipient	
							custom_tooltip = POD_weather_magic.duel.tt.attacker_wins
						}
					}
					
					hidden_effect = {
						scope:recipient = {
							POD_weather_magic_specific_effect = yes
							every_vassal_or_below = {
								limit = { NOT = { primary_title.tier = tier_barony } }
								POD_weather_magic_specific_effect = yes
							}
						}
					}
				}
				
				50 = {
					desc = POD_weather_magic.duel.failure

					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier
					}
					
					scope:actor = {
						send_interface_toast = {
							title = POD_weather_magic.duel.failure
							left_icon = scope:actor
							right_icon = scope:recipient
							custom_tooltip = POD_weather_magic.duel.tt.attacker_lost
						}
					}
				}
			}
	   }
    }
	
	ai_frequency = 3600
	
	ai_targets = {
		ai_recipients = primary_war_enemies
	}
	
	ai_will_do = {
		base = 0

			add = 45

			    is_at_war_with = scope:recipient
			}
			piety > omega_piety_gain
		}
		modifier = {
		    add = 10
			scope:actor = {
