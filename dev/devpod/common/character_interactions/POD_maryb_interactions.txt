######################################################
# Mary the Black content — Character interactions
######################################################
#
# POD_mary_michael_reembrace_interaction — "Receive into the Dream"
#
# Michael, restored to Mary's body in the Second Dream resolution, recognizes
# his own. Targets are existing vampires sworn to the Codex of Legacies — the
# faith that preserves his doctrine. The act is part diplomacy and part
# theology: the target is folded into Clan Toreador and into house_nephilim,
# the founder's own house. Their previous clan affiliation dies on the spot
# — Michael does not negotiate this part.
#
# NB: the `nephilim` trait is NOT used here. Despite sharing a name with
# house_nephilim, that trait is the demon-children splat (offspring of the
# infernal), unrelated to Michael's Toreador house. Eligibility and
# acceptance key off the Codex of Legacies faith, not the trait.
#
# Cost: 100 piety, paid by the actor. Cooldown: one calendar year per target.
#
# The recipient's faith is not otherwise altered — they already keep the
# Codex, so only the body's lineage changes.
#
######################################################

POD_mary_michael_reembrace_interaction = {
	category = interaction_category_vampire
	icon = icon_dynasty
	desc = POD_mary_michael_reembrace_interaction_desc

	cooldown_against_recipient = { years = 1 }

	# Actor must be a vampire who carries Michael's restored authority —
	# either Mary herself after the Second Dream, or any subsequent member
	# of house_nephilim still walking under the Codex. The second branch
	# keeps the interaction useful for Michael's heirs.
	is_shown = {
		scope:actor != scope:recipient
		# Player-only. Mary-Michael is always a PC; an AI in the founder's
		# seat never reaches for this.
		scope:actor = { is_ai = no }
		scope:actor = { POD_is_vampire_trigger = yes }
		scope:actor = {
			OR = {
				has_character_flag = mary_resolved_second_dream
				AND = {
					house = house:house_nephilim
					faith = faith:codexoflegacies
				}
			}
		}
		# Recipient must be a vampire who already keeps Michael's doctrine —
		# the Codex of Legacies. (Re-embracing mortals or other faiths is out
		# of scope for this pass; a true embrace would need the full
		# POD_embrace flow.)
		scope:recipient = {
			POD_is_vampire_trigger = yes
			faith = faith:codexoflegacies
			# Skip the obvious no-op: already in the destination state.
			NOT = {
				AND = {
					has_trait = toreador
					house = house:house_nephilim
				}
			}
		}
		POD_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		POD_interaction_is_valid_showing_failures_only = yes
	}

	cost = {
		piety = 100
	}

	# Acceptance. Recognition into the Dream is half invitation, half
	# compulsion — the breakdown below is what the target weighs. Net >= 0
	# accepts. Those who already keep the Codex or carry the angelic blood
	# come gladly; the proud, the crowned, and those who head their own
	# house resist surrendering their lineage to the founder. Dread leans
	# on the rest.
	ai_accept = {
		base = 0

		# Opinion of the founder (standard).
		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.5
			desc = AI_OPINION_REASON
		}

		# --- More likely to accept ---

		# Already keeps Michael's doctrine — this is fulfillment, not loss.
		modifier = {
			add = 40
			desc = mary_reembrace_accept_codex
			scope:recipient = { faith = faith:codexoflegacies }
		}
		# Already Clan Toreador — only the house changes; little to give up.
		modifier = {
			add = 25
			desc = mary_reembrace_accept_toreador
			scope:recipient = { has_trait = toreador }
		}
		# A true believer in the restored Dream.
		modifier = {
			add = 15
			desc = mary_reembrace_accept_zealous
			scope:recipient = {
				has_trait = zealous
				faith = faith:codexoflegacies
			}
		}
		# No proud lineage to surrender — eager for any at all.
		modifier = {
			add = 20
			desc = mary_reembrace_accept_weak_lineage
			scope:recipient = {
				OR = {
					has_trait = caitiff
					POD_is_thinblooded_trigger = yes
				}
			}
		}
		# Already kept in the founder's court.
		modifier = {
			add = 15
			desc = mary_reembrace_accept_courtier
			scope:recipient = { is_courtier_of = scope:actor }
		}
		# Bound to the founder by love.
		modifier = {
			add = 20
			desc = mary_reembrace_accept_lover
			scope:recipient = { has_relation_lover = scope:actor }
		}

		# Dread compels (standard halved reasons, stacking at higher levels).
		modifier = {
			add = intimidated_halved_reason_value
			desc = INTIMIDATED_REASON
			scope:recipient = {
				has_dread_level_towards = {
					target = scope:actor
					level = 1
				}
			}
		}
		modifier = {
			add = cowed_halved_reason_value
			desc = COWED_REASON
			scope:recipient = {
				has_dread_level_towards = {
					target = scope:actor
					level = 2
				}
			}
		}

		# --- Less likely to accept ---

		# Heads their own house and will not dissolve it into another's.
		modifier = {
			add = -50
			desc = mary_reembrace_decline_house_head
			scope:recipient = { is_house_head = yes }
		}
		# An independent ruler bows to no founder.
		modifier = {
			add = -30
			desc = mary_reembrace_decline_independent
			scope:recipient = { is_independent_ruler = yes }
		}
		# Too proud to take another's name.
		modifier = {
			add = -25
			desc = mary_reembrace_decline_arrogant
			scope:recipient = { has_trait = arrogant }
		}
		# Craves their own glory, not the founder's.
		modifier = {
			add = -15
			desc = mary_reembrace_decline_ambitious
			scope:recipient = { has_trait = ambitious }
		}
		# Devout to another god entirely.
		modifier = {
			add = -30
			desc = mary_reembrace_decline_other_zealot
			scope:recipient = {
				has_trait = zealous
				NOT = { faith = faith:codexoflegacies }
			}
		}
		# Names the founder a rival.
		modifier = {
			add = -25
			desc = mary_reembrace_decline_rival
			scope:recipient = { has_relation_rival = scope:actor }
		}
		# A crowned head with a kingdom's worth of pride to lose.
		modifier = {
			add = -20
			desc = mary_reembrace_decline_highborn_crown
			scope:recipient = { highest_held_title_tier >= tier_kingdom }
		}
	}

	on_accept = {
		scope:recipient = {
			# Strip whichever clan trait the recipient currently carries.
			# Mirrors PoD's POD_character_vampire_inherit_clan_effect pattern
			# (random_character_trait + vamp_clan flag) so it works against
			# bloodline traits as well as the base clan trait.
			random_character_trait = {
				limit = { has_trait_flag = vamp_clan }
				save_scope_as = previous_clan
			}
			if = {
				limit = { exists = scope:previous_clan }
				remove_trait = scope:previous_clan
			}
			add_trait = baali
			set_house = house:house_nephilim
		}
	}

	# Player-driven only. With no ai_frequency / ai_frequency_by_tier and no
	# ai_potential block, the AI never initiates this interaction (and the
	# is_shown is_ai = no guard makes that explicit). The ai_accept block
	# above still governs how an AI *recipient* decides when a player offers.
}
