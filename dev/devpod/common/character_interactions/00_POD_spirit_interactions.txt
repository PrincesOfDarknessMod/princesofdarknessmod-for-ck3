	target_filter = recipient_artifacts

		scope:target = { 
			NOT = { has_variable = possessed_artifact }

			NOT = { has_variable = wraith_jade }
			NOT = { has_variable = alchemy }
			NOT = { has_variable = amulet }
			NOT = { has_variable = active_effigy }
			NOT = { has_variable = soultaker } # Probably not a good idea
			NOT = { has_variable = soulforge } # Probably not a good idea
			NOT = { artifact_type = animal }
		}
	}
	
	is_shown = {
		scope:actor = scope:recipient
		scope:recipient = {
			has_trait = spirit
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = { 
			custom_description = {
				text = "demand_artifact_interaction_valid_tt_1"
				has_any_artifact = yes
			}
		}
	}
	
	on_accept = {
		scope:recipient = {
			save_scope_as = killed
			scope:target = { 
				set_variable = { name = fetish value = scope:recipient } 
				set_variable = possessed_artifact 
				add_soulforged_artifact_modifier_effect = { KILLED = scope:recipient }
				save_scope_as = this_artifact
				set_artifact_name = artifact_new_fetish_name
				set_artifact_description = artifact_new_fetish_desc
			}
			set_variable = { name = stone_prison value = scope:target }
		}
	}
	
	auto_accept = yes

		base = 0
	}
}
