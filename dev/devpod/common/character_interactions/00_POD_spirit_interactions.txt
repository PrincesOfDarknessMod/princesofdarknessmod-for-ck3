
POD_spirit_possess_artifact_interaction = {
	category = interaction_category_werewolf
	icon = POD_spirit_interaction
	desc = POD_spirit_possess_artifact_interaction_desc
	
	target_type = artifact
	target_filter = recipient_artifacts
	
	can_be_picked = {
		scope:target = { 
			NOT = { has_variable = artifact_pieces } 
			NOT = { has_variable = possessed_artifact }
			NOT = { has_variable = summoned_artifact }
			NOT = { has_variable = wraith_jade }
			NOT = { has_variable = soultaker } # Probably not a good idea
			NOT = { has_variable = soulforge } # Probably not a good idea
			NOT = { artifact_type = animal }
		}
	}
	
	is_shown = {
		scope:actor = scope:recipient
		scope:recipient = {
			has_trait = spirit
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = { 
			custom_description = {
				text = "demand_artifact_interaction_valid_tt_1"
				has_any_artifact = yes
			}
		}
	}
	
	on_accept = {
		scope:recipient = {
			save_scope_as = killed
			scope:target = { 
				set_variable = { name = fetish value = scope:recipient } 
				set_variable = possessed_artifact 
				add_soulforged_artifact_modifier_effect = { KILLED = scope:recipient }
				save_scope_as = this_artifact
				set_artifact_name = artifact_new_fetish_name
				set_artifact_description = artifact_new_fetish_desc
			}
			set_variable = { name = stone_prison value = scope:target }
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_spirit_depossess_artifact_interaction = {
	category = interaction_category_werewolf
	icon = POD_spirit_interaction
	desc = POD_spirit_depossess_artifact_interaction_desc
	
	is_shown = {
		scope:actor = scope:recipient
		scope:recipient = {
			has_trait = spirit
			has_variable = stone_prison
		}
	}

	is_valid_showing_failures_only = {
	}
	
	on_accept = {
		scope:recipient = {
			save_scope_as = killed
			var:stone_prison = {
				save_scope_as = this_artifact
				set_artifact_description = artifact_old_fetish_desc 
				remove_variable = fetish
			}
			remove_variable = stone_prison
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_create_fetish_interaction = {
	category = interaction_category_werewolf
	icon = POD_spirit_interaction
	desc = POD_create_fetish_interaction_desc

	cost = { gold = scope:actor.medium_gold_value }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { has_trait = spirit }
	}

	is_valid_showing_failures_only = {
		scope:actor = { has_relation_spirit_companion = scope:recipient }
		POD_cross_umbra_interaction_valid_trigger = yes										
	}

	send_option = {
		flag = weapon
		localization = add_artifact_interaction_weapon
	}
	send_option = {
		flag = armor
		localization = add_artifact_interaction_armor
	}
	send_option = {
		flag = shinies
		localization = add_artifact_interaction_shinies
	}
	send_option = {
		is_shown = { scope:actor = { has_royal_court = yes } }
		flag = book
		localization = add_artifact_interaction_book
	}
	send_option = {
		is_shown = { scope:actor = { has_royal_court = yes } }
		flag = furniture
		localization = add_artifact_interaction_furniture
	}
	send_option = {
		flag = trinket
		localization = add_artifact_interaction_trinket
	}

	on_accept = {
		scope:actor = {
			soulforged_artifact_quality_tooltip_effect = yes
			POD_universal_energy_decrease_multiple_effect = { COUNT = 2 }
		}
		create_artifact_custom_artifact_effect = {
			OWNER = scope:actor
			CREATOR = scope:actor
			TEMPLATE = scope:recipient
			PROCESS = fetish
		}
	}

	auto_accept = yes

	ai_will_do = {
		base = 0
	}
}

POD_summon_spirit_on_death_interaction = {
    category = interaction_category_werewolf
	icon = POD_spirit_interaction
	desc = POD_summon_spirit_on_death_interaction_desc
	common_interaction = yes
	
	cost = { piety = medium_piety_value }
	
	cooldown = { years = 1 }
	cooldown_against_recipient = { years = 10 }
	
	is_shown = {
		scope:actor = { POD_can_summon_spirit_trigger = yes }
	}
	
	is_valid = {
		scope:recipient = {
			has_trait = shapeshifter
		}
	}
	
	is_valid_showing_failures_only = {
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes										
	}
	
	on_accept = {
		scope:actor = { POD_universal_energy_decrease_multiple_effect = { COUNT = 2 } }
		scope:recipient = {
			set_variable = {
				name = marked_for_return
				value = scope:actor
				years = 10
			}
		}
		custom_tooltip = POD_summon_spirit_on_death_interaction_desc
	}
	
	auto_accept = yes
}

		}

	}

		localization = add_artifact_interaction_weapon
	}
	send_option = {
		flag = armor
		localization = add_artifact_interaction_armor
	}
	send_option = {
		flag = shinies
		localization = add_artifact_interaction_shinies

		is_shown = { scope:actor = { has_royal_court = yes } }

		is_shown = { scope:actor = { has_royal_court = yes } }
		flag = furniture
		localization = add_artifact_interaction_furniture
	}
	send_option = {
		flag = trinket
		localization = add_artifact_interaction_trinket
	}

	on_accept = {
		scope:actor = {
			soulforged_artifact_quality_tooltip_effect = yes
			POD_universal_energy_decrease_multiple_effect = { COUNT = 2 }
		}

	ai_max_reply_days = 0

	ai_accept = {
		base = 0 # Should be 0 for all interactions
		spirit_ai_accept_modifier = yes
	}
