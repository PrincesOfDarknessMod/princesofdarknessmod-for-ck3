POD_summon_spirit_on_death_interaction = {
	category = interaction_category_umbra
	icon = POD_spirit_interaction
	desc = POD_summon_spirit_on_death_interaction_desc
	common_interaction = yes
	
	cost = { piety = medium_piety_value }
	
	cooldown = { years = 1 }
	cooldown_against_recipient = { years = 10 }
	
	is_shown = {
		scope:actor = { POD_can_summon_spirit_trigger = yes }
	}
	
	is_valid = {
		scope:recipient = {
			has_trait = fera
		}
	}
	
	is_valid_showing_failures_only = {
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes										
	}
	
	on_accept = {
		scope:actor = { POD_universal_energy_decrease_multiple_effect = { COUNT = 2 } }
		scope:recipient = {
			set_variable = {
				name = marked_for_return
				value = scope:actor
				years = 10
			}
		}
		custom_tooltip = POD_summon_spirit_on_death_interaction_desc
	}
	
	auto_accept = yes
}

POD_personal_pact_interaction = {

	desc = POD_personal_pact_interaction_desc
	common_interaction = yes
	icon = POD_spirit_interaction
	
	cooldown = { years = 5 } 
	cost = { prestige = medium_prestige_value }
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { has_trait = spirit }
		scope:actor = { NOT = { has_relation_spirit_companion = scope:recipient } }
	}
	
	is_valid = {
	}
	
	is_valid_showing_failures_only = {
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
			POD_is_not_clone_trigger = yes
		}
		POD_cross_umbra_interaction_valid_trigger = yes								
	}
	
	on_accept = {
		scope:actor = { 
			set_relation_spirit_companion = scope:recipient 
			if = {
				limit = {
					OR = {
						POD_gains_faceofthegods_dharma_xp = yes
						POD_gains_faceofthegoddess_dharma_xp = yes
					}
				}
				POD_give_dharma_xp_effect = { VALUE = POD_kueijin_dharma_xp_value_very_high }
			}
		}		
	}

	ai_accept = {
		base = 0 # Should be 0 for all interactions
		
		spirit_ai_accept_modifier = yes
	}

}

POD_set_spirit_nushi_interaction = {
	category = interaction_category_vampire
	desc = POD_set_spirit_nushi_interaction_desc
	common_interaction = yes
	icon = POD_spirit_interaction
	
	cooldown = { years = 5 } 
	cost = { prestige = medium_prestige_value }
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_trait = kueijin }
		scope:recipient = { has_trait = spirit }
		scope:actor.house ?= { has_nushi_modifier_trigger = no }
		NOT = { scope:recipient = { has_variable = patron_house } }
	}
	
	is_valid = {
	}
	
	is_valid_showing_failures_only = {
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
			POD_is_not_clone_trigger = yes
		}
		POD_cross_umbra_interaction_valid_trigger = yes								
	}
	
	on_accept = {
		scope:actor = { 
			custom_tooltip = nushi_duty_tt 
			house = { 
				set_variable = {
					name = nushi
					value = scope:recipient
				}
			}
		}
		scope:recipient = { 
			if = { 
				limit = { has_education_rank_5_trigger = yes }
				scope:actor.house = { add_house_modifier = nushi_patron_5_modifer } # different modifier depending of nushi?
			}
			else_if = { 
				limit = { has_education_rank_4_trigger = yes }
				scope:actor.house = { add_house_modifier = nushi_patron_4_modifer } # different modifier depending of nushi?
			}
			else_if = { 
				limit = { has_education_rank_3_trigger = yes }
				scope:actor.house = { add_house_modifier = nushi_patron_3_modifer } # different modifier depending of nushi?
			}
			else_if = { 
				limit = { has_education_rank_2_trigger = yes }
				scope:actor.house = { add_house_modifier = nushi_patron_2_modifer } # different modifier depending of nushi?
			}
			else = { 
				scope:actor.house = { add_house_modifier = nushi_patron_1_modifer } # different modifier depending of nushi?
			}
			set_variable = {
				name = patron_house 
				value = scope:actor.house
			}
		}
	}

	ai_accept = {
		base = 0 # Should be 0 for all interactions
		
		spirit_ai_accept_modifier = yes
	}

}

POD_remove_spirit_nushi_interaction = {
	category = interaction_category_vampire
	desc = POD_remove_spirit_nushi_interaction_desc
	common_interaction = yes
	icon = POD_spirit_interaction
	
	cost = { prestige = medium_prestige_value }
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_trait = kueijin }
		scope:recipient = { has_trait = spirit }
		scope:recipient = { var:patron_house ?= scope:actor.house }
	}
	
	is_valid = {
		scope:actor = {
			custom_tooltip = {
				text = has_unfinished_nushi_duty
				has_nushi_duty_trigger = no
			}
		}
	}
	
	is_valid_showing_failures_only = {
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
			POD_is_not_clone_trigger = yes
		}
		POD_cross_umbra_interaction_valid_trigger = yes								
	}
	
	on_accept = {
		scope:actor = { 
			house = { 
				remove_variable = nushi
			}
		}
		scope:recipient = { 
			var:patron_house = {
				remove_house_modifier = nushi_patron_5_modifer 
				remove_house_modifier = nushi_patron_4_modifer 
				remove_house_modifier = nushi_patron_3_modifer 
				remove_house_modifier = nushi_patron_2_modifer 
				remove_house_modifier = nushi_patron_1_modifer
			}
			remove_variable = patron_house
			add_opinion = {
				target = scope:actor
				modifier = ask_spirit_opinion
				opinion = -25
			}
		}
	}

	auto_accept = yes
}
