
POD_summon_spirit_on_death_interaction = {
    category = interaction_category_werewolf
	icon = POD_spirit_interaction
	desc = POD_summon_spirit_on_death_interaction_desc
	common_interaction = yes
	
	cost = { piety = medium_piety_value }
	
	cooldown = { years = 1 }
	cooldown_against_recipient = { years = 10 }
	
	is_shown = {
		scope:actor = { POD_can_summon_spirit_trigger = yes }
	}
	
	is_valid = {
		scope:recipient = {
			has_trait = shapeshifter
		}
	}
	
	is_valid_showing_failures_only = {
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes										
	}
	
	on_accept = {
		scope:actor = { POD_universal_energy_decrease_multiple_effect = { COUNT = 2 } }
		scope:recipient = {
			set_variable = {
				name = marked_for_return
				value = scope:actor
				years = 10
			}
		}
		custom_tooltip = POD_summon_spirit_on_death_interaction_desc
	}
	
	auto_accept = yes
}

POD_absorb_fetish_interaction = {
	category = interaction_category_werewolf
	icon = POD_spirit_interaction
	desc = POD_absorption_interaction_desc

	target_type = artifact
	target_filter = actor_artifacts
	
	cooldown = { years = 5 }
	cost = { piety = scope:actor.major_piety_value }
	
	can_be_picked_artifact = {
		scope:target = { 
			has_variable = possessed_artifact
		}
	}
	
	is_shown = {
		scope:actor = scope:recipient 
		scope:actor = { has_trait = spirit }
		scope:actor = { has_perk = absorb_fetish_perk }
	}
	
	is_valid = {
	}
	
	is_valid_showing_failures_only = {
		scope:recipient = { NOT = { has_character_modifier = POD_gnosis_0_modifier } }
		scope:actor = { 
			custom_description = {
				text = "demand_artifact_interaction_valid_tt_1"
				has_any_artifact = yes
			}
		}
	}

	on_accept = {
		scope:actor = { POD_gift_cost_effect = { COUNT = 3 } }
		scope:target = {
			if = {
				limit = { is_unique = yes }
				scope:recipient = { add_stewardship_skill = 1 }
			}
			else_if = {
				limit = { rarity = illustrious }
				scope:recipient = { add_martial_skill = 1 }
			}
			else_if = {
				limit = { rarity = famed }
				scope:recipient = { add_intrigue_skill = 1 }
			}
			else_if = {
				limit = { rarity = masterwork }
				scope:recipient = { add_diplomacy_skill = 1 }
			}
			else = {
				scope:recipient = { add_learning_skill = 1 }
			}
			POD_destroy_artifact_side_effect = yes
			destroy_artifact = this
		}	
	}
	
	auto_accept = yes
	
	ai_frequency = 0

	ai_will_do = {
		base = 0
	}
}
