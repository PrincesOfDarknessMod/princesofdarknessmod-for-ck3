
POD_spirit_possess_artifact_interaction = {
	category = interaction_category_werewolf
	icon = POD_spirit_interaction
	desc = POD_spirit_possess_artifact_interaction_desc
	
	target_type = artifact
	target_filter = recipient_artifacts
	
	can_be_picked = {
		scope:target = { 
			NOT = { has_variable = artifact_pieces } 
			NOT = { has_variable = possessed_artifact }
			NOT = { has_variable = summoned_artifact }
			NOT = { has_variable = wraith_jade }
			NOT = { has_variable = soultaker } # Probably not a good idea
			NOT = { has_variable = soulforge } # Probably not a good idea
			NOT = { artifact_type = animal }
		}
	}
	
	is_shown = {
		scope:actor = scope:recipient
		scope:recipient = {
			has_trait = spirit
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = { 
			custom_description = {
				text = "demand_artifact_interaction_valid_tt_1"
				has_any_artifact = yes
			}
		}
	}
	
	on_accept = {
		scope:recipient = {
			save_scope_as = killed
			scope:target = { 
				set_variable = { name = fetish value = scope:recipient } 
				set_variable = possessed_artifact 
				add_soulforged_artifact_modifier_effect = { KILLED = scope:recipient }
				save_scope_as = this_artifact
				set_artifact_name = artifact_new_fetish_name
				set_artifact_description = artifact_new_fetish_desc
			}
			set_variable = { name = stone_prison value = scope:target }
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_absorb_fetish_interaction = {
	category = interaction_category_werewolf
	icon = POD_spirit_interaction
	desc = POD_absorption_interaction_desc

	target_type = artifact
	target_filter = actor_artifacts
	
	cooldown = { years = 5 }
	cost = { piety = scope:actor.major_piety_value }
	
	can_be_picked_artifact = {
		scope:target = { 
			has_variable = possessed_artifact
		}
	}
	
	is_shown = {
		scope:actor = scope:recipient 
		scope:actor = { has_trait = spirit }
		scope:actor = { has_perk = absorb_fetish_perk }
	}
	
	is_valid = {
	}
	
	is_valid_showing_failures_only = {
		scope:recipient = { NOT = { has_character_modifier = POD_gnosis_0_modifier } }
		scope:actor = { 
			custom_description = {
				text = "demand_artifact_interaction_valid_tt_1"
				has_any_artifact = yes
			}
		}
	}

	on_accept = {
		scope:actor = { POD_gift_cost_effect = { COUNT = 3 } }
		scope:target = {
			if = {
				limit = { is_unique = yes }
				scope:recipient = { add_stewardship_skill = 1 }
			}
			else_if = {
				limit = { rarity = illustrious }
				scope:recipient = { add_martial_skill = 1 }
			}
			else_if = {
				limit = { rarity = famed }
				scope:recipient = { add_intrigue_skill = 1 }
			}
			else_if = {
				limit = { rarity = masterwork }
				scope:recipient = { add_diplomacy_skill = 1 }
			}
			else = {
				scope:recipient = { add_learning_skill = 1 }
			}
			POD_destroy_artifact_side_effect = yes
			destroy_artifact = this
		}	
	}
	
	auto_accept = yes
	
	ai_frequency = 0

	ai_will_do = {
		base = 0
	}
}
