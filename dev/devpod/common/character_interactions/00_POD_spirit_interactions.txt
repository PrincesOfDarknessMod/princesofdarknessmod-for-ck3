
POD_summon_spirit_on_death_interaction = {
    category = interaction_category_werewolf
	icon = POD_spirit_interaction
	desc = POD_summon_spirit_on_death_interaction_desc
	common_interaction = yes
	
	cost = { piety = medium_piety_value }
	
	cooldown = { years = 1 }
	cooldown_against_recipient = { years = 10 }
	
	is_shown = {
		scope:actor = { POD_can_summon_spirit_trigger = yes }
	}
	
	is_valid = {
		scope:recipient = {
			has_trait = shapeshifter
		}
	}
	
	is_valid_showing_failures_only = {
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }
		}
		POD_cross_umbra_interaction_valid_trigger = yes										
	}
	
	on_accept = {
		scope:actor = { POD_universal_energy_decrease_multiple_effect = { COUNT = 2 } }
		scope:recipient = {
			set_variable = {
				name = marked_for_return
				value = scope:actor
				years = 10
			}
		}
		custom_tooltip = POD_summon_spirit_on_death_interaction_desc
	}
	
	auto_accept = yes
}

POD_create_forced_fetish_interaction = {
	category = interaction_category_werewolf
	icon = POD_spirit_interaction
	desc = POD_create_forced_fetish_interaction_desc
	
	cost = { gold = scope:actor.medium_gold_value }
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { has_trait = spirit }
	}

	is_valid_showing_failures_only = {
		scope:recipient = { is_imprisoned_by = scope:actor }
		scope:recipient = {
			NOT = { has_trait = torpor }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }	
			POD_is_not_clone_trigger = yes
		}
		POD_cross_umbra_interaction_valid_trigger = yes										
	}
	
	send_option = {
		flag = weapon
		localization = add_artifact_interaction_weapon
	}
	send_option = {
		flag = armor
		localization = add_artifact_interaction_armor
	}
	send_option = {
		flag = shinies
		localization = add_artifact_interaction_shinies
	}
	send_option = {
		is_shown = { scope:actor = { has_royal_court = yes } }
		flag = book
		localization = add_artifact_interaction_book
	}
	send_option = {
		is_shown = { scope:actor = { has_royal_court = yes } }
		flag = furniture
		localization = add_artifact_interaction_furniture
	}
	send_option = {
		flag = trinket
		localization = add_artifact_interaction_trinket
	}

	on_accept = {
		scope:actor = {
			soulforged_artifact_quality_tooltip_effect = yes
			POD_universal_energy_decrease_multiple_effect = { COUNT = 2 }
		}
		create_artifact_custom_artifact_effect = {
			OWNER = scope:actor
			CREATOR = scope:actor
			TEMPLATE = scope:recipient
			PROCESS = forced_fetish
		}
		scope:recipient = {
			death = {
				death_reason = death_forced_fetish
				killer = scope:actor
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
