
POD_tempt_interaction = {
	icon = icon_personal
	category = interaction_category_demon
	interface_priority = 90
	scheme = POD_tempt_scheme

	send_name = START_SCHEME
	ignores_pending_interaction_block = yes
	
	ai_frequency = 0
	
	cooldown_against_recipient = { years = 20 } 

	is_shown = {
		scope:actor = { has_trait = demon }
		scope:actor = { religion = religion:demons_religion }
		NOT = { scope:actor = scope:recipient }
		scope:recipient = { is_adult = yes }
		scope:actor = { is_adult = yes }
		can_be_tempted_trigger = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			NOT = { is_at_war_with = scope:recipient }
			can_start_scheme = {
				type = POD_tempt_scheme
				target = scope:recipient
			}
		}
		scope:recipient = {
			NOT = { has_trait = truefaith }
			NOT = { has_trait = beatific }
		}
		scope:recipient = {
			NOT = { has_trait = torpor }
			NOT = { has_trait = frozen }
			NOT = { has_trait = statue }
			NOT = { has_trait = wight }
			NOT = { has_trait = barghest }
			custom_description = { text = POD_NOT_AVAILABLE NOT = { has_character_flag = POD_do_not_disturb_flag } }		
		}
	}

	desc = POD_tempt_interaction_desc

	on_accept = {
		hidden_effect = {
			scope:actor = {
				start_scheme = {
					type = POD_tempt_scheme
					target = scope:recipient
				}
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
	
}
