POD_sorcery_summon_interaction = {
	category = interaction_category_mage
	icon = POD_sorcery_interaction
    desc = POD_sorcery_summon_interaction_desc
	
	cooldown = { months = 6 }
	cost = { piety = medium_piety_value }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { has_trait = sorcerer }
		scope:actor = { has_perk = summoning_warding_binding_perk }
		NOT = { scope:actor = { location = scope:recipient.location } }
		scope:recipient = { is_ruler = no }
	}

	is_valid_showing_failures_only = {
		scope:actor = { POD_can_interact_with_skinlands_trigger = { RECIPIENT = scope:recipient } }
		scope:recipient = { 
			NOT = { has_trait = torpor }
			NOT = { has_trait = statue }
			custom_description = { text = POD_do_not_disturb_text NOT = { has_character_flag = POD_do_not_disturb_flag } }	
			POD_is_not_clone_trigger = yes
			POD_is_not_trapped_in_umbra_trigger = { ACTOR = scope:actor }
		}
		custom_description = {
            object = scope:recipient
            text = "strong_mental_defenses_trigger"
            scope:recipient = { strong_mental_defenses_trigger = no }
        }				
	}
	
	on_accept = {
		scope:actor = {
			duel = {
				skill = diplomacy
				target = scope:recipient
				30 = {
					desc = POD_sorcery_summon_interaction_sucess
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					scope:recipient = {
						set_employer = scope:actor
						return_to_court = yes
						add_character_flag = {
							flag = blocked_from_leaving
							years = 5
						}
					}
				}
				30 = {
					desc = POD_sorcery_summon_interaction_fail
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier 
					}
				}
			}
		}
	}

	ai_frequency = 0
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

	is_shown = {
		NOT = { scope:actor = scope:recipient }

		scope:recipient = { is_landed = yes }
		scope:recipient.highest_held_title_tier > tier_barony
	}
	
	is_valid_showing_failures_only = {
	}
	
	send_option = {
		flag = flow
		localization = POD_weather_magic_flow
	}

		flag = storm
		localization = POD_weather_magic_storm
	}

		flag = earthquake
		localization = POD_weather_magic_earthquake
	}
	
	send_option = {
		flag = flood
		localization = POD_weather_magic_flood
	}
	
	send_option = {
		flag = drought
		localization = POD_weather_magic_drought
	}

	on_accept = {
	   scope:actor = {
			POD_willpower_rouse_check_effect = yes	
			
			if = {
				limit = { 
					scope:recipient.highest_held_title_tier = tier_empire
				}
				add_piety = -2000
			}
			else_if = {
				limit = { 
					scope:recipient.highest_held_title_tier >= tier_kingdom 
				}
				add_piety = -1000
			}
			else_if = {
				limit = { 
					scope:recipient.highest_held_title_tier >= tier_duchy 
				}
				add_piety = -500
			}
				
			if = {
				limit = { 
					scope:flow = yes
				}
				custom_tooltip = POD_weather_magic_interaction.tt.flow
			}
			if = {
				limit = { 
					scope:storm = yes
				}
				custom_tooltip = POD_weather_magic_interaction.tt.storm
			}
			if = {
				limit = { 
					scope:earthquake = yes
				}
				custom_tooltip = POD_weather_magic_interaction.tt.earthquake
			}
			if = {
				limit = { 
					scope:flood = yes
				}
				custom_tooltip = POD_weather_magic_interaction.tt.flood
			}
			if = {
				limit = { 
					scope:drought = yes
				}
				custom_tooltip = POD_weather_magic_interaction.tt.drought
			}
	
			duel = {
				skill = learning
				value = 10
				desc = POD_weather_magic.duel.desc
			
				50 = {
					desc = POD_weather_magic.duel.success
					
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					
					scope:actor = {
						send_interface_toast = {
							title = POD_weather_magic.duel.success
							left_icon = scope:actor
							right_icon = scope:recipient	
							custom_tooltip = POD_weather_magic.duel.tt.attacker_wins
						}
					}
					
					hidden_effect = {
						scope:recipient = {
							POD_weather_magic_specific_effect = yes
							every_vassal_or_below = {
								limit = { NOT = { primary_title.tier = tier_barony } }
								POD_weather_magic_specific_effect = yes
							}
						}
					}
				}

					desc = POD_weather_magic.duel.failure

						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier

							title = POD_weather_magic.duel.failure

							right_icon = scope:recipient
							custom_tooltip = POD_weather_magic.duel.tt.attacker_lost
						}
					}
				}
			}
		}
		scope:recipient = { 
			progress_towards_rival_effect = {
				CHARACTER = scope:actor
				REASON = rival_cast_spell

			}
