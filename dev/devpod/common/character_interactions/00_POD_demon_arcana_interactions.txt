POD_regeneration_interaction = {
	category = interaction_category_demon
	common_interaction = yes
	interface_priority = 110
	icon = POD_arcana_interaction
	desc = POD_regeneration_interaction_desc
	options_heading = POD_regeneration_interaction_options_heading

	cooldown = { months = 3 }	
	
	is_shown = {
		scope:actor = scope:recipient	
		scope:actor = {
			OR = {
				has_trait = demon
				has_trait = laham
				has_trait = nephilim
				is_a_thrall = yes
			}
		}
		scope:actor = { has_perk = arcana_regeneration_perk }
	}

	is_valid_showing_failures_only = {
		scope:actor = { 
			trigger_if = { 
				limit = { NOT = { has_perk = soul_burn_perk } }
				NOT = { has_character_modifier = resolve0_modifier } 
			}
		}
		scope:recipient = {
			NOT = { has_trait_with_flag = can_not_cure }
			NOT = { has_character_modifier = eye_of_sakhmet_modifier }
			custom_description = {
				text = POD_has_injury_trait
				object = scope:recipient
				OR = {
					has_trait = scarred
					has_trait = disfigured
					has_trait = blind
					has_trait = one_eyed
					has_trait = maimed
					has_trait = one_legged
					has_trait = eunuch
				
					has_trait = wounded_1
					has_trait = wounded_2
					has_trait = wounded_3
				}
			}
		}
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = scarred
			}
		}	
		is_valid = {
			scope:recipient = { 
				custom_description = {
					text = POD_permanent_scarred
					object = scope:recipient
					NOT = { has_character_flag = POD_permanent_scarring_flag }
				}
			}
		}
		flag = scarred
		localization = POD_heal_scarred
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = disfigured
			}
		}	
		is_valid = {
			scope:recipient = { 
				custom_description = {
					text = POD_permanent_disfigured
					object = scope:recipient
					NOT = { has_character_flag = POD_permanent_disfigured_flag }
				}
			}
		}
		flag = disfigured
		localization = POD_heal_disfigured
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				OR = {
					has_trait = wounded_1
					has_trait = wounded_2
					has_trait = wounded_3
				}
			}
		}

	send_option = {

			}
		}
		flag = one_eyed
		localization = POD_heal_one_eyed
	}
	
	send_option = {
		is_shown = {
			scope:recipient = { 
				has_trait = eunuch
				NOT = {  has_character_flag = POD_permanent_eunuch }
			}
		}	
		flag = eunuch
		localization = POD_heal_eunuch
	}
	
	on_accept = {
		scope:actor = { 
			POD_willpower_rouse_check_effect = yes
		}
		scope:recipient = {
			play_music_cue = "mx_cue_peace_ensues"

			if = {
				limit = {
					scope:scarred = yes
				}
				scope:recipient = {
					remove_trait = scarred
				}
				scope:actor = { POD_resolve_decrease_multiple_effect = { COUNT = 1 } }
			}	

			if = {
				limit = {
					scope:disfigured = yes
				}
				scope:recipient = {
					remove_trait = disfigured
				}
				scope:actor = { POD_resolve_decrease_multiple_effect = { COUNT = 2 } }
			}

				}

					remove_trait = wounded_3
					remove_trait = wounded_2
					remove_trait = wounded_1
				}

			}		

			if = {
				limit = {
					scope:one_eyed = yes
				}
				scope:recipient = {
					remove_trait = one_eyed
				}
				scope:actor = { POD_resolve_decrease_multiple_effect = { COUNT = 2 } }
			}	

			if = {
				limit = {
					scope:blind = yes
				}
				scope:recipient = {
					remove_trait = blind
					add_trait = one_eyed
				}
				scope:actor = { POD_resolve_decrease_multiple_effect = { COUNT = 2 } }
			}	

			if = {
				limit = {
					scope:one_legged = yes
				}
				scope:recipient = {
					remove_trait = one_legged
				}
				scope:actor = { POD_resolve_decrease_multiple_effect = { COUNT = 3 } }
			}	

			if = {
				limit = {
					scope:maimed = yes
				}
				scope:recipient = {
					remove_trait = maimed
				}
				scope:actor = { POD_resolve_decrease_multiple_effect = { COUNT = 3 } }
			}

			if = {
				limit = {
					scope:eunuch = yes
				}
				scope:recipient = {
					remove_trait = eunuch
				}
				scope:actor = { POD_resolve_decrease_multiple_effect = { COUNT = 3 } }
			}
		}
	}
	
	auto_accept = yes

	ai_potential = {
		# The basic requirements
		is_imprisoned = no
		is_landed = yes
	}
	ai_targets = {
		ai_recipients = self
	}
	ai_frequency = 12
	ai_will_do = {
		base = 100
	}
