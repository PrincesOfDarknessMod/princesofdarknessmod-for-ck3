POD_inspire_life_interaction = {
	category = interaction_category_vampire
	icon = POD_art_interaction
	desc = POD_inspire_life_interaction_desc

	target_type = artifact
	target_filter = actor_artifacts
	
	cost = { gold = scope:actor.medium_gold_value }
	
	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { 
			OR = {
				has_trait = kueijin 
				has_trait = dhampyr
			}
		}
		scope:actor = { has_perk = the_spark_of_life_perk }
		scope:actor = { has_any_artifact = yes }
		scope:recipient = { has_trait = wraith }
	}

	can_be_picked = {
		scope:target = {
			NOT = { has_variable = artifact_pieces } 
			NOT = { has_variable = possessed_artifact }
			NOT = { has_variable = tainted_relic }
			NOT = { has_variable = reliquarie } 
			NOT = { has_variable = wraith_jade }
		}
	}

	is_valid_showing_failures_only= {
		scope:actor = {
			custom_tooltip = {
				text = "i_am_empty_of_chi"
				NAND = { 
					has_character_modifier = yinchi0_modifier
					has_character_modifier = yangchi0_modifier
				}
			}
		}
		custom_description = {
			text = POD_not_in_possession
			scope:recipient = { num_of_relation_possessed_target = 0 }
		}
		scope:recipient = { has_relation_necromancer = scope:actor }
	}

	on_accept = {
		scope:actor = { POD_willpower_rouse_check_effect = yes }
		scope:actor = { POD_yangchi_decrease_multiple_effect = { COUNT = 5 } }
		scope:target = {
			add_soulforged_artifact_modifier_effect = { KILLED = scope:recipient }
			set_variable = possessed_artifact
		}
		scope:recipient = {
			death = {
				death_reason = death_soulforging
				killer = scope:actor
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_goblin_portal_interaction = {
	category = interaction_category_vampire
	icon = POD_art_interaction
	desc = POD_goblin_portal_interaction_desc
	
	cooldown = { months = 1 }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = { 
			OR = {
				has_trait = kueijin 
				has_trait = dhampyr
			}
		}
		scope:actor = { has_perk = goblin_portal_perk }
		scope:actor = { is_commanding_army = yes }
		scope:recipient = { exists = location }
	}

	is_valid = {
		scope:actor.location = {
			OR = {
				terrain = desert
				terrain = desert_mountains
			}
		}
		scope:recipient.location = {
			OR = {
				terrain = desert
				terrain = desert_mountains
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			wall_too_strong_trigger = no
		}
		scope:actor = {
			custom_tooltip = {
				text = "i_am_empty_of_chi"
				NAND = { 
					has_character_modifier = yinchi0_modifier
					has_character_modifier = yangchi0_modifier
				}
			}
		}
	}

	on_accept = {
		scope:actor = { 		
			POD_chi_decrease_multiple_effect = { COUNT = 5 }
			POD_willpower_rouse_check_effect = yes
			commanding_army = {
				set_army_location = scope:recipient.location
			}
		}
	}

	ai_frequency = 0
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
