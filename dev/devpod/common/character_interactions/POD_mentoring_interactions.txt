
POD_learn_arcanos_interaction = {
	category = interaction_category_wraith
	desc = POD_learn_arcanos_interaction_desc
	icon = icon_culture

	#ai_frequency = 0

	is_shown = {
		scope:actor = {
			POD_splat_has_access_to_arcanoi = yes
			is_adult = yes
		}
		scope:recipient = {
			has_trait = wraith
			is_adult = yes
		}
		scope:actor != scope:recipient
		POD_interaction_is_shown = yes
	}
	
	cooldown = { years = 5 }

	is_valid_showing_failures_only = {
		custom_tooltip = {
			text = "He_must_be_my_friend"
			scope:recipient = {
				OR = {
					has_relation_lover = scope:actor
					has_relation_friend = scope:actor
					has_relation_best_friend = scope:actor
					has_relation_soulmate = scope:actor
					scope:actor = {
						AND = {
							has_strong_hook = scope:recipient
							has_usable_hook = scope:recipient
						}
					}
				}
			}
		}
		scope:actor = {
			NOT = { is_at_war_with = scope:recipient }
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	on_accept = {
		scope:actor = {
			custom_tooltip = POD_learn_arcanos_interaction_accept
			if = {
				limit = {
					NOR = {
						has_relation_lover = scope:recipient
						has_relation_friend = scope:recipient
						has_relation_best_friend = scope:recipient
						has_relation_soulmate = scope:recipient
					}
				}
				use_hook = scope:recipient
			}
		}
		hidden_effect = {
			scope:actor = {
				trigger_event = POD_mentoring_disciplines.0002
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

POD_ask_learn_arcanos_interaction = {
	category = interaction_category_wraith
	desc = POD_ask_learn_arcanos_interaction_desc
	icon = icon_culture

	#ai_frequency = 0

	is_shown = {
		scope:actor = { num_of_relation_bound_wraith >= 1 }
		scope:recipient = { has_trait = wraith }
		POD_interaction_is_shown = yes
	}

	is_valid_showing_failures_only = {
		custom_tooltip = {
			text = "He_must_be_my_friend"
			scope:recipient = {
				OR = {
					has_relation_lover = scope:actor
					has_relation_friend = scope:actor
					has_relation_best_friend = scope:actor
					has_relation_soulmate = scope:actor
					scope:actor = {
						AND = {
							has_strong_hook = scope:recipient
							has_usable_hook = scope:recipient
						}
					}
				}
			}
		}
		scope:actor = {
			NOT = { is_at_war_with = scope:recipient }
		}
		trigger_if = {
			limit = { exists = scope:secondary_actor }
			POD_interaction_is_shown_custom = {
				ACTOR = scope:secondary_actor
				RECIPIENT = scope:recipient
			}
			POD_interaction_is_shown_custom = {
				ACTOR = scope:actor
				RECIPIENT = scope:secondary_actor
			}
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}

	populate_actor_list = {
		scope:actor = {
			every_relation = {
				type = bound_wraith
				limit = { NOT = { this = scope:recipient } }
				add_to_list = characters
			}
		}
	}
	
	on_accept = {
		scope:actor = {
			custom_tooltip = POD_ask_learn_arcanos_interaction_accept
			if = {
				limit = {
					NOR = {
						has_relation_lover = scope:recipient
						has_relation_friend = scope:recipient
						has_relation_best_friend = scope:recipient
						has_relation_soulmate = scope:recipient
					}
				}
				use_hook = scope:recipient
			}
		}
		hidden_effect = {
			scope:actor = { save_scope_as = necromancer }
			scope:secondary_actor = { save_scope_as = actor }
			scope:necromancer = {
				trigger_event = POD_mentoring_disciplines.0002
			}
		}
	}
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}
