POD_choose_sire_interaction = {
	category = interaction_category_friendly
	common_interaction = yes
	use_diplomatic_range = no
	icon = icon_dynasty

	is_shown = {
		scope:actor = {
			has_character_flag = chargen_choosing_sire_flag
			POD_is_vampire_trigger = yes
		}
		NOT = { scope:actor = scope:recipient }
		scope:recipient = {
			POD_is_vampire_trigger = yes
		}
	}
	
	is_valid_showing_failures_only = {
		POD_can_embrace = {
			SIRE   = scope:recipient
			CHILDE = scope:actor
		}
	}
	
	auto_accept = yes

	desc = POD_choose_sire_interaction_desc

	on_accept = {
		custom_tooltip = POD_choose_parent_interaction.tt
		scope:recipient = { save_scope_as = vampiresire }
		scope:actor = { save_scope_as = created_vampirechilde }
		scope:actor = { trigger_event = POD_charjoin.200 }
	}
}

POD_choose_sifu_interaction = {
	category = interaction_category_friendly
	common_interaction = yes
	use_diplomatic_range = no
	icon = icon_dynasty

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			is_kueijin_trigger = yes
			OR = {
				has_character_flag = chargen_choosing_sire_flag
				has_trait = dharma0
			}
		}
		scope:recipient = {
			is_kueijin_trigger = yes
		}
	}
	auto_accept = yes

	desc = POD_choose_sire_interaction_desc

	on_accept = {
		custom_tooltip = POD_choose_parent_interaction.tt
		scope:recipient = { save_scope_as = vampiresire }
		scope:actor = { save_scope_as = created_vampirechilde }
		scope:actor = {
			if = {
				limit = { has_trait = dharma0 }
				remove_trait = wight
				remove_trait = dharma0
				add_trait_force_tooltip = dharma1
			}
			trigger_event = POD_charjoin.700
		}
	}
}

POD_charjoin_become_vampire_interaction = {
	category = interaction_category_friendly
	common_interaction = yes
	icon = icon_dynasty

	desc = POD_charjoin_become_vampire_interaction_desc

	is_shown = {
		scope:actor = {
			OR = {
				has_character_flag = charjoin_become_vampire_flag
				AND = {
					POD_charjoin_decisions_enabled = yes
					POD_can_be_embraced_scheme_trigger = yes
				}
			}
		}
		scope:recipient = {
			POD_is_vampire_trigger = yes
		}
		POD_can_embrace = {
			SIRE   = scope:recipient
			CHILDE = scope:actor
		}
		POD_hostile_mental_interaction_is_shown = yes
	}
	
	is_valid_showing_failures_only = {
		trigger_if = {
			limit = { scope:actor = { NOT = { has_character_flag = charjoin_become_vampire_flag } } }
			custom_tooltip = {
				text = "He_must_be_my_friend"
				scope:recipient = {
					OR = {
						has_relation_lover = scope:actor
						has_relation_friend = scope:actor
						has_relation_best_friend = scope:actor
						has_relation_soulmate = scope:actor
						scope:actor = {
							AND = {
								has_strong_hook = scope:recipient
								has_usable_hook = scope:recipient
							}
						}
					}
				}
			}
		}
		scope:actor = {
			NOT = { is_at_war_with = scope:recipient }
		}
		scope:recipient = {
			is_vampgeneration16_trigger = no
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}
	
	auto_accept = yes

	on_accept = {
		custom_tooltip = POD_choose_parent_interaction.tt
		scope:recipient = { save_scope_as = vampiresire }
		scope:actor = {
			if = {
				limit = {
					NOR = {
						has_relation_lover = scope:recipient
						has_relation_friend = scope:recipient
						has_relation_best_friend = scope:recipient
						has_relation_soulmate = scope:recipient
					}
				}
				use_hook = scope:recipient
			}
			save_scope_as = created_vampirechilde
			remove_character_flag = charjoin_become_vampire_flag
			
			trigger_event = POD_charjoin.100
		}
	}
}

POD_charjoin_choose_mentor_interaction = {
	category = interaction_category_friendly
	common_interaction = yes
	icon = icon_dynasty

	is_shown = {
		scope:actor = {
			has_character_flag = independent_char_join_order_flag
		}
		scope:recipient = {
			POD_is_supehunter_trigger = yes
		}
		POD_hostile_mental_interaction_is_shown = yes
	}
	
	is_valid_showing_failures_only = {
		custom_tooltip = {
			text = "He_must_be_my_friend"
			scope:recipient = {
				OR = {
					has_relation_lover = scope:actor
					has_relation_friend = scope:actor
					has_relation_best_friend = scope:actor
					has_relation_soulmate = scope:actor
					scope:actor = {
						AND = {
							has_strong_hook = scope:recipient
							has_usable_hook = scope:recipient
						}
					}
				}
			}
		}
		scope:actor = {
			NOT = { is_at_war_with = scope:recipient }
		}
		trigger_if = {
			limit = { scope:recipient = { dynasty ?= dynasty:13000003 } }
			scope:actor = { is_female = yes }
		}
		trigger_if = {
			limit = { scope:recipient = { dynasty ?= dynasty:13000011 } }
			scope:actor = { is_female = yes }
		}
		scope:recipient = {
			NOT = { dynasty ?= dynasty:13000004 }
			NOT = { dynasty ?= dynasty:13000007 }
		}
		POD_interaction_is_valid_showing_failures_only = yes
	}
	
	auto_accept = yes

	desc = POD_charjoin_choose_mentor_interaction_desc

	on_accept = {
		custom_tooltip = POD_choose_parent_interaction.tt
		scope:recipient = { save_scope_as = huntermentor }
		scope:actor = {
			if = {
				limit = {
					NOR = {
						has_relation_lover = scope:recipient
						has_relation_friend = scope:recipient
						has_relation_best_friend = scope:recipient
						has_relation_soulmate = scope:recipient
					}
				}
				use_hook = scope:recipient
			}
			save_scope_as = hunterpupil
			remove_character_flag = independent_char_join_order_flag
	
			if = {
				limit = { is_mortal_trigger = yes }
				if = {
					limit = { scope:recipient = { has_trait = order_member } }
					trigger_event = POD_charjoin.310 	# Divorce branch
				}
				else = {
					trigger_event = POD_charjoin.300
				}
			}
			else_if = {
				limit = { POD_is_supehunter_trigger = yes }
				if = {
					limit = { scope:recipient = { has_trait = order_member } }
					trigger_event = POD_charjoin.410 	# Divorce branch
				}
				else = {
					trigger_event = POD_charjoin.400
				}
			}
		}
	}
}
