	#interface_priority = 95
	icon = POD_serpentis_interaction
    desc = POD_heart_of_darkness_interaction_desc
	
	ai_frequency = 0

	is_shown = {
		scope:actor = scope:recipient
		scope:actor = { is_adult = yes }
		scope:actor = { is_ai = no }
		scope:actor = { has_POD_vamp_blood_trigger = yes }
		scope:actor = {
		    OR = {
			    has_trait = serpentisadvanced
				has_perk = the_heart_of_darkness_perk
			}
		}
		scope:actor = { NOT = { has_character_modifier = heart_of_darkness_modifier } }
	}

		scope:actor = {
		    OR = {
		        NOT = { has_character_modifier = bloodhunger4_modifier }
				NOT = { has_character_modifier = bloodhunger5_modifier }
			}
		}
	}

	on_accept = {
	   custom_tooltip = POD_heart_of_darkness_interaction.tt
	   scope:actor = {
	       trigger_event = POD_heart_of_darkness.001
	   }
    }
	
	# ai_targets = {
		# ai_recipients = self
	# }
	
	auto_accept = yes
	
	cooldown = { years = 20 }
	
	ai_will_do = {
		base = 0
		
		modifier = {
			add = 15
			piety > 1200
		}
		modifier = {
		    add = 50
			has_trait = callous
		}
		modifier = {
		    factor = 0.2
			has_trait = compassionate
		}
		modifier = {
		    add = 30
			has_trait = paranoid
		}
		modifier = {
		    add = 15
			has_trait = sadistic
		}
	}  	
}

POD_heart_of_darkness_manage_hearts_interaction = {
    category = interaction_category_vampire
	#common_interaction = yes
	#interface_priority = 96
	icon = POD_serpentis_interaction
    desc = POD_heart_of_darkness_interaction_manage_hearts_desc
	
	ai_frequency = 0
	
	is_shown = {
		scope:actor = scope:recipient
		scope:recipient = { is_adult = yes }
		scope:recipient = { is_vampire_trigger = yes  }
		#scope:actor = { has_perk = heart_thief_perk }
		scope:actor = {
		    num_of_relation_heart_of_darkness_heartless > 0
		}
		scope:actor = { is_ai = no }		
	}
	
	populate_recipient_list = {
		scope:actor = {
			every_relation = {
				type = heart_of_darkness_heartless
				limit = {
				    is_alive = yes
				}
				
				add_to_list = characters
			
			}
		}
	}
	
	send_option = {	
		flag = kill
		
		is_shown = {
		    exists = scope:secondary_recipient
		}
		
		localization = POD_heart_of_darkness_kill_option
	}	
	
	send_option = {
		is_shown = {
			exists = scope:secondary_recipient
			scope:secondary_recipient = {
			    is_in_torpor_trigger = no
			}
			
		}	
		flag = forcetorpor
		localization = POD_heart_of_darkness_forcetorpor_option
	}
	
	send_option = {
		is_shown = {
			exists = scope:secondary_recipient
		}	
		flag = release
		localization = POD_heart_of_darkness_release_option
	}
	
	on_accept = {
	    
		
		if = {
		    limit = {
			    scope:kill = yes
			}
			scope:secondary_recipient = {
			    death = {
				    death_reason = death_execution
				}
			}
			scope:actor = {
			    add_piety = -1000
			}
			custom_tooltip = POD_heart_of_darkness_kill_option.tt
			
		}
		if = {
			limit = {
				scope:forcetorpor = yes
			}
            
            scope:secondary_recipient = {
                depose = yes
				add_trait = torpor
			}
			custom_tooltip = POD_heart_of_darkness_forcetorpor_option.tt
		}
		
		if = {
			limit = {
				scope:release = yes
			}
            
            scope:secondary_recipient = {
                remove_character_modifier = heart_of_darkness_modifier
				remove_opinion = {
			        target = scope:heart_attacker
                    modifier = POD_stole_heart_opinion
                }
			}
			hidden_effect = {
			    scope:secondary_recipient = {
				    remove_relation_heart_of_darkness_owner = scope:actor
				}
			}
			custom_tooltip = POD_heart_of_darkness_release_option.tt
		}
		
	}
	
	auto_accept = yes
}
