		scope:actor = { has_perk = teleport_perk }
		OR = {
			POD_teleport_interaction_army_is_shown_trigger = yes
			POD_teleport_interaction_travel_trigger = yes
		}
	}

	is_valid = {
		trigger_if = {
			limit = { POD_teleport_interaction_travel_trigger = no }
			POD_teleport_interaction_army_is_valid_trigger = yes
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = { 
			trigger_if = { 
				limit = { NOT = { has_perk = soul_burn_perk } }
				NOT = { has_character_modifier = resolve0_modifier } 
			}
		}
	}
	
	send_option = {
		is_shown = { POD_teleport_interaction_army_is_shown_trigger = yes }
		is_valid = { POD_teleport_interaction_army_is_valid_trigger = yes }
		flag = teleport_army
		localization = POD_teleport_army
	}
	
	send_option = {
		is_shown = { POD_teleport_interaction_travel_trigger = yes }
		flag = teleport_travel
		localization = POD_teleport_travel	
	}

	on_accept = {
		scope:actor = { 		
			POD_willpower_rouse_check_effect = yes
			POD_resolve_cost_effect = { COUNT = 5 }
			switch = {
				trigger = yes
				scope:teleport_army   = { POD_teleport_army_interaction_effect   = yes }
				scope:teleport_travel = { POD_teleport_travel_interaction_effect = yes }
			}
		}
	}

	ai_frequency = 0
	
	auto_accept = yes
	
	ai_will_do = {
		base = 0
	}
}

			}
		}
	}

				has_trait = demon

			} 
			has_perk = co_locate_perk
		}
		NOT = { scope:actor = scope:recipient }
		scope:actor = { is_at_war = yes }
		scope:recipient = { exists = location }
	}

	}

			}
		}
