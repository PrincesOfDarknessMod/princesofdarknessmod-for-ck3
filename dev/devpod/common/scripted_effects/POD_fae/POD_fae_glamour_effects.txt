POD_set_glamour_to_effect = {
	hidden_effect = { POD_remove_glamour_modifiers_effect = yes }
	save_scope_value_as = {
		name = clamp
		value = {
			value = $GLAMOUR$
			floor = yes
			min = 0
			max = 5 # corresponds to maximum_glamour script_value
		}
	}
	switch = {
		trigger = scope:clamp
		0 = { add_character_modifier = glamour0_modifier }
		1 = { add_character_modifier = glamour1_modifier }
		2 = { add_character_modifier = glamour2_modifier }
		3 = { add_character_modifier = glamour3_modifier }
		4 = { add_character_modifier = glamour4_modifier }
		5 = { add_character_modifier = glamour5_modifier }
	}
}

POD_glamour_change_effect = {
	if = {
		limit = { NOT = { $COUNT$ = 0 } }
		custom_description = {
			text = POD_glamour_change_desc
			value = $COUNT$
			subject = this
		}
		save_scope_value_as = {
			name = new_glamour
			value = {
				value = glamour_level
				add = $COUNT$
			}
		}
		POD_set_glamour_to_effect = { GLAMOUR = scope:new_glamour }
	}
}

POD_glamour_increase_effect = {
	POD_glamour_change_effect = { COUNT = $COUNT$ }
}

POD_glamour_decrease_effect = {
	save_scope_value_as = {
		name = glamour_cost
		value = {
			value = 0
			subtract = $COUNT$
		}
	}
	POD_glamour_change_effect = { COUNT = scope:glamour_cost }
}

POD_glamour_cost_effect = {
	save_scope_value_as = {
		name = glamour_cost
		value = {
			value = $COUNT$
			multiply = cantrip_cost_multiplier
		}
	}
	POD_energy_cost = {
		COUNT = scope:glamour_cost
		SPLAT_MAXIMUM_ENERGY = maximum_glamour
	}
}

POD_glamour_income_effect = {
	POD_energy_income = {
		COUNT = $COUNT$
		SPLAT_MAXIMUM_ENERGY = maximum_glamour
	}
}

POD_yearly_glamour_pulse_effect = {
	POD_revelry_pulse_effect = yes
	POD_reverie_pulse_effect = yes
	POD_fae_domicile_pulse_effect = yes
}

POD_revelry_pulse_effect = {
	if = {
		limit = { 
			save_temporary_scope_value_as = {
				name = biyearly_countdown
				value = { value = current_year modulo = 2 }
			}
			scope:biyearly_countdown = 0
			POD_can_use_revelry = yes
			POD_is_not_full_on_energy_including_vessels = yes

			switch = {
				trigger = has_trait_with_flag
				revelry_default = { piety_level >= 4 }
				revelry_sidhe = { is_landed = yes }
				revelry_boggan = { has_trait_xp = { trait = pod_crafter track = enhancing value >= 40 } }
				revelry_clurichaun = { stress = 0 }
				revelry_eshu = { 
					OR = {
						has_trait_xp = { trait = lifestyle_traveler track = travel value >= 80 } 
						has_trait_xp = { trait = lifestyle_traveler track = danger value >= 80 } 
					}
				}
				revelry_nocker = { has_trait_xp = { trait = pod_crafter track = forging value >= 40 } }
				revelry_piskey = { 
					calc_true_if = {
						amount = 2
						num_of_relation_best_friend >= 1
						num_of_relation_friend >= 1
						num_of_relation_friend >= 2
					}
				}
				revelry_pooka = { has_POD_shapeshifted_form_trigger = yes }
				revelry_troll = { is_landed = yes is_independent_ruler = no }
				revelry_redcap = { 
					calc_true_if = {
						amount = 2
						num_of_relation_nemesis >= 1
						num_of_relation_rival >= 1
						num_of_relation_rival >= 2
					}
				}
				revelry_satyr = { num_of_relation_lover >= 2 }
				revelry_sluagh = { 
					any_scheme = {
						count >= 1
						hostile_scheme_trigger = yes
					}
				}
				revelry_glome = {
					OR = {
						diplomacy_lifestyle_perk_points >= 1
						martial_lifestyle_perk_points >= 1
						stewardship_lifestyle_perk_points >= 1
						intrigue_lifestyle_perk_points >= 1
						learning_lifestyle_perk_points >= 1
						common_arts_lifestyle_perk_points >= 1
						noble_arts_lifestyle_perk_points >= 1
						slivers_lifestyle_perk_points >= 1
						wu_tan_lifestyle_perk_points >= 1
					}
				}
				revelry_kubera = { has_trait_xp = { trait = pod_crafter track = brewing value >= 40 } }
				revelry_ondine = { 
					location ?= {
						OR = {
							is_coastal = yes
							is_riverside_province = yes
						}
					}
				}
				revelry_solimond = { is_at_war = yes }
				revelry_mannikin = {
					OR = {
						has_trait_xp = { trait = pod_crafter track = forging value >= 40 }
						has_trait_xp = { trait = pod_crafter track = brewing value >= 40 }
						has_trait_xp = { trait = pod_crafter track = enhancing value >= 40 }
					}
				}
				revelry_kinain = { is_alive = yes }
				revelry_lilin = { num_of_relation_lover >= 2 }
			}
		}
		send_interface_message = {
			type = event_generic_good
			title = POD_revelry_trigger
			custom_tooltip = POD_revelry_trigger.tt
			POD_glamour_income_effect = { COUNT = POD_revelry_glamour }
		}
	}
}
POD_reverie_pulse_effect = {
	if = {
		limit = { 
			POD_can_use_reverie = yes
		}
		every_relation = {
			type = friend
			limit = { has_trait = mortal }
			add_to_temporary_list = reverie_list
		}
		every_relation = {
			type = best_friend
			limit = { has_trait = mortal }
			add_to_temporary_list = reverie_list
		}
		every_relation = {
			type = lover
			limit = { has_trait = mortal }
			add_to_temporary_list = reverie_list
		}
		every_relation = {
			type = soulmate
			limit = { has_trait = mortal }
			add_to_temporary_list = reverie_list
		}
		every_relation = {
			type = rival
			limit = { has_trait = mortal }
			add_to_temporary_list = reverie_list
		}
		every_relation = {
			type = nemesis
			limit = { has_trait = mortal }
			add_to_temporary_list = reverie_list
		}
		every_relation = {
			type = guardian
			limit = { has_trait = mortal }
			add_to_temporary_list = reverie_list
		}
		every_relation = {
			type = ward
			limit = { has_trait = mortal }
			add_to_temporary_list = reverie_list
		}
		every_relation = {
			type = student
			limit = { has_trait = mortal }
			add_to_temporary_list = reverie_list
		}
		save_scope_value_as = { name = glamour_gained value = 0 }
		save_scope_value_as = { name = realm_xp_gained value = 0 }
		save_scope_value_as = { name = art_xp_gained value = 0 }
		every_in_list = {
			list = reverie_list
			limit = { epiphanies_quality > 0 }
			if = {
				limit = { 
					save_temporary_scope_value_as = {
						name = biyearly_countdown
						value = { 
							value = current_year 
							modulo = {
								value = 5
								subtract = epiphanies_quality
								if = {
									limit = { PREV = { POD_improved_reverie_trigger = yes } }
									add = -1
								}
								min = 1
							} 
						}
					}
					scope:biyearly_countdown = 0
				}
				save_scope_value_as = {
					name = glamour_gained
					value = {
						value = scope:glamour_gained
						add = {
							value = 1
							multiply = PREV.multiplier_for_reverie
						}
					}
				}
				save_scope_value_as = {
					name = realm_xp_gained
					value = {
						value = scope:realm_xp_gained
						add = {
							value = 20
							multiply = PREV.multiplier_for_reverie
						}
					}
				}
				save_scope_value_as = {
					name = art_xp_gained
					value = {
						value = scope:art_xp_gained
						add = {
							value = 10
							multiply = PREV.multiplier_for_reverie
						}
					}
				}
			}
		}
		if = { 
			limit = { NOT = { scope:glamour_gained = 0 } } # This should cover everything
			send_interface_message = {
				type = event_generic_good
				title = POD_reverie_trigger
				left_icon = PREV
				custom_tooltip = POD_reverie_trigger.tt
				POD_glamour_income_effect = { COUNT = scope:glamour_gained }
				add_realm_lifestyle_xp = scope:realm_xp_gained
				add_common_arts_lifestyle_xp = scope:art_xp_gained
			}
		}
	}
}
POD_fae_domicile_pulse_effect = {
	if = {
		limit = { 
			years_to_gain_glamour_from_domicile >= 1
			save_temporary_scope_value_as = {
				name = biyearly_countdown
				value = { value = current_year modulo = years_to_gain_glamour_from_domicile }
			}
			scope:biyearly_countdown = 0
			POD_is_not_full_on_energy_including_vessels = yes
		}
		send_interface_message = {
			type = event_generic_good
			title = POD_domicile_fae
			POD_glamour_income_effect = { COUNT = 1 }
		}
	}
}
