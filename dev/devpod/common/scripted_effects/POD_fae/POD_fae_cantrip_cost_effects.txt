POD_cantrip_interaction_cost = {
	save_scope_as = cantrip_actor
	save_scope_value_as = {
		name = nature_level
		value = $NATURE_LEVEL$
	}
	$RECIPIENT$ = { save_scope_as = cantrip_recipient }
	$TARGET$   ?= { save_scope_as = cantrip_target }
	$PROVINCE$ ?= { save_scope_as = cantrip_province }
	save_scope_value_as = {
		name = cantrip_cost
		value = {
			value = $COUNT$
			if = {
				limit = { always = $REALM_ACTOR$ }
				add = cantrip_interaction_additional_realm_actor_cost
			}
			if = {
				limit = { always = $REALM_FAE$ }
				add = cantrip_interaction_additional_realm_fae_cost
			}
			if = {
				limit = { always = $REALM_NATURE$ }
				add = cantrip_interaction_additional_realm_nature_cost
			}
			if = {
				limit = { always = $REALM_PROP$ }
				add = cantrip_interaction_additional_realm_prop_cost
			}
			if = {
				limit = { always = $REALM_SCENE$ }
				add = cantrip_interaction_additional_realm_scene_cost
			}
			if = {
				limit = { always = $REALM_SCENE_AREA$ }
				add = cantrip_interaction_additional_realm_scene_area_cost
			}
			if = {
				limit = { always = $REALM_SCENE_PROVINCE$ }
				add = cantrip_interaction_additional_realm_scene_province_cost
			}
		}
	}
	if = {
		limit = { scope:cantrip_cost > $COUNT$ }
		
		if = {
			limit = { always = $REALM_ACTOR$ }
			POD_additional_realm_actor_cost_effect_tt = yes
		}
		if = {
			limit = { always = $REALM_FAE$ }
			POD_additional_realm_fae_cost_effect_tt = yes
		}
		if = {
			limit = { always = $REALM_NATURE$ }
			POD_additional_realm_nature_cost_effect_tt = yes
		}
		if = {
			limit = { always = $REALM_PROP$ }
			POD_additional_realm_prop_cost_effect_tt  = yes
		}
		if = {
			limit = { always = $REALM_SCENE$ }
			POD_additional_realm_scene_cost_effect_tt = yes
		}
		if = {
			limit = { always = $REALM_SCENE_AREA$ }
			POD_additional_realm_scene_area_cost_effect_tt = yes
		}
		if = {
			limit = { always = $REALM_SCENE_PROVINCE$ }
			POD_additional_realm_scene_province_cost_effect_tt = yes
		}
	}
	POD_glamour_cost_effect = { COUNT = scope:cantrip_cost }
}

POD_additional_realm_actor_cost_effect_tt = {
	if = {
		limit = { cantrip_interaction_additional_realm_actor_cost > 0 }
		
		custom_description = {
			text = POD_additional_realm_actor_cost_effect
			subject = scope:cantrip_actor
			object = scope:cantrip_recipient
		}
	}
}

POD_additional_realm_scene_area_cost_effect_tt = {
	if = {
		limit = { cantrip_interaction_additional_realm_scene_area_cost > 0 }
		
		if = {
			limit = { exists = scope:cantrip_target_continent }
			
			custom_description = {
				text = POD_additional_realm_scene_continent_cost_effect
				subject = scope:cantrip_actor
			}
		}
		else_if = {
			limit = { exists = scope:cantrip_target }
			
			custom_description = {
				text = POD_additional_realm_scene_area_cost_effect
				subject = scope:cantrip_actor
				object = scope:cantrip_target
			}
		}
	}
}
