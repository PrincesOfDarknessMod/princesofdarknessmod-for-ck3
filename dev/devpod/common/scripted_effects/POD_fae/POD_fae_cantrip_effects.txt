POD_stratus_modifier_apply = {
	save_scope_as = recipient
	$ACTOR$ = {
		if = {
			limit = { has_perk = slivers_stratus_perfect_perk }
			scope:recipient = { add_character_modifier = { modifier = cantrip_perfect_stratus_modifier years = 3 } }
		}
		else_if = {
			limit = { has_perk = slivers_stratus_stable_perk }
			scope:recipient = { add_character_modifier = { modifier = cantrip_stable_stratus_modifier years = 3 } }
		}
		else_if = {
			limit = { has_perk = slivers_stratus_semi_stable_perk }
			scope:recipient = { add_character_modifier = { modifier = cantrip_semi_stable_stratus_modifier years = 3 } }
		}
		else_if = {
			limit = { has_perk = slivers_stratus_unstable_perk }
			scope:recipient = { add_character_modifier = { modifier = cantrip_unstable_stratus_modifier years = 3 } }
		}
		else_if = {
			limit = { has_perk = slivers_stratus_uncontrolled_perk }
			scope:recipient = { add_character_modifier = { modifier = cantrip_uncontrolled_stratus_modifier years = 3 } }
		}
		else = {
			scope:recipient = { add_character_modifier = { modifier = cantrip_basic_stratus_modifier years = 3 } }
		}
	}
}
