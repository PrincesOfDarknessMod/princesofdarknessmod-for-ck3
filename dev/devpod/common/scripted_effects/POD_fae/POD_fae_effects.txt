POD_fae_initialization_effect = {
	if = { 
		limit = { 
			OR = { 
				exists = scope:character_being_changed
				is_from_ruler_designer = yes
			}
		}
		POD_fae_chargen_starting_xp_effect = yes
	}
	else_if = {
		limit = { has_kith_trigger = no }
		random_list = {
			50 = {
				trigger = { 
					exists = mother
					mother = {
						OR = {
							has_trait = fae
							has_trait = changeling
						}
					}
				}
				POD_custom_copy_kith_traits_effect = { ORIGIN = mother }
			}
			50 = {
				trigger = { 
					exists = real_father
					real_father = {
						OR = {
							has_trait = fae
							has_trait = changeling
						}
					}
				}
				POD_custom_copy_kith_traits_effect = { ORIGIN = real_father }
			}
		}
	}
	if = {
		limit = { has_kith_trigger = no }
		hidden_effect = { POD_character_fae_kith_initiation_effect = yes }
	}
	if = {
		limit = { 
			NOT = { religion = { is_in_family = rf_fae } } 
			is_a_cultist = no
			is_a_neteru = no
			POD_is_supehunter_trigger = no
		}
		POD_character_fae_faith_initiation_effect = yes
	}
	POD_initialize_energy_modifier = yes
	if = {
		limit = { NOT = { has_character_flag = immune_to_disease } }
		POD_remove_disease_traits_effect = yes
		add_character_flag = immune_to_disease
	}
	if = {
		limit = {
			religion = religion:spring_court_religion
			NOT = { knows_language = language_spring }
		}
		learn_language = language_spring
	}
	if = {
		limit = {
			religion = religion:summer_court_religion
			NOT = { knows_language = language_summer }
		}
		learn_language = language_summer
	}
	if = {
		limit = {
			religion = religion:autumn_court_religion
			NOT = { knows_language = language_autumn }
		}
		learn_language = language_autumn
	}
	if = {
		limit = {
			religion = religion:winter_court_religion
			NOT = { knows_language = language_winter }
		}
		learn_language = language_winter
	}
	if = {
		limit = {
			has_title = title:d_crystal_fortress
			NOT = { knows_language = language_ancient_arcadian }
		}
		learn_language = language_ancient_arcadian
	}
}

POD_set_glamour_to_effect = {
	hidden_effect = { POD_remove_glamour_modifiers_effect = yes }
	save_scope_value_as = {
		name = clamp
		value = {
			value = $GLAMOUR$
			floor = yes
			min = 0
			max = 5 # corresponds to maximum_glamour script_value
		}
	}
	switch = {
		trigger = scope:clamp
		0 = { add_character_modifier = glamour0_modifier }
		1 = { add_character_modifier = glamour1_modifier }
		2 = { add_character_modifier = glamour2_modifier }
		3 = { add_character_modifier = glamour3_modifier }
		4 = { add_character_modifier = glamour4_modifier }
		5 = { add_character_modifier = glamour5_modifier }
	}
}

POD_glamour_change_effect = {
	if = {
		limit = { NOT = { $COUNT$ = 0 } }
		custom_description = {
			text = POD_glamour_change_desc
			value = $COUNT$
			subject = this
		}
		save_scope_value_as = {
			name = new_glamour
			value = {
				value = glamour_level
				add = $COUNT$
			}
		}
		POD_set_glamour_to_effect = { GLAMOUR = scope:new_glamour }
	}
}

POD_copy_artifact_effect = {
	save_scope_as = owner
	POD_copy_artifact_rarity_effect = { TARGET = scope:target }
	
	POD_artifact_type_injector = {
		ARTIFACT = scope:target
		SCOPE    = scope:owner
		EFFECT   = POD_copy_artifact_final_effect
	}
	
	hidden_effect_new_object = { 
		scope:newly_created_artifact = {
			remove_artifact_modifier = artifact_placeholder_modifier
			copy_artifact_modifiers = scope:target
			set_variable = summoned_artifact
			set_max_durability = 10
			flag_as_trash_artifact = yes
			set_artifact_name = copied_artifact_name
		}
		if = {
			limit = { scope:target = { has_variable = enhance_amount } }
			scope:newly_created_artifact = {
				set_variable = {
					name = enhance_amount
					value = scope:target.var:enhance_amount
				}
			}
		}
	}
}

		visuals = $VISUALS$
		wealth = scope:wealth
		quality = scope:quality
		modifier = artifact_placeholder_modifier
		save_scope_as = newly_created_artifact
	}
}

POD_create_random_inspiration_effect = {
	if = {
		limit = { NOT = { exists = inspiration } }
		save_temporary_scope_as = inspired_character
		# More likely to get an certain Inspiration type if we will produce higher quality Artifacts of that type.
		random_list = {
			10 = {
				compare_modifier = {
					value = weapon_inspiration_average_skill_value
					multiplier = 6
				}
				modifier = { #Some cultures are more likely to pick this vocation than others
					add = 5
					culture = {
						has_cultural_tradition = tradition_metal_craftsmanship
					}
				}
				create_inspiration = weapon_inspiration
				custom_tooltip = POD_inspire_weapon
			}
			10 = { # =Slightly increased weight due to a low number of Armorers appearing in the world.
				compare_modifier = {
					value = armor_inspiration_average_skill_value
					multiplier = 7.2 # A character with an average of 20 in relevant skills will receive a +120 bonus
				}
				modifier = { #Some cultures are more likely to pick this vocation than others
					add = 5
					culture = {
						has_cultural_tradition = tradition_frugal_armorsmiths
					}
				}
				create_inspiration = armor_inspiration
				custom_tooltip = POD_inspire_armor
			}
			10 = { #Adventurers will on average produce better quality items, handle with care.
				compare_modifier = {
					value = adventure_inspiration_average_skill_value
					multiplier = 9
				}
				modifier = {
					add = 5
					culture = { #Some cultures are more likely to pick this vocation than others
						OR = {
							has_cultural_parameter = more_likely_to_leave_court #Swords for Hire & Caravaneers
							has_cultural_tradition = tradition_chivalry
						}
					}
				}
				create_inspiration = adventure_inspiration
				custom_tooltip = POD_inspire_adventure
			}
			12 = { # Significantly increased weight due to the wide variety of artifact types this inspiration can produce (regalia, crown, trinket).
				compare_modifier = {
					value = smith_inspiration_average_skill_value
					multiplier = 15
				}
				modifier = { #Some cultures are more likely to pick this vocation than others
					add = 10
					culture = {
						OR = {
							has_cultural_tradition = tradition_metal_craftsmanship
							has_cultural_tradition = tradition_ancient_miners
						}
					}
				}
				create_inspiration = smith_inspiration
				custom_tooltip = POD_inspire_smith
			}
			10 = {
				compare_modifier = {
					value = book_inspiration_average_skill_value
					multiplier = 6
				}
				modifier = {
					has_court_position = court_poet_court_position
					add = 15
				}
				modifier = {
					culture = { #Some cultures are more likely to pick this vocation than others
						OR = {
							has_cultural_tradition = tradition_language_scholars
							has_cultural_tradition = tradition_philosopher_culture
							has_cultural_tradition = tradition_legalistic
						}
					}
					add = 5
				}
				create_inspiration = book_inspiration
				custom_tooltip = POD_inspire_book
			}
			11 = { # Increased weight due to a low number of Weavers appearing in the world (probably a lack of high-diplomacy courtiers).
				compare_modifier = {
					value = weaver_inspiration_average_skill_value
					multiplier = 8.4
				}
				modifier = {
					culture = { #Some cultures are more likely to pick this vocation than others
						has_cultural_parameter = improved_weaver_inspiration
					}
					add = 10
				}
				create_inspiration = weaver_inspiration
				custom_tooltip = POD_inspire_weaver
			}
			14 = { # Moderately increased weight due to the variety of artifact types this inspiration can produce (throne, sculpture). 
				compare_modifier = {
					value = artisan_inspiration_average_skill_value
					multiplier = 12
				}
				modifier = {
					has_court_position = court_poet_court_position
					add = 5
				}
				modifier = {
					culture = { #Some cultures are more likely to pick this vocation than others
						has_cultural_parameter = improved_artisan_inspiration
					}
					add = 15
				}
				create_inspiration = artisan_inspiration
				custom_tooltip = POD_inspire_artisan
			}
			1 = {
				modifier = {
					add = 30
					has_trait = lifestyle_mystic
				}
				modifier = { #Alchemy arrived to Europe in the 13th century, but existed basically everywhere else before that
					add = -20
					culture_has_european_heritage_pillar_trigger = yes
				}
				compare_modifier = {
					value = learning
					multiplier = 2
				}
				create_inspiration = alchemy_inspiration
				custom_tooltip = POD_inspire_alchemy
			}
		}
	}
}

POD_changeling_effect = {
	$ADOPTER$ = {
		adopt = $CHILD$
		$CHILD$ = {
			set_house = $ADOPTER$.house
			# Remove bastard traits
			if = {
				limit = {
					has_any_bastard_trait_trigger = yes
				}
				remove_all_bastard_traits = yes
			}
			if = {
				limit = { exists = $ADOPTER$.capital_province } # Ward heads to landed Guardian
				$ADOPTER$.capital_province = { save_scope_as = destination_province }
			}
			else_if = { # Ward heads to Guardian's host
				limit = { exists = $ADOPTER$.host.capital_province }
				$ADOPTER$.host.capital_province = { save_scope_as = destination_province }
			}
			else = {
				$ADOPTER$.location = { save_scope_as = destination_province }
			}
			save_scope_as = adoptee
			$ADOPTER$ = {
				save_scope_as = adopter
			}
			if = {
				limit = {
					any_relation = {
						type = guardian
						NOT = { this = $ADOPTER$ }
					}
				}
				random_relation = {
					type = guardian
					limit = { NOT = { this = $ADOPTER$ } }
					save_scope_as = old_guardian
				}
				remove_guardian_effect = {
					GUARDIAN = scope:old_guardian
					WARD = scope:adoptee
					RETURN_WARD = no
					HIDE_OPINION = no
				}
			}
			if = {
				limit = { NOT = { scope:destination_province = $CHILD$.location } }
				if = {
					limit = { $ADOPTER$ = { is_ruler = yes } }
					visit_court_of = $ADOPTER$
				}
				else = {
					visit_court_of = $ADOPTER$.liege_or_court_owner
				}
			}
			
			# SPLAT #
			if = {
				limit = { NOT = { has_trait = changeling } }
				remove_trait = fae
				POD_become_changeling_effect = yes
			}
		}
	}
}
