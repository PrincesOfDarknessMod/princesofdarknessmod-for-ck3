POD_fae_initialization_effect = {
	if = { 
		limit = { 
			OR = { 
				exists = scope:character_being_changed
				is_from_ruler_designer = yes
			}
		}
		POD_fae_chargen_starting_xp_effect = yes
	}
	else_if = {
		limit = { has_kith_trigger = no }
		random_list = {
			50 = {
				trigger = { 
					exists = mother
					mother = {
						OR = {
							has_trait = fae
							has_trait = changeling
						}
					}
				}
				POD_custom_copy_kith_traits_effect = { ORIGIN = mother }
			}
			50 = {
				trigger = { 
					exists = real_father
					real_father = {
						OR = {
							has_trait = fae
							has_trait = changeling
						}
					}
				}
				POD_custom_copy_kith_traits_effect = { ORIGIN = real_father }
			}
		}
	}
	if = {
		limit = { has_kith_trigger = no }
		hidden_effect = { POD_character_fae_kith_initiation_effect = yes }
	}
	if = {
		limit = { 
			NOT = { religion = { is_in_family = rf_fae } } 
			is_a_cultist = no
			is_a_neteru = no
			POD_is_supehunter_trigger = no
		}
		POD_character_fae_faith_initiation_effect = yes
	}
	POD_initialize_energy_modifier = yes
	if = {
		limit = { NOT = { has_character_flag = immune_to_disease } }
		POD_remove_disease_traits_effect = yes
		add_character_flag = immune_to_disease
	}
	if = {
		limit = { has_trait = kithain }
		if = {
			limit = {
				religion = religion:spring_court_religion
				NOT = { knows_language = language_spring }
			}
			learn_language = language_spring
		}
		if = {
			limit = {
				religion = religion:summer_court_religion
				NOT = { knows_language = language_summer }
			}
			learn_language = language_summer
		}
		if = {
			limit = {
				religion = religion:autumn_court_religion
				NOT = { knows_language = language_autumn }
			}
			learn_language = language_autumn
		}
		if = {
			limit = {
				religion = religion:winter_court_religion
				NOT = { knows_language = language_winter }
			}
			learn_language = language_winter
		}
		if = {
			limit = {
				has_title = title:d_crystal_fortress
				NOT = { knows_language = language_ancient_arcadian }
			}
			learn_language = language_ancient_arcadian
		}
	}
	if = {
		limit = {
			has_trait_with_flag = affected_by_banality
			NOT = { has_trait = banality }
		}
		add_trait_force_tooltip = banality
	}
}

POD_change_banality_effect = {
	if = {
		limit = { has_trait = banality }
		if = {
			limit = { $VALUE$ >= 0 }
			save_scope_value_as = {
				value = {
					value = $VALUE$
					multiply = banality_gain_mult
				}
				name = banality_gain
			}
		}
		else = {
			save_scope_value_as = {
				value = {
					value = $VALUE$
					# multiply = banality_loss_mult
				}
				name = banality_gain
			}
		}
		custom_description = {
			text = POD_change_banality
			value = scope:banality_gain
			add_trait_xp = {
				trait = banality
				value = scope:banality_gain
			}
		}
		if = {
			limit = { fae_era = 2 }
			if = {
				limit = { NOT = { exists = global_var:fae_killed_by_banality } }
				set_global_variable = {
					name = fae_killed_by_banality
					value = {
						every_living_character = {
							limit = { has_trait_with_flag = affected_by_banality }
							add = 1
						}
						divide = 4
					}
				}
			}
		}
		else = { remove_global_variable = fae_killed_by_banality }
		if = {
			limit = {  
				save_temporary_scope_value_as = {
					value = {
						value = "this.has_trait_xp(banality)"
						add = scope:banality_gain
					}
					name = total_banality
				}
				scope:total_banality >= 100
			}
			change_global_variable = {
				name = fae_killed_by_banality
				subtract = 1
			}
			if = { 
				limit = { 
					global_var:fae_killed_by_banality <= 1 # Remember about subtracting 1
					NOT = {
						is_target_in_global_variable_list = {
							name = unavailable_unique_decisions
							target = flag:POD_war_of_seasons_war_victory_objective
						}
					}
					NOT = {
						is_target_in_global_variable_list = {
							name = unavailable_unique_decisions
							target = flag:POD_war_of_seasons_peace_victory_objective
						}
					}
				} 
				trigger_event = POD_fae_maintenance.20
			}
			
			death = {
				death_reason = death_unmade
			}
		}
	}
}
