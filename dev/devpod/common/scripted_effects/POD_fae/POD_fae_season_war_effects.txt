#SEASON - Religion
POD_pulse_season_leader = {
	switch = {
		trigger = $SEASON$
		religion:spring_court_religion = {
			ordered_ruler = {
				limit = {
					faith = { POD_is_faith_spring_court = yes }
					has_POD_fae_blood_trigger = yes
				}
				max = 1
				order_by = {
					value = 1
					add = popular_opinion_spring
				}
				save_scope_as = spring_ruler
			}
			custom_tooltip = {
				text = POD_pulse_season_leader_spring
				POD_become_season_leader_effect = { COURT = spring CHAR = scope:spring_ruler }
			}
		}
		religion:summer_court_religion = {
			ordered_ruler = {
				limit = {
					faith = { POD_is_faith_summer_court = yes }
					has_POD_fae_blood_trigger = yes
				}
				max = 1
				order_by = {
					value = {
						value = 0
						every_vassal = { add = 1 }
					}
				}
				save_scope_as = summer_ruler
			}
			if = {
				limit = { situation:PoD_war_of_seasons.var:summer_leader != scope:summer_ruler }
				POD_become_season_leader_effect = { COURT = summer CHAR = scope:summer_ruler }
			}
		}
		religion:autumn_court_religion = {
			ordered_ruler = {
				limit = {
					faith = { POD_is_faith_autumn_court = yes }
					has_POD_fae_blood_trigger = yes
				}
				max = 1
				order_by = {
					value = 0
					add = intrigue
				}
				save_scope_as = autumn_ruler
			}
			if = {
				limit = { situation:PoD_war_of_seasons.var:autumn_leader != scope:autumn_ruler }
				POD_become_season_leader_effect = { COURT = autumn CHAR = scope:autumn_ruler }
			}
		}
		religion:winter_court_religion = {
			ordered_ruler = {
				limit = {
					faith = { POD_is_faith_winter_court = yes }
					has_POD_fae_blood_trigger = yes
				}
				max = 1
				order_by = {
					value = 0
					add = dread
				}
				save_scope_as = winter_ruler
			}
			if = {
				limit = { situation:PoD_war_of_seasons.var:winter_leader != scope:winter_ruler }
				POD_become_season_leader_effect = { COURT = winter CHAR = scope:winter_ruler }
			}
		}
		religion:solstice_court_religion = {
			if = {
				limit = { exists = primary_heir }
				primary_heir = { save_scope_as = solstice_ruler }
			}
			else = {
				random_ruler = { save_scope_as = solstice_ruler }
			}
			if = {
				limit = { situation:PoD_war_of_seasons.var:solstice_leader != scope:solstice_ruler }
				POD_become_season_leader_effect = { COURT = solstice CHAR = scope:solstice_ruler }
			}
		}
	}
}

POD_become_season_leader_effect = {
	POD_save_tooltip = { LOC = $COURT$ }
	$CHAR$ ?= { 
		save_scope_as = character 
		custom_tooltip = {
			text = POD_become_season_leader_effect
			situation:PoD_war_of_seasons = { 
				set_variable = { 
					name = $COURT$_leader 
					value = $CHAR$ 
				} 
			}
		}
	}
	hidden_effect = {
		every_player = {
			limit = { PoD_is_member_of_seasonal_courts = yes }
			send_interface_message = {
				title = PoD_war_of_seasons
				$CHAR$ ?= { custom_tooltip = POD_become_season_leader_effect }
			}
		}
	}
}

POD_claim_leadership_scheme_prep_effect = {
	save_scope_as = scheme
	save_scope_value_as = {
		name = follow_up_event
		value = event_id:POD_claimleadership_events.1 
	}
	if = {
		limit = {
			NOT = { exists = scope:suppress_next_event }
		}
		scheme_owner = { trigger_event = scheme_critical_moments.0002 }
	}
}

POD_claim_leadership_outcome_roll_effect = {
	#SUCCESS ROLL
	random = {
		chance = scope:scheme.scheme_success_chance
		custom_tooltip = POD_claim_leadership_successful_roll_tt
		scope:owner = { mandala_trickster_increment_successful_schemes_effect = yes }
		save_scope_value_as = {
			name = scheme_successful
			value = yes
		}
	}
	### End Success roll
	#DISCOVERY ROLL
	custom_description_no_bullet = {
		text = claim_throne_success_discovery_warning_effect
	}
	random = {
		chance = scope:discovery_chance
		custom_tooltip = POD_claim_leadership_become_discovered_roll_tt
		save_scope_value_as = {
			name = scheme_discovered
			value = yes
		}
	}
	### End Discovery roll

	hidden_effect = {
		#FIRE CORRECT ON ACTIONS
		if = {
			limit = {
				exists = scope:scheme_successful
			}
			trigger_event = POD_claimleadership_events.2
		}
		else = {
			trigger_event = POD_claimleadership_events.3
		}
	}
}

#Used in the immediate of the Scheme Owner's success event
POD_claim_leadership_success_effect = {
	if = {
		limit = {
			exists = scope:scheme_discovered
		}
		# Trigger the response event for the target (actual opinion is set here)
		scope:target = {
			trigger_event = POD_claimleadership_events.2
		}
		
		show_as_tooltip = { # Target dislikes you
			scope:target = {
				add_opinion = {
					target = scope:owner
					modifier = claim_my_throne_success_discovered_opinion
				}
			}
		}
	}
	else = {
		# Trigger the response event for the target (actual opinion is set here)
		scope:target = {
			trigger_event = POD_claimleadership_events.2
		}
		show_as_tooltip = {
			scope:target = {
				add_opinion = {
					target = scope:owner
					modifier = claim_my_throne_success_undiscovered_opinion
					years = 10
				}
			}
		}
	}
}

POD_claim_leadership_failure_effect = {
	add_prestige = -500
	scope:target = {
		# Fire notification event for target
		trigger_event = POD_claimleadership_events.3
		show_as_tooltip = {
			add_opinion = {
				target = scope:owner
				modifier = claim_my_throne_failure_opinion
				years = 10
			}
			if = {
				limit = { exists = scope:set_rivalry }
				set_relation_rival = {
					target = scope:owner
					reason = rival_claimed_throne_failed
				}
			}
		}
	}
}

successful_POD_claim_leadership_event_option_effect = {
	if = {
		limit = {
			exists = scope:scheme_discovered
		}
		add_prestige = -500
	}
	show_as_tooltip = {
		scope:target = {
			if = {
				limit = { exists = scope:set_rivalry }
				set_relation_rival = {
					target = scope:owner
					reason = rival_claimed_throne
				}
			}
			if = {
				limit = { exists = scope:scheme_discovered }
				add_opinion = {
					target = scope:owner
					modifier = claim_my_throne_success_discovered_opinion
				}
			}
			else = {
				add_opinion = {
					target = scope:owner
					modifier = claim_my_throne_success_undiscovered_opinion
					years = 10
				}
			}
		}
	}

	scope:scheme = { end_scheme = yes }
	save_scope_as = character
	POD_become_season_leader_effect = {
		COURT = autumn
		CHAR = scope:character
	}
}

POD_war_of_seasons_peace_victory_objective_effect = {
	situation:PoD_war_of_seasons = { end_situation = yes }
	add_prestige = massive_prestige_value
	dynasty = { add_dynasty_prestige = excessive_dynasty_prestige_value }
	custom_tooltip = POD_war_of_seasons_peace_victory_objective_effect0
	custom_tooltip = POD_war_of_seasons_peace_victory_objective_effect1
	add_character_modifier = seasons_truce_bringer_modifier
	if = {
		limit = { is_landed = yes }
		hidden_effect = {
			every_held_title = { #Should shift all dejure of all Empires owned at the time.
				limit = {
					tier = tier_empire
				}
				every_in_de_jure_hierarchy = {
					limit = {
						tier = tier_kingdom
					}
					set_de_jure_liege_title = title:h_united_seasons

				}
			}
			every_held_title = { #Completely Controlled Kingdoms as well.
				limit = {
					tier = tier_kingdom
					root = { completely_controls = prev }
				}
				set_de_jure_liege_title = title:h_united_seasons
			}
		}
		create_title_and_vassal_change = {
			type = created
			save_scope_as = change
			add_claim_on_loss = no
		}
		title:h_united_seasons = {
			change_title_holder = {
				holder = root
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change
		hidden_effect = { set_primary_title_to = title:h_united_seasons }
		every_held_title = {
			limit = {
				tier = tier_hegemony
				NOT = { this = title:h_united_seasons }
			}
			root = { destroy_title = prev }
		}
	}
	else = {
		custom_tooltip = POD_war_of_seasons_peace_victory_objective_effect2
	}
	custom_tooltip = unlock_arcadia_restoration
}
