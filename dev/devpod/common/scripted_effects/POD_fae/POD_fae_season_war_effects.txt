winter_victory_ender_effect = {
	dynasty = { add_dynasty_prestige = 10000 }
	give_nickname = nick_herald_of_winter
	add_character_modifier = herald_of_winter_modifier
	custom_tooltip = winter_victory_doctrine_effect_tooltip
	
	hidden_effect = {
		religion:winter_court_religion = {
			every_faith = {
				add_doctrine = doctrine_winter_time
			}
		}
		religion:spring_court_religion = {
			every_faith = {
				remove_doctrine = doctrine_spring_time
			}
		}
		religion:summer_court_religion = {
			every_faith = {
				remove_doctrine = doctrine_summer_time
			}
		}
		religion:autumn_court_religion = {
			every_faith = {
				remove_doctrine = doctrine_autumn_time
			}
		}
		set_global_variable = end_seasons_rotate
	
		every_held_title = { #Should shift all dejure of all Empires owned at the time.
			limit = {
				tier >= tier_empire
			}
			every_in_de_jure_hierarchy = {
				limit = {
					tier = tier_kingdom
				}
				set_de_jure_liege_title = title:e_winter_court
				
			}
		}
		every_held_title = { #Completely Controlled Kingdoms as well.
			limit = {
				tier = tier_kingdom
				root = { completely_controls = prev }
			}
			set_de_jure_liege_title = title:e_winter_court
		}
		title:k_ireland = { set_de_jure_liege_title = title:e_winter_court }
		title:e_spring_court = {
			every_in_de_jure_hierarchy = {
				limit = { tier = tier_kingdom }
				set_de_jure_liege_title = title:e_season_courts
			}
		}
		title:e_summer_court = {
			every_in_de_jure_hierarchy = {
				limit = { tier = tier_kingdom }
				set_de_jure_liege_title = title:e_season_courts
			}
		}
		title:e_autumn_court = {
			every_in_de_jure_hierarchy = {
				limit = { tier = tier_kingdom }
				set_de_jure_liege_title = title:e_season_courts
			}
		}
	}
	
	destroy_title = title:e_spring_court
	destroy_title = title:e_summer_court
	destroy_title = title:e_autumn_court
	religion:spring_court_religion = {
		every_faith = {
			limit = { has_doctrine = doctrine_temporal_head }
			remove_doctrine = doctrine_temporal_head
			add_doctrine = doctrine_no_head
		}
	}
	religion:summer_court_religion = {
		every_faith = {
			limit = { has_doctrine = doctrine_temporal_head }
			remove_doctrine = doctrine_temporal_head
			add_doctrine = doctrine_no_head
		}
	}
	religion:autumn_court_religion = {
		every_faith = {
			limit = { has_doctrine = doctrine_temporal_head }
			remove_doctrine = doctrine_temporal_head
			add_doctrine = doctrine_no_head
		}
	}
	
	every_held_title = { #Should destroy all other Empires owned at the time.
		limit = {
			tier >= tier_empire
			NOT = { this = title:e_winter_court }
		}
		root = { destroy_title = prev }
	}

	add_to_global_variable_list = {
		name = unavailable_unique_decisions
		target = flag:seasons_struggle_objective_decision
	}
}

seasons_struggle_compromise_modifier_rewards_effect = {
	struggle:seasons_struggle = {
		show_as_tooltip = {
			every_involved_ruler = {
				limit = { this = root }
				custom = fp2_struggle_compromise_house_rewards_tt
				seasons_struggle_compromise_modifier_rewards_personal_house_effect = yes
			}
			every_involved_ruler = {
				limit = { this = root }
				custom = fp2_struggle_compromise_defensive_rewards_tt
				seasons_struggle_compromise_modifier_rewards_personal_county_effect = yes
			}
		}
		hidden_effect = {
			every_involved_ruler = {
				limit = {
					is_independent_ruler = yes
					primary_title = { is_mercenary_company = no }
					exists = house
					NOT = {
						house = { has_house_modifier = fp2_struggle_compromise_house_reward_modifier }
					}
				}
				seasons_struggle_compromise_modifier_rewards_personal_house_effect = yes
			}
			every_involved_ruler = {
				limit = {
					is_independent_ruler = yes
					primary_title = { is_mercenary_company = no }
				}
				seasons_struggle_compromise_modifier_rewards_personal_county_effect = yes
			}
		}
	}

	add_to_global_variable_list = {
		name = unavailable_unique_decisions
		target = flag:seasons_struggle_objective_decision
	}
}
