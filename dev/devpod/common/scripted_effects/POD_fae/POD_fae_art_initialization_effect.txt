POD_fae_dynasty_trait_assign_effect = {
	if = {
		limit = {
			exists = dynasty
			dynasty = { has_dynasty_perk = seelie_1_legacy_5 }
			NOT = { has_trait = lifestyle_poet }
		}
		add_trait = lifestyle_poet
	}
	if = {
		limit = {
			exists = dynasty
			dynasty = { has_dynasty_perk = seelie_3_legacy_4 }
			NOT = { has_trait = lifestyle_mystic }
		}
		add_trait = lifestyle_mystic
	}
	if = {
		limit = {
			exists = dynasty
			dynasty = { has_dynasty_perk = seelie_3_legacy_5 }
		}
		if = {
			limit = {
				dynasty = { exists = var:seelie_3_legacy_5_lifestyle_blademaster }
				NOT = { has_trait = lifestyle_blademaster }
			}
			add_trait = lifestyle_blademaster
		}
		else_if = {
			limit = {
				dynasty = { exists = var:seelie_3_legacy_5_lifestyle_hunter }
				NOT = { has_trait = lifestyle_hunter }
			}
			add_trait = lifestyle_hunter
		}
		else_if = {
			limit = {
				dynasty = { exists = var:seelie_3_legacy_5_lifestyle_physician }
				NOT = { has_trait = lifestyle_physician }
			}
			add_trait = lifestyle_physician
		}
		else_if = {
			limit = {
				dynasty = { exists = var:seelie_3_legacy_5_lifestyle_herbalist }
				NOT = { has_trait = lifestyle_herbalist }
			}
			add_trait = lifestyle_herbalist
		}
		else_if = {
			limit = {
				dynasty = { exists = var:seelie_3_legacy_5_lifestyle_gardener }
				NOT = { has_trait = lifestyle_gardener }
			}
			add_trait = lifestyle_gardener
		}
	}
}
