# Run after every mental break event
# Before vanilla effects, so the event flags are still present
POD_stress_threshold_event_aftereffects = {
	if = {
		limit = { POD_splat_can_have_wyrm_taint = yes }
		if = {
			limit = { has_character_flag = apply_stress_threshold_cooldown_level_3 }
			POD_change_wyrm_taint_effect = { COUNT = wyrm_taint_massive_value }
		}
		else_if = {
			limit = { has_character_flag = apply_stress_threshold_cooldown_level_2 }
			POD_change_wyrm_taint_effect = { COUNT = wyrm_taint_medium_value }
		}
		else = {
			POD_change_wyrm_taint_effect = { COUNT = wyrm_taint_minor_value }
		}
	}
	
	if = {
		limit = {
			has_trait = dharma
		}
		POD_diao_effect = yes
	}
	
	if = {
		limit = { POD_is_on_journey_trigger = { TYPE = breaking_the_blond_bond } }
		POD_abandon_journeys_of_type_effect = { TYPE = breaking_the_blond_bond }
	}
	
	#POD_possessed_stress_threshold_event_aftereffects = yes
}

POD_give_additional_education_level_effect = {
	if = {
		limit = { faith ?= { has_doctrine_parameter = codified_teachings_parameter } }
		POD_give_additional_education_level_macro = { VALUE = 1 }
	}
}
POD_give_additional_education_level_macro = {
	if = {
		limit = { has_focus = education_diplomacy }
		change_variable = { name = education_diplomacy_variable add = $VALUE$ }
	}
	else_if = {
		limit = { has_focus = education_martial }
		change_variable = { name = education_martial_variable add = $VALUE$ }
	}
	else_if = {
		limit = { has_focus = education_stewardship }
		change_variable = { name = education_stewardship_variable add = $VALUE$ }
	}
	else_if = {
		limit = { has_focus = education_intrigue }
		change_variable = { name = education_intrigue_variable add = $VALUE$ }
	}
	else_if = {
		limit = { has_focus = education_learning }
		change_variable = { name = education_learning_variable add = $VALUE$ }
	}
}
