POD_clean_yinyang_effect = {
	remove_trait = yin_imbalanced
	remove_trait = yang_imbalanced
	remove_trait = yinyang_balanced
}

POD_set_yinyang_effect = {

	random_list = {
		2 = {
			modifier = {
				add = 50
				has_religion = religion:thousandwhispers_religion
			}
			add_trait = yin_imbalanced
			add_character_flag = immune_to_disease
		}
		2 = {
			modifier = {
				add = 50
				has_religion = religion:boneflowers_religion
			}
			add_trait = yang_imbalanced
		}
		1 = {
			modifier = {
				add = 50
				has_religion = religion:thrashingdragons_religion
			}
			add_trait = yinyang_balanced
			add_character_flag = immune_to_disease
		}
	}
}

POD_turn_to_yin_effect = {
		
	hidden_effect = {
		if = {
			limit = {
				has_trait = yinyang_balanced
			}
			remove_trait = yinyang_balanced
		}
		if = {
			limit = {
				has_trait = yang_imbalanced
			}
			remove_trait = yang_imbalanced
		}
		remove_disease_trait_effect = yes
		add_character_flag = immune_to_disease
	}
	
	add_trait = yin_imbalanced
}

POD_turn_to_yang_effect = {
	
	hidden_effect = {
		if = {
			limit = {
				has_trait = yinyang_balanced
			}
			remove_trait = yinyang_balanced
		}
		if = {
			limit = {
				has_trait = yin_imbalanced
			}
			remove_trait = yin_imbalanced
		}
		remove_character_flag = immune_to_disease
	}

	add_trait = yang_imbalanced
}

POD_turn_to_balanced_effect = {
	
	hidden_effect = {
		if = {
			limit = {
				has_trait = yang_imbalanced
			}
			remove_trait = yang_imbalanced
		}
		if = {
			limit = {
				has_trait = yin_imbalanced
			}
			remove_trait = yin_imbalanced
		}
		remove_disease_trait_effect = yes
		add_character_flag = immune_to_disease
	}
	
	add_trait = yinyang_balanced
}

POD_check_yinyang_effect = {

	if = {
		limit = {
			yin_imbalanced_trigger = yes
		}
		POD_turn_to_yin_effect = yes
	}
	else_if = {
		limit = {
			yang_imbalanced_trigger = yes
		}
		POD_turn_to_yang_effect = yes
	}
	else = {
		POD_turn_to_balanced_effect = yes
	}
	
	POD_chi_hunger_effect = yes
	POD_chi_fill_effect = yes
	POD_yinchi_overdose_effect = yes
	POD_yangchi_overdose_effect = yes
}
