		}
	}
	
	every_relation = {
		type = ghoul_domitor
		remove_hook = {

		scope:victim = {
			remove_relation_ghoul_domitor = prev
			remove_character_flag = blocked_from_leaving
			remove_character_flag = blood_bonded
		}
	}
	
	every_relation = {
		type = demonic_master
		scope:victim = {
			remove_relation_demonic_master = prev
		}
		if = {
			limit = {
				has_variable = host
				var:host = scope:victim
			}
			scope:victim = { save_scope_as = dead_character }
			send_interface_message = {
				type = POD_event_days_of_fire
				title = host_death_notification_title
				desc = death_management.host.desc
				left_icon = scope:dead_character
