		# Resolve Blood Hunger Slaking from Sip
		if = {
			limit = {
				OR = {
					has_character_modifier = bloodpotency0_modifier
					has_character_modifier = bloodpotency1_modifier

					has_character_modifier = bloodhunger2_modifier
					has_character_modifier = bloodhunger3_modifier
					has_character_modifier = bloodhunger4_modifier
					has_character_modifier = bloodhunger5_modifier
				}
			}
			if = {
				limit = {
					has_character_modifier = bloodhunger2_modifier
				}
					hidden_effect = {
						remove_character_modifier = bloodhunger2_modifier

					add_character_modifier = {
						modifier = bloodhunger1_modifier					
					}
			}
			if = {
				limit = {
					has_character_modifier = bloodhunger3_modifier

					limit = { has_perk = quicken_the_mortals_blood_perk }
					add_character_modifier = bloodhunger1_modifier
				}
				else = {
					add_character_modifier = {
						modifier = bloodhunger2_modifier
					}
				}
			}	
			if = {
				limit = {

POD_hunger_feeding_sip_from_victim_effect = {
	if = {
		limit = { 
			OR = { 
				NOT = { has_trait = methuselahs_thirst }
				$VICTIM$ = { has_supernatural_race_trigger = yes }
			}
			OR = { 
				can_eat_wraith_trigger = yes
				$VICTIM$ = { can_be_killed_physically_trigger = yes }
			}
		}
		if = {
			limit = { $VICTIM$ = { POD_chi_user_trigger = no } }
			POD_hunger_feeding_sip_effect = yes
		}	
		else = {
			if = {
				limit = { $VICTIM$ = { chi_points_value >= 2 } }
				$VICTIM$ = { POD_chi_decrease_pre_effect = { COUNT = 2 } }
				POD_hunger_feeding_sip_effect = yes
			}
		}
	}
		
	if = {
		limit = {
			$VICTIM$ = {
				OR = {
					has_trait = werewolf		#Increased chance of frenzy (stress gain maybe?)
					has_trait = fae				#Highly random effects ranging from torpor to stat changes or being blood bound to the fae. At least in lore. Who knows what if anything we end up with?
					has_trait = changeling
				}
			}
		}
		add_stress = minor_stress_gain
	}
		
	if = {
		limit = {
			$VICTIM$ = {
				has_trait = mummy
			}
		}
		if = {
			limit = {
				faith = { 
					OR = {
						has_doctrine = doctrine_enlightenment_virtues_conscience_and_selfcontrol
						has_doctrine = doctrine_enlightenment_virtues_conscience_and_instinct
					}
				}
			}
			add_piety = minor_piety_gain
		}
		else = {
			add_piety = minor_piety_loss
		}
		add_character_modifier = {
			modifier = lifeblood_addiction_modifier
			years = 1
		}
	}
}
