POD_become_yamaking_effect = {
	save_temporary_scope_value_as = {
		name = saved_immortal_age
		value = POD_visual_age
	}
	POD_save_energy_level_before_splat_change = yes
	hidden_effect = { POD_remove_splat_traits_effect = yes }
	add_trait_force_tooltip = yamaking
	if = {
		limit = { exists = scope:saved_immortal_age }
		hidden_effect = { set_immortal_age = scope:saved_immortal_age }
	}
	hidden_effect = {
		POD_remove_wrong_supernatural_traits_on_existing_character = yes
		POD_remove_supernatural_relation_effect = yes
		POD_side_effects_from_changing_splats = yes
	}
	hidden_effect = {
		POD_yamaking_initialization_effect = yes
	}
	if = {
		limit = { is_ai = no }
		trigger_event = podintroductions.5102
	}
}
POD_yamaking_initialization_effect = {
	if = {
		limit = {
			OR = {
				exists = scope:character_being_changed
				is_from_ruler_designer = yes
			}
		}
		POD_change_government_if_landed = { GOVERNMENT = yomi_wan_government }
	}
	if = {
		limit = {
			is_a_cultist = no
			POD_is_supehunter_trigger = no
			NOT = { religion = religion:thousandhells_religion }
		}
		set_character_faith = faith:wayofthetenthousandscreams
	}
	POD_assing_age_modifier_effect = yes
	POD_update_yamaking_barrier = yes
	POD_setup_contested_yomi_effect = yes
	add_character_flag = immune_to_disease
}

POD_update_yamaking_barrier = {
	every_realm_county = {
		if = {
			limit = {
				title_province ?= { NOT = { POD_province_is_in_middle_kingdom_trigger = yes } }
				POD_is_middle_kingdom_breached_trigger = no
			}
			add_county_modifier = POD_yamaking_outside_middle_kingdom_modifier
			change_county_control = -100
		}
		else = {
			remove_county_modifier = POD_yamaking_outside_middle_kingdom_modifier
		}
	}
}

POD_yearly_defiled_xp_pulse = {
	send_interface_message = {
		type = POD_event_yamaking
		title = POD_yamaking_maintenance.1.t
		desc = POD_yamaking_maintenance.1.desc
		left_icon = root
		add_panoplies_lifestyle_xp = ROOT.panoply_xp_gain_value
		dynasty ?= { add_dynasty_prestige = ROOT.yamaking_renown_gain_value }
	}
}

POD_grant_defilement_generalized_cost = {
	if = {
		limit = {
			OR = {
				defilement_of_$TYPE$_granting_cost = 1000
			}
		}
		add_panoplies_lifestyle_perk_points = -1
	}
	else_if = {
		limit = {
			panoplies_lifestyle_xp < defilement_of_$TYPE$_granting_cost
			panoplies_lifestyle_perk_points >= 1
		}
		add_panoplies_lifestyle_perk_points = -1
		add_panoplies_lifestyle_xp = {
			value = {
				value = 1000
				subtract = defilement_of_$TYPE$_granting_cost
			}
		}
	}
	else = {
		add_panoplies_lifestyle_xp = {
			value = {
				value = defilement_of_$TYPE$_granting_cost
				multiply = -1
			}
		}
	}
}
POD_grant_defilement_cost = {
	switch = {
		trigger = flag:$TYPE$
		flag:taint = { POD_grant_defilement_generalized_cost = { TYPE = taint } }
		flag:war = { POD_grant_defilement_generalized_cost = { TYPE = war } }
		flag:greed = { POD_grant_defilement_generalized_cost = { TYPE = greed } }
		flag:shadows = { POD_grant_defilement_generalized_cost = { TYPE = shadows } }
		flag:rage = { POD_grant_defilement_generalized_cost = { TYPE = rage } }
	}
}

POD_panoply_cost_effect = {
	save_scope_value_as = {
		name = panoply_cost
		value = {
			value = $COUNT$
			multiply = panoply_cost_mult
			ceiling = yes
			min = 0
		}
	}
	POD_splat_agnostic_energy_cost = {
		COUNT = scope:panoply_cost
	}
}
POD_panoply_income_effect = {
	POD_splat_agnostic_energy_income = {
		COUNT = $COUNT$
	}
}

POD_yamaking_defiled_trait_add_effect = {
	trait:$TRAIT$ ?= { save_scope_as = trait }
	custom_description = {
		text = POD_yamaking_defiled_trait_add_effect_loca
		object = scope:trait
		add_to_variable_list = { name = POD_defiled_traits target = scope:trait }
	}
}
POD_yamaking_defiled_trait_check_effect = {
	save_scope_as = scope1
	every_relation = {
		type = $RELATION$
		save_scope_as = scope2
		scope:scope1 = {
			POD_give_defiled_traits_effect = { DEFILED = scope:scope2 }
		}
	}
}
POD_give_defiled_traits_effect = {
	every_in_list = {
		variable = POD_defiled_traits
		save_scope_as = trait
		$DEFILED$ = { add_trait = scope:trait }
	}
}
POD_setup_contested_yomi_effect = {
	if = { limit = { has_trait = yamaking } add_character_modifier = contested_yomi_highly_modifier }
}
POD_change_contested_yomi_effect = {
	save_scope_value_as = { name = calc value = { value = contested_yomi_level add = $COUNT$ min = 1 max = 4 } }
	hidden_effect = { POD_remove_contested_yomi_modifiers_effect = yes }
	switch = {
		trigger = scope:calc
		1 = { add_character_modifier = contested_yomi_highly_modifier }
		2 = { add_character_modifier = contested_yomi_modifier }
		3 = { add_character_modifier = contested_yomi_slightly_modifier }
		4 = { add_character_modifier = contested_yomi_safe_modifier }
	}
}
