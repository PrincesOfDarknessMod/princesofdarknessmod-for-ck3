POD_become_defiled = {
	add_trait_force_tooltip = defiled
	set_character_faith = faith:wayofthetenthousandscreams
	scope:yamaking ?= {
		PREV = { 
			set_character_faith = scope:yamaking.faith 
			set_relation_worshipped_yamaking = scope:yamaking
		}
	}
}

POD_concession_check_tooltip = {
	save_scope_value_as = { name = concession_required value = 0 }
	switch = {
		trigger = $LEVEL$
		1 = { save_scope_value_as = { name = concession_required value = 0 } }
		2 = { save_scope_value_as = { name = concession_required value = 5 } }
		3 = { save_scope_value_as = { name = concession_required value = 10 } }
		4 = { save_scope_value_as = { name = concession_required value = 20 } }
		5 = { save_scope_value_as = { name = concession_required value = 30 } }
		6 = { save_scope_value_as = { name = concession_required value = 50 } }
	}
	if = { 
		limit = { NOT = { has_variable = main_defilement } } 
		set_variable = {
			name = main_defilement
			value = flag:$DEFILEMENT$
		}
		switch = {
			trigger = flag:$DEFILEMENT$
			flag:taint = { custom_tooltip = defilements_of_taint_concession_gain }
			flag:war = { custom_tooltip = defilements_of_war_concession_gain }
			flag:greed = { custom_tooltip = defilements_of_greed_concession_gain }
			flag:shadows = { custom_tooltip = defilements_of_shadows_concession_gain }
			flag:rage = { custom_tooltip = defilements_of_rage_concession_gain }
		}
	}
	else_if = { 
		limit = { NOT = { var:main_defilement ?= flag:$DEFILEMENT$ } }
		save_scope_value_as = { 
			name = concession_required 
			value = { 
				value = scope:concession_required 
				multiply = 2 
				min = 1 
			} 
		}
	}
	if = { limit = { scope:concession_required >= 1 } 
		custom_description_no_bullet = { text = POD_concession_check_tooltip value = scope:concession_required }
		save_scope_value_as = { 
			name = concession_required 
			value = { 
				value = scope:concession_required 
				multiply = -1
			} 
		}
		hidden_effect = { POD_chance_concession_effect = { VALUE = scope:concession_required } }
	}
}

POD_chance_concession_effect = {
	if = { limit = { $VALUE$ >= 0 } 
		custom_description = {
			text = POD_chance_concession_effect_positive
			value = $VALUE$
		}
	}
	else = {
		custom_description = {
			text = POD_chance_concession_effect_negative
			value = $VALUE$
		}
	}
	hidden_effect = {
		POD_set_or_change_variable = {
			VAR = defiled_concession_var
			VAL = $VALUE$
		}
	}
}

POD_taint_defilement_concession_effect = {
	if = { 
		limit = { var:main_defilement ?= flag:taint } 
		POD_chance_concession_effect = { VALUE = $VALUE$ }
	}
}
