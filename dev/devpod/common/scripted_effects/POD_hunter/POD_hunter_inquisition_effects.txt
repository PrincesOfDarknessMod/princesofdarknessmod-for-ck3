# Reusable effects

# Need to pass TITLE
POD_inquisition_find_gm_title_and_assign_to_gm_effect = {
	create_title_and_vassal_change = {
		type = created
		save_scope_as = change
		add_claim_on_loss = no
	}
	
	$TITLE$ = {
		change_title_holder = {
			holder = prev
			change = scope:change
			take_baronies = yes
		}
		add_county_modifier = {
			modifier = POD_heart_of_rebellion_modifier
			years = 10
		}
	}
	resolve_title_and_vassal_change = scope:change
}

POD_inquisition_gm_after_creation_effect = {
	#add_character_modifier = POD_war_fervor_modifier
	add_character_flag = immune_to_disease
	
	house ?= { POD_set_colorscheme_white = yes }
	
	add_piety_level = 4
	add_gold = 5000
	add_prestige = 1000
	add_piety = 5000
	
	# Start declaring wars on the highest difficulty
	trigger_event = {
		id = PODinquisition.1000
		days = 5
	}
	
	if = {
		limit = {
			NOT = {
				has_game_rule = age_of_inquisition_difficulty_easy
			}
		}
		add_trait = conqueror
	}
	if = {
		limit = {
			has_game_rule = age_of_inquisition_difficulty_hard
		}
		add_character_modifier = {
			modifier = ai_extreme_conqueror_modifier
		}
	}
	
	# Become independent
	hidden_effect = {
		if = {
			limit = { is_imprisoned = yes }
			release_from_prison = yes
		}
	}
	create_title_and_vassal_change = {
		type = independency
		save_scope_as = going_independent
		add_claim_on_loss = no
	}
	becomes_independent = {
		change = scope:going_independent
	}
	resolve_title_and_vassal_change = scope:going_independent
	
	# for naughty players who switch game rules while inquisitors are spawning
	POD_achiev_redlist_eligibility_check = yes
}

find_title_for_frenchinquisition_spawn_effect = {
	if = { # The important thing is to not mess too much with players who can't resist
		limit = {
			title:c_uses = {
				holder = { is_ai = yes }
				OR = {
					duchy = { is_title_created = yes }
					duchy.holder = { is_ai = yes }
				}
			}
		}
		title:c_uses = {
			save_scope_as = frenchinquisition_county
		}
	}
	else = {
		title:k_grand_court = {
			random_in_de_jure_hierarchy = {
				continue = { tier > tier_county }
				limit = {
					tier = tier_county
					holder = { is_ai = yes } # Let's not murder players
					save_temporary_scope_as = potential_frenchinquisition_county
					trigger_if = {
						limit = { exists = title:k_grand_court.holder }
						title:k_grand_court.holder = {
							OR = {
								this = scope:potential_frenchinquisition_county.holder
								any_sub_realm_county = {
									this = scope:potential_frenchinquisition_county
								}
							}
						}
					}
					# Either this should be the holder's only title, or it should NOT be their capital
					trigger_if = {
						limit = {
							holder = {
								any_sub_realm_county = { count > 1 }
							}
						}
						NOT = {
							this = holder.capital_county
						}
					}
				}
				weight = {
					base = 1
					# Massively prefer to avoid a player-held Duchies if possible
					modifier = {
						factor = 0.01
						exists = duchy.holder
						duchy.holder = { is_ai = no }
					}
				}
				save_scope_as = frenchinquisition_county
			}
		}
	}
	
	if = { # Discard stuff from here on out
		limit = { NOT = { exists = scope:frenchinquisition_county } }
		debug_log = "No frenchinquisition title found, frenchinquisition won't spawn"
	}
	# Convert the County and warn relevant people
	else = {
		scope:frenchinquisition_county = {
			set_county_faith = faith:shadowinquisition
		}
		
		every_ruler = {
			if = {
				limit = {
					is_ai = no
					capital_province = { geographical_region = world_europe_west_francia }
				}
				trigger_event = PODinquisition.1001
			}
		}
		
		trigger_event = { # Random amount of time to prepare
			id = PODinquisition.0003
			days = { 5 10 }
		}
	}
}

		}

			random_in_de_jure_hierarchy = {
				continue = { tier > tier_county }
				limit = {
					tier = tier_county
					holder = { is_ai = yes } # Let's not murder players
					save_temporary_scope_as = potential_iberianinquisition_county

						limit = { exists = title:k_portugal.holder }

								any_sub_realm_county = {
									this = scope:potential_iberianinquisition_county
								}
							}
						}
					}

					trigger_if = {
						limit = {
							holder = {
								any_sub_realm_county = { count > 1 }
							}
						}
						NOT = {
							this = holder.capital_county
						}
					}
				}
				weight = {
					base = 1
					# Massively prefer to avoid a player-held Duchies if possible
					modifier = {
						factor = 0.01
						exists = duchy.holder
						duchy.holder = { is_ai = no }
					}
				}
				save_scope_as = iberianinquisition_county
			}
		}
	}
	
	if = { # Discard stuff from here on out
		limit = { NOT = { exists = scope:iberianinquisition_county } }
		debug_log = "No iberianinquisition title found, iberianinquisition won't spawn"
	}
	# Convert the County and warn relevant people
	else = {
		scope:iberianinquisition_county = {
			set_county_faith = faith:shadowinquisition
		}
		
		every_ruler = {
			if = {
				limit = {
					is_ai = no
					capital_province = { geographical_region = world_europe_west_iberia }
				}
				trigger_event = PODinquisition.1005
			}
		}
		
		trigger_event = { # Random amount of time to prepare
			id = PODinquisition.0009
			days = { 5 10 }
		}
	}

find_title_for_maghrebinquisition_spawn_effect = {
	if = { # The important thing is to not mess too much with players who can't resist
		limit = {
			title:c_annaba = {
				holder = { is_ai = yes }
				OR = {
					duchy = { is_title_created = yes }
					duchy.holder = { is_ai = yes }
				}
			}
		}
		title:c_annaba = {
			save_scope_as = maghrebinquisition_county
		}
	}
	else = {
		title:k_africa = {
			random_in_de_jure_hierarchy = {
				continue = { tier > tier_county }
				limit = {
					tier = tier_county
					holder = { is_ai = yes } # Let's not murder players
					save_temporary_scope_as = potential_maghrebinquisition_county
					trigger_if = {
						limit = { exists = title:k_africa.holder }
						title:k_africa.holder = {
							OR = {
								this = scope:potential_maghrebinquisition_county.holder
								any_sub_realm_county = {
									this = scope:potential_maghrebinquisition_county
								}
							}
						}
					}
					# Either this should be the holder's only title, or it should NOT be their capital
					trigger_if = {
						limit = {
							holder = {
								any_sub_realm_county = { count > 1 }
							}
						}
						NOT = {
							this = holder.capital_county
						}
					}
				}
				weight = {
					base = 1
					# Massively prefer to avoid a player-held Duchies if possible
					modifier = {
						factor = 0.01
						exists = duchy.holder
						duchy.holder = { is_ai = no }
					}
				}
				save_scope_as = maghrebinquisition_county
			}
		}
	}
	
	if = { # Discard stuff from here on out
		limit = { NOT = { exists = scope:maghrebinquisition_county } }
		debug_log = "No maghrebinquisition title found, maghrebinquisition won't spawn"
	}
	# Convert the County and warn relevant people
	else = {
		scope:maghrebinquisition_county = {
			set_county_faith = faith:ikhwan_al_safa
		}
		
		every_ruler = {
			if = {
				limit = {
					is_ai = no
					capital_province = { geographical_region = ghw_region_north_africa  }
				}
				trigger_event = PODinquisition.1012
			}
		}
		
		trigger_event = { # Random amount of time to prepare
			id = PODinquisition.0023
			days = { 5 10 }
		}
	}
}
