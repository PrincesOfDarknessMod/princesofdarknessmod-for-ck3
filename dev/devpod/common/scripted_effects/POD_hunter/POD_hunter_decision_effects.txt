embrace_hebrew_culture_effect = {
	# Scopes saves for localization
	save_scope_as = embracer_israel_hebrew

	#Convert your, and your whole family's, culture
	set_culture = culture:hebrew
	if = {
		limit = { any_spouse = { is_playable_character = no } }
		every_spouse = {
			limit = {
				is_playable_character = no
			}
			set_culture = culture:hebrew
		}
	}
	if = {
		limit = {
			any_close_family_member = {
				is_playable_character = no
				NOT = { is_spouse_of = root }
			}
		}
		every_close_family_member = {
			limit = {
				is_playable_character = no
				NOT = { is_spouse_of = root }
			}
			custom = all_unlanded_family_members
			set_culture = culture:hebrew
		}
	}
	if = {
		limit = {
			any_child = {
				NOT = { is_spouse_of = root }
			}
		}
		every_child = {
			limit = {
				NOT = { is_spouse_of = root }
			}
			custom = all_children_custom
			set_culture = culture:hebrew
		}
	}

	#Convert appropriate vassals, and their family
	if = {
		limit = {
			any_vassal_or_below = {
				is_ai = yes
				primary_title.title_province = {
					geographical_region = pod_judges_promised_land
				}
			}
		}
		every_vassal_or_below = {
			custom = embrace_hebrew_culture_vassals_custom
			limit = {
				is_ai = yes
				primary_title.title_province = {
					geographical_region = pod_judges_promised_land
				}
			}
			set_culture = culture:hebrew
			hidden_effect = {
				if = {
					limit = { any_spouse = { is_playable_character = no } }
					every_spouse = {
						limit = {
							is_playable_character = no
						}
						set_culture = culture:hebrew
					}
				}
				if = {
					limit = {
						any_close_family_member = {
							is_playable_character = no
							NOT = { is_spouse_of = prev }
						}
					}
					every_close_family_member = {
						limit = {
							is_playable_character = no
							NOT = { is_spouse_of = prev }
						}
						custom = all_unlanded_family_members
						set_culture = culture:hebrew
					}
				}
				if = {
					limit = {
						any_child = {
							NOT = { is_spouse_of = prev }
						}
					}
					every_child = {
						limit = {
							NOT = { is_spouse_of = prev }
						}
						custom = all_children_custom
						set_culture = culture:hebrew
					}
				}
				if = {
					limit = {
						any_vassal_or_below = {
							is_ai = yes
							primary_title.title_province = {
								geographical_region = pod_judges_promised_land
							}
						}
					}
					every_vassal_or_below = {
						custom = embrace_hebrew_culture_vassals_custom
						limit = {
							is_ai = yes
							primary_title.title_province = {
								geographical_region = pod_judges_promised_land
							}
						}
						set_culture = culture:hebrew
						hidden_effect = {
							if = {
								limit = { any_spouse = { is_playable_character = no } }
								every_spouse = {
									limit = {
										is_playable_character = no
									}
									set_culture = culture:hebrew
								}
							}
							if = {
								limit = {
									any_close_family_member = {
										is_playable_character = no
										NOT = { is_spouse_of = prev }
									}
								}
								every_close_family_member = {
									limit = {
										is_playable_character = no
										NOT = { is_spouse_of = prev }
									}
									custom = all_unlanded_family_members
									set_culture = culture:hebrew
								}
							}
							if = {
								limit = {
									any_child = {
										NOT = { is_spouse_of = prev }
									}
								}
								every_child = {
									limit = {
										NOT = { is_spouse_of = prev }
									}
									custom = all_children_custom
									set_culture = culture:hebrew
								}
							}
						}
					}
				}
			}
		}
	}

	#Flip Counties
	every_county_in_region = {
		region = pod_judges_promised_land
		custom = embrace_hebrew_culture_counties_custom
		limit = {
			target_is_de_facto_liege_or_above = root.primary_title
		}
		random = {
			chance = 40
			set_county_culture = culture:hebrew
			set_county_faith = root.faith
		}
	}

	#Bring in the modern innovations
	culture:hebrew = {
		custom_tooltip = decision_embrace_hebrew_culture_get_innovations
		hidden_effect = {
			reset_culture_creation_date = yes
			get_all_innovations_from = culture:ashkenazi
			get_all_innovations_from = culture:sephardi
			get_all_innovations_from = culture:radhanite
			get_all_innovations_from = culture:kochinim
			get_all_innovations_from = culture:bavlim
		}
	}
}
