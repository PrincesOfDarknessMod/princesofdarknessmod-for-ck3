
POD_shih_initialization_effect = {
	if = {
		limit = {
			NOT = { has_trait = shih }
		}
		add_trait = shih
	}
	if = {
		limit = { is_a_cultist = no }
		set_character_faith = faith:shih_faith
	}
}

POD_remove_shih_atribute_effect = {
	remove_trait = shih
	POD_qiao_elder_penalty_remove_effect = yes
}

POD_qiao_elder_penalty_effect = {
	if = { limit = { POD_qiao_elder_penalty_trigger = no } 
		custom_tooltip = POD_qiao_elder_penalty_effect
		switch = {
			trigger = flag:$QIAO$
			flag:i_shen = { add_character_modifier = shih_i_shen_penalty_modifier }
			flag:mo_kung = { add_character_modifier = shih_mo_kung_penalty_modifier }
			flag:yu_an = { add_character_modifier = shih_yu_an_penalty_modifier }
			flag:feng = { add_character_modifier = shih_feng_penalty_modifier }
			flag:shi = { add_character_modifier = shih_shi_penalty_modifier }
			flag:chien = { add_character_modifier = shih_chien_penalty_modifier }
			flag:long_ling = { add_character_modifier = shih_long_ling_penalty_modifier }
			flag:meng = { add_character_modifier = shih_meng_penalty_modifier }
			flag:zhu_mao = { add_character_modifier = shih_zhu_mao_penalty_modifier }
		}
	}
}
POD_qiao_elder_penalty_remove_effect = {
	hidden_effect = {
		remove_all_character_modifier_instances = shih_i_shen_penalty_modifier
		remove_all_character_modifier_instances = shih_mo_kung_penalty_modifier
		remove_all_character_modifier_instances = shih_yu_an_penalty_modifier
		remove_all_character_modifier_instances = shih_feng_penalty_modifier
		remove_all_character_modifier_instances = shih_shi_penalty_modifier
		remove_all_character_modifier_instances = shih_chien_penalty_modifier
		remove_all_character_modifier_instances = shih_long_ling_penalty_modifier
		remove_all_character_modifier_instances = shih_meng_penalty_modifier
		remove_all_character_modifier_instances = shih_zhu_mao_penalty_modifier
	}
}
