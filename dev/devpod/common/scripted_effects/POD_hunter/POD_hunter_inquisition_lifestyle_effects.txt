POD_add_curia_support = {
	if = {
		limit = {
			faith = faith:shadowinquisition
			is_landless_ruler = yes
			NOT = { has_trait = excommunicated }
		}
		custom_description = {
			text = POD_add_curia_support
			value = $VALUE$
			if = {
				limit = { NOT = { has_variable = curia_support_var } }
				set_variable = {
					name = curia_support_var
					value = 0
				}
			}
			# error suppression
			if = {
				limit = { has_variable = curia_support_var }
				change_variable = {
					name = curia_support_var
					add = $VALUE$
				}
			}
		}
	}
}

	}

	add_trait = excommunicated
}
