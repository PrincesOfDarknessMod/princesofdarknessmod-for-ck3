POD_create_coa_dummy_character = {
	create_character = {
		name = "Jean" # the unsung hero
		gender = male
		employer = $EMPLOYER$
		trait = "content"
		trait = "shy"
		trait = "lazy"
		trait = "mortal"
		trait = "immortal"
		trait = "torpor"
		trait = "education_diplomacy_1"
		diplomacy = 0
		stewardship = 0
		martial = 0
		intrigue = 0
		learning = 0
		random_traits = no
		faith = faith:innocence
		culture = culture:french
		dynasty = generate
		save_scope_as = $SAVE_AS$
		after_creation = {
			add_character_flag = POD_do_not_disturb_flag
			
			add_character_flag = POD_shadeskin_form_flag
			add_character_flag = POD_dde_hre_war_low_nobles_flag
			add_character_flag = POD_portrait_hermetic_mage_hood_male_flag
			add_character_flag = POD_ep1_adventurer_cloak_flag

			dynasty = { add_dynasty_prestige = 13000 } # better looking coa frame

			set_variable = is_coa_dummy
		}
	}
}

POD_create_personal_coa_effect = {
	save_scope_as = actor
	add_character_flag = POD_has_personal_coa_flag
	if = {
		limit = {
			scope:actor = { is_playable_character = yes }
		}
		POD_create_coa_dummy_character = {
			EMPLOYER = scope:actor
			SAVE_AS = recipient
		}
	}
	else = {
		POD_create_coa_dummy_character = {
			EMPLOYER = character:POD_wraith_01
			SAVE_AS = recipient
		}
	}
	
	hidden_effect_new_object = {
		scope:recipient = {
			set_global_variable = {
				name = coa_new_name
				value = scope:actor
			}
			scope:recipient.house = {
				set_house_name = coa_new_name
			}
			remove_global_variable = coa_new_name
			set_variable = {
				name = my_coa_belongs_to
				value = scope:actor
			}
		}
		if = {
			limit = {
				scope:actor = {
					is_lowborn = yes
				}
			}
			scope:recipient.house = { generate_coa = yes }
		} else = {
			scope:recipient.house = { set_coa = scope:actor.house }
		}
		scope:actor = {
			set_variable = {
				name = my_personal_coa
				value = scope:recipient
			}
			set_relation_pod_personal_coa_slave = scope:recipient
			scope:recipient = { save_scope_as = exiled_dummy }
			if = {
				limit = {
					scope:actor = {
						is_ai = yes
					}
				}
				# Send away the COA Dummy Immediately since the AI won't be editing their own COAs
				trigger_event = POD_personal_coat_of_arms.100
			} else = {
				# Send away the COA Dummy a few days after
				trigger_event = { id = POD_personal_coat_of_arms.100 days = 1 }
			}
		}
	}
}

POD_create_personal_coa_for_historical_ai_effect = {
	$ACTOR$ = {
		save_scope_as = actor
		add_character_flag = POD_has_personal_coa_flag
		POD_create_coa_dummy_character = {
			EMPLOYER = character:POD_wraith_01
			SAVE_AS = recipient
		}
		hidden_effect_new_object = {
			scope:recipient = {
				set_global_variable = {
					name = coa_new_name
					value = scope:actor
				}
				scope:recipient.house = {
					set_house_name = coa_new_name
					set_coa = $COA_KEY$
				}
				remove_global_variable = coa_new_name
				set_variable = {
					name = my_coa_belongs_to
					value = scope:actor
				}
			}
			scope:actor = {
				set_variable = {
					name = my_personal_coa
					value = scope:recipient
				}
				set_relation_pod_personal_coa_slave = scope:recipient
				scope:recipient = { save_scope_as = exiled_dummy } # Send away the COA Dummy Immediately since the AI won't be editing their own COAs
				trigger_event = POD_personal_coat_of_arms.100
			}
		}
	}
}

POD_prune_personal_coa_effect = {
	var:my_personal_coa = {
		death = {
			death_reason = death_vanished
		}
	}
}
