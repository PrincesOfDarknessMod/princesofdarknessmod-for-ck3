POD_change_umbra_progress_effect = {
	$VICTIM$ = { change_variable = { name = umbra_exploration_progress add = umbra_aptitude } }
	if = {
		limit = { 
			$VICTIM$ = { var:umbra_exploration_progress >= umbra_stationed_location_size }
			NOT = { has_umbra_level_5_in_trigger = { LOCATION = $VICTIM$.var:umbra_stationed_location } }
		}
		$VICTIM$ = { set_variable = { name = umbra_exploration_progress value = 0 } }
		POD_change_umbra_level_effect = { LOCATION = $VICTIM$.var:umbra_stationed_location VICTIM = $VICTIM$ }
	}
}

POD_gain_umbra_progress_effect = {
	$VICTIM$ = { change_variable = { name = umbra_exploration_progress add = $VALUE$ } }
	if = {
		limit = { 
			$VICTIM$ = { var:umbra_exploration_progress >= umbra_stationed_location_size }
			NOT = { has_umbra_level_5_in_trigger = { LOCATION = $VICTIM$.var:umbra_stationed_location } }
		}
		$VICTIM$ = { set_variable = { name = umbra_exploration_progress value = 0 } }
		POD_change_umbra_level_effect = { LOCATION = $VICTIM$.var:umbra_stationed_location VICTIM = $VICTIM$ }
	}
}

POD_lose_umbra_progress_effect = {
	var:local_umbra_explorer = { change_variable = { name = umbra_exploration_progress subtract = $VALUE$ } }
	if = {
		limit = { var:local_umbra_explorer = { var:umbra_exploration_progress < 0 } }
		var:local_umbra_explorer = { set_variable = { name = umbra_exploration_progress value = 0 } }
	}
}

POD_lose_umbra_progress_with_overflow_effect = { # Won't remove gained bonus
	var:local_umbra_explorer = { change_variable = { name = umbra_exploration_progress subtract = $VALUE$ } }
	if = {
		limit = { var:local_umbra_explorer = { var:umbra_exploration_progress < 0 } }
		var:local_umbra_explorer = { set_variable = { name = umbra_exploration_progress value = umbra_stationed_location_size } }
		if = {
			limit = { is_target_in_variable_list = { name = umbra_level_5_list target = var:local_umbra_explorer.var:umbra_stationed_location } }
			remove_list_variable = {
				name = umbra_level_5_list
				target = var:local_umbra_explorer.var:umbra_stationed_location
			}
			add_to_variable_list = {
				name = umbra_level_4_list
				target = var:local_umbra_explorer.var:umbra_stationed_location
			}
		}
		else_if = {
			limit = { is_target_in_variable_list = { name = umbra_level_4_list target = var:local_umbra_explorer.var:umbra_stationed_location } }
			remove_list_variable = {
				name = umbra_level_4_list
				target = var:local_umbra_explorer.var:umbra_stationed_location
			}
			add_to_variable_list = {
				name = umbra_level_3_list
				target = var:local_umbra_explorer.var:umbra_stationed_location
			}
		}
		else_if = {
			limit = { is_target_in_variable_list = { name = umbra_level_3_list target = var:local_umbra_explorer.var:umbra_stationed_location } }
			remove_list_variable = {
				name = umbra_level_3_list
				target = var:local_umbra_explorer.var:umbra_stationed_location
			}
			add_to_variable_list = {
				name = umbra_level_2_list
				target = var:local_umbra_explorer.var:umbra_stationed_location
			}
		}
		else_if = {
			limit = { is_target_in_variable_list = { name = umbra_level_2_list target = var:local_umbra_explorer.var:umbra_stationed_location } }
			remove_list_variable = {
				name = umbra_level_2_list
				target = var:local_umbra_explorer.var:umbra_stationed_location
			}
			add_to_variable_list = {
				name = umbra_level_1_list
				target = var:local_umbra_explorer.var:umbra_stationed_location
			}
		}
		else_if = {
			limit = { is_target_in_variable_list = { name = umbra_level_1_list target = var:local_umbra_explorer.var:umbra_stationed_location } }
			remove_list_variable = {
				name = umbra_level_1_list
				target = var:local_umbra_explorer.var:umbra_stationed_location
			}
		}
	}
}

POD_change_umbra_level_effect = {
	set_variable = { 
		name = umbra_location_to_change
		value = $LOCATION$
	}
	if = {
		limit = { is_target_in_variable_list = { name = umbra_level_4_list target = $LOCATION$ } }
		send_interface_toast = {
			title = umbra_level_5_reached_tt
			left_icon = root
			right_icon = $VICTIM$
			
			custom_tooltip = umbra_max_level_reached_tt
			add_prestige = minor_prestige_gain
			POD_gain_umbra_level_5_effect = { LOCATION = $LOCATION$ }
		}
	}
	else_if = {
		limit = { is_target_in_variable_list = { name = umbra_level_3_list target = $LOCATION$ } }
		send_interface_toast = {
			title = umbra_level_4_reached_tt
			left_icon = root
			right_icon = $VICTIM$
			
			add_prestige = minor_prestige_gain
			POD_gain_umbra_level_4_effect = { LOCATION = $LOCATION$ }
		}
	}
	else_if = {
		limit = { is_target_in_variable_list = { name = umbra_level_2_list target = $LOCATION$ } }
		send_interface_toast = {
			title = umbra_level_3_reached_tt
			left_icon = root
			right_icon = $VICTIM$
			
			add_prestige = minor_prestige_gain
			POD_gain_umbra_level_3_effect = { LOCATION = $LOCATION$ }
		}
	}
	else_if = {
		limit = { is_target_in_variable_list = { name = umbra_level_1_list target = $LOCATION$ } }
		send_interface_toast = {
			title = umbra_level_2_reached_tt
			left_icon = root
			right_icon = $VICTIM$
			
			add_prestige = minor_prestige_gain
			POD_gain_umbra_level_2_effect = { LOCATION = $LOCATION$ }
		}
	}
	else = {
		send_interface_toast = {
			title = umbra_level_1_reached_tt
			left_icon = root
			right_icon = $VICTIM$
			
			add_prestige = minor_prestige_gain
			POD_gain_umbra_level_1_effect = { LOCATION = $LOCATION$ }
		}
	}
	if = {
		limit = { has_variable = umbra_location_to_change }
		remove_variable = umbra_location_to_change
	}
	save_scope_as = player
	$VICTIM$ = { POD_umbra_set_expedition_interest_effect = yes }
}

POD_inherit_umbra_level_effect = {
	save_temporary_scope_as = heir
	clear_variable_list = umbra_level_5_list
	clear_variable_list = umbra_level_4_list
	clear_variable_list = umbra_level_3_list
	clear_variable_list = umbra_level_2_list
	clear_variable_list = umbra_level_1_list
	$TARGET$ = {
		every_in_list = {
			variable = umbra_level_5_list
			scope:heir = {
				POD_gain_umbra_level_1_effect = { LOCATION = prev }
				POD_gain_umbra_level_2_effect = { LOCATION = prev }
				POD_gain_umbra_level_3_effect = { LOCATION = prev }
				POD_gain_umbra_level_4_effect = { LOCATION = prev }
				POD_gain_umbra_level_5_effect = { LOCATION = prev }
			}
		}
		every_in_list = {
			variable = umbra_level_4_list
			scope:heir = {
				POD_gain_umbra_level_1_effect = { LOCATION = prev }
				POD_gain_umbra_level_2_effect = { LOCATION = prev }
				POD_gain_umbra_level_3_effect = { LOCATION = prev }
				POD_gain_umbra_level_4_effect = { LOCATION = prev }
			}
		}
		every_in_list = {
			variable = umbra_level_3_list
			scope:heir = {
				POD_gain_umbra_level_1_effect = { LOCATION = prev }
				POD_gain_umbra_level_2_effect = { LOCATION = prev }
				POD_gain_umbra_level_3_effect = { LOCATION = prev }
			}
		}
		every_in_list = {
			variable = umbra_level_2_list
			scope:heir = {
				POD_gain_umbra_level_1_effect = { LOCATION = prev }
				POD_gain_umbra_level_2_effect = { LOCATION = prev }
			}
		}
		every_in_list = {
			variable = umbra_level_1_list
			scope:heir = {
				POD_gain_umbra_level_1_effect = { LOCATION = prev }
			}
		}
	}
}

	else_if = {
		limit = { NOT = { has_perk = savage_kiss_perk } }

		limit = { NOT = { has_perk = luck_of_yomi_perk } }

		limit = { NOT = { has_perk = ebon_fountain_perk } }
