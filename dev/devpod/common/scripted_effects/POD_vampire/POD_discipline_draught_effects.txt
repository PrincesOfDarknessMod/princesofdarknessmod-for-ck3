POD_draught_of_might_effect = {
	scope:actor = {
		if = {
			 limit = {
				has_trait = potenceadvanced 
			}
			scope:recipient = {
				add_character_modifier = {
					modifier = POD_greater_draught_of_might_modifier
					days = 360
				}	
				POD_moderate_risk_of_increasing_masquerade_exposure = yes
			}
		}
		else = {
			scope:recipient = {
				add_character_modifier = {
					modifier = POD_lesser_draught_of_might_modifier
					days = 360
				}	
				POD_small_risk_of_increasing_masquerade_exposure = yes	
			}
		}
	}
}
