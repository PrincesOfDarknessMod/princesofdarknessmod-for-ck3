POD_gangrel_frenzy_effect = {
	if = {
		limit = {has_character_modifier = POD_gangrel_frenzy_modifier_heavy}
		POD_moderate_risk_of_increasing_masquerade_exposure = yes
	}
	else_if = {
		limit = {has_character_modifier = POD_gangrel_frenzy_modifier_medium}
		hidden_effect = {
			remove_character_modifier = POD_gangrel_frenzy_modifier_medium
		}
		add_character_modifier = {
			modifier = POD_gangrel_frenzy_modifier_heavy
			years = 10
		}
	}
	else_if = {
		limit = {has_character_modifier = POD_gangrel_frenzy_modifier_light}
		hidden_effect = {
			remove_character_modifier = POD_gangrel_frenzy_modifier_light
		}
		add_character_modifier = {
			modifier = POD_gangrel_frenzy_modifier_medium
			years = 10
		}
	}
	else = {
		add_character_modifier = {
			modifier = POD_gangrel_frenzy_modifier_light
			years = 10
		}
	}

}
