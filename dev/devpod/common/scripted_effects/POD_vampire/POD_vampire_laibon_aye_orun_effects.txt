# Laibon Aye/Orun System
# Core XP Management Effects
# Single trait (laibon_spirit) with two tracks (aye, orun)
# Track scale: 0-100, tiers at 25/50/75
##############################

### XP Addition ###

# adds aye/orun xp and properly previews any consequences
# e.g. modifiers or traits gained/lost
POD_laibon_add_aye_and_orun_xp = {
	save_scope_value_as = {
		name = new_aye_xp
		value = {
			value = POD_aye_xp
			add = $AYE$
			min = 0
			max = 100
		}
	}
	save_scope_value_as = {
		name = new_orun_xp
		value = {
			value = POD_orun_xp
			add = $ORUN$
			min = 0
			max = 100
		}
	}
	if = {
		limit = { $AYE$ != 0 }
		add_trait_xp = {
			trait = laibon_spirit
			track = aye
			value = $AYE$
		}
	}
	if = {
		limit = { $ORUN$ != 0 }
		add_trait_xp = {
			trait = laibon_spirit
			track = orun
			value = $ORUN$
		}
	}
	POD_laibon_check_aye_disconnected = {
		AYE = scope:new_aye_xp
	}
	POD_laibon_check_orun_disconnected = {
		ORUN = scope:new_orun_xp
	}
	POD_laibon_check_walker = {
		AYE  = scope:new_aye_xp
		ORUN = scope:new_orun_xp
	}
	POD_laibon_check_hollow = {
		AYE  = scope:new_aye_xp
		ORUN = scope:new_orun_xp
	}
}

### Disconnected Modifier Management ###

POD_laibon_check_aye_disconnected = {
	# Add penalty modifier if Aye < 25 (disconnected)
	if = {
		limit = {
			$AYE$ < POD_laibon_spirit_xp_for_rank_1
			NOT = { has_character_modifier = aye_disconnected_modifier }
		}
		add_character_modifier = aye_disconnected_modifier
	}
	# Remove penalty modifier if Aye >= 25
	else_if = {
		limit = {
			$AYE$ >= POD_laibon_spirit_xp_for_rank_1
			has_character_modifier = aye_disconnected_modifier
		}
		remove_character_modifier = aye_disconnected_modifier
	}
}

POD_laibon_check_orun_disconnected = {
	if = {
		limit = {
			$ORUN$ < POD_laibon_spirit_xp_for_rank_1
			NOT = { has_character_modifier = orun_disconnected_modifier }
		}
		add_character_modifier = orun_disconnected_modifier
	}
	else_if = {
		limit = {
			$ORUN$ >= POD_laibon_spirit_xp_for_rank_1
			has_character_modifier = orun_disconnected_modifier
		}
		remove_character_modifier = orun_disconnected_modifier
	}
}

### Combined State Checks ###

POD_laibon_check_walker = {
	if = {
		limit = {
			$AYE$  >= POD_laibon_spirit_xp_for_rank_3
			$ORUN$ >= POD_laibon_spirit_xp_for_rank_3
			NOT = { has_trait = walker_between_worlds }
		}
		add_trait = walker_between_worlds
		trigger_event = pod_laibon.300
	}
	else_if = {
		limit = {
			has_trait = walker_between_worlds
			OR = {
				$AYE$  < POD_laibon_spirit_xp_for_rank_3
				$ORUN$ < POD_laibon_spirit_xp_for_rank_3
			}
		}
		remove_trait = walker_between_worlds
	}
}

POD_laibon_check_hollow = {
	if = {
		limit = {
			$AYE$  < POD_laibon_spirit_xp_for_rank_1
			$ORUN$ < POD_laibon_spirit_xp_for_rank_1
			NOT = { has_trait = the_hollow }
		}
		add_trait = the_hollow
	}
	else_if = {
		limit = {
			has_trait = the_hollow
			OR = {
				$AYE$  >= POD_laibon_spirit_xp_for_rank_1
				$ORUN$ >= POD_laibon_spirit_xp_for_rank_1
			}
		}
		remove_trait = the_hollow
	}
}

### Initialization and Removal ###

POD_laibon_spirit_initialize = {
	# Add the laibon_spirit trait and set both tracks to tier 1 (33 XP each)
	if = {
		limit = { NOT = { has_trait = laibon_spirit } }
		add_trait = laibon_spirit
		add_trait_xp = {
			trait = laibon_spirit
			track = aye
			value = 33
		}
		add_trait_xp = {
			trait = laibon_spirit
			track = orun
			value = 33
		}
		if = {
			limit = { is_ai = no }
			trigger_event = { id = pod_laibon.400 days = 30 }
		}
	}
}

POD_laibon_spirit_cleanup = {
	remove_trait = laibon_spirit
	remove_trait = walker_between_worlds
	remove_trait = the_hollow
	POD_remove_character_modifier = { MODIFIER = aye_disconnected_modifier }
	POD_remove_character_modifier = { MODIFIER = orun_disconnected_modifier }
}
