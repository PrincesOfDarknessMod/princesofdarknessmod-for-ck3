### XP Addition ###

# adds aye/orun xp and properly previews any consequences
# e.g. modifiers or traits gained/lost
POD_laibon_add_aye_and_orun_xp = {
	POD_save_guruhi_beauty_level_as = {
		AYE  = POD_aye_xp
		ORUN = POD_orun_xp
		SAVE_AS = old_guruhi_beauty_level
	}
	save_scope_value_as = {
		name = new_aye_xp
		value = {
			value = POD_aye_xp
			add = $AYE$
			min = 0
			max = 100
		}
	}
	save_scope_value_as = {
		name = new_orun_xp
		value = {
			value = POD_orun_xp
			add = $ORUN$
			min = 0
			max = 100
		}
	}
	if = {
		limit = { $AYE$ != 0 }
		add_trait_xp = {
			trait = laibon_spirit
			track = aye
			value = $AYE$
		}
	}
	if = {
		limit = { $ORUN$ != 0 }
		add_trait_xp = {
			trait = laibon_spirit
			track = orun
			value = $ORUN$
		}
	}
	POD_laibon_check_aye_disconnected = {
		AYE = scope:new_aye_xp
	}
	POD_laibon_check_orun_disconnected = {
		ORUN = scope:new_orun_xp
	}
	POD_laibon_check_walker = {
		AYE  = scope:new_aye_xp
		ORUN = scope:new_orun_xp
	}
	POD_laibon_check_hollow = {
		AYE  = scope:new_aye_xp
		ORUN = scope:new_orun_xp
	}
	if = {
		limit = { is_guruhi_trigger = yes }
		POD_save_guruhi_beauty_level_as = {
			AYE  = scope:new_aye_xp
			ORUN = scope:new_orun_xp
			SAVE_AS = new_guruhi_beauty_level
		}
		if = {
			limit = { scope:old_guruhi_beauty_level != scope:new_guruhi_beauty_level }
			POD_clear_guruhi_beauty_modifiers = yes
			POD_set_guruhi_beauty_modifier = { LEVEL = scope:new_guruhi_beauty_level }
		}
	}
}

### Disconnected Modifier Management ###

POD_laibon_check_aye_disconnected = {
	# Add penalty modifier if Aye < 25 (disconnected)
	if = {
		limit = {
			$AYE$ < POD_laibon_spirit_xp_for_rank_1
			NOT = { has_character_modifier = aye_disconnected_modifier }
		}
		add_character_modifier = aye_disconnected_modifier
	}
	# Remove penalty modifier if Aye >= 25
	else_if = {
		limit = {
			$AYE$ >= POD_laibon_spirit_xp_for_rank_1
			has_character_modifier = aye_disconnected_modifier
		}
		remove_character_modifier = aye_disconnected_modifier
	}
}

POD_laibon_check_orun_disconnected = {
	if = {
		limit = {
			$ORUN$ < POD_laibon_spirit_xp_for_rank_1
			NOT = { has_character_modifier = orun_disconnected_modifier }
		}
		add_character_modifier = orun_disconnected_modifier
	}
	else_if = {
		limit = {
			$ORUN$ >= POD_laibon_spirit_xp_for_rank_1
			has_character_modifier = orun_disconnected_modifier
		}
		remove_character_modifier = orun_disconnected_modifier
	}
}

### Combined State Checks ###

POD_laibon_check_walker = {
	if = {
		limit = {
			$AYE$  >= POD_laibon_spirit_xp_for_rank_3
			$ORUN$ >= POD_laibon_spirit_xp_for_rank_3
			NOT = { has_trait = walker_between_worlds }
		}
		add_trait = walker_between_worlds
		trigger_event = pod_laibon.300
	}
	else_if = {
		limit = {
			has_trait = walker_between_worlds
			OR = {
				$AYE$  < POD_laibon_spirit_xp_for_rank_3
				$ORUN$ < POD_laibon_spirit_xp_for_rank_3
			}
		}
		remove_trait = walker_between_worlds
	}
}

POD_laibon_check_hollow = {
	if = {
		limit = {
			$AYE$  < POD_laibon_spirit_xp_for_rank_1
			$ORUN$ < POD_laibon_spirit_xp_for_rank_1
			NOT = { has_trait = the_hollow }
		}
		add_trait = the_hollow
	}
	else_if = {
		limit = {
			has_trait = the_hollow
			OR = {
				$AYE$  >= POD_laibon_spirit_xp_for_rank_1
				$ORUN$ >= POD_laibon_spirit_xp_for_rank_1
			}
		}
		remove_trait = the_hollow
	}
}

# arguments: AYE, ORUN, SAVE_AS
POD_save_guruhi_beauty_level_as = {
	save_scope_value_as = {
		name = $SAVE_AS$
		value = {
			if = {
				limit = {
					# alternatively: take the average of the two values?
					$AYE$  >= POD_laibon_spirit_xp_for_rank_3
					$ORUN$ >= POD_laibon_spirit_xp_for_rank_3
				}
				value = 4
			}
			else_if = {
				limit = {
					$AYE$  >= POD_laibon_spirit_xp_for_rank_2
					$ORUN$ >= POD_laibon_spirit_xp_for_rank_2
				}
				value = 3
			}
			else_if = {
				limit = {
					$AYE$  >= POD_laibon_spirit_xp_for_rank_1
					$ORUN$ >= POD_laibon_spirit_xp_for_rank_1
				}
				value = 2
			}
			else = {
				value = 1
			}
		}
	}
}

POD_set_guruhi_beauty_modifier = {
	switch = {
		trigger = $LEVEL$
		4 = { add_character_modifier = guruhi_beauty_4_modifier }
		3 = { add_character_modifier = guruhi_beauty_3_modifier }
		2 = { add_character_modifier = guruhi_beauty_2_modifier }
		1 = { add_character_modifier = guruhi_beauty_1_modifier }
	}
}

# arguments: OLD_LEVEL, NEW_LEVEL
POD_reload_guruhi_beauty_modifiers = {
	if = {
		limit = { $OLD_LEVEL$ != $NEW_LEVEL$ }
		POD_clear_guruhi_beauty_modifiers = yes
		POD_set_guruhi_beauty_modifier = { LEVEL = $NEW_LEVEL$ }
	}
}

POD_clear_guruhi_beauty_modifiers = {
	POD_remove_character_modifier = { MODIFIER = guruhi_beauty_4_modifier }
	POD_remove_character_modifier = { MODIFIER = guruhi_beauty_3_modifier }
	POD_remove_character_modifier = { MODIFIER = guruhi_beauty_2_modifier }
	POD_remove_character_modifier = { MODIFIER = guruhi_beauty_1_modifier }
}

### Initialization and Removal ###

POD_set_starting_aye_and_orun = {
	set_variable = {
		name  = canon_aye
		value = $AYE$
	}
	set_variable = {
		name  = canon_orun
		value = $ORUN$
	}
}

POD_laibon_spirit_initialize = {
	if = {
		limit = { NOT = { has_trait = laibon_spirit } }
		# save these before adding the trait so the trait doesn't interfere with the script_values
		save_scope_value_as = {
			name  = initial_aye
			value = POD_aye_xp
		}
		save_scope_value_as = {
			name  = initial_orun
			value = POD_orun_xp
		}
		
		add_trait = laibon_spirit
		POD_laibon_add_aye_and_orun_xp = {
			AYE  = scope:initial_aye
			ORUN = scope:initial_orun
		}
		if = {
			limit = { is_ai = no }
			trigger_event = { id = pod_laibon.400 days = 30 }
		}
	}
}

POD_laibon_spirit_cleanup = {
	# save values for posterity
	POD_set_starting_aye_and_orun = {
		AYE  = POD_aye_xp
		ORUN = POD_orun_xp
	}
	remove_trait = laibon_spirit
	remove_trait = walker_between_worlds
	remove_trait = the_hollow
	POD_remove_character_modifier = { MODIFIER = aye_disconnected_modifier }
	POD_remove_character_modifier = { MODIFIER = orun_disconnected_modifier }
}
