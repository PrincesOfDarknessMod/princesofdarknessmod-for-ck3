POD_character_vampire_discipline_initialization_effect = {
	# special case: custom bloodline
	if = {
		limit = { is_custom_bloodline_trigger = yes }
		save_scope_as = discipline_char
		if = {
			limit = { age >= 500 }
			dynasty = {
				every_in_list = {
					variable = custom_bloodline_advanced_disciplines
					scope:discipline_char = {
						add_trait = prev
					}
				}
			}
		}
		else = {
			dynasty = {
				every_in_list = {
					variable = custom_bloodline_disciplines
					scope:discipline_char = {
						add_trait = prev
					}
				}
			}
		}
	}
	# all other clans and bloodlines
	else = {
		if = {
			limit = {
				age < 500
				POD_select_clan_disciplines_for_character = {
					TRIGGER = POD_save_clan_disciplines
				}
			}
		}
		else_if = {
			limit = {
				POD_select_clan_disciplines_for_character = {
					TRIGGER = POD_save_advanced_clan_disciplines
				}
			}
		}
		if = {
			limit = { exists = scope:discipline_1 }
			add_trait = scope:discipline_1
			add_trait = scope:discipline_2
			add_trait = scope:discipline_3
		}
		# prevent errors from temporary scopes sticking around
		clear_saved_scope = discipline_1
		clear_saved_scope = discipline_2
		clear_saved_scope = discipline_3
	}
	
	# caitiff disciplines as fallback if no disciplines were added
	if = {
		limit = { has_POD_vampire_discipline_trigger = no }
		if = {
			limit = { age >= 500 }
			POD_caitiff_random_discipline_initialization_effect = { ADVANCED = yes }
			POD_caitiff_random_discipline_initialization_effect = { ADVANCED = yes }
			POD_caitiff_random_discipline_initialization_effect = { ADVANCED = yes }
		}
		else = {
			POD_caitiff_random_discipline_initialization_effect = { ADVANCED = no }
			POD_caitiff_random_discipline_initialization_effect = { ADVANCED = no }
			POD_caitiff_random_discipline_initialization_effect = { ADVANCED = no }
		}
	}
	
	# add fallback traditions for clans with no innate ones (e.g. maeghar)
	if = {
		limit = {
			has_trait = bloodsorcerydiscipline
			OR = {
				has_any_bloodmagic_path = no
				is_tremere_trigger = yes #Added because if you are a ghoul with BS and you become a tremere, you don't get thauma otherwise
			}
		}
		add_character_flag = has_thaumaturgy_bloodmagic_flag
	}
	if = {
		limit = {
			has_trait = necromancydiscipline
			has_any_necromancy_path = no
		}
		add_character_flag = has_cappadocian_necromancy_flag
	}
	
	# conditional bonus discipline
	if = {
		limit = {
			is_gargoyle_trigger = yes
			POD_visceratika_available_trigger = yes
		}
		add_trait = visceratika
	}
	
	POD_sorcerer_embrace_effect = yes
}
