POD_character_vampire_discipline_initialization_macro = {
	if = {
		limit = { age >= 500 }
		add_trait = $TRAIT1$advanced
		add_trait = $TRAIT2$advanced
		add_trait = $TRAIT3$advanced
	}
	else = {
		add_trait = $TRAIT1$
		add_trait = $TRAIT2$
		add_trait = $TRAIT3$
	}
}
