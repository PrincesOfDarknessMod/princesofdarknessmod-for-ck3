POD_become_ghoul_effect = {
	save_temporary_scope_value_as = {
		name = saved_immortal_age
		value = POD_visual_age
	}
	remove_trait = mortal
	# if = {
	# 	limit = { POD_splat_mixing_haight_mode = no }
	# 	remove_trait = kinfolk
	# 	remove_trait = gnosis
	# 	remove_trait = kinain
	# }
	
	add_trait_force_tooltip = ghoul
	if = {
		limit = { exists = scope:saved_immortal_age }
		hidden_effect = { set_immortal_age = scope:saved_immortal_age }
	}
	hidden_effect = {
		POD_remove_energy_modifiers_effect = yes
		POD_character_masquerade_initialization_effect = yes
		POD_side_effects_from_changing_splats = yes
		POD_set_bloodpotency_effect = { POTENCY = 0 }
		POD_set_hunger_modifier_effect = { AMOUNT = 4 }
		POD_ghoul_new_generation_ingested_effect = { GENERATION = 13 }
		if = { # So Historical Ghouls won't get any non-historical disciplines
			limit = { 
				has_global_variable = POD_game_after_lobby 
				NOT =  { has_character_flag = POD_ghoul_shouldnt_get_new_disciplines }
			}
			POD_trait_injector_disciplines = { EFFECT = POD_roll_for_random_discipline }
		}
		add_character_flag = POD_ghoul_shouldnt_get_new_disciplines
	}
	if = {
		limit = { is_ai = no }
		trigger_event = podintroductions.2002
	}
	add_character_flag = { flag = recent_ghoulify days = 280 }
}

POD_cease_being_ghoul_effect = {

			add_trait = mortal
		}
	}
	POD_lose_unaccessible_discipline_perks_effect = yes
	
}

POD_become_masterless_ghoul_effect = {
	remove_variable = POD_ghoul_mission
	remove_variable = POD_number_of_completed_ghoul_missions
	remove_variable = ghoul_mission_target
	every_relation = {
		type = ghoul_domitor
		prev = {
			remove_relation_ghoul_domitor = prev
			remove_character_flag = blocked_from_leaving
			remove_character_flag = blood_bonded
		}
	}
}

POD_become_bound_ghoul_effect = {
	$DOMITOR$ = { save_scope_as = domitor }
	if = { 
		limit = {
			scope:domitor.religion = {
				OR = { 
					is_in_family = rf_kindred
					this = religion:calomenasforsaken_religion
				}
			}
		}
		set_character_faith = scope:domitor.faith
	}
	else_if = {
		limit = { POD_can_participate_in_vampire_politics = yes }
		POD_character_revenant_faith_initialization_effect = yes
	}
	POD_clear_resonance_intensity_effect = yes
	POD_set_resonance_effect = yes
	if = { 
		limit = { scope:domitor = { has_perk = heavens_gate_perk } }
		add_trait_force_tooltip = revenant
	}
	if = {
		limit = { scope:domitor = { has_character_flag = drunk_grail_flag } }
		custom_tooltip = drink_grail_tt4

		add_prowess_skill = 5
		if = {
			limit = { NOT = { has_trait = physique_good } }
			add_trait = physique_good_1
		}
	}

	if = { limit = { is_ai = yes } add_character_flag = blocked_from_leaving }

	add_character_flag = blood_bonded

	POD_become_masterless_ghoul_effect = yes # To prevent multiple domitors
	set_relation_ghoul_domitor = scope:domitor

	POD_ghoul_new_generation_ingested_effect = { GENERATION = scope:domitor.POD_vampgeneration }

	scope:domitor = {
		add_hook = {
			target = PREV
			type = strong_bloodoath_hook
			years = -1
		}
		if = {
			limit = { faith = { has_doctrine_parameter = vampireghouling_illegal } }
			add_piety = major_piety_loss
		}
		else_if = {
			limit = { faith = { has_doctrine_parameter = vampireghouling_shunned } }
			add_piety = medium_piety_loss
		}
	}

	if = {
		limit = { POD_is_supehunter_trigger = yes }
		remove_trait = supehunter
		POD_hunterembrace_effect = yes	
	}

	if = {
		limit = {
			OR = {
				father ?= { 
					is_supehunter_trigger = yes
					NOT = { faith = faith:calomenasforsaken }
				}
				mother ?= { 
					is_supehunter_trigger = yes
					NOT = { faith = faith:calomenasforsaken }
				}
			}
		}
		add_trait = disinherited
	}

	if = {
		limit = { dynasty ?= { has_variable = POD_kill_upon_being_bound } }
		death = {
			death_reason = death_murder
			killer = scope:domitor
		}
	}

	save_scope_as = ghoul
	if = { # So Historical Ghouls won't get any non-historical disciplines
		limit = { has_global_variable = POD_game_after_lobby }
		custom_tooltip = {
			text = POD_become_bound_ghoul_effect_domitor_disciplines
			scope:domitor = { POD_trait_injector_disciplines = { EFFECT = POD_roll_for_domitor_discipline } }
		}
	}
}

POD_ghoul_drink_from_vampire_effect = {
	$VAMPIRE$ = { save_scope_as = vampire }
	if = {
		limit = { has_trait = ghoul }
		POD_drink_from_victim_effect = { VICTIM = scope:vampire AMOUNT = 1 }
	}
	if = {
		limit = { POD_can_be_ghouled_scheme_trigger = yes }
		POD_become_ghoul_effect = yes
		if = {
			limit = { POD_is_supehunter_trigger = yes }
			add_piety_level = -2
		}
	}

	POD_ghoul_new_generation_ingested_effect = { GENERATION = scope:vampire.POD_vampgeneration }
	
	scope:vampire = {
		remove_character_flag = POD_was_drunk_from
		add_character_flag = {
			flag = POD_was_drunk_from
			years = POD_duration_for_soft_cooldown_drinking
		}
		set_variable = {
			name = POD_was_drunk_from_times
			value = {
				value = 1
				if = { limit = { var:POD_was_drunk_from_times ?= 1 } value = 2 }
				if = { limit = { var:POD_was_drunk_from_times ?= 2 } value = 3 }
			}
			years = POD_duration_for_soft_cooldown_drinking
		}
	}
	if = { # Yes I know, this will happen even if ghoul that drinks will change. But this is only way to avoid Story Cycles I think. - VecsonON
		limit = { 
			scope:vampire = { has_character_flag = POD_was_drunk_from }
			is_unbondable_trigger = no
		}
		custom_tooltip = {
			text = POD_ghoul_drink_from_vampire_effect_the_same_vampire
			if = {
				limit = { scope:vampire.var:POD_was_drunk_from_times ?= 2 }
				random_list = {
					1 = { scope:vampire = { release_from_prison = yes } }
					1 = {}
				}
			}
			else_if = {
				limit = { scope:vampire.var:POD_was_drunk_from_times ?= 3 }
				scope:vampire = { release_from_prison = yes }
				random_list = {
					1 = { 
						POD_ghoul_drink_from_vampire_force_submission_effect = { CHANCE = 100 VAMPIRE = scope:vampire }
					}
					1 = {
						death = {
							death_reason = death_murder
							killer = scope:vampire
						}
					}
				}
			}
		}
	}
	if = {
		limit = {
			scope:vampire = {
				OR = {
					has_trait = dominatediscipline
					has_trait = presencediscipline
				}
				NOT = { has_character_modifier = bloodhunger5_modifier }
			}
			strong_mental_defenses_trigger = no
			is_unbondable_trigger = no
		}
		custom_tooltip = POD_ghoul_drink_from_vampire_effect_mental_discipline_risk
		POD_ghoul_drink_from_vampire_force_submission_effect = { CHANCE = POD_chance_for_vampire_to_use_disciplines_during_feeding VAMPIRE = scope:vampire }
	}
	if = {
		limit = { scope:vampire = { has_character_modifier = bloodhunger5_modifier } }
		scope:vampire = {
			death = {
				death_reason = death_bloodloss
				killer = PREV
			}
		}
	}
}

POD_ghoul_drink_from_vampire_force_submission_effect = {
	$VAMPIRE$ = { save_scope_as = vampire }
	if = {
		limit = { is_unbondable_trigger = no }
		random = {
			chance = $CHANCE$
			custom_tooltip = POD_ghoul_drink_from_vampire_force_submission_effect
			POD_become_bound_ghoul_effect = { DOMITOR = $VAMPIRE$ }
			$VAMPIRE$ = { 
				if = {
					limit = { is_imprisoned = yes }
					release_from_prison = yes 
				}
			}
		}
	}
}

POD_roll_for_random_discipline = {
	if = {
		limit = { NOT = { has_trait = $TRAIT$discipline } }
		save_scope_value_as = {
			name = roll_chance
			value = {
				switch = {
					trigger = trait:$TRAIT$
					trait:potence = { value = 50 }
					trait:celerity = { value = 50 }
					trait:fortitude = { value = 50 }
					trait:animalism = { value = 5 }
					trait:auspex = { value = 5 }
					trait:dominate = { value = 5 }
					trait:presence = { value = 5 }
					fallback = { value = 0 }
				}
			}
		}
		random = {
			chance = scope:roll_chance
			POD_add_trait_without_event_icon = { TRAIT = trait:$TRAIT$ }
		}
	}
}

POD_roll_for_domitor_discipline = {
	if = {
		limit = { 
			has_trait = $TRAIT$discipline
			scope:ghoul = { NOT = { has_trait = $TRAIT$discipline } }
		}
		save_scope_value_as = {
			name = roll_chance
			value = {
				if = {
					limit = { trait:$TRAIT$ = { has_trait_flag = discipline_advanced } }
					value = 40
				}
				else = { value = 20 }
			}
		}
		random = {
			chance = scope:roll_chance
			scope:ghoul = { 
				POD_add_trait_without_event_icon = { TRAIT = trait:$TRAIT$ }
				hidden_effect = {
					if = {
						limit = { has_trait = bloodsorcerydiscipline }
						POD_learn_targets_known_bloodmagic_paths = {
							TARGET = PREV
							CHANCE = 100
						}
					}
					if = {
						limit = { has_trait = necromancydiscipline }
						POD_learn_targets_known_necromancy_paths = {
							TARGET = PREV
							CHANCE = 100
						}
					}
				}
			}
		}
	}
}

}

POD_complete_ghoul_mission_effect = {
	if = {
		limit = { var:POD_ghoul_mission ?= $MISSION$ }
		var:POD_ghoul_mission ?= { save_scope_as = saved_mission }
		remove_variable = POD_ghoul_mission
		send_interface_message = {
			title = game_concept_bound_ghoul_mission
			custom_tooltip = POD_complete_ghoul_mission_effect
			POD_blood_income_effect = { COUNT = POD_vitae_gained_from_completed_mission }
			random_relation = {
				type = ghoul_domitor
				add_opinion = {
					target = PREV
					modifier = pleased_opinion
					opinion = 10
				}
				if = {
					limit = {
						NOT = {
							has_hook_of_type = {
								type = strong_bloodoath_hook
								target = PREV
							}
						}
					}
					remove_hook = { target = PREV }
					add_hook = {
						type = strong_bloodoath_hook
						target = PREV
						years = -1
					}
				}
				save_scope_as = domitor
			}
			
			##prias bonus
			if = {
				limit = {
					this = character:555022
					scope:domitor = character:790179
				}
			POD_resonance_from_feeding_on_victim_effect = { VICTIM = scope:domitor AMOUNT = 5 }
			}
			else = {
				POD_resonance_from_feeding_on_victim_effect = { VICTIM = scope:domitor AMOUNT = POD_vitae_gained_from_completed_mission }
			}
			POD_ghoul_new_generation_ingested_effect = { GENERATION = scope:domitor.POD_vampgeneration }

			if = {
				limit = { var:current_journey ?= { var:journey_type = flag:breaking_the_blond_bond } }
				POD_add_journey_progress = { AMOUNT = -10 } 
			}
			if = {
				limit = {
					any_in_list = {
						variable = paused_journeys
						count > 0
						var:journey_type = flag:breaking_the_blond_bond
						save_temporary_scope_as = journey
					}
				}
				POD_abandon_journey = { JOURNEY = scope:journey }
				custom_tooltip = POD_complete_ghoul_mission_effect_paused_breaking_bond_abandon
			}

			POD_set_or_change_variable = { VAR = POD_number_of_completed_ghoul_missions VAL = 1 }
		}
	}
}

POD_fail_ghoul_mission_effect = {
	if = {
		limit = { var:POD_ghoul_mission ?= $MISSION$ }
		var:POD_ghoul_mission ?= { save_scope_as = saved_mission }
		remove_variable = POD_ghoul_mission
		send_interface_message = {
			title = game_concept_bound_ghoul_mission
			custom_tooltip = POD_fail_ghoul_mission_effect
			random_relation = {
				type = ghoul_domitor
				add_opinion = {
					target = PREV
					modifier = respect_opinion
					opinion = -20
				}
				if = {
					limit = {
						opinion = { target = PREV value < 0 }
					}
					custom_tooltip = POD_fail_ghoul_mission_effect_risk
				}
				if = {
					limit = {
						opinion = { target = PREV value < -25 }
					}
					PREV = {
						death = {
							death_reason = death_murder
							killer = PREV
						}
					}
				}
			}
		}
	}
}
