POD_commander_trait_initialization_effect = {
    if = {
		limit = {
		    is_ventrue_clan_or_bloodline_trigger = yes
		}
		POD_ventrue_commander_trait_effect = yes
	}
	if = {
	    limit = {
	        is_brujah_trigger = yes
		}
		POD_brujah_commander_trait_effect = yes
	}
	if = {
	    limit = {
		    is_gangrel_clan_or_bloodline_trigger = yes
		}
		POD_gangrel_commander_trait_effect = yes
	}
	if = {
	    limit = {
		    is_assamite_trigger = yes
		}
		POD_banuhaqim_commander_trait_effect = yes
	}
	if = {
	    limit = {
		    is_ravnos_clan_or_bloodline_trigger = yes
		}
		POD_ravnos_commander_trait_effect = yes
	}
	if = {
	    limit = {
		    is_nosferatu_clan_or_bloodline_trigger = yes
		}
		POD_nosferatu_commander_trait_effect = yes
	}
	if = {
	    limit = {
		    is_salubri_trigger = yes
		}
		POD_salubri_commander_trait_effect = yes
	}
	if = {
	    limit = {
		    is_lasombra_trigger = yes
		}
		POD_lasombra_commander_trait_effect = yes
	}
	if = {
	    limit = {
		    is_lasombra_trigger = yes
		}
		POD_lasombra_commander_trait_effect = yes
	}
	if = {
	    limit = {
		    OR = {
			    is_caitiff_or_bushi_trigger = yes
				is_custom_bloodline_trigger = yes
			}
		}
		POD_caitiff_commander_trait_effect = yes
	}
	if = {
		limit = { is_bushi_clan_or_bloodline_trigger = yes }
		POD_bushi_lifestyle_trait_effect = yes
	}
}

POD_ventrue_commander_trait_effect = {
    if = {
	    limit = {
			dynasty ?= { has_dynasty_perk = ventrue_legacy_2 }
			has_trait = education_martial		
		}
		add_trait = leading_front
	}
}

POD_brujah_commander_trait_effect = {

	    limit = {
			dynasty ?= { has_dynasty_perk = brujah_legacy_2 }
			has_trait = education_martial		
		}
		add_trait = swift_movement
	}
}

POD_bushi_lifestyle_trait_effect = {
    if = {
		limit = {
			dynasty ?= { has_dynasty_perk = bushi_legacy_2 }
			OR = {
				has_trait = education_martial
				has_trait = education_martial_prowess
			}
		}
		add_trait = lifestyle_blademaster
	}
}
