POD_gain_grail_chivalry = {
	if = {
		limit = { has_trait = POD_grail_knight }
		# Exists for tooltip purposes
		save_temporary_scope_value_as = {
			name = new_chivalry_value
			value = {
				value = "this.has_trait_xp(POD_grail_knight)"
				add = $VALUE$
			}
		}
		add_trait_xp = {
			trait = POD_grail_knight
			value = $VALUE$
		}

		# Remove the trait if conditions are met
		if = {
			limit = {
				scope:new_chivalry_value <= 0
			}
			custom_tooltip = POD_grail_knight_loss_alert
			remove_trait = POD_grail_knight
		}
		# Update any modifiers granted by perks
		hidden_effect = {
			if = {
				limit = { has_character_modifier = POD_gkboons_inner_light_perk_modifier }
				remove_character_modifier = POD_gkboons_inner_light_perk_modifier
			}
			if = {
				limit = { POD_can_use_gkboons_inner_light_perk = yes }
				add_character_modifier = POD_gkboons_inner_light_perk_modifier
			}
		}
	}
}

# needs scope:character saved on the TopScope
# root scope is empty (global) instead of a character to prevent displaying the third-person perspective in the tooltip
POD_grail_knight_list_all_perk_effects_on_character = {
	if = {
		limit = { scope:character = { POD_has_no_gkboons_perks = yes } }
		custom_tooltip = POD_grail_knight_trait_tt_no_perks
	}
	else = {
		if = {
			limit = {
				scope:character = {
					NOT = {
						POD_has_no_gkboons_perks_of_type = { TYPE = inherited_vampire }
					}
				}
			}
			POD_save_tooltip = { LOC = gkboons_inherited_vampire }
			custom_tooltip = POD_grail_knight_trait_tt_header
		}
		POD_gkboons_type_injector_inherited_vampire  = { EFFECT = POD_grail_knight_display_perk_effect }
	}
}

# needs scope:character saved on the TopScope
# root scope is empty (global) instead of a character to prevent displaying the third-person perspective in the tooltip
POD_grail_knight_display_perk_effect = {
	if = {
		limit = { scope:character = { has_perk = $PERK$ } }
		POD_save_tooltip = { LOC = $PERK$ }
		custom_tooltip = POD_saved_tooltip_perk_effect_medium
	}
}
