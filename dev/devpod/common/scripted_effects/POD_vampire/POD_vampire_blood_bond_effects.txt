POD_bloodbond_effect = {
	POD_bloodbond_without_blood_cost_effect = {
		THRALL = $THRALL$
		REGENT = $REGENT$
	}
	$REGENT$ = { POD_hunger_rouse_check_effect = yes }
}

POD_bloodbond_without_blood_cost_effect = {
	if = {
		limit = {
			OR = {
				$REGENT$ = {
					has_hook_of_type = {
						target = $THRALL$
						type = weak_bloodoath_hook
					}
				}
				AND = {
					$THRALL$ = {
						OR = {
							AND = {
								is_tremere_trigger = yes
								NOT = { has_global_variable = POD_tremere_bane_change }
							}
							is_banpaia_trigger = yes
						}
						POD_suppress_clan_curse_trigger = no
					}
				}
			}
			
			$THRALL$ = { NOT = { has_character_flag = blood_bonded } }
			
		}
		custom_tooltip = POD_bond_prisoner_interaction_growingbloodbond
		$REGENT$ = {
			add_hook = {
				target = $THRALL$
				type = strong_bloodoath_hook
			}
		}
	}
	else = {
		$REGENT$ = {
			add_hook = {
				target = $THRALL$
				type = weak_bloodoath_hook
			}
		}
	}
}

POD_faith_consequences_of_bloodbond = {
	if = {
		limit = { faith = { has_doctrine_parameter = vampirebloodoath_illegal } }
		add_piety = major_piety_loss
	}
	else_if = {
		limit = { faith = { has_doctrine_parameter = vampirebloodoath_shunned } }
		add_piety = medium_piety_loss
	}
}
