# arguments: CHARACTER
POD_open_fleshcrafting_gui = {
	if = {
		limit = { has_variable = fleshcraft_target }
		POD_close_fleshcrafting_gui = yes
	}
	if = {
		limit = { $CHARACTER$ ?= { is_alive = yes } }
		hidden_effect_new_object = {
			
			##### step 1: set up graphics
			
			$CHARACTER$ = {
				add_character_flag = POD_fleshcraft_portrait
				add_character_flag = is_naked
				save_scope_as = fleshcraft_target
				set_variable = {
					name  = fleshcrafted_by
					value = prev
				}
			}
			
			# create a dummy character who looks like the original character's skeleton
			# the 2 characters get layered on top of each other in the GUI
			create_character = {
				save_scope_as = skeleton_dummy
				name = "Jean" # the unsung hero
				gender = scope:fleshcraft_target
				employer = character:POD_wraith_01
				random_traits = no
				age = 25 # TODO FLESHCRAFT: sometimes causes issues with very old or young characters
				faith = scope:fleshcraft_target.faith
				culture = scope:fleshcraft_target.culture
				dynasty = none
				after_creation = {
					copy_traits = scope:fleshcraft_target
					copy_inheritable_appearance_from = scope:fleshcraft_target
					
					if = {
						limit = {
							scope:fleshcraft_target = {
								POD_is_portrait_affected_by_weight = yes
							}
						}
						change_current_weight = scope:fleshcraft_target.current_weight
					}
					else = {
						add_trait = immortal
					}
					
					# remove traits that add portrait modifiers
					remove_trait = disfigured
					remove_trait = blind
					remove_trait = one_eyed
					
					add_character_flag = POD_fleshcraft_portrait
					add_character_flag = is_naked
					add_character_flag = POD_force_nudity_despite_settings
					
					if = {
						# yorak is a scrunklo and the regular skeleton's skull clips through his head
						# so he gets the headless skeleton
						limit = { scope:fleshcraft_target = character:790247 }
						add_character_flag = POD_portrait_skeleton_body_form_flag
					}
					else = {
						add_character_flag = POD_portrait_skeleton_form_flag
					}
					
					#add_character_flag = POD_portrait_skeleton_guts_form_flag
					add_character_flag = POD_portrait_no_hat_flag
					add_character_flag = POD_portrait_no_crown_flag
					add_character_flag = POD_no_beard_flag
					add_character_flag = POD_portrait_no_hair_flag
					
					add_character_flag = POD_do_not_disturb_flag
				}
			}
			
			##### step 2: set up the body parts
			# each layer of each body part has a variable attached to it
			# e.g. var:fc_torso_flesh
			# if the fleshcrafting target has a perk associated with that layer, save it as a flag
			# e.g. var:fc_torso_flesh = flag:fc_torso_adrenal_heart_perk
			# if that layer is unmodified, set the variable to flag:fc_unmodified_perk
			
			POD_fleshcrafting_init_all_layers = yes
			
			##### final step: set the variables needed to display the window itself
			
			set_variable = {
				name  = fleshcraft_target
				value = scope:fleshcraft_target
			}
			set_variable = {
				name  = fleshcraft_skeleton
				value = scope:skeleton_dummy
			}
		}
	}
}

POD_fleshcrafting_init_all_layers = {
	if = {
		limit = { has_variable = fc_selected_layer }
		remove_variable = fc_selected_layer
		clear_variable_list = fc_layer_perks
	}
	POD_fleshcrafting_layer_injector = {
		EFFECT = POD_init_fleshcrafting_layer_variable
	}
	# also add the perks to a variable list, which is easier to iterate over in some circumstances
	POD_fleshcrafting_refresh_list_of_selected_perks = yes
}

# needs scope:fleshcraft_target on the TopScope
POD_init_fleshcrafting_layer_variable = {
	save_scope_value_as = {
		name  = fc_perk_for_current_layer
		value = flag:fc_unmodified_perk
	}
	scope:fleshcraft_target = {
		POD_perk_injector_$LAYER$ = { EFFECT = POD_save_fleshcrafting_perk_as_scope_value }
	}
	set_variable = {
		name  = $LAYER$
		value = scope:fc_perk_for_current_layer
	}
}

POD_save_fleshcrafting_perk_as_scope_value = {
	if = {
		limit = { has_perk = $PERK$ }
		save_scope_value_as = {
			name  = fc_perk_for_current_layer
			value = flag:$PERK$
		}
	}
}

POD_clear_fleshcrafting_layer_variable = {
	remove_variable = $LAYER$
}

POD_fleshcrafting_refresh_list_of_selected_perks = {
	clear_variable_list = fc_all_selected_perks
	POD_fleshcrafting_layer_injector = {
		EFFECT = POD_fleshcrafting_add_perk_to_list_of_selections
	}
}

POD_fleshcrafting_add_perk_to_list_of_selections = {
	if = {
		limit = { var:$LAYER$ != flag:fc_unmodified_perk }
		add_to_variable_list = {
			name   = fc_all_selected_perks
			target = var:$LAYER$
		}
	}
}

POD_fleshcrafting_select_layer_for_modification = {
	clear_variable_list = fc_layer_perks
	if = {
		limit = { var:fc_selected_layer ?= $LAYER$ }
		remove_variable = fc_selected_layer
	}
	else = {
		set_variable = {
			name  = fc_selected_layer
			value = $LAYER$
		}
		add_to_variable_list = {
			name   = fc_layer_perks
			target = flag:fc_unmodified_perk
		}
		# sequence of injectors:
		# 1. the layer is passed to this effect as a flag, e.g. flag:fc_torso_flesh
		# 2. from that flag, select the right injector. e.g. POD_perk_injector_fc_torso_flesh
		# 3. all perks in that injector get added to the fc_layer_perks list (as flags)
		POD_fleshcrafting_layer_injector = {
			EFFECT = POD_fleshcrafting_select_layer_from_variable
		}
	}
}

POD_fleshcrafting_select_layer_from_variable = {
	if = {
		limit = { var:fc_selected_layer = flag:$LAYER$ }
		POD_perk_injector_$LAYER$ = {
			EFFECT = POD_fleshcrafting_init_perk_choices_for_layer
		}
	}
}

POD_fleshcrafting_init_perk_choices_for_layer = {
	add_to_variable_list = {
		name   = fc_layer_perks
		target = flag:$PERK$
	}
	POD_error_suppression_unused_flag = { FLAG = $PERK$ }
}

# needs scope:layer on the TopScope
POD_fleshcrafting_clear_layer_perk_from_specific_layer = {
	if = {
		limit = { var:fc_selected_layer ?= scope:layer }
		remove_variable = fc_selected_layer
		clear_variable_list = fc_layer_perks
	}
	POD_fleshcrafting_layer_injector = {
		EFFECT = POD_fleshcrafting_clear_layer_variable_from_specific_layer
	}
	POD_fleshcrafting_refresh_list_of_selected_perks = yes
}

POD_fleshcrafting_clear_layer_variable_from_specific_layer = {
	if = {
		limit = { scope:layer = flag:$LAYER$ }
		set_variable = {
			name  = $LAYER$
			value = flag:fc_unmodified_perk
		}
	}
}

POD_fleshcrafting_clear_all_layer_perks = {
	if = {
		limit = { has_variable = fc_selected_layer }
		remove_variable = fc_selected_layer
		clear_variable_list = fc_layer_perks
	}
	POD_fleshcrafting_layer_injector = {
		EFFECT = POD_fleshcrafting_clear_layer_variable
	}
	POD_fleshcrafting_refresh_list_of_selected_perks = yes
}

POD_fleshcrafting_clear_layer_variable = {
	set_variable = {
		name  = $LAYER$
		value = flag:fc_unmodified_perk
	}
}

# needs scope:flag on the TopScope
POD_fleshcrafting_choose_perk = {
	POD_fleshcrafting_layer_injector = {
		EFFECT = POD_fleshcrafting_set_variable_from_layer
	}
	remove_variable = fc_selected_layer
	clear_variable_list = fc_layer_perks
	POD_fleshcrafting_refresh_list_of_selected_perks = yes
}

POD_fleshcrafting_set_variable_from_layer = {
	if = {
		limit = { var:fc_selected_layer = flag:$LAYER$ }
		set_variable = {
			name  = $LAYER$
			value = scope:flag
		}
	}
	POD_error_suppression_unset_flag = { FLAG = $LAYER$ }
}

POD_fleshcrafting_gui_preview = {
	if = {
		limit = { POD_fleshcrafting_target_has_all_selected_perks = yes }
		custom_description_no_bullet = {
			text = POD_fleshcrafting_tutorial
		}
	}
	else_if = {
		limit = { POD_can_fleshcraft = yes }
		
		POD_fleshcrafting_effect = {
			SHOW_SUCCESS_CHANCE = no
		}
	}
}

# argument: SHOW_SUCCESS_CHANCE
POD_fleshcrafting_effect = {
	POD_willpower_rouse_check_effect = yes
	POD_discipline_cost_effect = {
		COUNT = POD_fleshcrafting_blood_cost
	}
	
	if = {
		limit = {
			var:fleshcraft_target = {
				this != prev
				NOT = { has_relation_ghoul_domitor = prev }
				NOT = { has_character_modifier = recently_tortured }
				NOT = { has_character_flag = is_being_tortured }
			}
		}
		custom_description_no_bullet = { text = POD_linebreak }
		custom_description_no_bullet = {
			text = POD_fleshcrafting_torture_consequences
			object = var:fleshcraft_target
			save_scope_as = actor
			var:fleshcraft_target = { save_scope_as = recipient }
			POD_torture_interaction_no_actor_effect = yes
		}
	}
	
	custom_description_no_bullet = { text = POD_linebreak }
	
	duel = {
		skill = learning
		# arbitrary value, actually unused. the duel is just for tooltip purposes
		value = 10
		#value = POD_fleshcrafting_difficulty
		
		0 = {
			show_chance = $SHOW_SUCCESS_CHANCE$
			desc = POD_fleshcrafting_success
			
			modifier = {
				add = POD_fleshcrafting_success_chance
			}
			
			# compare_modifier = {
			# 	value = scope:duel_value
			# 	multiplier = medium_positive_duel_skill_multiplier # 5
			# }
			# modifier = {
			# 	add = POD_fleshcrafting_flat_success_bonuses
			# }
			# modifier = {
			# 	add = 9999999
			# 	has_perk = perfect_essence_perk
			# 	this = var:fleshcraft_target
			# }
			
			send_interface_message = {
				type = send_interface_message_good
				title = POD_fleshcrafting_success_message
				#left_icon = this
				right_icon = var:fleshcraft_target

				#########################################
				#######   FLESHCRAFTING SUCCESS   #######
				#########################################
				
				# remove all perks
				# TODO FLESHCRAFT: refactor so removed perks are visible
				hidden_effect = {
					var:fleshcraft_target = {
						POD_perk_injector_fleshcrafting = {
							EFFECT = POD_remove_perk
						}
					}
				}
				
				# add fleshcrafted trait + xp if at least one perk was selected
				if = {
					limit = { any_in_list = { variable = fc_all_selected_perks } }
					var:fleshcraft_target = {
						add_trait = fleshcrafted
						
						save_scope_value_as = {
							name = fleshcrafted_xp_delta
							value = {
								value = prev.POD_fleshcrafting_current_complexity
								subtract = "this.has_trait_xp(fleshcrafted)"
							}
						}
						if = {
							limit = { scope:fleshcrafted_xp_delta != 0 }
							add_trait_xp = {
								trait = fleshcrafted
								value = scope:fleshcrafted_xp_delta
							}
						}
					}
				}
				# remove fleshcrafted trait if the target has no perks afterwards
				else = {
					var:fleshcraft_target = {
						remove_trait = fleshcrafted
					}
				}
				
				# applying the actual perks
				POD_fleshcrafting_layer_injector = {
					EFFECT = POD_set_fleshcrafting_perk_from_layer_variable
				}
				
				hidden_effect = {
					#For Spirit Lobe
					var:fleshcraft_target = {
						every_relation = {
							type = umbra_employer
							trigger_event = { id = POD_umbra_expedition.7000 days = 1 }
						}
					}
					
					# close gui (but only on a success)
					POD_close_fleshcrafting_gui = yes
				}
			}
		}
		
		100 = {
			show_chance = $SHOW_SUCCESS_CHANCE$
			desc = POD_fleshcrafting_failure
			
			modifier = {
				add = {
					value = POD_fleshcrafting_success_chance
					multiply = -1
				}
			}
			
			# compare_modifier = {
			# 	value = scope:duel_value
			# 	multiplier = low_negative_duel_skill_multiplier # -2.5
			# }
			# modifier = {
			# 	add = {
			# 		value = POD_fleshcrafting_flat_success_bonuses
			# 		multiply = -1
			# 	}
			# }
			# modifier = {
			# 	factor = 0
			# 	has_perk = perfect_essence_perk
			# 	this = var:fleshcraft_target
			# }
			
			send_interface_toast = {
				type = event_toast_effect_bad
				title = POD_fleshcrafting_failure_message
				left_icon = this
				right_icon = var:fleshcraft_target
				
				#########################################
				#######   FLESHCRAFTING FAILURE   #######
				#########################################

				# yowch
				var:fleshcraft_target = {
					POD_increase_wounds_hostile_effect = {
						ATTACKER = prev
						REASON = fleshcrafting
						COUNT = POD_base_wound_damage
					}
				}
			}
		}
	}
}

POD_set_fleshcrafting_perk_from_layer_variable = {
	var:$LAYER$ = { save_scope_as = fc_perk_to_add }
	var:fleshcraft_target = {
		POD_perk_injector_$LAYER$ = {
			EFFECT = POD_add_fleshcrafting_perk
		}
	}
	clear_saved_scope = fc_perk_to_add
}

POD_close_fleshcrafting_gui = {
	var:fleshcraft_target ?= {
		remove_character_flag = POD_fleshcraft_portrait
		remove_character_flag = is_naked
		remove_variable = fleshcrafted_by
	}
	var:fleshcraft_skeleton ?= {
		death = { death_reason = death_disappearance }
	}
	remove_variable = fleshcraft_target
	remove_variable = fleshcraft_skeleton
	remove_variable = fc_selected_layer
	clear_variable_list = fc_layer_perks
	clear_variable_list = fc_all_selected_perks
	POD_fleshcrafting_layer_injector = {
		EFFECT = POD_clear_fleshcrafting_layer_variable
	}
}

POD_fleshcrafting_gui_cleanup_on_death = {
	if = {
		limit = { has_variable = fleshcrafted_by }
		var:fleshcrafted_by = { POD_close_fleshcrafting_gui = yes }
	}
	if = {
		limit = { has_variable = fleshcraft_target }
		POD_close_fleshcrafting_gui = yes
	}
}

POD_transmogrify_set_rarity_effect = {
	if = {
		limit = { scope:recipient.$SKILL$ >= POD_transmogrify_very_high_skill }
		set_artifact_rarity = illustrious
	}
	else_if = {
		limit = { scope:recipient.$SKILL$ >= POD_transmogrify_high_skill }
		set_artifact_rarity = famed
	}
	else_if = {
		limit = { scope:recipient.$SKILL$ >= POD_transmogrify_medium_skill }
		set_artifact_rarity = masterwork
	}
	#Common by default
}
