POD_character_vampire_inherit_clan_effect = {
	scope:created_vampirechilde = {
		if = { 
			limit = { has_POD_vampire_clan_trigger = no }
			scope:vampiresire = {
				random_character_trait = {
					limit = { has_trait_flag = vamp_clan }
					save_scope_as = sire_clan
				}
			}
			if = {
				limit = { NOT = { exists = scope:sire_clan } }
				trait:caitiff = { save_scope_as = sire_clan }
			}
			add_trait = scope:sire_clan
			if = {
				limit = { scope:sire_clan = trait:gargoyle }
				POD_character_vampire_inherit_gargoyle_caste_effect = yes
			}
		}
	}
}
