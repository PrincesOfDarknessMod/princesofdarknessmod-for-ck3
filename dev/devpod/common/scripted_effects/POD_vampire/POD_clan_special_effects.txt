POD_character_vampire_malkavian_initialization_effect = {
	if = {
		limit = {
			is_malkavian_trigger = yes
		}
		# Make madness traits congenital so they can't be removed by events/interactions
		if = {
			limit = { has_trait = lunatic_1 }
			set_trait_rank = {
				trait = lunatic
				rank = 2
			}
		}
		else_if = {
			limit = { has_trait = possessed_1 }
			set_trait_rank = {
				trait = possessed
				rank = 2
			}
		}
		else_if = {
			limit = { has_trait = depressed_1 }
			set_trait_rank = {
				trait = depressed
				rank = 2
			}
		}
		else_if = {
			limit = {
				# If they already have a madness trait, don't give them another one
				NOR = {
					has_trait = lunatic_genetic
					has_trait = possessed_genetic
					has_trait = depressed_genetic
				}
			}
			POD_character_vampire_madness_initialization_effect = yes
		}
	}
}
