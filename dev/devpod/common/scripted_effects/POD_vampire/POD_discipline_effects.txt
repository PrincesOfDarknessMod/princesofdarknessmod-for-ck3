	save_scope_as = current
	if = {
		limit = {
			is_alive = yes
			NOT = { has_trait = golconda }
		}
		$CHARACTER$ = {
			add_to_variable_list = {
				name = bloodline
				target = scope:current
			}
		}
	}
	every_child = {
		even_if_dead = yes
		add_children_to_list = { CHARACTER = $CHARACTER$ }
	}
}

POD_dominate_effect = {
	if = {
		limit = {
			has_hook_of_type = {
				target = $TARGET$
				type = conditioning_hook
			}
		}
		remove_hook = {
			target = $TARGET$
			type = conditioning_hook
		}			
		add_hook = { # refresh cooldown
			target = $TARGET$
			type = conditioning_hook
			days = dominate_duraction
		}
	}
	else_if = {
		limit = {
			has_hook_of_type = {
				target = $TARGET$
				type = rationalize_hook
			}
		}
		remove_hook = {
			target = $TARGET$
			type = rationalize_hook
		}	
		add_hook = {
			target = $TARGET$
			type = conditioning_hook
			days = dominate_duraction
		}
	}
	else_if = {
		limit = {
			has_hook_of_type = {
				target = $TARGET$
				type = mesmerize_hook
			}
		}
		remove_hook = {
			target = $TARGET$
			type = mesmerize_hook
		}	
		add_hook = {
			target = $TARGET$
			type = rationalize_hook
			days = dominate_duraction
		}
	}
	else = {
		remove_hook = { target = $TARGET$ }
		add_hook = {
			target = $TARGET$
			type = mesmerize_hook
			days = dominate_duraction
		}
	}
	if = {
		limit = { NOT = { has_relation_dominate_target = $TARGET$ } }
		set_relation_dominate_target = $TARGET$
	}
	$TARGET$ = { save_scope_as = dominate_target }
	trigger_event = { id = POD_mesmerize.700 days = dominate_duraction }
}

POD_can_obtain_discipline_of_level_effect = {
	if = {
		limit = {
			NOT = { POD_can_obtain_discipline_of_level_trigger = { DISCIPLINE = $DISCIPLINE$ LEVEL = $LEVEL$ } }
		}
		if = {
			limit = { NOT = { has_trait = $DISCIPLINE$discipline } }
			custom_description_no_bullet = { text = NEEDS_DISCIPLINE_TEXT }
		}
		if = {
			limit = { 
				NOT = { POD_can_obtain_discipline_of_level_elder_advanced_effect = { DISCIPLINE = $DISCIPLINE$ LEVEL = $LEVEL$ } } 
				NOR = { # Exceptions
					trait:$DISCIPLINE$ = trait:valeren
				}
			}
			custom_description_no_bullet = { text = POD_can_obtain_discipline_of_level_elder_advanced_effect object = trait:$DISCIPLINE$advanced }
		}
		if = {
			limit = { NOT = { POD_can_obtain_discipline_of_level_ghoul_trigger = { DISCIPLINE = $DISCIPLINE$ LEVEL = $LEVEL$ } } }
			switch = {
				trigger = has_trait
				ghoul = { custom_description_no_bullet = { text = POD_can_obtain_discipline_of_level_ghoul_effect value = $LEVEL$ } }
				revenant = { custom_description_no_bullet = { text = POD_can_obtain_discipline_of_level_revenant_effect value = $LEVEL$ } }
			}
		}
		if = {
			limit = { NOT = { POD_can_obtain_discipline_of_level_vampire_trigger = { LEVEL = $LEVEL$ } } }
			save_scope_value_as = {
				name = generation_conversion
				value = {
					switch = {
						trigger = $LEVEL$
						7 = { value = 7 }
						8 = { value = 6 }
						9 = { value = 5 }
						10 = { value = 4 }
						11 = { value = 3 }
						fallback = { value = 13 }
					}
				}
			}
			custom_description_no_bullet = { text = POD_can_obtain_discipline_of_level_vampire_effect value = scope:generation_conversion }
		}
	}
}

POD_deduct_extra_perk_cost_effect = {
	if = {
		limit = {
			POD_save_discipline_perk_cost_values = {
				LEVEL = $LEVEL$
				TRAIT = trait:$DISCIPLINE$
			}
		}
	}
	save_scope_value_as = {
		name = extra_perk_points_deducted
		value = {
			value = POD_added_discipline_perk_cost
			multiply = -1
		}
	}
	# prevent errors from temporary scopes sticking around
	clear_saved_scope = discipline_1
	clear_saved_scope = discipline_2
	clear_saved_scope = discipline_3
	if = {
		limit = { scope:extra_perk_points_deducted < 0 }
		custom_tooltip = POD_scaling_discipline_perk_cost
		hidden_effect = {
			POD_add_resonance_perk_points_for_trait = {
				TRAIT = trait:$DISCIPLINE$
				COUNT = scope:extra_perk_points_deducted
			}
		}
	}
}
