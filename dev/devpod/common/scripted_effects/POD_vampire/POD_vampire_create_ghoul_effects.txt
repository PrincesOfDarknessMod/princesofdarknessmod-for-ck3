POD_player_character_create_ghoul_effect = {
	scope:ghoulthrall = {
		# faith first
		# (so that calomena's forsaken don't get a masquerade modifier in the ghoul init effect)
		if = { 
			limit = {
				scope:domitor.religion = {
					OR = { 
						is_in_family = rf_kindred
						this = religion:calomenasforsaken_religion
					}
				}
			}
			set_character_faith = scope:domitor.faith
		}
		else_if = {
			limit = { POD_can_participate_in_vampire_politics = yes }
			POD_character_revenant_faith_initialization_effect = yes
		}
		POD_become_ghoul_effect = yes
		POD_clear_resonance_intensity_effect = yes
		POD_set_resonance_effect = yes
		if = { 
			limit = {
				scope:domitor = { has_perk = heavens_gate_perk }
			}
			add_trait_force_tooltip = revenant
		}
	}

	hidden_effect = {
		scope:ghoulthrall = {
			if = {
				limit = { has_POD_vampire_discipline_trigger = no }
				POD_character_ghoul_discipline_initialization_effect = yes
			}
		}
	}
		
	scope:domitor = {
		if = {
			limit = { faith = { has_doctrine_parameter = vampireghouling_illegal } }
			add_piety = major_piety_loss
		}
		else_if = {
			limit = { faith = { has_doctrine_parameter = vampireghouling_shunned } }
			add_piety = medium_piety_loss
		}
	}
}

POD_become_ghoul_effect = {
	save_temporary_scope_value_as = {
		name = saved_immortal_age
		value = POD_visual_age
	}
	remove_trait = mortal
	if = { 
		limit = {
			has_trait = supehunter 
			NOT = { has_character_flag = chargen_changing_splat_flag }
			scope:domitor ?= { POD_is_supehunter_trigger = no }
		}
		scope:domitor = { save_scope_as = vampiresire }
		remove_trait = supehunter
		POD_hunterembrace_effect = yes
	}
	if = {
		limit = { POD_splat_mixing_haight_mode = no }
		if = {
			limit = { has_trait = kinfolk }
			remove_trait = kinfolk
			if = {
				limit = { has_trait = gnosis }
				remove_trait = gnosis
			}
		}
		remove_trait = kinain
	}
	if = {
		limit = { has_trait = changeling }
		add_trait = lunatic_genetic # Dauntain
	}
	
	add_trait_force_tooltip = ghoul
	if = {
		limit = { exists = scope:saved_immortal_age }
		hidden_effect = { set_immortal_age = scope:saved_immortal_age }
	}
	hidden_effect = {
		POD_remove_wrong_supernatural_traits_on_existing_character = yes
		POD_fully_refill_energy = yes
		POD_remove_supernatural_relation_effect = yes
		POD_side_effects_from_changing_splats = yes
	}
	if = {
		limit = {
			has_masquerade_risk_trigger = no
			POD_splat_has_masquerade_trigger = yes
		}
		POD_character_masquerade_initialization_effect = yes
	}
	if = {
		limit = { has_any_blood_potency_modifier = no }
		add_character_modifier = bloodpotency0_modifier
	}
	if = {
		limit = { is_ai = no }
		trigger_event = podintroductions.2002
	}
	add_character_flag = {
		flag = recent_ghoulify
		days = 280
	}
}

POD_stop_being_ghoul = {
	if = {
		limit = { has_trait = ghoul }
		remove_trait = ghoul
		POD_clear_ghoul_domitor_relations = yes
		hidden_effect = {
			POD_remove_wrong_supernatural_traits_on_existing_character = yes
		}
	}
}

POD_clear_ghoul_domitor_relations = {
	every_relation = {
		type = ghoul_domitor
		remove_hook = {
			target = prev
			type = strong_bloodoath_hook
		}
		prev = {
			remove_relation_ghoul_domitor = prev
			remove_character_flag = blocked_from_leaving
			remove_character_flag = blood_bonded
		}
	}
}
