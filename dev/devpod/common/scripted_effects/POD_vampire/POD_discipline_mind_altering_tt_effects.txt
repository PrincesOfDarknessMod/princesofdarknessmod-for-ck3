POD_mesmerize_attacker_tt_effect = {
	if = {
		limit = { 
			scope:attacker = {
				OR = {
					has_perk = rationalize_perk
				} 
			} 
		}
		custom_tooltip = POD_mesmerize.tt.pc.rationalize_perk
	}
	if = {
		limit = { scope:attacker = { has_trait = dominateadvanced } }
		custom_tooltip = POD_mesmerize.tt.pc.dominateadvanced
	}
	if = {
		limit = { scope:attacker = { has_perk = will_conquers_perk } }
		custom_tooltip = POD_mesmerize.tt.pc.will_conquers_perk
	}
	if = {
		limit = {
			scope:defender = { has_POD_vampire_generation_trigger = yes }
			scope:attacker.generation_value > scope:defender.generation_value
		}
		custom_tooltip = POD_mesmerize_tt_attacker_generation_adv
	}
	#mortals have lowered defenses
	if = {
		limit = { scope:defender = { has_trait = mortal } }
		custom_tooltip = POD_mesmerize.tt.mortal
	}
	#personality bonus or malus
	if = {
		limit = { scope:defender = { has_personality_submissive_trigger = yes } }
		custom_tooltip = POD_mesmerize.tt.submissive_personality
	}
}
