POD_character_vampire_bloodpotency_initialization_effect = {
	save_scope_value_as = {
		name  = generation
		value = POD_vampgeneration
	}
	if = {
		limit = { scope:generation = 0 } # safeguard for characters with no generation trait
		add_character_modifier = bloodpotency0_modifier
	}
	else_if = {
		limit = { scope:generation = 1 }
		add_character_modifier = bloodpotency10_modifier
	}
	else_if = {
		limit = { scope:generation = 2 }
		add_character_modifier = bloodpotency9_modifier
	}
	else_if = {
		limit = { scope:generation = 3 }
		add_character_modifier = bloodpotency8_modifier
	}
	else_if = {
		limit = { scope:generation = 4 }
		add_character_modifier = bloodpotency5_modifier
	}
	else_if = {
		limit = { scope:generation = 5 }
		add_character_modifier = bloodpotency4_modifier
	}
	else_if = {
		limit = { scope:generation < 8 }
		add_character_modifier = bloodpotency3_modifier
	}
	else_if = {
		limit = { scope:generation < 10 }
		add_character_modifier = bloodpotency2_modifier
	}
	else_if = {
		limit = { scope:generation < 14 }
		add_character_modifier = bloodpotency1_modifier
	}
	else = {
		add_character_modifier = bloodpotency0_modifier
	}
}

POD_set_bloodpotency_effect = {
	hidden_effect = { POD_remove_bloodpotency_modifiers_effect = yes }
	switch = {
		trigger = $POTENCY$
		0 = { add_character_modifier = bloodpotency0_modifier }
		1 = { add_character_modifier = bloodpotency1_modifier }
		2 = { add_character_modifier = bloodpotency2_modifier }
		3 = { add_character_modifier = bloodpotency3_modifier }
		4 = { add_character_modifier = bloodpotency4_modifier }
		5 = { add_character_modifier = bloodpotency5_modifier }
		6 = { add_character_modifier = bloodpotency6_modifier }
		7 = { add_character_modifier = bloodpotency7_modifier }
		8 = { add_character_modifier = bloodpotency8_modifier }
		9 = { add_character_modifier = bloodpotency9_modifier }
		10 = { add_character_modifier = bloodpotency10_modifier }
	}
}

POD_bloodpotency_cooldown_effect = {
	add_character_flag = {
		flag = bloodpotency_increased
		years = 30
	}
	add_character_flag = {
		flag = bloodpotency_decreased
		years = 30
	}
}

POD_bloodpotency_decrease_effect = {
	save_scope_value_as = {
		name = new_bloodpotency
		value = {
			value = blood_potency_value
			subtract = 1
			min = blood_potency_minimum
		}
	}
	POD_set_bloodpotency_effect = {
		POTENCY = scope:new_bloodpotency
	}
}
