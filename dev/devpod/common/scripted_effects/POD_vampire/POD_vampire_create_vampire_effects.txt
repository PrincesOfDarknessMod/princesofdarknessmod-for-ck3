	}

			POD_character_vampire_inherit_generation_effect = yes
			POD_character_vampire_inherit_clan_effect = yes

			#POD_clear_resonance_effect = yes
			POD_set_resonance_effect = yes
			
			POD_commander_trait_initialization_effect = yes
			POD_salubri_caste_new_vampire_initialization_effect = yes
			
			if = {
				limit = { POD_splat_has_disease_immunity = yes }
				POD_remove_disease_traits_effect = yes
			}
			
			copy_inheritable_appearance_from = scope:created_vampirechilde_dna
			
			if = {
				limit = {
					house.house_head ?= { is_amenti_trigger = yes }
				}
				set_relation_worshipped_reborn = scope:child.house.house_head
			}
				
			if = {
				limit = { scope:vampiresire = { is_a_cultist = yes } }

			}
		}

		clear_saved_scope = created_vampirechilde_dna
		clear_saved_scope = vampire_gender
		
		scope:vampiresire = {
			if = {
				limit = { highest_held_title_tier >= tier_county }
				add_courtier = scope:created_vampirechilde
				scope:created_vampirechilde = { return_to_court = yes }
			}
		}
	}

fera_embrace_effect = {
	add_trait = abomination
	if = {
		limit = {
			OR = {
				POD_splat_mixing_haight_mode = yes
				has_trait = werewolf 
				has_trait = ajaba
				has_trait = bastet # TODO:trigger death without gnosis
				has_trait = gurahl
			}
		}
		POD_save_fera_shape_effect = yes
		POD_remove_fera_on_embrace_effect = yes
	}
	else_if = {
		limit = {
			OR = {
				has_trait = mokole # TODO:trigger death with frenzy
				has_trait = rokea
			}
		}
		POD_save_fera_shape_effect = yes
		POD_remove_fera_on_embrace_effect = yes
		POD_become_wight_effect = yes
	}
	else_if = {
		limit = { has_trait = corax }
		death = { death_reason = death_sun }
		if = {
			limit = { 
				exists = scope:vampiresire 
				scope:vampiresire = { POD_sunny_again_trigger = no }
			}
			scope:vampiresire = { pod_death_no_killer = { death_reason = death_sun } }
		}
	}
	else_if = {
		limit = { has_trait = kitsune }
		death = { death_reason = death_fire }
		if = {
			limit = { 
				exists = scope:vampiresire 
				scope:vampiresire = { POD_immune_to_fire_trigger = no }
			}
			scope:vampiresire = { pod_death_no_killer = { death_reason = death_fire } }
		}
	}
	else_if = {
		limit = { has_trait = ratkin }
		death = { death_reason = death_old_age }
	}
	else = {
		death = { death_reason = death_failed_embrace }
	}
}

	    }
	}
}

		text = POD_create_random_vampire_effect
		value = $GEN$
		save_scope_as = created_vampirechilde
		random_living_character = {
			limit = { 
				has_trait = vampgeneration$GEN$
				age > PREV.age
			}
			weight = { # Closer Sire, Better Chance
				base = 50
				modifier = {
					add = {
						add = "scope:created_vampirechilde.location.squared_distance(THIS)"
						multiply = -1
					}
				}

			save_scope_as = vampiresire
		}
		POD_remove_vampire_generation_traits_effect = yes
		POD_remove_vampire_clan_traits_effect = yes

	}
