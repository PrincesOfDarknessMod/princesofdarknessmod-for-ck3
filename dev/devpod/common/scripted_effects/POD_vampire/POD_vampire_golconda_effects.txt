POD_achieve_golconda_effect = {
	if = {
		limit = { has_trait = golconda_seeker }
		remove_trait = golconda_seeker
	}
	if = {
		limit = { NOT = { has_trait = golconda } }
		add_trait = golconda
	}
	hidden_effect = {
		POD_remove_hunger_modifiers_effect = yes
		add_character_modifier = bloodhunger0_modifier
		
		every_relation = {
			type = herd_member
			prev = { POD_remove_herd_no_masquerade_effect = { TARGET = prev } }
		}

		POD_remove_clan_curse_effect = yes
		
		POD_abandon_journeys_of_type_effect = { TYPE = golconda_2 }

		if = {
			limit = {
				is_ai = no
				this = character:790236
			}
			set_global_variable = pod_achiev_qawiyya # Achievement: Make Peace, not War
		}
	}
}

POD_become_wight_effect = {
	add_trait = wight
	add_prestige = massive_prestige_loss

	if = {
		limit = {
			OR = {
				has_trait = bardo
				has_trait = bardoadvanced
			}
		}
		remove_trait = bardo
		remove_trait = bardoadvanced
		refund_perks = melancholy_lifestyle
	}

	if = {
		limit = { is_kueijin_trigger = yes }
		hidden_effect = {
			POD_clean_dharma_effect = yes
		}
		add_trait = dharma0
	}

	hidden_effect = {
		if = {
			limit = { has_trait = golconda_seeker }
			add_trait_xp = { trait = golconda_seeker value = -100 }
		}
		every_relation = {
			type = friend
			remove_relation_friend = prev
		}
		every_relation = {
			type = best_friend
			remove_relation_best_friend = prev
		}
		every_relation = {
			type = lover
			remove_relation_lover = prev
		}
		every_relation = {
			type = soulmate
			remove_relation_soulmate = prev
		}
	}

	POD_depose_effect = yes
}

POD_remove_clan_curse_effect = {
	POD_remove_visual_clan_curse_effect = yes
	if = {
		limit = { is_tzimisce_trigger = yes }
		add_character_modifier = uncursed_tzimisce_modifier
	}
	else_if = {
		limit = { is_malkavian_trigger = yes }
		POD_suppress_clan_curse_trait = { TRAIT = trait:lunatic_genetic }
		POD_suppress_clan_curse_trait = { TRAIT = trait:possessed_genetic }
		POD_suppress_clan_curse_trait = { TRAIT = trait:depressed_genetic }
	}
}

POD_remove_visual_clan_curse_effect = {
	if = {
		limit = { is_nosferatu_trigger = yes }
		add_character_modifier = uncursed_nosferatu_modifier
		hidden_effect = {
			if = {
				limit = { NOT = { has_trait = beauty_good } }
				POD_suppress_visual_clan_curse_trait = { TRAIT = trait:beauty_bad_1 }
				POD_suppress_visual_clan_curse_trait = { TRAIT = trait:beauty_bad_2 }
				POD_suppress_visual_clan_curse_trait = { TRAIT = trait:beauty_bad_3 }
				# Normalize Genes
				add_trait = beauty_good_3
				remove_trait = beauty_good_3
			}
			POD_suppress_visual_clan_curse_trait = { TRAIT = trait:scaly }
			POD_suppress_visual_clan_curse_trait = { TRAIT = trait:disfigured }
			POD_suppress_visual_clan_curse_trait = { TRAIT = trait:spindly }
			POD_suppress_visual_clan_curse_trait = { TRAIT = trait:hunchbacked }
			POD_suppress_clan_curse_flag = { FLAG = POD_pale_skin_look_flag }
			POD_suppress_clan_curse_flag = { FLAG = POD_ivory_skin_flag }
			POD_suppress_clan_curse_flag = { FLAG = POD_bald_look_flag }
		}
	}
}

POD_suppress_clan_curse_trait = {
	if = {
		limit = { has_trait = $TRAIT$ }
		add_to_variable_list = {
			name   = suppressed_clan_curse_traits
			target = $TRAIT$
		}
		remove_trait = $TRAIT$
	}
}

	}

POD_suppress_clan_curse_flag = {
	if = {
		limit = { has_character_flag = $FLAG$ }
		remove_character_flag = $FLAG$
		add_character_flag = $FLAG$_disabled
	}
}

POD_reapply_clan_curse_effect = {
	POD_reapply_visual_clan_curse_effect = yes
	if = {
		limit = { POD_suppress_clan_curse_trigger = no }
		if = {
			limit = { has_character_modifier = uncursed_tzimisce_modifier }
			remove_character_modifier = uncursed_tzimisce_modifier
		}
		every_in_list = {
			variable = suppressed_clan_curse_traits
			prev = { add_trait = prev }
		}
		clear_variable_list = suppressed_clan_curse_traits
	}
}

POD_reapply_visual_clan_curse_effect = {
	if = {
		limit = {
			is_nosferatu_trigger = yes
			POD_suppress_visual_clan_curse_trigger = no
		}
		if = {
			limit = { has_character_modifier = uncursed_nosferatu_modifier }
			remove_character_modifier = uncursed_nosferatu_modifier
		}
		hidden_effect = {
			remove_trait = beauty_good_1
			remove_trait = beauty_good_2
			remove_trait = beauty_good_3
			every_in_list = {
				variable = suppressed_visual_clan_curse_traits
				prev = { add_trait = prev }
			}
			clear_variable_list = suppressed_visual_clan_curse_traits
			POD_reapply_clan_curse_flag = { FLAG = POD_pale_skin_look_flag }
			POD_reapply_clan_curse_flag = { FLAG = POD_ivory_skin_flag }
			POD_reapply_clan_curse_flag = { FLAG = POD_bald_look_flag }
		}
	}
}

POD_reapply_clan_curse_flag = {
	if = {
		limit = { has_character_flag = $FLAG$_disabled }
		remove_character_flag = $FLAG$_disabled
		add_character_flag = $FLAG$
	}
}

POD_lose_golconda_effect = {
	if = {
		limit = {
			has_trait = golconda
		}
		remove_trait_force_tooltip = golconda
		add_trait_force_tooltip = golconda_seeker
		POD_golconda_seeker_maintenance_effect = yes
		POD_lose_elder_perks_effect = yes
		POD_reapply_clan_curse_effect = yes
		trigger_event = POD_golconda_journey.1000
	}
	if = {
		limit = {
			OR = {
				POD_is_on_golconda_journey_trigger = yes
				AND = {
					has_trait = golconda_seeker
					has_trait_xp = { trait = golconda_seeker value > 0 }
				}
			}
		}
		if = {
			limit = { POD_is_on_golconda_journey_trigger = yes }
			POD_abandon_journeys_of_type_effect = { TYPE = golconda_2 }
		}
		if = {
			limit = {
				has_trait = golconda_seeker
				has_trait_xp = { trait = golconda_seeker value > 0 }
			}
			add_trait_xp = { trait = golconda_seeker value = -100 }
		}
		trigger_event = POD_golconda_journey.1001
	}
}
