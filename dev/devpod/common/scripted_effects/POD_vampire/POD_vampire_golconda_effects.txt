}

POD_reapply_clan_curse_effect = {
	POD_reapply_visual_clan_curse_effect = yes
	
	if = {
		limit = { POD_suppress_clan_curse_trigger = no }
		
		POD_remove_character_modifier = { MODIFIER = uncursed_tzimisce_modifier }
		POD_remove_character_modifier = { MODIFIER = uncursed_kizoku_modifier }
		
		if = {
			limit = {
				is_anh_trigger = yes
				# disabled for the AI until we can make them smarter about choosing their domain
				is_ai = no
			}
			every_held_title = {
				POD_apply_anh_clan_curse = yes
			}
		}
		
		every_in_list = {
			variable = suppressed_clan_curse_traits
			prev = { add_trait = prev }
		}
		clear_variable_list = suppressed_clan_curse_traits
	}
}

POD_reapply_visual_clan_curse_effect = {
	if = {
		limit = {
			is_nosferatu_trigger = yes
			POD_suppress_visual_clan_curse_trigger = no
		}
		POD_remove_character_modifier = { MODIFIER = uncursed_nosferatu_modifier }
		hidden_effect = {
			remove_trait = beauty_good_1
			remove_trait = beauty_good_2
			remove_trait = beauty_good_3
			every_in_list = {
				variable = suppressed_visual_clan_curse_traits
				prev = { add_trait = prev }
			}
			clear_variable_list = suppressed_visual_clan_curse_traits
			POD_flag_injector_nosferatu_appearance = {
				EFFECT = POD_reapply_clan_curse_flag
			}
		}
	}
	else_if = {
		limit = {
			is_gargoyle_trigger = yes
			POD_suppress_visual_clan_curse_trigger = no
		}
		hidden_effect = {
			if = {
				limit = {
					any_in_list = {
						variable = suppressed_visual_clan_curse_traits
						OR = {
							this = trait:beauty_bad_1
							this = trait:beauty_bad_2
							this = trait:beauty_bad_3
						}
					}
				}
				remove_trait = beauty_good_1
				remove_trait = beauty_good_2
				remove_trait = beauty_good_3
			}
			every_in_list = {
				variable = suppressed_visual_clan_curse_traits
				prev = { add_trait = prev }
			}
			clear_variable_list = suppressed_visual_clan_curse_traits
			POD_flag_injector_gargoyle_appearance = {
				EFFECT = POD_reapply_clan_curse_flag
			}
		}
	}
}

# if ADD_MORTAL_TRAIT = yes, also add the mortal trait (if no other splat trait present) and depose
# if ADD_MORTAL_TRAIT = no, character will end up with no splat trait (will need to be added manually)
POD_make_vampire_mortal = {
	hidden_effect = { 
		set_age = POD_visual_age
	}
	POD_shapeshift_revert_all_effect = yes
	POD_remove_visual_clan_curse_when_switching_splats = yes
	remove_trait = vampire
	if = {
		limit = {
			$ADD_MORTAL_TRAIT$ = yes
			any_character_trait = {
				has_trait_flag = splat
				NOT = { this = trait:vampire }
				# no other splat traits left
				count = 0
			}
		}
		add_trait_force_tooltip = mortal
		if = {
			limit = {
				is_ruler = yes
				can_hold_title_as_mortal_trigger = no
			}
			# not using POD_depose_effect since mortals shouldn't be playable
			# TODO MORTAL: change to landed mortal handler
			depose = yes
		}
	}
	POD_remove_wrong_supernatural_traits_on_existing_character = yes
	hidden_effect = {
		POD_remove_supernatural_relation_effect = yes
		POD_side_effects_from_changing_splats = yes
		POD_remove_child_relation_effect = yes
		POD_remove_parent_relation_effect = yes
		POD_character_new_dynasty_based_on_existing_house_effect = yes
		force_character_skill_recalculation = yes
		# losing immortality seems to set your health to 2.5 (poor)
		# so we add a +2.5 health modifier
		POD_reapply_scaling_character_modifier = {
			MODIFIER = golconda_second_chance_modifier
		}
	}
}

POD_lose_incompatible_splat_relationships = {
	# divorce spouses who have an incompatible splat
	every_spouse = {
		limit = {
			POD_custom_incompatible_splats_trigger = {
				ACTOR = this
				RECIPIENT = prev
			}
		}
		divorce_effect = {
			DIVORCER = this
			DIVORCEE = prev
		}
	}
	
	# coterie handling
	# case 1: all coterie members have incompatible splats -> leave coterie
	if = {
		limit = {
			num_of_relation_coterie_member > 0
			any_relation = {
				type = coterie_member
				count = 0
				NOT = {
					POD_custom_incompatible_splats_trigger = {
						ACTOR = this
						RECIPIENT = prev
					}
				}
			}
		}
		POD_leave_coterie_effect = yes
	}
	# case 2: at least 1 coterie member is compatible -> kick out the ones who aren't
	else_if = {
		limit = {
			any_relation = {
				type = coterie_member
				count > 0
				NOT = {
					POD_custom_incompatible_splats_trigger = {
						ACTOR = this
						RECIPIENT = prev
					}
				}
			}
		}
		every_relation = {
			type = coterie_member
			limit = {
				POD_custom_incompatible_splats_trigger = {
					ACTOR = this
					RECIPIENT = prev
				}
			}
			POD_leave_coterie_effect = yes
		}
	}
}
