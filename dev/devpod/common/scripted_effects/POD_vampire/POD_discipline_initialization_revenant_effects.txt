POD_character_revenant_discipline_initialization_effect = {
	if = {
		limit = { POD_is_obertus_trigger = yes }
		add_trait = auspex
		add_trait = obfuscate
		add_trait = vicissitude
	}

	else_if = {
		limit = { POD_is_vlaszy_trigger = yes }
		add_trait = animalism
		add_trait = presence
		add_trait = potence
	}

	else_if = {
		limit = { POD_is_basarab_trigger = yes }
		add_trait = animalism
		add_trait = dominate
		add_trait = presence
	}

	else_if = {
		limit = { POD_is_grimaldi_trigger = yes }
		add_trait = celerity
		add_trait = dominate
		add_trait = fortitude
	}
	
	else_if = {
		limit = { POD_is_premsyl_trigger = yes }
		add_trait = auspex
		add_trait = potence
		add_trait = vicissitude
	}	

	else_if = {
		limit = { POD_is_bratovitch_trigger = yes }
		
		add_trait = animalism
		add_trait = vicissitude
		add_trait = fortitude
		
		if = {
			limit = { NOT = { has_trait = wrathful } }
			random = {
				chance = 75
				remove_trait = calm
				add_trait_force_tooltip = wrathful
			}
		}
		if = {
			limit = { NOT = { has_trait = giant } }
			random = {
				chance = 75
				add_trait = giant
			}
		}
		if = {
			limit = { is_cannibal_trigger = no }
			random = {
				chance = 50
				give_cannibal_secret_or_trait_effect = yes
			}
		}
	}

	else_if = {
		limit = { POD_is_szantovich_trigger = yes }
		add_trait = auspex
		add_trait = presence
		add_trait = vicissitude
	}	

	else_if = {
		limit = { POD_is_danislav_trigger = yes }
		add_trait = auspex
		add_trait = protean
		add_trait = vicissitude
	}	

	else_if = {
		limit = { POD_is_khazi_trigger = yes }
		add_trait = animalism
		add_trait = obfuscate
		add_trait = vicissitude
		if = {
			limit = { NOT = { has_trait = scaly } }
			add_trait = scaly
		}
		if = {
			limit = { NOT = { has_trait = albino } }
			add_trait = albino
		}
	}	

	else_if = {
		limit = { POD_is_krevcheski_trigger = yes }
		add_trait = auspex
		add_trait = bloodsorcery
		add_trait = dominate
		add_character_flag = has_thaumaturgy_bloodmagic_flag
	}

	else_if = {
		limit = { POD_is_oprichniki_trigger = yes }
		add_trait = animalism
		add_trait = obfuscate
		add_trait = presence
	}	

	else_if = {
		limit = { POD_is_narov_trigger = yes }
		add_trait = auspex
		add_trait = dominate
		add_trait = vicissitude
	}	
	
	else_if = {
		limit = { POD_is_rustovitchi_trigger = yes }
		add_trait = animalism
		add_trait = fortitude
		add_trait = vicissitude
	}	

	else_if = {
		limit = { POD_is_ruthvenski_trigger = yes }
		add_trait = animalism
		add_trait = bloodsorcery
		add_trait = vicissitude
		add_character_flag = has_koldunic_bloodmagic_flag
	}	
	
	else_if = {
		limit = { POD_is_enrathi_trigger = yes }
		add_trait = dominate
		add_trait = obfuscate
		add_trait = potence
	}		

	else_if = {
		limit = { POD_is_marijava_trigger = yes }
		add_trait = celerity
		add_trait = obfuscate
		add_trait = presence
	}

	else_if = {
		limit = { POD_is_rafastio_trigger = yes }
		add_trait = animalism
		add_trait = auspex
		add_trait = bloodsorcery
		add_character_flag = has_thaumaturgy_bloodmagic_flag
	}
	
	else_if = {
		limit = { POD_is_dhabi_trigger = yes }
		add_trait = daimonion
		add_trait = obfuscate
		add_trait = presence
	}	
	
	else_if = {
		limit = { POD_is_keskinen_trigger = yes }
		add_trait = bloodsorcery
		add_trait = dominate
		add_trait = obfuscate
		add_character_flag = has_thaumaturgy_bloodmagic_flag
		if = {
			limit = {
				NOT = { has_trait = lunatic_genetic }
				NOT = { has_trait = possessed_genetic }
				NOT = { has_trait = depressed_genetic }
				NOT = { has_trait = paranoid }
			}
			POD_character_vampire_madness_initialization_effect = yes
		}
	}		

	else_if = {
		limit = { POD_is_rosselini_trigger = yes }
		add_trait = dominate
		add_trait = fortitude
		add_trait = necromancy
		add_character_flag = has_rosselini_necromancy_flag
	}	

	else_if = {
		limit = { POD_is_rawan_trigger = yes }
		add_trait = auspex
		add_trait = obfuscate
		add_trait = presence
	}	

	else_if = {
		limit = { POD_is_kairouan_trigger = yes }
		add_trait = auspex
		add_trait = celerity
		add_trait = obfuscate
	}	
	
	else_if = {
		limit = { POD_is_dimitriou_trigger = yes }
		add_trait = auspex
		add_trait = potence
		add_trait = presence
	}	
	
	else_if = {
		limit = { POD_is_slatikov_trigger = yes }
		add_trait = animalism
		add_trait = bloodsorcery
		add_trait = vicissitude
		add_character_flag = has_koldunic_bloodmagic_flag
	}	

	else_if = {
		limit = { POD_is_nirriti_trigger = yes }
		add_trait = bloodsorcery
		add_trait = obfuscate
		add_trait = presence
		add_character_flag = has_sadhana_bloodmagic_flag
	}

	else_if = {
		limit = { POD_is_crassus_trigger = yes }
		add_trait = dominate
		add_trait = potence
		add_trait = presence
	}

	else_if = {
		limit = { POD_is_gravenor_trigger = yes }
		add_trait = animalism
		add_trait = obfuscate
		add_trait = potence
	}

	else_if = {
		limit = { POD_is_angustri_trigger = yes }
		add_trait = auspex
		add_trait = obfuscate
		add_trait = celerity
	}

	else_if = {
		limit = { POD_is_kurulas_trigger = yes }
		add_trait = animalism
		add_trait = fortitude
		add_trait = potence
	}

	else_if = {
		limit = { POD_is_qarnaim_trigger = yes }
		add_trait = celerity
		add_trait = potence
		add_trait = presence
		if = {
			limit = { NOT = { has_trait = wrathful } }
			random = {
				chance = 75
				remove_trait = calm
				add_trait_force_tooltip = wrathful
			}
		}
	}

	else_if = {
		limit = { POD_is_bima_trigger = yes }
		add_trait = potence
		add_trait = presence
		add_trait = serpentis
	}

	else_if = {
		limit = { POD_is_bomkazi_trigger = yes }
		add_trait = fortitude
		add_trait = necromancy
		add_trait = presence
		add_character_flag = has_impundulu_necromancy_flag
	}
	
	else_if = {
		limit = {
			exists = mother
			exists = real_father
		}
		
		POD_roll_revenant_discipline = {
			TRAIT = potence
			BASE_CHANCE = 80
			CHANCE_PER_PARENT = 40
		}
		POD_roll_revenant_discipline = {
			TRAIT = fortitude
			BASE_CHANCE = 50
			CHANCE_PER_PARENT = 40
		}
		POD_roll_revenant_discipline = {
			TRAIT = celerity
			BASE_CHANCE = 30
			CHANCE_PER_PARENT = 40
		}
		
		POD_roll_common_revenant_discipline = { TRAIT = animalism }
		POD_roll_common_revenant_discipline = { TRAIT = auspex }
		POD_roll_common_revenant_discipline = { TRAIT = dominate }
		POD_roll_common_revenant_discipline = { TRAIT = obfuscate }
		POD_roll_common_revenant_discipline = { TRAIT = presence }
		
		POD_roll_rare_revenant_discipline = { TRAIT = abombwe }
		POD_roll_rare_revenant_discipline = { TRAIT = bardo }
		POD_roll_rare_revenant_discipline = { TRAIT = bloodsorcery }
		POD_roll_rare_revenant_discipline = { TRAIT = chimerstry }
		POD_roll_rare_revenant_discipline = { TRAIT = daimonion }
		POD_roll_rare_revenant_discipline = { TRAIT = dementation }
		POD_roll_rare_revenant_discipline = { TRAIT = flight }
		POD_roll_rare_revenant_discipline = { TRAIT = kai }
		POD_roll_rare_revenant_discipline = { TRAIT = melpominee }
		POD_roll_rare_revenant_discipline = { TRAIT = mytherceria }
		POD_roll_rare_revenant_discipline = { TRAIT = necromancy }
		POD_roll_rare_revenant_discipline = { TRAIT = oblivion }
		POD_roll_rare_revenant_discipline = { TRAIT = obtenebration }
		POD_roll_rare_revenant_discipline = { TRAIT = protean }
		POD_roll_rare_revenant_discipline = { TRAIT = quietus }
		POD_roll_rare_revenant_discipline = { TRAIT = rift }
		POD_roll_rare_revenant_discipline = { TRAIT = serpentis }
		POD_roll_rare_revenant_discipline = { TRAIT = spiritus }
		POD_roll_rare_revenant_discipline = { TRAIT = temporis }
		POD_roll_rare_revenant_discipline = { TRAIT = valeren }
		POD_roll_rare_revenant_discipline = { TRAIT = vicissitude }
		POD_roll_rare_revenant_discipline = { TRAIT = visceratika }
		
		if = {
			limit = { has_trait = bloodsorcery }
			
			POD_learn_targets_known_bloodmagic_paths = {
				TARGET = real_father
				CHANCE = 100
			}
			POD_learn_targets_known_bloodmagic_paths = {
				TARGET = mother
				CHANCE = 100
			}
		}
		
		if = {
			limit = { has_trait = necromancy }

			POD_learn_targets_known_necromancy_paths = {
				TARGET = real_father
				CHANCE = 100
			}
			POD_learn_targets_known_necromancy_paths = {
				TARGET = mother
				CHANCE = 100
			}
		}
	}
	
	else = {
		add_trait = potence
		add_trait = celerity
		add_trait = fortitude
	}
}

POD_roll_common_revenant_discipline = {
	POD_roll_revenant_discipline = {
		TRAIT = $TRAIT$
		BASE_CHANCE = 20
		CHANCE_PER_PARENT = 40
	}
}

POD_roll_rare_revenant_discipline = {
	POD_roll_revenant_discipline = {
		TRAIT = $TRAIT$
		BASE_CHANCE = 0
		CHANCE_PER_PARENT = 45
	}
}

POD_roll_revenant_discipline = {
	random = {
		chance = $BASE_CHANCE$
		
		modifier = {
			add = $CHANCE_PER_PARENT$
			mother = { has_trait = $TRAIT$discipline }
		}
		modifier = {
			add = $CHANCE_PER_PARENT$
			real_father = { has_trait = $TRAIT$discipline }
		}
		modifier = {
			factor = 2
			dynasty ?= { has_dynasty_perk = aspiration_legacy_5 }
		}
		
		add_trait = $TRAIT$
	}
}
