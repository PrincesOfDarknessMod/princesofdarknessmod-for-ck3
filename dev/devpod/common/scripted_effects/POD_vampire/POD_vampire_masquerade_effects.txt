POD_character_masquerade_initialization_effect = {
	if = {
		limit = { POD_splat_has_masquerade_trigger = yes }
		add_character_modifier = masquerade0_modifier
	}
}

POD_set_masquerade_modifier_effect = {
	if = {
		limit = { masquerade_exposure_level != $AMOUNT$ }
		hidden_effect = { POD_remove_masquerade_modifiers_effect = yes }
		switch = {
			trigger  = $AMOUNT$
			0        = { add_character_modifier = masquerade0_modifier }
			1        = { add_character_modifier = masquerade1_modifier }
			2        = { add_character_modifier = masquerade2_modifier }
			3        = { add_character_modifier = masquerade3_modifier }
			4        = { add_character_modifier = masquerade4_modifier }
			fallback = { 
				add_character_modifier = masquerade5_modifier

				# Reduce AI masquerade after a few years if it is at max
				trigger_event = {
					id = POD_maintenance.26
					years = { 1 5 }
				}
			}
		}
	}
}

POD_change_masquerade_exposure_effect = {
	save_temporary_scope_value_as = {
		name = new_exposure
		value = {
			value = masquerade_exposure_level
			add = $DELTA$
			min = minimum_masquerade_exposure
			max = 5
		}
	}
	if = {
		limit = { exists = scope:new_exposure } # error suppression
		POD_set_masquerade_modifier_effect = { AMOUNT = scope:new_exposure }
	}
}

POD_masquerade_exposure_increase_effect = {
	if = {
		limit = {
			POD_splat_has_masquerade_trigger = yes
			NOR = {
				has_trait = exposed
				POD_is_forever_infamous_trigger = yes
				has_character_modifier = masquerade5_modifier
			}
			
		}
		custom_tooltip = masquerade_exposure_increase_effect.tt

		if = {
			limit = {
				NOT = { has_character_flag = is_in_POD_vampire_masquerade_breach_100 }
			}
			hidden_effect = {
				random = {
					chance = 10
					modifier = { factor = masquerade_exposure_level }
					trigger_event = {
						id = POD_vampire_masquerade_breach.100
						days = 7
					}
				}
			}
		}

		POD_change_masquerade_exposure_effect = { DELTA = 1 }
	}
}

POD_masquerade_exposure_decrease_effect = {
	if = {
		limit = {

		if = {

POD_piety_loss_from_massembrace = {

	if = { 
		limit = {
			faith = {
				OR = {
					has_doctrine_parameter = vampirism_illegal
					has_doctrine_parameter = hurt_mortal_lose_piety
					has_doctrine_parameter = vampirethemasquerade_consent
					has_doctrine_parameter = vampirethemasquerade_strict
				}
			}
		}
		add_piety = massive_piety_loss
	}

	else_if = { 
		limit = {
			faith = {
				OR = {
					has_doctrine_parameter = vampirism_shunned
					has_doctrine_parameter = vampirethemasquerade_family
				}
			}
		}
		add_piety = major_piety_loss
	}
	
	else_if = { 
		limit = {
			faith = {
				OR = {
					has_doctrine_parameter = vampirethemasquerade_lax
					has_doctrine_parameter = vampirethemasquerade_cult
				}
			}
		}
		add_piety = medium_piety_loss
	}
}

POD_piety_loss_from_massghouling = {
	if = { 
		limit = {

		}
		add_piety = medium_piety_loss
	}
	
	else_if = { 
		limit = {

				faith = { has_doctrine_parameter = vampireghouling_shunned }
				faith = { has_doctrine_parameter = vampirethemasquerade_lax }
			}
		}
		add_piety = minor_piety_loss
	}
}

POD_destroy_masquerade_effect = {
	custom_tooltip = forever_indomitable_decision_effect
	hidden_effect = {
		set_global_variable = destroyed_masquerade
		# Remove Masquerade modifiers from everyone
		every_living_character = {
			POD_remove_masquerade_modifiers_effect = yes
		}
		# Remove Masquerade doctrines
		every_religion_global = {
			every_faith = {
				limit = {
					OR = {
						has_doctrine = doctrine_vampirethemasquerade_strict
						has_doctrine = doctrine_vampirethemasquerade_lax
						has_doctrine = doctrine_vampirethemasquerade_cult
						has_doctrine = doctrine_vampirethemasquerade_family
						has_doctrine = doctrine_vampirethemasquerade_consent
					}
				}
				switch = {
					trigger = has_doctrine
					doctrine_vampirethemasquerade_strict  = { remove_doctrine = doctrine_vampirethemasquerade_strict  }
					doctrine_vampirethemasquerade_lax     = { remove_doctrine = doctrine_vampirethemasquerade_lax     }
					doctrine_vampirethemasquerade_cult    = { remove_doctrine = doctrine_vampirethemasquerade_cult    }
					doctrine_vampirethemasquerade_family  = { remove_doctrine = doctrine_vampirethemasquerade_family  }
					doctrine_vampirethemasquerade_consent = { remove_doctrine = doctrine_vampirethemasquerade_consent }
					fallback = { }
				}
				add_doctrine = doctrine_vampirethemasquerade_none
			}
		}
	}
}
