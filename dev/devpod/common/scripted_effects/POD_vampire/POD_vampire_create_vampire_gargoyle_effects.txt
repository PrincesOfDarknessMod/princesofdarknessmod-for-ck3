######################################################
# Hideout Upgrades — Chantry Annex Gargoyle Bonus
# Called during gargoyle creation to apply stat and opinion bonuses
# Scope: scope:created_vampirechilde (the new gargoyle)
# Expects: scope:gargoyle_experiment_thaumaturgist (the creator)
######################################################

POD_hideout_gargoyle_chantry_bonus_effect = {
	# Only applies if the creator has a Chantry Annex
	scope:gargoyle_experiment_thaumaturgist = {
		if = {
			limit = { domicile ?= { has_domicile_building_or_higher = vampire_tremere_chantry_annex_01 } }

			# Determine tier and apply stat modifier to the gargoyle
			scope:created_vampirechilde = {
				if = {
					limit = { scope:gargoyle_experiment_thaumaturgist = { domicile ?= { has_domicile_building_or_higher = vampire_tremere_chantry_annex_06 } } }
					add_character_modifier = POD_hideout_chantry_gargoyle_t6
				}
				else_if = {
					limit = { scope:gargoyle_experiment_thaumaturgist = { domicile ?= { has_domicile_building_or_higher = vampire_tremere_chantry_annex_05 } } }
					add_character_modifier = POD_hideout_chantry_gargoyle_t5
				}
				else_if = {
					limit = { scope:gargoyle_experiment_thaumaturgist = { domicile ?= { has_domicile_building_or_higher = vampire_tremere_chantry_annex_04 } } }
					add_character_modifier = POD_hideout_chantry_gargoyle_t4
				}
				else_if = {
					limit = { scope:gargoyle_experiment_thaumaturgist = { domicile ?= { has_domicile_building_or_higher = vampire_tremere_chantry_annex_03 } } }
					add_character_modifier = POD_hideout_chantry_gargoyle_t3
				}
				else_if = {
					limit = { scope:gargoyle_experiment_thaumaturgist = { domicile ?= { has_domicile_building_or_higher = vampire_tremere_chantry_annex_02 } } }
					add_character_modifier = POD_hideout_chantry_gargoyle_t2
				}
				else = {
					add_character_modifier = POD_hideout_chantry_gargoyle_t1
				}
			}

			# Apply opinion bonus — +5 per building tier
			scope:created_vampirechilde = {
				if = {
					limit = { scope:gargoyle_experiment_thaumaturgist = { domicile ?= { has_domicile_building_or_higher = vampire_tremere_chantry_annex_06 } } }
					add_opinion = { target = scope:gargoyle_experiment_thaumaturgist modifier = POD_hideout_chantry_gargoyle_creator_opinion opinion = 30 }
				}
				else_if = {
					limit = { scope:gargoyle_experiment_thaumaturgist = { domicile ?= { has_domicile_building_or_higher = vampire_tremere_chantry_annex_05 } } }
					add_opinion = { target = scope:gargoyle_experiment_thaumaturgist modifier = POD_hideout_chantry_gargoyle_creator_opinion opinion = 25 }
				}
