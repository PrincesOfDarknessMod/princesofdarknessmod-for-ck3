POD_rage_increase_effect = {
	if = {
		limit = {
			POD_splat_has_rage_trigger = yes
			NOT = { has_character_modifier = POD_rage_5_modifier }	
		}
		custom_tooltip = POD_rage_increase_effect.tt
		switch = {
			trigger = has_character_modifier
			POD_rage_0_modifier = {
				hidden_effect = { remove_character_modifier = POD_rage_0_modifier }
				add_character_modifier = POD_rage_1_modifier
			}
			POD_rage_1_modifier = {
				hidden_effect = { remove_character_modifier = POD_rage_1_modifier }
				add_character_modifier = POD_rage_2_modifier
			}
			POD_rage_2_modifier = {
				hidden_effect = { remove_character_modifier = POD_rage_2_modifier }
				add_character_modifier = POD_rage_3_modifier
			}
			POD_rage_3_modifier = {
				hidden_effect = { remove_character_modifier = POD_rage_3_modifier }
				add_character_modifier = POD_rage_4_modifier
			}
			POD_rage_4_modifier = {
				hidden_effect = { remove_character_modifier = POD_rage_4_modifier }
				add_character_modifier = POD_rage_5_modifier
			}
			fallback = { POD_assign_initial_rage_effect = yes }
		}
	}
}

POD_rage_decrease_effect = {
	if = {
		limit = {
			POD_splat_has_rage_trigger = yes
			NOT = { has_character_modifier = POD_rage_0_modifier }	
		}
		custom_tooltip = POD_rage_decrease_effect.tt
		switch = {
			trigger = has_character_modifier
			POD_rage_1_modifier = {
				hidden_effect = { remove_character_modifier = POD_rage_1_modifier }
				add_character_modifier = POD_rage_0_modifier
				
				POD_revert_to_breed_form_effect = yes
				if = {
					limit = {
						NOT = { stress_level >= 4 }
					}
					random = {
						chance = 5
						POD_rage_lose_the_wolf_effect = yes
						POD_remove_wrong_supernatural_traits_effect = yes
						POD_side_effects_from_changing_splats = yes
					}
				}
			}
			POD_rage_2_modifier = {
				hidden_effect = { remove_character_modifier = POD_rage_2_modifier }
				add_character_modifier = POD_rage_1_modifier
			}
			POD_rage_3_modifier = {
				hidden_effect = { remove_character_modifier = POD_rage_3_modifier }
				add_character_modifier = POD_rage_2_modifier
			}
			POD_rage_4_modifier = {
				hidden_effect = { remove_character_modifier = POD_rage_4_modifier }
				add_character_modifier = POD_rage_3_modifier
			}
			POD_rage_5_modifier = {
				hidden_effect = { remove_character_modifier = POD_rage_5_modifier }
				add_character_modifier = POD_rage_4_modifier
			}
			fallback = { POD_assign_initial_rage_effect = yes }
		}
	}
}


POD_rage_growing_maintenance_effect = {
	if = {

			POD_splat_has_rage_trigger = yes
			NOT = {
				has_character_modifier = POD_rage_5_modifier

}

POD_assign_initial_rage_effect = {
	if = {
		limit = {
			POD_has_rage_modifier_trigger = no
			POD_splat_has_rage_trigger = yes
		}

		if = {
			limit = {
				has_trait = spirit
			}
			add_character_modifier = POD_rage_5_modifier
		}

		else_if = {
			limit = {
				POD_has_auspice_trigger = no
			}
			add_character_modifier = {
				modifier = POD_rage_1_modifier			
			}
		}
		
		else = { 
			switch = {
				trigger = has_trait
				ragabash = { add_character_modifier = POD_rage_1_modifier }
				theurge = { add_character_modifier = POD_rage_2_modifier }
				philodox = { add_character_modifier = POD_rage_3_modifier }
				galliard = { add_character_modifier = POD_rage_4_modifier }
				ahroun = { add_character_modifier = POD_rage_5_modifier }
			}
		}
	}
}
