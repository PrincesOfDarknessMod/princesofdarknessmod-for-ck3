
POD_add_local_umbra_buff_side_effect = { # scope county
	if = {
		limit = { is_once_time_spirit_wilds_buff_trigger = { BUFF = $BUFF$ } }
		$VICTIM$ = {
			remove_list_variable = {
				name = spirit_wilds_buff_list
				target = prev.$BUFF$
			}
		}
	}
	else_if = {
		limit = { is_once_time_shadowlands_buff_trigger = { BUFF = $BUFF$ } }
		$VICTIM$ = {
			remove_list_variable = {
				name = shadowlands_buff_list
				target = prev.$BUFF$
			}
		}
	}
	else_if = {
		limit = { is_umbra_exploration_buff_trigger = { BUFF = $BUFF$ } }
		$VICTIM$ = {
			POD_increase_umbra_buff_count_effect = {
				BUFF = prev.$BUFF$
			}
		}
	}
}

POD_remove_local_umbra_buff_side_effect = { # scope county
	if = {
		limit = { is_once_time_spirit_wilds_buff_trigger = { BUFF = $BUFF$ } }
		$VICTIM$ = {
			add_to_variable_list = {
				name = spirit_wilds_buff_list
				target = prev.$BUFF$
			}
		}
	}
	else_if = {
		limit = { is_once_time_shadowlands_buff_trigger = { BUFF = $BUFF$ } }
		$VICTIM$ = {
			add_to_variable_list = {
				name = shadowlands_buff_list
				target = prev.$BUFF$
			}
		}
	}
	else_if = {
		limit = { is_umbra_exploration_buff_trigger = { BUFF = $BUFF$ } }
		$VICTIM$ = {
			POD_decrease_umbra_buff_count_effect = {
				BUFF = prev.$BUFF$
			}
		}
	}
}

POD_increase_umbra_buff_count_effect = {
	if = {
		limit = { is_target_in_variable_list = { name = umbra_buff_count_5_list target = $BUFF$ } }
		# DO NOTHING
	}
	else_if = {
		limit = { is_target_in_variable_list = { name = umbra_buff_count_4_list target = $BUFF$ } }
		remove_list_variable = {
			name = umbra_buff_count_4_list
			target = $BUFF$
		}
		add_to_variable_list = {
			name = umbra_buff_count_5_list
			target = $BUFF$
		}
	}
	else_if = {
		limit = { is_target_in_variable_list = { name = umbra_buff_count_3_list target = $BUFF$ } }
		remove_list_variable = {
			name = umbra_buff_count_3_list
			target = $BUFF$
		}
		add_to_variable_list = {
			name = umbra_buff_count_4_list
			target = $BUFF$
		}
	}
	else_if = {
		limit = { is_target_in_variable_list = { name = umbra_buff_count_2_list target = $BUFF$ } }
		remove_list_variable = {
			name = umbra_buff_count_2_list
			target = $BUFF$
		}
		add_to_variable_list = {
			name = umbra_buff_count_3_list
			target = $BUFF$
		}
	}
	else_if = {
		limit = { is_target_in_variable_list = { name = umbra_buff_count_1_list target = $BUFF$ } }
		remove_list_variable = {
			name = umbra_buff_count_1_list
			target = $BUFF$
		}
		add_to_variable_list = {
			name = umbra_buff_count_2_list
			target = $BUFF$
		}
	}
	else = {
		add_to_variable_list = {
			name = umbra_buff_count_1_list
			target = $BUFF$
		}
	}
}

POD_umbra_change_spirit_wilds_controller_effect = {
	if = {
		limit = { has_variable = local_spirit_wilds_buff }
		POD_remove_local_umbra_buff_effect = {
			BUFF = var:local_spirit_wilds_buff
			VICTIM = var:spirit_wilds_controller
		}
		if = {
			limit = { var:spirit_wilds_controller = { is_ai = no } }
			POD_remove_local_umbra_buff_side_effect = {
				BUFF = var:local_spirit_wilds_buff
				VICTIM = var:spirit_wilds_controller
			}
		}
		remove_variable = local_spirit_wilds_buff #Option1: clean the buff
		# POD_add_local_umbra_buff_effect = { #Option2: let player get free buff
			# BUFF = var:local_spirit_wilds_buff
			# VICTIM = $CHRACTER$
		# }
	}
	set_variable = {
		name = spirit_wilds_controller
		value = $CHRACTER$
	}
}
