blood_hunt_call_interaction_effect = {
	if = {
		limit = {
			exists = scope:target
			scope:recipient = {
				NOR = {
					is_at_war_with = scope:target.primary_defender
					is_at_war_with = scope:target.primary_attacker
				}
			}
		}
		scope:target = {
			hidden_effect = {
				set_called_to = scope:recipient
			}
			if = {
				limit = {
					is_attacker = scope:actor
				}
				add_attacker = scope:recipient
			}
			else = {
				add_defender = scope:recipient
			}
		}
	}
}

become_red_alastor_effect = {
	if = {
		limit = { NOT = { has_character_modifier = red_alastor_modifier } }
		if = {
			limit = { NOT = { has_variable = killed_anathema } }
			set_variable = {
				name = killed_anathema
				value = 1
			}
			custom_tooltip = red_alastor_tooltip
		}
		else_if = {
			limit = { var:killed_anathema = 4 }
			remove_variable = killed_anathema
			add_character_modifier = red_alastor_modifier
		}
		else = {
			change_variable = {
				name = killed_anathema
				add = 1
			}
			custom_tooltip = red_alastor_tooltip
		}
	}
}

set_justicar_effect = {
	if = {
		limit = { exists = scope:old_justicar }
		if = {
			limit = { $CHARACTER$ = scope:old_justicar }
			scope:old_justicar = {
				send_interface_toast = {
					type = event_generic_good
					title = hold_justicar_notification
					left_icon = scope:old_justicar
					right_icon = $CHARACTER$
					custom_tooltip = hold_justicar_notification_tooltip
				}
			}
		}
		else = {
			scope:old_justicar = {
				send_interface_toast = {
					type = event_generic_bad
					title = remove_justicar_notification
					left_icon = scope:old_justicar
					right_icon = $CHARACTER$
					custom_tooltip = remove_justicar_notification_tooltip
				}
			}
			$CHARACTER$ = {
				save_temporary_scope_as = new_justicar
				send_interface_toast = {
					type = event_generic_good
					title = set_justicar_notification
					left_icon = scope:old_justicar
					right_icon = $CHARACTER$
					custom_tooltip = set_justicar_notification_tooltip
				}
			}
			remove_list_variable = {
				name = justicars
				target = scope:old_justicar 
			}
			add_to_variable_list = { name = justicars target = $CHARACTER$ }
		}
	}
	else = {
		$CHARACTER$ = {
			save_temporary_scope_as = new_justicar
			send_interface_toast = {
				type = event_generic_good
				title = set_justicar_notification
				right_icon = $CHARACTER$
				custom_tooltip = set_justicar_notification_tooltip
			}
		}
		add_to_variable_list = { name = justicars target = $CHARACTER$ }
	}
}

		}
		random_in_list = {
			variable = anathema
			limit = { 
				NOT = { this = scope:red_list_candidate_1 }
				NOT = { this = scope:red_list_candidate_2 }
				NOT = { this = scope:red_list_candidate_3 }
				NOT = { this = scope:red_list_candidate_4 }
			}
			weight = {
				base = 0.01
				candidate_chance_modifier = yes
			}
			save_scope_as = red_list_candidate_5
		}
	}
}

set_anathema_effect = {
	if = {
		limit = {
			faith:camahumanity = {
				NOT = { is_target_in_variable_list = { name = red_list target = prev } }
			}
		}
		faith:camahumanity = { add_to_variable_list = { name = anathema target = prev } }
		if = {
			limit = { POD_is_anathema_trigger = no }
			add_trait = anathema
		}
	}
}

set_red_list_effect = { # Initiation only
	faith:camahumanity = { add_to_variable_list = { name = red_list target = prev } }
	set_variable = { name = trophy_clan value = trait:$TROPHY_CLAN$ }
	if = {
		limit = { POD_is_anathema_trigger = no }
		add_trait = anathema
	}
}

remove_anathema_effect = {
	faith:camahumanity = { remove_list_variable = { name = anathema target = prev } }
	remove_trait = anathema
}
