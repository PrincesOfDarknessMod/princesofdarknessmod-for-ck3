POD_change_character_debate_force_effect = {
	custom_description = {
		text = POD_change_character_debate_force
		value = $COUNT$
		POD_set_or_change_variable = {
			VAR = character_debate_force
			VAL = $COUNT$
		}
	}
}

POD_change_opponents_debate_force_effect = {
	custom_description = {
		text = POD_change_opponents_debate_force
		value = $COUNT$
		POD_set_or_change_variable = {
			VAR = opponents_debate_force
			VAL = $COUNT$
		}
	}
}

POD_reset_debate_force_effect = {
	set_variable = {
		name = character_debate_force
		value = 0
	}
	set_variable = {
		name = opponents_debate_force
		value = 0
	}
}

# Effect to automatically start the entire debate system
# Args:
#	- CHARACTER: takes the player as a scope
#	- OPPONENTS: takes a variable list containing various character scopes
#	- STARTER_EVENT: takes the namespace of the event that will be triggered; assumes that this event will trigger the others as appropriate
POD_start_debate_effect = {
	$CHARACTER$ = { save_scope_as = debate_leader }
	POD_reset_debate_force_effect = yes
	# Calculate initial strength of player
	scope:debate_leader = {
		# Using this variable because I don't know a better way to do multiply
		# Account for your and your chancellor's diplomacy
		set_variable = {
			name = character_initial_strength
			value = {
				value = diplomacy
				multiply = 25
			}
		}
		if = {
			limit = {
				has_council = yes
			}
			change_variable = {
				name = character_initial_strength
				add = {
					value = scope:debate_leader.cp:councillor_chancellor.diplomacy
					multiply = 10
				}
			}
		}
		# Account for ability to prove knowledge and communicate more easily
		change_variable = {
			name = character_initial_strength
			add = {
				value = scope:debate_leader.num_of_known_languages
				multiply = 35
			}
		}
		# Account for scholarly nature and so familiarity with debates
		if = {
			limit = {
				POD_is_scholarly = yes
			}
			change_variable = {
				name = character_initial_strength
				add = 350
			}
		}
		# Account for fame that makes your words more convincing
		change_variable = {
			name = character_initial_strength
			add = {
				value = scope:debate_leader.piety_level
				multiply = 30
			}
		}
		change_variable = {
			name = character_initial_strength
			add = {
				value = scope:debate_leader.prestige_level
				multiply = 25
			}
		}
		change_variable = {
			name = character_initial_strength
			add = {
				value = scope:debate_leader.dynasty.dynasty_prestige_level
				multiply = 30
			}
		}
		# Account for personality
		change_variable = {
			name = character_initial_strength
			add = {
				value = scope:debate_leader.ai_sociability  # Sociable and charismatic
				multiply = 10
			}
		}
		change_variable = {
			name = character_initial_strength
			add = {
				value = scope:debate_leader.ai_rationality  # Present a logical argument
				multiply = 5
			}
		}
		# Account for virtues
		change_variable = {
			name = opponent_initial_strength
			add = {
				value = "scope:debate_leader.num_virtuous_traits(scope:debate_leader.faith)"
				multiply = 35
			}
		}
	}
	# Set a minimum value
	if = {
		limit = { var:character_initial_strength < 100 }
		set_variable = {
			name = character_initial_strength
			value = 100
		}
	}
	POD_change_character_debate_force_effect = {
		COUNT = var:character_initial_strength
	}
	remove_variable = character_initial_strength

	# Calculate initial strength of opponents
	every_in_list = {
		list = $OPPONENTS$
		save_scope_as = debater
		
		scope:debate_leader = {
			# Account for their diplomacy
			set_variable = {
				name = opponent_initial_strength
				value = {
					value = scope:debater.diplomacy
					multiply = 25
				}
			}
			# Account for faith hostility
			# Creates curve (33x^2 - 16x) so Astray = 17, Hostile = 100, Evil = 249
			change_variable = {
				name = opponent_initial_strength
				add = {
					value = "scope:debate_leader.faith.faith_hostility_level(scope:debater.faith)"
					multiply = {
						value = "scope:debate_leader.faith.faith_hostility_level(scope:debater.faith)"
						multiply = 33
					}
				}
			}
			change_variable = {
				name = opponent_initial_strength
				subtract = {
					value = "scope:debate_leader.faith.faith_hostility_level(scope:debater.faith)"
					multiply = 16
				}
			}
			# Account for shared court lang making communication easier
			if = {
				limit = {
					has_same_court_language = scope:debater
				}
				change_variable = {
					name = opponent_initial_strength
					subtract = 50
				}
			}
			# Account for opinion stuff
			# Curve (x^2/16) so -100 increases difficulty (+625) substantially while +10 provides tiny reduction (-6)
			if = {
				limit = { "scope:debater.opinion(scope:debate_leader)" > 0 }
				change_variable = {
					name = opponent_initial_strength
					subtract = {
						value = "scope:debater.opinion(scope:debate_leader)"
						multiply = {
							value = "scope:debater.opinion(scope:debate_leader)"
							divide = 16
						}
						floor = yes
					}
				}
			}
			else = {
				change_variable = {
					name = opponent_initial_strength
					add = {
						value = "scope:debater.opinion(scope:debate_leader)"
						multiply = {
							value = "scope:debater.opinion(scope:debate_leader)"
							divide = 16
						}
						floor = yes
					}
				}
			}

			POD_change_opponents_debate_force_effect = {
				COUNT = var:opponent_initial_strength
			}

			remove_variable = opponent_initial_strength
		}
	}
	# Harder to convince a group than one person
	# 5x^2 + 20 for nice curve of 25 (at 1) to 520 (at 10)
	set_variable = {
		name = num_opponents_strength
		value = {
			value = 20
			add = {
				value = 5
				multiply = {
					value = list_size:$OPPONENTS$
					multiply = list_size:$OPPONENTS$
				}
			}
		}
	}
	# Set a minimum value
	if = {
		limit = { var:opponents_debate_force < 100 }
		set_variable = {
			name = opponents_debate_force
			value = 100
		}
	}
	POD_change_opponents_debate_force_effect = {
		COUNT = var:num_opponents_strength
	}
	remove_variable = num_opponents_strength

	# Trigger first event
	trigger_event = {
		id = $STARTER_EVENT$
		days = { 10 15 }
	}
}

# If no subsplat, get splat
POD_retrieve_subsplat_trait = {
	hidden_effect = {
		# Try to get a proper subsplat
		random_character_trait = {
			limit = { 
				OR = {
					has_trait_flag = vamp_clan 
					AND = {
						has_trait_flag = fera_type
						has_trait_flag = splat
					}
					AND = {
						has_trait_flag = fae_trait
						has_trait_flag = subsplat
					}
					AND = {
						has_trait_flag = wraith_trait
						has_trait_flag = subsplat
					}
					AND = {
						has_trait_flag = mummy_trait
						has_trait_flag = subsplat
					}
					has_trait_flag = subsplat 
				}
			}

			save_scope_as = subsplat_trait
		}
		# If no subsplat trait, get splat
		if = {
			limit = { NOT = { exists = scope:subsplat_trait } }
			random_character_trait = {
				limit = { 
					has_trait_flag = splat
				}

				save_scope_as = subsplat_trait
			}
		}
	}
}

# Only use for the POD_debate_select_opponent_from_* effects
POD_debate_select_opponent = {
	# Save to list of opponents
	add_to_list = opponents_list
	# Save as scope so I can use for events
	switch = {
		trigger = exists
		scope:debate_opponent_4 = {
			save_scope_as = debate_opponent_5
		}
		scope:debate_opponent_3 = {
			save_scope_as = debate_opponent_4
		}
		scope:debate_opponent_2 = {
			save_scope_as = debate_opponent_3
		}
		scope:debate_opponent_1 = {
			save_scope_as = debate_opponent_2
		}
		fallback = {
			save_scope_as = debate_opponent_1
		}
	}
}

# POD_debate_select_opponent_from_* effects take an argument CHARACTER which should contain `scope:character`
# Uses random_ruler rather than ordered_ruler to introduce more variation
POD_debate_select_opponent_from_dynasty = {
	$CHARACTER$.dynasty = {
		random_dynasty_member = {
			limit = {
				is_adult = yes
				NOT = { has_trait = torpor }
				is_alive = yes
				NOR = {
					is_target_in_variable_list = {
						name = opponents_list
						target = this
					}
					this = $CHARACTER$
					trigger_if = {
						limit = { exists = scope:debate_opponent_1 }
						this = scope:debate_opponent_1
					}
					trigger_if = {
						limit = { exists = scope:debate_opponent_2 }
						this = scope:debate_opponent_2
					}
					trigger_if = {
						limit = { exists = scope:debate_opponent_3 }
						this = scope:debate_opponent_3
					}
					trigger_if = {
						limit = { exists = scope:debate_opponent_4 }
						this = scope:debate_opponent_4
					}
					trigger_if = {
						limit = { exists = scope:debate_opponent_5 }
						this = scope:debate_opponent_5
					}
				}
			}
			weight = {
				base = 0
				debate_opponent_selection_modifier = yes
				modifier = {
					add = 250
					house = $CHARACTER$.house
				}
				# Apply additional bonuses for these
				modifier = {
					add = 100
					is_house_head = yes
				}
				modifier = {
					add = 500
					is_dynast = yes
				}
			}
			# Save to list of opponents
			POD_debate_select_opponent = yes		
		}
	}
}

POD_debate_select_opponent_from_faith = {

				is_adult = yes
				NOT = { has_trait = torpor }
            	is_alive = yes

					is_target_in_variable_list = {

						target = this
					}
					this = $CHARACTER$
					trigger_if = {
						limit = { exists = scope:debate_opponent_1 }
						this = scope:debate_opponent_1
					}
					trigger_if = {
						limit = { exists = scope:debate_opponent_2 }
						this = scope:debate_opponent_2
					}
					trigger_if = {
						limit = { exists = scope:debate_opponent_3 }

					}
					trigger_if = {
						limit = { exists = scope:debate_opponent_4 }

					}
