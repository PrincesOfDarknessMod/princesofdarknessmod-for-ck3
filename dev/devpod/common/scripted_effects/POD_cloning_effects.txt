
# Effects in this file copy various things from scope:cloned_character into current scope

#POD_remove_vanilla_traits_effect - removes vanilla not physical traits
#POD_remove_physical_traits_effect - removes vanilla congenital traits and various traits affecting physical condition in one way or another
#POD_remove_vanilla_secrets_effect - removes vanilla secrets
#POD_remove_supernatural_secrets_effect - removes POD secrets
#POD_copy_vanilla_traits_effect - copies vanilla not physical traits
#POD_copy_physical_traits_effect - copies vanilla congenital traits and various traits affecting physical condition in one way or another
#POD_copy_vanilla_secrets_effect - copies vanilla secrets
#POD_copy_supernatural_secrets_effect - copies POD secrets
#POD_copy_physical_appearance_effect - copies physical appearance, right now immortal age is set to 30 by default
#POD_copy_base_skills_effect - copies base skills, should be used before traits, modifiers and perks are copied
#POD_copy_supernatural_lifestyle_perks_effect - copies POD lifestyle perks
#POD_copy_vanilla_lifestyle_perks_effect - copies vanilla lifestyle perks
#POD_copy_vampire_modifiers_effect - copies POD vampire modifiers, such as blood potency, hunger, and masquerade exposure

POD_flag_as_mindless_clone = {
	add_character_flag = mindless_clone
}

POD_create_shadow_twin_effect = {
	save_scope_as = cloned_character
	 
	create_character = {
		save_scope_as = shadow_twin
		age = scope:cloned_character.age
		faith = scope:cloned_character.faith
		culture = scope:cloned_character.culture
		gender = scope:cloned_character
		random_traits = no
		dynasty = none
		employer = scope:cloned_character
	}
	if = {
		limit = {
			exists = scope:cloned_character
			exists = scope:shadow_twin
		}
		scope:shadow_twin = {
			POD_copy_base_skills_effect = {
				CLONED_CHAR = scope:cloned_character
				CLONE = scope:shadow_twin
			}
			copy_traits = scope:cloned_character
			POD_clear_invalid_clone_traits = yes
			POD_copy_vampire_modifiers_effect = yes
			POD_copy_supernatural_secrets_effect = yes
			POD_copy_trait_xp_for_clone_effect = yes
			POD_copy_sexuality_effect = { ORIGIN = scope:cloned_character }
			change_first_name = { template_character = scope:cloned_character }
			POD_copy_physical_appearance_effect = yes
			POD_copy_vanilla_lifestyle_perks_effect = yes
			POD_copy_supernatural_lifestyle_perks_effect = yes
			POD_flag_as_mindless_clone = yes
			add_character_modifier = shadow_twin_modifier
			give_nickname = nick_shadow_twin
			set_relation_shadow_twin_master = scope:cloned_character
			add_character_flag = blocked_from_leaving
		}
		scope:cloned_character = {
			add_hook = {
				target = scope:shadow_twin
				type = shadow_twin_bind_hook
			}
		}
	}
}

				CLONED_CHAR = scope:cloned_character
				CLONE = scope:next_hotei
			}
			copy_traits = scope:cloned_character
			POD_clear_invalid_clone_traits = yes
			POD_copy_supernatural_secrets_effect = yes
			POD_copy_trait_xp_for_clone_effect = yes
			POD_copy_sexuality_effect = { ORIGIN = scope:cloned_character }
			change_first_name = { template_character = scope:cloned_character }
			POD_copy_physical_appearance_effect = yes
			POD_copy_vanilla_lifestyle_perks_effect = yes
			POD_copy_supernatural_lifestyle_perks_effect = yes
			POD_set_parent_effect = { PARENT = scope:cloned_character }
			add_character_flag = blocked_from_leaving
			
			POD_initialize_energy_modifier = yes
			
			if = {
				limit = { exists = scope:idol_mummy }
				set_relation_worshipped_reborn = scope:idol_mummy

				list = demonic_masters

			}
		}
		if = {
			limit = { is_ruler = yes }
			set_designated_heir = scope:next_hotei
		}
	}

POD_clear_invalid_clone_traits = {
	# Remove wound traits.
	# There is the alternative of basing the clone's health track on the original, and copying wounds...
	# ...but eh. maybe when clone relations are refactored
	POD_trait_injector_wound = { EFFECT = POD_remove_trait }
	remove_variable = POD_health_damage_num
	remove_variable = POD_health_aggravated_damage_num
}

# Remove vanilla traits that affect personality in one way or another
POD_remove_vanilla_traits_effect = {
	every_character_trait = {
		limit = {
			OR = {
				has_trait_category = personality
				has_trait_category = education
				has_trait_category = childhood
				has_trait_category = commander
				has_trait_category = winter_commander
				has_trait_category = lifestyle
				has_trait_category = court_type
				has_trait_category = fame
			}
		}
		prev = { remove_trait = prev }
	}
}

# Remove vanilla congenital traits and various traits affecting physical condition in one way or another
POD_remove_physical_traits_effect = {
	every_character_trait = {
		limit = { 
			OR = { 
				has_trait_category = health
				has_trait_flag = reweaving_physical_trait
			}
		}
		save_temporary_scope_as = this_trait
		prev = { remove_trait = prev }
	}
}

POD_remove_personality_traits_effect = {
	save_temporary_scope_as = this_character
	every_character_trait = {
		limit = { has_trait_category = personality }
		save_temporary_scope_as = this_trait
		scope:this_character = { remove_trait = scope:this_trait }
	}
}

# Remove vanilla secrets
POD_remove_vanilla_secrets_effect = {
	POD_secret_injector_vanilla = { EFFECT = POD_remove_secret }
}

# Removes POD secrets
POD_remove_supernatural_secrets_effect = {
	if = {
		limit = { any_secret = { secret_type = secret_demonology } }
		remove_variable = secret_demonologist_xp
		remove_variable = secret_thrall_xp
		remove_variable = secret_diabolist_xp
	}
	POD_secret_injector_supernatural = { EFFECT = POD_remove_secret }
}

#Copy physical traits
POD_copy_physical_traits_effect = {
	save_scope_as = trait_receiver
	scope:cloned_character = {
		every_character_trait = {
			limit = {
				OR = {
					has_trait_category = health
					has_trait_flag = reweaving_physical_trait
				}
				NOT = { this = trait:pregnant } # no.
			}
			scope:trait_receiver = {
				add_trait = prev
			}
		}
	}
}

POD_copy_secret_to_clone = {
	if = {
		limit = {
			scope:cloned_character = {
				any_secret = { secret_type = $SECRET$ }
			}
		}
		add_secret = {
			type = $SECRET$
		}
	}
}

#Copy POD secrets
POD_copy_supernatural_secrets_effect = {
	if = {
		limit = {
			scope:cloned_character = {
				any_secret = { secret_type = secret_demonology }
			}
		}
		if = {
			limit = { scope:cloned_character = { has_variable = secret_demonologist_xp } }
			set_variable = {
				name  = secret_demonologist_xp
				value = scope:cloned_character.var:secret_demonologist_xp
			}
		}
		else = {
			set_variable = {
				name  = secret_demonologist_xp
				value = 0
			}
		}
		if = {
			limit = { scope:cloned_character = { has_variable = secret_thrall_xp } }
			set_variable = {
				name  = secret_thrall_xp
				value = scope:cloned_character.var:secret_thrall_xp
			}
		}
		else = {
			set_variable = {
				name  = secret_thrall_xp
				value = 0
			}
		}
		if = {
			limit = { scope:cloned_character = { has_variable = secret_diabolist_xp } }
			set_variable = {
				name  = secret_diabolist_xp
				value = scope:cloned_character.var:secret_diabolist_xp
			}
		}
		else = {
			set_variable = {
				name  = secret_diabolist_xp
				value = 0
			}
		}
	}
	
	POD_secret_injector_supernatural = { EFFECT = POD_copy_secret_to_clone }
}

POD_get_base_skill_effect = {
	hidden_effect = {
		if = { limit = { $SKILL$ >= POD_$SKILL$_cap } add_character_modifier = save_real_$SKILL$_negate_modifier }
		while = {
			limit = { $SKILL$ >= POD_$SKILL$_cap }
			add_character_modifier = save_real_$SKILL$_modifier
			force_character_skill_recalculation = yes
		}
		save_scope_value_as = { name = saved_skill value = $SKILL$ }
		add_$SKILL$_skill = -10000
		force_character_skill_recalculation = yes
		save_scope_value_as = { name = modifier_skill value = $SKILL$ }
		save_scope_value_as = { name = base_$SKILL$ value = 0 }
		while = {
			limit = { $SKILL$ < scope:saved_skill }
			add_$SKILL$_skill = 1
			force_character_skill_recalculation = yes
			save_scope_value_as = { name = base_$SKILL$ value = { value = scope:base_$SKILL$ add = 1 } }
		}
		remove_character_modifier = save_real_$SKILL$_negate_modifier
		remove_all_character_modifier_instances = save_real_$SKILL$_modifier
		force_character_skill_recalculation = yes
	}
}

POD_copy_sexuality_effect = {
	if = {
		limit = { $ORIGIN$ = { has_sexuality = asexual } }
		set_sexuality = asexual
	}
	else_if = {
		limit = { $ORIGIN$ = { has_sexuality = bisexual } }
		set_sexuality = bisexual
	}
	else_if = {
		limit = { $ORIGIN$ = { has_sexuality = heterosexual } }
		if = {
			limit = {
				OR = {
					AND = {
						is_female = yes
						$ORIGIN$ = { is_female = yes } 
					}
					AND = {
						is_female = no
						$ORIGIN$ = { is_female = no } 
					}
				}
			}
			set_sexuality = heterosexual
		}
		else = {
			set_sexuality = homosexual
		}
	}
	else_if = {
		limit = { $ORIGIN$ = { has_sexuality = homosexual } }
		if = {
			limit = {
				OR = {
					AND = {
						is_female = yes
						$ORIGIN$ = { is_female = yes } 
					}
					AND = {
						is_female = no
						$ORIGIN$ = { is_female = no } 
					}
				}
			}
			set_sexuality = homosexual
		}
		else = {
			set_sexuality = heterosexual
		}
	}
}
