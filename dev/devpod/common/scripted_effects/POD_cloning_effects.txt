
# Effects in this file copy various things from scope:cloned_character into current scope

#POD_remove_vanilla_traits_effect - removes vanilla not physical traits
#POD_remove_physical_traits_effect - removes vanilla congenital traits and various traits affecting physical condition in one way or another
#POD_remove_vanilla_secrets_effect - removes vanilla secrets
#POD_remove_supernatural_secrets_effect - removes POD secrets
#POD_copy_vanilla_traits_effect - copies vanilla not physical traits
#POD_copy_physical_traits_effect - copies vanilla congenital traits and various traits affecting physical condition in one way or another
#POD_copy_vanilla_secrets_effect - copies vanilla secrets
#POD_copy_supernatural_secrets_effect - copies POD secrets
#POD_copy_physical_appearance_effect - copies physical appearance, right now immortal age is set to 30 by default
#POD_copy_base_skills_effect - copies base skills, should be used before traits, modifiers and perks are copied
#POD_copy_supernatural_lifestyle_perks_effect - copies POD lifestyle perks
#POD_copy_vanilla_lifestyle_perks_effect - copies vanilla lifestyle perks
#POD_copy_vampire_modifiers_effect - copies POD vampire modifiers, such as blood potency, hunger, and masquerade exposure


POD_create_shadow_twin_effect = {
	save_scope_as = cloned_character
	 
	create_character = {
		save_scope_as = shadow_twin
		age = scope:cloned_character.age
		faith = scope:cloned_character.faith
		culture = scope:cloned_character.culture
		gender = scope:cloned_character
		random_traits = no
		dynasty = none
		employer = scope:cloned_character
	}
	if = {
		limit = {
			exists = scope:cloned_character
			exists = scope:shadow_twin
		}
		scope:shadow_twin = {
			POD_copy_base_skills_effect = {
				CLONED_CHAR = scope:cloned_character
				CLONE = scope:shadow_twin
			}
			copy_traits = scope:cloned_character
			POD_copy_vampire_modifiers_effect = yes
			POD_copy_supernatural_secrets_effect = yes
			POD_copy_trait_xp_for_clone_effect = yes
			POD_copy_sexuality_effect = { ORIGIN = scope:cloned_character }
			change_first_name =  { template_character = scope:cloned_character }
			POD_copy_physical_appearance_effect = yes
			POD_copy_vanilla_lifestyle_perks_effect = yes
			POD_copy_supernatural_lifestyle_perks_effect = yes
			add_character_modifier = shadow_twin_modifier
			give_nickname = nick_shadow_twin
			set_relation_shadow_twin_master = scope:cloned_character
			add_character_flag = blocked_from_leaving
		}
		scope:cloned_character = {
			add_hook = {
				target = scope:shadow_twin
				type = shadow_twin_bind_hook
			}
		}
	}
}

			POD_copy_supernatural_lifestyle_perks_effect = yes

POD_get_base_martial_effect = {
	if = {
		limit = { martial >= 100 }
		add_character_modifier = save_real_martial_negate_modifier
		add_character_modifier = save_real_martial_modifier
		force_character_skill_recalculation = yes
		while = {
			limit = { martial >= 100 }
			add_character_modifier = save_real_martial_modifier
			force_character_skill_recalculation = yes
		}
	}
	set_local_variable = { name = base_martial value = martial }
	add_martial_skill = -256
	add_martial_skill = 128
	force_character_skill_recalculation = yes
	change_local_variable = { name = base_martial subtract = martial }
	add_martial_skill = local_var:base_martial
	if = {
		limit = { has_character_modifier = save_real_martial_negate_modifier }
		remove_character_modifier = save_real_martial_negate_modifier
		remove_all_character_modifier_instances = save_real_martial_modifier
	}
	force_character_skill_recalculation = yes
	save_scope_value_as = { name = base_martial value = local_var:base_martial }
}

			limit = {
				OR = {
					AND = {
						is_female = yes

					AND = {
						is_female = no
						$ORIGIN$ = { is_female = no }

		}

			set_sexuality = homosexual
		}
	}

		limit = { $ORIGIN$ = { has_sexuality = homosexual } }
		if = {
			limit = {
				OR = {
					AND = {
						is_female = yes

					}
					AND = {
						is_female = no

				}
