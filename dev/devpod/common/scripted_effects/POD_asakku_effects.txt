######################################################
# Depth Management
######################################################

# Set asakku_depth to a specific value
POD_asakku_set_depth_effect = {
	set_variable = {
		name = asakku_depth
		value = $VALUE$
	}
	POD_asakku_sync_depth_to_story_effect = yes
}

# Mirrors var:asakku_depth onto the depth_accumulation story
# so the story-cycle visualization (basic_counter) renders correctly.
# Called from set/increment/decrement effects below.
POD_asakku_sync_depth_to_story_effect = {
	every_owned_story = {
		limit = { story_type = POD_asakku_depth_accumulation }
		set_variable = {
			name = depth
			value = prev.var:asakku_depth
		}
	}
}

# Increment asakku_depth by a given amount (clamped to 100)
POD_asakku_increment_depth_effect = {
	# AI characters cap at depth 25 — once they become Asakku, no further growth
	if = {
		limit = {
			is_ai = yes
			has_variable = asakku_depth
			var:asakku_depth >= 25
		}
		# No-op: AI depth stays at 25
	}
	else = {
		if = {
			limit = { NOT = { has_variable = asakku_depth } }
			set_variable = {
				name = asakku_depth
				value = 0
			}
		}
		change_variable = {
			name = asakku_depth
			add = $AMOUNT$
		}
		if = {
			limit = {
				has_variable = asakku_depth
				var:asakku_depth > 100
			}
			set_variable = {
				name = asakku_depth
				value = 100
			}
		}
		POD_asakku_sync_depth_to_story_effect = yes
	}
}

# Decrement asakku_depth by a given amount (clamped to 0)
POD_asakku_decrement_depth_effect = {
	if = {
		limit = { has_variable = asakku_depth }
		change_variable = {
			name = asakku_depth
			add = {
				value = $AMOUNT$
				multiply = -1
			}
		}
		if = {
			limit = { var:asakku_depth < 0 }
			set_variable = {
				name = asakku_depth
				value = 0
			}
		}
		POD_asakku_sync_depth_to_story_effect = yes
	}
}

######################################################
# Threshold Check - fires appropriate events
######################################################

POD_asakku_check_depth_thresholds_effect = {
	if = {
		limit = { has_variable = asakku_depth }
		# Depth 100 - Iryi Ascendant
		if = {
			limit = {
				var:asakku_depth >= 100
				NOT = { has_trait = POD_asakku_iryi_ascendant }
			}
			POD_asakku_grant_iryi_ascendant_effect = yes
		}
		# Depth 75 - Third threshold
		else_if = {
			limit = {
				var:asakku_depth >= 75
				NOT = { has_character_flag = POD_asakku_depth_75_reached }
			}
			add_character_flag = POD_asakku_depth_75_reached
			trigger_event = POD_asakku.0012
		}
		# Depth 50 - Second threshold
		else_if = {
			limit = {
				var:asakku_depth >= 50
				NOT = { has_character_flag = POD_asakku_depth_50_reached }
			}
			add_character_flag = POD_asakku_depth_50_reached
			trigger_event = POD_asakku.0011
		}
		# Depth 25 - First threshold (Asakku)
		else_if = {
			limit = {
				var:asakku_depth >= 25
				NOT = { has_character_flag = POD_asakku_depth_25_reached }
			}
			add_character_flag = POD_asakku_depth_25_reached
			trigger_event = POD_asakku.0010
		}
	}
}

# Trait Grants
######################################################

# Grant Soul Infection trait and initialise depth
POD_asakku_grant_soul_infection_effect = {
	if = {
		limit = { NOT = { has_trait = POD_asakku_soul_infection } }
		add_trait = POD_asakku_soul_infection
	}
	# Create story cycle for depth accumulation
	if = {
		limit = {
			NOT = { has_character_flag = POD_asakku_has_depth_story }
		}
		add_character_flag = POD_asakku_has_depth_story
		create_story = POD_asakku_depth_accumulation
	}
	if = {
		limit = { NOT = { has_variable = asakku_depth } }
		POD_asakku_set_depth_effect = { VALUE = 1 }
	}
	else_if = {
		limit = { var:asakku_depth < 1 }
		POD_asakku_set_depth_effect = { VALUE = 1 }
	}
}

# Grant Asakku trait (replaces Soul Infection)
# Also grants possession and sets Corrupted Way faith
POD_asakku_grant_asakku_effect = {
	# Remove Soul Infection if present
	if = {
		limit = { has_trait = POD_asakku_soul_infection }
		remove_trait = POD_asakku_soul_infection
	}
	# Grant Asakku
	if = {
		limit = { NOT = { has_trait = POD_asakku_asakku } }
		add_trait = POD_asakku_asakku
	}
	# Set Corrupted Way faith
	if = {
		limit = {
			NOT = { faith = faith:corruptedway }
		}
		set_character_faith = faith:corruptedway
	}				
	# Remove Elioud if present (you can't be both hunter and hunted)
	if = {
		limit = { has_trait = POD_asakku_elioud }
		remove_trait = POD_asakku_elioud
	}
	# Remove any Elioud rivalries - they've switched sides
	every_relation = {
		type = rival
		limit = {
			has_trait = POD_asakku_elioud
		}
		remove_relation_rival = prev
	}
	# Grant possessed trait
	if = {
		limit = {
			NOT = { has_trait = possessed_1 }
		}
		add_trait = possessed_1
	}
	# Create story cycle for depth accumulation if not already present
	if = {
		limit = {
			NOT = { has_character_flag = POD_asakku_has_depth_story }
		}
		add_character_flag = POD_asakku_has_depth_story
		create_story = POD_asakku_depth_accumulation
	}
	# Ensure depth is at least 25
	if = {
		limit = { NOT = { has_variable = asakku_depth } }
		POD_asakku_set_depth_effect = { VALUE = 25 }
	}
	else_if = {
		limit = { var:asakku_depth < 25 }
		POD_asakku_set_depth_effect = { VALUE = 25 }
	}
}

# Grant Iryi Ascendant (capstone at depth 100)
POD_asakku_grant_iryi_ascendant_effect = {
	# Remove lower progression traits
	if = {
		limit = { has_trait = POD_asakku_soul_infection }
		remove_trait = POD_asakku_soul_infection
	}
	if = {
		limit = { has_trait = POD_asakku_asakku }
		remove_trait = POD_asakku_asakku
	}
	# Grant capstone
	if = {
		limit = { NOT = { has_trait = POD_asakku_iryi_ascendant } }
		add_trait = POD_asakku_iryi_ascendant
	}
	# Set depth to exactly 100
	POD_asakku_set_depth_effect = { VALUE = 100 }
	# Fire capstone event
	trigger_event = POD_asakku.0013
}

######################################################
# Sinister Mastery Management
######################################################

POD_asakku_apply_sinister_mastery_effect = {
	if = {
		limit = { var:asakku_depth >= 50 }
		# Upgrade to tier 2
		if = {
			limit = { has_character_modifier = POD_asakku_sinister_mastery_1 }
			remove_character_modifier = POD_asakku_sinister_mastery_1
		}
		if = {
			limit = { NOT = { has_character_modifier = POD_asakku_sinister_mastery_2 } }
			add_character_modifier = POD_asakku_sinister_mastery_2
		}
	}
	else_if = {
		limit = { var:asakku_depth >= 25 }
		# Apply tier 1
		if = {
			limit = { NOT = { has_character_modifier = POD_asakku_sinister_mastery_1 } }
			add_character_modifier = POD_asakku_sinister_mastery_1
		}
	}
}

# Elioud Rivalry - create rivalry with Andeleon

POD_asakku_elioud_rivalry_with_andeleon_effect = {

		limit = {
			has_trait = POD_asakku_elioud
			in_diplomatic_range = character:790612
			NOT = { has_relation_rival = character:790612 }
		}
		set_relation_rival = {
			reason = rival_asakku_threat
			target = character:790612
		}
	}
}
