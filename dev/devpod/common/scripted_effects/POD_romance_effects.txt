POD_had_sex_with_effect = {
	$CHARACTER$ = {
		if = {
			limit = { 
				has_character_modifier = oath_of_truehearts_modifier
				NOT = { 
					has_opinion_modifier = {
						modifier = oath_of_truehearts_opinion
						target = scope:had_sex_root_character
					}
				}
			}
			every_relation = { 
				type = lover
				POD_oathbreaking_effect = {
					TARGET = scope:had_sex_with_effect_partner
					THIRD = scope:had_sex_root_character
					OATH = oath_of_truehearts
				}
			}
			remove_character_modifier = oath_of_truehearts_modifier
			add_trait = oathbreaker
		}
	}
	if = {
		limit = { 
			has_character_modifier = oath_of_truehearts_modifier
			NOT = { 
				has_opinion_modifier = {
					modifier = oath_of_truehearts_opinion
					target = scope:had_sex_with_effect_partner
				}
			}
		}
		every_relation = { 
			type = lover
			POD_oathbreaking_effect = {
				TARGET = scope:had_sex_root_character
				THIRD = scope:had_sex_with_effect_partner
				OATH = oath_of_truehearts
			}
		}
		remove_character_modifier = oath_of_truehearts_modifier
		add_trait = oathbreaker
	}
}
POD_had_sex_with_unknown_effect = {
	if = {
		limit = { has_character_modifier = oath_of_truehearts_modifier }
		save_scope_as = cheater
		every_relation = { 
			type = lover
			POD_oathbreaking_effect = {
				TARGET = scope:cheater
				THIRD = scope:cheater
				OATH = oath_of_truehearts
			}
		}
		remove_character_modifier = oath_of_truehearts_modifier
		add_trait = oathbreaker
	}
}
