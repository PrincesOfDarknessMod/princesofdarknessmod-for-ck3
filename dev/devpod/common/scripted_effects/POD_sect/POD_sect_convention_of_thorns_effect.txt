unvote_tenet_effect = {
	if = {
		limit = {
			is_target_in_variable_list = { 
				name = voted_tenet 
				target = $TENET$
			}
		}
		remove_list_variable = { name = voted_tenet target = $TENET$ }
		faith:camahumanity = { down_the_vote_effect = { TENET = $TENET$ } }
	}
}

vote_tenet_effect = {
	if = {
		limit = {
			NOT = {
				is_target_in_variable_list = { 
					name = voted_tenet 
					target = $TENET$ 
				}
			}
		}
		add_to_variable_list = { name = voted_tenet target = $TENET$ }
		faith:camahumanity = { up_the_vote_effect = { TENET = $TENET$ } }
	}
}
