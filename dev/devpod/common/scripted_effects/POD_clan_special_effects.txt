POD_character_vampire_malkavian_initialization_effect = {
	if = {
		limit = {
			is_malkavian_trigger = yes
		}
		# Make madness traits congenital so they can't be removed by events/interactions
		if = {
			limit = { has_trait = lunatic_1 }
			set_trait_rank = {
				trait = lunatic
				rank = 2
			}
		}
		else_if = {
			limit = { has_trait = possessed_1 }
			set_trait_rank = {
				trait = possessed
				rank = 2
			}
		}
		else_if = {
			limit = { has_trait = depressed_1 }
			set_trait_rank = {
				trait = depressed
				rank = 2
			}
		}
		else_if = {
			limit = {
				# If they already have a madness trait, don't give them another one
				NOR = {
					has_trait = lunatic_genetic
					has_trait = possessed_genetic
					has_trait = depressed_genetic
				}
			}
			POD_character_vampire_madness_initialization_effect = yes
		}
	}
}

POD_character_vampire_tzimisce_initialization_effect = {
	if = {
		limit = {
			is_tzimisce_trigger = yes
			NOT = { has_character_flag = had_POD_character_tzimisce_looks }
			#OR = {
			#	has_perk = malleable_visage_perk
			#	has_trait = vicissitudeadvanced
			#}
			NOT = { has_trait = beauty_good }
			NOT = { dynasty ?= dynasty:10000015 }
		}
		
		random_list = {
			# Make the Tzimisce hotter
			20 = {
				random_list = {
					70 = { add_trait = beauty_good_1 }
					25 = { add_trait = beauty_good_2 }
					5 = { add_trait = beauty_good_3 }
				}
			}
			# But most are still weird
			80 = {
				POD_shapeshift_vicissitude_face_randomize_effect = yes
				random = {
					chance = 50
					POD_shapeshift_vicissitude_face_randomize_effect = yes
				}
				random = {
					chance = 25
					POD_shapeshift_vicissitude_face_randomize_effect = yes
				}
				random = {
					chance = 5
					POD_shapeshift_vicissitude_face_randomize_effect = yes
				}
				# and a lucky few are weird and hot
				random = {
					chance = 10
					add_trait = beauty_good_1
				}
			}
		}		

		add_character_flag = had_POD_character_tzimisce_looks
	}

	if = {
		limit = {
			is_tzimisce_trigger = yes
			is_landed = yes
			is_ai = yes
		}
		random = {
			chance = 10
			POD_give_massregnant_secret_effect = yes
		}
	}
}
