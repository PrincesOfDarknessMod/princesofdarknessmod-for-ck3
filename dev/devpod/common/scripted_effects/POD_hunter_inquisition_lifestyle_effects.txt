	if = {
		limit = {

				has_trait_xp = { trait = beatific value = 19 }

				has_trait_xp = { trait = beatific value = 59 }
				has_trait_xp = { trait = beatific value = 79 }
				has_trait_xp = { trait = beatific value = 99 }
			}
		}
		add_trait_xp = { trait = beatific value = 1 }
	}
}

POD_advance_beatific_multiple_effect = {
	if = {
		limit = { 
			OR = {
				has_trait_xp = { trait = beatific value = 19 }
				has_trait_xp = { trait = beatific value = 39 }
				has_trait_xp = { trait = beatific value = 59 }
				has_trait_xp = { trait = beatific value = 79 }
				has_trait_xp = { trait = beatific value = 99 }
			}
		}
		# Do Nothing
	}
	else_if = {
		limit = { has_trait_xp = { trait = beatific value < 20 } }
		add_trait_xp = { trait = beatific value = $VALUE$ }
		if = {
			limit = { has_trait_xp = { trait = beatific value >= 20 } }
			add_trait_xp = { trait = beatific value = -100 }
			add_trait_xp = { trait = beatific value = 19 }
		}
	}
	else_if = {
		limit = { has_trait_xp = { trait = beatific value < 40 } }
		add_trait_xp = { trait = beatific value = $VALUE$ }
		if = {
			limit = { has_trait_xp = { trait = beatific value >= 40 } }
			add_trait_xp = { trait = beatific value = -100 }
			add_trait_xp = { trait = beatific value = 39 }
		}
	}
	else_if = {
		limit = { has_trait_xp = { trait = beatific value < 60 } }
		add_trait_xp = { trait = beatific value = $VALUE$ }
		if = {
			limit = { has_trait_xp = { trait = beatific value >= 60 } }
			add_trait_xp = { trait = beatific value = -100 }
			add_trait_xp = { trait = beatific value = 59 }
		}
	}
	else_if = {
		limit = { has_trait_xp = { trait = beatific value < 80 } }
		add_trait_xp = { trait = beatific value = $VALUE$ }
		if = {
			limit = { has_trait_xp = { trait = beatific value >= 80 } }
			add_trait_xp = { trait = beatific value = -100 }
			add_trait_xp = { trait = beatific value = 79 }

	else_if = {
		limit = { has_trait_xp = { trait = beatific value < 100 } }

	if = {
		limit = {
			scope:defender = { is_vampire_trigger = yes }
		}
		if = {
			limit = { scope:defender = { age >= 1000 } }
			add_prestige = 200
			dynasty ?= { add_dynasty_prestige = major_dynasty_prestige_gain }
		}
		else_if = {
			limit = { scope:defender = { age >= 400 } }
			add_prestige = 150
			dynasty ?= { add_dynasty_prestige = medium_dynasty_prestige_gain }
		}
		else_if = {
			limit = { scope:defender = { age >= 200 } }
			add_prestige = 100
			dynasty ?= { add_dynasty_prestige = minor_dynasty_prestige_gain }
		}
		else = {
			add_prestige = 50
			dynasty ?= { add_dynasty_prestige = miniscule_dynasty_prestige_gain }
		}
	}
	
	else_if = {
		limit = {
			scope:defender = { is_werewolf_trigger = yes }
		}
		add_prestige = 100
		dynasty ?= { add_dynasty_prestige = minor_dynasty_prestige_gain }
	}
	
	else_if = {
		limit = {
			scope:defender = { is_demon_trigger = yes }
		}
		add_prestige = 150
		dynasty ?= { add_dynasty_prestige = medium_dynasty_prestige_gain }
	}

	else_if = {
		limit = {
			scope:defender = { is_fae_trigger = yes }
		}
		add_prestige = 100
		dynasty ?= { add_dynasty_prestige = minor_dynasty_prestige_gain }
	}

	else = {
		add_prestige = 50
		dynasty ?= { add_dynasty_prestige = miniscule_dynasty_prestige_gain }
