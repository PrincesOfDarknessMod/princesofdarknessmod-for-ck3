# Personal COA by DYLAN/AGOT

POD_create_personal_coa_effect = {
	save_scope_as = actor
	add_character_flag = POD_has_personal_coa_flag
	if = {
		limit = {
			scope:actor = { is_landed = yes }
		}
		create_character = {
			#name = "Dummy"
			gender = male
			employer = scope:actor
			trait = "content"
			trait = "shy"
			trait = "lazy"
			trait = "wraith"
			trait = "torpor"
			trait = "education_diplomacy_1"
			trait = "diplomatic_court_1"
			diplomacy = 0
			stewardship = 0
			martial = 0
			intrigue = 0
			learning = 0
			random_traits = no
			faith = faith:innocence
			culture = title:e_stygia.holder.culture
			dynasty = generate
			save_scope_as = recipient
		}
	}
	else_if = {
		limit = {
			scope:actor = {	is_landed = no }
		}
		create_character = {
			#name = "Dummy"
			gender = male
			employer = character:POD_wraith_01 #Charon
			trait = "content"
			trait = "shy"
			trait = "lazy"
			trait = "wraith"
			trait = "torpor"
			trait = "education_diplomacy_1"
			trait = "diplomatic_court_1"
			diplomacy = 0
			stewardship = 0
			martial = 0
			intrigue = 0
			learning = 0
			random_traits = no
			faith = faith:innocence
			culture = title:e_stygia.holder.culture
			dynasty = generate
			save_scope_as = recipient
		}
	}
	
	scope:recipient = {
		set_global_variable = {
			name = coa_new_name
			value = scope:actor
		}
		scope:recipient.house = {
			set_house_name = coa_new_name
		}
		remove_global_variable = coa_new_name
		set_variable = {
			name = my_coa_belongs_to
			value = scope:actor
		}
		add_character_flag = POD_dde_hre_war_low_nobles_flag
		add_character_flag = POD_portrait_hermetic_mage_hood_male_flag
		add_character_flag = POD_ep1_adventurer_cloak_flag
	}
	if = {
		limit = {
			scope:actor = {
				is_lowborn = yes
			}
		}
		scope:recipient.house = { generate_coa = yes }
	} else = {
		scope:recipient.house = { set_coa = scope:actor.house }
	}
	scope:actor = {
		set_variable = {
			name = my_personal_coa
			value = scope:recipient
		}
		set_relation_pod_personal_coa_slave = scope:recipient
		if = {
			limit = {
				scope:actor = {
					is_ai = yes
				}
			}
			scope:recipient = { save_scope_as = exiled_dummy } # Send away the COA Dummy Immediately since the AI won't be editing their own COAs
			trigger_event = POD_personal_coat_of_arms.100
		} else = {
			scope:recipient = { save_scope_as = exiled_dummy } # Send away the COA Dummy a few days after
			trigger_event = { id = POD_personal_coat_of_arms.100 days = 14 }
		}
	}
}

}

POD_prune_personal_coa_effect = {
	var:my_personal_coa = {
		death = {
			death_reason = death_vanished
		}
	}
}
