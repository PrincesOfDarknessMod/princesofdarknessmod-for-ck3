POD_become_hideous_effect = { 
	switch = {
		trigger = has_trait
		beauty_good_3 = { remove_trait = beauty_good_3 add_trait = beauty_good_2 }
		beauty_good_2 = { remove_trait = beauty_good_2 add_trait = beauty_good_1 }
		beauty_bad_1 = { remove_trait = beauty_good_1 }
		fallback = { add_trait = beauty_bad_1 }
		beauty_bad_1 = { remove_trait = beauty_bad_1 add_trait = beauty_bad_2 }
		beauty_bad_2 = { remove_trait = beauty_bad_2 add_trait = beauty_bad_3 }
	}
}
