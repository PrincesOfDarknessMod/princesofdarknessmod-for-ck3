######################################################################
# One Holding Rule - effects
#
# Capacity (the would-be domain limit the tax bonus scales on) is read from a
# DEFERRED event, not inline because domain_limit cannot be re-read while 
# POD_onehold_cap is applied. It would read the capped value of 1. Removing 
# the cap and reading in the same effect also returns the stale cached value, 
# so a tick must pass first.
#
# Driven on game start, title gain, 5-year pulse.
######################################################################

# Uncap now and queue a deferred read. No-op if the character should not be capped
# (rule off, exempt by tier, or not landed) or a recalc is already pending.
POD_onehold_request_recalc_effect = {
	if = {
		limit = {
			POD_onehold_should_cap_trigger = yes
		}
		remove_character_modifier = POD_onehold_cap
		POD_onehold_remove_all_tax_rungs_effect = yes
		trigger_event = {
			id = POD_onehold.1
			days = 1
		}
	}
}

POD_onehold_refresh_tax_rung_effect = {
	POD_onehold_remove_all_tax_rungs_effect = yes
	switch = {
		trigger = domain_limit
		13 = { add_character_modifier = POD_onehold_tax_12 }
		12 = { add_character_modifier = POD_onehold_tax_11 }
		11 = { add_character_modifier = POD_onehold_tax_10 }
		10 = { add_character_modifier = POD_onehold_tax_9 }
		9 = { add_character_modifier = POD_onehold_tax_8 }
		8 = { add_character_modifier = POD_onehold_tax_7 }
		7 = { add_character_modifier = POD_onehold_tax_6 }
		6 = { add_character_modifier = POD_onehold_tax_5 }
		5 = { add_character_modifier = POD_onehold_tax_4 }
		4 = { add_character_modifier = POD_onehold_tax_3 }
		3 = { add_character_modifier = POD_onehold_tax_2 }
		2 = { add_character_modifier = POD_onehold_tax_1 }
	}

}

POD_onehold_remove_all_tax_rungs_effect = {
	remove_character_modifier = POD_onehold_tax_1
	remove_character_modifier = POD_onehold_tax_2
	remove_character_modifier = POD_onehold_tax_3
	remove_character_modifier = POD_onehold_tax_4
	remove_character_modifier = POD_onehold_tax_5
	remove_character_modifier = POD_onehold_tax_6
	remove_character_modifier = POD_onehold_tax_7
	remove_character_modifier = POD_onehold_tax_8
	remove_character_modifier = POD_onehold_tax_9
	remove_character_modifier = POD_onehold_tax_10
	remove_character_modifier = POD_onehold_tax_11
	remove_character_modifier = POD_onehold_tax_12
}

# Undo everything (used when the rule is found OFF on a character who was
# previously set up, e.g. the rule was disabled mid-game).
POD_onehold_cleanup_character_effect = {
	if = {
		limit = { has_character_modifier = POD_onehold_cap }
		remove_character_modifier = POD_onehold_cap
		POD_onehold_remove_all_tax_rungs_effect = yes
	}
}

# Per-character maintenance from the pulse: re-read capacity (so later
# house-modifier / law / tier changes are picked up) for characters that should be
# capped, or tear down for those that should not (rule off, or now tier-exempt).
POD_onehold_pulse_effect = {
	if = {
		limit = { POD_onehold_should_cap_trigger = yes }
		POD_onehold_request_recalc_effect = yes
	}
	else = {
		POD_onehold_cleanup_character_effect = yes
	}
}
