POD_gnosis_increase_effect = {
	if = {
		limit = {
			POD_splat_has_gnosis_trigger = yes
			NOT = { stress_level >= 3 }
			NOT = { has_character_modifier = POD_gnosis_5_modifier }	
		}
		custom_tooltip = POD_gnosis_increase_effect.tt

			if = {
				limit = {
					has_character_modifier = POD_gnosis_4_modifier
					POD_is_kinfolk_trigger = no
				}
					hidden_effect = {
						remove_character_modifier = POD_gnosis_4_modifier
					}
					add_character_modifier = {
						modifier = POD_gnosis_5_modifier						
					}
			}

			if = {
				limit = {
					has_character_modifier = POD_gnosis_3_modifier
					POD_is_kinfolk_trigger = no
				}
					hidden_effect = {
						remove_character_modifier = POD_gnosis_3_modifier
					}
					add_character_modifier = {
						modifier = POD_gnosis_4_modifier					
					}
			}

			if = {
				limit = {
					has_character_modifier = POD_gnosis_2_modifier
				}
					hidden_effect = {
						remove_character_modifier = POD_gnosis_2_modifier
					}
					add_character_modifier = {
						modifier = POD_gnosis_3_modifier						
					}
			}

			if = {
				limit = {
					has_character_modifier = POD_gnosis_1_modifier
				}
					hidden_effect = {
						remove_character_modifier = POD_gnosis_1_modifier
					}
					add_character_modifier = {
						modifier = POD_gnosis_2_modifier						
					}
			}

			if = {
				limit = {
					has_character_modifier = POD_gnosis_0_modifier
				}
					hidden_effect = {
						remove_character_modifier = POD_gnosis_0_modifier

					}

	}
}

POD_gnosis_growing_maintenance_effect = {
	if = {
		limit = {
			POD_splat_has_gnosis_trigger = yes
			OR = {
				AND = {
					POD_is_kinfolk_trigger = no
					NOT = { has_character_modifier = POD_gnosis_5_modifier }
				}
				AND = {
					POD_is_kinfolk_trigger = yes
					NOT = { has_character_modifier = POD_gnosis_3_modifier }
				}
			}
			NOT = { stress_level >= 3 }
		}
		
		random = {
			chance = 15
		
			modifier = {
				add = 10
				has_trait = homid_born
			}
			modifier = {
				add = 20
				has_trait = metis
			}
			modifier = {
				add = 30
				has_trait = beast_born
			}	
			modifier = {
				add = 5
				piety_level = 1
			}	
			modifier = {
				add = 10
				piety_level = 2
			}		
			modifier = {
				add = 15
				piety_level = 3
			}		
			modifier = {
				add = 20
				piety_level = 4
			}		
			modifier = {
				add = 25
				piety_level >= 5
			}		
			modifier = {
				POD_is_kinfolk_trigger = yes
				POD_splat_has_gnosis_trigger = yes
				add = 5
			}
			
			POD_gnosis_increase_effect = yes
		}
	}
}

			change_variable = { name = stored_sekhem subtract = 5 }
			$OWNER$ = { custom_tooltip = POD_extract_gnosis_5_tooltip }
		}
	}
	if = {
		limit = { NOT = { has_variable = temporary_vessel } }
		set_vessel_artifact_name_effect = yes
	}
	if = {
		limit = { 
			OR = {
				has_variable = temporary_vessel
				var:stored_sekhem <= 0 
			}
			NOT = { has_variable = permanent_vessel }
		}
		$OWNER$ = {
			send_interface_toast = {
				right_icon = prev
				title = gnosis_vessel_damaged
				prev = { degrade_vessel_durability_effect = yes }
			}
		}
	}
}

POD_gift_cost_effect = {
	if = {
		limit = {
			any_character_artifact = {
				has_variable = stored_sekhem
				var:stored_sekhem > 0
			}
		}
		ordered_character_artifact = {
			limit = {
				has_variable = stored_sekhem 
				var:stored_sekhem > 0
			}
			order_by = {
				value = 0
				add = var:stored_sekhem
			}
			max = 1
			POD_gnosis_vessel_decrease_multiple_effect = { COUNT = $COUNT$ OWNER = prev }
		}
	}
	else = {
		POD_gnosis_decrease_multiple_effect = { COUNT = $COUNT$ }
	}
}
