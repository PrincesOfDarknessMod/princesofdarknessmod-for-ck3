POD_gnosis_gain_check_effect = {
	custom_tooltip = POD_gnosis_gain_check_effect.tt

	}

POD_gnosis_increase_effect = {
	if = {
		limit = {
			POD_splat_has_gnosis_trigger = yes
			NOT = { stress_level >= 3 }
			NOT = { has_character_modifier = POD_gnosis_5_modifier }	
		}
		custom_tooltip = POD_gnosis_increase_effect.tt

			if = {
				limit = {
					has_character_modifier = POD_gnosis_4_modifier
					POD_is_kinfolk_trigger = no
				}
					hidden_effect = {
						remove_character_modifier = POD_gnosis_4_modifier
					}
					add_character_modifier = {
						modifier = POD_gnosis_5_modifier						
					}
			}

			if = {
				limit = {
					has_character_modifier = POD_gnosis_3_modifier
					POD_is_kinfolk_trigger = no
				}
					hidden_effect = {
						remove_character_modifier = POD_gnosis_3_modifier
					}
					add_character_modifier = {
						modifier = POD_gnosis_4_modifier					
					}
			}

			if = {
				limit = {
					has_character_modifier = POD_gnosis_2_modifier
				}
					hidden_effect = {
						remove_character_modifier = POD_gnosis_2_modifier
					}
					add_character_modifier = {
						modifier = POD_gnosis_3_modifier						
					}
			}

			if = {
				limit = {
					has_character_modifier = POD_gnosis_1_modifier
				}
					hidden_effect = {
						remove_character_modifier = POD_gnosis_1_modifier
					}
					add_character_modifier = {
						modifier = POD_gnosis_2_modifier						
					}
			}

			if = {
				limit = {
					has_character_modifier = POD_gnosis_0_modifier
				}
					hidden_effect = {
						remove_character_modifier = POD_gnosis_0_modifier
					}
					add_character_modifier = {
						modifier = POD_gnosis_1_modifier			
					}
			}
	}
}

POD_gnosis_growing_maintenance_effect = {
	if = {
		limit = {
			POD_splat_has_gnosis_trigger = yes
			OR = {
				AND = {
					POD_is_kinfolk_trigger = no
					NOT = { has_character_modifier = POD_gnosis_5_modifier }
				}
				AND = {
					POD_is_kinfolk_trigger = yes
					NOT = { has_character_modifier = POD_gnosis_3_modifier }
				}
			}
			NOT = { stress_level >= 3 }
		}
		
		random = {
			chance = 15
		
			modifier = {
				add = 10
				has_trait = homid_born
			}
			modifier = {
				add = 20
				has_trait = metis
			}
			modifier = {
				add = 30
				has_trait = lupus_born
			}	
			modifier = {
				add = 5
				piety_level = 1
			}	
			modifier = {
				add = 10
				piety_level = 2
			}		
			modifier = {
				add = 15
				piety_level = 3
			}		
			modifier = {
				add = 20
				piety_level = 4
			}		
			modifier = {
				add = 25
				piety_level >= 5
			}		
			modifier = {
				POD_is_kinfolk_trigger = yes
				POD_splat_has_gnosis_trigger = yes
				add = 5
			}
			
			POD_gnosis_increase_effect = yes
		}
	}
}

POD_assign_initial_gnosis_effect = {
	if = {
		limit = {
			POD_has_gnosis_modifier_trigger = no
			POD_splat_has_gnosis_trigger = yes
		}

		if = { 
			limit = {
				POD_is_kinfolk_trigger = yes
			}
			add_character_modifier = {
				modifier = POD_gnosis_0_modifier			
			}
		}
		
		if = { 
			limit = {
				has_trait = homid_born
			}
			add_character_modifier = {
				modifier = POD_gnosis_1_modifier			
			}
		}

		if = { 
			limit = {
				has_trait = metis
			}
			add_character_modifier = {
				modifier = POD_gnosis_2_modifier			
			}
		}

		if = { 
			limit = {
				has_trait = lupus_born
			}
			add_character_modifier = {
				modifier = POD_gnosis_3_modifier			
			}
		}
	}
}
