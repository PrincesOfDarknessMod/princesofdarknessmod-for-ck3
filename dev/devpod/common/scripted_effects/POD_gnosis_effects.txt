POD_gnosis_increase_effect = {
	if = {
		limit = {
			POD_splat_has_gnosis_trigger = yes
			NOT = { stress_level >= 3 }
			NOT = { has_character_modifier = POD_gnosis_5_modifier }	
		}
		custom_tooltip = POD_gnosis_increase_effect.tt

		if = {
			limit = {has_character_modifier = POD_gnosis_0_modifier}
			hidden_effect = { remove_character_modifier = POD_gnosis_0_modifier }
			add_character_modifier = POD_gnosis_1_modifier
		}
		else_if = {
			limit = {has_character_modifier = POD_gnosis_1_modifier}
			hidden_effect = { remove_character_modifier = POD_gnosis_1_modifier }
			add_character_modifier = POD_gnosis_2_modifier
		}
		else_if = {
			limit = {has_character_modifier = POD_gnosis_2_modifier}
			hidden_effect = { remove_character_modifier = POD_gnosis_2_modifier }
			add_character_modifier = POD_gnosis_3_modifier
		}
		else_if = {
			limit = {
				has_character_modifier = POD_gnosis_3_modifier
				POD_is_kinfolk_trigger = no
			}
			hidden_effect = { remove_character_modifier = POD_gnosis_3_modifier }
			add_character_modifier = POD_gnosis_4_modifier
		}
		else_if = {

POD_gnosis_vessel_decrease_multiple_effect = {
	if = {
		limit = { var:stored_sekhem >= $COUNT$ }
		custom_description = {
			text = POD_extract_gnosis
			subject = $OWNER$
			value = $COUNT$
			change_variable = { name = stored_sekhem subtract = $COUNT$ }
		}
	}
	else = {
		save_scope_value_as = {
			name = excess_gnosis
			value = {
				value = $COUNT$
				subtract = var:stored_sekhem
			}
		}
		custom_description = {
			text = POD_extract_gnosis
			subject = $OWNER$
			value = var:stored_sekhem
			change_variable = { name = stored_sekhem subtract = var:stored_sekhem }
		}
		$OWNER$ = {
			POD_gnosis_decrease_multiple_effect = { COUNT = scope:excess_gnosis }
		}
	}
	if = {
		limit = { NOT = { has_variable = temporary_vessel } }
		set_vessel_artifact_name_effect = yes
	}
	if = {
		limit = { 
			OR = {
				has_variable = temporary_vessel
				var:stored_sekhem <= 0 
			}
			NOT = { has_variable = permanent_vessel }
		}
		$OWNER$ = {
			send_interface_toast = {
				right_icon = prev
				title = gnosis_vessel_damaged
				prev = { degrade_vessel_durability_effect = yes }
			}
		}
	}
}

POD_gnosis_vessel_increase_multiple_effect = {
	if = {
		limit = { vessel_difference >= $COUNT$ }
		custom_description = {
			text = POD_store_gnosis
			subject = $OWNER$
			value = $COUNT$
			change_variable = { name = stored_sekhem add = $COUNT$ }
		}
	}
	else = {
		custom_description = {
			text = POD_store_gnosis
			subject = $OWNER$
			value = vessel_difference
			change_variable = { name = stored_sekhem add = vessel_difference }
		}
	}
	set_vessel_artifact_name_effect = yes
}

POD_gift_cost_effect = {
	if = {
		limit = {
			any_character_artifact = {
				has_variable = stored_sekhem
				var:stored_sekhem > 0
			}
		}
		ordered_character_artifact = {
			limit = {
				has_variable = stored_sekhem 
				var:stored_sekhem > 0
			}
			order_by = {
				value = 0
				add = var:stored_sekhem
			}
			position = 0
			POD_gnosis_vessel_decrease_multiple_effect = { COUNT = $COUNT$ OWNER = prev }
		}
	}
	else = {
		POD_gnosis_decrease_multiple_effect = { COUNT = $COUNT$ }
	}
}

POD_gnosis_income_effect = {
	if = {
		limit = {
			OR = {
				gnosis_difference > $COUNT$
				NOT = {
					any_character_artifact = {
						NOT = { has_variable = temporary_vessel }
						has_variable = stored_sekhem
						vessel_difference > 0
					}
				}
			}
		}
		POD_gnosis_increase_multiple_effect = { COUNT = $COUNT$ }
	}
	else = {
		save_scope_value_as = {
			name = excess_gnosis
			value = {
				value = $COUNT$
				subtract = gnosis_difference
			}
		}
		POD_gnosis_increase_multiple_effect = { COUNT = gnosis_difference }
		ordered_character_artifact = {
			limit = {
				NOT = { has_variable = temporary_vessel }
				has_variable = stored_sekhem 
				vessel_difference > 0
			}
			order_by = {
				value = 0
				add = vessel_difference
			}
			position = 0
			POD_gnosis_vessel_increase_multiple_effect = { COUNT = scope:excess_gnosis OWNER = prev }
		}
	}
}
