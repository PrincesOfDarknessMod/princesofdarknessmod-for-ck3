
POD_automaton_initiate_ingredient_effect = {
	POD_ingredient_clear_effect = yes
	add_to_variable_list = { name = ingredient_list target = flag:water }
	add_to_variable_list = { name = ingredient_list target = flag:fuel }
	add_to_variable_list = { name = ingredient_list target = flag:mineral }
	switch = {
		trigger = $RECIPE$
		flag:simple_servitor = {
			add_to_variable_list = { name = ingredient_list target = flag:sekhem }
		}
		flag:overseer = {
			add_to_variable_list = { name = ingredient_list target = flag:sekhem }
		}
		flag:guard = {
			add_to_variable_list = { name = ingredient_list target = flag:sekhem }
		}
	}
	if = {
		limit = { is_target_in_variable_list = { name = ingredient_list target = flag:water } }
		add_to_variable_list = { name = alt_ingredient_list target = flag:exotic_water }
		add_to_variable_list = { name = alt_ingredient_list target = flag:magic_water }
	}
	if = {
		limit = { is_target_in_variable_list = { name = ingredient_list target = flag:fuel } }
		add_to_variable_list = { name = alt_ingredient_list target = flag:exotic_fuel }
		add_to_variable_list = { name = alt_ingredient_list target = flag:magic_fuel }
	}
	if = {
		limit = { is_target_in_variable_list = { name = ingredient_list target = flag:mineral } }
		add_to_variable_list = { name = alt_ingredient_list target = flag:exotic_mineral }
		add_to_variable_list = { name = alt_ingredient_list target = flag:magic_mineral }
	}
}

POD_automaton_create_effect = {
	remove_short_term_gold = ingredient_total_cost
	POD_cost_sekhem_effect = yes
	POD_cost_material_effect = { ALCHEMY = exotic_water }
	POD_cost_material_effect = { ALCHEMY = exotic_fuel }
	POD_cost_material_effect = { ALCHEMY = exotic_mineral }
	POD_cost_material_effect = { ALCHEMY = magic_water }
	POD_cost_material_effect = { ALCHEMY = magic_fuel }
	POD_cost_material_effect = { ALCHEMY = magic_mineral }
	random = {
		chance = automaton_success_chance
		POD_automaton_create_character_effect = yes
	}
	if = {
		limit = { has_perk = create_object_perk }
		random = {
			chance = 5
			add_gold = ingredient_total_cost
		}
	}
}

POD_automaton_create_character_effect = { 
	root = { save_scope_as = owner }
	var:craft_creator = { save_scope_as = creator }
	
	switch = {
		trigger = var:current_recipe
		flag:simple_servitor = {
			create_effigy_character_effect = { TYPE = servitor }
			scope:creator = { add_hekau_effigy_lifestyle_xp = miniscule_lifestyle_xp }
		}
		flag:overseer = {
			create_effigy_character_effect = { TYPE = overseer }
			scope:creator = { add_hekau_effigy_lifestyle_xp = medium_lifestyle_xp }
		}
		flag:guard = {
			create_effigy_character_effect = { TYPE = guard }
			scope:creator = { add_hekau_effigy_lifestyle_xp = major_lifestyle_xp }
		}
	}
}

get_effigy_character_quality_effect = {
	save_scope_value_as = {
		name = quality
		value = {
			value = automaton_success_chance
			min = 1
		}
	}
}

create_effigy_character_effect = {
	get_effigy_character_quality_effect = yes
	get_artifact_wealth_effect = yes
	
	if = {
		limit = { rarity_to_create >= 180 }
		generate_effigy_character_effect = { RARITY = illustrious TYPE = $TYPE$ }
	}
	else_if = {
		limit = { rarity_to_create >= 130 }
		generate_effigy_character_effect = { RARITY = famed TYPE = $TYPE$ }
	}
	else_if = {
		limit = { rarity_to_create >= 50 }
		generate_effigy_character_effect = { RARITY = masterwork TYPE = $TYPE$ }
	}
	else = {
		generate_effigy_character_effect = { RARITY = common TYPE = $TYPE$ }
	}
}
	
generate_effigy_character_effect = {
	create_character = {
		location = root.capital_province
		template = POD_$RARITY$_$TYPE$_effigy_character
		save_scope_as = effigy_character
	}
	hidden_effect_new_object = {
		add_hook = {
			target = scope:effigy_character
			type = effigy_hook
		}
		scope:effigy_character = { 
			set_relation_effigy_creator = root
			add_character_flag = had_POD_character_initialization
			set_employer = root
			return_to_court = yes
			add_character_flag = blocked_from_leaving
			add_character_flag = immune_to_disease
			set_sexuality = asexual
		}
		trigger_event = podmummy.3100
	}
}

POD_change_automaton_effect = {

	POD_automaton_initiate_ingredient_effect = { RECIPE = scope:recipe }
