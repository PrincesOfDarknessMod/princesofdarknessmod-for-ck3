# arguments: TRAIT
POD_init_preferred_victim_trait = {
	if = {
		limit = {
			$TRAIT$ = { save_temporary_scope_as = new_trait }
			any_in_list = {
				count = 0
				variable = preferred_victim_traits
				this = scope:new_trait
			}
		}
		hidden_effect_new_object = {
			add_to_variable_list = {
				name   = event_preferred_victim_trait_choices
				target = $TRAIT$
			}
		}
	}
}

POD_add_preferred_victim_trait_from_selection = {
	POD_add_preferred_victim_trait_to_list = {
		TRAIT = $TRAIT$
		LIST  = preferred_victim_traits_to_add
	}
	
	every_in_list = {
		list = preferred_victim_traits_to_add
		prev = {
			POD_add_preferred_victim_trait = { TRAIT = prev }
		}
	}
	
	POD_clear_list = { LIST = preferred_victim_traits_to_add }
	
	# increment this each time you choose, not for each trait
	# because some choices add multiple traits
	set_variable = {
		name = preferred_victim_traits_chosen_var
		value = {
			value = preferred_victim_traits_chosen
			add = 1
		}
	}
}

POD_add_preferred_victim_trait = {
	custom_description = {
		text = POD_add_preferred_victim_trait_tt
		object = $TRAIT$
		add_to_variable_list = {
			name = preferred_victim_traits
			target = $TRAIT$
		}
	}
	if = {
		limit = {
			$TRAIT$ = trait:vampire
			POD_is_vampire_trigger = yes
		}
		custom_tooltip = POD_add_preferred_victim_trait_warning_vampire
	}
	else_if = {
		limit = {
			$TRAIT$ = { has_trait_flag = vamp_clan }
			has_trait = $TRAIT$
		}
		custom_tooltip = POD_add_preferred_victim_trait_warning_vampire
	}
	else_if = {
		limit = {
			$TRAIT$ = trait:ghoul
			POD_splat_can_create_ghouls_naturally = yes
		}
		custom_tooltip = POD_add_preferred_victim_trait_warning_ghoul
	}
	else_if = {
		limit = {
			$TRAIT$ = trait:revenant
			POD_splat_can_create_ghouls_naturally = yes
			has_perk = heavens_gate_perk
		}
		custom_tooltip = POD_add_preferred_victim_trait_warning_revenant
	}
	else_if = {
		limit = {
			$TRAIT$ = trait:gargoyle
			OR = {
				POD_can_create_gargoyle_trigger = yes
				is_tremere_clan_or_bloodline_trigger = yes
			}
		}
		custom_tooltip = POD_add_preferred_victim_trait_warning_gargoyle
	}
	else_if = {
		limit = {
			$TRAIT$ = trait:supehunter
			POD_is_supehunter_trigger = yes
		}
		custom_tooltip = POD_add_preferred_victim_trait_warning_supehunter
	}
	else_if = {
		limit = {
			$TRAIT$ = trait:witch
			is_witch_trigger = yes
		}
		custom_tooltip = POD_add_preferred_victim_trait_warning_infernalist
	}
	# TODO MAGE: mage/sorcerer creation
}

POD_init_stalk_option = {
	add_to_variable_list = {
		name = stalk_options
		target = flag:$OPTION$
	}
	set_variable = {
		name = $OPTION$
		value = 0
	}
	POD_error_suppression_unused_flag = {
		FLAG = $OPTION$
	}
}

POD_cleanup_stalk_option = {
	remove_variable = $OPTION$
}

	POD_stalk_option_injector = {
		EFFECT = POD_init_stalk_option
	}

POD_stalk_torture_event_after = {
	remove_variable = stalk_actions_var
	clear_variable_list = stalk_options
	POD_stalk_option_injector = {
		EFFECT = POD_cleanup_stalk_option
	}
	
	if = {
		limit = {
			scope:recipient = {
				NOT = { has_character_modifier = recently_tortured }
				NOT = { has_character_flag = is_being_tortured }
			}
		}
		if = {
			limit = { NOT = { exists = scope:actor } }
			save_scope_as = actor
		}
		save_scope_value_as = {
			name  = torture_method
			value = flag:whip
		}
		POD_torture_interaction_no_actor_effect = yes
	}
}

			value = {
				value = POD_physique_level
				add = $COUNT$

			}
		}
		if = {
			limit = { scope:new_physique_level != POD_physique_level }
			remove_trait = physique_good_2
			remove_trait = physique_good_1
			remove_trait = physique_bad_1
			remove_trait = physique_bad_2
			remove_trait = physique_bad_3
			POD_get_physique_trait_by_level = {
				LEVEL   = scope:new_physique_level
				SAVE_AS = new_physique_trait
			}
			if = {
				limit = { exists = scope:new_physique_trait }
				add_trait_force_tooltip = scope:new_physique_trait
			}
		}
	}
}

	if = {
		limit = { $COUNT$ > 0 }
		scope:recipient = {
			add_character_modifier = {
				modifier = POD_dollmaker_modifier
				years = POD_stalk_dollmaker_duration_years
			}
		}
		if = {
			limit = { NOT = { scope:recipient = { is_imprisoned_by = prev } } }
			scope:recipient = {
				add_character_flag = { flag = lost_duel days = 1 }
			}
			rightfully_imprison_character_less_verbose_effect = {
				TARGET = scope:recipient
				IMPRISONER = this
			}
		}
	}
}
