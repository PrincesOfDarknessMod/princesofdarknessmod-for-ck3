# arguments: LOVER_1, LOVER_2
POD_possessed_on_set_relation_lover = {
	$LOVER_1$ = { save_scope_as = lover_1 }
	$LOVER_2$ = { save_scope_as = lover_2 }
	scope:lover_1 = {
		if = {
			limit = { has_perk = possessed_enticer_perk }
			POD_possessed_gain_true_relief = {
				VALUE = POD_true_relief_enticer
			}
		}
		if = {
			limit = {
				has_perk = possessed_mother_of_monsters_perk
				scope:lover_2 = { has_trait = fomori }
			}
			add_hook = {
				type = eternal_loyalty_hook
				target = scope:lover_2
			}
		}
	}
}
