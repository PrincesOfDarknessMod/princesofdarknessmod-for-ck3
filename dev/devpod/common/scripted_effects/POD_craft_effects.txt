POD_replace_ingredient_effect = {
	if = {
		limit = { is_target_in_variable_list = { name = ingredient_list target = flag:$ALCHEMY$ } }
		remove_list_variable = { name = ingredient_list target = flag:$ALCHEMY$ }
		add_to_variable_list = { name = alt_ingredient_list target = flag:$ALCHEMY$ }
	}
}

POD_cost_elixir_effect = {
	if = {
		limit = { is_target_in_variable_list = { name = ingredient_list target = flag:$ALCHEMY$ } }
		while = {
			count = $ALCHEMY$_amount
			random_character_artifact = {
				limit = { 
					var:alchemy ?= {
						OR = {
							this = flag:$ALCHEMY$_diplomacy
							this = flag:$ALCHEMY$_martial
							this = flag:$ALCHEMY$_stewardship
							this = flag:$ALCHEMY$_intrigue
							this = flag:$ALCHEMY$_learning
							this = flag:$ALCHEMY$_prowess
						}
					}
				}
				destroy_artifact = this
			}
		}
	}
}

POD_cost_alchemy_effect = {
	if = {
		limit = { is_target_in_variable_list = { name = ingredient_list target = flag:$ALCHEMY$ } }
		while = {
			count = $ALCHEMY$_amount
			random_character_artifact = {
				limit = { var:alchemy ?= flag:$ALCHEMY$ }
				destroy_artifact = this
			}
		}
	}
}

POD_cost_material_effect = {
	if = {
		limit = { is_target_in_variable_list = { name = ingredient_list target = flag:$ALCHEMY$ } }
		custom_tooltip = {
			text = lose_$ALCHEMY$
			change_variable = {
				name = $ALCHEMY$_reserve
				subtract = $ALCHEMY$_amount
			}
		}
	}
}

POD_cost_sekhem_effect = {
	if = {
		limit = { is_target_in_variable_list = { name = ingredient_list target = flag:sekhem } }
		if = {
			limit = { is_east_mummy_trigger = yes }
			POD_yangchi_ongoing_cost_effect = { COUNT = sekhem_amount }
		}
		else = {
			POD_sekhem_ongoing_cost_effect = { COUNT = sekhem_amount }
		}
	}
}

POD_cost_vampire_blood_effect = {
	if = {
		limit = { has_vampire_blood_ingredient_trigger = yes }
		var:character_material ?= { POD_hunger_increase_multiple_effect = { COUNT = vampire_blood_amount } }
	}
}

POD_cost_counterfeit_effect = {
	if = {
		limit = { is_target_in_variable_list = { name = ingredient_list target = flag:counterfeit_$ALCHEMY$ } }
		while = {
			count = counterfeit_$ALCHEMY$_amount
			random_character_artifact = {
				limit = { 
					var:alchemy ?= {
						OR = {
							this = flag:animalism_counterfeit_$ALCHEMY$
							this = flag:auspex_counterfeit_$ALCHEMY$

							this = flag:dominate_counterfeit_$ALCHEMY$

}


POD_craft_monthly_tick = {
	if = {
		limit = { exotic_water_reserve < exotic_water_max_reserve }
		set_variable = {
			name = exotic_water_reserve
			value = exotic_water_max_reserve
		}
	}
	if = {
		limit = { exotic_fuel_reserve < exotic_fuel_max_reserve }
		set_variable = {
			name = exotic_fuel_reserve
			value = exotic_fuel_max_reserve
		}
	}
	if = {
		limit = { exotic_mineral_reserve < exotic_mineral_max_reserve }
		set_variable = {
			name = exotic_mineral_reserve
			value = exotic_mineral_max_reserve
		}
	}
	if = {
		limit = { exotic_fuel_reserve < exotic_herb_max_reserve }
		set_variable = {
			name = exotic_herb_reserve
			value = exotic_herb_max_reserve
		}
	}
	if = {
		limit = { magic_water_reserve < magic_water_max_reserve }
		if = {
			limit = { has_variable = magic_water_reserve }
			change_variable = {
				name = magic_water_reserve
				add = magic_water_income
			}
		}
		else = {
			set_variable = {
				name = magic_water_reserve
				value = magic_water_income
			}
		}
	}
	if = {
		limit = { magic_fuel_reserve < magic_fuel_max_reserve }
		if = {
			limit = { has_variable = magic_fuel_reserve }
			change_variable = {
				name = magic_fuel_reserve
				add = magic_fuel_income
			}
		}
		else = {
			set_variable = {
				name = magic_fuel_reserve
				value = magic_fuel_income
			}
		}
	}
	if = {
		limit = { magic_herb_reserve < magic_herb_max_reserve }
		if = {
			limit = { has_variable = magic_herb_reserve }
			change_variable = {
				name = magic_herb_reserve
				add = magic_herb_income
			}
		}
		else = {
			set_variable = {
				name = magic_herb_reserve
				value = magic_herb_income
			}
		}
	}
	if = {
		limit = { magic_mineral_reserve < magic_mineral_max_reserve }
		if = {
			limit = { has_variable = magic_mineral_reserve }
			change_variable = {
				name = magic_mineral_reserve
				add = magic_mineral_income
			}
		}
		else = {
			set_variable = {
				name = magic_mineral_reserve
				value = magic_mineral_income
			}
		}
	}
}
