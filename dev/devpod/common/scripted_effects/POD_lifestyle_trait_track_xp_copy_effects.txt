POD_copy_lifestyle_trait_xp_single_track = {
	trait:$TRAIT$ = { save_scope_as = trait_to_copy }
	if = {
		limit = {
			has_trait = $TRAIT$
			scope:cloned_character = { has_trait = $TRAIT$ }
		}
		add_trait_xp = { 
			trait = $TRAIT$
			value = "scope:cloned_character.has_trait_xp(scope:trait_to_copy)"
		}
	}
}
