# Umbra Modifier
POD_add_umbra_modifier_effect = {
	if = {
		limit = { in_astral_reaches_trigger = { LOCATION = $LOCATION$ } }
		add_character_modifier = traveling_astral_reaches_modifier
	}
	else_if = {
		limit = { in_spirit_wilds_trigger = { LOCATION = $LOCATION$ } }
		add_character_modifier = traveling_spirit_wilds_modifier
	}
	else_if = {
		limit = { in_underworld_trigger = { LOCATION = $LOCATION$ } }
		add_character_modifier = traveling_underworld_modifier
	}
	else_if = {
		limit = { in_dreaming_trigger = { LOCATION = $LOCATION$ } }
		add_character_modifier = traveling_dreaming_modifier
	}
	else_if = {
		limit = { in_deep_umbra_trigger = { LOCATION = $LOCATION$ } }
		add_character_modifier = traveling_deep_umbra_modifier
	}
}

POD_change_umbra_modifier_effect = {
	if = {
		limit = { in_astral_reaches_trigger = { LOCATION = $START$ } }
		if = {
			limit = { NOT = { in_astral_reaches_trigger = { LOCATION = $END$ } } }
			remove_character_modifier = traveling_astral_reaches_modifier
			POD_add_umbra_modifier_effect = { LOCATION = $END$ }
		}
	}
	else_if = {
		limit = { in_spirit_wilds_trigger = { LOCATION = $START$ } }
		if = {
			limit = { NOT = { in_spirit_wilds_trigger = { LOCATION = $END$ } } }
			remove_character_modifier = traveling_spirit_wilds_modifier
			POD_add_umbra_modifier_effect = { LOCATION = $END$ }
		}
	}
	else_if = {
		limit = { in_underworld_trigger = { LOCATION = $START$ } }
		if = {
			limit = { NOT = { in_underworld_trigger = { LOCATION = $END$ } } }
			remove_character_modifier = traveling_underworld_modifier
			POD_add_umbra_modifier_effect = { LOCATION = $END$ }
		}
	}
	else_if = {
		limit = { in_dreaming_trigger = { LOCATION = $START$ } }
		if = {
			limit = { NOT = { in_dreaming_trigger = { LOCATION = $END$ } } }
			remove_character_modifier = traveling_dreaming_modifier
			POD_add_umbra_modifier_effect = { LOCATION = $END$ }
		}
	}
	else_if = {
		limit = { in_deep_umbra_trigger = { LOCATION = $START$ } }
		if = {
			limit = { NOT = { in_deep_umbra_trigger = { LOCATION = $END$ } } }
			remove_character_modifier = traveling_deep_umbra_modifier
			POD_add_umbra_modifier_effect = { LOCATION = $END$ }
		}
	}
	else_if = {
		limit = { in_penumbra_trigger = { LOCATION = $START$ } }
		if = {
			limit = { NOT = { in_penumbra_trigger = { LOCATION = $END$ } } }
			pay_for_spirit_travel_effect = { END = $END$ }
			POD_add_umbra_modifier_effect = { LOCATION = $END$ }
			if = {
				limit = { 
					has_POD_shapeshifted_form_trigger = yes 
					NOT = { has_perk = aspect_of_duat_perk }
				}
				POD_shapeshift_revert_effect = yes
			}
			if = {
				limit = { has_character_modifier = reversa_of_fortune_perk_modifier }
				remove_character_modifier = reversa_of_fortune_perk_modifier
			}
		}
	}
	else_if = {
		limit = { in_zone_trigger = { LOCATION = $START$ } }
	}
	else = {
		POD_add_umbra_modifier_effect = { LOCATION = $END$ }
	}
}
