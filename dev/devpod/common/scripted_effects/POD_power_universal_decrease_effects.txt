POD_quinary_change_power_cost_change_effect = {
	if = {
		limit = {
			OR = {
				$COUNT$ = 1
				$COUNT$ = 2
			}
		}
		POD_$TYPE$_effect = { COUNT = 1 }
	}
	else_if = {
		limit = {
			OR = {
				$COUNT$ = 3
				$COUNT$ = 4
			}
		}
		POD_$TYPE$_effect = { COUNT = 2 }
	}
	else_if = {
		limit = {
			OR = {
				$COUNT$ = 5
				$COUNT$ = 6
			}
		}
		POD_$TYPE$_effect = { COUNT = 3 }
	}
	else_if = {
		limit = {
			OR = {
				$COUNT$ = 7
				$COUNT$ = 8
			}
		}
		POD_$TYPE$_effect = { COUNT = 4 }
	}
	else_if = {
		limit = {
			OR = {
				$COUNT$ = 9
				$COUNT$ = 10
			}
		}
		POD_$TYPE$_effect = { COUNT = 5 }
	}
}

POD_hunger_increase_with_bloodloss_multiple_effect = {	
	if = {
		limit = { POD_has_infinite_blood_trigger = no }
		
		save_scope_value_as = {
			name = hunger_increase
			value = {
				value = $COUNT$
				max = blood_level
			}
		}
		
		save_scope_value_as = {
			name = bloodloss_amount
			value = {
				value = blood_level
				subtract = $COUNT$
				max = 0
				multiply = -1
			}
		}
		
		if = {
			limit = { scope:bloodloss_amount > 0 }
			POD_power_cost_bloodloss_effect = { COUNT = scope:bloodloss_amount }
		}
		if = {
			limit = { scope:hunger_increase > 0 }
			POD_discipline_cost_effect = { COUNT = scope:hunger_increase }
		}
	}
}
