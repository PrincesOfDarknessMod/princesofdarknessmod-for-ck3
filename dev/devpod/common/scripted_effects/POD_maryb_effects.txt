######################################
# Story cycle sync
######################################
#
# Pushes Mary's current character variables (mary_michael_balance,
# mary_tablets_recovered) into the matching story-scope variables on her
# POD_maryb_journey story. Called from every event that mutates those
# character variables (podintroductions.380, tablet events, deal outcomes).
# Idempotent — safe to call when no story exists yet.
#
POD_maryb_sync_story_effect = {
	every_owned_story = {
		limit = { story_type = POD_maryb_journey }
		if = {
			limit = { prev = { has_variable = mary_michael_balance } }
			set_variable = {
				name = michael_balance
				value = prev.var:mary_michael_balance
			}
		}
		if = {
			limit = { prev = { has_variable = mary_tablets_recovered } }
			set_variable = {
				name = tablets_recovered
				value = prev.var:mary_tablets_recovered
			}
		}
	}
}

# Creates Mary's journey story if it doesn't already exist. Called once from
# podintroductions.380's immediate block; subsequent calls are no-ops.
POD_maryb_create_journey_story_effect = {
	if = {
		limit = {
			NOT = {
				any_owned_story = { story_type = POD_maryb_journey }
			}
		}
		create_story = POD_maryb_journey
	}
}


######################################
# Ash and Gold — ascendancy reroll
######################################
#
# Called once on resolution (POD_maryb.230) and every few years thereafter
# from POD_maryb.50's yearly pulse. Removes whichever ascendancy modifier is
# currently present and randomly applies one of the two. If the side actually
# flips (and the character is player-controlled), a toast surfaces so the
# player feels the tide turn rather than discovering it by accident.
#
# The synthesis horror: the player does not choose which half is uppermost,
# and cannot keep one half down for long. The body decides.
#
POD_maryb_ash_gold_reroll_effect = {
	# Remember who was up before the roll, so we only notify on a real flip.
	if = {
		limit = { has_character_modifier = mary_ash_and_gold_michael_modifier }
		set_variable = { name = mary_ash_gold_prev value = flag:michael }
	}
	else_if = {
		limit = { has_character_modifier = mary_ash_and_gold_mary_modifier }
		set_variable = { name = mary_ash_gold_prev value = flag:mary }
	}
	else = {
		set_variable = { name = mary_ash_gold_prev value = flag:none }
	}

	# Clear both (only one is present; removing both is safe and idempotent).
	if = {
		limit = { has_character_modifier = mary_ash_and_gold_michael_modifier }
		remove_character_modifier = mary_ash_and_gold_michael_modifier
	}
	if = {
		limit = { has_character_modifier = mary_ash_and_gold_mary_modifier }
		remove_character_modifier = mary_ash_and_gold_mary_modifier
	}

	random_list = {
		50 = {
			add_character_modifier = mary_ash_and_gold_michael_modifier
			if = {
				limit = {
					is_ai = no
					NOT = { var:mary_ash_gold_prev = flag:michael }
				}
				send_interface_toast = {
					type = event_generic_neutral
					title = mary_ash_gold_michael_surfaces.toast
					left_icon = root
				}
			}
		}
		50 = {
			add_character_modifier = mary_ash_and_gold_mary_modifier
			if = {
				limit = {
					is_ai = no
					NOT = { var:mary_ash_gold_prev = flag:mary }
				}
				send_interface_toast = {
					type = event_generic_neutral
					title = mary_ash_gold_mary_surfaces.toast
					left_icon = root
				}
			}
		}
	}

	remove_variable = mary_ash_gold_prev
}


######################################
# Starting pact creation (inline)
######################################
#
# Creates a single demonic pact on the calling character, mirroring PoD's
# POD_story_pact shape exactly so the panel display, the yearly tick effect,
# and all other PoD pact mechanics work without modification.
#
# Why inline instead of delegating to PoD's POD_create_pact_from_stipulation_stories:
#
#   PoD's pact creator computes `perks_gained` via a script_value
#   (`save_scope_value_as = { name = perks_gained value = { add = $OBLIGATION_STORY$.var:level ... } }`)
#   that the engine evaluates during tooltip-build. Looking up a stipulation
#   story by `var:id = flag:X` with random_in_list (or any iteration) is
#   non-deterministic from the tooltip-builder's perspective, so the scope
#   is unset when the script_value tries to read it, producing
#   "Event target link 'scope' returned an unset scope" errors. PoD's own
#   GUI bypasses this by passing `var:selected_X` (a direct variable read),
#   which the tooltip-builder can resolve.
#
#   We could mimic the GUI pattern (write the stipulation story refs to
#   temporary character variables, then pass var:X), but the resulting
#   complexity exceeds the cost of inlining — the surface area we depend on
#   is the pact-story variable shape (`obligation`, `frequency`, `consequence`,
#   `demon`, `perk_points_gained`, `completed`), which is the story_cycle
#   type definition itself and is stable PoD call.
#
# Parameters:
#   DEMON        : character scope (e.g. character:9201500)
#   OBLIGATION   : flag literal (e.g. flag:sacrifice_of_blood) — stored
#                  directly as the variable value, mirroring how
#                  POD_pact_gui_init_stipulation stores `id`. The pact
#                  display reads it back with `.GetFlagKey`.
#   FREQUENCY    : numeric (years) — the pact's effect_group reads
#                  `var:frequency` as a number, not a flag.
#   CONSEQUENCE  : flag literal (e.g. flag:health) — same form as OBLIGATION.
#   PERKS_GAINED : numeric. Per PoD's formula in
#                  POD_create_pact_from_stipulation_stories:
#                    obligation_level + consequence_level + frequency_level + POD_baali_pit_bonus
#                  For Mary's starting pacts at fresh game start (no pit, bonus=0):
#                    Sacrifice of Blood (L2) / 8_years (L-2) / Health    (L2) = 2
#                    Tribute of Agony   (L2) / 8_years (L-2) / Knowledge (L3) = 3
#
POD_maryb_create_starting_pact = {
	# Mary needs investiture lifestyle access to actually use the perk points
	# she'll receive. PoD grants this flag inside diabolism flows we're not
	# touching, so we set it explicitly the first time.
	if = {
		limit = { NOT = { has_character_flag = access_to_investiture } }
		add_character_flag = access_to_investiture
	}

	$DEMON$ = { save_scope_as = demon }
	add_investiture_lifestyle_perk_points = $PERKS_GAINED$

	create_story = {
		type = POD_story_pact
		save_scope_as = pact
	}
	hidden_effect = {
		scope:pact ?= {
			set_variable = { name = obligation         value = $OBLIGATION$ }
			set_variable = { name = frequency          value = $FREQUENCY$ }
			set_variable = { name = consequence        value = $CONSEQUENCE$ }
			set_variable = { name = perk_points_gained value = $PERKS_GAINED$ }
			set_variable = { name = demon              value = scope:demon }
			set_variable = { name = completed          value = no }
		}
		add_to_variable_list = {
			name   = demonic_pacts
			target = scope:pact
		}
	}
}
