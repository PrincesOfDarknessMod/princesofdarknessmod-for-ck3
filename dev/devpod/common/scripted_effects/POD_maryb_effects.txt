######################################
# Story cycle sync
######################################
#
# Pushes Mary's current character variables (mary_michael_balance,
# mary_tablets_recovered) into the matching story-scope variables on her
# POD_maryb_story story. Called from every event that mutates those
# character variables (podintroductions.380, tablet events, deal outcomes).
#
POD_maryb_sync_story_effect = {
	every_owned_story = {
		limit = { story_type = POD_maryb_story }
		# Only mirror while the arc is unresolved. After a path is chosen the
		# resolution / oscillation effects own the counters directly (and retire
		# some of them), so the generic mirror must not re-create or clobber them.
		if = {
			limit = { prev = { POD_maryb_resolved_trigger = no } }
			if = {
				limit = { prev = { has_variable = mary_michael_balance } }
				set_variable = {
					name = michael_balance
					value = prev.var:mary_michael_balance
				}
			}
			if = {
				limit = { prev = { has_variable = mary_tablets_recovered } }
				set_variable = {
					name = tablets_recovered
					value = prev.var:mary_tablets_recovered
				}
			}
		}
	}
}

######################################
# Story cycle — retire / repurpose counters on resolution
######################################
#
# The journey panel's two counter rows (tug-of-war michael_balance, basic
# tablets_recovered) render only while their backing story variable exists
# (window_situation_list.gui gates each row on HasFlag(variable_name)). So
# removing the story variable cleanly hides that counter once its part of the
# arc is finished. One counter per call so the Ash and Gold path can retire the
# tablet counter while keeping the balance bar (repurposed below).
#
POD_maryb_hide_story_counter_effect = {
	every_owned_story = {
		limit = { story_type = POD_maryb_story }
		if = {
			limit = { has_variable = $COUNTER$ }
			remove_variable = $COUNTER$
		}
	}
}

# Ash and Gold: drive the surviving tug-of-war bar from the current ascendancy
# rather than the (now retired) balance. Full swing to whichever half looks out
# (+100 Mary / -100 Michael), so the bar visibly lurches side to side as the
# synthesis turns. Called from POD_maryb_ash_gold_reroll_effect, so every reroll
# updates it. Exactly one ascendancy modifier is present when this runs.
POD_maryb_sync_ascendancy_counter_effect = {
	every_owned_story = {
		limit = { story_type = POD_maryb_story }
		if = {
			limit = { prev = { has_character_modifier = mary_ash_and_gold_mary_modifier } }
			set_variable = { name = michael_balance value = 100 }
		}
		else_if = {
			limit = { prev = { has_character_modifier = mary_ash_and_gold_michael_modifier } }
			set_variable = { name = michael_balance value = -100 }
		}
	}
}

# Creates Mary's journey story if it doesn't already exist. Called once from
# podintroductions.380's immediate block; subsequent calls are no-ops.
POD_maryb_create_journey_story_effect = {
	if = {
		limit = {
			NOT = {
				any_owned_story = { story_type = POD_maryb_story }
			}
		}
		create_story = POD_maryb_story
	}
}


######################################
# Ash and Gold — ascendancy reroll
######################################
#
# Called once on resolution (POD_maryb.230) and every few years thereafter
# from POD_maryb.50's yearly pulse. Removes whichever ascendancy modifier is
# currently present and randomly applies one of the two. If the side actually
# flips (and the character is player-controlled), a toast surfaces so the
# player feels the tide turn rather than discovering it by accident.
#
POD_maryb_ash_gold_reroll_effect = {
	# Wrapped in hidden_effect: this is a pure mechanical swap (and a random
	# roll), so it should never render in a tooltip. It also reads
	# var:mary_ash_gold_prev, set earlier in this same effect, inside the
	# random_list branches; the tooltip-builder does NOT apply that earlier
	# set_variable, so without hidden_effect it errors on the var read when an
	# option that calls this effect (e.g. POD_maryb.230) builds its tooltip.
	hidden_effect = {
		# Remember who was up before the roll, so we only notify on a real flip.
		if = {
			limit = { has_character_modifier = mary_ash_and_gold_michael_modifier }
			set_variable = { name = mary_ash_gold_prev value = flag:michael }
		}
		else_if = {
			limit = { has_character_modifier = mary_ash_and_gold_mary_modifier }
			set_variable = { name = mary_ash_gold_prev value = flag:mary }
		}
		else = {
			set_variable = { name = mary_ash_gold_prev value = flag:none }
		}

		# Clear both (only one is present; removing both is safe and idempotent).
		if = {
			limit = { has_character_modifier = mary_ash_and_gold_michael_modifier }
			remove_character_modifier = mary_ash_and_gold_michael_modifier
		}
		if = {
			limit = { has_character_modifier = mary_ash_and_gold_mary_modifier }
			remove_character_modifier = mary_ash_and_gold_mary_modifier
		}

		random_list = {
			50 = {
				add_character_modifier = mary_ash_and_gold_michael_modifier
				if = {
					limit = {
						is_ai = no
						exists = var:mary_ash_gold_prev
						NOT = { var:mary_ash_gold_prev = flag:michael }
					}
					send_interface_toast = {
						type = event_generic_neutral
						title = mary_ash_gold_michael_surfaces.toast
						left_icon = root
					}
				}
			}
			50 = {
				add_character_modifier = mary_ash_and_gold_mary_modifier
				if = {
					limit = {
						is_ai = no
						exists = var:mary_ash_gold_prev
						NOT = { var:mary_ash_gold_prev = flag:mary }
					}
					send_interface_toast = {
						type = event_generic_neutral
						title = mary_ash_gold_mary_surfaces.toast
						left_icon = root
					}
				}
			}
		}

		remove_variable = mary_ash_gold_prev

		# Swing the journey panel's tug-of-war bar to the new ascendant half.
		POD_maryb_sync_ascendancy_counter_effect = yes
	}
}


######################################
# Well of Sacrifice — found the Well
######################################
#
# Called from POD_maryb_found_well_decision. Places tier 1 of the Well building
# in Constantinople's holding (title:b_constantinople.title_province) via
# add_building, which bypasses the building's can_construct — tier 1 is never
# offered in the build UI, so this decision is the sole way to found the Well.
#
POD_maryb_found_well_effect = {
	if = {
		limit = { exists = title:b_constantinople }
		title:b_constantinople.title_province = { add_building = maryb_well_of_sacrifice_01 }
	}
}


######################################
# Well of Sacrifice — sacrifice development (the offering gate)
######################################
#
# Called from POD_maryb_well_sacrifice_decision (which carries the cooldown).
# Pours a fixed slice of Constantinople's own development down the shaft, then
# grants the OFFERING modifier for the next unbuilt Well tier on c_byzantion
# (mary_well_offering_t2/t3/t4). The Well building's can_construct reads that
# modifier, so the offering is the key that lets the player raise the next tier
# in the build UI. One sacrifice unlocks one tier.
#
POD_maryb_well_sacrifice_effect = {
	# Pour a slice of the city's development down the shaft.
	if = {
		limit = { exists = title:c_byzantion }
		title:c_byzantion = { change_development_level = -3 }
	}
	# Grant the offering for the next unbuilt tier, read from what is built. The
	# decision blocks this rite while any offering is still pending, and each
	# offering is consumed when its tier is built (POD_maryb_well_offering_consume
	# in POD_maryb_on_actions.txt), so exactly one offering is ever live at a time.
	if = {
		limit = { exists = title:b_constantinople }
		if = {
			# Tier 3 built, tier 4 not — offer for the open gateway.
			limit = { title:b_constantinople.title_province = { has_building_or_higher = maryb_well_of_sacrifice_03 } }
			title:c_byzantion = { add_county_modifier = { modifier = mary_well_offering_t4 } }
		}
		else_if = {
			# Tier 2 built — offer for tier 3.
			limit = { title:b_constantinople.title_province = { has_building_or_higher = maryb_well_of_sacrifice_02 } }
			title:c_byzantion = { add_county_modifier = { modifier = mary_well_offering_t3 } }
		}
		else = {
			# Tier 1 built (guaranteed by the decision's is_shown) — offer for tier 2.
			title:c_byzantion = { add_county_modifier = { modifier = mary_well_offering_t2 } }
		}
	}
}


######################################
# Starting pact creation (inline)
######################################
#
# Creates a single demonic pact on the calling character, mirroring PoD's
# POD_story_pact shape exactly so the panel display, the yearly tick effect,
# and all other PoD pact mechanics work without modification.
#
# Parameters:
#   DEMON        : character scope (e.g. character:9201500)
#   OBLIGATION   : flag literal (e.g. flag:sacrifice_of_blood) — stored
#                  directly as the variable value, mirroring how
#                  POD_pact_gui_init_stipulation stores `id`. The pact
#                  display reads it back with `.GetFlagKey`.
#   FREQUENCY    : numeric (years) — the pact's effect_group reads
#                  `var:frequency` as a number, not a flag.
#   CONSEQUENCE  : flag literal (e.g. flag:health) — same form as OBLIGATION.
#   PERKS_GAINED : numeric. Per PoD's formula in
#                  POD_create_pact_from_stipulation_stories:
#                    obligation_level + consequence_level + frequency_level + POD_baali_pit_bonus
#                  For Mary's starting pacts at fresh game start (no pit, bonus=0):
#                    Sacrifice of Blood (L2) / 8_years (L-2) / Health    (L2) = 2
#                    Tribute of Agony   (L2) / 8_years (L-2) / Knowledge (L3) = 3
#
POD_maryb_create_starting_pact = {
	# Mary needs investiture lifestyle access to actually use the perk points
	# she'll receive. PoD grants this flag inside diabolism flows we're not
	# touching, so we set it explicitly the first time.
	if = {
		limit = { NOT = { has_character_flag = access_to_investiture } }
		add_character_flag = access_to_investiture
	}

	$DEMON$ = { save_scope_as = demon }
	add_investiture_lifestyle_perk_points = $PERKS_GAINED$

	create_story = {
		type = POD_story_pact
		save_scope_as = pact
	}
	hidden_effect = {
		scope:pact ?= {
			set_variable = { name = obligation         value = $OBLIGATION$ }
			set_variable = { name = frequency          value = $FREQUENCY$ }
			set_variable = { name = consequence        value = $CONSEQUENCE$ }
			set_variable = { name = perk_points_gained value = $PERKS_GAINED$ }
			set_variable = { name = demon              value = scope:demon }
			set_variable = { name = completed          value = no }
		}
		add_to_variable_list = {
			name   = demonic_pacts
			target = scope:pact
		}
	}
}
