POD_character_vampire_malkavian_initialization_effect = {
	if = {
		limit = {
			is_malkavian_trigger = yes
			# If they already have a madness trait, don't give them another one
			NOR = {
				has_trait = lunatic
				has_trait = possessed
				has_trait = depressed
			}
		}
		POD_character_vampire_madness_initialization_effect = yes
	}
}

POD_character_vampire_madness_initialization_effect = {
	random_list = {
			
		60 = {
			add_trait = "lunatic_genetic"		
		}

		20 = {
			add_trait = "possessed_genetic"		
		}
			
		20 = {
			add_trait = "depressed_genetic"			
		}			
	}
}
