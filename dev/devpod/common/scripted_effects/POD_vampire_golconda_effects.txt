POD_reapply_clan_curse_effect = {
	POD_reapply_visual_clan_curse_effect = yes
	if = {
		limit = { POD_suppress_clan_curse_trigger = no }
		if = {
			limit = { has_character_modifier = uncursed_tzimisce_modifier }
			remove_character_modifier = uncursed_tzimisce_modifier
		}
		every_in_list = {
			variable = suppressed_clan_curse_traits
			prev = { add_trait = prev }
		}
		clear_variable_list = suppressed_clan_curse_traits
	}
}

POD_lose_golconda_effect = {
	if = {
		limit = {
			has_trait = golconda
		}
		remove_trait_force_tooltip = golconda
		add_trait_force_tooltip = golconda_seeker
		POD_golconda_seeker_maintenance_effect = yes
		POD_lose_elder_perks_effect = yes
		POD_reapply_clan_curse_effect = yes
		trigger_event = POD_golconda_journey.1000
	}
	if = {
		limit = {
			OR = {
				POD_is_on_golconda_journey_trigger = yes
				AND = {
					has_trait = golconda_seeker
					has_trait_xp = { trait = golconda_seeker value > 0 }
				}
			}
		}
		if = {
			limit = { POD_is_on_golconda_journey_trigger = yes }
			POD_abandon_journeys_of_type_effect = { TYPE = golconda_2 }
		}
		if = {
			limit = {
				has_trait = golconda_seeker
				has_trait_xp = { trait = golconda_seeker value > 0 }
			}
			add_trait_xp = { trait = golconda_seeker value = -100 }
		}
		trigger_event = POD_golconda_journey.1001
	}
}
