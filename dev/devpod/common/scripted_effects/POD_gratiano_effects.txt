# Route all Trust changes here. $DELTA$ must be a single token: number, script value or scope value.
POD_gratiano_change_trust_effect = {
	save_scope_value_as = {
		name = pod_gratiano_new_trust
		value = {
			value = POD_gratiano_trust_value
			add = $DELTA$
			min = 0
			max = 100
		}
	}
	set_variable = {
		name = POD_gratiano_trust
		value = scope:pod_gratiano_new_trust
	}
	every_owned_story = {
		# Leave mirrors unset during Proclamation to hide its bars.
		limit = {
			story_type = POD_gratiano_story
			var:phase >= 2
		}
		set_variable = { name = trust_mirror value = prev.POD_gratiano_trust_value }
	}
	POD_gratiano_update_trust_penalty_effect = yes
	if = {
		limit = {
			POD_gratiano_trust_value <= 0
			is_ai = no
			NOT = { has_character_flag = POD_gratiano_exposure_pending }
			NOT = { has_character_flag = POD_gratiano_exposure_cooldown }
			NOT = {
				is_target_in_global_variable_list = {
					name = unavailable_unique_decisions
					target = flag:POD_gratiano_strike_the_castle
				}
			}
			NOT = { has_global_variable = variable_POD_lasombra_dead }
		}
		hidden_effect = {
			add_character_flag = {
				flag = POD_gratiano_exposure_pending
				days = 30		# Deduplication only; 0003 sets the lockout.
			}
			trigger_event = {
				id = POD_gratiano.0003
				days = 1
			}
		}
	}
	else_if = {
		limit = {
			POD_gratiano_trust_value <= 20
			POD_gratiano_trust_value > 0		# Match 0002: suppressed exposure at 0 must not consume its warning.
			is_ai = no
			NOT = { has_character_flag = POD_gratiano_trust_warning_pending }
			NOT = { has_character_flag = POD_gratiano_trust_warning_cooldown }
		}
		hidden_effect = {
			add_character_flag = {
				flag = POD_gratiano_trust_warning_pending
				days = 5		# Deduplication only; 0002 sets the cooldown.
			}
			trigger_event = {
				id = POD_gratiano.0002
				days = 1
			}
		}
	}
}

POD_gratiano_update_trust_penalty_effect = {
	hidden_effect = {
		remove_character_modifier = POD_gratiano_low_standing_1_modifier
		remove_character_modifier = POD_gratiano_low_standing_2_modifier
		remove_character_modifier = POD_gratiano_low_standing_3_modifier
		if = {
			limit = { POD_gratiano_trust_value <= 25 }
			add_character_modifier = POD_gratiano_low_standing_3_modifier
		}
		else_if = {
			limit = { POD_gratiano_trust_value <= 50 }
			add_character_modifier = POD_gratiano_low_standing_2_modifier
		}
		else_if = {
			limit = { POD_gratiano_trust_value <= 75 }
			add_character_modifier = POD_gratiano_low_standing_1_modifier
		}
	}
}

# Removing the story variable hides its counter row.
POD_gratiano_hide_story_counter_effect = {
	every_owned_story = {
		limit = { story_type = POD_gratiano_story }

		}
	}
}

# Quarterly sync, called in story_owner scope.
POD_gratiano_story_sync_effect = {
	every_owned_story = {
		limit = { story_type = POD_gratiano_story }
		# 1 Proclamation -> 2 Courtier -> 3 Liberator -> 4 Kingslayer.
		# Separate ifs allow catch-up across phases in one pass.
		if = {
			limit = {
				var:phase ?= 1
				prev = { NOT = { has_character_modifier = bad_gratiano_modifier } }
			}
			set_variable = { name = phase value = 2 }
			prev ?= {
				if = {
					limit = { NOT = { has_character_flag = POD_gratiano_intro_fired } }
					add_character_flag = POD_gratiano_intro_fired
					trigger_event = POD_gratiano.0001
				}
			}
		}
		if = {
			limit = {
				var:phase ?= 2
				prev = {
					has_character_flag = POD_gratiano_giovanni_cleared
					has_character_flag = POD_gratiano_moncada_dealt
					has_character_flag = POD_gratiano_nahir_dealt
					has_character_flag = POD_gratiano_ventrue_cleared
					has_character_flag = POD_amici_noctis_member
				}
			}
			set_variable = { name = phase value = 3 }
		}
		if = {
			limit = {
				var:phase ?= 3
				is_target_in_global_variable_list = {
					name = unavailable_unique_decisions
					target = flag:POD_gratiano_strike_the_castle
				}
			}
			set_variable = { name = phase value = 4 }
		}
		if = {
			limit = { var:phase >= 2 }
			set_variable = { name = trust_mirror value = prev.POD_gratiano_trust_value }
			set_variable = { name = conspiracy_mirror value = prev.POD_gratiano_conspiracy_value }
		}
		# Catches what the membership hooks cannot see: members who died, and coterie
		# departures driven by PoD's own quarterly rivalry pulse.
		POD_gratiano_story_roster_refresh_effect = yes
	}
	# Cleared rivals can regain land before Plan I is sealed.
	POD_gratiano_giovanni_reeval_effect = yes
	POD_gratiano_ventrue_reeval_effect = yes
	if = {
		limit = {
			has_character_flag = POD_gratiano_giovanni_cleared
			has_character_flag = POD_gratiano_ventrue_cleared
			has_character_flag = POD_gratiano_moncada_dealt
			has_character_flag = POD_gratiano_nahir_dealt
			NOT = { has_character_flag = POD_amici_noctis_member }
		}
		trigger_event = { id = POD_gratiano.0056 days = 3 }
	}
	# Sybil's terms.
	if = {
		limit = {
			NOT = { has_character_flag = POD_gratiano_sybil_asked }
			character:505140 = { is_alive = yes NOT = { has_trait = torpor } }
		}
		trigger_event = { id = POD_gratiano.0254 days = { 3 7 } }
	}
	# Re-arm the convocation after torpor.
	if = {
		limit = {
			has_global_variable = variable_POD_lasombra_dead
			NOT = { has_character_flag = POD_gratiano_convened }
			NOT = { has_trait = torpor }
		}
		trigger_event = { id = POD_gratiano.0280 days = { 3 7 } }
	}
	POD_gratiano_check_elders_framed_effect = yes
}

# Story scope. Rebuilt in full on every call, so the panel sheds the dead and the
# sold-out without any call site having to reconcile anything. The coterie and the pool
# are disjoint by the induct/remove contract, but a coterie invite taken outside that
# contract would otherwise portrait the same member twice.
# Gated here, not at the call sites: Gratiano may already hold a coterie during the
# Proclamation, and the panel stays bare until Born to Rule opens the Courtier phase.
POD_gratiano_story_roster_refresh_effect = {
	if = {
		limit = { var:phase >= 2 }
		save_scope_as = gratiano_roster_story
		clear_variable_list = conspiracy_inner_circle
		clear_variable_list = conspiracy_circle

		story_owner ?= {
			save_scope_as = gratiano_roster_owner

			var:coterie ?= {
				every_in_list = {
					variable = members
					limit = {
						is_alive = yes
						NOT = { this = scope:gratiano_roster_owner }
					}
					scope:gratiano_roster_story = {
						add_to_variable_list = { name = conspiracy_inner_circle target = prev }
					}
				}
			}

			# 18 fills the panel's three portrait rows; the pool is not expected to reach it.
			ordered_in_list = {
				variable = POD_gratiano_recruit_pool
				order_by = intrigue
				max = 18
				check_range_bounds = no
				limit = {
					is_alive = yes
					NOT = { this = scope:gratiano_roster_owner }
					NOT = {
						scope:gratiano_roster_story = {
							is_target_in_variable_list = { name = conspiracy_inner_circle target = prev }
						}
					}
				}
				scope:gratiano_roster_story = {
					add_to_variable_list = { name = conspiracy_circle target = prev }
				}
			}
		}
	}
}

# Gratiano scope. Every membership change routes through here; the panel lists live
# on the story, so no call site has to reach into it or know the panel exists.
POD_gratiano_conspiracy_changed_effect = {
	hidden_effect = {
		every_owned_story = {
			limit = { story_type = POD_gratiano_story }
			POD_gratiano_story_roster_refresh_effect = yes
		}
	}
}

# Called in Gratiano scope by the story sync; ROOT may be the story.

POD_gratiano_giovanni_reeval_effect = {
	if = {
		limit = {
			POD_gratiano_map_arcs_active_trigger = yes
			has_character_flag = POD_gratiano_giovanni_arc_intro
		}
		if = {
			limit = {
				POD_gratiano_plan_i_sealed_trigger = no
				has_character_flag = POD_gratiano_giovanni_cleared
				POD_gratiano_giovanni_cleared_trigger = no
			}
			remove_character_flag = POD_gratiano_giovanni_cleared
			hidden_effect = {
				# Any blocker can represent the reopened objective.
				every_independent_ruler = {
					limit = {
						is_giovanni_trigger = yes
						any_realm_county = { POD_gratiano_italian_sphere_trigger = yes }
					}
					save_scope_as = POD_gratiano_giovanni_blocker
				}
				# Pending flags outlive dropped events. Restart unconditionally on reopening;
				# a duplicate council instance collapses when the objective clears again.
				trigger_event = { id = POD_gratiano.0061 days = { 30 60 } }
			}
			trigger_event = POD_gratiano.0074
		}
		else_if = {
			limit = {
				NOT = { has_character_flag = POD_gratiano_giovanni_cleared }
				POD_gratiano_giovanni_cleared_trigger = yes
			}
			add_character_flag = POD_gratiano_giovanni_cleared
			# A later re-clear must not release another rival arc's foreground slot.
			if = {
				limit = { has_character_flag = POD_gratiano_giovanni_holding_slot }
				remove_character_flag = POD_gratiano_giovanni_holding_slot
				remove_character_flag = POD_gratiano_rival_in_focus
			}
			trigger_event = POD_gratiano.0073
		}
		if = {
			limit = {
				character:505018 ?= { is_alive = no }
				NOT = { has_character_flag = POD_gratiano_giovanni_blowback_done }
			}
			add_character_flag = POD_gratiano_giovanni_blowback_done
			trigger_event = { id = POD_gratiano.0076 days = { 30 90 } }
		}
	}
}

POD_gratiano_ventrue_reeval_effect = {
	if = {
		limit = {
			POD_gratiano_map_arcs_active_trigger = yes
			has_character_flag = POD_gratiano_ventrue_arc_intro
		}
		# Seed before comparing so the initial count cannot register as progress.
		if = {
			limit = { NOT = { has_variable = POD_gratiano_ventrue_last_count } }
			set_variable = {
				name = POD_gratiano_ventrue_last_count
				value = POD_gratiano_ventrue_blockers_value
			}
		}
		if = {
			limit = {
				NOT = { has_character_flag = POD_gratiano_ventrue_cleared }
				var:POD_gratiano_ventrue_last_count > POD_gratiano_ventrue_blockers_value
				POD_gratiano_ventrue_blockers_value >= 1
			}
			send_interface_toast = {
				type = event_generic_neutral
				# ROOT may be the story; use Gratiano explicitly.
				title = POD_gratiano_ventrue_progress_toast
				left_icon = character:505144
			}
		}
		set_variable = {
			name = POD_gratiano_ventrue_last_count
			value = POD_gratiano_ventrue_blockers_value
		}
		if = {
			limit = {
				POD_gratiano_plan_i_sealed_trigger = no
				has_character_flag = POD_gratiano_ventrue_cleared
				POD_gratiano_ventrue_cleared_trigger = no
			}
			remove_character_flag = POD_gratiano_ventrue_cleared
			hidden_effect = {
				every_independent_ruler = {
					limit = {
						is_ventrue_trigger = yes
						any_realm_county = { POD_gratiano_italian_sphere_trigger = yes }
					}
					save_scope_as = POD_gratiano_ventrue_blocker
				}
				# Restart unconditionally, as in the Giovanni re-evaluation.
				trigger_event = { id = POD_gratiano.0091 days = { 30 60 } }
				trigger_event = { id = POD_gratiano.0092 days = { 60 120 } }
			}
			trigger_event = POD_gratiano.0104
		}
		else_if = {
			limit = {
				NOT = { has_character_flag = POD_gratiano_ventrue_cleared }
				POD_gratiano_ventrue_cleared_trigger = yes
			}
			add_character_flag = POD_gratiano_ventrue_cleared
			if = {
				limit = { has_character_flag = POD_gratiano_ventrue_holding_slot }
				remove_character_flag = POD_gratiano_ventrue_holding_slot
				remove_character_flag = POD_gratiano_rival_in_focus
			}
			trigger_event = POD_gratiano.0103
		}
	}
}

# Pool add/remove contract: Gratiano scope; $TARGET$ is the member.
# Route membership changes here to keep member flags in sync.
POD_gratiano_add_to_pool_effect = {
	if = {
		limit = {
			$TARGET$ = { is_alive = yes }
			NOT = {
				is_target_in_variable_list = {
					name = POD_gratiano_recruit_pool
					target = $TARGET$
				}
			}
		}
		add_to_variable_list = {
			name = POD_gratiano_recruit_pool
			target = $TARGET$
		}
		POD_gratiano_conspiracy_changed_effect = yes
	}
}

POD_gratiano_remove_from_pool_effect = {
	remove_list_variable = {
		name = POD_gratiano_recruit_pool
		target = $TARGET$
	}
	if = {
		limit = { $TARGET$ = { is_alive = yes } }
		$TARGET$ = {
			remove_character_flag = POD_gratiano_conscript_disloyal
			remove_character_flag = POD_gratiano_academy_zealot
		}
	}
	POD_gratiano_conspiracy_changed_effect = yes
}

POD_gratiano_conscript_to_pool_effect = {
	POD_gratiano_add_to_pool_effect = { TARGET = $TARGET$ }
	$TARGET$ = { add_character_flag = POD_gratiano_conscript_disloyal }
}

		age = 30
		gender_female_chance = 50
		dynasty = none
		faith = root.faith
		culture = root.culture
		random_traits = no
		trait = lasombra
		trait = $EDU$

		trait = $T2$
		$FOCUS$ = $FOCUSVAL$
		prowess = $PROW$

			POD_character_create_vampirelord_effect = yes
		}
	}
}
