POD_character_initialization_for_nobles_effect = {

	POD_set_resonance_effect = yes

	if = {
		limit = { NOT = { has_character_flag = had_POD_character_initialization } }
		
	if = {
		limit = { exists = liege }
		liege = { save_scope_as = template_ruler }
	}
	else_if = { # give same rank title making indepent
		limit = { exists = capital_province.county.previous_holder }
		capital_province.county.previous_holder = { save_scope_as = template_ruler }
	}
	
	if = {
		limit = {
			scope:template_ruler ?= { is_forsaken_or_gantenbein_trigger = yes }
		}
		POD_character_initialization_for_forsaken_barons_effect = yes
	}
	else_if = {
		limit = {
			scope:template_ruler ?= { 
				OR = { 
					has_trait = supehunter
					is_child_of_supehunter_trigger = yes
				} 
			}
		}
		POD_character_initialization_for_supehunter_barons_effect = yes
	}
	else_if = {
		limit = {
			scope:template_ruler ?= { 
				OR = { 
					has_trait = vampire
					has_trait = risen 
				}
			}
		}
		POD_character_initialization_for_vampire_nobles_effect = yes
	}
	else_if = {
		limit = {
			scope:template_ruler ?= { has_trait = shapeshifter }
		}
		POD_character_initialization_for_shapeshifter_barons_effect = yes
	}
	else_if = {
		limit = {
			scope:template_ruler ?= { has_trait = wraith }
		}
		POD_character_initialization_for_wraith_nobles_effect = yes
	}
	else_if = {
		limit = {
			scope:template_ruler ?= { has_trait = fae }
		}
		POD_character_initialization_for_fae_nobles_effect = yes
	}
	else_if = {
		limit = {
			scope:template_ruler ?= { has_POD_kj_blood_trigger = yes }
		}
		POD_character_initialization_for_kueijin_nobles_effect = yes
	}
	else_if = { # delay to here for ghoul-ww
		limit = {
			scope:template_ruler ?= { has_trait = ghoul }
		}
		POD_character_initialization_for_vampire_nobles_effect = yes
	}
	else_if = {
		limit = {
			scope:template_ruler ?= { 
				has_trait = revenant 
				religion = { is_in_family = rf_kindred }
			}
		}
		trigger_event = POD_ghoul_maintenance.300
	}
	else_if = {
		limit = {
			scope:template_ruler ?= { 
				OR = { 
					has_trait = changeling
					has_trait = kinain 
				}
				religion = { is_in_family = rf_fae }
			}
		}
		POD_character_initialization_for_fae_nobles_effect = yes
	}
	else_if = {
		limit = {
			scope:template_ruler ?= { 
				has_trait = kinfolk 
				OR = { 
					religion = { is_in_family = rf_gaia } 
					religion = { is_in_family = rf_wyrm } 
				}
			}
		}
		POD_character_initialization_for_shapeshifter_barons_effect = yes
	}
	
	else_if = {
		limit = {
			exists = capital_province
			capital_province = { religion = religion:calomenasforsaken_religion }
		}
		POD_character_initialization_for_forsaken_barons_effect = yes
	}
	else_if = {
		limit = {
			exists = capital_province
			capital_province = { religion = { is_in_family = rf_hunters } }
		}
		POD_character_initialization_for_supehunter_barons_effect = yes
	}

			exists = capital_province
			capital_province = { religion = { is_in_family = rf_kueijin } }
		}
		POD_character_initialization_for_kueijin_nobles_effect = yes
	}
	else_if = {
		limit = {
			exists = capital_province
			capital_province = { religion = { is_in_family = rf_kindred } }
		}
		POD_character_initialization_for_vampire_nobles_effect = yes
	}
	else_if = {
		limit = {
			exists = capital_province
			capital_province = { 
				OR = {
					religion = { is_in_family = rf_gaia }
					religion = { is_in_family = rf_wyrm }
				}
			}
		}
		POD_character_initialization_for_shapeshifter_barons_effect = yes
	}	
	else_if = {
		limit = {
			exists = capital_province
			capital_province = { religion = { is_in_family = rf_fae } }
		}
		POD_character_initialization_for_fae_nobles_effect = yes
	}
	else_if = {
		limit = {
			exists = capital_province
			capital_province = { religion = { is_in_family = rf_wraiths } }
		}
		POD_character_initialization_for_wraith_nobles_effect = yes

		POD_character_initialization_for_random_nobles_effect = yes
	}
	
		if = {
			limit = { has_amenti_liege_trigger = yes }
			POD_character_initialization_for_neteru_barons_effect = yes
		}
		if = {
			limit = { has_demonic_liege_trigger = yes }
			POD_character_initialization_for_thrall_barons_effect = yes
		}
		trigger_event = POD_maintenance.2
	}

		}

				} 
			}
			add_trait = supehunter

				limit = { liege_or_court_owner ?= { has_trait = shih } }
				add_trait = shih
				trigger_event = POD_character_canon_modifier_initalization.210
			}
		}
		else_if = {
			limit = { 
				liege_or_court_owner ?= {
					OR = {
						has_trait = shapeshifter 
						has_trait = kinfolk 
					}
				}
			}
			add_trait = kinfolk
			trigger_event = POD_character_canon_modifier_initalization.600
		}
		else_if = {
			limit = { 
				liege_or_court_owner ?= { 
					OR = {
						has_trait = fae 
						has_trait = changeling 
					}
				}
			}
			add_trait = changeling
			trigger_event = POD_character_canon_modifier_initalization.910
		}
		else_if = {
			limit = { liege_or_court_owner ?= { has_trait = wraith } }
			add_trait = wraith
			POD_arcanoi_initiation_effect = yes
			trigger_event = POD_character_canon_modifier_initalization.400
		}
		else = {

		}
	}
}

give_unbondable_secret_or_trait_effect = {
	save_temporary_scope_as = unbondable
	if = {
		limit = { #Not already a unbondable
			NOR = {
				has_trait = unbondable
				any_secret = { secret_type = secret_unbondable }
			}
		}
		if = {
			limit = {
				trait_is_shunned_or_criminal_in_my_or_lieges_faith_trigger = { TRAIT = unbondable GENDER_CHARACTER = scope:unbondable }
			}
			add_secret = {
				type = secret_unbondable
			}
			# random_secret = {
				# limit = { secret_type = secret_unbondable }
				# save_scope_as = unbondable_secret
			# }
		}
		else = {
			add_trait = unbondable
		}
	}

		limit = {
			POD_is_rokea_trigger = yes
			POD_has_breed_trigger = no
		}	
		
		# Assign Breed at Random 
		random_list = {
			10 = { add_trait = beast_born }
			#5 = { add_trait = metis  }
			10 = { add_trait = homid_born } 
		}
	}
}

POD_character_initialization_for_fae_nobles_effect = {
	random_list = {
		80 = { 
			trigger = { has_demonic_liege_trigger = no }
			add_trait = fae
			modifier = {
				add = -75
				has_global_variable = shattering
			}
		}
		20 = { add_trait = changeling }
	}
	
	if = {
		limit = { has_trait = fae }
		POD_fae_initialization_effect = yes
	}
	if = {
		limit = { has_trait = changeling }
		POD_changeling_initialization_effect = yes
	}
	trigger_event = POD_fae_maintenance.3
	
	add_character_flag = {
		flag = had_POD_character_initialization
	}
}

POD_character_initialization_for_fae_barons_effect = {
	if = {
		limit = {
			current_date < 1348.1.1
		}
		random_list = {
			70 = { 
				trigger = { has_demonic_liege_trigger = no }
				add_trait = fae
			}
			30 = { add_trait = changeling }
		}
	}
	else = {
		random_list = {
			30 = {
				trigger = { has_demonic_liege_trigger = no }
				add_trait = fae
			}
			70 = { add_trait = changeling }
		}
	}
	
	if = {
		limit = { has_trait = fae }
		POD_fae_initialization_effect = yes
	}
	if = {
		limit = { has_trait = changeling }
		POD_changeling_initialization_effect = yes
	}
	trigger_event = POD_fae_maintenance.3
	
	add_character_flag = {
		flag = had_POD_character_initialization
	}
}

POD_character_initialization_for_fae_commoners_effect = {
	random_list = {
		10 = { add_trait = fae }
		20 = { add_trait = changeling }
		70 = { add_trait = kinain }
	}
	
	if = {
		limit = {
			has_trait = fae
		}
		POD_fae_initialization_effect = yes
	}
	if = {
		limit = {
			has_trait = changeling
		}
		POD_changeling_initialization_effect = yes
	}
	if = {
		limit = {
			has_trait = kinain
		}
		random = {
			chance = 1
			add_trait = psychic
			POD_resonance_increase_intensity_effect = yes
		}
		random = {
			chance = 1
			give_sorcerer_secret_or_trait_effect = yes
		}
		POD_kinain_initialization_effect = yes
	}
	trigger_event = POD_fae_maintenance.3
	
	add_character_flag = {
		flag = had_POD_character_initialization
	}
}
