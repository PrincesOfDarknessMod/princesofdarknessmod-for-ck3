			add_trait = vampire
			trigger_event = POD_character_canon_modifier_initalization.101
		}
		else_if = {
			limit = { liege_or_court_owner ?= { has_trait = kueijin } }
			add_trait = kueijin
			trigger_event = POD_character_canon_modifier_initalization.200
		}
		else_if = {
			limit = { 
				liege_or_court_owner ?= { 
					has_trait = supehunter 
					NOT = { has_trait = ghoul }
				} 
			}
			add_trait = supehunter
			if = {
				limit = { liege_or_court_owner ?= { has_trait = shih } }
				add_trait = shih
				trigger_event = POD_character_canon_modifier_initalization.210
			}
		}
		else_if = {
			limit = { 
				liege_or_court_owner ?= {
					NOT = { has_trait = entomos }
					OR = {
						has_trait = fera 
						has_trait = kinfolk 
					}
				}
			}
			add_trait = kinfolk
			trigger_event = POD_character_canon_modifier_initalization.600
		}
		else_if = {
			limit = { 
				liege_or_court_owner ?= { 
					OR = {
						has_trait = fae 
						has_trait = changeling 
					}
				}
			}
			add_trait = changeling
			trigger_event = POD_character_canon_modifier_initalization.910
		}
		else_if = {
			limit = { liege_or_court_owner ?= { has_trait = wraith } }
			add_trait = wraith
			POD_arcanoi_initiation_effect = yes
			trigger_event = POD_character_canon_modifier_initalization.400
		}
		else = {
			add_trait = mortal
		}
	}
}
