	}

		limit = { scope:attacker = { has_trait = presenceadvanced } }
		custom_tooltip = POD_entrance.tt.pc.presenceadvanced

	if = {
		limit = { scope:attacker = { has_perk = ironclad_command_perk } }
		custom_tooltip = POD_entrance.tt.pc.ironclad_command_perk
	}
	if = {
		limit = { scope:attacker = { has_perk = as_below_so_above_perk }	}
		custom_tooltip = POD_entrance.tt.pc.as_below_so_above_perk
	}
}

POD_mesmerize_defender_tt_effect = {
	if = {
		limit = { scope:defender = { has_character_flag = willpower_spent } }
		custom_tooltip = POD_entrance.tt.willpower_spent
	}
	if = {
		limit = { scope:defender = { has_perk = unswayable_mind_perk } }
		custom_tooltip = POD_entrance.tt.unswayable_mind_perk
	}
	if = {
		limit = { scope:defender = { has_perk = fortify_the_inner_facade_perk } }
		custom_tooltip = POD_entrance.tt.fortify_the_inner_facade_perk
	}
	if = {
		limit = { scope:defender = { has_character_flag = song_of_jupiter } }
		custom_tooltip = POD_entrance.tt.song_of_jupiter
	}
	if = {
		limit = {
			is_vampire_trigger = yes
			scope:attacker.generation_value < scope:defender.generation_value
		}
		custom_tooltip = POD_mesmerize_tt_defender_generation_adv
	}
}
