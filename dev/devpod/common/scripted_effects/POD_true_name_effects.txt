POD_learn_true_name_of_effect = {
        set_relation_known_true_name = $TARGET$
}
POD_forget_true_name_of_effect = {
        remove_relation_known_true_name = $TARGET$
}

# True Names Non Character

POD_learn_true_name_of_trait_effect = {
	custom_description = {
		text = POD_learn_true_name_of_trait_effect
		object = $TRAIT$
		add_to_variable_list = { name = known_true_name target = $TRAIT$ }
	}
}
POD_forget_true_name_of_trait_effect = {
	custom_description = {
		text = POD_forget_true_name_of_trait_effect
		object = $TRAIT$
		remove_list_variable = { name = known_true_name target = $TRAIT$ }
	}
}

POD_learn_true_name_of_nonstandard_effect = {
	save_scope_value_as = { name = true_name_scope value = $TARGET$ }
	custom_tooltip = {
		text = POD_LEARN_TRUE_NAME_OF_NONSTANDARD_EFFECT
		add_to_variable_list = { name = known_true_name target = $TARGET$ }
	}
}
POD_forget_true_name_of_nonstandard_effect = {
	save_scope_value_as = { name = true_name_scope value = $TARGET$ }
	custom_tooltip = {
		text = POD_FORGET_TRUE_NAME_OF_NONSTANDARD_EFFECT
		remove_list_variable = { name = known_true_name target = $TARGET$ }
	}
}

POD_learn_random_true_name_of_trait_type_effect = {
	save_scope_as = learner
	random_trait = {
		limit = { 
			scope:learner = { NOT = { POD_knows_true_name_of_trait_trigger = { TARGET = PREV } } }
			has_trait_category = $TYPE$
		}
		save_scope_as = trait
		scope:learner = { POD_learn_true_name_of_trait_effect = { TRAIT = scope:trait } }
	}
}
POD_learn_random_true_name_of_trait_flag_effect = {
	save_scope_as = learner
	random_trait = {
		limit = { 
			scope:learner = { NOT = { POD_knows_true_name_of_trait_trigger = { TARGET = PREV } } }
			has_trait_flag = reweaving_$FLAG$_trait 
		}
		save_scope_as = trait
		scope:learner = { POD_learn_true_name_of_trait_effect = { TRAIT = scope:trait } }
	}
}
