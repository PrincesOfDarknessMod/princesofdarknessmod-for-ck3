POD_assemble_brujah_cadre_effect = {
	
	save_scope_as = leader
	
	POD_dismiss_brujah_cadre_effect = yes
	every_knight = {
		limit = { has_trait = brujah }
		add_hook_no_toast = { target = scope:leader type = cadre_hook}
		add_opinion = {  modifier = POD_cadre_opinion target = scope:leader } 
		scope:leader = { 
			remove_hook = { target = prev }
			add_hook_no_toast = { target = prev type = cadre_hook }
			add_character_modifier = { modifier = POD_brujah_cadre_modifier years = 10 }
			add_opinion = {  modifier = POD_cadre_opinion target = prev } 
		}
	}

}

POD_dismiss_brujah_cadre_effect = {
	save_scope_as = leader
	
	
	every_hooked_character = {
		limit = { 
			has_opinion_modifier = { target = scope:leader modifier = POD_cadre_opinion } 
		}
		remove_opinion = { modifier = POD_cadre_opinion target = scope:leader }
		remove_hook = { type = cadre_hook target = scope:leader }
		scope:leader = {
			hidden_effect = { remove_character_modifier = POD_brujah_cadre_modifier }
			remove_hook = { type = cadre_hook target = prev }
			remove_opinion = { modifier = POD_cadre_opinion target = prev }
			
		}
	}

}
