POD_possessed_spirit_leaves_body = {
	if = {
		limit = {
			has_trait = fomori # TODO POSSESSED: other types
			POD_is_not_a_mindless_clone = yes
			# safeguard to skip historical characters that get culled on non-fomori bookmarks
			any_owned_story = { type = POD_story_possessed }
		}
		
		save_scope_as = cloned_character
		
		if = {
			limit = { is_ruler = yes }
			save_scope_as = spirit_employer
		}
		else_if = {
			limit = { exists = location }
			location = { save_scope_as = spirit_location }
		}
		else = {
			random_independent_ruler = {
				limit = {
					exists = capital_province
					capital_province = { local_pool_is_full_trigger = no }
				}
				alternative_limit = { exists = capital_province }
				capital_province = { save_scope_as = spirit_location }
			}
		}
		
		hidden_effect_new_object = {
			# TODO POSSESSED: ask purple about overhauling seeder banes
			if = {
				limit = { exists = scope:spirit_employer }
				create_character = {
					age = scope:cloned_character.age
					faith = faith:wyrmbound
					culture = scope:cloned_character.culture
					gender = scope:cloned_character
					random_traits = no
					dynasty = none
					employer = scope:spirit_employer
					save_scope_as = spawned_spirit
				}
			}
			else = {
				create_character = {
					age = scope:cloned_character.age
					faith = faith:wyrmbound
					culture = scope:cloned_character.culture
					gender = scope:cloned_character
					random_traits = no
					dynasty = none
					location = scope:spirit_location
					save_scope_as = spawned_spirit
				}
			}
			scope:spawned_spirit = {
				POD_copy_base_skills_effect = {
					CLONED_CHAR = scope:cloned_character
					CLONE = scope:spawned_spirit
				}
				copy_traits = scope:cloned_character
				POD_remove_splat_traits_effect = yes
				add_trait = spirit
				POD_remove_wrong_supernatural_traits_on_existing_character = yes
				POD_clear_invalid_clone_traits = yes
				POD_copy_trait_xp_for_clone_effect = yes
				POD_copy_sexuality_effect = { ORIGIN = scope:cloned_character }
				change_first_name = { template_character = scope:cloned_character }
				POD_copy_physical_appearance_effect = yes
				POD_copy_vanilla_lifestyle_perks_effect = yes
				
				set_variable = { name = spirit_type value = flag:spirit_seeder }
				POD_spirit_initialization_effect = yes
				
				# TODO POSSESSED: track completed urges for fomarch
			}
			send_interface_message = {
				type = POD_spirit_message
				desc = POD_spawn_spirit_effect_desc
				left_icon = scope:spawned_spirit
			}
		}
		set_variable = {
			name = spawned_spirit
			days = 1
		}
		
		# TODO POSSESSED: alert event for heir
		
		POD_clear_possession = yes
	}
}

POD_possessed_stress_threshold_event_post_immediate = {
	if = {
		limit = {
			any_owned_story = {
				type = POD_story_possessed
				save_temporary_scope_as = possessed_story
			}
		}
		scope:possessed_story = {
			POD_set_or_change_variable = {
				VAR = total_mental_breaks
				VAL = 1
			}
		}
	}
	if = {
		limit = { has_perk = possessed_dissociative_identity_perk }
		POD_shuffle_personality_effect = yes
	}
	if = {
		limit = { has_perk = possessed_fading_perk }
		if = {
			limit = { prestige >= 0 }
			add_prestige = {
				value = prestige
				if = {
					limit = { POD_total_possessed_perks >= POD_possessed_stronger_taint_threshold }
					multiply = -0.66
				}
				else = {
					multiply = -0.33
				}
			}
		}
		else = {
			add_prestige_level = -1
		}
	}
}
