POD_diablerie_unconquered_soul_effect = {
	if = {
		limit = {
			has_trait_rank = {
				trait = vampgeneration
				rank < 10
			}	
		}
		
		custom_description = { 
			value = POD_unconquered_soul_chance
			text = unconquered_soul_warning
		}

		hidden_effect = {
			random = {
				chance = POD_unconquered_soul_chance
				
				scope:victim = {
					save_scope_as = unconquered_soul
				}
				add_character_flag = POD_in_unconquered_soul_event
				#trigger_event = { 
				#	id = POD_diablerize.810 days = { 2 5 }
				#}
				trigger_event = { 
					id = POD_diablerize.800 years = { 5 25 }
				}
			}	
		}
	}
}
