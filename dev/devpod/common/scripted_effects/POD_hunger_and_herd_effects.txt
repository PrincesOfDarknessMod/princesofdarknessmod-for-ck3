#Effect to add or increase the level of bloodloss a character suffers by one step. This already exists in the form of POD_hunger_feeding_bloodloss_effect (and that one is fancier) but that one is used in a lot of places beside herds and uses different numbers for the duration than what I want to use here.
POD_herd_bloodloss_effect = {
	#If the herdmember is a vampire then we don't want them to get anemia since that isn't a thing for vampires. They should get more hungry though.
	if = {
		limit = {
			is_vampire_trigger = yes
		}
		if = {
			limit = {
				has_character_modifier = bloodhunger4_modifier
			}
			remove_character_modifier = bloodhunger4_modifier
			add_character_modifier = bloodhunger5_modifier
		}
		else_if = {
			limit = {
				has_character_modifier = bloodhunger3_modifier
			}
			remove_character_modifier = bloodhunger3_modifier
			add_character_modifier = bloodhunger4_modifier
		}
		else_if = {
			limit = {
				has_character_modifier = bloodhunger2_modifier
			}
			remove_character_modifier = bloodhunger2_modifier
			add_character_modifier = bloodhunger3_modifier
		}
		else_if = {
			limit = {
				has_character_modifier = bloodhunger1_modifier
			}
			remove_character_modifier = bloodhunger1_modifier
			add_character_modifier = bloodhunger2_modifier
		}
		else_if = {
			limit = {
				has_character_modifier = bloodhunger0_modifier
			}
			remove_character_modifier = bloodhunger0_modifier
			add_character_modifier = bloodhunger1_modifier
		}
	}
	#Here's where we handle mortals (and others that can suffer bloodloss)
	#Feeding on a mortal while they're suffering from catastrophic bloodloss leads to death of the victim
	else_if = {
		limit = {
			has_character_modifier = catastrophicbloodloss_modifier
		}
		unknown_murder_effect = {
			VICTIM = scope:herdmember
			MURDERER = scope:herd_owner
			REASON = death_mysterious
		}
	}
	#Major bloodloss upgrades to catastrophic bloodloss
	else_if = {
		limit = {
			has_character_modifier = majorbloodloss_modifier
		}
		remove_character_modifier = majorbloodloss_modifier
		add_character_modifier = {
			modifier = catastrophicbloodloss_modifier
			months = 12
		}
	}
	#Small bloodloss upgrades to major bloodloss
	else_if = {
		limit = {
			has_character_modifier = smallbloodloss_modifier
		}
		remove_character_modifier = smallbloodloss_modifier
		add_character_modifier = {
			modifier = majorbloodloss_modifier
			months = 6
		}
	}
	#Slight bloodloss upgrades to small bloodloss
	else_if = {
		limit = {
			has_character_modifier = slightbloodloss_modifier
		}
		remove_character_modifier = slightbloodloss_modifier
		add_character_modifier = {
			modifier = smallbloodloss_modifier
			months = 3
		}
	}
	#If there is no bloodloss then add slight bloodloss for 29 days (to protect against accidental overlap during a 30-day feeding cycle)
	else = {
		add_character_modifier = {
			modifier = slightbloodloss_modifier
			days = 29
		}
	}
}

#This effect handles granting resonance xp when feeding on your herd by checking what resonance_xp_value evaluates to for the herdmember and then adds that number as lifestyle xp to the herd owner.
POD_herd_resonance_xp_effect = {
	#Choleric
	if = {
		limit = {
			scope:herdmember = { has_character_flag = choleric_resonance }
		}
		scope:herd_owner = { add_choleric_lifestyle_xp = scope:herdmember.resonance_xp_value }
	}
	#Melancholic/Melancholy
	else_if = {
		limit = {
			scope:herdmember = { has_character_flag = melancholic_resonance }
		}
		scope:herd_owner = { add_melancholy_lifestyle_xp = scope:herdmember.resonance_xp_value }
	}
	#Phlegmatic
	else_if = {
		limit = {
			scope:herdmember = { has_character_flag = phlegmatic_resonance }
		}
		scope:herd_owner = { add_phlegmatic_lifestyle_xp = scope:herdmember.resonance_xp_value }
	}
	#Sanguine
	else_if = {
		limit = {
			scope:herdmember = { has_character_flag = sanguine_resonance }
		}
		scope:herd_owner = { add_sanguine_lifestyle_xp = scope:herdmember.resonance_xp_value }
	}
	#Animal
	#This isn't a else_if because vampires can have both one of the regular resonances and animal resonance at the same time.
	if = {
		limit = {
			scope:herdmember = { has_animal_resonance_trigger = yes }
		}
		scope:herd_owner = { add_animal_lifestyle_xp = scope:herdmember.resonance_xp_value }
	}
}


#Main feeding effect that gets executed on each herdmember via a recurring event.

POD_herd_feed_effect = {
	#First we check what we're feeding on and handle hunger decreases and bloodloss accordingly.

	#Herdmember is a mortal (or at least something that increased blood potency makes less nourishing)
	if = {
		limit = {
			#This is where we add all the traits that indicate that someone is mortal because higher blood potencies mean more drinking when mortals are involved.
			scope:herdmember = { has_supernatural_race_trigger = no }
		}
		#Blood Potency 0-3 This is where a vampire can still reduce hunger by taking a small sip from a mortal
		if = {
			limit = {
				OR = {
					scope:herd_owner = { has_character_modifier = bloodpotency0_modifier }
					scope:herd_owner = { has_character_modifier = bloodpotency1_modifier }
					scope:herd_owner = { has_character_modifier = bloodpotency2_modifier }
					scope:herd_owner = { has_character_modifier = bloodpotency3_modifier }
				}
			}

			#Reduce hunger
			scope:herd_owner = { POD_herd_hunger_decrease_effect = yes }
			#Reduce hunger more if the character has the right perk
			if = {
				limit = {
					scope:herd_owner = { has_perk = quicken_the_mortals_blood_perk }
				}
				scope:herd_owner = { POD_herd_hunger_decrease_effect = yes }
			}
		}
		#Blood Potency 4 Reduces hunger only if you have the right perk
		else_if = {
			limit = {
				scope:herd_owner = { has_perk = quicken_the_mortals_blood_perk }
				scope:herd_owner = { has_character_modifier = bloodpotency4_modifier }
			}
			scope:herd_owner = { POD_herd_hunger_decrease_effect = yes }
		}
		#Blood Potency 5 Reduces hunger only if you have the right perk and only to a certain level
		#Blood Potency 6+ No more hunger reduction when sipping even with the right perk so there's no need for a check for that.
		else_if = {
			limit = {
				scope:herd_owner = { has_perk = quicken_the_mortals_blood_perk }
				scope:herd_owner = { has_character_modifier = bloodpotency5_modifier }
				OR = {
					scope:herd_owner = { has_character_modifier = bloodhunger3_modifier }
					scope:herd_owner = { has_character_modifier = bloodhunger4_modifier }
