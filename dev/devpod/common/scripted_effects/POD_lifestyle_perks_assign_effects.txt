##########   CORE MACROS   ##########
#####################################

POD_assign_perk = {
	if = {
		limit = { NOT = { has_perk = $PERK$ } }
		add_perk = $PERK$
	}
}

POD_copy_perk_to_clone = {
	if = {
		limit = {
			scope:cloned_character = {
				has_perk = $PERK$
			}
		}
		add_perk = $PERK$
	}
}

POD_assign_lifestyle = {
	POD_perk_injector_$LIFESTYLE$ = {
		EFFECT = POD_assign_perk
	}
}

POD_assign_lifestyle_with_tooltip = {
	POD_save_tooltip = { LOC = $LIFESTYLE$ }
 	custom_tooltip = POD_assign_perk_tree_tt
	hidden_effect = {
		POD_perk_injector_$LIFESTYLE$ = {
			EFFECT = POD_assign_perk
		}
	}
}

POD_assign_lifestyle_with_custom_tooltip = {
	custom_tooltip = $TOOLTIP$
	hidden_effect = {
		POD_perk_injector_$LIFESTYLE$ = {
			EFFECT = POD_assign_perk
		}
	}
}

POD_copy_lifestyle_to_clone = {
	POD_perk_injector_$LIFESTYLE$ = {
		EFFECT = POD_copy_perk_to_clone
	}
}

POD_assign_elder_perk = {
	if = {
		limit = {
			scope:minimum_generation <= $GEN$
			NOT = { has_perk = $PERK$ }
		}
		add_perk = $PERK$
	}
}

POD_assign_elder_lifestyle = {
	save_scope_value_as = {
		name  = minimum_generation
		value = $GEN$
	}
	POD_perk_injector_$LIFESTYLE$_elder = {
		EFFECT = POD_assign_elder_perk
	}
}


POD_assign_perk_and_decrement_counter = {
	if = {
		limit = {
			scope:perk_counter > 0
			NOT = { has_perk = $PERK$ }
		}
		add_perk = $PERK$
		POD_decrement_scope_value = { SCOPE = perk_counter }
	}
}

POD_gain_multiple_perks_in_lifestyle = {
	save_scope_value_as = {
		name  = perk_counter
		value = $AMOUNT$
	}
	POD_perk_injector_$LIFESTYLE$ = {
		EFFECT = POD_assign_perk_and_decrement_counter
	}
}

POD_gain_next_perk_in_lifestyle = {
	POD_gain_multiple_perks_in_lifestyle = {
		LIFESTYLE = $LIFESTYLE$
		AMOUNT = 1
	}
}
