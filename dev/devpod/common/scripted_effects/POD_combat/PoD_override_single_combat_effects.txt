##################################################
# POD COMBAT Single Combat Effect Override
##################################################
# Intercept enabled, non-fixed duels with exactly one PC. Other valid duels use Part 4;
# self-matches invalidate in Part 2. Parameters match PoD's original effect.
# First fights defer through pod_combat.0200; pod_combat.9999 handles outcomes and follow-up.
# Wounds are written live. Scope hand-off: COMBAT_REFERENCE.info section 10.
##################################################

configure_start_single_combat_effect = {
	# ── Part 1: PoD-standard scope setup (identical to original) ──
	$SC_INITIATOR$ = { save_scope_as = sc_initiator }
	$SC_ATTACKER$ = { save_scope_as = sc_attacker }
	$SC_DEFENDER$ = { save_scope_as = sc_defender }
	if = {
		limit = { exists = scope:sc_finished }
		clear_saved_scope = sc_finished
	}
	save_scope_value_as = {
		name = follow_up_event
		value = event_id:$OUTPUT_EVENT$
	}
	save_scope_value_as = {
		name = invalidation_event
		value = event_id:$INVALIDATION_EVENT$
	}
	# Hoisted before the per-branch saves so the Part-3 gate can read
	# scope:fixed (FIXED duels route to vanilla, which honors the winner).
	save_scope_value_as = { name = fixed value = flag:$FIXED$ }

	# ── Part 2: Self-match invalidation (same as original) ──
	if = {
		limit = { scope:sc_attacker = scope:sc_defender }
		scope:sc_initiator = {
			trigger_event = { saved_event_id = scope:invalidation_event }
		}
		error_log = debug_log.single_combat_setup.attacker_and_defender_same_person
		debug_log = debug_log.single_combat_setup.attacker_and_defender_same_person.notice
		debug_log_scopes = yes
	}

	# -- Part 3: PC-involved branch - launch the Combat-PoD minigame --
	# Fall-through cases (rule off, FIXED, PC-vs-PC) are covered in the
	# file header and the limit block below.
	else_if = {
		limit = {
			has_game_rule = pod_combat_enabled
			# Exactly one side is PC. PC-vs-PC duels can't run the
			# minigame - its UI is single-perspective and only one
			# PC would see it. Route them to the vanilla body instead.
			OR = {
				AND = {
					scope:sc_attacker = { is_ai = no }
					scope:sc_defender = { is_ai = yes }
				}
				AND = {
					scope:sc_attacker = { is_ai = yes }
					scope:sc_defender = { is_ai = no }
				}
			}
			# Check actual seating: player's controlling proxy character hold a session without occupying a combat slot.
			scope:sc_attacker = { NOT = { pod_combat_is_seated_trigger = yes } }
			scope:sc_defender = { NOT = { pod_combat_is_seated_trigger = yes } }
			# Bypass interception while an intro is pending to avoid launching a second minigame underneath it.
			scope:sc_attacker = { NOT = { has_variable = pod_combat_intro_pending } }
			scope:sc_defender = { NOT = { has_variable = pod_combat_intro_pending } }
			# Don't intercept story-predetermined duels - PoD's vanilla
			# body in Part 4 already honors FIXED correctly.
			NOT = { scope:fixed = flag:sc_attacker }
			NOT = { scope:fixed = flag:sc_defender }
		}
		hidden_effect = { remove_single_combat_info_effect = yes }
		scope:sc_attacker = { set_variable = { name = engaged_in_single_combat value = yes } }
		scope:sc_defender = { set_variable = { name = engaged_in_single_combat value = yes } }

		# Preserve the rules as scope values for any downstream code that
		# inspects them (hold-court/duel event descriptions, etc.).
		save_scope_value_as = { name = fatality value = flag:$FATALITY$ }
		save_scope_value_as = { name = fixed    value = flag:$FIXED$ }
		save_scope_value_as = { name = locale   value = flag:$LOCALE$ }

		# Capture terrain before combat can remove the defender's location; guard landless principals.
		if = {
			limit = { scope:locale = flag:terrain_scope }
			if = {
				limit = { exists = scope:sc_defender.location }
				scope:sc_defender.location = { save_scope_as = background_terrain_scope }
			}
			else_if = {
				limit = { exists = scope:sc_attacker.location }
				scope:sc_attacker.location = { save_scope_as = background_terrain_scope }
			}
		}


		# Launch in PC scope on either side. Part 2 already handles self-match invalidation.
		if = {
			limit = { scope:sc_attacker = { is_ai = no } }
			scope:sc_attacker = {
				pod_combat_sc_intro_or_launch_effect = {
					OPPONENT = scope:sc_defender
					OUTPUT_EVENT = $OUTPUT_EVENT$
					FATALITY = $FATALITY$
					FIXED = $FIXED$
					LOCALE = $LOCALE$
				}
			}
		}
		else = {
			# PC is the defender. (PC-vs-PC duels are excluded by the
			# limit clause above and routed to vanilla instead.)
			scope:sc_defender = {
				pod_combat_sc_intro_or_launch_effect = {
					OPPONENT = scope:sc_attacker
					OUTPUT_EVENT = $OUTPUT_EVENT$
					FATALITY = $FATALITY$
					FIXED = $FIXED$
					LOCALE = $LOCALE$
				}
			}
		}
	}

	####### PART 4: POD FALL-THROUGH
	# PoD's attacker!=defender body, with weapon/shirtlessness setup moved to Part 1.
	else = {
		hidden_effect = { remove_single_combat_info_effect = yes }
		scope:sc_attacker = {
			set_variable = {
				name = engaged_in_single_combat
				value = yes
			}
		}
		scope:sc_defender = {
			set_variable = {
				name = engaged_in_single_combat
				value = yes
			}
		}
		single_combat_apply_default_shirtlessness_effect = {
			ATTACKER = scope:sc_attacker
			DEFENDER = scope:sc_defender
		}
		save_scope_value_as = {
			name = fatality
			value = flag:$FATALITY$
		}
		save_scope_value_as = {
			name = fixed
			value = flag:$FIXED$
		}
		save_scope_value_as = {
			name = locale
			value = flag:$LOCALE$
		}
		if = {
			limit = { scope:locale = flag:terrain_scope }
			scope:sc_defender.location = { save_scope_as = background_terrain_scope }
		}
		scope:sc_attacker = {
			set_variable = {
				name = sc_attacker_injury_bonus
				value = starting_injury_bonus
			}
		}
		scope:sc_defender = {
			set_variable = {
				name = sc_defender_injury_bonus
				value = starting_injury_bonus
			}
			set_variable = {
				name = success_threshold
				value = starting_success_threshold
			}
		}
		scope:sc_attacker = {
			set_variable = {
				name = sc_attacker_injury_risk_score
				value = 0
			}
			set_variable = {
				name = sc_attacker_duel_success_score
				value = 0
			}
			set_local_variable = {
				name = current_duel_edge_buff
				value = 0
			}
			set_local_variable = {
				name = current_duel_edge_debuff
				value = 0
			}
			set_signature_weapon_effect = yes
		}
		scope:sc_defender = {
			set_variable = {
				name = sc_defender_injury_risk_score
				value = 0
			}
			set_variable = {
				name = sc_defender_duel_success_score
				value = 0
			}
			set_local_variable = {
				name = current_duel_edge_buff
				value = 0
			}
			set_local_variable = {
				name = current_duel_edge_debuff
				value = 0
			}
			set_variable = {
				name = current_round
				value = 1
			}
			set_signature_weapon_effect = yes
		}
		# PoD additions (warform + knight discipline buffs).
		scope:sc_attacker = {
			if = {
				limit = {
					has_trait = fera
					is_ai = yes
					POD_can_shapeshift_trigger = yes
				}
				POD_shapeshift_fera_warform_effect = yes
			}
			if = {
				limit = { POD_is_exalted_trigger = yes is_ai = yes }
				POD_select_war_charm_loadout_effect = yes
			}
			POD_enemy_knight_set_discipline_battle_effect = { CASTER = scope:sc_defender }
			POD_ally_knight_set_discipline_battle_effect  = { CASTER = scope:sc_attacker }
		}
		scope:sc_defender = {
			if = {
				limit = {
					has_trait = fera
					is_ai = yes
					POD_can_shapeshift_trigger = yes
				}
				POD_shapeshift_fera_warform_effect = yes
			}
			if = {
				limit = { POD_is_exalted_trigger = yes is_ai = yes }
				POD_select_war_charm_loadout_effect = yes
			}
			POD_enemy_knight_set_discipline_battle_effect = { CASTER = scope:sc_attacker }
			POD_ally_knight_set_discipline_battle_effect  = { CASTER = scope:sc_defender }
		}
		hidden_effect = {
			scope:sc_attacker = { force_character_skill_recalculation = yes }
			scope:sc_defender = { force_character_skill_recalculation = yes }
		}
		scope:sc_defender = { trigger_event = single_combat.0001 }
	}
}

##################################################
# Combat-PoD minigame launcher
##################################################

pod_combat_sc_launch_minigame_effect = {
	# PC scope. $OPPONENT$ = enemy leader; $OUTPUT_EVENT$ = follow-up on initiator;
	# $FATALITY$/$FIXED$/$LOCALE$ = literal rule tokens. Invalidation is handled before launch.
	# Preserve the caller's scope table until create_duel opens the shell.
	# Hide setup and bounded AI loops from the initiating option's tooltip.
	custom_tooltip = pod_combat_duel_begins_tt
	hidden_effect = {
		POD_combat_create_duel_effect = {
			INITIATOR    = scope:sc_initiator
			ATTACKER     = scope:sc_attacker
			DEFENDER     = scope:sc_defender
			FATALITY     = $FATALITY$
			FIXED        = $FIXED$
			LOCALE       = $LOCALE$
			OUTPUT_EVENT = $OUTPUT_EVENT$
		}
		POD_combat_duel_add_enemy_effect = { CHARACTER = $OPPONENT$ }
		POD_combat_finalize_duel_effect = yes
	}
}

##################################################
# First-fight gate - intro popup or straight to the launcher
##################################################

pod_combat_sc_intro_or_launch_effect = {
	# PC scope, with the launcher's parameters. Defer the first fight so loadouts can be edited before combat.
	# The intro preserves caller scopes; Begin clears the pending marker on success or invalidation.
	# Post-launch title/team-name setters cannot reach a deferred fight, so its labels use defaults.
	# Both branches share one tooltip line.
	custom_tooltip = pod_combat_duel_begins_tt
	hidden_effect = {
		if = {
			limit = { NOT = { has_character_flag = pod_combat_intro_event_flag } }
			add_character_flag = pod_combat_intro_event_flag
			set_variable = { name = pod_combat_intro_pending value = yes }
			trigger_event = { id = pod_combat.0200 }
		}
		else = {
			pod_combat_sc_launch_minigame_effect = {
				OPPONENT     = $OPPONENT$
				OUTPUT_EVENT = $OUTPUT_EVENT$
				FATALITY     = $FATALITY$
				FIXED        = $FIXED$
				LOCALE       = $LOCALE$
			}
		}
	}
}
