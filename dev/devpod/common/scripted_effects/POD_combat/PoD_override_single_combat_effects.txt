##################################################
# Combat-PoD — Single Combat Effect Override
##################################################
# Redirects PoD/vanilla duels to the Combat-PoD minigame when exactly one
# side is the PC, the pod_combat_enabled rule is on, and FIXED is unset. All
# other cases fall through to PoD's original body (Part 4): NPC-vs-NPC,
# PC-vs-PC (minigame UI is single-perspective), rule off, and FIXED duels
# (vanilla honors the story-specified winner). Params match the original.
#
# Post-combat flow lives in pod_combat.9999 (reads the PC's pod_combat_sc_context
# pointer to restore scopes + fire the follow-up). Wounds are written LIVE during
# move execution, so end-of-combat applies death-only outcomes; full handoff in
# COMBAT_REFERENCE.info.
##################################################

configure_start_single_combat_effect = {
	# ── Part 1: PoD-standard scope setup (identical to original) ──
	$SC_INITIATOR$ = { save_scope_as = sc_initiator }
	$SC_ATTACKER$ = { save_scope_as = sc_attacker }
	$SC_DEFENDER$ = { save_scope_as = sc_defender }
	if = {
		limit = { exists = scope:sc_finished }
		clear_saved_scope = sc_finished
	}
	save_scope_value_as = {
		name = follow_up_event
		value = event_id:$OUTPUT_EVENT$
	}
	save_scope_value_as = {
		name = invalidation_event
		value = event_id:$INVALIDATION_EVENT$
	}
	# Hoisted before the per-branch saves so the Part-3 gate can read
	# scope:fixed (FIXED duels route to vanilla, which honors the winner).
	save_scope_value_as = { name = fixed value = flag:$FIXED$ }

	# ── Part 2: Self-match invalidation (same as original) ──
	if = {
		limit = { scope:sc_attacker = scope:sc_defender }
		scope:sc_initiator = {
			trigger_event = { saved_event_id = scope:invalidation_event }
		}
		error_log = debug_log.single_combat_setup.attacker_and_defender_same_person
		debug_log = debug_log.single_combat_setup.attacker_and_defender_same_person.notice
		debug_log_scopes = yes
	}

	# -- Part 3: PC-involved branch - launch the Combat-PoD minigame --
	# Fall-through cases (rule off, FIXED, PC-vs-PC) are covered in the
	# file header and the limit block below.
	else_if = {
		limit = {
			has_game_rule = pod_combat_enabled
			# Exactly one side is PC. PC-vs-PC duels can't run the
			# minigame — its UI is single-perspective and only one
			# PC would see it. Route them to the vanilla body instead.
			OR = {
				AND = {
					scope:sc_attacker = { is_ai = no }
					scope:sc_defender = { is_ai = yes }
				}
				AND = {
					scope:sc_attacker = { is_ai = yes }
					scope:sc_defender = { is_ai = no }
				}
			}
			# Guard against re-entry if a previous combat somehow left state.
			scope:sc_attacker = { NOT = { has_variable = pod_combat_active } }
			scope:sc_defender = { NOT = { has_variable = pod_combat_active } }
			# Don't intercept story-predetermined duels — PoD's vanilla
			# body in Part 4 already honors FIXED correctly.
			NOT = { scope:fixed = flag:sc_attacker }
			NOT = { scope:fixed = flag:sc_defender }
		}
		hidden_effect = { remove_single_combat_info_effect = yes }
		scope:sc_attacker = { set_variable = { name = engaged_in_single_combat value = yes } }
		scope:sc_defender = { set_variable = { name = engaged_in_single_combat value = yes } }

		# Preserve the rules as scope values for any downstream code that
		# inspects them (hold-court/duel event descriptions, etc.).
		save_scope_value_as = { name = fatality value = flag:$FATALITY$ }
		save_scope_value_as = { name = fixed    value = flag:$FIXED$ }
		save_scope_value_as = { name = locale   value = flag:$LOCALE$ }

		# Fera shapeshifting and discipline combat buffs are handled inside
		# pod_combat_init_slot_from_character_effect for all combatants.
		# The PC is excluded from auto-shift (they may prefer human form
		# or have already shifted manually).

		# One branch per side so the duel builder runs in the PC's scope.
		# INVALIDATION_EVENT isn't passed: Part 2 already fired it on the
		# attacker==defender self-match, and the launcher doesn't reuse it.
		if = {
			limit = { scope:sc_attacker = { is_ai = no } }
			scope:sc_attacker = {
				pod_combat_sc_launch_minigame_effect = {
					OPPONENT = scope:sc_defender
					OUTPUT_EVENT = $OUTPUT_EVENT$
					FATALITY = $FATALITY$
					FIXED = $FIXED$
					LOCALE = $LOCALE$
				}
			}
		}
		else = {
			# PC is the defender. (PC-vs-PC duels are excluded by the
			# limit clause above and routed to vanilla instead.)
			scope:sc_defender = {
				pod_combat_sc_launch_minigame_effect = {
					OPPONENT = scope:sc_attacker
					OUTPUT_EVENT = $OUTPUT_EVENT$
					FATALITY = $FATALITY$
					FIXED = $FIXED$
					LOCALE = $LOCALE$
				}
			}
		}
	}

	# ────────────────────────────────────────────────────────────────────
	# Part 4: Fall-through - PoD's original duel body, ADAPTED. Verbatim port of
	# PoD's attacker!=defender clause with two deltas:
	#   1. Both POD_shapeshift_fera_warform_effect sites add an is_ai=yes guard so
	#      PC-side fera don't auto-shapeshift in this fallback.
	#   2. Signature-weapon / shirtlessness setup reordered to fold with the
	#      Part-1 scope hoist.
	# Fires for every non-minigame case (see header).
	# ────────────────────────────────────────────────────────────────────
	else = {
		hidden_effect = { remove_single_combat_info_effect = yes }
		scope:sc_attacker = {
			set_variable = {
				name = engaged_in_single_combat
				value = yes
			}
		}
		scope:sc_defender = {
			set_variable = {
				name = engaged_in_single_combat
				value = yes
			}
		}
		single_combat_apply_default_shirtlessness_effect = {
			ATTACKER = scope:sc_attacker
			DEFENDER = scope:sc_defender
		}
		save_scope_value_as = {
			name = fatality
			value = flag:$FATALITY$
		}
		save_scope_value_as = {
			name = fixed
			value = flag:$FIXED$
		}
		save_scope_value_as = {
			name = locale
			value = flag:$LOCALE$
		}
		if = {
			limit = { scope:locale = flag:terrain_scope }
			scope:sc_defender.location = { save_scope_as = background_terrain_scope }
		}
		scope:sc_attacker = {
			set_variable = {
				name = sc_attacker_injury_bonus
				value = starting_injury_bonus
			}
		}
		scope:sc_defender = {
			set_variable = {
				name = sc_defender_injury_bonus
				value = starting_injury_bonus
			}
			set_variable = {
				name = success_threshold
				value = starting_success_threshold
			}
		}
		scope:sc_attacker = {
			set_variable = {
				name = sc_attacker_injury_risk_score
				value = 0
			}
			set_variable = {
				name = sc_attacker_duel_success_score
				value = 0
			}
			set_local_variable = {
				name = current_duel_edge_buff
				value = 0
			}
			set_local_variable = {
				name = current_duel_edge_debuff
				value = 0
			}
			set_signature_weapon_effect = yes
		}
		scope:sc_defender = {
			set_variable = {
				name = sc_defender_injury_risk_score
				value = 0
			}
			set_variable = {
				name = sc_defender_duel_success_score
				value = 0
			}
			set_local_variable = {
				name = current_duel_edge_buff
				value = 0
			}
			set_local_variable = {
				name = current_duel_edge_debuff
				value = 0
			}
			set_variable = {
				name = current_round
				value = 1
			}
			set_signature_weapon_effect = yes
		}
		# PoD bolt-ons (warform + knight discipline buffs).
		scope:sc_attacker = {
			if = {
				limit = {
					has_trait = fera
					is_ai = yes
					POD_can_shapeshift_trigger = yes
				}
				POD_shapeshift_fera_warform_effect = yes
			}
			POD_enemy_knight_set_discipline_battle_effect = { CASTER = scope:sc_defender }
			POD_ally_knight_set_discipline_battle_effect  = { CASTER = scope:sc_attacker }
		}
		scope:sc_defender = {
			if = {
				limit = {
					has_trait = fera
					is_ai = yes
					POD_can_shapeshift_trigger = yes
				}
				POD_shapeshift_fera_warform_effect = yes
			}
			POD_enemy_knight_set_discipline_battle_effect = { CASTER = scope:sc_attacker }
			POD_ally_knight_set_discipline_battle_effect  = { CASTER = scope:sc_defender }
		}
		hidden_effect = {
			scope:sc_attacker = { force_character_skill_recalculation = yes }
			scope:sc_defender = { force_character_skill_recalculation = yes }
		}
		scope:sc_defender = { trigger_event = single_combat.0001 }
	}
}

##################################################
# Combat-PoD minigame launcher — called from configure_start_single_combat_effect
##################################################
#
# Runs in the PC's scope; builds the POD_combat_sc_context story (sc_*
# refs, rule flags, follow-up id, parent aliases) and hands off to the
# duel builder. pod_combat.9999 reads it back at combat end.
# Param contract is documented on pod_combat_sc_launch_minigame_effect below.
##################################################

##################################################
# Parent-event alias stash / restore
##################################################
#
# The duel system's parent events routinely save custom scope aliases
# (scope:spirit, scope:actor, scope:my_rival, scope:og_root, ...) that
# their follow-up events then read. Under the original vanilla duel
# chain those scopes propagate naturally through the event graph. The
# minigame's scripted-GUI execute boundary breaks that chain, so unless
# we capture the parent's named scopes at launch and restore them in
# pod_combat.9999, the follow-up event fires with an empty scope table
# and throws "Undefined event target" errors (see POD_umbra_expedition
# and friends).
#
# This effect is called once at launch (inside the parent event's scope
# chain — root scope is the parent event's actor / PC), and writes each
# alias as a story variable on scope:sc_ctx (the POD_combat_sc_context
# story created by the launcher). Any alias that is not present in the
# parent scope is silently skipped. The matching restore block in
# pod_combat.9999 rebuilds the scopes from those story vars.
#
# All writes go to scope:sc_ctx (the context story). The launcher must
# save_scope_as the new context story BEFORE calling this effect so
# scope:sc_ctx is bound. Each branch flips into the story scope to
# write — set_variable inside scope:sc_ctx attaches the variable to the
# story, not to the current actor character.
#
# Whitelist coverage
# Re-audit on  PoD update: grep PoD for new
# configure_start_single_combat_effect callers and add their saved
# scopes here if they reference fresh aliases in the follow-up event.
#   Characters (PoD): spirit, alastor, wyrm_champion, vampire_attacker
#   Characters (van): actor, parent, aggressor_vassal, distant_knight,
#                     snitch, strong_leader, warrior_monk, attacked_rival,
#                     my_rival, rival_burner, rival_on_sight, rivals_child,
#                     og_root, root_for_loc, recidivist
#   Artifacts:        shadowflame, burned_artifact
#   Schemes:          scheme  (POD_diablerize.600 reads scope:scheme to
#                              end_scheme after the duel resolves)
#   Scheme actors:    owner   (scheme owner / action initiator; widely
#                              used by vanilla `murder_success_kill_effect`
#                              and its callees, which pass MURDERER =
#                              scope:owner down into unknown_murder_effect /
#                              pod_death. Missing scope:owner cascades into
#                              scope:pod_killer errors because pod_death does
#                              `$killer$ = { save_temporary_scope_as =
#                              pod_killer }` and $killer$ resolves from it.)
#
# Still NOT stashed (add as they surface):
#   - scope:vassal_war  — used by epidemic_events_2.txt, but that
#     file's duel call doesn't pass OUTPUT_EVENT, so it isn't on
#     the minigame intercept path. Add if PoD updates wire it as
#     a duel follow-up scope.
#   - scope:laamp_title — only a single read in
#     ep3_laamp_flavour_ewan_events.txt, and that read isn't in
#     a duel follow-up event. Add if it surfaces.
#   - scope:fp2_*_gate  — only `exists` checks in
#     fp2_struggle_events.txt; not currently in any duel follow-up.
#     Raw / non-typed scope values would also need a different
#     detection mechanism than `exists = scope:X`.
#   - scope:duel_legal  — read by ep3_laamp_flavour_ewan.2092
#     (a real duel follow-up), BUT grep across CK3/PoD references
#     finds no `save_scope_as = duel_legal` anywhere — the scope
#     is never actually set, so the check
#     `NOT = { exists = scope:duel_legal }` is dead defense and
#     stashing it would capture nothing. Add if a future caller
#     starts setting it.
##################################################

pod_combat_sc_stash_parent_aliases_effect = {
	# Character aliases — each test is independent; missing scopes are skipped.
	# Each write switches into scope:sc_ctx (the context story) so the
	# variable lands on the story, not on the parent-scope character.
	#
	# `scope:X ?= { ... }` is CK3's safe-scope-navigation operator: it
	# switches into scope:X and runs the body if scope:X exists, and
	# is a silent no-op otherwise. Inside the body, saved scopes
	# (including scope:X itself) are still globally resolvable, so the
	# `value = scope:X` read on the right-hand side still works after
	# the scope switch. 
	scope:spirit           ?= { scope:sc_ctx = { set_variable = { name = alias_spirit           value = scope:spirit } } }
	scope:alastor          ?= { scope:sc_ctx = { set_variable = { name = alias_alastor          value = scope:alastor } } }
	scope:wyrm_champion    ?= { scope:sc_ctx = { set_variable = { name = alias_wyrm_champion    value = scope:wyrm_champion } } }
	scope:vampire_attacker ?= { scope:sc_ctx = { set_variable = { name = alias_vampire_attacker value = scope:vampire_attacker } } }
	scope:actor            ?= { scope:sc_ctx = { set_variable = { name = alias_actor            value = scope:actor } } }
	scope:parent           ?= { scope:sc_ctx = { set_variable = { name = alias_parent           value = scope:parent } } }
	scope:aggressor_vassal ?= { scope:sc_ctx = { set_variable = { name = alias_aggressor_vassal value = scope:aggressor_vassal } } }
	scope:defender_vassal  ?= { scope:sc_ctx = { set_variable = { name = alias_defender_vassal  value = scope:defender_vassal } } }
	scope:distant_knight   ?= { scope:sc_ctx = { set_variable = { name = alias_distant_knight   value = scope:distant_knight } } }
	scope:snitch           ?= { scope:sc_ctx = { set_variable = { name = alias_snitch           value = scope:snitch } } }
	scope:strong_leader    ?= { scope:sc_ctx = { set_variable = { name = alias_strong_leader    value = scope:strong_leader } } }
	scope:warrior_monk     ?= { scope:sc_ctx = { set_variable = { name = alias_warrior_monk     value = scope:warrior_monk } } }
	scope:attacked_rival   ?= { scope:sc_ctx = { set_variable = { name = alias_attacked_rival   value = scope:attacked_rival } } }
	scope:my_rival         ?= { scope:sc_ctx = { set_variable = { name = alias_my_rival         value = scope:my_rival } } }
	scope:rival_burner     ?= { scope:sc_ctx = { set_variable = { name = alias_rival_burner     value = scope:rival_burner } } }
	scope:rival_on_sight   ?= { scope:sc_ctx = { set_variable = { name = alias_rival_on_sight   value = scope:rival_on_sight } } }
	scope:rivals_child     ?= { scope:sc_ctx = { set_variable = { name = alias_rivals_child     value = scope:rivals_child } } }
	scope:og_root          ?= { scope:sc_ctx = { set_variable = { name = alias_og_root          value = scope:og_root } } }
	scope:root_for_loc     ?= { scope:sc_ctx = { set_variable = { name = alias_root_for_loc     value = scope:root_for_loc } } }
	scope:recidivist       ?= { scope:sc_ctx = { set_variable = { name = alias_recidivist       value = scope:recidivist } } }
	scope:owner            ?= { scope:sc_ctx = { set_variable = { name = alias_owner            value = scope:owner } } }
	# Artifact aliases — stored the same way as character refs.
	scope:shadowflame      ?= { scope:sc_ctx = { set_variable = { name = alias_shadowflame      value = scope:shadowflame } } }
	scope:burned_artifact  ?= { scope:sc_ctx = { set_variable = { name = alias_burned_artifact  value = scope:burned_artifact } } }
	# Scheme scope — typed scope objects store fine in story variables.
	# POD_diablerize.600 runs end_scheme on scope:scheme after the duel
	# resolves; without this stash, its option tooltips throw "Undefined
	# event target 'scheme'" and the final end_scheme call fails.
	scope:scheme           ?= { scope:sc_ctx = { set_variable = { name = alias_scheme           value = scope:scheme } } }
	# Title alias — vanilla hold_court.8210 (the vassal-rebellion
	# legal-duel chain) saves scope:vassal_claim before launching the
	# duel; hold_court.8211 (OUTPUT_EVENT) reads it twice to transfer
	# the title from the loser to scope:sc_victor. The PC can be
	# either party in this chain (aggressor_vassal or defender_vassal),
	# so the minigame intercepts the duel and the follow-up fires via
	# pod_combat.9999. Without this stash, hold_court.8211 throws
	# "Undefined event target 'vassal_claim'" and the title transfer
	# silently fails. Typed scope (a title), stores fine in a story
	# variable.
	scope:vassal_claim     ?= { scope:sc_ctx = { set_variable = { name = alias_vassal_claim     value = scope:vassal_claim } } }
}

pod_combat_sc_launch_minigame_effect = {
	# Builds a 1v1 (PC vs OPPONENT) through the duel builder, then
	# finalizes to open the combat window. Runs in the PC's scope.
	#
	# POD_combat_create_duel_effect owns the duel story: it creates the
	# POD_combat_sc_context story on Charon, snapshots the sc_* parties +
	# rule flags + follow-up event id onto it, stashes the parent-event
	# aliases (pod_combat_sc_stash_parent_aliases_effect), and pins the
	# pointer var on the PC. pod_combat.9999 reads through that pointer at
	# combat end to restore scopes and fire the follow-up. See
	# COMBAT_REFERENCE.info §7 for the new-object-wrap rationale.
	#
	# Parameters:
	#   $OPPONENT$     — the non-PC combatant (seated as the enemy/0 leader)
	#   $OUTPUT_EVENT$ — follow-up event fired on the initiator at combat end
	#   $FATALITY$ / $FIXED$ / $LOCALE$ — duel rule flags
	# (INVALIDATION_EVENT is handled in the override's Part 2 before the
	#  launcher runs, so it isn't a launcher parameter.)
	POD_combat_create_duel_effect = {
		INITIATOR    = scope:sc_initiator
		ATTACKER     = scope:sc_attacker
		DEFENDER     = scope:sc_defender
		FATALITY     = $FATALITY$
		FIXED        = $FIXED$
		LOCALE       = $LOCALE$
		OUTPUT_EVENT = $OUTPUT_EVENT$
	}
	POD_combat_duel_add_enemy_effect = { CHARACTER = $OPPONENT$ }
	POD_combat_finalize_duel_effect = yes
}
