# Character scope (the new spawn). The roster launcher runs the BUILD effect it was given on each
# spawn after create_character and before seating. 
#
# Sub-identity comes from scopes the caller saves before the launcher runs:
#   trait:brujah = { save_scope_as = pod_combat_boss_clan }
#   faith:getoffenris = { save_scope_as = pod_combat_boss_tribe }
#   trait:ahroun = { save_scope_as = pod_combat_boss_auspice }
# An absent scope falls through to PoD's own random picker for that splat.

# Every launcher entry passes a BUILD; this one does nothing. TIER is stored so the argument is read.
POD_combat_boss_build_none_effect = {
	save_scope_value_as = { name = pod_combat_boss_build_tier value = $TIER$ }
}

# T1: a typical elder, the clan's three advanced disciplines (the template is 600 years old, so PoD's
# discipline initializer picks the advanced trio). T2: a great methuselah, two more advanced
# disciplines with Fortitude first for soak, the elder trees at GEN 4.
POD_combat_boss_build_vampire_effect = {
	save_scope_value_as = { name = pod_combat_boss_build_tier value = $TIER$ }
	if = {
		limit = { exists = scope:pod_combat_boss_clan }
		add_trait = scope:pod_combat_boss_clan
	}
	else = { POD_character_vampire_clan_initialization_effect = yes }
	# PoD's NPC vampire sequence: clan traits, faith by clan and location, clan looks.
	POD_character_vampire_clan_traits_effect = yes
	if = {
		limit = {
			is_a_cultist = no
			NOT = { religion = { is_in_family = rf_kindred } }
		}
		POD_character_vampire_faith_initialization_effect = yes
	}
	POD_character_vampire_clan_special_effect = yes
	if = {
		limit = { scope:pod_combat_boss_build_tier = 2 }
		add_trait = vampgeneration5
	}
	else = { add_trait = vampgeneration8 }
	POD_character_vampire_discipline_initialization_effect = yes
	if = {
		limit = { scope:pod_combat_boss_build_tier = 2 }
		if = {
			limit = { NOT = { has_trait = fortitudeadvanced } }
			add_trait = fortitudeadvanced
		}
		else = { POD_caitiff_random_discipline_initialization_effect = { ADVANCED = yes } }
		POD_caitiff_random_discipline_initialization_effect = { ADVANCED = yes }
	}
	POD_combat_initialize_boss_effect = { TIER = $TIER$ }
	POD_combat_boss_finish_creation_effect = yes
	if = {
		limit = { scope:pod_combat_boss_build_tier = 2 }
		POD_combat_boss_grant_elder_trees_effect = { GEN = 4 }
	}
	save_scope_value_as = { name = pod_combat_boss_blood_potency value = pod_combat_boss_blood_potency_value }
	POD_set_bloodpotency_effect = { POTENCY = scope:pod_combat_boss_blood_potency }
	POD_combat_boss_prowess_topup_effect = yes
}

# T1: an Adren (rank 3), the first three breed gifts and the whole auspice tree. T2: a Legend
# (rank 6, prestige level 4 for the rank track's prowess), every breed, auspice and tribal gift.
# PoD's werewolf initializer fills breed, auspice and tribe only where they are missing.
POD_combat_boss_build_werewolf_effect = {
	save_scope_value_as = { name = pod_combat_boss_build_tier value = $TIER$ }
	if = {
		limit = { exists = scope:pod_combat_boss_tribe }
		set_character_faith = scope:pod_combat_boss_tribe
	}
	if = {
		limit = { exists = scope:pod_combat_boss_auspice }
		add_trait = scope:pod_combat_boss_auspice
	}
	POD_werewolf_initialization_effect = yes
	if = {
		limit = { scope:pod_combat_boss_build_tier = 2 }
		add_trait_xp = { trait = werewolf value = 100 }
	}
	else = { add_trait_xp = { trait = werewolf value = 40 } }
	if = {
		limit = { has_trait = beast_born }
		if = {
			limit = { scope:pod_combat_boss_build_tier = 2 }
			POD_assign_lifestyle = { LIFESTYLE = beastborngifts }
		}
		else = { POD_gain_multiple_perks_in_lifestyle = { AMOUNT = 3 LIFESTYLE = beastborngifts } }
	}
	else = {
		if = {
			limit = { scope:pod_combat_boss_build_tier = 2 }
			POD_assign_lifestyle = { LIFESTYLE = homidgifts }
		}
		else = { POD_gain_multiple_perks_in_lifestyle = { AMOUNT = 3 LIFESTYLE = homidgifts } }
	}
	POD_combat_boss_grant_auspice_tree_effect = yes
	if = {
		limit = { scope:pod_combat_boss_build_tier = 2 }
		POD_combat_boss_grant_werewolf_tribal_tree_effect = yes
	}
	# Crinos: +15 prowess and the form the innate moves gate on.
	POD_shapeshift_fera_warform_effect = yes
	POD_combat_initialize_boss_effect = { TIER = $TIER$ }
	POD_combat_boss_finish_creation_effect = yes
	POD_combat_boss_prowess_topup_effect = yes
}
