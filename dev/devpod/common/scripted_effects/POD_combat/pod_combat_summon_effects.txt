######################################################
######   POD COMBAT - MID-COMBAT COMBATANT ADDITION + SUMMONS
######################################################
# Seats a brand-new combatant into a LIVE fight (the duel builder only runs
# pre-combat). See COMBAT_REFERENCE.info for the slot / leader-pointer model.
#
#   pod_combat_add_ally_midfight_effect            = { CHARACTER }             -> player, now
#   pod_combat_add_enemy_midfight_effect           = { CHARACTER }             -> enemy, now
#   pod_combat_schedule_ally_reinforcement_effect  = { CHARACTER JOIN_ROUND }  -> player, at round N
#   pod_combat_schedule_enemy_reinforcement_effect = { CHARACTER JOIN_ROUND }  -> enemy, at round N
# add_* enters immediately; schedule_* enters at the start of JOIN_ROUND (a round <= the
# current one means now), seated by pod_combat_group_end_round_effect's per-round sweep.
# Team-specific entry points, not one { TEAM }: CK3 can't index the per-team counter /
# leader pointer / combatant-add effect by a parse-time token, so the caller resolves it.
# Every mid-fight seat auto-posts a named "<who> joins the fray" log row, so callers
# (reinforcements, summons, events) never wire their own arrival message.

###################
###### GENERAL: ADD A COMBATANT MID-FIGHT
###################

pod_combat_add_ally_midfight_effect = {
	# Host scope. $CHARACTER$ must be saved because init re-evaluates it after switching scope.
	# Finalize/log only when add_ally advances the team-size counter.
	if = {
		limit = {
			var:pod_combat_session ?= { has_variable = pod_combat_player_team_size }
			var:pod_combat_session ?= { has_variable = pod_combat_enemy_leader_slot }
			var:pod_combat_session.var:pod_combat_enemy_leader_slot ?= { has_variable = slot_char }
		}
		# Snapshot landing position BEFORE the add (add_ally seats at team_size then bumps it).
		save_scope_value_as = { name = pod_combat_summon_pos value = var:pod_combat_session.var:pod_combat_player_team_size }
		# Leader pointers retain position 0 after death. For the baseline, prefer the lowest-position
		# living opponent; retain the original leader if none survives.
		var:pod_combat_session.var:pod_combat_enemy_leader_slot.var:slot_char ?= {
			save_scope_as = pod_combat_summon_opp_leader
			save_scope_value_as = { name = pod_combat_summon_opp_prowess value = prowess }
		}
		if = {
			limit = { NOT = { var:pod_combat_session.var:pod_combat_enemy_leader_slot ?= { pod_combat_slot_char_is_alive_trigger = yes } } }
			var:pod_combat_session ?= {
				ordered_in_list = {
					variable = pod_combat_active_slots
					limit = {
						pod_combat_slot_is_enemy_trigger = yes
						pod_combat_slot_char_is_alive_trigger = yes
					}
					order_by = { value = 0 subtract = var:team_position }
					position = 0
					var:slot_char = {
						save_scope_as = pod_combat_summon_opp_leader
						save_scope_value_as = { name = pod_combat_summon_opp_prowess value = prowess }
					}
				}
			}
		}
		POD_combat_duel_add_ally_effect = { CHARACTER = $CHARACTER$ }
		# A capped add leaves the counter unchanged; do not finalize, log or consume that reinforcement.
		if = {
			limit = { var:pod_combat_session ?= { var:pod_combat_player_team_size > scope:pod_combat_summon_pos } }
			# Apply death/practice gates and snapshot wounds before the discipline battery deals damage.
			if = {
				limit = { pod_combat_uses_death_gate_trigger = yes }
				$CHARACTER$ = { add_character_flag = pod_combat_nonfatal_active }
			}
			# Practice: snapshot the late arrival's wounds so they restore at combat end.
			if = {
				limit = { pod_combat_is_practice_trigger = yes }
				$CHARACTER$ = { pod_combat_snapshot_one_practice_wounds_effect = yes }
			}
			pod_combat_init_one_midfight_slot_effect = { TEAM = player }
			pod_combat_push_system_log_named_effect = { MESSAGE_KEY = combatant_joins  CHARACTER = $CHARACTER$ }
		}
	}
}

pod_combat_add_enemy_midfight_effect = {
	# Enemy-team counterpart of pod_combat_add_ally_midfight_effect; opposing leader = PC.
	if = {
		limit = {
			var:pod_combat_session ?= { has_variable = pod_combat_enemy_team_size }
			var:pod_combat_session ?= { has_variable = pod_combat_pc_slot }
			var:pod_combat_session.var:pod_combat_pc_slot ?= { has_variable = slot_char }
		}
		save_scope_value_as = { name = pod_combat_summon_pos value = var:pod_combat_session.var:pod_combat_enemy_team_size }
		# Use the player leader, falling back to the lowest-position living teammate if dead.
		var:pod_combat_session.var:pod_combat_pc_slot.var:slot_char ?= {
			save_scope_as = pod_combat_summon_opp_leader
			save_scope_value_as = { name = pod_combat_summon_opp_prowess value = prowess }
		}
		if = {
			limit = { NOT = { var:pod_combat_session.var:pod_combat_pc_slot ?= { pod_combat_slot_char_is_alive_trigger = yes } } }
			var:pod_combat_session ?= {
				ordered_in_list = {
					variable = pod_combat_active_slots
					limit = {
						pod_combat_slot_is_player_team_trigger = yes
						pod_combat_slot_char_is_alive_trigger = yes
					}
					order_by = { value = 0 subtract = var:team_position }
					position = 0
					var:slot_char = {
						save_scope_as = pod_combat_summon_opp_leader
						save_scope_value_as = { name = pod_combat_summon_opp_prowess value = prowess }
					}
				}
			}
		}
		POD_combat_duel_add_enemy_effect = { CHARACTER = $CHARACTER$ }
		# Success gate - see the ally twin above for the rationale.
		if = {
			limit = { var:pod_combat_session ?= { var:pod_combat_enemy_team_size > scope:pod_combat_summon_pos } }
			# Death gate + practice wound snapshot for mid-fight seats, both ahead of
			# the seat init - see the ally twin above.
			if = {
				limit = { pod_combat_uses_death_gate_trigger = yes }
				$CHARACTER$ = { add_character_flag = pod_combat_nonfatal_active }
			}
			if = {
				limit = { pod_combat_is_practice_trigger = yes }
				$CHARACTER$ = { pod_combat_snapshot_one_practice_wounds_effect = yes }
			}
			pod_combat_init_one_midfight_slot_effect = { TEAM = enemy }
			pod_combat_push_system_log_named_effect = { MESSAGE_KEY = combatant_joins  CHARACTER = $CHARACTER$ }
		}
	}
}

pod_combat_init_one_midfight_slot_effect = {
	# $TEAM$ = player|enemy. Initialize before calculating hit chance.
	# Save CHARACTER before init re-evaluates it in character scope.
	# Mid-fight seats do not run the AI warform pass; discipline init still applies opposing-leader Delirium.
	var:pod_combat_session ?= {
		every_in_list = {
			variable = pod_combat_active_slots
			limit = {
				var:combat_team = flag:$TEAM$
				var:team_position = scope:pod_combat_summon_pos
			}
			var:slot_char = { save_scope_as = pod_combat_summon_slot_char }
			pod_combat_init_slot_from_character_effect = { CHARACTER = scope:pod_combat_summon_slot_char ENEMY_LEADER = scope:pod_combat_summon_opp_leader }
			pod_combat_calc_one_hit_chance_effect = { OPP_PROWESS = scope:pod_combat_summon_opp_prowess }
			set_variable = { name = slot_has_acted value = 1 }
			# Seed after init for the selector's unlock proof. Keep character scope switches last.
			if = {
				limit = {
					scope:pod_combat_summon_slot_char = {
						is_ai = yes
						pod_combat_equipped_count = 0
					}
				}
				scope:pod_combat_summon_slot_char = { POD_apply_ai_full_loadout = yes }
			}
		}
	}
}

###################
###### DEFERRED REINFORCEMENTS
###################

pod_combat_schedule_ally_reinforcement_effect = {
	# $CHARACTER$ (saved scope) joins the PLAYER team at $JOIN_ROUND$ (absolute round).
	# A round <= the current one seats immediately; a future round defers to the
	# per-round sweep in pod_combat_group_end_round_effect. Host (PC) scope.
	if = {
		limit = {
			var:pod_combat_session ?= {
				has_variable = pod_combat_round
				var:pod_combat_round < $JOIN_ROUND$
			}
		}
		pod_combat_defer_reinforcement_effect = { CHARACTER = $CHARACTER$  JOIN_ROUND = $JOIN_ROUND$  TEAM = player }
	}
	else = { pod_combat_add_ally_midfight_effect = { CHARACTER = $CHARACTER$ } }
}

pod_combat_schedule_enemy_reinforcement_effect = {
	# Enemy-team counterpart of pod_combat_schedule_ally_reinforcement_effect.
	if = {
		limit = {
			var:pod_combat_session ?= {
				has_variable = pod_combat_round
				var:pod_combat_round < $JOIN_ROUND$
			}
		}
		pod_combat_defer_reinforcement_effect = { CHARACTER = $CHARACTER$  JOIN_ROUND = $JOIN_ROUND$  TEAM = enemy }
	}
	else = { pod_combat_add_enemy_midfight_effect = { CHARACTER = $CHARACTER$ } }
}

pod_combat_defer_reinforcement_effect = {
	# Registers $CHARACTER$ to enter on $JOIN_ROUND$ (absolute round). Host scope.
	# Join round + team live on the character (they describe that character's
	# entry, and it may not be seated yet); the roster of who is waiting is the
	# fight's, and drives the per-round sweep in pod_combat_group_end_round_effect.
	$CHARACTER$ = {
		set_variable = { name = pod_combat_join_round value = $JOIN_ROUND$ }
		set_variable = { name = pod_combat_join_team  value = flag:$TEAM$ }
	}
	var:pod_combat_session ?= {
		add_to_variable_list = { name = pod_combat_pending_reinforcements target = $CHARACTER$ }
	}
}

pod_combat_process_pending_reinforcements_effect = {
	# Host scope. Seat due reinforcements before initiative resets acted flags.
	# Use two passes to avoid modifying the list during iteration.
	# Iterate in fight scope; re-enter reinf_host for seating.
	save_scope_as = reinf_host
	# ?=-bound: tooltip-build can reach this with no fight created yet, and the
	# body below is skipped wholesale when the pointer does not resolve.
	var:pod_combat_session ?= { save_temporary_scope_as = pod_fight }
	var:pod_combat_session ?= {
		if = {
			limit = { has_variable_list = pod_combat_pending_reinforcements }
			every_in_list = {
				variable = pod_combat_pending_reinforcements
				limit = { var:pod_combat_join_round <= scope:pod_fight.var:pod_combat_round }
				save_temporary_scope_as = due_char
				scope:pod_fight = { add_to_variable_list = { name = pod_combat_reinf_due target = scope:due_char } }
			}
			if = {
				limit = { has_variable_list = pod_combat_reinf_due }
				every_in_list = {
					variable = pod_combat_reinf_due
					save_temporary_scope_as = due_char
					scope:pod_fight = {
						remove_list_variable = { name = pod_combat_pending_reinforcements target = scope:due_char }
					}
					scope:reinf_host = {
						# Seat on the stored team. A reinforcement killed before its round is
						# just dropped (removed above, not seated).
						if = {
							limit = { scope:due_char = { is_alive = yes  var:pod_combat_join_team = flag:player } }
							pod_combat_add_ally_midfight_effect = { CHARACTER = scope:due_char }
						}
						else_if = {
							limit = { scope:due_char = { is_alive = yes  var:pod_combat_join_team = flag:enemy } }
							pod_combat_add_enemy_midfight_effect = { CHARACTER = scope:due_char }
						}
					}
					# is_alive: var mutation on a dead reference errors; a reinforcement
					# killed before its round was dropped above, its vars die with it.
					if = {
						limit = { scope:due_char = { is_alive = yes } }
						scope:due_char = {
							remove_variable = pod_combat_join_round
							remove_variable = pod_combat_join_team
						}
					}
				}
				clear_variable_list = pod_combat_reinf_due
				# Each reinforcement self-announces a named "joins the fray" row as it seats
				# (via pod_combat_add_ally_midfight_effect / pod_combat_add_enemy_midfight_effect),
				# so no batch arrival message.
			}
		}
	}
}

###################
###### MOVE-DRIVEN SUMMONS
###################

pod_combat_spawn_spirit_family_helper = {
	# Scope = summoner. Suppress PoD's strategic summon toast; combat posts its
	# own named arrival row. Explicitly disable is_summoned_spirit so combat
	# summons use the templates' base ranks instead of strategic caster scaling.
	save_scope_as = spawning_character
	save_scope_value_as = { name = spirit_spawn_notification value = no }
	save_scope_value_as = { name = is_summoned_spirit value = no }
	if = {
		limit = { $MOVE_STORY$ = { var:summon_recipe = flag:bane } }
		POD_spawn_bane_spirit_effect = yes
	}
	else_if = {
		limit = { $MOVE_STORY$ = { var:summon_recipe = flag:gaian_spirit } }
		POD_spawn_gaian_spirit_effect = yes
	}
	else_if = {
		limit = { $MOVE_STORY$ = { var:summon_recipe = flag:elemental_bane } }
		POD_spawn_elemental_bane_spirit_effect = yes
	}
	else = {
		debug_log = "pod_combat: non-spirit recipe reached spirit-family helper"
	}
	if = {
		limit = { exists = scope:spawned_spirit }
		scope:spawned_spirit = { save_scope_as = pod_combat_new_summon }
	}
}

pod_combat_summon_combatant_glue_effect = {
	# Host scope; $MOVE_STORY$ carries summon_recipe. The recipe creates pod_combat_new_summon.
	# summon_hostile reverses the destination team. Hide creation from tooltip preview.
	hidden_effect = {
		var:pod_combat_session ?= { var:pod_combat_actor_char = { save_scope_as = pod_combat_summoner } }
		# Check destination capacity and duel permission before creating a character to avoid orphaned summons.
		if = {
			limit = {
				pod_combat_summon_recipe_is_known_trigger = { MOVE_STORY = $MOVE_STORY$ }
				NOT = { pod_combat_duel_bans_summons_trigger = yes }
				OR = {
					AND = {
						pod_combat_summon_destination_is_player_team_trigger = { MOVE_STORY = $MOVE_STORY$ }
						var:pod_combat_session ?= { var:pod_combat_player_team_size < pod_combat_max_team_size }
					}
					AND = {
						pod_combat_summon_destination_is_enemy_team_trigger = { MOVE_STORY = $MOVE_STORY$ }
						var:pod_combat_session ?= { var:pod_combat_enemy_team_size < pod_combat_max_team_size }
					}
				}
			}
			pod_combat_apply_summon_recipe_effect = { MOVE_STORY = $MOVE_STORY$ }
			if = {
				limit = {
					exists = scope:pod_combat_new_summon
					scope:pod_combat_new_summon = { is_alive = yes }
				}
				# Summons always KO instead of dying even in fatal duels. Keeping them
				# alive guarantees both cleanup paths can remove the generated character.
				scope:pod_combat_new_summon = {
					add_character_flag = pod_combat_nonfatal_active
					add_character_flag = spirit_pact_block
				}
				# Register before seating so teardown can find the summon even if seating fails or the host dies.
				var:pod_combat_session ?= {
					add_to_variable_list = { name = pod_combat_summoned_chars target = scope:pod_combat_new_summon }
				}
				if = {
					limit = { pod_combat_summon_destination_is_player_team_trigger = { MOVE_STORY = $MOVE_STORY$ } }
					pod_combat_add_ally_midfight_effect = { CHARACTER = scope:pod_combat_new_summon }
				}
				else_if = {
					limit = { pod_combat_summon_destination_is_enemy_team_trigger = { MOVE_STORY = $MOVE_STORY$ } }
					pod_combat_add_enemy_midfight_effect = { CHARACTER = scope:pod_combat_new_summon }
				}
			}
			else = {
				debug_log = "pod_combat: known summon recipe returned no living character"
			}
		}
	}
}

pod_combat_apply_summon_recipe_effect = {
	# Each recipe branch must return one initialized, living
	# character as scope:pod_combat_new_summon.
	if = {
		limit = { $MOVE_STORY$ = { var:summon_recipe = flag:wraith } }
		# The combat template keeps PoD's wraith identity while omitting the
		# campaign initializer's relocation and persistent maintenance state.
		create_character = {
			location = scope:pod_combat_summoner.location
			template = POD_combat_summoned_wraith_character
			save_scope_as = pod_combat_new_summon
		}
	}
	else_if = {
		limit = { $MOVE_STORY$ = { var:summon_recipe = flag:controlled_animal } }
		if = {
			limit = { scope:pod_combat_summoner = { has_perk = plague_of_beasts_perk } }
			create_character = {
				location = scope:pod_combat_summoner.location
				template = POD_combat_summoned_controlled_animal_swarm_character
				save_scope_as = pod_combat_new_summon
			}
		}
		else = {
			create_character = {
				location = scope:pod_combat_summoner.location
				template = POD_combat_summoned_controlled_animal_character
				save_scope_as = pod_combat_new_summon
			}
		}
	}
	else_if = {
		limit = { $MOVE_STORY$ = { var:summon_recipe = flag:wolf } }
		create_character = {
			location = scope:pod_combat_summoner.location
			template = POD_combat_summoned_wolf_character
			save_scope_as = pod_combat_new_summon
		}
	}
	else_if = {
		limit = {
			$MOVE_STORY$ = {
				OR = {
					var:summon_recipe = flag:bane
					var:summon_recipe = flag:gaian_spirit
					var:summon_recipe = flag:elemental_bane
				}
			}
		}
		scope:pod_combat_summoner = {
			pod_combat_spawn_spirit_family_helper = { MOVE_STORY = $MOVE_STORY$ }
		}
	}
	else = {
		# The known-recipe pre-check makes this unreachable in normal execution.
		debug_log = "pod_combat: unknown summon_recipe reached creation dispatch"
	}
}
