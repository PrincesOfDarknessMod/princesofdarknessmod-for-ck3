######################################################
######   POD COMBAT — MID-COMBAT COMBATANT ADDITION + SUMMONS
######################################################
# Seats a brand-new combatant into a LIVE fight (the duel builder only runs
# pre-combat). See COMBAT_REFERENCE.info for the slot / leader-pointer model.
#
#   pod_combat_add_ally_midfight_effect            = { CHARACTER }             -> player, now
#   pod_combat_add_enemy_midfight_effect           = { CHARACTER }             -> enemy, now
#   pod_combat_schedule_ally_reinforcement_effect  = { CHARACTER JOIN_ROUND }  -> player, at round N
#   pod_combat_schedule_enemy_reinforcement_effect = { CHARACTER JOIN_ROUND }  -> enemy, at round N
# add_* enters immediately; schedule_* enters at the start of JOIN_ROUND (a round <= the
# current one means now), seated by pod_combat_group_end_round_effect's per-round sweep.
# Team-specific entry points, not one { TEAM }: CK3 can't index the per-team counter /
# leader pointer / add wrapper by a parse-time token, so the caller resolves it.
# Every mid-fight seat auto-posts a named "<who> joins the fray" log row, so callers
# (reinforcements, summons, events) never wire their own arrival message.

###################
###### GENERAL: ADD A COMBATANT MID-FIGHT
###################

pod_combat_add_ally_midfight_effect = {
	# $CHARACTER$ must be a SAVED SCOPE — init re-evaluates it in the character's
	# own scope, so a var: chain would throw. Runs in the combat-host (PC) scope.
	# Opposing leader (discipline CASTER + hit-chance baseline) = enemy leader.
	# limit bail also keeps the var reads tooltip-build-safe. add_ally self-guards
	# the soft cap; if hit, no slot is created and the success gate below (F4) skips
	# the seat finalize/log so nothing phantom-joins.
	if = {
		limit = {
			has_variable = pod_combat_player_team_size
			has_variable = pod_combat_enemy_leader_slot
			var:pod_combat_enemy_leader_slot = { has_variable = slot_char }
		}
		# Snapshot landing position BEFORE the add (add_ally seats at team_size then bumps it).
		save_scope_value_as = { name = pod_combat_summon_pos value = var:pod_combat_player_team_size }
		# Opposing-leader baseline (discipline CASTER + hit-chance prowess). The enemy
		# leader pointer is the ORIGINAL position-0 slot, never re-pointed on leader
		# death, so a dead leader would baseline this seat against a corpse (C31). Seed
		# the leader unconditionally (always set), then if it is DEAD override with the
		# rear-most LIVING enemy; if none lives, the seeded (dead) leader stays.
		save_scope_value_as = { name = pod_combat_summon_opp_prowess value = var:pod_combat_enemy_leader_slot.var:slot_prowess }
		var:pod_combat_enemy_leader_slot.var:slot_char = { save_scope_as = pod_combat_summon_opp_leader }
		if = {
			limit = { NOT = { var:pod_combat_enemy_leader_slot = { pod_combat_slot_char_is_alive_trigger = yes } } }
			ordered_in_list = {
				variable = pod_combat_active_slots
				limit = {
					pod_combat_slot_is_enemy_trigger = yes
					pod_combat_slot_char_is_alive_trigger = yes
				}
				order_by = { value = 0 subtract = var:team_position }
				position = 0
				save_scope_value_as = { name = pod_combat_summon_opp_prowess value = var:slot_prowess }
				var:slot_char = { save_scope_as = pod_combat_summon_opp_leader }
			}
		}
		POD_combat_duel_add_ally_effect = { CHARACTER = $CHARACTER$ }
		# Add_ally drops a seat past pod_combat_max_team_size and leaves the team-size
		# counter put. Gate the finalize/log/flag on the counter actually bumping past the
		# snapshot - a capped add otherwise posts a phantom "joins the fray" row and
		# death-gate-flags a character who never entered the fight (and a deferred
		# reinforcement would be silently consumed).
		if = {
			limit = { var:pod_combat_player_team_size > scope:pod_combat_summon_pos }
			pod_combat_init_one_midfight_slot_effect = { TEAM = player }
			pod_combat_push_system_log_named_effect = { MESSAGE_KEY = combatant_joins  CHARACTER = $CHARACTER$ }
			# Non-fatal / practice duel: a mid-fight seat must also carry the death gate
			# (the finalize stamp only covers the initial roster), or it could DIE from a
			# filled wound track - practice now deals real damage. Host scope, sc_context bare.
			# End-combat strip iterates pod_combat_active_slots (now incl. this slot).
			if = {
				limit = { pod_combat_uses_death_gate_trigger = yes }
				$CHARACTER$ = { add_character_flag = pod_combat_nonfatal_active }
			}
			# Practice: snapshot the late arrival's wounds so they restore at combat end.
			if = {
				limit = { pod_combat_is_practice_trigger = yes }
				$CHARACTER$ = { pod_combat_snapshot_one_practice_wounds_effect = yes }
			}
		}
	}
}

pod_combat_add_enemy_midfight_effect = {
	# Enemy-team counterpart of pod_combat_add_ally_midfight_effect; opposing leader = PC.
	if = {
		limit = {
			has_variable = pod_combat_enemy_team_size
			has_variable = pod_combat_pc_slot
			var:pod_combat_pc_slot = { has_variable = slot_char }
		}
		save_scope_value_as = { name = pod_combat_summon_pos value = var:pod_combat_enemy_team_size }
		# Opposing-leader baseline - mirror of the ally twin (C31). Opposing side is the
		# player team; the PC slot is the leader pointer. Seed the PC, then if the PC is
		# dead override with the rear-most LIVING player-team member (the PC is normally
		# alive mid-fight - its death ends combat - so this is defensive).
		save_scope_value_as = { name = pod_combat_summon_opp_prowess value = var:pod_combat_pc_slot.var:slot_prowess }
		var:pod_combat_pc_slot.var:slot_char = { save_scope_as = pod_combat_summon_opp_leader }
		if = {
			limit = { NOT = { var:pod_combat_pc_slot = { pod_combat_slot_char_is_alive_trigger = yes } } }
			ordered_in_list = {
				variable = pod_combat_active_slots
				limit = {
					pod_combat_slot_is_player_team_trigger = yes
					pod_combat_slot_char_is_alive_trigger = yes
				}
				order_by = { value = 0 subtract = var:team_position }
				position = 0
				save_scope_value_as = { name = pod_combat_summon_opp_prowess value = var:slot_prowess }
				var:slot_char = { save_scope_as = pod_combat_summon_opp_leader }
			}
		}
		POD_combat_duel_add_enemy_effect = { CHARACTER = $CHARACTER$ }
		# Success gate - see the ally twin above for the rationale.
		if = {
			limit = { var:pod_combat_enemy_team_size > scope:pod_combat_summon_pos }
			pod_combat_init_one_midfight_slot_effect = { TEAM = enemy }
			pod_combat_push_system_log_named_effect = { MESSAGE_KEY = combatant_joins  CHARACTER = $CHARACTER$ }
			# Non-fatal / practice death gate + practice wound snapshot for mid-fight
			# seats - see the ally twin above.
			if = {
				limit = { pod_combat_uses_death_gate_trigger = yes }
				$CHARACTER$ = { add_character_flag = pod_combat_nonfatal_active }
			}
			if = {
				limit = { pod_combat_is_practice_trigger = yes }
				$CHARACTER$ = { pod_combat_snapshot_one_practice_wounds_effect = yes }
			}
		}
	}
}

pod_combat_init_one_midfight_slot_effect = {
	# $TEAM$ = player | enemy. Re-finds the slot add_ally/add_enemy just created and
	# runs finalize's per-slot work. Order matters: init writes slot_prowess that
	# calc_one_hit_chance then reads. Snapshot var:slot_char first — init re-evaluates
	# CHARACTER in the char's own scope. Summon is NOT fera-shapeshifted (the upfront
	# crinos pass doesn't re-run mid-fight) — fine for non-fera summons.
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			var:combat_team = flag:$TEAM$
			var:team_position = scope:pod_combat_summon_pos
		}
		var:slot_char = { save_scope_as = pod_combat_summon_slot_char }
		pod_combat_init_slot_from_character_effect = { CHARACTER = scope:pod_combat_summon_slot_char ENEMY_LEADER = scope:pod_combat_summon_opp_leader }
		pod_combat_calc_one_hit_chance_effect = { OPP_PROWESS = scope:pod_combat_summon_opp_prowess }
		set_variable = { name = slot_has_acted value = 1 }
		# Seed the AI loadout at seat time (slot init just cleared it):
		# mirrors the finalize pre-seed, so loadout membership stays a valid
		# unlock proof for the AI selector's dynamic-only validator. Last in
		# the body: a scope-switch after writes is the safe loop position.
		if = {
			limit = {
				scope:pod_combat_summon_slot_char = {
					is_ai = yes
					pod_combat_equipped_count = 0
				}
			}
			scope:pod_combat_summon_slot_char = { POD_apply_ai_full_loadout = yes }
		}
	}
}

###################
###### DEFERRED REINFORCEMENTS
###################

pod_combat_schedule_ally_reinforcement_effect = {
	# $CHARACTER$ (saved scope) joins the PLAYER team at $JOIN_ROUND$ (absolute round).
	# A round <= the current one seats immediately; a future round defers to the
	# per-round sweep in pod_combat_group_end_round_effect. Host (PC) scope.
	if = {
		limit = {
			has_variable = pod_combat_round
			var:pod_combat_round < $JOIN_ROUND$
		}
		pod_combat_defer_reinforcement_effect = { CHARACTER = $CHARACTER$  JOIN_ROUND = $JOIN_ROUND$  TEAM = player }
	}
	else = { pod_combat_add_ally_midfight_effect = { CHARACTER = $CHARACTER$ } }
}

pod_combat_schedule_enemy_reinforcement_effect = {
	# Enemy-team counterpart of pod_combat_schedule_ally_reinforcement_effect.
	if = {
		limit = {
			has_variable = pod_combat_round
			var:pod_combat_round < $JOIN_ROUND$
		}
		pod_combat_defer_reinforcement_effect = { CHARACTER = $CHARACTER$  JOIN_ROUND = $JOIN_ROUND$  TEAM = enemy }
	}
	else = { pod_combat_add_enemy_midfight_effect = { CHARACTER = $CHARACTER$ } }
}

pod_combat_defer_reinforcement_effect = {
	# Registers $CHARACTER$ to enter on $JOIN_ROUND$ (absolute round). Host scope.
	# Join round + team live on the character; the host list
	# pod_combat_pending_reinforcements drives the per-round sweep in
	# pod_combat_group_end_round_effect.
	$CHARACTER$ = {
		set_variable = { name = pod_combat_join_round value = $JOIN_ROUND$ }
		set_variable = { name = pod_combat_join_team  value = flag:$TEAM$ }
	}
	add_to_variable_list = { name = pod_combat_pending_reinforcements target = $CHARACTER$ }
}

pod_combat_process_pending_reinforcements_effect = {
	# Host scope. Seats every pending reinforcement whose join round has arrived, then
	# unregisters it. Called from pod_combat_group_end_round_effect after the round
	# increment and before calculate_initiative, so a freshly-seated slot rolls into the
	# new round's initiative (its seat-time slot_has_acted = 1 is reset to 0 there, so it
	# acts this round). Two passes: a list can't be modified while every_in_list walks it.
	if = {
		limit = { has_variable_list = pod_combat_pending_reinforcements }
		save_scope_as = reinf_host
		every_in_list = {
			variable = pod_combat_pending_reinforcements
			limit = { var:pod_combat_join_round <= scope:reinf_host.var:pod_combat_round }
			save_temporary_scope_as = due_char
			scope:reinf_host = { add_to_variable_list = { name = pod_combat_reinf_due target = scope:due_char } }
		}
		if = {
			limit = { has_variable_list = pod_combat_reinf_due }
			every_in_list = {
				variable = pod_combat_reinf_due
				save_temporary_scope_as = due_char
				scope:reinf_host = {
					remove_list_variable = { name = pod_combat_pending_reinforcements target = scope:due_char }
					# Seat on the stored team. A reinforcement killed before its round is
					# just dropped (removed above, not seated).
					if = {
						limit = { scope:due_char = { is_alive = yes  var:pod_combat_join_team = flag:player } }
						pod_combat_add_ally_midfight_effect = { CHARACTER = scope:due_char }
					}
					else_if = {
						limit = { scope:due_char = { is_alive = yes  var:pod_combat_join_team = flag:enemy } }
						pod_combat_add_enemy_midfight_effect = { CHARACTER = scope:due_char }
					}
				}
				# is_alive: var mutation on a dead reference errors; a reinforcement
				# killed before its round was dropped above, its vars die with it.
				if = {
					limit = { scope:due_char = { is_alive = yes } }
					scope:due_char = {
						remove_variable = pod_combat_join_round
						remove_variable = pod_combat_join_team
					}
				}
			}
			clear_variable_list = pod_combat_reinf_due
			# Each reinforcement self-announces a named "joins the fray" row as it seats
			# (via pod_combat_add_ally/enemy_midfight_effect), so no batch arrival message.
		}
	}
}

###################
###### MOVE-DRIVEN SUMMONS (generic; raise_wraith is the first user)
###################

pod_combat_summon_combatant_glue_effect = {
	# Fired from section 14b of POD_execute_combat_move on a summons_combatant move;
	# runs in combat-host scope. Creates a fresh combatant, applies the move's
	# summon_recipe, and seats it on the caster's team; despawned at combat end via
	# pod_combat_summoned_chars. The seat self-logs a named "joins the fray" row.
	# $MOVE_STORY$ = the executing move story (carries summon_recipe).
	# WHOLE BODY in hidden_effect: the confirm-tooltip walks the effect tree without
	# applying state, and a bare create_character / create_story / list-add errors
	# in that preview ("Undefined event target").
	hidden_effect = {
		var:pod_combat_actor_char = { save_scope_as = pod_combat_summoner }
		# Capacity pre-check BEFORE creating anything: create-then-fail-to-seat would
		# orphan an un-cleaned character. Same threshold add_ally/add_enemy use.
		if = {
			limit = {
				OR = {
					AND = {
						var:pod_combat_current_actor_team = flag:player
						var:pod_combat_player_team_size < pod_combat_max_team_size
					}
					AND = {
						var:pod_combat_current_actor_team = flag:enemy
						var:pod_combat_enemy_team_size < pod_combat_max_team_size
					}
				}
			}
			# NO template: avoids template root-scope reads; explicit culture/faith/gender.
			create_character = {
				location = scope:pod_combat_summoner.location
				culture  = scope:pod_combat_summoner.culture
				faith    = scope:pod_combat_summoner.faith
				gender_female_chance = 50
				save_scope_as = pod_combat_new_summon
			}
			pod_combat_apply_summon_recipe_effect = { CHARACTER = scope:pod_combat_new_summon  MOVE_STORY = $MOVE_STORY$ }
			# Team-specific entry points, so branch on the actor's team flag.
			if = {
				limit = { var:pod_combat_current_actor_team = flag:player }
				pod_combat_add_ally_midfight_effect = { CHARACTER = scope:pod_combat_new_summon }
			}
			else = {
				pod_combat_add_enemy_midfight_effect = { CHARACTER = scope:pod_combat_new_summon }
			}
			add_to_variable_list = { name = pod_combat_summoned_chars target = scope:pod_combat_new_summon }
		}
	}
}

pod_combat_apply_summon_recipe_effect = {
	# Splat-init dispatch for a freshly-created summon shell. $CHARACTER$ = the shell
	# (saved scope); $MOVE_STORY$ supplies var:summon_recipe (a flag) selecting the
	# recipe. A new creature type = one more else_if branch here; nothing else changes.
	$CHARACTER$ = {
		if = {
			limit = { $MOVE_STORY$ = { var:summon_recipe = flag:wraith } }
			# SAFE wraith recipe, NOT POD_wraith_initialization_effect, which relocates
			# to the Shadowlands (POD_send_to_dark_kingdom_effect) + fires POD_wraith_maintenance.1.
			# add_trait wraith FIRST so energy init picks the pathos pool.
			add_trait = wraith
			POD_initialize_energy_modifier = yes
			POD_arcanoi_initiation_effect = yes
			add_character_flag = had_POD_character_initialization
		}
	}
}
