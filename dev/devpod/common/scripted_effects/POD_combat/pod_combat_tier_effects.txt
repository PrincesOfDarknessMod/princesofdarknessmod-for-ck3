######################################################
######   POD COMBAT TIER EFFECTS
######################################################
# Static infrastructure for the tiered buff/debuff system.
#
# Data model:
#   pod_combat_{side,c{N}}_base_soak        — soak before any tier effects
#   pod_combat_{side,c{N}}_base_hit_chance  — hit chance before any tier effects
#   pod_combat_{side,c{N}}_soak             — live soak (base + buff mags - debuff mags)
#   pod_combat_{side,c{N}}_hit_chance       — live hit chance (same formula)
#   pod_combat_{side,c{N}}_{stat}_{direction}_{duration}_tier  — current tier (1/2/3)
#
# Tier magnitudes: small = 10, medium = 20, large = 30
#
# Apply/revert flow (recalc-from-base, no deltas):
#   - Applying tier N: set the tier var, then call a recalc effect. The
#     stat is rebuilt from base + sum of active tier magnitudes, clamped
#     on write. Clamping losses do NOT accumulate because each apply and
#     each revert starts from the pristine base.
#   - Round end: round-duration tier vars are cleared, then every slot
#     is recalculated. Combat-duration tiers persist.
#   - Combat end: all tier vars cleared.
#
# Edit this file directly if you change:
#   - The number of slots (currently 10)
#   - The tier stat categories (currently soak, hit)
#   - Tier magnitudes (must remain in lockstep with the move-side
#     tier setters in pod_combat_move_init_effects.txt)

pod_combat_recalc_soak_player_effect = {
	if = {
		limit = { has_variable = pod_combat_player_base_soak }
		save_scope_value_as = {
			name = new_soak_player
			value = {
				value = var:pod_combat_player_base_soak
				if = { limit = { has_variable = pod_combat_player_soak_buff_round_tier var:pod_combat_player_soak_buff_round_tier = 1 } add = @small_buff_mag }
				if = { limit = { has_variable = pod_combat_player_soak_buff_round_tier var:pod_combat_player_soak_buff_round_tier = 2 } add = @medium_buff_mag }
				if = { limit = { has_variable = pod_combat_player_soak_buff_round_tier var:pod_combat_player_soak_buff_round_tier = 3 } add = @large_buff_mag }
				if = { limit = { has_variable = pod_combat_player_soak_buff_combat_tier var:pod_combat_player_soak_buff_combat_tier = 1 } add = @small_buff_mag }
				if = { limit = { has_variable = pod_combat_player_soak_buff_combat_tier var:pod_combat_player_soak_buff_combat_tier = 2 } add = @medium_buff_mag }
				if = { limit = { has_variable = pod_combat_player_soak_buff_combat_tier var:pod_combat_player_soak_buff_combat_tier = 3 } add = @large_buff_mag }
				if = { limit = { has_variable = pod_combat_player_soak_debuff_round_tier var:pod_combat_player_soak_debuff_round_tier = 1 } subtract = @small_buff_mag }
				if = { limit = { has_variable = pod_combat_player_soak_debuff_round_tier var:pod_combat_player_soak_debuff_round_tier = 2 } subtract = @medium_buff_mag }
				if = { limit = { has_variable = pod_combat_player_soak_debuff_round_tier var:pod_combat_player_soak_debuff_round_tier = 3 } subtract = @large_buff_mag }

###### PAIR RECALC (actor vs target)
############################################################
# Rebuild both workspace base_hit_chance values from the current
# actor / target prowess pair, then layer the active hit-tier
# modifiers on top via the per-side recalc effects below.
#
# Why this exists: pod_combat_calculate_all_hit_chances_effect
# bakes each slot's hit chance once at combat init, against the
# primary leader (slot 0 for enemies, slot 5 for players). When
# an AI from slot 7 attacks slot 2 — or when the AoE iterator
# walks every enemy in turn — the workspace would otherwise carry
# the leader-vs-leader baseline through to the damage roll,
# which is the symptom the assessment in #11 flagged. Calling
# this from pod_combat_load_target_from_story_effect makes every
# pair load (workspace load, AI target swap, AoE per-target
# iteration) refresh the workspace numbers against the actual
# pair before any damage roll reads them.
#
# Slot writeback is deliberate-omission: we don't touch
# pod_combat_cN_base_hit_chance / hit_chance because (a) AoE
# would scribble over the actor's slot once per target leaving
# it pinned to the last iteration, and (b) the slot value is
# what the UI display reads — leaving it as the leader-baseline
# keeps the per-slot display consistent across the team. The
# PC retargeting path keeps its own explicit slot writeback in
# pod_combat_recalc_player_hit_chance_effect because the PC's
# selected_target IS persistent and the display should follow.
pod_combat_recalc_pair_hit_chance_effect = {
	if = {
		limit = {
			has_variable = pod_combat_player_prowess
			has_variable = pod_combat_opponent_prowess
		}
		# Actor side: actor prowess minus target prowess.
		save_scope_value_as = {
			name = recalc_pair_player_base_hit
			value = {
				value = pod_combat_hit_chance_base
				add = {
					value = var:pod_combat_player_prowess
					subtract = var:pod_combat_opponent_prowess
					divide = 2
				}
				min = 20
				max = 95
			}
		}
		set_variable = { name = pod_combat_player_base_hit_chance value = scope:recalc_pair_player_base_hit }
		pod_combat_recalc_hit_player_effect = yes

		# Target side: same formula mirrored. Retaliation rolls
		# read pod_combat_opponent_hit_chance, so this keeps the
		# counter-attack hit math against the current pair too
		# (instead of the target's stored leader-vs-PC baseline).
		save_scope_value_as = {
			name = recalc_pair_opponent_base_hit
			value = {
				value = pod_combat_hit_chance_base
				add = {
					value = var:pod_combat_opponent_prowess
					subtract = var:pod_combat_player_prowess
					divide = 2
				}
				min = 20
				max = 95
			}
		}
		set_variable = { name = pod_combat_opponent_base_hit_chance value = scope:recalc_pair_opponent_base_hit }
		pod_combat_recalc_hit_opponent_effect = yes
	}
}

###### PER-SLOT RECALC (parametric)

# Story-scoped recalc — caller scope = slot story. Reads the slot's
# base_soak / base_hit_chance and the eight tier-magnitude vars
# directly (no host-side pointer dereference), writes back the
# rebuilt slot_soak / slot_hit_chance. Used by the host-scope wrapper
# below (for existing $SLOT$-parametric callers) AND by every_in_list
# iterations over pod_combat_slots, where current scope is already
# the slot story (e.g. pod_combat_revert_round_tiers_effect).
pod_combat_recalc_slot_stats_via_story_effect = {
	if = {
		limit = { has_variable = slot_base_soak }
		save_scope_value_as = {
			name = new_soak
			value = {
				value = var:slot_base_soak
				if = { limit = { has_variable = slot_soak_buff_round_tier    var:slot_soak_buff_round_tier    = 1 } add      = @small_buff_mag  }
				if = { limit = { has_variable = slot_soak_buff_round_tier    var:slot_soak_buff_round_tier    = 2 } add      = @medium_buff_mag }
				if = { limit = { has_variable = slot_soak_buff_round_tier    var:slot_soak_buff_round_tier    = 3 } add      = @large_buff_mag  }
				if = { limit = { has_variable = slot_soak_buff_combat_tier   var:slot_soak_buff_combat_tier   = 1 } add      = @small_buff_mag  }
				if = { limit = { has_variable = slot_soak_buff_combat_tier   var:slot_soak_buff_combat_tier   = 2 } add      = @medium_buff_mag }
				if = { limit = { has_variable = slot_soak_buff_combat_tier   var:slot_soak_buff_combat_tier   = 3 } add      = @large_buff_mag  }
				if = { limit = { has_variable = slot_soak_debuff_round_tier  var:slot_soak_debuff_round_tier  = 1 } subtract = @small_buff_mag  }
				if = { limit = { has_variable = slot_soak_debuff_round_tier  var:slot_soak_debuff_round_tier  = 2 } subtract = @medium_buff_mag }
				if = { limit = { has_variable = slot_soak_debuff_round_tier  var:slot_soak_debuff_round_tier  = 3 } subtract = @large_buff_mag  }
				if = { limit = { has_variable = slot_soak_debuff_combat_tier var:slot_soak_debuff_combat_tier = 1 } subtract = @small_buff_mag  }
				if = { limit = { has_variable = slot_soak_debuff_combat_tier var:slot_soak_debuff_combat_tier = 2 } subtract = @medium_buff_mag }
				if = { limit = { has_variable = slot_soak_debuff_combat_tier var:slot_soak_debuff_combat_tier = 3 } subtract = @large_buff_mag  }
				min = 0
				max = 100
			}
		}
		set_variable = { name = slot_soak value = scope:new_soak }
	}
	if = {
		limit = { has_variable = slot_base_hit_chance }
		save_scope_value_as = {
			name = new_hit
			value = {
				value = var:slot_base_hit_chance
				if = { limit = { has_variable = slot_hit_buff_round_tier    var:slot_hit_buff_round_tier    = 1 } add      = @small_buff_mag  }
				if = { limit = { has_variable = slot_hit_buff_round_tier    var:slot_hit_buff_round_tier    = 2 } add      = @medium_buff_mag }
				if = { limit = { has_variable = slot_hit_buff_round_tier    var:slot_hit_buff_round_tier    = 3 } add      = @large_buff_mag  }
				if = { limit = { has_variable = slot_hit_buff_combat_tier   var:slot_hit_buff_combat_tier   = 1 } add      = @small_buff_mag  }
				if = { limit = { has_variable = slot_hit_buff_combat_tier   var:slot_hit_buff_combat_tier   = 2 } add      = @medium_buff_mag }
				if = { limit = { has_variable = slot_hit_buff_combat_tier   var:slot_hit_buff_combat_tier   = 3 } add      = @large_buff_mag  }
				if = { limit = { has_variable = slot_hit_debuff_round_tier  var:slot_hit_debuff_round_tier  = 1 } subtract = @small_buff_mag  }
				if = { limit = { has_variable = slot_hit_debuff_round_tier  var:slot_hit_debuff_round_tier  = 2 } subtract = @medium_buff_mag }
				if = { limit = { has_variable = slot_hit_debuff_round_tier  var:slot_hit_debuff_round_tier  = 3 } subtract = @large_buff_mag  }
				if = { limit = { has_variable = slot_hit_debuff_combat_tier var:slot_hit_debuff_combat_tier = 1 } subtract = @small_buff_mag  }
				if = { limit = { has_variable = slot_hit_debuff_combat_tier var:slot_hit_debuff_combat_tier = 2 } subtract = @medium_buff_mag }
				if = { limit = { has_variable = slot_hit_debuff_combat_tier var:slot_hit_debuff_combat_tier = 3 } subtract = @large_buff_mag  }
				min = 20
				max = 95
			}
		}
		set_variable = { name = slot_hit_chance value = scope:new_hit }
	}
}

# Host-scope wrapper — existing call sites pass SLOT = 0..9. Flips into
# the slot story and delegates to the story-scoped helper above. The
# wrapper exists so the int-parametric callers (calc_one_hit_chance,
# recalc_player_hit_chance) stay unchanged through this refactor.
# Rebuild c$SLOT$_soak and c$SLOT$_hit_chance from base + active tiers.
# Used by the round-end revert and anywhere slot state changes outside
# the workspace dispatch.
pod_combat_recalc_slot_stats_effect = {
	# $SLOT$ = 0-9
	if = {
		limit = { has_variable = pod_combat_slot_$SLOT$ }
		var:pod_combat_slot_$SLOT$ = {
			pod_combat_recalc_slot_stats_via_story_effect = yes
		}
	}
}

############################################################
###### ROUND-END REVERT
############################################################

# Called from pod_combat_group_end_round_effect. Clears round-duration
# tier vars on every filled slot, then recalculates each slot's live
# stats from base + remaining combat-duration tiers. Because the live
# stats are rebuilt from scratch, clamping losses do not accumulate.
#
# Iterates over pod_combat_slots — the iteration body runs in slot-story
# scope, so var:slot_X reads/writes target the slot story directly and
# var:slot_char switches into the participant character without going
# through the host's per-slot pointer var.
pod_combat_revert_round_tiers_effect = {
	every_in_list = {
		variable = pod_combat_slots
		limit = { has_variable = slot_char }
		# Strip round-duration tier vars on the slot story.
		remove_variable = slot_soak_buff_round_tier
		remove_variable = slot_soak_debuff_round_tier
		remove_variable = slot_hit_buff_round_tier
		remove_variable = slot_hit_debuff_round_tier
		# Clear the char-scope soak tier bridge vars (read by
		# POD_soak_chance). Only round-tier soak vars are stripped
		# here; combat-tier persists until combat-end cleanup.
		var:slot_char = {
			remove_variable = pod_combat_tier_soak_buff_round
			remove_variable = pod_combat_tier_soak_debuff_round
		}
		# Recalc this slot's live stats from base + remaining tiers.
		pod_combat_recalc_slot_stats_via_story_effect = yes
	}
}

############################################################
###### COMBAT-END TIER CLEAR
############################################################

# Called from pod_combat_cleanup_effect. Strips all tier vars from every
# slot. The underlying stat vars are removed separately by
# pod_combat_group_cleanup_effect.
#
# Runs BEFORE pod_combat_clear_slot_stories_effect (which tears down the
# slot stories themselves at the end of the cleanup chain), so the slot
# stories are still alive while we strip their tier vars. Iterates
# pod_combat_slots — iteration body runs in slot-story scope, so
# `remove_variable = slot_X` writes the slot story directly.
pod_combat_clear_combat_tiers_effect = {
	every_in_list = {
		variable = pod_combat_slots
		# Strip every tier var on the slot story (current scope).
		remove_variable = slot_soak_buff_round_tier
		remove_variable = slot_soak_buff_combat_tier
		remove_variable = slot_soak_debuff_round_tier
		remove_variable = slot_soak_debuff_combat_tier
		remove_variable = slot_hit_buff_round_tier
		remove_variable = slot_hit_buff_combat_tier
		remove_variable = slot_hit_debuff_round_tier
		remove_variable = slot_hit_debuff_combat_tier
		# Clear the char-scope soak tier bridge vars (read by
		# POD_soak_chance). Covers both round and combat tiers since
		# this runs at combat end.
		if = {
			limit = { has_variable = slot_char }
			var:slot_char = {
				remove_variable = pod_combat_tier_soak_buff_round
				remove_variable = pod_combat_tier_soak_buff_combat
				remove_variable = pod_combat_tier_soak_debuff_round
				remove_variable = pod_combat_tier_soak_debuff_combat
			}
		}
	}
}
