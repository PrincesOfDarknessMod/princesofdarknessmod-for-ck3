######################################################
######   POD COMBAT TIER EFFECTS
######################################################
# Static infrastructure for the tiered buff/debuff system on soak and
# hit_chance. Lives downstream of POD_execute_combat_move §11. Edit this
# file directly when changing slot count, tier stat categories, or tier
# magnitudes (must stay in lockstep with the setters in
# pod_combat_move_init_effects.txt).
#
# Data model, magnitudes, and the recalc-from-base apply/revert flow:
# COMBAT_REFERENCE.info §5.
#
# Repetition note: each recalc effect below is a 12-branch if-chain
# checking {buff,debuff} × {round,combat} × tier {1,2,3} for ONE
# stat. The four effects split {soak,hit} × {actor,target} —
# pod_combat_recalc_soak_actor_effect, _soak_target_effect,
# _hit_actor_effect, _hit_target_effect. Every branch is mechanically
# identical (`has_variable = X_tier var:X_tier = N` →
# `add = @{small,medium,large}_buff_mag`); only the workspace prefix
# (actor / target), the stat (soak / hit), and the clamp bounds
# differ across the 4 effects. If you're scanning the file, expect
# the same pattern in all four bodies — only the prefix, stat, and
# clamp change.

############################################################
###### PAIR RECALC (actor vs target)
############################################################
# Rebuild both workspace base_hit_chance values from the current
# actor / target prowess pair, then layer the active hit-tier
# modifiers on top via the per-side recalc effects below.
#
# Why this exists: pod_combat_calculate_all_hit_chances_effect
# bakes each slot's hit chance once at combat init, against the
# primary leader (player/0 for enemies, enemy/0 for players). When
# an AI back-ranker attacks a non-leader target, or when the AoE iterator
# walks every enemy in turn, the workspace would otherwise carry
# the leader-vs-leader baseline through to the damage roll,
# which is the symptom the assessment in #11 flagged. Calling
# this from pod_combat_load_target_from_story_effect makes every
# pair load (workspace load, AI target swap, AoE per-target
# iteration) refresh the workspace numbers against the actual
# pair before any damage roll reads them.
#
# Slot writeback is deliberate-omission: we don't touch
# slot_base_hit_chance / slot_hit_chance because (a) AoE
# would scribble over the actor's slot once per target leaving
# it pinned to the last iteration, and (b) the slot value is
# what the UI display reads — leaving it as the leader-baseline
# keeps the per-slot display consistent across the team. The
# PC retargeting path keeps its own explicit slot writeback in
# pod_combat_recalc_pc_hit_chance_effect because the PC's
# selected_target IS persistent and the display should follow.
pod_combat_recalc_pair_hit_chance_effect = {
	if = {
		limit = {
			has_variable = pod_combat_actor_prowess
			has_variable = pod_combat_target_prowess
		}
		# Actor side: actor prowess minus target prowess.
		save_scope_value_as = {
			name = recalc_pair_actor_base_hit
			value = {
				value = pod_combat_hit_chance_base
				add = {
					value = var:pod_combat_actor_prowess
					subtract = var:pod_combat_target_prowess
					divide = 2
				}
				min = 20
				max = 95
			}
		}
		set_variable = { name = pod_combat_actor_base_hit_chance value = scope:recalc_pair_actor_base_hit }
		pod_combat_recalc_hit_actor_effect = yes

		# Target side: same formula mirrored. Keeps the target's hit chance
		# current against this pair — for the target's OWN turn (when it
		# acts) and for the slot-card Hit display — instead of its stored
		# leader-vs-PC baseline. 
		save_scope_value_as = {
			name = recalc_pair_target_base_hit
			value = {
				value = pod_combat_hit_chance_base
				add = {
					value = var:pod_combat_target_prowess
					subtract = var:pod_combat_actor_prowess
					divide = 2
				}
				min = 20
				max = 95
			}
		}
		set_variable = { name = pod_combat_target_base_hit_chance value = scope:recalc_pair_target_base_hit }
		pod_combat_recalc_hit_target_effect = yes
	}
}

# Scope-based entry: runs on the CURRENT slot story. Callers are either in a
# slot-iteration body (calc_one_hit_chance) or have scope-switched into a
# leader pointer first (pod_combat_recalc_pc_hit_chance_effect via
# pod_combat_pc_slot). Rebuilds slot_soak and slot_hit_chance from base +
# active tiers. Used by the round-end revert and anywhere slot state changes
# outside the workspace dispatch.
pod_combat_recalc_slot_stats_effect = {
	pod_combat_recalc_slot_stats_via_story_effect = yes
}

############################################################
###### ROUND-END REVERT
############################################################

# Called from pod_combat_group_end_round_effect. Clears round-duration
# tier vars on every filled slot, then recalculates each slot's live
# stats from base + remaining combat-duration tiers. Because the live
# stats are rebuilt from scratch, clamping losses do not accumulate.
#
# Iterates pod_combat_active_slots (filled-only), so the per-slot
# scope is always a populated slot story. var:slot_X reads/writes
# target the slot story directly and var:slot_char switches into the
# participant character without going through the host's per-slot
# pointer var.
pod_combat_revert_round_tiers_effect = {
	every_in_list = {
		variable = pod_combat_active_slots
		# Strip round-duration tier vars on the slot story.
		remove_variable = slot_soak_buff_round_tier
		remove_variable = slot_soak_debuff_round_tier
		remove_variable = slot_hit_buff_round_tier
		remove_variable = slot_hit_debuff_round_tier
		# Clear the char-scope soak tier bridge vars (read by
		# POD_soak_chance). Only round-tier soak vars are stripped
		# here; combat-tier persists until combat-end cleanup.
		var:slot_char = {
			remove_variable = pod_combat_tier_soak_buff_round
			remove_variable = pod_combat_tier_soak_debuff_round
		}
		# Recalc this slot's live stats from base + remaining tiers.
		pod_combat_recalc_slot_stats_via_story_effect = yes
	}
}

############################################################
###### COMBAT-END TIER CLEAR
############################################################

# Called from pod_combat_cleanup_effect. Strips all tier vars from every
# slot. The underlying stat vars are removed separately by
# pod_combat_group_cleanup_effect.
#
# Runs BEFORE pod_combat_clear_slot_stories_effect (which tears down the
# slot stories themselves at the end of the cleanup chain), so the slot
# stories are still alive while we strip their tier vars. Iterates
# pod_combat_active_slots — iteration body runs in slot-story scope, so
# `remove_variable = slot_X` writes the slot story directly. The list
# is filled-only by construction, so the char-scope bridge clear no
# longer needs a `has_variable = slot_char` guard.
pod_combat_clear_combat_tiers_effect = {
	every_in_list = {
		variable = pod_combat_active_slots
		# Strip every tier var on the slot story (current scope).
		remove_variable = slot_soak_buff_round_tier
		remove_variable = slot_soak_buff_combat_tier
		remove_variable = slot_soak_debuff_round_tier
		remove_variable = slot_soak_debuff_combat_tier
		remove_variable = slot_hit_buff_round_tier
		remove_variable = slot_hit_buff_combat_tier
		remove_variable = slot_hit_debuff_round_tier
		remove_variable = slot_hit_debuff_combat_tier
		# Clear the char-scope soak tier bridge vars (read by
		# POD_soak_chance). Covers both round and combat tiers since
		# this runs at combat end.
