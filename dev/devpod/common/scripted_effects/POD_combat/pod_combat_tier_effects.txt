######################################################
######   POD COMBAT TIER EFFECTS (hand-maintained)
######################################################
# Static infrastructure for the tiered buff/debuff system. These effects
# do NOT regenerate when skills are added or modified.
#
# Data model:
#   pod_combat_{side,c{N}}_base_soak        — soak before any tier effects
#   pod_combat_{side,c{N}}_base_hit_chance  — hit chance before any tier effects
#   pod_combat_{side,c{N}}_soak             — live soak (base + buff mags - debuff mags)
#   pod_combat_{side,c{N}}_hit_chance       — live hit chance (same formula)
#   pod_combat_{side,c{N}}_{stat}_{direction}_{duration}_tier  — current tier (1/2/3)
#
# Tier magnitudes: small = 10, medium = 20, large = 30
#
# Apply/revert flow (recalc-from-base, no deltas):
#   - Applying tier N: set the tier var, then call a recalc effect. The
#     stat is rebuilt from base + sum of active tier magnitudes, clamped
#     on write. Clamping losses do NOT accumulate because each apply and
#     each revert starts from the pristine base.
#   - Round end: round-duration tier vars are cleared, then every slot
#     is recalculated. Combat-duration tiers persist.
#   - Combat end: all tier vars cleared.
#
# Edit this file directly if you change:
#   - The number of slots (currently 10)
#   - The tier stat categories (currently soak, hit)
#   - Tier magnitudes (match TIER_MAGNITUDE in generate_combat_files.py)

pod_combat_recalc_soak_player_effect = {
	if = {
		limit = { has_variable = pod_combat_player_base_soak }
		save_scope_value_as = {
			name = new_soak_player
			value = {
				value = var:pod_combat_player_base_soak
				if = { limit = { has_variable = pod_combat_player_soak_buff_round_tier var:pod_combat_player_soak_buff_round_tier = 1 } add = 10 }
				if = { limit = { has_variable = pod_combat_player_soak_buff_round_tier var:pod_combat_player_soak_buff_round_tier = 2 } add = 20 }
				if = { limit = { has_variable = pod_combat_player_soak_buff_round_tier var:pod_combat_player_soak_buff_round_tier = 3 } add = 30 }
				if = { limit = { has_variable = pod_combat_player_soak_buff_combat_tier var:pod_combat_player_soak_buff_combat_tier = 1 } add = 10 }
				if = { limit = { has_variable = pod_combat_player_soak_buff_combat_tier var:pod_combat_player_soak_buff_combat_tier = 2 } add = 20 }
				if = { limit = { has_variable = pod_combat_player_soak_buff_combat_tier var:pod_combat_player_soak_buff_combat_tier = 3 } add = 30 }
				if = { limit = { has_variable = pod_combat_player_soak_debuff_round_tier var:pod_combat_player_soak_debuff_round_tier = 1 } subtract = 10 }
				if = { limit = { has_variable = pod_combat_player_soak_debuff_round_tier var:pod_combat_player_soak_debuff_round_tier = 2 } subtract = 20 }
				if = { limit = { has_variable = pod_combat_player_soak_debuff_round_tier var:pod_combat_player_soak_debuff_round_tier = 3 } subtract = 30 }
				if = { limit = { has_variable = pod_combat_player_soak_debuff_combat_tier var:pod_combat_player_soak_debuff_combat_tier = 1 } subtract = 10 }
				if = { limit = { has_variable = pod_combat_player_soak_debuff_combat_tier var:pod_combat_player_soak_debuff_combat_tier = 2 } subtract = 20 }
				if = { limit = { has_variable = pod_combat_player_soak_debuff_combat_tier var:pod_combat_player_soak_debuff_combat_tier = 3 } subtract = 30 }
				min = 0
				max = 100
			}
		}
		set_variable = { name = pod_combat_player_soak value = scope:new_soak_player }
	}
}
