######################################################
######   POD COMBAT TIER EFFECTS
######################################################
# Tiered buff/debuff system on soak and hit_chance. Tier magnitudes must
# stay in lockstep with the setters in pod_combat_move_init_effects.txt.
# Data model + recalc-from-base apply/revert flow: COMBAT_REFERENCE.info §5.
#
# The 12-branch tier chain ({buff,debuff} x {round,combat} x tier {1,2,3})
# lives once in pod_combat_recalc_stat_core_effect; the four public recalc
# wrappers differ only in token values (side, stat, clamps, stealth bonus).

############################################################
###### WORKSPACE RECALC CORE (parametric)
############################################################

# Shared tier-stacking math behind the four public recalc wrappers.
# Params: SIDE = actor|target, STAT = soak|hit (tier-var stem),
# OUT = soak|hit_chance (base/output var stem), MIN/MAX = clamp bounds,
# STEALTH_BONUS = pod_combat_stealth_hit_bonus on hit, 0 on soak (stealth
# only boosts hit; the zero add keeps the formula shared).
pod_combat_recalc_stat_core_effect = {
	if = {
		limit = { has_variable = pod_combat_$SIDE$_base_$OUT$ }
		save_scope_value_as = {
			name = new_$STAT$_$SIDE$
			value = {
				value = var:pod_combat_$SIDE$_base_$OUT$
				if = { limit = { has_variable = pod_combat_$SIDE$_$STAT$_buff_round_tier var:pod_combat_$SIDE$_$STAT$_buff_round_tier = 1 } add = combat_tier_small_mag }
				if = { limit = { has_variable = pod_combat_$SIDE$_$STAT$_buff_round_tier var:pod_combat_$SIDE$_$STAT$_buff_round_tier = 2 } add = combat_tier_medium_mag }
				if = { limit = { has_variable = pod_combat_$SIDE$_$STAT$_buff_round_tier var:pod_combat_$SIDE$_$STAT$_buff_round_tier = 3 } add = combat_tier_large_mag }
				if = { limit = { has_variable = pod_combat_$SIDE$_$STAT$_buff_combat_tier var:pod_combat_$SIDE$_$STAT$_buff_combat_tier = 1 } add = combat_tier_small_mag }
				if = { limit = { has_variable = pod_combat_$SIDE$_$STAT$_buff_combat_tier var:pod_combat_$SIDE$_$STAT$_buff_combat_tier = 2 } add = combat_tier_medium_mag }
				if = { limit = { has_variable = pod_combat_$SIDE$_$STAT$_buff_combat_tier var:pod_combat_$SIDE$_$STAT$_buff_combat_tier = 3 } add = combat_tier_large_mag }
				if = { limit = { has_variable = pod_combat_$SIDE$_$STAT$_debuff_round_tier var:pod_combat_$SIDE$_$STAT$_debuff_round_tier = 1 } subtract = combat_tier_small_mag }
				if = { limit = { has_variable = pod_combat_$SIDE$_$STAT$_debuff_round_tier var:pod_combat_$SIDE$_$STAT$_debuff_round_tier = 2 } subtract = combat_tier_medium_mag }
				if = { limit = { has_variable = pod_combat_$SIDE$_$STAT$_debuff_round_tier var:pod_combat_$SIDE$_$STAT$_debuff_round_tier = 3 } subtract = combat_tier_large_mag }
				if = { limit = { has_variable = pod_combat_$SIDE$_$STAT$_debuff_combat_tier var:pod_combat_$SIDE$_$STAT$_debuff_combat_tier = 1 } subtract = combat_tier_small_mag }
				if = { limit = { has_variable = pod_combat_$SIDE$_$STAT$_debuff_combat_tier var:pod_combat_$SIDE$_$STAT$_debuff_combat_tier = 2 } subtract = combat_tier_medium_mag }
				if = { limit = { has_variable = pod_combat_$SIDE$_$STAT$_debuff_combat_tier var:pod_combat_$SIDE$_$STAT$_debuff_combat_tier = 3 } subtract = combat_tier_large_mag }
				# Stealth folds into the recalc so the bonus survives workspace
				# loads and pair rebuilds; apply/break only toggle the flag.
				if = { limit = { has_variable = pod_combat_$SIDE$_is_stealthed } add = $STEALTH_BONUS$ }
				min = $MIN$
				max = $MAX$
			}
		}
		set_variable = { name = pod_combat_$SIDE$_$OUT$ value = scope:new_$STAT$_$SIDE$ }
	}
}

############################################################
###### WORKSPACE RECALC (ACTOR)
############################################################

pod_combat_recalc_soak_actor_effect = {
	pod_combat_recalc_stat_core_effect = { SIDE = actor STAT = soak OUT = soak MIN = 0 MAX = 100 STEALTH_BONUS = 0 }
}

pod_combat_recalc_hit_actor_effect = {
	pod_combat_recalc_stat_core_effect = { SIDE = actor STAT = hit OUT = hit_chance MIN = 20 MAX = 95 STEALTH_BONUS = pod_combat_stealth_hit_bonus }
	# Actor-only extra: sync the live-roll mirror, gated like the core so a
	# missing base leaves it untouched.
	if = {
		limit = { has_variable = pod_combat_actor_base_hit_chance }
		set_variable = { name = pod_combat_current_hit_chance value = scope:new_hit_actor }
	}
}

############################################################
###### WORKSPACE RECALC (TARGET)
############################################################

pod_combat_recalc_soak_target_effect = {
	pod_combat_recalc_stat_core_effect = { SIDE = target STAT = soak OUT = soak MIN = 0 MAX = 100 STEALTH_BONUS = 0 }
}

############################################################
###### PAIR RECALC (actor vs target)
############################################################
# Rebuild both workspace base_hit_chance values from the current
# actor/target prowess pair, then re-layer hit tiers via the per-side
# recalc effects.
#
# Why: hit chances are baked at combat init against the primary leader.
# Without this, an AI back-ranker hitting a non-leader, or the AoE
# iterator walking every enemy, carries the leader-vs-leader baseline
# into the damage roll. Called from pod_combat_load_target_from_story_effect
# so every pair load refreshes against the actual pair first.
#
# Slot writeback deliberately omitted: AoE would pin the actor's slot to
# the last iteration, and the slot value is what the UI reads — leaving it
# at the leader baseline keeps the per-slot display consistent. The PC
# retargeting path (pod_combat_recalc_pc_hit_chance_effect) DOES write back
# because the PC's selected_target is persistent.
pod_combat_recalc_pair_hit_chance_effect = {
	if = {
		limit = {
			has_variable = pod_combat_actor_prowess
			has_variable = pod_combat_target_prowess
		}
		save_scope_value_as = {
			name = recalc_pair_actor_base_hit
			value = {
				value = pod_combat_hit_chance_base
				add = {
					value = var:pod_combat_actor_prowess
					subtract = var:pod_combat_target_prowess
					divide = 2
				}
				min = 20
				max = 95
			}
		}
		set_variable = { name = pod_combat_actor_base_hit_chance value = scope:recalc_pair_actor_base_hit }
		pod_combat_recalc_hit_actor_effect = yes

		# Target side mirrored: keeps the target current against this pair for
		# its own turn and the slot-card Hit display, not its stored baseline.
		save_scope_value_as = {
			name = recalc_pair_target_base_hit
			value = {
				value = pod_combat_hit_chance_base
				add = {
					value = var:pod_combat_target_prowess
					subtract = var:pod_combat_actor_prowess
					divide = 2
				}
				min = 20
				max = 95
			}
		}
		set_variable = { name = pod_combat_target_base_hit_chance value = scope:recalc_pair_target_base_hit }
		pod_combat_recalc_hit_target_effect = yes
	}
}


pod_combat_recalc_hit_target_effect = {
	# Target stealth hit bonus feeds own-turn rolls + slot display only;
	# retaliation skips the hit roll entirely.
	pod_combat_recalc_stat_core_effect = { SIDE = target STAT = hit OUT = hit_chance MIN = 20 MAX = 95 STEALTH_BONUS = pod_combat_stealth_hit_bonus }
}

############################################################
###### PER-SLOT RECALC (parametric)
############################################################

# Caller scope = slot story: reads slot_base_* and the eight tier vars
# directly (no host pointer deref), writes back slot_soak / slot_hit_chance.
# Used by the host-scope wrapper below and by every_in_list iterations over
# pod_combat_active_slots (where current scope is already the slot story).
pod_combat_recalc_slot_stats_via_story_effect = {
	if = {
		limit = { has_variable = slot_base_soak }
		save_scope_value_as = {
			name = new_soak
			value = {
				value = var:slot_base_soak
				if = { limit = { has_variable = slot_soak_buff_round_tier    var:slot_soak_buff_round_tier    = 1 } add      = combat_tier_small_mag  }
				if = { limit = { has_variable = slot_soak_buff_round_tier    var:slot_soak_buff_round_tier    = 2 } add      = combat_tier_medium_mag }
				if = { limit = { has_variable = slot_soak_buff_round_tier    var:slot_soak_buff_round_tier    = 3 } add      = combat_tier_large_mag }
				if = { limit = { has_variable = slot_soak_buff_combat_tier   var:slot_soak_buff_combat_tier   = 1 } add      = combat_tier_small_mag  }
				if = { limit = { has_variable = slot_soak_buff_combat_tier   var:slot_soak_buff_combat_tier   = 2 } add      = combat_tier_medium_mag }
				if = { limit = { has_variable = slot_soak_buff_combat_tier   var:slot_soak_buff_combat_tier   = 3 } add      = combat_tier_large_mag }
				if = { limit = { has_variable = slot_soak_debuff_round_tier  var:slot_soak_debuff_round_tier  = 1 } subtract = combat_tier_small_mag  }
				if = { limit = { has_variable = slot_soak_debuff_round_tier  var:slot_soak_debuff_round_tier  = 2 } subtract = combat_tier_medium_mag }
				if = { limit = { has_variable = slot_soak_debuff_round_tier  var:slot_soak_debuff_round_tier  = 3 } subtract = combat_tier_large_mag }
				if = { limit = { has_variable = slot_soak_debuff_combat_tier var:slot_soak_debuff_combat_tier = 1 } subtract = combat_tier_small_mag  }
				if = { limit = { has_variable = slot_soak_debuff_combat_tier var:slot_soak_debuff_combat_tier = 2 } subtract = combat_tier_medium_mag }
				if = { limit = { has_variable = slot_soak_debuff_combat_tier var:slot_soak_debuff_combat_tier = 3 } subtract = combat_tier_large_mag }
				min = 0
				max = 100
			}
		}
		set_variable = { name = slot_soak value = scope:new_soak }
	}
	if = {
		limit = { has_variable = slot_base_hit_chance }
		save_scope_value_as = {
			name = new_hit
			value = {
				value = var:slot_base_hit_chance
				if = { limit = { has_variable = slot_hit_buff_round_tier    var:slot_hit_buff_round_tier    = 1 } add      = combat_tier_small_mag  }
				if = { limit = { has_variable = slot_hit_buff_round_tier    var:slot_hit_buff_round_tier    = 2 } add      = combat_tier_medium_mag }
				if = { limit = { has_variable = slot_hit_buff_round_tier    var:slot_hit_buff_round_tier    = 3 } add      = combat_tier_large_mag }
				if = { limit = { has_variable = slot_hit_buff_combat_tier   var:slot_hit_buff_combat_tier   = 1 } add      = combat_tier_small_mag  }
				if = { limit = { has_variable = slot_hit_buff_combat_tier   var:slot_hit_buff_combat_tier   = 2 } add      = combat_tier_medium_mag }
				if = { limit = { has_variable = slot_hit_buff_combat_tier   var:slot_hit_buff_combat_tier   = 3 } add      = combat_tier_large_mag }
				if = { limit = { has_variable = slot_hit_debuff_round_tier  var:slot_hit_debuff_round_tier  = 1 } subtract = combat_tier_small_mag  }
				if = { limit = { has_variable = slot_hit_debuff_round_tier  var:slot_hit_debuff_round_tier  = 2 } subtract = combat_tier_medium_mag }
				if = { limit = { has_variable = slot_hit_debuff_round_tier  var:slot_hit_debuff_round_tier  = 3 } subtract = combat_tier_large_mag }
				if = { limit = { has_variable = slot_hit_debuff_combat_tier var:slot_hit_debuff_combat_tier = 1 } subtract = combat_tier_small_mag  }
				if = { limit = { has_variable = slot_hit_debuff_combat_tier var:slot_hit_debuff_combat_tier = 2 } subtract = combat_tier_medium_mag }
				if = { limit = { has_variable = slot_hit_debuff_combat_tier var:slot_hit_debuff_combat_tier = 3 } subtract = combat_tier_large_mag }
				# Stealth hit bonus, slot-scope mirror (see actor branch). Round-end
				# revert calls this per slot, so the bonus must follow
				# slot_is_stealthed across the round transition.
				if = { limit = { has_variable = slot_is_stealthed } add = pod_combat_stealth_hit_bonus }
				min = 20
				max = 95
			}
		}
		set_variable = { name = slot_hit_chance value = scope:new_hit }
	}
}

# Thin alias running on the current slot story (callers already in a
# slot-iteration body or scope-switched into a leader pointer).
pod_combat_recalc_slot_stats_effect = {
	pod_combat_recalc_slot_stats_via_story_effect = yes
}

############################################################
###### ROUND-END REVERT
############################################################

# Clears round-duration tier vars on every filled slot, then recalcs each
# slot's stats from base + remaining combat tiers. Rebuilding from scratch
# means clamping losses don't accumulate across rounds.
#
# Iterates pod_combat_active_slots (filled-only), so per-slot scope is the
# slot story; var:slot_char switches into the participant character.
pod_combat_revert_round_tiers_effect = {
	every_in_list = {
		variable = pod_combat_active_slots
		remove_variable = slot_soak_buff_round_tier
		remove_variable = slot_soak_debuff_round_tier
		remove_variable = slot_hit_buff_round_tier
		remove_variable = slot_hit_debuff_round_tier
		# Char-scope soak bridge vars (read by POD_soak_chance). Only
		# round-tier here; combat-tier persists until combat-end cleanup.
		var:slot_char = {
			remove_variable = pod_combat_tier_soak_buff_round
			remove_variable = pod_combat_tier_soak_debuff_round
		}
		pod_combat_recalc_slot_stats_via_story_effect = yes
	}
}

############################################################
###### COMBAT-END TIER CLEAR
############################################################

# Strips all tier vars from every slot; the underlying stat vars are
# removed separately by pod_combat_group_cleanup_effect.
#
# Must run BEFORE pod_combat_clear_slot_stories_effect so the slot stories
# are still alive while we strip their vars. Iterates the filled-only
# pod_combat_active_slots in slot-story scope.
pod_combat_clear_combat_tiers_effect = {
	every_in_list = {
		variable = pod_combat_active_slots
		remove_variable = slot_soak_buff_round_tier
		remove_variable = slot_soak_buff_combat_tier
		remove_variable = slot_soak_debuff_round_tier
		remove_variable = slot_soak_debuff_combat_tier
		remove_variable = slot_hit_buff_round_tier
		remove_variable = slot_hit_buff_combat_tier
		remove_variable = slot_hit_debuff_round_tier
		remove_variable = slot_hit_debuff_combat_tier
		# Char-scope soak bridge vars (read by POD_soak_chance); both
		# round and combat tiers, since this runs at combat end.
		var:slot_char = {
			remove_variable = pod_combat_tier_soak_buff_round
			remove_variable = pod_combat_tier_soak_buff_combat
			remove_variable = pod_combat_tier_soak_debuff_round
			remove_variable = pod_combat_tier_soak_debuff_combat
		}
	}
}
