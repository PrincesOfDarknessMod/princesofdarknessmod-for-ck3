##############################################################################################
# GROUP COMBAT INFRASTRUCTURE EFFECTS
# Handles slot-based workspace load/save, dispatch, initialization, assignment, and cleanup
# for the group combat system (two teams of arbitrary size; soft cap
# pod_combat_max_team_size).
#
# Roster layout (filled by the duel builder - create / add_ally / add_enemy).
# Every combatant carries (combat_team, team_position); position 0 = leader:
#   player / 0    = PC (always present)
#   player / 1+   = Ally slots
#   enemy  / 0    = Primary opponent / enemy leader (present in any fight)
#   enemy  / 1+   = Enemy reinforcements
#
# Terminology:
#   actor / target - workspace-relative, follows the action not team
#                      position (on enemy turns the enemy is "actor", the
#                      PC may be "target"). Drives the pod_combat_actor_* /
#                      pod_combat_target_* mirrors and the $ACTOR_SIDE$ /
#                      $TARGET_SIDE$ parametric values.
#   player / enemy - team membership (combat_team = flag:player/enemy).
#                      No fixed positional cap (only the soft cap above) -
#                      every slot is reached by iterating
#                      pod_combat_active_slots and matching
#                      (combat_team, team_position). The ONLY pointer vars
#                      are the two leaders, pod_combat_pc_slot (player/0)
#                      and pod_combat_enemy_leader_slot (enemy/0).
##############################################################################################

##############################################################################################
# SECTION 1: WORKSPACE LOAD EFFECTS
# Copy a slot's state into the actor or target workspace, via dispatch
# effects keyed on the current-actor / selected-target host pointers.
##############################################################################################

# Parametric workspace-load helper: reads the slot story's slot_X fields
# into the pod_combat_$SIDE$_* mirror on scope:host. Param SIDE = actor/target.
#
# Caller contract: current scope is the slot story; scope:host is the
# workspace owner. The top save_scope_as stashes the slot story so the
# post-switch `scope:host = { ... }` body can still reach slot fields via
# scope:slot_story.var:slot_X.
#
# New mirrored field: one line here + the writeback in
# pod_combat_save_side_to_story_helper + a remove_variable in the workspace
# cleanup; full recipe in COMBAT_REFERENCE.info (the workspace section).
#
# Prowess is NOT a mirrored field - it is stashed live from the character by
# pod_combat_stash_side_live_stats_effect below.
pod_combat_load_side_from_story_helper = {
	if = {
		limit = { has_variable = slot_char }
		save_scope_as = slot_story
		# The mirror lives on the FIGHT, so this one switch relocates the whole
		# helper. Every value below is an absolute scope:slot_story chain and
		# resolves the same from here as it did from the host - which is what
		# makes a 30-field move a one-line change.
		scope:host.var:pod_combat_session ?= {
			if = { limit = { scope:slot_story = { has_variable = slot_base_hit_chance } } set_variable = { name = pod_combat_$SIDE$_base_hit_chance value = scope:slot_story.var:slot_base_hit_chance } }
			set_variable = { name = pod_combat_$SIDE$_morale value = scope:slot_story.var:slot_morale }
			set_variable = { name = pod_combat_$SIDE$_hit_chance value = scope:slot_story.var:slot_hit_chance }
			# slot_energy_used default-inits to 0 at story creation, so always
			# present (no gate). Actor-side feeds AI energy-attrition weighting.
			set_variable = { name = pod_combat_$SIDE$_energy_used value = scope:slot_story.var:slot_energy_used }
			# Stance flags (presence-as-signal)
			if = {
				limit = { scope:slot_story = { has_variable = slot_is_guarding } }
				set_variable = { name = pod_combat_$SIDE$_is_guarding value = yes }
			}
			else = { remove_variable = pod_combat_$SIDE$_is_guarding }
			if = {
				limit = { scope:slot_story = { has_variable = slot_is_dodging } }
				set_variable = { name = pod_combat_$SIDE$_is_dodging value = yes }
			}
			else = { remove_variable = pod_combat_$SIDE$_is_dodging }
			if = {
				limit = { scope:slot_story = { has_variable = slot_is_stealthed } }
				set_variable = { name = pod_combat_$SIDE$_is_stealthed value = yes }
			}
			else = { remove_variable = pod_combat_$SIDE$_is_stealthed }
			# Spirit-damage capability + incorporeality flags. Read by
			# pod_combat_resolve_contact_effect (gates checks_spirit_gate
			# moves) and pod_combat_apply_retaliation_effect (sides swap for
			# the retaliator -> original-attacker gate).
			if = {
				limit = { scope:slot_story = { has_variable = slot_can_damage_spirit } }
				set_variable = { name = pod_combat_$SIDE$_can_damage_spirit value = yes }
			}
			else = { remove_variable = pod_combat_$SIDE$_can_damage_spirit }
			if = {
				limit = { scope:slot_story = { has_variable = slot_incorporeal_spirit } }
				set_variable = { name = pod_combat_$SIDE$_incorporeal_spirit value = yes }
			}
			else = { remove_variable = pod_combat_$SIDE$_incorporeal_spirit }
			# Initiative bonus + AP drain. Consumed at round-end (init_bonus)
			# or the target's next turn start (ap_drain).
			if = {
				limit = { scope:slot_story = { has_variable = slot_init_bonus } }
				set_variable = { name = pod_combat_$SIDE$_init_bonus value = scope:slot_story.var:slot_init_bonus }
			}
			else = { remove_variable = pod_combat_$SIDE$_init_bonus }
			if = {
				limit = { scope:slot_story = { has_variable = slot_ap_drain } }
				set_variable = { name = pod_combat_$SIDE$_ap_drain value = scope:slot_story.var:slot_ap_drain }
			}
			else = { remove_variable = pod_combat_$SIDE$_ap_drain }
			# Retaliation stances (12 vars: hit / miss / any x value / kind /
			# kind_from_weapon / morale flag)
			if = {
				limit = { scope:slot_story = { has_variable = slot_retaliate_hit } }
				set_variable = { name = pod_combat_$SIDE$_retaliate_hit value = scope:slot_story.var:slot_retaliate_hit }
			}
			else = { remove_variable = pod_combat_$SIDE$_retaliate_hit }
			if = {
				limit = { scope:slot_story = { has_variable = slot_retaliate_hit_kind } }
				set_variable = { name = pod_combat_$SIDE$_retaliate_hit_kind value = scope:slot_story.var:slot_retaliate_hit_kind }
			}
			else = { remove_variable = pod_combat_$SIDE$_retaliate_hit_kind }
			if = {
				limit = { scope:slot_story = { has_variable = slot_retaliate_hit_kind_from_weapon } }
				set_variable = { name = pod_combat_$SIDE$_retaliate_hit_kind_from_weapon value = yes }
			}
			else = { remove_variable = pod_combat_$SIDE$_retaliate_hit_kind_from_weapon }
			if = {
				limit = { scope:slot_story = { has_variable = slot_retaliate_hit_morale } }
				set_variable = { name = pod_combat_$SIDE$_retaliate_hit_morale value = yes }
			}
			else = { remove_variable = pod_combat_$SIDE$_retaliate_hit_morale }
			if = {
				limit = { scope:slot_story = { has_variable = slot_retaliate_miss } }
				set_variable = { name = pod_combat_$SIDE$_retaliate_miss value = scope:slot_story.var:slot_retaliate_miss }
			}
			else = { remove_variable = pod_combat_$SIDE$_retaliate_miss }
			if = {
				limit = { scope:slot_story = { has_variable = slot_retaliate_miss_kind } }
				set_variable = { name = pod_combat_$SIDE$_retaliate_miss_kind value = scope:slot_story.var:slot_retaliate_miss_kind }
			}
			else = { remove_variable = pod_combat_$SIDE$_retaliate_miss_kind }
			if = {
				limit = { scope:slot_story = { has_variable = slot_retaliate_miss_kind_from_weapon } }
				set_variable = { name = pod_combat_$SIDE$_retaliate_miss_kind_from_weapon value = yes }
			}
			else = { remove_variable = pod_combat_$SIDE$_retaliate_miss_kind_from_weapon }
			if = {
				limit = { scope:slot_story = { has_variable = slot_retaliate_miss_morale } }
				set_variable = { name = pod_combat_$SIDE$_retaliate_miss_morale value = yes }
			}
			else = { remove_variable = pod_combat_$SIDE$_retaliate_miss_morale }
			if = {
				limit = { scope:slot_story = { has_variable = slot_retaliate_any } }
				set_variable = { name = pod_combat_$SIDE$_retaliate_any value = scope:slot_story.var:slot_retaliate_any }
			}
			else = { remove_variable = pod_combat_$SIDE$_retaliate_any }
			if = {
				limit = { scope:slot_story = { has_variable = slot_retaliate_any_kind } }
				set_variable = { name = pod_combat_$SIDE$_retaliate_any_kind value = scope:slot_story.var:slot_retaliate_any_kind }
			}
			else = { remove_variable = pod_combat_$SIDE$_retaliate_any_kind }
			if = {
				limit = { scope:slot_story = { has_variable = slot_retaliate_any_kind_from_weapon } }
				set_variable = { name = pod_combat_$SIDE$_retaliate_any_kind_from_weapon value = yes }
			}
			else = { remove_variable = pod_combat_$SIDE$_retaliate_any_kind_from_weapon }
			if = {
				limit = { scope:slot_story = { has_variable = slot_retaliate_any_morale } }
				set_variable = { name = pod_combat_$SIDE$_retaliate_any_morale value = yes }
			}
			else = { remove_variable = pod_combat_$SIDE$_retaliate_any_morale }
			# Hit tier tracking vars (buff/debuff x round/combat). Presence on
			# slot = active tier; absence-as-signal preserved. Soak tiers are not
			# mirrored: they live on the combatant as pod_combat_tier_soak_*.
			if = {
				limit = { scope:slot_story = { has_variable = slot_hit_buff_round_tier } }
				set_variable = { name = pod_combat_$SIDE$_hit_buff_round_tier value = scope:slot_story.var:slot_hit_buff_round_tier }
			}
			else = { remove_variable = pod_combat_$SIDE$_hit_buff_round_tier }
			if = {
				limit = { scope:slot_story = { has_variable = slot_hit_buff_combat_tier } }
				set_variable = { name = pod_combat_$SIDE$_hit_buff_combat_tier value = scope:slot_story.var:slot_hit_buff_combat_tier }
			}
			else = { remove_variable = pod_combat_$SIDE$_hit_buff_combat_tier }
			if = {
				limit = { scope:slot_story = { has_variable = slot_hit_debuff_round_tier } }
				set_variable = { name = pod_combat_$SIDE$_hit_debuff_round_tier value = scope:slot_story.var:slot_hit_debuff_round_tier }
			}
			else = { remove_variable = pod_combat_$SIDE$_hit_debuff_round_tier }
			if = {
				limit = { scope:slot_story = { has_variable = slot_hit_debuff_combat_tier } }
				set_variable = { name = pod_combat_$SIDE$_hit_debuff_combat_tier value = scope:slot_story.var:slot_hit_debuff_combat_tier }
			}
			else = { remove_variable = pod_combat_$SIDE$_hit_debuff_combat_tier }
			# Character ref. Read by AI trait gates (actor-side) and the
			# damage pipeline as var:pod_combat_$TARGET_SIDE$_char.
			set_variable = { name = pod_combat_$SIDE$_char value = scope:slot_story.var:slot_char }
		}
		# scope:host, NOT the fight: the stash reaches the combatant THROUGH the
		# host's pod_combat_session pointer, so it needs a character in scope. The
		# ambient scope here is the SLOT story (the block above closed), so an
		# unwrapped call would miss by one object the same way a fight-scoped one
		# does. Same idiom as the pair-hit recalc below.
		scope:host = { pod_combat_stash_side_live_stats_effect = { SIDE = $SIDE$ } }
	}
}

# Per-action live stat stash. Caller scope = a CHARACTER whose pod_combat_session
# points at this fight; pod_combat_$SIDE$_char must already be written on the
# fight. Every line below reaches the fight THROUGH that pointer, so calling this
# with the fight itself in scope writes nothing at all: the limit's guarded deref
# resolves against an empty scope, and the else branch's removes address the same
# dead pointer and skip. Silent both ways - it shipped that way for a release.
#
# Soak and prowess are read live off the combatant rather than snapshotted at
# combat start: POD_soak_chance is the number PoD's wound roll actually rolls,
# and wound-track penalties move prowess during the fight. The stash exists
# because the AI weighting reads both inside random_list modifiers (once per
# candidate move) - hoisting the read to the once-per-load boundary keeps that
# a plain var read instead of re-walking POD_soak_chance per candidate.
pod_combat_stash_side_live_stats_effect = {
	if = {
		limit = {
			var:pod_combat_session ?= { has_variable = pod_combat_$SIDE$_char }
			exists = var:pod_combat_session.var:pod_combat_$SIDE$_char
		}
		var:pod_combat_session.var:pod_combat_$SIDE$_char ?= {
			# Floored like pod_combat_slot_prowess_value: stacked wound-track
			# penalties drive prowess below zero, and the pair hit formula
			# subtracts this raw.
			save_scope_value_as = { name = live_prowess value = { value = pod_combat_char_prowess_value min = 0 } }
			save_scope_value_as = { name = live_soak value = POD_soak_chance }
		}
		var:pod_combat_session ?= { set_variable = { name = pod_combat_$SIDE$_prowess value = scope:live_prowess } }
		var:pod_combat_session ?= { set_variable = { name = pod_combat_$SIDE$_soak value = scope:live_soak } }
	}
	else = {
		var:pod_combat_session ?= { remove_variable = pod_combat_$SIDE$_prowess }
		var:pod_combat_session ?= { remove_variable = pod_combat_$SIDE$_soak }
	}
}

# Thin wrapper - loads actor workspace from current slot story.
# Caller scope = slot story; scope:host must be saved before delegating.
pod_combat_load_actor_from_story_effect = {
	pod_combat_load_side_from_story_helper = { SIDE = actor }
}

# Thin wrapper - loads target workspace from current slot story.
# Caller scope = slot story; scope:host must be saved before delegating.
#
# Plus a target-specific pair-hit-chance recalc: slot-stored hit chance
# is leader-baseline (set once at init in
# pod_combat_calculate_all_hit_chances_effect), so without this every AI
# swap / AoE per-target iteration / load would push the wrong number into
# the damage roll. Reads pod_combat_actor_prowess (live-stashed by the
# preceding actor load) and pod_combat_target_prowess, then layers active hit
# tiers - so a combatant worn down by wounds loses accuracy as the fight runs.
pod_combat_load_target_from_story_effect = {
	pod_combat_load_side_from_story_helper = { SIDE = target }
	if = {
		limit = { has_variable = slot_char }
		scope:host = { pod_combat_recalc_pair_hit_chance_effect = yes }
	}
}

##############################################################################################
# SECTION 1b: KO MARKER + ALLY-DOWN MORALE SHOCK
# The single alive->down transition point. Every in-combat KO flip site calls
# pod_combat_mark_slot_ko_effect instead of writing slot_alive directly, so the
# per-KO side effect (the ally-down morale shock) fires exactly once even though
# several sites can re-detect the same KO. The init-time roster correction in
# POD_combat_finalize_duel_effect keeps a bare flip on purpose: a combatant who
# walks in already KO'd never fell mid-fight, so no shock and no log row.
##############################################################################################

# Scope = a slot story whose KO condition the caller has already verified
# (pod_combat_slot_should_be_marked_ko_trigger). The var:slot_alive = 1 gate
# makes the flip a real transition: re-detections on an already-down slot are
# no-ops. Caller contract: scope:host = the workspace owner (PC), as for the
# load/save helpers (the teammate walk and the log push re-enter it).
pod_combat_mark_slot_ko_effect = {
	if = {
		limit = { var:slot_alive = 1 }
		set_variable = { name = slot_alive value = 0 }
		# Ally-down morale shock: every still-living teammate loses
		# pod_combat_ally_down_morale_penalty morale, floored at 1. Morale 0 is
		# itself a KO source, so the floor both keeps the design rule (an ally
		# falling demoralizes but never breaks you outright) and guarantees the
		# shock can never trigger a KO of its own.
		# Char-dependent shock/log: skip when the combatant's character was removed
		# from the game (fatal-duel cleanup / despawned summon). The slot still flips
		# KO above; there is just no char to read or name.
		if = {
			limit = { exists = var:slot_char }
			save_scope_as = ally_down_slot
			var:slot_char = { save_scope_as = ally_down_char }
			set_variable = { name = pod_combat_ally_shock_hits value = 0 }
			# Literal team dispatch: combat_team is a flag var, so the teammate
			# filter compares literals through the parametric helper instead of
			# lifting the flag (lifted flags degrade to 'value' type and every
			# later flag compare silently type-errors).
			if = {
				limit = { var:combat_team = flag:player }
				pod_combat_ally_down_shock_team_helper = { TEAM = player }
			}
			else = {
				pod_combat_ally_down_shock_team_helper = { TEAM = enemy }
			}
			# Lift the hit count into a scope value (numbers survive that fine) and
			# clear the scratch var BEFORE the host re-entry, keeping this body
			# "slot-local writes first, scope switches last" like the DoT tick loop.
			save_scope_value_as = { name = ally_shock_hits value = var:pod_combat_ally_shock_hits }
			remove_variable = pod_combat_ally_shock_hits
			# One system row per KO that actually shook somebody. In a 1v1 (or when
			# every teammate already sits at the floor) nothing happened, so no row.
			if = {
				limit = { scope:ally_shock_hits > 0 }
				scope:host = {
					pod_combat_push_system_log_named_effect = { MESSAGE_KEY = ally_down_shock CHARACTER = scope:ally_down_char }
				}
			}
		}
	}
}

# Teammate walk for the marker above. $TEAM$ = player|enemy (literal). The
# downed slot is already flipped, so the alive filter excludes it without a
# self-check. Caller scope is the downed SLOT STORY, but pod_combat_active_slots
# lives on the HOST, so the walk must re-enter scope:host first: a list read off
# a scope that lacks it iterates zero times with no error.
pod_combat_ally_down_shock_team_helper = {
	scope:host = {
		scope:host.var:pod_combat_session ?= {
			every_in_list = {
				variable = pod_combat_active_slots
				limit = {
					var:combat_team = flag:$TEAM$
					pod_combat_slot_is_alive_trigger = yes
				}
				pod_combat_apply_ally_down_shock_to_slot_effect = yes
			}
		}
	}
}

# Per-teammate shock body (scope = a living teammate slot story). At 1 morale
# there is nothing left to take, so the slot does not count as hit. The
# scope:ally_down_slot counter bump lives inside this call so the teammate
# loop's iteration scope never sees an inline scope switch.
pod_combat_apply_ally_down_shock_to_slot_effect = {
	if = {
		limit = { var:slot_morale > 1 }
		set_variable = { name = slot_morale value = { value = var:slot_morale subtract = pod_combat_ally_down_morale_penalty min = 1 } }
		scope:ally_down_slot = { change_variable = { name = pod_combat_ally_shock_hits add = 1 } }
		# Mirror the direct slot write into the live workspace when this shocked
		# teammate is the current actor / selected target, or the end-of-turn
		# re-save (pod_combat_execute_ai_turn_effect) flushes a stale pre-shock
		# morale back over the slot - a dominated actor KO-ing a teammate on its
		# final AP would otherwise lose its own shock. scope:host is the
		# workspace owner (guaranteed by the mark-KO caller contract); the
		# matcher triggers self-guard on it, so they fail closed off-combat.
		# The post-shock morale rides a scope value (a number, so no flag
		# degradation) to cross the scope:host switch cleanly.
		save_scope_value_as = { name = ally_shock_new_morale value = var:slot_morale }
		if = {
			limit = { pod_combat_slot_is_current_actor_trigger = yes }
			scope:host.var:pod_combat_session ?= { set_variable = { name = pod_combat_actor_morale value = scope:ally_shock_new_morale } }
		}
		if = {
			limit = { pod_combat_slot_is_selected_target_trigger = yes }
			scope:host.var:pod_combat_session ?= { set_variable = { name = pod_combat_target_morale value = scope:ally_shock_new_morale } }
		}
	}
}

##############################################################################################
# SECTION 2: WORKSPACE SAVE EFFECTS
# Two effects that copy the actor or target workspace back into a slot story.
##############################################################################################

# Parametric workspace-save helper: reads pod_combat_$SIDE$_* from
# scope:host into the slot story's slot_X fields. Param SIDE = actor/target.
# Includes the alive-flip for BOTH sides - an actor can KO itself via
# self-damage or incoming retaliation just as a target can.
#
# Caller contract: current scope is the slot story; scope:host is the
# workspace owner. The top `has_variable = pod_combat_$SIDE$_char` gate
# skips slots whose mirrors weren't loaded.
#
# New mirrored field: see the recipe on the load helper above.
pod_combat_save_side_to_story_helper = {
	if = {
		limit = { scope:host.var:pod_combat_session ?= { has_variable = pod_combat_$SIDE$_char } }
		# Flush post-action morale to the slot FIRST so the shared KO trigger reads it.
		set_variable = { name = slot_morale value = scope:host.var:pod_combat_session.var:pod_combat_$SIDE$_morale }
		# Alive-flip on any KO condition, both sides, via the shared
		# pod_combat_slot_should_be_marked_ko_trigger (4 KO sources documented
		# there). The marker owns the transition gate + ally-down shock (§1b).
		if = {
			limit = { pod_combat_slot_should_be_marked_ko_trigger = yes }
			pod_combat_mark_slot_ko_effect = yes
		}
		if = { limit = { scope:host.var:pod_combat_session ?= { has_variable = pod_combat_$SIDE$_base_hit_chance } } set_variable = { name = slot_base_hit_chance value = scope:host.var:pod_combat_session.var:pod_combat_$SIDE$_base_hit_chance } }
		# slot_morale is written at the top of this block (before the KO check).
		set_variable = { name = slot_hit_chance     value = scope:host.var:pod_combat_session.var:pod_combat_$SIDE$_hit_chance }
		if = {
			limit = { scope:host.var:pod_combat_session ?= { has_variable = pod_combat_$SIDE$_energy_used } }
			set_variable = { name = slot_energy_used value = scope:host.var:pod_combat_session.var:pod_combat_$SIDE$_energy_used }
		}
		# Stance flags (presence-as-signal)
		if = {
			limit = { scope:host.var:pod_combat_session ?= { has_variable = pod_combat_$SIDE$_is_guarding } }
			set_variable    = { name = slot_is_guarding value = yes }
		}
		else = { remove_variable = slot_is_guarding }
		if = {
			limit = { scope:host.var:pod_combat_session ?= { has_variable = pod_combat_$SIDE$_is_dodging } }
			set_variable    = { name = slot_is_dodging value = yes }
		}
		else = { remove_variable = slot_is_dodging }
		if = {
			limit = { scope:host.var:pod_combat_session ?= { has_variable = pod_combat_$SIDE$_is_stealthed } }
			set_variable    = { name = slot_is_stealthed value = yes }
		}
		else = { remove_variable = slot_is_stealthed }
		# Initiative bonus writeback. Round-end initiative recalc reads
		# slot_init_bonus directly outside any workspace lifecycle.
		if = {
			limit = { scope:host.var:pod_combat_session ?= { has_variable = pod_combat_$SIDE$_init_bonus } }
			set_variable    = { name = slot_init_bonus value = scope:host.var:pod_combat_session.var:pod_combat_$SIDE$_init_bonus }
		}

pod_combat_load_workspace_effect = {
	# Load current actor -> actor workspace, then selected target ->
	# target workspace, by iterating pod_combat_active_slots and
	# matching each slot's (combat_team, team_position) against the
	# current-actor / selected-target host pointers. The iteration limit fires for at
	# most one slot per pass - the load helper inside the body is the
	# story-scoped variant, which writes to scope:host (this scope).
	save_scope_as = host
	var:pod_combat_session ?= {
		every_in_list = {
			variable = pod_combat_active_slots
			limit = {
				pod_combat_slot_is_current_actor_trigger = yes
			}
			pod_combat_load_actor_from_story_effect = yes
		}
	}
	var:pod_combat_session ?= {
		every_in_list = {
			variable = pod_combat_active_slots
			limit = {
				pod_combat_slot_is_selected_target_trigger = yes
			}
			pod_combat_load_target_from_story_effect = yes
		}
	}

	# Push the pair-recomputed hit chances back to the actor / target slot
	# stories so the per-slot GUI display matches the upcoming damage roll.
	var:pod_combat_session ?= {
		every_in_list = {
			variable = pod_combat_active_slots
			limit = {
				pod_combat_slot_is_current_actor_trigger = yes
			}
			if = {
				limit = { scope:host.var:pod_combat_session ?= { has_variable = pod_combat_actor_base_hit_chance } }
				set_variable = { name = slot_base_hit_chance value = scope:host.var:pod_combat_session.var:pod_combat_actor_base_hit_chance }
			}
			if = {
				limit = { scope:host.var:pod_combat_session ?= { has_variable = pod_combat_actor_hit_chance } }
				set_variable = { name = slot_hit_chance value = scope:host.var:pod_combat_session.var:pod_combat_actor_hit_chance }
			}
		}
	}
	var:pod_combat_session ?= {
		every_in_list = {
			variable = pod_combat_active_slots
			limit = {
				pod_combat_slot_is_selected_target_trigger = yes
			}
			if = {
				limit = { scope:host.var:pod_combat_session ?= { has_variable = pod_combat_target_base_hit_chance } }
				set_variable = { name = slot_base_hit_chance value = scope:host.var:pod_combat_session.var:pod_combat_target_base_hit_chance }
			}
			if = {
				limit = { scope:host.var:pod_combat_session ?= { has_variable = pod_combat_target_hit_chance } }
				set_variable = { name = slot_hit_chance value = scope:host.var:pod_combat_session.var:pod_combat_target_hit_chance }
			}
		}
	}

	if = {
		limit = { var:pod_combat_session ?= { has_variable = pod_combat_actor_hit_chance } }
		var:pod_combat_session ?= { set_variable = { name = pod_combat_current_hit_chance value = var:pod_combat_actor_hit_chance } }
	}
}

# -- PC unlock bake  --
# Adds each unlocked move to the ACTOR's per-character pod_combat_unlocked_moves
# list, so per-frame surfaces (POD_can_execute_combat_move_trigger,
# pod_combat_loadout_option) test cheap list membership instead of re-running the
# unlock dispatcher every frame. The list lives on the viewing/fighting PC,
# NOT as a flag on the globally-shared move stories - so one player opening the
# editor can no longer rewrite another player's unlock state. The builders clear
# the list before refilling, so this only ever adds. FULL dispatcher per move
# preserves the multi-splat union.
pod_combat_bake_one_unlock_effect = {
	# $MOVE$ = saved-scope move story ; $ACTOR$ = saved-scope PC. BOTH are saved
	# scopes (NOT `this`/`root`): STORY so the dispatcher's $STORY$.var reads stay
	# stable across its internal $ACTOR$ scope-switch, and ACTOR because `root` is
	# NOT the PC at the finalize call site (it can be an unset scope - the launcher
	# reaches finalize via scope switches that leave `this` = host but root unbound)
	# and is also unset during a tooltip-build walk of the debug launcher. The
	# exists guard makes a scopeless call a no-op instead of erroring on the
	# dispatcher's actor scope-switch (which is what broke the gated move buttons).
	if = {
		limit = {
			exists = $ACTOR$
			pod_combat_actor_unlocks_move_trigger = { STORY = $MOVE$ ACTOR = $ACTOR$ }
		}
		$ACTOR$ = { add_to_variable_list = { name = pod_combat_unlocked_moves target = $MOVE$ } }
	}
}

# uses_weapon / ap_refund per-effect marks) and pod_combat_ai_eff_ap_drain.
#
# Those marks are the one piece of per-fight state that cannot live on the
# fight story. The roll is a random_in_list over the loadout, so `this` is a
# move story when the band clause reads the score - a fight-scoped scalar is
# not addressable from there. They stay on the catalogue, and are cleaned up
# here instead.
#
# Within a fight the scoring walk is unconditional and set-or-clear, which is
# what makes a stale mark unreachable (see the NOTE on the roll's limit). This
# effect covers the other end: without it the last action's marks outlive the
# fight on SHARED, save-persistent catalogue stories.
#
# Char scope. Called from pod_combat_duel_end_effect and the dead-host release,
# NOT from cleanup, and that placement is the whole point: both callers run while
# every combatant is still a real character.
#
# CK3 garbage-collects an unlanded duel loser, and the resulting weak reference is
# not a character scope at all - has_variable_list below then errors with "This
# scope doesn't support variables. Scope: <name> weak". Neither `?=` nor
# `exists = var:slot_char` screens that: both still resolve on a same-tick weak
# reference (measured, 2026-08-21 - the KO predicate's exists clause catches a
# LATER-flushed reference, not this one). Only `is_alive` reads false, and gating
# on liveness would skip a corpse whose marks we can still clear.
#
# So the marks are cleared BEFORE anything dies. Reading the list off a corpse is
# fine (reads are tolerated on a dead-but-present character, only mutation errors),
# and every mutation here lands on a move story, which is legal whatever state the
# owning character is in - so the callers' own is_alive gates are the only screen
# this needs.
pod_combat_clear_ai_move_marks_effect = {
	if = {
		limit = { has_variable_list = pod_combat_loadout }
		every_in_list = {
			variable = pod_combat_loadout
			remove_variable = pod_combat_ai_admitted
			remove_variable = pod_combat_ai_score
			remove_variable = pod_combat_ai_bonus_met
			remove_variable = pod_combat_ai_bonus_met_count
			remove_variable = pod_combat_ai_bonus_met_uses_weapon
			remove_variable = pod_combat_ai_bonus_met_ap_refund
			# The one mark with no per-pass clear twin (its precompute branch is
			# static per move), so this is the only place it is ever removed.
			remove_variable = pod_combat_ai_eff_ap_drain
		}
	}
}

pod_combat_init_slot_from_character_effect = {
	# Runs in the SLOT-STORY scope (called once per slot from the
	# pod_combat_init_all_slots_effect loop). Params:
	#   $CHARACTER$    = the slot's occupant (the loop passes var:slot_char).
	#   $ENEMY_LEADER$ = the OPPOSING team's leader - the CASTER for the
	#     "enemy knight" discipline buff: scope:pod_combat_target for
	#     player-team slots, scope:pod_combat_player_leader for enemy-team
	#     slots. Supplied by the caller so this effect needs no team branch.
	#
	# AI fera shapeshift runs upfront in POD_combat_finalize_duel_effect (before
	# this), so every AI CASTER already has its crinos modifier here. A human who
	# shifts LATER is not covered by this pass at all - pod_combat_delirium_effects.txt
	# re-derives the Delirium half of the battery at the turn boundaries.
	#
	# Is a human driving this slot? Captured HERE, in slot scope, because
	# slot_human_controller is a slot var and everything below runs inside
	# $CHARACTER$ - a test in there would silently ask the character instead.
	save_scope_value_as = { name = pod_combat_init_slot_is_driven value = 0 }
	if = {
		limit = { has_variable = slot_human_controller }
		save_scope_value_as = { name = pod_combat_init_slot_is_driven value = 1 }
	}
	$CHARACTER$ = {
		# Discipline combat buffs. Named scopes (not root) to match PoD's duel
		# pattern and dodge text-substitution issues in nested parametric effects.
		#
		# Both battery halves also deal real wounds off caster-perk branches with no
		# victim filter. That damage is INTENTIONAL and stays: the perk descriptions
		# promise it explicitly for duels ("or to your foe during [duels]" in
		# LIONS_BITE_PERK_EFFECT / WHITE_FIRE_PERK_EFFECT / BS_HEAT_WAVE_PERK_EFFECT /
		# MADDENING_HOWL_PERK_EFFECT and friends), so suppressing it inside the
		# minigame would silently break the advertised contract of every one of them.
		# The fatality = no writeback is what keeps it survivable - see
		# pod_combat_finalize_combat_health_effect.
		POD_enemy_knight_set_discipline_battle_effect = { CASTER = $ENEMY_LEADER$ }
		# The battery's wounds can no longer kill here (the death gate is stamped for
		# every mode in POD_combat_finalize_duel_effect, ahead of this), but a launcher
		# can still seat a character who was already dead. Everything below therefore
		# re-checks: var and flag mutation ERRORS on a dead reference (it silently
		# no-ops only on a live character that lacks the var), so an unguarded tail
		# here is error spam, not a harmless miss.
		if = {
			limit = { is_alive = yes }
			POD_ally_knight_set_discipline_battle_effect = { CASTER = $CHARACTER$ }
			force_character_skill_recalculation = yes
			# Clear any soak tier left over from an earlier fight. The combat-end
			# clear cannot reach a combatant who died during that fight, so this is
			# what stops a resurrected character starting a new fight still buffed.
			remove_variable = pod_combat_tier_soak_buff_round
			remove_variable = pod_combat_tier_soak_buff_combat
			remove_variable = pod_combat_tier_soak_debuff_round
			remove_variable = pod_combat_tier_soak_debuff_combat
			# Same reasoning for move-applied marker modifiers: teardown misses a
			# combatant who died mid-fight, and an abandoned fight never runs
			# teardown at all, so strip them on the way IN as well.
			remove_character_modifier = pod_combat_anima_flare
			remove_character_modifier = pod_combat_predator_focus
			remove_character_modifier = pod_combat_kai_focus
			# And for the weapon-attack marker, which is what tells PoD's
			# classifier a wound came from a swing. It drops at the end of every
			# dispatch, so only a fight abandoned mid-action can strand one.
			remove_variable = pod_combat_weapon_attack
			remove_variable = pod_combat_weapon_soak_pen
			# Clear per-fight AI scratch so a fresh fight starts clean.
			# Human-occupied slots skipped (persistent loadout, no AI history).
			if = {
				limit = { is_ai = yes }
				pod_combat_clear_ai_scratch_effect = yes
			}
			# The loadout is a separate question from the AI's memory: wipe it only
			# where nothing human is driving, so a launcher can curate a proxy's or
			# a clone's repertoire before finalize and have it survive. Left alone,
			# the finalize seed still fills an empty one with the AI's full
			# repertoire, so an uncurated body is armed either way.
			#
			# scope: value, not a bare has_variable: this block is CHARACTER scope
			# and slot_human_controller is a SLOT var, so testing it here would ask
			# the character, always answer no, and change nothing at all.
			if = {
				limit = {
					is_ai = yes
					scope:pod_combat_init_slot_is_driven = 0
				}
				clear_variable_list = pod_combat_loadout
			}
		}
	}
	# Soak and prowess are NOT snapshotted here: both are read live off
	# var:slot_char wherever they are needed, like HP. The discipline buffs above
	# are character modifiers held to duel end, so a live read still includes
	# them, and a live read additionally tracks the wound penalties a combatant
	# picks up during the fight.
	#
	# Shared starting-morale SV (cap or tuned override, + the designer hook,
	# floor 1) - same source as the slot constructor, so the hook survives init.
	pod_combat_set_slot_var_effect = { NAME = morale    VALUE = pod_combat_starting_morale_value }
	pod_combat_set_slot_alive_effect = { VALUE = 1 }
	# AI energy-use counter: the AI move picker layers stacking weight penalties
	# on it to bias away from energy spends as the pool depletes (see
	# pod_combat_ai_effects.txt). Selection bias only; the AI still pays the
	# real splat energy via POD_splat_agnostic_energy_cost.
	pod_combat_set_slot_var_effect = { NAME = energy_used VALUE = 0 }
	# ── Spirit / incorporeality capability flags ─────────────────
	# Two per-slot capability flags captured once at init, mirroring PoD's
	# POD_damage_spirit_custom_trigger semantics (full explanation at the load
	# helper / resolver): incorporeal_spirit (target unhittable by mundane
	# physical attacks, via POD_can_be_killed_physically_trigger = no) and
	# can_damage_spirit (actor can strike spirits, via
	# POD_can_damage_spirit_trigger). Gate the checks_spirit_gate check in
	# pod_combat_resolve_contact_effect.
	pod_combat_remove_slot_var_effect = { NAME = incorporeal_spirit }
	if = {
		limit = { $CHARACTER$ = { POD_can_be_killed_physically_trigger = no } }
		pod_combat_set_slot_var_effect = { NAME = incorporeal_spirit VALUE = yes }
	}
	pod_combat_remove_slot_var_effect = { NAME = can_damage_spirit }
	if = {
		limit = { $CHARACTER$ = { POD_can_damage_spirit_trigger = yes } }
		pod_combat_set_slot_var_effect = { NAME = can_damage_spirit VALUE = yes }
	}
	pod_combat_remove_slot_var_effect = { NAME = is_guarding }
	pod_combat_remove_slot_var_effect = { NAME = is_dodging }
	# ── State marker modifier ────────────────────────────────────
	# pod_combat_duel_active lets other systems (feeding, schemes, travel,
	# other-mod activity events) gate on has_character_modifier and defer mid-duel.
	# Cleared in pod_combat_duel_end_effect; days = 30 is a safety net (combat
	# lasts seconds of game time, so it never clips a real duel).
	$CHARACTER$ = {
		add_character_modifier = {
			modifier = pod_combat_duel_active
			days     = 30
		}
	}
	# Hit chance calculated later after all combatants are initialized

	# HP display mirror. Seed the wound-bar mirror
	# vars (slot scope here) for the first render and any mid-fight summon seat;
	# this effect runs per-slot via init_all_slots (finalize) AND for mid-fight
	# seats. Refreshed thereafter at the save_workspace / tick_dots chokepoints.
	pod_combat_write_slot_hp_mirror_effect = yes
}

##############################################################################################
# SECTION 5: DUEL BUILDER - create / add combatant
#
# Callers build a fight imperatively:
#   <PC scope> = {
#       POD_combat_create_duel_effect    = { INITIATOR=.. ATTACKER=.. DEFENDER=.. FATALITY=.. FIXED=.. LOCALE=.. OUTPUT_EVENT=.. }
#       POD_combat_duel_add_ally_effect  = { CHARACTER = <ally>  }   # 0+ times
#       POD_combat_duel_add_enemy_effect = { CHARACTER = <enemy> }   # 1+ times (first -> enemy/0 leader)
#       POD_combat_finalize_duel_effect  = yes
#   }
#
# Slot layout / roster: see the file header. Team size bounded only by the
# soft cap pod_combat_max_team_size. PC is always the combat host; every
# builder runs in the PC scope.
#
# create_duel opens the shell card (pod_combat.9995) as its first act, so any
# caller - PoD's single-combat launcher or a direct builder call - hands its own
# saved scopes to the follow-up. Nothing may be saved into the caller's chain
# between the caller's own scope setup and create_duel that the follow-up must
# not see.
#
# Slot stat-init is DEFERRED to finalize (add_* records occupancy only):
# each combatant's discipline buff needs the OPPOSING team leader as CASTER,
# unknown until the full roster is placed. finalize derives the leaders then
# runs pod_combat_init_all_slots_effect once.
##############################################################################################

# Shared body of both builders. Do NOT call this directly - use
# POD_combat_create_duel_effect (the caller fights) or
# POD_combat_create_proxy_duel_effect (the caller directs combatants it is not).
# One core rather than two copies because everything except the caller's own seat
# is identical, and a forked ~160-line builder would drift.
pod_combat_create_duel_core_effect = {
	# Scaffolds host state, opens the shell card, creates the duel story (the
	# caller-context handoff), and optionally seats the caller in slot 0. Runs in
	# the caller's scope.
	#
	# Seating params:
	#   SEAT_HOST - yes: the caller takes player/0 and is marked as driving it.
	#       no: the caller holds the session pointer (so their window binds and
	#       they own the shell card) but occupies no slot; the player team is
	#       built entirely from add_ally, and the launcher marks one of those
	#       seats itself.
	#   INITIAL_PLAYER_SIZE - 1 when the caller is seated, 0 when not. This is
	#       what decides where the FIRST add_ally lands, and player/0 must end up
	#       occupied either way: the player-team leader pointer, the enemy team's
	#       discipline CASTER, the whole hit-chance baseline pass and the
	#       mid-fight enemy adds all key off it, and the hit-chance pass fails
	#       SILENTLY when it is missing.
	#
	# Params (all forwarded onto the duel story for pod_combat.9999):
	#   INITIATOR / ATTACKER / DEFENDER - the original duel parties in
	#       PoD's sc_* sense (NOT necessarily the PC / enemy leader - the PC may
	#       be the defender). pod_combat.9999 restores these so win/loss
	#       attribution and scope:sc_victor logic stay correct.
	#   FATALITY / FIXED / LOCALE - duel rule flags.
	#   OUTPUT_EVENT - follow-up event fired on INITIATOR at combat end.

	# ── The duel's canonical scope names, then the shell card ──
	# pod_combat.9995 inherits the caller's saved-scope table as it stands at the
	# trigger_event below and holds it open for the whole fight, which is what
	# lets the follow-up read the launching event's own aliases (scope:spirit, an
	# artifact scope:target, flag-typed scope values) instead of the degraded
	# story rebuild. COMBAT_REFERENCE.info §10 has the model.
	#
	# These saves have to precede the card: its conclude gate
	# (pod_combat_shell_owns_live_fight_trigger) and .9999's live path read the
	# sc_* principals off the inherited table, and the finalization step reads
	# scope:fatality. configure_start_single_combat_effect already saved these
	# exact names and values, so re-saving is a no-op on that route; a caller
	# that drives the builder directly gets a self-sufficient card for free.
	$INITIATOR$ = { save_scope_as = sc_initiator }
	$ATTACKER$  = { save_scope_as = sc_attacker }
	$DEFENDER$  = { save_scope_as = sc_defender }
	save_scope_value_as = { name = fatality value = flag:$FATALITY$ }
	save_scope_value_as = { name = fixed    value = flag:$FIXED$ }
	save_scope_value_as = { name = locale   value = flag:$LOCALE$ }

	# Equally, the card has to precede the rest of the builder: the builder's own
	# helpers save generic names into this same chain
	# (pod_combat_calc_turn_orders_effect writes scope:host and scope:me), which
	# would otherwise ride along and shadow same-named scopes the parent set - an
	# activity duel's scope:host is the activity host, and silently handing the
	# follow-up the combat host instead is worse than a missing-scope error.
	#
	# hidden_effect keeps "Triggers event" out of the duel option tooltips of the
	# ~90 callers routed through configure_start_single_combat_effect; it does not
	# affect scope propagation.
	hidden_effect = { trigger_event = { id = pod_combat.9995 } }

	save_scope_as = combat_pc

	# Save-compat: rebuild the move catalogue if the save predates the current
	# version (the configure-skills decision does the same). The empty-loadout
	# fallback below reseeds the PC after POD_reload_combat_moves wipes loadouts.
	hidden_effect = {
		if = {
			limit = {
				OR = {
					NOT = { exists = global_var:POD_combat_version }
					global_var:POD_combat_version != POD_current_combat_version
				}
				# Don't end/recreate the global stories while ANY player has a
				# live fight referencing them (they reload on the next no-combat open).
				NOT = { pod_combat_any_fight_active_trigger = yes }
			}
			POD_reload_combat_moves = yes
		}
	}
	# Only when the caller is going to fight. A director arming itself would be
	# seeding a loadout nothing reads.
	if = {
		limit = {
			always = $SEAT_HOST$
			pod_combat_no_moves_equipped_trigger = yes
		}
		POD_apply_default_combat_loadout = yes
	}

	# Close the config window if open. The sheet scratch list goes with it -
	# this is a close path like pod_combat_close_skills, and skipping the
	# clear would leave one story ref per catalogue move in the save.
	remove_variable = pod_combat_config_open
	clear_variable_list = pod_combat_sheet_rows

	# Defensive teardown of leftover state from a crashed / force-quit prior fight
	# (orphaned slot stories + stale workspace aliases). Keyed on the POINTER, not on
	# the fight's own pod_combat_active: bare presence is deliberately the widest
	# question we can ask, so this also reclaims a fight that ended without cleanup
	# and one whose story is already gone. That is what stops a stale pointer from
	# permanently blackballing its owner - every loop inside cleanup is ?=-guarded,
	# so a dangling pointer no-ops through and cleanup removes the pointer itself.
	if = {
		limit = { has_variable = pod_combat_session }
		# A stale fight also skipped its combat-end teardown, so the
		# pod_combat_nonfatal_active death-gate flag and the pod_combat_duel_active
		# marker can be stuck on the prior combatants. Both iterate the slot list,
		# which pod_combat_cleanup_effect destroys when it ends the fight story,
		# so strip them FIRST. We skip the wound-cap (the discarded fight is not being
		# finalized) and just clear the flag/marker.
		pod_combat_clear_nonfatal_flag_effect = yes
		pod_combat_duel_end_effect = yes
		pod_combat_cleanup_effect = yes
	}

	# ── The fight story ──
	# Anchored on Charon (immortal, off-map) so it survives the PC dying
	# mid-combat; participants hold one pointer (pod_combat_session), cleanup
	# is one end_story. Var writes go inside hidden_effect_new_object so
	# tooltip-build (which skips create_story) doesn't error on the scope
	# reads. COMBAT_REFERENCE §7.
	#
	# Created before any per-fight state is seeded, so that state has an owner
	# to be written to from the first line of the fight onward.
	character:POD_wraith_01 = {
		create_story = {
			type = POD_combat_fight
			save_scope_as = pod_combat_duel
		}
	}

	hidden_effect_new_object = {
		# Dual-named sc_ctx, which is how the .9999 fallback addresses the
		# story. Must live INSIDE the new-object block - tooltip-build skips
		# create_story, so a normal-path scope:pod_combat_duel reference would
		# error 'Undefined event target'.
		scope:pod_combat_duel = { save_scope_as = sc_ctx }

		scope:pod_combat_duel = {
			set_variable = { name = initiator value = $INITIATOR$ }
			set_variable = { name = attacker  value = $ATTACKER$ }
			set_variable = { name = defender  value = $DEFENDER$ }
			set_variable = { name = fatality  value = flag:$FATALITY$ }
			set_variable = { name = fixed     value = flag:$FIXED$ }
			set_variable = { name = locale    value = flag:$LOCALE$ }

			# Optional duel rule: summoning. Default allowed; barred when the
			# launching chain ran POD_combat_ban_duel_summons_effect (contract on
			# that effect). Both writes are flag LITERALS - copying the opt-in
			# through a value read would degrade the flag, and every later
			# comparison against it would then be a silently-skipped type error.
			set_variable = { name = summons value = flag:allowed }
			if = {
				limit = { exists = scope:pod_combat_ban_summons }
				set_variable = { name = summons value = flag:banned }
			}

			POD_save_event_as_var = { VAR = follow_up ID = $OUTPUT_EVENT$ }

			# The character this fight's events are addressed to: the
			# pod_combat.9995 shell recipient, and the fallback recipient for
			# anything else that needs a live character. Not "the owner of the
			# fight's state" - that is this story.
			set_variable = { name = ctx_host value = scope:combat_pc }

			# Team sizes = next free team_position per side; PC occupies player/0
			# (seated below) so player starts at 1, enemy starts at 0 (first
			# add_enemy = enemy/0). Written through the bound scope rather than
			# the host's pod_combat_session, which is not set until below.
			set_variable = { name = pod_combat_round value = 1 }
			set_variable = { name = pod_combat_player_team_size value = $INITIAL_PLAYER_SIZE$ }
			set_variable = { name = pod_combat_enemy_team_size  value = 0 }
		}

		# Every participant points at the fight through this one var; the host
		# gets it here, mid-fight arrivals get it as they seat. It is also what
		# answers "is this character in a fight" now that the answer has to
		# survive the fight's state living somewhere other than on them.
		scope:combat_pc = {
			set_variable = { name = pod_combat_session value = scope:pod_combat_duel }
		}
	}

	# ── caller -> player/0 (occupancy only; stat init deferred to finalize) ──
	# ...and that seat is the one this human plays. Stated here by the BUILDER
	# rather than inferred at finalize from "player team, position 0": a launcher
	# that wants a second played body, or wants the human to direct combatants it
	# is not, says so with its own mark call and needs no engine change.
	#
	# Skipped entirely in proxy mode, where the player team starts empty and the
	# launcher does both the seating and the marking.
	if = {
		limit = { always = $SEAT_HOST$ }
		pod_combat_set_slot_char_effect = { TEAM = player POSITION = 0 CHARACTER = scope:combat_pc }
		pod_combat_mark_slot_controlled_effect = { TEAM = player POSITION = 0 CONTROLLER = scope:combat_pc }
	}

	# This fight has a shell card holding its scope table, so the combat window's
	# scope-less End Combat button stays hidden and .9999 expects the live route.
	# Written after the stale-fight teardown so the pointer is the NEW fight's; the
	# flag cannot survive a prior fight, since that fight's story is a different
	# object that got end_story'd.
	var:pod_combat_session ?= { set_variable = { name = pod_combat_sc_shell value = yes } }
}

# ── Optional duel rule: bar summoning ──
# Run in the launching chain BEFORE the duel starts - ahead of
# configure_start_single_combat_effect on the standard route, or ahead of
# POD_combat_create_duel_effect on a raw builder call:
#
#   POD_combat_ban_duel_summons_effect = yes
#   configure_start_single_combat_effect = { ... }
#
# Every summons_combatant move is then barred for BOTH sides for that fight: the
# shared move gate greys the PC's button with a reason and refuses the AI's pick,
# and the summon glue repeats the check. Event-driven arrivals
# (pod_combat_add_*_midfight_effect and the scheduled reinforcements) are NOT
# affected - the rule is about what a combatant may do, not who may enter.
#
# A saved scope value rather than a builder parameter, for two reasons. The
# first-fight intro DEFERS the launch into a later event, and only the scope table
# crosses that boundary - the same constraint that forces the fatality ladder in
# pod_combat_intro_effects.txt, which a second rule would have multiplied. And
# staying optional is what lets the ~90 existing duel call sites go untouched;
# CK3 has no default for an unsupplied macro argument.
#
# The value rides the shell card's inherited table into the follow-up, so a REMATCH
# launched from that follow-up inherits the bar. That is usually what a caller
# wants; POD_combat_allow_duel_summons_effect drops it when it is not.
POD_combat_ban_duel_summons_effect = {
	save_scope_value_as = { name = pod_combat_ban_summons value = yes }
}

# Drops the bar again, so the next duel launched from this chain allows summoning.
POD_combat_allow_duel_summons_effect = {
	# Guarded: clear_saved_scope on a name that was never saved errors.
	if = {
		limit = { exists = scope:pod_combat_ban_summons }
		clear_saved_scope = pod_combat_ban_summons
	}
}

	pod_combat_create_duel_core_effect = {
		SEAT_HOST = yes
		INITIAL_PLAYER_SIZE = 1
		INITIATOR = $INITIATOR$
		ATTACKER  = $ATTACKER$
		DEFENDER  = $DEFENDER$
		FATALITY  = $FATALITY$
		FIXED     = $FIXED$
		LOCALE    = $LOCALE$
		OUTPUT_EVENT = $OUTPUT_EVENT$
	}
}

# The director builder: the launching human DIRECTS the fight without joining it.
# Their window binds (they hold the session pointer and own the shell card) but
# they occupy no slot, take no wounds, and are not "in combat" for the busy-gates
# - pod_combat_is_seated_trigger reads false for them.
#
# The caller MUST then, before finalize:
#   1. seat the player team with POD_combat_duel_add_ally_effect - the FIRST call
#      lands at player/0 and becomes the player-team leader, which several
#      finalize steps require (one of them fails silently without it);
#   2. mark one seat with pod_combat_mark_slot_controlled_effect, passing the
#      position that seat ALREADY holds (add_ally bumps the counter after it
#      seats, so re-reading the counter is off by one) and the DIRECTOR as
#      CONTROLLER;
#   3. seat the enemy team and call POD_combat_finalize_duel_effect.
#
# Name the lead proxy as ATTACKER or DEFENDER, not the director: outcome
# attribution records the player team's LEADER occupant, and the sc_ victor snap
# resolves against the principals. INITIATOR should be the director, so the
# follow-up event fires on the human.
POD_combat_create_proxy_duel_effect = {
	pod_combat_create_duel_core_effect = {
		SEAT_HOST = no
		INITIAL_PLAYER_SIZE = 0
		INITIATOR = $INITIATOR$
		ATTACKER  = $ATTACKER$
		DEFENDER  = $DEFENDER$
		FATALITY  = $FATALITY$
		FIXED     = $FIXED$
		LOCALE    = $LOCALE$
		OUTPUT_EVENT = $OUTPUT_EVENT$
	}
}

POD_combat_duel_add_ally_effect = {
	# $CHARACTER$ = next ally; appended at player/<pod_combat_player_team_size>,
	# counter bumped. Occupancy only; stat init runs at finalize.
	# Runs in the PC (host) scope, so the first ally lands at player/1.
	#
	# pod_combat_max_team_size is a runaway guardrail; an add past it is dropped.
	#
	# The outer gate keeps this build-safe: during the confirm-tooltip preview
	# create_duel's create_story hasn't run, so neither the session pointer nor
	# the counter exists, and a bare read in a comparison errors during
	# tooltip-build (has_variable doesn't short-circuit a sibling comparison, so
	# the read must live in a nested if). The ?= now covers both the missing
	# pointer and the missing counter; the nesting is kept because it is what
	# documents the tooltip-build path.
	if = {
		limit = { var:pod_combat_session ?= { has_variable = pod_combat_player_team_size } }
		if = {
			limit = { var:pod_combat_session ?= { var:pod_combat_player_team_size < pod_combat_max_team_size } }
			# POSITION resolves in host scope (set_slot_char captures it before its
			# new-object block), so the chained read is the right shape here.
			pod_combat_set_slot_char_effect = { TEAM = player POSITION = var:pod_combat_session.var:pod_combat_player_team_size CHARACTER = $CHARACTER$ }
			var:pod_combat_session ?= { change_variable = { name = pod_combat_player_team_size add = 1 } }
		}
	}
}

POD_combat_duel_add_enemy_effect = {
	# $CHARACTER$ = next enemy; appended at enemy/<pod_combat_enemy_team_size>,
	# counter bumped. The FIRST enemy lands at enemy/0 = primary opponent /
	# enemy leader (player-team hit-chance baseline + PC default target).
	#
	# Occupancy only; finalize runs the stat init. The outer has_variable gate
	# keeps the cap comparison build-safe (see add_ally for the rationale).
	if = {
		limit = { var:pod_combat_session ?= { has_variable = pod_combat_enemy_team_size } }
		if = {
			limit = { var:pod_combat_session ?= { var:pod_combat_enemy_team_size < pod_combat_max_team_size } }
			pod_combat_set_slot_char_effect = { TEAM = enemy POSITION = var:pod_combat_session.var:pod_combat_enemy_team_size CHARACTER = $CHARACTER$ }
			var:pod_combat_session ?= { change_variable = { name = pod_combat_enemy_team_size add = 1 } }
		}
	}
}


##############################################################################################
# SECTION 6: SLOT-INIT DISPATCHER + DUEL FINALIZE
#
# pod_combat_init_all_slots_effect runs the per-slot stat / discipline init
# for every occupied slot by iterating pod_combat_active_slots.
#
# POD_combat_finalize_duel_effect completes a duel built via
# create_duel + add_ally/add_enemy: derives the leader scopes, shapeshifts
# AI fera, inits every slot, computes hit chances, runs the init-time KO /
# victory check, opens the combat window (pod_combat_active = yes), and
# starts round 1.
##############################################################################################

pod_combat_init_all_slots_effect = {
	# Init every combatant by iterating pod_combat_active_slots. finalize must have
	# saved scope:pod_combat_target (enemy leader) and scope:pod_combat_player_leader,
	# passed as the discipline-buff CASTER: player-team slots vs the enemy leader,
	# enemy-team slots vs the player leader. Both are DERIVED from the leader slots,
	# not assumed to be the host - a proxy launcher need not seat it at player/0.
	#
	# init_slot_from_character runs in each slot's own scope (the iteration
	# body) and writes the slot's vars directly. CHARACTER must still be a
	# SAVED SCOPE, not var:slot_char: init does `$CHARACTER$ = { ... CASTER =
	# $CHARACTER$ }`, which switches into the character's scope and re-evaluates
	# $CHARACTER$ there - a var:slot_char chain would re-resolve against the
	# occupant (no such var) and throw, while a saved scope resolves globally.
	# So snapshot the occupant to scope:init_slot_char first (overwritten each
	# iteration) before delegating.
	var:pod_combat_session ?= {
		every_in_list = {
			variable = pod_combat_active_slots
			var:slot_char = { save_scope_as = init_slot_char }
			if = {
				limit = { pod_combat_slot_is_player_team_trigger = yes }
				pod_combat_init_slot_from_character_effect = { CHARACTER = scope:init_slot_char ENEMY_LEADER = scope:pod_combat_target }
			}
			else = {
				pod_combat_init_slot_from_character_effect = { CHARACTER = scope:init_slot_char ENEMY_LEADER = scope:pod_combat_player_leader }
			}
		}
	}
}

POD_combat_finalize_duel_effect = {
	# Completes a duel built via create_duel + add_ally/add_enemy, in the PC
	# (host) scope. The window opens (pod_combat_active) only after this whole
	# effect returns (one atomic tick), so the GUI never sees a half-init roster.

	save_scope_as = combat_pc

	# Derive the two leader pointers (each team's position-0 slot). Every other
	# slot is reached by iterating the active list; exactly one match per team.
	#
	# Symmetric between the teams, and deliberately SEPARATE from the control
	# marker below: a leader is a board position, a controller is not. They
	# coincide today only because the builder seats the host at player/0.
	var:pod_combat_session ?= {
		every_in_list = {
			variable = pod_combat_active_slots
			limit = { pod_combat_slot_is_player_team_trigger = yes pod_combat_slot_is_leader_trigger = yes }
			save_scope_as = pc_leader_story
			scope:combat_pc.var:pod_combat_session ?= { set_variable = { name = pod_combat_pc_slot value = scope:pc_leader_story } }
		}
	}
	var:pod_combat_session ?= {
		every_in_list = {
			variable = pod_combat_active_slots
			limit = { pod_combat_slot_is_enemy_trigger = yes pod_combat_slot_is_leader_trigger = yes }
			save_scope_as = enemy_leader_story
			scope:combat_pc.var:pod_combat_session ?= { set_variable = { name = pod_combat_enemy_leader_slot value = scope:enemy_leader_story } }
		}
	}

	# Seed the display pointer at the first marked slot in roster order. The MARK
	# itself is no longer set here - the builder states it at seat time via
	# pod_combat_mark_slot_controlled_effect - so this pass only has to answer
	# "which played slot does the window show before anyone has had a turn".
	#
	# Persistent display pointer: the slot this human drives. Re-pointed at every
	# hand-off, never cleared mid-fight - the target readout, the TARGETED badges
	# and PODCombatMyCombatant (the move bar's loadout) all read through it and
	# must survive AI turns. Distinct from pod_combat_control_turn_live, which is
	# the per-turn latch; conflating them would blank the readout on every AI turn.
	#
	# First-match by the NOT-exists idiom, as the retarget scans do. Roster order
	# is arbitrary with several marked slots, which is correct for the one thing
	# this seeds: a starting view, replaced at the first hand-off.
	#
	# This one is FIGHT-scoped and therefore a singleton: it names the slot whose
	# turn the engine is currently servicing, which is a property of the fight, not
	# of any viewer. Its per-viewer twin is pod_combat_my_control_slot, seeded in
	# the pass below - see that pass for which readers use which.
	var:pod_combat_session ?= {
		every_in_list = {
			variable = pod_combat_active_slots
			limit = {
				has_variable = slot_human_controller
				NOT = { exists = scope:pod_seed_control_slot }
			}
			save_scope_as = pod_seed_control_slot
		}
	}
	if = {
		limit = { exists = scope:pod_seed_control_slot }
		var:pod_combat_session ?= { set_variable = { name = pod_combat_active_control_slot value = scope:pod_seed_control_slot } }
	}

	# Per-MARKED-SLOT seeding: the driver's own display pointer, and that slot's
	# standing target. Both are per-slot facts, so this pass has NO first-match
	# guard - every marked slot seeds its own, which is what makes two drivers (or
	# one driver with two bodies) independent.
	#
	# pod_combat_my_control_slot lives on the CONTROLLING CHARACTER, not on the
	# fight. That is the whole point: a fight var can only ever name one slot, so
	# two distinct humans sharing a fight would each see whichever of them was
	# handed off most recently. Every per-viewer surface (the move bar through
	# PODCombatMyCombatant, the target readout, the TARGETED badges, the hit
	# breakdown, self-target suppression) reads this one; the fight var above stays
	# for the turn machinery, which must follow the LIVE slot regardless of viewer.
	#
	# Unlike a story var, a character var does not die with end_story - see the
	# release blocks in pod_combat_duel_end_effect and the host-death path.
	var:pod_combat_session ?= {
		every_in_list = {
			variable = pod_combat_active_slots
			limit = { has_variable = slot_human_controller }
			save_scope_as = pod_marked_slot
			# Standing target defaults to enemy/0; the player can switch via the
			# Target button on any card. slot_target_char is the mirror the top-row
			# 'Target: X' readout reads, rewritten wherever the target changes.
			set_variable = { name = slot_target_team     value = flag:enemy }
			set_variable = { name = slot_target_position value = 0 }
			if = {
				limit = {
					scope:combat_pc.var:pod_combat_session ?= { has_variable = pod_combat_enemy_leader_slot }
					scope:combat_pc.var:pod_combat_session.var:pod_combat_enemy_leader_slot ?= { has_variable = slot_char }
				}
				set_variable = { name = slot_target_char value = scope:combat_pc.var:pod_combat_session.var:pod_combat_enemy_leader_slot.var:slot_char }
			}
			# Scope switch is the body's LAST statement, keeping the iteration scope
			# clean. ?= because the marker is a character ref that a mid-fight unmark
			# could clear.
			var:slot_human_controller ?= { set_variable = { name = pod_combat_my_control_slot value = scope:pod_marked_slot } }
		}
	}

	# scope:combat_pc = the PC (root). scope:pod_combat_target = the enemy
	# leader's occupant (enemy/0, the primary opponent) - the player team's
	# discipline-init CASTER and the post-combat dispatch target.
	# scope:pod_combat_player_leader is its twin, the ENEMY-team slots' CASTER.
	# Both are derived from the leader pointers rather than assumed: the player
	# leader is scope:combat_pc only while the builder seats the host at
	# player/0, which a proxy launcher would not.
	if = {
		limit = {
			var:pod_combat_session ?= { has_variable = pod_combat_pc_slot }
			var:pod_combat_session.var:pod_combat_pc_slot ?= { has_variable = slot_char }
		}
		var:pod_combat_session.var:pod_combat_pc_slot.var:slot_char = { save_scope_as = pod_combat_player_leader }
	}
	if = {
		limit = {
			var:pod_combat_session ?= { has_variable = pod_combat_enemy_leader_slot }
			var:pod_combat_session.var:pod_combat_enemy_leader_slot ?= { has_variable = slot_char }
		}
		var:pod_combat_session.var:pod_combat_enemy_leader_slot.var:slot_char = { save_scope_as = pod_combat_target }
	}

	var:pod_combat_session ?= { set_variable = { name = pod_combat_active value = yes } }

	scope:combat_pc = {
		if = {
			limit = {
				has_perk = honour_bound_perk
				POD_has_coterie_members_trigger = yes
				var:coterie ?= {
					any_in_list = {
						variable = members
						NOT = { this = scope:combat_pc }
						is_imprisoned = no
						in_diplomatic_range = scope:combat_pc
					}
				}
			}
			var:coterie ?= {
				random_in_list = {
					variable = members
					limit = {
						NOT = { this = scope:combat_pc }
						is_imprisoned = no
						in_diplomatic_range = scope:combat_pc
					}
					save_scope_as = honourbound_ally
				}
			}
			POD_combat_duel_add_ally_effect = { CHARACTER = scope:honourbound_ally }
		}
	}
	
	scope:pod_combat_target = {
		if = {
			limit = {
				has_perk = honour_bound_perk
				POD_has_coterie_members_trigger = yes
				var:coterie ?= {
					any_in_list = {
						variable = members
						NOT = { this = scope:pod_combat_target }
						is_imprisoned = no
						in_diplomatic_range = scope:pod_combat_target
					}
				}
			}
			var:coterie ?= {
				random_in_list = {
					variable = members
					limit = {
						NOT = { this = scope:pod_combat_target }
						is_imprisoned = no
						in_diplomatic_range = scope:pod_combat_target
					}
					save_scope_as = honourbound_ally
				}
			}
			POD_combat_duel_add_enemy_effect = { CHARACTER = scope:honourbound_ally }
		}
	}

	#Snapshot of starting enemy leader prowess for Ahroun story
	scope:combat_pc = {
		if = {
			limit = { has_trait = ahroun has_trait_xp = { trait = ahroun value < 100 } }
			set_variable = { name = ahroun_target_prowess value = scope:pod_combat_target.prowess }
		}
	}

	# Seed the fight's step_mode from the toggling player's persistent preference
	# (pod_combat_manual_step_pref, present = manual; absent = auto, the default),
	# alongside pod_combat_active so the window's first draw can read it. The pref
	# stays on the character - only the receiver moved - so the limit reads the
	# host and only the write goes to the fight. Deterministic on both branches:
	# the explicit remove on the auto path means a stale step_mode can never leak.
	if = {
		limit = { has_variable = pod_combat_manual_step_pref }
		var:pod_combat_session ?= { set_variable = { name = pod_combat_step_mode value = yes } }
	}
	else = {
		var:pod_combat_session ?= { remove_variable = pod_combat_step_mode }
	}

	# -- Pre-pass: shapeshift every AI fera combatant BEFORE slot init --
	# The flag is the exit half's authority (pod_combat_revert_fight_granted_warform_effect):
	# teardown reverts exactly the characters this pass shifted. Cleared at the seat
	# point, not here, so mid-fight seats are covered too.
	#
	# Gating the revert on PoD's own POD_maintenance.3 / .15 rule instead would miss
	# the common case: an at-war AI that was not yet shifted gets shifted here, and
	# `is_at_war = no` would then refuse to revert it for the rest of the war.
	#
	# The already-in-warform check is what keeps this from re-shifting (and, via the
	# revert_all inside the warform effect, churning) a form that is already up. It
	# does NOT preserve non-warform forms: POD_shapeshift_fera_warform_effect opens
	# with POD_shapeshift_revert_all_effect on any AI, so a gift, totem or animal
	# form the character wore into the fight is wiped on entry and is not restored
	# on exit. That entry behaviour is PoD's, and predates the exit half.
	var:pod_combat_session ?= {
		every_in_list = {
			variable = pod_combat_active_slots
			limit = {
				var:slot_char = {
					is_ai = yes
					has_trait = fera
					POD_can_shapeshift_trigger = yes
					NOT = { POD_is_shapeshifted_warform_trigger = yes }
				}
			}
			var:slot_char = {
				POD_shapeshift_fera_warform_effect = yes
				add_character_flag = pod_combat_fight_granted_warform
			}
		}
	}

	# -- Every duel: gate PoD's inline death check --
	# Stamp pod_combat_nonfatal_active on every combatant, whatever the fatality mode.
	# POD_can_die_from_wounds (delta in POD_disciplines_triggers.txt) reads it
	# and returns no, so the death check never reaches pod_death mid-fight - which is
	# what stops a guaranteed-resurrection opponent from full-healing on every killing
	# blow. Stripped post-combat by pod_combat_finalize_combat_health_effect, which then
	# lets pod_combat_resolve_suppressed_deaths_effect settle the fatal modes' deaths in
	# one pass at the writeback.
	#
	# MUST precede init_all_slots: slot init opens with PoD's discipline battery
	# (POD_enemy_knight_set_discipline_battle_effect), which applies real wounds off
	# caster-perk branches that carry no victim filter - enough to kill a low-track
	# mortal before round 1. Gating after that pass let a fatality = no duel end at
	# setup. Both gates only need pod_combat_active_slots, populated by add_ally /
	# add_enemy, which is why init_all_slots can iterate it here too.
	if = {
		limit = { pod_combat_uses_death_gate_trigger = yes }
		pod_combat_apply_nonfatal_flag_effect = yes
	}
	# Practice spar: snapshot every combatant's wound track now so it can be
	# restored at combat end (pod_combat_finalize_combat_health_effect practice branch).
	# Also ahead of init for the same reason: the snapshot is the restore baseline,
	# so taking it after the discipline battery would bake that damage in permanently,
	# and pod_combat_practice_active must exist before the wounds land or PoD's
	# practice guards (balefire / infection / treatment / Clay-Man) fail open.
	if = {
		limit = { pod_combat_is_practice_trigger = yes }
		pod_combat_snapshot_practice_wounds_effect = yes
	}

	# -- Per-slot stat / discipline init for every occupied slot --
	# Deferred from add_* (see Section 5 header for why).
	pod_combat_init_all_slots_effect = yes

	# Leader-baseline hit chances (formula in the effect). Pair-specific recalc
	# against the selected target happens later via pod_combat_recalc_pc_hit_chance_effect.
	pod_combat_calculate_all_hit_chances_effect = yes

	# -- Set up initial target / workspace defaults --
	# Seed pod_combat_target_char so the damage helper's
	# var:pod_combat_$TARGET_SIDE$_char lookup resolves before the first target load.
	if = {
		limit = { var:pod_combat_session ?= { has_variable = pod_combat_enemy_leader_slot } var:pod_combat_session.var:pod_combat_enemy_leader_slot ?= { has_variable = slot_char } }
		var:pod_combat_session ?= { set_variable = { name = pod_combat_target_char value = var:pod_combat_enemy_leader_slot.var:slot_char } }
	}
	var:pod_combat_session ?= { set_variable = { name = pod_combat_selected_target_team     value = flag:enemy } }
	var:pod_combat_session ?= { set_variable = { name = pod_combat_selected_target_position value = 0 } }
	# (The per-slot standing target is seeded in the marked-slot pass above, so that
	# every controlled slot gets one rather than only the display pointer's.)
	# AP / drain setup happens in the PC-goes-first branch below (AI turns do
	# their own). apply_ap_drain reads the workspace, so it must follow load_workspace.

	# Seed the last-damage result transient read by the first log push.
	var:pod_combat_session ?= { set_variable = { name = pod_combat_last_damage value = 0 } }
	# AI anti-repetition history lives on the AI char, not the host, so no
	# PC-side init here (cleanup: slot init + group cleanup).

	# System log: combat begins. Must precede the KO/victory check below, not just
	# the turn-order block: the log list is appended oldest-first and rendered
	# bottomtotop, so pushing after the victory row would stack "combat begins"
	# ABOVE it. Unconditional on purpose - a fight decided by the init-time KO
	# check still gets a start row, so a one-sided open reads as a real (if
	# instant) fight instead of a bare "victory" with no context.
	pod_combat_push_system_log_effect = { MESSAGE_KEY = combat_started }

	# -- Init-time KO check --
	# init sets slot_alive = 1 unconditionally, but HP / morale / state read
	# live, so a combatant who walks in already KO'd (filled track, torpor,
	# morale-broken, scripted incap) would be marked alive. Apply the shared KO
	# predicate, then victory-check so a one-sided start resolves immediately.
	# Deliberately a bare flip, NOT pod_combat_mark_slot_ko_effect (pre-fight
	# roster correction; rationale in the section 1b header).
	var:pod_combat_session ?= {
		every_in_list = {
			variable = pod_combat_active_slots
			limit = {
				pod_combat_slot_should_be_marked_ko_trigger = yes
			}
			set_variable = { name = slot_alive value = 0 }
		}
	}
	pod_combat_check_group_victory_effect = yes

	# -- Determine round 1 turn order --
	# Unified flow: every combat (1v1 = degenerate NvN) rolls initiative and
	# processes pre-PC turns so nobody gets a free first move. Skipped if the
	# init-time KO check already ended combat.
	if = {
		limit = { NOT = { var:pod_combat_session ?= { has_variable = pod_combat_is_over } } }
	# -- Seed AI loadouts upfront (perf) --
	# Primary seed site (mid-fight seats seed in
	# pod_combat_init_one_midfight_slot_effect; pod_combat_ai_ensure_loadout_effect
	# is the empty-list backstop). Must run after slot init (which clears AI
	# loadouts) and the warform pre-pass (so unlocks read the fighting form);
	# inside the is_over guard so an init-time KO finish skips it. Moves the
	# per-AI catalogue scan out of the first End Turn click and makes loadout
	# membership the unlock proof the AI selector's dynamic-only validator
	# (pod_combat_move_is_usable_now_trigger) relies on.
	var:pod_combat_session ?= {
		every_in_list = {
			variable = pod_combat_active_slots
			limit = {
				pod_combat_slot_is_alive_trigger = yes
				var:slot_char = {
					is_ai = yes
					pod_combat_equipped_count = 0
				}
			}
			var:slot_char = { POD_apply_ai_full_loadout = yes }
		}
	}

	# Bake the PC's loadout-move unlocks for the cheap per-frame combat-bar check
	# (POD_can_execute_combat_move_trigger tests pod_combat_unlocked_moves membership instead of re-running the unlock
	# dispatcher every GUI frame). After the loadout seed + warform pre-pass so it
	# reads the fighting form.
	# Bake for EVERY human-driven combatant, not just the host. The move bar reads
	# the driven combatant's loadout and POD_can_execute_combat_move_trigger proves
	# against that same character's pod_combat_unlocked_moves - and that proof
	# fails CLOSED, so an unbaked controlled slot renders a full move bar with
	# every button dead. Identical to the old single call while one slot is marked.
	#
	# The scope switch is the iteration body's last statement, which is what keeps
	# an inline switch from corrupting the walk.
	var:pod_combat_session ?= {
		every_in_list = {
			variable = pod_combat_active_slots
			limit = { has_variable = slot_human_controller }
			var:slot_char ?= { pod_combat_bake_pc_loadout_unlocks_effect = yes }
		}
	}

	pod_combat_calculate_initiative_effect = yes
	pod_combat_find_next_actor_effect = yes

	if = {
		limit = { pod_combat_current_actor_is_controlled_trigger = yes }
		# A controlled slot goes first in initiative order.
		pod_combat_setup_pc_turn_effect = yes
	}
	else = {
		# an AI goes first on round 1.
		# STEP mode resolves NOTHING here, unlike the round-transition branch in
		# pod_combat_group_end_round_effect. The combat window opens only AFTER
		# this effect returns, so executing the opening AI turn now would resolve
		# it invisibly. The preceding find_next already primed current_actor on the
		# first AI, so the window opens with the Advance button live and the
		# player's first Advance click resolves that opening turn.
		# AUTO mode walks the pre-PC AI turns now; run_pre_pc primes its own
		# find_next, so no pre-call here (idempotent re-select).
		if = {
			limit = { NOT = { var:pod_combat_session ?= { has_variable = pod_combat_step_mode } } }
			# Hidden for the tooltip builder, not the game: the drain's
			# hang-safe `while count` (pod_combat_ai_outer_turn_cap_value = 200)
			# makes CK3 render the loop body once per allowed iteration when it
			# describes whatever launched the duel. Callers that reach finalize
			# outside the single-combat launcher (the debug interactions) rely
			# on this guard.
			hidden_effect = { pod_combat_run_pre_pc_ai_turns_effect = yes }
		}
	}
	}  # end of NOT pod_combat_is_over guard (init-time KO check)
}

# Per-slot helper: compute one slot's base hit chance from prowess and
# apply it. Runs in the SLOT-STORY scope (called once per slot from the
# pod_combat_calculate_all_hit_chances_effect loop).
pod_combat_calc_one_hit_chance_effect = {
	# $OPP_PROWESS$ = the opposing team leader's prowess (a value the caller
	#   captured in host scope). Base hit = pod_combat_hit_chance_base +
	#   (own prowess - opposing leader prowess) / 2, clamped to the shared
	#   pod_combat_hit_chance_min/max window, then the live hit_chance is
	#   rebuilt from base + active tiers.
	if = {
		limit = { has_variable = slot_char exists = var:slot_char }
		var:slot_char = {
			save_scope_value_as = { name = own_prowess value = prowess }
		}
		save_scope_value_as = {
			name = base_hit
			value = {
				value = pod_combat_hit_chance_base
				add = {
					value = scope:own_prowess
					subtract = $OPP_PROWESS$
					divide = 2
				}
				min = pod_combat_hit_chance_min
				max = pod_combat_hit_chance_max
			}
		}
		pod_combat_set_slot_var_effect = { NAME = base_hit_chance VALUE = scope:base_hit }
		pod_combat_recalc_slot_hit_effect = yes
	}
}

pod_combat_calculate_all_hit_chances_effect = {
	# Compute every slot's base hit chance by iterating pod_combat_active_slots.
	# base = pod_combat_hit_chance_base + (own prowess - opposing leader
	# prowess) / 2, clamped to the pod_combat_hit_chance_min/max window.
	# Player-team slots measure against the enemy
	# leader; enemy-team slots against the PC. recalc_slot_hit then rebuilds
	# live hit_chance from base + active tiers (so tier debuffs survive
	# the round-end reset).
	#
	# Capture each leader's live prowess in host scope first so the per-slot
	# calc reads a saved value, not the host pointer chain from slot scope.
	#
	# Guarded on both leaders and their characters existing: during the tooltip
	# preview no slot stories exist, so the chained reads would warn.
	if = {
		limit = {
			var:pod_combat_session ?= { has_variable = pod_combat_enemy_leader_slot }
			var:pod_combat_session ?= { has_variable = pod_combat_pc_slot }
			var:pod_combat_session.var:pod_combat_enemy_leader_slot ?= { has_variable = slot_char }
			var:pod_combat_session.var:pod_combat_pc_slot ?= { has_variable = slot_char }
		}
		var:pod_combat_session.var:pod_combat_enemy_leader_slot.var:slot_char ?= {
			save_scope_value_as = { name = enemy_leader_prowess value = prowess }
		}
		var:pod_combat_session.var:pod_combat_pc_slot.var:slot_char ?= {
			save_scope_value_as = { name = pc_prowess value = prowess }
		}
		var:pod_combat_session ?= {
			every_in_list = {
				variable = pod_combat_active_slots
				if = {
					limit = { pod_combat_slot_is_player_team_trigger = yes }
					pod_combat_calc_one_hit_chance_effect = { OPP_PROWESS = scope:enemy_leader_prowess }
				}
				else = {
					pod_combat_calc_one_hit_chance_effect = { OPP_PROWESS = scope:pc_prowess }
				}
			}
		}
	}
}

##############################################################################################
# SECTION 6b: PLAYER HIT CHANCE RECALC ON TARGET SWAP
# Rebuild the PC's hit chance against a newly selected target. See
# pod_combat_recalc_pc_hit_chance_effect below.
##############################################################################################

pod_combat_recalc_pc_hit_chance_effect = {
	# Rebuild the acting slot's base_hit_chance from the loaded target's prowess,
	# then full tier recalc so active hit buffs/debuffs reapply on the new
	# base. Called on target swap (pod_combat_select_target) and from the
	# tier-recalc path. Human-controlled turns only (the `_pc_` name encodes
	# that); the AI path recalculates per slot. Round-duration tiers and
	# stealth's hit bonus are handled elsewhere.
	if = {
		limit = {
			var:pod_combat_session ?= { has_variable = pod_combat_actor_prowess }
			var:pod_combat_session ?= { has_variable = pod_combat_target_prowess }
		}
		save_scope_value_as = {
			name = pc_recalc_base_hit
			value = {
				value = pod_combat_hit_chance_base
				add = {
					value = var:pod_combat_session.var:pod_combat_actor_prowess
					subtract = var:pod_combat_session.var:pod_combat_target_prowess
					divide = 2
				}
				min = pod_combat_hit_chance_min
				max = pod_combat_hit_chance_max
			}
		}
		var:pod_combat_session ?= { set_variable = { name = pod_combat_actor_base_hit_chance value = scope:pc_recalc_base_hit } }
		# Stash both sides' prowess (display-only host vars) so the PC hit-chance
		# breakdown tooltip can show the matchup without re-reading the transient
		# workspace; refreshed here on every target swap / PC turn setup, so they
		# report the same live values the base hit above was built from.
		var:pod_combat_session ?= { set_variable = { name = pod_combat_pc_hit_own_prowess    value = var:pod_combat_actor_prowess } }
		var:pod_combat_session ?= { set_variable = { name = pod_combat_pc_hit_target_prowess value = var:pod_combat_target_prowess } }
		var:pod_combat_session.var:pod_combat_active_control_slot ?= { pod_combat_set_slot_var_effect = { NAME = base_hit_chance VALUE = scope:pc_recalc_base_hit } }
		pod_combat_recalc_hit_actor_effect = yes
		var:pod_combat_session.var:pod_combat_active_control_slot ?= { pod_combat_recalc_slot_hit_effect = yes }
	}
}

##############################################################################################
# SECTION 7: GROUP CLEANUP
# Remove all per-slot variables (every slot in pod_combat_active_slots) and global group state.
# Called in addition to the existing workspace cleanup.
##############################################################################################

# Clears per-fight AI scratch on every living non-PC AI combatant so the NPC's
# next fight starts unbiased. Dead-safe: iterates the slot list and mutates only
# living AI slot_chars (a dead/removed participant would error). MUST run while
# the slot stories still resolve, and BEFORE any summon despawn (death_disappearance
# removes the char, so a later slot_char read hits a weak ref). Shared by the
# normal group cleanup and the dead-host release (pod_combat_host_died_release_effect).
pod_combat_clear_all_ai_scratch_effect = {
	var:pod_combat_session ?= {
		every_in_list = {
			variable = pod_combat_active_slots
			# Marker, not board position: the question is "did the AI drive this
			# slot", and a human-driven clone reads is_ai = yes while never having
			# run the selector. Identical at arity 1 - player/0 is the only marked
			# slot and the only slot the positional predicate matched.
			limit = {
				NOT = { has_variable = slot_human_controller }
				var:slot_char = { is_ai = yes  is_alive = yes }
			}
			var:slot_char = { pod_combat_clear_ai_scratch_effect = yes }
		}
	}
	# Third pass: the two LISTS a human-driven AI body accumulated. The loops
	# above deliberately skip marked slots, which is right for the AI memory they
	# never wrote - but the loadout the launcher curated and the unlock list the
	# bake built are per-fight scaffolding on an NPC, and nothing else clears
	# either, so without this they ride the save forever. Only lists, never the
	# AI history vars.
	var:pod_combat_session ?= {
		every_in_list = {
			variable = pod_combat_active_slots
			limit = {
				has_variable = slot_human_controller
				var:slot_char = { is_ai = yes  is_alive = yes }
			}
			var:slot_char = {
				clear_variable_list = pod_combat_loadout
				clear_variable_list = pod_combat_unlocked_moves
			}
		}
	}
	# The move-story marks are NOT cleared here. They cannot be: this effect runs
	# from cleanup, which is downstream of the writeback's deferred kills, and a
	# garbage-collected loser is a weak reference that has_variable_list cannot be
	# pointed at. They are cleared one step earlier instead, from
	# pod_combat_duel_end_effect, where nothing has died yet - see
	# pod_combat_clear_ai_move_marks_effect.
}

# Strips the scheduling vars off pending (never-seated) reinforcements. Dead-safe:
# per-char mutation on living reinforcements only. The roster itself is the
# fight's and dies with it, so there is no list clear to split out here any more.
pod_combat_strip_pending_reinforcement_vars_effect = {
	var:pod_combat_session ?= {
		if = {
			limit = { has_variable_list = pod_combat_pending_reinforcements }
			every_in_list = {
				variable = pod_combat_pending_reinforcements
				limit = { is_alive = yes }
				remove_variable = pod_combat_join_round
				remove_variable = pod_combat_join_team
			}
		}
	}
}

pod_combat_group_cleanup_effect = {
	# ── Unregister reinforcements that never entered ─────────────
	# Scheduled-but-not-yet-seated combatants (combat ended before their join round).
	# Summoned combatants are despawned separately below; here just strip the
	# scheduling vars off the characters. The roster list needs no clear: it lives
	# on the fight story and goes when that story ends.
	pod_combat_strip_pending_reinforcement_vars_effect = yes

	# ── AI per-fight transient state cleanup ─────────────────────
	# Shared dead-safe sub-effect (also called by the dead-host release path).
	# Runs before the slot teardown below so slot_char still resolves.
	pod_combat_clear_all_ai_scratch_effect = yes

	# ── Despawn summoned temporary combatants ────────────────────
	# Mid-combat summons (e.g. the necromancer's raised wraith) are temporary.
	# Kill any still alive (is_alive guard avoids double-removal in a fatal duel);
	# their slot stories are torn down just below.
	# MUST run AFTER the AI-scratch clear above: death_disappearance REMOVES the
	# created character, so a clear loop reading var:slot_char on the summon's slot
	# would otherwise hit a removed (weak) reference and error
	# ("scope doesn't support variables"). With this order the clear runs while the
	# summon is still alive; then it is despawned, then its slot story is torn down.
	var:pod_combat_session ?= {
		if = {
			limit = { has_variable_list = pod_combat_summoned_chars }
			every_in_list = {
				variable = pod_combat_summoned_chars
				limit = { is_alive = yes }
				death = { death_reason = death_disappearance }
			}
			# No clear: the registry dies with the fight story.
		}
	}

	# ── Workspace mirrors for the spirit capability flags ────────
	# Host-scope mirrors of the spirit capability flags (the slot-story copies
	# were torn down above).

	# Remaining host-side target/hit workspace. The team sizes, the current actor
	# and the find_next scan scratch are gone from here: they live on the fight
	# story and die with it.
}

# SECTION 8: INITIATIVE SYSTEM
##############################################################################################

# Roll each alive slot's initiative base, fold in pending slot_init_bonus,
# clear slot_has_acted, then stamp turn-order ranks.
pod_combat_calculate_initiative_effect = {
	# Initiative is STATIC by default: the base is rolled once per combatant and
	# holds for the fight. The reroll knob (global_var:pod_combat_tune_init_reroll)
	# restores per-round rolling.
	#   slot_initiative_base = base (prowess, or a flat tunable baseline when the
	#                          decouple-prowess toggle is set)
	#                        + integer_range[1, pod_combat_initiative_roll_max]
	#                        + initiative_modifier_value
	#   slot_initiative      = slot_initiative_base + init_bonus (carried from last
	#                          round, then cleared)
	# Base + roll cap are tunable knobs in pod_combat_values.txt.
	save_scope_as = host
	var:pod_combat_session ?= {
		every_in_list = {
			variable = pod_combat_active_slots
			limit = { pod_combat_slot_is_alive_trigger = yes }
			# A fresh mid-fight seat always rolls here, whatever the knob says.
			if = {
				limit = {
					OR = {
						exists = global_var:pod_combat_tune_init_reroll
						NOT = { has_variable = slot_initiative_base }
					}
				}
				set_variable = {
					name = slot_initiative_base
					value = {
						value = pod_combat_initiative_base_value
						add = {
							integer_range = { min = 1 max = pod_combat_initiative_roll_max }
						}
						# Designer hook (pod_combat_values.txt): per-character initiative
						# modifier, 0 unless filled in. Evaluated on this slot (var:slot_char).
						add = pod_combat_initiative_modifier_value
					}
				}
			}
			# Recomposed every round, so a folded init_bonus stays a ONE-round shift even
			# when the base underneath it never moves. Also keeps slot_initiative set on
			# every alive slot: find_next_actor reads it without a has_variable guard.
			set_variable = { name = slot_initiative value = var:slot_initiative_base }
			if = {
				limit = { has_variable = slot_init_bonus }
				change_variable = { name = slot_initiative add = var:slot_init_bonus }
				remove_variable = slot_init_bonus
			}
			set_variable = { name = slot_has_acted value = 0 }
			# once_per = round resets each round; the combat list persists (dies with the slot).
			clear_variable_list = slot_moves_used_round
		}
	}

	# Stamp turn-order rank as a stored slot var so the stats tooltip can show
	# 'Turn N' directly: a per-slot SV can't reach the host's other-slot pointers,
	# and GUI bindings don't support dynamic SV names.
	pod_combat_calc_turn_orders_effect = yes
	pod_combat_build_turn_order_list_effect = yes
}

# Compute slot_turn_order on every alive slot. Rank 1 = highest initiative,
# ties broken by lower team ordinal (player before enemy, lower position first).
# Nested every_in_list: outer picks a slot, inner counts how many others beat
# it. scope:host lets the inner loop re-iterate the list; scope:me writes back.
pod_combat_calc_turn_orders_effect = {
	save_scope_as = host
	var:pod_combat_session ?= {
		every_in_list = {
			variable = pod_combat_active_slots
			limit = {
				has_variable = slot_initiative
				pod_combat_slot_is_alive_trigger = yes
			}
			save_scope_as = me
			save_scope_value_as = { name = my_init       value = var:slot_initiative }
			# Tiebreak key is (is_enemy asc, then team_position asc) - player team
			# before enemy team, lower positions first. A direct two-field compare
			# (not a single ordinal) keeps the ordering correct for unbounded teams.
			save_scope_value_as = { name = my_is_enemy   value = pod_combat_slot_is_enemy_value }
			save_scope_value_as = { name = my_position   value = var:team_position }
			# Start at rank 1; bump for each other slot that ranks above us.
			set_variable = { name = slot_turn_order value = 1 }
			# Re-enter host scope for the inner iteration over the same list.
			scope:host.var:pod_combat_session ?= {
				every_in_list = {
					variable = pod_combat_active_slots
					limit = {
						has_variable = slot_initiative
						pod_combat_slot_is_alive_trigger = yes
						# Exclude self.
						NOT = { this = scope:me }
						# Strictly higher initiative, OR equal initiative and earlier on the
						# (is_enemy, team_position) tiebreak. Strict comparisons so two distinct
						# slots never both count each other.
						OR = {
							var:slot_initiative > scope:my_init
							AND = {
								var:slot_initiative = scope:my_init
								OR = {
									pod_combat_slot_is_enemy_value < scope:my_is_enemy
									AND = {
										pod_combat_slot_is_enemy_value = scope:my_is_enemy
										var:team_position < scope:my_position
									}
								}
							}
						}
					}
					# Bump the OUTER slot's rank - write back via scope:me.
					scope:me = {
						change_variable = { name = slot_turn_order add = 1 }
					}
				}
			}
		}
	}
}

# GUI-facing turn-order list for the initiative track: same membership as the
# alive ranked slots, ordered by slot_turn_order (GUI datamodels render list
# order; they cannot sort). Rebuilt every round right after the ranks are
# stamped. One pass per rank; ranks are contiguous 1..N over alive slots, so
# the while exits when the cursor passes the last rank. The alive filter also
# screens out STALE slot_turn_order values on slots KO'd in earlier rounds
# (the stamp pass only writes alive slots, it never clears dead ones).
pod_combat_build_turn_order_list_effect = {
	# The outer has_variable gate keeps this build-safe: during a duel option's
	# confirm-tooltip preview the fight story has not been created (tooltip-build
	# skips create_story), so the session pointer is unset and both the deref and
	# every scope:pod_fight use below would error. At real execution it is always
	# set. Same gate as POD_combat_duel_add_ally_effect uses on the team counter.
	if = {
		limit = { has_variable = pod_combat_session }
		var:pod_combat_session = { save_temporary_scope_as = pod_fight }
		scope:pod_fight = {
			clear_variable_list = pod_combat_turn_order_list
			set_variable = { name = pod_combat_rank_cursor value = 1 }
		}
		while = {
			limit = {
				scope:pod_fight = {
					any_in_list = {
						variable = pod_combat_active_slots
						pod_combat_slot_is_alive_trigger = yes
						has_variable = slot_turn_order
						var:slot_turn_order >= scope:pod_fight.var:pod_combat_rank_cursor
					}
				}
			}
			scope:pod_fight = {
				every_in_list = {
					variable = pod_combat_active_slots
					limit = {
						pod_combat_slot_is_alive_trigger = yes
						has_variable = slot_turn_order
						var:slot_turn_order = scope:pod_fight.var:pod_combat_rank_cursor
					}
					# Append to the track. The inline scope switch is the body's LAST
					# statement (iteration-scope rule) - and it is still needed even
					# though the loop already runs in the fight, because the body's
					# own scope is the slot story being ranked.
					save_scope_as = ranked_slot
					scope:pod_fight = {
						add_to_variable_list = { name = pod_combat_turn_order_list target = scope:ranked_slot }
					}
				}
			}
			scope:pod_fight = { change_variable = { name = pod_combat_rank_cursor add = 1 } }
		}
		scope:pod_fight = { remove_variable = pod_combat_rank_cursor }
	}
}


##############################################################################################
# SECTION 9: FIND NEXT ACTOR
# Walk pod_combat_active_slots (filled-only by construction) and pick
# the slot with the highest initiative that hasn't acted yet.
##############################################################################################

pod_combat_find_next_actor_effect = {
	# Find the alive combatant with highest unprocessed initiative. The seed sits
	# below every reachable initiative because slot_init_bonus can go NEGATIVE
	# (stacked initiative_debuff), and a -1 floor would skip any slot landing
	# <= -1 on the strict > compare, stealing turns from debuffed combatants.
	var:pod_combat_session ?= {
		set_variable = { name = pod_combat_best_init value = -999 }
		# Tiebreak ordinal of the running best (is_enemy asc, then team_position asc),
		# matching pod_combat_calc_turn_orders_effect so the initiative track's predicted
		# order is authoritative. Seeded above the max real is_enemy so the first
		# candidate always wins via the best_found gate, never via this compare.
		set_variable = { name = pod_combat_best_is_enemy value = 2 }
		# best_found presence = a candidate was picked; absence after the scan = none.
		remove_variable = pod_combat_best_found
	}

	# Iterate and pick the highest-initiative unacted slot. On an initiative tie, break
	# by (is_enemy asc, team_position asc) - the SAME key as the initiative track
	# (pod_combat_calc_turn_orders_effect), so a post-reinforcement player slot
	# appended past the enemy slots still acts in its shown order, not insertion order.
	var:pod_combat_session ?= {
		# The scan scratch lives on the fight, so the slot loop needs the fight itself
		# back as a scope inside the body (where `this` is the slot).
		save_temporary_scope_as = pod_fight
		every_in_list = {
			variable = pod_combat_active_slots
			limit = {
				pod_combat_slot_is_alive_trigger = yes
				pod_combat_slot_has_not_acted_trigger = yes
			}
			save_scope_value_as = { name = current_init value = var:slot_initiative }
			# Slot-local reads FIRST (iteration-scope rule): is_enemy via the same SV the
			# track uses, then the slot story (combat_team is a flag, uncapturable by
			# save_scope_value_as) for the fight block. save_scope_as is the LAST slot read.
			save_scope_value_as = { name = current_is_enemy value = pod_combat_slot_is_enemy_value }
			save_scope_as = candidate_slot
			scope:pod_fight = {
				# New best if strictly higher initiative, OR (once a best exists) an init
				# tie that ranks earlier on the (is_enemy, team_position) tiebreak. The
				# best_found gate lets the first candidate seed best_* with no compare.
				if = {
					limit = {
						OR = {
							scope:current_init > var:pod_combat_best_init
							AND = {
								has_variable = pod_combat_best_found
								scope:current_init = var:pod_combat_best_init
								OR = {
									scope:current_is_enemy < var:pod_combat_best_is_enemy
									AND = {
										scope:current_is_enemy = var:pod_combat_best_is_enemy
										scope:candidate_slot.var:team_position < var:pod_combat_best_position
									}
								}
							}
						}
					}
					set_variable = { name = pod_combat_best_init value = scope:current_init }
					set_variable = { name = pod_combat_best_is_enemy value = scope:current_is_enemy }
					set_variable = { name = pod_combat_best_team value = scope:candidate_slot.var:combat_team }
					set_variable = { name = pod_combat_best_position value = scope:candidate_slot.var:team_position }
					set_variable = { name = pod_combat_best_found value = yes }
				}
			}
		}
	}

	# Absent best_found = every alive slot has already acted, so the else clears
	# current_actor and the round is over. Scratch and result both live on the fight,
	# so this whole hand-off is one block - the value expressions resolve there too.
	var:pod_combat_session ?= {
		if = {
			limit = { has_variable = pod_combat_best_found }
			set_variable = { name = pod_combat_current_actor_team value = var:pod_combat_best_team }
			set_variable = { name = pod_combat_current_actor_position value = var:pod_combat_best_position }
		}
		else = {
			remove_variable = pod_combat_current_actor_team
			remove_variable = pod_combat_current_actor_position
		}

		# Cleared per scan, not just at teardown: the scan re-runs every turn and the
		# seed above does not reset best_team / best_position.
		remove_variable = pod_combat_best_init
		remove_variable = pod_combat_best_is_enemy
		remove_variable = pod_combat_best_found
		remove_variable = pod_combat_best_team
		remove_variable = pod_combat_best_position
	}
}

# resolve the acting AI char into scope:focus_actor so the selectors' focus-fire modifier can read
# its stored mark ONCE per call rather than per candidate. The workspace
# pod_combat_actor_char still points at the PRIOR actor here (load_workspace
# runs after target selection), so match on the current-actor pointers. Caller
# scope = host; the saved scope is read by the selector weight block.
pod_combat_ai_resolve_focus_actor_effect = {
	if = {
		limit = { var:pod_combat_session ?= { has_variable = pod_combat_current_actor_team } }
		var:pod_combat_session ?= {
			save_temporary_scope_as = pod_fight
			every_in_list = {
				variable = pod_combat_active_slots
				limit = {
					var:combat_team = scope:pod_fight.var:pod_combat_current_actor_team
					var:team_position = scope:pod_fight.var:pod_combat_current_actor_position
				}
				var:slot_char = { save_scope_as = focus_actor }
			}
		}
	}
}

# Write site, called once per AI turn from pod_combat_execute_ai_turn_effect (after load_workspace, so
# pod_combat_actor_char is the current actor). Records the turn's chosen target
# as the actor's persistent focus, or clears it on a controlled turn (a stolen
# turn inverts the target side, so the mark would point at the wrong team next
# turn). The focus TEAM flag is copied straight off the chosen slot's combat_team
# (the literal-flag source), one hop, mirroring the proven PC pc_target_team
# write - NOT chained through selected_target_team, to dodge flag-copy degrade.
pod_combat_ai_record_focus_target_effect = {
	save_scope_as = host
	if = {
		limit = { var:pod_combat_session ?= { has_variable = pod_combat_actor_char } }
		if = {
			limit = {
				NOT = { var:pod_combat_session ?= { has_variable = pod_combat_turn_is_controlled } }
				var:pod_combat_session ?= { has_variable = pod_combat_selected_target_team }
			}
			var:pod_combat_session ?= {
				every_in_list = {
					variable = pod_combat_active_slots
					limit = {
						var:combat_team = scope:host.var:pod_combat_session.var:pod_combat_selected_target_team
						var:team_position = scope:host.var:pod_combat_session.var:pod_combat_selected_target_position
					}
					save_scope_as = focus_slot
				}
			}
			if = {
				limit = { exists = scope:focus_slot }
				var:pod_combat_session.var:pod_combat_actor_char ?= {
					set_variable = { name = pod_combat_ai_focus_team     value = scope:focus_slot.var:combat_team }
					set_variable = { name = pod_combat_ai_focus_position value = scope:focus_slot.var:team_position }
					# Display-only char ref of the focus target so the slot card can name
					# it (PODCombatSlotFocusTarget) without resolving (team,position) at
					# render time. Kept in lockstep with the pair above.
					set_variable = { name = pod_combat_ai_focus_char     value = scope:focus_slot.var:slot_char }
				}
			}
		}
		else_if = {
			limit = { var:pod_combat_session ?= { has_variable = pod_combat_turn_is_controlled } }
			var:pod_combat_session.var:pod_combat_actor_char ?= {
				remove_variable = pod_combat_ai_focus_team
				remove_variable = pod_combat_ai_focus_position
				remove_variable = pod_combat_ai_focus_char
			}
		}
	}
}

# Folds ONE attackable slot into the host's finishable summary. this = the slot,
# $HOST$ = the combat host, and scope:fin_can_hp / scope:fin_can_mor carry the
# actor's per-kind capability (scopes the caller saved are visible in here).
#
# The caller's every_in_list has already applied alive + the side-aware attackable
# filter, so every slot arriving here is one this actor could legally hit; the only
# questions left are whether it is low enough on either track and whether the actor
# owns a move of that kind.
pod_combat_ai_fold_finishable_slot_effect = {
	# Re-seeded per slot: these are ordinary saved scopes, so they survive from the
	# previous iteration of the caller's walk.
	save_scope_value_as = { name = f_hit value = 0 }
	if = {
		limit = {
			scope:fin_can_hp = 1
			var:slot_char.POD_empty_damage_boxes <= pod_combat_ai_hp_finish_threshold
		}
		save_scope_value_as = { name = f_hit value = 1 }
		$HOST$.var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_enemy_finishable_hp value = 1 } }
	}
	if = {
		limit = {
			scope:fin_can_mor = 1
			var:slot_morale <= pod_combat_ai_morale_finish_threshold
		}
		save_scope_value_as = { name = f_hit value = 1 }
		$HOST$.var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_enemy_finishable_mor value = 1 } }
	}
	if = {
		limit = { scope:f_hit = 1 }
		# Position rides a scope value: inside the $HOST$ blocks below, a bare
		# var:team_position would resolve against the HOST, not this slot.
		save_scope_value_as = { name = f_pos value = var:team_position }
		$HOST$.var:pod_combat_session ?= { change_variable = { name = pod_combat_ai_fin_n add = 1 } }
		# Identity of the FIRST qualifying slot, for the trace only. team_n 0 means
		# nothing stamped yet, so this writes once per walk.
		if = {
			limit = { $HOST$.var:pod_combat_session ?= { var:pod_combat_ai_fin_team_n = 0 } }
			if = {
				limit = { var:combat_team = flag:player }
				$HOST$.var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_fin_team_n value = 1 } }
			}
			else = { $HOST$.var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_fin_team_n value = 2 } } }
			$HOST$.var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_fin_pos value = scope:f_pos } }
		}
		# Whether the slot the actor is already pointed at is the finishable one.
		# Tested on the slot itself rather than by comparing the selected target
		# against the stamp above, so it stays correct when several slots qualify.
		if = {
			limit = {
				$HOST$.var:pod_combat_session ?= { has_variable = pod_combat_selected_target_team }
				var:combat_team = $HOST$.var:pod_combat_session.var:pod_combat_selected_target_team
				var:team_position = $HOST$.var:pod_combat_session.var:pod_combat_selected_target_position
			}
			$HOST$.var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_cur_target_finishable value = 1 } }
		}
	}
}

# Per-ACTION precompute (called from pod_combat_ai_precompute_action_summary_effect,
# so its affordability reads see live AP). THE definition of "finishable" for the
# whole AI: an attackable slot at or under the HP or morale finish threshold
# (pod_combat_ai_hp_finish_threshold / pod_combat_ai_morale_finish_threshold) while
# the actor owns an AP-affordable move of that kind. That is a priority heuristic,
# not proof of a kill: it tests no energy, once-per availability, damage magnitude,
# soak, immunity, hit roll or resolved result. Consumers read the output flags
# rather than re-deriving them, with one exception: the kill-snap admit arm in
# pod_combat_ai_target_candidate_trigger mirrors the per-slot threshold tests
# below and has to move with them.
#
# Split BY KIND, because the two kinds do not overlap for most actors: most
# morale-damage moves deal no HP damage at all, so a morale specialist and a
# bruiser are looking at different enemies. Side-aware, so a controlled
# (dominated) actor evaluates the side it has been turned against.
#
# Outputs, all host vars and ALL written every call so readers need no guard:
#   pod_combat_ai_enemy_finishable       1 if either kind fired. The combined
#                                        flag, read by the intent chooser's
#                                        emergency gate, the finishable /
#                                        finish-focus adds on all four lottery
#                                        branches, and the critical-heal modifier.
#   pod_combat_ai_enemy_finishable_hp    the two halves.
#   pod_combat_ai_enemy_finishable_mor
#   pod_combat_ai_cur_target_finishable  1 if the currently selected target is
#                                        itself finishable.
#   pod_combat_ai_fin_n                  how many slots qualified (trace).
#   pod_combat_ai_fin_team_n             first qualifying slot: team as 1 = player /
#   pod_combat_ai_fin_pos                2 = enemy, plus its position. Trace only,
#                                        and NUMERIC - a loc string cannot safely
#                                        read an unset flag var, and a copied flag
#                                        degrades to a plain value anyway.
#
# The walk is one every_in_list whose body is a single effect CALL. Per-slot work
# has to switch scope to reach the character's wound track, and an INLINE switch
# inside every_in_list corrupts the iteration scope; a called effect does not.
pod_combat_ai_flag_finishable_enemy_effect = {
	save_scope_as = host
	# Capability once per call rather than once per slot. Deliberately affordability
	# only, NOT pod_combat_move_is_usable_now_trigger: widening it here would quietly
	# move the intent lottery, which reads the combined flag on every branch.
	save_scope_value_as = { name = fin_can_hp value = 0 }
	save_scope_value_as = { name = fin_can_mor value = 0 }
	if = {
		limit = {
			var:pod_combat_session.var:pod_combat_actor_char ?= {
				any_in_list = {
					variable = pod_combat_loadout
					has_variable = hp_damage
					trigger_if = {
						limit = { has_variable = ap_cost }
						scope:host.var:pod_combat_session.var:pod_combat_current_ap >= var:ap_cost
					}
				}
			}
		}
		save_scope_value_as = { name = fin_can_hp value = 1 }
	}
	if = {
		limit = {
			var:pod_combat_session.var:pod_combat_actor_char ?= {
				any_in_list = {
					variable = pod_combat_loadout
					has_variable = morale_damage
					trigger_if = {
						limit = { has_variable = ap_cost }
						scope:host.var:pod_combat_session.var:pod_combat_current_ap >= var:ap_cost
					}
				}
			}
		}
		save_scope_value_as = { name = fin_can_mor value = 1 }
	}
	var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_enemy_finishable_hp value = 0 } }
	var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_enemy_finishable_mor value = 0 } }
	var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_cur_target_finishable value = 0 } }
	var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_fin_n value = 0 } }
	var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_fin_team_n value = 0 } }
	var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_fin_pos value = 0 } }
	var:pod_combat_session ?= {
		every_in_list = {
			variable = pod_combat_active_slots
			limit = {
				pod_combat_slot_is_alive_trigger = yes
				# The set the actor can attack THIS turn: normally the opposing team; on a
				# controlled (dominated) turn the betrayed own side.
				trigger_if = {
					limit = { scope:host.var:pod_combat_session ?= { has_variable = pod_combat_turn_is_controlled } }
					var:combat_team = scope:host.var:pod_combat_session.var:pod_combat_current_actor_team
					NOT = { var:team_position = scope:host.var:pod_combat_session.var:pod_combat_current_actor_position }
				}
				trigger_else = {
					NOT = { var:combat_team = scope:host.var:pod_combat_session.var:pod_combat_current_actor_team }
				}
			}
			pod_combat_ai_fold_finishable_slot_effect = { HOST = scope:host }
		}
	}
	if = {
		limit = {
			OR = {
				var:pod_combat_session ?= { var:pod_combat_ai_enemy_finishable_hp = 1 }
				var:pod_combat_session ?= { var:pod_combat_ai_enemy_finishable_mor = 1 }
			}
		}
		var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_enemy_finishable value = 1 } }
	}
	else = { var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_enemy_finishable value = 0 } } }
}

# Controlled-aware target selection, shared by the turn-start path
# (pod_combat_execute_ai_turn_effect) and the per-action re-pick
# (pod_combat_ai_refresh_target_if_dead_effect). A dominated (controlled) actor
# targets its OWN side - the selection inverts; otherwise it targets the opposing
# side as normal. Both selectors self-exclude the actor, so a dominated
# actor never picks itself. Caller scope = the combat host; both the current
# actor and the mind-control mark are the fight's and reached through it.
pod_combat_ai_select_target_for_turn_effect = {
	if = {
		limit = { var:pod_combat_session ?= { has_variable = pod_combat_current_actor_team  var:pod_combat_current_actor_team = flag:player } }
		if = {
			limit = { var:pod_combat_session ?= { has_variable = pod_combat_turn_is_controlled } }
			pod_combat_ai_select_player_target_effect = yes
		}
		else = { pod_combat_ai_select_enemy_target_effect = yes }
	}
	else = {
		if = {
			limit = { var:pod_combat_session ?= { has_variable = pod_combat_turn_is_controlled } }
			pod_combat_ai_select_enemy_target_effect = yes
		}
		else = { pod_combat_ai_select_player_target_effect = yes }
	}
}

# Controlled-turn fizzle check: a dominated actor with NO alive own-side teammate
# to betray (sole survivor of its inverted side at turn start, OR it just KO'd its
# last teammate mid-turn) sets pod_combat_turn_control_fizzled so it stops acting
# rather than self-striking / hitting a corpse. Roster-based, so it composes with
# the selectors' self-exclusion. Gated on controlled = a no-op on a
# normal turn. Caller scope = the combat host.
pod_combat_ai_check_control_fizzle_effect = {
	save_scope_as = host
	if = {
		limit = {
			var:pod_combat_session ?= { has_variable = pod_combat_turn_is_controlled }
			NOT = {
				scope:host.var:pod_combat_session ?= {
					any_in_list = {
						variable = pod_combat_active_slots
						pod_combat_slot_is_alive_trigger = yes
						var:combat_team = scope:host.var:pod_combat_session.var:pod_combat_current_actor_team
						NOT = { var:team_position = scope:host.var:pod_combat_session.var:pod_combat_current_actor_position }
					}
				}
			}
		}
		var:pod_combat_session ?= { set_variable = { name = pod_combat_turn_control_fizzled value = 1 } }
	}
}

# Does the acting AI own an AP-affordable move of each damage kind this turn?
# Always-written host vars pod_combat_ai_can_dmg_hp / _mor (0 or 1). Ownership
# plus affordability only, like the finishable scan - no energy, once-per or
# magnitude test. The two price affordability differently on purpose (below).
#
# Two reads here are deliberately NOT the obvious ones, because the selector is
# entered from three paths that disagree about what is loaded (turn start, the
# re-pick after the mark died, and the alternate-target retry). Each read is the one
# that is correct on ALL THREE:
#   - the actor comes from scope:focus_actor, resolved off the current-actor
#     pointers, not from var:pod_combat_actor_char. On the turn-start path the
#     workspace has not been loaded yet, so actor_char still names the PRIOR actor
#     and would price the wrong character's loadout. The two mid-turn paths do have
#     it right; focus_actor is right on all of them.
#   - affordability is measured against the turn's AP budget, not
#     var:pod_combat_current_ap. Turn start runs before the AP seed, so current_ap is
#     still the previous actor's leftover. Pricing against the cap also keeps the
#     answer stable across a turn rather than flipping as AP is spent, so one actor
#     cannot gain and lose a capability between its own actions.
# The cap is snapshotted in host scope first because the SV reads host vars, so it
# would not resolve inside the loadout walk.
#
# The sibling pod_combat_ai_flag_finishable_enemy_effect DOES read actor_char and
# live AP, and is right to: it runs only from the per-action precompute, where both
# are current.
#
# Guarded on exists = scope:focus_actor, the idiom the selector's own weight arms
# use. The resolve cannot silently miss for a live actor - the current-actor pointers
# are written in lockstep and every selector entry is gated on them - so this writes
# a real answer rather than inheriting an earlier one.
pod_combat_ai_flag_actor_capability_effect = {
	save_scope_as = host
	var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_can_dmg_hp value = 0 } }
	var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_can_dmg_mor value = 0 } }
	save_scope_value_as = { name = cap_ap value = pod_combat_effective_ap_cap_value }
	if = {
		limit = {
			exists = scope:focus_actor
			scope:focus_actor = {
				any_in_list = {
					variable = pod_combat_loadout
					has_variable = hp_damage
					trigger_if = {
						limit = { has_variable = ap_cost }
						var:ap_cost <= scope:cap_ap
					}
				}
			}
		}
		var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_can_dmg_hp value = 1 } }
	}
	if = {
		limit = {
			exists = scope:focus_actor
			scope:focus_actor = {
				any_in_list = {
					variable = pod_combat_loadout
					has_variable = morale_damage
					trigger_if = {
						limit = { has_variable = ap_cost }
						var:ap_cost <= scope:cap_ap
					}
				}
			}
		}
		var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_can_dmg_mor value = 1 } }
	}
}

# Tallies one candidate for the trace: the total, plus how many sit in each of the
# kill factor's two bands. An effect rather than an inline scope switch because the
# tallies live on the host while the walk is in slot scope, and an inline switch
# inside an every_in_list body corrupts the iteration.
#
# The two band tests MIRROR THE KILL FACTOR'S ARMS EXACTLY, capability gate included
# (pod_combat_ai_select_target_core_effect). That is the whole point of recording
# them: without a count of what the factor could fire on, a flat result cannot be
# told apart from an absent opportunity. Change the factor and these change with it.
pod_combat_ai_trace_count_candidate_effect = {
	$HOST$.var:pod_combat_session ?= { change_variable = { name = pod_combat_ai_trace_sel_cand add = 1 } }
	if = {
		limit = {
			$HOST$.var:pod_combat_session ?= { var:pod_combat_ai_can_dmg_hp = 1 }
			var:slot_char.POD_empty_damage_boxes <= pod_combat_ai_kill_priority_boxes
		}
		$HOST$.var:pod_combat_session ?= { change_variable = { name = pod_combat_ai_trace_sel_kb_hp add = 1 } }
	}
	if = {
		limit = {
			$HOST$.var:pod_combat_session ?= { var:pod_combat_ai_can_dmg_mor = 1 }
			var:slot_morale <= pod_combat_ai_morale_kill_priority
		}
		$HOST$.var:pod_combat_session ?= { change_variable = { name = pod_combat_ai_trace_sel_kb_mor add = 1 } }
	}
}

# Stamps which call path is about to run the target selector, so a per-selection
# record can be read as a turn-start pick, a re-pick after the mark died, a
# last-resort retry, or a kill-snap. Host scope. This comment is the ONLY legend for
# the numeric ctx field in the trace - keep it exhaustive.
# 1 = turn start, 2 = target died, 3 = alternate target retry, 4 = kill-snap.
pod_combat_ai_trace_sel_ctx_effect = {
	if = {
		limit = { exists = global_var:pod_combat_tune_ai_trace }
		var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_trace_sel_ctx value = $CTX$ } }
	}
}

# Shared weighted target roll behind the two selectors below: random_in_list
# over alive slots passing $TEAM_TRIGGER$ (minus the acting slot), weighted
# toward low HP (kill-priority floor), cross-turn focus-fire, and temperament
# preferences; the winner lands in the host's selected-target pointers.
# Caller scope = the combat host.
pod_combat_ai_select_target_core_effect = {
	save_scope_as = host
	# This-pass proof: clear here, set only when the roll below actually matches a
	# slot. Guards the consumer against reusing a scope:chosen_target left over from
	# an earlier selection this turn (e.g. the alternate-target retry rolling empty),
	# which would otherwise silently re-target the stale slot.
	var:pod_combat_session ?= { remove_variable = pod_combat_ai_target_picked }
	pod_combat_ai_resolve_focus_actor_effect = yes
	pod_combat_ai_flag_actor_capability_effect = yes
	var:pod_combat_session ?= {
		random_in_list = {
			variable = pod_combat_active_slots
			limit = { pod_combat_ai_target_candidate_trigger = { TEAM_TRIGGER = $TEAM_TRIGGER$ } }
			weight = {
				base = 1
				modifier = { add = 50 }
				# Kill priority (factor 3): converge on an enemy inside the kill band on
				# a track this actor can damage. The band (pod_combat_ai_kill_priority_boxes
				# / pod_combat_ai_morale_kill_priority) sits one step looser than the finish
				# thresholds, so this fires before a target is finishable.
				#
				# ONE factor with an OR, never two stacking factors. An enemy low on both
				# tracks does not die any faster - the actor kills it through one of them
				# - so x9 would reward a state rather than an opportunity, and at 459 it
				# would drown focus-fire and the pref adds entirely.
				#
				# Each arm is gated on the matching capability, which is what stops a
				# morale specialist being dragged toward an HP-dying enemy it has no way
				# to hurt: most morale-damage moves deal no HP damage at all.
				# The gate only bites on a specialist, since almost any loadout with an
				# attack clears the AP cap - that narrowness is the point, not a defect.
				#
				# Weight budget, which the arms below must not break: healthy 51, +wounded
				# pref 71, +focus 91, +focus +threat pref 116 (the worst non-kill), against
				# 153 for anything this factor fires on. Kill > focus > prefs, margin 37.
				modifier = {
					factor = 3
					OR = {
						AND = {
							scope:host.var:pod_combat_session ?= { var:pod_combat_ai_can_dmg_hp = 1 }
							var:slot_char.POD_empty_damage_boxes <= pod_combat_ai_kill_priority_boxes
						}
						AND = {
							scope:host.var:pod_combat_session ?= { var:pod_combat_ai_can_dmg_mor = 1 }
							var:slot_morale <= pod_combat_ai_morale_kill_priority
						}
					}
				}
				# Cross-turn focus-fire: stick to last turn's mark. Kept ADDITIVE and
				# strictly below the gap the kill factor opens (focus < 51), so a fresh
				# kill-band enemy on a different slot still outranks a healthy focused
				# stick - see the weight budget above. One identity match per candidate,
				# no list-walk.
				modifier = {
					add = 40
					exists = scope:focus_actor
					scope:focus_actor = { has_variable = pod_combat_ai_focus_team }
					var:combat_team = scope:focus_actor.var:pod_combat_ai_focus_team
					var:team_position = scope:focus_actor.var:pod_combat_ai_focus_position
				}
				# Temperament target preference (precomputed flag on the actor; one
				# gated read via the already-resolved scope:focus_actor, no per-candidate
				# trait walk). wounded/threat are mutually exclusive. wounded's threshold
				# (pod_combat_ai_target_pref_wounded_threshold, one past the kill
				# trigger) means a sadist also leans toward moderately-hurt targets
				# outside the kill band; the factor-3 arm still ranks kill-band
				# targets above them.
				modifier = {
					add = pod_combat_ai_target_pref_wounded_weight
					exists = scope:focus_actor
					scope:focus_actor = {
						has_variable = pod_combat_ai_target_pref
						var:pod_combat_ai_target_pref = flag:wounded
					}
					var:slot_char.POD_empty_damage_boxes <= pod_combat_ai_target_pref_wounded_threshold
				}
				modifier = {
					add = pod_combat_ai_target_pref_threat_weight
					exists = scope:focus_actor
					scope:focus_actor = {
						has_variable = pod_combat_ai_target_pref
						var:pod_combat_ai_target_pref = flag:threat
					}
					exists = var:slot_char
					var:slot_char.pod_combat_char_prowess_value >= pod_combat_ai_target_pref_threat_threshold
				}
			}
			save_scope_as = chosen_target
			scope:host.var:pod_combat_session ?= { set_variable = pod_combat_ai_target_picked }
		}
	}
	if = {
		limit = { scope:host.var:pod_combat_session ?= { has_variable = pod_combat_ai_target_picked } }
		scope:host = {
			var:pod_combat_session ?= { set_variable = { name = pod_combat_selected_target_team     value = scope:chosen_target.var:combat_team } }
			var:pod_combat_session ?= { set_variable = { name = pod_combat_selected_target_position value = scope:chosen_target.var:team_position } }
		}
	}
	# Per-selection trace record. Everything, the candidate re-count included, sits
	# inside the one-clause gate, so with the trace off this costs a single
	# global-var existence check. Emitted even when the roll picks NOTHING: an empty
	# roll leaves no pick record at all, and the retry ladder is made of exactly
	# those. Identities ride numerically (1 = player, 2 = enemy) because a loc
	# string cannot safely read a flag var that may be unset.
	if = {
		limit = { exists = global_var:pod_combat_tune_ai_trace }
		var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_trace_sel_cand value = 0 } }
		var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_trace_sel_kb_hp value = 0 } }
		var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_trace_sel_kb_mor value = 0 } }
		var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_trace_sel_actor_t value = 0 } }
		if = {
			limit = {
				var:pod_combat_session ?= {
					has_variable = pod_combat_current_actor_team
					var:pod_combat_current_actor_team = flag:player
				}
			}
			var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_trace_sel_actor_t value = 1 } }
		}
		else_if = {
			limit = { var:pod_combat_session ?= { has_variable = pod_combat_current_actor_team } }
			var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_trace_sel_actor_t value = 2 } }
		}
		var:pod_combat_session ?= {
			every_in_list = {
				variable = pod_combat_active_slots
				limit = { pod_combat_ai_target_candidate_trigger = { TEAM_TRIGGER = $TEAM_TRIGGER$ } }
				pod_combat_ai_trace_count_candidate_effect = { HOST = scope:host }
			}
		}
		var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_trace_sel_picked value = 0 } }
		var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_trace_sel_team value = 0 } }
		var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_trace_sel_pos value = 0 } }
		var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_trace_sel_hp value = 0 } }
		var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_trace_sel_mor value = 0 } }
		if = {
			# scope:chosen_target is only this pass's winner while the proof var is
			# set - the head of this effect clears it, so an empty roll cannot stage
			# an earlier selection's slot.
			limit = { var:pod_combat_session ?= { has_variable = pod_combat_ai_target_picked } }
			var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_trace_sel_picked value = 1 } }
			var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_trace_sel_pos value = scope:chosen_target.var:team_position } }
			if = {
				limit = { scope:chosen_target = { var:combat_team = flag:player } }
				var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_trace_sel_team value = 1 } }
			}
			else = { var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_trace_sel_team value = 2 } } }
			if = {
				limit = { scope:chosen_target = { has_variable = slot_morale } }
				var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_trace_sel_mor value = scope:chosen_target.var:slot_morale } }
			}
			if = {
				limit = { scope:chosen_target = { exists = var:slot_char } }
				scope:chosen_target.var:slot_char = { save_scope_value_as = { name = sel_hp value = POD_empty_damage_boxes } }
				var:pod_combat_session ?= { set_variable = { name = pod_combat_ai_trace_sel_hp value = scope:sel_hp } }
			}
		}
		debug_log = pod_combat_trace_target
	}
}

##############################################################################################
# SECTION 10b: SLOT TARGET PERSISTENCE (UNIFIED)
#
# slot_target_team / _position on a slot story point at that slot's standing
# target (ally OR enemy); the skill's target-type is enforced by its can_use
# trigger, not by separate vars. A slot is never a valid target for itself.
#
# Called at turn start AND after every player move on the slot reached through
# pod_combat_active_control_slot: if the target got KO'd it auto-retargets to the
# first alive other slot (enemies first, then allies).
#
##############################################################################################

pod_combat_restore_slot_target_effect = {
	# Retarget if no alive slot matches the acting slot's stored target. The OR
	# also covers the unset case and the self-target case.
	save_scope_as = host
	var:pod_combat_session.var:pod_combat_active_control_slot ?= { save_scope_as = acting_slot }
	if = {
		limit = {
			exists = scope:acting_slot
			OR = {
				NOT = { scope:acting_slot = { has_variable = slot_target_team } }
				AND = {
					scope:acting_slot.var:slot_target_team = scope:acting_slot.var:combat_team
					scope:acting_slot.var:slot_target_position = scope:acting_slot.var:team_position
				}
				NOT = {
					scope:host.var:pod_combat_session ?= {
						any_in_list = {
							variable = pod_combat_active_slots
							var:combat_team = scope:acting_slot.var:slot_target_team
							var:team_position = scope:acting_slot.var:slot_target_position
							pod_combat_slot_is_alive_trigger = yes
						}
					}
				}
			}
		}
	# Auto-retarget: first alive enemy, else first alive ally that is not the
	# actor. The NOT = { exists = scope:chosen } in each limit makes the iteration
	# first-match. Both scans exclude the actor by identity rather than by board
	# position, so a controlled slot on either team retargets correctly.
		var:pod_combat_session ?= {
			every_in_list = {
				variable = pod_combat_active_slots
				limit = {
					NOT = { exists = scope:chosen }
					pod_combat_slot_is_enemy_trigger = yes
					pod_combat_slot_differs_from_trigger = { SLOT = scope:acting_slot }
					pod_combat_slot_is_alive_trigger = yes
				}
				save_scope_as = chosen
			}
		}
		var:pod_combat_session ?= {
			every_in_list = {
				variable = pod_combat_active_slots
				limit = {
					NOT = { exists = scope:chosen }
					pod_combat_slot_is_player_team_trigger = yes
					pod_combat_slot_differs_from_trigger = { SLOT = scope:acting_slot }
					pod_combat_slot_is_alive_trigger = yes
				}
				save_scope_as = chosen
			}
		}
		if = {
			limit = { exists = scope:chosen }
			scope:acting_slot = {
				set_variable = { name = slot_target_team     value = scope:chosen.var:combat_team }
				set_variable = { name = slot_target_position value = scope:chosen.var:team_position }
				set_variable = { name = slot_target_char     value = scope:chosen.var:slot_char }
			}
			scope:host = {
				var:pod_combat_session ?= { set_variable = { name = pod_combat_selected_target_team      value = scope:chosen.var:combat_team } }
				var:pod_combat_session ?= { set_variable = { name = pod_combat_selected_target_position  value = scope:chosen.var:team_position } }
			}
			scope:chosen = {
				pod_combat_load_target_from_story_effect = yes
			}
		}
	}
	else_if = {
		# Stored target is still alive - mirror it into selected_target (an AI may
		# have clobbered that pair with its own target during the previous turn) and
		# reload the target workspace with this slot's choice.
		limit = { exists = scope:acting_slot }
		var:pod_combat_session ?= {
			set_variable = { name = pod_combat_selected_target_team     value = scope:acting_slot.var:slot_target_team }
			set_variable = { name = pod_combat_selected_target_position value = scope:acting_slot.var:slot_target_position }
			every_in_list = {
				variable = pod_combat_active_slots
				limit = {
					var:combat_team = scope:acting_slot.var:slot_target_team
					var:team_position = scope:acting_slot.var:slot_target_position
				}
				pod_combat_load_target_from_story_effect = yes
			}
		}
	}
}

##############################################################################################
# SECTION 11: GROUP TURN LOOP
# The main group turn processing effect. Called after the PC ends their turn (or at round
# start for non-PC actors). Processes all remaining actors in initiative order.
##############################################################################################

# PC-turn setup. current_actor must already point at player/0. Order matters:
# apply_ap_drain reads the workspace, so the load must precede it.
# The recalc is intentionally redundant with load_workspace's pair-hit
# writeback (without it the PC slot-card hit chance refreshed only on a target
# click). Stance flags are NOT reset here: load_workspace already set/removes
# them per slot; stance persists until consumed.
# Seed the acting combatant's per-turn AP budget on the host.
# Default: full refill to the effective cap (pod_combat_effective_ap_cap_value,
# which also carries the escalating-cap tuning) + the per-character designer
# hook, floored at 0. When the partial-regen knob is tuned
# (global_var:pod_combat_tune_ap_regen), unspent AP snapshotted on the slot at
# turn end carries over: AP = min(cap, leftover + regen) + hook. A slot with
# no snapshot yet (first turn, fresh summon, knob toggled mid-fight) seeds the
# full bar. Caller scope = the combat host; runs after load_workspace so
# var:pod_combat_actor_char is current.
pod_combat_seed_turn_ap_effect = {
	save_scope_as = host
	if = {
		limit = { exists = global_var:pod_combat_tune_ap_regen }
		var:pod_combat_session ?= {
			every_in_list = {
				variable = pod_combat_active_slots
				limit = { pod_combat_slot_is_current_actor_trigger = yes }
				save_scope_as = pod_combat_ap_seed_slot
			}
		}
	}
	# The cap SV is host-scope only (it reads the host's session pointer), so it has
	# to be resolved to a number HERE - inside the fight block below it would ask the
	# fight story for a session pointer and silently yield 0.
	save_scope_value_as = { name = ap_seed_cap value = pod_combat_effective_ap_cap_value }
	# ?= keeps the deref exists-safe if the actor slot somehow failed to match.
	if = {
		limit = { scope:pod_combat_ap_seed_slot ?= { has_variable = slot_ap_leftover } }
		var:pod_combat_session ?= {
			set_variable = {
				name = pod_combat_current_ap
				value = {
					value = scope:pod_combat_ap_seed_slot.var:slot_ap_leftover
					add = global_var:pod_combat_tune_ap_regen
					max = scope:ap_seed_cap
					# The workspace actor pointer stays on the host.
					scope:host.var:pod_combat_session.var:pod_combat_actor_char = { add = pod_combat_ap_modifier_value }
					min = 0
				}
			}
		}
	}
	else = {
		var:pod_combat_session ?= {
			set_variable = {
				name = pod_combat_current_ap
				value = {
					value = scope:ap_seed_cap
					scope:host.var:pod_combat_session.var:pod_combat_actor_char = { add = pod_combat_ap_modifier_value }
					min = 0
				}
			}
		}
	}
	pod_combat_apply_ap_drain_effect = yes
}

pod_combat_setup_pc_turn_effect = {
	# HAND-OFF. Publish the acting slot and raise the player-turn latch before
	# anything else, because the target restore below reaches the slot's stored
	# target through the pointer this sets.
	#
	# current_actor must already point at the controlled slot. The one caller
	# that can arrive with NO actor (after_pre_pc_step_settle's defensive else)
	# matches nothing here, so no latch is raised and the pointer keeps its
	# previous value - the same dead, disabled UI that path produced before.
	save_scope_as = host
	if = {
		limit = { var:pod_combat_session ?= { has_variable = pod_combat_current_actor_team } }
		var:pod_combat_session ?= {
			every_in_list = {
				variable = pod_combat_active_slots
				limit = { pod_combat_slot_is_current_actor_trigger = yes }
				save_scope_as = pod_handoff_slot
				# Both writes inside the matched iteration, not hoisted out behind an
				# `exists = scope:pod_handoff_slot` gate: a saved scope outlives the
				# call, so a later no-actor call in the same effect chain would see the
				# PREVIOUS hand-off's slot and raise the latch on a turn nobody owns.
				# Scope switch is the body's last statement, keeping the iteration
				# scope clean (the mind-control matcher in execute_ai_turn is the same
				# shape).
				# The driver's own pointer follows the hand-off too, so their client
				# keeps resolving the body they are playing. Written before the
				# fight-side switch so the scope switch below stays the last
				# statement in the iteration.
				var:slot_human_controller ?= { set_variable = { name = pod_combat_my_control_slot value = scope:pod_handoff_slot } }
				scope:host.var:pod_combat_session ?= {
					set_variable = { name = pod_combat_active_control_slot value = scope:pod_handoff_slot }
					set_variable = { name = pod_combat_control_turn_live value = 1 }
				}
			}
		}
	}
	pod_combat_restore_slot_target_effect = yes
	# Round-duration buffs expire at the START of the holder's turn (a self soak buff
	# guarded the prior enemy phase; it lapses now). pod_combat_tier_effects.txt.
	pod_combat_clear_actor_round_buffs_effect = yes

	# Mid-fight form changes: re-derive the Delirium before the workspace load below,
	# so the stash and the hit recalc that follow read the corrected prowess on the
	# first frame of this turn rather than one turn late.
	pod_combat_refresh_delirium_effect = yes
	pod_combat_load_workspace_effect = yes
	pod_combat_recalc_pc_hit_chance_effect = yes
	pod_combat_seed_turn_ap_effect = yes
}

# Atomic AI step (find_next-first): select the next actor, then run exactly
# one AI turn. Used by both the main post-PC loop and the pre-PC loop so the
# pointer-advance count is identical in every context.
#
# The not-controlled guard is load-bearing in the PRE-PC context only: there the
# controlled slot is still unacted, so find_next can legitimately land on it. A
# bare execute would then run the player's turn as if it were AI. The guard keeps
# execute to genuine AI actors; the pre-PC loop's own limit stops the moment the
# pointer reaches a controlled slot. In the post-PC main loop that slot is already
# acted, so find_next never returns it and the guard simply never fires there.
# An ally AI on the player team is unmarked, so it IS a real AI step and still
# executes - the marker, not the team, is what excludes.
#
# The guard sits OUTSIDE the fight block: the trigger is character-scoped (it
# walks the roster from the caller's own session pointer) and carries its own
# no-actor guard, so it is safe as a sibling.
pod_combat_advance_one_ai_turn_effect = {
	pod_combat_find_next_actor_effect = yes
	if = {
		limit = {
			NOT = { pod_combat_current_actor_is_controlled_trigger = yes }
			var:pod_combat_session ?= {
				has_variable = pod_combat_current_actor_team
				NOT = { has_variable = pod_combat_is_over }
			}
		}
		pod_combat_execute_ai_turn_effect = yes
	}
}

# Post-PC main loop: drain every remaining AI in initiative order over the
# alive + unacted + not-over set, bodied by the atomic step. Terminates because
# each step marks one slot acted.
pod_combat_run_remaining_ai_turns_effect = {
	# Prime the actor pointer, exactly as the pre-PC pump does. Load-bearing since
	# the while limit below started reading current_actor: the caller commits with
	# the pointer still parked on the controlled slot it just spent (the commit
	# marks it acted but nothing re-points), so without this the stop-at-controller
	# is true on pass 0, the drain runs ZERO turns, and the tail hands the player
	# back the turn they just took. The old limit never read the pointer - it asked
	# only "does an alive unacted slot exist" - which is why no prime was needed
	# before and why adding one is not optional now.
	#
	# Idempotent: advance_one_ai_turn re-runs find_next anyway. Skipped once the
	# fight is over so a decided fight is not re-pointed at a live slot.
	if = {
		limit = { NOT = { var:pod_combat_session ?= { has_variable = pod_combat_is_over } } }
		pod_combat_find_next_actor_effect = yes
	}
	while = {
		# Hang-safe ceiling (each step marks one slot acted, so real termination is
		# <= combatant count; only trips on a malformed no-progress state).
		count = pod_combat_ai_outer_turn_cap_value
		limit = {
			# Stop the moment the walk reaches a controlled slot, exactly like the
			# pre-PC pump. Without this the loop and the atomic step disagree:
			# find_next selects the controlled slot, advance_one_ai_turn refuses to
			# execute it, nobody marks it acted, and the limit stays true - so the
			# loop spins to its ceiling and the slot is silently skipped for the
			# round. Inert while one slot is marked (it has already committed by the
			# time this runs, and find_next only picks unacted slots), which is why
			# the bug was invisible.
			NOT = { pod_combat_current_actor_is_controlled_trigger = yes }
			var:pod_combat_session ?= {
				NOT = { has_variable = pod_combat_is_over }
				any_in_list = {
					variable = pod_combat_active_slots
					pod_combat_slot_is_alive_trigger = yes
					pod_combat_slot_has_not_acted_trigger = yes
				}
			}
		}
		pod_combat_advance_one_ai_turn_effect = yes
	}

	# Stopped at a controlled slot -> hand off, same tail as the pre-PC pump.
	# Callers must not end the round after this; see the stopped-for-a-human gate
	# at both AUTO call sites.
	#
	# The alive + not-acted terms are belt to the prime above. A legitimate stop
	# always lands on an alive unacted slot by construction (find_next selects
	# nothing else), so they can never suppress a real hand-off - but they DO stop
	# a stale pointer from re-granting a spent turn, which is precisely how this
	# effect broke when the limit first learned to read the pointer.
	if = {
		limit = {
			pod_combat_current_actor_is_controlled_trigger = yes
			var:pod_combat_session ?= {
				NOT = { has_variable = pod_combat_is_over }
				any_in_list = {
					variable = pod_combat_active_slots
					has_variable = slot_human_controller
					pod_combat_slot_is_alive_trigger = yes
					pod_combat_slot_has_not_acted_trigger = yes
				}
			}
		}
		pod_combat_setup_pc_turn_effect = yes
	}

	# No-progress diagnostic: bounded loop stopped with alive unacted slots while
	# the fight is not over and nothing is waiting on a human -> a step failed to
	# mark progress.
	if = {
		limit = {
			NOT = { pod_combat_current_actor_is_controlled_trigger = yes }
			var:pod_combat_session ?= {
				NOT = { has_variable = pod_combat_is_over }
				any_in_list = {
					variable = pod_combat_active_slots
					pod_combat_slot_is_alive_trigger = yes
					pod_combat_slot_has_not_acted_trigger = yes
				}
			}
		}
		debug_log = "pod_combat: run_remaining_ai_turns hit iteration cap with unacted alive slots"
	}
}

# Pre-PC loop (combat start / round start): walk AI actors in initiative
# order until the PC's slot comes up, leaving current_actor on the PC (or
# cleared if the round exhausts). Each step is exactly one find_next + at most
# one execute. The single prime find_next selects (does not consume) the first
# actor so the while limit can read current_actor on the first pass; a caller's
# own earlier find_next is a harmless idempotent re-select.
pod_combat_run_pre_pc_ai_turns_effect = {
	pod_combat_find_next_actor_effect = yes
	while = {
		# Hang-safe ceiling; see pod_combat_run_remaining_ai_turns_effect.
		count = pod_combat_ai_outer_turn_cap_value
		limit = {
			# Stop the moment the walk reaches a controlled slot (same guard the
			# original execute-first loop carried). Hoisted out of the fight block
			# because it is character-scoped; see advance_one_ai_turn.
			NOT = { pod_combat_current_actor_is_controlled_trigger = yes }
			var:pod_combat_session ?= {
				has_variable = pod_combat_current_actor_team
				NOT = { has_variable = pod_combat_is_over }
				any_in_list = {
					variable = pod_combat_active_slots
					pod_combat_slot_is_alive_trigger = yes
					pod_combat_slot_has_not_acted_trigger = yes
				}
			}
		}
		pod_combat_advance_one_ai_turn_effect = yes
	}

	# No-progress diagnostic: bounded pre-PC loop stopped before reaching a
	# controlled slot while an AI actor remained and the fight was not over.
	if = {
		limit = {
			NOT = { pod_combat_current_actor_is_controlled_trigger = yes }
			var:pod_combat_session ?= {
				has_variable = pod_combat_current_actor_team
				NOT = { has_variable = pod_combat_is_over }
				any_in_list = {
					variable = pod_combat_active_slots
					pod_combat_slot_is_alive_trigger = yes
					pod_combat_slot_has_not_acted_trigger = yes
				}
			}
		}
		debug_log = "pod_combat: run_pre_pc_ai_turns hit iteration cap before reaching the PC"
	}

	# Controlled slot reached (and combat not over) -> set up its turn. The loop
	# leaves current_actor there, so this fires exactly when the walk stopped at a
	# marked slot rather than at is_over.
	if = {
		limit = {
			pod_combat_current_actor_is_controlled_trigger = yes
			var:pod_combat_session ?= { NOT = { has_variable = pod_combat_is_over } }
		}
		pod_combat_setup_pc_turn_effect = yes
	}
}

pod_combat_process_group_turns_effect = {
	save_scope_as = host

	# Mark the current actor as acted; snapshot unspent AP for the regen-knob
	# carry-over. Named scope rather than prev: the AP now sits two hops away
	# (host -> fight), and prev only ever meant the one hop.
	#
	# Keyed on the current-actor pointers (the AI commit's matcher shape) rather
	# than pod_combat_pc_slot: this runs only while a controlled turn is live, and
	# a live turn means the pointers are parked on the controlled slot, so the two
	# name the same slot today. The matcher stays right when that slot is not
	# player/0.
	if = {
		limit = { var:pod_combat_session ?= { has_variable = pod_combat_current_actor_team } }
		var:pod_combat_session ?= {
			every_in_list = {
				variable = pod_combat_active_slots
				limit = { pod_combat_slot_is_current_actor_trigger = yes }
				pod_combat_set_slot_var_effect = { NAME = has_acted VALUE = 1 }
				pod_combat_set_slot_var_effect = { NAME = ap_leftover VALUE = scope:host.var:pod_combat_session.var:pod_combat_current_ap }
			}
		}
	}

	pod_combat_save_workspace_effect = yes

	# COMMIT. The controlled turn is spent: drop the hand-off latch before any AI
	# runs below, so the GUI reads "not my turn" for the whole AI drain. The
	# display pointer (pod_combat_active_control_slot) deliberately survives - the
	# target badge and readout must keep rendering through the AI phase.
	var:pod_combat_session ?= { remove_variable = pod_combat_control_turn_live }

	# Discard the human's damage accounting. The trace counters are incremented by
	# the shared executor, so a human attack adds to them, but only the AI turn
	# emitter clears them - so without this the player's damage lands on the NEXT
	# AI's row and corrupts per-actor attribution. Cleared rather than emitted
	# because the human has no turn record of its own; giving it one is the better
	# fix and a larger change.
	if = {
		limit = { exists = global_var:pod_combat_tune_ai_trace }
		var:pod_combat_session ?= { remove_variable = pod_combat_ai_trace_soaked }
		var:pod_combat_session ?= { remove_variable = pod_combat_ai_trace_landed }
		var:pod_combat_session ?= { remove_variable = pod_combat_ai_trace_hits }
	}

	# Round-duration debuffs expire at the END of the holder's turn (they afflicted
	# this whole turn). pod_combat_tier_effects.txt.
	pod_combat_clear_actor_round_debuffs_effect = yes

	# Same refresh, on the PC-commit side: a form change made during the turn just
	# committed has to land before the AI drain below, or every AI action this round
	# swings against the pre-shift numbers.
	pod_combat_refresh_delirium_effect = yes

	# the prologue above (mark PC acted + save PC
	# workspace) is the PC-commit, run EXACTLY ONCE per PC turn here. This is the
	# only effect that carries it; mid-round Advance clicks route through
	# pod_combat_advance_turn (prologue-free), so it never re-fires.
	if = {
		limit = { var:pod_combat_session ?= { has_variable = pod_combat_step_mode } }
		# STEP: take only the FIRST remaining AI turn, then return to the GUI.
		# after_step_settle re-points current_actor at the next pending actor (or
		# ends the round) and hands control back to the PC if it is up next.
		pod_combat_advance_one_ai_turn_effect = yes
		pod_combat_after_step_settle_effect = yes
	}
	else = {
		# AUTO: drain every remaining AI, then end the round if combat is not over
		# AND the drain actually ran out rather than stopping to hand a second
		# controlled slot its turn. Ending the round there would consume the turn
		# the drain just set up.
		pod_combat_run_remaining_ai_turns_effect = yes
		if = {
			limit = {
				NOT = { pod_combat_current_actor_is_controlled_trigger = yes }
				NOT = { var:pod_combat_session ?= { has_variable = pod_combat_is_over } }
			}
			pod_combat_group_end_round_effect = yes
		}
	}
}

# post-step branch for the IN-ROUND step path (after
# the PC has committed). Decides, without draining the rest of the AI, whether
# the round just ended, the PC is up next, or combat is over. ALWAYS re-runs
# find_next first so current_actor reflects the NEXT pending actor (never the
# just-consumed one) before the GUI reads it for the End-Turn/Advance label.
pod_combat_after_step_settle_effect = {
	if = {
		limit = { var:pod_combat_session ?= { has_variable = pod_combat_is_over } }
		# Nothing: GUI swaps End Turn / Advance -> End Combat via is_over gating.
	}
	else = {
		# Re-point at the next pending actor, or clear it if the round is over.
		pod_combat_find_next_actor_effect = yes
		if = {
			limit = { var:pod_combat_session ?= { has_variable = pod_combat_current_actor_team } }
			if = {
				# Next actor is controlled -> set up its turn, hand control back.
				# The button relabels to End Turn and the move bar lights up.
				limit = { pod_combat_current_actor_is_controlled_trigger = yes }
				pod_combat_setup_pc_turn_effect = yes
			}
			# else: an AI is still pending -> leave current_actor on it; the
			# Advance button stays live for the next click.
		}
		else = {
			# find_next cleared current_actor -> round exhausted. Run the normal
			# transition; group_end_round (mode-aware) sets up the PC turn, begins
			# stepping the next round's pre-PC AI, or ends combat. This is the last
			# statement, so on a DoT-KO boundary control falls back to the GUI with
			# is_over set and no PC turn queued.
			pod_combat_group_end_round_effect = yes
		}
	}
}

# post-step branch for the PRE-PC step path (combat
# start / round start, BEFORE the PC has acted). Mirrors after_step_settle but
# its done-stepping target is "current actor == player/0" (the pre-PC native
# exit), and it likewise ALWAYS re-runs find_next once before branching.
pod_combat_after_pre_pc_step_settle_effect = {
	if = {
		limit = { var:pod_combat_session ?= { has_variable = pod_combat_is_over } }
		# Nothing; GUI handles via is_over gating.
	}
	else = {
		# Repoint at the next pending actor / clear.
		pod_combat_find_next_actor_effect = yes
		if = {
			limit = { pod_combat_current_actor_is_controlled_trigger = yes }
			# Controlled slot reached -> hand control back.
			pod_combat_setup_pc_turn_effect = yes
		}
		else_if = {
			# An AI is still pending -> leave current_actor on it; Advance stays live.
			limit = { var:pod_combat_session ?= { has_variable = pod_combat_current_actor_team } }
		}
		else = {
			# Defensive: a fresh pre-PC round can't exhaust without reaching the PC
			# (the round was just seeded with the PC unacted), but if find_next ever
			# clears the pointer, fall through to PC setup rather than stall.
			pod_combat_setup_pc_turn_effect = yes
		}
	}
}

pod_combat_execute_ai_turn_effect = {

	save_scope_as = combat_host

	# Belt to the commit's clear in process_group_turns: an AI is about to act, so
	# no controlled turn can be live. Covers any entry path that reaches an AI turn
	# without passing the commit (the pre-PC drain, a mid-round Advance).
	var:pod_combat_session ?= { remove_variable = pod_combat_control_turn_live }

	# STEP mode lets the player shapeshift between two Advance clicks, so re-derive
	# before target selection and the workspace load: the AI threat weighting and the
	# pair hit chance both read live prowess.
	pod_combat_refresh_delirium_effect = yes

	# Mind control: detect a stolen turn before target selection. The resolve
	# loop mirrors the mark-acted matcher at the bottom; the host flag gates
	# the selector inversion here and the AI move filter's controlled-turn
	# restrictions.
	if = {
		limit = { var:pod_combat_session ?= { has_variable = pod_combat_current_actor_team } }
		var:pod_combat_session ?= {
			every_in_list = {
				variable = pod_combat_active_slots
				limit = {
					var:combat_team = scope:combat_host.var:pod_combat_session.var:pod_combat_current_actor_team
					var:team_position = scope:combat_host.var:pod_combat_session.var:pod_combat_current_actor_position
				}
				if = {
					limit = { has_variable = slot_controlled }
					scope:combat_host.var:pod_combat_session ?= { set_variable = { name = pod_combat_turn_is_controlled value = 1 } }
				}
			}
		}
	}

	# Hoist the per-AI personality precompute to BEFORE the first target
	# selection. The bias -> target-pref flag (pod_combat_ai_target_pref) is read by
	# the GROUP target selectors below; running it lazily at move-selection time left
	# each AI's FIRST target pick of a combat without its temperament pref. Same
	# once-per-combat guard (pers_profile_done, on the AI char). Actor = the slot
	# matching the current-actor pointers (pod_combat_actor_char is not written until
	# load_workspace below); single-match loop, and the scope switch into var:slot_char
	# is the body's last statement so the iteration scope stays clean. Mirrors the
	# controlled-turn matcher above.
	if = {
		limit = { var:pod_combat_session ?= { has_variable = pod_combat_current_actor_team } }
		var:pod_combat_session ?= {
			every_in_list = {
				variable = pod_combat_active_slots
				limit = {
					var:combat_team = scope:combat_host.var:pod_combat_session.var:pod_combat_current_actor_team
					var:team_position = scope:combat_host.var:pod_combat_session.var:pod_combat_current_actor_position
				}
				var:slot_char = {
					if = {
						limit = { NOT = { has_variable = pod_combat_pers_profile_done } }
						pod_combat_apply_personality_profile_effect = yes
						set_variable = pod_combat_pers_profile_done
					}
				}
			}
		}
	}

	# Fresh AI-turn scratch: clear the progress/stall flags and any leftover
	# alternate-target exclusion so a prior actor's stall can't block this turn.
	var:pod_combat_session ?= { remove_variable = pod_combat_ai_turn_stalled }
	var:pod_combat_session ?= { remove_variable = pod_combat_ai_action_executed }
	# Belt to try_action's braces: that clear sits inside a gate an already-stalled
	# or downed actor never enters, so a lock could otherwise outlive its turn.
	var:pod_combat_session ?= { remove_variable = pod_combat_ai_intent_lock }
	var:pod_combat_session ?= { remove_variable = pod_combat_ai_excluded_target_team }
	var:pod_combat_session ?= { remove_variable = pod_combat_ai_excluded_target_position }
	# Kill-snap scratch. snap_blocked is turn-scoped by design; the other two are
	# per-attempt and per-call, cleared here as a belt so a turn that ended mid-attempt
	# cannot leak one into the next actor.
	var:pod_combat_session ?= { remove_variable = pod_combat_ai_snap_blocked }
	var:pod_combat_session ?= { remove_variable = pod_combat_ai_snap_done }
	var:pod_combat_session ?= { remove_variable = pod_combat_ai_snapped_this_turn }
	var:pod_combat_session ?= { remove_variable = pod_combat_ai_snap_restrict }

	# Select target (controlled-aware) + check the controlled-turn fizzle. The
	# inverted-aware selection is SHARED with the per-action re-pick in
	# pod_combat_ai_refresh_target_if_dead_effect, so a multi-AP stolen turn keeps
	# betraying its own side instead of reverting to the normal enemy side.
	# The fizzle is roster-based (no alive own-side teammate to betray), so it
	# composes with the selectors' self-exclusion.
	pod_combat_ai_trace_sel_ctx_effect = { CTX = 1 }
	pod_combat_ai_select_target_for_turn_effect = yes
	pod_combat_ai_check_control_fizzle_effect = yes

	# Round-duration buffs expire at the START of the holder's turn (a self soak buff
	# guarded the prior enemy phase; it lapses now). pod_combat_tier_effects.txt.
	pod_combat_clear_actor_round_buffs_effect = yes

	# Load workspace (actor -> actor workspace, target -> target workspace)
	pod_combat_load_workspace_effect = yes

	# Record this turn's chosen target as the
	# actor's persistent focus, ONCE per turn (here, not the per-action loop).
	# pod_combat_actor_char is now the current actor (load_workspace wrote it).
	# Rationale for char-var storage and the controlled-turn clear lives on
	# pod_combat_ai_record_focus_target_effect.
	pod_combat_ai_record_focus_target_effect = yes

	# The finishable-enemy flag is raised per action, not here, so its affordability
	# check reads live AP (pod_combat_ai_precompute_action_summary_effect).

	pod_combat_seed_turn_ap_effect = yes

	# Run AI moves (uses existing pod_combat_ai_take_turn_effect which reads
	# from workspace). A fizzled stolen turn skips the moves entirely.
	if = {
		limit = { var:pod_combat_session ?= { has_variable = pod_combat_turn_control_fizzled } }
		var:pod_combat_session ?= { var:pod_combat_actor_char = { save_scope_as = mind_log_victim } }
		pod_combat_push_mind_log_effect = { MESSAGE_KEY = mind_fizzle }
	}
	else = {
		if = {
			limit = { var:pod_combat_session ?= { has_variable = pod_combat_turn_is_controlled } }
			var:pod_combat_session ?= { var:pod_combat_actor_char = { save_scope_as = mind_log_victim } }
			pod_combat_push_mind_log_effect = { MESSAGE_KEY = mind_betrayal }
		}
		pod_combat_ai_take_turn_effect = yes
	}

	pod_combat_save_workspace_effect = yes

	# Round-duration debuffs expire at the END of the holder's turn (they afflicted
	# this whole turn). pod_combat_tier_effects.txt.
	pod_combat_clear_actor_round_debuffs_effect = yes

	# Mark the actor acted (slot_has_acted = 1 on the matching slot). scope:host
	# persists from the save_workspace above; the limit matches at most one slot.
	if = {
		limit = { var:pod_combat_session ?= { has_variable = pod_combat_current_actor_team } }
		var:pod_combat_session ?= {
			every_in_list = {
				variable = pod_combat_active_slots
				limit = {
					var:combat_team = scope:host.var:pod_combat_session.var:pod_combat_current_actor_team
					var:team_position = scope:host.var:pod_combat_session.var:pod_combat_current_actor_position
				}
				set_variable = { name = slot_has_acted value = 1 }
				# Snapshot unspent AP for the regen-knob carry-over.
				set_variable = { name = slot_ap_leftover value = scope:host.var:pod_combat_session.var:pod_combat_current_ap }
				# One stolen turn exactly: the control marker dies with the turn
				# it consumed, fizzled or not.
				if = {
					limit = { scope:host.var:pod_combat_session ?= { has_variable = pod_combat_turn_is_controlled } }
					remove_variable = slot_controlled
				}
			}
		}
	}
	var:pod_combat_session ?= {
		remove_variable = pod_combat_turn_is_controlled
		remove_variable = pod_combat_turn_control_fizzled
	}

	pod_combat_check_group_victory_effect = yes
}

##############################################################################################
# SECTION 12: GROUP VICTORY CHECK
# Check if all enemies are knocked out (player wins) or all allies are knocked out
# (player loses).
##############################################################################################

pod_combat_check_group_victory_effect = {
	# Victory conditions (evaluated in order, first match ends combat):
	#   1. All enemies KO'd              -> player team wins
	#   2. Every human-driven slot down  -> player loses (nothing left to play)
	#   3. All player-team slots down    -> player loses

	# scope:combat_pc is the CALLER: a bare `root` in the set_variable arms below
	# errors with "Event target link 'root' returned unset scope". It is no longer
	# what gets recorded as winner or loser - see the principal helper below.
	save_scope_as = combat_pc
	# The player side's stand-in for the outcome pair: the player team's leader
	# occupant, which is the caller in an ordinary duel and a directed combatant
	# in a proxy fight. Saved once here so all three checks name the same
	# character, and saved BEFORE any check so the arms can read it.
	pod_combat_save_player_principal_effect = yes

	# --- Check 1: All enemies knocked out ---
	# All three checks express a team wipe as 'no alive slot of that team via
	# any_in_list'. pod_combat_active_slots is filled-only, so no slot_char gate.
	if = {
		limit = {
			NOT = {
				var:pod_combat_session ?= {
					any_in_list = {
						variable = pod_combat_active_slots
						pod_combat_slot_is_enemy_trigger = yes
						pod_combat_slot_is_alive_trigger = yes
					}
				}
			}
			var:pod_combat_session ?= { has_variable = pod_combat_enemy_leader_slot } var:pod_combat_session.var:pod_combat_enemy_leader_slot ?= { has_variable = slot_char }
			NOT = { var:pod_combat_session ?= { has_variable = pod_combat_is_over } }
		}
		# Player team wins! The loser chain is re-rooted through scope:combat_pc:
		# inside the fight block a bare var:pod_combat_session would ask the fight
		# story for its own session pointer, which does not exist.
		var:pod_combat_session ?= {
			set_variable = { name = pod_combat_winner value = scope:pod_combat_player_principal }
			set_variable = { name = pod_combat_loser value = scope:combat_pc.var:pod_combat_session.var:pod_combat_enemy_leader_slot.var:slot_char }
			set_variable = { name = pod_combat_is_over value = yes }
		}
		# Stays in host scope: the saved scope must name the character.
		var:pod_combat_session.var:pod_combat_enemy_leader_slot.var:slot_char = { save_scope_as = pod_combat_target }
		# System log: victory. Pushed AFTER is_over so it lands at the TOP of the
		# newest-first log.
		pod_combat_push_system_log_effect = { MESSAGE_KEY = combat_ended_victory }
		# Combat window stays open - player clicks "End Combat" to dismiss
	}

	# --- Check 2: every human-driven slot is down (even if allies still stand) ---
	# Nothing left for the player to play = loss; end so the post-fight event resolves.
	#
	# Keyed on the control marker, not on player/0. Identical while one slot is
	# marked, and it is what stops the window going dead under clones: find_next
	# only walks ALIVE slots and the drain only hands off when the walk STOPS at a
	# controlled slot, so with every controlled slot dead and this check still
	# asking about the leader, the round would exhaust with no actor, nothing to
	# end it, and both is_player_turn and can_advance reading false.
	#
	# The any_in_list is negated rather than asking "the marked slot is KO'd" so
	# that a fight with no marked slot at all (a launcher that never marked one)
	# does not read as an instant loss.
	if = {
		limit = {
			var:pod_combat_session ?= {
				any_in_list = {
					variable = pod_combat_active_slots
					has_variable = slot_human_controller
				}
				NOT = {
					any_in_list = {
						variable = pod_combat_active_slots
						has_variable = slot_human_controller
						pod_combat_slot_is_alive_trigger = yes
					}
				}
			}
			var:pod_combat_session ?= { has_variable = pod_combat_enemy_leader_slot } var:pod_combat_session.var:pod_combat_enemy_leader_slot ?= { has_variable = slot_char }
			NOT = { var:pod_combat_session ?= { has_variable = pod_combat_is_over } }
		}
		# Never record a downed/dead enemy leader as the victor - credit
		# the lowest-position ALIVE enemy (the same helper the concede and
		# auto-resolve paths use); the leader stays the fallback on a full wipe.
		pod_combat_save_enemy_victor_effect = yes
		var:pod_combat_session ?= {
			set_variable = { name = pod_combat_winner value = scope:pod_combat_enemy_victor }
			set_variable = { name = pod_combat_loser value = scope:pod_combat_player_principal }
			set_variable = { name = pod_combat_is_over value = yes }
		}
		scope:pod_combat_enemy_victor = { save_scope_as = pod_combat_target }
		pod_combat_push_system_log_effect = { MESSAGE_KEY = combat_ended_defeat }
		# Combat window stays open - player clicks "End Combat" to dismiss
	}

	# --- Check 3: All allies knocked out --- symmetric to Check 1, player
	# team; covers a bypassed PC death check.
	if = {
		limit = {
			NOT = {
				var:pod_combat_session ?= {
					any_in_list = {
						variable = pod_combat_active_slots
						pod_combat_slot_is_player_team_trigger = yes
						pod_combat_slot_is_alive_trigger = yes
					}
				}
			}
			var:pod_combat_session ?= { has_variable = pod_combat_enemy_leader_slot } var:pod_combat_session.var:pod_combat_enemy_leader_slot ?= { has_variable = slot_char }
			NOT = { var:pod_combat_session ?= { has_variable = pod_combat_is_over } }
		}
		# Enemy team wins: credit an ALIVE enemy, not a downed leader
		# (see Check 2).
		pod_combat_save_enemy_victor_effect = yes
		var:pod_combat_session ?= {
			set_variable = { name = pod_combat_winner value = scope:pod_combat_enemy_victor }
			set_variable = { name = pod_combat_loser value = scope:pod_combat_player_principal }
			set_variable = { name = pod_combat_is_over value = yes }
		}
		scope:pod_combat_enemy_victor = { save_scope_as = pod_combat_target }
		pod_combat_push_system_log_effect = { MESSAGE_KEY = combat_ended_defeat }
		# Combat window stays open - player clicks "End Combat" to dismiss
	}
}

##############################################################################################
# SECTION 13: GROUP ROUND END
# Increment round, calculate new initiative, expire round-duration tier buffs,
# and set up the next round.
##############################################################################################

# Round-end Damage-over-Time tick. Hooked into pod_combat_group_end_round_effect
# after the round increment, before reinforcements/initiative. Walks every alive
# slot carrying a burn (slot_dot_rounds > 0): logs the tick, decrements, clears at
# 0, then deals the damage. HP ticks bypass soak/guard via the workspace round-trip
# (pod_combat_deal_dot_tick_from_host_effect); morale ticks erode slot_morale.
# Then re-marks KO (shared predicate) + re-checks victory once: a burn can end the
# fight between turns, so a downed slot must be out before the new round seats.
pod_combat_tick_dots_effect = {
	# combat_host: read by the log push + as the wound ATTACKER. host: contract name for
	# the workspace load/save helpers (same dual-save the AoE dispatchers use).
	save_scope_as = combat_host
	save_scope_as = host
	var:pod_combat_session ?= {
		every_in_list = {
			variable = pod_combat_active_slots
			limit = {
				var:slot_alive = 1
				has_variable = slot_dot_rounds
				var:slot_dot_rounds > 0
			}
			# Phase A: slot-local reads and writes only, BEFORE any inline scope switch (an inline
			# switch corrupts the iteration scope for every later statement in the body - one root
			# cause of the long DoT-damage hunt). The amount is lifted into a scope VALUE (numbers
			# survive that fine); the KIND is NOT - reading a flag var through save_scope_value_as
			# degrades it to 'value' type and every later flag comparison TYPE-ERRORS (the other
			# root cause). The host dispatch reads the kind through the saved slot object instead.
			save_scope_as = dot_slot
			save_scope_value_as = { name = dot_tick_amount value = var:slot_dot_amount }
			var:slot_char = { save_scope_as = dot_log_victim }
			# Resolve the burn's attacker - the caster (slot_dot_source) if still
			# alive, else self-attribute to the victim (always valid, never mis-credits
			# a live third party). Slot-local, before any inline scope switch.
			if = {
				limit = { has_variable = slot_dot_source  var:slot_dot_source = { is_alive = yes } }
				var:slot_dot_source = { save_scope_as = dot_attacker }
			}
			else = {
				var:slot_char = { save_scope_as = dot_attacker }
			}
			change_variable = { name = slot_dot_rounds subtract = 1 }
			# Phase B: deal the tick. HP ticks use the proven AoE per-target shape: load the slot
			# into the TARGET workspace, wound from HOST scope, save back, KO-check. The fork
			# compares the slot VAR (a literal flag since the §13b reseed). The tick's log row
			# is pushed AFTER this, resolved, by pod_combat_push_dot_tick_resolved_log_effect.
			if = {
				limit = { var:slot_dot_kind = flag:morale }
				pod_combat_deal_slot_dot_morale_tick_effect = yes
			}
			else = {
				pod_combat_load_target_from_story_effect = yes
				scope:host = { pod_combat_deal_dot_tick_from_host_effect = yes }
				pod_combat_save_target_to_story_effect = yes
				pod_combat_aoe_check_target_alive_effect = yes
			}
			# Log the tick AFTER it resolves, with the actual applied amount (a
			# CALL: reads only saved scopes, safe after Phase B's inline switches).
			pod_combat_push_dot_tick_resolved_log_effect = yes
			# Phase C: expiry, AFTER the wound (the host dispatch reads slot_dot_kind off the slot,
			# so it must survive the final tick). A scripted-effect CALL: calls stay reliable after
			# phase B's inline switches, inline statements do not.
			pod_combat_expire_dot_if_done_effect = yes
		}
	}
	scope:host = { pod_combat_load_workspace_effect = yes }
	var:pod_combat_session ?= {
		every_in_list = {
			variable = pod_combat_active_slots
			limit = {
				var:slot_alive = 1
				pod_combat_slot_should_be_marked_ko_trigger = yes
			}
			pod_combat_mark_slot_ko_effect = yes
		}
	}
	pod_combat_check_group_victory_effect = yes

	# HP display mirror. Round-end DoT batch is the
	# only HP path the per-action save_workspace refresh misses (its trailing
	# load_workspace only reloads the actor/target pair), so refresh all slots here.
	pod_combat_refresh_hp_mirror_effect = yes
}

pod_combat_group_end_round_effect = {
	# Guard: skip if combat already ended this round (vars may be cleaned up).
	if = {
		limit = { NOT = { var:pod_combat_session ?= { has_variable = pod_combat_round } } }
		# Combat already cleaned up - nothing to do
	}
	else = {

	var:pod_combat_session ?= { change_variable = { name = pod_combat_round add = 1 } }

	# System log: round transition. Pushed AFTER the increment so entry.var:round
	# captures the NEW round number.
	pod_combat_push_system_log_effect = { MESSAGE_KEY = round_begins }

	# Tick active Damage-over-Time burns before reinforcements/initiative so a slot
	# the burn just downed is out before the new round seats and rolls turn order.
	# The tick runs the victory check, which can set pod_combat_is_over on a killing
	# round-end burn; everything after it is guarded so a DoT-ended combat doesn't
	# seat reinforcements / roll initiative / set up a phantom next turn (matches the
	# init path's is_over guard).
	pod_combat_tick_dots_effect = yes

	if = {
		limit = { NOT = { var:pod_combat_session ?= { has_variable = pod_combat_is_over } } }

	# Seat reinforcements scheduled to arrive this round, before initiative so they
	# roll into the new round's turn order and act this round.
	pod_combat_process_pending_reinforcements_effect = yes

	# Backstop for any form change the two turn-side sites missed, placed after the
	# reinforcement seat (its own battery already afflicted the new arrival, so the
	# candidate filter skips it) and before initiative, which is prowess-derived.
	pod_combat_refresh_delirium_effect = yes

	pod_combat_calculate_initiative_effect = yes
	pod_combat_find_next_actor_effect = yes

	# Round-duration tiers clear at each combatant's own turn boundary (buffs
	# at turn start, debuffs at turn end), not here.

	# If a controlled slot goes first, set up workspace and wait for player input
	if = {
		limit = { pod_combat_current_actor_is_controlled_trigger = yes }
		pod_combat_setup_pc_turn_effect = yes
	}
	else = {
		# AI goes first next round. Mode-aware.
		if = {
			limit = { var:pod_combat_session ?= { has_variable = pod_combat_step_mode } }
			# STEP: take only the first AI turn this round, then return to the GUI.
			# advance_one_ai_turn re-runs find_next (idempotent re-select of the same
			# actor the preceding find_next already chose), so no extra pre-call here.
			pod_combat_advance_one_ai_turn_effect = yes
			pod_combat_after_pre_pc_step_settle_effect = yes
		}
		else = {
			# AUTO: walk pre-PC AI turns until the PC reaches its slot or the round
			# ends. run_pre_pc_ai_turns primes its own find_next (idempotent re-select
			# of the preceding find_next).
			pod_combat_run_pre_pc_ai_turns_effect = yes
		}
	}

	} # end of NOT pod_combat_is_over guard (round-end DoT victory)

	} # end else (combat not yet cleaned up)
}
