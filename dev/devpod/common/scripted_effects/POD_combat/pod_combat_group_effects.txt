##############################################################################################
# GROUP COMBAT INFRASTRUCTURE EFFECTS
# Handles slot-based workspace load/save, dispatch, initialization, assignment, and cleanup
# for the group combat system (two teams of arbitrary size; soft cap
# pod_combat_max_team_size).
#
# Roster layout (filled by the duel builder — create / add_ally / add_enemy).
# Every combatant carries (combat_team, team_position); position 0 = leader:
#   player / 0    = PC (player character, always present)
#   player / 1+   = Ally slots (seated by POD_combat_duel_add_ally_effect)
#   enemy  / 0    = Primary opponent (first enemy seated; present in any fight)
#   enemy  / 1+   = Enemy reinforcements (seated by POD_combat_duel_add_enemy_effect)
#
# Terminology:
#   actor / target   — workspace-relative. "actor" = whoever is ACTING
#                      this action; "target" = whoever is being TARGETED.
#                      Used by pod_combat_actor_* / pod_combat_target_*
#                      workspace mirrors, by recalc helper names, and
#                      as the literal $ACTOR_SIDE$ / $TARGET_SIDE$
#                      parametric values (so e.g.
#                      pod_combat_$TARGET_SIDE$_char resolves to
#                      pod_combat_target_char). The "actor/target"
#                      framing follows the action, not team position:
#                      on enemy turns the enemy's slot loads into the
#                      "actor" workspace, and the PC may load into
#                      "target" if the AI is hitting them.
#   player / enemy   — team membership, read off each slot's combat_team
#                      flag (flag:player / flag:enemy). The player team is
#                      the PC + ally positions; the enemy team is the
#                      primary opponent + reinforcements. Used for the
#                      active_slots team filters (var:combat_team =
#                      flag:enemy etc.), AoE filters, and victory-check
#                      team wipes. Teams are UNBOUNDED — slots are reached
#                      by iterating pod_combat_active_slots and matching
#                      (combat_team, team_position); the only pointer vars
#                      are the two leaders, pod_combat_pc_slot (player/0)
#                      and pod_combat_enemy_leader_slot (enemy/0).
##############################################################################################

##############################################################################################
# SECTION 1: WORKSPACE LOAD EFFECTS
# Two parametric effects that copy a given slot's state into the actor or target workspace.
# Called via dispatch effects keyed on the current-actor / selected-target host pointers (each a combat_team + team_position pair).
##############################################################################################

# Parametric workspace-load helper. Reads the current slot story's
# fields and writes them into the pod_combat_$SIDE$_* workspace mirror
# on scope:host. Called via the actor / target thin wrappers below.
#
# Caller contract: current scope is the slot story being loaded;
# scope:host must be the workspace owner (combat host). save_scope_as
# at the top stashes the slot story so the post-scope-switch
# `scope:host = { ... }` body can still reach the slot's fields via
# scope:slot_story.var:slot_X.
#
# Param: SIDE — "actor" or "target".
#
# If you add a new mirrored field, add ONE line here (and the matching
# writeback in pod_combat_save_side_to_story_helper). See
# COMBAT_REFERENCE.info §11 "Adding a new workspace mirror field" for
# the full checklist.
pod_combat_load_side_from_story_helper = {
	if = {
		limit = { has_variable = slot_char }
		save_scope_as = slot_story
		scope:host = {
			set_variable = { name = pod_combat_$SIDE$_soak value = scope:slot_story.var:slot_soak }
			set_variable = { name = pod_combat_$SIDE$_base_soak value = scope:slot_story.var:slot_base_soak }
			if = { limit = { scope:slot_story = { has_variable = slot_base_hit_chance } } set_variable = { name = pod_combat_$SIDE$_base_hit_chance value = scope:slot_story.var:slot_base_hit_chance } }
			set_variable = { name = pod_combat_$SIDE$_morale value = scope:slot_story.var:slot_morale }
			set_variable = { name = pod_combat_$SIDE$_prowess value = scope:slot_story.var:slot_prowess }
			set_variable = { name = pod_combat_$SIDE$_hit_chance value = scope:slot_story.var:slot_hit_chance }
			# slot_energy_used is default-init'd to 0 at story creation,
			# so always present — no has_variable gate needed. Mirrored
			# both sides for symmetry; actor-side reader is the AI energy-
			# attrition weighting, target-side is no-op overhead.
			set_variable = { name = pod_combat_$SIDE$_energy_used value = scope:slot_story.var:slot_energy_used }
			# Stance flags (presence-as-signal)
			if = {
				limit = { scope:slot_story = { has_variable = slot_is_guarding } }
				set_variable = { name = pod_combat_$SIDE$_is_guarding value = yes }
			}
			else = { remove_variable = pod_combat_$SIDE$_is_guarding }
			if = {
				limit = { scope:slot_story = { has_variable = slot_is_dodging } }
				set_variable = { name = pod_combat_$SIDE$_is_dodging value = yes }
			}
			else = { remove_variable = pod_combat_$SIDE$_is_dodging }
			if = {
				limit = { scope:slot_story = { has_variable = slot_is_stealthed } }
				set_variable = { name = pod_combat_$SIDE$_is_stealthed value = yes }
			}
			else = { remove_variable = pod_combat_$SIDE$_is_stealthed }
			# Spirit-damage capability + incorporeality flags. Both read
			# by pod_combat_resolve_contact_effect (target-side
			# incorporeal_spirit gates checks_spirit_gate moves;
			# can_damage_spirit lets actors land hits) and by
			# pod_combat_apply_retaliation_helper (actor/target sides
			# swap roles for the retaliator → original-attacker gate).
			if = {
				limit = { scope:slot_story = { has_variable = slot_can_damage_spirit } }
				set_variable = { name = pod_combat_$SIDE$_can_damage_spirit value = yes }
			}
			else = { remove_variable = pod_combat_$SIDE$_can_damage_spirit }
			if = {
				limit = { scope:slot_story = { has_variable = slot_incorporeal_spirit } }
				set_variable = { name = pod_combat_$SIDE$_incorporeal_spirit value = yes }
			}
			else = { remove_variable = pod_combat_$SIDE$_incorporeal_spirit }
			# Initiative bonus + AP drain (mirrored both sides for load/save
			# symmetry). POD_execute_combat_move §14 writes target-side via
			# workspace mirror; consumed at round-end (init_bonus) or
			# target's next turn start (ap_drain) via slot or
			# actor-side-after-load reads.
			if = {
				limit = { scope:slot_story = { has_variable = slot_init_bonus } }
				set_variable = { name = pod_combat_$SIDE$_init_bonus value = scope:slot_story.var:slot_init_bonus }
			}
			else = { remove_variable = pod_combat_$SIDE$_init_bonus }
			if = {
				limit = { scope:slot_story = { has_variable = slot_ap_drain } }
				set_variable = { name = pod_combat_$SIDE$_ap_drain value = scope:slot_story.var:slot_ap_drain }
			}
			else = { remove_variable = pod_combat_$SIDE$_ap_drain }
			# Retaliation stances (4 vars: hit / miss × value / aggravated flag)
			if = {
				limit = { scope:slot_story = { has_variable = slot_retaliate_hit } }
				set_variable = { name = pod_combat_$SIDE$_retaliate_hit value = scope:slot_story.var:slot_retaliate_hit }
			}
			else = { remove_variable = pod_combat_$SIDE$_retaliate_hit }
			if = {
				limit = { scope:slot_story = { has_variable = slot_retaliate_hit_agg } }
				set_variable = { name = pod_combat_$SIDE$_retaliate_hit_agg value = yes }
			}
			else = { remove_variable = pod_combat_$SIDE$_retaliate_hit_agg }
			if = {
				limit = { scope:slot_story = { has_variable = slot_retaliate_miss } }
				set_variable = { name = pod_combat_$SIDE$_retaliate_miss value = scope:slot_story.var:slot_retaliate_miss }
			}
			else = { remove_variable = pod_combat_$SIDE$_retaliate_miss }
			if = {
				limit = { scope:slot_story = { has_variable = slot_retaliate_miss_agg } }
				set_variable = { name = pod_combat_$SIDE$_retaliate_miss_agg value = yes }
			}
			else = { remove_variable = pod_combat_$SIDE$_retaliate_miss_agg }
			# Tier tracking vars (8 total: soak/hit × buff/debuff × round/combat).
			# Presence on slot = active tier; absence-as-signal preserved.
			if = {
				limit = { scope:slot_story = { has_variable = slot_soak_buff_round_tier } }
				set_variable = { name = pod_combat_$SIDE$_soak_buff_round_tier value = scope:slot_story.var:slot_soak_buff_round_tier }
			}
			else = { remove_variable = pod_combat_$SIDE$_soak_buff_round_tier }
			if = {
				limit = { scope:slot_story = { has_variable = slot_soak_buff_combat_tier } }
				set_variable = { name = pod_combat_$SIDE$_soak_buff_combat_tier value = scope:slot_story.var:slot_soak_buff_combat_tier }
			}
			else = { remove_variable = pod_combat_$SIDE$_soak_buff_combat_tier }
			if = {
				limit = { scope:slot_story = { has_variable = slot_soak_debuff_round_tier } }
				set_variable = { name = pod_combat_$SIDE$_soak_debuff_round_tier value = scope:slot_story.var:slot_soak_debuff_round_tier }
			}
			else = { remove_variable = pod_combat_$SIDE$_soak_debuff_round_tier }
			if = {
				limit = { scope:slot_story = { has_variable = slot_soak_debuff_combat_tier } }
				set_variable = { name = pod_combat_$SIDE$_soak_debuff_combat_tier value = scope:slot_story.var:slot_soak_debuff_combat_tier }
			}
			else = { remove_variable = pod_combat_$SIDE$_soak_debuff_combat_tier }
			if = {
				limit = { scope:slot_story = { has_variable = slot_hit_buff_round_tier } }
				set_variable = { name = pod_combat_$SIDE$_hit_buff_round_tier value = scope:slot_story.var:slot_hit_buff_round_tier }
			}
			else = { remove_variable = pod_combat_$SIDE$_hit_buff_round_tier }
			if = {
				limit = { scope:slot_story = { has_variable = slot_hit_buff_combat_tier } }
				set_variable = { name = pod_combat_$SIDE$_hit_buff_combat_tier value = scope:slot_story.var:slot_hit_buff_combat_tier }
			}
			else = { remove_variable = pod_combat_$SIDE$_hit_buff_combat_tier }
			if = {
				limit = { scope:slot_story = { has_variable = slot_hit_debuff_round_tier } }
				set_variable = { name = pod_combat_$SIDE$_hit_debuff_round_tier value = scope:slot_story.var:slot_hit_debuff_round_tier }
			}
			else = { remove_variable = pod_combat_$SIDE$_hit_debuff_round_tier }
			if = {
				limit = { scope:slot_story = { has_variable = slot_hit_debuff_combat_tier } }
				set_variable = { name = pod_combat_$SIDE$_hit_debuff_combat_tier value = scope:slot_story.var:slot_hit_debuff_combat_tier }
			}
			else = { remove_variable = pod_combat_$SIDE$_hit_debuff_combat_tier }
			# Character ref. Read by AI trait gates (actor-side) and the
			# damage pipeline as var:pod_combat_$TARGET_SIDE$_char in
			# retaliation / self-damage contexts.
			set_variable = { name = pod_combat_$SIDE$_char value = scope:slot_story.var:slot_char }
			# Team identity (combat_team / team_position). Immutable for the
			# whole combat, so mirrored LOAD-ONLY (no writeback in the save
			# helper). Lets "same team as the actor" checks read
			# pod_combat_actor_team instead of deriving a team from an index,
			# and "is the leader" checks read pod_combat_$SIDE$_position = 0.
			set_variable = { name = pod_combat_$SIDE$_team     value = scope:slot_story.var:combat_team }
			set_variable = { name = pod_combat_$SIDE$_position value = scope:slot_story.var:team_position }
		}
	}
}

# Thin wrapper — loads actor workspace from current slot story.
# Caller scope = slot story; scope:host must be saved before delegating.
# Used by pod_combat_load_workspace_effect's iteration over
# pod_combat_active_slots (finds the slot whose (combat_team, team_position)
# matches the current-actor host pointer), and by any future single-slot caller
# that needs to save scope:host and flip into a slot story before
# delegating.
pod_combat_load_actor_from_story_effect = {
	pod_combat_load_side_from_story_helper = { SIDE = actor }
}

# Thin wrapper — loads target workspace from current slot story.
# Caller scope = slot story; scope:host must be saved before delegating.
# Called from every_in_list iterations on pod_combat_active_slots in
# the AoE dispatchers and the workspace dispatch (where the iteration
# body's current scope IS the slot story), and from
# pod_combat_select_target / pod_combat_ai_swap_to_pc_target_effect for
# their single-slot host→story flips.
#
# Plus a target-specific pair-hit-chance recalc after the workspace
# fields are loaded. Slot-stored hit chance is leader-baseline (computed
# once at combat init in pod_combat_calculate_all_hit_chances_effect),
# so without this every AI swap, AoE per-target iteration, and
# workspace-load would push the wrong number into the damage roll. The
# recalc reads pod_combat_actor_prowess (set by the actor load that
# runs before this) and pod_combat_target_prowess (set by the helper
# above), then layers the active hit-tier modifiers. No slot writeback
pod_combat_load_target_from_story_effect = {
	pod_combat_load_side_from_story_helper = { SIDE = target }
	if = {
		limit = { has_variable = slot_char }
		scope:host = { pod_combat_recalc_pair_hit_chance_effect = yes }
	}
}

# SECTION 2: WORKSPACE SAVE EFFECTS
# Two effects that copy the actor or target workspace back into a slot story.
##############################################################################################

# Parametric workspace-save helper. Reads pod_combat_$SIDE$_* from
# scope:host and writes the slot story's slot_X fields. Includes the
# alive-flip (death / incapacitation / wound-track-full / morale-broken)
# for BOTH sides — symmetric since an actor can KO itself via
# self-damage or incoming retaliation just as a target can. Called via
# the actor / target thin wrappers below.
#
# Caller contract: current scope is the slot story being written;
# scope:host must be the workspace owner. The `has_variable =
# pod_combat_$SIDE$_char` gate at the top ensures we don't write to
# slots whose workspace mirrors weren't loaded.
#
# Param: SIDE — "actor" or "target".
#
# If you add a new mirrored field, add ONE writeback here (and the
# matching load in pod_combat_load_side_from_story_helper above). See
# COMBAT_REFERENCE.info §11 "Adding a new workspace mirror field" for
# the full checklist.
pod_combat_save_side_to_story_helper = {
	if = {
		limit = { scope:host = { has_variable = pod_combat_$SIDE$_char } }
		# Flush the post-action morale to the slot FIRST so the shared KO
		# trigger below reads it from the slot field (var:slot_morale), the
		# same way the post-flush actor/target KO checks do.
		set_variable = { name = slot_morale value = scope:host.var:pod_combat_$SIDE$_morale }
		# Alive-flip — flips slot_alive = 0 on any KO condition, for BOTH
		# sides, via the shared pod_combat_slot_should_be_marked_ko_trigger
		# (one definition, also used by the actor/target KO checks in
		# pod_combat_save_workspace_effect; the four KO sources are
		# documented on the trigger).
		if = {
			limit = { pod_combat_slot_should_be_marked_ko_trigger = yes }
			set_variable = { name = slot_alive value = 0 }
		}
		# Workspace -> slot writes. The slot story is current scope so
		# writes are direct `set_variable = { name = slot_X ... }`;
		# reads come from scope:host.var:pod_combat_$SIDE$_X.
		set_variable = { name = slot_soak           value = scope:host.var:pod_combat_$SIDE$_soak }
		set_variable = { name = slot_base_soak      value = scope:host.var:pod_combat_$SIDE$_base_soak }
		if = { limit = { scope:host = { has_variable = pod_combat_$SIDE$_base_hit_chance } } set_variable = { name = slot_base_hit_chance value = scope:host.var:pod_combat_$SIDE$_base_hit_chance } }
		# slot_morale is written at the top of this block (before the KO check).
		set_variable = { name = slot_prowess        value = scope:host.var:pod_combat_$SIDE$_prowess }
		set_variable = { name = slot_hit_chance     value = scope:host.var:pod_combat_$SIDE$_hit_chance }
		if = {
			limit = { scope:host = { has_variable = pod_combat_$SIDE$_energy_used } }
			set_variable = { name = slot_energy_used value = scope:host.var:pod_combat_$SIDE$_energy_used }
		}
		# Stance flags (presence-as-signal)
		if = {
			limit = { scope:host = { has_variable = pod_combat_$SIDE$_is_guarding } }
			set_variable    = { name = slot_is_guarding value = yes }
		}
		else = { remove_variable = slot_is_guarding }
		if = {
			limit = { scope:host = { has_variable = pod_combat_$SIDE$_is_dodging } }
			set_variable    = { name = slot_is_dodging value = yes }
		}
		else = { remove_variable = slot_is_dodging }
		if = {
			limit = { scope:host = { has_variable = pod_combat_$SIDE$_is_stealthed } }
			set_variable    = { name = slot_is_stealthed value = yes }
		}
		else = { remove_variable = slot_is_stealthed }
		# Initiative bonus writeback. Round-end initiative recalc reads
		# slot_init_bonus directly outside any workspace lifecycle.
		if = {
			limit = { scope:host = { has_variable = pod_combat_$SIDE$_init_bonus } }
			set_variable    = { name = slot_init_bonus value = scope:host.var:pod_combat_$SIDE$_init_bonus }
		}
		else = { remove_variable = slot_init_bonus }
		# AP drain writeback. Consumed at the affected combatant's next
		# turn start by pod_combat_apply_ap_drain_effect.
		if = {
			limit = { scope:host = { has_variable = pod_combat_$SIDE$_ap_drain } }
			set_variable    = { name = slot_ap_drain   value = scope:host.var:pod_combat_$SIDE$_ap_drain }
		}
		else = { remove_variable = slot_ap_drain }
		# Retaliation stances (4 vars: hit / miss × value / aggravated flag)
		if = {
			limit = { scope:host = { has_variable = pod_combat_$SIDE$_retaliate_hit } }
			set_variable    = { name = slot_retaliate_hit value = scope:host.var:pod_combat_$SIDE$_retaliate_hit }
		}
		else = { remove_variable = slot_retaliate_hit }
		if = {
			limit = { scope:host = { has_variable = pod_combat_$SIDE$_retaliate_hit_agg } }
			set_variable    = { name = slot_retaliate_hit_agg value = yes }
		}
		else = { remove_variable = slot_retaliate_hit_agg }
		if = {
			limit = { scope:host = { has_variable = pod_combat_$SIDE$_retaliate_miss } }
			set_variable    = { name = slot_retaliate_miss value = scope:host.var:pod_combat_$SIDE$_retaliate_miss }
		}
		else = { remove_variable = slot_retaliate_miss }
		if = {
			limit = { scope:host = { has_variable = pod_combat_$SIDE$_retaliate_miss_agg } }
			set_variable    = { name = slot_retaliate_miss_agg value = yes }
		}
		else = { remove_variable = slot_retaliate_miss_agg }
		# Tier tracking vars (8 total: soak/hit × buff/debuff × round/combat).
		# Presence on workspace = active tier on slot; absence-as-signal preserved.
		if = {
			limit = { scope:host = { has_variable = pod_combat_$SIDE$_soak_buff_round_tier } }
			set_variable    = { name = slot_soak_buff_round_tier    value = scope:host.var:pod_combat_$SIDE$_soak_buff_round_tier }
		}
		else = { remove_variable = slot_soak_buff_round_tier    }
		if = {
			limit = { scope:host = { has_variable = pod_combat_$SIDE$_soak_buff_combat_tier } }
			set_variable    = { name = slot_soak_buff_combat_tier   value = scope:host.var:pod_combat_$SIDE$_soak_buff_combat_tier }
		}
		else = { remove_variable = slot_soak_buff_combat_tier   }
		if = {
			limit = { scope:host = { has_variable = pod_combat_$SIDE$_soak_debuff_round_tier } }
			set_variable    = { name = slot_soak_debuff_round_tier  value = scope:host.var:pod_combat_$SIDE$_soak_debuff_round_tier }
		}
		else = { remove_variable = slot_soak_debuff_round_tier  }
		if = {
			limit = { scope:host = { has_variable = pod_combat_$SIDE$_soak_debuff_combat_tier } }
			set_variable    = { name = slot_soak_debuff_combat_tier value = scope:host.var:pod_combat_$SIDE$_soak_debuff_combat_tier }
		}
		else = { remove_variable = slot_soak_debuff_combat_tier }
		if = {
			limit = { scope:host = { has_variable = pod_combat_$SIDE$_hit_buff_round_tier } }
			set_variable    = { name = slot_hit_buff_round_tier     value = scope:host.var:pod_combat_$SIDE$_hit_buff_round_tier }
		}
		else = { remove_variable = slot_hit_buff_round_tier     }
		if = {
			limit = { scope:host = { has_variable = pod_combat_$SIDE$_hit_buff_combat_tier } }
			set_variable    = { name = slot_hit_buff_combat_tier    value = scope:host.var:pod_combat_$SIDE$_hit_buff_combat_tier }
		}
		else = { remove_variable = slot_hit_buff_combat_tier    }
		if = {
			limit = { scope:host = { has_variable = pod_combat_$SIDE$_hit_debuff_round_tier } }
			set_variable    = { name = slot_hit_debuff_round_tier   value = scope:host.var:pod_combat_$SIDE$_hit_debuff_round_tier }
		}
		else = { remove_variable = slot_hit_debuff_round_tier   }
		if = {
			limit = { scope:host = { has_variable = pod_combat_$SIDE$_hit_debuff_combat_tier } }
			set_variable    = { name = slot_hit_debuff_combat_tier  value = scope:host.var:pod_combat_$SIDE$_hit_debuff_combat_tier }
		}
		else = { remove_variable = slot_hit_debuff_combat_tier  }
	}
}

# Thin wrapper — saves actor workspace back to current slot story.
# Caller scope = slot story; scope:host must be saved before delegating.
# Called from pod_combat_save_workspace_effect's iteration over
# pod_combat_active_slots.
pod_combat_save_actor_to_story_effect = {
	pod_combat_save_side_to_story_helper = { SIDE = actor }
}

# ─── AOE TARGET-ALIVE CHECK ───────────────────────────────────────
# Post-iteration alive-flip used by the damage-side AoE dispatchers
# in pod_combat_move_effects.txt. Runs after pod_combat_save_target_to_story_effect
# has written the slot's freshly-resolved combat state, so this can
# read both the character's PoD wound state AND the just-written
# slot_morale value.
#
# Flips slot_alive = 0 when ANY of:
#   * The character died (is_alive = no — fatal duel + lethal blow).
#   * The character is in an incapacitated states.
#   * Combat morale broke (slot_morale <= 0). This is the only check
#     pod_combat_save_target_to_story_effect's internal alive-check
#     can't make pre-write — slot_morale isn't set until that effect
#     writes the workspace value to the slot.
#   * The wound track is full (any combination of superficial +
#     aggravated reaching track length). Catches combat-HP exhaustion
#     in non-fatal duels where PoD's death branch is gated and is_alive
#     stays yes.
#
# Called from per-target body inside the AoE iterators in
# pod_combat_move_effects.txt. Implicit scope must be the iterated
# slot story (var:slot_morale + var:slot_char live there).
#
# Centralised here so a future incapacitation clause only needs to be
# added in one site — the AoE iterators in pod_combat_move_effects.txt
# both call this rather than each inlining the alive checks.
pod_combat_aoe_check_target_alive_effect = {
	if = {
		limit = {
			pod_combat_slot_should_be_marked_ko_trigger = yes
		}
		set_variable = { name = slot_alive value = 0 }
	}
}


# Thin wrapper — saves target workspace back to current slot story.
# Caller scope = slot story; scope:host must be saved before delegating.
# Called from every_in_list iterations on pod_combat_active_slots in
# the AoE dispatchers and the workspace dispatch
# (pod_combat_save_workspace_effect), and from the AI swap-target
# effects' restore path (pod_combat_ai_restore_enemy_target_effect).
# The full alive-flip (incl. wound-track-full + morale-broken) lives in
# the shared helper now, applied to both sides.
pod_combat_save_target_to_story_effect = {
	pod_combat_save_side_to_story_helper = { SIDE = target }
}

##############################################################################################
# SECTION 3: WORKSPACE DISPATCH EFFECTS
# Match the current-actor / selected-target host pointers (each a
# combat_team + team_position pair) to call the correct parametric
# load/save effect for the appropriate slot.
##############################################################################################

pod_combat_load_workspace_effect = {
	# Load current actor → actor workspace, then selected target →
	# target workspace, by iterating pod_combat_active_slots and
	# matching each slot's (combat_team, team_position) against the
	# current-actor / selected-target host pointers. The iteration limit fires for at
	# most one slot per pass — the load helper inside the body is the
	# story-scoped variant, which writes to scope:host (this scope).
	save_scope_as = host
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			scope:host = { has_variable = pod_combat_current_actor_team }
			var:combat_team = scope:host.var:pod_combat_current_actor_team
			var:team_position = scope:host.var:pod_combat_current_actor_position
		}
		pod_combat_load_actor_from_story_effect = yes
	}
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			scope:host = { has_variable = pod_combat_selected_target_team }
			var:combat_team = scope:host.var:pod_combat_selected_target_team
			var:team_position = scope:host.var:pod_combat_selected_target_position
		}
		pod_combat_load_target_from_story_effect = yes
	}

	# Push the freshly pair-recomputed hit chances back to the actor's
	# and target's slot stories so the per-slot GUI display matches the
	# values that will actually drive the upcoming damage roll.
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			scope:host = { has_variable = pod_combat_current_actor_team }
			var:combat_team = scope:host.var:pod_combat_current_actor_team
			var:team_position = scope:host.var:pod_combat_current_actor_position
		}
		if = {
			limit = { scope:host = { has_variable = pod_combat_actor_base_hit_chance } }
			set_variable = { name = slot_base_hit_chance value = scope:host.var:pod_combat_actor_base_hit_chance }
		}
		if = {
			limit = { scope:host = { has_variable = pod_combat_actor_hit_chance } }
			set_variable = { name = slot_hit_chance value = scope:host.var:pod_combat_actor_hit_chance }
		}
	}
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			scope:host = { has_variable = pod_combat_selected_target_team }
			var:combat_team = scope:host.var:pod_combat_selected_target_team
			var:team_position = scope:host.var:pod_combat_selected_target_position
		}
		if = {
			limit = { scope:host = { has_variable = pod_combat_target_base_hit_chance } }
			set_variable = { name = slot_base_hit_chance value = scope:host.var:pod_combat_target_base_hit_chance }
		}
		if = {
			limit = { scope:host = { has_variable = pod_combat_target_hit_chance } }
			set_variable = { name = slot_hit_chance value = scope:host.var:pod_combat_target_hit_chance }
		}
	}

	# Set the current_hit_chance for the damage pipeline
	if = {
		limit = { has_variable = pod_combat_actor_hit_chance }
		set_variable = { name = pod_combat_current_hit_chance value = var:pod_combat_actor_hit_chance }
	}
}

# Stamp pod_combat_nonfatal_active on every active combatant's char
# so POD_can_die_from_wounds returns no while combat is active. Paired
# with pod_combat_clear_nonfatal_flag_effect (combat-end strip).
pod_combat_apply_nonfatal_flag_effect = {

		variable = pod_combat_active_slots
		var:slot_char = {
			add_character_flag = pod_combat_nonfatal_active
		}
	}
}

# Strip pod_combat_nonfatal_active after the post-combat wound cap
# has guaranteed the char has at least one empty damage box.
pod_combat_clear_nonfatal_flag_effect = {
	every_in_list = {
		variable = pod_combat_active_slots
		var:slot_char = {
			remove_character_flag = pod_combat_nonfatal_active
		}
	}
}

pod_combat_save_workspace_effect = {
	# Mirror image of pod_combat_load_workspace_effect: iterate
	# pod_combat_active_slots and write the actor / target workspace
	# back to whichever slot's (combat_team, team_position) matches the
	# current-actor / selected-target host pointers.
	save_scope_as = host
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			scope:host = { has_variable = pod_combat_current_actor_team }
			var:combat_team = scope:host.var:pod_combat_current_actor_team
			var:team_position = scope:host.var:pod_combat_current_actor_position
		}
		pod_combat_save_actor_to_story_effect = yes
	}
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			scope:host = { has_variable = pod_combat_selected_target_team }
			var:combat_team = scope:host.var:pod_combat_selected_target_team
			var:team_position = scope:host.var:pod_combat_selected_target_position
		}
		pod_combat_save_target_to_story_effect = yes
	}

	# KO check on the action's target — flips slot_alive=0 via the shared
	# pod_combat_slot_should_be_marked_ko_trigger (four KO sources
	# documented there). Iterates pod_combat_active_slots, matching only
	# the slot whose (combat_team, team_position) equals the selected-target pointer.
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			scope:host = { has_variable = pod_combat_selected_target_team }
			var:combat_team = scope:host.var:pod_combat_selected_target_team
			var:team_position = scope:host.var:pod_combat_selected_target_position
			pod_combat_slot_should_be_marked_ko_trigger = yes
		}
		set_variable = { name = slot_alive value = 0 }
	}

	# KO check on the actor (counter-damage during their own action
	# a retaliating target's morale damage / chip damage can finish them
	# off). Same KO predicates as above; iterates the same list.
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			scope:host = { has_variable = pod_combat_current_actor_team }
			var:combat_team = scope:host.var:pod_combat_current_actor_team
			var:team_position = scope:host.var:pod_combat_current_actor_position
			pod_combat_slot_should_be_marked_ko_trigger = yes
		}
		set_variable = { name = slot_alive value = 0 }
	}

	# Victory check at the post-flush boundary. Slot _alive flags and
	# morale are fresh here (the workspace just flushed back to slot
	# vars), so team-wipe detection works. This is one of two call
	# sites — the other is at the end of pod_combat_execute_ai_turn_effect
	# (covering AI action boundaries). Per-damage victory checks are
	# NOT used: an AoE move would fire the check after each victim's
	# damage call, and a multi-victim wipe would emit one cleanup per
	# kill. Checking only at action boundaries keeps cleanup atomic.
	pod_combat_check_group_victory_effect = yes
}

##############################################################################################
# SECTION 4: SLOT INITIALIZATION FROM CHARACTER
# Reads a character's health track, soak, and prowess and stores them in slot variables.
##############################################################################################

pod_combat_init_slot_from_character_effect = {
	# Runs in the SLOT-STORY scope (called once per slot from the
	# pod_combat_init_all_slots_effect loop). Params:
	#   $CHARACTER$    = the slot's occupant (the loop passes var:slot_char).
	#   $ENEMY_LEADER$ = the OPPOSING team's leader — the CASTER for the
	#     "enemy knight" discipline buff: scope:pod_combat_target for
	#     player-team slots, scope:combat_pc for enemy-team slots. Supplied
	#     by the caller so this effect needs no team branch.
	# Applies the discipline combat buffs and reads health/soak/prowess into
	# the current slot's variables (the scope-based accessors write current
	# scope).
	#
	# Shapeshifting for AI fera is NOT done here. It runs upfront in
	# POD_combat_finalize_duel_effect (the active-slots shapeshift pass)
	# before this dispatcher, so every potential CASTER already has its
	# crinos modifier when the discipline effects query it (especially for
	# the delirium check, which reads has_character_modifier on the CASTER).
	$CHARACTER$ = {
		# Discipline combat buffs. Uses named scopes (not root) to match
		# PoD's own duel pattern and avoid text-substitution issues in
		# nested parametric effects. The opposing-team leader (the "enemy
		# knight" CASTER) is supplied as $ENEMY_LEADER$ by the caller.
		POD_enemy_knight_set_discipline_battle_effect = { CASTER = $ENEMY_LEADER$ }
		POD_ally_knight_set_discipline_battle_effect = { CASTER = $CHARACTER$ }
		force_character_skill_recalculation = yes
	}
	# Read soak and prowess from the character and snapshot them as
	# saved scopes so the parametric pod_combat_set_slot_var_effect
	# calls below can pass them as VALUE without re-resolving against
	# the slot story scope. HP / wound state isn't snapshotted
	# pod_combat_slot_max_hp_value / _total_damage_sv / _agg_damage_sv
	# read live from the participant's PoD wound vars at GUI render
	# time.
	$CHARACTER$ = {
		save_scope_value_as = {
			name = slot_soak
			value = POD_soak_chance
		}
		save_scope_value_as = {
			name = slot_prowess
			value = prowess
		}
	}

	pod_combat_set_slot_var_effect = { NAME = soak      VALUE = scope:slot_soak }
	pod_combat_set_slot_var_effect = { NAME = base_soak VALUE = scope:slot_soak }
	pod_combat_set_slot_var_effect = { NAME = morale    VALUE = pod_combat_max_morale }
	pod_combat_set_slot_var_effect = { NAME = prowess   VALUE = scope:slot_prowess }
	pod_combat_set_slot_alive_effect = { VALUE = 1 }
	# AI energy-use counter: increments on every energy-cost skill this
	# combatant fires. The AI move picker layers stacking ADDITIVE weight
	# penalties on top of this counter (-8 / -10 / -12 / -14 at thresholds
	# 1 / 2 / 3 / 4+, cumulating to 0 / -8 / -18 / -30 / -44) to bias
	# selection AWAY from energy spends as the pool depletes. The counter
	# only drives selection bias; the AI still pays the actual splat
	# energy on top, via the same POD_spend_splat_agnostic_energy path
	# the PC uses.
	pod_combat_set_slot_var_effect = { NAME = energy_used VALUE = 0 }
	# ── Spirit / incorporeality capability flags ─────────────────
	# Mirrors PoD's POD_damage_spirit_custom_trigger semantics for the
	# minigame. Two per-slot flags are captured once at init and never
	# change during combat:
	#   incorporeal_spirit: target is an incorporeal spirit (ghost,
	#       Umbral entity, non-Risen wraith, etc.) that can't be harmed
	#       by mundane physical attacks. Checked via PoD's own
	#       POD_can_be_killed_physically_trigger = no.
	#   can_damage_spirit: actor has the supernatural means to strike
	#       incorporeal spirits (fera/spirit/fae/wraith/true_faith, or
	#       specific perks / artifacts). Checked via PoD's own
	#       POD_can_damage_spirit_trigger = yes.
	# These gate the spirit-capability check inside
	# pod_combat_resolve_contact_effect. The check fires when the move's
	# checks_spirit_gate capability flag is set (default-on for physical
	# attack_type, opt-in for supernatural moves that should respect
	# incorporeality). If the target is incorporeal and the actor lacks
	# pod_combat_$ACTOR_SIDE$_can_damage_spirit, the attack auto-misses.
	# Moves without checks_spirit_gate hit incorporeal targets normally.
	pod_combat_remove_slot_var_effect = { NAME = incorporeal_spirit }
	if = {
		limit = { $CHARACTER$ = { POD_can_be_killed_physically_trigger = no } }
		pod_combat_set_slot_var_effect = { NAME = incorporeal_spirit VALUE = yes }
	}
	pod_combat_remove_slot_var_effect = { NAME = can_damage_spirit }
	if = {
		limit = { $CHARACTER$ = { POD_can_damage_spirit_trigger = yes } }
		pod_combat_set_slot_var_effect = { NAME = can_damage_spirit VALUE = yes }
	}
	pod_combat_remove_slot_var_effect = { NAME = is_guarding }
	pod_combat_remove_slot_var_effect = { NAME = is_dodging }
	# ── AI per-fight transient state ─────────────────────────────
	# AI loadouts and anti-repetition history (last / prev / prev2
	# move) are per-fight scratch state. Clear them on slot init so
	# every fight seeds from the current move catalogue and starts
	# with empty history. PC slot (slot 0) is skipped. The player's
	# loadout is persistent equipment configured via the loadout
	# decision, and the PC has no AI history vars to clear.
	if = {
		limit = { $CHARACTER$ = { is_ai = yes } }
		$CHARACTER$ = {
			clear_variable_list = pod_combat_loadout
			remove_variable = pod_combat_ai_last_move
			remove_variable = pod_combat_ai_prev_move
			remove_variable = pod_combat_ai_prev2_move
		}
	}
	# ── State marker modifier ────────────────────────────────────
	# Apply pod_combat_duel_active to every combatant so other
	# systems (feeding, schemes, travel, activity events from
	# other mods) can gate on `has_character_modifier` and defer
	# firing mid-duel. Normal cleanup happens inside
	# pod_combat_duel_end_effect (scripted effect called from
	# pod_combat_end_combat_effect); `days = 30` is a safety net
	# for edge cases (crash, save corruption, mod conflict blocking
	# the inline cleanup). Combat minigames end in seconds of game
	# time, so 30 days never clips a real duel and keeps stale state
	# bounded if anything goes wrong with cleanup.
	$CHARACTER$ = {
		add_character_modifier = {
			modifier = pod_combat_duel_active
			days     = 30
		}
	}
	# Hit chance calculated later after all combatants are initialized
}

##############################################################################################
# SECTION 5: DUEL BUILDER — create / add combatant
#
# Callers build a fight imperatively:
#   <PC scope> = {
#       POD_combat_create_duel_effect    = { INITIATOR=.. ATTACKER=.. DEFENDER=.. FATALITY=.. FIXED=.. LOCALE=.. OUTPUT_EVENT=.. }
#       POD_combat_duel_add_ally_effect  = { CHARACTER = <ally>  }   # 0+ times
#       POD_combat_duel_add_enemy_effect = { CHARACTER = <enemy> }   # 1+ times (first → enemy/0 leader)
#       POD_combat_finalize_duel_effect  = yes
#   }
#
# Slot layout: PC = player/0, allies = player/1+, primary opponent = enemy/0,
# reinforcements = enemy/1+ — team size bounded only by the soft cap
# pod_combat_max_team_size. The PC is always the combat host; every builder
# effect runs in the PC's scope.
#
# Why slot stat-init is DEFERRED to finalize (add_* only records
# occupancy): pod_combat_init_slot_from_character_effect applies each
# combatant's discipline battle buff using the OPPOSING team leader as
# CASTER (player-team combatants read scope:pod_combat_target = the
# enemy/0 leader; enemy-team combatants read scope:combat_pc). The opposing leader isn't
# known until the full roster is placed, so add_* defers init and
# finalize runs it once (pod_combat_init_all_slots_effect) after deriving
# the leader scopes.
##############################################################################################

POD_combat_create_duel_effect = {
	# Scaffolds host state, creates the duel story (the caller-context
	# handoff), and seats the PC in slot 0. Runs in the PC's scope.
	#
	# Params (all forwarded onto the duel story for pod_combat.9999):
	#   INITIATOR / ATTACKER / DEFENDER — the original duel parties in
	#       PoD's sc_* sense (NOT necessarily the PC / enemy leader — the PC may
	#       be the defender). pod_combat.9999 restores these so win/loss
	#       attribution and scope:sc_victor logic stay correct.
	#   FATALITY / FIXED / LOCALE — duel rule flags.
	#   OUTPUT_EVENT — follow-up event fired on INITIATOR at combat end.
	save_scope_as = combat_pc

	# Save-compat version check — mirrors the pod_combat_configure_skills
	# decision's version reload, so a player who loads an older save and
	# enters combat WITHOUT opening the configure window first doesn't run
	# with stale move-story references. POD_reload_combat_moves rebuilds
	# the global catalogue and wipes loadouts; the empty-loadout fallback
	# below reseeds the PC with starter moves.
	hidden_effect = {
		if = {
			limit = {
				OR = {
					NOT = { exists = global_var:POD_combat_version }
					global_var:POD_combat_version != POD_current_combat_version
				}
			}
			POD_reload_combat_moves = yes
		}
	}
	if = {
		limit = { pod_combat_no_moves_equipped_trigger = yes }
		POD_apply_default_combat_loadout = yes
	}

	# Close the config window if open.
	remove_variable = pod_combat_config_open

	# Defensive teardown of leftover combat state (a prior fight that
	# crashed / was force-quit / regressed cleanup would leave
	# pod_combat_active set with orphaned slot stories + stale workspace
	# aliases). pod_combat_cleanup_effect is the same teardown used at the
	# normal end of combat; running it here is idempotent when nothing's
	# leftover. pod_combat_active is NOT set by create — it's set at
	# finalize, so this check only fires for genuinely stale state.
	if = {
		limit = { has_variable = pod_combat_active }
		pod_combat_cleanup_effect = yes
	}
	remove_variable = pod_combat_is_over

	set_variable = { name = pod_combat_round value = 1 }
	# Team-size counters track the next free team_position for each side.
	# The PC already occupies player/0 (seated below), so the player team
	# starts at size 1; the enemy team starts empty and its first add_enemy
	# seats the primary opponent (the enemy leader) at enemy/0.
	set_variable = { name = pod_combat_player_team_size value = 1 }
	set_variable = { name = pod_combat_enemy_team_size  value = 0 }

	# ── The duel story: caller-context handoff, saved as a scope ──
	# Anchored on Charon (immortal, off-map) so it survives the PC dying
	# mid-combat. The PC keeps one pointer var (pod_combat_sc_context);
	# cleanup is a single end_story. create_story runs in the normal path
	# so its save_scope_as binds; the var writes go inside
	# hidden_effect_new_object so tooltip-build (which skips not-yet-
	# existing objects) doesn't error on the scope reads. See
	# COMBAT_REFERENCE §7 for the new-object-wrap rationale.
	character:POD_wraith_01 = {
		create_story = {
			type = POD_combat_sc_context
			save_scope_as = pod_combat_duel
		}
	}

	hidden_effect_new_object = {
		# Dual-name the story as sc_ctx so pod_combat_sc_stash_parent_aliases_effect
		# (which writes to scope:sc_ctx) keeps working unchanged. This lives
		# INSIDE the new-object block: tooltip-build skips create_story, so a
		# scope:pod_combat_duel reference in the normal path would error with
		# "Undefined event target 'pod_combat_duel'".
		scope:pod_combat_duel = { save_scope_as = sc_ctx }

		scope:pod_combat_duel = {
			set_variable = { name = initiator value = $INITIATOR$ }
			set_variable = { name = attacker  value = $ATTACKER$ }
			set_variable = { name = defender  value = $DEFENDER$ }
			set_variable = { name = fatality  value = flag:$FATALITY$ }
			set_variable = { name = fixed     value = flag:$FIXED$ }
			set_variable = { name = locale    value = flag:$LOCALE$ }
			POD_save_event_as_var = { VAR = follow_up ID = $OUTPUT_EVENT$ }
		}

		# Snapshot any parent-event aliases (scope:spirit, scope:scheme,
		# scope:vassal_claim, …) onto the duel story for post-combat scope
		# restore in pod_combat.9999. Each write is guarded by ?= safe
		# navigation, so this is a silent no-op for callers (e.g. the
		# debug interactions) with no such aliases in scope.
		pod_combat_sc_stash_parent_aliases_effect = yes

		# Pin the duel story on the PC for cleanup access.
		scope:combat_pc = {
			set_variable = { name = pod_combat_sc_context value = scope:pod_combat_duel }
		}
	}

	# ── PC → player/0 (occupancy only; stat init deferred to finalize) ──
	pod_combat_set_slot_char_effect = { TEAM = player POSITION = 0 CHARACTER = scope:combat_pc }
}

POD_combat_duel_add_ally_effect = {
	# $CHARACTER$ = character to seat as the next ally on the player team.
	# Appends at player/<current pod_combat_player_team_size> and bumps the
	# counter, so the team grows without bound. Occupancy only — the stat /
	# discipline init runs once at finalize (pod_combat_init_all_slots_effect)
	# after the full roster is placed. Runs in the PC (host) scope; the PC
	# already holds player/0, so the first ally lands at player/1.
	#
	# pod_combat_max_team_size is a runaway guardrail, not a design cap; an
	# add past it is silently dropped (see the constant in pod_combat_values).
	#
	# The outer has_variable gate keeps this build-safe: during the duel
	# interaction's confirm-tooltip preview, create_duel's set_variable is not
	# applied, so pod_combat_player_team_size is unset — and a bare var: read
	# in a comparison errors during tooltip-build (has_variable does not
	# short-circuit a sibling comparison in the same limit, so the read must
	# live in a nested if). At real execution the counter is always set.
	if = {
		limit = { has_variable = pod_combat_player_team_size }
		if = {
			limit = { var:pod_combat_player_team_size < pod_combat_max_team_size }
			pod_combat_set_slot_char_effect = { TEAM = player POSITION = var:pod_combat_player_team_size CHARACTER = $CHARACTER$ }
			change_variable = { name = pod_combat_player_team_size add = 1 }
		}
	}
}

POD_combat_duel_add_enemy_effect = {
	# $CHARACTER$ = character to seat as the next enemy. Appends at
	# enemy/<current pod_combat_enemy_team_size> and bumps the counter. The
	# FIRST enemy added lands at enemy/0 — the primary opponent / enemy
	# leader, which is the player team's hit-chance baseline and the PC's
	# default target. Occupancy only; finalize runs the stat init.
	#
	# pod_combat_max_team_size is a runaway guardrail, not a design cap. The
	# outer has_variable gate keeps the cap comparison build-safe (see
	# POD_combat_duel_add_ally_effect for the tooltip-build rationale).
	if = {
		limit = { has_variable = pod_combat_enemy_team_size }
		if = {
			limit = { var:pod_combat_enemy_team_size < pod_combat_max_team_size }
			pod_combat_set_slot_char_effect = { TEAM = enemy POSITION = var:pod_combat_enemy_team_size CHARACTER = $CHARACTER$ }
			change_variable = { name = pod_combat_enemy_team_size add = 1 }
		}
	}
}

##############################################################################################
# SECTION 6: SLOT-INIT DISPATCHER + DUEL FINALIZE
#
# pod_combat_init_all_slots_effect runs the per-slot stat / discipline init
# for every occupied slot. $SLOT$ is a parse-time macro (it builds
# var:pod_combat_slot_$SLOT$), so it can't be a runtime loop variable —
# hence 10 guarded literal calls, mirroring pod_combat_calculate_all_hit_chances_effect.
#
# POD_combat_finalize_duel_effect completes a duel built via
# create_duel + add_ally/add_enemy: derives the leader scopes, shapeshifts
# AI fera, inits every slot, computes hit chances, runs the init-time KO /
# victory check, opens the combat window (pod_combat_active = yes), and
# starts round 1.
##############################################################################################

pod_combat_init_all_slots_effect = {
	# Initialise every combatant by iterating pod_combat_active_slots. The
	# caller (finalize) must have saved scope:pod_combat_target (the enemy
	# leader) and scope:combat_pc (the PC) — passed here as the discipline-buff
	# CASTER ($ENEMY_LEADER$): player-team slots buff against the enemy leader,
	# enemy-team slots against the PC.
	#
	# init_slot_from_character runs in each slot's own scope (the iteration
	# body) and writes the slot's vars directly. CHARACTER must still be a
	# SAVED SCOPE, not var:slot_char: init does `$CHARACTER$ = { ... CASTER =
	# $CHARACTER$ }`, which switches into the character's scope and re-evaluates
	# $CHARACTER$ there — a var:slot_char chain would re-resolve against the
	# occupant (no such var) and throw, while a saved scope resolves globally.
	# So snapshot the occupant to scope:init_slot_char first (overwritten each
	# iteration — the loop is sequential) before delegating.
	every_in_list = {
		variable = pod_combat_active_slots
		var:slot_char = { save_scope_as = init_slot_char }
		if = {
			limit = { var:combat_team = flag:player }
			pod_combat_init_slot_from_character_effect = { CHARACTER = scope:init_slot_char ENEMY_LEADER = scope:pod_combat_target }
		}
		else = {
			pod_combat_init_slot_from_character_effect = { CHARACTER = scope:init_slot_char ENEMY_LEADER = scope:combat_pc }
		}
	}
}

POD_combat_finalize_duel_effect = {
	# Completes a duel built via create_duel + add_ally/add_enemy. Runs in
	# the PC (host) scope. The combat window opens (pod_combat_active) only
	# after this whole effect returns — it's one atomic tick, so the GUI
	# never sees a half-initialized roster.

	save_scope_as = combat_pc

	# Derive the two leader pointers by scanning pod_combat_active_slots for
	# each team's position-0 slot. These are the ONLY per-slot pointer vars;
	# every other slot is reached by iterating the active list. There is
	# exactly one match per team — the PC for player, the primary opponent
	# for enemy.
	every_in_list = {
		variable = pod_combat_active_slots
		limit = { var:combat_team = flag:player var:team_position = 0 }
		save_scope_as = pc_leader_story
		scope:combat_pc = { set_variable = { name = pod_combat_pc_slot value = scope:pc_leader_story } }
	}
	every_in_list = {
		variable = pod_combat_active_slots
		limit = { var:combat_team = flag:enemy var:team_position = 0 }
		save_scope_as = enemy_leader_story
		scope:combat_pc = { set_variable = { name = pod_combat_enemy_leader_slot value = scope:enemy_leader_story } }
	}

	# scope:combat_pc = the PC (root). scope:pod_combat_target = the enemy
	# leader's occupant (enemy/0, the primary opponent) — the player team's
	# discipline-init CASTER and the post-combat dispatch target.
	if = {
		limit = {
			has_variable = pod_combat_enemy_leader_slot
			var:pod_combat_enemy_leader_slot = { has_variable = slot_char }
		}
		var:pod_combat_enemy_leader_slot.var:slot_char = { save_scope_as = pod_combat_target }
	}

	set_variable = { name = pod_combat_active value = yes }

	# -- Pre-pass: shapeshift every AI fera combatant BEFORE slot init --
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			var:slot_char = {
				is_ai = yes
				has_trait = fera
				POD_can_shapeshift_trigger = yes
				NOT = { POD_is_shapeshifted_warform_trigger = yes }
			}
		}
		var:slot_char = { POD_shapeshift_fera_warform_effect = yes }
	}

	# -- Per-slot stat / discipline init for every occupied slot --
	# Deferred from the add_* builders to here so each combatant's
	# discipline buff reads the final opposing-team leader (scope:combat_pc
	# / scope:pod_combat_target, both saved above) rather than a partial
	# roster.
	pod_combat_init_all_slots_effect = yes

	# -- Calculate hit chances for all combatants --
	# Formula: base 60 + (own prowess - primary opposing leader's prowess) / 2
	# "Primary opposing leader" = the enemy/0 leader for player-team combatants,
	# and player/0 (the PC) for enemy-team combatants. NOT an average across the enemy
	# team — pair-specific recalculation kicks in later when an actor/target pair is
	# loaded into the workspace via pod_combat_recalc_pc_hit_chance_effect, which
	# re-derives hit_chance against the *currently selected* target instead of the leader.
	pod_combat_calculate_all_hit_chances_effect = yes

	# -- Set up initial target / workspace defaults --
	# Seed the TARGET_SIDE-addressable alias (pod_combat_target_char) so
	# the damage helper's var:pod_combat_$TARGET_SIDE$_char lookup
	# resolves correctly from the moment combat initializes, even before
	# the first target_load effect runs.
	if = {
		limit = { has_variable = pod_combat_enemy_leader_slot var:pod_combat_enemy_leader_slot = { has_variable = slot_char } }
		set_variable = { name = pod_combat_target_char value = var:pod_combat_enemy_leader_slot.var:slot_char }
	}
	set_variable = { name = pod_combat_selected_target_team     value = flag:enemy }
	set_variable = { name = pod_combat_selected_target_position value = 0 }
	# Initialize the unified PC target: defaults to the primary opponent
	# (enemy/0). The player can switch targets mid-combat via the Target
	# button on any combat card (allies or other enemies). The (team,
	# position) pair covers both team-enemy and single-ally targeting; the
	# can_use triggers enforce that each skill fires against a valid slot.
	set_variable = { name = pod_combat_pc_target_team     value = flag:enemy }
	set_variable = { name = pod_combat_pc_target_position value = 0 }
	# Mirror the target's character into pod_combat_pc_target_char for the
	# top-row "Target: X" display, so it reads one var instead of resolving
	# the target slot at render time. Re-written wherever pc_target changes.
	if = {
		limit = { has_variable = pod_combat_enemy_leader_slot var:pod_combat_enemy_leader_slot = { has_variable = slot_char } }
		set_variable = { name = pod_combat_pc_target_char value = var:pod_combat_enemy_leader_slot.var:slot_char }
	}
	# AP / drain setup happens in the "PC goes first" branch below (and
	# AI turns set their own AP via pod_combat_execute_ai_turn_effect):
	# pod_combat_apply_ap_drain_effect reads pod_combat_actor_ap_drain
	# from the workspace, so the drain call must follow load_workspace.

	# Initialize combat log
	set_variable = { name = pod_combat_last_damage value = 0 }
	# AI anti-repetition history (pod_combat_ai_last_move /
	# _prev_move / _prev2_move) lives on the AI character, not on
	# the combat host (the selector switches into
	# var:pod_combat_actor_char before reading/writing). No PC-side
	# init or remove needed here. Per-fight cleanup of those vars
	# happens in pod_combat_init_slot_from_character_effect (slot
	# entry) and pod_combat_group_cleanup_effect (combat exit).

	# -- Non-fatal duel: gate PoD's inline death check --
	# For fatality = no we add pod_combat_nonfatal_active as a character
	# flag to every combatant. POD_can_die_from_wounds (overridden in
	# common/scripted_triggers/POD_combat/pod_combat_death_gate_overrides.txt)
	# reads the flag and returns no, so POD_health_death_check_effect
	# never enters its pod_death branch during the combat. The flag is
	# stripped post-combat by pod_combat_end_combat_effect AFTER
	# pod_combat_apply_nonfatal_effect has capped everyone to at
	# least one empty box. Fatal duels deliberately don't gate the
	# death check. Combatants can die mid-fight; KO via wound-track-
	# full happens regardless via the slot alive-flip check below.
	if = {
		limit = {
			has_variable = pod_combat_sc_context
			var:pod_combat_sc_context.var:fatality = flag:no
		}
		pod_combat_apply_nonfatal_flag_effect = yes
	}

	# -- Init-time KO check --
	# pod_combat_init_slot_from_character_effect sets slot_alive = 1
	# unconditionally for every populated slot. HP / morale / state are
	# all read live from the character at runtime, so a character who
	# walks into combat already KO'd (filled wound track from prior
	# damage, scripted incapacitation, torpor, morale-broken from a
	# prior scene) would be marked alive at init and could act / be
	# targeted before the first save_workspace check flipped their
	# flag. Apply the shared pod_combat_slot_should_be_marked_ko_trigger
	# (same predicate the action-boundary checks use) and then run the
	# victory check
	# so a degenerate "everyone on one side is already KO'd" combat
	# resolves immediately instead of running a phantom first round.
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			pod_combat_slot_should_be_marked_ko_trigger = yes
		}
		set_variable = { name = slot_alive value = 0 }
	}
	pod_combat_check_group_victory_effect = yes

	# -- Determine round 1 turn order --
	# Unified flow: every combat — including 1v1 — rolls initiative and
	# processes any pre-PC turns so nobody gets a free first move. A 1v1
	# is just a degenerate NvN with player/0 and enemy/0 populated. Skipped
	# when the init-time KO check already ended combat (everyone on
	# one side started KO'd).
	if = {
		limit = { NOT = { has_variable = pod_combat_is_over } }
	# System log: combat begins. Pushed BEFORE turn order so it
	# ends up at the BOTTOM of the newest-first log — reading bottom-up
	# the row tells "combat began, then these things happened…" Skipped
	# when the init-time KO check above already ended combat (no point
	# bookending an aborted combat).
	pod_combat_push_system_log_effect = { MESSAGE_KEY = combat_started }

	pod_combat_calculate_initiative_effect = yes
	pod_combat_find_next_actor_effect = yes

	if = {
		limit = {
			has_variable = pod_combat_current_actor_team
			var:pod_combat_current_actor_team = flag:player
			var:pod_combat_current_actor_position = 0
		}
		# PC goes first in initiative order — restore PC target, load
		# workspace, then set AP + apply drain (load must precede
		# drain so the workspace mirror is populated).
		pod_combat_restore_pc_target_effect = yes
		pod_combat_load_workspace_effect = yes
		# Explicit PC slot 0 hit-chance recalc — mirrors the proven
		# pod_combat_select_target SGUI sequence (load_workspace +
		# recalc_pc_hit_chance). The load_workspace writeback already
		# pushes the recomputed pair hit chance to slot 0, so this is
		# a redundant guarantee that PC's slot card displays the
		# correct PC-vs-pc_target hit chance from the moment the move
		# bar becomes interactive. Without this, intermittent staleness
		# was observed where PC's hit chance display only refreshed
		# after the player clicked a target, even when load_workspace
		# had run on entry.
		pod_combat_recalc_pc_hit_chance_effect = yes
		set_variable = { name = pod_combat_current_ap value = pod_combat_base_ap }
		pod_combat_apply_ap_drain_effect = yes
	}
	else = {
		# An AI goes first — process AI turns until it's the PC's turn
		# or combat ends. The PC is blocked from acting because the
		# turn-side check fires against current_actor != 0.
		# AI turn setup (load + set AP + drain) happens inside
		# pod_combat_execute_ai_turn_effect.
		pod_combat_process_pre_pc_turns_effect = yes
	}
	}  # end of NOT pod_combat_is_over guard (init-time KO check)
}

# Per-slot helper: compute one slot's base hit chance from prowess and
# apply it. Runs in the SLOT-STORY scope (called once per slot from the
# pod_combat_calculate_all_hit_chances_effect loop).
pod_combat_calc_one_hit_chance_effect = {
	# $OPP_PROWESS$ = the opposing team leader's prowess (a value the caller
	#   snapshotted in host scope). Base hit = pod_combat_hit_chance_base +
	#   (own prowess - opposing leader prowess) / 2, clamped 20-95, then the
	#   live hit_chance is rebuilt from base + active tiers.
	save_scope_value_as = {
		name = base_hit
		value = {
			value = pod_combat_hit_chance_base
			add = {
				value = var:slot_prowess
				subtract = $OPP_PROWESS$
				divide = 2
			}
			min = 20
			max = 95
		}
	}
	pod_combat_set_slot_var_effect = { NAME = base_hit_chance VALUE = scope:base_hit }
	pod_combat_recalc_slot_stats_effect = yes
}

pod_combat_calculate_all_hit_chances_effect = {
	# Compute every slot's base hit chance by iterating pod_combat_active_slots.
	# base = pod_combat_hit_chance_base + (own prowess - opposing leader
	# prowess) / 2, clamped 20-95. Player-team slots measure against the enemy
	# leader; enemy-team slots against the PC. recalc_slot_stats then rebuilds
	# the live hit_chance from base + active tier magnitudes (soak too, so tier
	# debuffs carry cleanly across the round-end reset). Pair-specific recalc
	# against the *currently selected* target happens later via
	# pod_combat_recalc_pc_hit_chance_effect.
	#
	# Snapshot each leader's prowess in host scope first (one read each) so the
	# per-slot script value references a saved value rather than re-walking the
	# host leader-pointer chain from slot scope.
	#
	# Guarded on both leaders existing: during the duel builder's tooltip
	# preview no slot stories are created, so the leader pointers are unset and
	# the chained .var:slot_prowess reads would log "variable not set". At real
	# execution (post-finalize, and on target-swap recalc) both leaders exist.
	if = {
		limit = {
			has_variable = pod_combat_enemy_leader_slot
			has_variable = pod_combat_pc_slot
		}
		save_scope_value_as = { name = enemy_leader_prowess value = var:pod_combat_enemy_leader_slot.var:slot_prowess }
		save_scope_value_as = { name = pc_prowess           value = var:pod_combat_pc_slot.var:slot_prowess }
		every_in_list = {
			variable = pod_combat_active_slots
			if = {
				limit = { var:combat_team = flag:player }
				pod_combat_calc_one_hit_chance_effect = { OPP_PROWESS = scope:enemy_leader_prowess }
			}
			else = {
				pod_combat_calc_one_hit_chance_effect = { OPP_PROWESS = scope:pc_prowess }
			}
		}
	}
}

##############################################################################################
# SECTION 6b: PLAYER HIT CHANCE RECALC ON TARGET SWAP
# Called by pod_combat_select_target (the unified target-select SGUI)
# after loading a new enemy into pod_combat_target_*. Rebuilds the
# PC's base_hit_chance from the freshly loaded target's prowess and
# propagates to slot 0's slot_base_hit_chance via
# pod_combat_set_slot_var_effect, then runs the workspace and per-slot
# tier recalc so any active hit buff/debuff tiers reapply cleanly
# against the new base. Stealth's +30 hit bonus is outside the tier
# system and stays tied to the stealth flag (consumed via the damage
# pipeline).
##############################################################################################

pod_combat_recalc_pc_hit_chance_effect = {
	# PC-specific hit-chance recalc: rebuilds slot 0's base_hit_chance
	# from the currently-loaded target's prowess, then calls the full
	# tier recalc so any active hit buffs / debuffs reapply cleanly
	# against the new base. Called when the PC interactively swaps
	# their target via pod_combat_select_target (the SGUI invokes this
	# right after loading the new enemy into pod_combat_target_*), and
	# also from the tier-recalc path when stat changes invalidate the
	# previously-computed hit chance.
	#
	# player/0 is hardcoded inside this effect. The function is
	# PC-only by design (target swap is a PC interaction; AI selects
	# moves and targets together each turn) — the `_pc_` in the
	# effect name encodes that contract.
	#
	# Round-duration tiers carry over (they expire at round end, not on
	# target swap). Stealth's +30 is a separate mechanic outside the
	# tier system and is handled by its own consume paths.
	if = {
		limit = {
			has_variable = pod_combat_actor_prowess
			has_variable = pod_combat_target_prowess
		}
		save_scope_value_as = {
			name = pc_recalc_base_hit
			value = {
				value = pod_combat_hit_chance_base
				add = {
					value = var:pod_combat_actor_prowess
					subtract = var:pod_combat_target_prowess
					divide = 2
				}
				min = 20
				max = 95
			}
		}
		set_variable = { name = pod_combat_actor_base_hit_chance value = scope:pc_recalc_base_hit }
		var:pod_combat_pc_slot = { pod_combat_set_slot_var_effect = { NAME = base_hit_chance VALUE = scope:pc_recalc_base_hit } }
		pod_combat_recalc_hit_actor_effect = yes
		var:pod_combat_pc_slot = { pod_combat_recalc_slot_stats_effect = yes }
	}
}

##############################################################################################
# SECTION 7: GROUP CLEANUP
# Remove all per-slot variables (every slot in pod_combat_active_slots) and global group state.
# Called in addition to the existing workspace cleanup.
##############################################################################################

pod_combat_group_cleanup_effect = {
	# ── Despawn summoned temporary combatants ────────────────────
	# Mid-combat summons (e.g. the necromancer's raised wraith) are temporary —
	# they vanish when the fight ends. Each was registered in
	# pod_combat_summoned_chars by pod_combat_summon_wraith_glue_effect. Kill any
	# still alive (a fatal duel may have killed some already — the is_alive guard
	# avoids double-removal); their slot stories are end_story'd by the teardown
	# below, and the AI-state cleanup that follows tolerates a dead slot_char.
	if = {
		limit = { has_variable = pod_combat_summoned_chars }
		every_in_list = {
			variable = pod_combat_summoned_chars
			limit = { is_alive = yes }
			death = { death_reason = death_disappearance }
		}
		clear_variable_list = pod_combat_summoned_chars
	}

	# ── AI per-fight transient state cleanup ─────────────────────
	# Two kinds of state on AI characters that should be per-fight:
	#   - pod_combat_loadout: the AI's scratch loadout list (one entry
	#     per generated move — full-catalogue snapshot since AI isn't
	#     restricted by category caps).
	#     Cleared so save data doesn't carry it between fights.
	#   - pod_combat_ai_last_move / _prev_move / _prev2_move: the
	#     three-deep anti-repetition history. Cleared so the same
	#     NPC's next fight doesn't open with -35/-20/-12 penalties
	#     against the moves they used last combat (penalty values
	#     live in pod_combat_ai_effects.txt's selector weight block).
	# Slot 0 (PC) is skipped. the player's loadout is persistent
	# equipment, and the AI history vars are never written on the PC
	# (the selector switches into var:pod_combat_actor_char before
	# touching them, which is always an AI on AI turns).
	#
	# Runs FIRST so the slot stories still exist (and slot_char is
	# still readable) when we reach into each slot's character to
	# clear its per-fight scratch state. The slot-story teardown
	# below tears down everything the chained read needs.
	#
	# Pairs with the slot-init clear in
	# pod_combat_init_slot_from_character_effect (handles legacy
	# NPCs on the way INTO combat; this handles save bloat / cross-
	# fight bias on the way OUT).
	# Iterate active slots, skipping the PC (player/0) and any non-AI
	# combatants. var:slot_alive isn't checked — even KO'd AIs have
	# accumulated cross-fight bias that should be cleared.
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			NOT = { AND = { var:combat_team = flag:player var:team_position = 0 } }
			var:slot_char = { is_ai = yes }
		}
		var:slot_char = {
			clear_variable_list = pod_combat_loadout
			remove_variable = pod_combat_ai_last_move
			remove_variable = pod_combat_ai_prev_move
			remove_variable = pod_combat_ai_prev2_move
		}
	}

	# ── Slot story_cycle teardown ────────────────────────────────
	# end_story every POD_combat_slot story (iterating
	# pod_combat_active_slots — the single slot list), drop the two leader
	# pointer vars (pod_combat_pc_slot / pod_combat_enemy_leader_slot), and
	# clear pod_combat_active_slots. Runs AFTER the AI cleanup
	# above so the slots' var:slot_char reads
	# still resolve. See pod_combat_slot_effects.txt for the lifecycle
	# helper definitions.
	pod_combat_clear_slot_stories_effect = yes

	# ── Workspace mirrors for the spirit capability flags ────────
	# Per-slot slot_incorporeal_spirit / slot_can_damage_spirit are
	# torn down with the rest of the slot story above; these are the
	# host-scope workspace mirrors that
	# pod_combat_load_actor_from_story_effect /
	# pod_combat_load_target_from_story_effect populate.
	remove_variable = pod_combat_actor_incorporeal_spirit
	remove_variable = pod_combat_actor_can_damage_spirit
	remove_variable = pod_combat_target_incorporeal_spirit
	remove_variable = pod_combat_target_can_damage_spirit

	# Global group state. (pod_combat_mode was retired with the duel
	# builder — the roster is explicit now, not a 0/1/2/3 shape code — so
	# there's nothing to remove for it.)
	remove_variable = pod_combat_enemy_team_size
	remove_variable = pod_combat_player_team_size
	remove_variable = pod_combat_current_actor_team
	remove_variable = pod_combat_current_actor_position
	remove_variable = pod_combat_selected_target_team
	remove_variable = pod_combat_selected_target_position
	remove_variable = pod_combat_pc_target_team
	remove_variable = pod_combat_pc_target_position
	remove_variable = pod_combat_pc_target_char
	remove_variable = pod_combat_best_init
	remove_variable = pod_combat_best_found
	remove_variable = pod_combat_best_team
	remove_variable = pod_combat_best_position
}

##############################################################################################
# SECTION 8: INITIATIVE SYSTEM
##############################################################################################

# Roll initiative for every filled+alive slot in one pass over
# pod_combat_active_slots, fold in each slot's pending slot_init_bonus,
# and clear slot_has_acted ready for the new round.
pod_combat_calculate_initiative_effect = {
	# Roll initiative for every alive combatant and stamp turn-order
	# rank. Initiative = prowess + roll + init_bonus, where roll is a
	# uniform integer in [1, pod_combat_initiative_roll_max] sampled via
	# the engine's `integer_range` script-math primitive. The upper
	# bound is the only tunable — defined in
	# pod_combat_values.txt (search `pod_combat_initiative_roll_max`)
	# with a comment block explaining the gameplay knob:
	#   higher → more chaos in turn order
	#   lower  → more prowess-deterministic
	# init_bonus comes from buff/debuff moves applied previous round;
	# positive = faster (buff), negative = slower (debuff). Cleared
	# after application.
	#
	# Iterates pod_combat_active_slots — the iteration body runs in
	# slot-story scope so var:slot_X reads/writes target the slot
	# story directly.
	save_scope_as = host
	every_in_list = {
		variable = pod_combat_active_slots
		limit = { var:slot_alive = 1 }
		set_variable = {
			name = slot_initiative
			value = {
				value = var:slot_prowess
				add = {
					integer_range = { min = 1 max = pod_combat_initiative_roll_max }
				}
			}
		}
		if = {
			limit = { has_variable = slot_init_bonus }
			change_variable = { name = slot_initiative add = var:slot_init_bonus }
			remove_variable = slot_init_bonus
		}
		set_variable = { name = slot_has_acted value = 0 }
	}

	# Stamp each slot's turn-order rank as a stored slot var so the
	# datamodel-driven slot card widget can display "Turn N" without
	# needing a per-slot SV that compares against all other slots
	# (the slot story can't reach the host's other-slot pointers from
	# its own scope, and dynamic SV-name construction isn't supported
	# in GUI bindings).
	pod_combat_calc_turn_orders_effect = yes
}

# Compute slot_turn_order on every alive combatant after initiatives
# are rolled. Rank semantics: highest initiative = rank 1, ties broken
# by lower team ordinal (so e.g. player/0 beats enemy/0 if both rolled
# the same number).
#
# Implementation is a nested every_in_list: outer loop iterates each
# alive slot story to compute its rank; inner loop counts how many
# OTHER alive slots beat it on the (initiative, -ordinal)
# lexicographic order. Saving scope:host before iterating lets the
# inner loop re-enter host scope to iterate the active list again.
# Saving scope:me lets the inner loop write back into the outer's
# slot story.
pod_combat_calc_turn_orders_effect = {
	save_scope_as = host
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			has_variable = slot_initiative
			var:slot_alive = 1
		}
		save_scope_as = me
		save_scope_value_as = { name = my_init       value = var:slot_initiative }
		# Tiebreak key is (is_enemy asc, then team_position asc) — player team
		# before enemy team, lower positions first. A direct two-field compare
		# (not a single ordinal) keeps the ordering correct for unbounded teams.
		save_scope_value_as = { name = my_is_enemy   value = pod_combat_slot_is_enemy_value }
		save_scope_value_as = { name = my_position   value = var:team_position }
		# Start at rank 1; bump for each other slot that ranks above us.
		set_variable = { name = slot_turn_order value = 1 }
		# Re-enter host scope for the inner iteration over the same list.
		scope:host = {
			every_in_list = {
				variable = pod_combat_active_slots
				limit = {
					has_variable = slot_initiative
					var:slot_alive = 1
					# Exclude self.
					NOT = { this = scope:me }
					# Strictly higher initiative, OR same initiative and the
					# other slot ranks earlier on the (is_enemy, team_position)
					# tiebreak: other is player while I'm enemy, OR same team
					# with a lower position. Strict comparisons so two distinct
					# slots never both count each other (the legacy >= vs >
					# asymmetry).
					OR = {
						var:slot_initiative > scope:my_init
						AND = {
							var:slot_initiative = scope:my_init
							OR = {
								pod_combat_slot_is_enemy_value < scope:my_is_enemy
								AND = {
									pod_combat_slot_is_enemy_value = scope:my_is_enemy
									var:team_position < scope:my_position
								}
							}
						}
					}
				}
				# Bump the OUTER slot's rank — write back via scope:me.
				scope:me = {
					change_variable = { name = slot_turn_order add = 1 }
				}
			}
		}
	}
}

##############################################################################################
# SECTION 9: FIND NEXT ACTOR
# Walk pod_combat_active_slots (filled-only by construction) and pick
# the slot with the highest initiative that hasn't acted yet.
##############################################################################################

pod_combat_find_next_actor_effect = {
	# Find the alive combatant with highest unprocessed initiative.
	# best_init starts below the most negative initiative any combatant
	# can realistically roll — the per-slot init roll is `prowess +
	# random_list bucket [1,30] + slot_init_bonus`, and slot_init_bonus
	# can be NEGATIVE via stacked initiative_debuff moves. A -1 sentinel
	# would silently skip any slot whose final initiative landed at or
	# below -1 (the strict `>` compare in the body never triggers),
	# ending the round early and stealing turns from heavily-debuffed
	# combatants. -999 stays safely below any practical clamp; if init
	# math ever grows past that range, push this further down.
	set_variable = { name = pod_combat_best_init value = -999 }
	# best_found presence replaces the old best_slot = -1 sentinel: a slot
	# is addressed by (combat_team, team_position) now, and there's no
	# "-1 means none" pair. pod_combat_best_found is set only when a
	# candidate is picked; absence after the scan = "no actor this pass".
	remove_variable = pod_combat_best_found

	# Iterate the active slot list and pick the highest-initiative slot
	# that hasn't acted yet. Iteration order matches the slot stories'
	# insertion order (PC at player/0, then allies player/1-4, then enemies
	# enemy/0-4 — the same ordering the old slot_index 0..9 produced), so
	# among ties the first-inserted slot wins because the strict `>` check
	# inside the body only updates best_* on a STRICTLY higher init.
	save_scope_as = host
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			var:slot_alive = 1
			var:slot_has_acted = 0
		}
		save_scope_value_as = { name = current_init value = var:slot_initiative }
		# Save the slot story so the host-scope block below can read its
		# combat_team (a flag — can't be captured via save_scope_value_as).
		save_scope_as = candidate_slot
		scope:host = {
			if = {
				limit = { scope:current_init > var:pod_combat_best_init }
				set_variable = { name = pod_combat_best_init value = scope:current_init }
				set_variable = { name = pod_combat_best_team value = scope:candidate_slot.var:combat_team }
				set_variable = { name = pod_combat_best_position value = scope:candidate_slot.var:team_position }
				set_variable = { name = pod_combat_best_found value = yes }
			}
		}
	}

	# Set the current actor from the best slot found, or clear the pointer
	# if nobody qualified (all alive slots have acted → round is over).
	if = {
		limit = { has_variable = pod_combat_best_found }
		set_variable = { name = pod_combat_current_actor_team value = var:pod_combat_best_team }
		set_variable = { name = pod_combat_current_actor_position value = var:pod_combat_best_position }
	}
	else = {
		remove_variable = pod_combat_current_actor_team
		remove_variable = pod_combat_current_actor_position
	}

	# Clean up temp vars
	remove_variable = pod_combat_best_init
	remove_variable = pod_combat_best_found
	remove_variable = pod_combat_best_team
	remove_variable = pod_combat_best_position
}

##############################################################################################
# SECTION 10: AI TARGET SELECTION
# Two effects: one for selecting an enemy target (for player team AI), one for selecting
# a player team target (for enemy AI). Uses weighted random favoring low-HP targets.
##############################################################################################

pod_combat_ai_select_enemy_target_effect = {
	# Select a target from the enemy team for an ally AI actor.
	# random_in_list iterates the active slots filtered to combat_team =
	# enemy; weighting biases toward low-HP targets (×2 if 2 or fewer empty
	# wound boxes left).
	save_scope_as = host
	random_in_list = {
		variable = pod_combat_active_slots
		limit = {
			var:slot_alive = 1
			var:combat_team = flag:enemy
		}
		weight = {
			base = 1
			modifier = { add = 50 }
			modifier = {
				factor = 2
				var:slot_char.POD_empty_damage_boxes <= 2
			}
		}
		save_scope_as = chosen_target
	}
	if = {
		limit = { exists = scope:chosen_target }
		scope:host = {
			set_variable = { name = pod_combat_selected_target_team     value = scope:chosen_target.var:combat_team }
			set_variable = { name = pod_combat_selected_target_position value = scope:chosen_target.var:team_position }
		}
	}
}

pod_combat_ai_select_player_target_effect = {
	# Select a target from the player team for an enemy AI actor. Same
	# weight shape as the enemy-target effect; just the team filter inverts.
	save_scope_as = host
	random_in_list = {
		variable = pod_combat_active_slots
		limit = {
			var:slot_alive = 1
			var:combat_team = flag:player
		}
		weight = {
			base = 1
			modifier = { add = 50 }
			modifier = {
				factor = 2
				var:slot_char.POD_empty_damage_boxes <= 2
			}
		}
		save_scope_as = chosen_target
	}
	if = {
		limit = { exists = scope:chosen_target }
		scope:host = {
			set_variable = { name = pod_combat_selected_target_team     value = scope:chosen_target.var:combat_team }
			set_variable = { name = pod_combat_selected_target_position value = scope:chosen_target.var:team_position }
		}
	}
}

##############################################################################################
# SECTION 10b: PC TARGET PERSISTENCE (UNIFIED)
#
# pod_combat_pc_target_team / _position together point at whomever the PC has
# currently selected via the Target button on any combat card: ally OR enemy. A skill's
# target-type (single-enemy / single-ally / team-*) is enforced by the can_use trigger,
# not by having separate target variables.
#
# The PC (player/0) is never a valid stored target — self-only skills use
# targeting = self, not pc_target = player/0.
#
# This effect is called at PC turn start AND after every PC move, so if the PC's target
# just got KO'd, it auto-retargets to the first alive non-PC slot (preferring enemies
# first, then allies). Combat should have ended before we run out of candidates.
##############################################################################################

pod_combat_restore_pc_target_effect = {
	# Detect dead/missing pc_target via any_in_list — if NO active
	# slot matches pc_target with slot_alive = 1, we need to retarget.
	# The OR also covers the unset case (no pc_target var) and the
	# self-targeting case (pc_target = 0 = PC).
	save_scope_as = host
	if = {
		limit = {
			OR = {
				NOT = { has_variable = pod_combat_pc_target_team }
				AND = { var:pod_combat_pc_target_team = flag:player var:pod_combat_pc_target_position = 0 }
				NOT = {
					any_in_list = {
						variable = pod_combat_active_slots
						var:combat_team = scope:host.var:pod_combat_pc_target_team
						var:team_position = scope:host.var:pod_combat_pc_target_position
						var:slot_alive = 1
					}
				}
			}
		}
		# Auto-retarget: prefer alive enemies first, then non-PC allies.
		# Two-pass iteration over pod_combat_active_slots: first pass finds
		# the first alive enemy; second pass falls back to the first alive
		# ally (team_position >= 1, i.e. not the PC) if no enemy was found.
		# The `NOT = { exists = scope:chosen }` clause in each limit makes
		# the iteration first-match (the slot iteration order is insertion
		# order — player/0..4 then enemy/0..4). If nothing is alive we fall
		# through silently — combat is effectively over.
		every_in_list = {
			variable = pod_combat_active_slots
			limit = {
				NOT = { exists = scope:chosen }
				var:combat_team = flag:enemy
				var:slot_alive = 1
			}
			save_scope_as = chosen
		}
		every_in_list = {
			variable = pod_combat_active_slots
			limit = {
				NOT = { exists = scope:chosen }
				var:combat_team = flag:player
				var:team_position >= 1
				var:slot_alive = 1
			}
			save_scope_as = chosen
		}
		if = {
			limit = { exists = scope:chosen }
			scope:host = {
				set_variable = { name = pod_combat_pc_target_team           value = scope:chosen.var:combat_team }
				set_variable = { name = pod_combat_pc_target_position        value = scope:chosen.var:team_position }
				set_variable = { name = pod_combat_pc_target_char           value = scope:chosen.var:slot_char }
				set_variable = { name = pod_combat_selected_target_team      value = scope:chosen.var:combat_team }
				set_variable = { name = pod_combat_selected_target_position  value = scope:chosen.var:team_position }
			}
			scope:chosen = {
				pod_combat_load_target_from_story_effect = yes
			}
		}
	}
	else = {
		# Current PC target is still alive — mirror pc_target into selected_target
		# (in case an AI clobbered it with their own target during the previous turn)
		# and reload the target workspace with the PC's choice.
		set_variable = { name = pod_combat_selected_target_team     value = var:pod_combat_pc_target_team }
		set_variable = { name = pod_combat_selected_target_position value = var:pod_combat_pc_target_position }
		every_in_list = {
			variable = pod_combat_active_slots
			limit = {
				var:combat_team = scope:host.var:pod_combat_pc_target_team
				var:team_position = scope:host.var:pod_combat_pc_target_position
			}
			pod_combat_load_target_from_story_effect = yes
		}
	}
}

##############################################################################################
# SECTION 11: GROUP TURN LOOP
# The main group turn processing effect. Called after the PC ends their turn (or at round
# start for non-PC actors). Processes all remaining actors in initiative order.
##############################################################################################

pod_combat_process_group_turns_effect = {
	# Process all remaining actors in initiative order
	# This is called after the PC acts (slot 0) to run all AI turns

	# Mark current actor (PC) as acted
	var:pod_combat_pc_slot = { pod_combat_set_slot_var_effect = { NAME = has_acted VALUE = 1 } }

	# Save PC's workspace back to the PC slot
	pod_combat_save_workspace_effect = yes

	# Process every remaining (post-PC) AI in initiative order. A real loop,
	# not a fixed unroll: team size is unbounded now, so the old 9-iteration
	# cap (10 slots minus the PC) stranded every combatant past the 9th.
	# Terminates because each execute_ai_turn marks exactly one slot
	# has_acted, so the alive-and-unacted set strictly shrinks; the inner
	# victory check can also set pod_combat_is_over.
	while = {
		limit = {
			NOT = { has_variable = pod_combat_is_over }
			any_in_list = {
				variable = pod_combat_active_slots
				var:slot_alive = 1
				var:slot_has_acted = 0
			}
		}
		pod_combat_find_next_actor_effect = yes
		if = {
			limit = { has_variable = pod_combat_current_actor_team NOT = { has_variable = pod_combat_is_over } }
			pod_combat_execute_ai_turn_effect = yes
		}
	}

	# All actors have acted — end round if combat not over
	if = {
		limit = { NOT = { has_variable = pod_combat_is_over } }
		pod_combat_group_end_round_effect = yes
	}
}

pod_combat_execute_ai_turn_effect = {
	# Execute one AI turn for the current actor
	# 1. Select target based on actor's team
	# 2. Load workspace
	# 3. Run AI move selection
	# 4. Save workspace
	# 5. Mark as acted
	# 6. Check victory

	# Select target
	if = {
		limit = { has_variable = pod_combat_current_actor_team var:pod_combat_current_actor_team = flag:player }
		# Actor is on player team — target an enemy
		pod_combat_ai_select_enemy_target_effect = yes
	}
	else = {
		# Actor is on enemy team — target a player
		pod_combat_ai_select_player_target_effect = yes
	}

	# Load workspace (actor -> actor workspace, target -> target workspace)
	pod_combat_load_workspace_effect = yes

	# Set AP for this actor's turn
	set_variable = { name = pod_combat_current_ap value = pod_combat_base_ap }
	pod_combat_apply_ap_drain_effect = yes

	# Run AI moves (uses existing pod_combat_ai_take_turn_effect which reads from workspace)
	pod_combat_ai_take_turn_effect = yes

	# Save workspace back to slots
	pod_combat_save_workspace_effect = yes

	# Mark actor as acted. Writes slot_has_acted = 1 to the slot whose
	# (combat_team, team_position) matches the current actor. scope:host
	# was already saved by pod_combat_save_workspace_effect above and
	# persists through the rest of this effect chain. Iterates
	# pod_combat_active_slots; the limit matches at most one slot (the
	# team+position pair is unique).
	if = {
		limit = { has_variable = pod_combat_current_actor_team }
		every_in_list = {
			variable = pod_combat_active_slots
			limit = {
				var:combat_team = scope:host.var:pod_combat_current_actor_team
				var:team_position = scope:host.var:pod_combat_current_actor_position
			}
			set_variable = { name = slot_has_acted value = 1 }
		}
	}

	# Check victory
	pod_combat_check_group_victory_effect = yes
}

##############################################################################################
# SECTION 12: GROUP VICTORY CHECK
# Check if all enemies are knocked out (player wins) or all allies are knocked out
# (player loses).
##############################################################################################

pod_combat_check_group_victory_effect = {
	# Victory conditions (evaluated in order, first match ends combat):
	#   1. All enemies KO'd           → player team wins
	#   2. PC (player/0) KO'd            → player loses (PC can't act, so combat cannot continue)
	#   3. All allies KO'd             → player loses

	# Save the PC scope so we can reference it via scope:combat_pc in set_variable
	# (using bare `root` there triggers "Event target link 'root' returned unset scope")
	save_scope_as = combat_pc

	# --- Check 1: All enemies knocked out ---
	# Expressed as "no alive enemy slot exists" — `any_in_list` over
	# the active list filtered by combat_team = flag:enemy and
	# slot_alive = 1. pod_combat_active_slots is filled-only by
	# construction, so no separate slot_char presence gate is needed.
	if = {
		limit = {
			NOT = {
				any_in_list = {
					variable = pod_combat_active_slots
					var:combat_team = flag:enemy
					var:slot_alive = 1
				}
			}
			has_variable = pod_combat_enemy_leader_slot var:pod_combat_enemy_leader_slot = { has_variable = slot_char }
			NOT = { has_variable = pod_combat_is_over }
		}
		# Player team wins!
		set_variable = { name = pod_combat_winner value = scope:combat_pc }
		set_variable = { name = pod_combat_loser value = var:pod_combat_enemy_leader_slot.var:slot_char }
		set_variable = { name = pod_combat_is_over value = yes }
		var:pod_combat_enemy_leader_slot.var:slot_char = { save_scope_as = pod_combat_target }
		# System log: combat ends — victory. Pushed AFTER setting
		# is_over so the bookend row appears at the TOP of the
		# newest-first log (most recent), framing the row above as the
		# resolved outcome.
		pod_combat_push_system_log_effect = { MESSAGE_KEY = combat_ended_victory }
		# Combat window stays open — player clicks "End Combat" to dismiss
	}

	# --- Check 2: PC knocked out (even if allies still standing) ---
	# The PC is special — the player has no way to continue the fight if they are down.
	# End combat as a loss so the post-fight event can resolve wounds and outcomes.
	if = {
		limit = {
			has_variable = pod_combat_pc_slot
			var:pod_combat_pc_slot = { has_variable = slot_char var:slot_alive = 0 }
			has_variable = pod_combat_enemy_leader_slot var:pod_combat_enemy_leader_slot = { has_variable = slot_char }
			NOT = { has_variable = pod_combat_is_over }
		}
		set_variable = { name = pod_combat_winner value = var:pod_combat_enemy_leader_slot.var:slot_char }
		set_variable = { name = pod_combat_loser value = scope:combat_pc }
		set_variable = { name = pod_combat_is_over value = yes }
		var:pod_combat_enemy_leader_slot.var:slot_char = { save_scope_as = pod_combat_target }
		pod_combat_push_system_log_effect = { MESSAGE_KEY = combat_ended_defeat }
		# Combat window stays open — player clicks "End Combat" to dismiss
	}

	# --- Check 3: All allies knocked out (covers edge cases where PC death check was bypassed) ---
	# Symmetric to Check 1: combat_team = flag:player filters to player team
	# (PC + ally slots 1-4). See Check 1 comment for the count-to-
	# any_in_list rewrite rationale.
	if = {
		limit = {
			NOT = {
				any_in_list = {
					variable = pod_combat_active_slots
					var:combat_team = flag:player
					var:slot_alive = 1
				}
			}
			has_variable = pod_combat_enemy_leader_slot var:pod_combat_enemy_leader_slot = { has_variable = slot_char }
			NOT = { has_variable = pod_combat_is_over }
		}
		# Enemy team wins
		set_variable = { name = pod_combat_winner value = var:pod_combat_enemy_leader_slot.var:slot_char }
		set_variable = { name = pod_combat_loser value = scope:combat_pc }
		set_variable = { name = pod_combat_is_over value = yes }
		var:pod_combat_enemy_leader_slot.var:slot_char = { save_scope_as = pod_combat_target }
		pod_combat_push_system_log_effect = { MESSAGE_KEY = combat_ended_defeat }
		# Combat window stays open — player clicks "End Combat" to dismiss
	}
}

##############################################################################################
# SECTION 13: GROUP ROUND END
# Increment round, calculate new initiative, expire round-duration tier buffs,
# and set up the next round.
##############################################################################################

pod_combat_group_end_round_effect = {
	# Guard: if combat ended during this round (victory check fired
	# pod_combat_end_combat_effect which may have cleaned up variables),
	# skip the round-end logic entirely.
	if = {
		limit = { NOT = { has_variable = pod_combat_round } }
		# Combat already cleaned up — nothing to do
	}
	else = {

	change_variable = { name = pod_combat_round add = 1 }

	# System log: round transition. Pushed AFTER the increment
	# so entry.var:round captures the NEW round number — the log
	# row reads "— Round 2 begins —" at the top of the freshly-
	# started round's action sequence.
	pod_combat_push_system_log_effect = { MESSAGE_KEY = round_begins }

	# Calculate initiative for next round
	pod_combat_calculate_initiative_effect = yes

	# Find the first actor (highest initiative)
	pod_combat_find_next_actor_effect = yes

	# Expire all round-duration buff/debuff tiers on every slot, then
	# rebuild each slot's live soak and hit_chance from base + remaining
	# combat-duration tier magnitudes via
	# pod_combat_recalc_slot_stats_via_story_effect (covers slot_soak
	# revert too — no separate soak-reset pass needed).
	pod_combat_revert_round_tiers_effect = yes

	# If PC (slot 0) goes first, set up workspace and wait for player input
	if = {
		limit = { has_variable = pod_combat_current_actor_team var:pod_combat_current_actor_team = flag:player var:pod_combat_current_actor_position = 0 }
		# Order matters: pod_combat_apply_ap_drain_effect reads
		# pod_combat_actor_ap_drain from the workspace, which is
		# populated by pod_combat_load_actor_from_story_effect inside
		# load_workspace. Restoring the target first and loading
		# before draining mirrors the AI turn-start ordering at the
		# top of pod_combat_execute_ai_turn_effect.
		pod_combat_restore_pc_target_effect = yes
		pod_combat_load_workspace_effect = yes
		# Explicit PC slot 0 hit-chance recalc — mirrors the proven
		# pod_combat_select_target SGUI sequence. See the matching
		# call in POD_combat_finalize_duel_effect for the full rationale.
		pod_combat_recalc_pc_hit_chance_effect = yes
		set_variable = { name = pod_combat_current_ap value = pod_combat_base_ap }
		pod_combat_apply_ap_drain_effect = yes
	}
	else = {
		# Non-PC goes first — process all AI turns until it's PC's turn or round ends
		pod_combat_process_pre_pc_turns_effect = yes
	}

	} # end else (combat not yet cleaned up)
}

pod_combat_process_pre_pc_turns_effect = {
	# Process AI turns that happen before the PC's turn, in initiative order.
	# The caller has already set current_actor via find_next. A real loop, not
	# a fixed unroll: with unbounded team sizes more than 9 combatants can out-
	# roll the PC, and the old 9-iteration cap then left the PC's turn un-set-up
	# (a soft-lock: current_actor stranded on an AI nobody runs). Each pass runs
	# the current AI then advances; the loop exits the moment find_next lands on
	# the PC (current_actor = player/0), after which the PC-setup block below
	# takes over. Terminates because every execute_ai_turn marks one slot
	# has_acted, so find_next must eventually reach the still-unacted PC.
	while = {
		limit = {
			has_variable = pod_combat_current_actor_team
			NOT = { AND = { var:pod_combat_current_actor_team = flag:player var:pod_combat_current_actor_position = 0 } }
			NOT = { has_variable = pod_combat_is_over }
		}
		pod_combat_execute_ai_turn_effect = yes
		pod_combat_find_next_actor_effect = yes
	}

	# If PC is next (or combat is over), set up workspace for PC
	if = {
		limit = { has_variable = pod_combat_current_actor_team var:pod_combat_current_actor_team = flag:player var:pod_combat_current_actor_position = 0 NOT = { has_variable = pod_combat_is_over } }
		# HP / max / agg_damage derive live from each slot character's
		# PoD wound vars via SVs (pod_combat_slot_hp_value et al. in
		# pod_combat_values.txt) — there is no slot-stored HP cache to
		# resync, so the wound bars in the combat window read the
		# current values automatically once the workspace is loaded.
		# See COMBAT_REFERENCE.info §7 "HP cache rejected" for the
		# rationale.

		# Order matters: pod_combat_apply_ap_drain_effect reads
		# pod_combat_actor_ap_drain from the workspace, which is
		# populated by load_actor_from_story_effect inside
		# load_workspace. Restore target + load workspace must happen
		# BEFORE the AP / drain pair.
		pod_combat_restore_pc_target_effect = yes
		pod_combat_load_workspace_effect = yes
		# Explicit PC slot 0 hit-chance recalc — mirrors the proven
		# pod_combat_select_target SGUI sequence. See the matching
		# call in POD_combat_finalize_duel_effect for the full rationale.
		# Especially important on this path because every AI turn that
		# attacked the PC pushed "PC-vs-that-AI" into slot 0 via
		# save_workspace; load_workspace's writeback should refresh it
		# against PC's current target, but the redundant explicit recalc
		# matches what the player-click path does and was observed to
		# fix intermittent display staleness.
		pod_combat_recalc_pc_hit_chance_effect = yes
		set_variable = { name = pod_combat_current_ap value = pod_combat_base_ap }
		pod_combat_apply_ap_drain_effect = yes
		# DO NOT reset is_guarding / is_dodging / soak here. Stance
		# state persists until consumed (see comments in damage
		# pipeline), and pod_combat_load_workspace_effect already
		# performs the correct set/remove based on each slot story's
		# state — see pod_combat_load_actor_from_story_effect /
		# _target_from_story_effect, where every stance flag has both
		# a `set_variable = yes` branch and an `else = remove_variable`
		# branch keyed on the slot story's slot_is_X presence.
	}
}
