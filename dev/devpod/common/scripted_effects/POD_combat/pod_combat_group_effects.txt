##############################################################################################
# GROUP COMBAT INFRASTRUCTURE EFFECTS
# Handles slot-based workspace load/save, dispatch, initialization, assignment, and cleanup
# for the group combat system (up to 10 combatants in slots 0-9).
#
# Slot layout:
#   0    = PC (player character, always present)
#   1-4  = Ally slots (coterie members, modes 2/3)
#   5    = Primary opponent (always present)
#   6-9  = Enemy slots (enemy knights, modes 1/3)
#
# Modes:
#   0 = 1v1, 1 = 1vN, 2 = Nv1, 3 = NvN
#
# Terminology:
#   actor / target   — workspace-relative. "actor" = whoever is ACTING
#                      this action; "target" = whoever is being TARGETED.
#                      Used by pod_combat_actor_* / pod_combat_target_*
#                      workspace mirrors, by recalc helper names, and
#                      as the literal $ACTOR_SIDE$ / $TARGET_SIDE$
#                      parametric values (so e.g.
#                      pod_combat_$TARGET_SIDE$_char resolves to
#                      pod_combat_target_char). The "actor/target"
#                      framing follows the action, not team position:
#                      on enemy turns the enemy's slot loads into the
#                      "actor" workspace, and the PC may load into
#                      "target" if the AI is hitting them.
#   player / enemy   — positional, slot-index based. "player team" =
#                      slots 0-4 (PC + ally knights); "enemy team" =
#                      slots 5-9 (primary opponent + enemy knights).
#                      Used for slot range filters in active_slots
#                      iteration (var:slot_index <= 4 vs >= 5), team
#                      AoE filters, and victory-check team wipes.
#                      There is NO pod_combat_player_* or
#                      pod_combat_enemy_* var name — positional naming
#                      lives in slot_index ranges, not var prefixes.
##############################################################################################

##############################################################################################
# SECTION 1: WORKSPACE LOAD EFFECTS
# Two parametric effects that copy a given slot's state into the actor or target workspace.
# Called via dispatch effects that switch on pod_combat_current_actor / pod_combat_selected_target.
##############################################################################################

# Caller scope = the slot story; caller must have saved the workspace
# owner as scope:host before delegating. Reads the slot's vars
# directly (current scope = slot story) and writes pod_combat_actor_*
# on scope:host.
#
# MIRROR of pod_combat_load_target_from_story_effect (below) — same
# field-by-field copy, writes pod_combat_actor_* instead of
# pod_combat_target_*. If you add a field to one, add it to both.
#
# Used by pod_combat_load_workspace_effect's iteration over
# pod_combat_active_slots (finds the slot whose slot_index matches
# pod_combat_current_actor), and by any future single-slot caller
# that needs to save scope:host and flip into a slot story before
# delegating.
pod_combat_load_actor_from_story_effect = {
	if = {
		limit = { has_variable = slot_char }
		save_scope_as = slot_story
		scope:host = {
			set_variable = { name = pod_combat_actor_soak value = scope:slot_story.var:slot_soak }
			set_variable = { name = pod_combat_actor_base_soak value = scope:slot_story.var:slot_base_soak }
			if = { limit = { scope:slot_story = { has_variable = slot_base_hit_chance } } set_variable = { name = pod_combat_actor_base_hit_chance value = scope:slot_story.var:slot_base_hit_chance } }
			set_variable = { name = pod_combat_actor_morale value = scope:slot_story.var:slot_morale }
			set_variable = { name = pod_combat_actor_prowess value = scope:slot_story.var:slot_prowess }
			set_variable = { name = pod_combat_actor_hit_chance value = scope:slot_story.var:slot_hit_chance }
			# slot_energy_used is default-init'd to 0 at story creation, so
			# always present — no has_variable gate needed. Named
			# pod_combat_actor_energy_used (not _player_) — see comment
			# in pod_combat_move_effects.txt §3 for the convention break.
			set_variable = { name = pod_combat_actor_energy_used value = scope:slot_story.var:slot_energy_used }
			if = {
				limit = { scope:slot_story = { has_variable = slot_is_guarding } }
				set_variable = { name = pod_combat_actor_is_guarding value = yes }
			}
			else = {
				remove_variable = pod_combat_actor_is_guarding
			}
			if = {
				limit = { scope:slot_story = { has_variable = slot_is_dodging } }
				set_variable = { name = pod_combat_actor_is_dodging value = yes }
			}
			else = {
				remove_variable = pod_combat_actor_is_dodging
			}
			if = {
				limit = { scope:slot_story = { has_variable = slot_is_stealthed } }
				set_variable = { name = pod_combat_actor_is_stealthed value = yes }
			}
			else = {
				remove_variable = pod_combat_actor_is_stealthed
			}
			# Spirit-damage capability (static per combat). The actor needs
			# this flag to land attacks on incorporeal spirit targets 
			# read by the spirit gate inside pod_combat_resolve_contact_effect
			# (gated on the move's checks_spirit_gate capability flag) via
			# pod_combat_$ACTOR_SIDE$_can_damage_spirit.
			if = {
				limit = { scope:slot_story = { has_variable = slot_can_damage_spirit } }
				set_variable = { name = pod_combat_actor_can_damage_spirit value = yes }
			}
			else = {
				remove_variable = pod_combat_actor_can_damage_spirit
			}
			# Incorporeality on the actor side is read by the retaliation
			# spirit gate in pod_combat_apply_retaliation_helper: when the
			# original attacker (= retaliation target) is incorporeal, the
			# counter-strike only lands if the retaliator has the matching
			# can_damage_spirit flag.
			if = {
				limit = { scope:slot_story = { has_variable = slot_incorporeal_spirit } }
				set_variable = { name = pod_combat_actor_incorporeal_spirit value = yes }
			}
			else = {
				remove_variable = pod_combat_actor_incorporeal_spirit
			}
			# Initiative bonus is not mirrored on the workspace — slot_init_bonus
			# is set by mid-execution moves and consumed at round-end by the
			# initiative recalc, both of which read the slot directly. The
			# workspace mirror would carry no information.
			#
			# AP drain (applied at turn start, then cleared).
			if = {
				limit = { scope:slot_story = { has_variable = slot_ap_drain } }
				set_variable = { name = pod_combat_actor_ap_drain value = scope:slot_story.var:slot_ap_drain }
			}
			else = { remove_variable = pod_combat_actor_ap_drain }
			# Retaliation stances
			if = {
				limit = { scope:slot_story = { has_variable = slot_retaliate_hit } }
				set_variable = { name = pod_combat_actor_retaliate_hit value = scope:slot_story.var:slot_retaliate_hit }
			}
			else = { remove_variable = pod_combat_actor_retaliate_hit }
			if = {
				limit = { scope:slot_story = { has_variable = slot_retaliate_hit_agg } }
				set_variable = { name = pod_combat_actor_retaliate_hit_agg value = yes }
			}
			else = { remove_variable = pod_combat_actor_retaliate_hit_agg }
			if = {
				limit = { scope:slot_story = { has_variable = slot_retaliate_miss } }
				set_variable = { name = pod_combat_actor_retaliate_miss value = scope:slot_story.var:slot_retaliate_miss }
			}
			else = { remove_variable = pod_combat_actor_retaliate_miss }
			if = {
				limit = { scope:slot_story = { has_variable = slot_retaliate_miss_agg } }
				set_variable = { name = pod_combat_actor_retaliate_miss_agg value = yes }
			}
			else = { remove_variable = pod_combat_actor_retaliate_miss_agg }
			# Tier tracking vars (slot -> player workspace)
			if = {
				limit = { scope:slot_story = { has_variable = slot_soak_buff_round_tier } }
				set_variable = { name = pod_combat_actor_soak_buff_round_tier value = scope:slot_story.var:slot_soak_buff_round_tier }
			}
			else = { remove_variable = pod_combat_actor_soak_buff_round_tier }
			if = {
				limit = { scope:slot_story = { has_variable = slot_soak_buff_combat_tier } }
				set_variable = { name = pod_combat_actor_soak_buff_combat_tier value = scope:slot_story.var:slot_soak_buff_combat_tier }
			}
			else = { remove_variable = pod_combat_actor_soak_buff_combat_tier }
			if = {
				limit = { scope:slot_story = { has_variable = slot_soak_debuff_round_tier } }
				set_variable = { name = pod_combat_actor_soak_debuff_round_tier value = scope:slot_story.var:slot_soak_debuff_round_tier }
			}
			else = { remove_variable = pod_combat_actor_soak_debuff_round_tier }
			if = {
				limit = { scope:slot_story = { has_variable = slot_soak_debuff_combat_tier } }
				set_variable = { name = pod_combat_actor_soak_debuff_combat_tier value = scope:slot_story.var:slot_soak_debuff_combat_tier }
			}
			else = { remove_variable = pod_combat_actor_soak_debuff_combat_tier }
			if = {
				limit = { scope:slot_story = { has_variable = slot_hit_buff_round_tier } }
				set_variable = { name = pod_combat_actor_hit_buff_round_tier value = scope:slot_story.var:slot_hit_buff_round_tier }
			}
			else = { remove_variable = pod_combat_actor_hit_buff_round_tier }
			if = {
				limit = { scope:slot_story = { has_variable = slot_hit_buff_combat_tier } }
				set_variable = { name = pod_combat_actor_hit_buff_combat_tier value = scope:slot_story.var:slot_hit_buff_combat_tier }
			}
			else = { remove_variable = pod_combat_actor_hit_buff_combat_tier }
			if = {
				limit = { scope:slot_story = { has_variable = slot_hit_debuff_round_tier } }
				set_variable = { name = pod_combat_actor_hit_debuff_round_tier value = scope:slot_story.var:slot_hit_debuff_round_tier }
			}
			else = { remove_variable = pod_combat_actor_hit_debuff_round_tier }
			if = {
				limit = { scope:slot_story = { has_variable = slot_hit_debuff_combat_tier } }
				set_variable = { name = pod_combat_actor_hit_debuff_combat_tier value = scope:slot_story.var:slot_hit_debuff_combat_tier }
			}
			else = { remove_variable = pod_combat_actor_hit_debuff_combat_tier }
			# Actor character reference. Read by:
			#   * The AI selection effect to run trait gates against the
			#     current actor (e.g. has_trait = fera) so enemy AIs only
			#     pick moves they actually qualify for.
			#   * The damage pipeline as var:pod_combat_$TARGET_SIDE$_char
			#     when TARGET_SIDE="actor" (single-target self-damage,
			#     retaliation against the original attacker, etc.).
			set_variable = { name = pod_combat_actor_char value = scope:slot_story.var:slot_char }
		}
	}
}

# Caller scope = the slot story; caller must have saved the workspace
# owner as scope:host before delegating. Reads the slot's vars
# directly (current scope = slot story) and writes pod_combat_target_*
# on scope:host.
#
# The save-then-re-enter dance lets us read slot fields via
# scope:slot_story.var:slot_X from inside scope:host. Without it,
# every read inside scope:host = { ... } would resolve against host
# scope and the slot fields wouldn't be visible.
#
# Called from every_in_list iterations on pod_combat_active_slots in
# the AoE dispatchers and the workspace dispatch (where the iteration
# body's current scope IS the slot story), and from
# pod_combat_select_target / pod_combat_ai_swap_to_pc_target_effect for
# their single-slot host→story flips.
#
# MIRROR of pod_combat_load_actor_from_story_effect (above) same
# field-by-field copy, writes pod_combat_target_* instead of
# pod_combat_actor_*. If you add a field to one, add it to both.
pod_combat_load_target_from_story_effect = {
	if = {
		limit = { has_variable = slot_char }
		save_scope_as = slot_story
		scope:host = {
			set_variable = { name = pod_combat_target_soak value = scope:slot_story.var:slot_soak }
			set_variable = { name = pod_combat_target_base_soak value = scope:slot_story.var:slot_base_soak }
			if = { limit = { scope:slot_story = { has_variable = slot_base_hit_chance } } set_variable = { name = pod_combat_target_base_hit_chance value = scope:slot_story.var:slot_base_hit_chance } }
			set_variable = { name = pod_combat_target_morale value = scope:slot_story.var:slot_morale }
			set_variable = { name = pod_combat_target_prowess value = scope:slot_story.var:slot_prowess }
			set_variable = { name = pod_combat_target_hit_chance value = scope:slot_story.var:slot_hit_chance }
			# Target-side slot_energy_used is not mirrored — the AI energy-
			# attrition weighting only reads the actor side.
			if = {
				limit = { scope:slot_story = { has_variable = slot_is_guarding } }
				set_variable = { name = pod_combat_target_is_guarding value = yes }
			}
			else = {
				remove_variable = pod_combat_target_is_guarding
			}
			if = {
				limit = { scope:slot_story = { has_variable = slot_is_dodging } }
				set_variable = { name = pod_combat_target_is_dodging value = yes }
			}
			else = {
				remove_variable = pod_combat_target_is_dodging
			}
			if = {
				limit = { scope:slot_story = { has_variable = slot_is_stealthed } }
				set_variable = { name = pod_combat_target_is_stealthed value = yes }
			}
			else = {
				remove_variable = pod_combat_target_is_stealthed
			}
			# Incorporeality on the target side — read by the spirit gate
			# inside pod_combat_resolve_contact_effect when the move
			# carries the checks_spirit_gate capability flag (default-on
			# for physical attack_type, opt-in for supernatural). The
			# attack auto-misses when this flag is set and the actor
			# lacks pod_combat_$ACTOR_SIDE$_can_damage_spirit (mirroring
			# PoD's POD_damage_spirit_custom_trigger).
			if = {
				limit = { scope:slot_story = { has_variable = slot_incorporeal_spirit } }
				set_variable = { name = pod_combat_target_incorporeal_spirit value = yes }
			}
			else = {
				remove_variable = pod_combat_target_incorporeal_spirit
			}
			# Target's spirit-damage capability — read by the retaliation
			# spirit gate in pod_combat_apply_retaliation_helper, where the
			# target side becomes the retaliator and must have this flag to
			# land a counter-strike on an incorporeal original-attacker.
			if = {
				limit = { scope:slot_story = { has_variable = slot_can_damage_spirit } }
				set_variable = { name = pod_combat_target_can_damage_spirit value = yes }
			}
			else = {
				remove_variable = pod_combat_target_can_damage_spirit
			}
			# Initiative bonus and AP drain are not mirrored on the target
			# side. Both were write-only mirrors: a single move-execution
			# write site (POD_execute_combat_move §14) would stage the
			# value for save_workspace to push to the slot, but nothing on
			# the target side ever read them — so the §14 writes now hit
			# the target slot directly via the active_slots iterator.
			# Retaliation stances on the target side
			if = {
				limit = { scope:slot_story = { has_variable = slot_retaliate_hit } }
				set_variable = { name = pod_combat_target_retaliate_hit value = scope:slot_story.var:slot_retaliate_hit }
			}
			else = { remove_variable = pod_combat_target_retaliate_hit }
			if = {
				limit = { scope:slot_story = { has_variable = slot_retaliate_hit_agg } }
				set_variable = { name = pod_combat_target_retaliate_hit_agg value = yes }
			}
			else = { remove_variable = pod_combat_target_retaliate_hit_agg }
			if = {
				limit = { scope:slot_story = { has_variable = slot_retaliate_miss } }
				set_variable = { name = pod_combat_target_retaliate_miss value = scope:slot_story.var:slot_retaliate_miss }
			}
			else = { remove_variable = pod_combat_target_retaliate_miss }
			if = {
				limit = { scope:slot_story = { has_variable = slot_retaliate_miss_agg } }
				set_variable = { name = pod_combat_target_retaliate_miss_agg value = yes }
			}
			else = { remove_variable = pod_combat_target_retaliate_miss_agg }
			# Tier tracking vars (slot -> target workspace)
			if = {
				limit = { scope:slot_story = { has_variable = slot_soak_buff_round_tier } }
				set_variable = { name = pod_combat_target_soak_buff_round_tier value = scope:slot_story.var:slot_soak_buff_round_tier }
			}
			else = { remove_variable = pod_combat_target_soak_buff_round_tier }
			if = {
				limit = { scope:slot_story = { has_variable = slot_soak_buff_combat_tier } }
				set_variable = { name = pod_combat_target_soak_buff_combat_tier value = scope:slot_story.var:slot_soak_buff_combat_tier }
			}
			else = { remove_variable = pod_combat_target_soak_buff_combat_tier }
			if = {
				limit = { scope:slot_story = { has_variable = slot_soak_debuff_round_tier } }
				set_variable = { name = pod_combat_target_soak_debuff_round_tier value = scope:slot_story.var:slot_soak_debuff_round_tier }
			}
			else = { remove_variable = pod_combat_target_soak_debuff_round_tier }
			if = {
				limit = { scope:slot_story = { has_variable = slot_soak_debuff_combat_tier } }
				set_variable = { name = pod_combat_target_soak_debuff_combat_tier value = scope:slot_story.var:slot_soak_debuff_combat_tier }
			}
			else = { remove_variable = pod_combat_target_soak_debuff_combat_tier }
			if = {
				limit = { scope:slot_story = { has_variable = slot_hit_buff_round_tier } }
				set_variable = { name = pod_combat_target_hit_buff_round_tier value = scope:slot_story.var:slot_hit_buff_round_tier }
			}
			else = { remove_variable = pod_combat_target_hit_buff_round_tier }
			if = {
				limit = { scope:slot_story = { has_variable = slot_hit_buff_combat_tier } }
				set_variable = { name = pod_combat_target_hit_buff_combat_tier value = scope:slot_story.var:slot_hit_buff_combat_tier }
			}
			else = { remove_variable = pod_combat_target_hit_buff_combat_tier }
			if = {
				limit = { scope:slot_story = { has_variable = slot_hit_debuff_round_tier } }
				set_variable = { name = pod_combat_target_hit_debuff_round_tier value = scope:slot_story.var:slot_hit_debuff_round_tier }
			}
			else = { remove_variable = pod_combat_target_hit_debuff_round_tier }
			if = {
				limit = { scope:slot_story = { has_variable = slot_hit_debuff_combat_tier } }
				set_variable = { name = pod_combat_target_hit_debuff_combat_tier value = scope:slot_story.var:slot_hit_debuff_combat_tier }
			}
			else = { remove_variable = pod_combat_target_hit_debuff_combat_tier }
			# Alias matching $TARGET_SIDE$ = "target" — parallels
			# pod_combat_actor_char (see actor-load above). Lets the damage
			# pipeline reference var:pod_combat_$TARGET_SIDE$_char uniformly.
			set_variable = { name = pod_combat_target_char value = scope:slot_story.var:slot_char }

			# Refresh workspace hit chances for the new actor/target pair.
			# Slot-stored hit chance is leader-baseline (computed once at
			# combat init in pod_combat_calculate_all_hit_chances_effect),
			# so without this every AI swap, AoE per-target iteration, and
			# workspace-load would push the wrong number into the damage
			# roll. The pair recalc reads the workspace's pod_combat_actor_prowess
			# (set by the actor load that runs before this) and the
			# pod_combat_target_prowess we just wrote above, then layers
			# the active hit-tier modifiers via the per-side recalc
			# effects. No slot writeback — that's intentional, see the
			# comment on the recalc effect itself in
			# pod_combat_tier_effects.txt.
			pod_combat_recalc_pair_hit_chance_effect = yes
		}
	}
}

##############################################################################################
# SECTION 2: WORKSPACE SAVE EFFECTS
# Two effects that copy the actor or target workspace back into a slot story.
##############################################################################################

# Caller scope = slot story; caller must have saved the workspace
# owner as scope:host. Reads workspace vars on scope:host and writes
# the slot story's vars directly.
#
# Mirrors pod_combat_save_target_to_story_effect but reads the actor
# side (pod_combat_actor_*). Called from
# pod_combat_save_workspace_effect's iteration over pod_combat_active_slots.
pod_combat_save_actor_to_story_effect = {
	if = {
		limit = { scope:host = { has_variable = pod_combat_actor_char } }
		# Alive-flip on death / incapacitation. Reads slot_char from
		# current scope (slot story); writes slot_alive on current scope.
		if = {
			limit = {
				OR = {
					var:slot_char = { is_alive = no }
					var:slot_char = { pod_combat_is_incapacitated_trigger = yes }
				}
			}
			set_variable = { name = slot_alive value = 0 }
		}
		# Workspace -> slot writes. The slot story is current scope so
		# writes are direct `set_variable = { name = slot_X ... }`;
		# reads come from scope:host.var:pod_combat_actor_X.
		set_variable = { name = slot_soak           value = scope:host.var:pod_combat_actor_soak }
		set_variable = { name = slot_base_soak      value = scope:host.var:pod_combat_actor_base_soak }
		if = { limit = { scope:host = { has_variable = pod_combat_actor_base_hit_chance } } set_variable = { name = slot_base_hit_chance value = scope:host.var:pod_combat_actor_base_hit_chance } }
		set_variable = { name = slot_morale         value = scope:host.var:pod_combat_actor_morale }
		set_variable = { name = slot_prowess        value = scope:host.var:pod_combat_actor_prowess }
		set_variable = { name = slot_hit_chance     value = scope:host.var:pod_combat_actor_hit_chance }
		if = {
			limit = { scope:host = { has_variable = pod_combat_actor_energy_used } }
			set_variable = { name = slot_energy_used value = scope:host.var:pod_combat_actor_energy_used }
		}
		if = {
			limit = { scope:host = { has_variable = pod_combat_actor_is_guarding } }
			set_variable    = { name = slot_is_guarding value = yes }
		}
		else = {
			remove_variable = slot_is_guarding
		}
		if = {
			limit = { scope:host = { has_variable = pod_combat_actor_is_dodging } }
			set_variable    = { name = slot_is_dodging value = yes }
		}
		else = {
			remove_variable = slot_is_dodging
		}
		if = {
			limit = { scope:host = { has_variable = pod_combat_actor_is_stealthed } }
			set_variable    = { name = slot_is_stealthed value = yes }
		}
		else = {
			remove_variable = slot_is_stealthed
		}
		# slot_init_bonus is not mirrored on the actor side — no
		# workspace reader. The slot value is updated directly by the
		# initiative-debuff move payload (which targets the enemy
		# slot) and consumed by the round-end initiative recalc;
		# neither path touches the workspace.
		#
		# AP drain persists across workspace cycles via the actor mirror
		# (one reader: pod_combat_apply_ap_drain_effect at turn start).
		if = {
			limit = { scope:host = { has_variable = pod_combat_actor_ap_drain } }
			set_variable    = { name = slot_ap_drain   value = scope:host.var:pod_combat_actor_ap_drain }
		}
		else = { remove_variable = slot_ap_drain }

# ─── AOE TARGET-ALIVE CHECK ───────────────────────────────────────
# Post-iteration alive-flip used by the damage-side AoE dispatchers
# in pod_combat_move_effects.txt. Runs after pod_combat_save_target_to_story_effect
# has written the slot's freshly-resolved combat state, so this can
# read both the character's PoD wound state AND the just-written
# slot_morale value.
#
# Flips slot_alive = 0 when ANY of:
#   * The character died (is_alive = no — fatal duel + lethal blow).
#   * The character is incapacitated (PoD's torpor / wight / dharma0 /
#     shadow / zombu / incapable predicate).
#   * Combat morale broke (slot_morale <= 0). This is the only check
#     pod_combat_save_target_to_story_effect's internal alive-check
#     can't make pre-write — slot_morale isn't set until that effect
#     writes the workspace value to the slot.
#   * The wound track is full (any combination of superficial +
#     aggravated reaching track length). Catches combat-HP exhaustion
#     in non-fatal duels where PoD's death branch is gated and is_alive
#     stays yes.
#
# Called from per-target body inside the AoE iterators in
# pod_combat_move_effects.txt — implicit scope must be the iterated
# slot story (var:slot_morale + var:slot_char live there).
#
# Centralised here so a future incapacitation clause only needs to be
# added in one site — the AoE iterators in pod_combat_move_effects.txt
# both call this rather than each inlining the alive checks.
pod_combat_aoe_check_target_alive_effect = {
	if = {
		limit = {
			OR = {
				var:slot_char = { is_alive = no }
				var:slot_char = { pod_combat_is_incapacitated_trigger = yes }
				var:slot_morale <= 0
				var:slot_char = { POD_total_damage_boxes >= POD_health_track_length }
			}
		}
		set_variable = { name = slot_alive value = 0 }
	}
}


# Caller scope = slot story; caller must have saved the workspace
# owner as scope:host. Reads workspace vars on scope:host and writes
# the slot story's vars directly.
#
# Called from every_in_list iterations on pod_combat_active_slots in
# the AoE dispatchers and the workspace dispatch
# (pod_combat_save_workspace_effect), and from the AI swap-target
# effects' restore path (pod_combat_ai_restore_enemy_target_effect).
pod_combat_save_target_to_story_effect = {
	if = {
		limit = { scope:host = { has_variable = pod_combat_target_char } }
		# Alive-flip on death / incapacitation / wound-track-full.
		# The wound-track-full clause is what catches combat-HP
		# exhaustion in non-fatal duels (where PoD's death check is
		# gated, so is_alive stays yes and the trait checks won't
		# fire) — without it those targets would stay marked alive
		# even with a full wound track. In fatal duels it's redundant
		# with is_alive=no for mortal characters but still useful for
		# the brief window between wound-track-fill and PoD's death
		# check resolving. Reads slot_char from current scope (slot
		# story); writes slot_alive on current scope.
		if = {
			limit = {
				OR = {
					var:slot_char = { is_alive = no }
					var:slot_char = { pod_combat_is_incapacitated_trigger = yes }
					var:slot_char = { POD_total_damage_boxes >= POD_health_track_length }
					# Morale-broken target — added for parity with the
					# other three KO check sites (init-time KO check,
					# save_workspace's actor + target passes, and
					# aoe_check_target_alive). The save_target path was
					# missing this clause; without it, a morale-only
					# attack that reduced a target to slot_morale = 0
					# wouldn't flip slot_alive until the next save tick
					# touched the slot through one of the other paths.
					# Reads workspace mirror (pod_combat_target_morale)
					# instead of var:slot_morale because we haven't
					# written the new morale into the slot yet — the
					# workspace value IS the post-attack number.
					scope:host.var:pod_combat_target_morale <= 0
				}
			}
			set_variable = { name = slot_alive value = 0 }
		}
		# Workspace -> slot writes. The slot story is current scope so
		# writes are direct `set_variable = { name = slot_X ... }`;
		# reads come from scope:host.var:pod_combat_target_X.
		set_variable = { name = slot_soak           value = scope:host.var:pod_combat_target_soak }
		set_variable = { name = slot_base_soak      value = scope:host.var:pod_combat_target_base_soak }
		if = { limit = { scope:host = { has_variable = pod_combat_target_base_hit_chance } } set_variable = { name = slot_base_hit_chance value = scope:host.var:pod_combat_target_base_hit_chance } }
		set_variable = { name = slot_morale         value = scope:host.var:pod_combat_target_morale }
		set_variable = { name = slot_prowess        value = scope:host.var:pod_combat_target_prowess }
		set_variable = { name = slot_hit_chance     value = scope:host.var:pod_combat_target_hit_chance }
		# slot_energy_used is not written on the target side — the AI
		# energy-attrition counter is updated only when an actor SPENDS
		# energy on their own move (via pod_combat_actor_energy_used in
		# POD_execute_combat_move §3). The target-side workspace mirror
		# is therefore absent (write-only would have no consumer).
		if = {
			limit = { scope:host = { has_variable = pod_combat_target_is_guarding } }
			set_variable    = { name = slot_is_guarding value = yes }
		}
		else = {
			remove_variable = slot_is_guarding
		}
		if = {
			limit = { scope:host = { has_variable = pod_combat_target_is_dodging } }
			set_variable    = { name = slot_is_dodging value = yes }
		}
		else = {
			remove_variable = slot_is_dodging
		}
		if = {
			limit = { scope:host = { has_variable = pod_combat_target_is_stealthed } }
			set_variable    = { name = slot_is_stealthed value = yes }
		}
		else = {
			remove_variable = slot_is_stealthed
		}
		# Initiative bonus and AP drain are not mirrored on the target
		# side — both would be write-only with no reader.
		# POD_execute_combat_move §14 writes slot_init_bonus /
		# slot_ap_drain directly on the target slot via an active_slots
		# lookup, so save_target has nothing to push back here.
		# Retaliation stances
		if = {
			limit = { scope:host = { has_variable = pod_combat_target_retaliate_hit } }
			set_variable    = { name = slot_retaliate_hit value = scope:host.var:pod_combat_target_retaliate_hit }
		}
		else = { remove_variable = slot_retaliate_hit }
		if = {
			limit = { scope:host = { has_variable = pod_combat_target_retaliate_hit_agg } }
			set_variable    = { name = slot_retaliate_hit_agg value = yes }
		}
		else = { remove_variable = slot_retaliate_hit_agg }
		if = {
			limit = { scope:host = { has_variable = pod_combat_target_retaliate_miss } }
			set_variable    = { name = slot_retaliate_miss value = scope:host.var:pod_combat_target_retaliate_miss }
		}
		else = { remove_variable = slot_retaliate_miss }
		if = {
			limit = { scope:host = { has_variable = pod_combat_target_retaliate_miss_agg } }
			set_variable    = { name = slot_retaliate_miss_agg value = yes }
		}
		else = { remove_variable = slot_retaliate_miss_agg }
		# Tier tracking vars (target workspace -> slot)
		if = {
			limit = { scope:host = { has_variable = pod_combat_target_soak_buff_round_tier } }
			set_variable    = { name = slot_soak_buff_round_tier    value = scope:host.var:pod_combat_target_soak_buff_round_tier }
		}
		else = { remove_variable = slot_soak_buff_round_tier    }
		if = {
			limit = { scope:host = { has_variable = pod_combat_target_soak_buff_combat_tier } }
			set_variable    = { name = slot_soak_buff_combat_tier   value = scope:host.var:pod_combat_target_soak_buff_combat_tier }
		}
		else = { remove_variable = slot_soak_buff_combat_tier   }
		if = {
			limit = { scope:host = { has_variable = pod_combat_target_soak_debuff_round_tier } }
			set_variable    = { name = slot_soak_debuff_round_tier  value = scope:host.var:pod_combat_target_soak_debuff_round_tier }
		}
		else = { remove_variable = slot_soak_debuff_round_tier  }
		if = {
			limit = { scope:host = { has_variable = pod_combat_target_soak_debuff_combat_tier } }
			set_variable    = { name = slot_soak_debuff_combat_tier value = scope:host.var:pod_combat_target_soak_debuff_combat_tier }
		}
		else = { remove_variable = slot_soak_debuff_combat_tier }
		if = {
			limit = { scope:host = { has_variable = pod_combat_target_hit_buff_round_tier } }
			set_variable    = { name = slot_hit_buff_round_tier     value = scope:host.var:pod_combat_target_hit_buff_round_tier }
		}
		else = { remove_variable = slot_hit_buff_round_tier     }
		if = {
			limit = { scope:host = { has_variable = pod_combat_target_hit_buff_combat_tier } }
			set_variable    = { name = slot_hit_buff_combat_tier    value = scope:host.var:pod_combat_target_hit_buff_combat_tier }
		}
		else = { remove_variable = slot_hit_buff_combat_tier    }
		if = {
			limit = { scope:host = { has_variable = pod_combat_target_hit_debuff_round_tier } }
			set_variable    = { name = slot_hit_debuff_round_tier   value = scope:host.var:pod_combat_target_hit_debuff_round_tier }
		}
		else = { remove_variable = slot_hit_debuff_round_tier   }
		if = {
			limit = { scope:host = { has_variable = pod_combat_target_hit_debuff_combat_tier } }
			set_variable    = { name = slot_hit_debuff_combat_tier  value = scope:host.var:pod_combat_target_hit_debuff_combat_tier }
		}
		else = { remove_variable = slot_hit_debuff_combat_tier  }
	}
}

##############################################################################################
# SECTION 3: WORKSPACE DISPATCH EFFECTS
# Switch on pod_combat_current_actor / pod_combat_selected_target to call the correct
# parametric load/save effect for the appropriate slot.
##############################################################################################

pod_combat_load_workspace_effect = {
	# Load current actor → actor workspace, then selected target →
	# target workspace, by iterating pod_combat_active_slots and
	# matching slot_index against pod_combat_current_actor /
	# pod_combat_selected_target. The iteration limit fires for at
	# most one slot per pass — the load helper inside the body is the
	# story-scoped variant, which writes to scope:host (this scope).
	save_scope_as = host
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			scope:host = { has_variable = pod_combat_current_actor }
			var:slot_index = scope:host.var:pod_combat_current_actor
		}
		pod_combat_load_actor_from_story_effect = yes
	}
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			scope:host = { has_variable = pod_combat_selected_target }
			var:slot_index = scope:host.var:pod_combat_selected_target
		}
		pod_combat_load_target_from_story_effect = yes
	}

	# Push the freshly pair-recomputed hit chances back to the actor's
	# and target's slot stories so the per-slot GUI display matches the
	# values that will actually drive the upcoming damage roll. Without
	# this, slot_hit_chance keeps whatever the previous turn's
	# save_workspace / save_target wrote — most visibly: an AI attacking
	# the PC saves "PC vs that AI" into slot 0, then PC turn start runs
	# this effect (workspace gets pair-refreshed against PC's selected
	# target via pod_combat_recalc_pair_hit_chance_effect inside
	# load_target), but slot 0 stays on the AI-vs-PC value until the
	# player clicks an attack and the post-move save_workspace finally
	# writes the live workspace back. Result was the displayed hit chance
	# changing the moment the player took their move, even though only
	# the slot's display was catching up. Symmetric with the writeback
	# inside save_actor / save_target — same flow, just at the load
	# tick instead of the save tick. Single-pair only (no AoE concerns)
	# because the AoE iterators call pod_combat_load_target_from_story_effect
	# directly per target, bypassing this dispatcher entirely.
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			scope:host = { has_variable = pod_combat_current_actor }
			var:slot_index = scope:host.var:pod_combat_current_actor
		}
		if = {
			limit = { scope:host = { has_variable = pod_combat_actor_base_hit_chance } }
			set_variable = { name = slot_base_hit_chance value = scope:host.var:pod_combat_actor_base_hit_chance }
		}
		if = {
			limit = { scope:host = { has_variable = pod_combat_actor_hit_chance } }
			set_variable = { name = slot_hit_chance value = scope:host.var:pod_combat_actor_hit_chance }
		}
	}
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			scope:host = { has_variable = pod_combat_selected_target }
			var:slot_index = scope:host.var:pod_combat_selected_target
		}
		if = {
			limit = { scope:host = { has_variable = pod_combat_target_base_hit_chance } }
			set_variable = { name = slot_base_hit_chance value = scope:host.var:pod_combat_target_base_hit_chance }
		}
		if = {
			limit = { scope:host = { has_variable = pod_combat_target_hit_chance } }
			set_variable = { name = slot_hit_chance value = scope:host.var:pod_combat_target_hit_chance }
		}
	}

	# Set the current_hit_chance for the damage pipeline
	if = {
		limit = { has_variable = pod_combat_actor_hit_chance }
		set_variable = { name = pod_combat_current_hit_chance value = var:pod_combat_actor_hit_chance }
	}
}

# Stamp pod_combat_nonfatal_active on every active combatant's char
# so POD_can_die_from_wounds returns no while combat is active. Paired

pod_combat_apply_nonfatal_flag_effect = {

		}
	}

# Strip pod_combat_nonfatal_active after the post-combat wound cap

pod_combat_clear_nonfatal_flag_effect = {
	every_in_list = {
		variable = pod_combat_active_slots

			remove_character_flag = pod_combat_nonfatal_active
		}
	}

pod_combat_save_workspace_effect = {
	# Mirror image of pod_combat_load_workspace_effect: iterate
	# pod_combat_active_slots and write the actor / target workspace
	# back to whichever slot's slot_index matches pod_combat_current_actor
	# / pod_combat_selected_target. 
	save_scope_as = host
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			scope:host = { has_variable = pod_combat_current_actor }
			var:slot_index = scope:host.var:pod_combat_current_actor
		}
		pod_combat_save_actor_to_story_effect = yes
	}
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			scope:host = { has_variable = pod_combat_selected_target }
			var:slot_index = scope:host.var:pod_combat_selected_target
		}
		pod_combat_save_target_to_story_effect = yes
	}

	# KO check on the action's target. Same OR'd KO conditions across
	# four sources (PoD is_alive, pod_combat_is_incapacitated_trigger,
	# slot_morale ≤ 0, wound-track-full) see comment block above for
	# rationale. Iterates pod_combat_active_slots and flips slot_alive=0
	# only on the slot whose slot_index matches selected_target. 
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			scope:host = { has_variable = pod_combat_selected_target }
			var:slot_index = scope:host.var:pod_combat_selected_target
			OR = {
				var:slot_char = { is_alive = no }
				var:slot_char = { pod_combat_is_incapacitated_trigger = yes }
				var:slot_morale <= 0
				var:slot_char = { POD_total_damage_boxes >= POD_health_track_length }
			}
		}
		set_variable = { name = slot_alive value = 0 }
	}

	# KO check on the actor (counter-damage during their own action
	# a retaliating target's morale damage / chip damage can finish them
	# off). Same KO predicates as above; iterates the same list.
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			scope:host = { has_variable = pod_combat_current_actor }
			var:slot_index = scope:host.var:pod_combat_current_actor
			OR = {
				var:slot_char = { is_alive = no }
				var:slot_char = { pod_combat_is_incapacitated_trigger = yes }
				var:slot_morale <= 0
				var:slot_char = { POD_total_damage_boxes >= POD_health_track_length }
			}
		}
		set_variable = { name = slot_alive value = 0 }
	}

	# Victory check at the post-flush boundary. Slot _alive flags and
	# morale are fresh here (the workspace just flushed back to slot
	# vars), so team-wipe detection works. This is one of two call
	# sites — the other is at the end of pod_combat_execute_ai_turn_effect
	# (covering AI action boundaries). Per-damage victory checks are
	# NOT used: an AoE move would fire the check after each victim's
	# damage call, and a multi-victim wipe would emit one cleanup per
	# kill. Checking only at action boundaries keeps cleanup atomic.
	# The check is guarded on NOT = pod_combat_is_over so cascading
	# calls (AoE save_workspace) are idempotent.
	pod_combat_check_group_victory_effect = yes
}

# SECTION 4: SLOT INITIALIZATION FROM CHARACTER
# Reads a character's health track, soak, and prowess and stores them in slot variables.
##############################################################################################

pod_combat_init_slot_from_character_effect = {
	# $SLOT$ = 0-9, $CHARACTER$ = character scope
	# Applies the discipline combat buffs and reads health/soak/prowess
	# into the slot variables.
	#
	# Shapeshifting for AI fera is NOT done here — it runs upfront in
	# pod_combat_initialize_effect as a pre-pass, so every potential
	# CASTER already has its crinos modifier when the discipline effects
	# query it (especially for the delirium check, which reads
	# has_character_modifier on the CASTER).
	#
	# Discipline buff CASTER assignment:
	#   Slots 0-4 (player team) → scope:pod_combat_target  (enemy leader)
	#   Slots 5-9 (enemy team)  → scope:combat_pc          (the PC)
	# Ally buff uses the character themselves as CASTER.
	$CHARACTER$ = {
		# Discipline combat buffs. Uses named scopes (not root) to match
		# PoD's own duel pattern and avoid text-substitution issues in
		# nested parametric effects.
		if = {
			limit = { $SLOT$ <= 4 }
			# Player team — enemy leader is the opponent passed into group init
			POD_enemy_knight_set_discipline_battle_effect = { CASTER = scope:pod_combat_target }
		}
		else = {
			# Enemy team — enemy leader is the PC
			POD_enemy_knight_set_discipline_battle_effect = { CASTER = scope:combat_pc }
		}
		POD_ally_knight_set_discipline_battle_effect = { CASTER = $CHARACTER$ }
		force_character_skill_recalculation = yes
	}
	# Read soak and prowess from the character and snapshot them as
	# saved scopes so the parametric pod_combat_set_slot_var_effect
	# calls below can pass them as VALUE without re-resolving against
	# the slot story scope. HP / wound state isn't snapshotted
	# pod_combat_slot_max_hp_value / _total_damage_sv / _agg_damage_sv
	# read live from the participant's PoD wound vars at GUI render
	# time.
	$CHARACTER$ = {
		save_scope_value_as = {
			name = slot_soak
			value = POD_soak_chance
		}
		save_scope_value_as = {
			name = slot_prowess
			value = prowess
		}
	}

	pod_combat_set_slot_var_effect = { SLOT = $SLOT$ NAME = soak      VALUE = scope:slot_soak }
	pod_combat_set_slot_var_effect = { SLOT = $SLOT$ NAME = base_soak VALUE = scope:slot_soak }
	pod_combat_set_slot_var_effect = { SLOT = $SLOT$ NAME = morale    VALUE = pod_combat_max_morale }
	pod_combat_set_slot_var_effect = { SLOT = $SLOT$ NAME = prowess   VALUE = scope:slot_prowess }
	pod_combat_set_slot_alive_effect = { SLOT = $SLOT$ VALUE = 1 }
	# AI energy-use counter: increments on every energy-cost skill this
	# combatant fires. The AI move picker layers stacking ADDITIVE weight
	# penalties on top of this counter (-8 / -10 / -12 / -14 at thresholds
	# 1 / 2 / 3 / 4+, cumulating to 0 / -8 / -18 / -30 / -44) to bias
	# selection AWAY from energy spends as the pool depletes. The counter
	# only drives selection bias; the AI still pays the actual splat
	# energy on top, via the same POD_spend_splat_agnostic_energy path
	# the PC uses.
	pod_combat_set_slot_var_effect = { SLOT = $SLOT$ NAME = energy_used VALUE = 0 }
	# ── Spirit / incorporeality capability flags ─────────────────
	# Mirrors PoD's POD_damage_spirit_custom_trigger semantics for the
	# minigame. Two per-slot flags are captured once at init and never
	# change during combat:
	#   incorporeal_spirit: target is an incorporeal spirit (ghost,
	#       Umbral entity, non-Risen wraith, etc.) that can't be harmed
	#       by mundane physical attacks. Checked via PoD's own
	#       POD_can_be_killed_physically_trigger = no.
	#   can_damage_spirit: actor has the supernatural means to strike
	#       incorporeal spirits (fera/spirit/fae/wraith/true_faith, or
	#       specific perks / artifacts). Checked via PoD's own
	#       POD_can_damage_spirit_trigger = yes.
	# These gate the spirit-capability check inside
	# pod_combat_resolve_contact_effect. The check fires when the move's
	# checks_spirit_gate capability flag is set (default-on for physical
	# attack_type, opt-in for supernatural moves that should respect
	# incorporeality). If the target is incorporeal and the actor lacks
	# pod_combat_$ACTOR_SIDE$_can_damage_spirit, the attack auto-misses.
	# Moves without checks_spirit_gate hit incorporeal targets normally.
	pod_combat_remove_slot_var_effect = { SLOT = $SLOT$ NAME = incorporeal_spirit }
	if = {
		limit = { $CHARACTER$ = { POD_can_be_killed_physically_trigger = no } }
		pod_combat_set_slot_var_effect = { SLOT = $SLOT$ NAME = incorporeal_spirit VALUE = yes }
	}
	pod_combat_remove_slot_var_effect = { SLOT = $SLOT$ NAME = can_damage_spirit }
	if = {
		limit = { $CHARACTER$ = { POD_can_damage_spirit_trigger = yes } }
		pod_combat_set_slot_var_effect = { SLOT = $SLOT$ NAME = can_damage_spirit VALUE = yes }
	}
	pod_combat_remove_slot_var_effect = { SLOT = $SLOT$ NAME = is_guarding }
	pod_combat_remove_slot_var_effect = { SLOT = $SLOT$ NAME = is_dodging }
	# ── AI per-fight transient state ─────────────────────────────
	# AI loadouts and anti-repetition history (last / prev / prev2
	# move) are per-fight scratch state. Clear them on slot init so
	# every fight seeds from the current move catalogue and starts
	# with empty history. PC slot (slot 0) is skipped. The player's
	# loadout is persistent equipment configured via the loadout
	# decision, and the PC has no AI history vars to clear.
	if = {
		limit = { $CHARACTER$ = { is_ai = yes } }
		$CHARACTER$ = {
			clear_variable_list = pod_combat_loadout
			remove_variable = pod_combat_ai_last_move
			remove_variable = pod_combat_ai_prev_move
			remove_variable = pod_combat_ai_prev2_move
		}
	}
	# ── State marker modifier ────────────────────────────────────
	# Apply pod_combat_duel_active to every combatant so other
	# systems (feeding, schemes, travel, activity events from
	# other mods) can gate on `has_character_modifier` and defer
	# firing mid-duel. Normal cleanup happens inside
	# pod_combat_duel_end_effect (scripted effect called from
	# pod_combat_end_combat_effect); `days = 30` is a safety net
	# for edge cases (crash, save corruption, mod conflict blocking
	# the inline cleanup). Combat minigames end in seconds of game
	# time, so 30 days never clips a real duel and keeps stale state
	# bounded if anything goes wrong with cleanup.
	$CHARACTER$ = {
		add_character_modifier = {
			modifier = pod_combat_duel_active
			days     = 30
		}
	}
	# Hit chance calculated later after all combatants are initialized
}

##############################################################################################
# SECTION 5: SLOT ASSIGNMENT EFFECTS
# Assign characters to the next available ally (1-4) or enemy (6-9) slot.
##############################################################################################

pod_combat_assign_ally_slot_effect = {
	# $CHARACTER$ = character to add to next available ally slot (1-4)
	# Team affiliation is derived from slot index — slots 0-4 = player
	# team, 5-9 = enemy team; not stored as a per-slot variable. No
	# pod_combat_actor_team_size counter is kept either; iterations
	# over pod_combat_active_slots filtered by slot_index handle every
	# reader that would have needed the count.
	#
	# Occupancy check: gate on `has_variable = pod_combat_slot_$N$`
	# rather than `var:pod_combat_slot_$N$ = { has_variable = slot_char }`.
	# Under lazy creation the pointer var doesn't exist until the slot
	# is first filled, and a raw `var:` lookup against an unset pointer
	# throws "Failed to fetch variable" + "Failed context switch" at
	# runtime. The pointer-var and slot_char presence are coextensive
	# (set together by pod_combat_set_slot_char_effect), so the
	# pointer-var check is sufficient.
	if = {
		limit = { NOT = { has_variable = pod_combat_slot_1 } }
		pod_combat_set_slot_char_effect = { SLOT = 1 CHARACTER = $CHARACTER$ }
		pod_combat_init_slot_from_character_effect = { SLOT = 1 CHARACTER = $CHARACTER$ }
	}
	else_if = {
		limit = { NOT = { has_variable = pod_combat_slot_2 } }
		pod_combat_set_slot_char_effect = { SLOT = 2 CHARACTER = $CHARACTER$ }
		pod_combat_init_slot_from_character_effect = { SLOT = 2 CHARACTER = $CHARACTER$ }
	}
	else_if = {
		limit = { NOT = { has_variable = pod_combat_slot_3 } }
		pod_combat_set_slot_char_effect = { SLOT = 3 CHARACTER = $CHARACTER$ }
		pod_combat_init_slot_from_character_effect = { SLOT = 3 CHARACTER = $CHARACTER$ }
	}
	else_if = {
		limit = { NOT = { has_variable = pod_combat_slot_4 } }
		pod_combat_set_slot_char_effect = { SLOT = 4 CHARACTER = $CHARACTER$ }
		pod_combat_init_slot_from_character_effect = { SLOT = 4 CHARACTER = $CHARACTER$ }
	}
}

pod_combat_assign_enemy_slot_effect = {
	# $CHARACTER$ = character to add to next available enemy slot (6-9)
	# pod_combat_enemy_team_size IS live. The assignment / fill-pass
	# logic in pod_combat_initialize_effect reads it to gate
	# how many enemies still fit. Per-slot _team is dead (computed
	# from slot index everywhere).
	#
	# Occupancy check: see pod_combat_assign_ally_slot_effect — same
	# rationale, the pointer-var check survives the not-yet-created
	# case that `var:pod_combat_slot_$N$` blows up on.
	if = {
		limit = { NOT = { has_variable = pod_combat_slot_6 } }
		pod_combat_set_slot_char_effect = { SLOT = 6 CHARACTER = $CHARACTER$ }
		pod_combat_init_slot_from_character_effect = { SLOT = 6 CHARACTER = $CHARACTER$ }
		change_variable = { name = pod_combat_enemy_team_size add = 1 }
	}
	else_if = {
		limit = { NOT = { has_variable = pod_combat_slot_7 } }
		pod_combat_set_slot_char_effect = { SLOT = 7 CHARACTER = $CHARACTER$ }
		pod_combat_init_slot_from_character_effect = { SLOT = 7 CHARACTER = $CHARACTER$ }
		change_variable = { name = pod_combat_enemy_team_size add = 1 }
	}
	else_if = {
		limit = { NOT = { has_variable = pod_combat_slot_8 } }
		pod_combat_set_slot_char_effect = { SLOT = 8 CHARACTER = $CHARACTER$ }
		pod_combat_init_slot_from_character_effect = { SLOT = 8 CHARACTER = $CHARACTER$ }
		change_variable = { name = pod_combat_enemy_team_size add = 1 }
	}
	else_if = {
		limit = { NOT = { has_variable = pod_combat_slot_9 } }
		pod_combat_set_slot_char_effect = { SLOT = 9 CHARACTER = $CHARACTER$ }
		pod_combat_init_slot_from_character_effect = { SLOT = 9 CHARACTER = $CHARACTER$ }
		change_variable = { name = pod_combat_enemy_team_size add = 1 }
	}
}

##############################################################################################
# SECTION 6: GROUP INITIALIZATION
# Sets up the full group combat: PC in slot 0, primary opponent in slot 5,
# allies in slots 1-4 (modes 2/3), enemy reinforcements in slots 6-9 (modes 1/3).
# Calculates hit chances and loads initial workspace.
##############################################################################################

pod_combat_initialize_effect = {
	# $OPPONENT$ = primary target character
	# $MODE$ = 0 (1v1), 1 (1vN), 2 (Nv1), 3 (NvN)

	$OPPONENT$ = { save_scope_as = pod_combat_target }

	# Save-compat version check — mirrors the pod_combat_configure_skills
	# decision's version reload. The configure decision was the only
	# version-check site, so a player loading an older save and entering
	# combat WITHOUT opening the configure window first would keep stale
	# move-story references in their loadout (and AI loadouts in
	# particular pull from the global catalogue, so AI selection would
	# see the old stories too). Running the check here closes that gap.
	# POD_reload_combat_moves clears the global list, rebuilds every
	# story from the generator-emitted POD_initialize_combat_moves, and
	# wipes every character's pod_combat_loadout — the empty-loadout
	# fallback below then reseeds the PC with starter moves.
	hidden_effect = {
		if = {
			limit = {
				OR = {
					NOT = { exists = global_var:POD_combat_version }
					global_var:POD_combat_version != POD_current_combat_version
				}
			}
			POD_reload_combat_moves = yes
		}
	}

	# Default loadout if no moves equipped (list-based after the refactor;
	# POD_apply_default_combat_loadout iterates POD_combat_moves and adds
	# every story with is_starter_move set).
	if = {
		limit = { pod_combat_no_moves_equipped_trigger = yes }
		POD_apply_default_combat_loadout = yes
	}

	# Close config window if open
	remove_variable = pod_combat_config_open

	# Defensive cleanup for stale combat state: a prior combat that
	# crashed, was force-quit, or left state via cleanup regression
	# would leave pod_combat_active set with orphaned slot stories,
	# old log entries, and stale workspace aliases. Without this
	# preamble, the new fight inherits all of that: old log rows
	# render in the new combat, slot stories on Charon leak (new
	# ones are created without ending the old ones), and the
	# workspace aliases (pod_combat_actor_char / pod_combat_target_char)
	# point at characters from the prior fight until the new
	# load_workspace overwrites them.
	#
	# pod_combat_cleanup_effect is the same teardown used at the
	# normal end of combat; running it defensively here is
	# idempotent when nothing's leftover.
	if = {
		limit = { has_variable = pod_combat_active }
		pod_combat_cleanup_effect = yes
	}

	# Defensive: clear is_over flag in case the cleanup branch above
	# didn't fire (no prior pod_combat_active) but a stale is_over
	# somehow survived from earlier code paths.
	remove_variable = pod_combat_is_over

	set_variable = { name = pod_combat_mode value = $MODE$ }
	set_variable = { name = pod_combat_active value = yes }
	set_variable = { name = pod_combat_round value = 1 }

	# Per-slot story_cycle scaffolding is lazy: each
	# pod_combat_set_slot_char_effect call below creates the matching
	# POD_combat_slot story on Charon on first use, anchors a pointer
	# on the host (pod_combat_slot_$N$), and registers the slot in
	# pod_combat_active_slots. Slots the current mode never fills
	# never get a story_cycle scope. pod_combat_init_slot_from_character_effect
	# (called per combatant below) writes the per-slot stat / stance /
	# tier fields onto each freshly populated slot story. Cleanup walks
	# pod_combat_active_slots in pod_combat_group_cleanup_effect at
	# combat end. See pod_combat_slot_effects.txt for the lifecycle
	# helpers.

	# The PC also gets registered as slot 0's slot_char
	# (via pod_combat_set_slot_char_effect) below.
	save_scope_as = combat_pc

	# -- Pre-pass: shapeshift all AI fera combatants BEFORE any slot init --
	# This must run before the slot loop because each slot's init calls
	# POD_enemy_knight_set_discipline_battle_effect with the opposing team
	# leader as CASTER. That effect's delirium check queries
	# $CASTER$ = { POD_give_delirium_trigger = yes } which requires the
	# crinos character modifier to already be present. If we waited and
	# shifted inside each slot's init, player-team slots 0-4 would query
	# the un-shifted opponent leader (slot 5 hasn't run yet), and slot 5
	# would similarly miss any allied fera that shifted in their own slot.
	# Shapeshifting upfront ensures every CASTER has its warform modifier
	# before any delirium / discipline query runs.
	#
	# The PC is EXPLICITLY EXCLUDED from auto-shapeshift.
	scope:pod_combat_target = {
		if = {
			limit = {
				is_ai = yes
				has_trait = fera
				POD_can_shapeshift_trigger = yes
				NOT = { POD_is_shapeshifted_warform_trigger = yes }
			}
			POD_shapeshift_fera_warform_effect = yes
		}
	}
	# Coterie allies (modes 2 and 3 only — matches the ally pickup loop below)
	if = {
		limit = {
			OR = {
				$MODE$ = 2
				$MODE$ = 3
			}
		}
		every_relation = {
			type = coterie_member
			limit = {
				is_alive = yes
				NOT = { has_trait = incapable }
				is_ai = yes
				has_trait = fera
				POD_can_shapeshift_trigger = yes
				NOT = { POD_is_shapeshifted_warform_trigger = yes }
			}
			POD_shapeshift_fera_warform_effect = yes
		}
	}
	# Opponent's knights (modes 1 and 3 only — matches the knight pickup loop below)
	if = {
		limit = {
			OR = {
				$MODE$ = 1
				$MODE$ = 3
			}
		}
		scope:pod_combat_target = {
			every_knight = {
				limit = {
					is_alive = yes
					NOT = { has_trait = incapable }
					opinion = {
						target = scope:pod_combat_target
						value >= 0
					}
					is_ai = yes
					NOT = { this = scope:combat_pc }
					has_trait = fera
					POD_can_shapeshift_trigger = yes
					NOT = { POD_is_shapeshifted_warform_trigger = yes }
				}
				POD_shapeshift_fera_warform_effect = yes
			}
		}
	}

	# -- Slot 0: PC (always) --
	pod_combat_set_slot_char_effect = { SLOT = 0 CHARACTER = scope:combat_pc }
	pod_combat_init_slot_from_character_effect = { SLOT = 0 CHARACTER = scope:combat_pc }

	# -- Slots 1-4: Coterie allies (modes 2 and 3) --
	if = {
		limit = {
			has_variable = pod_combat_mode
			OR = {
				var:pod_combat_mode = 2
				var:pod_combat_mode = 3
			}
		}
		every_relation = {
			type = coterie_member
			limit = {
				is_alive = yes
				is_adult = yes
				NOT = { has_trait = incapable }
			}
			save_scope_as = ally_candidate
			scope:combat_pc = {
				pod_combat_assign_ally_slot_effect = { CHARACTER = scope:ally_candidate }
			}
		}
	}

	# -- Slot 5: Primary opponent (always) --
	pod_combat_set_slot_char_effect = { SLOT = 5 CHARACTER = scope:pod_combat_target }
	pod_combat_init_slot_from_character_effect = { SLOT = 5 CHARACTER = scope:pod_combat_target }
	set_variable = { name = pod_combat_enemy_team_size value = 1 }

	# -- Slots 6-9: Opponent's knights (modes 1 and 3) --
	if = {
		limit = {
			has_variable = pod_combat_mode
			OR = {
				var:pod_combat_mode = 1
				var:pod_combat_mode = 3
			}
		}
		scope:pod_combat_target = {
			every_knight = {
				limit = {
					is_alive = yes
					NOT = { has_trait = incapable }
					opinion = {
						target = scope:pod_combat_target
						value >= 0
					}
				}
				save_scope_as = enemy_candidate
				scope:combat_pc = {
					if = {
						limit = { var:pod_combat_enemy_team_size < 5 }
						pod_combat_assign_enemy_slot_effect = { CHARACTER = scope:enemy_candidate }
					}
				}
			}
		}
	}

	# -- Calculate hit chances for all combatants --
	# Formula: base 60 + (own prowess - primary opposing leader's prowess) / 2
	# "Primary opposing leader" = slot 5 (the enemy leader) for player-team combatants 0-4,
	# and slot 0 (the PC) for enemy-team combatants 5-9. NOT an average across the enemy
	# team — pair-specific recalculation kicks in later when an actor/target pair is
	# loaded into the workspace via pod_combat_recalc_pc_hit_chance_effect, which
	# re-derives hit_chance against the *currently selected* target instead of the leader.
	pod_combat_calculate_all_hit_chances_effect = yes

	# -- Set up initial target / workspace defaults --
	# Seed the TARGET_SIDE-addressable alias (pod_combat_target_char) so
	# the damage helper's var:pod_combat_$TARGET_SIDE$_char lookup
	# resolves correctly from the moment combat initializes, even before
	# the first target_load effect runs.
	if = {
		limit = { has_variable = pod_combat_slot_5 var:pod_combat_slot_5 = { has_variable = slot_char } }
		set_variable = { name = pod_combat_target_char value = var:pod_combat_slot_5.var:slot_char }
	}
	set_variable = { name = pod_combat_selected_target value = 5 }
	# Initialize the unified PC target: defaults to the primary opponent
	# (slot 5). The player can switch targets mid-combat via the Target
	# button on any combat card (allies 1-4 or enemies 5-9). A single
	# variable covers both team-enemy and single-ally targeting; the
	# can_use triggers enforce that each skill fires against a valid slot.
	set_variable = { name = pod_combat_pc_target value = 5 }
	# AP / drain setup happens in the "PC goes first" branch below (and
	# AI turns set their own AP via pod_combat_execute_ai_turn_effect):
	# pod_combat_apply_ap_drain_effect reads pod_combat_actor_ap_drain
	# from the workspace, so the drain call must follow load_workspace.

	# Initialize combat log
	set_variable = { name = pod_combat_last_damage value = 0 }
	# AI anti-repetition history (pod_combat_ai_last_move /
	# _prev_move / _prev2_move) lives on the AI character, not on
	# the combat host (the selector switches into
	# var:pod_combat_actor_char before reading/writing). No PC-side
	# init or remove needed here. Per-fight cleanup of those vars
	# happens in pod_combat_init_slot_from_character_effect (slot
	# entry) and pod_combat_group_cleanup_effect (combat exit).

	# -- Non-fatal duel: gate PoD's inline death check --
	# For fatality = no we add pod_combat_nonfatal_active as a character
	# flag to every combatant. POD_can_die_from_wounds (overridden in
	# common/scripted_triggers/POD_combat/pod_combat_death_gate_overrides.txt)
	# reads the flag and returns no, so POD_health_death_check_effect
	# never enters its pod_death branch during the combat. The flag is
	# stripped post-combat by pod_combat_end_combat_effect AFTER
	# pod_combat_apply_nonfatal_effect has capped everyone to at
	# least one empty box. Fatal duels deliberately don't gate the
	# death check. Combatants can die mid-fight; KO via wound-track-
	# full happens regardless via the slot alive-flip check below.
	if = {
		limit = {
			has_variable = pod_combat_sc_context
			var:pod_combat_sc_context.var:fatality = flag:no
		}
		pod_combat_apply_nonfatal_flag_effect = yes
	}

	# -- Init-time KO check --
	# pod_combat_init_slot_from_character_effect sets slot_alive = 1
	# unconditionally for every populated slot. HP / morale / state are
	# all read live from the character at runtime, so a character who
	# walks into combat already KO'd (filled wound track from prior
	# damage, scripted incapacitation, torpor, morale-broken from a
	# prior scene) would be marked alive at init and could act / be
	# targeted before the first save_workspace check flipped their
	# flag. Apply the same 4-way KO predicate the action-boundary
	# checks use (see lines ~841-872) and then run the victory check
	# so a degenerate "everyone on one side is already KO'd" combat
	# resolves immediately instead of running a phantom first round.
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			OR = {
				var:slot_char = { is_alive = no }
				var:slot_char = { pod_combat_is_incapacitated_trigger = yes }
				var:slot_morale <= 0
				var:slot_char = { POD_total_damage_boxes >= POD_health_track_length }
			}
		}
		set_variable = { name = slot_alive value = 0 }
	}
	pod_combat_check_group_victory_effect = yes

	# -- Determine round 1 turn order --
	# Unified flow: every combat — including 1v1 — rolls initiative and
	# processes any pre-PC turns so nobody gets a free first move. A 1v1
	# is just a degenerate NvN with slots 0 and 5 populated. Skipped
	# when the init-time KO check already ended combat (everyone on
	# one side started KO'd).
	if = {
		limit = { NOT = { has_variable = pod_combat_is_over } }
	# System log: combat begins. Pushed BEFORE turn order so it
	# ends up at the BOTTOM of the newest-first log — reading bottom-up
	# the row tells "combat began, then these things happened…" Skipped
	# when the init-time KO check above already ended combat (no point
	# bookending an aborted combat).
	pod_combat_push_system_log_effect = { MESSAGE_KEY = combat_started }

	pod_combat_calculate_initiative_effect = yes
	pod_combat_find_next_actor_effect = yes

	if = {
		limit = {
			has_variable = pod_combat_current_actor
			var:pod_combat_current_actor = 0
		}
		# PC goes first in initiative order — restore PC target, load
		# workspace, then set AP + apply drain (load must precede
		# drain so the workspace mirror is populated).
		pod_combat_restore_pc_target_effect = yes
		pod_combat_load_workspace_effect = yes
		# Explicit PC slot 0 hit-chance recalc — mirrors the proven
		# pod_combat_select_target SGUI sequence (load_workspace +
		# recalc_pc_hit_chance). The load_workspace writeback already
		# pushes the recomputed pair hit chance to slot 0, so this is
		# a redundant guarantee that PC's slot card displays the
		# correct PC-vs-pc_target hit chance from the moment the move
		# bar becomes interactive. Without this, intermittent staleness
		# was observed where PC's hit chance display only refreshed
		# after the player clicked a target, even when load_workspace
		# had run on entry.
		pod_combat_recalc_pc_hit_chance_effect = yes
		set_variable = { name = pod_combat_current_ap value = pod_combat_base_ap }
		pod_combat_apply_ap_drain_effect = yes
	}
	else = {
		# An AI goes first — process AI turns until it's the PC's turn
		# or combat ends. The PC is blocked from acting because the
		# turn-side check fires against current_actor != 0.
		# AI turn setup (load + set AP + drain) happens inside
		# pod_combat_execute_ai_turn_effect.
		pod_combat_process_pre_pc_turns_effect = yes
	}
	}  # end of NOT pod_combat_is_over guard (init-time KO check)
}

# Per-slot helper: compute one slot's base hit chance from prowess and
# apply it. Body identical for every slot — only the comparison-target
# slot differs (player team measures against slot 5's prowess, enemy
# team measures against slot 0's). Lifted out of
# pod_combat_calculate_all_hit_chances_effect so the dispatcher is just
# 10 parametric calls instead of a 60-line copy/paste block.
pod_combat_calc_one_hit_chance_effect = {
	# $SLOT$       = the slot to compute (0..9)
	# $ENEMY_SLOT$ = the comparison-target slot (5 for player team,
	#                0 for enemy team)
	#
	# Outer gate uses the slot pointer + slot_char chain so tooltip-
	# build (no slot stories) and unfilled slots both short-circuit
	# before any chained `var:pod_combat_slot_$SLOT$.var:slot_prowess`
	# read fires. The ENEMY_SLOT pointer is gated symmetrically — in
	# current callers it's always 0 or 5 (always populated for any
	# combat the minigame intercepts), but writing the check defensively
	# lets a future caller pass a non-leader ENEMY_SLOT without runtime-
	# erroring on the chained prowess read below.
	if = {
		limit = {
			has_variable = pod_combat_slot_$SLOT$
			var:pod_combat_slot_$SLOT$ = { has_variable = slot_char }
			has_variable = pod_combat_slot_$ENEMY_SLOT$
			var:pod_combat_slot_$ENEMY_SLOT$ = { has_variable = slot_char }
		}
		save_scope_value_as = {
			name = base_hit
			value = {
				value = pod_combat_hit_chance_base
				add = {
					value = var:pod_combat_slot_$SLOT$.var:slot_prowess
					subtract = var:pod_combat_slot_$ENEMY_SLOT$.var:slot_prowess
					divide = 2
				}
				min = 20
				max = 95
			}
		}
		pod_combat_set_slot_var_effect = { SLOT = $SLOT$ NAME = base_hit_chance VALUE = scope:base_hit }
		pod_combat_recalc_slot_stats_effect = { SLOT = $SLOT$ }
	}
}

pod_combat_calculate_all_hit_chances_effect = {
	# For each slot, compute the base hit chance from prowess:
	#   base = pod_combat_hit_chance_base + (own prowess - primary enemy prowess) / 2,
	#  clamped 20–95. Player team (0-4) measured against slot 5's prowess.
	# Enemy team (5-9) measured against slot 0's (PC) prowess.
	# After setting each slot's base_hit_chance, pod_combat_recalc_slot_stats_effect
	# rebuilds the live hit_chance from base + any active tier magnitudes.
	# (Soak is recalculated the same way so tier debuffs carry cleanly across
	# the round-end reset.)

	# Player team — measures against primary opponent (slot 5) prowess.
	pod_combat_calc_one_hit_chance_effect = { SLOT = 0 ENEMY_SLOT = 5 }
	pod_combat_calc_one_hit_chance_effect = { SLOT = 1 ENEMY_SLOT = 5 }
	pod_combat_calc_one_hit_chance_effect = { SLOT = 2 ENEMY_SLOT = 5 }
	pod_combat_calc_one_hit_chance_effect = { SLOT = 3 ENEMY_SLOT = 5 }
	pod_combat_calc_one_hit_chance_effect = { SLOT = 4 ENEMY_SLOT = 5 }

	# Enemy team — measures against PC (slot 0) prowess.
	pod_combat_calc_one_hit_chance_effect = { SLOT = 5 ENEMY_SLOT = 0 }
	pod_combat_calc_one_hit_chance_effect = { SLOT = 6 ENEMY_SLOT = 0 }
	pod_combat_calc_one_hit_chance_effect = { SLOT = 7 ENEMY_SLOT = 0 }
	pod_combat_calc_one_hit_chance_effect = { SLOT = 8 ENEMY_SLOT = 0 }
	pod_combat_calc_one_hit_chance_effect = { SLOT = 9 ENEMY_SLOT = 0 }
}

##############################################################################################
# SECTION 6b: PLAYER HIT CHANCE RECALC ON TARGET SWAP
# Called by pod_combat_select_target (the unified target-select SGUI)
# after loading a new enemy into pod_combat_target_*. Rebuilds the
# PC's base_hit_chance from the freshly loaded target's prowess and
# propagates to slot 0's slot_base_hit_chance via
# pod_combat_set_slot_var_effect, then runs the workspace and per-slot
# tier recalc so any active hit buff/debuff tiers reapply cleanly
# against the new base. Stealth's +30 hit bonus is outside the tier
# system and stays tied to the stealth flag (consumed via the damage
# pipeline).
##############################################################################################

pod_combat_recalc_pc_hit_chance_effect = {
	# PC-specific hit-chance recalc: rebuilds slot 0's base_hit_chance
	# from the currently-loaded target's prowess, then calls the full
	# tier recalc so any active hit buffs / debuffs reapply cleanly
	# against the new base. Called when the PC interactively swaps
	# their target via pod_combat_select_target (the SGUI invokes this
	# right after loading the new enemy into pod_combat_target_*), and
	# also from the tier-recalc path when stat changes invalidate the
	# previously-computed hit chance.
	#
	# SLOT = 0 is hardcoded inside this effect. The function is
	# PC-only by design (target swap is a PC interaction; AI selects
	# moves and targets together each turn) — the `_pc_` in the
	# effect name encodes that contract.
	#
	# Round-duration tiers carry over (they expire at round end, not on
	# target swap). Stealth's +30 is a separate mechanic outside the
	# tier system and is handled by its own consume paths.
	if = {
		limit = {
			has_variable = pod_combat_actor_prowess
			has_variable = pod_combat_target_prowess
		}
		save_scope_value_as = {
			name = pc_recalc_base_hit
			value = {
				value = pod_combat_hit_chance_base
				add = {
					value = var:pod_combat_actor_prowess
					subtract = var:pod_combat_target_prowess
					divide = 2
				}
				min = 20
				max = 95
			}
		}
		set_variable = { name = pod_combat_actor_base_hit_chance value = scope:pc_recalc_base_hit }
		pod_combat_set_slot_var_effect = { SLOT = 0 NAME = base_hit_chance VALUE = scope:pc_recalc_base_hit }
		pod_combat_recalc_hit_actor_effect = yes
		pod_combat_recalc_slot_stats_effect = { SLOT = 0 }
	}
}

##############################################################################################
# SECTION 7: GROUP CLEANUP
# Remove all per-slot variables (slots 0-9) and global group state.
# Called in addition to the existing workspace cleanup.
##############################################################################################

pod_combat_group_cleanup_effect = {
	# ── AI per-fight transient state cleanup ─────────────────────
	# Two kinds of state on AI characters that should be per-fight:
	#   - pod_combat_loadout: the AI's scratch loadout list (one entry
	#     per generated move — full-catalogue snapshot since AI isn't
	#     restricted by category caps).
	#     Cleared so save data doesn't carry it between fights.
	#   - pod_combat_ai_last_move / _prev_move / _prev2_move: the
	#     three-deep anti-repetition history. Cleared so the same
	#     NPC's next fight doesn't open with -35/-20/-12 penalties
	#     against the moves they used last combat (penalty values
	#     live in pod_combat_ai_effects.txt's selector weight block).
	# Slot 0 (PC) is skipped. the player's loadout is persistent
	# equipment, and the AI history vars are never written on the PC
	# (the selector switches into var:pod_combat_actor_char before
	# touching them, which is always an AI on AI turns).
	#
	# Runs FIRST so the slot stories still exist (and slot_char is
	# still readable) when we reach into each slot's character to
	# clear its per-fight scratch state. The slot-story teardown
	# below tears down everything the chained read needs.
	#
	# Pairs with the slot-init clear in
	# pod_combat_init_slot_from_character_effect (handles legacy
	# NPCs on the way INTO combat; this handles save bloat / cross-
	# fight bias on the way OUT).
	# Iterate active slots, skipping the PC (slot 0) and any non-AI
	# combatants. var:slot_alive isn't checked — even KO'd AIs have
	# accumulated cross-fight bias that should be cleared. Replaced
	# nine hardcoded `if = { limit = { ...slot_N... } var:slot_N.var:slot_char = { ... } }`
	# blocks with a single iteration.
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			var:slot_index > 0
			var:slot_char = { is_ai = yes }
		}
		var:slot_char = {
			clear_variable_list = pod_combat_loadout
			remove_variable = pod_combat_ai_last_move
			remove_variable = pod_combat_ai_prev_move
			remove_variable = pod_combat_ai_prev2_move
		}
	}

	# ── Slot story_cycle teardown ────────────────────────────────
	# end_story every POD_combat_slot story (iterating
	# pod_combat_active_slots — the only slot list now that creation
	# is lazy), drop the host's pointer vars (pod_combat_slot_0..9),
	# and clear pod_combat_active_slots. Runs AFTER the AI cleanup
	# above so the chained var:pod_combat_slot_$N$.var:slot_char reads
	# still resolve. See pod_combat_slot_effects.txt for the lifecycle
	# helper definitions.
	pod_combat_clear_slot_stories_effect = yes

	# ── Workspace mirrors for the spirit capability flags ────────
	# Per-slot slot_incorporeal_spirit / slot_can_damage_spirit are
	# torn down with the rest of the slot story above; these are the
	# host-scope workspace mirrors that
	# pod_combat_load_actor_from_story_effect /
	# pod_combat_load_target_from_story_effect populate.
	remove_variable = pod_combat_actor_incorporeal_spirit
	remove_variable = pod_combat_actor_can_damage_spirit
	remove_variable = pod_combat_target_incorporeal_spirit
	remove_variable = pod_combat_target_can_damage_spirit

	# Global group state
	remove_variable = pod_combat_mode
	remove_variable = pod_combat_enemy_team_size
	remove_variable = pod_combat_current_actor
	remove_variable = pod_combat_selected_target
	remove_variable = pod_combat_pc_target
	remove_variable = pod_combat_best_init
	remove_variable = pod_combat_best_slot
}

##############################################################################################
# SECTION 8: INITIATIVE SYSTEM
# For each alive combatant slot (0-9), calculate initiative as
# prowess + (one of seven discrete buckets {1, 5, 10, 15, 20, 25, 30})
# + init_bonus. NOT a uniform random(1-30) — the seven buckets are
# uniformly weighted (1 each in random_list) but the values are
# discrete, which produces a coarser distribution than continuous 1-30
# would. Stored on each slot story as slot_initiative; slot_has_acted
# is cleared on the same pass.
##############################################################################################

# Roll initiative for every filled+alive slot in one pass over
# pod_combat_active_slots, fold in each slot's pending slot_init_bonus,
# and clear slot_has_acted ready for the new round. Iteration body
# runs in slot-story scope so the random_list bucket pick saves to
# scope:init_roll and the per-slot writes are direct.
pod_combat_calculate_initiative_effect = {
	# Roll initiative for every alive combatant and stamp turn-order
	# rank. Initiative = prowess + bucket_pick + init_bonus, where
	# bucket_pick is one of seven discrete values {1, 5, 10, 15, 20,
	# 25, 30} chosen uniformly via random_list with weight 1 each.
	# Tuning the variance means changing the bucket VALUES (not
	# switching to a uniform range), adding more buckets extends the
	# curve, narrower spreads (e.g. {5, 10, 15}) make it more
	# prowess-deterministic.
	# init_bonus comes from buff/debuff moves applied previous round;
	# positive = faster (buff), negative = slower (debuff). Cleared
	# after application.
	#
	# Iterates pod_combat_active_slots — the iteration body runs in
	# slot-story scope so var:slot_X reads/writes target the slot
	# story directly.
	save_scope_as = host
	every_in_list = {
		variable = pod_combat_active_slots
		limit = { var:slot_alive = 1 }
		random_list = {
			1 = { save_scope_value_as = { name = init_roll value = { value = var:slot_prowess add = 1  } } }
			1 = { save_scope_value_as = { name = init_roll value = { value = var:slot_prowess add = 5  } } }
			1 = { save_scope_value_as = { name = init_roll value = { value = var:slot_prowess add = 10 } } }
			1 = { save_scope_value_as = { name = init_roll value = { value = var:slot_prowess add = 15 } } }
			1 = { save_scope_value_as = { name = init_roll value = { value = var:slot_prowess add = 20 } } }
			1 = { save_scope_value_as = { name = init_roll value = { value = var:slot_prowess add = 25 } } }
			1 = { save_scope_value_as = { name = init_roll value = { value = var:slot_prowess add = 30 } } }
		}
		set_variable = { name = slot_initiative value = scope:init_roll }
		if = {
			limit = { has_variable = slot_init_bonus }
			change_variable = { name = slot_initiative add = var:slot_init_bonus }
			remove_variable = slot_init_bonus
		}
		set_variable = { name = slot_has_acted value = 0 }
	}

	# Stamp each slot's turn-order rank as a stored slot var so the
	# datamodel-driven slot card widget can display "Turn N" without
	# needing a per-slot SV that compares against all other slots
	# (the slot story can't reach the host's other-slot pointers from
	# its own scope, and dynamic SV-name construction isn't supported
	# in GUI bindings).
	pod_combat_calc_turn_orders_effect = yes
}

# Compute slot_turn_order on every alive combatant after initiatives
# are rolled. Rank semantics: highest initiative = rank 1, ties broken
# by lower slot_index (so e.g. slot 0 beats slot 5 if both rolled the
# same number).
#
# Implementation is a nested every_in_list: outer loop iterates each
# alive slot story to compute its rank; inner loop counts how many
# OTHER alive slots beat it on the (initiative, -slot_index)
# lexicographic order. Saving scope:host before iterating lets the
# inner loop re-enter host scope to iterate the active list again.
# Saving scope:me lets the inner loop write back into the outer's
# slot story.
pod_combat_calc_turn_orders_effect = {
	save_scope_as = host
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			has_variable = slot_initiative
			var:slot_alive = 1
		}
		save_scope_as = me
		save_scope_value_as = { name = my_init  value = var:slot_initiative }
		save_scope_value_as = { name = my_index value = var:slot_index }
		# Start at rank 1; bump for each other slot that ranks above us.
		set_variable = { name = slot_turn_order value = 1 }
		# Re-enter host scope for the inner iteration over the same list.
		scope:host = {
			every_in_list = {
				variable = pod_combat_active_slots
				limit = {
					has_variable = slot_initiative
					var:slot_alive = 1
					# Exclude self.
					NOT = { this = scope:me }
					# Strictly higher initiative, OR same initiative with
					# lower slot_index (matches the legacy SVs' >= vs >
					# asymmetric tiebreaker).
					OR = {
						var:slot_initiative > scope:my_init
						AND = {
							var:slot_initiative = scope:my_init
							var:slot_index < scope:my_index
						}
					}
				}
				# Bump the OUTER slot's rank — write back via scope:me.
				scope:me = {
					change_variable = { name = slot_turn_order add = 1 }
				}
			}
		}
	}
}

##############################################################################################
# SECTION 9: FIND NEXT ACTOR
# Walk pod_combat_active_slots (filled-only by construction) and pick
# the slot with the highest initiative that hasn't acted yet.
##############################################################################################

pod_combat_find_next_actor_effect = {
	# Find the alive combatant with highest unprocessed initiative.
	# best_init starts below the most negative initiative any combatant
	# can realistically roll — the per-slot init roll is `prowess +
	# random_list bucket [1,30] + slot_init_bonus`, and slot_init_bonus
	# can be NEGATIVE via stacked initiative_debuff moves. A -1 sentinel
	# would silently skip any slot whose final initiative landed at or
	# below -1 (the strict `>` compare in the body never triggers),
	# ending the round early and stealing turns from heavily-debuffed
	# combatants. -999 stays safely below any practical clamp; if init
	# math ever grows past that range, push this further down.
	set_variable = { name = pod_combat_best_init value = -999 }
	set_variable = { name = pod_combat_best_slot value = -1 }

	# Iterate the active slot list and pick the highest-initiative
	# slot that hasn't acted yet. Iteration order matches the slot
	# stories' insertion order (slots 0..9), so among ties the
	# lowest-indexed slot wins because the strict `>` check inside the
	# body only updates best_* on a STRICTLY higher init.
	#
	# The max-comparison lives inside the body, NOT in the iteration
	# limit. CK3's `every_in_list` doesn't reliably re-evaluate `limit`
	# against host-var mutations made by earlier iteration bodies — the
	# list is captured up-front and a host-var-cross-iteration check in
	# `limit` ends up letting every slot through, so whichever iterated
	# LAST wins regardless of initiative. With pod_combat_active_slots
	# filled in [0, 1, 5, 6, 7] order, that bug surfaced as "highest
	# slot index always acts first," e.g. NPC slot 7 beating PC slot 0
	# even when PC's initiative was higher. Moving the strict `>`
	# compare into the body via `if = { limit = { ... } }` reads
	# `var:pod_combat_best_init` against host's CURRENT state each
	# iteration, since effect blocks run synchronously.
	save_scope_as = host
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			var:slot_alive = 1
			var:slot_has_acted = 0
		}
		save_scope_value_as = { name = current_init  value = var:slot_initiative }
		save_scope_value_as = { name = current_index value = var:slot_index }
		scope:host = {
			if = {
				limit = { scope:current_init > var:pod_combat_best_init }
				set_variable = { name = pod_combat_best_init value = scope:current_init }
				set_variable = { name = pod_combat_best_slot value = scope:current_index }
			}
		}
	}

	# Set the current actor from the best slot found
	set_variable = { name = pod_combat_current_actor value = var:pod_combat_best_slot }

	# Clean up temp vars
	remove_variable = pod_combat_best_init
	remove_variable = pod_combat_best_slot
}

##############################################################################################
# SECTION 10: AI TARGET SELECTION
# Two effects: one for selecting an enemy target (for player team AI), one for selecting
# a player team target (for enemy AI). Uses weighted random favoring low-HP targets.
##############################################################################################

pod_combat_ai_select_enemy_target_effect = {
	# Select a target from the enemy team (slots 5-9) for an ally
	# AI actor. random_in_list iterates the active slots filtered
	# to slot_index >= 5; weighting biases toward low-HP targets
	# (×2 if 2 or fewer empty wound boxes left).
	save_scope_as = host
	random_in_list = {
		variable = pod_combat_active_slots
		limit = {
			var:slot_alive = 1
			var:slot_index >= 5
		}
		weight = {
			base = 1
			modifier = { add = 50 }
			modifier = {
				factor = 2
				var:slot_char.POD_empty_damage_boxes <= 2
			}
		}
		save_scope_as = chosen_target
	}
	if = {
		limit = { exists = scope:chosen_target }
		scope:host = {
			set_variable = { name = pod_combat_selected_target value = scope:chosen_target.var:slot_index }
		}
	}
}

pod_combat_ai_select_player_target_effect = {
	# Select a target from the player team (slots 0-4) for an enemy
	# AI actor. Same weight shape as the enemy-target effect; just
	# the team filter inverts.
	save_scope_as = host
	random_in_list = {
		variable = pod_combat_active_slots
		limit = {
			var:slot_alive = 1
			var:slot_index <= 4
		}
		weight = {
			base = 1
			modifier = { add = 50 }
			modifier = {
				factor = 2
				var:slot_char.POD_empty_damage_boxes <= 2
			}
		}
		save_scope_as = chosen_target
	}
	if = {
		limit = { exists = scope:chosen_target }
		scope:host = {
			set_variable = { name = pod_combat_selected_target value = scope:chosen_target.var:slot_index }
		}
	}
}

##############################################################################################
# SECTION 10b: PC TARGET PERSISTENCE (UNIFIED)
#
# pod_combat_pc_target is a single slot index (1-9) representing whomever the PC has
# currently selected via the Target button on any combat card: ally OR enemy. A skill's
# target-type (single-enemy / single-ally / team-*) is enforced by the can_use trigger,
# not by having separate target variables.
#
# Slot 0 (the PC themselves) is never a valid stored target — self-only skills use
# targeting = self, not pc_target = 0.
#
# This effect is called at PC turn start AND after every PC move, so if the PC's target
# just got KO'd, it auto-retargets to the first alive non-PC slot (preferring enemies
# first, then allies). Combat should have ended before we run out of candidates.
##############################################################################################

pod_combat_restore_pc_target_effect = {
	# Detect dead/missing pc_target via any_in_list, if NO active
	# slot matches pc_target with slot_alive = 1, we need to retarget.
	# The OR also covers the unset case (no pc_target var) and the
	# self-targeting case (pc_target = 0 = PC). Replaced a 10-AND OR
	# block keyed on integer slot index.
	save_scope_as = host
	if = {
		limit = {
			OR = {
				NOT = { has_variable = pod_combat_pc_target }
				var:pod_combat_pc_target = 0
				NOT = {
					any_in_list = {
						variable = pod_combat_active_slots
						var:slot_index = scope:host.var:pod_combat_pc_target
						var:slot_alive = 1
					}
				}
			}
		}
		# Auto-retarget: prefer alive enemies (5-9) first, then allies
		# (1-4). Two-pass iteration over pod_combat_active_slots: first
		# pass finds the lowest-indexed alive enemy; second pass falls
		# back to the lowest-indexed alive ally if no enemy was found.
		# The `NOT = { exists = scope:chosen }` clause in each limit
		# makes the iteration first-match (the slot iteration order is
		# insertion order, 0..9). If nothing is alive we fall through
		# silently — combat is effectively over.
		every_in_list = {
			variable = pod_combat_active_slots
			limit = {
				NOT = { exists = scope:chosen }
				var:slot_index >= 5
				var:slot_alive = 1
			}
			save_scope_as = chosen
		}
		every_in_list = {
			variable = pod_combat_active_slots
			limit = {
				NOT = { exists = scope:chosen }
				var:slot_index >= 1
				var:slot_index <= 4
				var:slot_alive = 1
			}
			save_scope_as = chosen
		}
		if = {
			limit = { exists = scope:chosen }
			scope:host = {
				set_variable = { name = pod_combat_pc_target value = scope:chosen.var:slot_index }
				set_variable = { name = pod_combat_selected_target value = scope:chosen.var:slot_index }
			}
			scope:chosen = {
				pod_combat_load_target_from_story_effect = yes
			}
		}
	}
	else = {
		# Current PC target is still alive — mirror pc_target into selected_target
		# (in case an AI clobbered it with their own target during the previous turn)
		# and reload the target workspace with the PC's choice.
		set_variable = { name = pod_combat_selected_target value = var:pod_combat_pc_target }
		every_in_list = {
			variable = pod_combat_active_slots
			limit = { var:slot_index = scope:host.var:pod_combat_pc_target }
			pod_combat_load_target_from_story_effect = yes
		}
	}
}

##############################################################################################
# SECTION 11: GROUP TURN LOOP
# The main group turn processing effect. Called after the PC ends their turn (or at round
# start for non-PC actors). Processes all remaining actors in initiative order.
##############################################################################################

pod_combat_process_group_turns_effect = {
	# Process all remaining actors in initiative order
	# This is called after the PC acts (slot 0) to run all AI turns

	# Mark current actor (PC) as acted
	pod_combat_set_slot_var_effect = { SLOT = 0 NAME = has_acted VALUE = 1 }

	# Save PC's workspace back to slot 0
	pod_combat_save_workspace_effect = yes

	# Loop: find next actor, run their turn, repeat until all have acted

	# Iteration 1
	pod_combat_find_next_actor_effect = yes
	if = {
		limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 NOT = { has_variable = pod_combat_is_over } }
		pod_combat_execute_ai_turn_effect = yes
	}
	# Iteration 2
	if = {
		limit = { NOT = { has_variable = pod_combat_is_over } }
		pod_combat_find_next_actor_effect = yes
		if = {
			limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 NOT = { has_variable = pod_combat_is_over } }
			pod_combat_execute_ai_turn_effect = yes
		}
	}
	# Iteration 3
	if = {
		limit = { NOT = { has_variable = pod_combat_is_over } }
		pod_combat_find_next_actor_effect = yes
		if = {
			limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 NOT = { has_variable = pod_combat_is_over } }
			pod_combat_execute_ai_turn_effect = yes
		}
	}
	# Iteration 4
	if = {
		limit = { NOT = { has_variable = pod_combat_is_over } }
		pod_combat_find_next_actor_effect = yes
		if = {
			limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 NOT = { has_variable = pod_combat_is_over } }
			pod_combat_execute_ai_turn_effect = yes
		}
	}
	# Iteration 5
	if = {
		limit = { NOT = { has_variable = pod_combat_is_over } }
		pod_combat_find_next_actor_effect = yes
		if = {
			limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 NOT = { has_variable = pod_combat_is_over } }
			pod_combat_execute_ai_turn_effect = yes
		}
	}
	# Iteration 6
	if = {
		limit = { NOT = { has_variable = pod_combat_is_over } }
		pod_combat_find_next_actor_effect = yes
		if = {
			limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 NOT = { has_variable = pod_combat_is_over } }
			pod_combat_execute_ai_turn_effect = yes
		}
	}
	# Iteration 7
	if = {
		limit = { NOT = { has_variable = pod_combat_is_over } }
		pod_combat_find_next_actor_effect = yes
		if = {
			limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 NOT = { has_variable = pod_combat_is_over } }
			pod_combat_execute_ai_turn_effect = yes
		}
	}
	# Iteration 8
	if = {
		limit = { NOT = { has_variable = pod_combat_is_over } }
		pod_combat_find_next_actor_effect = yes
		if = {
			limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 NOT = { has_variable = pod_combat_is_over } }
			pod_combat_execute_ai_turn_effect = yes
		}
	}
	# Iteration 9
	if = {
		limit = { NOT = { has_variable = pod_combat_is_over } }
		pod_combat_find_next_actor_effect = yes
		if = {
			limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 NOT = { has_variable = pod_combat_is_over } }
			pod_combat_execute_ai_turn_effect = yes
		}
	}

	# All actors have acted — end round if combat not over
	if = {
		limit = { NOT = { has_variable = pod_combat_is_over } }
		pod_combat_group_end_round_effect = yes
	}
}

pod_combat_execute_ai_turn_effect = {
	# Execute one AI turn for the current actor
	# 1. Select target based on actor's team
	# 2. Load workspace
	# 3. Run AI move selection
	# 4. Save workspace
	# 5. Mark as acted
	# 6. Check victory

	# Select target
	if = {
		limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor <= 4 }
		# Actor is on player team — target an enemy
		pod_combat_ai_select_enemy_target_effect = yes
	}
	else = {
		# Actor is on enemy team — target a player
		pod_combat_ai_select_player_target_effect = yes
	}

	# Load workspace (actor -> actor workspace, target -> target workspace)
	pod_combat_load_workspace_effect = yes

	# Set AP for this actor's turn
	set_variable = { name = pod_combat_current_ap value = pod_combat_base_ap }
	pod_combat_apply_ap_drain_effect = yes

	# Run AI moves (uses existing pod_combat_ai_take_turn_effect which reads from workspace)
	pod_combat_ai_take_turn_effect = yes

	# Save workspace back to slots
	pod_combat_save_workspace_effect = yes

	# Mark actor as acted. Writes slot_has_acted = 1 to the slot
	# whose slot_index matches pod_combat_current_actor. scope:host
	# was already saved by pod_combat_save_workspace_effect above
	# and persists through the rest of this effect chain. Iterates
	# pod_combat_active_slots; the limit matches at most one slot
	# (slot_index is unique) so the write semantics are identical
	# to the prior 10-way else_if dispatch.
	if = {
		limit = { has_variable = pod_combat_current_actor }
		every_in_list = {
			variable = pod_combat_active_slots
			limit = { var:slot_index = scope:host.var:pod_combat_current_actor }
			set_variable = { name = slot_has_acted value = 1 }
		}
	}

	# Check victory
	pod_combat_check_group_victory_effect = yes
}

##############################################################################################
# SECTION 12: GROUP VICTORY CHECK
# Check if all enemies are knocked out (player wins) or all allies are knocked out
# (player loses).
##############################################################################################

pod_combat_check_group_victory_effect = {
	# Victory conditions (evaluated in order, first match ends combat):
	#   1. All enemies KO'd           → player team wins
	#   2. PC (slot 0) KO'd            → player loses (PC can't act, so combat cannot continue)
	#   3. All allies KO'd             → player loses

	# Save the PC scope so we can reference it via scope:combat_pc in set_variable
	# (using bare `root` there triggers "Event target link 'root' returned unset scope")
	save_scope_as = combat_pc

	# --- Check 1: All enemies knocked out ---
	# Expressed as "no alive enemy slot exists" — `any_in_list` over
	# the active list filtered by slot_index >= 5 (enemy team) and
	# slot_alive = 1. pod_combat_active_slots is filled-only by
	# construction, so no separate slot_char presence gate is needed.
	if = {
		limit = {
			NOT = {
				any_in_list = {
					variable = pod_combat_active_slots
					var:slot_index >= 5
					var:slot_alive = 1
				}
			}
			has_variable = pod_combat_slot_5 var:pod_combat_slot_5 = { has_variable = slot_char }
			NOT = { has_variable = pod_combat_is_over }
		}
		# Player team wins!
		set_variable = { name = pod_combat_winner value = scope:combat_pc }
		set_variable = { name = pod_combat_loser value = var:pod_combat_slot_5.var:slot_char }
		set_variable = { name = pod_combat_is_over value = yes }
		var:pod_combat_slot_5.var:slot_char = { save_scope_as = pod_combat_target }
		# System log: combat ends — victory. Pushed AFTER setting
		# is_over so the bookend row appears at the TOP of the
		# newest-first log (most recent), framing the row above as the
		# resolved outcome.
		pod_combat_push_system_log_effect = { MESSAGE_KEY = combat_ended_victory }
		# Combat window stays open — player clicks "End Combat" to dismiss
	}

	# --- Check 2: PC knocked out (even if allies still standing) ---
	# The PC is special — the player has no way to continue the fight if they are down.
	# End combat as a loss so the post-fight event can resolve wounds and outcomes.
	if = {
		limit = {
			has_variable = pod_combat_slot_0
			var:pod_combat_slot_0 = { has_variable = slot_char var:slot_alive = 0 }
			has_variable = pod_combat_slot_5 var:pod_combat_slot_5 = { has_variable = slot_char }
			NOT = { has_variable = pod_combat_is_over }
		}
		set_variable = { name = pod_combat_winner value = var:pod_combat_slot_5.var:slot_char }
		set_variable = { name = pod_combat_loser value = scope:combat_pc }
		set_variable = { name = pod_combat_is_over value = yes }
		var:pod_combat_slot_5.var:slot_char = { save_scope_as = pod_combat_target }
		pod_combat_push_system_log_effect = { MESSAGE_KEY = combat_ended_defeat }
		# Combat window stays open — player clicks "End Combat" to dismiss
	}

	# --- Check 3: All allies knocked out (covers edge cases where PC death check was bypassed) ---
	# Symmetric to Check 1: slot_index <= 4 filters to player team
	# (PC + ally slots 1-4). See Check 1 comment for the count-to-
	# any_in_list rewrite rationale.
	if = {
		limit = {
			NOT = {
				any_in_list = {
					variable = pod_combat_active_slots
					var:slot_index <= 4
					var:slot_alive = 1
				}
			}
			has_variable = pod_combat_slot_5 var:pod_combat_slot_5 = { has_variable = slot_char }
			NOT = { has_variable = pod_combat_is_over }
		}
		# Enemy team wins
		set_variable = { name = pod_combat_winner value = var:pod_combat_slot_5.var:slot_char }
		set_variable = { name = pod_combat_loser value = scope:combat_pc }
		set_variable = { name = pod_combat_is_over value = yes }
		var:pod_combat_slot_5.var:slot_char = { save_scope_as = pod_combat_target }
		pod_combat_push_system_log_effect = { MESSAGE_KEY = combat_ended_defeat }
		# Combat window stays open — player clicks "End Combat" to dismiss
	}
}

##############################################################################################
# SECTION 13: GROUP ROUND END
# Increment round, calculate new initiative, expire round-duration tier buffs,
# and set up the next round.
#
# No health write-back happens here. Wounds are applied LIVE during each move
# via POD_increase_wounds_hostile_effect, not deferred to round end. The
# fatality-mode finalization (wound cap + nonfatal-flag strip) runs once at
# combat END via pod_combat_apply_nonfatal_effect called from
# pod_combat_end_combat_effect. Calling it from here would strip
# pod_combat_nonfatal_active after round 1, silently breaking fatality = no
# from round 2 onward.
##############################################################################################

pod_combat_group_end_round_effect = {
	# Guard: if combat ended during this round (victory check fired
	# pod_combat_end_combat_effect which may have cleaned up variables),
	# skip the round-end logic entirely.
	if = {
		limit = { NOT = { has_variable = pod_combat_round } }
		# Combat already cleaned up — nothing to do
	}
	else = {

	change_variable = { name = pod_combat_round add = 1 }

	# System log: round transition. Pushed AFTER the increment
	# so entry.var:round captures the NEW round number — the log
	# row reads "— Round 2 begins —" at the top of the freshly-
	# started round's action sequence.
	pod_combat_push_system_log_effect = { MESSAGE_KEY = round_begins }

	# Calculate initiative for next round
	pod_combat_calculate_initiative_effect = yes

	# Find the first actor (highest initiative)
	pod_combat_find_next_actor_effect = yes

	# Expire all round-duration buff/debuff tiers on every slot, then
	# rebuild each slot's live soak and hit_chance from base + remaining
	# combat-duration tier magnitudes via
	# pod_combat_recalc_slot_stats_via_story_effect (covers slot_soak
	# revert too — no separate soak-reset pass needed). Recalc-from-base
	# avoids the clamping loss a delta-based revert would accumulate.
	# Guard / Dodge / Stealth flags are intentionally NOT cleared here;
	# stances persist until consumed by an attack or wiped at combat end
	# (pod_combat_cleanup_effect).
	pod_combat_revert_round_tiers_effect = yes

	# If PC (slot 0) goes first, set up workspace and wait for player input
	if = {
		limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor = 0 }
		# Order matters: pod_combat_apply_ap_drain_effect reads
		# pod_combat_actor_ap_drain from the workspace, which is
		# populated by pod_combat_load_actor_from_story_effect inside
		# load_workspace. Restoring the target first and loading
		# before draining mirrors the AI turn-start ordering at the
		# top of pod_combat_execute_ai_turn_effect.
		pod_combat_restore_pc_target_effect = yes
		pod_combat_load_workspace_effect = yes
		# Explicit PC slot 0 hit-chance recalc — mirrors the proven
		# pod_combat_select_target SGUI sequence. See the matching
		# call in pod_combat_initialize_effect for the full rationale.
		pod_combat_recalc_pc_hit_chance_effect = yes
		set_variable = { name = pod_combat_current_ap value = pod_combat_base_ap }
		pod_combat_apply_ap_drain_effect = yes
	}
	else = {
		# Non-PC goes first — process all AI turns until it's PC's turn or round ends
		pod_combat_process_pre_pc_turns_effect = yes
	}

	} # end else (combat not yet cleaned up)
}

pod_combat_process_pre_pc_turns_effect = {
	# Process AI turns that happen before PC's turn in initiative order
	# Unrolled loop: up to 9 iterations

	# Current actor is already set by find_next_actor
	if = {
		limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 var:pod_combat_current_actor != 0 NOT = { has_variable = pod_combat_is_over } }
		pod_combat_execute_ai_turn_effect = yes

		# Find next actor
		pod_combat_find_next_actor_effect = yes
		if = {
			limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 var:pod_combat_current_actor != 0 NOT = { has_variable = pod_combat_is_over } }
			pod_combat_execute_ai_turn_effect = yes

			pod_combat_find_next_actor_effect = yes
			if = {
				limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 var:pod_combat_current_actor != 0 NOT = { has_variable = pod_combat_is_over } }
				pod_combat_execute_ai_turn_effect = yes

				pod_combat_find_next_actor_effect = yes
				if = {
					limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 var:pod_combat_current_actor != 0 NOT = { has_variable = pod_combat_is_over } }
					pod_combat_execute_ai_turn_effect = yes

					pod_combat_find_next_actor_effect = yes
					if = {
						limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 var:pod_combat_current_actor != 0 NOT = { has_variable = pod_combat_is_over } }
						pod_combat_execute_ai_turn_effect = yes

						pod_combat_find_next_actor_effect = yes
						if = {
							limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 var:pod_combat_current_actor != 0 NOT = { has_variable = pod_combat_is_over } }
							pod_combat_execute_ai_turn_effect = yes

							pod_combat_find_next_actor_effect = yes
							if = {
								limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 var:pod_combat_current_actor != 0 NOT = { has_variable = pod_combat_is_over } }
								pod_combat_execute_ai_turn_effect = yes

								pod_combat_find_next_actor_effect = yes
								if = {
									limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 var:pod_combat_current_actor != 0 NOT = { has_variable = pod_combat_is_over } }
									pod_combat_execute_ai_turn_effect = yes

									pod_combat_find_next_actor_effect = yes
									if = {
										limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 var:pod_combat_current_actor != 0 NOT = { has_variable = pod_combat_is_over } }
										pod_combat_execute_ai_turn_effect = yes
									}
								}
							}
						}
					}
				}
			}
		}
	}

	# If PC is next (or combat is over), set up workspace for PC
	if = {
		limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor = 0 NOT = { has_variable = pod_combat_is_over } }
		# HP / max / agg_damage derive live from each slot character's
		# PoD wound vars via SVs (pod_combat_slot_hp_value et al. in
		# pod_combat_values.txt) — there is no slot-stored HP cache to
		# resync, so the wound bars in the combat window read the
		# current values automatically once the workspace is loaded.
		# See COMBAT_REFERENCE.info §7 "HP cache rejected" for the
		# rationale.

		# Order matters: pod_combat_apply_ap_drain_effect reads
		# pod_combat_actor_ap_drain from the workspace, which is
		# populated by load_actor_from_story_effect inside
		# load_workspace. Restore target + load workspace must happen
		# BEFORE the AP / drain pair.
		pod_combat_restore_pc_target_effect = yes
		pod_combat_load_workspace_effect = yes
		# Explicit PC slot 0 hit-chance recalc — mirrors the proven
		# pod_combat_select_target SGUI sequence. See the matching
		# call in pod_combat_initialize_effect for the full rationale.
		# Especially important on this path because every AI turn that
		# attacked the PC pushed "PC-vs-that-AI" into slot 0 via
		# save_workspace; load_workspace's writeback should refresh it
		# against PC's current target, but the redundant explicit recalc
		# matches what the player-click path does and was observed to
		# fix intermittent display staleness.
		pod_combat_recalc_pc_hit_chance_effect = yes
		set_variable = { name = pod_combat_current_ap value = pod_combat_base_ap }
		pod_combat_apply_ap_drain_effect = yes
		# DO NOT reset is_guarding / is_dodging / soak here. Stance
		# state persists until consumed (see comments in damage
		# pipeline), and pod_combat_load_workspace_effect already
		# performs the correct set/remove based on each slot story's
		# state — see pod_combat_load_actor_from_story_effect /
		# _target_from_story_effect, where every stance flag has both
		# a `set_variable = yes` branch and an `else = remove_variable`
		# branch keyed on the slot story's slot_is_X presence.
		#
		# A previous version had:
		#   remove_variable = pod_combat_actor_is_guarding
		#   remove_variable = pod_combat_target_is_guarding
		#   remove_variable = pod_combat_actor_is_dodging
		#   remove_variable = pod_combat_target_is_dodging
		#   set_variable = { name = pod_combat_actor_soak value = var:pod_combat_actor_base_soak }
		#   set_variable = { name = pod_combat_target_soak value = var:pod_combat_target_base_soak }
		# That block ran AFTER load_workspace and unconditionally
		# wiped stance flags + zeroed tier soak bonuses. After the
		# PC turn ended, pod_combat_save_workspace_effect saved the
		# wiped workspace back to the slot stories, permanently
		# erasing slot_is_guarding / slot_is_dodging on both PC and
		# the selected target, and stripping tier-driven soak from
		# slot_soak until the next round-end recalc rebuilt it.
		# Removed to fix the stance/soak persistence bug.
	}
}
