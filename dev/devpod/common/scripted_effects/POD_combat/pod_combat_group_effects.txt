##############################################################################################
# GROUP COMBAT INFRASTRUCTURE EFFECTS
# File: pod_combat_group_effects.txt
# Handles slot-based workspace load/save, dispatch, initialization, assignment, and cleanup
# for the group combat system (up to 10 combatants in slots 0-9).
#
# Slot layout:
#   0    = PC (player character, always present)
#   1-4  = Ally slots (coterie members, modes 2/3)
#   5    = Primary opponent (always present)
#   6-9  = Enemy slots (opponent knights, modes 1/3)
#
# Modes:
#   0 = 1v1, 1 = 1vN, 2 = Nv1, 3 = NvN
##############################################################################################

##############################################################################################
# SECTION 1: WORKSPACE LOAD EFFECTS
# Two parametric effects that copy a given slot's state into the player or opponent workspace.
# Called via dispatch effects that switch on pod_combat_current_actor / pod_combat_selected_target.
##############################################################################################

pod_combat_load_actor_from_slot_effect = {
	# Copies slot $SLOT$ data -> pod_combat_player_* workspace
	# The whole body is guarded by has_variable so tooltip-preview walks
	# don't spam warnings when the slot is unpopulated.
	#
	# HP / max_hp / agg_damage / initial_hp / agg_total are NOT loaded
	# into workspace anymore — wound state is read live from the
	# character's PoD variables via var:pod_combat_player_char.POD_*.
	if = {
		limit = { has_variable = pod_combat_c$SLOT$_char }
		set_variable = { name = pod_combat_player_soak value = var:pod_combat_c$SLOT$_soak }
		set_variable = { name = pod_combat_player_base_soak value = var:pod_combat_c$SLOT$_base_soak }
		if = { limit = { has_variable = pod_combat_c$SLOT$_base_hit_chance } set_variable = { name = pod_combat_player_base_hit_chance value = var:pod_combat_c$SLOT$_base_hit_chance } }
		set_variable = { name = pod_combat_player_morale value = var:pod_combat_c$SLOT$_morale }
		set_variable = { name = pod_combat_player_prowess value = var:pod_combat_c$SLOT$_prowess }
		set_variable = { name = pod_combat_player_hit_chance value = var:pod_combat_c$SLOT$_hit_chance }
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_energy_used }
			set_variable = { name = pod_combat_player_energy_used value = var:pod_combat_c$SLOT$_energy_used }
		}
		else = { set_variable = { name = pod_combat_player_energy_used value = 0 } }
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_is_guarding }
			set_variable = { name = pod_combat_player_is_guarding value = yes }
		}
		else = {
			remove_variable = pod_combat_player_is_guarding
		}
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_is_dodging }
			set_variable = { name = pod_combat_player_is_dodging value = yes }
		}
		else = {
			remove_variable = pod_combat_player_is_dodging
		}
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_is_stealthed }
			set_variable = { name = pod_combat_player_is_stealthed value = yes }
		}
		else = {
			remove_variable = pod_combat_player_is_stealthed
		}
		# Spirit-damage capability (static per combat). The actor needs
		# this flag to land Physical attacks on incorporeal spirit targets.
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_can_damage_spirit }
			set_variable = { name = pod_combat_player_can_damage_spirit value = yes }
		}
		else = {
			remove_variable = pod_combat_player_can_damage_spirit
		}
		# Incorporeality on the actor side is also tracked so retaliation /
		# mirrored flows can inspect it, though the main spirit-damage gate
		# lives on the opponent (target) side.
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_incorporeal_spirit }
			set_variable = { name = pod_combat_player_incorporeal_spirit value = yes }
		}
		else = {
			remove_variable = pod_combat_player_incorporeal_spirit
		}
		# Initiative bonus/penalty (persists into next round's roll)
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_init_bonus }
			set_variable = { name = pod_combat_player_init_bonus value = var:pod_combat_c$SLOT$_init_bonus }
		}
		else = { remove_variable = pod_combat_player_init_bonus }
		# AP drain (applied at turn start, then cleared)
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_ap_drain }
			set_variable = { name = pod_combat_player_ap_drain value = var:pod_combat_c$SLOT$_ap_drain }
		}
		else = { remove_variable = pod_combat_player_ap_drain }
		# Retaliation stances
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_retaliate_hit }
			set_variable = { name = pod_combat_player_retaliate_hit value = var:pod_combat_c$SLOT$_retaliate_hit }
		}
		else = { remove_variable = pod_combat_player_retaliate_hit }
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_retaliate_hit_agg }
			set_variable = { name = pod_combat_player_retaliate_hit_agg value = yes }
		}
		else = { remove_variable = pod_combat_player_retaliate_hit_agg }
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_retaliate_miss }
			set_variable = { name = pod_combat_player_retaliate_miss value = var:pod_combat_c$SLOT$_retaliate_miss }
		}
		else = { remove_variable = pod_combat_player_retaliate_miss }
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_retaliate_miss_agg }
			set_variable = { name = pod_combat_player_retaliate_miss_agg value = yes }
		}
		else = { remove_variable = pod_combat_player_retaliate_miss_agg }
		# Tier tracking vars (slot -> player workspace)
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_soak_buff_round_tier }
			set_variable = { name = pod_combat_player_soak_buff_round_tier value = var:pod_combat_c$SLOT$_soak_buff_round_tier }
		}
		else = { remove_variable = pod_combat_player_soak_buff_round_tier }
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_soak_buff_combat_tier }
			set_variable = { name = pod_combat_player_soak_buff_combat_tier value = var:pod_combat_c$SLOT$_soak_buff_combat_tier }
		}
		else = { remove_variable = pod_combat_player_soak_buff_combat_tier }
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_soak_debuff_round_tier }
			set_variable = { name = pod_combat_player_soak_debuff_round_tier value = var:pod_combat_c$SLOT$_soak_debuff_round_tier }
		}
		else = { remove_variable = pod_combat_player_soak_debuff_round_tier }
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_soak_debuff_combat_tier }
			set_variable = { name = pod_combat_player_soak_debuff_combat_tier value = var:pod_combat_c$SLOT$_soak_debuff_combat_tier }
		}
		else = { remove_variable = pod_combat_player_soak_debuff_combat_tier }
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_hit_buff_round_tier }
			set_variable = { name = pod_combat_player_hit_buff_round_tier value = var:pod_combat_c$SLOT$_hit_buff_round_tier }
		}
		else = { remove_variable = pod_combat_player_hit_buff_round_tier }
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_hit_buff_combat_tier }
			set_variable = { name = pod_combat_player_hit_buff_combat_tier value = var:pod_combat_c$SLOT$_hit_buff_combat_tier }
		}
		else = { remove_variable = pod_combat_player_hit_buff_combat_tier }
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_hit_debuff_round_tier }
			set_variable = { name = pod_combat_player_hit_debuff_round_tier value = var:pod_combat_c$SLOT$_hit_debuff_round_tier }
		}
		else = { remove_variable = pod_combat_player_hit_debuff_round_tier }
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_hit_debuff_combat_tier }
			set_variable = { name = pod_combat_player_hit_debuff_combat_tier value = var:pod_combat_c$SLOT$_hit_debuff_combat_tier }
		}
		else = { remove_variable = pod_combat_player_hit_debuff_combat_tier }
		# Actor character reference — mirrors pod_combat_opponent set in the
		# target load effect. Used by the AI selection effect to run trait
		# gates against the current actor (e.g. has_trait = fera) so enemy
		# AIs only pick moves they actually qualify for.
		set_variable = { name = pod_combat_actor_char value = var:pod_combat_c$SLOT$_char }
		# Alias matching $TARGET_SIDE$ = "player" — lets the damage
		# pipeline look up the target character uniformly as
		# var:pod_combat_$TARGET_SIDE$_char regardless of which side is
		# being damaged (used by retaliation when the actor is the victim).
		set_variable = { name = pod_combat_player_char value = var:pod_combat_c$SLOT$_char }
	}
}

pod_combat_load_target_from_slot_effect = {
	# Copies slot $SLOT$ data -> pod_combat_opponent_* workspace
	# The whole body is guarded by has_variable so tooltip-preview walks
	# don't spam warnings when the slot is unpopulated.
	#
	# Same as actor-load: HP / max_hp / agg_damage come from the
	# character's PoD wound vars at read time, not the workspace.
	if = {
		limit = { has_variable = pod_combat_c$SLOT$_char }
		set_variable = { name = pod_combat_opponent_soak value = var:pod_combat_c$SLOT$_soak }
		set_variable = { name = pod_combat_opponent_base_soak value = var:pod_combat_c$SLOT$_base_soak }
		if = { limit = { has_variable = pod_combat_c$SLOT$_base_hit_chance } set_variable = { name = pod_combat_opponent_base_hit_chance value = var:pod_combat_c$SLOT$_base_hit_chance } }
		set_variable = { name = pod_combat_opponent_morale value = var:pod_combat_c$SLOT$_morale }
		set_variable = { name = pod_combat_opponent_prowess value = var:pod_combat_c$SLOT$_prowess }
		set_variable = { name = pod_combat_opponent_hit_chance value = var:pod_combat_c$SLOT$_hit_chance }
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_energy_used }
			set_variable = { name = pod_combat_opponent_energy_used value = var:pod_combat_c$SLOT$_energy_used }
		}
		else = { set_variable = { name = pod_combat_opponent_energy_used value = 0 } }
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_is_guarding }
			set_variable = { name = pod_combat_opponent_is_guarding value = yes }
		}
		else = {
			remove_variable = pod_combat_opponent_is_guarding
		}
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_is_dodging }
			set_variable = { name = pod_combat_opponent_is_dodging value = yes }
		}
		else = {
			remove_variable = pod_combat_opponent_is_dodging
		}
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_is_stealthed }
			set_variable = { name = pod_combat_opponent_is_stealthed value = yes }
		}
		else = {
			remove_variable = pod_combat_opponent_is_stealthed
		}
		# Incorporeality on the target side — gates Physical damage in the
		# four pod_combat_deal_physical*_hp_damage_effect pipelines. The
		# attack auto-misses when this flag is set and the actor lacks
		# pod_combat_player_can_damage_spirit (mirroring PoD's
		# POD_damage_spirit_custom_trigger).
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_incorporeal_spirit }
			set_variable = { name = pod_combat_opponent_incorporeal_spirit value = yes }
		}
		else = {
			remove_variable = pod_combat_opponent_incorporeal_spirit
		}
		# Also track target's spirit-damage capability (useful for symmetric
		# retaliation checks / future mirrored logic).
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_can_damage_spirit }
			set_variable = { name = pod_combat_opponent_can_damage_spirit value = yes }
		}
		else = {
			remove_variable = pod_combat_opponent_can_damage_spirit
		}
		# Initiative bonus/penalty and AP drain on the target side
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_init_bonus }
			set_variable = { name = pod_combat_opponent_init_bonus value = var:pod_combat_c$SLOT$_init_bonus }
		}
		else = { remove_variable = pod_combat_opponent_init_bonus }
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_ap_drain }
			set_variable = { name = pod_combat_opponent_ap_drain value = var:pod_combat_c$SLOT$_ap_drain }
		}
		else = { remove_variable = pod_combat_opponent_ap_drain }
		# Retaliation stances on the target side
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_retaliate_hit }
			set_variable = { name = pod_combat_opponent_retaliate_hit value = var:pod_combat_c$SLOT$_retaliate_hit }
		}
		else = { remove_variable = pod_combat_opponent_retaliate_hit }
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_retaliate_hit_agg }
			set_variable = { name = pod_combat_opponent_retaliate_hit_agg value = yes }
		}
		else = { remove_variable = pod_combat_opponent_retaliate_hit_agg }
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_retaliate_miss }
			set_variable = { name = pod_combat_opponent_retaliate_miss value = var:pod_combat_c$SLOT$_retaliate_miss }
		}
		else = { remove_variable = pod_combat_opponent_retaliate_miss }
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_retaliate_miss_agg }
			set_variable = { name = pod_combat_opponent_retaliate_miss_agg value = yes }
		}
		else = { remove_variable = pod_combat_opponent_retaliate_miss_agg }
		# Tier tracking vars (slot -> opponent workspace)
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_soak_buff_round_tier }
			set_variable = { name = pod_combat_opponent_soak_buff_round_tier value = var:pod_combat_c$SLOT$_soak_buff_round_tier }
		}
		else = { remove_variable = pod_combat_opponent_soak_buff_round_tier }
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_soak_buff_combat_tier }
			set_variable = { name = pod_combat_opponent_soak_buff_combat_tier value = var:pod_combat_c$SLOT$_soak_buff_combat_tier }
		}
		else = { remove_variable = pod_combat_opponent_soak_buff_combat_tier }
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_soak_debuff_round_tier }
			set_variable = { name = pod_combat_opponent_soak_debuff_round_tier value = var:pod_combat_c$SLOT$_soak_debuff_round_tier }
		}
		else = { remove_variable = pod_combat_opponent_soak_debuff_round_tier }
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_soak_debuff_combat_tier }
			set_variable = { name = pod_combat_opponent_soak_debuff_combat_tier value = var:pod_combat_c$SLOT$_soak_debuff_combat_tier }
		}
		else = { remove_variable = pod_combat_opponent_soak_debuff_combat_tier }
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_hit_buff_round_tier }
			set_variable = { name = pod_combat_opponent_hit_buff_round_tier value = var:pod_combat_c$SLOT$_hit_buff_round_tier }
		}
		else = { remove_variable = pod_combat_opponent_hit_buff_round_tier }
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_hit_buff_combat_tier }
			set_variable = { name = pod_combat_opponent_hit_buff_combat_tier value = var:pod_combat_c$SLOT$_hit_buff_combat_tier }
		}
		else = { remove_variable = pod_combat_opponent_hit_buff_combat_tier }
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_hit_debuff_round_tier }
			set_variable = { name = pod_combat_opponent_hit_debuff_round_tier value = var:pod_combat_c$SLOT$_hit_debuff_round_tier }
		}
		else = { remove_variable = pod_combat_opponent_hit_debuff_round_tier }
		if = {
			limit = { has_variable = pod_combat_c$SLOT$_hit_debuff_combat_tier }
			set_variable = { name = pod_combat_opponent_hit_debuff_combat_tier value = var:pod_combat_c$SLOT$_hit_debuff_combat_tier }
		}
		else = { remove_variable = pod_combat_opponent_hit_debuff_combat_tier }
		# Also set the opponent character reference for counter checks
		set_variable = { name = pod_combat_opponent value = var:pod_combat_c$SLOT$_char }
		# Alias matching $TARGET_SIDE$ = "opponent" — parallels
		# pod_combat_player_char (see actor-load above). Lets the damage
		# pipeline reference var:pod_combat_$TARGET_SIDE$_char uniformly
		# instead of branching to pod_combat_opponent vs pod_combat_actor_char.
		set_variable = { name = pod_combat_opponent_char value = var:pod_combat_c$SLOT$_char }

		# Refresh workspace hit chances for the new actor/target pair.
		# Slot-stored hit chance is leader-baseline (computed once at
		# combat init in pod_combat_calculate_all_hit_chances_effect),
		# so without this every AI swap, AoE per-target iteration, and
		# workspace-load would push the wrong number into the damage
		# roll. The pair recalc reads the workspace's player_prowess
		# (set by the actor load that runs before this) and the
		# opponent_prowess we just wrote above, then layers the active
		# hit-tier modifiers via the per-side recalc effects. No slot
		# writeback — that's intentional, see the comment on the recalc
		# effect itself in pod_combat_tier_effects.txt.
		pod_combat_recalc_pair_hit_chance_effect = yes
	}
}

##############################################################################################
# SECTION 2: WORKSPACE SAVE EFFECTS
# Two parametric effects that copy the player or opponent workspace back into a given slot.
##############################################################################################

pod_combat_save_actor_to_slot_effect = {
	# Copies pod_combat_player_* workspace -> slot $SLOT$.
	# HP, max HP, and aggravated damage are NOT synced here — they are
	# live script values derived from the slot character's PoD wound
	# vars. The _alive flag IS synced because AI / victory-check paths read it directly.
	if = {
		limit = { has_variable = pod_combat_player_char }
		if = {
			limit = { OR = { var:pod_combat_c$SLOT$_char = { is_alive = no } var:pod_combat_c$SLOT$_char = { pod_combat_is_incapacitated_trigger = yes } } }
			set_variable = { name = pod_combat_c$SLOT$_alive value = 0 }
		}
		set_variable = { name = pod_combat_c$SLOT$_soak value = var:pod_combat_player_soak }
		set_variable = { name = pod_combat_c$SLOT$_base_soak value = var:pod_combat_player_base_soak }
		if = { limit = { has_variable = pod_combat_player_base_hit_chance } set_variable = { name = pod_combat_c$SLOT$_base_hit_chance value = var:pod_combat_player_base_hit_chance } }
		set_variable = { name = pod_combat_c$SLOT$_morale value = var:pod_combat_player_morale }
		set_variable = { name = pod_combat_c$SLOT$_prowess value = var:pod_combat_player_prowess }
		set_variable = { name = pod_combat_c$SLOT$_hit_chance value = var:pod_combat_player_hit_chance }
		if = {
			limit = { has_variable = pod_combat_player_energy_used }
			set_variable = { name = pod_combat_c$SLOT$_energy_used value = var:pod_combat_player_energy_used }
		}
		if = {
			limit = { has_variable = pod_combat_player_is_guarding }
			set_variable = { name = pod_combat_c$SLOT$_is_guarding value = yes }
		}
		else = {
			remove_variable = pod_combat_c$SLOT$_is_guarding
		}
		if = {
			limit = { has_variable = pod_combat_player_is_dodging }
			set_variable = { name = pod_combat_c$SLOT$_is_dodging value = yes }
		}
		else = {
			remove_variable = pod_combat_c$SLOT$_is_dodging
		}
		if = {
			limit = { has_variable = pod_combat_player_is_stealthed }
			set_variable = { name = pod_combat_c$SLOT$_is_stealthed value = yes }
		}
		else = {
			remove_variable = pod_combat_c$SLOT$_is_stealthed
		}
		# Initiative bonus and AP drain persist across workspace cycles
		if = {
			limit = { has_variable = pod_combat_player_init_bonus }
			set_variable = { name = pod_combat_c$SLOT$_init_bonus value = var:pod_combat_player_init_bonus }
		}
		else = { remove_variable = pod_combat_c$SLOT$_init_bonus }
		if = {
			limit = { has_variable = pod_combat_player_ap_drain }
			set_variable = { name = pod_combat_c$SLOT$_ap_drain value = var:pod_combat_player_ap_drain }
		}
		else = { remove_variable = pod_combat_c$SLOT$_ap_drain }
		# Retaliation stances
		if = {
			limit = { has_variable = pod_combat_player_retaliate_hit }
			set_variable = { name = pod_combat_c$SLOT$_retaliate_hit value = var:pod_combat_player_retaliate_hit }
		}
		else = { remove_variable = pod_combat_c$SLOT$_retaliate_hit }
		if = {
			limit = { has_variable = pod_combat_player_retaliate_hit_agg }
			set_variable = { name = pod_combat_c$SLOT$_retaliate_hit_agg value = yes }
		}
		else = { remove_variable = pod_combat_c$SLOT$_retaliate_hit_agg }
		if = {
			limit = { has_variable = pod_combat_player_retaliate_miss }
			set_variable = { name = pod_combat_c$SLOT$_retaliate_miss value = var:pod_combat_player_retaliate_miss }
		}
		else = { remove_variable = pod_combat_c$SLOT$_retaliate_miss }
		if = {
			limit = { has_variable = pod_combat_player_retaliate_miss_agg }
			set_variable = { name = pod_combat_c$SLOT$_retaliate_miss_agg value = yes }
		}
		else = { remove_variable = pod_combat_c$SLOT$_retaliate_miss_agg }
		# Tier tracking vars (player workspace -> slot)
		if = {
			limit = { has_variable = pod_combat_player_soak_buff_round_tier }
			set_variable = { name = pod_combat_c$SLOT$_soak_buff_round_tier value = var:pod_combat_player_soak_buff_round_tier }
		}

	if = {
		limit = { has_variable = pod_combat_opponent_char }
		# Alive-flip on death / incapacitation / wound-track-full.
		# The wound-track-full clause is what catches combat-HP
		# exhaustion in non-fatal duels (where PoD's death check is
		# gated, so is_alive stays yes and the trait checks won't fire)

		# resolving.
		if = {
			limit = { OR = { var:pod_combat_c$SLOT$_char = { is_alive = no } var:pod_combat_c$SLOT$_char = { pod_combat_is_incapacitated_trigger = yes } var:pod_combat_c$SLOT$_char = { POD_total_damage_boxes >= POD_health_track_length } } }

##############################################################################################
# SECTION 4: SLOT INITIALIZATION FROM CHARACTER
# Reads a character's health track, soak, and prowess and stores them in slot variables.
##############################################################################################

pod_combat_init_slot_from_character_effect = {
	# $SLOT$ = 0-9, $CHARACTER$ = character scope
	# Applies the discipline combat buffs and reads health/soak/prowess
	# into the slot variables.
	#
	# Shapeshifting for AI fera is NOT done here — it runs upfront in
	# pod_combat_initialize_effect as a pre-pass, so every potential
	# CASTER already has its crinos modifier when the discipline effects
	# query it (especially for the delirium check, which reads
	# has_character_modifier on the CASTER).
	#
	# Discipline buff CASTER assignment:
	#   Slots 0-4 (player team) → scope:pod_combat_target  (enemy leader)
	#   Slots 5-9 (enemy team)  → scope:combat_pc          (the PC)
	# Ally buff uses the character themselves as CASTER.
	$CHARACTER$ = {
		# Discipline combat buffs. Uses named scopes (not root) to match
		# PoD's own duel pattern and avoid text-substitution issues in
		# nested parametric effects.
		if = {
			limit = { $SLOT$ <= 4 }
			# Player team — enemy leader is the opponent passed into group init
			POD_enemy_knight_set_discipline_battle_effect = { CASTER = scope:pod_combat_target }
		}
		else = {
			# Enemy team — enemy leader is the PC
			POD_enemy_knight_set_discipline_battle_effect = { CASTER = scope:combat_pc }
		}
		POD_ally_knight_set_discipline_battle_effect = { CASTER = $CHARACTER$ }
		force_character_skill_recalculation = yes
	}
	# Reads health track, soak, prowess from the character and stores in slot variables
	$CHARACTER$ = {
		save_scope_value_as = {
			name = slot_track_length
			value = POD_health_track_length
		}
		save_scope_value_as = {
			name = slot_empty_boxes
			value = POD_empty_damage_boxes
		}
		save_scope_value_as = {
			name = slot_soak
			value = POD_soak_chance
		}
		save_scope_value_as = {
			name = slot_prowess
			value = prowess
		}
	}

	set_variable = { name = pod_combat_c$SLOT$_soak value = scope:slot_soak }
	set_variable = { name = pod_combat_c$SLOT$_base_soak value = scope:slot_soak }
	set_variable = { name = pod_combat_c$SLOT$_morale value = pod_combat_max_morale }
	set_variable = { name = pod_combat_c$SLOT$_prowess value = scope:slot_prowess }
	set_variable = { name = pod_combat_c$SLOT$_alive value = 1 }
	# AI energy-use counter — increments on every energy-cost skill this
	# combatant fires. The AI move picker layers stacking ADDITIVE weight
	# penalties on top of this counter (-8 / -10 / -12 / -14 at thresholds
	# 1 / 2 / 3 / 4+, cumulating to 0 / -8 / -18 / -30 / -44) to bias
	# selection AWAY from energy spends as the pool depletes. This is
	# additive selection-bias. The AI may actualy pays the splat energy 
	# on top of this. Energy-cost moves route through the same
	# POD_spend_splat_agnostic_energy path the PC uses.
	set_variable = { name = pod_combat_c$SLOT$_energy_used value = 0 }
	# ── Spirit / incorporeality capability flags ─────────────────
	# Mirrors PoD's POD_damage_spirit_custom_trigger semantics for the
	# minigame. Two per-slot flags are captured once at init and never
	# change during combat:
	#   incorporeal_spirit — target is an incorporeal spirit (ghost,
	#       Umbral entity, non-Risen wraith, etc.) that can't be harmed
	#       by mundane physical attacks. Checked via PoD's own
	#       POD_can_be_killed_physically_trigger = no.
	#   can_damage_spirit  — actor has the supernatural means to strike
	#       incorporeal spirits (fera/spirit/fae/wraith/true_faith, or
	#       specific perks / artifacts). Checked via PoD's own
	#       POD_can_damage_spirit_trigger = yes.
	# These gate the four Physical damage pipelines at the top: if the
	# target is incorporeal and the actor can't damage spirits, the
	# attack auto-misses. Supernatural / Mental / Fire / Sunlight /
	# Silver / Gold / Holy pipelines are unaffected — they connect
	# through incorporeality normally.
	remove_variable = pod_combat_c$SLOT$_incorporeal_spirit
	if = {
		limit = { $CHARACTER$ = { POD_can_be_killed_physically_trigger = no } }
		set_variable = { name = pod_combat_c$SLOT$_incorporeal_spirit value = yes }
	}
	remove_variable = pod_combat_c$SLOT$_can_damage_spirit
	if = {
		limit = { $CHARACTER$ = { POD_can_damage_spirit_trigger = yes } }
		set_variable = { name = pod_combat_c$SLOT$_can_damage_spirit value = yes }
	}
	remove_variable = pod_combat_c$SLOT$_is_guarding
	remove_variable = pod_combat_c$SLOT$_is_dodging
	# ── AI per-fight transient state ─────────────────────────────
	# AI loadouts and anti-repetition history (last / prev / prev2
	# move) are per-fight scratch state. Clear them on slot init so
	# every fight seeds from the current move catalogue and starts
	# with empty history. PC slot (slot 0) is skipped — the player's
	# loadout is persistent equipment configured via the loadout
	# decision, and the PC has no AI history vars to clear.
	if = {
		limit = { $CHARACTER$ = { is_ai = yes } }
		$CHARACTER$ = {
			clear_variable_list = pod_combat_loadout
			remove_variable = pod_combat_ai_last_move
			remove_variable = pod_combat_ai_prev_move
			remove_variable = pod_combat_ai_prev2_move
		}
	}
	# ── State marker modifier ────────────────────────────────────
	# Apply pod_combat_duel_active to every combatant so other
	# systems (feeding, schemes, travel, activity events from
	# other mods) can gate on `has_character_modifier` and defer
	# firing mid-duel. Normal cleanup happens via the
	# pod_on_combat_duel_end on_action; `days = 30` is a safety
	# net for edge cases (crash, save corruption, mod conflict
	# blocking the on_action). Combat minigames end in seconds of
	# game time, so 30 days never clips a real duel and keeps
	# stale state bounded if anything goes wrong with cleanup.
	$CHARACTER$ = {
		add_character_modifier = {
			modifier = pod_combat_duel_active
			days     = 30
		}
	}
	# Hit chance calculated later after all combatants are initialized
}

##############################################################################################
# SECTION 6: GROUP INITIALIZATION
# Sets up the full group combat: PC in slot 0, primary opponent in slot 5,
# allies in slots 1-4 (modes 2/3), enemy reinforcements in slots 6-9 (modes 1/3).
# Calculates hit chances and loads initial workspace.
##############################################################################################

pod_combat_initialize_effect = {
	# $OPPONENT$ = primary target character
	# $MODE$ = 0 (1v1), 1 (1vN), 2 (Nv1), 3 (NvN)

	$OPPONENT$ = { save_scope_as = pod_combat_target }

	# Default loadout if no moves equipped (list-based after the refactor;
	# POD_apply_default_combat_loadout iterates POD_combat_moves and adds
	# every story with is_starter_move set).
	if = {
		limit = { pod_combat_no_moves_equipped_trigger = yes }
		POD_apply_default_combat_loadout = yes
	}

	# Close config window if open
	remove_variable = pod_combat_config_open

	# Defensive: clear is_over flag in case a previous combat left it stale
	# (e.g. old save file, crash, or cleanup regression)
	remove_variable = pod_combat_is_over

	# Set mode
	set_variable = { name = pod_combat_mode value = $MODE$ }
	set_variable = { name = pod_combat_active value = yes }
	set_variable = { name = pod_combat_round value = 1 }

	# Save the PC scope early so the shapeshift pre-pass can exclude them
	# explicitly. The PC also gets saved as pod_combat_c0_char below.
	save_scope_as = combat_pc

	# -- Pre-pass: shapeshift all AI fera combatants BEFORE any slot init --
	# This must run before the slot loop because each slot's init calls
	# POD_enemy_knight_set_discipline_battle_effect with the opposing team
	# leader as CASTER. That effect's delirium check queries
	# $CASTER$ = { POD_give_delirium_trigger = yes } which requires the
	# crinos character modifier to already be present. If we waited and
	# shifted inside each slot's init, player-team slots 0-4 would query
	# the un-shifted opponent leader (slot 5 hasn't run yet), and slot 5
	# would similarly miss any allied fera that shifted in their own slot.
	# Shapeshifting upfront ensures every CASTER has its warform modifier
	# before any delirium / discipline query runs.
	#
	# The PC is EXPLICITLY EXCLUDED from auto-shapeshift. They may prefer
	# to fight in human form (e.g., to preserve social options post-combat)
	# or may have already shifted manually via an interaction.
	scope:pod_combat_target = {
		if = {
			limit = {
				is_ai = yes
				has_trait = fera
				POD_can_shapeshift_trigger = yes
				NOT = { POD_is_shapeshifted_warform_trigger = yes }
			}
			POD_shapeshift_fera_warform_effect = yes
		}
	}
	# Coterie allies (modes 2 and 3 only — matches the ally pickup loop below)
	if = {
		limit = {
			OR = {
				$MODE$ = 2
				$MODE$ = 3
			}
		}
		every_relation = {
			type = coterie_member
			limit = {
				is_alive = yes
				NOT = { has_trait = incapable }
				is_ai = yes
				has_trait = fera
				POD_can_shapeshift_trigger = yes
				NOT = { POD_is_shapeshifted_warform_trigger = yes }
			}
			POD_shapeshift_fera_warform_effect = yes
		}
	}
	# Opponent's knights (modes 1 and 3 only — matches the knight pickup loop below)
	if = {
		limit = {
			OR = {
				$MODE$ = 1
				$MODE$ = 3
			}
		}
		scope:pod_combat_target = {
			every_knight = {
				limit = {
					is_alive = yes
					NOT = { has_trait = incapable }
					opinion = {
						target = scope:pod_combat_target
						value >= 0
					}
					is_ai = yes
					NOT = { this = scope:combat_pc }
					has_trait = fera
					POD_can_shapeshift_trigger = yes
					NOT = { POD_is_shapeshifted_warform_trigger = yes }
				}
				POD_shapeshift_fera_warform_effect = yes
			}
		}
	}

	# -- Slot 0: PC (always) --
	set_variable = { name = pod_combat_c0_char value = scope:combat_pc }
	pod_combat_init_slot_from_character_effect = { SLOT = 0 CHARACTER = scope:combat_pc }

	# -- Slots 1-4: Coterie allies (modes 2 and 3) --
	if = {
		limit = {
			has_variable = pod_combat_mode
			OR = {
				var:pod_combat_mode = 2
				var:pod_combat_mode = 3
			}
		}
		# Add coterie members sorted by prowess
		every_relation = {
			type = coterie_member
			limit = {
				is_alive = yes
				is_adult = yes
				NOT = { has_trait = incapable }
			}
			save_scope_as = ally_candidate
			scope:combat_pc = {
				pod_combat_assign_ally_slot_effect = { CHARACTER = scope:ally_candidate }
			}
		}
	}

	# -- Slot 5: Primary opponent (always) --
	set_variable = { name = pod_combat_c5_char value = scope:pod_combat_target }
	pod_combat_init_slot_from_character_effect = { SLOT = 5 CHARACTER = scope:pod_combat_target }
	set_variable = { name = pod_combat_enemy_team_size value = 1 }

	# -- Slots 6-9: Opponent's knights (modes 1 and 3) --
	if = {
		limit = {
			has_variable = pod_combat_mode
			OR = {
				var:pod_combat_mode = 1
				var:pod_combat_mode = 3
			}
		}
		scope:pod_combat_target = {
			every_knight = {
				limit = {
					is_alive = yes
					NOT = { has_trait = incapable }
					opinion = {
						target = scope:pod_combat_target
						value >= 0
					}
				}
				save_scope_as = enemy_candidate
				scope:combat_pc = {
					if = {
						limit = { var:pod_combat_enemy_team_size < 5 }
						pod_combat_assign_enemy_slot_effect = { CHARACTER = scope:enemy_candidate }
					}
				}
			}
		}
	}

	# -- Calculate hit chances for all combatants --
	# Formula: base 60 + (own prowess - primary opposing leader's prowess) / 2
	# "Primary opposing leader" = slot 5 (the enemy leader) for player-team combatants 0-4,
	# and slot 0 (the PC) for enemy-team combatants 5-9. NOT an average across the enemy
	# team — pair-specific recalculation kicks in later when an actor/target pair is
	# loaded into the workspace via pod_combat_recalc_player_hit_chance_effect, which
	# re-derives hit_chance against the *currently selected* target instead of the leader.
	pod_combat_calculate_all_hit_chances_effect = yes

	# -- Set up initial target / workspace defaults --
	if = {
		limit = { has_variable = pod_combat_c5_char }
		set_variable = { name = pod_combat_opponent value = var:pod_combat_c5_char }
		# Keep the TARGET_SIDE-addressable alias (pod_combat_opponent_char)
		# in sync with pod_combat_opponent so the damage helper's
		# var:pod_combat_$TARGET_SIDE$_char lookup resolves correctly
		# from the moment combat initializes, even before the first
		# target_load effect runs.
		set_variable = { name = pod_combat_opponent_char value = var:pod_combat_c5_char }
	}
	set_variable = { name = pod_combat_selected_target value = 5 }
	# Initialize the unified PC target — defaults to the primary opponent
	# (slot 5). The player can switch targets mid-combat via the Target
	# button on any combat card (allies 1-4 or enemies 5-9). A single
	# variable covers both team-enemy and single-ally targeting; the
	# can_use triggers enforce that each skill fires against a valid slot.
	set_variable = { name = pod_combat_pc_target value = 5 }
	set_variable = { name = pod_combat_current_ap value = pod_combat_base_ap }
	pod_combat_apply_ap_drain_effect = yes

	# Initialize combat log
	set_variable = { name = pod_combat_last_damage value = 0 }
	# AI anti-repetition history (pod_combat_ai_last_move /
	# _prev_move / _prev2_move) lives on the AI character, not on
	# the combat host (the selector switches into
	# var:pod_combat_actor_char before reading/writing). No PC-side
	# init or remove needed here. Per-fight cleanup of those vars
	# happens in pod_combat_init_slot_from_character_effect (slot
	# entry) and pod_combat_group_cleanup_effect (combat exit).

	# -- Non-fatal duel: gate PoD's inline death check --
	# For fatality = no we add pod_combat_nonfatal_active as a character
	# flag to every combatant. POD_can_die_from_wounds (overridden in
	# common/scripted_triggers/POD_combat/pod_combat_death_gate_overrides.txt)
	# reads the flag and returns no, so POD_health_death_check_effect
	# never enters its pod_death branch during the combat. The flag is
	# stripped post-combat by pod_combat_end_combat_effect AFTER
	# pod_combat_cap_nonfatal_slot_effect has capped everyone to at
	# least one empty box. Fatal duels deliberately don't gate the
	# death check — combatants can die mid-fight; KO via wound-track-
	# full happens regardless via the slot alive-flip check below.
	if = {
		limit = {
			has_variable = pod_combat_sc_context
			var:pod_combat_sc_context.var:fatality = flag:no
		}
		pod_combat_apply_nonfatal_flag_effect = yes
	}

	# -- Determine round 1 turn order --
	# Unified flow: every combat — including 1v1 — rolls initiative and
	# processes any pre-PC turns so nobody gets a free first move. A 1v1
	# is just a degenerate NvN with slots 0 and 5 populated.
	pod_combat_calculate_initiative_effect = yes
	pod_combat_find_next_actor_effect = yes

	if = {
		limit = {
			has_variable = pod_combat_current_actor

		}
		# PC goes first in initiative order — restore PC target, load workspace
		set_variable = { name = pod_combat_current_turn_side value = 0 }
		pod_combat_restore_pc_target_effect = yes
		pod_combat_load_workspace_effect = yes
	}
	else = {
		# An AI goes first — process AI turns until it's the PC's turn
		# or combat ends. The PC is blocked from acting because the
		# turn-side check fires against current_actor != 0.
		pod_combat_process_pre_pc_turns_effect = yes
	}
}

# SECTION 6b: PLAYER HIT CHANCE RECALC ON TARGET SWAP
# Called by pod_combat_select_target_N (slots 5-9) after loading the new
# target into pod_combat_opponent_*. Rebuilds the PC's base_hit_chance
# from the freshly loaded opponent's prowess and propagates to
# pod_combat_c0_base_hit_chance, then runs the workspace and per-slot
# tier recalc so any active hit buff/debuff tiers reapply cleanly against
# the new base. Stealth's +30 hit bonus is outside the tier system and
# stays tied to the stealth flag (consumed via the damage pipeline).
##############################################################################################

pod_combat_recalc_player_hit_chance_effect = {
	# Called when the PC swaps targets. Rebuilds the PC's base_hit_chance
	# from the freshly loaded opponent's prowess, then calls the full tier
	# recalc so any active hit buffs / debuffs reapply cleanly.
	# Round-duration tiers carry over (they expire at round end, not on
	# target swap). Stealth's +30 is a separate mechanic outside the tier
	# system and is handled by its own consume paths.
	if = {
		limit = {
			has_variable = pod_combat_player_prowess
			has_variable = pod_combat_opponent_prowess
		}
		save_scope_value_as = {
			name = pc_recalc_base_hit
			value = {
				value = pod_combat_hit_chance_base
				add = {
					value = var:pod_combat_player_prowess
					subtract = var:pod_combat_opponent_prowess
					divide = 2
				}
				min = 20
				max = 95
			}
		}
		set_variable = { name = pod_combat_player_base_hit_chance value = scope:pc_recalc_base_hit }
		set_variable = { name = pod_combat_c0_base_hit_chance value = scope:pc_recalc_base_hit }
		pod_combat_recalc_hit_player_effect = yes
		pod_combat_recalc_slot_stats_effect = { SLOT = 0 }
	}
}

# uniformly weighted (1 each in random_list) but the values are
# discrete, which produces a coarser distribution than continuous 1-30

##############################################################################################
# SECTION 10b: PC TARGET PERSISTENCE (UNIFIED)
#
# pod_combat_pc_target is a single slot index (1-9) representing whomever the PC has
# currently selected via the Target button on any combat card — ally OR enemy. A skill's
# target-type (single-enemy / single-ally / team-*) is enforced by the can_use trigger,
# not by having separate target variables.
#
# Slot 0 (the PC themselves) is never a valid stored target — self-only skills use
# targeting = self, not pc_target = 0.
#
# This effect is called at PC turn start AND after every PC move, so if the PC's target
# just got KO'd, it auto-retargets to the first alive non-PC slot (preferring enemies
# first, then allies). Combat should have ended before we run out of candidates.
##############################################################################################

pod_combat_restore_pc_target_effect = {
	if = {
		limit = {
			OR = {
				NOT = { has_variable = pod_combat_pc_target }
				var:pod_combat_pc_target = 0
				AND = {
					var:pod_combat_pc_target = 1
					OR = { NOT = { has_variable = pod_combat_c1_char } var:pod_combat_c1_alive = 0 }
				}
				AND = {
					var:pod_combat_pc_target = 2
					OR = { NOT = { has_variable = pod_combat_c2_char } var:pod_combat_c2_alive = 0 }
				}
				AND = {
					var:pod_combat_pc_target = 3
					OR = { NOT = { has_variable = pod_combat_c3_char } var:pod_combat_c3_alive = 0 }
				}
				AND = {
					var:pod_combat_pc_target = 4
					OR = { NOT = { has_variable = pod_combat_c4_char } var:pod_combat_c4_alive = 0 }
				}
				AND = {
					var:pod_combat_pc_target = 5
					OR = { NOT = { has_variable = pod_combat_c5_char } var:pod_combat_c5_alive = 0 }
				}
				AND = {
					var:pod_combat_pc_target = 6
					OR = { NOT = { has_variable = pod_combat_c6_char } var:pod_combat_c6_alive = 0 }
				}
				AND = {
					var:pod_combat_pc_target = 7
					OR = { NOT = { has_variable = pod_combat_c7_char } var:pod_combat_c7_alive = 0 }
				}
				AND = {
					var:pod_combat_pc_target = 8
					OR = { NOT = { has_variable = pod_combat_c8_char } var:pod_combat_c8_alive = 0 }
				}
				AND = {
					var:pod_combat_pc_target = 9
					OR = { NOT = { has_variable = pod_combat_c9_char } var:pod_combat_c9_alive = 0 }
				}
			}
		}
		# Auto-retarget: prefer alive enemies (5-9) first, then allies (1-4).
		# If nothing is alive we fall through silently; combat is effectively over.
		if = {
			limit = { has_variable = pod_combat_c5_char var:pod_combat_c5_alive = 1 }
			set_variable = { name = pod_combat_pc_target value = 5 }
			set_variable = { name = pod_combat_selected_target value = 5 }
			pod_combat_load_target_from_slot_effect = { SLOT = 5 }
		}
		else_if = {
			limit = { has_variable = pod_combat_c6_char var:pod_combat_c6_alive = 1 }
			set_variable = { name = pod_combat_pc_target value = 6 }
			set_variable = { name = pod_combat_selected_target value = 6 }
			pod_combat_load_target_from_slot_effect = { SLOT = 6 }
		}
		else_if = {
			limit = { has_variable = pod_combat_c7_char var:pod_combat_c7_alive = 1 }
			set_variable = { name = pod_combat_pc_target value = 7 }
			set_variable = { name = pod_combat_selected_target value = 7 }
			pod_combat_load_target_from_slot_effect = { SLOT = 7 }
		}
		else_if = {
			limit = { has_variable = pod_combat_c8_char var:pod_combat_c8_alive = 1 }
			set_variable = { name = pod_combat_pc_target value = 8 }
			set_variable = { name = pod_combat_selected_target value = 8 }
			pod_combat_load_target_from_slot_effect = { SLOT = 8 }
		}
		else_if = {
			limit = { has_variable = pod_combat_c9_char var:pod_combat_c9_alive = 1 }
			set_variable = { name = pod_combat_pc_target value = 9 }
			set_variable = { name = pod_combat_selected_target value = 9 }
			pod_combat_load_target_from_slot_effect = { SLOT = 9 }
		}
		else_if = {
			limit = { has_variable = pod_combat_c1_char var:pod_combat_c1_alive = 1 }
			set_variable = { name = pod_combat_pc_target value = 1 }
			set_variable = { name = pod_combat_selected_target value = 1 }
			pod_combat_load_target_from_slot_effect = { SLOT = 1 }
		}
		else_if = {
			limit = { has_variable = pod_combat_c2_char var:pod_combat_c2_alive = 1 }
			set_variable = { name = pod_combat_pc_target value = 2 }
			set_variable = { name = pod_combat_selected_target value = 2 }
			pod_combat_load_target_from_slot_effect = { SLOT = 2 }
		}
		else_if = {
			limit = { has_variable = pod_combat_c3_char var:pod_combat_c3_alive = 1 }
			set_variable = { name = pod_combat_pc_target value = 3 }
			set_variable = { name = pod_combat_selected_target value = 3 }
			pod_combat_load_target_from_slot_effect = { SLOT = 3 }
		}
		else_if = {
			limit = { has_variable = pod_combat_c4_char var:pod_combat_c4_alive = 1 }
			set_variable = { name = pod_combat_pc_target value = 4 }
			set_variable = { name = pod_combat_selected_target value = 4 }
			pod_combat_load_target_from_slot_effect = { SLOT = 4 }
		}
	}
	else = {
		# Current PC target is still alive — mirror pc_target into selected_target
		# (in case an AI clobbered it with their own target during the previous turn)
		# and reload the target workspace with the PC's choice.
		set_variable = { name = pod_combat_selected_target value = var:pod_combat_pc_target }
		if = { limit = { var:pod_combat_pc_target = 1 } pod_combat_load_target_from_slot_effect = { SLOT = 1 } }
		else_if = { limit = { var:pod_combat_pc_target = 2 } pod_combat_load_target_from_slot_effect = { SLOT = 2 } }
		else_if = { limit = { var:pod_combat_pc_target = 3 } pod_combat_load_target_from_slot_effect = { SLOT = 3 } }
		else_if = { limit = { var:pod_combat_pc_target = 4 } pod_combat_load_target_from_slot_effect = { SLOT = 4 } }
		else_if = { limit = { var:pod_combat_pc_target = 5 } pod_combat_load_target_from_slot_effect = { SLOT = 5 } }
		else_if = { limit = { var:pod_combat_pc_target = 6 } pod_combat_load_target_from_slot_effect = { SLOT = 6 } }
		else_if = { limit = { var:pod_combat_pc_target = 7 } pod_combat_load_target_from_slot_effect = { SLOT = 7 } }
		else_if = { limit = { var:pod_combat_pc_target = 8 } pod_combat_load_target_from_slot_effect = { SLOT = 8 } }
		else_if = { limit = { var:pod_combat_pc_target = 9 } pod_combat_load_target_from_slot_effect = { SLOT = 9 } }
	}
}

# SECTION 11: GROUP TURN LOOP
# The main group turn processing effect. Called after the PC ends their turn (or at round
# start for non-PC actors). Processes all remaining actors in initiative order.
##############################################################################################

pod_combat_process_group_turns_effect = {
	# Process all remaining actors in initiative order
	# This is called after the PC acts (slot 0) to run all AI turns

	# Mark current actor (PC) as acted
	set_variable = { name = pod_combat_c0_has_acted value = 1 }

	# Save PC's workspace back to slot 0
	pod_combat_save_workspace_effect = yes

	# Loop: find next actor, run their turn, repeat until all have acted

	# Iteration 1
	pod_combat_find_next_actor_effect = yes
	if = {
		limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 NOT = { has_variable = pod_combat_is_over } }
		pod_combat_execute_ai_turn_effect = yes
	}
	# Iteration 2
	if = {
		limit = { NOT = { has_variable = pod_combat_is_over } }
		pod_combat_find_next_actor_effect = yes
		if = {
			limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 NOT = { has_variable = pod_combat_is_over } }
			pod_combat_execute_ai_turn_effect = yes
		}
	}
	# Iteration 3
	if = {
		limit = { NOT = { has_variable = pod_combat_is_over } }
		pod_combat_find_next_actor_effect = yes
		if = {
			limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 NOT = { has_variable = pod_combat_is_over } }
			pod_combat_execute_ai_turn_effect = yes
		}
	}
	# Iteration 4
	if = {
		limit = { NOT = { has_variable = pod_combat_is_over } }
		pod_combat_find_next_actor_effect = yes
		if = {
			limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 NOT = { has_variable = pod_combat_is_over } }
			pod_combat_execute_ai_turn_effect = yes
		}
	}
	# Iteration 5
	if = {
		limit = { NOT = { has_variable = pod_combat_is_over } }
		pod_combat_find_next_actor_effect = yes
		if = {
			limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 NOT = { has_variable = pod_combat_is_over } }
			pod_combat_execute_ai_turn_effect = yes
		}
	}
	# Iteration 6
	if = {
		limit = { NOT = { has_variable = pod_combat_is_over } }
		pod_combat_find_next_actor_effect = yes
		if = {
			limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 NOT = { has_variable = pod_combat_is_over } }
			pod_combat_execute_ai_turn_effect = yes
		}
	}
	# Iteration 7
	if = {
		limit = { NOT = { has_variable = pod_combat_is_over } }
		pod_combat_find_next_actor_effect = yes
		if = {
			limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 NOT = { has_variable = pod_combat_is_over } }
			pod_combat_execute_ai_turn_effect = yes
		}
	}
	# Iteration 8
	if = {
		limit = { NOT = { has_variable = pod_combat_is_over } }
		pod_combat_find_next_actor_effect = yes
		if = {
			limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 NOT = { has_variable = pod_combat_is_over } }
			pod_combat_execute_ai_turn_effect = yes
		}
	}
	# Iteration 9
	if = {
		limit = { NOT = { has_variable = pod_combat_is_over } }
		pod_combat_find_next_actor_effect = yes
		if = {
			limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 NOT = { has_variable = pod_combat_is_over } }
			pod_combat_execute_ai_turn_effect = yes
		}
	}

	# All actors have acted — end round if combat not over
	if = {
		limit = { NOT = { has_variable = pod_combat_is_over } }
		pod_combat_group_end_round_effect = yes
	}
}

pod_combat_execute_ai_turn_effect = {
	# Execute one AI turn for the current actor
	# 1. Select target based on actor's team
	# 2. Load workspace
	# 3. Run AI move selection
	# 4. Save workspace
	# 5. Mark as acted
	# 6. Check victory

	# Select target
	if = {
		limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor <= 4 }
		# Actor is on player team — target an enemy
		pod_combat_ai_select_enemy_target_effect = yes
	}
	else = {
		# Actor is on enemy team — target a player
		pod_combat_ai_select_player_target_effect = yes
	}

	# Load workspace (actor -> player workspace, target -> opponent workspace)
	pod_combat_load_workspace_effect = yes

	# Set AP for this actor's turn
	set_variable = { name = pod_combat_current_ap value = pod_combat_base_ap }
	pod_combat_apply_ap_drain_effect = yes

	# Set turn side for backward compat with AI effects
	if = {
		limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor <= 4 }
		set_variable = { name = pod_combat_current_turn_side value = 0 }
	}
	else = {
		set_variable = { name = pod_combat_current_turn_side value = 1 }
	}

	# Run AI moves (uses existing pod_combat_ai_take_turn_effect which reads from workspace)
	pod_combat_ai_take_turn_effect = yes

	# Save workspace back to slots
	pod_combat_save_workspace_effect = yes

	# Mark actor as acted
	if = { limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor = 0 } set_variable = { name = pod_combat_c0_has_acted value = 1 } }
	else_if = { limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor = 1 } set_variable = { name = pod_combat_c1_has_acted value = 1 } }
	else_if = { limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor = 2 } set_variable = { name = pod_combat_c2_has_acted value = 1 } }
	else_if = { limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor = 3 } set_variable = { name = pod_combat_c3_has_acted value = 1 } }
	else_if = { limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor = 4 } set_variable = { name = pod_combat_c4_has_acted value = 1 } }
	else_if = { limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor = 5 } set_variable = { name = pod_combat_c5_has_acted value = 1 } }
	else_if = { limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor = 6 } set_variable = { name = pod_combat_c6_has_acted value = 1 } }
	else_if = { limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor = 7 } set_variable = { name = pod_combat_c7_has_acted value = 1 } }
	else_if = { limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor = 8 } set_variable = { name = pod_combat_c8_has_acted value = 1 } }
	else_if = { limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor = 9 } set_variable = { name = pod_combat_c9_has_acted value = 1 } }

	# Check victory
	pod_combat_check_group_victory_effect = yes
}

##############################################################################################
# SECTION 12: GROUP VICTORY CHECK
# Check if all enemies are knocked out (player wins) or all allies are knocked out
# (player loses).
##############################################################################################

pod_combat_check_group_victory_effect = {
	# Victory conditions (evaluated in order, first match ends combat):
	#   1. All enemies KO'd           → player team wins
	#   2. PC (slot 0) KO'd            → player loses (PC can't act, so combat cannot continue)
	#   3. All allies KO'd             → player loses

	# Save the PC scope so we can reference it via scope:combat_pc in set_variable
	# (using bare `root` there triggers "Event target link 'root' returned unset scope")
	save_scope_as = combat_pc

	# --- Check 1: All enemies knocked out ---
	save_scope_value_as = { name = enemies_alive value = 0 }
	if = { limit = { has_variable = pod_combat_c5_char var:pod_combat_c5_alive = 1 } save_scope_value_as = { name = enemies_alive value = { value = scope:enemies_alive add = 1 } } }
	if = { limit = { has_variable = pod_combat_c6_char var:pod_combat_c6_alive = 1 } save_scope_value_as = { name = enemies_alive value = { value = scope:enemies_alive add = 1 } } }
	if = { limit = { has_variable = pod_combat_c7_char var:pod_combat_c7_alive = 1 } save_scope_value_as = { name = enemies_alive value = { value = scope:enemies_alive add = 1 } } }
	if = { limit = { has_variable = pod_combat_c8_char var:pod_combat_c8_alive = 1 } save_scope_value_as = { name = enemies_alive value = { value = scope:enemies_alive add = 1 } } }
	if = { limit = { has_variable = pod_combat_c9_char var:pod_combat_c9_alive = 1 } save_scope_value_as = { name = enemies_alive value = { value = scope:enemies_alive add = 1 } } }

	if = {
		limit = {
			scope:enemies_alive <= 0
			has_variable = pod_combat_c5_char
			NOT = { has_variable = pod_combat_is_over }
		}
		# Player team wins!
		set_variable = { name = pod_combat_winner value = scope:combat_pc }
		set_variable = { name = pod_combat_loser value = var:pod_combat_c5_char }
		set_variable = { name = pod_combat_is_over value = yes }
		var:pod_combat_c5_char = { save_scope_as = pod_combat_target }
		# Combat window stays open — player clicks "End Combat" to dismiss
	}

	# --- Check 2: PC knocked out (even if allies still standing) ---
	# The PC is special — the player has no way to continue the fight if they are down.
	# End combat as a loss so the post-fight event can resolve wounds and outcomes.
	if = {
		limit = {
			has_variable = pod_combat_c0_alive
			var:pod_combat_c0_alive = 0
			has_variable = pod_combat_c5_char
			NOT = { has_variable = pod_combat_is_over }
		}
		set_variable = { name = pod_combat_winner value = var:pod_combat_c5_char }
		set_variable = { name = pod_combat_loser value = scope:combat_pc }
		set_variable = { name = pod_combat_is_over value = yes }
		var:pod_combat_c5_char = { save_scope_as = pod_combat_target }
		# Combat window stays open — player clicks "End Combat" to dismiss
	}

	# --- Check 3: All allies knocked out (covers edge cases where PC death check was bypassed) ---
	save_scope_value_as = { name = allies_alive value = 0 }
	if = { limit = { has_variable = pod_combat_c0_char var:pod_combat_c0_alive = 1 } save_scope_value_as = { name = allies_alive value = { value = scope:allies_alive add = 1 } } }
	if = { limit = { has_variable = pod_combat_c1_char var:pod_combat_c1_alive = 1 } save_scope_value_as = { name = allies_alive value = { value = scope:allies_alive add = 1 } } }
	if = { limit = { has_variable = pod_combat_c2_char var:pod_combat_c2_alive = 1 } save_scope_value_as = { name = allies_alive value = { value = scope:allies_alive add = 1 } } }
	if = { limit = { has_variable = pod_combat_c3_char var:pod_combat_c3_alive = 1 } save_scope_value_as = { name = allies_alive value = { value = scope:allies_alive add = 1 } } }
	if = { limit = { has_variable = pod_combat_c4_char var:pod_combat_c4_alive = 1 } save_scope_value_as = { name = allies_alive value = { value = scope:allies_alive add = 1 } } }

	if = {
		limit = {
			scope:allies_alive <= 0
			has_variable = pod_combat_c5_char
			NOT = { has_variable = pod_combat_is_over }
		}
		# Enemy team wins
		set_variable = { name = pod_combat_winner value = var:pod_combat_c5_char }
		set_variable = { name = pod_combat_loser value = scope:combat_pc }
		set_variable = { name = pod_combat_is_over value = yes }
		var:pod_combat_c5_char = { save_scope_as = pod_combat_target }
		# Combat window stays open — player clicks "End Combat" to dismiss
	}
}

##############################################################################################
# SECTION 12B: PER-TURN SLOT STATE RESET
# Resets round-only buffs (soak → base_soak) on ALL 10 slots at the
# start of each new round. Without this, round-only soak buffs leak
# across rounds because the old workspace-only clear didn't write back
# to the slots.
##############################################################################################

pod_combat_clear_per_turn_slot_state_effect = {
	# For each slot: reset live soak to base_soak. Only act on slots
	# that exist (have a character assigned).
	#
	# NOTE: guard / dodge / stealth stances are NOT cleared here — they
	# persist across rounds and are only consumed by an enemy attack (in
	# the damage pipeline) or wiped at combat end (pod_combat_cleanup_effect).
	# Stealth's +30 hit bonus similarly stays until consumed, since it's
	# tied to the stealth flag and the damage pipeline reverts it when
	# the stealth is triggered.
	if = { limit = { has_variable = pod_combat_c0_char }
		if = { limit = { has_variable = pod_combat_c0_base_soak }
			set_variable = { name = pod_combat_c0_soak value = var:pod_combat_c0_base_soak }
		}
	}
	if = { limit = { has_variable = pod_combat_c1_char }
		if = { limit = { has_variable = pod_combat_c1_base_soak }
			set_variable = { name = pod_combat_c1_soak value = var:pod_combat_c1_base_soak }
		}
	}
	if = { limit = { has_variable = pod_combat_c2_char }
		if = { limit = { has_variable = pod_combat_c2_base_soak }
			set_variable = { name = pod_combat_c2_soak value = var:pod_combat_c2_base_soak }
		}
	}
	if = { limit = { has_variable = pod_combat_c3_char }
		if = { limit = { has_variable = pod_combat_c3_base_soak }
			set_variable = { name = pod_combat_c3_soak value = var:pod_combat_c3_base_soak }
		}
	}
	if = { limit = { has_variable = pod_combat_c4_char }
		if = { limit = { has_variable = pod_combat_c4_base_soak }
			set_variable = { name = pod_combat_c4_soak value = var:pod_combat_c4_base_soak }
		}
	}
	if = { limit = { has_variable = pod_combat_c5_char }
		if = { limit = { has_variable = pod_combat_c5_base_soak }
			set_variable = { name = pod_combat_c5_soak value = var:pod_combat_c5_base_soak }
		}
	}
	if = { limit = { has_variable = pod_combat_c6_char }
		if = { limit = { has_variable = pod_combat_c6_base_soak }
			set_variable = { name = pod_combat_c6_soak value = var:pod_combat_c6_base_soak }
		}
	}
	if = { limit = { has_variable = pod_combat_c7_char }
		if = { limit = { has_variable = pod_combat_c7_base_soak }
			set_variable = { name = pod_combat_c7_soak value = var:pod_combat_c7_base_soak }
		}
	}
	if = { limit = { has_variable = pod_combat_c8_char }
		if = { limit = { has_variable = pod_combat_c8_base_soak }
			set_variable = { name = pod_combat_c8_soak value = var:pod_combat_c8_base_soak }
		}
	}
	if = { limit = { has_variable = pod_combat_c9_char }
		if = { limit = { has_variable = pod_combat_c9_base_soak }
			set_variable = { name = pod_combat_c9_soak value = var:pod_combat_c9_base_soak }
		}
	}
}

##############################################################################################
# SECTION 13: GROUP ROUND END + HEALTH WRITE-BACK
# Apply health track damage for all alive combatants, increment round, calculate new
# initiative, and set up the next round.
##############################################################################################

pod_combat_group_end_round_effect = {
	# Guard: if combat ended during this round (victory check fired
	# pod_combat_end_combat_effect which may have cleaned up variables),
	# skip the round-end logic entirely.
	if = {
		limit = { NOT = { has_variable = pod_combat_round } }
		# Combat already cleaned up — nothing to do
	}
	else = {

	# Apply health track damage for all alive combatants
	# Routed through the wrapper so fatality mode (practice/no) is respected.
	pod_combat_apply_nonfatal_effect = yes

	# Increment round
	change_variable = { name = pod_combat_round add = 1 }

	# Calculate initiative for next round
	pod_combat_calculate_initiative_effect = yes

	# Find the first actor (highest initiative)
	pod_combat_find_next_actor_effect = yes

	# Reset live soak to base_soak on every slot. Guard / Dodge /
	# Stealth flags are intentionally NOT cleared here — those stances
	# persist until consumed by an attack or combat ends. See
	# pod_combat_clear_per_turn_slot_state_effect for details.
	pod_combat_clear_per_turn_slot_state_effect = yes

	# Expire all round-duration buff/debuff tiers on every slot, then
	# rebuild each slot's live soak and hit_chance from base + remaining
	# combat-duration tier magnitudes. Recalc-from-base avoids any
	# clamping loss that a delta-based revert would accumulate.
	pod_combat_revert_round_tiers_effect = yes

	# If PC (slot 0) goes first, set up workspace and wait for player input
	if = {
		limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor = 0 }
		set_variable = { name = pod_combat_current_ap value = pod_combat_base_ap }
		pod_combat_apply_ap_drain_effect = yes
		# Restore the PC's chosen target (may have been clobbered by AI turns last round)
		pod_combat_restore_pc_target_effect = yes
		# Load workspace for PC acting against current selected target
		pod_combat_load_workspace_effect = yes
	}
	else = {
		# Non-PC goes first — process all AI turns until it's PC's turn or round ends
		pod_combat_process_pre_pc_turns_effect = yes
	}

	} # end else (combat not yet cleaned up)
}

pod_combat_process_pre_pc_turns_effect = {
	# Process AI turns that happen before PC's turn in initiative order
	# Unrolled loop: up to 9 iterations

	# Current actor is already set by find_next_actor
	if = {
		limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 var:pod_combat_current_actor != 0 NOT = { has_variable = pod_combat_is_over } }
		pod_combat_execute_ai_turn_effect = yes

		# Find next actor
		pod_combat_find_next_actor_effect = yes
		if = {
			limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 var:pod_combat_current_actor != 0 NOT = { has_variable = pod_combat_is_over } }
			pod_combat_execute_ai_turn_effect = yes

			pod_combat_find_next_actor_effect = yes
			if = {
				limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 var:pod_combat_current_actor != 0 NOT = { has_variable = pod_combat_is_over } }
				pod_combat_execute_ai_turn_effect = yes

				pod_combat_find_next_actor_effect = yes
				if = {
					limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 var:pod_combat_current_actor != 0 NOT = { has_variable = pod_combat_is_over } }
					pod_combat_execute_ai_turn_effect = yes

					pod_combat_find_next_actor_effect = yes
					if = {
						limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 var:pod_combat_current_actor != 0 NOT = { has_variable = pod_combat_is_over } }
						pod_combat_execute_ai_turn_effect = yes

						pod_combat_find_next_actor_effect = yes
						if = {
							limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 var:pod_combat_current_actor != 0 NOT = { has_variable = pod_combat_is_over } }
							pod_combat_execute_ai_turn_effect = yes

							pod_combat_find_next_actor_effect = yes
							if = {
								limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 var:pod_combat_current_actor != 0 NOT = { has_variable = pod_combat_is_over } }
								pod_combat_execute_ai_turn_effect = yes

								pod_combat_find_next_actor_effect = yes
								if = {
									limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 var:pod_combat_current_actor != 0 NOT = { has_variable = pod_combat_is_over } }
									pod_combat_execute_ai_turn_effect = yes

									pod_combat_find_next_actor_effect = yes
									if = {
										limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor >= 0 var:pod_combat_current_actor != 0 NOT = { has_variable = pod_combat_is_over } }
										pod_combat_execute_ai_turn_effect = yes
									}
								}
							}
						}
					}
				}
			}
		}
	}

	# If PC is next (or combat is over), set up workspace for PC
	if = {
		limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor = 0 NOT = { has_variable = pod_combat_is_over } }
		set_variable = { name = pod_combat_current_ap value = pod_combat_base_ap }
		pod_combat_apply_ap_drain_effect = yes
		# Pull fresh HP / max / agg_damage for every slot from the
		# characters' PoD wound state, so the combat window shows the
		# right bars when the move bar becomes interactive. AI turns
		# mutated char wounds directly (no slot refresh between AI
		# actions), so the slot vars need a pre-render sync.

		# Restore the PC's chosen target (AI turns just clobbered selected_target)
		pod_combat_restore_pc_target_effect = yes
		pod_combat_load_workspace_effect = yes
		remove_variable = pod_combat_player_is_guarding
		remove_variable = pod_combat_opponent_is_guarding
		remove_variable = pod_combat_player_is_dodging
		remove_variable = pod_combat_opponent_is_dodging
		set_variable = { name = pod_combat_player_soak value = var:pod_combat_player_base_soak }
		set_variable = { name = pod_combat_opponent_soak value = var:pod_combat_opponent_base_soak }
	}
}
