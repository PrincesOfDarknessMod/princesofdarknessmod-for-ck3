######################################################
######   POD COMBAT - SC-CONTEXT HAND-OFF HELPERS
######################################################
# Helpers for pod_combat.9999, the post-combat dispatcher. All run on the
# combat host (PC), which carries a `var:pod_combat_session` pointer to
# the fight's POD_combat_fight story.
#
# Two paths reach .9999, and they differ in what is already in the scope
# table when they arrive:
#
#   Live path - the shell event (pod_combat.9995) inherited the launching
#   event's entire scope table and held it open for the fight, so the sc_*
#   principals, the rule flags and every parent alias (scope:spirit,
#   scope:task_contract, the artifact scope:target, ...) are simply present.
#   Only the outcome, which combat itself produced, has to be derived.
#
#   Fallback path - .9999 was reached from a raw GUI execution, where the
#   scope table is empty. The principals and rule flags are then rebuilt from
#   the context story; the parent's own scopes are simply gone, so the
#   follow-up runs degraded. Nothing routes here today: every fight is built
#   through POD_combat_create_duel_effect, which opens the card. It stands as
#   the defensive arm for a fight seated without the builder, and a
#   card-carrying fight reaching it is a defect the dispatcher error_logs.
#
# The dispatcher tells the paths apart by testing `exists = scope:sc_attacker`
# and binds scope:sc_ctx itself before either branch, since both need it.

##############################################################################
# 1. Derive the outcome (both paths)
##############################################################################

# sc_victor / sc_loser come from combat-owned host vars, so they are derived
# identically whichever path arrived. Victor/loser are absent for draws and
# mutual KOs, hence the guard. sc_finished / victory_type are the constants
# every duel follow-up expects to read.
POD_combat_derive_sc_outcome_effect = {
	if = {
		limit = {
			var:pod_combat_session ?= {
				has_variable = pod_combat_winner
				has_variable = pod_combat_loser
			}
		}
		var:pod_combat_session.var:pod_combat_winner = { save_scope_as = sc_victor }
		var:pod_combat_session.var:pod_combat_loser  = { save_scope_as = sc_loser }
		# Snap a non-principal victor to the winning side's principal.
		# The outcome pair can record a SUMMON as winner (PC lost, first
		# alive enemy was a summoned combatant), but vanilla duels only
		# ever name the two duelists and every follow-up identity compare
		# (sc_victor = scope:recidivist, = scope:spirit, ...) assumes
		# that. The loser is always a principal (the PC or the enemy
		# leader), so the victor is whichever principal the loser isn't.
		# The snapped principal can be dead (fell before its summon
		# finished the fight); follow-ups already tolerate dead victors.
		if = {
			limit = {
				NOT = { scope:sc_victor = scope:sc_attacker }
				NOT = { scope:sc_victor = scope:sc_defender }
			}
			if = {
				limit = { scope:sc_loser = scope:sc_attacker }
				scope:sc_defender = { save_scope_as = sc_victor }
			}
			else = {
				scope:sc_attacker = { save_scope_as = sc_victor }
			}
		}
	}

	save_scope_value_as = { name = sc_finished  value = yes }
	save_scope_value_as = { name = victory_type value = flag:skill }
}


##############################################################################
# 2. Fallback: rebuild the principals from the context story
##############################################################################

# Degraded path only (see the file header). The alias saves are fabrications
# standing in for whatever the parent event actually named: on the live path
# the parent's own scope:target / scope:victim / ... arrive intact and these
# approximations never run, which is what makes the artifact-duel class of
# clobber bug structurally impossible there.
POD_combat_restore_sc_context_effect = {
	scope:sc_ctx.var:initiator = { save_scope_as = sc_initiator }
	scope:sc_ctx.var:attacker  = { save_scope_as = sc_attacker }
	scope:sc_ctx.var:defender  = { save_scope_as = sc_defender }

	# Defender aliases
	scope:sc_ctx.var:defender = { save_scope_as = victim }
	scope:sc_ctx.var:defender = { save_scope_as = recipient }
	scope:sc_ctx.var:defender = { save_scope_as = defender }
	scope:sc_ctx.var:defender = { save_scope_as = target }
	scope:sc_ctx.var:defender = { save_scope_as = adversary }

	# Attacker aliases
	scope:sc_ctx.var:attacker = { save_scope_as = attacker }
	scope:sc_ctx.var:attacker = { save_scope_as = challenger }
}


##############################################################################
# 3. Fallback: reconstruct the rule-flag scope values
##############################################################################

# Degraded path only - on the live path fatality / fixed / locale ride the
# shell event as the parent's own scope values, flag type intact.
#
# Why the enum chains: save_scope_value_as takes a literal flag value
# (`flag:X`) but rejects a flag-typed variable as its `value` - it evaluates
# value as a script-value expression and errors "Got value of type 'flag'".
# So every flag round-trips through an if/else_if chain against each known
# literal. Vanilla's `00_single_combat_effects.txt` writes them the same way.
#
# `fixed` always resolves to flag:no here - configure_start_*_combat_effect
# routes FIXED duels around the minigame entirely. Round-tripped only so the
# follow-up handler reads a consistent rule set whatever path resolved the duel.
#
# The fatality chain below has a WRITE-side twin in
# pod_combat_intro_effects.txt: the first-fight intro's deferred launch dispatches
# the same five literals into the builder for the same reason (a flag cannot ride a
# macro arg). Keep both in step - a FATALITY literal listed here and missing there
# silently turns a practice spar into a duel to the death.
#
# Known gap, degraded-route only: four live callers pass LOCALE = wilderness_scope,
# which the locale chain does not enumerate, so this restore hands them
# flag:battlefield. Cosmetic (background art) and pre-existing.
POD_combat_restore_sc_rule_flags_effect = {
	# Fatality
	if = {
		limit = { scope:sc_ctx.var:fatality = flag:always }
		save_scope_value_as = { name = fatality value = flag:always }
	}
	else_if = {
		limit = { scope:sc_ctx.var:fatality = flag:default }
		save_scope_value_as = { name = fatality value = flag:default }
	}
	else_if = {
		limit = { scope:sc_ctx.var:fatality = flag:possible }
		save_scope_value_as = { name = fatality value = flag:possible }
	}
	else_if = {
		limit = { scope:sc_ctx.var:fatality = flag:practice }
		save_scope_value_as = { name = fatality value = flag:practice }
	}
	else = {
		save_scope_value_as = { name = fatality value = flag:no }
	}

	# Fixed
	if = {
		limit = { scope:sc_ctx.var:fixed = flag:sc_attacker }
		save_scope_value_as = { name = fixed value = flag:sc_attacker }
	}
	else_if = {
		limit = { scope:sc_ctx.var:fixed = flag:sc_defender }
		save_scope_value_as = { name = fixed value = flag:sc_defender }
	}
	else = {
		save_scope_value_as = { name = fixed value = flag:no }
	}

	# Locale
	if = {
		limit = { scope:sc_ctx.var:locale = flag:battlefield }
		save_scope_value_as = { name = locale value = flag:battlefield }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:terrain_scope }
		save_scope_value_as = { name = locale value = flag:terrain_scope }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:throne_room }
		save_scope_value_as = { name = locale value = flag:throne_room }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:army_camp }
		save_scope_value_as = { name = locale value = flag:army_camp }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:tavern }
		save_scope_value_as = { name = locale value = flag:tavern }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:feast }
		save_scope_value_as = { name = locale value = flag:feast }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:sitting_room }
		save_scope_value_as = { name = locale value = flag:sitting_room }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:random_nice_day }
		save_scope_value_as = { name = locale value = flag:random_nice_day }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:corridor_night }
		save_scope_value_as = { name = locale value = flag:corridor_night }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:alley_night }
		save_scope_value_as = { name = locale value = flag:alley_night }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:alley_day }
		save_scope_value_as = { name = locale value = flag:alley_day }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:corridor_day }
		save_scope_value_as = { name = locale value = flag:corridor_day }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:temple }
		save_scope_value_as = { name = locale value = flag:temple }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:courtyard }
		save_scope_value_as = { name = locale value = flag:courtyard }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:dungeon }
		save_scope_value_as = { name = locale value = flag:dungeon }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:docks }
		save_scope_value_as = { name = locale value = flag:docks }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:market }
		save_scope_value_as = { name = locale value = flag:market }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:ep2_travel_bridge }
		save_scope_value_as = { name = locale value = flag:ep2_travel_bridge }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:apocalyptic }
		save_scope_value_as = { name = locale value = flag:apocalyptic }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:duel_locale }
		save_scope_value_as = { name = locale value = flag:duel_locale }
	}
	else = {
		# Keep this enum synced with PoD's single_combat_events.txt background
		# overrides - an unlisted locale falls through here and loses its
		# original visual context. battlefield is the safe universal default.
		# The first-fight intro's deferred launch passes this same literal for a
		# different reason (see COMBAT_REFERENCE section 10), so a fight deferred
		# behind the intro reads battlefield on both routes rather than disagreeing.
		save_scope_value_as = { name = locale value = flag:battlefield }
	}

	# terrain_scope backgrounds: PoD reads the defender's location. Unlike the
	# launcher (defender always landed at setup), this restore runs at .9999,
	# where a lethal hit may already have killed sc_defender, so its location
	# link can be unset and a bare deref errors. Guard it and fall back to the
	# combat host's location (root = the PC at .9999) so background_terrain_scope
	# stays populated: a follow-up that renders the terrain_scope background does
	# so unconditionally (not gated on scope:locale), so an unset scope would
	# error at render regardless. Both branches exist-guarded for a landless PC.
	if = {
		limit = { scope:locale = flag:terrain_scope }
		if = {
			limit = { exists = scope:sc_defender.location }
			scope:sc_defender.location = { save_scope_as = background_terrain_scope }
		}
		else_if = {
			limit = { exists = root.location }
			root.location = { save_scope_as = background_terrain_scope }
		}
	}
}


##############################################################################
# 4. Apply duel-end finalization side-effects
##############################################################################

# Four side-effect phases: release engaged flags, bump the victor's
# artifact_duel_var, apply the mode-specific forced kills, then settle every
# death the in-fight gate deferred. See each block below.
POD_combat_apply_duel_finalization_effect = {
	# ── Side principals, captured up front ──
	# Consumed by the deferred-death sweep at the bottom, which needs a killer per
	# victim. Taken HERE, before any forced kill below: both savers read the fight
	# through the host, and the host is exactly who a fatality = always loss turns
	# into a corpse. Captured early they are also still pre-kill, so the enemy
	# saver's first-living fallback picks a combatant who survived the fight.
	pod_combat_save_player_principal_effect = yes
	pod_combat_save_enemy_victor_effect = yes

	# ── Release engaged flags ──
	scope:sc_attacker = {
		if = {
			limit = { is_alive = yes }
			remove_variable = engaged_in_single_combat
		}
	}
	scope:sc_defender = {
		if = {
			limit = { is_alive = yes }
			remove_variable = engaged_in_single_combat
		}
	}

	# -- Victor weapon: artifact_duel_var increment --
	# Feeds the artifact.4080 "weapon has fought N duels" chain. Verbatim port from
	# PoD's finalise_combat_results_effect (single_combat_events.txt ~line 2622),
	# which the minigame replaces; without it intercepted victors lose duel-count
	# progression. random_character_artifact picks one weapon (mirrors vanilla).
	if = {
		limit = { exists = scope:sc_victor }
		scope:sc_victor = {
			if = {
				limit = {
					any_character_artifact = {
						is_equipped = yes
						artifact_slot_type = primary_armament
						NOT = { has_variable = artifact_duel_var }
					}
				}
				random_character_artifact = {
					limit = {
						is_equipped = yes
						artifact_slot_type = primary_armament
						NOT = { has_variable = artifact_duel_var }
					}
					set_variable = {
						name  = artifact_duel_var
						value = 1
					}
				}
			}
			else_if = {
				limit = {
					any_character_artifact = {
						is_equipped = yes
						artifact_slot_type = primary_armament
						has_variable = artifact_duel_var
					}
				}
				random_character_artifact = {
					limit = {
						is_equipped = yes
						artifact_slot_type = primary_armament
						has_variable = artifact_duel_var
					}
					change_variable = {
						name = artifact_duel_var
						add  = 1
					}
				}
			}
		}
	}

	# -- Apply death-only outcomes --
	# Wounds were written live but PoD's inline death check was gated for the whole
	# fight, so nobody has died of them yet - the sweep at the end of this effect
	# settles that. This block only handles the modes that kill the loser REGARDLESS
	# of wound math (always, default-tribal). No wound re-application here: that
	# would double-hit on top of the live damage.
	if = {
		limit = { exists = scope:sc_victor }

		if = {
			limit = { scope:fatality = flag:practice }
		}
		else = {
			# POD_fera_add_renown_effect rewards its execution scope. We enter
			# from the PC combat host, so the bare call would land ferocity on
			# the player even when an AI won - wrap into the actual victor.
			scope:sc_victor = {
				POD_fera_add_renown_effect = { TYPE = ferocity }
			}
		}

		if = {
			limit = { scope:fatality = flag:default }
			# Tribal governments: tribal duels are always to the death.
			if = {
				limit = { pod_combat_fatality_default_will_die_trigger = yes }
				# is_alive guard: the loser can already be a corpse (seated dead
				# by a launcher, or killed mid-fight by something outside the
				# wound pipeline), and re-calling pod_death on one spams the
				# error log and risks duplicate downstream effects (kinslayer,
				# doctrine triggers). Same pattern at the flag:always branch.
				if = {
					limit = { scope:sc_loser = { is_alive = yes } }
					scope:sc_loser = {
						pod_death = { killer = scope:sc_victor death_reason = death_duel }
					}
				}
				if = {
					limit = { scope:sc_victor = { faith = { has_doctrine = tenet_cranial_trophies } } }
					tgp_cranial_trophies_beheading_effect = { KILLER = scope:sc_victor DEAD = scope:sc_loser }
				}
			}
			# Non-tribal default: trust minigame damage (no forced kill).
		}
		else_if = {
			limit = { scope:fatality = flag:always }
			scope:sc_victor = {
				if = {
					limit = {
						this = scope:sc_attacker
						NOT = { has_execute_reason = scope:sc_loser }
					}
					add_kinslayer_trait_or_nothing_effect = { VICTIM = scope:sc_loser }
				}
			}
			# is_alive guard - see tribal-default branch above.
			if = {
				limit = { scope:sc_loser = { is_alive = yes } }
				scope:sc_loser = {
					pod_death = { killer = scope:sc_victor death_reason = death_duel }
				}
			}
			if = {
				limit = { scope:sc_victor = { faith = { has_doctrine = tenet_cranial_trophies } } }
				tgp_cranial_trophies_beheading_effect = { KILLER = scope:sc_victor DEAD = scope:sc_loser }
			}
		}
		# flag:no / flag:practice - no forced kill, and the sweep below skips them.
		# flag:possible and non-tribal flag:default - no forced kill either; whether
		# the loser dies is decided purely by the wounds they took, which is what
		# the sweep below asks.
	}

	# -- Settle the deaths the in-fight gate deferred --
	# Last, so the forced kills above keep their own death_reason and their
	# kinslayer / cranial-trophies side effects, and this only catches whoever is
	# still standing on a lethal track.
	pod_combat_resolve_suppressed_deaths_effect = yes
}


# Live wounds land every hit, but pod_combat_nonfatal_active suppresses PoD's
# inline death check for the whole fight (see pod_combat_uses_death_gate_trigger)
# - otherwise a guaranteed-resurrection opponent full-heals on every killing blow
# and the duel can never end. A combatant whose track filled was therefore KO'd,
# not killed. This runs PoD's own check once per combatant now that
# pod_combat_finalize_combat_health_effect has stripped the gate, which is the
# model base PoD uses (work_out_wounds_or_death_effect, single_combat_events.txt).
#
# Host scope, called from .9999 only - the host dying in here is already the
# expected case there (see the dead-host arm at the end of pod_combat.9999).
pod_combat_resolve_suppressed_deaths_effect = {
	if = {
		limit = { NOT = { pod_combat_duel_is_bloodless_trigger = yes } }
		# Read the roster off scope:sc_ctx, not var:pod_combat_session: .9999 binds
		# it once and a story cannot become a corpse part-way through this loop.
		scope:sc_ctx ?= {
			every_in_list = {
				variable = pod_combat_active_slots
				limit = {
					exists = var:slot_char
					pod_combat_slot_char_is_alive_trigger = yes
					pod_combat_slot_char_is_summon_trigger = no
				}
				# Killer = the OPPOSING side's principal, so murder attribution
				# lands on someone who actually fought this victim rather than on
				# their own leader. A victim whose opposite number is unset or
				# already dead is left standing instead: pod_death would hand that
				# corpse to POD_commit_murder_effect, and PoD settles a full track
				# at the next POD_refresh_health_track anyway.
				# trigger_if, not sibling clauses - Jomini triggers do not
				# short-circuit, so an unset scope: read would still evaluate.
				if = {
					limit = {
						pod_combat_slot_is_enemy_trigger = yes
						trigger_if = {
							limit = { exists = scope:pod_combat_player_principal }
							scope:pod_combat_player_principal = { is_alive = yes }
						}
						trigger_else = { always = no }
					}
					scope:pod_combat_player_principal = { save_scope_as = wound_attacker }
					var:slot_char = { pod_combat_run_deferred_death_check_effect = yes }
				}
				else_if = {
					limit = {
						pod_combat_slot_is_player_team_trigger = yes
						trigger_if = {
							limit = { exists = scope:pod_combat_enemy_victor }
							scope:pod_combat_enemy_victor = { is_alive = yes }
						}
						trigger_else = { always = no }
					}
					scope:pod_combat_enemy_victor = { save_scope_as = wound_attacker }
					var:slot_char = { pod_combat_run_deferred_death_check_effect = yes }
				}
			}
		}
	}
}

# Character scope. PoD's own wound-death gate, run once against the final track:
# it re-tests track-full AND aggravated >= POD_aggravated_damage_threshold_for_death
# AND POD_can_die_from_wounds, so a combatant beaten down on superficial damage
# alone survives, and an immortal without guaranteed resurrection still takes the
# POD_long_depose_effect off-ramp instead of dying. No-ops on a survivable track.
# Reads scope:wound_attacker for the killer, which the caller sets per victim.
pod_combat_run_deferred_death_check_effect = {
	POD_health_death_check_effect = {
		TOTAL_SUPERFICIAL_DAMAGE = POD_superficial_damage_boxes
		TOTAL_AGGRAVATED_DAMAGE  = POD_aggravated_damage_boxes
		REASON                   = duel_aggravated
	}
}


##############################################################################
# 5. Fire the follow-up event with the initiator -> defender -> attacker fallback chain
##############################################################################

# Matches vanilla's initiator -> defender -> attacker fallback chain (see
# events/single_combat_events.txt) so the follow-up's `root` is the character
# vanilla would have used. The chain matters because trigger_event silently
# no-ops on a dead host and combatants here may have died - and because the
# intercept gate only requires PC in {sc_attacker, sc_defender}, NOT
# PC = sc_initiator: AI-initiated attacks on the PC (alastor challenges,
# spirit attacks) fire with sc_initiator = AI while the PC is the defender.
POD_combat_fire_sc_followup_effect = {
	if = {
		limit = { scope:sc_initiator = { is_alive = yes } }
		scope:sc_initiator = {
			trigger_event = { saved_event_id = scope:sc_ctx.var:follow_up }
		}
	}
	else_if = {
		limit = { scope:sc_defender = { is_alive = yes } }
		scope:sc_defender = {
			trigger_event = { saved_event_id = scope:sc_ctx.var:follow_up }
		}
	}
	else = {
		scope:sc_attacker = {
			trigger_event = { saved_event_id = scope:sc_ctx.var:follow_up }
		}
	}
}
