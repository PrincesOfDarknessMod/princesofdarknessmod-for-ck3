######################################################
######   POD COMBAT — SC-CONTEXT RESTORE HELPERS
######################################################
# Helpers consumed by pod_combat.9999 (the duel-finalize follow-up event)
# to restore the launcher's POD_combat_sc_context story_cycle into the
# scope shape the original PoD duel pipeline expected. Each helper owns
# one finalization phase; pod_combat.9999 is a thin orchestrator calling
# them in order.
#
# All helpers run in the same scope: the immediate block of
# pod_combat.9999, fired on the combat host (PC). They expect the PC
# to carry a `var:pod_combat_sc_context` pointer to the launcher's
# context story.
#
# Restore order (called by pod_combat.9999):
#   1. POD_combat_restore_sc_context_effect       — sc_initiator / attacker /
#                                                   defender + aliases + sc_victor / loser
#   2. pod_combat_sc_restore_parent_aliases_effect — 24 custom parent-event scopes
#   3. POD_combat_restore_sc_rule_flags_effect    — fatality / fixed / locale +
#                                                   sc_finished / victory_type
#   4. POD_combat_apply_duel_finalization_effect  — engaged flags, artifact_duel_var,
#                                                   death-only outcomes
#   5. POD_combat_fire_sc_followup_effect         — follow-up event with the
#                                                   initiator → defender → attacker
#                                                   fallback chain
#   6. pod_combat_cleanup_effect                  — end the context story, clear
#                                                   the pointer var (existing helper)

##############################################################################
# 1. Restore sc_initiator / sc_attacker / sc_defender + their aliases
##############################################################################

# Reads through `var:pod_combat_sc_context` on the PC and reconstructs the
# canonical character scopes vanilla / PoD callers expect (sc_initiator,
# sc_attacker, sc_defender), plus the alias scopes the duel system writes
# during launcher setup (victim / recipient / defender / target / adversary
# for the loser side; attacker / challenger for the attacker side).
#
# Also reads `pod_combat_winner` / `pod_combat_loser` from the PC into
# sc_victor / sc_loser when present (absent for draws / mutual KOs).
POD_combat_restore_sc_context_effect = {
	var:pod_combat_sc_context = { save_scope_as = sc_ctx }

	# Canonical character scopes
	scope:sc_ctx.var:initiator = { save_scope_as = sc_initiator }
	scope:sc_ctx.var:attacker  = { save_scope_as = sc_attacker }
	scope:sc_ctx.var:defender  = { save_scope_as = sc_defender }

	# Defender aliases (covers victim, recipient, defender, target, adversary)
	scope:sc_ctx.var:defender = { save_scope_as = victim }
	scope:sc_ctx.var:defender = { save_scope_as = recipient }
	scope:sc_ctx.var:defender = { save_scope_as = defender }
	scope:sc_ctx.var:defender = { save_scope_as = target }
	scope:sc_ctx.var:defender = { save_scope_as = adversary }

	# Attacker aliases (covers attacker, challenger)
	scope:sc_ctx.var:attacker = { save_scope_as = attacker }
	scope:sc_ctx.var:attacker = { save_scope_as = challenger }

	# Victor / loser from minigame result (absent for draws / mutual KOs)
	if = {
		limit = {
			has_variable = pod_combat_winner
			has_variable = pod_combat_loser
		}
		var:pod_combat_winner = { save_scope_as = sc_victor }
		var:pod_combat_loser  = { save_scope_as = sc_loser }
	}
}

##############################################################################
# 3. Reconstruct rule-flag scope values
##############################################################################

# Round-trips the duel rule flags (fatality, fixed, locale) from
# variable-backed flags on sc_ctx into saved-scope flag values that the
# follow-up event handler reads.
#
# Why the enum chains: save_scope_value_as accepts literal flag values
# (`flag:X`) but does NOT accept a flag-typed variable as its `value`
# it evaluates value as a script-value expression and rejects the flag
# with "Got value of type 'flag'". So every flag has to round-trip
# through an enumerated if/else_if chain comparing the stored variable
# against each known literal. Vanilla's `00_single_combat_effects.txt`
# writes the flags this way too; we just have to read them back the
# same way we'd write them.
#
# Locale enum must stay synchronised with PoD's
# single_combat_events.txt background overrides; missing locales fall
# through to battlefield and lose the original visual context.
#
# NOTE on `fixed`: round-tripped for follow-up event compatibility (any
# handler reading scope:fixed sees what the parent event passed).
# Reaching this restore path always means scope:fixed = flag:no. The
# configure_start_*_combat_effect overrides route FIXED = sc_attacker /
# sc_defender duels around the minigame entirely. Restoring the value
# here keeps the follow-up event reading a consistent rule set
# regardless of which path resolved the duel.
POD_combat_restore_sc_rule_flags_effect = {
	# Fatality
	if = {
		limit = { scope:sc_ctx.var:fatality = flag:always }
		save_scope_value_as = { name = fatality value = flag:always }
	}
	else_if = {
		limit = { scope:sc_ctx.var:fatality = flag:default }
		save_scope_value_as = { name = fatality value = flag:default }
	}
	else_if = {
		limit = { scope:sc_ctx.var:fatality = flag:possible }
		save_scope_value_as = { name = fatality value = flag:possible }
	}
	else_if = {
		limit = { scope:sc_ctx.var:fatality = flag:practice }
		save_scope_value_as = { name = fatality value = flag:practice }
	}
	else = {
		save_scope_value_as = { name = fatality value = flag:no }
	}

	# Fixed
	if = {
		limit = { scope:sc_ctx.var:fixed = flag:sc_attacker }
		save_scope_value_as = { name = fixed value = flag:sc_attacker }
	}
	else_if = {
		limit = { scope:sc_ctx.var:fixed = flag:sc_defender }
		save_scope_value_as = { name = fixed value = flag:sc_defender }
	}
	else = {
		save_scope_value_as = { name = fixed value = flag:no }
	}

	# Locale
	if = {
		limit = { scope:sc_ctx.var:locale = flag:battlefield }
		save_scope_value_as = { name = locale value = flag:battlefield }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:terrain_scope }
		save_scope_value_as = { name = locale value = flag:terrain_scope }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:throne_room }
		save_scope_value_as = { name = locale value = flag:throne_room }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:army_camp }
		save_scope_value_as = { name = locale value = flag:army_camp }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:tavern }
		save_scope_value_as = { name = locale value = flag:tavern }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:feast }
		save_scope_value_as = { name = locale value = flag:feast }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:sitting_room }
		save_scope_value_as = { name = locale value = flag:sitting_room }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:random_nice_day }
		save_scope_value_as = { name = locale value = flag:random_nice_day }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:corridor_night }
		save_scope_value_as = { name = locale value = flag:corridor_night }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:alley_night }
		save_scope_value_as = { name = locale value = flag:alley_night }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:alley_day }
		save_scope_value_as = { name = locale value = flag:alley_day }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:corridor_day }
		save_scope_value_as = { name = locale value = flag:corridor_day }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:temple }
		save_scope_value_as = { name = locale value = flag:temple }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:courtyard }
		save_scope_value_as = { name = locale value = flag:courtyard }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:dungeon }
		save_scope_value_as = { name = locale value = flag:dungeon }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:docks }
		save_scope_value_as = { name = locale value = flag:docks }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:market }
		save_scope_value_as = { name = locale value = flag:market }
	}
	else_if = {
		limit = { scope:sc_ctx.var:locale = flag:ep2_travel_bridge }
		save_scope_value_as = { name = locale value = flag:ep2_travel_bridge }

		limit = { scope:sc_ctx.var:locale = flag:duel_locale }
		save_scope_value_as = { name = locale value = flag:duel_locale }
	}
	else = {
		# Unrecognised / unset locale — fall back to battlefield, the
		# safe default that every duel template handles. Keep this list
		# synchronised with PoD's single_combat_events.txt background
		# overrides; missing locales fall through to battlefield and
		# lose the original visual context.
		save_scope_value_as = { name = locale value = flag:battlefield }
	}

	# If the locale uses terrain_scope, follow PoD's pattern of saving
	# the defender's location as the background.
	if = {
		limit = { scope:locale = flag:terrain_scope }
		scope:sc_defender.location = { save_scope_as = background_terrain_scope }
	}

	# Constants the follow-up handler always expects to see.
	save_scope_value_as = { name = sc_finished  value = yes }
	save_scope_value_as = { name = victory_type value = flag:skill }
}


##############################################################################
# 4. Apply duel-end finalization side-effects
##############################################################################

# Three side-effect phases:
#   - Release the engaged-in-single-combat flag from both combatants
#     (alive guard prevents log spam on corpses).
#   - Bump the victor's primary armament artifact_duel_var (verbatim
#     port from PoD's finalise_combat_results_effect — without porting
#     this, intercepted-duel victors silently lose artifact duel-count
#     progression even though the rest of the result pipeline works).
#   - Apply death-only outcomes based on fatality mode.
POD_combat_apply_duel_finalization_effect = {
	# ── Release engaged flags ──
	scope:sc_attacker = {
		if = {
			limit = { is_alive = yes }
			remove_variable = engaged_in_single_combat
		}
	}
	scope:sc_defender = {
		if = {
			limit = { is_alive = yes }
			remove_variable = engaged_in_single_combat
		}
	}

	# ── Victor weapon: artifact_duel_var increment ──
	# Tracks how many duels the victor's primary armament has won,
	# feeding the artifact.4080 narrative chain ("your weapon has fought
	# N duels" lore events). First win on a given artifact sets the var
	# to 1; subsequent wins increment.
	#
	# Verbatim port from PoD's finalise_combat_results_effect
	# (POD Reference/events/single_combat_events.txt ~line 2622), which
	# the minigame intercepts and replaces — without porting this block,
	# intercepted-duel victors silently lose artifact duel-count
	# progression even when the rest of the result pipeline (fera
	# ferocity, fatality death, follow-up) fires correctly.
	#
	# Two-branch structure handles "first duel for this artifact" (sets
	# the var to 1) vs "subsequent duel" (change_variable add = 1).
	# random_character_artifact picks a single weapon when the victor
	# has multiple equipped primary armaments — mirrors vanilla rather
	# than incrementing every weapon.
	#
	# Defensive `exists = scope:sc_victor` guard matches the death-
	# outcomes block below: a draw / mutual KO leaves sc_victor unset,
	# in which case no artifact gets the progression bump.
	if = {
		limit = { exists = scope:sc_victor }
		scope:sc_victor = {
			if = {
				limit = {
					any_character_artifact = {
						is_equipped = yes
						artifact_slot_type = primary_armament
						NOT = { has_variable = artifact_duel_var }
					}
				}
				random_character_artifact = {
					limit = {
						is_equipped = yes
						artifact_slot_type = primary_armament
						NOT = { has_variable = artifact_duel_var }
					}
					set_variable = {
						name  = artifact_duel_var
						value = 1
					}
				}
			}
			else_if = {
				limit = {
					any_character_artifact = {
						is_equipped = yes
						artifact_slot_type = primary_armament
						has_variable = artifact_duel_var
					}
				}
				random_character_artifact = {
					limit = {
						is_equipped = yes
						artifact_slot_type = primary_armament
						has_variable = artifact_duel_var
					}
					change_variable = {
						name = artifact_duel_var
						add  = 1
					}
				}
			}
		}
	}

	# ── Apply death-only outcomes ──
	# NOTE: Wound damage was written to PoD health tracks live during
	# move execution (the damage helpers in pod_combat_effects.txt call
	# POD_increase_wounds_* immediately, not deferred). PoD's wound
	# pipeline invokes POD_health_death_check_effect inline whenever
	# wounds land — see POD_apply_wound_boxes in PoD's
	# POD_health_track_effects.txt and the override note at the top of
	# pod_combat_death_gate_overrides.txt — so a lethal hit kills the
	# loser the moment it resolves, subject to the
	# pod_combat_nonfatal_active gate for non-fatal duels. Re-applying
	# POD_increase_wounds_hostile_effect here would double-damage the
	# loser and risk a duplicate death trigger.
	#
	# For fatality modes that guarantee a kill (always, default-tribal),
	# we still call pod_death directly so the outcome is deterministic
	# even when the wound math fell short of the death threshold.
	if = {
		limit = { exists = scope:sc_victor }

		if = {
			limit = { scope:fatality = flag:practice }
		}
		else = {
			# POD_fera_add_renown_effect grants renown to its execution
			# scope (see PoD's POD_fera_renown_effects.txt). PoD's own
			# duel finalizer fires this inside the attacker context;
			# pod_combat.9999 enters from the PC combat host instead,
			# so without the scope:sc_victor wrap the bare call lands
			# the ferocity reward on the player even when an AI won.
			# Wrap into the actual victor — fera ferocity tracks the
			# combatant who closed the kill, regardless of which side
			# initiated.
			scope:sc_victor = {
				POD_fera_add_renown_effect = { TYPE = ferocity }
			}
		}

		if = {
			limit = { scope:fatality = flag:default }
			# Tribal governments: tribal duels are always to the death.
			if = {
				limit = {
					scope:sc_attacker = {
						OR = {
							AND = {
								is_ruler = yes
								government_has_flag = government_is_tribal
							}
							AND = {
								is_ruler = no
								exists = court_owner
								scope:sc_attacker.court_owner = { government_has_flag = government_is_tribal }
							}
							AND = {
								is_ruler = no
								exists = host
								scope:sc_attacker.host = { government_has_flag = government_is_tribal }
							}
							AND = {
								is_ruler = no
								NOR = { exists = court_owner exists = host }
								scope:sc_defender = {
									OR = {
										AND = { is_ruler = yes government_has_flag = government_is_tribal }
										AND = { is_ruler = no exists = court_owner scope:sc_defender.court_owner = { government_has_flag = government_is_tribal } }
										AND = { is_ruler = no exists = host scope:sc_defender.host = { government_has_flag = government_is_tribal } }
									}
								}
							}
						}
					}
				}
				# is_alive guard: under fatality = no this branch can't
				# fire (the path is gated on `else_if` against
				# flag:default), but live minigame damage can still kill
				# the loser during fatality = default tribal fights —
				# re-calling pod_death on a corpse spams the error log
				# and risks duplicate downstream effects (kinslayer,
				# doctrine triggers). Same pattern at the flag:always
				# branch below.
				if = {
					limit = { scope:sc_loser = { is_alive = yes } }
					scope:sc_loser = {
						pod_death = { killer = scope:sc_victor death_reason = death_duel }
					}
				}
				if = {
					limit = { scope:sc_victor = { faith = { has_doctrine = tenet_cranial_trophies } } }
					tgp_cranial_trophies_beheading_effect = { KILLER = scope:sc_victor DEAD = scope:sc_loser }
				}
			}
			# Non-tribal default: trust minigame damage (no forced kill).
		}
		else_if = {
			limit = { scope:fatality = flag:always }
			scope:sc_victor = {
				if = {
					limit = {
						this = scope:sc_attacker
						NOT = { has_execute_reason = scope:sc_loser }
					}
					add_kinslayer_trait_or_nothing_effect = { VICTIM = scope:sc_loser }
				}
			}
			# is_alive guard — see comment on the tribal-default branch
			# above. fatality = always can still race with in-combat
			# lethal damage (especially with the nonfatal round-end strip
			# now fixed); the loser may already be down by the time
			# pod_combat.9999 fires.
			if = {
				limit = { scope:sc_loser = { is_alive = yes } }
				scope:sc_loser = {
					pod_death = { killer = scope:sc_victor death_reason = death_duel }
				}
			}
			if = {
				limit = { scope:sc_victor = { faith = { has_doctrine = tenet_cranial_trophies } } }
				tgp_cranial_trophies_beheading_effect = { KILLER = scope:sc_victor DEAD = scope:sc_loser }
			}
		}
		# flag:no       — writeback already capped loser's track at 1 empty box; no further action.
		# flag:possible — trust minigame damage; periodic death check handles full-track kills.
	}
}


##############################################################################
# 5. Fire the follow-up event with the initiator → defender → attacker fallback chain
##############################################################################

# Matches vanilla's three-step fallback chain (see
# events/single_combat_events.txt for the canonical pattern) so the
# follow-up's `root` is the same character vanilla would have used.
#
# The intercept gate only requires the PC to be one of {sc_attacker,
# sc_defender}. It does NOT require PC = sc_initiator. AI-initiated
# interactions targeting the PC (Sabbat alastor challenges, spirit
# attacks, etc.) fire with sc_initiator = AI even though PC is the
# defender, so the fallback chain matters.
#
# Fallback chain: initiator → defender → attacker. The chain exists
# because trigger_event silently no-ops if the host character is dead,
# and combatants in this duel may have died. Vanilla's comment is
# explicit: "Doesn't hugely matter technically, since the output event
# should be from a neutral scope, but if they're dead then the event
# won't fire and the chain silently invalidates."
POD_combat_fire_sc_followup_effect = {
	if = {
		limit = { scope:sc_initiator = { is_alive = yes } }
		scope:sc_initiator = {
			trigger_event = { saved_event_id = scope:sc_ctx.var:follow_up }
		}
	}
	else_if = {
		limit = { scope:sc_defender = { is_alive = yes } }
		scope:sc_defender = {
			trigger_event = { saved_event_id = scope:sc_ctx.var:follow_up }
		}
	}
	else = {
		scope:sc_attacker = {
			trigger_event = { saved_event_id = scope:sc_ctx.var:follow_up }
		}
	}
}
