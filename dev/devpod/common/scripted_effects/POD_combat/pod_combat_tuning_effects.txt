# Runtime balancing knobs for the debug tuning window
# (pod_combat_tuning_window.gui + pod_combat_tuning_guis.txt).
# Every knob is a global_var:pod_combat_tune_* override read by the
# matching script value in pod_combat_values.txt; unset = the shipped
# default. Globals persist in the save - promote settled numbers into
# the defaults and Reset All before release.

# Shared stepper body. $VAR$ = pod_combat_tune_* suffix; $DELTA$ = the
# signed step passed by the GUI arrows as scope:delta; $DEFAULT$ = the
# seed on first touch (the knob's current effective value, so stepping
# starts from what the game is using); $MIN$ / $MAX$ = clamp bounds.
pod_combat_tune_edit_effect = {
	if = {
		limit = { NOT = { exists = global_var:pod_combat_tune_$VAR$ } }
		set_global_variable = { name = pod_combat_tune_$VAR$ value = $DEFAULT$ }
	}
	change_global_variable = { name = pod_combat_tune_$VAR$ add = $DELTA$ }
	if = {
		limit = { global_var:pod_combat_tune_$VAR$ < $MIN$ }
		set_global_variable = { name = pod_combat_tune_$VAR$ value = $MIN$ }
	}
	if = {
		limit = { global_var:pod_combat_tune_$VAR$ > $MAX$ }
		set_global_variable = { name = pod_combat_tune_$VAR$ value = $MAX$ }
	}
}

# Back to shipped defaults: remove every tuning global (remove on an
