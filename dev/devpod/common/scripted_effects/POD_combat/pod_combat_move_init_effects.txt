######################################################
######   POD COMBAT MOVE INIT EFFECTS
######################################################
# Modular setter effects + the POD_init_combat_move constructor used
# by POD_initialize_combat_moves in pod_combat_move_list_effects.txt.
# Together these are the "authoring layer" for defining a combat move
# as a story_cycle with a handful of variables.
#
# Data flow:
#   1. on_game_start fires POD_initialize_combat_moves.
#   2. That effect calls POD_init_combat_move per move, creating a
#      POD_combat_move story on Charon, adding it to the global list
#      POD_combat_moves, and setting mandatory variables.
#   3. Inside scope:new_combat_move, optional setters (POD_combat_move_set_*)
#      are called for attributes that apply to that specific move.
#   4. At execution, POD_execute_combat_move reads the story's variables
#      and applies consequences. POD_can_execute_combat_move validates.
#
# All setters run inside the story scope. Presence of a variable on the
# story means the attribute applies; absence means skip.


###################
###### CONSTRUCTOR
###################

# Creates one combat-move story_cycle on Charon, sets mandatory vars,
# adds it to the global list, and saves it on Charon by its decision
# name for the data_binding macro to find.
#
# Parameters:
#   DECISION   — the hidden decision holding this move's unlock reqs
#   SKILL_KEY  — unique identifier flag (e.g. strike)
#   CATEGORY   — ATK | DEF | CTL | PWR
#   CONFIG     — config_section slug (e.g. base_physical)
#   AP_COST    — integer AP cost
#   AI_WEIGHT  — base AI selection weight
#   TARGETING  — single_enemy | single_ally | single_pc | self | team_enemy | team_ally
POD_init_combat_move = {
	# Use Charon as the off-map anchor. He's immortal, hidden, and alive
	# in every bookmark — matches the DUEL-Refactor pattern.
	character:POD_wraith_01 = {
		create_story = {
			type = POD_combat_move
			save_scope_as = new_combat_move
		}

		scope:new_combat_move = {
			set_variable = { name = decision        value = decision:$DECISION$ }
			set_variable = { name = skill_key       value = flag:$SKILL_KEY$ }
			set_variable = { name = combat_category value = flag:$CATEGORY$ }
			set_variable = { name = config_section  value = flag:$CONFIG$ }
			set_variable = { name = ap_cost         value = $AP_COST$ }
			set_variable = { name = ai_weight       value = $AI_WEIGHT$ }
			set_variable = { name = targeting       value = flag:$TARGETING$ }
		}

		# Add to the global catalogue (used by GUI + AI selection)
		add_to_global_variable_list = {
			name   = POD_combat_moves
			target = scope:new_combat_move
		}

		# Also save by decision name so PODCombatMoveDesc macro can find it
		set_variable = {
			name  = $DECISION$
			value = scope:new_combat_move
		}

		# Error-log suppression. The following flags and variables are set
		# by setters below but may not appear in every move. This gated
		# block silences "unused flag" / "unset variable" warnings for the
		# common ones. Uses POD_error_suppression_unused_flag.
		#
		# The has_variable = $DECISION$ check on Charon (inside the limit
		# below) suppresses the "POD_combat_move_X is set but never used"
		# warnings the analyzer raises for the per-move pointer variable
		# stashed at line ~65. The actual read happens through the
		# PODCombatMoveDesc macro chain (Var(combatmove).Story.MakeScope),
		# which the analyzer can't trace — without this never-run read it
		# would log one warning per registered move every game start.
		if = {
			limit = {
				always = no
				has_variable = $DECISION$
				scope:new_combat_move = {
					POD_can_execute_combat_move = { STORY = scope:new_combat_move ACTOR = root }
					has_variable = skill_key
					has_variable = combat_category
					has_variable = config_section
					has_variable = targeting
				}
			}
			POD_error_suppression_unused_flag = { FLAG = $SKILL_KEY$ }
			POD_error_suppression_unused_flag = { FLAG = $CATEGORY$ }
			POD_error_suppression_unused_flag = { FLAG = $CONFIG$ }
			POD_error_suppression_unused_flag = { FLAG = $TARGETING$ }
			POD_error_suppression_unused_flag = { FLAG = single_enemy }
			POD_error_suppression_unused_flag = { FLAG = single_ally }
			POD_error_suppression_unused_flag = { FLAG = single_pc }
			POD_error_suppression_unused_flag = { FLAG = self }
			POD_error_suppression_unused_flag = { FLAG = team_enemy }
			POD_error_suppression_unused_flag = { FLAG = team_ally }
			POD_error_suppression_unused_flag = { FLAG = superficial }
			POD_error_suppression_unused_flag = { FLAG = aggravated }
			POD_error_suppression_unused_flag = { FLAG = fire }
			POD_error_suppression_unused_flag = { FLAG = silver }
			POD_error_suppression_unused_flag = { FLAG = gold }
			POD_error_suppression_unused_flag = { FLAG = holy }
			POD_error_suppression_unused_flag = { FLAG = sunlight }
			POD_error_suppression_unused_flag = { FLAG = ATK }
			POD_error_suppression_unused_flag = { FLAG = DEF }
			POD_error_suppression_unused_flag = { FLAG = CTL }
			POD_error_suppression_unused_flag = { FLAG = PWR }
			POD_error_suppression_unused_flag = { FLAG = wounded }
			POD_error_suppression_unused_flag = { FLAG = healthy }
			POD_error_suppression_unused_flag = { FLAG = round_robin }
			POD_error_suppression_unused_flag = { FLAG = threat }
			# Rider-condition flag values — read by
			# pod_combat_rider_condition_passes_trigger via flag-equality
			# checks. The analyzer can't follow flag literals into
			# trigger_else_if branches, so list them here to suppress
			# the "set but never used" warning.
			POD_error_suppression_unused_flag = { FLAG = always }
			POD_error_suppression_unused_flag = { FLAG = on_hp_damage }
			POD_error_suppression_unused_flag = { FLAG = on_contact }
			POD_error_suppression_unused_flag = { FLAG = on_miss }
		}
	}
}

###################
###### DAMAGE & HEALING SETTERS
###################

# $VALUE$  = hp damage amount
# Damage-source flavor → maps to a PoD death_reason suffix via the
# combat dispatcher. Allowed values:
#   fire | cold | lightning | sun | silver | gold | holy | balefire
# Absent (variable never set) = generic physical/supernatural damage
# (REASON = duel, subject to soak, not aggravated unless
# damage_type = aggravated overrides).
POD_combat_move_set_damage_source = {
	set_variable = { name = damage_source value = flag:$VALUE$ }
}

# $VALUE$ = tier / raw damage amount. hp_damage defaults to
# superficial; override with POD_combat_move_set_damage_type_aggravated
# if the move deals forcibly-aggravated damage regardless of source.
POD_combat_move_set_hp_damage = {
	set_variable = { name = hp_damage   value = $VALUE$ }
	set_variable = { name = damage_type value = flag:superficial }
}

POD_combat_move_set_damage_type_aggravated = {
	set_variable = { name = damage_type value = flag:aggravated }
}

POD_combat_move_set_morale_damage = {
	set_variable = { name = morale_damage value = $VALUE$ }
}

POD_combat_move_set_self_damage = {
	set_variable = { name = self_damage value = $VALUE$ }
}

# $VALUE$ = hp restored
# $TYPE$  = superficial | aggravated (controls which heal effect fires —
#          aggravated heal reduces the agg_damage counter too)
POD_combat_move_set_hp_heal = {
	set_variable = { name = hp_heal      value = $VALUE$ }
	set_variable = { name = heal_type    value = flag:$TYPE$ }

POD_combat_move_set_morale_heal = {
	set_variable = { name = morale_heal value = $VALUE$ }
}


###################
###### RIDER CONDITION SETTERS
###################
# Gate flags that control when rider effects (heal / morale damage /
# self-damage) fire based on the §6 HP-damage outcome. The runtime
# trigger pod_combat_rider_condition_passes_trigger reads these via
# trigger_else_if branches and consults pod_combat_last_result on
# the host. See that trigger's header for the full result-code → flag
# mapping (always / on_hp_damage / on_contact / on_miss).
#
# Setter is only called when the move overrides the runtime default
# (always); no setter → runtime treats the absent variable
# as "always" via the trigger's first branch.
#
# $VALUE$ must be one of: always | on_hp_damage | on_contact | on_miss.
# The generator validates against RIDER_CONDITIONS so unknown values
# never reach runtime.
POD_combat_move_set_heal_condition = {
	set_variable = { name = heal_condition value = flag:$VALUE$ }
}

POD_combat_move_set_morale_damage_condition = {
	set_variable = { name = morale_damage_condition value = flag:$VALUE$ }
}

POD_combat_move_set_self_damage_condition = {
	set_variable = { name = self_damage_condition value = flag:$VALUE$ }
}

###################
###### DEFAULT LOADOUT SEEDER
###################
# Equip every move with is_starter_move set. Called from the configure
# decision and from combat init when a character has no loadout yet.
# Works for any character (PC or AI); up to 8 moves carry is_starter_move.
POD_apply_default_combat_loadout = {
	# PC-side default: seed the moves tagged with is_starter_move.
	# Intended as a starting point the player then customises via
	# the loadout editor.
	#
	# Inside every_in_global_list the scope becomes the current entry
	# (the story), so add_to_variable_list would target the story's
	# own list instead of the loadout owner's. Save the owner scope
	# before iterating and switch back explicitly to write the list.
	save_scope_as = pod_combat_loadout_owner
	clear_variable_list = pod_combat_loadout
	every_in_global_list = {
		variable = POD_combat_moves
		limit = { has_variable = is_starter_move }
		scope:pod_combat_loadout_owner = {
			add_to_variable_list = {
				name   = pod_combat_loadout
				target = prev
			}
		}
	}
}


POD_apply_ai_full_loadout = {
	# AI-side seed: dump the entire move catalogue into the actor's
	# pod_combat_loadout. Filtering happens at PICK time, not at seed
	# time — the selection trigger POD_can_execute_combat_move takes
	# the actor as a scope parameter and gates each move via
	# can_execute_decision against the actor's traits/perks/faiths,
	# so unqualified moves simply never win the weighted roll. 
	#
	# Called lazily from pod_combat_ai_ensure_loadout_effect when the
	# AI walks into combat with an empty loadout. Works for any
	# character.
	save_scope_as = pod_combat_loadout_owner
	clear_variable_list = pod_combat_loadout
	every_in_global_list = {
		variable = POD_combat_moves
		scope:pod_combat_loadout_owner = {
			add_to_variable_list = {
				name   = pod_combat_loadout
				target = prev
			}
		}
	}
}


###################
###### VERSION-MISMATCH RELOAD
###################
# Called from the pod_combat_configure_skills decision when the player
# loads a save whose POD_combat_version variable doesn't match the
# script-value POD_current_combat_version. Clears every character's
# loadout and the global catalogue, then rebuilds from scratch. Saved
# starter-flagged moves are re-equipped by the subsequent default-loadout
# pass in the configure effect.
#
# AI loadout cleanup: every_living_character iterates ALL characters,
# not just human players, so any NPC that fought before the version
# bump gets their stale pod_combat_loadout cleared too. Without this,
# AI characters retained refs to the old (now end_story-d) move
# catalogue and the next time they fought their selector would try
# to pick from dangling story references. Slot-init also clears AI
# loadouts when they enter combat, so this reload-time clear is
# belt-and-suspenders cleanup that reduces save bloat between fights.
#
# Cost: every_living_character walks the whole character pool. Runs
# only on version mismatch (rare — only when the player reopens the
# configure decision after the move dataset has changed).
POD_reload_combat_moves = {
	every_living_character = {
		clear_variable_list = pod_combat_loadout
	}
	every_in_global_list = {
		variable = POD_combat_moves
		end_story = yes
	}
	clear_global_variable_list = POD_combat_moves
	POD_initialize_combat_moves = yes
	# Re-seed PC starter loadouts. NPCs are seeded lazily at fight
	# start (slot init clears + first AI turn re-seeds), so they
	# don't need a re-seed pass here.
	every_player = {
		POD_apply_default_combat_loadout = yes
	}
	set_global_variable = {
		name  = POD_combat_version
		value = POD_current_combat_version
	}
}
