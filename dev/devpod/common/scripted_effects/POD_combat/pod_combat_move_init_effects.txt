######################################################
######   POD COMBAT MOVE INIT EFFECTS
######################################################
# Authoring layer for combat moves: the POD_init_combat_move
# constructor + modular POD_combat_move_set_* setters, driven by
# POD_initialize_combat_moves (pod_combat_move_list_effects.txt).
# A move is a POD_combat_move story_cycle whose variables the
# execution engine (pod_combat_move_effects.txt) reads at runtime.
#
# All setters run inside the story scope; presence of a variable =
# attribute applies, absence = skip. Some setters are unreferenced by
# any current move row but retain full runtime support.

###################
###### CONSTRUCTOR
###################

# Parameters:
#   DECISION   — the hidden decision holding this move's unlock reqs
#   SKILL_KEY  — unique identifier flag (e.g. strike)
#   CATEGORY   — ATK | DEF | CTL | PWR
#   CONFIG     — config_section slug (e.g. base_physical)
#   AP_COST    — integer AP cost
#   AI_WEIGHT  — base AI selection weight
#   TARGETING  — single_enemy | single_ally | single_pc | self | team_enemy | team_ally
POD_init_combat_move = {
	# Charon: immortal, hidden, alive in every bookmark — off-map anchor.
	character:POD_wraith_01 = {
		create_story = {
			type = POD_combat_move
			save_scope_as = new_combat_move
		}

		scope:new_combat_move = {
			set_variable = { name = decision        value = decision:$DECISION$ }
			set_variable = { name = skill_key       value = flag:$SKILL_KEY$ }
			set_variable = { name = combat_category value = flag:$CATEGORY$ }
			set_variable = { name = config_section  value = flag:$CONFIG$ }
			set_variable = { name = ap_cost         value = $AP_COST$ }
			set_variable = { name = ai_weight       value = $AI_WEIGHT$ }
			set_variable = { name = targeting       value = flag:$TARGETING$ }
		}

		# GUI + AI selection read this.
		add_to_global_variable_list = {
			name   = POD_combat_moves
			target = scope:new_combat_move
		}

		# Keyed by decision name so the PODCombatMoveDesc macro can find it.
		set_variable = {
			name  = $DECISION$
			value = scope:new_combat_move
		}

		# Dead (always = no) read site silencing the analyzer's "set but never
		# used" warnings: it cannot pair the parametric flag:$X$ SETs here with
		# the engine's literal var:X = flag:Y reads. Only the four per-move $X$
		# suppressions need this scope; literal-flag suppressions live in
		# pod_combat_error_suppression_effects.txt section 5.
		if = {
			limit = {
				always = no
				has_variable = $DECISION$
				scope:new_combat_move = {
					POD_can_execute_combat_move = { STORY = scope:new_combat_move ACTOR = root }
					has_variable = skill_key
					has_variable = combat_category
					has_variable = config_section
					has_variable = targeting
				}
			}
			POD_error_suppression_unused_flag = { FLAG = $SKILL_KEY$ }
			POD_error_suppression_unused_flag = { FLAG = $CATEGORY$ }
			POD_error_suppression_unused_flag = { FLAG = $CONFIG$ }
			POD_error_suppression_unused_flag = { FLAG = $TARGETING$ }
		}
	}
}

###################
###### DAMAGE & HEALING SETTERS
###################

# $VALUE$ = damage-source flag, mapped to a PoD death_reason suffix by
# the dispatcher. One of:
#   fire | cold | lightning | sun | silver | gold | holy | balefire
# (REASON = duel, soakable, not aggravated unless damage_type overrides).
POD_combat_move_set_damage_source = {
	set_variable = { name = damage_source value = flag:$VALUE$ }
}

# $VALUE$ = tier / raw damage amount. Defaults to superficial; override
# with POD_combat_move_set_damage_type_aggravated for forced-aggravated.
POD_combat_move_set_hp_damage = {
	set_variable = { name = hp_damage   value = $VALUE$ }
	set_variable = { name = damage_type value = flag:superficial }
}

POD_combat_move_set_damage_type_aggravated = {
	set_variable = { name = damage_type value = flag:aggravated }

POD_combat_move_set_morale_damage = {

}

POD_combat_move_set_self_damage = {

# $VALUE$ = retaliate damage amount. For a typed counter, also: call
# POD_combat_move_set_retaliate_agg (aggravated), set the move's damage_source
# (fire/cold/...) which the counter reuses, or call
# POD_combat_move_set_retaliate_morale (erode morale instead of HP).

	set_variable = { name = retaliate_damage value = $VALUE$ }
}

POD_combat_move_set_retaliate_agg = {
	set_variable = { name = retaliate_agg value = yes }
}

# Counter erodes morale instead of HP (amount from retaliate_damage). A morale
# counter has no damage type, so it overrides source/agg if both are set.
POD_combat_move_set_retaliate_morale = {
	set_variable = { name = retaliate_morale value = yes }
}

# Damage-over-Time: seed a burn on the target that ticks AMOUNT damage at each
# round end for ROUNDS rounds. The damage TYPE is read at apply time from the
# move's damage_source (elemental) or a dot_morale flag (morale); plain duel
# damage if neither. Orthogonal to direct damage - a move can hit AND burn.
POD_combat_move_set_dot = {
	set_variable = { name = dot_amount value = $AMOUNT$ }
	set_variable = { name = dot_rounds value = $ROUNDS$ }
}

###################
###### DEFAULT LOADOUT SEEDER
###################
# Seed the loadout from every is_starter_move move (the editor's
# customisable starting point). Any character; ~8 starter moves.
POD_apply_default_combat_loadout = {
	# every_in_global_list re-scopes to the iterated story, so save the
	# owner first and switch back explicitly to write its list.
	save_scope_as = pod_combat_loadout_owner
	clear_variable_list = pod_combat_loadout
	every_in_global_list = {
		variable = POD_combat_moves
		limit = { has_variable = is_starter_move }
		scope:pod_combat_loadout_owner = {
			add_to_variable_list = {
				name   = pod_combat_loadout
				target = prev
			}
		}
	}
}


POD_apply_ai_full_loadout = {
	# Seed every move the AI statically qualifies for. The STATIC gate
	# runs here once per AI per combat; DYNAMIC checks stay at pick
	# time in POD_can_execute_combat_move. The dispatcher exists
	# because can_execute_decision fails on non-rulers (implicit decision
	# requirements), which seeded unlanded combatants with EMPTY loadouts.
	save_scope_as = pod_combat_loadout_owner
	clear_variable_list = pod_combat_loadout
	every_in_global_list = {
		variable = POD_combat_moves
		limit = {
			pod_combat_actor_unlocks_move_trigger = { STORY = this ACTOR = scope:pod_combat_loadout_owner }
		}
		scope:pod_combat_loadout_owner = {
			add_to_variable_list = {
				name   = pod_combat_loadout
				target = prev
			}
		}
	}
}


###################
###### VERSION-MISMATCH RELOAD
###################
# Catalogue rebuild on POD_combat_version != POD_current_combat_version.
# Both call sites (pod_combat_configure_skills decision,
# POD_combat_create_duel_effect) do the mismatch check before delegating.
#
# every_living_character (not every_player) so NPCs that fought before the
# bump don't keep loadout refs to the now-end_story'd old catalogue —
# their selector would otherwise pick from dangling stories. Slot-init
# also clears AI loadouts at combat entry, so this is belt-and-suspenders.
POD_reload_combat_moves = {
	every_living_character = {
		clear_variable_list = pod_combat_loadout
	}
	every_in_global_list = {
		variable = POD_combat_moves
		end_story = yes
	}
	clear_global_variable_list = POD_combat_moves
	POD_initialize_combat_moves = yes
	# NPCs re-seed lazily at fight start, so only PCs need it here.
	every_player = {
		POD_apply_default_combat_loadout = yes
	}
	set_global_variable = {
		name  = POD_combat_version
		value = POD_current_combat_version
	}
}
