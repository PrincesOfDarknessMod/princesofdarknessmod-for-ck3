######################################################
######   POD COMBAT MOVE INIT EFFECTS
######################################################
# Authoring layer for combat moves: the POD_init_combat_move
# constructor + modular POD_combat_move_set_* setters, driven by
# POD_initialize_combat_moves (pod_combat_move_list_effects.txt).
# A move is a POD_combat_move story_cycle whose variables the
# execution engine (pod_combat_move_effects.txt) reads at runtime.
#
# All setters run inside the story scope; presence of a variable =
# attribute applies, absence = skip. Some setters are unreferenced by
# any current move row but retain full runtime support.

###################
###### CONSTRUCTOR
###################

# Parameters:
#   DECISION - hidden presentation decision (art + shared-rule failure text)
#   SKILL_KEY - unique identifier flag (e.g. strike)
#   CATEGORY - ATK | DEF | CTL | PWR
#   CONFIG - config_section slug (e.g. base_physical)
#   GROUP - splat family for the show-all display filter (universal | vampire |
#           werewolf | bastet | mokole | fera | fae | demon | wraith | mummy |
#           kueijin | hunter | sorcerer | psychic | theurgy | truefaith)
#   AP_COST - integer AP cost
#   AI_WEIGHT - base AI selection weight
#   TARGETING - single_enemy | single_ally | single_pc | self | team_enemy | team_ally | team_all
#               (team_all hits both teams, the caster included)
POD_init_combat_move = {
	# Charon: immortal, hidden, alive in every bookmark - off-map anchor.
	character:POD_wraith_01 = {
		create_story = {
			type = POD_combat_move
			save_scope_as = new_combat_move
		}

		scope:new_combat_move = {
			set_variable = { name = decision          value = decision:$DECISION$ }
			set_variable = { name = skill_key         value = flag:$SKILL_KEY$ }
			set_variable = { name = combat_category   value = flag:$CATEGORY$ }
			set_variable = { name = config_section    value = flag:$CONFIG$ }
			set_variable = { name = move_unlock_group value = flag:$GROUP$ }
			set_variable = { name = ap_cost           value = $AP_COST$ }
			set_variable = { name = ai_weight         value = $AI_WEIGHT$ }
			set_variable = { name = targeting         value = flag:$TARGETING$ }
		}

		# GUI + AI selection read this.
		add_to_global_variable_list = {
			name   = POD_combat_moves
			target = scope:new_combat_move
		}

		# Keyed by decision name so the PODCombatMoveDesc macro can find it.
		set_variable = {
			name  = $DECISION$
			value = scope:new_combat_move
		}

		# Dead (always = no) read site silencing the analyzer's "set but never
		# used" warnings: it cannot pair the parametric flag:$X$ SETs here with
		# the engine's literal var:X = flag:Y reads. Only the five per-move $X$
		# suppressions need this scope; literal-flag suppressions live in
		# pod_combat_error_suppression_effects.txt section 4.
		if = {
			limit = {
				always = no
				has_variable = $DECISION$
				scope:new_combat_move = {
					POD_can_execute_combat_move_trigger = { STORY = scope:new_combat_move ACTOR = root }
					has_variable = skill_key
					has_variable = combat_category
					has_variable = config_section
					has_variable = move_unlock_group
					has_variable = targeting
				}
			}
			POD_error_suppression_unused_flag = { FLAG = $SKILL_KEY$ }
			POD_error_suppression_unused_flag = { FLAG = $CATEGORY$ }
			POD_error_suppression_unused_flag = { FLAG = $CONFIG$ }
			POD_error_suppression_unused_flag = { FLAG = $GROUP$ }
			POD_error_suppression_unused_flag = { FLAG = $TARGETING$ }
		}
	}
}

###################
###### MARKER-MODIFIER SETTER
###################
# Applies a marker character modifier to the move's own user; section 12c of
# POD_execute_combat_move fires it. Read back by the has_modifier condition
# predicate, which is what lets a LATER move gate or bonus off the state this
# move left behind (the anima-flare / charm AP-refund loop).
# $KEY$ must be a key defined in common/modifiers/pod_combat_modifiers.txt and
# listed in the generator's MARKER_MODIFIERS: CK3 needs a literal for both
# add_character_modifier and has_character_modifier, so section 12c and the
# generated sub-dispatch each switch over the key rather than interpolating it.
# The user is the only supported recipient - see the authoring guide on why a
# team_ally marker move is rejected.
POD_combat_move_set_marker_modifier = {
	set_variable = { name = marker_modifier value = flag:$KEY$ }
}

# Strips a marker character modifier instead of applying one.
# $KEY$    = a MARKER_MODIFIERS key (combat-owned only - a move must never strip
#            a PoD-owned modifier such as a shapeshift form or discipline buff,
#            which is why this list is MARKER_MODIFIERS and not the wider
#            COND_READABLE_MODIFIERS the has_modifier condition may name).
# $TARGET$ = user | target. `target` reads the loaded target workspace, so the
#            move's own targeting decides whether that is an ally or an enemy;
#            the generator rejects `target` on team targeting because this
#            writes one workspace side and does not broadcast.
POD_combat_move_set_remove_marker = {
	set_variable = { name = remove_marker        value = flag:$KEY$ }
	set_variable = { name = remove_marker_target value = flag:$TARGET$ }
}

# Generalized move condition (gate or bonus).
# $MODE$ = gate | bonus  ;  $TARGET$ = user | target | global | relational
# $PREDICATE$ = the state predicate (see pod_combat_move_condition_met_trigger)
# $THRESHOLD$ = the X for value predicates (pass 0 for boolean ones)
POD_combat_move_set_condition = {
	set_variable = { name = cond_mode      value = flag:$MODE$ }
	set_variable = { name = cond_target    value = flag:$TARGET$ }
	set_variable = { name = cond_predicate value = flag:$PREDICATE$ }
	set_variable = { name = cond_threshold value = $THRESHOLD$ }
}

# Identifier-predicate param: has_trait -> trait key, target_known_secret ->
# secret_type. Read by the pod_combat_cond_*_trigger sub-dispatches.
# Mode-only twin of POD_combat_move_set_condition, for a move whose row-level
# condition is nothing but the mirror of bonus clause 1. cond_mode is still
# read on a bonus move by the sheet/compact "if+" glyphs (PODCombatSheetFxCond,
# PODCombatCLCond) and by the sheet's conditional filter, but cond_target /
# cond_predicate / cond_threshold / cond_param are not: every reader of those
# is reached only through cond_mode = flag:gate, and the bonus path evaluates
# the CLAUSE story instead. Writing them anyway would duplicate clause 1 with
# no reader and give the two copies room to drift.
# Display projection for the sheet AP column and the compact line, which are
# customizable_localization on the MOVE story and cannot index bonus_clauses to
# fetch a magnitude. Written only when clause 1 grants ap_drain (the generator
# position-locks that effect to clause 1 for exactly this reason), so the number
# those two columns print is always the clause they describe. Not gameplay data:
# execution reads the clause, never this.
POD_combat_move_set_sheet_ap_drain = {
	set_variable = { name = sheet_ap_drain value = $VALUE$ }
}

###################
###### BONUS-CLAUSE SETTERS
###################
# A move may carry N stackable conditional bonuses, each its own condition and
# its own payload, on top of the single move-level gate the cond_* scalars hold.
# One clause = one POD_combat_move_clause story carrying the SAME cond_* var
# names the move-level gate uses, which is what lets
# pod_combat_move_condition_met_trigger evaluate a clause verbatim with
# STORY = the clause story. Execution §2b walks the list once and folds every
# met clause into per-effect fight transients, so same-effect clauses sum and
# each phase still applies exactly once.
#
# Runs in MOVE-story scope, like every other setter here.
# $TARGET$ = user | target | global | relational
# $PREDICATE$ = the state predicate (see pod_combat_move_condition_met_trigger)
# $THRESHOLD$ = the X for value predicates (pass 0 for boolean ones)
# $EFFECT$ = the bonus primitive (the bonus_effect branches in §2b)
# $VALUE$ = its magnitude (pass 0 for the magnitude-less effects)
POD_combat_move_add_bonus_clause = {
	# Charon owns the clause stories for the same reason he owns the moves.
	character:POD_wraith_01 = {
		create_story = {
			type = POD_combat_move_clause
			save_scope_as = pod_new_clause
		}
	}
	scope:pod_new_clause = {
		set_variable = { name = cond_target    value = flag:$TARGET$ }
		set_variable = { name = cond_predicate value = flag:$PREDICATE$ }
		set_variable = { name = cond_threshold value = $THRESHOLD$ }
		set_variable = { name = bonus_effect   value = flag:$EFFECT$ }
		set_variable = { name = bonus_value    value = $VALUE$ }
	}
	# Teardown handle. POD_reload_combat_moves ends the move stories that hold
	# the bonus_clauses lists, so this global list is the only reference that
	# still reaches a clause by the time the rebuild needs to end it.
	add_to_global_variable_list = {
		name   = POD_combat_move_clauses
		target = scope:pod_new_clause
	}
	add_to_variable_list = {
		name   = bonus_clauses
		target = scope:pod_new_clause
	}
}

# The three optional clause fields. Separate setters rather than arguments on
# the add above because CK3 macros have no optional arguments and an empty
# argument is a parse error; each reads scope:pod_new_clause, so a call must
# follow its own POD_combat_move_add_bonus_clause immediately.

# Identifier-predicate param, the clause twin of POD_combat_move_set_cond_param.
POD_combat_move_clause_set_cond_param = {
	scope:pod_new_clause = { set_variable = { name = cond_param value = flag:$PARAM$ } }
}

POD_combat_move_clause_set_cond_param_trait = {
	scope:pod_new_clause = { set_variable = { name = cond_param_trait value = $PARAM$ } }
}

# change_class element, or the independent class for a typed damage bonus. The
# move-level twin is gone: nothing outside the clause list reads it.
POD_combat_move_clause_set_bonus_param = {
	scope:pod_new_clause = { set_variable = { name = bonus_param value = flag:$PARAM$ } }
}

# hide_condition_if_unfulfilled: the stat card hides this clause's condition
# line AND its effect line together until the viewer meets the condition, so an
# unearned rider never advertises itself. Presence var - no value to carry.
POD_combat_move_clause_set_hide = {
	scope:pod_new_clause = { set_variable = hide_if_unmet }
}

###### DEFAULT LOADOUT SEEDER
###################
# Seed the loadout from every is_starter_move move (the editor's
# customisable starting point). Any character; ~8 starter moves.
POD_apply_default_combat_loadout = {
	# every_in_global_list re-scopes to the iterated story, so save the
	# owner first and switch back explicitly to write its list.
	save_scope_as = pod_combat_loadout_owner
	clear_variable_list = pod_combat_loadout
	every_in_global_list = {
		variable = POD_combat_moves
		limit = { has_variable = is_starter_move }
		scope:pod_combat_loadout_owner = {
			add_to_variable_list = {
				name   = pod_combat_loadout
				target = prev
			}
		}
	}
}


POD_apply_ai_full_loadout = {
	# Seed every move the AI statically qualifies for. The STATIC unlock
	# gate runs ONLY here, once per AI per combat (eagerly at fight
	# finalize / mid-fight seat); pick-time validation is dynamic-only
	# (pod_combat_move_is_usable_now_trigger), so loadout membership IS
	# the unlock proof - never seed an AI list around this dispatcher
	# filter. The dispatcher exists because can_execute_decision fails on
	# non-rulers (implicit decision requirements), which seeded unlanded
	# combatants with EMPTY loadouts.
	save_scope_as = pod_combat_loadout_owner
	clear_variable_list = pod_combat_loadout
	every_in_global_list = {
		variable = POD_combat_moves
		limit = {
			pod_combat_actor_unlocks_move_trigger = { STORY = this ACTOR = scope:pod_combat_loadout_owner }
		}
		scope:pod_combat_loadout_owner = {
			add_to_variable_list = {
				name   = pod_combat_loadout
				target = prev
			}
		}
	}
}


###################
###### SAVED LOADOUT SLOTS (player presets)
###################
# Four loadout-preset slots per character. Each slot is a variable list
# pod_combat_saved_loadout_$SLOT$ (1-4) of move SKILL_KEY flags - NOT story refs.
# Story refs die when POD_reload_combat_moves rebuilds the catalogue (a version /
# save reload), which silently empties the slot; skill_key is stable across rebuilds,
# so presets survive. Save copies the live loadout's skill_keys into a slot; Load
# rebuilds the loadout from the catalogue, taking every move whose skill_key is in the
# slot AND that the PC still unlocks (a move the PC no longer qualifies for is skipped).
# Caps are inherently preserved (a slot was saved from a cap-valid loadout; Load only
# drops moves). $SLOT$ text-substitutes into the literal list name.

pod_combat_save_loadout_slot_effect = {
	# this = the PC. Store each equipped move's stable skill_key (reload-proof).
	save_scope_as = pod_combat_loadout_owner
	clear_variable_list = pod_combat_saved_loadout_$SLOT$
	every_in_list = {
		variable = pod_combat_loadout
		scope:pod_combat_loadout_owner = {
			add_to_variable_list = { name = pod_combat_saved_loadout_$SLOT$ target = prev.var:skill_key }
		}
	}
}

pod_combat_load_loadout_slot_effect = {
	# this = the PC. Rebuild the loadout from the catalogue: take each move whose
	# skill_key is in the slot AND that the PC still unlocks. Iterating the catalogue
	# (not the slot's flags) keeps it a single scope-trap-free pass and re-binds the
	# stored skill_keys to the CURRENT move stories.
	save_scope_as = pod_combat_loadout_owner
	clear_variable_list = pod_combat_loadout
	every_in_global_list = {
		variable = POD_combat_moves
		limit = {
			scope:pod_combat_loadout_owner = {
				is_target_in_variable_list = { name = pod_combat_saved_loadout_$SLOT$ target = prev.var:skill_key }
			}
			pod_combat_actor_unlocks_move_trigger = { STORY = this ACTOR = scope:pod_combat_loadout_owner }
		}
		scope:pod_combat_loadout_owner = {
			add_to_variable_list = { name = pod_combat_loadout target = prev }
		}
	}
	# Keep the editor's count cache (pip strips / cap checks) in step.
	pod_combat_refresh_loadout_counts_effect = yes
}

pod_combat_clear_loadout_slot_effect = {
	clear_variable_list = pod_combat_saved_loadout_$SLOT$
}

###################
###### MOVE-SHEET SCRATCH LIST (spreadsheet pool view)
###################
# The sheet renders a per-character copy of the catalogue instead of the
# global list: GUI datamodels cannot sort or filter, so both happen here at
# build time and the list's insertion order IS the display order (same
# load-bearing fact the combat log rides on). Rebuilt on: sheet view
# selected, editor open (if sheet view persisted), every filter/sort click,
# and equip toggles while the equipped-only filter is on. Cleared at editor
# close (one scratch story ref per catalogue move is save bloat) and by
# POD_reload_combat_moves (story refs die with the catalogue).

pod_combat_sheet_rebuild_effect = {
	# this = the viewing PC. Seed sort defaults first: the sort SV and the
	# header active-column SGUIs compare var:pod_combat_sheet_sort, and a
	# compare on an unset var logs errors.
	if = {
		limit = { NOT = { has_variable = pod_combat_sheet_sort } }
		set_variable = { name = pod_combat_sheet_sort value = flag:ap }
		set_variable = { name = pod_combat_sheet_asc value = yes }
	}
	# Iterators re-scope to the iterated story; save the owner to filter
	# against and write back to (same trap-note as the loadout copies above).
	save_scope_as = pod_combat_sheet_owner
	clear_variable_list = pod_combat_sheet_rows
	ordered_in_global_list = {
		variable = POD_combat_moves
		limit = { pod_combat_sheet_row_match_trigger = yes }
		order_by = pod_combat_sheet_sort_value
		# No position: iterate every match in order. max only stops the ordered
		# iterator being read as pick-one; 5000 sits far above any catalogue size.
		max = 5000
		check_range_bounds = no
		scope:pod_combat_sheet_owner = {
			add_to_variable_list = { name = pod_combat_sheet_rows target = prev }
		}
	}
}

# Header click: new column sorts ascending; re-clicking the active column
# flips direction (ledger behavior).
# $SORT$ = ap/energy/damage/heal/morale/dot/apswing.
pod_combat_sheet_set_sort_effect = {
	if = {
		limit = {
			has_variable = pod_combat_sheet_sort
			var:pod_combat_sheet_sort = flag:$SORT$
		}
		if = {
			limit = { has_variable = pod_combat_sheet_asc }
			remove_variable = pod_combat_sheet_asc
		}
		else = {
			set_variable = { name = pod_combat_sheet_asc value = yes }
		}
	}
	else = {
		set_variable = { name = pod_combat_sheet_sort value = flag:$SORT$ }
		set_variable = { name = pod_combat_sheet_asc value = yes }
	}
	pod_combat_sheet_rebuild_effect = yes
}

# Filter-popup Reset: sheet-specific filters off, sort back to AP ascending.
# Deliberately leaves the shared category filters and show-all alone - those
# belong to the whole editor, not just the sheet.
pod_combat_sheet_reset_effect = {
	remove_variable = pod_combat_sheet_f_equipped
	remove_variable = pod_combat_sheet_f_damage
	remove_variable = pod_combat_sheet_f_heal
	remove_variable = pod_combat_sheet_f_energy
	remove_variable = pod_combat_sheet_f_agg
	remove_variable = pod_combat_sheet_f_elemental
	remove_variable = pod_combat_sheet_f_conditional
	set_variable = { name = pod_combat_sheet_sort value = flag:ap }
	set_variable = { name = pod_combat_sheet_asc value = yes }
	pod_combat_sheet_rebuild_effect = yes
}

###################
###### VERSION-MISMATCH RELOAD
###################
# Catalogue rebuild on POD_combat_version != POD_current_combat_version.
# Both call sites (pod_combat_configure_skills decision,
# POD_combat_create_duel_effect) do the mismatch check before delegating.
#
# every_living_character (not every_player) so NPCs that fought before the
# bump don't keep loadout refs to the now-end_story'd old catalogue - 
# their selector would otherwise pick from dangling stories. Slot-init
# also clears AI loadouts at combat entry, so this is belt-and-suspenders.
POD_reload_combat_moves = {
	every_living_character = {
		clear_variable_list = pod_combat_loadout
		# The sheet scratch list holds refs to the stories end_story'd below.
		clear_variable_list = pod_combat_sheet_rows
		# The per-char unlock list holds refs to those same stories.
		clear_variable_list = pod_combat_unlocked_moves
	}
	# Clauses first: ending a move drops the bonus_clauses list that points at
	# them, so after the moves teardown below nothing could reach a clause story
	# to end it and the rebuild would leak one per clause per reload.
	every_in_global_list = {
		variable = POD_combat_move_clauses
		end_story = yes
	}
	clear_global_variable_list = POD_combat_move_clauses
	every_in_global_list = {
		variable = POD_combat_moves
		end_story = yes
	}
	clear_global_variable_list = POD_combat_moves
	POD_initialize_combat_moves = yes
	# NPCs re-seed lazily at fight start, so only PCs need it here.
	every_player = {
		POD_apply_default_combat_loadout = yes
	}
	set_global_variable = {
		name  = POD_combat_version
		value = POD_current_combat_version
	}
}
