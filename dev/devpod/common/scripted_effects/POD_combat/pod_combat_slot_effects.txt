# Per-slot story_cycle (POD_combat_slot) lifecycle + accessor helpers.
# Caller scope is the combat host; stories live on Charon (POD_wraith_01).
#
# Slot identity = (combat_team flag:player/enemy, team_position int 0+,
# 0 = leader). No flat slot_index. Addressing is by ITERATION over the
# single host list pod_combat_active_slots - no per-position pointers, so
# teams are unbounded. The only pointer vars are the two leaders
# (pod_combat_pc_slot = player/0, pod_combat_enemy_leader_slot = enemy/0),
# derived once at finalize for the many direct-leader call sites.
#
# Field catalogue / init defaults / absence-as-signal list: COMBAT_REFERENCE.info §3.


###################
###### CLEANUP
###################

# end_story frees the slot stories from Charon's owned-stories list (same
# teardown as log entries / sc_context). The leader-pointer removes are
# no-ops if a fight never set them. Caller must be in host scope.
pod_combat_clear_slot_stories_effect = {
	every_in_list = {
		variable = pod_combat_active_slots
		end_story = yes
	}
	clear_variable_list = pod_combat_active_slots
	clear_variable_list = pod_combat_turn_order_list
	remove_variable = pod_combat_pc_slot
	remove_variable = pod_combat_enemy_leader_slot
}


###################
###### ACCESSOR HELPERS
###################
# Write helpers operating on the CURRENT slot story. Callers are either
# inside an `every_in_list = pod_combat_active_slots` loop, or have
# scope-switched into a leader pointer via `var:pod_combat_pc_slot = { ... }`
# - the `var:` prefix is mandatory; without it CK3 reads the pointer name as
# an effect and silently drops the block.

pod_combat_set_slot_alive_effect = {
	# $VALUE$ = 0 or 1
	set_variable = { name = slot_alive value = $VALUE$ }
}

# Universal slot-var setter: writes slot_$NAME$ on the current slot story.
#   $NAME$  = stat short name (morale, prowess, soak, hit_chance, ...)
#   $VALUE$ = literal, scope:, or var: reference.
#
# $VALUE$ is captured via save_scope_value_as FIRST so scope:/var: refs
# resolve to the caller's value before the write flips into the slot story.
pod_combat_set_slot_var_effect = {
	save_scope_value_as = { name = pod_combat_slot_var_temp value = $VALUE$ }
	set_variable = { name = slot_$NAME$ value = scope:pod_combat_slot_var_temp }
}

# Removes slot_$NAME$ from the current slot story. Used for
# absence-as-signal fields whose presence carries meaning (stances,
# retaliate vars, init_bonus, ap_drain).
#   $NAME$ = field short name (is_guarding, retaliate_hit, init_bonus, ...)
pod_combat_remove_slot_var_effect = {
	remove_variable = slot_$NAME$
}

# Create a fresh slot story for one combatant and append it to the host's
# pod_combat_active_slots. The duel builder only ever APPENDS a new (team,
# position) from a runtime team-size value, so this ALWAYS creates - no
# per-position pointer, no create-guard. Caller must be in host scope.
#
# hidden_effect_new_object wrap: CK3's tooltip builder walks effect chains
# symbolically and skips state-mutating calls including create_story, so a
# later scope:new_slot read errors "Undefined event target" during tooltip
# preview. The wrap lets real execution run while preview skips the
# new-object block. Same pattern as PoD's POD_heist_create_node.
pod_combat_set_slot_char_effect = {
	# $TEAM$      = player | enemy  (parse-time literal)
	# $POSITION$  = team_position to assign (runtime value, 0 = leader)
	# $CHARACTER$ = character scope to assign to the slot
	#
	# Capture $POSITION$ in host scope BEFORE the new-object block so it stays
	# readable as a scope value after we flip into the new story.
	save_scope_value_as = { name = pod_combat_slot_pos_temp value = $POSITION$ }
	hidden_effect_new_object = {
		character:POD_wraith_01 = {
			create_story = {
				type = POD_combat_slot
				save_scope_as = new_slot
			}
		}
		scope:new_slot = {
			# Identity: combat_team is the source of truth, position 0 = leader.
			set_variable = { name = combat_team   value = flag:$TEAM$ }
			set_variable = { name = team_position value = scope:pod_combat_slot_pos_temp }
			set_variable = { name = slot_alive    value = 1 }
			# slot_char is set up front (before slot_morale) so the starting-morale
			# designer hook can read var:slot_char.
			set_variable = { name = slot_char        value = $CHARACTER$ }
			# Pure-value stats zero-defaulted so SV gates need only a slot
			# presence check, not per-field has_variable; gate-style fields
			# (base_soak, hit_chance, initiative, stances, tiers) stay
			# absent-until-written.
			#
			# slot_morale defaults to pod_combat_starting_morale_value (cap or
			# tuned override + the designer hook, floored at 1), not 0, so a
			# freshly-created slot reads alive before real init: the init-time
			# KO check treats var:slot_morale <= 0 as a KO predicate. Same SV
			# as the per-slot init seed in pod_combat_group_effects.txt.
			set_variable = {
				name = slot_morale
				value = pod_combat_starting_morale_value
			}
			set_variable = { name = slot_prowess     value = 0 }
			set_variable = { name = slot_soak        value = 0 }
			set_variable = { name = slot_energy_used value = 0 }
			set_variable = { name = slot_has_acted   value = 0 }
		}

		# Register on the host - scope inside the wrap is still the caller.
		# Iterated by the GUI datamodel and every dispatch loop; cleared in
		# pod_combat_clear_slot_stories_effect.
		add_to_variable_list = {
			name = pod_combat_active_slots
			target = scope:new_slot
		}
	}
}
