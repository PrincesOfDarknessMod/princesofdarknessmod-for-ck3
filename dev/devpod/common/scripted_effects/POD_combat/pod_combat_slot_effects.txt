# Per-slot lifecycle and variable-setting effects. Host scope unless specified; stories live on Charon.
# Slots are addressed by (combat_team, team_position), with position 0 as leader.
# The fight owns the roster and leader pointers; its on_end cascade ends the slots.
# Field defaults and absence-as-signal rules: COMBAT_REFERENCE.info section 3.




###################
###### SLOT VARIABLE EFFECTS
###################
# Slot-story scope. Pointer dereferences require var: (for example,
# var:pod_combat_session.var:pod_combat_pc_slot); bare names parse as effects.

pod_combat_set_slot_alive_effect = {
	# $VALUE$ = 0 or 1
	set_variable = { name = slot_alive value = $VALUE$ }
}

# $NAME$ = stat suffix; $VALUE$ = numeric literal, scope: or var: reference.
# Values pass through save_scope_value_as; do not use this setter for flag variables.
pod_combat_set_slot_var_effect = {
	save_scope_value_as = { name = pod_combat_slot_var_temp value = $VALUE$ }
	set_variable = { name = slot_$NAME$ value = scope:pod_combat_slot_var_temp }
}

# $NAME$ = field suffix. Remove presence-as-signal fields to deactivate them.
pod_combat_remove_slot_var_effect = {
	remove_variable = slot_$NAME$
}

# Mark an existing seat. Caller: character holding pod_combat_session.
# $TEAM$ = player|enemy literal; $POSITION$ = existing team_position;
# $CONTROLLER$ = driving character, who need not occupy a slot but needs a session pointer.
# Multiple controlled seats are supported.
# Call before finalize so the occupant's unlocks are baked. Late marks must call
# pod_combat_bake_pc_loadout_unlocks_effect on the occupant.
# add_ally increments the team-size counter after seating; capture the position before adding.
# To unmark, use pod_combat_remove_slot_var_effect = { NAME = human_controller }.
pod_combat_mark_slot_controlled_effect = {
	# Capture arguments before iteration changes scope.
	save_scope_value_as = { name = pod_mark_slot_pos value = $POSITION$ }
	$CONTROLLER$ = { save_scope_as = pod_mark_controller }
	var:pod_combat_session ?= {
		every_in_list = {
			variable = pod_combat_active_slots
			limit = {
				var:combat_team = flag:$TEAM$
				var:team_position = scope:pod_mark_slot_pos
			}
			set_variable = { name = slot_human_controller value = scope:pod_mark_controller }
		}
	}
}

# Host scope. Always creates a new slot; the caller supplies a fresh (team, position).
# hidden_effect_new_object prevents tooltip preview from reading the uncreated slot.
pod_combat_set_slot_char_effect = {
	# $TEAM$      = player | enemy  (parse-time literal)
	# $POSITION$  = team_position to assign (runtime value, 0 = leader)
	# $CHARACTER$ = character scope to assign to the slot
	#
	# Capture the position before entering the new story.
	save_scope_value_as = { name = pod_combat_slot_pos_temp value = $POSITION$ }

	# Clear at every seating, including mid-fight arrivals, so teardown cannot
	# remove a form inherited from an earlier fight.
	if = {
		limit = { $CHARACTER$ = { is_alive = yes } }
		$CHARACTER$ = { remove_character_flag = pod_combat_fight_granted_warform }
	}
	hidden_effect_new_object = {
		character:POD_wraith_01 = {
			create_story = {
				type = POD_combat_slot
				save_scope_as = new_slot
			}
		}
		scope:new_slot = {
			set_variable = { name = combat_team   value = flag:$TEAM$ }
			set_variable = { name = team_position value = scope:pod_combat_slot_pos_temp }
			set_variable = { name = slot_alive    value = 1 }
			# Set slot_char before morale so the starting-morale hook can read it.
			set_variable = { name = slot_char        value = $CHARACTER$ }
			# Morale must start above zero to avoid the init-time KO check.
			# Use the same value as slot init so designer adjustments survive initialization.
			# Stances and tiers remain absent until applied; character stats stay on slot_char.
			set_variable = {
				name = slot_morale
				value = pod_combat_starting_morale_value
			}
			set_variable = { name = slot_energy_used value = 0 }
			set_variable = { name = slot_has_acted   value = 0 }
		}

		# Read the fight through the host: mid-fight arrivals lack builder scopes.
		# Tooltip preview may have no session.
		if = {
			limit = { has_variable = pod_combat_session }
			var:pod_combat_session = { save_temporary_scope_as = pod_fight }
			scope:pod_fight = {
				add_to_variable_list = {
					name = pod_combat_active_slots
					target = scope:new_slot
				}
			}
			scope:new_slot = {
				set_variable = { name = fight value = scope:pod_fight }
			}
			# Every participant needs the session for busy gates and its client's GUI.
			# Both teardown paths clear it from living characters; dead characters cannot be mutated.
			$CHARACTER$ = {
				set_variable = { name = pod_combat_session value = scope:pod_fight }
			}
		}
	}
}
