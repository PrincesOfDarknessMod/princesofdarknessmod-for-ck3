###### TEMPORARY TROILE
######################################################

# New Troile's character scope, from the template's after_creation. Identity, grants and authored
# stats only; POD_combat_boss_finish_creation_effect supplies the rest from the creation hook.
POD_combat_troile_debug_initialize_effect = {
	POD_combat_initialize_boss_effect = { TIER = 3 }
	POD_combat_boss_grant_move_effect = { MOVE = POD_combat_move_troile_between_heartbeats }
	POD_combat_boss_grant_unlimited_energy_effect = yes
	# Elder perks have no initializer, so they stay explicit; the base trees come from the hook.
	POD_assign_elder_lifestyle = { LIFESTYLE = celerity GEN = 3 }
	POD_assign_elder_lifestyle = { LIFESTYLE = potence GEN = 3 }
	POD_assign_elder_lifestyle = { LIFESTYLE = presence GEN = 3 }
	# The potency effect switches on a trigger or saved numeric scope, not a literal number.
	save_scope_value_as = { name = pod_combat_boss_bloodpotency value = 10 }
	POD_set_bloodpotency_effect = { POTENCY = scope:pod_combat_boss_bloodpotency }
}

# clan, tribe or auspice scopes saved. TITLE and TEAM_NAME are localization keys, so the tier
# picks the launcher call rather than passing through it.

			TEAM_NAME = pod_combat_boss_vampire_t2_team_name
