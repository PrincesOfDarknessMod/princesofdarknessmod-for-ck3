# Called once per bonus clause by pod_combat_move_desc. Keep effects in scripted_effects/ for resolution.
# this = viewer; pod_clause = clause story; story = move story; nohide = optional log-card override.
# Default damage class comes from the move; clause magnitude comes from pod_clause.
# Execution counterpart: COMBAT_REFERENCE.info section 2b.

# Hide the condition and bonus together so the card cannot imply an unconditional effect.
pod_combat_card_clause_line_effect = {
	if = {
		limit = {
			# Nest guarded reads: raw_text evaluates even in hidden branches.
			trigger_if = {
				limit = { scope:pod_clause = { has_variable = hide_if_unmet } }
				trigger_if = {
					limit = { exists = scope:nohide }
					always = yes
				}
				# Evaluate viewer traits/perks/modifiers live. Execution uses the driven actor instead,
				# so proxy/clone cards can differ until the card receives an explicit actor scope.
				trigger_else_if = {
					limit = { scope:pod_clause = { var:cond_predicate = flag:has_perk } }
					pod_combat_cond_has_perk_trigger = { STORY = scope:pod_clause }
				}
				trigger_else_if = {
					limit = { scope:pod_clause = { var:cond_predicate = flag:has_trait } }
					pod_combat_cond_has_trait_trigger = { STORY = scope:pod_clause }
				}
				trigger_else_if = {
					limit = { scope:pod_clause = { var:cond_predicate = flag:has_modifier } }
					pod_combat_cond_has_modifier_trigger = { STORY = scope:pod_clause }
				}
				# Unsupported hidden conditions fail closed.
				trigger_else = { always = no }
			}
			trigger_else = { always = yes }
		}
		pod_combat_card_clause_separator_effect = yes
		pod_combat_card_clause_condition_line_effect = yes
		pod_combat_card_clause_bonus_line_effect = yes
	}
}

pod_combat_card_clause_separator_effect = {
	custom_tooltip = pod_combat_desc_clause_separator
}

# Effect line, per bonus_effect. change_type and the three stance lines keep the
# pod_combat_desc_bonus_* keys: those four strings read no scope at all, so
# there is nothing in them to rebind.
pod_combat_card_clause_bonus_line_effect = {
	if =      { limit = { scope:pod_clause = { var:bonus_effect = flag:self_heal } }     custom_tooltip = pod_combat_desc_clause_self_heal }
	else_if = { limit = { scope:pod_clause = { var:bonus_effect = flag:heal_target } }   custom_tooltip = pod_combat_desc_clause_heal_target }
	else_if = { limit = { scope:pod_clause = { var:bonus_effect = flag:morale_damage } } custom_tooltip = pod_combat_desc_clause_morale_damage }
	else_if = { limit = { scope:pod_clause = { var:bonus_effect = flag:self_morale_heal } } custom_tooltip = pod_combat_desc_clause_self_morale_heal }
	# Only recognized bonus_param classes form separate typed packets;
	# unknown values join the main packet and must use its description.
	else_if = {
		limit = {
			scope:pod_clause = {
				var:bonus_effect = flag:damage
				trigger_if = {
					limit = { has_variable = bonus_param }
					pod_combat_move_flag_is_damage_class_trigger = { FLAG = var:bonus_param }
				}
				trigger_else = { always = no }
			}
		}
		custom_tooltip = pod_combat_desc_clause_damage_typed
	}
	# Check elemental classes before aggravated: elements render without a wound-box icon.
	# For untyped clauses, read class from the move story.
	else_if = {
		limit = {
			scope:pod_clause = { var:bonus_effect = flag:damage }
			OR = {
				scope:story = { pod_combat_move_class_is_elemental_trigger = yes }
				# A change_class clause retypes the whole action, so the base class
				# printed above is not what lands. Boxless here too, or the card
				# reads "+1 superficial damage" beside "deals fire damage".
				scope:story = {
					any_in_list = {
						variable = bonus_clauses
						var:bonus_effect = flag:change_class
					}
				}
			}
		}
		custom_tooltip = pod_combat_desc_clause_damage_plain
	}
	else_if = {
		limit = {
			scope:pod_clause = { var:bonus_effect = flag:damage }
			scope:story = { pod_combat_class_always_aggravated_trigger = yes }
		}
		custom_tooltip = pod_combat_desc_clause_damage_agg
	}
	else_if = { limit = { scope:pod_clause = { var:bonus_effect = flag:damage } } custom_tooltip = pod_combat_desc_clause_damage }
	else_if = { limit = { scope:pod_clause = { var:bonus_effect = flag:change_type } } custom_tooltip = pod_combat_desc_bonus_change_type }
	else_if = { limit = { scope:pod_clause = { var:bonus_effect = flag:change_class } } custom_tooltip = pod_combat_desc_clause_change_class }
	else_if = { limit = { scope:pod_clause = { var:bonus_effect = flag:hit } }         custom_tooltip = pod_combat_desc_clause_hit }
	# bonus_value is tier 1-3 here. Buff wording follows self/ally targeting;
	# debuffs are restricted to single_enemy.
	else_if = {
		limit = { scope:pod_clause = { var:bonus_effect = flag:hit_buff_round } }
		if = {
			limit = { scope:story = { pod_combat_move_targets_ally_trigger = yes } }
			custom_tooltip = pod_combat_desc_clause_hit_buff_round_other
		}
		else = { custom_tooltip = pod_combat_desc_clause_hit_buff_round }
	}
	else_if = { limit = { scope:pod_clause = { var:bonus_effect = flag:hit_debuff_round } }  custom_tooltip = pod_combat_desc_clause_hit_debuff_round }
	else_if = {
		limit = { scope:pod_clause = { var:bonus_effect = flag:soak_buff_round } }
		if = {
			limit = { scope:story = { pod_combat_move_targets_ally_trigger = yes } }
			custom_tooltip = pod_combat_desc_clause_soak_buff_round_other
		}
		else = { custom_tooltip = pod_combat_desc_clause_soak_buff_round }
	}
	else_if = { limit = { scope:pod_clause = { var:bonus_effect = flag:soak_debuff_round } } custom_tooltip = pod_combat_desc_clause_soak_debuff_round }
	# Rest-of-combat twins of the four above; same self/ally split on the
	# buffs, same enemy-only debuffs.
	else_if = {
		limit = { scope:pod_clause = { var:bonus_effect = flag:hit_buff_combat } }
		if = {
			limit = { scope:story = { pod_combat_move_targets_ally_trigger = yes } }
			custom_tooltip = pod_combat_desc_clause_hit_buff_combat_other
		}
		else = { custom_tooltip = pod_combat_desc_clause_hit_buff_combat }
	}
	else_if = { limit = { scope:pod_clause = { var:bonus_effect = flag:hit_debuff_combat } } custom_tooltip = pod_combat_desc_clause_hit_debuff_combat }
	else_if = {
		limit = { scope:pod_clause = { var:bonus_effect = flag:soak_buff_combat } }
		if = {
			limit = { scope:story = { pod_combat_move_targets_ally_trigger = yes } }
			custom_tooltip = pod_combat_desc_clause_soak_buff_combat_other
		}
		else = { custom_tooltip = pod_combat_desc_clause_soak_buff_combat }
	}
	else_if = { limit = { scope:pod_clause = { var:bonus_effect = flag:soak_debuff_combat } } custom_tooltip = pod_combat_desc_clause_soak_debuff_combat }
	else_if = { limit = { scope:pod_clause = { var:bonus_effect = flag:ap_refund } }     custom_tooltip = pod_combat_desc_clause_ap_refund }
	else_if = { limit = { scope:pod_clause = { var:bonus_effect = flag:stance_guard } }   custom_tooltip = pod_combat_desc_bonus_stance_guard }
	else_if = { limit = { scope:pod_clause = { var:bonus_effect = flag:stance_dodge } }   custom_tooltip = pod_combat_desc_bonus_stance_dodge }
	else_if = { limit = { scope:pod_clause = { var:bonus_effect = flag:stance_stealth } } custom_tooltip = pod_combat_desc_bonus_stance_stealth }
	else_if = { limit = { scope:pod_clause = { var:bonus_effect = flag:ap_drain } }      custom_tooltip = pod_combat_desc_clause_ap_drain }
	else_if = { limit = { scope:pod_clause = { var:bonus_effect = flag:energy_gain } }  custom_tooltip = pod_combat_desc_clause_energy_gain }
	else_if = { limit = { scope:pod_clause = { var:bonus_effect = flag:energy_drain } } custom_tooltip = pod_combat_desc_clause_energy_drain }
	else_if = { limit = { scope:pod_clause = { var:bonus_effect = flag:dot } }          custom_tooltip = pod_combat_desc_clause_dot }
	else_if = { limit = { scope:pod_clause = { var:bonus_effect = flag:morale_dot } }   custom_tooltip = pod_combat_desc_clause_morale_dot }
	else_if = { limit = { scope:pod_clause = { var:bonus_effect = flag:retaliate_boost } } custom_tooltip = pod_combat_desc_clause_retaliate_boost }
	# Conditional weapon swing.
	else_if = {
		limit = { scope:pod_clause = { var:bonus_effect = flag:uses_weapon } }
		if = {
			limit = { pod_combat_actor_is_armed_trigger = yes }
			random_equipped_character_artifact = {
				limit = {
					artifact_slot_type = primary_armament
					prev = { can_benefit_from_artifact = prev }
				}
				save_scope_as = pod_weapon
			}
			if = {
				limit = { scope:pod_weapon = { POD_weapon_artifact_has_override_trigger = yes } }
				custom_tooltip = pod_combat_desc_weapon_empowered
			}
			else = {
				custom_tooltip = pod_combat_desc_uses_weapon_armed
			}
		}
		else = {
			custom_tooltip = pod_combat_desc_uses_weapon
		}
	}
}
