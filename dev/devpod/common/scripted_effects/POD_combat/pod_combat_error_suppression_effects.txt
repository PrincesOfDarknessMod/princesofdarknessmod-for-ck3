##############################################################################################
# POD COMBAT — VALIDATOR ERROR SUPPRESSION
##############################################################################################
# Single home for every "the validator can't see this set/use" workaround
# in the combat module. Called once at game start via POD_combat_on_game_start
# (in pod_combat_on_actions.txt) so the suppression effect itself is "used"
# from a real code path — without that the validator's "Setting it in an
# unused scripted trigger or effect does not count" rule would invalidate
# every suppression below.
#
# Pattern: every block guards on `limit = { always = no }`, so the body
# never executes at runtime. The parser still walks the body when checking
# set/use pairing, so a literal `set_variable` / `add_character_flag` /
# `has_variable` / `this = flag:X` inside an always=no branch counts as
# a set or use.
#
# Three parametric helpers below mirror PoD's
# POD_error_suppression_unused_flag / _unset_flag in
# POD Reference/common/scripted_effects/POD_error_suppression_effects.txt,
# but combat-flavored:
#   - POD_combat_suppress_optional_move_var — literal SET for an optional
#     move-story var whose setter writes parametrically (analyzer can't
#     pair `set_variable = { name = $X$ ... }` with literal reads).
#   - POD_combat_suppress_log_entry_var — literal USE for a log-entry
#     story var consumed only by GUI bindings and customizable_loc
#     dispatchers (both validator-invisible blind spots).
#   - POD_combat_suppress_flag — combined SET + USE for a flag literal
#     that appears as both a variable value (sink SET) and a custom_loc
#     dispatcher selector (USE).
#
# Adding a new combat var / flag / scope alias: add one call to the
# matching helper below, not a per-call site. Per-move parametric
# suppressions (POD_error_suppression_unused_flag with $SKILL_KEY$ /
# $CATEGORY$ / $CONFIG$ / $TARGETING$) stay in POD_init_combat_move
# since they can only resolve in the macro-substituted scope.
##############################################################################################


##############################################################################
# PARAMETRIC HELPERS
##############################################################################

# Provides a literal SET for an optional move-story var. The modular
# setters in pod_combat_move_init_effects.txt write these parametrically
# (`set_variable = { name = $X$ ... }`), which the analyzer can't pair
# with the literal `has_variable = X` reads scattered through the
# execution engine. Calling this once per optional var name gives the
# parser the SET it needs.
#
# Param: VAR — bare variable name (e.g. ap_cost, energy_cost, hp_damage).
POD_combat_suppress_optional_move_var = {
	if = {
		limit = { always = no }
		character:POD_wraith_01 = { set_variable = $VAR$ }
	}
}

# Provides a literal USE for a log-entry story var. These vars are
# written by pod_combat_push_log_effect and consumed by the GUI row
# template (Scope.Var('X').GetValue) or by PODCombatLog* customizable_loc
# dispatchers — neither of which the validator can see. Calling this
# once per log-entry var gives the parser the USE it needs.
#
# Param: VAR — bare variable name (e.g. outcome, attack_result, round).
POD_combat_suppress_log_entry_var = {
	if = {
		limit = {
			always = no
			character:POD_wraith_01 = { has_variable = $VAR$ }
		}
	}
}

# Provides BOTH a SET (via the POD_combat_flag_sink scratch var) and a
# USE (via literal `this = flag:X`) for a flag literal that appears on
# both axes — variable value (sink SET) and customizable_loc dispatcher
# selector (USE). Replaces the prior two-section split where each flag
# had to be listed once in §4 and again in §5.
#
# Param: FLAG — bare flag name (e.g. single_enemy, fire, ATK).
POD_combat_suppress_flag = {
	if = {
		limit = { always = no }
		character:POD_wraith_01 = {
			set_variable = { name = POD_combat_flag_sink value = flag:$FLAG$ }
			remove_variable = POD_combat_flag_sink
		}
	}
	if = {
		limit = {
			always = no
			this = flag:$FLAG$
		}
	}
}
