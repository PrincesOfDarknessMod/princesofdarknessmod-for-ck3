##############################################################################
# PARAMETRIC SCRIPTED EFFECTS
##############################################################################

# $VAR$ = bare variable name. Supplies a literal SET for parametric setters.
pod_combat_suppress_optional_move_var = {
	# The write targets Charon explicitly; caller scope cannot redirect it.
	if = {
		limit = { always = no }
		character:POD_wraith_01 = { set_variable = $VAR$ }
	}
}

# $VAR$ = bare variable name. Supplies a USE for GUI/customizable-localization reads invisible to the validator.
pod_combat_suppress_log_entry_var = {
	if = {
		limit = {
			always = no
			character:POD_wraith_01 = { has_variable = $VAR$ }
		}
	}
}

# $FLAG$ = bare flag name. Supplies both SET and USE.
pod_combat_suppress_flag = {
	if = {
		limit = { always = no }
		character:POD_wraith_01 = {
			set_variable = { name = POD_combat_flag_sink value = flag:$FLAG$ }
			remove_variable = POD_combat_flag_sink
		}
	}
	if = {
		limit = {
			always = no
			this = flag:$FLAG$
		}
	}
}
