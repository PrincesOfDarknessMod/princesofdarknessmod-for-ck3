##############################################################################################
# POD COMBAT — VALIDATOR ERROR SUPPRESSION
##############################################################################################
# Single home for every "the validator can't see this set/use" workaround
# in the combat module. Called once at game start via POD_combat_on_game_start
# (in pod_combat_on_actions.txt) so the suppression effect itself is "used"
# from a real code path — without that the validator's "Setting it in an
# unused scripted trigger or effect does not count" rule would invalidate
# every suppression below.
#
# Pattern: every block guards on `limit = { always = no }`, so the body
# never executes at runtime. The parser still walks the body when checking
# set/use pairing, so a literal `set_variable` / `add_character_flag` /
# `has_variable` / `this = flag:X` inside an always=no branch counts as
# a set or use. Mirrors PoD's own POD_error_suppression_unused_flag /
# _unset_flag helpers in
# POD Reference/common/scripted_effects/POD_error_suppression_effects.txt.
#
# Sections below cover, in order:
#   1. Debug character flags — console-only toggles + desync tripwire.
#   2. Log-entry story_cycle vars — consumed by the GUI row template
#      and the PODCombatLog* custom_loc dispatchers.
#   3. Optional move-story vars — written parametrically by the modular
#      setters in pod_combat_move_init_effects.txt; the analyzer can't
#      pair the `set_variable = { name = $X$ ... }` macro substitution
#      with the literal `has_variable = X` reads in the runtime engine.
#   4. Flag-sink — literal SETs for flag values the runtime only writes
#      via macro substitution (`flag:$CODE$`).
#   5. Flag uses — literal USEs for flag values the runtime only reads
#      via customizable_localization (custom_loc reads don't count per
#      the validator's localization exclusion).
#   6. Event target aliases — saved-scope names the custom_loc dispatcher
#      reads but that are only set in GUI raw_text (GuiScope.AddScope).
#
# Adding a new combat var / flag / scope alias: add it to the matching
# section here, not to a per-call site. Per-move parametric suppressions
# (POD_error_suppression_unused_flag with $SKILL_KEY$ / $CATEGORY$ /
# $CONFIG$ / $TARGETING$) stay in POD_init_combat_move since they can
# only resolve in the macro-substituted scope.
##############################################################################################

POD_combat_suppress_validator_warnings = {

	##############################################################################
	# 1. LOG-ENTRY STORY_CYCLE VARS
	##############################################################################

	# actor_side — set by pod_combat_push_log_effect on each new log
	# entry. Read by pod_combat_window.gui's row template to drive
	# ally / enemy border colour + actor-name tint
	# (Scope.Var('actor_side').GetValue). The validator can't see GUI
	# reads — same blind spot as localization. Provides the USE.
	#
	# The scope here is the host (PC), not a log_entry story, so the
	# has_variable check would always be false at runtime — but the
	# always=no outer guard means it never evaluates. The parser sees
	# the var name and counts it as used.
	if = {
		limit = {
			always = no
			has_variable = actor_side
		}
	}

	# outcome / attack_result / retal_fired — three-axis result vars on
	# each log entry, set by pod_combat_push_log_effect. Read by
	# PODCombatLogResultText (outcome + attack_result) and
	# PODCombatLogRetalRider (retal_fired) custom_loc dispatchers.
	# Customizable_localization reads don't count to the validator.
	#
	# aoe_burst_index — set by push_log when called from inside an AoE
	# iterator (0 = first row of the move's burst; 1+ = subsequent).
	# Read by pod_combat_window.gui's buff-rider visibility check to
	# suppress duplicate buff-rider lines across N per-target rows.
	# GUI Var reads are validator-invisible — same blind spot as
	# actor_side above.
	#
	# follow_up — set on the POD_combat_sc_context story by
	# POD_save_event_as_var, consumed in pod_combat.9999 via
	# `trigger_event = { saved_event_id = scope:sc_ctx.var:follow_up }`.
	# The trigger_event's saved_event_id parameter doesn't count as a use.
	if = {
		limit = { always = no }
		character:POD_wraith_01 = {
			if = { limit = { has_variable = outcome          } }
			if = { limit = { has_variable = attack_result    } }
			if = { limit = { has_variable = retal_fired      } }
			if = { limit = { has_variable = retal_agg        } }
			if = { limit = { has_variable = aoe_burst_index  } }
			if = { limit = { has_variable = system_message   } }
			if = { limit = { has_variable = round            } }
			if = { limit = { has_variable = follow_up        } }
		}
	}

	##############################################################################
	# 2. OPTIONAL MOVE-STORY VARS
	##############################################################################

	# Every optional variable the execution engine or AI dispatcher reads
	# via `has_variable` on a move story. Each one is written by a modular
	# setter in pod_combat_move_init_effects.txt — but only when a CSV
	# move exercises that attribute, so if no current move uses (say)
	# `bypass_dodge`, the analyzer otherwise raises "Variable bypass_dodge
	# is used but is never set." Provides the SET for each.
	#
	# When adding a new POD_combat_move_set_* setter:
	#   - If the setter is exercised by every move, no entry needed.
	#   - If it's optional (most are), add the var name below.
	if = {
		limit = { always = no }
		character:POD_wraith_01 = {
			# Cost vars
			set_variable = ap_cost
			set_variable = energy_cost
			set_variable = gold_cost
			set_variable = stress_cost
			set_variable = piety_cost
			set_variable = prestige_cost
			# Damage / heal vars
			set_variable = hp_damage
			set_variable = morale_damage
			set_variable = self_damage
			set_variable = hp_heal
			set_variable = morale_heal
			set_variable = heal_type
			set_variable = damage_source
			set_variable = damage_type
			# Tier vars (8 tuples)
			set_variable = soak_buff_round_tier
			set_variable = soak_buff_combat_tier
			set_variable = soak_debuff_round_tier
			set_variable = soak_debuff_combat_tier
			set_variable = hit_buff_round_tier
			set_variable = hit_buff_combat_tier
			set_variable = hit_debuff_round_tier
			set_variable = hit_debuff_combat_tier
			# Presence flags
			set_variable = set_guard
			set_variable = dodge
			set_variable = stealth
			set_variable = remove_guard
			set_variable = bypass_dodge
			set_variable = bypass_stealth
			set_variable = retaliate_if_hit
			set_variable = retaliate_if_missed
			set_variable = retaliate_agg
			set_variable = is_starter_move
			set_variable = drain
			# Parametric non-tier
			set_variable = ap_gain
			set_variable = ap_drain
			set_variable = initiative_debuff
			set_variable = retaliate_damage
			# Capability flags — resolved from attack_type defaults +
			# per-move override_* CSV columns. Read via has_variable
			# inside pod_combat_resolve_contact_effect and
			# pod_combat_check_guard_effect.
			set_variable = checks_spirit_gate
			set_variable = requires_hit_roll
			set_variable = checks_guard
			set_variable = checks_stance
			# Rider-condition story vars. heal_condition is set by the
			# generator on drain moves; the other two have setters
			# defined but no CSV row currently calls them.
			set_variable = heal_condition
			set_variable = morale_damage_condition
			set_variable = self_damage_condition
		}
	}

	##############################################################################
	# 3. FLAG LITERALS — SETs (flag-sink)
	##############################################################################

	# Literal flag SETs for values the runtime only writes via macro
	# substitution (`set_variable = { value = flag:$CODE$ }`). The
	# analyzer needs a literal `flag:X` call to pair with each `var:X =
	# flag:Y` read; parameterized substitutions don't count.
	#
	# Categories below match the flag's runtime role:
	#   - targeting:      var:targeting on the move story
	#   - damage source:  var:damage_source on the move story
	#   - damage type:    var:damage_type on the move story
	#   - category:       var:combat_category on the move story
	#   - rider condition: var:heal_condition / morale_damage_condition /
	#                     self_damage_condition on the move story
	#   - outcome axis:   var:pod_combat_last_outcome on the host
	#                     (entry.var:outcome on the log entry)
	#   - attack_result:  var:pod_combat_last_attack_result on the host
	#                     (entry.var:attack_result on the log entry)
	if = {
		limit = { always = no }
		character:POD_wraith_01 = {
			# Targeting
			set_variable = { name = POD_combat_flag_sink value = flag:single_enemy }
			set_variable = { name = POD_combat_flag_sink value = flag:single_ally }
			set_variable = { name = POD_combat_flag_sink value = flag:single_pc }
			set_variable = { name = POD_combat_flag_sink value = flag:self }
			set_variable = { name = POD_combat_flag_sink value = flag:team_enemy }
			set_variable = { name = POD_combat_flag_sink value = flag:team_ally }
			# Damage source (sunlight is a legacy CSV alias the generator
			# canonicalises to "sun" before emit; sinking both keeps the
			# alias support resilient to future changes)
			set_variable = { name = POD_combat_flag_sink value = flag:fire }
			set_variable = { name = POD_combat_flag_sink value = flag:cold }
			set_variable = { name = POD_combat_flag_sink value = flag:lightning }
			set_variable = { name = POD_combat_flag_sink value = flag:sun }
			set_variable = { name = POD_combat_flag_sink value = flag:sunlight }
			set_variable = { name = POD_combat_flag_sink value = flag:silver }
			set_variable = { name = POD_combat_flag_sink value = flag:gold }
			set_variable = { name = POD_combat_flag_sink value = flag:holy }
			set_variable = { name = POD_combat_flag_sink value = flag:balefire }
			# Damage type
			set_variable = { name = POD_combat_flag_sink value = flag:superficial }
			set_variable = { name = POD_combat_flag_sink value = flag:aggravated }
			# Category
			set_variable = { name = POD_combat_flag_sink value = flag:ATK }
			set_variable = { name = POD_combat_flag_sink value = flag:DEF }
			set_variable = { name = POD_combat_flag_sink value = flag:CTL }
			set_variable = { name = POD_combat_flag_sink value = flag:PWR }
			# Rider condition
			set_variable = { name = POD_combat_flag_sink value = flag:always }
			set_variable = { name = POD_combat_flag_sink value = flag:on_hp_damage }
			set_variable = { name = POD_combat_flag_sink value = flag:on_contact }
			set_variable = { name = POD_combat_flag_sink value = flag:on_miss }
			# Outcome axis
			set_variable = { name = POD_combat_flag_sink value = flag:attack }
			set_variable = { name = POD_combat_flag_sink value = flag:heal }
			set_variable = { name = POD_combat_flag_sink value = flag:morale_damage }
			set_variable = { name = POD_combat_flag_sink value = flag:morale_healed }
			set_variable = { name = POD_combat_flag_sink value = flag:utility }
			# Attack result axis
			set_variable = { name = POD_combat_flag_sink value = flag:hp_damage }
			set_variable = { name = POD_combat_flag_sink value = flag:soaked }
			set_variable = { name = POD_combat_flag_sink value = flag:missed }
			set_variable = { name = POD_combat_flag_sink value = flag:dodged }
			set_variable = { name = POD_combat_flag_sink value = flag:stealthed }
			set_variable = { name = POD_combat_flag_sink value = flag:guard_blocked }
			# System-message bookend flags — emitted by
			# pod_combat_push_system_log_effect, read by the
			# PODCombatLogSystemMessageText custom_loc dispatcher.
			set_variable = { name = POD_combat_flag_sink value = flag:combat_started }
			set_variable = { name = POD_combat_flag_sink value = flag:combat_ended_victory }
			set_variable = { name = POD_combat_flag_sink value = flag:combat_ended_defeat }
			set_variable = { name = POD_combat_flag_sink value = flag:round_begins }
			remove_variable = POD_combat_flag_sink
		}
	}

	##############################################################################
	# 4. FLAG LITERALS — USEs
	##############################################################################

	# Literal flag USEs for values the runtime only reads via
	# customizable_localization (entry.var:outcome = flag:X branches in
	# PODCombatLogResultText, etc.). The validator explicitly excludes
	# "use in localization" from set/use pairing, so the literal `this =
	# flag:X` pattern in an always=no branch provides the needed USE.
	#
	# Flags that ARE read script-side (e.g. flag:hp_damage in the
	# partition triggers, flag:attack in the rider gate) don't strictly
	# need a USE entry here — but listing every sunk flag keeps the
	# section symmetric with section 4 and ensures no future refactor
	# that drops a script-side reader silently triggers a warning.
	if = {
		limit = {
			always = no
			OR = {
				# Targeting
				this = flag:single_enemy
				this = flag:single_ally
				this = flag:single_pc
				this = flag:self
				this = flag:team_enemy
				this = flag:team_ally
				# Damage source
				this = flag:fire
				this = flag:cold
				this = flag:lightning
				this = flag:sun
				this = flag:sunlight
				this = flag:silver
				this = flag:gold
				this = flag:holy
				this = flag:balefire
				# Damage type
				this = flag:superficial
				this = flag:aggravated
				# Category
				this = flag:ATK
				this = flag:DEF
				this = flag:CTL
				this = flag:PWR
				# Rider condition
				this = flag:always
				this = flag:on_hp_damage
				this = flag:on_contact
				this = flag:on_miss
				# Outcome axis
				this = flag:attack
				this = flag:heal
				this = flag:morale_damage
				this = flag:morale_healed
				this = flag:utility
				# Attack result axis
				this = flag:hp_damage
				this = flag:soaked
				this = flag:missed
				this = flag:dodged
				this = flag:stealthed
				this = flag:guard_blocked
				# System-message bookend flags
				this = flag:combat_started
				this = flag:combat_ended_victory
				this = flag:combat_ended_defeat
				this = flag:round_begins
			}
		}
	}

	##############################################################################
	# 5. EVENT TARGET ALIASES
	##############################################################################

	# Saved-scope names the custom_loc dispatchers read (scope:entry /
	# scope:move) but that are only set in GUI raw_text via
	# `GuiScope.AddScope('entry', ...).AddScope('move', ...)` — outside
	# any scripted-effect, so the analyzer never sees the writes.
	# `save_scope_as` inside an always=no branch provides the SET.
	if = {
		limit = { always = no }
		save_scope_as = entry
		save_scope_as = move
	}
}
