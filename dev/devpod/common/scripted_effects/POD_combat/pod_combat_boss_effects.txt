######################################################
###### SHARED BOSS SETUP
######################################################

# Character scope. TIER is an integer from 1 to 3
POD_combat_initialize_boss_effect = {
	add_character_flag = pod_combat_boss
	set_variable = { name = pod_combat_boss_tier value = $TIER$ }
}

# Character scope. MOVE is the move's DECISION key, which is also its registry variable on
# Charon. Grant at spawn. A missing registry variable means the move
# is not in this save's catalogue (POD_current_combat_version not bumped) or the key is misspelled.
POD_combat_boss_grant_move_effect = {
	if = {
		limit = { exists = character:POD_wraith_01.var:$MOVE$ }
		add_to_variable_list = {
			name = pod_combat_boss_moves
			target = character:POD_wraith_01.var:$MOVE$
		}
	}
	else = {
		debug_log = "POD_combat_boss_grant_move_effect: $MOVE$ is not in the combat catalogue; bump POD_current_combat_version or check the DECISION key"
	}
}

# Character scope. The shared tail of boss creation reached from the on_character_created hook.
POD_combat_boss_finish_creation_effect = {
	if = {
		limit = { NOT = { has_character_flag = pod_combat_boss_creation_finished } }
		add_character_flag = pod_combat_boss_creation_finished
		# Splat-generic despite the name: it grants the trees for every *advanced trait and sets its own flag.
		if = {
			limit = { NOT = { has_character_flag = had_POD_character_lifestyle } }
			POD_character_vampire_lifestyle_initialization_effect = yes
		}
		POD_initialize_energy_modifier = yes
		if = {
			limit = { has_masquerade_risk_trigger = no }
			POD_character_masquerade_initialization_effect = yes
		}
		if = {
			limit = { POD_is_vampire_trigger = yes }
			POD_set_resonance_effect = yes
		}
		add_character_flag = had_POD_character_initialization
	}
}

# Character scope. POD_has_infinite_blood_trigger accepts the modifier. PoD has no such rule for other splats; 
# they are logged and unchanged.
POD_combat_boss_grant_unlimited_energy_effect = {
	if = {
		limit = { POD_splat_has_hunger_exclusively = yes }
		add_character_modifier = { modifier = pod_combat_boss_unlimited_energy_modifier }
	}
	else = {
		debug_log = "POD_combat_boss_grant_unlimited_energy_effect: PoD has an infinite-energy rule for hunger splats only; nothing granted"
	}
}

# Host scope.
POD_combat_boss_launch_failed_effect = {
	send_interface_toast = {
		title = pod_combat_boss_launch_unavailable
		left_icon = this
	}
}

######################################################
###### TEMPORARY BOSS ENCOUNTER LIFECYCLE
######################################################

# Player scope.
# TEMPLATE initializes the new character before seating.
# TITLE, TEAM_NAME and OUTPUT_EVENT are required localization/event keys.
# MUSIC is a track key from music/in_game; it is cued on the host once the fight is finalized.
# INCLUDE_COTERIE is yes/no; eligible companions are seated before finalization.
# DESPAWN is yes/no: yes registers the boss with combat cleanup, which removes it after the fight;
# ESCORT_COUNT is number of escorts to the boss
POD_combat_launch_boss_effect = {
	POD_combat_launch_boss_roster_effect = {
		TEMPLATE = $TEMPLATE$
		BUILD = POD_combat_boss_build_none_effect
		BUILD_TIER = 0
		TITLE = $TITLE$
		TEAM_NAME = $TEAM_NAME$
		FATALITY = $FATALITY$
		OUTPUT_EVENT = $OUTPUT_EVENT$
		INCLUDE_COTERIE = $INCLUDE_COTERIE$
		DESPAWN = $DESPAWN$
		MUSIC = $MUSIC$
		ESCORT_TEMPLATE = $TEMPLATE$
		ESCORT_COUNT = 0
		ESCORT_BUILD = POD_combat_boss_build_none_effect
		ESCORT_TIER = 0
	}
}

POD_combat_boss_create_roster_member_effect = {
	if = {
		limit = { exists = scope:pod_combat_boss_new_spawn }
		clear_saved_scope = pod_combat_boss_new_spawn
	}
	create_character = {
		template = $TEMPLATE$
		location = scope:pod_combat_boss_host.location
		save_scope_as = pod_combat_boss_new_spawn
	}
	hidden_effect_new_object = {
		if = {
			limit = {
				exists = scope:pod_combat_boss_new_spawn
				scope:pod_combat_boss_new_spawn ?= { is_alive = yes }
			}
			scope:pod_combat_boss_host = {
				add_to_variable_list = {
					name = pod_combat_boss_pending_roster
					target = scope:pod_combat_boss_new_spawn
				}
			}
			scope:pod_combat_boss_new_spawn = {
				if = {
					limit = {
						has_character_flag = pod_combat_boss
						NOT = { has_character_flag = pod_combat_boss_creation_finished }
					}
					debug_log = pod_combat_boss_hook_late_log
					POD_combat_boss_finish_creation_effect = yes
				}
				$BUILD$ = { TIER = $TIER$ }
			}
		}
		else = { save_scope_value_as = { name = pod_combat_boss_roster_complete value = no } }
	}
}

# Player host scope.
POD_combat_launch_boss_roster_effect = {
	save_scope_value_as = { name = pod_combat_boss_roster_size value = { value = $ESCORT_COUNT$ add = 1 } }
	if = {
		limit = {
			pod_combat_boss_launch_ready_trigger = yes
			scope:pod_combat_boss_roster_size >= 1
			scope:pod_combat_boss_roster_size <= pod_combat_max_team_size
		}
		# Template grants store move-story references; reload before creating any boss.
		hidden_effect = {
			if = {
				limit = {
					OR = {
						NOT = { exists = global_var:POD_combat_version }
						global_var:POD_combat_version != POD_current_combat_version
					}
					NOT = { pod_combat_any_fight_active_trigger = yes }
				}
				POD_reload_combat_moves = yes
			}
		}
		save_scope_as = pod_combat_boss_host
		save_scope_value_as = { name = pod_combat_boss_roster_complete value = yes }
		save_scope_value_as = { name = pod_combat_boss_roster_started value = no }
		clear_variable_list = pod_combat_boss_pending_roster
		if = {
			limit = { exists = scope:pod_combat_boss_spawn }
			clear_saved_scope = pod_combat_boss_spawn
		}
		POD_combat_boss_create_roster_member_effect = { TEMPLATE = $TEMPLATE$ BUILD = $BUILD$ TIER = $BUILD_TIER$ }
		hidden_effect_new_object = {
			scope:pod_combat_boss_new_spawn ?= { save_scope_as = pod_combat_boss_spawn }
			while = {
				count = $ESCORT_COUNT$
				POD_combat_boss_create_roster_member_effect = { TEMPLATE = $ESCORT_TEMPLATE$ BUILD = $ESCORT_BUILD$ TIER = $ESCORT_TIER$ }
			}
			if = {
				limit = {
					scope:pod_combat_boss_roster_complete = yes
					exists = scope:pod_combat_boss_spawn
					any_in_list = {
						variable = pod_combat_boss_pending_roster
						count = all
						is_alive = yes
					}
				}
				POD_combat_create_duel_effect = {
					INITIATOR = scope:pod_combat_boss_host
					ATTACKER = scope:pod_combat_boss_host
					DEFENDER = scope:pod_combat_boss_spawn
					FATALITY = $FATALITY$
					FIXED = no
					LOCALE = terrain_scope
					OUTPUT_EVENT = $OUTPUT_EVENT$
				}
				if = {
					limit = {
						has_variable = pod_combat_session
						exists = scope:pod_combat_duel
						scope:pod_combat_duel ?= { has_variable = pod_combat_enemy_team_size }
					}
					if = {
						limit = { always = $DESPAWN$ }
						every_in_list = {
							variable = pod_combat_boss_pending_roster
							save_scope_as = pod_combat_boss_roster_member
							scope:pod_combat_boss_host.var:pod_combat_session = {
								add_to_variable_list = {
									name = pod_combat_summoned_chars
									target = scope:pod_combat_boss_roster_member
								}
							}
						}
					}
					POD_combat_duel_add_enemy_effect = { CHARACTER = scope:pod_combat_boss_spawn }
					every_in_list = {
						variable = pod_combat_boss_pending_roster
						limit = { NOT = { this = scope:pod_combat_boss_spawn } }
						save_scope_as = pod_combat_boss_roster_member
						scope:pod_combat_boss_host = {
							POD_combat_duel_add_enemy_effect = { CHARACTER = scope:pod_combat_boss_roster_member }
						}
					}
					if = {
						limit = {
							any_in_list = {
								variable = pod_combat_boss_pending_roster
								count = all
								pod_combat_is_seated_trigger = yes
							}
						}
						if = {
							limit = { always = $INCLUDE_COTERIE$ }
							scope:pod_combat_boss_host.var:coterie ?= {
								every_in_list = {
									variable = members
									limit = { pod_combat_boss_ally_available_trigger = { LEADER = scope:pod_combat_boss_host } }
									save_scope_as = pod_combat_boss_ally
									scope:pod_combat_boss_host = {
										POD_combat_duel_add_ally_effect = { CHARACTER = scope:pod_combat_boss_ally }
									}
								}
							}
						}
						POD_combat_set_custom_title = { LOC = $TITLE$ }
						POD_combat_set_enemy_team_name = { LOC = $TEAM_NAME$ }
						POD_combat_finalize_duel_effect = yes
						save_scope_value_as = { name = pod_combat_boss_roster_started value = yes }
						play_music_cue = $MUSIC$
					}
				}
			}
			if = {
				limit = { scope:pod_combat_boss_roster_started = no }
				if = {
					limit = { has_variable = pod_combat_session }

					var:pod_combat_session = {
						every_in_list = {
							variable = pod_combat_active_slots
							var:slot_char ?= {
								if = {
									limit = { is_alive = yes }
									if = {
										limit = { NOT = { this = scope:pod_combat_boss_host } }
										remove_variable = pod_combat_session
									}
									remove_variable = pod_combat_my_control_slot
								}
							}
						}
						clear_variable_list = pod_combat_summoned_chars
					}
					pod_combat_cleanup_effect = yes
				}
				every_in_list = {
					variable = pod_combat_boss_pending_roster
					limit = {
						is_alive = yes
						NOT = { has_variable = pod_combat_session }
					}
					death = { death_reason = death_disappearance }
				}
				POD_combat_boss_launch_failed_effect = yes
			}
			clear_variable_list = pod_combat_boss_pending_roster
		}
	}
	else = { POD_combat_boss_launch_failed_effect = yes }
}
