	set_variable = { name = pod_combat_equipped_guard value = yes }
	set_variable = { name = pod_combat_equipped_feint value = yes }
	set_variable = { name = pod_combat_equipped_intimidate value = yes }
	set_variable = { name = pod_combat_equipped_rally value = yes }
	set_variable = { name = pod_combat_equipped_quick_jab value = yes }
	set_variable = { name = pod_combat_equipped_reckless_charge value = yes }
}
