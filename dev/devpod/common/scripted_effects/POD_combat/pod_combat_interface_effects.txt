# effects for loc flavorization, animation state inputs, etc

# sets a custom name for the combat window (to replace "COMBAT")
# assumes that scope:pod_combat_duel is present (e.g. after using POD_combat_create_duel_effect)
# arguments:
# - LOC: a flag name (used as the localization key)
POD_combat_set_custom_title = {
	hidden_effect_new_object = {
		scope:pod_combat_duel ?= {
			set_variable = {
				name = custom_title
				value = flag:$LOC$
			}
		}
	}
	POD_error_suppression_unused_flag = { FLAG = $LOC$ }
}
