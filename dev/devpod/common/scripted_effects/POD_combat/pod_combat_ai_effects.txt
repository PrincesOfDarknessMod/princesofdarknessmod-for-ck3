##########################################
######   POD COMBAT AI EFFECTS   ######
##########################################

# All combat modes use the same workspace bridge and the generic
# pod_combat_ai_select_and_execute_move_effect below. A 1v1 fight is
# a degenerate NvN with slots 0 and 5 populated; the same
# workspace-relative move effects handle both.
#
# AI model — one combatant's turn:
#   1. Turn start: pick an initial enemy target via
#      pod_combat_ai_select_{enemy,player}_target_effect (weighted
#      random favoring low-HP). Sets pod_combat_selected_target.
#   2. Up to 6 action attempts per turn (a hard cap, not strictly
#      one per AP — moves can refund AP via ap_gain, but the loop
#      caps at 6 unrolled calls regardless). Each attempt:
#        a. refresh_target_if_dead — re-pick if previous action KO'd
#           the original target; no-op when still alive.
#        b. select_and_execute_move — random_in_list over the actor's
#           loadout, weighted by base ai_weight + situational modifiers
#           (anti-repetition, energy attrition, target HP / morale
#           finish bonuses, hit-chance penalties, round-buff anti-spam
#           and late-turn hit-buff penalties). Moves failing
#           POD_can_execute_combat_move are filtered out.
#        c. swap_for_story — flip workspace to single_ally / single_pc
#           slot if the move's targeting demands; restore after.
#        d. save_workspace — propagate mutations and run victory check
#           (early exit if combat ended this action).


# ─────────────────────────────────────────────────────────────
# AI TUNING CONSTANTS — edit these to tweak AI behaviour
# ─────────────────────────────────────────────────────────────
# CK3 `@name = value` parse-time numeric literals, referenced
# from the weight block in
# pod_combat_ai_select_and_execute_move_effect below.

# Anti-repetition (three-deep history).
@ai_repeat_last         = -35
@ai_repeat_prev         = -20
@ai_repeat_prev2        = -12

# Need-based bonuses.
@ai_heal_needed         = 30   # hp_heal when applicable hp track has room
@ai_morale_heal_low     = 30   # morale_heal when morale below threshold
@ai_defense_when_hurt   = 15   # guard / dodge / stealth / soak-buff while hp < max
@ai_hitbuff_low_acc     = 15   # hit_buff while hit_chance ≤ threshold

# Round-buff lifecycle penalties.
@ai_round_buff_spam     = -8   # generic round-duration buff (any stat)
@ai_round_buff_lateap   = -20  # round hit-buff specifically, low AP remaining

# Remove-guard bias.
@ai_remove_guard_hit    = 15   # target IS guarding
@ai_remove_guard_miss   = -15  # target NOT guarding (strip would no-op)

# Target-aware offensive bonuses (single_enemy moves only).
@ai_bypass_dodge        = 15   # bypass_dodge vs dodging target
@ai_bypass_stealth      = 15   # bypass_stealth vs stealthed target
@ai_cash_in_stealth     = 20   # actor stealthed → spend the +30 hit window
@ai_aggravated_vs_soak  = 20   # high-soak target → bias aggravated
@ai_weakness_exploit    = 25   # damage_source matches target splat weakness

# Finish bonuses.
@ai_hp_finish_1box      = 25   # target near death, any damage finishes
@ai_hp_finish_2box      = 15   # target hurt, move deals ≥ ai_thresh_finish_damage
@ai_mor_finish_low      = 25   # target morale low, any morale damage finishes
@ai_mor_finish_mid      = 15   # target morale mid, move deals ≥ ai_thresh_finish_morale

# Hit-chance penalties (requires_hit_roll moves only).
@ai_hit_penalty_low     = -25  # hit_chance ≤ ai_thresh_hit_low
@ai_hit_penalty_mid     = -15  # hit_chance ≤ ai_thresh_hit_mid

# Energy attrition (stacking per step of actor_energy_used).
@ai_energy_step1        = -8
@ai_energy_step2        = -10
@ai_energy_step3        = -12
@ai_energy_step4        = -14

# Class / repertoire bias.
@ai_trait_gated         = 10   # non-starter move (perk/trait/splat-gated)

# Final-weight floor (random_in_list ignores items with weight ≤ 0).
@ai_weight_floor        = 1

# Comparison thresholds read by the modifier `limit`s.
@ai_thresh_morale_shaken   = 10   # morale-heal trigger
@ai_thresh_hit_low         = 30   # hit_penalty_low fires at-or-below
@ai_thresh_hit_mid         = 40   # hit_penalty_mid fires at-or-below
@ai_thresh_hit_buff_band   = 50   # hitbuff_low_acc fires at-or-below
@ai_thresh_high_soak       = 70   # aggravated_vs_soak fires at-or-above
@ai_thresh_late_ap         = 5    # round_buff_lateap fires at-or-below (base 6 − 1)
@ai_thresh_hp_finish_1     = 1    # empty_damage_boxes ≤ this → hp_finish_1box fires
@ai_thresh_hp_finish_2     = 2    # empty_damage_boxes ≤ this → hp_finish_2box fires
@ai_thresh_finish_damage   = 2    # hp_finish_2box also requires hp_damage ≥ this
@ai_thresh_mor_finish_low  = 2    # target morale ≤ this → mor_finish_low fires
@ai_thresh_mor_finish_mid  = 4    # target morale ≤ this → mor_finish_mid fires
@ai_thresh_finish_morale   = 2    # mor_finish_mid also requires morale_damage ≥ this


pod_combat_ai_refresh_target_if_dead_effect = {
	# Re-pick the AI's selected_target if the slot it currently points
	# at isn't alive anymore (most often: the previous try_action's
	# move just KO'd the original target and pod_combat_save_workspace_effect
	# flipped its _alive flag in the same try). When the current
	# target is still alive this no-ops and keeps the AI on its
	# original mark, so we don't thrash targeting between every
	# action of a multi-action turn.
	#
	# pod_combat_current_actor < 5 → AI is on the player team
	# (slots 0-4) and looks for an enemy in 5-9. >= 5 inverts. Same
	# split pod_combat_execute_ai_turn_effect uses at turn start.
	#
	# The "current target is alive" check is an any_in_list: if SOME
	# active slot's slot_index matches selected_target AND that slot
	# is alive, the target is fine and we no-op. NOT(any) = "no live
	# match found" → re-pick.
	save_scope_as = host
	if = {
		limit = {
			has_variable = pod_combat_selected_target
			NOT = {
				any_in_list = {
					variable = pod_combat_active_slots
					var:slot_index = scope:host.var:pod_combat_selected_target
					var:slot_alive = 1
				}
			}
		}
		# Re-pick from the appropriate team's live slots and reload
		# the workspace so the new target's data is in opponent_*.
		if = {
			limit = { has_variable = pod_combat_current_actor var:pod_combat_current_actor <= 4 }
			pod_combat_ai_select_enemy_target_effect = yes
		}
		else = {
			pod_combat_ai_select_player_target_effect = yes
		}
		pod_combat_load_workspace_effect = yes
	}
}


pod_combat_ai_take_turn_effect = {
	# Hard cap of 6 action attempts per turn — six unrolled try_action
	# calls, each gated by `pod_combat_current_ap >= 1`. AP refunds
	# via ap_gain CAN raise current AP mid-turn, but they can't
	# unlock a seventh attempt. Each call is a no-op when AP < 1 or
	# combat is over.
	pod_combat_ai_try_action_effect = yes
	pod_combat_ai_try_action_effect = yes
	pod_combat_ai_try_action_effect = yes
	pod_combat_ai_try_action_effect = yes
	pod_combat_ai_try_action_effect = yes
	pod_combat_ai_try_action_effect = yes
}

pod_combat_ai_try_action_effect = {
	if = {
		limit = {
			pod_combat_current_ap_value >= 1
			NOT = { pod_combat_is_over_trigger = yes }
		}
		# Refresh the AI's selected_target if the previous try killed
		# whoever was originally chosen. Without this, the second/third
		# action of a multi-action turn keeps firing single-enemy moves
		# at a corpse. The helper is a no-op when the current target
		# is still alive — so this only thrashes targeting when it has
		# to.
		pod_combat_ai_refresh_target_if_dead_effect = yes

		pod_combat_ai_select_and_execute_move_effect = yes

		# Persist this action's mutations to the slots and run the
		# victory check inside save_workspace_effect. Two consequences
		# the post-loop save in pod_combat_execute_ai_turn_effect
		# couldn't deliver on its own:
		#   1. Slot _alive flags get updated immediately, so the next
		#      try_action's AI swap effects + selector see the KO'd
		#      slot as dead instead of attacking a corpse.
		#   2. pod_combat_check_group_victory_effect fires (it's
		#      chained at the tail of save_workspace_effect), so if
		#      this hit ended combat the next try's
		#      `NOT = { pod_combat_is_over_trigger = yes }` gate
		#      short-circuits and the remaining try_action calls
		#      no-op.
		pod_combat_save_workspace_effect = yes
	}
}

pod_combat_ai_select_and_execute_move_effect = {
	# AI picks a move from its own pod_combat_loadout list using
	# random_in_list — selection is proportional to the move's
	# computed weight (story's ai_weight + situational modifiers).
	# Higher-weight moves win more often but lower-weight moves can
	# still come up. 
	#
	# The move's trait/perk gate is evaluated through the generic
	# POD_can_execute_combat_move trigger in the limit clause.
	#
	# After selection, POD_combat_ai_swap_for_story dispatches to the
	# matching target-swap effect, POD_execute_combat_move runs the
	# body, and POD_combat_ai_restore_after_move puts the turn-start
	# target back.

	# Seed a default loadout for AI NPCs that never opened the configure
	# window. pod_combat_actor_char is the character scope for the acting
	# combatant (written by pod_combat_group_effects.txt before this fires).
	pod_combat_ai_ensure_loadout_effect = yes

	if = {
		limit = { has_variable = pod_combat_actor_char }

		# Two scopes matter in this effect:
		#   scope:combat_host  — the PC character where combat state and
		#                        the workspace mirror vars live
		#                        (pod_combat_current_ap,
		#                        pod_combat_actor_*, pod_combat_target_*,
		#                        loaded from the acting slot each turn;
		#                        HP itself derives live from the slot
		#                        character's PoD wound vars, not cached
		#                        on the workspace). Required for
		#                        POD_execute_combat_move and all
		#                        affordability checks.
		#   scope:ai_actor     — the acting AI character, where the
		#                        AI's pod_combat_loadout list and
		#                        pod_combat_ai_last_move live.
		#                        Required for list iteration and
		#                        anti-repetition tracking.
		save_scope_as = combat_host

		var:pod_combat_actor_char = {
			save_scope_as = ai_actor

			# Iteration scope = story. The trigger needs named scope
			# references for both story and actor so its internal trait
			# check ($ACTOR$ = { ... }) and its workspace checks can't
			# get confused about which `prev` each piece refers to.
			# save_temporary_scope_as inside the limit binds the
			# current iteration scope (the story) so the trigger can
			# still reach it after switching into scope:combat_host.
			random_in_list = {
				variable = pod_combat_loadout
				limit = {
					save_temporary_scope_as = ai_candidate_move
					scope:combat_host = {
						POD_can_execute_combat_move = {
							STORY = scope:ai_candidate_move
							ACTOR = scope:ai_actor
						}

						# Redundancy guard — skip moves whose buff /
						# debuff is already at or above the level it
						# would apply, or whose stance flag is
						# already active. Without this the AI
						# happily re-applies the same Toughness or
						# Vanish every turn, burning AP for no
						# state change. Currently no damaging move
						# carries a tier or stance rider, so a
						# blanket exclude on pure buff/debuff/stance
						# moves doesn't strand any partially-useful
						# action. Tier checks use strict `<` so a
						# Tier-2 buff can still upgrade a Tier-1
						# stack but a Tier-1 buff can't downgrade a
						# Tier-2.
						#
						# All checks read against the workspace
						# (pod_combat_actor_* = acting AI,
						# pod_combat_target_* = selected target).
						# Self-targeted buffs go through
						# pod_combat_actor_*; target-side debuffs
						# through pod_combat_target_*.
						trigger_if = {
							limit = {
								scope:ai_candidate_move = { has_variable = soak_buff_round_tier }
								has_variable = pod_combat_actor_soak_buff_round_tier
							}
							var:pod_combat_actor_soak_buff_round_tier < scope:ai_candidate_move.var:soak_buff_round_tier
						}
						trigger_if = {
							limit = {
								scope:ai_candidate_move = { has_variable = soak_buff_combat_tier }
								has_variable = pod_combat_actor_soak_buff_combat_tier
							}
							var:pod_combat_actor_soak_buff_combat_tier < scope:ai_candidate_move.var:soak_buff_combat_tier
						}
						# Self hit tier buffs
						trigger_if = {
							limit = {
								scope:ai_candidate_move = { has_variable = hit_buff_round_tier }
								has_variable = pod_combat_actor_hit_buff_round_tier
							}
							var:pod_combat_actor_hit_buff_round_tier < scope:ai_candidate_move.var:hit_buff_round_tier
						}
						trigger_if = {
							limit = {
								scope:ai_candidate_move = { has_variable = hit_buff_combat_tier }
								has_variable = pod_combat_actor_hit_buff_combat_tier
							}
							var:pod_combat_actor_hit_buff_combat_tier < scope:ai_candidate_move.var:hit_buff_combat_tier
						}
						# Target soak tier debuffs
						trigger_if = {
							limit = {
								scope:ai_candidate_move = { has_variable = soak_debuff_round_tier }
								has_variable = pod_combat_target_soak_debuff_round_tier
							}
							var:pod_combat_target_soak_debuff_round_tier < scope:ai_candidate_move.var:soak_debuff_round_tier
						}
						trigger_if = {
							limit = {
								scope:ai_candidate_move = { has_variable = soak_debuff_combat_tier }
								has_variable = pod_combat_target_soak_debuff_combat_tier
							}
							var:pod_combat_target_soak_debuff_combat_tier < scope:ai_candidate_move.var:soak_debuff_combat_tier
						}
						# Target hit tier debuffs
						trigger_if = {
							limit = {
								scope:ai_candidate_move = { has_variable = hit_debuff_round_tier }
								has_variable = pod_combat_target_hit_debuff_round_tier
							}
							var:pod_combat_target_hit_debuff_round_tier < scope:ai_candidate_move.var:hit_debuff_round_tier
						}
						trigger_if = {
							limit = {
								scope:ai_candidate_move = { has_variable = hit_debuff_combat_tier }
								has_variable = pod_combat_target_hit_debuff_combat_tier
							}
							var:pod_combat_target_hit_debuff_combat_tier < scope:ai_candidate_move.var:hit_debuff_combat_tier
						}
						# Pure-heal redundancy — exclude heal-only moves
						# whose effects are wasted on the actor's current
						# state. A move is "heal-only" when it has hp_heal
						# or morale_heal but neither hp_damage nor
						# morale_damage; lifesteal-style hybrids stay
						# eligible because their damage portion still earns
						# the AP. The keep-condition mirrors the +30
						# heal-when-hurt weight modifiers below — superficial
						# heals need room below sup_ceiling, aggravated heals
						# need room below max_hp, morale heals need morale
						# below cap. If NONE of the move's heals can do
						# anything useful, drop the candidate before
						# weighting. Without this gate the AI sometimes
						# burned a turn casting a 3-AP regen on itself at
						# full HP — wasted AP and a confusing log entry.
						trigger_if = {
							limit = {
								scope:ai_candidate_move = {
									OR = {
										has_variable = hp_heal
										has_variable = morale_heal
									}
									NOT = { has_variable = hp_damage }
									NOT = { has_variable = morale_damage }
								}
							}
							OR = {
								# Superficial hp_heal — useful below
								# sup_ceiling (= max_hp − aggravated_boxes;
								# aggravated wounds pin the top of the
								# track, capping a superficial heal's
								# usefulness even when hp < max_hp).
								AND = {
									scope:ai_candidate_move = {
										has_variable = hp_heal
										var:heal_type = flag:superficial
									}
									pod_combat_actor_hp_value < pod_combat_actor_sup_ceiling_value
								}
								# Aggravated hp_heal — useful for any wound,
								# capped at max_hp. PoD's aggravated heal
								# effect heals aggravated first and spills
								# into superficial when no aggravated remain.
								AND = {
									scope:ai_candidate_move = {
										has_variable = hp_heal
										var:heal_type = flag:aggravated
									}
									pod_combat_actor_hp_value < pod_combat_actor_max_hp_value
								}
								# Morale heal — useful below cap.
								AND = {
									scope:ai_candidate_move = { has_variable = morale_heal }
									var:pod_combat_actor_morale < pod_combat_max_morale
								}
							}
						}
						# Self stance flags (presence-only, no tier).
						# NAND, NOT NOT — CK3's NOT = { A B } is NOR (all
						# must fail), so a NOT block listing both
						# conditions excludes the move whenever EITHER is
						# true. We want "exclude only when BOTH are true"
						# — that's NAND (at least one false). Earlier
						# revisions used NOT here, which silently
						# blacklisted every stance move and, when the
						# actor was already in a stance, every non-stance
						# move too — the AI's loadout returned nothing.
						NAND = {
							scope:ai_candidate_move = { has_variable = set_guard }
							has_variable = pod_combat_actor_is_guarding
						}
						NAND = {
							scope:ai_candidate_move = { has_variable = dodge }
							has_variable = pod_combat_actor_is_dodging
						}
						NAND = {
							scope:ai_candidate_move = { has_variable = stealth }
							has_variable = pod_combat_actor_is_stealthed
						}
						NAND = {
							scope:ai_candidate_move = { has_variable = retaliate_if_hit }
							has_variable = pod_combat_actor_retaliate_hit
						}
						NAND = {
							scope:ai_candidate_move = { has_variable = retaliate_if_missed }
							has_variable = pod_combat_actor_retaliate_miss
						}
					}
				}

				weight = {
					base = 0

					# Per-move base weight (8-15).
					modifier = {
						add = var:ai_weight
					}

					# Anti-repetition: three-deep history with
					# diminishing penalty. Tracking only the most
					# recent move caused the AI to oscillate between
					# two good moves — picking A penalised A, so B
					# won, which unpenalised A, so A won again, ad
					# infinitum. Three slots force the AI to look
					# further down its list before repeating.
					modifier = {
						add = @ai_repeat_last
						scope:ai_actor = { has_variable = pod_combat_ai_last_move }
						this = scope:ai_actor.var:pod_combat_ai_last_move
					}
					modifier = {
						add = @ai_repeat_prev
						scope:ai_actor = { has_variable = pod_combat_ai_prev_move }
						this = scope:ai_actor.var:pod_combat_ai_prev_move
					}
					modifier = {
						add = @ai_repeat_prev2
						scope:ai_actor = { has_variable = pod_combat_ai_prev2_move }
						this = scope:ai_actor.var:pod_combat_ai_prev2_move
					}

					# Heal when hurt — workspace on combat_host. Split by
					# heal_type so the +30 only fires when the move can
					# actually restore damage on the actor:
					#   flag:superficial — useful only when there's room
					#     below pod_combat_actor_sup_ceiling_value
					#     (= max_hp − aggravated_boxes). Aggravated wounds
					#     pin the top of the track, so a superficial heal
					#     whose ceiling has already been reached is a
					#     no-op even if hp < max_hp (wounds are all
					#     aggravated).
					#   flag:aggravated  — useful for any wound. PoD's
					#     aggravated heal effect heals aggravated first
					#     and spills into superficial when no aggravated
					#     remain, so hp < max_hp is the right gate.
					# A heal move with hp_heal but no heal_type set
					# triggers neither branch and stays at base weight,
					# which is the safe failure mode.
					modifier = {
						add = @ai_heal_needed
						has_variable = hp_heal
						has_variable = heal_type
						var:heal_type = flag:superficial
						scope:combat_host = {
							pod_combat_actor_hp_value < pod_combat_actor_sup_ceiling_value
						}
					}
					modifier = {
						add = @ai_heal_needed
						has_variable = hp_heal
						has_variable = heal_type
						var:heal_type = flag:aggravated
						scope:combat_host = {
							pod_combat_actor_hp_value < pod_combat_actor_max_hp_value
						}
					}
					# Morale heal when shaken.
					modifier = {
						add = @ai_morale_heal_low
						has_variable = morale_heal
						scope:combat_host = { var:pod_combat_actor_morale < @ai_thresh_morale_shaken }
					}
					# Guard when hurt.
					modifier = {
						add = @ai_defense_when_hurt
						has_variable = set_guard
						scope:combat_host = {
							pod_combat_actor_hp_value < pod_combat_actor_max_hp_value
						}
					}
					# dodge when hurt.
					modifier = {
						add = @ai_defense_when_hurt
						has_variable = dodge
						scope:combat_host = {
							pod_combat_actor_hp_value < pod_combat_actor_max_hp_value
						}
					}
					# stealth when hurt.
					modifier = {
						add = @ai_defense_when_hurt
						has_variable = stealth
						scope:combat_host = {
							pod_combat_actor_hp_value < pod_combat_actor_max_hp_value
						}
					}
					# Soak buff when hurt — defensive tier buffs become
					# valuable once the actor has taken wounds. The
					# tier-redundancy guard upstream already filters out
					# buffs at-or-below the actor's currently-held tier,
					# so any candidate that reaches here is a meaningful
					# upgrade. Either round-duration or combat-duration
					# soak buff qualifies.
					modifier = {
						add = @ai_defense_when_hurt
						OR = {
							has_variable = soak_buff_round_tier
							has_variable = soak_buff_combat_tier
						}
						scope:combat_host = {
							pod_combat_actor_hp_value < pod_combat_actor_max_hp_value
						}
					}
					# Hit buff when hit chance is low — offensive tier
					# buffs become valuable when the actor's hit chance
					# against the current workspace target is poor.
					# Threshold 50 catches "noticeably below the 60-base
					# parity case." Like the soak version, the
					# tier-redundancy guard upstream filters non-upgrades
					# so any candidate here is a meaningful tier raise.
					modifier = {
						add = @ai_hitbuff_low_acc
						OR = {
							has_variable = hit_buff_round_tier
							has_variable = hit_buff_combat_tier
						}
						scope:combat_host = {
							var:pod_combat_actor_hit_chance <= @ai_thresh_hit_buff_band
						}
					}
					# Round-buff anti-spam penalty — round-duration tier
					# buffs expire at end of round so the actor gets at
					# most one follow-up action to benefit. A flat -8
					# keeps them out of the default rotation while leaving
					# the +15 buff-when-needed bonuses above able to win
					# the tie (net +7) in situations where the buff
					# actually solves a problem. Combat-duration buffs are
					# untouched — they pay off across the whole fight
					# regardless of when in the turn they land.
					modifier = {
						add = @ai_round_buff_spam
						OR = {
							has_variable = soak_buff_round_tier
							has_variable = hit_buff_round_tier
						}
					}
					# Late-turn hit-buff penalty — a round hit buff with
					# no AP left to attack is pure waste; the buff vanishes
					# at end of round without ever being applied to a hit
					# roll. Threshold = base AP (6) − 1, so the modifier
					# fires the moment current_ap drops below the starting
					# allowance. Note this also fires on the first action
					# of an AP-drained turn (drain pulls current_ap below
					# 6 before any spend) — that's a deliberate false
					# positive: a drained actor has fewer follow-up attacks
					# and is even worse-positioned to capitalize on a
					# round hit buff, so the penalty stands. Stacks with
					# the -8 above for -28 total once the actor commits
					# to a turn.
					modifier = {
						add = @ai_round_buff_lateap
						has_variable = hit_buff_round_tier
						scope:combat_host = { var:pod_combat_current_ap <= @ai_thresh_late_ap }
					}
					# Remove-guard vs guarded target.
					modifier = {
						add = @ai_remove_guard_hit
						has_variable = remove_guard
						scope:combat_host = { has_variable = pod_combat_target_is_guarding }
					}
					modifier = {
						add = @ai_remove_guard_miss
						has_variable = remove_guard
						scope:combat_host = { NOT = { has_variable = pod_combat_target_is_guarding } }
					}

					# ── Target-aware modifiers ──────────────────────
					# Read workspace target state. Gated to single_enemy
					# moves because the workspace target at weight time
					# IS the eventual target for those (the AI's
					# turn-start enemy slot). Self / team / single_ally
					# / single_pc moves don't read the workspace target
					# meaningfully, so these modifiers skip them.

					# Bypass-dodge bonus when target is dodging.
					modifier = {
						add = @ai_bypass_dodge
						has_variable = bypass_dodge
						var:targeting = flag:single_enemy
						scope:combat_host = { has_variable = pod_combat_target_is_dodging }
					}

					# Bypass-stealth bonus when target is stealthed.
					modifier = {
						add = @ai_bypass_stealth
						has_variable = bypass_stealth
						var:targeting = flag:single_enemy
						scope:combat_host = { has_variable = pod_combat_target_is_stealthed }
					}

					# Capitalize-on-stealth bonus: when the AI is
					# currently stealthed (the +30 hit-chance bonus
					# is live), bias hard toward damage-dealing
					# moves so the actor cashes in the stealth
					# window instead of burning it on heals, buffs,
					# or stance changes. Stealth is consumed by the
					# next attack via
					# pod_combat_consume_actor_stealth_effect, so a
					# non-damage move both wastes the +30 AND leaves
					# the actor visible afterward (stealth flag
					# round-trips to slot when the next save fires
					# without the flag set, depending on path).
					# Single-enemy gate matches the other offense
					# biases — a self-targeted stealth move
					# shouldn't get the bonus.
					modifier = {
						add = @ai_cash_in_stealth
						has_variable = hp_damage
						var:targeting = flag:single_enemy
						scope:combat_host = { has_variable = pod_combat_actor_is_stealthed }
					}

					# Aggravated-vs-soak bonus: when the workspace target
					# has high soak (≥ 70%), bias toward aggravated-damage
					# moves. Aggravated bypasses the soak roll entirely for
					# the vast majority of targets (PoD's
					# POD_damage_type_is_soakable returns no for aggravated
					# death reasons; only Mokole undying-body, Demon
					# infernal-fortitude, Dragon ichor-temp, and Autumn-armor
					# combatants opt back in via POD_can_soak_aggravated_damage,
					# and even those roll a separate, usually narrower
					# aggravated-soak). A superficial move would mostly
					# bounce off; an aggravated alternative in the actor's
					# loadout is the right answer.
					modifier = {
						add = @ai_aggravated_vs_soak
						has_variable = hp_damage
						has_variable = damage_type
						var:damage_type = flag:aggravated
						var:targeting = flag:single_enemy
						scope:combat_host = {
							var:pod_combat_target_soak >= @ai_thresh_high_soak
						}
					}

					# Weakness-exploitation bonus: when the candidate's
					# damage_source matches a splat weakness of the
					# workspace target, +25. The mapping lives in
					# pod_combat_damage_source_weakness_triggers.txt; edit
					# the per-source triggers there to add or remove
					# weaknesses without touching this selector.
					#
					# Why +25: hitting a splat weakness either lands aggravated
					# wounds on a target that wouldn't otherwise take them
					# (silver vs werewolf, gold vs mokole, sun vs vampire)
					# or doubles the damage box count (fire vs bushi). Both
					# are stronger than the +20 aggravated-vs-soak bias
					# above, on par with the +25 HP-finish-tier-1 bonus.
					modifier = {
						add = @ai_weakness_exploit
						has_variable = hp_damage
						has_variable = damage_source
						var:targeting = flag:single_enemy
						scope:combat_host = {
							has_variable = pod_combat_target_char
						}
						pod_combat_move_exploits_target_weakness_trigger = yes
					}

					# HP-finish bonus tier 1: target near death (1 box
					# left), any damage move can finish them. Reads HP
					# directly off the opponent character's PoD wound
					# track since there's no opponent_hp_value SV.
					modifier = {
						add = @ai_hp_finish_1box
						has_variable = hp_damage
						var:targeting = flag:single_enemy
						scope:combat_host = {
							has_variable = pod_combat_target_char
							var:pod_combat_target_char.POD_empty_damage_boxes <= @ai_thresh_hp_finish_1
						}
					}

					# HP-finish bonus tier 2: target hurt (2 boxes left)
					# AND move deals at least 2 damage so a finish is
					# realistic.
					modifier = {
						add = @ai_hp_finish_2box
						has_variable = hp_damage
						var:hp_damage >= @ai_thresh_finish_damage
						var:targeting = flag:single_enemy
						scope:combat_host = {
							has_variable = pod_combat_target_char
							var:pod_combat_target_char.POD_empty_damage_boxes <= @ai_thresh_hp_finish_2
						}
					}

					# Morale-finish bonus tier 1: target almost broken
					# (morale 0-2), any morale-damage move finishes.
					modifier = {
						add = @ai_mor_finish_low
						has_variable = morale_damage
						var:targeting = flag:single_enemy
						scope:combat_host = {
							var:pod_combat_target_morale <= @ai_thresh_mor_finish_low
						}
					}

					# Morale-finish bonus tier 2: target shaken (morale
					# 3-4) AND move deals at least 2 morale damage.
					modifier = {
						add = @ai_mor_finish_mid
						has_variable = morale_damage
						var:morale_damage >= @ai_thresh_finish_morale
						var:targeting = flag:single_enemy
						scope:combat_host = {
							var:pod_combat_target_morale <= @ai_thresh_mor_finish_mid
						}
					}

					# Low-hit attack penalty: moves that actually roll a
					# hit chance and could miss. Auto-hit moves
					# (supernatural / mental, plus any physical move
					# with override_requires_hit_roll = no) skip this
					# penalty because they can't whiff. Gated on the
					# capability flag directly rather than the
					# `NOT has_variable = attack_type` proxy that used
					# to live here — the proxy missed the
					# physical+override_requires_hit_roll=no edge case.
					modifier = {
						add = @ai_hit_penalty_low
						has_variable = hp_damage
						has_variable = requires_hit_roll
						var:targeting = flag:single_enemy
						scope:combat_host = {
							var:pod_combat_actor_hit_chance <= @ai_thresh_hit_low
						}
					}

					# Mid-hit attack: softer penalty in the 31-40%
					# band, same gate (only fires for moves that roll
					# a hit chance).
					modifier = {
						add = @ai_hit_penalty_mid
						has_variable = hp_damage
						has_variable = requires_hit_roll
						var:targeting = flag:single_enemy
						scope:combat_host = {
							var:pod_combat_actor_hit_chance <= @ai_thresh_hit_mid
						}
					}

					# Energy attrition: gently deprioritise energy
					# spends as the pool drains this combat. The four
					# stacking add modifiers below are -8 / -10 / -12 / -14
					# (NOT a flat -4 per step), so the cumulative penalty
					# at 0 / 1 / 2 / 3 / 4+ energy used is 0 / -8 / -18 /
					# -30 / -44. The escalating per-step magnitude makes
					# the AI back off energy-spends faster as the pool
					# depletes than a flat curve would. Stacked add
					# modifiers (rather than a single `subtract = { value
					# = X multiply = 4 }`) because random_in_list weight
					# modifiers don't accept inline script-value math
					# only flat numeric add / factor.
					# NOTE: This is a SELECTION-BIAS penalty layered on
					# top of the real energy spend — the AI should
					# pays splat energy when it executes an energy-cost
					# move (the move runs through the same
					# POD_spend_splat_agnostic_energy path the PC uses).
					# pod_combat_actor_energy_used is the workspace
					# mirror of the current actor's slot_energy_used
					# (per-combatant, persists for the whole combat;
					# no per-turn reset). Named "actor" (not "player")
					# because it counts spends by whoever is currently
					# acting, PC or AI, so the "player_*" workspace
					# prefix would read as "PC-only" and mislead.
					modifier = {
						add = @ai_energy_step1
						has_variable = energy_cost
						scope:combat_host = { var:pod_combat_actor_energy_used >= 1 }
					}
					modifier = {
						add = @ai_energy_step2
						has_variable = energy_cost
						scope:combat_host = { var:pod_combat_actor_energy_used >= 2 }
					}
					modifier = {
						add = @ai_energy_step3
						has_variable = energy_cost
						scope:combat_host = { var:pod_combat_actor_energy_used >= 3 }
					}
					modifier = {
						add = @ai_energy_step4
						has_variable = energy_cost
						scope:combat_host = { var:pod_combat_actor_energy_used >= 4 }
					}

					# Trait-gate bonus: moves with splat / perk /
					# trait requirements (disciplines, gifts, hekau,
					# etc.) represent the character's actual class
					# repertoire. Nudge them above generic physicals
					# so e.g. vampires favour their disciplines over
					# plain strikes. Starter moves have no required
					# trait, so a missing is_starter_move flag is the
					# proxy.
					modifier = {
						add = @ai_trait_gated
						NOT = { has_variable = is_starter_move }
					}

					# Floor the final weight so a move that the
					# situational modifiers have driven into negative
					# territory still has a non-zero chance of being
					# picked. random_in_list ignores items with
					# weight <= 0, which would silently exclude moves
					# the player has equipped but had a streak of bad
					# context — keeping `min = @ai_weight_floor` lets
					# the AI eventually fall back to anything in the
					# loadout.
					min = @ai_weight_floor
				}

				save_scope_as = chosen_move
			}

			# Record anti-repetition on the AI actor (still in ai_actor
			# scope). Shift a 3-deep ring: prev2 ← prev, prev ← last,
			# last ← chosen. Order matters — read each older slot
			# BEFORE overwriting it, so writes cascade cleanly in a
			# single pass.
			if = {
				limit = { exists = scope:chosen_move }
				if = {
					limit = { has_variable = pod_combat_ai_prev_move }
					set_variable = {
						name  = pod_combat_ai_prev2_move
						value = var:pod_combat_ai_prev_move
					}
				}
				if = {
					limit = { has_variable = pod_combat_ai_last_move }
					set_variable = {
						name  = pod_combat_ai_prev_move
						value = var:pod_combat_ai_last_move
					}
				}
				set_variable = {
					name  = pod_combat_ai_last_move
					value = scope:chosen_move
				}
			}
		}

		# Execute on combat_host so swap / execute / restore all write
		# to the workspace that's actually loaded there. Running this
		# inside var:pod_combat_actor_char would miss the workspace
		# entirely and the AI would "act" with no mechanical effect.
		#
		# scope:chosen_move can be stale: CK3 saved scopes persist up
		# the call stack, so a chosen_move saved by random_in_list in
		# a PRIOR try_action call is still visible here. If random_in_list
		# this call picks nothing (every move in the loadout failed
		# the limit e.g. AP=1 left and every move costs >=2),
		# `exists = scope:chosen_move` would still see the stale value
		# and re-execute the previous move. Re-checking executability
		# against the current workspace state inside the gate catches
		# that case: the same trigger random_in_list filtered with
		# above is invoked here, so the gate passes only if the
		# chosen_move is genuinely picked-able right now.
		if = {
			limit = {
				exists = scope:chosen_move
				scope:combat_host = {
					POD_can_execute_combat_move = {
						STORY = scope:chosen_move
						ACTOR = scope:ai_actor
					}
				}
			}
			POD_combat_ai_swap_for_story     = { STORY = scope:chosen_move }
			POD_execute_combat_move           = { STORY = scope:chosen_move }
			POD_combat_ai_restore_after_move = { STORY = scope:chosen_move }
		}
	}
}

pod_combat_ai_ensure_loadout_effect = {
	# AI loadouts are TRANSIENT per fight. 
	# Lifecycle:
	#   1. Slot init (pod_combat_init_slot_from_character_effect)
	#      clears pod_combat_loadout on every AI combatant entering
	#      combat — handles legacy NPCs with stale story refs from
	#      pre-version-reload catalogues, and refreshes the pool for
	#      mid-campaign trait changes.
	#   2. This effect runs at the top of every AI move-selection
	#      call. On the AI's first turn the loadout is empty (just
	#      cleared at slot init), so POD_apply_ai_full_loadout fires
	#      and dumps the entire move catalogue. On subsequent turns
	#      the gate fails (count > 0) and this is a cheap no-op.
	#   3. pod_combat_group_cleanup_effect clears the loadout again
	#      at combat end so save data between fights doesn't carry
	#      442 story refs per AI.
	#
	# AI opponents aren't limited to the 4-per-category cap so they
	# have their full trait-appropriate repertoire to pick from.
	# (Trait gating happens at selection time via the limit clause's
	# POD_can_execute_combat_move call, which itself uses
	# can_execute_decision against the move's hidden decision, not
	# at seed time. An unqualified move simply never wins the
	# weighted roll.)
	#
	# PC loadouts are seeded by the configure decision with the 8
	# starters and persist as equipment; the slot-init clear skips
	# slot 0 so the PC's list isn't touched.
	if = {
		limit = {
			has_variable = pod_combat_actor_char
		}
		var:pod_combat_actor_char = {
			if = {
				limit = { pod_combat_equipped_count = 0 }
				POD_apply_ai_full_loadout = yes
			}
		}
	}
}

POD_combat_ai_swap_for_story = {
	# $STORY$ = the chosen move
	# Dispatches to the matching swap effect based on the story's
	# targeting flag. Single_ally and single_pc swap into the right
	# slot before the move body runs; every other targeting mode
	# keeps the AI's turn-start enemy target (single_enemy default)
	# or handles its own slot iteration (team_enemy / team_ally /
	# self).

	if = {
		limit = { $STORY$ = { var:targeting = flag:single_ally } }
		pod_combat_ai_swap_to_ally_target_effect = yes
	}
	else_if = {
		limit = { $STORY$ = { var:targeting = flag:single_pc } }
		pod_combat_ai_swap_to_pc_target_effect = yes
	}
	# single_enemy, self, team_enemy, team_ally — no swap needed
	# (the workspace keeps the turn-start opponent target; AoE skills
	# do their own slot iteration inside the execution engine).
}

POD_combat_ai_restore_after_move = {
	# $STORY$ = the chosen move
	# Undoes any swap done by POD_combat_ai_swap_for_story so the next
	# action starts from the AI's turn-start enemy target. Only swaps
	# that saved pod_combat_ai_saved_enemy_target need restoration;
	# the existing swap effects all save it, so a blanket restore is safe.
	if = {
		limit = {
			OR = {
				$STORY$ = { var:targeting = flag:single_ally }
				$STORY$ = { var:targeting = flag:single_pc }
			}
			has_variable = pod_combat_ai_saved_enemy_target
		}
		pod_combat_ai_restore_enemy_target_effect = yes
	}
}

# ─── SINGLE-ALLY TARGET SWAP ──────────────────────────────────────
# Fired by the AI select effect right before a single_ally move body
# runs. Saves the AI's current enemy target (selected_target, set at
# AI turn start by pod_combat_ai_select_{enemy,player}_target_effect),
# picks the first alive non-self same-team ally, and loads that ally
# into the opponent workspace so the move effect's writes land on the
# right slot.
#
# Single_ally moves are filtered out of the selector pool when the
# acting AI has no alive same-team ally — POD_can_execute_combat_move's
# AI-side gate calls pod_combat_ai_has_available_ally_trigger for
# single_ally targeting and rejects the candidate if the trigger
# returns false. So when this swap effect runs, the if/else_if chain
# below is guaranteed to hit a match. 

pod_combat_ai_swap_to_ally_target_effect = {
	# Save the current selected_target so pod_combat_ai_restore_enemy_target_effect
	# can put it back after the single-ally move fires.
	set_variable = { name = pod_combat_ai_saved_enemy_target value = var:pod_combat_selected_target }

	# Pick any alive non-self same-team ally and load them into the
	# opponent workspace. Team membership is fixed by slot index: slots
	# 0-4 are team 0 (player team), 5-9 are team 1 (enemy team), so
	# "same team as actor" is the AND-of-(actor team, slot team) filter
	# below.
	save_scope_as = host
	random_in_list = {
		variable = pod_combat_active_slots
		limit = {
			var:slot_alive = 1
			NOT = { var:slot_index = scope:host.var:pod_combat_current_actor }
			OR = {
				AND = { scope:host = { var:pod_combat_current_actor <= 4 } var:slot_index <= 4 }
				AND = { scope:host = { var:pod_combat_current_actor >= 5 } var:slot_index >= 5 }
			}
		}
		weight = {
			base = 1
			modifier = { add = 50 }
		}
		save_scope_as = chosen_target
	}
	if = {
		limit = { exists = scope:chosen_target }
		scope:host = {
			set_variable = { name = pod_combat_selected_target value = scope:chosen_target.var:slot_index }
		}
		scope:chosen_target = {
			pod_combat_load_target_from_story_effect = yes
		}
	}
}

pod_combat_ai_swap_to_pc_target_effect = {
	# Save the current selected_target so restore_enemy_target can put it back
	set_variable = { name = pod_combat_ai_saved_enemy_target value = var:pod_combat_selected_target }

	# Swap to slot 0 (the PC) and load into the opponent workspace.
	# Snapshot host scope, flip into the slot story, delegate to
	# pod_combat_load_target_from_story_effect.
	if = {
		limit = { has_variable = pod_combat_slot_0 var:pod_combat_slot_0 = { has_variable = slot_char } }
		set_variable = { name = pod_combat_selected_target value = 0 }
		save_scope_as = host
		var:pod_combat_slot_0 = {
			pod_combat_load_target_from_story_effect = yes
		}
	}
}

pod_combat_ai_restore_enemy_target_effect = {
	# Restore the enemy target saved by either
	# pod_combat_ai_swap_to_ally_target_effect (single_ally moves) or
	# pod_combat_ai_swap_to_pc_target_effect (single_pc moves) and reload
	# it into the opponent workspace. Called from
	# POD_combat_ai_restore_after_move whenever the move's targeting was
	# single_ally or single_pc, so subsequent AI actions in the same turn
	# continue to aim at the AI's original chosen enemy.
	#
	# CRITICAL: Save the ally slot first. The single-ally move effect wrote
	# buffs into the opponent workspace, and the workspace currently holds
	# the buffed ally. If we reload the enemy slot without saving first, the
	# buff would be overwritten by the stale slot data and effectively erased.
	if = {
		limit = { has_variable = pod_combat_ai_saved_enemy_target }
		save_scope_as = host
		# Save the ally workspace back to its slot before we reload the enemy
		every_in_list = {
			variable = pod_combat_active_slots
			limit = { var:slot_index = scope:host.var:pod_combat_selected_target }
			pod_combat_save_target_to_story_effect = yes
		}

		# Now swap selected_target back to the enemy and reload that slot's data
		set_variable = { name = pod_combat_selected_target value = var:pod_combat_ai_saved_enemy_target }
		every_in_list = {
			variable = pod_combat_active_slots
			limit = { var:slot_index = scope:host.var:pod_combat_selected_target }
			pod_combat_load_target_from_story_effect = yes
		}
		remove_variable = pod_combat_ai_saved_enemy_target
	}
}
