# 1v1 is a degenerate NvN (player/0 + enemy/0); same workspace-relative
# move effects handle every roster.


# ─────────────────────────────────────────────────────────────
# AI TUNING CONSTANTS — edit these to tweak AI behaviour
# ─────────────────────────────────────────────────────────────
# Designer-tuned biases for AI move choice, applied by the weight block in
# pod_combat_ai_select_and_execute_move_effect. Weight deltas first,
# comparison thresholds at the bottom.

# Anti-repetition (three-deep history).
@ai_repeat_last         = -35
@ai_repeat_prev         = -20
@ai_repeat_prev2        = -12

# Need-based bonuses.
@ai_morale_heal_low     = 30   # morale_heal when morale below threshold
@ai_morale_heal_early   = 12   # smaller proactive morale top-up in the pre-shaken band
@ai_defense_when_hurt   = 15   # guard / dodge / stealth / soak-buff while hp < max
@ai_hitbuff_low_acc     = 15   # hit_buff while hit_chance ≤ threshold

# Round-buff lifecycle penalties.
@ai_round_buff_spam     = -8   # round HIT buff only (soak round buffs now guard the enemy phase, not penalized)
# Separate from the generic spam gate: stacking both penalties floored a
# round hit-buff out of contention, so the AI never refreshed an expired one
# late-turn. -12 keeps a late refresh pickable while still discouraging it.
@ai_round_buff_lateap   = -6   # round hit-buff specifically, low AP remaining (softened from -12: with -8 spam the old -20 stack buried a genuinely-needed late refresh)

# Tier-size preference (buffs AND debuffs) — tier 3 buys 3× the
# effect for the same AP cost, so bias toward it. Tier 2 is pinned
# at 0 as the neutral baseline; shifting it slides the whole curve.
@ai_tier_small          = -5   # tier 1 (small  / 10%)
@ai_tier_medium         = 0    # tier 2 (medium / 20%) — neutral baseline
@ai_tier_large          = 10   # tier 3 (large  / 30%)
@ai_buff_undersized     = -8   # extra penalty on a buff/debuff when an affordable, strictly-larger tier of the SAME stat is held (steer to the bigger version)
@ai_spirit_gate_wasted    = -25  # physical attack vs an incorporeal target the actor can't hurt (guaranteed whiff)
@ai_spirit_gate_viable    = 15   # damage move that CAN connect with an incorporeal target
@ai_aoe_multi_enemy       = 12   # team_enemy damage while 2+ enemies of the actor are alive
@ai_hitdebuff_advantage   = 10   # hit_debuff aimed at an accurate enemy (drag a real threat down)
@ai_combat_buff_reward    = 8    # combat-duration buff/debuff early in the fight (outlasts a round buff)
@ai_cheap_finish_tempo    = 12   # affordable damage when AP is running low (press, don't dawdle on utility)
@ai_energy_high_cost_lowap = -15 # heavy move (energy_cost >= 3) when AP is tight (avoid stranding the turn)
@ai_thresh_lowap_expensive = 3   # "AP tight" ceiling for the heavy-move penalty
@ai_control_defend        = 10   # control (mind-control / AP-drain) while wounded vs an accurate threat
@ai_bonus_condition_met   = 12   # a bonus move's condition is met at scoring -> nudge to trigger it

# Remove-guard bias.
@ai_remove_guard_hit    = 15   # target IS guarding
@ai_remove_guard_miss   = -15  # target NOT guarding (strip would no-op)

# Target-aware offensive bonuses (single_enemy moves only).
@ai_bypass_dodge        = 15   # bypass_dodge vs dodging target
@ai_bypass_stealth      = 15   # bypass_stealth vs stealthed target
@ai_cash_in_stealth     = 20   # actor stealthed → spend the +30 hit window
@ai_aggravated_vs_soak  = 20   # high-soak target → bias aggravated
@ai_superficial_vs_soak = -30  # mundane superficial attack (no aggravated, no elemental source) vs a >=70% soak wall
@ai_weakness_exploit    = 25   # damage_source matches target splat weakness

# Finish bonuses.
@ai_hp_finish_1box      = 25   # target near death, any damage finishes
@ai_hp_finish_2box      = 15   # target hurt, move deals ≥ ai_thresh_finish_damage
@ai_mor_finish_low      = 25   # target morale low, any morale damage finishes
@ai_mor_finish_mid      = 15   # target morale mid, move deals ≥ ai_thresh_finish_morale

# Hit-chance penalties (requires_hit_roll moves only).
@ai_hit_penalty_low     = -25  # hit_chance ≤ ai_thresh_hit_low
@ai_hit_penalty_mid     = -15  # hit_chance ≤ ai_thresh_hit_mid

# Energy attrition (stacking per step of actor_energy_used). Step 4 holds at
# -12 (not steeper) so a dying AI can still afford its last expensive move to
# close a kill instead of being floored out of contention.
@ai_energy_step1        = -8
@ai_energy_step2        = -10
@ai_energy_step3        = -12
@ai_energy_step4        = -12

# Class / repertoire bias.
# +18 (was +10) so a class move reaches a real fallback share (~26) vs the
# heal's +38; learned powers felt unused at +10. Non-starter moves only and
# independent of heal/finish, so it stays under the +30 heal / +25 finish caps.
@ai_trait_gated         = 18   # non-starter move (perk/trait/splat-gated)

# Final-weight floor (random_in_list ignores items with weight ≤ 0).
@ai_weight_floor        = 1

# Comparison thresholds read by the modifier `limit`s.
@ai_thresh_morale_shaken   = 10   # morale-heal trigger (below = shaken, full bonus)
@ai_thresh_morale_early    = 15   # morale-heal early band ceiling (shaken..this = proactive)
@ai_thresh_hit_low         = 30   # hit_penalty_low fires at-or-below
@ai_thresh_hit_mid         = 40   # hit_penalty_mid fires at-or-below
@ai_thresh_hit_buff_band   = 50   # hitbuff_low_acc fires at-or-below
@ai_thresh_high_soak       = 70   # aggravated_vs_soak fires at-or-above
@ai_thresh_late_ap         = 5    # round_buff_lateap fires at-or-below (base 6 − 1)
@ai_thresh_hp_finish_1     = 1    # empty_damage_boxes ≤ this → hp_finish_1box fires
@ai_thresh_hp_finish_2     = 2    # empty_damage_boxes ≤ this → hp_finish_2box fires
@ai_thresh_finish_damage   = 2    # hp_finish_2box also requires hp_damage ≥ this
@ai_thresh_mor_finish_low  = 2    # target morale ≤ this → mor_finish_low fires
@ai_thresh_mor_finish_mid  = 4    # target morale ≤ this → mor_finish_mid fires
@ai_thresh_finish_morale   = 2    # mor_finish_mid also requires morale_damage ≥ this

# ─────────────────────────────────────────────────────────────
# PERSONALITY + PRIMITIVE-SITUATIONAL CONSTANTS
# ─────────────────────────────────────────────────────────────
# Personality bias magnitudes, consumed by pod_combat_apply_personality_profile_effect
# (the precompute) and the PERSONALITY BAND read-modifiers. Pre-clamp per-trait
# contributions; cross-axis sums are bounded by the @ai_pers_clamp_* values below.
@ai_pers_aggressive     = 8
@ai_pers_bold           = 8
@ai_pers_savage         = 5
@ai_pers_defensive      = 8
@ai_pers_timid          = 10
@ai_pers_passive        = 5
@ai_pers_dominating     = 8
@ai_pers_cruel          = 8
@ai_pers_cruel_light    = 3
@ai_pers_wary           = 8
@ai_pers_disciplined    = 5
@ai_pers_reckless       = 5
@ai_pers_guile          = 5

# Cross-axis post-sum clamp bounds (symmetric: var clamped to [-N, +N]).
# Negated twins are precomputed defines (CK3 has no clamp_variable, and the
# negative-bound compares/writes need a literal, not inline use-site math).
@ai_pers_clamp_dot      = 8
@ai_pers_clamp_control  = 10
@ai_pers_clamp_savage   = 8
@ai_pers_clamp_dot_neg      = @[ai_pers_clamp_dot * -1]
@ai_pers_clamp_control_neg  = @[ai_pers_clamp_control * -1]
@ai_pers_clamp_savage_neg   = @[ai_pers_clamp_savage * -1]

# Category post-sum clamp (atk/def/ctl/pwr). Compatible same-category traits
# stack unbounded (~+76 worst-case ATK), enough to swamp the +30 heal-need and
# let a dying aggressive AI fight to its death. The heal-vs-attack inequality
# works out to C < 11.5; clamp at 10 for margin. A single strong trait
# (timid +10, bold/aggressive +8) still reads in full.
@ai_pers_clamp_cat      = 10
@ai_pers_clamp_cat_neg      = @[ai_pers_clamp_cat * -1]

# Primitive situational weights (DoT / mind control / stun / summon / retaliate).
@ai_dot_healthy_target    = 10   # DoT vs a target with boxes to burn
@ai_dot_overkill_penalty  = -10  # DoT ramp-up wasted on a near-dead target
@ai_mind_control_value    = 15   # dominate a dominatable AI
@ai_stun_value            = 15   # full stun (ap_drain ≥ @ai_thresh_stun) vs AI; aligned with mind_control's 15 (the two control primitives)
@ai_ap_drain_value        = 6    # partial ap_drain vs AI (flat, no target-AP read)
@ai_summon_outnumbered    = 12   # summon while down a teammate (KO'd same-team slot)
@ai_retaliate_when_exposed = 10  # arm a retaliation while hp < max

# Personality/primitive thresholds.
@ai_thresh_dot_healthy  = 4    # DoT bonus needs target empty_damage_boxes ≥ this
@ai_thresh_stun         = 6    # ap_drain ≥ this counts as a full stun (vs partial)

# ── PERSONALITY TARGET PREFERENCE ─────────────────────────
# Temperament biases WHO gets hit, not just which move. Derived ONCE per combat
# in pod_combat_apply_personality_profile_effect (a single mutually-exclusive
# pod_combat_ai_target_pref flag off the summed/clamped bias vars), then read as
# ONE gated additive in the two GROUP target selectors via scope:focus_actor.
# Both magnitudes stay below the near-dead kill gap (factor 3 widens that gap to
# +102 over the 51 baseline), so kill-priority always overrides target preference.
#
# NOTE: the target-pref ADD weights and candidate thresholds are now shared
# script_values (pod_combat_ai_target_pref_{wounded,threat}_{weight,threshold} in
# pod_combat_values.txt), referenced directly by the selectors in
# pod_combat_group_effects.txt - no @-define literal-mirror to keep in sync (G09).
# The three bias thresholds below ARE read by the precompute in THIS file.
@ai_thresh_pref_savage    = 5    # bias_savage ≥ this picks the wounded pref
@ai_thresh_pref_dot       = 5    # bias_dot ≥ this picks the wounded pref
@ai_thresh_pref_control   = 8    # bias_control ≥ this picks the threat pref (if not wounded)

# ── B3: CRITICAL HEAL ──────────────────────────────────────────
# A dying actor (at its last box) with no finishable enemy to kill instead gets
# +20 heal weight ON TOP of the scaled base heal (the pod_combat_ai_heal_weight_* host vars,
# ~+30 when dying -> effective ~+50), so it tops up
# rather than trading a doomed strike. enemy_finishable is precomputed once per
# turn in GROUP (pod_combat_ai_flag_finishable_enemy_effect).
@ai_heal_critical         = 20   # extra heal weight for a dying actor with no kill available
@ai_thresh_heal_critical  = 1    # actor_hp_value ≤ this (at its last box) = "dying"


# per-actor personality precompute.
# Runs ONCE per combat on scope:ai_actor (the acting NPC), writing the seven bias
# vars the PERSONALITY BAND reads at candidate-scoring time. Cleared at the
# AI-character cleanup site (slot init + group teardown), NOT host teardown.
#
# Contract: seed all seven biases to 0, sum per-trait contributions, then clamp
# LAST. The three cross-axis vars (_dot/_control/_savage) clamp because multiple
# compatible traits write them (e.g. sadistic + ambitious both writing _control);
# the four category vars clamp because many opposed-pair-independent traits stack
# them (~+76 worst-case ATK unclamped) - see @ai_pers_clamp_cat for why that would
# otherwise let a dying aggressive AI skip the heal. Magnitudes are the @ai_pers_*
# constants in the tuning header.
pod_combat_apply_personality_profile_effect = {
	# Seed all seven biases to 0 -> branch-free, safe reads in the weight block.
	set_variable = { name = pod_combat_pers_bias_atk     value = 0 }
	set_variable = { name = pod_combat_pers_bias_def     value = 0 }
	set_variable = { name = pod_combat_pers_bias_ctl     value = 0 }
	set_variable = { name = pod_combat_pers_bias_pwr     value = 0 }
	set_variable = { name = pod_combat_pers_bias_dot     value = 0 }
	set_variable = { name = pod_combat_pers_bias_control value = 0 }
	set_variable = { name = pod_combat_pers_bias_savage  value = 0 }

	# Opposed-pair traits are mutually exclusive, so the wrathful/calm and
	# brave/craven contradictions can't co-occur on one character.
	# wrathful: berserker - presses damage, savors aggravated + DoT, scorns defense.
	if = {
		limit = { has_trait = wrathful }
		change_variable = { name = pod_combat_pers_bias_atk    add = @ai_pers_aggressive }
		change_variable = { name = pod_combat_pers_bias_savage add = @ai_pers_savage }
		change_variable = { name = pod_combat_pers_bias_dot    add = @ai_pers_savage }
		change_variable = { name = pod_combat_pers_bias_def    subtract = @ai_pers_defensive }
	}
	# calm: disciplined - stances/buffs and measured control, no berserk lunge.
	if = {
		limit = { has_trait = calm }
		change_variable = { name = pod_combat_pers_bias_def    add = @ai_pers_defensive }
		change_variable = { name = pod_combat_pers_bias_ctl    add = @ai_pers_disciplined }
		change_variable = { name = pod_combat_pers_bias_savage subtract = @ai_pers_savage }
	}
	# brave: frontline - presses ATK, won't skulk. The DEF lean is category-wide
	# (the mechanism can't single out disengage stances); a brave actor still
	# heals when truly hurt (the heal-need gate is untouched by personality).
	if = {
		limit = { has_trait = brave }
		change_variable = { name = pod_combat_pers_bias_atk add = @ai_pers_bold }
		change_variable = { name = pod_combat_pers_bias_def subtract = @ai_pers_bold }
	}
	# craven: marquee turtle - hoards guard/dodge/heal, flees damage.
	if = {
		limit = { has_trait = craven }
		change_variable = { name = pod_combat_pers_bias_def add = @ai_pers_timid }
		change_variable = { name = pod_combat_pers_bias_atk subtract = @ai_pers_timid }
	}
	# sadistic: cruelty - DoT + domination + morale-break. The morale-break lean
	# rides the _ctl bias rather than a dedicated axis (morale-attack moves are
	# CTL category), so no separate morale var is needed.
	if = {
		limit = { has_trait = sadistic }
		change_variable = { name = pod_combat_pers_bias_dot     add = @ai_pers_cruel }
		change_variable = { name = pod_combat_pers_bias_control add = @ai_pers_cruel }
		change_variable = { name = pod_combat_pers_bias_ctl     add = @ai_pers_cruel }
	}
	# ambitious: dominator - control plays (mind_control/ap_drain) + ATK pressure.
	if = {
		limit = { has_trait = ambitious }
		change_variable = { name = pod_combat_pers_bias_control add = @ai_pers_dominating }
		change_variable = { name = pod_combat_pers_bias_ctl     add = @ai_pers_dominating }
		change_variable = { name = pod_combat_pers_bias_atk     add = @ai_pers_bold }
	}

	# content: disengaged - mild defense, reluctant to escalate.
	if = {
		limit = { has_trait = content }
		change_variable = { name = pod_combat_pers_bias_def add = @ai_pers_passive }
		change_variable = { name = pod_combat_pers_bias_atk subtract = @ai_pers_passive }
	}
	# arbitrary: tyrant - control + morale-break. Morale-break rides _ctl (morale
	# moves are CTL category), same as sadistic; no dedicated morale axis.
	if = {
		limit = { has_trait = arbitrary }
		change_variable = { name = pod_combat_pers_bias_ctl     add = @ai_pers_dominating }
		change_variable = { name = pod_combat_pers_bias_control add = @ai_pers_dominating }
	}
	# just: lawful - straightforward ATK, no underhanded control.
	if = {
		limit = { has_trait = just }
		change_variable = { name = pod_combat_pers_bias_atk add = @ai_pers_disciplined }
		change_variable = { name = pod_combat_pers_bias_ctl subtract = @ai_pers_disciplined }
	}
	# compassionate: reluctant - mild defense, avoids cruelty primitives.
	if = {
		limit = { has_trait = compassionate }
		change_variable = { name = pod_combat_pers_bias_def add = @ai_pers_passive }
		change_variable = { name = pod_combat_pers_bias_dot subtract = @ai_pers_passive }
	}
	# callous: efficient lethality - kills rather than savors. Control routes to
	# _control (not _ctl) so the band de-dup and clamp apply.
	if = {
		limit = { has_trait = callous }
		change_variable = { name = pod_combat_pers_bias_atk     add = @ai_pers_aggressive }
		change_variable = { name = pod_combat_pers_bias_control add = @ai_pers_cruel_light }
	}
	# vengeful: hunter - ATK + DoT pursuit. The DEF lean is category-wide ("disengage
	# only" isn't expressible), same as brave; the finish gate already rewards kills.
	if = {
		limit = { has_trait = vengeful }
		change_variable = { name = pod_combat_pers_bias_atk add = @ai_pers_bold }
		change_variable = { name = pod_combat_pers_bias_dot add = @ai_pers_cruel_light }
		change_variable = { name = pod_combat_pers_bias_def subtract = @ai_pers_bold }
	}
	# forgiving: de-escalating - mild defense, no pursuit.
	if = {
		limit = { has_trait = forgiving }
		change_variable = { name = pod_combat_pers_bias_def add = @ai_pers_passive }
		change_variable = { name = pod_combat_pers_bias_atk subtract = @ai_pers_passive }
	}
	# paranoid: pre-emptive - guard/dodge + control (remove_guard/ap_drain), shies
	# from committal PWR.
	if = {
		limit = { has_trait = paranoid }
		change_variable = { name = pod_combat_pers_bias_def     add = @ai_pers_wary }
		change_variable = { name = pod_combat_pers_bias_control add = @ai_pers_wary }
		change_variable = { name = pod_combat_pers_bias_pwr     subtract = @ai_pers_wary }
	}
	# trusting: open - mild ATK, undervalues paranoia's control plays.
	if = {
		limit = { has_trait = trusting }
		change_variable = { name = pod_combat_pers_bias_atk     add = @ai_pers_passive }
		change_variable = { name = pod_combat_pers_bias_control subtract = @ai_pers_passive }
	}
	# diligent: methodical - combat-duration buff setup + control.
	if = {
		limit = { has_trait = diligent }
		change_variable = { name = pod_combat_pers_bias_ctl add = @ai_pers_disciplined }
		change_variable = { name = pod_combat_pers_bias_def add = @ai_pers_disciplined }
	}
	# lazy: low-effort - mild penalty to setup-heavy plays.
	if = {
		limit = { has_trait = lazy }
		change_variable = { name = pod_combat_pers_bias_ctl subtract = @ai_pers_passive }
		change_variable = { name = pod_combat_pers_bias_pwr subtract = @ai_pers_passive }
	}
	# patient: long-game - defense/control setup + DoT, avoids low-odds gambles.
	if = {
		limit = { has_trait = patient }
		change_variable = { name = pod_combat_pers_bias_def add = @ai_pers_disciplined }
		change_variable = { name = pod_combat_pers_bias_ctl add = @ai_pers_disciplined }
		change_variable = { name = pod_combat_pers_bias_dot add = @ai_pers_cruel_light }
	}
	# impatient: burst-now - immediate ATK, penalizes slow setup/DoT.
	if = {
		limit = { has_trait = impatient }
		change_variable = { name = pod_combat_pers_bias_atk add = @ai_pers_aggressive }
		change_variable = { name = pod_combat_pers_bias_ctl subtract = @ai_pers_disciplined }
		change_variable = { name = pod_combat_pers_bias_dot subtract = @ai_pers_cruel_light }
	}
	# arrogant: overreach - flashy PWR/ATK, disdains careful defense (deliberately
	# suboptimal; heal-need still pulls it to safety when low).
	if = {
		limit = { has_trait = arrogant }
		change_variable = { name = pod_combat_pers_bias_pwr add = @ai_pers_reckless }
		change_variable = { name = pod_combat_pers_bias_atk add = @ai_pers_reckless }
		change_variable = { name = pod_combat_pers_bias_def subtract = @ai_pers_reckless }
	}
	# humble: solid - efficient ATK, avoids PWR overreach.
	if = {
		limit = { has_trait = humble }
		change_variable = { name = pod_combat_pers_bias_def add = @ai_pers_disciplined }
		change_variable = { name = pod_combat_pers_bias_pwr subtract = @ai_pers_reckless }
	}
	# deceitful: ambusher - stealth/bypass + control (stealth/bypass feed existing
	# situationals; this only tips the opening).
	if = {
		limit = { has_trait = deceitful }
		change_variable = { name = pod_combat_pers_bias_control add = @ai_pers_guile }
	}
	# honest: straightforward - direct ATK, shuns trickery.
	if = {
		limit = { has_trait = honest }
		change_variable = { name = pod_combat_pers_bias_atk     add = @ai_pers_disciplined }
		change_variable = { name = pod_combat_pers_bias_control subtract = @ai_pers_guile }
	}
	# zealous: crusader - ATK + PWR (faith-driven heavy plays).
	if = {
		limit = { has_trait = zealous }
		change_variable = { name = pod_combat_pers_bias_atk add = @ai_pers_bold }
		change_variable = { name = pod_combat_pers_bias_pwr add = @ai_pers_disciplined }
	}
	# cynical: pragmatic - slight ATK efficiency.
	if = {
		limit = { has_trait = cynical }
		change_variable = { name = pod_combat_pers_bias_atk add = @ai_pers_cruel_light }
	}

	# Cross-axis clamps. CK3 has no clamp_variable effect, so clamp via paired
	# if/set_variable to [-bound, +bound]. These vars take contributions from
	# several compatible (non-opposed) traits, so the sum needs bounding.
	if = {
		limit = { var:pod_combat_pers_bias_dot > @ai_pers_clamp_dot }
		set_variable = { name = pod_combat_pers_bias_dot value = @ai_pers_clamp_dot }
	}
	if = {
		limit = { var:pod_combat_pers_bias_dot < @ai_pers_clamp_dot_neg }
		set_variable = { name = pod_combat_pers_bias_dot value = @ai_pers_clamp_dot_neg }
	}
	if = {
		limit = { var:pod_combat_pers_bias_control > @ai_pers_clamp_control }
		set_variable = { name = pod_combat_pers_bias_control value = @ai_pers_clamp_control }
	}
	if = {
		limit = { var:pod_combat_pers_bias_control < @ai_pers_clamp_control_neg }
		set_variable = { name = pod_combat_pers_bias_control value = @ai_pers_clamp_control_neg }
	}
	if = {
		limit = { var:pod_combat_pers_bias_savage > @ai_pers_clamp_savage }
		set_variable = { name = pod_combat_pers_bias_savage value = @ai_pers_clamp_savage }
	}
	if = {
		limit = { var:pod_combat_pers_bias_savage < @ai_pers_clamp_savage_neg }
		set_variable = { name = pod_combat_pers_bias_savage value = @ai_pers_clamp_savage_neg }
	}

	# Category clamps (atk/def/ctl/pwr). Unlike the cross-axis vars, these
	# accumulate from many opposed-pair-independent traits (CK3 has no
	# 3-trait cap), so the worst compatible ATK pile reaches ~+76 unclamped.
	# The post-sum bound keeps that from drowning the heal-need (+30) and
	# finish (+25) incentives, so a dying aggressive AI still heals to safety.
	# NOTE: raising @ai_pers_clamp_cat above the heal/finish need-bonuses is
	# intentionally capped here - the heal-vs-attack inequality is the hard
	# ceiling, so category bias does NOT scale linearly past 10. A designer
	# expecting linear scaling above the clamp would misread this block.
	if = {
		limit = { var:pod_combat_pers_bias_atk > @ai_pers_clamp_cat }
		set_variable = { name = pod_combat_pers_bias_atk value = @ai_pers_clamp_cat }
	}
	if = {
		limit = { var:pod_combat_pers_bias_atk < @ai_pers_clamp_cat_neg }
		set_variable = { name = pod_combat_pers_bias_atk value = @ai_pers_clamp_cat_neg }
	}
	if = {
		limit = { var:pod_combat_pers_bias_def > @ai_pers_clamp_cat }
		set_variable = { name = pod_combat_pers_bias_def value = @ai_pers_clamp_cat }
	}
	if = {
		limit = { var:pod_combat_pers_bias_def < @ai_pers_clamp_cat_neg }
		set_variable = { name = pod_combat_pers_bias_def value = @ai_pers_clamp_cat_neg }
	}
	if = {
		limit = { var:pod_combat_pers_bias_ctl > @ai_pers_clamp_cat }
		set_variable = { name = pod_combat_pers_bias_ctl value = @ai_pers_clamp_cat }
	}
	if = {
		limit = { var:pod_combat_pers_bias_ctl < @ai_pers_clamp_cat_neg }
		set_variable = { name = pod_combat_pers_bias_ctl value = @ai_pers_clamp_cat_neg }
	}
	if = {
		limit = { var:pod_combat_pers_bias_pwr > @ai_pers_clamp_cat }
		set_variable = { name = pod_combat_pers_bias_pwr value = @ai_pers_clamp_cat }
	}
	if = {
		limit = { var:pod_combat_pers_bias_pwr < @ai_pers_clamp_cat_neg }
		set_variable = { name = pod_combat_pers_bias_pwr value = @ai_pers_clamp_cat_neg }
	}

	# derive a single mutually-exclusive target
	# preference from the FINAL clamped biases, read by the GROUP target selectors.
	# Wounded takes precedence (lethality reads first): a cruel/sadistic actor
	# (high _dot or _savage) finishes the hurt; an ambitious/dominating actor (high
	# _control) hunts the dangerous. Left unset otherwise -> selectors stay neutral.
	if = {
		limit = { OR = { var:pod_combat_pers_bias_dot >= @ai_thresh_pref_dot  var:pod_combat_pers_bias_savage >= @ai_thresh_pref_savage } }
		set_variable = { name = pod_combat_ai_target_pref value = flag:wounded }
	}
	else_if = {
		limit = { var:pod_combat_pers_bias_control >= @ai_thresh_pref_control }
		set_variable = { name = pod_combat_ai_target_pref value = flag:threat }
	}
}


pod_combat_ai_refresh_target_if_dead_effect = {
	# Re-pick selected_target only if it's no longer alive (usually the
	# prior try_action KO'd it). No-op while alive, so the AI stays on
	# its mark across a multi-action turn instead of thrashing targets.
	save_scope_as = host
	if = {
		limit = {
			has_variable = pod_combat_selected_target_team
			NOT = {
				any_in_list = {
					variable = pod_combat_active_slots
					var:combat_team = scope:host.var:pod_combat_selected_target_team
					var:team_position = scope:host.var:pod_combat_selected_target_position
					pod_combat_slot_is_alive_trigger = yes
				}
			}
		}
		# Re-pick from the appropriate team's live slots and reload
		# the workspace so the new target's data is in pod_combat_target_*.
		# Re-pick controlled-aware (C02): on a stolen turn the dominated actor must
		# keep targeting its OWN side, so use the shared inverted-aware selector
		# rather than a plain enemy/player branch (which reverted to the normal side
		# and wasted the steal once the first betrayal target died). Then re-check
		# the sole-survivor fizzle - it may have just KO'd its last teammate - so the
		# remaining actions stop (try_action gates on the fizzle flag) instead of
		# striking a corpse, before reloading the workspace.
		pod_combat_ai_select_target_for_turn_effect = yes
		pod_combat_ai_check_control_fizzle_effect = yes
		pod_combat_load_workspace_effect = yes
	}
}


pod_combat_ai_take_turn_effect = {
	# Up to 6 actions/turn. A while (count = iteration cap, limit = stop-early)
	# rather than a 6x unroll: net-AP-positive moves (e.g. between_the_ticks,
	# ap_gain > ap_cost) refund AP mid-turn, so the limit stops as soon as the AI
	# is out of AP / the fight ended / a controlled turn fizzled, while count = 6
	# keeps the hard per-turn ceiling (hang-safe). try_action self-gates on the
	# same conditions, so the limit is belt-and-suspenders. Host (PC) scope - all
	# three gate vars are host-scoped (C33).
	while = {
		count = 6
		limit = {
			pod_combat_current_ap_value >= 1
			NOT = { pod_combat_is_over_trigger = yes }
			NOT = { has_variable = pod_combat_turn_control_fizzled }
		}
		pod_combat_ai_try_action_effect = yes
	}
}

pod_combat_ai_try_action_effect = {
	if = {
		limit = {
			pod_combat_current_ap_value >= 1
			NOT = { pod_combat_is_over_trigger = yes }
		}
		# (refresh helper is a no-op while the current target still lives)
		pod_combat_ai_refresh_target_if_dead_effect = yes

		# A controlled (dominated) turn that has run out of own-side targets to
		# betray (refresh just set pod_combat_turn_control_fizzled) stops acting for
		# the rest of the turn rather than executing a move against a dead slot.
		# No-op on a normal turn - the flag is never set. (C02/C27 mid-turn fizzle.)
		if = {
			limit = { NOT = { has_variable = pod_combat_turn_control_fizzled } }
			pod_combat_ai_select_and_execute_move_effect = yes

			# Save per-action (not once post-loop): flips slot _alive flags so
			# the next try sees a KO'd target as dead, and runs the victory
			# check at its tail so a fight-ending hit trips the remaining
			# tries' is_over gate.
			pod_combat_save_workspace_effect = yes
		}
	}
}

# Caches whether ONE bonus-mode loadout move's condition is currently met, as
# the flag pod_combat_ai_bonus_met on the move story, for the
# @ai_bonus_condition_met weight modifier to read as a flat has_variable.
# Called per bonus move from the selector's precompute loop (below) - in effect
# scope each move is a saved scope (scope:precompute_bonus_move), so the
# condition trigger's STORY stays stable inside its own $ACTOR$ / target
# scope-switches, which prev/this would NOT in a weight modifier.
# The move story is shared, but AI turns are sequential and the loop refreshes
# every bonus move right before that actor scores, so no stale cross-actor read.
pod_combat_ai_precompute_bonus_met_effect = {
	# this = a bonus-mode POD_combat_move; HOST = combat host, ACTOR = acting AI.
	save_temporary_scope_as = precompute_bonus_move
	if = {
		limit = {
			$HOST$ = {
				pod_combat_move_condition_met_trigger = { STORY = scope:precompute_bonus_move  ACTOR = $ACTOR$  PREDICATE = scope:precompute_bonus_move.var:cond_predicate }
			}
		}
		set_variable = pod_combat_ai_bonus_met
	}
	else = {
		remove_variable = pod_combat_ai_bonus_met
	}
}

pod_combat_ai_select_and_execute_move_effect = {
	# Weighted random_in_list over the loadout, then swap / execute /
	# restore. pod_combat_actor_char (the acting combatant) is written by
	# pod_combat_group_effects.txt before this fires.
	pod_combat_ai_ensure_loadout_effect = yes

	if = {
		limit = { has_variable = pod_combat_actor_char }

		# Two scopes used below:
		#   scope:combat_host — PC holding the workspace mirror vars; needed
		#                       for POD_execute_combat_move + affordability.
		#   scope:ai_actor    — acting AI; holds the loadout list and the
		#                       anti-repetition history.
		save_scope_as = combat_host

		# Deficit-scaled heal weights, precomputed in host scope (where the actor HP
		# script_values resolve). A random_in_list weight `add` takes only flat numbers /
		# @consts / var refs - NOT a named script_value or inline math - so the curve is
		# computed here and the heal modifiers read it via add = scope:combat_host.var:X.
		# Recomputed each AI action; curve = base + (damaged fraction of the track) * span.
		if = {
			limit = { pod_combat_actor_sup_ceiling_value > 0 }
			set_variable = {
				name = pod_combat_ai_heal_weight_superficial
				value = {
					value = pod_combat_actor_sup_ceiling_value
					subtract = pod_combat_actor_hp_value
					divide = pod_combat_actor_sup_ceiling_value
					multiply = pod_combat_ai_heal_span_value
					add = pod_combat_ai_heal_base_value
				}
			}
		}
		else = { set_variable = { name = pod_combat_ai_heal_weight_superficial  value = pod_combat_ai_heal_base_value } }
		if = {
			limit = { pod_combat_actor_max_hp_value > 0 }
			set_variable = {
				name = pod_combat_ai_heal_weight_aggravated
				value = {
					value = pod_combat_actor_max_hp_value
					subtract = pod_combat_actor_hp_value
					divide = pod_combat_actor_max_hp_value
					multiply = pod_combat_ai_heal_span_value
					add = pod_combat_ai_heal_base_value
				}
			}
		}
		else = { set_variable = { name = pod_combat_ai_heal_weight_aggravated  value = pod_combat_ai_heal_base_value } }

		var:pod_combat_actor_char = {
			save_scope_as = ai_actor

			# Personality profile precompute (fallback). The PRIMARY site is the
			# hoist in pod_combat_execute_ai_turn_effect, which runs this BEFORE the
			# first target selection (C13); this guarded copy is a no-op once that
			# fired and only computes here if the hoist somehow did not. Runs once per
			# combat (guard var on the AI char, cleared at the slot-init / group-
			# teardown AI cleanup) so the weight block only READS the bias vars.
			scope:ai_actor = {
				if = {
					limit = { NOT = { has_variable = pod_combat_pers_profile_done } }
					pod_combat_apply_personality_profile_effect = yes
					set_variable = pod_combat_pers_profile_done
				}
			}

			# Bonus-condition met-status precompute (before the random_in_list).
			# The @ai_bonus_condition_met weight modifier can't call the condition
			# trigger itself: a weight modifier can't bind a saved scope, and
			# prev/this re-anchor inside the trigger's own $ACTOR$ / target
			# scope-switches (so $STORY$.var reads there fetch the wrong scope).
			# Evaluate it here, where each move is a saved scope, and cache a flat
			# flag the modifier reads. (Same shape as the heal-weight precompute.)
			every_in_list = {
				variable = pod_combat_loadout
				limit = { has_variable = cond_mode  var:cond_mode = flag:bonus }
				pod_combat_ai_precompute_bonus_met_effect = { HOST = scope:combat_host  ACTOR = scope:ai_actor }
			}

			# Iteration scope = the candidate story. save_temporary_scope_as
			# binds it as ai_candidate_move so the trigger can still reach it
			# after we switch into scope:combat_host — named refs for both
			# story and actor keep its trait/workspace checks from confusing
			# which `prev` each piece means.
			random_in_list = {
				variable = pod_combat_loadout
				limit = {
					save_temporary_scope_as = ai_candidate_move
					scope:combat_host = {
						# Dynamic-only validation: loadout membership already
						# proves the unlock (seeded through the dispatcher at
						# fight finalize / mid-fight seat), so skip the
						# unlock switch per candidate.
						pod_combat_move_is_usable_now_trigger = {
							STORY = scope:ai_candidate_move
							ACTOR = scope:ai_actor
						}

						# A stolen (dominated) turn is single-target offense only:
						# team-relative AoE filters would hit the wrong side, and
						# healing the side being betrayed wastes the steal.
						trigger_if = {
							limit = { has_variable = pod_combat_turn_is_controlled }
							NOR = {
								scope:ai_candidate_move = { var:targeting = flag:team_enemy }
								scope:ai_candidate_move = { var:targeting = flag:team_ally }
								scope:ai_candidate_move = { var:targeting = flag:single_ally }
								# C28: single_pc hard-targets the PC (swap_to_pc forces player/0),
								# escaping the betrayal inversion. Latent today (no single_pc move
								# shipped) but excluded for correctness.
								scope:ai_candidate_move = { var:targeting = flag:single_pc }
								# G05: a dominated unit casting mind_control on a same-team target
								# would cascade the steal across the victim's own team. Stolen turns
								# are single-target OFFENSE only - no control re-propagation.
								scope:ai_candidate_move = { has_variable = mind_control }
							}
						}
						# Mind-control waste gate: never pick the move when the
						# current target can never be dominated (a player, or a
						# defended target this caster cannot bypass).
						trigger_if = {
							limit = { scope:ai_candidate_move = { has_variable = mind_control } }
							has_variable = pod_combat_target_char
							var:pod_combat_target_char = { is_ai = yes }
							trigger_if = {
								limit = { var:pod_combat_target_char = { POD_has_strong_mental_defenses_trigger = yes } }
								scope:ai_actor = { can_bypass_mental_defenses_trigger = yes }
							}
						}

						# Tier / stance / heal redundancy guards live in
						# pod_combat_move_is_usable_now_trigger (shared with
						# the PC validator); only AI-specific gates are inline.
						# Pure-heal redundancy — drop a heal-only move (heal
						# but no damage; lifesteal hybrids stay) when none of
						# its heals can do anything on the actor's current
						# state. AI-specific: broader limit than the
						# validator's self-targeting-only heal gate (covers
						# every targeting mode except single_ally), so NOT a
						# duplicate. Keep-condition mirrors the heal-when-hurt
						# weight modifiers below.
						trigger_if = {
							limit = {
								scope:ai_candidate_move = {
									OR = {
										has_variable = hp_heal
										has_variable = morale_heal
									}
									NOT = { has_variable = hp_damage }
									NOT = { has_variable = morale_damage }
									# single_ally / team_ally heals depend on the
									# TARGETED slots' state, not the actor's. single_ally
									# gates on ally-need in pod_combat_move_is_usable_now_trigger;
									# team_ally is gated by the team_ally trigger_if below
									# (which DOES include the actor, since a team heal hits it).
									NOT = { var:targeting = flag:single_ally }
									NOT = { var:targeting = flag:team_ally }
								}
							}
							OR = {
								# Superficial — useful below sup_ceiling.
								AND = {
									scope:ai_candidate_move = {
										has_variable = hp_heal
										var:heal_type = flag:superficial
									}
									pod_combat_actor_hp_value < pod_combat_actor_sup_ceiling_value
								}
								# Aggravated — useful for any wound (capped at max_hp).
								AND = {
									scope:ai_candidate_move = {
										has_variable = hp_heal
										var:heal_type = flag:aggravated
									}
									pod_combat_actor_hp_value < pod_combat_actor_max_hp_value
								}
								# Morale heal — useful below cap.
								AND = {
									scope:ai_candidate_move = { has_variable = morale_heal }
									var:pod_combat_actor_morale < pod_combat_max_morale
								}
							}
						}

						# team_ally pure-heal redundancy - a team heal hits the
						# whole team INCLUDING the actor, so keep it iff any alive
						# same-team slot (actor included) can benefit. Uses the
						# actor-inclusive team-heal-need trigger + the shared
						# per-slot predicate, so AI filtering matches the runtime team-ally
						# admit. One team scan per heal candidate per AI turn.
						trigger_if = {
							limit = {
								scope:ai_candidate_move = {
									OR = {
										has_variable = hp_heal
										has_variable = morale_heal
									}
									NOT = { has_variable = hp_damage }
									NOT = { has_variable = morale_damage }
									var:targeting = flag:team_ally
								}
							}
							pod_combat_ai_has_team_member_needing_heal_trigger = { STORY = scope:ai_candidate_move }
						}
					}
				}

				weight = {
					base = 0

					# Per-move base weight (8-15).
					modifier = {
						add = var:ai_weight
					}

					# Anti-repetition: three-deep history
					modifier = {
						add = @ai_repeat_last
						scope:ai_actor = { has_variable = pod_combat_ai_last_move }
						this = scope:ai_actor.var:pod_combat_ai_last_move
					}
					modifier = {
						add = @ai_repeat_prev
						scope:ai_actor = { has_variable = pod_combat_ai_prev_move }
						this = scope:ai_actor.var:pod_combat_ai_prev_move
					}
					modifier = {
						add = @ai_repeat_prev2
						scope:ai_actor = { has_variable = pod_combat_ai_prev2_move }
						this = scope:ai_actor.var:pod_combat_ai_prev2_move
					}
					# Conditional bonus would fire right now -> nudge toward triggering it
					# (the move's base effect is weighted separately). met-status is
					# precomputed per loadout move into the flag pod_combat_ai_bonus_met
					# (pod_combat_ai_precompute_bonus_met_effect, run just before this
					# random_in_list) because a weight modifier can't pass a saved scope to
					# the condition trigger and prev/this re-anchor inside the trigger's own
					# scope-switches; the modifier just reads the cached flat flag here.
					modifier = {
						add = @ai_bonus_condition_met
						has_variable = pod_combat_ai_bonus_met
					}

					# Heal when hurt, split by heal_type so the scaled weight only fires
					# where the heal can do something: superficial caps at
					# sup_ceiling, aggravated caps at max_hp. A heal with no
					# heal_type fires neither branch (safe default).
					modifier = {
						add = scope:combat_host.var:pod_combat_ai_heal_weight_superficial
						has_variable = hp_heal
						has_variable = heal_type
						var:heal_type = flag:superficial
						scope:combat_host = {
							pod_combat_actor_hp_value < pod_combat_actor_sup_ceiling_value
						}
					}
					modifier = {
						add = scope:combat_host.var:pod_combat_ai_heal_weight_aggravated
						has_variable = hp_heal
						has_variable = heal_type
						var:heal_type = flag:aggravated
						scope:combat_host = {
							pod_combat_actor_hp_value < pod_combat_actor_max_hp_value
						}
					}
					# B3 critical heal: actor at its last box and no finishable enemy to
					# kill instead -> stack an extra heal incentive over the scaled base heal
					# (so a dying AI tops up rather than trading a doomed strike). Gated
					# on hp_heal + heal_type so it rides the same heals the base bonus
					# does. enemy_finishable is precomputed once per turn in GROUP
					# (pod_combat_ai_flag_finishable_enemy_effect), not per candidate.
					# The NOT = { var:... = 1 } read is safe because the precompute
					# always writes the var (0 or 1) at turn start before move selection.
					modifier = {
						add = @ai_heal_critical
						has_variable = hp_heal
						has_variable = heal_type
						scope:combat_host = {
							pod_combat_actor_hp_value <= @ai_thresh_heal_critical
							NOT = { var:pod_combat_ai_enemy_finishable = 1 }
						}
					}
					# Morale heal when shaken.
					modifier = {
						add = @ai_morale_heal_low
						has_variable = morale_heal
						scope:combat_host = { var:pod_combat_actor_morale < @ai_thresh_morale_shaken }
					}
					# Proactive morale top-up in the pre-shaken band (shaken..early): top
					# up before it breaks, but smaller than the shaken bonus so a genuine
					# crisis still outweighs it.
					modifier = {
						add = @ai_morale_heal_early
						has_variable = morale_heal
						scope:combat_host = {
							var:pod_combat_actor_morale >= @ai_thresh_morale_shaken
							var:pod_combat_actor_morale <= @ai_thresh_morale_early
						}
					}
					# Guard when hurt.
					modifier = {
						add = @ai_defense_when_hurt
						has_variable = set_guard
						scope:combat_host = {
							pod_combat_actor_hp_value < pod_combat_actor_max_hp_value
						}
					}
					# dodge when hurt.
					modifier = {
						add = @ai_defense_when_hurt
						has_variable = dodge
						scope:combat_host = {
							pod_combat_actor_hp_value < pod_combat_actor_max_hp_value
						}
					}
					# stealth when hurt.
					modifier = {
						add = @ai_defense_when_hurt
						has_variable = stealth
						scope:combat_host = {
							pod_combat_actor_hp_value < pod_combat_actor_max_hp_value
						}
					}
					# Soak buff when hurt
					modifier = {
						add = @ai_defense_when_hurt
						OR = {
							has_variable = soak_buff_round_tier
							has_variable = soak_buff_combat_tier
						}
						scope:combat_host = {
							pod_combat_actor_hp_value < pod_combat_actor_max_hp_value
						}
					}
					# Hit buff when hit chance is low (≤50, below 60-base
					# parity). Upstream tier-redundancy guard already
					# ensures any candidate here is a real tier raise.
					modifier = {
						add = @ai_hitbuff_low_acc
						OR = {
							has_variable = hit_buff_round_tier
							has_variable = hit_buff_combat_tier
						}
						scope:combat_host = {
							var:pod_combat_actor_hit_chance <= @ai_thresh_hit_buff_band
						}
					}
					# Hit round-buff anti-spam. A hit buff only sharpens the
					# caster's OWN attacks and clears at their next turn-start
					# (before they act again), so it is worth at most this turn's
					# remaining actions. Soak round buffs are deliberately NOT
					# penalized: they now guard the whole enemy phase until the
					# holder's next turn, so the buff-when-needed bonus governs them.
					modifier = {
						add = @ai_round_buff_spam
						has_variable = hit_buff_round_tier
					}
					# Late-turn hit-buff penalty — a round hit buff with no AP
					# left to spend is wasted (expires unused). Threshold =
					# base AP − 1, i.e. anything past the first action.
					modifier = {
						add = @ai_round_buff_lateap
						has_variable = hit_buff_round_tier
						scope:combat_host = { var:pod_combat_current_ap <= @ai_thresh_late_ap }
					}

					# ── Tier-size preference ───────────────────────
					# Bias toward higher-tier buffs/debuffs. Tier vars hold
					# magnitude 1/2/3 (small/medium/large).
					modifier = {
						add = @ai_tier_small
						OR = {
							AND = { has_variable = soak_buff_round_tier    var:soak_buff_round_tier    = 1 }
							AND = { has_variable = soak_buff_combat_tier   var:soak_buff_combat_tier   = 1 }
							AND = { has_variable = soak_debuff_round_tier  var:soak_debuff_round_tier  = 1 }
							AND = { has_variable = soak_debuff_combat_tier var:soak_debuff_combat_tier = 1 }
							AND = { has_variable = hit_buff_round_tier     var:hit_buff_round_tier     = 1 }
							AND = { has_variable = hit_buff_combat_tier    var:hit_buff_combat_tier    = 1 }
							AND = { has_variable = hit_debuff_round_tier   var:hit_debuff_round_tier   = 1 }
							AND = { has_variable = hit_debuff_combat_tier  var:hit_debuff_combat_tier  = 1 }
						}
					}
					modifier = {
						add = @ai_tier_medium
						OR = {
							AND = { has_variable = soak_buff_round_tier    var:soak_buff_round_tier    = 2 }
							AND = { has_variable = soak_buff_combat_tier   var:soak_buff_combat_tier   = 2 }
							AND = { has_variable = soak_debuff_round_tier  var:soak_debuff_round_tier  = 2 }
							AND = { has_variable = soak_debuff_combat_tier var:soak_debuff_combat_tier = 2 }
							AND = { has_variable = hit_buff_round_tier     var:hit_buff_round_tier     = 2 }
							AND = { has_variable = hit_buff_combat_tier    var:hit_buff_combat_tier    = 2 }
							AND = { has_variable = hit_debuff_round_tier   var:hit_debuff_round_tier   = 2 }
							AND = { has_variable = hit_debuff_combat_tier  var:hit_debuff_combat_tier  = 2 }
						}
					}
					modifier = {
						add = @ai_tier_large
						OR = {
							AND = { has_variable = soak_buff_round_tier    var:soak_buff_round_tier    = 3 }
							AND = { has_variable = soak_buff_combat_tier   var:soak_buff_combat_tier   = 3 }
							AND = { has_variable = soak_debuff_round_tier  var:soak_debuff_round_tier  = 3 }
							AND = { has_variable = soak_debuff_combat_tier var:soak_debuff_combat_tier = 3 }
							AND = { has_variable = hit_buff_round_tier     var:hit_buff_round_tier     = 3 }
							AND = { has_variable = hit_buff_combat_tier    var:hit_buff_combat_tier    = 3 }
							AND = { has_variable = hit_debuff_round_tier   var:hit_debuff_round_tier   = 3 }
							AND = { has_variable = hit_debuff_combat_tier  var:hit_debuff_combat_tier  = 3 }
						}
					}

					# ── Undersized-buff downgrade ───────────────────
					# On top of the flat tier preference: if the candidate buff/debuff
					# has an affordable strictly-larger tier of the SAME stat in the
					# loadout this action, push the AI off the smaller one toward the
					# bigger version. Per-stat so the affordable-larger scan only looks
					# at the matching kind.
					modifier = {
						add = @ai_buff_undersized
						OR = {
							AND = { has_variable = soak_buff_round_tier    pod_combat_ai_has_affordable_larger_buff_tier_trigger = { STAT_VAR = soak_buff_round_tier } }
							AND = { has_variable = soak_buff_combat_tier   pod_combat_ai_has_affordable_larger_buff_tier_trigger = { STAT_VAR = soak_buff_combat_tier } }
							AND = { has_variable = soak_debuff_round_tier  pod_combat_ai_has_affordable_larger_buff_tier_trigger = { STAT_VAR = soak_debuff_round_tier } }
							AND = { has_variable = soak_debuff_combat_tier pod_combat_ai_has_affordable_larger_buff_tier_trigger = { STAT_VAR = soak_debuff_combat_tier } }
							AND = { has_variable = hit_buff_round_tier     pod_combat_ai_has_affordable_larger_buff_tier_trigger = { STAT_VAR = hit_buff_round_tier } }
							AND = { has_variable = hit_buff_combat_tier    pod_combat_ai_has_affordable_larger_buff_tier_trigger = { STAT_VAR = hit_buff_combat_tier } }
							AND = { has_variable = hit_debuff_round_tier   pod_combat_ai_has_affordable_larger_buff_tier_trigger = { STAT_VAR = hit_debuff_round_tier } }
							AND = { has_variable = hit_debuff_combat_tier  pod_combat_ai_has_affordable_larger_buff_tier_trigger = { STAT_VAR = hit_debuff_combat_tier } }
						}
					}

					# Remove-guard vs guarded target.
					modifier = {
						add = @ai_remove_guard_hit
						has_variable = remove_guard
						scope:combat_host = { has_variable = pod_combat_target_is_guarding }
					}
					modifier = {
						add = @ai_remove_guard_miss
						has_variable = remove_guard
						scope:combat_host = { NOT = { has_variable = pod_combat_target_is_guarding } }
					}

					# ── Target-aware modifiers ──────────────────────
					# Gated to single_enemy: only there is the weight-time
					# workspace target the eventual target. Other targeting
					# modes don't read it meaningfully.

					# Bypass-dodge bonus when target is dodging.
					modifier = {
						add = @ai_bypass_dodge
						has_variable = bypass_dodge
						var:targeting = flag:single_enemy
						scope:combat_host = { has_variable = pod_combat_target_is_dodging }
					}

					# Bypass-stealth bonus when target is stealthed.
					modifier = {
						add = @ai_bypass_stealth
						has_variable = bypass_stealth
						var:targeting = flag:single_enemy
						scope:combat_host = { has_variable = pod_combat_target_is_stealthed }
					}

					# Cash in stealth: while stealthed (+30 hit live), bias
					# toward damage so the window isn't spent on heals/buffs.
					modifier = {
						add = @ai_cash_in_stealth
						has_variable = hp_damage
						var:targeting = flag:single_enemy
						scope:combat_host = { has_variable = pod_combat_actor_is_stealthed }
					}

					# Aggravated vs high-soak (≥70%) target — aggravated
					# bypasses the soak roll.
					modifier = {
						add = @ai_aggravated_vs_soak
						has_variable = hp_damage
						has_variable = damage_type
						var:damage_type = flag:aggravated
						var:targeting = flag:single_enemy
						scope:combat_host = {
							var:pod_combat_target_soak >= @ai_thresh_high_soak
						}
					}

					# Complement of the aggravated bias above: a mundane superficial
					# attack (superficial damage_type, no elemental/special damage_source)
					# mostly bounces off a high-soak wall, so steer hard off it when the
					# actor has anything better. damage_type is always set on an hp_damage
					# move (POD_combat_move_set_hp_damage sets it), so the read is safe;
					# the weight floor keeps it pickable as a last resort.
					modifier = {
						add = @ai_superficial_vs_soak
						has_variable = hp_damage
						has_variable = damage_type
						var:damage_type = flag:superficial
						var:targeting = flag:single_enemy
						NOT = { has_variable = damage_source }
						scope:combat_host = {
							var:pod_combat_target_soak >= @ai_thresh_high_soak
						}
					}

					# Weakness exploit: damage_source matches the target's
					# splat weakness. Mapping lives in
					# pod_combat_damage_source_weakness_triggers.txt.
					modifier = {
						add = @ai_weakness_exploit
						has_variable = hp_damage
						has_variable = damage_source
						var:targeting = flag:single_enemy
						scope:combat_host = {
							has_variable = pod_combat_target_char
						}
						pod_combat_move_exploits_target_weakness_trigger = yes
					}

					# HP-finish tier 1: target at 1 box, any damage finishes.
					# Reads the wound track directly (no target_hp_value SV).
					modifier = {
						add = @ai_hp_finish_1box
						has_variable = hp_damage
						var:targeting = flag:single_enemy
						scope:combat_host = {
							has_variable = pod_combat_target_char
							var:pod_combat_target_char.POD_empty_damage_boxes <= @ai_thresh_hp_finish_1
						}
					}

					# HP-finish tier 2: target at 2 boxes and move deals ≥2.
					modifier = {
						add = @ai_hp_finish_2box
						has_variable = hp_damage
						var:hp_damage >= @ai_thresh_finish_damage
						var:targeting = flag:single_enemy
						scope:combat_host = {
							has_variable = pod_combat_target_char
							var:pod_combat_target_char.POD_empty_damage_boxes <= @ai_thresh_hp_finish_2
						}
					}

					# ===== PRIMITIVE SITUATIONALS (DoT / control / stun) =====
					# Keyed only on signals that exist. Target reads mirror the HP-finish
					# idiom above (scope:combat_host + pod_combat_target_char), and are
					# personality-independent (they fire for any actor holding the move).

					# DoT vs a target with room to burn: reward a >=2-round burn on a
					# single enemy that still has boxes to lose.
					modifier = {
						add = @ai_dot_healthy_target
						has_variable = dot_rounds
						var:dot_rounds >= 2
						var:targeting = flag:single_enemy
						scope:combat_host = {
							has_variable = pod_combat_target_char
							var:pod_combat_target_char.POD_empty_damage_boxes >= @ai_thresh_dot_healthy
						}
					}
					# DoT overkill: a ramp-up is wasted on a target a burst would finish
					# now. Reuse the 2-box finish threshold for "near dead".
					modifier = {
						add = @ai_dot_overkill_penalty
						has_variable = dot_amount
						var:targeting = flag:single_enemy
						scope:combat_host = {
							has_variable = pod_combat_target_char
							var:pod_combat_target_char.POD_empty_damage_boxes <= @ai_thresh_hp_finish_2
						}
					}
					# Mind control vs a dominatable AI. Pure positive weight: the
					# random_in_list limit already waste-gates players and
					# undominatable targets, so this only ever lands on a real steal.
					modifier = {
						add = @ai_mind_control_value
						has_variable = mind_control
						var:targeting = flag:single_enemy
						scope:combat_host = {
							has_variable = pod_combat_target_char
							var:pod_combat_target_char = { is_ai = yes }
						}
					}
					# Full stun (ap_drain >= @ai_thresh_stun) vs an AI with a turn to lose.
					# No target-AP read exists, so gate on is_ai only.
					modifier = {
						add = @ai_stun_value
						has_variable = ap_drain
						var:ap_drain >= @ai_thresh_stun
						var:targeting = flag:single_enemy
						scope:combat_host = {
							has_variable = pod_combat_target_char
							var:pod_combat_target_char = { is_ai = yes }
						}
					}
					# Partial ap_drain (< @ai_thresh_stun): small flat bonus vs an AI.
					modifier = {
						add = @ai_ap_drain_value
						has_variable = ap_drain
						var:ap_drain < @ai_thresh_stun
						var:targeting = flag:single_enemy
						scope:combat_host = {
							has_variable = pod_combat_target_char
							var:pod_combat_target_char = { is_ai = yes }
						}
					}

					# ===== SUMMON + RETALIATE SITUATIONALS =====
					# Both flags are presence-only move vars (summons_combatant /
					# retaliate_if_hit), so has_variable alone gates them.

					# Summon while outnumbered. Summon moves are self-targeted, so NO
					# single_enemy gate. Outnumbered signal: no per-team alive-count
					# exists and a clean two-pass count per candidate is too costly
					# here, so use the "a teammate has fallen" proxy - one
					# short-circuiting any_in_list over the host's pod_combat_active_slots
					# for a same-team KO'd slot. The actor's team is read back through
					# scope:combat_host inside the iteration (the ally-target-swap idiom).
					modifier = {
						add = @ai_summon_outnumbered
						has_variable = summons_combatant
						scope:combat_host = {
							any_in_list = {
								variable = pod_combat_active_slots
								var:combat_team = scope:combat_host.var:pod_combat_current_actor_team
								pod_combat_slot_is_ko_trigger = yes
							}
						}
					}
					# Retaliate while exposed: a counter pays off only if the actor
					# can still be struck. Reuse the exact @ai_defense_when_hurt
					# actor-HP-below-max idiom (host-scoped SV compare).
					modifier = {
						add = @ai_retaliate_when_exposed
						has_variable = retaliate_if_hit
						scope:combat_host = {
							pod_combat_actor_hp_value < pod_combat_actor_max_hp_value
						}
					}

					# ===== scenario sharpening =====
					# All additive nudges (constants in the header block). Target/host
					# workspace reads (incorporeal, hit_chance, AP, energy_used) are all
					# loaded at scoring time by the group load helper.

					# Spirit gate: a physical attack auto-whiffs an incorporeal target
					# unless the actor can damage spirits. Steer hard off the wasted swing.
					modifier = {
						add = @ai_spirit_gate_wasted
						has_variable = hp_damage
						has_variable = checks_spirit_gate
						var:targeting = flag:single_enemy
						scope:combat_host = {
							has_variable = pod_combat_target_incorporeal_spirit
							NOT = { has_variable = pod_combat_actor_can_damage_spirit }
						}
					}
					# ...and toward a damage move that CAN connect with it (either the
					# move bypasses the gate, or the actor can hurt spirits).
					modifier = {
						add = @ai_spirit_gate_viable
						has_variable = hp_damage
						var:targeting = flag:single_enemy
						scope:combat_host = { has_variable = pod_combat_target_incorporeal_spirit }
						OR = {
							NOT = { has_variable = checks_spirit_gate }
							scope:combat_host = { has_variable = pod_combat_actor_can_damage_spirit }
						}
					}
					# AoE pays off with multiple live targets. No per-team alive count
					# exists, so count enemy-of-actor live slots inline (count >= 2),
					# reusing the summon block's team-compare idiom.
					modifier = {
						add = @ai_aoe_multi_enemy
						has_variable = hp_damage
						var:targeting = flag:team_enemy
						scope:combat_host = {
							any_in_list = {
								variable = pod_combat_active_slots
								count >= 2
								NOT = { var:combat_team = scope:combat_host.var:pod_combat_current_actor_team }
								pod_combat_slot_is_alive_trigger = yes
							}
						}
					}
					# Drag a genuinely accurate enemy's hit chance down (debuff-side
					# mirror of the self hit-buff-when-low bonus).
					modifier = {
						add = @ai_hitdebuff_advantage
						OR = { has_variable = hit_debuff_round_tier  has_variable = hit_debuff_combat_tier }
						var:targeting = flag:single_enemy
						scope:combat_host = { has_variable = pod_combat_target_hit_chance  var:pod_combat_target_hit_chance >= @ai_thresh_hit_buff_band }
					}
					# A combat-long buff/debuff outlasts a round one, so reward setting
					# it up early (before much energy is spent).
					modifier = {
						add = @ai_combat_buff_reward
						OR = {
							has_variable = soak_buff_combat_tier
							has_variable = hit_buff_combat_tier
							has_variable = soak_debuff_combat_tier
							has_variable = hit_debuff_combat_tier
						}
						scope:combat_host = { var:pod_combat_actor_energy_used < 2 }
					}
					# Low on AP: press affordable damage rather than spend the dwindling
					# action budget on utility (candidates are already affordability-gated,
					# so the surviving damage here is the cheap kind).
					modifier = {
						add = @ai_cheap_finish_tempo
						has_variable = hp_damage
						scope:combat_host = { var:pod_combat_current_ap <= @ai_thresh_late_ap }
					}
					# Conversely, avoid committing a heavy move when AP is tight enough
					# that it would strand the rest of the turn.
					modifier = {
						add = @ai_energy_high_cost_lowap
						has_variable = energy_cost
						var:energy_cost >= 3
						scope:combat_host = { var:pod_combat_current_ap <= @ai_thresh_lowap_expensive }
					}
					# Control as defense: when wounded and facing an accurate threat,
					# disabling it (dominate / drain AP) is worth the turn. Target AP is
					# untracked, so accuracy stands in for "dangerous" (cf. is_ai gate above).
					modifier = {
						add = @ai_control_defend
						OR = { has_variable = mind_control  has_variable = ap_drain }
						var:targeting = flag:single_enemy
						scope:combat_host = {
							pod_combat_actor_hp_value < pod_combat_actor_max_hp_value
							has_variable = pod_combat_target_hit_chance
							var:pod_combat_target_hit_chance >= @ai_thresh_hit_buff_band
						}
					}

					# Morale-finish tier 1: target morale ≤2, any morale damage finishes.
					modifier = {
						add = @ai_mor_finish_low
						has_variable = morale_damage
						var:targeting = flag:single_enemy
						scope:combat_host = {
							var:pod_combat_target_morale <= @ai_thresh_mor_finish_low
						}
					}

					# Morale-finish tier 2: target morale ≤4 and move deals ≥2.
					modifier = {
						add = @ai_mor_finish_mid
						has_variable = morale_damage
						var:morale_damage >= @ai_thresh_finish_morale
						var:targeting = flag:single_enemy
						scope:combat_host = {
							var:pod_combat_target_morale <= @ai_thresh_mor_finish_mid
						}
					}

					# Low-hit penalty (hit_chance ≤30). requires_hit_roll gate
					# spares auto-hit moves, which can't whiff.
					modifier = {
						add = @ai_hit_penalty_low
						has_variable = hp_damage
						has_variable = requires_hit_roll
						var:targeting = flag:single_enemy
						scope:combat_host = {
							var:pod_combat_actor_hit_chance <= @ai_thresh_hit_low
						}
					}

					# Mid-hit penalty: softer, 31-40% band, same gate.
					modifier = {
						add = @ai_hit_penalty_mid
						has_variable = hp_damage
						has_variable = requires_hit_roll
						var:targeting = flag:single_enemy
						scope:combat_host = {
							var:pod_combat_actor_hit_chance <= @ai_thresh_hit_mid
						}
					}

					# Energy attrition — four stacking steps -8/-10/-12/-12
					# (escalating then flat at step 4; see @ai_energy_step4),
					# so cumulative at 0/1/2/3/4+ energy used is
					# 0/-8/-18/-30/-42.
					modifier = {
						add = @ai_energy_step1
						has_variable = energy_cost
						scope:combat_host = { var:pod_combat_actor_energy_used >= 1 }
					}
					modifier = {
						add = @ai_energy_step2
						has_variable = energy_cost
						scope:combat_host = { var:pod_combat_actor_energy_used >= 2 }
					}
					modifier = {
						add = @ai_energy_step3
						has_variable = energy_cost
						scope:combat_host = { var:pod_combat_actor_energy_used >= 3 }
					}
					modifier = {
						add = @ai_energy_step4
						has_variable = energy_cost
						scope:combat_host = { var:pod_combat_actor_energy_used >= 4 }
					}

					# Trait-gate bonus — nudge class moves (disciplines,
					# gifts, hekau…) above generic physicals. Non-starter is
					# the proxy: only gated moves lack is_starter_move.
					modifier = {
						add = @ai_trait_gated
						NOT = { has_variable = is_starter_move }
					}

					# ===== PERSONALITY BAND =====
					# Reads the precomputed scope:ai_actor bias vars and dispatches
					# on the CANDIDATE's own combat_category flag and primitive
					# has_variable s - no has_trait at candidate time. Suppressed on
					# a stolen (dominated) turn so the dominator intent drives the
					# steal, not the victim temperament; the control flag is
					# HOST-scoped.
					modifier = {
						add = scope:ai_actor.var:pod_combat_pers_bias_atk
						var:combat_category = flag:ATK
						scope:combat_host = { NOT = { has_variable = pod_combat_turn_is_controlled } }
					}
					modifier = {
						add = scope:ai_actor.var:pod_combat_pers_bias_def
						var:combat_category = flag:DEF
						scope:combat_host = { NOT = { has_variable = pod_combat_turn_is_controlled } }
					}
					# De-dup: a control PRIMITIVE candidate reads _control, not its
					# CTL category bias. So suppress _ctl on mind_control / ap_drain
					# moves to avoid double-counting.
					modifier = {
						add = scope:ai_actor.var:pod_combat_pers_bias_ctl
						var:combat_category = flag:CTL
						NOT = { has_variable = mind_control }
						NOT = { has_variable = ap_drain }
						scope:combat_host = { NOT = { has_variable = pod_combat_turn_is_controlled } }
					}
					modifier = {
						add = scope:ai_actor.var:pod_combat_pers_bias_pwr
						var:combat_category = flag:PWR
						scope:combat_host = { NOT = { has_variable = pod_combat_turn_is_controlled } }
					}
					modifier = {
						add = scope:ai_actor.var:pod_combat_pers_bias_dot
						has_variable = dot_rounds
						scope:combat_host = { NOT = { has_variable = pod_combat_turn_is_controlled } }
					}
					# _control covers BOTH control primitives (one modifier each,
					# so a both-flag move reads it twice - intentional; the var is
					# clamped at +/-10).
					modifier = {
						add = scope:ai_actor.var:pod_combat_pers_bias_control
						has_variable = mind_control
						scope:combat_host = { NOT = { has_variable = pod_combat_turn_is_controlled } }
					}
					modifier = {
						add = scope:ai_actor.var:pod_combat_pers_bias_control
						has_variable = ap_drain
						scope:combat_host = { NOT = { has_variable = pod_combat_turn_is_controlled } }
					}
					# Savage compounds ATK for a berserker (intentional, bounded by
					# the +8 clamp); gates on aggravated damage_type. The
					# has_variable guard MUST precede the flag compare: damage_type
					# is set only on damaging moves, so a bare var: read errors on
					# buff/heal/utility candidates (AND short-circuits in order).
					modifier = {
						add = scope:ai_actor.var:pod_combat_pers_bias_savage
						has_variable = damage_type
						var:damage_type = flag:aggravated
						scope:combat_host = { NOT = { has_variable = pod_combat_turn_is_controlled } }
					}

					# Floor: random_in_list silently drops items with weight
					# ≤ 0, so a move buried by situational penalties would
					# become unpickable. Floor keeps it as a fallback.
					min = @ai_weight_floor
				}

				save_scope_as = chosen_move
			}
		}

		# Execute on combat_host — that's where the workspace lives. Inside
		# var:pod_combat_actor_char the AI would "act" with no effect.
		if = {
			limit = {
				exists = scope:chosen_move
				scope:combat_host = {
					# Dynamic-only recheck: same unlock proof as the selector limit.
					pod_combat_move_is_usable_now_trigger = {
						STORY = scope:chosen_move
						ACTOR = scope:ai_actor
					}
				}
			}
			pod_combat_ai_swap_for_story     = { STORY = scope:chosen_move }
			POD_execute_combat_move           = { STORY = scope:chosen_move }
			pod_combat_ai_restore_after_move = { STORY = scope:chosen_move }

			# Record anti-repetition only after the move actually executed
			# (inside the same gate), else a stale chosen_move would pollute
			# history. Shift the 3-deep ring reading each older slot before
			# overwriting it.
			scope:ai_actor = {
				if = {
					limit = { has_variable = pod_combat_ai_prev_move }
					set_variable = {
						name  = pod_combat_ai_prev2_move
						value = var:pod_combat_ai_prev_move
					}
				}
				if = {
					limit = { has_variable = pod_combat_ai_last_move }
					set_variable = {
						name  = pod_combat_ai_prev_move
						value = var:pod_combat_ai_last_move
					}
				}
				set_variable = {
					name  = pod_combat_ai_last_move
					value = scope:chosen_move
				}
			}
		}
	}
}

pod_combat_ai_ensure_loadout_effect = {
	# Empty-list BACKSTOP only; primary seeding is eager, at fight setup
	# (POD_combat_finalize_duel_effect, after the fera warform pre-pass) and
	# at mid-fight seat (pod_combat_init_one_midfight_slot_effect). AI
	# loadouts are TRANSIENT per fight (cleared at slot init and combat
	# cleanup); if a path reaches the selector unseeded anyway, seed here
	# (count>0 no-ops). AI gets its full trait-appropriate repertoire (no
	# 4-per-category cap). Seeding always filters through the unlock
	# dispatcher (POD_apply_ai_full_loadout), which is what lets the
	# selector validate picks with the dynamic-only trigger. PCs are seeded
	# by the configure decision and skip the slot-init clear (slot 0), so
	# their equipped list survives.
	if = {
		limit = {
			has_variable = pod_combat_actor_char
		}
		var:pod_combat_actor_char = {
			if = {
				limit = { pod_combat_equipped_count = 0 }
				POD_apply_ai_full_loadout = yes
			}
		}
	}
}

pod_combat_ai_swap_for_story = {
	# $STORY$ = the chosen move. Only single_ally / single_pc need a
	# target swap; other modes keep the turn-start enemy or self-iterate.
	if = {
		limit = { $STORY$ = { var:targeting = flag:single_ally } }
		pod_combat_ai_swap_to_ally_target_effect = { STORY = $STORY$ }
	}
	else_if = {
		limit = { $STORY$ = { var:targeting = flag:single_pc } }
		pod_combat_ai_swap_to_pc_target_effect = yes
	}
	# else: no swap (AoE modes iterate slots in the execution engine).
}

pod_combat_ai_restore_after_move = {
	# $STORY$ = the chosen move. Undoes the swap so the next action aims
	# at the turn-start enemy again. Both swaps save the enemy target, so
	# gating on its presence is a safe blanket restore.
	if = {
		limit = {
			OR = {
				$STORY$ = { var:targeting = flag:single_ally }
				$STORY$ = { var:targeting = flag:single_pc }
			}
			has_variable = pod_combat_ai_saved_enemy_target_team
		}
		pod_combat_ai_restore_enemy_target_effect = yes
	}
}

# ─── SINGLE-ALLY TARGET SWAP ──────────────────────────────────────
# Saves the enemy target, picks an alive non-self same-team ally, loads
# it into the target workspace. The selector already rejects single_ally
# when no ally is available (pod_combat_ai_has_available_ally_trigger),
# so a match here is guaranteed.

pod_combat_ai_swap_to_ally_target_effect = {
	# $STORY$ = the single_ally move being executed (used by the G04 heal-useful
	# gate in the limit below); passed by pod_combat_ai_swap_for_story.
	# Save selected_target so restore_enemy_target can put it back.
	set_variable = { name = pod_combat_ai_saved_enemy_target_team     value = var:pod_combat_selected_target_team }
	set_variable = { name = pod_combat_ai_saved_enemy_target_position value = var:pod_combat_selected_target_position }

	# Pick any alive non-self same-team ally into the target workspace.
	save_scope_as = host
	random_in_list = {
		variable = pod_combat_active_slots
		limit = {
			pod_combat_slot_is_alive_trigger = yes
			# not the actor itself (same team + same position)
			NOT = {
				AND = {
					var:combat_team = scope:host.var:pod_combat_current_actor_team
					var:team_position = scope:host.var:pod_combat_current_actor_position
				}
			}
			# same team as the actor
			var:combat_team = scope:host.var:pod_combat_current_actor_team
			# G04: for a single_ally HEAL move, admit only allies the heal can
			# ACTUALLY help (heal-type-aware), so the most-wounded weight below can't
			# steer a superficial heal onto an aggravated-only ally (heals 0). A
			# non-heal single_ally move (no hp_heal/morale_heal) skips this gate and
			# admits any alive ally as before. Matches the selector's C05 heal-need
			# gate, so for a heal admitted there a helpable ally is guaranteed here.
			trigger_if = {
				limit = {
					$STORY$ = {
						OR = {
							has_variable = hp_heal
							has_variable = morale_heal
						}
					}
				}
				pod_combat_slot_heal_is_useful_trigger = { SLOT = this STORY = $STORY$ }
			}
		}
		weight = {
			base = 1
			modifier = { add = 50 }
			# Prefer the most-wounded ally: same low-HP factor the enemy/player
			# target selectors use, so heals land where they matter most.
			modifier = {
				factor = 2
				var:slot_char.POD_empty_damage_boxes <= 2
			}
		}
		save_scope_as = chosen_target
	}
	# `this` is chosen_target here, so the load runs without an explicit
	# scope-switch; host/chosen_target stay resolvable for the write.
	scope:chosen_target ?= {
		scope:host = {
			set_variable = { name = pod_combat_selected_target_team     value = scope:chosen_target.var:combat_team }
			set_variable = { name = pod_combat_selected_target_position value = scope:chosen_target.var:team_position }
		}
		pod_combat_load_target_from_story_effect = yes
	}
}

pod_combat_ai_swap_to_pc_target_effect = {
	# Save the current selected_target so restore_enemy_target can put it back
	set_variable = { name = pod_combat_ai_saved_enemy_target_team     value = var:pod_combat_selected_target_team }
	set_variable = { name = pod_combat_ai_saved_enemy_target_position value = var:pod_combat_selected_target_position }

	# Swap to player/0 (the PC) and load into the target workspace.
	if = {
		limit = { has_variable = pod_combat_pc_slot var:pod_combat_pc_slot = { has_variable = slot_char } }
		set_variable = { name = pod_combat_selected_target_team     value = flag:player }
		set_variable = { name = pod_combat_selected_target_position value = 0 }
		save_scope_as = host
		var:pod_combat_pc_slot = {
			pod_combat_load_target_from_story_effect = yes
		}
	}
}

pod_combat_ai_restore_enemy_target_effect = {
	# Restore the saved enemy target into the workspace so the rest of the
	# turn aims at it again.
	#
	# CRITICAL: save the ally slot FIRST — the workspace still holds the
	# buffed ally the move just wrote, and reloading the enemy without
	# saving would overwrite (erase) that buff with stale slot data.
	if = {
		limit = { has_variable = pod_combat_ai_saved_enemy_target_team }
		save_scope_as = host
		# Save the ally workspace back to its slot before we reload the enemy
		every_in_list = {
			variable = pod_combat_active_slots
			limit = { var:combat_team = scope:host.var:pod_combat_selected_target_team var:team_position = scope:host.var:pod_combat_selected_target_position }
			pod_combat_save_target_to_story_effect = yes
		}

		# Now swap selected_target back to the enemy and reload that slot's data
		set_variable = { name = pod_combat_selected_target_team     value = var:pod_combat_ai_saved_enemy_target_team }
		set_variable = { name = pod_combat_selected_target_position value = var:pod_combat_ai_saved_enemy_target_position }
		every_in_list = {
			variable = pod_combat_active_slots
			limit = { var:combat_team = scope:host.var:pod_combat_selected_target_team var:team_position = scope:host.var:pod_combat_selected_target_position }
			pod_combat_load_target_from_story_effect = yes
		}
		remove_variable = pod_combat_ai_saved_enemy_target_team
		remove_variable = pod_combat_ai_saved_enemy_target_position
	}
}
