# Unified AI flow — every combat (1v1 and NvN alike) runs through the
# workspace bridge and picks moves from the CSV-generated selector in
# pod_combat_ai_generated_effects.txt. There is no longer a legacy
# flipped-side selector; a 1v1 is just a degenerate NvN with slots 0 and 5
# populated and the same workspace-relative move effects work correctly.

pod_combat_ai_take_turn_effect = {
	# AI acts up to 6 times (max AP=6 / min cost=1)
	# Each call is a no-op if AP < 1 or combat is over
	pod_combat_ai_try_action_effect = yes
	pod_combat_ai_try_action_effect = yes
	pod_combat_ai_try_action_effect = yes
	pod_combat_ai_try_action_effect = yes
	pod_combat_ai_try_action_effect = yes
	pod_combat_ai_try_action_effect = yes
}

pod_combat_ai_try_action_effect = {
	if = {
		limit = {
			pod_combat_current_ap_value >= 1
			NOT = { pod_combat_is_over_trigger = yes }
		}
		pod_combat_ai_select_and_execute_move_effect = yes
	}
}

pod_combat_ai_select_and_execute_move_effect = {
	# All AI decisions go through the CSV-generated selector, which uses
	# pod_combat_can_use_{key}_trigger gates + trait checks against the
	# current actor and reuses the standard pod_combat_use_{key}_effect
	# move effects. See pod_combat_ai_generated_effects.txt for the body.
	pod_combat_ai_csv_select_effect = yes
}

# ─── SINGLE-ALLY TARGET SWAP ──────────────────────────────────────
# Fired by the AI select effect right before a single-ally move body runs.
# Saves the AI's current enemy target (selected_target, set at AI turn
# start by pod_combat_ai_select_{enemy,player}_target_effect), picks the
# first alive non-self same-team ally, and loads that ally into the
# opponent workspace so the move effect's writes land on the right slot.
#
# The AI select generator emits pod_combat_ai_has_available_ally_trigger
# as a trigger gate, so this effect is only called when at least one
# alive non-self ally exists on the actor's team. If the gate is working
# correctly, the if/else_if chain below always hits a match.

pod_combat_ai_swap_to_ally_target_effect = {
	# Save the current selected_target so pod_combat_ai_restore_enemy_target_effect
	# can put it back after the single-ally move fires.
	set_variable = { name = pod_combat_ai_saved_enemy_target value = var:pod_combat_selected_target }

	# Pick the first alive non-self same-team ally and load them into the
	# opponent workspace. Team membership is fixed by slot index: slots 0-4
	# are team 0 (player team), 5-9 are team 1 (enemy team), so "same team
	# as actor" is expressed via a compile-time gate on pod_combat_current_actor.
	if = {
		limit = {
			has_variable = pod_combat_c0_char
			var:pod_combat_c0_alive = 1
			var:pod_combat_current_actor <= 4
			NOT = { var:pod_combat_current_actor = 0 }
		}
		set_variable = { name = pod_combat_selected_target value = 0 }
		pod_combat_load_target_from_slot_effect = { SLOT = 0 }
	}
	else_if = {
		limit = {
			has_variable = pod_combat_c1_char
			var:pod_combat_c1_alive = 1
			var:pod_combat_current_actor <= 4
			NOT = { var:pod_combat_current_actor = 1 }
		}
		set_variable = { name = pod_combat_selected_target value = 1 }
		pod_combat_load_target_from_slot_effect = { SLOT = 1 }
	}
	else_if = {
		limit = {
			has_variable = pod_combat_c2_char
			var:pod_combat_c2_alive = 1
			var:pod_combat_current_actor <= 4
			NOT = { var:pod_combat_current_actor = 2 }
		}
		set_variable = { name = pod_combat_selected_target value = 2 }
		pod_combat_load_target_from_slot_effect = { SLOT = 2 }
	}
	else_if = {
		limit = {
			has_variable = pod_combat_c3_char
			var:pod_combat_c3_alive = 1
			var:pod_combat_current_actor <= 4
			NOT = { var:pod_combat_current_actor = 3 }
		}
		set_variable = { name = pod_combat_selected_target value = 3 }
		pod_combat_load_target_from_slot_effect = { SLOT = 3 }
	}
	else_if = {
		limit = {
			has_variable = pod_combat_c4_char
			var:pod_combat_c4_alive = 1
			var:pod_combat_current_actor <= 4
			NOT = { var:pod_combat_current_actor = 4 }
		}
		set_variable = { name = pod_combat_selected_target value = 4 }
		pod_combat_load_target_from_slot_effect = { SLOT = 4 }
	}
	else_if = {
		limit = {
			has_variable = pod_combat_c5_char
			var:pod_combat_c5_alive = 1
			var:pod_combat_current_actor >= 5
			NOT = { var:pod_combat_current_actor = 5 }
		}
		set_variable = { name = pod_combat_selected_target value = 5 }
		pod_combat_load_target_from_slot_effect = { SLOT = 5 }
	}
	else_if = {
		limit = {
			has_variable = pod_combat_c6_char
			var:pod_combat_c6_alive = 1
			var:pod_combat_current_actor >= 5
			NOT = { var:pod_combat_current_actor = 6 }
		}
		set_variable = { name = pod_combat_selected_target value = 6 }
		pod_combat_load_target_from_slot_effect = { SLOT = 6 }
	}
	else_if = {
		limit = {
			has_variable = pod_combat_c7_char
			var:pod_combat_c7_alive = 1
			var:pod_combat_current_actor >= 5
			NOT = { var:pod_combat_current_actor = 7 }
		}
		set_variable = { name = pod_combat_selected_target value = 7 }
		pod_combat_load_target_from_slot_effect = { SLOT = 7 }
	}
	else_if = {

# ─── SINGLE-PC TARGET SWAP ────────────────────────────────────────
# Fired by the AI select effect right before a single-pc move body runs.
# Saves the AI's current enemy target and loads slot 0 (the PC) into the
# opponent workspace so the move effect applies to the PC. After the
# move fires, pod_combat_ai_restore_enemy_target_effect puts the enemy
# target back (the same restore effect used by single-ally).
#
# Unlike the ally swap, there's no slot iteration — slot 0 is always
# the PC, always populated during combat, and always valid as a target
# for single-pc skills. The defensive `has_variable = pod_combat_c0_char`
# guard is redundant but harmless if combat is improperly initialized.

pod_combat_ai_swap_to_pc_target_effect = {
	# Save the current selected_target so restore_enemy_target can put it back
	set_variable = { name = pod_combat_ai_saved_enemy_target value = var:pod_combat_selected_target }

	# Swap to slot 0 (the PC) and load into the opponent workspace
	if = {
		limit = { has_variable = pod_combat_c0_char }
		set_variable = { name = pod_combat_selected_target value = 0 }
		pod_combat_load_target_from_slot_effect = { SLOT = 0 }
	}
}

pod_combat_ai_restore_enemy_target_effect = {
	# Restore the enemy target saved by pod_combat_ai_swap_to_ally_target_effect
	# and reload it into the opponent workspace. Called after each single-ally
	# move fires so subsequent AI actions in the same turn continue to aim
	# at the AI's original chosen enemy.
	#
	# CRITICAL: Save the ally slot first. The single-ally move effect wrote
	# buffs into the opponent workspace, and the workspace currently holds
	# the buffed ally. If we reload the enemy slot without saving first, the
	# buff would be overwritten by the stale slot data and effectively erased.
	# pod_combat_save_target_to_slot_effect uses pod_combat_selected_target
	# to pick the destination slot, which still points at the ally at this point.
	if = {
		limit = { has_variable = pod_combat_ai_saved_enemy_target }
		# Save the ally workspace back to its slot before we reload the enemy
		if = { limit = { var:pod_combat_selected_target = 0 } pod_combat_save_target_to_slot_effect = { SLOT = 0 } }
		else_if = { limit = { var:pod_combat_selected_target = 1 } pod_combat_save_target_to_slot_effect = { SLOT = 1 } }
		else_if = { limit = { var:pod_combat_selected_target = 2 } pod_combat_save_target_to_slot_effect = { SLOT = 2 } }
		else_if = { limit = { var:pod_combat_selected_target = 3 } pod_combat_save_target_to_slot_effect = { SLOT = 3 } }
		else_if = { limit = { var:pod_combat_selected_target = 4 } pod_combat_save_target_to_slot_effect = { SLOT = 4 } }
		else_if = { limit = { var:pod_combat_selected_target = 5 } pod_combat_save_target_to_slot_effect = { SLOT = 5 } }
		else_if = { limit = { var:pod_combat_selected_target = 6 } pod_combat_save_target_to_slot_effect = { SLOT = 6 } }
		else_if = { limit = { var:pod_combat_selected_target = 7 } pod_combat_save_target_to_slot_effect = { SLOT = 7 } }
		else_if = { limit = { var:pod_combat_selected_target = 8 } pod_combat_save_target_to_slot_effect = { SLOT = 8 } }
		else_if = { limit = { var:pod_combat_selected_target = 9 } pod_combat_save_target_to_slot_effect = { SLOT = 9 } }

		# Now swap selected_target back to the enemy and reload that slot's data
		set_variable = { name = pod_combat_selected_target value = var:pod_combat_ai_saved_enemy_target }
		if = { limit = { var:pod_combat_selected_target = 0 } pod_combat_load_target_from_slot_effect = { SLOT = 0 } }
		else_if = { limit = { var:pod_combat_selected_target = 1 } pod_combat_load_target_from_slot_effect = { SLOT = 1 } }
		else_if = { limit = { var:pod_combat_selected_target = 2 } pod_combat_load_target_from_slot_effect = { SLOT = 2 } }
		else_if = { limit = { var:pod_combat_selected_target = 3 } pod_combat_load_target_from_slot_effect = { SLOT = 3 } }
		else_if = { limit = { var:pod_combat_selected_target = 4 } pod_combat_load_target_from_slot_effect = { SLOT = 4 } }
		else_if = { limit = { var:pod_combat_selected_target = 5 } pod_combat_load_target_from_slot_effect = { SLOT = 5 } }
		else_if = { limit = { var:pod_combat_selected_target = 6 } pod_combat_load_target_from_slot_effect = { SLOT = 6 } }
		else_if = { limit = { var:pod_combat_selected_target = 7 } pod_combat_load_target_from_slot_effect = { SLOT = 7 } }
		else_if = { limit = { var:pod_combat_selected_target = 8 } pod_combat_load_target_from_slot_effect = { SLOT = 8 } }
		else_if = { limit = { var:pod_combat_selected_target = 9 } pod_combat_load_target_from_slot_effect = { SLOT = 9 } }
		remove_variable = pod_combat_ai_saved_enemy_target
	}
}
