# Weapon overrides for static and conditional uses_weapon moves. Profiles live
# on the fight for execution phases 2c-17; shared move stories must not hold
# per-combatant state. See COMBAT_REFERENCE.info section 2.

# Host scope, once per action. $STORY$ = the move story.
# Clear before testing eligibility so a failed conditional swing cannot reuse
# the previous action's profile.
pod_combat_seed_actor_weapon_profile_effect = {
	var:pod_combat_session ?= { remove_variable = pod_combat_weapon_class }
	var:pod_combat_session ?= { remove_variable = pod_combat_weapon_bonus_damage }
	var:pod_combat_session ?= { remove_variable = pod_combat_weapon_undodgeable }
	var:pod_combat_session ?= { remove_variable = pod_combat_weapon_heal_on_damage }
	var:pod_combat_session ?= { remove_variable = pod_combat_weapon_energy_on_damage }
	var:pod_combat_session ?= { remove_variable = pod_combat_weapon_soak_penalty }

	if = {
		limit = {
			pod_combat_move_swings_weapon_trigger = { STORY = $STORY$ }
			var:pod_combat_session ?= { has_variable = pod_combat_actor_char }
			var:pod_combat_session.var:pod_combat_actor_char ?= { pod_combat_actor_is_armed_trigger = yes }
		}
		# Refresh the scope each action so nested reactions cannot reuse
		# another actor's weapon.
		var:pod_combat_session.var:pod_combat_actor_char ?= {
			random_equipped_character_artifact = {
				limit = {
					artifact_slot_type = primary_armament
					prev = { can_benefit_from_artifact = prev }
				}
				save_scope_as = pod_combat_weapon
			}
		}

		# Weapon retype applies only to the superficial damage; move elements
		# and aggravated take precedence.
		if = {
			limit = {
				# Keep the damage_type read nested: OR evaluates it even when
				# the variable is absent.
				$STORY$ = {
					trigger_if = {
						limit = { has_variable = damage_type }
						var:damage_type = flag:superficial
					}
					trigger_else = { always = yes }
				}
			}
			# Only damage class may fall back to another equipped artifact, without a
			# weapon-type or slot filter. Other overrides come from
			# primary_armament. Resolve each source immediately before reading
			# it to avoid stale saved scopes.
			if = {
				limit = { scope:pod_combat_weapon = { has_variable = artifact_weapon_damage_class } }
				pod_combat_read_weapon_class_effect = { SOURCE = scope:pod_combat_weapon }
			}
			else_if = {
				limit = {
					var:pod_combat_session.var:pod_combat_actor_char ?= {
						any_equipped_character_artifact = {
							prev = { can_benefit_from_artifact = prev }
							has_variable = artifact_weapon_damage_class
						}
					}
				}
				var:pod_combat_session.var:pod_combat_actor_char ?= {
					random_equipped_character_artifact = {
						limit = {
							prev = { can_benefit_from_artifact = prev }
							has_variable = artifact_weapon_damage_class
						}
						save_scope_as = pod_combat_class_source
					}
				}
				pod_combat_read_weapon_class_effect = { SOURCE = scope:pod_combat_class_source }
			}
		}

		if = {
			limit = { scope:pod_combat_weapon = { has_variable = artifact_weapon_bonus_damage } }
			var:pod_combat_session ?= { set_variable = { name = pod_combat_weapon_bonus_damage value = scope:pod_combat_weapon.var:artifact_weapon_bonus_damage } }
		}
		if = {
			limit = { scope:pod_combat_weapon = { has_variable = artifact_weapon_undodgeable } }
			var:pod_combat_session ?= { set_variable = pod_combat_weapon_undodgeable }
		}
		# Recovery riders apply once in phase 9c, after damage resolves.
		if = {
			limit = { scope:pod_combat_weapon = { has_variable = artifact_weapon_heal_on_damage } }
			var:pod_combat_session ?= { set_variable = { name = pod_combat_weapon_heal_on_damage value = scope:pod_combat_weapon.var:artifact_weapon_heal_on_damage } }
		}
		if = {
			limit = { scope:pod_combat_weapon = { has_variable = artifact_weapon_energy_on_damage } }
			var:pod_combat_session ?= { set_variable = { name = pod_combat_weapon_energy_on_damage value = scope:pod_combat_weapon.var:artifact_weapon_energy_on_damage } }
		}
		# The damage effect copies this penalty onto the actor for PoD's wound effect.
		if = {
			limit = { scope:pod_combat_weapon = { has_variable = artifact_weapon_soak_penalty } }
			var:pod_combat_session ?= { set_variable = { name = pod_combat_weapon_soak_penalty value = scope:pod_combat_weapon.var:artifact_weapon_soak_penalty } }
		}
	}
}

# Host scope. $SOURCE$ must be an artifact scope carrying
# artifact_weapon_damage_class. Write the actual flags; lifting a flag variable
# through a scope value loses its type.
pod_combat_read_weapon_class_effect = {
	var:pod_combat_session ?= {
		if = { limit = { $SOURCE$ = { var:artifact_weapon_damage_class = flag:fire } }      set_variable = { name = pod_combat_weapon_class value = flag:fire } }
		if = { limit = { $SOURCE$ = { var:artifact_weapon_damage_class = flag:cold } }      set_variable = { name = pod_combat_weapon_class value = flag:cold } }
		if = { limit = { $SOURCE$ = { var:artifact_weapon_damage_class = flag:lightning } } set_variable = { name = pod_combat_weapon_class value = flag:lightning } }
		if = { limit = { $SOURCE$ = { var:artifact_weapon_damage_class = flag:sun } }       set_variable = { name = pod_combat_weapon_class value = flag:sun } }
		if = { limit = { $SOURCE$ = { var:artifact_weapon_damage_class = flag:silver } }    set_variable = { name = pod_combat_weapon_class value = flag:silver } }
		if = { limit = { $SOURCE$ = { var:artifact_weapon_damage_class = flag:gold } }      set_variable = { name = pod_combat_weapon_class value = flag:gold } }
		if = { limit = { $SOURCE$ = { var:artifact_weapon_damage_class = flag:holy } }      set_variable = { name = pod_combat_weapon_class value = flag:holy } }
		if = { limit = { $SOURCE$ = { var:artifact_weapon_damage_class = flag:balefire } }  set_variable = { name = pod_combat_weapon_class value = flag:balefire } }
		if = { limit = { $SOURCE$ = { var:artifact_weapon_damage_class = flag:poison } }    set_variable = { name = pod_combat_weapon_class value = flag:poison } }
		if = { limit = { $SOURCE$ = { var:artifact_weapon_damage_class = flag:cold_iron } } set_variable = { name = pod_combat_weapon_class value = flag:cold_iron } }
	}
}

# Host scope. Section 17 teardown.
pod_combat_clear_actor_weapon_profile_effect = {
	var:pod_combat_session ?= { remove_variable = pod_combat_weapon_class }
	var:pod_combat_session ?= { remove_variable = pod_combat_weapon_bonus_damage }
	var:pod_combat_session ?= { remove_variable = pod_combat_weapon_undodgeable }
	var:pod_combat_session ?= { remove_variable = pod_combat_weapon_heal_on_damage }
	var:pod_combat_session ?= { remove_variable = pod_combat_weapon_energy_on_damage }
	var:pod_combat_session ?= { remove_variable = pod_combat_weapon_soak_penalty }
}

# $KIND_VAR$ is written on the current scope from the active weapon profile.
# No retype leaves it unchanged. Literal flags avoid scope-value degradation.
pod_combat_override_kind_from_weapon_effect = {
	if = {
		limit = { scope:combat_host.var:pod_combat_session ?= { has_variable = pod_combat_weapon_class } }
		if = { limit = { scope:combat_host.var:pod_combat_session ?= { var:pod_combat_weapon_class = flag:fire } }      set_variable = { name = $KIND_VAR$ value = flag:fire } }
		if = { limit = { scope:combat_host.var:pod_combat_session ?= { var:pod_combat_weapon_class = flag:cold } }      set_variable = { name = $KIND_VAR$ value = flag:cold } }
		if = { limit = { scope:combat_host.var:pod_combat_session ?= { var:pod_combat_weapon_class = flag:lightning } } set_variable = { name = $KIND_VAR$ value = flag:lightning } }
		if = { limit = { scope:combat_host.var:pod_combat_session ?= { var:pod_combat_weapon_class = flag:sun } }       set_variable = { name = $KIND_VAR$ value = flag:sun } }
		if = { limit = { scope:combat_host.var:pod_combat_session ?= { var:pod_combat_weapon_class = flag:silver } }    set_variable = { name = $KIND_VAR$ value = flag:silver } }
		if = { limit = { scope:combat_host.var:pod_combat_session ?= { var:pod_combat_weapon_class = flag:gold } }      set_variable = { name = $KIND_VAR$ value = flag:gold } }
		if = { limit = { scope:combat_host.var:pod_combat_session ?= { var:pod_combat_weapon_class = flag:holy } }      set_variable = { name = $KIND_VAR$ value = flag:holy } }
		if = { limit = { scope:combat_host.var:pod_combat_session ?= { var:pod_combat_weapon_class = flag:balefire } }  set_variable = { name = $KIND_VAR$ value = flag:balefire } }
		if = { limit = { scope:combat_host.var:pod_combat_session ?= { var:pod_combat_weapon_class = flag:poison } }    set_variable = { name = $KIND_VAR$ value = flag:poison } }
		if = { limit = { scope:combat_host.var:pod_combat_session ?= { var:pod_combat_weapon_class = flag:cold_iron } } set_variable = { name = $KIND_VAR$ value = flag:cold_iron } }
	}
}
