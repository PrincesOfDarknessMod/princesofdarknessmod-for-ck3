# The uses_weapon path: a move flagged as swinging the actor's own weapon
# lets the equipped artifact retype its damage, add to it, or make it
# undodgeable. Everything here is a per-action host transient seeded in
# execution section 2c and torn down in section 17 - the move stories are a
# SHARED global catalogue, so nothing per-combatant may be stamped on them.
# Phase contract: COMBAT_REFERENCE.info section 2.

# Host scope, once per action. $STORY$ = the move story.
# Unconditional set-or-clear: these must never survive into the next action,
# so every path clears first and only an armed weapon-move re-sets.
pod_combat_seed_actor_weapon_profile_effect = {
	remove_variable = pod_combat_weapon_class
	remove_variable = pod_combat_weapon_bonus_damage
	remove_variable = pod_combat_weapon_undodgeable
	remove_variable = pod_combat_weapon_heal_on_damage
	remove_variable = pod_combat_weapon_energy_on_damage
	remove_variable = pod_combat_weapon_soak_penalty

	if = {
		limit = {
			$STORY$ = { has_variable = uses_weapon }
			has_variable = pod_combat_actor_char
			var:pod_combat_actor_char = { pod_combat_actor_is_armed_trigger = yes }
		}
		# One primary_armament slot exists, so the random_ pick is deterministic.
		# Re-read on every armed weapon action and consumed immediately below,
		# so a nested reaction cannot act on the outer actor's weapon.
		var:pod_combat_actor_char = {
			random_equipped_character_artifact = {
				limit = {
					artifact_slot_type = primary_armament
					prev = { can_benefit_from_artifact = prev }
				}
				save_scope_as = pod_combat_weapon
			}
		}

		# Ammunition and the like: when the armament has no class of its own,
		# carried kit may supply one - a quiver of silvered arrows makes a plain
		# bow strike silver. The armament always wins, so this only ever fills a
		# gap. Saved separately because everything else on the profile (bonus
		# damage, undodgeable, the bane tags) still belongs to the weapon.
		if = {
			limit = {
				scope:pod_combat_weapon = { NOT = { has_variable = artifact_weapon_damage_class } }
			}
			var:pod_combat_actor_char = {
				random_equipped_character_artifact = {
					limit = {
						prev = { can_benefit_from_artifact = prev }
						has_variable = artifact_weapon_damage_class
					}
					save_scope_as = pod_combat_class_source
				}
			}
		}

		# Retype replaces the default class only. An authored element or
		# `aggravated` is the move's own identity - a lightning katana neither
		# overwrites a silvered bolt nor downgrades an aggravated strike.
		if = {
			limit = {
				# trigger_if, not an OR of has_variable / var reads: OR does not
				# short-circuit, so the var read would still fire (and log) on a
				# move that has no damage_type at all.
				$STORY$ = {
					trigger_if = {
						limit = { has_variable = damage_type }
						var:damage_type = flag:superficial
					}
					trigger_else = { always = yes }
				}
			}
			# Literal if-chain, never a var copy - flag values degrade to plain
			# 'value' through scope-value copies.
			# The armament answers first; carried kit only fills a gap. Each branch
			# re-derives its source immediately before reading it, so a stale saved
			# scope from a previous action can never be read.
			if = {
				limit = { scope:pod_combat_weapon = { has_variable = artifact_weapon_damage_class } }
				pod_combat_read_weapon_class_effect = { SOURCE = scope:pod_combat_weapon }
			}
			else_if = {
				limit = {
					var:pod_combat_actor_char = {
						any_equipped_character_artifact = {
							prev = { can_benefit_from_artifact = prev }
							has_variable = artifact_weapon_damage_class
						}
					}
				}
				var:pod_combat_actor_char = {
					random_equipped_character_artifact = {
						limit = {
							prev = { can_benefit_from_artifact = prev }
							has_variable = artifact_weapon_damage_class
						}
						save_scope_as = pod_combat_class_source
					}
				}
				pod_combat_read_weapon_class_effect = { SOURCE = scope:pod_combat_class_source }
			}
		}

		if = {
			limit = { scope:pod_combat_weapon = { has_variable = artifact_weapon_bonus_damage } }
			set_variable = { name = pod_combat_weapon_bonus_damage value = scope:pod_combat_weapon.var:artifact_weapon_bonus_damage }
		}
		if = {
			limit = { scope:pod_combat_weapon = { has_variable = artifact_weapon_undodgeable } }
			set_variable = pod_combat_weapon_undodgeable
		}
		# Recovery riders, consumed once per action in §9c once the damage
		# results are final. Numeric copies, so no flag-degradation concern.
		if = {
			limit = { scope:pod_combat_weapon = { has_variable = artifact_weapon_heal_on_damage } }
			set_variable = { name = pod_combat_weapon_heal_on_damage value = scope:pod_combat_weapon.var:artifact_weapon_heal_on_damage }
		}
		if = {
			limit = { scope:pod_combat_weapon = { has_variable = artifact_weapon_energy_on_damage } }
			set_variable = { name = pod_combat_weapon_energy_on_damage value = scope:pod_combat_weapon.var:artifact_weapon_energy_on_damage }
		}
		# Read at the dispatch site rather than here: the soak roll happens
		# inside PoD's wound call, which can only reach the ACTOR character.
		if = {
			limit = { scope:pod_combat_weapon = { has_variable = artifact_weapon_soak_penalty } }
			set_variable = { name = pod_combat_weapon_soak_penalty value = scope:pod_combat_weapon.var:artifact_weapon_soak_penalty }
		}
	}
}

# Writes the host's pod_combat_weapon_class from $SOURCE$'s damage class.
# Literal if-chain, never a var copy - flag values degrade to plain 'value'
# through scope-value copies. Callers guarantee $SOURCE$ carries the var.
pod_combat_read_weapon_class_effect = {
	if = { limit = { $SOURCE$ = { var:artifact_weapon_damage_class = flag:fire } }      set_variable = { name = pod_combat_weapon_class value = flag:fire } }
	if = { limit = { $SOURCE$ = { var:artifact_weapon_damage_class = flag:cold } }      set_variable = { name = pod_combat_weapon_class value = flag:cold } }
	if = { limit = { $SOURCE$ = { var:artifact_weapon_damage_class = flag:lightning } } set_variable = { name = pod_combat_weapon_class value = flag:lightning } }
	if = { limit = { $SOURCE$ = { var:artifact_weapon_damage_class = flag:sun } }       set_variable = { name = pod_combat_weapon_class value = flag:sun } }
	if = { limit = { $SOURCE$ = { var:artifact_weapon_damage_class = flag:silver } }    set_variable = { name = pod_combat_weapon_class value = flag:silver } }
	if = { limit = { $SOURCE$ = { var:artifact_weapon_damage_class = flag:gold } }      set_variable = { name = pod_combat_weapon_class value = flag:gold } }
	if = { limit = { $SOURCE$ = { var:artifact_weapon_damage_class = flag:holy } }      set_variable = { name = pod_combat_weapon_class value = flag:holy } }
	if = { limit = { $SOURCE$ = { var:artifact_weapon_damage_class = flag:balefire } }  set_variable = { name = pod_combat_weapon_class value = flag:balefire } }
	if = { limit = { $SOURCE$ = { var:artifact_weapon_damage_class = flag:poison } }    set_variable = { name = pod_combat_weapon_class value = flag:poison } }
	if = { limit = { $SOURCE$ = { var:artifact_weapon_damage_class = flag:cold_iron } } set_variable = { name = pod_combat_weapon_class value = flag:cold_iron } }
}

# Host scope. Section 17 teardown.
pod_combat_clear_actor_weapon_profile_effect = {
	remove_variable = pod_combat_weapon_class
	remove_variable = pod_combat_weapon_bonus_damage
	remove_variable = pod_combat_weapon_undodgeable
	remove_variable = pod_combat_weapon_heal_on_damage
	remove_variable = pod_combat_weapon_energy_on_damage
	remove_variable = pod_combat_weapon_soak_penalty
}

# Re-stamps $KIND_VAR$ on the CURRENT scope from the weapon retype, so the
# payloads seeded during this action (retaliation kind, DoT kind, the
# team_all caster share) match the hit that produced them. Self-guarding:
# a no-op when no retype is live. Literal if-chain for the same reason as
# pod_combat_seed_kind_from_class_effect.
pod_combat_override_kind_from_weapon_effect = {
	if = {
		limit = { scope:combat_host ?= { has_variable = pod_combat_weapon_class } }
		if = { limit = { scope:combat_host = { var:pod_combat_weapon_class = flag:fire } }      set_variable = { name = $KIND_VAR$ value = flag:fire } }
		if = { limit = { scope:combat_host = { var:pod_combat_weapon_class = flag:cold } }      set_variable = { name = $KIND_VAR$ value = flag:cold } }
		if = { limit = { scope:combat_host = { var:pod_combat_weapon_class = flag:lightning } } set_variable = { name = $KIND_VAR$ value = flag:lightning } }
		if = { limit = { scope:combat_host = { var:pod_combat_weapon_class = flag:sun } }       set_variable = { name = $KIND_VAR$ value = flag:sun } }
		if = { limit = { scope:combat_host = { var:pod_combat_weapon_class = flag:silver } }    set_variable = { name = $KIND_VAR$ value = flag:silver } }
		if = { limit = { scope:combat_host = { var:pod_combat_weapon_class = flag:gold } }      set_variable = { name = $KIND_VAR$ value = flag:gold } }
		if = { limit = { scope:combat_host = { var:pod_combat_weapon_class = flag:holy } }      set_variable = { name = $KIND_VAR$ value = flag:holy } }
		if = { limit = { scope:combat_host = { var:pod_combat_weapon_class = flag:balefire } }  set_variable = { name = $KIND_VAR$ value = flag:balefire } }
		if = { limit = { scope:combat_host = { var:pod_combat_weapon_class = flag:poison } }    set_variable = { name = $KIND_VAR$ value = flag:poison } }
		if = { limit = { scope:combat_host = { var:pod_combat_weapon_class = flag:cold_iron } } set_variable = { name = $KIND_VAR$ value = flag:cold_iron } }
	}
}
