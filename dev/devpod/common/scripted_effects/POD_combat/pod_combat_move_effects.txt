# $STORY$ = move story. Seed $KIND_VAR$ on the current scope; superficial or
# absent class defaults to duel. Literal flags avoid variable-to-scope-value
# degradation.
pod_combat_seed_kind_from_class_effect = {
	# Caller selects the owner: fight for retaliation and team_all
	# self-damage, slot for DoT. Do not add an internal scope switch.
	set_variable = { name = $KIND_VAR$ value = flag:duel }
	if = {
		limit = { $STORY$ = { has_variable = damage_type } }
		if = { limit = { $STORY$ = { var:damage_type = flag:aggravated } } set_variable = { name = $KIND_VAR$ value = flag:aggravated } }
		if = { limit = { $STORY$ = { var:damage_type = flag:fire } }      set_variable = { name = $KIND_VAR$ value = flag:fire } }
		if = { limit = { $STORY$ = { var:damage_type = flag:cold } }      set_variable = { name = $KIND_VAR$ value = flag:cold } }
		if = { limit = { $STORY$ = { var:damage_type = flag:lightning } } set_variable = { name = $KIND_VAR$ value = flag:lightning } }
		if = { limit = { $STORY$ = { var:damage_type = flag:sun } }       set_variable = { name = $KIND_VAR$ value = flag:sun } }
		if = { limit = { $STORY$ = { var:damage_type = flag:silver } }    set_variable = { name = $KIND_VAR$ value = flag:silver } }
		if = { limit = { $STORY$ = { var:damage_type = flag:gold } }      set_variable = { name = $KIND_VAR$ value = flag:gold } }
		if = { limit = { $STORY$ = { var:damage_type = flag:holy } }      set_variable = { name = $KIND_VAR$ value = flag:holy } }
		if = { limit = { $STORY$ = { var:damage_type = flag:balefire } }  set_variable = { name = $KIND_VAR$ value = flag:balefire } }
		if = { limit = { $STORY$ = { var:damage_type = flag:poison } }    set_variable = { name = $KIND_VAR$ value = flag:poison } }
		if = { limit = { $STORY$ = { var:damage_type = flag:cold_iron } } set_variable = { name = $KIND_VAR$ value = flag:cold_iron } }
	}
}

# Apply a fired change_class to $KIND_VAR$ on the current scope. No matching
# clause leaves the kind unchanged; literal flags avoid scope-value
# degradation.
pod_combat_override_kind_from_bonus_class_effect = {
	# Read through combat_host because callers may be in fight or slot scope.
	if = {
		limit = { scope:combat_host.var:pod_combat_session ?= { has_variable = pod_combat_bonus_class } }
		if = { limit = { scope:combat_host.var:pod_combat_session ?= { var:pod_combat_bonus_class = flag:fire } }      set_variable = { name = $KIND_VAR$ value = flag:fire } }
		if = { limit = { scope:combat_host.var:pod_combat_session ?= { var:pod_combat_bonus_class = flag:cold } }      set_variable = { name = $KIND_VAR$ value = flag:cold } }
		if = { limit = { scope:combat_host.var:pod_combat_session ?= { var:pod_combat_bonus_class = flag:lightning } } set_variable = { name = $KIND_VAR$ value = flag:lightning } }
		if = { limit = { scope:combat_host.var:pod_combat_session ?= { var:pod_combat_bonus_class = flag:sun } }       set_variable = { name = $KIND_VAR$ value = flag:sun } }
		if = { limit = { scope:combat_host.var:pod_combat_session ?= { var:pod_combat_bonus_class = flag:silver } }    set_variable = { name = $KIND_VAR$ value = flag:silver } }
		if = { limit = { scope:combat_host.var:pod_combat_session ?= { var:pod_combat_bonus_class = flag:gold } }      set_variable = { name = $KIND_VAR$ value = flag:gold } }
		if = { limit = { scope:combat_host.var:pod_combat_session ?= { var:pod_combat_bonus_class = flag:holy } }      set_variable = { name = $KIND_VAR$ value = flag:holy } }
		if = { limit = { scope:combat_host.var:pod_combat_session ?= { var:pod_combat_bonus_class = flag:balefire } }  set_variable = { name = $KIND_VAR$ value = flag:balefire } }
		if = { limit = { scope:combat_host.var:pod_combat_session ?= { var:pod_combat_bonus_class = flag:poison } }    set_variable = { name = $KIND_VAR$ value = flag:poison } }
		if = { limit = { scope:combat_host.var:pod_combat_session ?= { var:pod_combat_bonus_class = flag:cold_iron } } set_variable = { name = $KIND_VAR$ value = flag:cold_iron } }
	}
}

# Typed bonuses share the contact result but use separate wound packets per
# class. Same-class clauses sum. $BUCKET_VAR$ = literal class bucket; $AMOUNT$
# = magnitude.
pod_combat_accumulate_typed_bonus_damage_effect = {
	# Do not create zero-total buckets: a zero-count wound call can emit
	# immunity toasts and incorrectly promote the aggregate result to contact.
	if = {
		limit = { $AMOUNT$ > 0 }
		scope:combat_host.var:pod_combat_session ?= {
			if = {
				limit = { has_variable = $BUCKET_VAR$ }
				change_variable = { name = $BUCKET_VAR$ add = $AMOUNT$ }
			}
			else = {
				set_variable = { name = $BUCKET_VAR$ value = $AMOUNT$ }
			}
		}
	}
}

###### MAIN ENTRY POINT

POD_execute_combat_move = {
	# $STORY$ = the POD_combat_move story_cycle

	# ── 0. Capture combat host as a saved scope ──────────────────
	# Save the host before AoE scope switches; root can be unset inside the AI
	# per-target chain.
	save_scope_as = combat_host

	# ── 1. AP spend ──────────────────────────────────────────────
	if = {
		limit = { $STORY$ = { has_variable = ap_cost } }
		pod_combat_spend_ap_effect = { AMOUNT = $STORY$.var:ap_cost }
	}

	# ── 1b. once_per: mark this move used on the actor's slot ─────
	# Match by occupant, as the once-per usability gate does.
	if = {
		limit = {
			$STORY$ = { has_variable = once_per }
			scope:combat_host.var:pod_combat_session ?= { has_variable = pod_combat_actor_char }
		}
		var:pod_combat_session ?= {
			every_in_list = {
				variable = pod_combat_active_slots
				limit = { var:slot_char = { this = scope:combat_host.var:pod_combat_session.var:pod_combat_actor_char } }
				if = {
					limit = { $STORY$ = { var:once_per = flag:round } }
					add_to_variable_list = { name = slot_moves_used_round target = $STORY$ }
				}
				else = {
					add_to_variable_list = { name = slot_moves_used_combat target = $STORY$ }
				}
			}
		}
	}

	# ── 2. Result reset + log metadata ───────────────────────────
	pod_combat_clear_last_result_vars_effect = yes
	var:pod_combat_session ?= { set_variable = { name = pod_combat_last_damage value = 0 } }

	# ── 2b. Bonus clause pre-evaluation ──────────────────────────
	# Accumulate matching clauses before applying each effect once; per-clause
	# application would change energy caps, DoT replacement, and damage
	# totals. Usability triggers enforce gates.
	#
	# Clear outside the loop, including moves without clauses, to discard
	# state left by aborted actions. Phase 17 clears the same set after
	# completion.
	var:pod_combat_session ?= {
		remove_variable = pod_combat_bonus_damage
		remove_variable = pod_combat_bonus_typed_damage_superficial
		remove_variable = pod_combat_bonus_typed_damage_aggravated
		remove_variable = pod_combat_bonus_typed_damage_fire
		remove_variable = pod_combat_bonus_typed_damage_cold
		remove_variable = pod_combat_bonus_typed_damage_lightning
		remove_variable = pod_combat_bonus_typed_damage_sun
		remove_variable = pod_combat_bonus_typed_damage_silver
		remove_variable = pod_combat_bonus_typed_damage_gold
		remove_variable = pod_combat_bonus_typed_damage_holy
		remove_variable = pod_combat_bonus_typed_damage_balefire
		remove_variable = pod_combat_bonus_typed_damage_poison
		remove_variable = pod_combat_bonus_typed_damage_cold_iron
		remove_variable = pod_combat_bonus_hit
		remove_variable = pod_combat_bonus_morale_damage
		remove_variable = pod_combat_bonus_self_heal
		remove_variable = pod_combat_bonus_heal_target
		remove_variable = pod_combat_bonus_self_morale_heal
		remove_variable = pod_combat_bonus_energy_gain
		remove_variable = pod_combat_bonus_energy_drain
		remove_variable = pod_combat_bonus_dot
		remove_variable = pod_combat_bonus_morale_dot
		remove_variable = pod_combat_bonus_ap_refund
		remove_variable = pod_combat_bonus_ap_drain
		remove_variable = pod_combat_bonus_retaliate_boost
		remove_variable = pod_combat_bonus_hit_buff_round
		remove_variable = pod_combat_bonus_hit_debuff_round
		remove_variable = pod_combat_bonus_soak_buff_round
		remove_variable = pod_combat_bonus_soak_debuff_round
		remove_variable = pod_combat_bonus_hit_buff_combat
		remove_variable = pod_combat_bonus_hit_debuff_combat
		remove_variable = pod_combat_bonus_soak_buff_combat
		remove_variable = pod_combat_bonus_soak_debuff_combat
		remove_variable = pod_combat_bonus_aggravated
		remove_variable = pod_combat_bonus_stance_guard
		remove_variable = pod_combat_bonus_stance_dodge
		remove_variable = pod_combat_bonus_stance_stealth
		remove_variable = pod_combat_bonus_uses_weapon
		remove_variable = pod_combat_bonus_class
	}
	if = {
		limit = {
			var:pod_combat_session ?= { has_variable = pod_combat_actor_char }
			$STORY$ = { has_variable_list = bonus_clauses }
		}
		$STORY$ = {
			every_in_list = {
				variable = bonus_clauses
				save_temporary_scope_as = pod_clause
				# Use combat_host: evaluator arms read the session relative to
				# their caller, and root/prev are unreliable after iteration
				# scope switches.
				scope:combat_host = {
					if = {
						limit = {
							pod_combat_move_condition_met_trigger = { STORY = scope:pod_clause ACTOR = var:pod_combat_session.var:pod_combat_actor_char PREDICATE = scope:pod_clause.var:cond_predicate }
						}
						var:pod_combat_session ?= {
							# Keep literal transient names so the scope
							# checker can resolve their ownership.
							switch = {
								trigger = scope:pod_clause.var:bonus_effect
								flag:damage = {
									if = {
										limit = { scope:pod_clause = { has_variable = bonus_param } }
										# Keep separate class buckets; invalid
										# parameters fall back to the main
										# packet.
										switch = {
											trigger = scope:pod_clause.var:bonus_param
											flag:superficial = { pod_combat_accumulate_typed_bonus_damage_effect = { BUCKET_VAR = pod_combat_bonus_typed_damage_superficial AMOUNT = scope:pod_clause.var:bonus_value } }
											flag:aggravated = { pod_combat_accumulate_typed_bonus_damage_effect = { BUCKET_VAR = pod_combat_bonus_typed_damage_aggravated AMOUNT = scope:pod_clause.var:bonus_value } }
											flag:fire        = { pod_combat_accumulate_typed_bonus_damage_effect = { BUCKET_VAR = pod_combat_bonus_typed_damage_fire AMOUNT = scope:pod_clause.var:bonus_value } }
											flag:cold        = { pod_combat_accumulate_typed_bonus_damage_effect = { BUCKET_VAR = pod_combat_bonus_typed_damage_cold AMOUNT = scope:pod_clause.var:bonus_value } }
											flag:lightning   = { pod_combat_accumulate_typed_bonus_damage_effect = { BUCKET_VAR = pod_combat_bonus_typed_damage_lightning AMOUNT = scope:pod_clause.var:bonus_value } }
											flag:sun         = { pod_combat_accumulate_typed_bonus_damage_effect = { BUCKET_VAR = pod_combat_bonus_typed_damage_sun AMOUNT = scope:pod_clause.var:bonus_value } }
											flag:silver      = { pod_combat_accumulate_typed_bonus_damage_effect = { BUCKET_VAR = pod_combat_bonus_typed_damage_silver AMOUNT = scope:pod_clause.var:bonus_value } }
											flag:gold        = { pod_combat_accumulate_typed_bonus_damage_effect = { BUCKET_VAR = pod_combat_bonus_typed_damage_gold AMOUNT = scope:pod_clause.var:bonus_value } }
											flag:holy        = { pod_combat_accumulate_typed_bonus_damage_effect = { BUCKET_VAR = pod_combat_bonus_typed_damage_holy AMOUNT = scope:pod_clause.var:bonus_value } }
											flag:balefire    = { pod_combat_accumulate_typed_bonus_damage_effect = { BUCKET_VAR = pod_combat_bonus_typed_damage_balefire AMOUNT = scope:pod_clause.var:bonus_value } }
											flag:poison      = { pod_combat_accumulate_typed_bonus_damage_effect = { BUCKET_VAR = pod_combat_bonus_typed_damage_poison AMOUNT = scope:pod_clause.var:bonus_value } }
											flag:cold_iron   = { pod_combat_accumulate_typed_bonus_damage_effect = { BUCKET_VAR = pod_combat_bonus_typed_damage_cold_iron AMOUNT = scope:pod_clause.var:bonus_value } }
											fallback = {
												# Report invalid classes
												# because the stat card still
												# advertises a typed bonus.
												debug_log = "pod_combat: damage clause bonus_param is not a damage class; folded into the main packet"
												if = { limit = { has_variable = pod_combat_bonus_damage } change_variable = { name = pod_combat_bonus_damage add = scope:pod_clause.var:bonus_value } }
												else = { set_variable = { name = pod_combat_bonus_damage value = scope:pod_clause.var:bonus_value } }
											}
										}
									}
									else = {
										if = { limit = { has_variable = pod_combat_bonus_damage } change_variable = { name = pod_combat_bonus_damage add = scope:pod_clause.var:bonus_value } }
										else = { set_variable = { name = pod_combat_bonus_damage value = scope:pod_clause.var:bonus_value } }
									}
								}
								flag:hit = {
									if = { limit = { has_variable = pod_combat_bonus_hit }  change_variable = { name = pod_combat_bonus_hit add = scope:pod_clause.var:bonus_value } }
									else = { set_variable = { name = pod_combat_bonus_hit value = scope:pod_clause.var:bonus_value } }
								}
								flag:morale_damage = {
									if = { limit = { has_variable = pod_combat_bonus_morale_damage }  change_variable = { name = pod_combat_bonus_morale_damage add = scope:pod_clause.var:bonus_value } }
									else = { set_variable = { name = pod_combat_bonus_morale_damage value = scope:pod_clause.var:bonus_value } }
								}
								flag:self_heal = {
									if = { limit = { has_variable = pod_combat_bonus_self_heal }  change_variable = { name = pod_combat_bonus_self_heal add = scope:pod_clause.var:bonus_value } }
									else = { set_variable = { name = pod_combat_bonus_self_heal value = scope:pod_clause.var:bonus_value } }
								}
								flag:heal_target = {
									if = { limit = { has_variable = pod_combat_bonus_heal_target }  change_variable = { name = pod_combat_bonus_heal_target add = scope:pod_clause.var:bonus_value } }
									else = { set_variable = { name = pod_combat_bonus_heal_target value = scope:pod_clause.var:bonus_value } }
								}
								flag:self_morale_heal = {
									if = { limit = { has_variable = pod_combat_bonus_self_morale_heal }  change_variable = { name = pod_combat_bonus_self_morale_heal add = scope:pod_clause.var:bonus_value } }
									else = { set_variable = { name = pod_combat_bonus_self_morale_heal value = scope:pod_clause.var:bonus_value } }
								}
								flag:energy_gain = {
									if = { limit = { has_variable = pod_combat_bonus_energy_gain }  change_variable = { name = pod_combat_bonus_energy_gain add = scope:pod_clause.var:bonus_value } }
									else = { set_variable = { name = pod_combat_bonus_energy_gain value = scope:pod_clause.var:bonus_value } }
								}
								flag:energy_drain = {
									if = { limit = { has_variable = pod_combat_bonus_energy_drain }  change_variable = { name = pod_combat_bonus_energy_drain add = scope:pod_clause.var:bonus_value } }
									else = { set_variable = { name = pod_combat_bonus_energy_drain value = scope:pod_clause.var:bonus_value } }
								}
								flag:dot = {
									if = { limit = { has_variable = pod_combat_bonus_dot }  change_variable = { name = pod_combat_bonus_dot add = scope:pod_clause.var:bonus_value } }
									else = { set_variable = { name = pod_combat_bonus_dot value = scope:pod_clause.var:bonus_value } }
								}
								flag:morale_dot = {
									if = { limit = { has_variable = pod_combat_bonus_morale_dot }  change_variable = { name = pod_combat_bonus_morale_dot add = scope:pod_clause.var:bonus_value } }
									else = { set_variable = { name = pod_combat_bonus_morale_dot value = scope:pod_clause.var:bonus_value } }
								}
								flag:ap_refund = {
									if = { limit = { has_variable = pod_combat_bonus_ap_refund }  change_variable = { name = pod_combat_bonus_ap_refund add = scope:pod_clause.var:bonus_value } }
									else = { set_variable = { name = pod_combat_bonus_ap_refund value = scope:pod_clause.var:bonus_value } }
								}
								flag:ap_drain = {
									if = { limit = { has_variable = pod_combat_bonus_ap_drain }  change_variable = { name = pod_combat_bonus_ap_drain add = scope:pod_clause.var:bonus_value } }
									else = { set_variable = { name = pod_combat_bonus_ap_drain value = scope:pod_clause.var:bonus_value } }
								}
								flag:retaliate_boost = {
									if = { limit = { has_variable = pod_combat_bonus_retaliate_boost }  change_variable = { name = pod_combat_bonus_retaliate_boost add = scope:pod_clause.var:bonus_value } }
									else = { set_variable = { name = pod_combat_bonus_retaliate_boost value = scope:pod_clause.var:bonus_value } }
								}
								# Tier grants (bonus_value is 1/2/3 = small/medium/large, not a
								# magnitude), so these accumulate TAKE-MAX: met clauses at tier 1 and
								# tier 2 must leave tier 2, where the guarded-add above would invent a
								# tier 3 the magnitude tables do not define. Nested rather than an
								# "absent OR lower" pair of siblings: CK3 does not short-circuit, so the
								# var: read would still evaluate on the first grant, where it is absent.
								flag:hit_buff_round = {
									if = {
										limit = { has_variable = pod_combat_bonus_hit_buff_round }

											limit = { var:pod_combat_bonus_hit_buff_round < scope:pod_clause.var:bonus_value }
											set_variable = { name = pod_combat_bonus_hit_buff_round value = scope:pod_clause.var:bonus_value }
										}
									}
									else = { set_variable = { name = pod_combat_bonus_hit_buff_round value = scope:pod_clause.var:bonus_value } }
								}
								flag:hit_debuff_round = {
									if = {
										limit = { has_variable = pod_combat_bonus_hit_debuff_round }
										if = {
											limit = { var:pod_combat_bonus_hit_debuff_round < scope:pod_clause.var:bonus_value }
											set_variable = { name = pod_combat_bonus_hit_debuff_round value = scope:pod_clause.var:bonus_value }

										if = {
											limit = { var:pod_combat_bonus_soak_buff_combat < scope:pod_clause.var:bonus_value }
											set_variable = { name = pod_combat_bonus_soak_buff_combat value = scope:pod_clause.var:bonus_value }
										}
									}
									else = { set_variable = { name = pod_combat_bonus_soak_buff_combat value = scope:pod_clause.var:bonus_value } }
								}
								flag:soak_debuff_combat = {
									if = {
										limit = { has_variable = pod_combat_bonus_soak_debuff_combat }
										if = {
											limit = { var:pod_combat_bonus_soak_debuff_combat < scope:pod_clause.var:bonus_value }
											set_variable = { name = pod_combat_bonus_soak_debuff_combat value = scope:pod_clause.var:bonus_value }
										}
									}
									else = { set_variable = { name = pod_combat_bonus_soak_debuff_combat value = scope:pod_clause.var:bonus_value } }
								}
								# change_type keeps its historical var name.
								flag:change_type    = { set_variable = pod_combat_bonus_aggravated }
								flag:stance_guard   = { set_variable = pod_combat_bonus_stance_guard }
								flag:stance_dodge   = { set_variable = pod_combat_bonus_stance_dodge }
								flag:stance_stealth = { set_variable = pod_combat_bonus_stance_stealth }
								flag:uses_weapon    = { set_variable = pod_combat_bonus_uses_weapon }
								# The element rides a literal if-chain, not a var
								# copy: a scope-value copy degrades a flag to a
								# plain value, and the §6 / counter / DoT switches
								# compare it against flag literals.
								flag:change_class = {
									if = { limit = { scope:pod_clause.var:bonus_param = flag:fire }      set_variable = { name = pod_combat_bonus_class value = flag:fire } }
									if = { limit = { scope:pod_clause.var:bonus_param = flag:cold }      set_variable = { name = pod_combat_bonus_class value = flag:cold } }
									if = { limit = { scope:pod_clause.var:bonus_param = flag:lightning } set_variable = { name = pod_combat_bonus_class value = flag:lightning } }
									if = { limit = { scope:pod_clause.var:bonus_param = flag:sun }       set_variable = { name = pod_combat_bonus_class value = flag:sun } }
									if = { limit = { scope:pod_clause.var:bonus_param = flag:silver }    set_variable = { name = pod_combat_bonus_class value = flag:silver } }
									if = { limit = { scope:pod_clause.var:bonus_param = flag:gold }      set_variable = { name = pod_combat_bonus_class value = flag:gold } }
									if = { limit = { scope:pod_clause.var:bonus_param = flag:holy }      set_variable = { name = pod_combat_bonus_class value = flag:holy } }
									if = { limit = { scope:pod_clause.var:bonus_param = flag:balefire }  set_variable = { name = pod_combat_bonus_class value = flag:balefire } }
									if = { limit = { scope:pod_clause.var:bonus_param = flag:poison }    set_variable = { name = pod_combat_bonus_class value = flag:poison } }
									if = { limit = { scope:pod_clause.var:bonus_param = flag:cold_iron } set_variable = { name = pod_combat_bonus_class value = flag:cold_iron } }
								}
							}
						}
					}
				}
			}
		}
	}

	# ── 2c. Weapon profile (uses_weapon moves) ───────────────────
	# Resolve the actor's equipped weapon ONCE per action into host
	# transients the damage phases read. Unconditional set-or-clear, so an
	# unarmed actor or a non-weapon move actively clears the previous
	# action's profile rather than inheriting it.
	# Runs after 2b on purpose: the uses_weapon conditional bonus grants the
	# swing through the latch set there, so the seeder's gate needs it decided.
	pod_combat_seed_actor_weapon_profile_effect = { STORY = $STORY$ }

	# ── 3. Energy cost (splat-agnostic) ──────────────────────────
	# POD_splat_agnostic_energy_cost routes to the CURRENT scope's splat pool, so
	# MUST scope-switch to the actor first - else every AI cast drains the PC pool.
	if = {
		limit = { $STORY$ = { has_variable = energy_cost } }
		if = {
			limit = { var:pod_combat_session ?= { has_variable = pod_combat_actor_char } }
			var:pod_combat_session.var:pod_combat_actor_char ?= {
				POD_splat_agnostic_energy_cost = { COUNT = $STORY$.var:energy_cost }
			}
		}
		else = {
			# actor_char not stamped yet (move fired before load_workspace);
			# current scope is the actor in those flows by construction.
			POD_splat_agnostic_energy_cost = { COUNT = $STORY$.var:energy_cost }
		}
		# AI energy-attrition counter, read by the selector on the next AI turn.
		if = {
			limit = { var:pod_combat_session ?= { has_variable = pod_combat_actor_energy_used } }
			var:pod_combat_session ?= { change_variable = { name = pod_combat_actor_energy_used add = 1 } }
		}
		else = {
			var:pod_combat_session ?= { set_variable = { name = pod_combat_actor_energy_used value = 1 } }
		}
	}

	# ── 4. CK3 resource costs + gains (controlled turns only) ────
	# Only a human-controlled actor pays CK3-side gold/stress/piety/prestige and
	# gains the steal_gold / gain_dread values; AI turns touch no CK3 resources.
	#
	# Payer identity splits on whether the primitive has an affordability gate.
	# gold / piety / prestige stay on root, the clicking player, because
	# pod_combat_move_dynamic_gates_trigger tests those three against root too:
	# charging the acting body instead would let a move the driver can afford push
	# a controled character's purse negative. Stress and the phase-4 gains have no
	# gate, so they follow the body that actually swung - a player watching a
	# proxy fight should not accrue its strain.
	# Pre-existing fail-open on either side: the gate is advisory, so a cost the
	# payer cannot meet still gets charged (remove_short_term_gold can go negative).
	if = {
		limit = { pod_combat_current_actor_is_controlled_trigger = yes }
		if = {
			limit = { $STORY$ = { has_variable = gold_cost } }
			remove_short_term_gold = $STORY$.var:gold_cost
		}
		if = {
			limit = { $STORY$ = { has_variable = piety_cost } }
			add_piety = {
				value = $STORY$.var:piety_cost
				multiply = -1
			}
		}
		if = {
			limit = { $STORY$ = { has_variable = prestige_cost } }
			add_prestige = {
				value = $STORY$.var:prestige_cost
				multiply = -1
			}
		}

		# Body-scoped half. Resolve the payee once instead of branching the whole
		# body twice: seed it with root, then let the actor overwrite it. Seeding
		# first is what makes the bare ?= safe - a move fired before load_workspace
		# has no actor_char, the deref skips, and root stands (the same fallback
		# the energy cost in section 3 spells out as an else). The actor is seated
		# by construction, so its own pod_combat_session reaches the same fight the
		# reads below need.
		save_temporary_scope_as = pod_combat_payee
		var:pod_combat_session.var:pod_combat_actor_char ?= { save_temporary_scope_as = pod_combat_payee }
		scope:pod_combat_payee = {
			if = {
				limit = { $STORY$ = { has_variable = stress_cost } }
				add_stress = $STORY$.var:stress_cost
			}

			# Phase 4 gains. steal_gold transfers from the target, clamped to what
			# it actually has so a broke victim never mints gold for the thief;
			# gain_dread adds to the actor.
			# Practice spars leave no lasting transfers (the contract that restores
			# wounds and skips strain), so the gains skip too - no risk-free
			# gold-theft or dread loop against sparring partners.
			if = {
				limit = { NOT = { pod_combat_is_practice_trigger = yes } }
				if = {
					limit = {
						$STORY$ = { has_variable = steal_gold }
						var:pod_combat_session ?= { has_variable = pod_combat_target_char }
					}
					var:pod_combat_session.var:pod_combat_target_char ?= {
						save_temporary_scope_value_as = {
							name = pod_combat_stolen_gold
							value = { value = $STORY$.var:steal_gold  max = gold  min = 0 }
						}
						# add_gold rejects negative values, so debit the victim with
						# remove_short_term_gold (the clamp above keeps it <= their gold).
						remove_short_term_gold = scope:pod_combat_stolen_gold
					}
					add_gold = scope:pod_combat_stolen_gold
				}
				if = {
					limit = { $STORY$ = { has_variable = gain_dread } }
					add_dread = $STORY$.var:gain_dread
				}
			}
		}
	}

	# ── 5. Target stance strip ──────────────────────────────────
	# Strip target dodge/stealth/guard BEFORE damage resolves, so a strip + attack
	# combo lands its own attack. bypass_dodge / bypass_stealth are plain move
	# presence flags (silent strips); remove_guard (below) also logs. bypass_guard has no
	# runtime var: the generator omits the checks_guard capability flag and the
	# contact resolver reads that omission directly.
	if = {
		limit = { $STORY$ = { has_variable = bypass_dodge } }
		var:pod_combat_session ?= { remove_variable = pod_combat_target_is_dodging }
	}
	if = {
		limit = { $STORY$ = { has_variable = bypass_stealth } }
		# Must also subtract the +30 stealth hit bonus baked into saved
		# hit_chance, else the target keeps a phantom offensive boost. Idempotent.
		pod_combat_break_stealth_workspace_effect = { SIDE = target }
	}
	# remove_guard is a move EFFECT (it logs seg_remove_guard), but it strips the
	# same kind of target stance, so it runs here too, BEFORE §6, so a strip-guard
	# + attack move lands its own attack instead of being guard-blocked. AoE strips
	# each enemy slot's guard in §5b; this workspace write is redundant for
	# team_enemy (like the §5c target-writes).
	if = {
		limit = {
			$STORY$ = {
				has_variable = remove_guard
				NOT = { has_variable = requires_control_contact }
			}
		}
		var:pod_combat_session ?= { remove_variable = pod_combat_target_is_guarding }
	}

	# ── 5b. AoE debuff pre-damage broadcast ──────────────────────
	# team_enemy: broadcast tier debuffs + stance-strips to ALL enemy slots
	# BEFORE the damage iteration, so each hit respects them on its own target.
	# §5's workspace strip only touches the pre-AoE target, which each iteration
	# overwrites. §5c's target-writes still run but are redundant for team_enemy.
	# team_all does the same across both teams, plus the caster's own tier share.
	if = {
		limit = { $STORY$ = { var:targeting = flag:team_enemy } }
		pod_combat_apply_team_enemy_debuffs_effect = { STORY = $STORY$ }
	}
	else_if = {
		limit = { $STORY$ = { var:targeting = flag:team_all } }
		pod_combat_apply_team_all_debuffs_effect = { STORY = $STORY$ }
		pod_combat_apply_team_all_self_debuffs_effect = { STORY = $STORY$ }
	}

	# ── 5b2. Pure-control contact delivery ────────────────────────
	# Explicitly typed single-target controls have no HP/morale phase to invoke
	# the contact resolver for them. The generator marks those rows with
	# requires_control_contact; resolve their advertised spirit/stance/hit/guard
	# rules here, before any target-side debuff/control payload is applied.
	if = {
		limit = { $STORY$ = { has_variable = requires_control_contact } }
		$STORY$ = { save_scope_as = current_move }
		pod_combat_resolve_contact_effect = {
			TARGET_SIDE = target
			ACTOR_SIDE  = actor
		}
		if = {
			limit = { NOT = { scope:contact_result = 0 } }
			var:pod_combat_session ?= { set_variable = pod_combat_control_delivery_failed }
		}
		# A hostile control attempt breaks the caster's stealth just like an HP
		# attack, whether the delivery lands or is evaded.
		pod_combat_consume_actor_stealth_effect = { ACTOR_SIDE = actor }
	}
	# A control-delivered guard strip is part of that delivery, unlike an
	# ordinary strip+attack move whose pre-damage strip remains in §5.
	if = {
		limit = {
			$STORY$ = {
				has_variable = remove_guard
				has_variable = requires_control_contact
			}
			NOT = { var:pod_combat_session ?= { has_variable = pod_combat_control_delivery_failed } }
		}
		var:pod_combat_session ?= { remove_variable = pod_combat_target_is_guarding }
	}

	# ── 5c. Tier buffs/debuffs (applied BEFORE the §6 damage) ──────────
	# Runs ahead of damage so a buff/debuff + attack in the SAME move lands on
	# THIS attack: a self hit buff raises pod_combat_current_hit_chance for the
	# §6 hit roll, and a target soak debuff sets the POD_soak_chance bridge var
	# before the §6 wound roll. AoE target debuffs are still broadcast to slots in
	# §5b; the workspace target-writes below stay redundant for team_enemy/team_all.
	# Buff tiers route to the actor for both team AoE modes (only single_ally and
	# single_pc redirect to the target), so a team_all buff would land on the caster alone.
	# Buffs normally apply to the actor. A single_ally move names a
	# beneficiary in the target workspace, so route every tier and its soak
	# bridge to that target instead. team_ally still seeds the actor here and
	# broadcasts to every allied slot in §12b.
	if = {
		limit = { NOT = { var:pod_combat_session ?= { has_variable = pod_combat_control_delivery_failed } } }
		if = {
			limit = { $STORY$ = { var:targeting = flag:single_ally } }
			pod_combat_apply_workspace_buff_tiers_effect = { STORY = $STORY$ SIDE = target }
		}
		else = {
			pod_combat_apply_workspace_buff_tiers_effect = { STORY = $STORY$ SIDE = actor }
		}
	}

	# A failed pure-control contact has no target-side payload. Unmarked moves
	# never set this tripwire, preserving the existing pre-damage debuff timing.
	if = {
		limit = { NOT = { var:pod_combat_session ?= { has_variable = pod_combat_control_delivery_failed } } }
		pod_combat_apply_workspace_debuff_tiers_effect = { STORY = $STORY$ SIDE = target }
	}

	# ── 5d. Conditional bonus hit chance ────────────────────────
	# The summed hit bump from every met hit clause (§2b) for THIS attack.
	# Sits on top of any tier recalc above; same ceiling as the recalc.
	if = {
		limit = {
			var:pod_combat_session ?= { has_variable = pod_combat_bonus_hit }
			var:pod_combat_session ?= { has_variable = pod_combat_current_hit_chance }
		}
		var:pod_combat_session ?= { change_variable = { name = pod_combat_current_hit_chance add = var:pod_combat_bonus_hit } }
		# Re-cap at the shared ceiling. No floor needed: the generator rejects
		# a negative clause value, so a bump can only raise the chance.
		if = {
			limit = { var:pod_combat_session.var:pod_combat_current_hit_chance > pod_combat_hit_chance_max }
			var:pod_combat_session ?= { set_variable = { name = pod_combat_current_hit_chance value = pod_combat_hit_chance_max } }
		}
	}

	# ── 6. HP damage dispatch ────────────────────────────────────
	# Routes single_enemy vs team AoE. See HP DAMAGE DISPATCH header below.
	if = {
		limit = { $STORY$ = { has_variable = hp_damage } }
		if = {
			limit = { $STORY$ = { var:targeting = flag:team_enemy } }
			pod_combat_apply_team_enemy_hp_damage_effect = { STORY = $STORY$ }
		}
		else_if = {
			limit = { $STORY$ = { var:targeting = flag:team_all } }
			pod_combat_apply_team_all_hp_damage_effect = { STORY = $STORY$ }
		}
		else = {
			# single_enemy (default, most common)
			pod_combat_apply_single_hp_damage_effect = { STORY = $STORY$ TARGET_SIDE = target }
		}
	}

	# ── 7. Morale damage ─────────────────────────────────────────
	# Own pipeline: auto-hit, no guard, no soak, dodge still works.
	# Rider gate (here and §8/§9): a *_condition var on the move makes the
	# payload fire only on hit/miss/contact, read off §6's outcome on the host.
	# Default (no condition) = fires unconditionally.
	# The conditional bonus (the summed morale_damage clauses, held at §2b) folds
	# into the SAME deal call as the base amount: one call per target consumes
	# the target's stance exactly once for the combined total (a second
	# independent call pierced the dodge/stealth the base call had just
	# consumed) and the logged amount is the true total. A bonus-only move (no
	# base morale_damage) takes the same path with a base of 0, including the
	# team_enemy broadcast (the team applier reads the combined scope value).
	if = {
		limit = {
			OR = {
				$STORY$ = { has_variable = morale_damage }
				var:pod_combat_session ?= { has_variable = pod_combat_bonus_morale_damage }
			}
			pod_combat_rider_condition_passes_trigger = {
				STORY     = $STORY$
				RIDER_VAR = morale_damage_condition
			}
		}
		save_scope_value_as = {
			name = pod_combat_morale_deal_total
			value = {
				value = 0
				if = { limit = { $STORY$ = { has_variable = morale_damage } } add = $STORY$.var:morale_damage }
				if = {
					limit = { var:pod_combat_session ?= { has_variable = pod_combat_bonus_morale_damage } }
					add = var:pod_combat_session.var:pod_combat_bonus_morale_damage
				}
			}
		}
		if = {
			limit = { $STORY$ = { var:targeting = flag:team_enemy } }
			pod_combat_apply_team_enemy_morale_damage_effect = { STORY = $STORY$ }
		}
		else_if = {
			limit = { $STORY$ = { var:targeting = flag:team_all } }
			pod_combat_apply_team_all_morale_damage_effect = { STORY = $STORY$ }
		}
		else = {
			pod_combat_deal_morale_damage_effect = { TARGET_SIDE = target AMOUNT = scope:pod_combat_morale_deal_total }
		}
	}

	# ── 8. Self-damage ───────────────────────────────────────────
	# Routed through the PoD pipeline like a real hit so wound state stays in
	# lockstep. REASON=duel (superficial, soakable) not self_mutilation - 
	# overextension, not a permanent wound. Rider-gated (see §7).
	if = {
		limit = {
			$STORY$ = { has_variable = self_damage }
			pod_combat_rider_condition_passes_trigger = {
				STORY     = $STORY$
				RIDER_VAR = self_damage_condition
			}
		}
		pod_combat_deal_hp_damage_bypass_guard_effect = {
			TARGET_SIDE = actor
			AMOUNT      = $STORY$.var:self_damage
			REASON      = duel
		}
		# Lift the post-soak recoil amount for the log's recoil segment.
		# scope:attack_result_flag stays discarded - recoil never overwrites the
		# main attack's outcome axes.
		if = {
			limit = { scope:damage_dealt > 0 }
			var:pod_combat_session ?= { set_variable = { name = pod_combat_last_self_damage value = scope:damage_dealt } }
		}
	}

	# ── 9. HP heal ───────────────────────────────────────────────
	# heal_type=aggravated routes to the aggravated variant (also reduces
	# agg_damage); else superficial (caps at max_hp - agg_damage).
	# Rider-gated (see §7) on heal_condition - drain heals get
	# heal_condition=on_hp_damage auto-emitted by the generator's `drain` token.
	# team_all has no broadcast arm here (nor in §10/§12b): a heal reaching both
	# sides would mend the enemies the same move just hit. It falls to the actor
	# self-heal else.
	if = {
		limit = {
			$STORY$ = { has_variable = hp_heal }
			pod_combat_rider_condition_passes_trigger = {
				STORY     = $STORY$
				RIDER_VAR = heal_condition
			}
		}
		if = {
			limit = { $STORY$ = { var:targeting = flag:team_ally } }
			pod_combat_apply_team_ally_hp_heal_effect = { STORY = $STORY$ }
		}
		else_if = {
			# single_ally / single_pc - the beneficiary is loaded into the target
			# workspace (the ally by the PC's picker, the PC by the AI swap), so
			# SIDE=target heals them instead of the actor.
			limit = { $STORY$ = { OR = { var:targeting = flag:single_ally  var:targeting = flag:single_pc } } }
			if = {
				limit = { $STORY$ = { has_variable = heal_type var:heal_type = flag:aggravated } }
				pod_combat_restore_hp_aggravated_effect = { SIDE = target AMOUNT = $STORY$.var:hp_heal }
			}
			else = {
				pod_combat_restore_hp_superficial_effect = { SIDE = target AMOUNT = $STORY$.var:hp_heal }
			}
		}
		else_if = {
			limit = { $STORY$ = { has_variable = heal_type var:heal_type = flag:aggravated } }
			pod_combat_restore_hp_aggravated_effect = { SIDE = actor AMOUNT = $STORY$.var:hp_heal }
		}
		else = {
			pod_combat_restore_hp_superficial_effect = { SIDE = actor AMOUNT = $STORY$.var:hp_heal }
		}
	}

	# ── 9b. Conditional bonus heals / morale / energy ───────────
	# Use absolute AMOUNT paths: the healing effect reads them after switching to
	# the recipient character.
	if = {
		limit = { var:pod_combat_session ?= { has_variable = pod_combat_bonus_self_heal } }
		pod_combat_restore_hp_superficial_effect = { SIDE = actor AMOUNT = scope:combat_host.var:pod_combat_session.var:pod_combat_bonus_self_heal }
	}
	if = {
		limit = { var:pod_combat_session ?= { has_variable = pod_combat_bonus_heal_target } }
		pod_combat_restore_hp_superficial_effect = { SIDE = target AMOUNT = scope:combat_host.var:pod_combat_session.var:pod_combat_bonus_heal_target }
	}
	# Bonus morale healing is actor-only; unconditional morale_heal supports
	# allies.
	if = {
		limit = { var:pod_combat_session ?= { has_variable = pod_combat_bonus_self_morale_heal } }
		pod_combat_restore_morale_effect = { SIDE = actor AMOUNT = var:pod_combat_session.var:pod_combat_bonus_self_morale_heal }
	}

	# Combine base and bonus energy before applying PoD's per-call ceiling.
	# Apply in actor scope, including AI, and read COUNT.
	if = {
		limit = { $STORY$ = { has_variable = energy_gain } }
		if = {
			limit = { var:pod_combat_session ?= { has_variable = pod_combat_bonus_energy_gain } }
			var:pod_combat_session ?= { change_variable = { name = pod_combat_bonus_energy_gain add = $STORY$.var:energy_gain } }
		}
		else = {
			var:pod_combat_session ?= { set_variable = { name = pod_combat_bonus_energy_gain value = $STORY$.var:energy_gain } }
		}
	}
	if = {
		limit = { var:pod_combat_session ?= { has_variable = pod_combat_bonus_energy_gain } }
		if = {
			limit = { var:pod_combat_session ?= { has_variable = pod_combat_actor_char } }
			var:pod_combat_session.var:pod_combat_actor_char ?= { POD_splat_agnostic_energy_gain = { COUNT = scope:combat_host.var:pod_combat_session.var:pod_combat_bonus_energy_gain } }
		}
		else = { POD_splat_agnostic_energy_gain = { COUNT = var:pod_combat_session.var:pod_combat_bonus_energy_gain } }
		# Log the requested amount scaled to the actor's resource. The scripted effect
		# does not expose pool-capped actual gain.
		pod_combat_scale_energy_to_splat_effect = { CHARACTER = var:pod_combat_session.var:pod_combat_actor_char COUNT = scope:combat_host.var:pod_combat_session.var:pod_combat_bonus_energy_gain }
		var:pod_combat_session ?= { set_variable = { name = pod_combat_last_energy_gain value = scope:pod_combat_scaled_energy } }
	}
	if = {
		limit = {
			var:pod_combat_session ?= { has_variable = pod_combat_bonus_energy_drain }
			var:pod_combat_session ?= { has_variable = pod_combat_target_char }
			# Practice contract: no lasting harm to the partner - a real
			# energy drain would persist past the spar.
			NOT = { pod_combat_is_practice_trigger = yes }
		}
		var:pod_combat_session.var:pod_combat_target_char ?= { POD_splat_agnostic_energy_loss = { COUNT = scope:combat_host.var:pod_combat_session.var:pod_combat_bonus_energy_drain } }
		# Log the requested drain in the target's resource units, only when
		# applied.
		pod_combat_scale_energy_to_splat_effect = { CHARACTER = var:pod_combat_session.var:pod_combat_target_char COUNT = scope:combat_host.var:pod_combat_session.var:pod_combat_bonus_energy_drain }
		var:pod_combat_session ?= { set_variable = { name = pod_combat_last_energy_drain value = scope:pod_combat_scaled_energy } }
	}

	# ── 9c. Weapon recovery riders ───────────────────────────────
	# Recovery applies once per action after all damage packets, including
	# AoE. Fully soaked hits do not qualify.
	if = {
		limit = {
			var:pod_combat_session ?= { has_variable = pod_combat_weapon_heal_on_damage }
			pod_combat_result_code_indicates_hp_damage_trigger = yes
		}
		# AMOUNT is read in recipient scope; keep the absolute path.
		pod_combat_restore_hp_superficial_effect = { SIDE = actor AMOUNT = scope:combat_host.var:pod_combat_session.var:pod_combat_weapon_heal_on_damage }
	}
	if = {
		limit = {
			var:pod_combat_session ?= { has_variable = pod_combat_weapon_energy_on_damage }
			pod_combat_result_code_indicates_hp_damage_trigger = yes
			var:pod_combat_session ?= { has_variable = pod_combat_actor_char }
		}
		# Apply to the actor's pool; COUNT is read in character scope.
		var:pod_combat_session.var:pod_combat_actor_char ?= { POD_splat_agnostic_energy_gain = { COUNT = scope:combat_host.var:pod_combat_session.var:pod_combat_weapon_energy_on_damage } }
		# Add to phase 9b's logged gain; both amounts use the actor's resource
		# units.
		pod_combat_scale_energy_to_splat_effect = { CHARACTER = var:pod_combat_session.var:pod_combat_actor_char COUNT = scope:combat_host.var:pod_combat_session.var:pod_combat_weapon_energy_on_damage }
		if = {
			limit = { var:pod_combat_session ?= { has_variable = pod_combat_last_energy_gain } }
			var:pod_combat_session ?= { change_variable = { name = pod_combat_last_energy_gain add = scope:pod_combat_scaled_energy } }
		}
		else = {
			var:pod_combat_session ?= { set_variable = { name = pod_combat_last_energy_gain value = scope:pod_combat_scaled_energy } }
		}
	}

	# ── 10. Morale heal ──────────────────────────────────────────
	if = {
		limit = { $STORY$ = { has_variable = morale_heal } }
		if = {
			limit = { $STORY$ = { var:targeting = flag:team_ally } }
			pod_combat_apply_team_ally_morale_heal_effect = { STORY = $STORY$ }
		}
		else_if = {
			limit = { $STORY$ = { OR = { var:targeting = flag:single_ally  var:targeting = flag:single_pc } } }
			pod_combat_restore_morale_effect = { SIDE = target AMOUNT = $STORY$.var:morale_heal }
		}
		else = {
			pod_combat_restore_morale_effect = { SIDE = actor AMOUNT = $STORY$.var:morale_heal }
		}
	}

	# ── 12. Non-tier presence flags ──────────────────────────────
	if = {
		limit = { $STORY$ = { var:targeting = flag:single_ally } }
		pod_combat_apply_workspace_stances_effect = { STORY = $STORY$ SIDE = target }
	}
	else = {
		pod_combat_apply_workspace_stances_effect = { STORY = $STORY$ SIDE = actor }
	}

	# Conditional stance bonuses are actor-only and may coexist.
	if = {
		limit = { var:pod_combat_session ?= { has_variable = pod_combat_bonus_stance_guard } }
		var:pod_combat_session ?= { set_variable = { name = pod_combat_actor_is_guarding value = yes } }
	}
	if = {
		limit = { var:pod_combat_session ?= { has_variable = pod_combat_bonus_stance_dodge } }
		var:pod_combat_session ?= { set_variable = { name = pod_combat_actor_is_dodging value = yes } }
	}
	if = {
		limit = { var:pod_combat_session ?= { has_variable = pod_combat_bonus_stance_stealth } }
		pod_combat_apply_stealth_workspace_effect = { SIDE = actor }
	}

	# ── 12b. AoE buff broadcast ──────────────────────────────────
	if = {
		limit = { $STORY$ = { var:targeting = flag:team_ally } }
		pod_combat_apply_team_ally_buffs_effect = { STORY = $STORY$ }
	}

	# ── 12b-marker. Marker removal ───────────────────────────
	# Remove after condition evaluation and before marker application, so a
	# move can consume one marker and apply another. Only combat-owned
	# MARKER_MODIFIERS may be removed.
	if = {
		limit = {
			$STORY$ = { has_variable = remove_marker }
			# Pure strip moves may require control contact; suppress removal
			# when delivery fails.
			NOT = { var:pod_combat_session ?= { has_variable = pod_combat_control_delivery_failed } }
		}
		if = {
			limit = {
				$STORY$ = { var:remove_marker_target = flag:target }
				var:pod_combat_session ?= { has_variable = pod_combat_target_char }
				# Misses prevent target stripping. User-marker consumption is
				# still paid on a miss. Use NOT(miss) so zero-HP strip moves
				# qualify.
				NOT = { pod_combat_result_code_indicates_miss_trigger = yes }
			}
			var:pod_combat_session.var:pod_combat_target_char ?= {
				pod_combat_strip_marker_effect = { STORY = $STORY$ }
			}
			# Log successful removal only.
			var:pod_combat_session ?= { set_variable = pod_combat_last_marker_removed }
		}
		else_if = {
			limit = {
				$STORY$ = { var:remove_marker_target = flag:user }
				var:pod_combat_session ?= { has_variable = pod_combat_actor_char }
			}
			var:pod_combat_session.var:pod_combat_actor_char ?= {
				pod_combat_strip_marker_effect = { STORY = $STORY$ }
			}
			# Log successful removal only.
			var:pod_combat_session ?= { set_variable = pod_combat_last_marker_removed }
		}
	}

	# ── 12c. Marker character modifier ───────────────────────────
	# Apply after phase 2b so a move cannot satisfy its own condition. New
	# literal keys also need condition-trigger, teardown-effect, and slot-init
	# cases. Cleanup normally removes markers; 30 days limits persistence if
	# cleanup fails.
	if = {
		limit = {
			$STORY$ = { has_variable = marker_modifier }
			var:pod_combat_session ?= { has_variable = pod_combat_actor_char }
		}
		switch = {
			trigger = $STORY$.var:marker_modifier
			flag:pod_combat_anima_flare = {
				var:pod_combat_session.var:pod_combat_actor_char ?= {
					add_character_modifier = {
						modifier = pod_combat_anima_flare
						days     = 30
					}
				}
			}
			flag:pod_combat_predator_focus = {
				var:pod_combat_session.var:pod_combat_actor_char ?= {
					add_character_modifier = { modifier = pod_combat_predator_focus days = 30 }
				}
			}
			flag:pod_combat_kai_focus = {
				var:pod_combat_session.var:pod_combat_actor_char ?= {
					add_character_modifier = { modifier = pod_combat_kai_focus days = 30 }
				}
			}
		}
	}

	# ── 13. Retaliation stances ──────────────────────────────────
	# Arming contract: pod_combat_arm_retaliation_effect. Contact/miss charges
	# take precedence over any-attack charges when fired by
	# pod_combat_check_retaliation_effect.
	if = {
		limit = { $STORY$ = { has_variable = retaliate_if_hit } }
		pod_combat_arm_retaliation_effect = { STORY = $STORY$ WHEN = hit }
	}
	if = {
		limit = { $STORY$ = { has_variable = retaliate_if_missed } }
		pod_combat_arm_retaliation_effect = { STORY = $STORY$ WHEN = miss }
	}
	if = {
		limit = { $STORY$ = { has_variable = retaliate_if_attacked } }
		pod_combat_arm_retaliation_effect = { STORY = $STORY$ WHEN = any }
	}

	# ── 13b. Damage-over-Time application ─────────────────────────
	# Single-target DoT, applied after direct damage. Replacement and kind
	# rules: pod_combat_seed_dot_on_slot_effect.
	if = {
		limit = {
			$STORY$ = { has_variable = dot_rounds }
			var:pod_combat_session ?= { has_variable = pod_combat_target_char }
			# Use NOT(miss): a morale-only DoT can land without an HP contact
			# result.
			NOT = { pod_combat_result_code_indicates_miss_trigger = yes }
		}
		save_scope_value_as = { name = pod_combat_dot_is_morale value = { value = 0  if = { limit = { $STORY$ = { has_variable = dot_morale } }  value = 1 } } }
		var:pod_combat_session ?= {
			every_in_list = {
				variable = pod_combat_active_slots
				# Skip a target the same move just killed/downed (no affliction row on a corpse).
				limit = { var:slot_char = { this = scope:combat_host.var:pod_combat_session.var:pod_combat_target_char }  NOT = { pod_combat_slot_should_be_marked_ko_trigger = yes } }
				pod_combat_seed_dot_on_slot_effect = { STORY = $STORY$ AMOUNT = $STORY$.var:dot_amount ROUNDS = $STORY$.var:dot_rounds CASTER = scope:combat_host.var:pod_combat_session.var:pod_combat_actor_char }
				# Application log: morale is explicit; HP stays class-neutral because
				# the wound class resolves per victim when each later tick lands.
				if = {
					limit = { $STORY$ = { has_variable = dot_morale } }
					pod_combat_push_dot_applied_log_effect = { MESSAGE_KEY = dot_applied_morale }
				}
				else = {
					pod_combat_push_dot_applied_log_effect = { MESSAGE_KEY = dot_applied }
				}
			}
		}
	}


	# ── 13b-bonus. Conditional bonus DoT ─────────────────────────
	# Bonus HP DoT uses pod_combat_bonus_dot_rounds. Set
	# pod_combat_dot_is_morale explicitly so a separate morale_dot clause
	# cannot change its kind.
	if = {
		limit = {
			var:pod_combat_session ?= { has_variable = pod_combat_bonus_dot }
			var:pod_combat_session ?= { has_variable = pod_combat_target_char }
			NOT = { pod_combat_result_code_indicates_miss_trigger = yes }
		}
		save_scope_value_as = { name = pod_combat_dot_is_morale value = 0 }
		var:pod_combat_session ?= {
			every_in_list = {
				variable = pod_combat_active_slots
				limit = { var:slot_char = { this = scope:combat_host.var:pod_combat_session.var:pod_combat_target_char }  NOT = { pod_combat_slot_should_be_marked_ko_trigger = yes } }
				pod_combat_seed_dot_on_slot_effect = { STORY = $STORY$ AMOUNT = scope:combat_host.var:pod_combat_session.var:pod_combat_bonus_dot ROUNDS = pod_combat_bonus_dot_rounds CASTER = scope:combat_host.var:pod_combat_session.var:pod_combat_actor_char }
				pod_combat_push_dot_applied_log_effect = { MESSAGE_KEY = dot_applied }
			}
		}
	}
	# ── 13b-bonus-morale. Conditional bonus morale DoT ───────────
	# Set morale kind independently of other bonuses. The generator rejects a
	# base DoT on this move because both would overwrite the same slot_dot_*
	# state.
	if = {
		limit = {
			var:pod_combat_session ?= { has_variable = pod_combat_bonus_morale_dot }
			var:pod_combat_session ?= { has_variable = pod_combat_target_char }
			NOT = { pod_combat_result_code_indicates_miss_trigger = yes }
		}
		save_scope_value_as = { name = pod_combat_dot_is_morale value = 1 }
		var:pod_combat_session ?= {
			every_in_list = {
				variable = pod_combat_active_slots
				limit = { var:slot_char = { this = scope:combat_host.var:pod_combat_session.var:pod_combat_target_char }  NOT = { pod_combat_slot_should_be_marked_ko_trigger = yes } }
				pod_combat_seed_dot_on_slot_effect = { STORY = $STORY$ AMOUNT = scope:combat_host.var:pod_combat_session.var:pod_combat_bonus_morale_dot ROUNDS = pod_combat_bonus_dot_rounds CASTER = scope:combat_host.var:pod_combat_session.var:pod_combat_actor_char }
				pod_combat_push_dot_applied_log_effect = { MESSAGE_KEY = dot_applied_morale }
			}
		}
	}
	# ── 13c. Mind control application ───────────────────────────
	# Single-target only: an AoE would affect only the loaded target. Players
	# are immune; PoD mental defenses can resist. Control reverses the
	# target's next-turn targeting once.
	if = {
		limit = {
			$STORY$ = { has_variable = mind_control }
			var:pod_combat_session ?= { has_variable = pod_combat_target_char }
			NOT = { var:pod_combat_session ?= { has_variable = pod_combat_control_delivery_failed } }
			# Morale-delivered mind control only lands when that morale attack was
			# not evaded. Pure mind-control rows use the control-contact tripwire.
			OR = {
				$STORY$ = { NOT = { has_variable = morale_damage } }
				var:pod_combat_session ?= { has_variable = pod_combat_last_morale_damage_amount }
			}
		}
		if = {
			limit = {
				pod_combat_target_is_dominatable_trigger = yes
				trigger_if = {
					limit = { var:pod_combat_session.var:pod_combat_target_char = { POD_has_strong_mental_defenses_trigger = yes } }
					var:pod_combat_session ?= { var:pod_combat_actor_char = { can_bypass_mental_defenses_trigger = yes } }
				}
			}
			var:pod_combat_session ?= {
				every_in_list = {
					variable = pod_combat_active_slots
					limit = { var:slot_char = { this = scope:combat_host.var:pod_combat_session.var:pod_combat_target_char } }
					set_variable = { name = slot_controlled value = 1 }
					var:slot_char = { save_scope_as = mind_log_victim }
					pod_combat_push_mind_log_effect = { MESSAGE_KEY = mind_seized }
				}
			}
		}
		else = {
			var:pod_combat_session ?= { var:pod_combat_target_char = { save_scope_as = mind_log_victim } }
			pod_combat_push_mind_log_effect = { MESSAGE_KEY = mind_resisted }
		}
	}

	# ── 14. Parametric non-tier flags ────────────────────────────
	if = {
		limit = { $STORY$ = { has_variable = ap_gain } }
		var:pod_combat_session ?= { change_variable = { name = pod_combat_current_ap add = $STORY$.var:ap_gain } }
	}

	# The generator keeps total refunds below ap_cost. Log only the refund
	# that fired.
	if = {
		limit = { var:pod_combat_session ?= { has_variable = pod_combat_bonus_ap_refund } }
		var:pod_combat_session ?= { change_variable = { name = pod_combat_current_ap add = var:pod_combat_bonus_ap_refund } }
		var:pod_combat_session ?= { set_variable = { name = pod_combat_last_ap_refund value = var:pod_combat_bonus_ap_refund } }
	}
	# AP drain is consumed at the target's next turn start; initiative debuffs
	# at round end. Sum base and bonus drain before comparing with the pending
	# drain, or a bonus could incorrectly augment another attacker's stronger
	# value.
	if = {
		limit = {
			NOT = { var:pod_combat_session ?= { has_variable = pod_combat_control_delivery_failed } }
			OR = {
				$STORY$ = { has_variable = ap_drain }
				var:pod_combat_session ?= { has_variable = pod_combat_bonus_ap_drain }
			}
		}
		save_scope_value_as = {
			name = effective_ap_drain
			value = {
				value = 0
				if = { limit = { $STORY$ = { has_variable = ap_drain } } add = $STORY$.var:ap_drain }
				if = {
					limit = { var:pod_combat_session ?= { has_variable = pod_combat_bonus_ap_drain } }
					add = var:pod_combat_session.var:pod_combat_bonus_ap_drain
				}
			}
		}
		# A weaker drain must not reduce a stronger pending drain.
		if = {
			limit = { NOT = { var:pod_combat_session ?= { has_variable = pod_combat_target_ap_drain } } }
			var:pod_combat_session ?= { set_variable = { name = pod_combat_target_ap_drain value = scope:effective_ap_drain } }
		}
		else_if = {
			limit = { var:pod_combat_session.var:pod_combat_target_ap_drain < scope:effective_ap_drain }
			var:pod_combat_session ?= { set_variable = { name = pod_combat_target_ap_drain value = scope:effective_ap_drain } }
		}
		# Log the combined drain inside the delivery gate so resisted control
		# reports no drain.
		if = {
			limit = { scope:effective_ap_drain > 0 }
			var:pod_combat_session ?= { set_variable = { name = pod_combat_last_ap_drain value = scope:effective_ap_drain } }
		}
	}
	if = {
		limit = {
			$STORY$ = { has_variable = initiative_debuff }
			NOT = { var:pod_combat_session ?= { has_variable = pod_combat_control_delivery_failed } }
		}
		save_scope_value_as = {
			name = init_debuff_neg
			value = {
				value = $STORY$.var:initiative_debuff
				multiply = -1
			}
		}
		# Accumulate initiative debuffs from all actions until the round-end
		# calculation.
		if = {
			limit = { NOT = { var:pod_combat_session ?= { has_variable = pod_combat_target_init_bonus } } }
			var:pod_combat_session ?= { set_variable = { name = pod_combat_target_init_bonus value = scope:init_debuff_neg } }
		}
		else = {
			var:pod_combat_session ?= { change_variable = { name = pod_combat_target_init_bonus add = scope:init_debuff_neg } }
		}
	}

	# ── 14b. Summon a combatant ──────────────────────────────────
	# Summon after damage so this action's AoE cannot hit the new unit. It
	# acts next round and cannot retaliate here.
	if = {
		limit = { $STORY$ = { has_variable = summons_combatant } }
		pod_combat_summon_combatant_glue_effect = { MOVE_STORY = $STORY$ }
	}

	# ── 15. Retaliation check ────────────────────────────────────
	# Skip team AoE: this check would select only the pre-AoE target's
	# reaction.
	if = {
		limit = {
			$STORY$ = { has_variable = hp_damage }
			$STORY$ = { NOR = { var:targeting = flag:team_enemy  var:targeting = flag:team_all } }
		}
		pod_combat_check_retaliation_effect = { TARGET_SIDE = target ACTOR_SIDE = actor }
	}

	# ── 16. Push combat log entry ────────────────────────────────
	# AoE already logs each target. Add a final row only for later effects
	# those rows cannot capture, including recoil, recovery, energy, and AP
	# changes.
	if = {
		limit = {
			OR = {
				NOT = { var:pod_combat_session ?= { has_variable = pod_combat_aoe_phase_pushed_rows } }
				var:pod_combat_session ?= { has_variable = pod_combat_last_heal_amount }
				var:pod_combat_session ?= { has_variable = pod_combat_last_morale_heal_amount }
				var:pod_combat_session ?= { has_variable = pod_combat_last_retal_fired }
				var:pod_combat_session ?= { has_variable = pod_combat_last_self_damage }
				var:pod_combat_session ?= { has_variable = pod_combat_last_energy_gain }
				var:pod_combat_session ?= { has_variable = pod_combat_last_energy_drain }
				var:pod_combat_session ?= { has_variable = pod_combat_last_ap_refund }
				var:pod_combat_session ?= { has_variable = pod_combat_last_ap_drain }
			}
		}
		pod_combat_push_log_effect = { MOVE_STORY = $STORY$ }
	}

	# ── 17. Bonus teardown ───────────────────────────────────────
	# Recalculate hit chance before clearing bonus state so phase 5d cannot
	# affect the next action.
	if = {
		limit = {
			var:pod_combat_session ?= { has_variable = pod_combat_bonus_hit }
			var:pod_combat_session ?= { has_variable = pod_combat_current_hit_chance }
		}
		pod_combat_recalc_hit_actor_effect = yes
	}
	# Keep this clear list aligned with phase 2b; apply sites test presence
	# only.
	var:pod_combat_session ?= {
		remove_variable = pod_combat_bonus_damage
		remove_variable = pod_combat_bonus_typed_damage_superficial
		remove_variable = pod_combat_bonus_typed_damage_aggravated
		remove_variable = pod_combat_bonus_typed_damage_fire
		remove_variable = pod_combat_bonus_typed_damage_cold
		remove_variable = pod_combat_bonus_typed_damage_lightning
		remove_variable = pod_combat_bonus_typed_damage_sun
		remove_variable = pod_combat_bonus_typed_damage_silver
		remove_variable = pod_combat_bonus_typed_damage_gold
		remove_variable = pod_combat_bonus_typed_damage_holy
		remove_variable = pod_combat_bonus_typed_damage_balefire
		remove_variable = pod_combat_bonus_typed_damage_poison
		remove_variable = pod_combat_bonus_typed_damage_cold_iron
		remove_variable = pod_combat_bonus_hit
		remove_variable = pod_combat_bonus_morale_damage
		remove_variable = pod_combat_bonus_self_heal
		remove_variable = pod_combat_bonus_heal_target
		remove_variable = pod_combat_bonus_self_morale_heal
		remove_variable = pod_combat_bonus_energy_gain
		remove_variable = pod_combat_bonus_energy_drain
		remove_variable = pod_combat_bonus_dot
		remove_variable = pod_combat_bonus_morale_dot
		remove_variable = pod_combat_bonus_ap_refund
		remove_variable = pod_combat_bonus_ap_drain
		remove_variable = pod_combat_bonus_retaliate_boost
		remove_variable = pod_combat_bonus_hit_buff_round
		remove_variable = pod_combat_bonus_hit_debuff_round
		remove_variable = pod_combat_bonus_soak_buff_round
		remove_variable = pod_combat_bonus_soak_debuff_round
		remove_variable = pod_combat_bonus_hit_buff_combat
		remove_variable = pod_combat_bonus_hit_debuff_combat
		remove_variable = pod_combat_bonus_soak_buff_combat
		remove_variable = pod_combat_bonus_soak_debuff_combat
		remove_variable = pod_combat_bonus_aggravated
		remove_variable = pod_combat_bonus_stance_guard
		remove_variable = pod_combat_bonus_stance_dodge
		remove_variable = pod_combat_bonus_stance_stealth
		remove_variable = pod_combat_bonus_uses_weapon
		remove_variable = pod_combat_bonus_class
	}
	pod_combat_clear_actor_weapon_profile_effect = yes
	# Clear any markers left by dispatch. Skip dead actors: retaliation may
	# have killed them, and variable mutations would error.
	var:pod_combat_session.var:pod_combat_actor_char ?= {
		if = {
			limit = { is_alive = yes }
			remove_variable = pod_combat_force_agg
			remove_variable = pod_combat_weapon_attack
			remove_variable = pod_combat_weapon_soak_pen
		}
	}
}


######################################################
###### HP DAMAGE DISPATCH (single target)
######################################################
# Two-stage dispatch:
#   1. pod_combat_apply_single_hp_damage_effect - maps the move's damage
#      class (var:damage_type) to a PoD death_reason suffix (REASON) and
#      handles the change_type bonus, then forwards.
#   2. pod_combat_dispatch_damage_pipeline_effect - runs the unified
#      pod_combat_resolve_contact_effect (capability-flag-driven) and
#      fires the primary and optional typed-damage effects on hit.

pod_combat_apply_single_hp_damage_effect = {
	# $STORY$ = move story
	# $TARGET_SIDE$ = "target" (the workspace side that takes the damage)
	#
	# Single-axis dispatch on the move's damage classification (see
	# POD_combat_move_set_damage_class): an element rides its REASON into
	# PoD core so immunities / downgrades / splat multipliers apply per
	# target; `aggravated` forces duel_aggravated; superficial = duel.
	# The switch's kind table is mirrored by the change_class and weapon-retype
	# switches just below it, the typed-bonus chain in
	# pod_combat_apply_typed_bonus_damage_effect, the retaliation effect
	# (pod_combat_deal_typed_retal_damage_effect) and the DoT tick switch
	# (pod_combat_deal_dot_tick_from_host_effect). Adding a class means touching
	# every one of them plus the §2b bucket arms - see "Adding a new damage
	# class" in COMBAT_REFERENCE.info.
	#
	# The change_class bonus (fight var pod_combat_bonus_class) converts a
	# superficial move's damage to a named element for one action when a clause
	# granted it. It outranks the weapon retype and, like it, declines against a
	# victim immune to the converted element rather than dealing nothing.
	#
	# The change_type bonus (fight var pod_combat_bonus_aggravated):
	#   - on a superficial move it upgrades the REASON to duel_aggravated;
	#   - on an elemental move the element must survive (immunity checks
	#     key off it), so the ACTOR gets a transient pod_combat_force_agg
	#     marker the core classifier reads via scope:wound_attacker (the
	#     devpod delta in POD_health_track_triggers.txt) and the elemental
	#     REASON rides through, force-classified aggravated.
	#
	# A uses_weapon move adds a second, actor-side source of classification:
	# the equipped weapon's retype (host pod_combat_weapon_class, seeded in
	# §2c) and the wielded-artifact aggravation list. The retype is only ever
	# seeded over a superficial move, so it cannot collide with the branches
	# above.
	#
	# Save the move as scope:current_move so the resolver (and the whole dispatch
	# chain below it) reads capability flags without re-threading $STORY$.
	$STORY$ = { save_scope_as = current_move }

	# Limit artifact aggravation to armed weapon dispatches. PoD's artifact
	# list has no slot/benefit filter, so the armed check is required. Recoil,
	# counters, and DoTs must not inherit this marker.
	if = {
		limit = {
			scope:combat_host = { pod_combat_move_swings_weapon_trigger = { STORY = scope:current_move } }
			scope:combat_host.var:pod_combat_session.var:pod_combat_actor_char ?= { pod_combat_actor_is_armed_trigger = yes }
		}
		scope:combat_host.var:pod_combat_session.var:pod_combat_actor_char ?= { set_variable = pod_combat_weapon_attack }
		# PoD reads soak penetration from the actor. Match the weapon marker's
		# lifetime so retaliation cannot inherit it.
		if = {
			limit = { scope:combat_host = { var:pod_combat_session ?= { has_variable = pod_combat_weapon_soak_penalty } } }
			scope:combat_host.var:pod_combat_session.var:pod_combat_actor_char ?= {
				set_variable = { name = pod_combat_weapon_soak_pen value = scope:combat_host.var:pod_combat_session.var:pod_combat_weapon_soak_penalty }
			}
		}
	}

	if = {
		limit = {
			OR = {
				$STORY$ = { has_variable = damage_type var:damage_type = flag:aggravated }
				AND = {
					scope:combat_host = { var:pod_combat_session ?= { has_variable = pod_combat_bonus_aggravated } }
					$STORY$ = { NOT = { pod_combat_move_class_is_elemental_trigger = yes } }
					# Preserve the elemental REASON for immunity checks.
					scope:combat_host = { NOT = { var:pod_combat_session ?= { has_variable = pod_combat_weapon_class } } }
				}
			}
		}
		pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = duel_aggravated }
	}
	else = {
		if = {
			limit = {
				scope:combat_host = { var:pod_combat_session ?= { has_variable = pod_combat_bonus_aggravated } }
				OR = {
					$STORY$ = { pod_combat_move_class_is_elemental_trigger = yes }
					scope:combat_host.var:pod_combat_session ?= { has_variable = pod_combat_weapon_class }
				}
			}
			scope:combat_host.var:pod_combat_session.var:pod_combat_actor_char ?= { set_variable = pod_combat_force_agg }
		}
		# change_class precedes weapon retype, including invalid data
		# containing both. Fall back per victim when the converted class is
		# immune. Run the test from combat_host because it reads the session
		# relative to its caller.
		if = {
			limit = {
				scope:combat_host = {
					var:pod_combat_session ?= { has_variable = pod_combat_bonus_class }
					NOT = {
						pod_combat_stamped_type_is_wasted_trigger = {
							KIND_VAR = pod_combat_bonus_class
							VICTIM   = var:pod_combat_session.var:pod_combat_$TARGET_SIDE$_char
						}
					}
				}
			}
			switch = {
				trigger = scope:combat_host.var:pod_combat_session.var:pod_combat_bonus_class
				flag:fire      = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = fire } }
				flag:cold      = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = cold } }
				flag:lightning = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = lightning } }
				flag:sun       = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = sun } }
				flag:silver    = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = silver } }
				flag:gold      = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = gold } }
				flag:holy      = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = true_faith } }
				flag:balefire  = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = balefire } }
				flag:poison    = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = poison } }
				flag:cold_iron = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = cold_iron } }
				fallback = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = duel } }
			}
		}
		# Retype applies only over superficial damage; keep the original class
		# against victims immune to the replacement.
		else_if = {
			limit = {
				scope:combat_host = {
					var:pod_combat_session ?= { has_variable = pod_combat_weapon_class }
					NOT = {
						pod_combat_weapon_retype_is_wasted_trigger = { VICTIM = var:pod_combat_session.var:pod_combat_$TARGET_SIDE$_char }
					}
				}
			}
			switch = {
				trigger = scope:combat_host.var:pod_combat_session.var:pod_combat_weapon_class
				flag:fire      = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = fire } }
				flag:cold      = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = cold } }
				flag:lightning = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = lightning } }
				flag:sun       = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = sun } }
				flag:silver    = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = silver } }
				flag:gold      = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = gold } }
				flag:holy      = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = true_faith } }
				flag:balefire  = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = balefire } }
				flag:poison    = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = poison } }
				flag:cold_iron = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = cold_iron } }
				fallback = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = duel } }
			}
		}
		else = {
			switch = {
				trigger = scope:current_move.var:damage_type
				flag:fire      = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = fire } }
				flag:cold      = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = cold } }
				flag:lightning = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = lightning } }
				flag:sun       = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = sun } }
				flag:silver    = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = silver } }
				flag:gold      = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = gold } }
				# holy keys PoD's aggravated promotion against true_faith.
				flag:holy      = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = true_faith } }
				flag:balefire  = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = balefire } }
				flag:poison    = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = poison } }
				flag:cold_iron = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = cold_iron } }
				# superficial (the hp default) or unset -> generic physical.
				fallback = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = duel } }
			}
		}
		# Classification is synchronous inside the wound call, so the
		# marker can drop immediately.
		scope:combat_host.var:pod_combat_session.var:pod_combat_actor_char ?= { remove_variable = pod_combat_force_agg }
	}
	# Same reasoning for the weapon marker, but it is set before the
	# aggravated branch so it has to drop outside the if/else.
	scope:combat_host.var:pod_combat_session.var:pod_combat_actor_char ?= {
		remove_variable = pod_combat_weapon_attack
		remove_variable = pod_combat_weapon_soak_pen
	}
}

pod_combat_apply_typed_bonus_damage_effect = {
	# $TARGET_SIDE$ = "actor" or "target"
	# A typed damage bonus shares the move's contact result, then resolves as its
	# own PoD wound packet - one per class bucket §2b filled, fired in the class
	# order below, which is therefore also the order the log slots are claimed.
	#
	# There is deliberately no immunity waste guard here (unlike change_class and
	# the weapon retype): a rider is additive, so an immune class costs the move
	# nothing and reads "Immune" on its own log segment.
	#
	# Host scope on every call path, so the bare pointer resolves - the same shape
	# the dispatch chain around this effect uses.
	if = {
		limit = {
			# Nested, never flat siblings: a Jomini AND evaluates every clause, so a
			# bare deref beside its own `exists` still errors on a dangling pointer.
			# Same shape as the primary route's guard in pod_combat_effects.txt.
			trigger_if = {
				limit = { exists = var:pod_combat_session }
				trigger_if = {
					limit = { exists = var:pod_combat_session.var:pod_combat_$TARGET_SIDE$_char }
					var:pod_combat_session.var:pod_combat_$TARGET_SIDE$_char = { is_alive = yes }
				}
				trigger_else = { always = no }
			}
			trigger_else = { always = no }
		}
		# ONE strip, ahead of every typed packet: the main packet's weapon retype,
		# soak shred and change_type marker must not reclassify an explicitly
		# authored rider. pod_combat_apply_single_hp_damage_effect re-stamps all
		# three per victim, so an AoE sweep's later victims keep theirs.
		var:pod_combat_session.var:pod_combat_actor_char ?= {
			remove_variable = pod_combat_force_agg
			remove_variable = pod_combat_weapon_attack
			remove_variable = pod_combat_weapon_soak_pen
		}
		pod_combat_apply_typed_bonus_class_effect = { TARGET_SIDE = $TARGET_SIDE$ BUCKET_VAR = pod_combat_bonus_typed_damage_superficial CLASS = superficial REASON = duel }
		pod_combat_apply_typed_bonus_class_effect = { TARGET_SIDE = $TARGET_SIDE$ BUCKET_VAR = pod_combat_bonus_typed_damage_aggravated  CLASS = aggravated  REASON = duel_aggravated }
		pod_combat_apply_typed_bonus_class_effect = { TARGET_SIDE = $TARGET_SIDE$ BUCKET_VAR = pod_combat_bonus_typed_damage_fire        CLASS = fire        REASON = fire }
		pod_combat_apply_typed_bonus_class_effect = { TARGET_SIDE = $TARGET_SIDE$ BUCKET_VAR = pod_combat_bonus_typed_damage_cold        CLASS = cold        REASON = cold }
		pod_combat_apply_typed_bonus_class_effect = { TARGET_SIDE = $TARGET_SIDE$ BUCKET_VAR = pod_combat_bonus_typed_damage_lightning   CLASS = lightning   REASON = lightning }
		pod_combat_apply_typed_bonus_class_effect = { TARGET_SIDE = $TARGET_SIDE$ BUCKET_VAR = pod_combat_bonus_typed_damage_sun         CLASS = sun         REASON = sun }
		pod_combat_apply_typed_bonus_class_effect = { TARGET_SIDE = $TARGET_SIDE$ BUCKET_VAR = pod_combat_bonus_typed_damage_silver      CLASS = silver      REASON = silver }
		pod_combat_apply_typed_bonus_class_effect = { TARGET_SIDE = $TARGET_SIDE$ BUCKET_VAR = pod_combat_bonus_typed_damage_gold        CLASS = gold        REASON = gold }
		pod_combat_apply_typed_bonus_class_effect = { TARGET_SIDE = $TARGET_SIDE$ BUCKET_VAR = pod_combat_bonus_typed_damage_holy        CLASS = holy        REASON = true_faith }
		pod_combat_apply_typed_bonus_class_effect = { TARGET_SIDE = $TARGET_SIDE$ BUCKET_VAR = pod_combat_bonus_typed_damage_balefire    CLASS = balefire    REASON = balefire }
		pod_combat_apply_typed_bonus_class_effect = { TARGET_SIDE = $TARGET_SIDE$ BUCKET_VAR = pod_combat_bonus_typed_damage_poison      CLASS = poison      REASON = poison }
		pod_combat_apply_typed_bonus_class_effect = { TARGET_SIDE = $TARGET_SIDE$ BUCKET_VAR = pod_combat_bonus_typed_damage_cold_iron   CLASS = cold_iron   REASON = cold_iron }
	}
}

# One class bucket -> one PoD wound packet. $BUCKET_VAR$ = the literal bucket
# name, $CLASS$ = the class key the log stamps as a flag, $REASON$ = the PoD
# death_reason suffix. Host scope, called once per class from the chain above.
pod_combat_apply_typed_bonus_class_effect = {
	if = {
		limit = {
			trigger_if = {
				limit = { exists = var:pod_combat_session }
				var:pod_combat_session = { has_variable = $BUCKET_VAR$ }
				# Re-checked per packet, not just once for the chain: deaths are
				# deferred for every ordinary combatant, but a clone dies INSIDE the
				# wound call, so an earlier packet can leave this victim a corpse.
				var:pod_combat_session.var:pod_combat_$TARGET_SIDE$_char = { is_alive = yes }
			}
			trigger_else = { always = no }
		}
		# Resolve amount before entering the victim scope, where
		# POD_increase_wounds_effect reads COUNT.
		var:pod_combat_session ?= {
			save_scope_value_as = { name = typed_bonus_amount value = var:$BUCKET_VAR$ }
		}
		pod_combat_apply_typed_bonus_damage_payload_effect = {
			TARGET_SIDE = $TARGET_SIDE$
			AMOUNT      = scope:typed_bonus_amount
			REASON      = $REASON$
			CLASS       = $CLASS$
		}
	}
}
pod_combat_dispatch_damage_pipeline_effect = {
	# $STORY$ = move story
	# $TARGET_SIDE$ = "actor" or "target"
	# $REASON$ = PoD death_reason suffix (passed through to the payload)
	# Runs the resolver, fires the payload on hit, consumes actor stealth.
	# Relies on scope:current_move (saved in pod_combat_apply_single_hp_damage_effect).
	pod_combat_resolve_contact_effect = {
		TARGET_SIDE = $TARGET_SIDE$
		ACTOR_SIDE  = actor
	}
	if = {
		limit = { scope:contact_result = 0 }
		# One successful contact, however many independently-routed wound packets
		# it carries, remains one attack for trace hit-rate accounting.
		if = {
			limit = { exists = global_var:pod_combat_tune_ai_trace }
			pod_combat_trace_add_effect = { NAME = hits VALUE = 1 }
		}
		save_scope_value_as = {
			name = effective_hp_damage
			value = {
				value = $STORY$.var:hp_damage
				if = {
					limit = { scope:combat_host = { var:pod_combat_session ?= { has_variable = pod_combat_bonus_damage } } }
					add = scope:combat_host.var:pod_combat_session.var:pod_combat_bonus_damage
				}
				if = {
					limit = { scope:combat_host.var:pod_combat_session ?= { has_variable = pod_combat_weapon_bonus_damage } }
					add = scope:combat_host.var:pod_combat_session.var:pod_combat_weapon_bonus_damage
				}
			}
		}
		pod_combat_apply_primary_damage_payload_effect = {
			TARGET_SIDE = $TARGET_SIDE$
			AMOUNT      = scope:effective_hp_damage
			REASON      = $REASON$
		}
		pod_combat_apply_typed_bonus_damage_effect = { TARGET_SIDE = $TARGET_SIDE$ }
	}
	# Every attack consumes stealth, including misses. AoE defers consumption
	# until after the sweep so all targets use the stealth accuracy bonus.
	if = {
		limit = { scope:combat_host = { NOT = { var:pod_combat_session ?= { has_variable = pod_combat_aoe_suppress_stealth_consume } } } }
		pod_combat_consume_actor_stealth_effect = { ACTOR_SIDE = actor }
	}
}


######################################################
###### AOE DISPATCH (team_enemy + team_ally)
######################################################
# Resolve ally/enemy relative to the actor. Save each damaged target before
# its KO check.
# ── Scope handling ──────────────────────────────────────────────
# Damage/heal AoE uses the workspace round-trip; tier/stance AoE writes slots
# directly. scope:host is the iterator-local host; scope:combat_host is the
# execution entry host used by nested damage. See COMBAT_REFERENCE.info
# section 7.


# $STORY$ = move story. Caller must save scope:host before iterating slots.
# Advance burst position before pushing, then clear per-target results.
pod_combat_aoe_push_per_target_log_effect = {
	scope:host = {
		if = {
			limit = { var:pod_combat_session ?= { has_variable = pod_combat_aoe_phase_pushed_rows } }
			var:pod_combat_session ?= { change_variable = { name = pod_combat_aoe_iter_burst_index add = 1 } }
		}
		else = {
			var:pod_combat_session ?= { set_variable = { name = pod_combat_aoe_iter_burst_index value = 0 } }
		}
		var:pod_combat_session ?= { set_variable = { name = pod_combat_aoe_phase_pushed_rows value = 1 } }
		pod_combat_push_log_effect = { MOVE_STORY = $STORY$ }
		pod_combat_clear_per_target_state_effect = yes
	}
}


pod_combat_apply_team_enemy_hp_damage_effect = {
	# $STORY$ = move story
	# Pushes one log row per target so the log reflects each enemy's outcome,
	# not just the last. Canonical AoE per-iteration sequence (other AoE
	# effects reference this one):
	#   1. save_target_to_story - persist per-target workspace state to the slot.
	#   2. aoe_check_target_alive - flip slot_alive on KO / morale-break.
	#   3. (inside the log effect) set pod_combat_aoe_phase_pushed_rows so §16 skips
	#      its wrap-up row unless actor-side state fired.
	#   4. push_log - the per-target row.
	#   5. clear per-target last_* (keeps the pushed-rows flag).
	save_scope_as = host
	# Defer stealth consumption until all targets receive its accuracy bonus.
	scope:combat_host.var:pod_combat_session ?= { set_variable = pod_combat_aoe_suppress_stealth_consume }
	var:pod_combat_session ?= {
		every_in_list = {
			variable = pod_combat_active_slots
			limit = {
				pod_combat_slot_is_alive_trigger = yes
				pod_combat_aoe_team_enemy_filter_trigger = yes
			}
			pod_combat_load_target_from_story_effect = yes
			scope:host = {
				pod_combat_apply_single_hp_damage_effect = { STORY = $STORY$ TARGET_SIDE = target }
				# Preserve any-HP-damage before the row push clears per-target

					var:pod_combat_session ?= { set_variable = pod_combat_aoe_any_hp_damage }
				}
			}
			pod_combat_save_target_to_story_effect = yes
			pod_combat_aoe_check_target_alive_effect = yes
			pod_combat_aoe_push_per_target_log_effect = { STORY = $STORY$ }
		}
	}
	# Consume stealth before target reload recalculates actor hit chance.
	# Target-only reload preserves the live actor workspace.
	scope:combat_host.var:pod_combat_session ?= { remove_variable = pod_combat_aoe_suppress_stealth_consume }
	pod_combat_consume_actor_stealth_effect = { ACTOR_SIDE = actor }
	scope:host = { pod_combat_reload_selected_target_effect = yes }
}

pod_combat_apply_team_all_hp_damage_effect = {
	# $STORY$ = move story
	# The team_enemy sequence (see pod_combat_apply_team_enemy_hp_damage_effect for
	# the per-iteration pattern and the burst-stealth window) run over
	# both teams, so the actor's own side takes the hit too. A KO mid-sweep does
	# not stop the loop - victory is only scored once the whole move completes.
	#
	# The CASTER is hit as well, but after the sweep and through the actor
	# workspace rather than from inside the loop. Sweeping its own slot would put
	# the same combatant in both workspace sides, and the end-of-move actor flush
	# rewrites every mirrored slot field from the actor workspace - so the loop's
	# slot-side writes for the caster would be silently reverted (morale most
	# visibly; HP survives only because it lives on PoD's wound track, not the
	# slot). Routing it through the actor side keeps one authoritative copy.
	save_scope_as = host
	scope:combat_host.var:pod_combat_session ?= { set_variable = pod_combat_aoe_suppress_stealth_consume }
	var:pod_combat_session ?= {
		every_in_list = {
			variable = pod_combat_active_slots
			limit = {
				pod_combat_slot_is_alive_trigger = yes
				pod_combat_aoe_team_all_sweep_filter_trigger = yes
			}
			pod_combat_load_target_from_story_effect = yes
			scope:host = {
				pod_combat_apply_single_hp_damage_effect = { STORY = $STORY$ TARGET_SIDE = target }
				if = {
					limit = { pod_combat_result_code_indicates_hp_damage_trigger = yes }
					var:pod_combat_session ?= { set_variable = pod_combat_aoe_any_hp_damage }
				}
			}
			pod_combat_save_target_to_story_effect = yes
			pod_combat_aoe_check_target_alive_effect = yes
			pod_combat_aoe_push_per_target_log_effect = { STORY = $STORY$ }
		}
	}
	# Self-damage bypasses contact resolution but retains PoD soak and
	# classification. Seed a literal kind, including the duel default.
	var:pod_combat_session ?= {
		pod_combat_seed_kind_from_class_effect = { STORY = $STORY$ KIND_VAR = pod_combat_team_all_self_kind }
		pod_combat_override_kind_from_weapon_effect = { KIND_VAR = pod_combat_team_all_self_kind }
	}
	# change_class takes precedence. Skip the immunity fallback for
	# self-damage: immunity should protect the caster.
	var:pod_combat_session ?= { pod_combat_override_kind_from_bonus_class_effect = { KIND_VAR = pod_combat_team_all_self_kind } }
	pod_combat_deal_typed_retal_damage_effect = {
		TARGET_SIDE = actor
		AMOUNT      = $STORY$.var:hp_damage
		KIND_VAR    = pod_combat_team_all_self_kind
	}
	var:pod_combat_session ?= { remove_variable = pod_combat_team_all_self_kind }
	# Add to recoil so the final AoE row logs self-damage without overwriting
	# a separate self_damage rider.
	if = {
		limit = { scope:damage_dealt > 0 }
		if = {
			limit = { var:pod_combat_session ?= { has_variable = pod_combat_last_self_damage } }
			var:pod_combat_session ?= { change_variable = { name = pod_combat_last_self_damage add = scope:damage_dealt } }
		}
		else = {
			var:pod_combat_session ?= { set_variable = { name = pod_combat_last_self_damage value = scope:damage_dealt } }
		}
	}
	scope:combat_host.var:pod_combat_session ?= { remove_variable = pod_combat_aoe_suppress_stealth_consume }
	pod_combat_consume_actor_stealth_effect = { ACTOR_SIDE = actor }
	scope:host = { pod_combat_reload_selected_target_effect = yes }
}

# Arm one retaliation stance on the actor's workspace. $STORY$ = move story;
# $WHEN$ = hit | miss | any (the pod_combat_actor_retaliate_<WHEN>* var family).
# Take-strongest compares within one family only, so an `any` charge never
# displaces - nor is displaced by - an armed hit/miss charge.
# Amount fallback: retaliate_damage -> hp_damage -> 1, staged into
# pod_combat_retal_arm_candidate so the take-strongest test and the write agree
# by construction, then boosted by a fired retaliate_boost bonus. Kind
# precedence: morale > retaliate_agg > the move's damage class element >
# a granted change_class element > the live weapon retype > plain duel.
# The extra _morale presence flag feeds the stance tooltip (a bare
# flag-compare on the slot-story custom-loc root never matches, so the
# customizable localization reads this).
pod_combat_arm_retaliation_effect = {
	# Compare and write the same boosted amount.
	if = {
		limit = { $STORY$ = { has_variable = retaliate_damage } }
		var:pod_combat_session ?= { set_variable = { name = pod_combat_retal_arm_candidate value = $STORY$.var:retaliate_damage } }
	}
	else_if = {
		limit = { $STORY$ = { has_variable = hp_damage } }
		var:pod_combat_session ?= { set_variable = { name = pod_combat_retal_arm_candidate value = $STORY$.var:hp_damage } }
	}
	else = {
		# An armed charge needs an amount even when the move declares no
		# damage.
		var:pod_combat_session ?= { set_variable = { name = pod_combat_retal_arm_candidate value = 1 } }
	}
	# Apply the summed boost before comparison, independently for each charge
	# family.
	if = {
		limit = { var:pod_combat_session ?= { has_variable = pod_combat_bonus_retaliate_boost } }
		var:pod_combat_session ?= { change_variable = { name = pod_combat_retal_arm_candidate add = var:pod_combat_bonus_retaliate_boost } }
	}
	# Equal or weaker amounts preserve the current kind as well as the amount.
	if = {
		limit = {
			trigger_if = {
				limit = { var:pod_combat_session ?= { has_variable = pod_combat_actor_retaliate_$WHEN$ } }
				var:pod_combat_session.var:pod_combat_actor_retaliate_$WHEN$ < var:pod_combat_session.var:pod_combat_retal_arm_candidate
			}
			trigger_else = { always = yes }
		}
		# Clear first: the kind chain has no final else for a plain counter.
		var:pod_combat_session ?= {
			remove_variable = pod_combat_actor_retaliate_$WHEN$_kind
			remove_variable = pod_combat_actor_retaliate_$WHEN$_kind_from_weapon
			remove_variable = pod_combat_actor_retaliate_$WHEN$_morale
		}
		var:pod_combat_session ?= { set_variable = { name = pod_combat_actor_retaliate_$WHEN$ value = var:pod_combat_retal_arm_candidate } }
		if = {
			limit = { $STORY$ = { has_variable = retaliate_morale } }
			var:pod_combat_session ?= { set_variable = { name = pod_combat_actor_retaliate_$WHEN$_kind value = flag:morale } }
			var:pod_combat_session ?= { set_variable = { name = pod_combat_actor_retaliate_$WHEN$_morale value = yes } }
		}
		else_if = {
			limit = { $STORY$ = { has_variable = retaliate_agg } }
			var:pod_combat_session ?= { set_variable = { name = pod_combat_actor_retaliate_$WHEN$_kind value = flag:aggravated } }
		}
		else_if = {
			limit = { $STORY$ = { pod_combat_move_class_is_elemental_trigger = yes } }
			# Literal-flags-only seeding
			var:pod_combat_session ?= { pod_combat_seed_kind_from_class_effect = { STORY = $STORY$ KIND_VAR = pod_combat_actor_retaliate_$WHEN$_kind } }
		}
		else_if = {
			# A change_class conversion a clause granted carries into the counter
			# it armed. Ranks below an authored class (which the generator anyway
			# forbids pairing with it) and above the weapon.
			limit = { var:pod_combat_session ?= { has_variable = pod_combat_bonus_class } }
			var:pod_combat_session ?= { pod_combat_override_kind_from_bonus_class_effect = { KIND_VAR = pod_combat_actor_retaliate_$WHEN$_kind } }
			# Same provenance marker as the weapon arm below: the name is
			# historical, what it means downstream is "transient-source kind, so
			# stand down against an immune victim rather than counter for zero".
			var:pod_combat_session ?= { set_variable = { name = pod_combat_actor_retaliate_$WHEN$_kind_from_weapon value = yes } }
		}
		else_if = {
			# A counter armed while holding a typed weapon strikes back with it.
			# Ranks below an authored class, which is the move's own identity.
			limit = { scope:combat_host.var:pod_combat_session ?= { has_variable = pod_combat_weapon_class } }
			var:pod_combat_session ?= { pod_combat_override_kind_from_weapon_effect = { KIND_VAR = pod_combat_actor_retaliate_$WHEN$_kind } }
			# Provenance, for the fire-time waste guard: only a WEAPON-derived kind
			# stands down against an immune victim. An authored elemental counter
			# keeps its identity and may legitimately read immune. The victim is
			# unknowable until the counter fires, so the test cannot run here; this
			# marker rides the slot round-trip with the other stance fields.
			var:pod_combat_session ?= { set_variable = { name = pod_combat_actor_retaliate_$WHEN$_kind_from_weapon value = yes } }
		}
	}
	var:pod_combat_session ?= { remove_variable = pod_combat_retal_arm_candidate }
}

# Team tier/stance broadcasts write slots and character soak bridges directly.
# Display refresh waits for workspace load; POD_soak_chance reads bridge
# changes immediately. Shared DoT and tier-write effects are below.

# Slot scope. Reapplication replaces amount, rounds, and kind. $STORY$ = move
# story; $CASTER$ = source character; $AMOUNT$/$ROUNDS$ = per-tick
# amount/count. Stage dot_applied_* scopes for the caller's log row. After
# this call, read $STORY$ or saved scopes, not inline slot vars.
#
# Caller sets scope:pod_combat_dot_is_morale to 1 for morale or 0 for HP
# before iteration. Keep it separate from fight bonuses so morale_dot cannot
# reclassify another DoT. A trigger argument containing $STORY$ would
# not expand.
pod_combat_seed_dot_on_slot_effect = {
	set_variable = { name = slot_dot_amount value = $AMOUNT$ }
	set_variable = { name = slot_dot_rounds value = $ROUNDS$ }
	# Retain the caster for kill credit and attacker-dependent wound
	# classification.
	set_variable = { name = slot_dot_source value = $CASTER$ }
	# The retype immunity checks below need the victim scope.
	var:slot_char = { save_scope_as = dot_applied_victim }
	pod_combat_seed_kind_from_class_effect = { STORY = $STORY$ KIND_VAR = slot_dot_kind }
	# Keep the authored kind when this victim is immune to the weapon retype.
	if = {
		limit = {
			scope:combat_host ?= {
				NOT = { pod_combat_weapon_retype_is_wasted_trigger = { VICTIM = scope:dot_applied_victim } }
			}
		}
		pod_combat_override_kind_from_weapon_effect = { KIND_VAR = slot_dot_kind }
	}
	# change_class overrides weapon retype unless the victim is immune. Run
	# its test from combat_host: the slot has no session pointer.
	if = {
		limit = {
			scope:combat_host ?= {
				NOT = { pod_combat_stamped_type_is_wasted_trigger = { KIND_VAR = pod_combat_bonus_class VICTIM = scope:dot_applied_victim } }
			}
		}
		pod_combat_override_kind_from_bonus_class_effect = { KIND_VAR = slot_dot_kind }
	}
	# Apply morale last so it overrides all HP classes.
	if = {
		limit = { scope:pod_combat_dot_is_morale = 1 }
		set_variable = { name = slot_dot_kind value = flag:morale }
	}
	save_scope_value_as = { name = dot_applied_amount value = $AMOUNT$ }
	save_scope_value_as = { name = dot_applied_rounds value = $ROUNDS$ }
}

# $STORY$ = move story; $SRC$ = its tier var; $DST$ = destination on the
# current scope. Enforce upgrades here as well as in selection gates,
# including bypass callers.
pod_combat_upgrade_tier_var_effect = {
	if = {
		limit = {
			$STORY$ = { has_variable = $SRC$ }
			# Nest the $DST$ read; OR would evaluate it on first application,
			# when absent.
			trigger_if = {
				limit = { has_variable = $DST$ }
				var:$DST$ < $STORY$.var:$SRC$
			}
			trigger_else = { always = yes }
		}
		set_variable = { name = $DST$ value = $STORY$.var:$SRC$ }
	}
}

# $FIGHT_SRC$ = bonus-tier transient read through combat_host
# $DST$ = tier var on the current scope. 
pod_combat_upgrade_tier_from_bonus_effect = {
	if = {
		limit = {
			# Nest both optional reads. Keep both trigger_else branches: an
			# unentered trigger_if returns true.
			trigger_if = {
				limit = { scope:combat_host.var:pod_combat_session ?= { has_variable = $FIGHT_SRC$ } }
				trigger_if = {
					limit = { has_variable = $DST$ }
					var:$DST$ < scope:combat_host.var:pod_combat_session.var:$FIGHT_SRC$
				}
				trigger_else = { always = yes }
			}
			trigger_else = { always = no }
		}
		set_variable = { name = $DST$ value = scope:combat_host.var:pod_combat_session.var:$FIGHT_SRC$ }
	}
}

# $SIDE$ = actor | target. Shared upgrade, soak-bridge, and recalculation path
# for self and selected-ally buffs.
pod_combat_apply_workspace_buff_tiers_effect = {
	# Tier-upgrade effects write to the current scope: fight for hit tiers,
	# character for soak bridges.
	var:pod_combat_session ?= {
		pod_combat_upgrade_tier_var_effect = { STORY = $STORY$ SRC = hit_buff_round_tier DST = pod_combat_$SIDE$_hit_buff_round_tier }
		pod_combat_upgrade_tier_from_bonus_effect = { FIGHT_SRC = pod_combat_bonus_hit_buff_round DST = pod_combat_$SIDE$_hit_buff_round_tier }
		pod_combat_upgrade_tier_var_effect = { STORY = $STORY$ SRC = hit_buff_combat_tier DST = pod_combat_$SIDE$_hit_buff_combat_tier }
		pod_combat_upgrade_tier_from_bonus_effect = { FIGHT_SRC = pod_combat_bonus_hit_buff_combat DST = pod_combat_$SIDE$_hit_buff_combat_tier }
	}
	if = {
		limit = { var:pod_combat_session ?= { has_variable = pod_combat_$SIDE$_char } }
		var:pod_combat_session.var:pod_combat_$SIDE$_char = {
			pod_combat_upgrade_tier_var_effect = { STORY = $STORY$ SRC = soak_buff_round_tier DST = pod_combat_tier_soak_buff_round }
			pod_combat_upgrade_tier_from_bonus_effect = { FIGHT_SRC = pod_combat_bonus_soak_buff_round DST = pod_combat_tier_soak_buff_round }
			pod_combat_upgrade_tier_var_effect = { STORY = $STORY$ SRC = soak_buff_combat_tier DST = pod_combat_tier_soak_buff_combat }
			pod_combat_upgrade_tier_from_bonus_effect = { FIGHT_SRC = pod_combat_bonus_soak_buff_combat DST = pod_combat_tier_soak_buff_combat }
		}
	}
	if = {
		limit = {
			OR = {
				$STORY$ = {
					OR = {
						has_variable = soak_buff_round_tier
						has_variable = soak_buff_combat_tier
					}
				}
				# A clause-granted tier writes no story var, so the story-only
				# gate would land the bridge write and skip its refresh.
				var:pod_combat_session ?= {
					OR = {
						has_variable = pod_combat_bonus_soak_buff_round
						has_variable = pod_combat_bonus_soak_buff_combat
					}
				}
			}
		}
		# Refresh before AI weighting and tracing read the changed soak.
		pod_combat_stash_side_live_stats_effect = { SIDE = $SIDE$ }
	}
	if = {
		limit = {
			OR = {
				$STORY$ = {
					OR = {
						has_variable = hit_buff_round_tier
						has_variable = hit_buff_combat_tier
					}
				}
				var:pod_combat_session ?= {
					OR = {
						has_variable = pod_combat_bonus_hit_buff_round
						has_variable = pod_combat_bonus_hit_buff_combat
					}
				}
			}
		}
		pod_combat_recalc_hit_$SIDE$_effect = yes
	}
}

# Target-side sibling of the buff effect. The caller owns the delivery gate;

pod_combat_apply_workspace_debuff_tiers_effect = {
	# Tier-upgrade effects write to the current scope: fight for hit tiers,
	# character for soak bridges.
	var:pod_combat_session ?= {
		pod_combat_upgrade_tier_var_effect = { STORY = $STORY$ SRC = hit_debuff_round_tier DST = pod_combat_$SIDE$_hit_debuff_round_tier }
		pod_combat_upgrade_tier_from_bonus_effect = { FIGHT_SRC = pod_combat_bonus_hit_debuff_round DST = pod_combat_$SIDE$_hit_debuff_round_tier }
		pod_combat_upgrade_tier_var_effect = { STORY = $STORY$ SRC = hit_debuff_combat_tier DST = pod_combat_$SIDE$_hit_debuff_combat_tier }
		pod_combat_upgrade_tier_from_bonus_effect = { FIGHT_SRC = pod_combat_bonus_hit_debuff_combat DST = pod_combat_$SIDE$_hit_debuff_combat_tier }
	}
	if = {
		limit = { var:pod_combat_session ?= { has_variable = pod_combat_$SIDE$_char } }
		var:pod_combat_session.var:pod_combat_$SIDE$_char = {
			pod_combat_upgrade_tier_var_effect = { STORY = $STORY$ SRC = soak_debuff_round_tier DST = pod_combat_tier_soak_debuff_round }
			pod_combat_upgrade_tier_from_bonus_effect = { FIGHT_SRC = pod_combat_bonus_soak_debuff_round DST = pod_combat_tier_soak_debuff_round }
			pod_combat_upgrade_tier_var_effect = { STORY = $STORY$ SRC = soak_debuff_combat_tier DST = pod_combat_tier_soak_debuff_combat }
			pod_combat_upgrade_tier_from_bonus_effect = { FIGHT_SRC = pod_combat_bonus_soak_debuff_combat DST = pod_combat_tier_soak_debuff_combat }
		}
	}
	if = {
		limit = {
			OR = {
				$STORY$ = {
					OR = {
						has_variable = soak_debuff_round_tier
						has_variable = soak_debuff_combat_tier
					}
				}
				var:pod_combat_session ?= {
					OR = {
						has_variable = pod_combat_bonus_soak_debuff_round
						has_variable = pod_combat_bonus_soak_debuff_combat
					}
				}
			}
		}
		pod_combat_stash_side_live_stats_effect = { SIDE = $SIDE$ }
	}
	if = {
		limit = {
			OR = {
				$STORY$ = {
					OR = {
						has_variable = hit_debuff_round_tier
						has_variable = hit_debuff_combat_tier
					}
				}
				var:pod_combat_session ?= {
					OR = {
						has_variable = pod_combat_bonus_hit_debuff_round
						has_variable = pod_combat_bonus_hit_debuff_combat
					}
				}
			}
		}
		pod_combat_recalc_hit_$SIDE$_effect = yes
	}
}

# Apply move-authored defensive stances to one loaded workspace side.
pod_combat_apply_workspace_stances_effect = {
	if = {
		limit = { $STORY$ = { has_variable = set_guard } }
		var:pod_combat_session ?= { set_variable = { name = pod_combat_$SIDE$_is_guarding value = yes } }
	}
	if = {
		limit = { $STORY$ = { has_variable = dodge } }
		var:pod_combat_session ?= { set_variable = { name = pod_combat_$SIDE$_is_dodging value = yes } }
	}
	if = {
		limit = { $STORY$ = { has_variable = stealth } }
		pod_combat_apply_stealth_workspace_effect = { SIDE = $SIDE$ }
	}
}

pod_combat_apply_team_ally_buffs_effect = {
	# $STORY$ = move story. Body runs in slot-story scope (set_variable writes the
	# slot directly; var:slot_char flips to the char for the soak bridge write).
	save_scope_as = host
	var:pod_combat_session ?= {
		every_in_list = {
			variable = pod_combat_active_slots
			limit = {
				pod_combat_slot_is_alive_trigger = yes
				pod_combat_aoe_team_ally_filter_trigger = yes
			}
			pod_combat_upgrade_tier_var_effect = { STORY = $STORY$ SRC = hit_buff_round_tier DST = slot_hit_buff_round_tier }
			pod_combat_upgrade_tier_var_effect = { STORY = $STORY$ SRC = hit_buff_combat_tier DST = slot_hit_buff_combat_tier }
			if = {
				limit = { $STORY$ = { has_variable = set_guard } }
				set_variable = { name = slot_is_guarding  value = yes }
			}
			if = {
				limit = { $STORY$ = { has_variable = dodge } }
				set_variable = { name = slot_is_dodging   value = yes }
			}
			if = {
				limit = { $STORY$ = { has_variable = stealth } }
				pod_combat_apply_stealth_slot_effect = yes
			}
			# Keep the character switch after all inline slot writes
			# (iteration scope corruption). Check upgrades per ally; selection
			# only checked the actor.
			var:slot_char = {
				pod_combat_upgrade_tier_var_effect = { STORY = $STORY$ SRC = soak_buff_round_tier DST = pod_combat_tier_soak_buff_round }
				pod_combat_upgrade_tier_var_effect = { STORY = $STORY$ SRC = soak_buff_combat_tier DST = pod_combat_tier_soak_buff_combat }
			}
		}
	}
}

pod_combat_apply_team_enemy_debuffs_effect = {
	# $STORY$ = move story
	# Mirrors pod_combat_apply_team_ally_buffs_effect but for the enemy-target
	# side: debuff tier vars + bypass/remove flag clearances.
	save_scope_as = host
	var:pod_combat_session ?= {

			variable = pod_combat_active_slots

				pod_combat_slot_is_alive_trigger = yes

			pod_combat_upgrade_tier_var_effect = { STORY = $STORY$ SRC = hit_debuff_combat_tier DST = slot_hit_debuff_combat_tier }

				remove_variable = slot_is_guarding
			}
			# Keep the character switch last; see the ally-buff iteration
			# contract.
			var:slot_char = {
				pod_combat_upgrade_tier_var_effect = { STORY = $STORY$ SRC = soak_debuff_round_tier DST = pod_combat_tier_soak_debuff_round }
				pod_combat_upgrade_tier_var_effect = { STORY = $STORY$ SRC = soak_debuff_combat_tier DST = pod_combat_tier_soak_debuff_combat }
			}
		}
	}
}
