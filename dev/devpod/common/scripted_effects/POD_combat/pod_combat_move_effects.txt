######################################################
######   POD COMBAT MOVE EXECUTION ENGINE
######################################################
# Single generic effect that executes any combat move by reading the
# attributes off its story_cycle. Called from the
# pod_combat_execute_move SGUI (PC) and pod_combat_ai_select_and_execute_move_effect
# (AI). Caller scope is the combat host; the workspace is assumed loaded.
# Damage / soak / retaliation helpers live in pod_combat_effects.txt;
# tier recalc in pod_combat_tier_effects.txt; slot load/save in
# pod_combat_group_effects.txt. This file just dispatches.
#
# Phase contract (§0 setup → §16 push log) and the load-bearing
# anti-invariants live in COMBAT_REFERENCE.info §2. Read that before
# reordering anything.


######################################################
###### MAIN ENTRY POINT
######################################################

POD_execute_combat_move = {
	# $STORY$ = the POD_combat_move story_cycle

	# ── 0. Capture combat host as a saved scope ──────────────────
	# Downstream damage routing (pod_combat_route_to_pod_damage)
	# needs scope:combat_host to find the workspace owner from
	# inside per-target scope-switches. The PC button-click path
	# enters this effect with root = PC = combat host, and the AI
	# path enters with root = combat host (the AI driver scope-
	# switches back to the host before this call). Saving it here
	# once means the AoE iteration helpers don't need their own
	# `root = { save_scope_as = combat_host }` dance — that
	# dance was crashing the AI AoE chain because the AoE inner
	# scope-switches make the engine treat root as unset when the
	# save executes, blowing up scope:combat_host before the
	# per-target damage call could read it.
	save_scope_as = combat_host

	# ── 1. AP spend ──────────────────────────────────────────────
	if = {
		limit = { $STORY$ = { has_variable = ap_cost } }
		pod_combat_spend_ap_effect = { AMOUNT = $STORY$.var:ap_cost }
	}

	# ── 2. Result reset + log metadata ───────────────────────────
	# Remove any stale result from the previous move so absence =
	# "no outcome yet" for this move's pipeline. Damage / heal /
	# stance helpers along the pipeline will write the outcome flag;
	# push_log defaults to flag:utility when nothing wrote.
	remove_variable = pod_combat_last_result
	# Parallel amount fields — each pipeline (damage / heal / morale
	# damage / morale heal) writes its own variable so a damage+heal
	# move can preserve the HP-damage value without the heal pipeline
	# clobbering it. Cleared here so a previous move's amounts don't
	# leak into this one.
	remove_variable = pod_combat_last_heal_amount
	remove_variable = pod_combat_last_morale_damage_amount
	remove_variable = pod_combat_last_morale_heal_amount
	# Retaliation damage from the previous move — captured by
	# pod_combat_apply_retaliation_helper. Cleared so a non-retaliation
	# move doesn't carry the previous turn's counter amount onto its
	# log row.
	remove_variable = pod_combat_last_retal_damage
	# Default last_damage=0 so utility moves log cleanly; damage helpers
	# overwrite this as needed.
	set_variable = { name = pod_combat_last_damage value = 0 }

	# Move ref is threaded explicitly through pod_combat_push_log_effect's
	# MOVE_STORY parameter at §16 — no host-var handoff required, no
	# shared mutable state between caller and callee.

	# ── 3. Energy cost (splat-agnostic) ──────────────────────────
	# PoD's splat-agnostic accessor routes to blood/gnosis/pathos/angst/
	# resolve/glamour/dread/willpower depending on the CURRENT scope's
	# splat. So, we MUST scope-switch to the actor before calling it.
	# Workspace state (incl. pod_combat_actor_char) is stored on
	# combat_host (PC), but the actor is set to the acting NPC for AI
	# turns and to PC itself for PC turns. Without the switch, every
	# AI cast would drain the PC's pool.
	if = {
		limit = { $STORY$ = { has_variable = energy_cost } }
		if = {
			limit = { has_variable = pod_combat_actor_char }
			var:pod_combat_actor_char = {
				POD_splat_agnostic_energy_cost = { COUNT = $STORY$.var:energy_cost }
			}
		}
		else = {
			# pod_combat_actor_char isn't set yet; happens when a move
			# fires before pod_combat_load_workspace_effect has stamped
			# the actor pointer. Fall back to paying on the current
			# scope, which is the actor in those flows by construction.
			POD_splat_agnostic_energy_cost = { COUNT = $STORY$.var:energy_cost }
		}
		# AI attrition counter stays on combat_host because the selection
		# weight reads it from there when the next AI turn picks a move.
		# pod_combat_actor_energy_used tracks "the current actor used
		# an energy-cost move" — true for PC and AI alike — and the
		# energy-attrition weighting in the AI selector keys off the
		# acting character's history.
		if = {
			limit = { has_variable = pod_combat_actor_energy_used }
			change_variable = { name = pod_combat_actor_energy_used add = 1 }
		}
		else = {
			set_variable = { name = pod_combat_actor_energy_used value = 1 }
		}
	}

	# ── 4. CK3 resource costs (PC only) ──────────────────────────
	# Only slot 0 (the player) pays CK3-side costs. AI combatants get
	# the move's effect at no CK3 cost (their AI turn is free).
	if = {
		limit = {
			has_variable = pod_combat_current_actor
			var:pod_combat_current_actor = 0
		}
		if = {
			limit = { $STORY$ = { has_variable = gold_cost } }
			remove_short_term_gold = $STORY$.var:gold_cost
		}
		if = {
			limit = { $STORY$ = { has_variable = stress_cost } }
			add_stress = $STORY$.var:stress_cost
		}
		if = {
			limit = { $STORY$ = { has_variable = piety_cost } }
			add_piety = {
				value = $STORY$.var:piety_cost
				multiply = -1
			}
		}
		if = {
			limit = { $STORY$ = { has_variable = prestige_cost } }
			add_prestige = {
				value = $STORY$.var:prestige_cost
				multiply = -1
			}
		}
	}

	# ── 5. Target stance bypass ──────────────────────────────────
	# Strip the target's dodge/stealth stances BEFORE damage resolves so
	# the damage pipeline treats the target as unprotected. bypass_guard
	# has no runtime variable — the generator OMITS the checks_guard
	# capability flag for moves with bypass_guard in `special`, and the
	# contact resolver reads that omission directly. No separate
	# dispatcher branch is needed here.
	if = {
		limit = { $STORY$ = { has_variable = bypass_dodge } }
		remove_variable = pod_combat_target_is_dodging
	}
	if = {
		limit = { $STORY$ = { has_variable = bypass_stealth } }
		# Tear down both the flag AND the +30 hit-chance bonus the
		# target's stealth granted. Without subtracting the bonus, a
		# bypass_stealth'd target keeps the +30 baked into their slot's
		# saved hit_chance and gets a phantom offensive boost on their
		# next attack. Helper is idempotent — no-op if target isn't
		# stealthed.
		pod_combat_break_stealth_workspace_effect = { SIDE = target SYNC_CURRENT = no }
	}

	# ── 5b. AoE debuff pre-damage broadcast ──────────────────────
	# For team_enemy moves, broadcast tier debuffs + stance-strip
	# flags to ALL enemy slots BEFORE the damage iteration so each
	# hit in the AoE respects bypass_dodge / bypass_stealth /
	# remove_guard on its own target (the workspace strip in
	# section 5 only touches the pre-AoE designated target's
	# workspace copy, which each AoE iteration overwrites when it
	# loads a new slot). Tier debuffs (soak_debuff / hit_debuff)
	# broadcast here persist on every enemy slot, not just the
	# pre-AoE target's workspace. Section 11/12's target-side
	# writes still run post-damage but are redundant for
	# team_enemy — the authoritative debuff state is on slot vars
	# set here.
	if = {
		limit = { $STORY$ = { var:targeting = flag:team_enemy } }
		POD_combat_apply_team_enemy_debuffs = { STORY = $STORY$ }
	}

	# ── 6. HP damage dispatch ────────────────────────────────────
	# Routes by source × type × targeting through POD_combat_apply_single_hp_damage
	# (or its team_enemy AoE wrapper). The single-target path saves the
	# move story as scope:current_move and forwards to
	# POD_combat_dispatch_damage_pipeline, which runs the unified
	# pod_combat_resolve_contact_effect (capability-flag-driven contact
	# resolution) followed by pod_combat_apply_damage_payload_effect on
	# hit. Pipeline behavior (hit roll / guard check / spirit gate) is
	# selected by the move's capability flags rather than by branching
	# to a dedicated effect. See pod_combat_effects.txt's UNIFIED
	# CONTACT RESOLVER block for the implementation.
	if = {
		limit = { $STORY$ = { has_variable = hp_damage } }
		if = {
			limit = { $STORY$ = { var:targeting = flag:team_enemy } }
			POD_combat_apply_team_enemy_hp_damage = { STORY = $STORY$ }
		}
		else = {
			# single_enemy (default and most common case)
			POD_combat_apply_single_hp_damage = { STORY = $STORY$ TARGET_SIDE = target }
		}
	}

	# ── 7. Morale damage ─────────────────────────────────────────
	# Morale damage uses its own pipeline (auto-hit, no guard, no soak,
	# dodge still works). Source is informational — routed the same way
	# regardless of damage_source.
	#
	# Rider gate: default fires unconditionally; designers can author
	# "morale damage only on hit" / "only on miss" / "only on contact"
	# by setting morale_damage_condition on the move. Trigger reads
	# the §6 outcome via pod_combat_last_result on
	# the host (this scope).
	if = {
		limit = {
			$STORY$ = { has_variable = morale_damage }
			pod_combat_rider_condition_passes_trigger = {
				STORY     = $STORY$
				RIDER_VAR = morale_damage_condition
			}
		}
		if = {
			limit = { $STORY$ = { var:targeting = flag:team_enemy } }
			POD_combat_apply_team_enemy_morale_damage = { STORY = $STORY$ }
		}
		else = {
			pod_combat_deal_morale_damage_effect = { TARGET_SIDE = target AMOUNT = $STORY$.var:morale_damage }
		}
	}

	# ── 8. Self-damage ───────────────────────────────────────────
	# Route self-damage through the same PoD pipeline as regular hits
	# so wound state stays in lockstep. Uses REASON=duel (superficial,
	# soakable) rather than self_mutilation, reckless-charge-style
	# self damage is physical overextension, not a permanent wound.
	# Soak still applies, so armor/fortitude can mitigate it. The
	# bypass_guard helper skips the guard check since self-damage
	# doesn't care about stances.
	#
	# Snapshot/restore around the call: pod_combat_route_to_pod_damage
	# writes pod_combat_last_damage with the per-call rolled damage
	# AND stamps pod_combat_last_result with the call's outcome
	# (flag:hp_damage on a landed hit, flag:soaked when soak ate it).
	# Both must be preserved across the self-damage call so the log
	# line reflects the MAIN hit's outcome,
	# not the self-damage's:
	#   * Without restoring last_damage, push_log's hp_damage line would
	#     read "PC hit Enemy for $self_damage$" instead of the main hit.
	#   * Without restoring last_result, a fully-soaked self-damage
	#     after a successful main hit would overwrite the main hit's
	#     flag:hp_damage with flag:soaked — the move would render as
	#     "soaked" despite the target taking real damage. Likewise
	#     self-damage that lands after a soaked main hit would flip
	#     flag:soaked into flag:hp_damage.
	#
	# Rider gate: default fires unconditionally; designers
	# can author "self-damage on miss" (reckless overswings) or
	# "self-damage on hit" (blood payment for landing the strike) by
	# setting self_damage_condition on the move. Trigger reads §6's
	# pod_combat_last_result on this scope (the combat host).
	if = {
		limit = {
			$STORY$ = { has_variable = self_damage }
			pod_combat_rider_condition_passes_trigger = {
				STORY     = $STORY$
				RIDER_VAR = self_damage_condition
			}
		}
		# Stash the main-hit value + result flag as real variables on
		# the combat host so they survive the self-damage pipeline
		# call. Using plain variables instead of save_scope_value_as
		# because the latter has been observed to fail to round-trip
		# through the deeper damage dispatch chain (PoD wound pipeline
		# + AoE scope-switches consume saved scopes mid-flight).
		#
		# has_variable guards: a move with self_damage but no hp_damage
		# never has a main-hit value written (§6 doesn't fire), so the
		# vars don't exist. Reading them with `value = var:X` returns
		# 0 / unset which then gets restored as 0 / a corrupt flag
		# the log line falls through to the — fallback instead of
		# rendering a meaningful result. Stash + restore both gate
		# on presence so the variables stay absent when they were
		# absent going in.
		if = {
			limit = { has_variable = pod_combat_last_damage }
			set_variable = { name = pod_combat_pre_self_last_damage value = var:pod_combat_last_damage }
		}
		if = {
			limit = { has_variable = pod_combat_last_result }
			set_variable = { name = pod_combat_pre_self_result value = var:pod_combat_last_result }
		}

		pod_combat_deal_hp_damage_bypass_guard_effect = {
			TARGET_SIDE = actor
			AMOUNT      = $STORY$.var:self_damage
			REASON      = duel
		}

		# Restore the main-hit value + result flag so push_log reads
		# the right number and renders the right result.
		# (No parallel pod_combat_self_last_damage field today. The
		# combat-log GUI doesn't render a self-damage rider. If that
		# changes, capture the per-call last_damage to a dedicated
		# var here BEFORE the restore; the snapshot/restore itself
		# is load-bearing either way.)
		if = {
			limit = { has_variable = pod_combat_pre_self_last_damage }
			set_variable = { name = pod_combat_last_damage value = var:pod_combat_pre_self_last_damage }
			remove_variable = pod_combat_pre_self_last_damage
		}
		else = {
			# No main-hit value to restore. Wipe the per-call value the
			# self-damage helper just wrote so the log doesn't render
			# self-damage as the move's "main" damage line.
			remove_variable = pod_combat_last_damage
		}
		if = {
			limit = { has_variable = pod_combat_pre_self_result }
			set_variable = { name = pod_combat_last_result value = var:pod_combat_pre_self_result }
			remove_variable = pod_combat_pre_self_result
		}
		else = {
			# Same rationale — discard the self-damage's outcome flag
			# so push_log falls through to the utility fallback.
			remove_variable = pod_combat_last_result
		}
	}

	# ── 9. HP heal ───────────────────────────────────────────────
	# heal_type (flag:aggravated) routes to the aggravated heal variant
	# which also reduces the agg_damage counter. Otherwise uses the
	# superficial heal (caps at max_hp − agg_damage).
	# team_ally heals iterate the actor's whole team via AoE dispatch;
	# single-target heals (and self heals via targeting=self) just run
	# on the player-side workspace.
	#
	# Rider gate: heal_condition controls whether the heal fires based
	# on §6's outcome. Default = always (utility heals). Drain heals
	# (vampire bite, spectre soul-drain, etc.) get heal_condition =
	# on_hp_damage emitted automatically by the generator when the
	# move has the `drain` token in `special`, so "only heal on a
	# landed hit" runs through the same machinery as every other
	# rider condition.
	if = {
		limit = {
			$STORY$ = { has_variable = hp_heal }
			pod_combat_rider_condition_passes_trigger = {
				STORY     = $STORY$
				RIDER_VAR = heal_condition
			}
		}
		if = {
			limit = { $STORY$ = { var:targeting = flag:team_ally } }
			POD_combat_apply_team_ally_hp_heal = { STORY = $STORY$ }
		}
		else_if = {
			limit = { $STORY$ = { has_variable = heal_type var:heal_type = flag:aggravated } }
			pod_combat_restore_hp_aggravated_effect = { SIDE = actor AMOUNT = $STORY$.var:hp_heal }
		}
		else = {
			pod_combat_restore_hp_superficial_effect = { SIDE = actor AMOUNT = $STORY$.var:hp_heal }
		}
	}

	# ── 10. Morale heal ──────────────────────────────────────────
	if = {
		limit = { $STORY$ = { has_variable = morale_heal } }
		if = {
			limit = { $STORY$ = { var:targeting = flag:team_ally } }
			POD_combat_apply_team_ally_morale_heal = { STORY = $STORY$ }
		}
		else = {
			pod_combat_restore_morale_effect = { SIDE = actor AMOUNT = $STORY$.var:morale_heal }
		}
	}

	# ── 11. Tier buffs/debuffs ───────────────────────────────────
	# Buffs land on the actor (pod_combat_actor_*), debuffs on the
	# target (pod_combat_target_*). Values 1/2/3 map to +/-10/20/30
	# in recalc.
	#
	# Soak tiers ALSO propagate to the acting/targeted character as
	# pod_combat_tier_soak_{buff,debuff}_{round,combat} variables.
	# PoD's POD_soak_chance SV (overridden in
	# POD_health_track_values.txt) reads these character vars so the
	# damage roll picks up the combat tier bonus. Workspace
	# copies stay for the hit-chance side (which still
	# runs through combat's own pipeline).
	if = {
		limit = { $STORY$ = { has_variable = soak_buff_round_tier } }
		set_variable = { name = pod_combat_actor_soak_buff_round_tier value = $STORY$.var:soak_buff_round_tier }
		if = {
			limit = { has_variable = pod_combat_actor_char }
			var:pod_combat_actor_char = {
				set_variable = { name = pod_combat_tier_soak_buff_round value = $STORY$.var:soak_buff_round_tier }
			}
		}
	}
	if = {
		limit = { $STORY$ = { has_variable = soak_buff_combat_tier } }
		set_variable = { name = pod_combat_actor_soak_buff_combat_tier value = $STORY$.var:soak_buff_combat_tier }
		if = {
			limit = { has_variable = pod_combat_actor_char }
			var:pod_combat_actor_char = {
				set_variable = { name = pod_combat_tier_soak_buff_combat value = $STORY$.var:soak_buff_combat_tier }
			}
		}
	}
	if = {
		limit = { $STORY$ = { has_variable = soak_debuff_round_tier } }
		set_variable = { name = pod_combat_target_soak_debuff_round_tier value = $STORY$.var:soak_debuff_round_tier }
		if = {
			limit = { has_variable = pod_combat_target_char }
			var:pod_combat_target_char = {
				set_variable = { name = pod_combat_tier_soak_debuff_round value = $STORY$.var:soak_debuff_round_tier }
			}
		}
	}
	if = {
		limit = { $STORY$ = { has_variable = soak_debuff_combat_tier } }
		set_variable = { name = pod_combat_target_soak_debuff_combat_tier value = $STORY$.var:soak_debuff_combat_tier }
		if = {
			limit = { has_variable = pod_combat_target_char }
			var:pod_combat_target_char = {
				set_variable = { name = pod_combat_tier_soak_debuff_combat value = $STORY$.var:soak_debuff_combat_tier }
			}
		}
	}
	if = {
		limit = { $STORY$ = { has_variable = hit_buff_round_tier } }
		set_variable = { name = pod_combat_actor_hit_buff_round_tier value = $STORY$.var:hit_buff_round_tier }
	}
	if = {
		limit = { $STORY$ = { has_variable = hit_buff_combat_tier } }
		set_variable = { name = pod_combat_actor_hit_buff_combat_tier value = $STORY$.var:hit_buff_combat_tier }
	}
	if = {
		limit = { $STORY$ = { has_variable = hit_debuff_round_tier } }
		set_variable = { name = pod_combat_target_hit_debuff_round_tier value = $STORY$.var:hit_debuff_round_tier }
	}
	if = {
		limit = { $STORY$ = { has_variable = hit_debuff_combat_tier } }
		set_variable = { name = pod_combat_target_hit_debuff_combat_tier value = $STORY$.var:hit_debuff_combat_tier }
	}
	# Recalc live stats only if the matching tier var changed.
	if = {
		limit = {
			OR = {
				$STORY$ = { has_variable = soak_buff_round_tier }
				$STORY$ = { has_variable = soak_buff_combat_tier }
			}
		}
		pod_combat_recalc_soak_actor_effect = yes
	}
	if = {
		limit = {
			OR = {
				$STORY$ = { has_variable = soak_debuff_round_tier }
				$STORY$ = { has_variable = soak_debuff_combat_tier }
			}
		}
		pod_combat_recalc_soak_target_effect = yes
	}
	if = {
		limit = {
			OR = {
				$STORY$ = { has_variable = hit_buff_round_tier }
				$STORY$ = { has_variable = hit_buff_combat_tier }
			}
		}
		pod_combat_recalc_hit_actor_effect = yes
	}
	if = {
		limit = {
			OR = {
				$STORY$ = { has_variable = hit_debuff_round_tier }
				$STORY$ = { has_variable = hit_debuff_combat_tier }
			}
		}
		pod_combat_recalc_hit_target_effect = yes
	}

	# ── 12. Non-tier presence flags ──────────────────────────────
	if = {
		limit = { $STORY$ = { has_variable = set_guard } }
		set_variable = { name = pod_combat_actor_is_guarding value = yes }
	}
	if = {
		limit = { $STORY$ = { has_variable = dodge } }
		set_variable = { name = pod_combat_actor_is_dodging value = yes }
	}
	if = {
		limit = { $STORY$ = { has_variable = stealth } }
		# Idempotent stealth applicator — guards on the existing flag
		# so a duplicate stealth move from a stealthed actor doesn't
		# double-bake the +30. The PC-side `POD_can_execute_combat_move`
		# also rejects stealth-while-stealthed, but the helper guard
		# keeps the slot-stealth invariant if any future path applies
		# stealth without going through can_execute. The bonus is
		# reverted by pod_combat_consume_actor_stealth_effect when the
		# stealth is later broken by attacking or being hit.
		pod_combat_apply_stealth_workspace_effect = { SIDE = actor SYNC_CURRENT = yes }
	}
	if = {
		limit = { $STORY$ = { has_variable = remove_guard } }
		remove_variable = pod_combat_target_is_guarding
	}

	# ── 12b. AoE buff broadcast ──────────────────────────────────
	# For team_ally moves, extend the actor-workspace buff writes
	# in sections 11 + 12 to ALL ally slots so every teammate
	# carries the tier buffs (soak/hit) and stance flags
	# (set_guard / dodge / stealth). The actor's own slot gets
	# written here; save_actor_to_slot at end of turn writes the
	# same values back from workspace, so there's no conflict.
	# Soak tier bridge vars are also set on each ally's character
	# so POD_soak_chance override picks up the buff right
	# away (not just after their next turn loads workspace).
	if = {
		limit = { $STORY$ = { var:targeting = flag:team_ally } }
		POD_combat_apply_team_ally_buffs = { STORY = $STORY$ }
	}

	# ── 13. Retaliation stances ──────────────────────────────────
	# Setting up a retaliation — arms a reaction that fires next turn
	# in §15 when the actor is attacked.
	#   retaliate_if_hit    — fires on contact (flag:hp_damage /
	#                         guard_blocked / soaked / hp_damage_retal /
	#                         soaked_retal). Thorns / spite semantics:
	#                         any attack that physically reaches the
	#                         defender triggers the counter, even if
	#                         the defender's soak or guard absorbed it.
	#   retaliate_if_missed — fires on evasion (flag:missed / dodged /
	#                         stealthed). Riposte semantics:
	#                         the attacker overcommitted on a whiff.
	# retaliate_damage gives the counter amount; otherwise the move's
	# hp_damage value is reused. retaliate_agg marks it as aggravated.

	# Amount fallback chain: explicit retaliate_damage → move's
	# hp_damage → final fallback of 1. The final `else` defaults
	# matter only when a move sets retaliate_if_hit / _missed without
	# either retaliate_damage OR hp_damage. Without the
	# fallback the workspace var stays unset and
	# pod_combat_apply_reaction_if_triggered_effect's
	# `has_variable = pod_combat_actor_retaliate_hit` gate fails
	# the retaliation silently doesn't fire even though the move
	# armed it.
	if = {
		limit = { $STORY$ = { has_variable = retaliate_if_hit } }
		if = {
			limit = { $STORY$ = { has_variable = retaliate_damage } }
			set_variable = { name = pod_combat_actor_retaliate_hit value = $STORY$.var:retaliate_damage }
		}
		else_if = {
			limit = { $STORY$ = { has_variable = hp_damage } }
			set_variable = { name = pod_combat_actor_retaliate_hit value = $STORY$.var:hp_damage }
		}
		else = {
			# No explicit amount and no hp_damage to reuse; default
			# to 1 so the reaction at least fires. Designer can
			# author a real amount via retaliate_damage when this
			# starts mattering.
			set_variable = { name = pod_combat_actor_retaliate_hit value = 1 }
		}
		if = {
			limit = { $STORY$ = { has_variable = retaliate_agg } }
			set_variable = { name = pod_combat_actor_retaliate_hit_agg value = yes }
		}
	}
	if = {
		limit = { $STORY$ = { has_variable = retaliate_if_missed } }
		if = {
			limit = { $STORY$ = { has_variable = retaliate_damage } }
			set_variable = { name = pod_combat_actor_retaliate_miss value = $STORY$.var:retaliate_damage }
		}
		else_if = {
			limit = { $STORY$ = { has_variable = hp_damage } }
			set_variable = { name = pod_combat_actor_retaliate_miss value = $STORY$.var:hp_damage }
		}
		else = {
			set_variable = { name = pod_combat_actor_retaliate_miss value = 1 }
		}
		if = {
			limit = { $STORY$ = { has_variable = retaliate_agg } }
			set_variable = { name = pod_combat_actor_retaliate_miss_agg value = yes }
		}
	}

	# ── 14. Parametric non-tier flags ────────────────────────────
	if = {
		limit = { $STORY$ = { has_variable = ap_gain } }
		change_variable = { name = pod_combat_current_ap add = $STORY$.var:ap_gain }
	}
	# AP-drain and initiative-debuff target writes bypass the workspace
	# target mirror and land on the target slot directly. Workspace's
	# pod_combat_target_ap_drain / _init_bonus were write-only mirrors
	# with no readers — the consumers (pod_combat_apply_ap_drain_effect
	# and the round-end initiative recalc) read slot_ap_drain /
	# slot_init_bonus directly, not the workspace — so the mirrors were
	# removed and the writes route through active_slots instead.
	# scope:combat_host is saved in §0; the active_slots iterator
	# filtered on slot_index = selected_target matches at most one slot
	# — the same slot save_target_to_story would have written to.
	if = {
		limit = { $STORY$ = { has_variable = ap_drain } }
		every_in_list = {
			variable = pod_combat_active_slots
			limit = { var:slot_index = scope:combat_host.var:pod_combat_selected_target }
			set_variable = { name = slot_ap_drain value = $STORY$.var:ap_drain }
		}
	}
	if = {
		limit = { $STORY$ = { has_variable = initiative_debuff } }
		save_scope_value_as = {
			name = init_debuff_neg
			value = {
				value = $STORY$.var:initiative_debuff
				multiply = -1
			}
		}
		every_in_list = {
			variable = pod_combat_active_slots
			limit = { var:slot_index = scope:combat_host.var:pod_combat_selected_target }
			set_variable = { name = slot_init_bonus value = scope:init_debuff_neg }
		}
	}

	# ── 15. Retaliation check ────────────────────────────────────
	# After any damage resolves, check if the target had a retaliation
	# stance set up from a previous turn.
	#
	# Gated on single-target damage: team_enemy AoE moves don't
	# trigger retaliation. The retaliation effect can only fire on
	# ONE target at a time (the workspace target, which the AoE
	# path restores to the pre-AoE designated target). Firing it
	# there would unfairly single out the designated target out of
	# all 5 AoE victims. Skipping retaliation for AoE is the
	# simplest consistent behavior.
	if = {
		limit = {
			$STORY$ = { has_variable = hp_damage }
			NOT = { $STORY$ = { var:targeting = flag:team_enemy } }
		}
		pod_combat_check_retaliation_effect = { TARGET_SIDE = target ACTOR_SIDE = actor }
	}

	# ── 16. Push combat log entry ────────────────────────────────
	# Move story passed explicitly so push_log doesn't depend on a
	# host-var handoff.
	pod_combat_push_log_effect = { MOVE_STORY = $STORY$ }
}


######################################################
###### HP DAMAGE DISPATCH (single target)
######################################################
# Three-stage dispatch:
#   1. POD_combat_apply_single_hp_damage    — handles mental no-op +
#      aggravated override + delegates to dispatch_by_source.
#   2. POD_combat_dispatch_by_source        — maps var:damage_source
#      flag to a PoD death_reason suffix (REASON), then forwards.
#   3. POD_combat_dispatch_damage_pipeline  — runs the unified
#      pod_combat_resolve_contact_effect (capability-flag-driven) and
#      fires pod_combat_apply_damage_payload_effect on hit.
#
# Capability flags drive what the resolver does: requires_hit_roll /
# checks_guard / checks_stance / checks_spirit_gate. The resolver IS
# the pipeline and branching happens via flag
# presence on scope:current_move.

POD_combat_apply_single_hp_damage = {
	# $STORY$ = move story
	# $TARGET_SIDE$ = "target" (the workspace side that takes the damage)
	#
	# attack_type (physical / supernatural / mental) and damage_source
	# (fire / silver / sun / ...) are independent dimensions:
	#   attack_type → capability flags (requires_hit_roll, checks_guard,
	#                  checks_stance, checks_spirit_gate). Resolved at
	#                  generation time and stamped on the story by
	#                  POD_init_combat_move.
	#   damage_source → PoD death_reason suffix. Mapped at runtime by
	#                  POD_combat_dispatch_by_source.
	#   damage_type = aggravated forces REASON = duel_aggravated,
	#                  overriding any source-specific aggravation rule.
	#
	# Save the move story as scope:current_move so the contact resolver
	# (and any other downstream helper) can read its capability flags
	# without each one threading $STORY$ through its signature.
	# CK3 saved scopes propagate down the call stack. The dispatch
	# chain (apply_single → dispatch_by_source → dispatch_damage_pipeline
	# → resolve_contact / apply_payload) all see the same scope:current_move.
	$STORY$ = { save_scope_as = current_move }

	# Aggravated override: damage_type = aggravated forces the
	# wound to be aggravated regardless of source or target splat.
	# Delegates straight to the unified dispatcher with the canonical
	# duel_aggravated reason — capability flags on the move story
	# (set at init from attack_type defaults) drive pipeline behavior.
	if = {
		limit = { $STORY$ = { has_variable = damage_type var:damage_type = flag:aggravated } }
		POD_combat_dispatch_damage_pipeline = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = duel_aggravated }
	}

	# Damage-source dispatch — delegated to POD_combat_dispatch_by_source
	# which maps the move's damage_source flag to a PoD death_reason
	# suffix and routes through POD_combat_dispatch_damage_pipeline.
	# Add a new damage source by extending that helper alone; this
	# branch stays untouched.
	else = {
		POD_combat_dispatch_by_source = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ }
	}
}

# Damage-source → death_reason dispatcher. Maps the move's
# var:damage_source (flag) to the PoD death_reason suffix expected by
# POD_damage_type_is_aggravated / POD_wound_multiplier_from_splat_weaknesses,
# then forwards to POD_combat_dispatch_damage_pipeline (which runs the
# unified contact resolver — pipeline behavior is read from capability
# flags on scope:current_move, not selected here).
#
# The flag IS the REASON suffix in 7 of 8 cases (identity mapping).
# `holy` is the lone exception — it maps to PoD's death_reason
# `true_faith` rather than `holy_damage` because PoD's wound system
# already keys aggravated promotion against true_faith for vampires
# and other faith-vulnerable splats.
#
# To add a new damage source:
#   1. Set the move's damage_source flag
#   2. Add a new else_if branch below mapping the flag to its REASON.
#   3. Update the loc / icon set in pod_combat_static_l_english.yml
#      and pod_combat_log_custom_loc.txt's PODCombatDamageSourceMarker.
# CK3 script can't pass `var:damage_source` as a REASON parameter
# directly (REASON is a token, not a value-lookup), so the chain
# stays — but it's contained here, and the multi-axis caller
# (POD_combat_apply_single_hp_damage) is no longer cluttered with it.
POD_combat_dispatch_by_source = {
	# $STORY$ = move story
	# $TARGET_SIDE$ = "actor" or "target"
	if = { limit = { $STORY$ = { has_variable = damage_source var:damage_source = flag:fire } }
		POD_combat_dispatch_damage_pipeline = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = fire }
	}
	else_if = { limit = { $STORY$ = { has_variable = damage_source var:damage_source = flag:cold } }
		POD_combat_dispatch_damage_pipeline = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = cold }
	}
	else_if = { limit = { $STORY$ = { has_variable = damage_source var:damage_source = flag:lightning } }
		POD_combat_dispatch_damage_pipeline = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = lightning }
	}
	else_if = { limit = { $STORY$ = { has_variable = damage_source var:damage_source = flag:sun } }
		POD_combat_dispatch_damage_pipeline = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = sun }
	}
	else_if = { limit = { $STORY$ = { has_variable = damage_source var:damage_source = flag:silver } }
		POD_combat_dispatch_damage_pipeline = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = silver }
	}
	else_if = { limit = { $STORY$ = { has_variable = damage_source var:damage_source = flag:gold } }
		POD_combat_dispatch_damage_pipeline = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = gold }
	}
	else_if = { limit = { $STORY$ = { has_variable = damage_source var:damage_source = flag:holy } }
		# Only non-identity mapping — PoD keys aggravated promotion
		# against true_faith, not holy.
		POD_combat_dispatch_damage_pipeline = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = true_faith }
	}
	else_if = { limit = { $STORY$ = { has_variable = damage_source var:damage_source = flag:balefire } }
		POD_combat_dispatch_damage_pipeline = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = balefire }
	}
	# No damage_source set → generic physical/supernatural damage.
	else = {
		POD_combat_dispatch_damage_pipeline = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = duel }
	}
}

POD_combat_dispatch_damage_pipeline = {
	# $STORY$ = move story
	# $TARGET_SIDE$ = "actor" or "target"
	# $REASON$ = PoD death_reason suffix (passed through to the payload)
	#
	# Unified contact-resolver + damage-payload flow. Capability flags
	# on the executing move drive branching inside
	# pod_combat_resolve_contact_effect. The dispatcher no longer
	# selects a pipeline; it just runs the resolver, fires the payload
	# on hit, and consumes the actor's stealth.
	#
	# scope:current_move is set by POD_combat_apply_single_hp_damage at
	# entry so the resolver can read the move's capability flags
	# (checks_spirit_gate / requires_hit_roll / checks_guard /
	# checks_stance) without each call site re-saving the scope.
	pod_combat_resolve_contact_effect = {
		TARGET_SIDE = $TARGET_SIDE$
		ACTOR_SIDE  = actor
	}
	if = {
		limit = { scope:contact_result = 0 }
		pod_combat_apply_damage_payload_effect = {
			TARGET_SIDE = $TARGET_SIDE$
			AMOUNT      = $STORY$.var:hp_damage
			REASON      = $REASON$
		}
	}
	# Actor stealth always consumed once the move resolves, regardless
	# of contact outcome — moving / committing the action breaks stealth
	# even on a miss.
	pod_combat_consume_actor_stealth_effect = { ACTOR_SIDE = actor }
}


######################################################
###### AOE DISPATCH (team_enemy + team_ally)
######################################################
# Apply the move's effects across a whole team. "Enemy" vs "ally"
# resolves RELATIVE to the current actor — slots 0-4 are the player
# team, 5-9 the enemy team, so an NPC actor in slot 7 treats 0-4 as
# its enemies and 5-9 as its allies.
#
# All six AoE dispatchers iterate pod_combat_active_slots filtered by
# slot_index against the actor's team:
#
#   team_enemy: actor on player team (current_actor < 5) → hit slots 5-9.
#               actor on enemy team (current_actor >= 5) → hit slots 0-4.
#   team_ally:  actor on player team → buff/heal slots 0-4 (includes self).
#               actor on enemy team → buff/heal slots 5-9.
#
# The team-side decision is encapsulated in two shared triggers:
#   pod_combat_aoe_team_enemy_filter_trigger
#   pod_combat_aoe_team_ally_filter_trigger
# Both read scope:host.var:pod_combat_current_actor and the iterated
# slot's slot_index. Single source of truth for the team decision so
# future AoE additions don't re-derive the index split.
#
# Damage-side dispatchers also call
# pod_combat_aoe_check_target_alive_effect after each save_target so
# the four KO clauses (death / incapacitation / morale-break /
# wound-track-full) live in one place. See that effect's header for
# the full clause list and ordering rationale.
#
# ── The host save dance explanation ─────────────────────────────
# Every AoE dispatcher (and pod_combat_push_log_effect) opens with
# `save_scope_as = host`. CK3's every_in_list switches the iteration
# body's scope to the iterated entry (a slot story, in our case),
# which means `root` and the calling scope's vars are NOT directly
# reachable from inside the body.
#
# Two scope names are used in this codebase:
#   scope:combat_host  — saved once at the top of POD_execute_combat_move
#                        (§0). Long-lived for the duration of the move,
#                        used by deeply-nested damage routing
#                        (pod_combat_route_to_pod_damage and friends)
#                        that need to read the workspace from inside
#                        per-target scope-switches.
#   scope:host         — function-local save inside an AoE / log
#                        iterator. Same target as combat_host (the
#                        combat host character) but bound to the
#                        iterator's body so it survives the
#                        every_in_list scope switch even when the call
#                        is reached without combat_host being the
#                        outer scope (e.g. push_log fires in places
#                        where the outer chain has already switched).
#
# Inside the iteration body:
#   - var:slot_X reads/writes target the iterated slot story directly.
#   - scope:host = { ... } re-enters host scope to read workspace vars
#     or call helpers that expect a character scope (single-target
#     damage / heal pipelines, victory checks, etc.).
#   - pod_combat_save_target_to_story_effect / _load_target_from_story_effect
#     bridge between the host's workspace and the iterated slot story.
#
# Damage / heal AoEs go through the workspace pattern: load each
# target into the TARGET workspace slot, run the existing single-
# target pipeline (which reads / writes the workspace), save the
# result back, then restore the pre-AoE workspace at the end.
# Buff / debuff AoEs skip the workspace and write the slot story's
# tier / stance vars + char-bridge bridge vars directly.
#
# This asymmetry is intentional: workspace-load adds ~30 var
# round-trips per target, which buys nothing for pure-write payloads
# (buff / debuff broadcasts have no soak/hit roll, no death check, no
# wound-system entry). The shared scaffolding (host save, slot iter,
# team filter) is now identical across both shapes; only the body
# differs. A future A12 (per-target retaliation) hook would attach
# inside the damage iterators specifically — buffs/debuffs don't
# arm reactions, so they don't need the workspace round-trip even
# under that extension.

POD_combat_apply_team_enemy_hp_damage = {
	# $STORY$ = move story
	save_scope_as = host
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			var:slot_alive = 1
			pod_combat_aoe_team_enemy_filter_trigger = yes
		}
		pod_combat_load_target_from_story_effect = yes
		scope:host = {
			POD_combat_apply_single_hp_damage = { STORY = $STORY$ TARGET_SIDE = target }
		}
		pod_combat_save_target_to_story_effect = yes
		# Post-hit alive flip (death / incapacitation / morale break /
		# wound-track full). Centralised in
		# pod_combat_aoe_check_target_alive_effect — see its header
		# for the full clause list.
		pod_combat_aoe_check_target_alive_effect = yes
	}
	scope:host = { pod_combat_load_workspace_effect = yes }
}

POD_combat_apply_team_enemy_morale_damage = {
	# $STORY$ = move story
	save_scope_as = host
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			var:slot_alive = 1
			pod_combat_aoe_team_enemy_filter_trigger = yes
		}
		pod_combat_load_target_from_story_effect = yes
		scope:host = {
			pod_combat_deal_morale_damage_effect = { TARGET_SIDE = target AMOUNT = $STORY$.var:morale_damage }
		}
		pod_combat_save_target_to_story_effect = yes
		# Same post-hit alive flip as the HP-damage iterator; the
		# morale clause is what makes this matter for the morale path.
		pod_combat_aoe_check_target_alive_effect = yes
	}
	scope:host = { pod_combat_load_workspace_effect = yes }
}

# Apply HP / morale healing across the actor's entire team. Loads
# each ally into the TARGET workspace slot (not actor) so
# pod_combat_restore_hp_*_effect / pod_combat_restore_morale_effect
# can run with SIDE=target without clobbering the actor's own
# buffs, morale, hit chance, etc. pod_combat_load_workspace_effect
# at the end restores the pre-AoE pair.

POD_combat_apply_team_ally_hp_heal = {
	# $STORY$ = move story
	# Skip KO'd allies — no point healing a dead slot; wound state on
	# the char is inaccessible once is_alive = no anyway.
	save_scope_as = host
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			var:slot_alive = 1
			pod_combat_aoe_team_ally_filter_trigger = yes

		pod_combat_load_target_from_story_effect = yes

			if = {
				limit = { $STORY$ = { has_variable = heal_type var:heal_type = flag:aggravated } }
				pod_combat_restore_hp_aggravated_effect = { SIDE = target AMOUNT = $STORY$.var:hp_heal }
			}
			else = {
				pod_combat_restore_hp_superficial_effect = { SIDE = target AMOUNT = $STORY$.var:hp_heal }
			}
		}
		pod_combat_save_target_to_story_effect = yes
	}
	scope:host = { pod_combat_load_workspace_effect = yes }
}

######################################################
###### AOE BUFF DISPATCH (team_ally / team_enemy)
######################################################
# Broadcast tier buffs / stance flags across a whole team. Unlike the
# damage + heal AoE paths, these don't touch the workspace — they
# write per-slot tier / flag vars + (for soak) the character-scope
# bridge var directly, so all teammates carry the buff regardless of
# who the current actor is.
#
# The single-target tier section in POD_execute_combat_move has
# ALREADY written the buffs onto the actor's workspace + char. These
# team broadcasts just extend that write to the actor's teammates
# (team_ally) or the other enemies (team_enemy). Calling the broadcast
# on a targeting=self or targeting=single_* move is a no-op.
# The POD_execute_combat_move dispatcher gates it on targeting.
#
# Slot soak / hit display won't visually update until the affected
# ally/enemy next loads into workspace (their next turn). The
# character-bridge var for soak IS live immediately, so mid-combat
# damage rolls pick up the buff right away.

POD_combat_apply_team_ally_buffs = {
	# $STORY$ = move story
	# Iteration body runs in slot-story scope so `set_variable = { name = slot_X ... }`
	# writes the slot story directly and `var:slot_char` flips into the
	# participant character for the soak-tier bridge var write.
	save_scope_as = host
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			var:slot_alive = 1
			pod_combat_aoe_team_ally_filter_trigger = yes
		}
		# Tier soak buffs: slot var + char var (PoD soak bridge).
		# Upgrade-only — the OR guard fires when the slot has no
		# current tier (first application) OR the new tier is strictly
		# higher than the existing one. Without this, a team_ally
		# buff with tier-1 lands on an ally who already had tier-3
		# from an earlier move and downgrades them. Mirrors the AI
		# selector's strict-< redundancy semantic but applied
		# per-slot at write time (the selection gate only checks the
		# actor's workspace tier, not every iterated ally's slot).
		if = {
			limit = {
				$STORY$ = { has_variable = soak_buff_round_tier }
				OR = {
					NOT = { has_variable = slot_soak_buff_round_tier }
					var:slot_soak_buff_round_tier < $STORY$.var:soak_buff_round_tier
				}
			}
			set_variable = { name = slot_soak_buff_round_tier value = $STORY$.var:soak_buff_round_tier }
			var:slot_char = {
				set_variable = { name = pod_combat_tier_soak_buff_round value = $STORY$.var:soak_buff_round_tier }
			}
		}
		if = {
			limit = {
				$STORY$ = { has_variable = soak_buff_combat_tier }
				OR = {
					NOT = { has_variable = slot_soak_buff_combat_tier }
					var:slot_soak_buff_combat_tier < $STORY$.var:soak_buff_combat_tier
				}
			}
			set_variable = { name = slot_soak_buff_combat_tier value = $STORY$.var:soak_buff_combat_tier }
			var:slot_char = {
				set_variable = { name = pod_combat_tier_soak_buff_combat value = $STORY$.var:soak_buff_combat_tier }
			}
		}
		# Tier hit buffs: slot var only (hit chance is workspace-local,
		# will be picked up when the ally next loads). Same upgrade-only
		# guard as the soak buffs above.
		if = {
			limit = {
				$STORY$ = { has_variable = hit_buff_round_tier }
				OR = {
					NOT = { has_variable = slot_hit_buff_round_tier }
					var:slot_hit_buff_round_tier < $STORY$.var:hit_buff_round_tier
				}
			}
			set_variable = { name = slot_hit_buff_round_tier value = $STORY$.var:hit_buff_round_tier }
		}
		if = {
			limit = {
				$STORY$ = { has_variable = hit_buff_combat_tier }
				OR = {
					NOT = { has_variable = slot_hit_buff_combat_tier }
					var:slot_hit_buff_combat_tier < $STORY$.var:hit_buff_combat_tier
				}
			}
			set_variable = { name = slot_hit_buff_combat_tier value = $STORY$.var:hit_buff_combat_tier }
		}
		# Stance flags: set_guard / dodge / stealth broadcast to each ally.
		if = {
			limit = { $STORY$ = { has_variable = set_guard } }
			set_variable = { name = slot_is_guarding  value = yes }
		}
		if = {
			limit = { $STORY$ = { has_variable = dodge } }
			set_variable = { name = slot_is_dodging   value = yes }
		}
		if = {
			limit = { $STORY$ = { has_variable = stealth } }
			# Centralised slot-stealth applicator — flag + +30
			# hit_chance, idempotent. Same helper used downstream by
			# any future slot-direct stealth path.
			pod_combat_apply_stealth_slot_effect = yes
		}
	}
}

POD_combat_apply_team_enemy_debuffs = {
	# $STORY$ = move story
	# Mirrors POD_combat_apply_team_ally_buffs but for the enemy-target
	# side: debuff tier vars + bypass/remove flag clearances.
	save_scope_as = host
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			var:slot_alive = 1
			pod_combat_aoe_team_enemy_filter_trigger = yes
		}
		# Upgrade-only — see the symmetric comment in
		# POD_combat_apply_team_ally_buffs above. A tier-1 debuff
		# move shouldn't "heal" an enemy who already had a tier-3
		# debuff from an earlier hit.
		if = {
			limit = {
				$STORY$ = { has_variable = soak_debuff_round_tier }
				OR = {
					NOT = { has_variable = slot_soak_debuff_round_tier }
					var:slot_soak_debuff_round_tier < $STORY$.var:soak_debuff_round_tier
				}
			}
			set_variable = { name = slot_soak_debuff_round_tier value = $STORY$.var:soak_debuff_round_tier }
			var:slot_char = {
				set_variable = { name = pod_combat_tier_soak_debuff_round value = $STORY$.var:soak_debuff_round_tier }
			}
		}
		if = {
			limit = {
				$STORY$ = { has_variable = soak_debuff_combat_tier }
				OR = {
					NOT = { has_variable = slot_soak_debuff_combat_tier }
					var:slot_soak_debuff_combat_tier < $STORY$.var:soak_debuff_combat_tier
				}
			}
			set_variable = { name = slot_soak_debuff_combat_tier value = $STORY$.var:soak_debuff_combat_tier }
			var:slot_char = {
				set_variable = { name = pod_combat_tier_soak_debuff_combat value = $STORY$.var:soak_debuff_combat_tier }
			}
		}
		if = {
			limit = {
				$STORY$ = { has_variable = hit_debuff_round_tier }
				OR = {
					NOT = { has_variable = slot_hit_debuff_round_tier }
					var:slot_hit_debuff_round_tier < $STORY$.var:hit_debuff_round_tier
				}
			}
			set_variable = { name = slot_hit_debuff_round_tier value = $STORY$.var:hit_debuff_round_tier }
		}
		if = {
			limit = {
				$STORY$ = { has_variable = hit_debuff_combat_tier }
				OR = {
					NOT = { has_variable = slot_hit_debuff_combat_tier }
					var:slot_hit_debuff_combat_tier < $STORY$.var:hit_debuff_combat_tier
				}
			}
			set_variable = { name = slot_hit_debuff_combat_tier value = $STORY$.var:hit_debuff_combat_tier }
		}
		# Bypass / remove flags — strip the target's stances.
		if = {
			limit = { $STORY$ = { has_variable = bypass_dodge } }
			remove_variable = slot_is_dodging
		}
		if = {
			limit = { $STORY$ = { has_variable = bypass_stealth } }
			# Centralised slot-stealth teardown — flag removal + −30
			# hit_chance, idempotent. Without the helper, every
			# bypass_stealth AoE used to strip the flag but leave the
			# +30 baked into slot_hit_chance.
			pod_combat_break_stealth_slot_effect = yes
		}
		if = {
			limit = { $STORY$ = { has_variable = remove_guard } }
			remove_variable = slot_is_guarding
		}
	}
}
