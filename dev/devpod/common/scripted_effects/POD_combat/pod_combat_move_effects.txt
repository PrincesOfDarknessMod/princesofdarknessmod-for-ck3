######################################################
######   POD COMBAT MOVE EXECUTION ENGINE
######################################################
# Generic dispatcher: executes any combat move by reading attributes off its
# story_cycle. Called from pod_combat_execute_move SGUI (PC) and
# pod_combat_ai_select_and_execute_move_effect (AI). Caller scope = combat host.
# Helpers live in pod_combat_effects.txt (damage/soak/retaliation),
# pod_combat_tier_effects.txt (tier recalc), pod_combat_group_effects.txt (slot load/save).
#
# Phase contract (§0 -> §17) and load-bearing anti-invariants: COMBAT_REFERENCE.info §2.
# Read that before reordering anything.


######################################################
###### DAMAGE-KIND SEEDING HELPER
######################################################

# Seeds $KIND_VAR$ on the CURRENT scope (host retaliate kinds, slot DoT kind)
# from $STORY$'s damage classification. Literal if-chain, never a var
# copy - flag values degrade to plain 'value' through scope-value copies.
# superficial (the hp default) and absent both fall to flag:duel.
pod_combat_seed_kind_from_class_effect = {
	set_variable = { name = $KIND_VAR$ value = flag:duel }
	if = {
		limit = { $STORY$ = { has_variable = damage_type } }
		if = { limit = { $STORY$ = { var:damage_type = flag:aggravated } } set_variable = { name = $KIND_VAR$ value = flag:aggravated } }
		if = { limit = { $STORY$ = { var:damage_type = flag:fire } }      set_variable = { name = $KIND_VAR$ value = flag:fire } }
		if = { limit = { $STORY$ = { var:damage_type = flag:cold } }      set_variable = { name = $KIND_VAR$ value = flag:cold } }
		if = { limit = { $STORY$ = { var:damage_type = flag:lightning } } set_variable = { name = $KIND_VAR$ value = flag:lightning } }
		if = { limit = { $STORY$ = { var:damage_type = flag:sun } }       set_variable = { name = $KIND_VAR$ value = flag:sun } }
		if = { limit = { $STORY$ = { var:damage_type = flag:silver } }    set_variable = { name = $KIND_VAR$ value = flag:silver } }
		if = { limit = { $STORY$ = { var:damage_type = flag:gold } }      set_variable = { name = $KIND_VAR$ value = flag:gold } }
		if = { limit = { $STORY$ = { var:damage_type = flag:holy } }      set_variable = { name = $KIND_VAR$ value = flag:holy } }
		if = { limit = { $STORY$ = { var:damage_type = flag:balefire } }  set_variable = { name = $KIND_VAR$ value = flag:balefire } }
		if = { limit = { $STORY$ = { var:damage_type = flag:poison } }    set_variable = { name = $KIND_VAR$ value = flag:poison } }
		if = { limit = { $STORY$ = { var:damage_type = flag:cold_iron } } set_variable = { name = $KIND_VAR$ value = flag:cold_iron } }
	}
}

# Re-stamps $KIND_VAR$ on the CURRENT scope from a FIRED change_class
# conversion, so the counter and burn a converted action leaves behind match
# the element it actually struck with. Same literal-chain reason as above.
# Self-guarding: a no-op unless the latch is set and this move is the
# change_class bonus, so call sites need no gate of their own.
pod_combat_override_kind_from_bonus_class_effect = {
	if = {
		limit = {
			scope:combat_host ?= { has_variable = pod_combat_bonus_active }
			$STORY$ = {
				# Guarded chain, not a bare AND: CK3 boolean blocks do not
				# short-circuit, and almost no move declares a bonus at all.
				trigger_if = {
					limit = { has_variable = bonus_effect  has_variable = bonus_param }
					var:bonus_effect = flag:change_class
				}
				trigger_else = { always = no }
			}
		}
		if = { limit = { $STORY$ = { var:bonus_param = flag:fire } }      set_variable = { name = $KIND_VAR$ value = flag:fire } }
		if = { limit = { $STORY$ = { var:bonus_param = flag:cold } }      set_variable = { name = $KIND_VAR$ value = flag:cold } }
		if = { limit = { $STORY$ = { var:bonus_param = flag:lightning } } set_variable = { name = $KIND_VAR$ value = flag:lightning } }
		if = { limit = { $STORY$ = { var:bonus_param = flag:sun } }       set_variable = { name = $KIND_VAR$ value = flag:sun } }
		if = { limit = { $STORY$ = { var:bonus_param = flag:silver } }    set_variable = { name = $KIND_VAR$ value = flag:silver } }
		if = { limit = { $STORY$ = { var:bonus_param = flag:gold } }      set_variable = { name = $KIND_VAR$ value = flag:gold } }
		if = { limit = { $STORY$ = { var:bonus_param = flag:holy } }      set_variable = { name = $KIND_VAR$ value = flag:holy } }
		if = { limit = { $STORY$ = { var:bonus_param = flag:balefire } }  set_variable = { name = $KIND_VAR$ value = flag:balefire } }
		if = { limit = { $STORY$ = { var:bonus_param = flag:poison } }    set_variable = { name = $KIND_VAR$ value = flag:poison } }
		if = { limit = { $STORY$ = { var:bonus_param = flag:cold_iron } } set_variable = { name = $KIND_VAR$ value = flag:cold_iron } }
	}
}


######################################################
###### MAIN ENTRY POINT
######################################################

POD_execute_combat_move = {
	# $STORY$ = the POD_combat_move story_cycle

	# ── 0. Capture combat host as a saved scope ──────────────────
	# Downstream damage routing (pod_combat_route_to_pod_damage_effect) needs
	# scope:combat_host from inside per-target scope-switches. Saving it once
	# here (root = host on both PC and AI entry) avoids a per-iteration
	# `root = { save_scope_as = combat_host }`, which CRASHED the AI AoE chain:
	# the AoE inner scope-switches make the engine treat root as unset when the
	# save runs, blowing up scope:combat_host before the per-target damage read.
	save_scope_as = combat_host

	# ── 1. AP spend ──────────────────────────────────────────────
	if = {
		limit = { $STORY$ = { has_variable = ap_cost } }
		pod_combat_spend_ap_effect = { AMOUNT = $STORY$.var:ap_cost }
	}

	# ── 1b. once_per: mark this move used on the actor's slot ─────
	# Add the move story to the actor slot's round / combat used-list (per the
	# move's once_per); the usability gates reject a move already in the list.
	# Match the actor's slot by its occupant (var:slot_char) - the same key the gate
	# uses; scope:combat_host (saved at §0) holds the actor-char pointer.
	if = {
		limit = {
			$STORY$ = { has_variable = once_per }
			scope:combat_host = { has_variable = pod_combat_actor_char }
		}
		every_in_list = {
			variable = pod_combat_active_slots
			limit = { var:slot_char = { this = scope:combat_host.var:pod_combat_actor_char } }
			if = {
				limit = { $STORY$ = { var:once_per = flag:round } }
				add_to_variable_list = { name = slot_moves_used_round target = $STORY$ }
			}
			else = {
				add_to_variable_list = { name = slot_moves_used_combat target = $STORY$ }
			}
		}
	}

	# ── 2. Result reset + log metadata ───────────────────────────
	# Clear stale pod_combat_last_* so absence = "no outcome yet". Clear list
	# lives in the helper so adding a "last" field is one edit shared across
	# here, push_log's post-capture clear, and combat-end cleanup.
	pod_combat_clear_last_result_vars_effect = yes
	set_variable = { name = pod_combat_last_damage value = 0 }

	# ── 2b. Bonus condition pre-evaluation ───────────────────────
	# cond_mode = bonus: evaluate the state condition ONCE, here in host scope
	# (root = host, acting char = pod_combat_actor_char). A set
	# pod_combat_bonus_active then lets each effect phase grant its bonus.
	# (gate mode is enforced pre-execution in the usability triggers, not here.)
	if = {
		limit = {
			has_variable = pod_combat_actor_char
			$STORY$ = { has_variable = cond_mode  var:cond_mode = flag:bonus }
			pod_combat_move_condition_met_trigger = { STORY = $STORY$ ACTOR = var:pod_combat_actor_char PREDICATE = $STORY$.var:cond_predicate }
		}
		set_variable = pod_combat_bonus_active
		# §6 bonus host vars, consumed by the damage pipeline / aggravated check.
		if = {
			limit = { $STORY$ = { var:bonus_effect = flag:damage } }
			set_variable = { name = pod_combat_bonus_damage value = $STORY$.var:bonus_value }
		}
		else_if = {
			limit = { $STORY$ = { var:bonus_effect = flag:change_type } }
			set_variable = pod_combat_bonus_aggravated
		}
	}

	# ── 2c. Weapon profile (uses_weapon moves) ───────────────────
	# Resolve the actor's equipped weapon ONCE per action into host
	# transients the damage phases read. Unconditional set-or-clear, so an
	# unarmed actor or a non-weapon move actively clears the previous
	# action's profile rather than inheriting it.
	# Runs after 2b on purpose: the uses_weapon conditional bonus grants the
	# swing through the latch set there, so the seeder's gate needs it decided.
	pod_combat_seed_actor_weapon_profile_effect = { STORY = $STORY$ }

	# ── 3. Energy cost (splat-agnostic) ──────────────────────────
	# POD_splat_agnostic_energy_cost routes to the CURRENT scope's splat pool, so
	# MUST scope-switch to the actor first - else every AI cast drains the PC pool.
	if = {
		limit = { $STORY$ = { has_variable = energy_cost } }
		if = {
			limit = { has_variable = pod_combat_actor_char }
			var:pod_combat_actor_char = {
				POD_splat_agnostic_energy_cost = { COUNT = $STORY$.var:energy_cost }
			}
		}
		else = {
			# actor_char not stamped yet (move fired before load_workspace);
			# current scope is the actor in those flows by construction.
			POD_splat_agnostic_energy_cost = { COUNT = $STORY$.var:energy_cost }
		}
		# AI energy-attrition counter, read by the selector on the next AI turn.
		if = {
			limit = { has_variable = pod_combat_actor_energy_used }
			change_variable = { name = pod_combat_actor_energy_used add = 1 }
		}
		else = {
			set_variable = { name = pod_combat_actor_energy_used value = 1 }
		}
	}

	# ── 4. CK3 resource costs + gains (PC only) ──────────────────
	# Only the PC (player/0) pays CK3-side gold/stress/piety/prestige and reaps the
	# steal_gold / gain_dread primitives; AI turns touch no CK3 resources.
	if = {
		limit = {
			has_variable = pod_combat_current_actor_team
			var:pod_combat_current_actor_team = flag:player
			var:pod_combat_current_actor_position = 0
		}
		if = {
			limit = { $STORY$ = { has_variable = gold_cost } }
			remove_short_term_gold = $STORY$.var:gold_cost
		}
		if = {
			limit = { $STORY$ = { has_variable = stress_cost } }
			add_stress = $STORY$.var:stress_cost
		}
		if = {
			limit = { $STORY$ = { has_variable = piety_cost } }
			add_piety = {
				value = $STORY$.var:piety_cost
				multiply = -1
			}
		}
		if = {
			limit = { $STORY$ = { has_variable = prestige_cost } }
			add_prestige = {
				value = $STORY$.var:prestige_cost
				multiply = -1
			}
		}

		# Phase 4 gains. steal_gold transfers from the target, clamped to what it
		# actually has so a broke victim never mints gold for the PC; gain_dread
		# adds to the actor. this = the player/0 actor here (the PC-only gate),
		# so the actor-side reads are direct, same as the costs above.
		# Practice spars leave no lasting transfers (the contract that restores
		# wounds and skips strain), so the gains skip too - no risk-free
		# gold-theft or dread loop against sparring partners.
		if = {
			limit = { NOT = { pod_combat_is_practice_trigger = yes } }
			if = {
				limit = {
					$STORY$ = { has_variable = steal_gold }
					has_variable = pod_combat_target_char
				}
				var:pod_combat_target_char = {
					save_temporary_scope_value_as = {
						name = pod_combat_stolen_gold
						value = { value = $STORY$.var:steal_gold  max = gold  min = 0 }
					}
					# add_gold rejects negative values, so debit the victim with
					# remove_short_term_gold (the clamp above keeps it <= their gold).
					remove_short_term_gold = scope:pod_combat_stolen_gold
				}
				add_gold = scope:pod_combat_stolen_gold
			}
			if = {
				limit = { $STORY$ = { has_variable = gain_dread } }
				add_dread = $STORY$.var:gain_dread
			}
		}
	}

	# ── 5. Target stance strip ──────────────────────────────────
	# Strip target dodge/stealth/guard BEFORE damage resolves, so a strip + attack
	# combo lands its own attack. bypass_dodge / bypass_stealth are plain move
	# presence flags (silent strips); remove_guard (below) also logs. bypass_guard has no
	# runtime var: the generator omits the checks_guard capability flag and the
	# contact resolver reads that omission directly.
	if = {
		limit = { $STORY$ = { has_variable = bypass_dodge } }
		remove_variable = pod_combat_target_is_dodging
	}
	if = {
		limit = { $STORY$ = { has_variable = bypass_stealth } }
		# Must also subtract the +30 stealth hit bonus baked into saved
		# hit_chance, else the target keeps a phantom offensive boost. Idempotent.
		pod_combat_break_stealth_workspace_effect = { SIDE = target }
	}
	# remove_guard is a move EFFECT (it logs seg_remove_guard), but it strips the
	# same kind of target stance, so it runs here too, BEFORE §6, so a strip-guard
	# + attack move lands its own attack instead of being guard-blocked. AoE strips
	# each enemy slot's guard in §5b; this workspace write is redundant for
	# team_enemy (like the §5c target-writes).
	if = {
		limit = {
			$STORY$ = {
				has_variable = remove_guard
				NOT = { has_variable = requires_control_contact }
			}
		}
		remove_variable = pod_combat_target_is_guarding
	}

	# ── 5b. AoE debuff pre-damage broadcast ──────────────────────
	# team_enemy: broadcast tier debuffs + stance-strips to ALL enemy slots
	# BEFORE the damage iteration, so each hit respects them on its own target.
	# §5's workspace strip only touches the pre-AoE target, which each iteration
	# overwrites. §5c's target-writes still run but are redundant for team_enemy.
	# team_all does the same across both teams, plus the caster's own tier share.
	if = {
		limit = { $STORY$ = { var:targeting = flag:team_enemy } }
		pod_combat_apply_team_enemy_debuffs_effect = { STORY = $STORY$ }
	}
	else_if = {
		limit = { $STORY$ = { var:targeting = flag:team_all } }
		pod_combat_apply_team_all_debuffs_effect = { STORY = $STORY$ }
		pod_combat_apply_team_all_self_debuffs_effect = { STORY = $STORY$ }
	}

	# ── 5b2. Pure-control contact delivery ────────────────────────
	# Explicitly typed single-target controls have no HP/morale phase to invoke
	# the contact resolver for them. The generator marks those rows with
	# requires_control_contact; resolve their advertised spirit/stance/hit/guard
	# rules here, before any target-side debuff/control payload is applied.
	if = {
		limit = { $STORY$ = { has_variable = requires_control_contact } }
		$STORY$ = { save_scope_as = current_move }
		pod_combat_resolve_contact_effect = {
			TARGET_SIDE = target
			ACTOR_SIDE  = actor
		}
		if = {
			limit = { NOT = { scope:contact_result = 0 } }
			set_variable = pod_combat_control_delivery_failed
		}
		# A hostile control attempt breaks the caster's stealth just like an HP
		# attack, whether the delivery lands or is evaded.
		pod_combat_consume_actor_stealth_effect = { ACTOR_SIDE = actor }
	}
	# A control-delivered guard strip is part of that delivery, unlike an
	# ordinary strip+attack move whose pre-damage strip remains in §5.
	if = {
		limit = {
			$STORY$ = {
				has_variable = remove_guard
				has_variable = requires_control_contact
			}
			NOT = { has_variable = pod_combat_control_delivery_failed }
		}
		remove_variable = pod_combat_target_is_guarding
	}

	# ── 5c. Tier buffs/debuffs (applied BEFORE the §6 damage) ──────────
	# Runs ahead of damage so a buff/debuff + attack in the SAME move lands on
	# THIS attack: a self hit buff raises pod_combat_current_hit_chance for the
	# §6 hit roll, and a target soak debuff sets the POD_soak_chance bridge var
	# before the §6 wound roll. AoE target debuffs are still broadcast to slots in
	# §5b; the workspace target-writes below stay redundant for team_enemy/team_all.
	# Buff tiers route to the actor for both team AoE modes (only single_ally and
	# single_pc redirect to the target), so a team_all buff would land on the
	# caster alone - the generator lint rejects support payloads on team_all.
	# Buffs normally apply to the actor. A single_ally move names a
	# beneficiary in the target workspace, so route every tier and its soak
	# bridge to that target instead. team_ally still seeds the actor here and
	# broadcasts to every allied slot in §12b.
	if = {
		limit = { NOT = { has_variable = pod_combat_control_delivery_failed } }
		if = {
			limit = { $STORY$ = { var:targeting = flag:single_ally } }
			pod_combat_apply_workspace_buff_tiers_effect = { STORY = $STORY$ SIDE = target }
		}
		else = {
			pod_combat_apply_workspace_buff_tiers_effect = { STORY = $STORY$ SIDE = actor }
		}
	}

	# A failed pure-control contact has no target-side payload. Unmarked moves
	# never set this tripwire, preserving the existing pre-damage debuff timing.
	if = {
		limit = { NOT = { has_variable = pod_combat_control_delivery_failed } }
		pod_combat_apply_workspace_debuff_tiers_effect = { STORY = $STORY$ SIDE = target }
	}

	# ── 5d. Conditional bonus hit chance ────────────────────────
	# +bonus_value hit for THIS attack when the move's condition held (§2b).
	# Sits on top of any tier recalc above; same ceiling as the recalc.
	if = {
		limit = {
			has_variable = pod_combat_bonus_active
			has_variable = pod_combat_current_hit_chance
			$STORY$ = { var:bonus_effect = flag:hit }
		}
		change_variable = { name = pod_combat_current_hit_chance add = $STORY$.var:bonus_value }
		# Re-cap at the shared ceiling. No floor needed: the generator rejects
		# a negative bonus_value, so a bump can only raise the chance.
		if = {
			limit = { var:pod_combat_current_hit_chance > pod_combat_hit_chance_max }
			set_variable = { name = pod_combat_current_hit_chance value = pod_combat_hit_chance_max }
		}
	}

	# ── 6. HP damage dispatch ────────────────────────────────────
	# Routes single_enemy vs team AoE. See HP DAMAGE DISPATCH header below.
	if = {
		limit = { $STORY$ = { has_variable = hp_damage } }
		if = {
			limit = { $STORY$ = { var:targeting = flag:team_enemy } }
			pod_combat_apply_team_enemy_hp_damage_effect = { STORY = $STORY$ }
		}
		else_if = {
			limit = { $STORY$ = { var:targeting = flag:team_all } }
			pod_combat_apply_team_all_hp_damage_effect = { STORY = $STORY$ }
		}
		else = {
			# single_enemy (default, most common)
			pod_combat_apply_single_hp_damage_effect = { STORY = $STORY$ TARGET_SIDE = target }
		}
	}

	# ── 7. Morale damage ─────────────────────────────────────────
	# Own pipeline: auto-hit, no guard, no soak, dodge still works.
	# Rider gate (here and §8/§9): a *_condition var on the move makes the
	# payload fire only on hit/miss/contact, read off §6's outcome on the host.
	# Default (no condition) = fires unconditionally.
	# The conditional bonus (bonus_effect = morale_damage, held at §2b) folds
	# into the SAME deal call as the base amount: one call per target consumes
	# the target's stance exactly once for the combined total (a second
	# independent call pierced the dodge/stealth the base call had just
	# consumed) and the logged amount is the true total. A bonus-only move (no
	# base morale_damage) takes the same path with a base of 0, including the
	# team_enemy broadcast (the team applier reads the combined scope value).
	if = {
		limit = {
			OR = {
				$STORY$ = { has_variable = morale_damage }
				AND = {
					has_variable = pod_combat_bonus_active
					$STORY$ = { var:bonus_effect = flag:morale_damage }
				}
			}
			pod_combat_rider_condition_passes_trigger = {
				STORY     = $STORY$
				RIDER_VAR = morale_damage_condition
			}
		}
		save_scope_value_as = {
			name = pod_combat_morale_deal_total
			value = {
				value = 0
				if = { limit = { $STORY$ = { has_variable = morale_damage } } add = $STORY$.var:morale_damage }
				if = {
					limit = { has_variable = pod_combat_bonus_active  $STORY$ = { var:bonus_effect = flag:morale_damage } }
					add = $STORY$.var:bonus_value
				}
			}
		}
		if = {
			limit = { $STORY$ = { var:targeting = flag:team_enemy } }
			pod_combat_apply_team_enemy_morale_damage_effect = { STORY = $STORY$ }
		}
		else_if = {
			limit = { $STORY$ = { var:targeting = flag:team_all } }
			pod_combat_apply_team_all_morale_damage_effect = { STORY = $STORY$ }
		}
		else = {
			pod_combat_deal_morale_damage_effect = { TARGET_SIDE = target AMOUNT = scope:pod_combat_morale_deal_total }
		}
	}

	# ── 8. Self-damage ───────────────────────────────────────────
	# Routed through the PoD pipeline like a real hit so wound state stays in
	# lockstep. REASON=duel (superficial, soakable) not self_mutilation - 
	# overextension, not a permanent wound. Rider-gated (see §7).
	if = {
		limit = {
			$STORY$ = { has_variable = self_damage }
			pod_combat_rider_condition_passes_trigger = {
				STORY     = $STORY$
				RIDER_VAR = self_damage_condition
			}
		}
		pod_combat_deal_hp_damage_bypass_guard_effect = {
			TARGET_SIDE = actor
			AMOUNT      = $STORY$.var:self_damage
			REASON      = duel
		}
		# Lift the post-soak recoil amount for the log's recoil segment.
		# scope:attack_result_flag stays discarded - recoil never overwrites the
		# main attack's outcome axes.
		if = {
			limit = { scope:damage_dealt > 0 }
			set_variable = { name = pod_combat_last_self_damage value = scope:damage_dealt }
		}
	}

	# ── 9. HP heal ───────────────────────────────────────────────
	# heal_type=aggravated routes to the aggravated variant (also reduces
	# agg_damage); else superficial (caps at max_hp - agg_damage).
	# Rider-gated (see §7) on heal_condition - drain heals get
	# heal_condition=on_hp_damage auto-emitted by the generator's `drain` token.
	# team_all has no broadcast arm here (nor in §10/§12b): a heal reaching both
	# sides would mend the enemies the same move just hit. It falls to the actor
	# self-heal else, and the generator lint rejects the combination outright.
	if = {
		limit = {
			$STORY$ = { has_variable = hp_heal }
			pod_combat_rider_condition_passes_trigger = {
				STORY     = $STORY$
				RIDER_VAR = heal_condition
			}
		}
		if = {
			limit = { $STORY$ = { var:targeting = flag:team_ally } }
			pod_combat_apply_team_ally_hp_heal_effect = { STORY = $STORY$ }
		}
		else_if = {
			# single_ally / single_pc - the beneficiary is loaded into the target
			# workspace (the ally by the PC's picker, the PC by the AI swap), so
			# SIDE=target heals them instead of the actor.
			limit = { $STORY$ = { OR = { var:targeting = flag:single_ally  var:targeting = flag:single_pc } } }
			if = {
				limit = { $STORY$ = { has_variable = heal_type var:heal_type = flag:aggravated } }
				pod_combat_restore_hp_aggravated_effect = { SIDE = target AMOUNT = $STORY$.var:hp_heal }
			}
			else = {
				pod_combat_restore_hp_superficial_effect = { SIDE = target AMOUNT = $STORY$.var:hp_heal }
			}
		}
		else_if = {
			limit = { $STORY$ = { has_variable = heal_type var:heal_type = flag:aggravated } }
			pod_combat_restore_hp_aggravated_effect = { SIDE = actor AMOUNT = $STORY$.var:hp_heal }
		}
		else = {
			pod_combat_restore_hp_superficial_effect = { SIDE = actor AMOUNT = $STORY$.var:hp_heal }
		}
	}

	# ── 9b. Conditional bonus heals / morale / energy ───────────
	# Bonus heal when the move's condition held (set §2b). Independent of the
	# base hp_heal gate; superficial. self_heal -> actor, heal_target -> target.
	if = {
		limit = { has_variable = pod_combat_bonus_active  $STORY$ = { var:bonus_effect = flag:self_heal } }
		pod_combat_restore_hp_superficial_effect = { SIDE = actor AMOUNT = $STORY$.var:bonus_value }
	}
	if = {
		limit = { has_variable = pod_combat_bonus_active  $STORY$ = { var:bonus_effect = flag:heal_target } }
		pod_combat_restore_hp_superficial_effect = { SIDE = target AMOUNT = $STORY$.var:bonus_value }
	}
	# Bonus morale heal, actor only - there is deliberately no target twin (the
	# unconditional morale_heal column already covers allies). The primitive
	# caps at the actor's own max and writes the ACTUAL delta to
	# pod_combat_last_morale_heal_amount, which the log entry capture, the
	# segment bake and the §16 wrap-up gate already read, so this needs no log
	# work of its own.
	if = {
		limit = { has_variable = pod_combat_bonus_active  $STORY$ = { var:bonus_effect = flag:self_morale_heal } }
		pod_combat_restore_morale_effect = { SIDE = actor AMOUNT = $STORY$.var:bonus_value }
	}

	# Bonus energy: restore to the actor (energy_gain) or drain the target
	# (energy_drain) when the move's condition held (§2b). Splat-agnostic
	# COUNT routes to the current scope's pool, so scope-switch first (mirror of
	# §3's energy_cost). Applies for AI actors too, so not PC-gated.
	if = {
		limit = { has_variable = pod_combat_bonus_active  $STORY$ = { var:bonus_effect = flag:energy_gain } }
		if = {
			limit = { has_variable = pod_combat_actor_char }
			var:pod_combat_actor_char = { POD_splat_agnostic_energy_gain = { COUNT = $STORY$.var:bonus_value } }
		}
		else = { POD_splat_agnostic_energy_gain = { COUNT = $STORY$.var:bonus_value } }
		# Lift for the log segment (push_log capture + the §16 wrap-up gate).
		# The REQUESTED amount: the splat-agnostic helpers don't expose the
		# applied gain, so a pool-capped partial still logs the full bonus.
		# Scaled to the actor's own resource so the row can read "+1 Blood".
		pod_combat_scale_energy_to_splat_effect = { CHARACTER = var:pod_combat_actor_char COUNT = $STORY$.var:bonus_value }
		set_variable = { name = pod_combat_last_energy_gain value = scope:pod_combat_scaled_energy }
	}
	if = {
		limit = {
			has_variable = pod_combat_bonus_active
			$STORY$ = { var:bonus_effect = flag:energy_drain }
			has_variable = pod_combat_target_char
			# Practice contract: no lasting harm to the partner - a real
			# energy drain would persist past the spar.
			NOT = { pod_combat_is_practice_trigger = yes }
		}
		var:pod_combat_target_char = { POD_splat_agnostic_energy_loss = { COUNT = $STORY$.var:bonus_value } }
		# Requested amount, like the gain lift above, but scaled to the TARGET:
		# it is their pool that shrinks, and their splat decides by how much.
		# Inside the practice gate, so a spar never logs a drain that was skipped.
		pod_combat_scale_energy_to_splat_effect = { CHARACTER = var:pod_combat_target_char COUNT = $STORY$.var:bonus_value }
		set_variable = { name = pod_combat_last_energy_drain value = scope:pod_combat_scaled_energy }
	}

	# ── 9c. Weapon recovery riders ───────────────────────────────
	# A weapon that feeds on the wounds it opens. Fires ONCE per action, not
	# once per victim: pod_combat_result_code_indicates_hp_damage_trigger reads
	# the live result pair on a single target and the AoE latch on a sweep, so
	# a three-target swing still returns one drink. Damage results are final by
	# here (§6 closed, both sweeps ran inside it) and §15 counters have not run
	# yet, so the actor is necessarily still alive - no is_alive guard, same as
	# the §9b bonus heals above.
	#
	# A fully-soaked hit is not "damage": the predicate excludes flag:soaked, so
	# armour that stops the blade also stops the feeding.
	if = {
		limit = {
			has_variable = pod_combat_weapon_heal_on_damage
			pod_combat_result_code_indicates_hp_damage_trigger = yes
		}
		# Absolute scope reference, not a bare var: the route helper consumes
		# $AMOUNT$ inside a switch to the healed CHARACTER, where a host-relative
		# read would find nothing (§9b passes $STORY$.var:bonus_value for the
		# same reason).
		pod_combat_restore_hp_superficial_effect = { SIDE = actor AMOUNT = scope:combat_host.var:pod_combat_weapon_heal_on_damage }
	}
	if = {
		limit = {
			has_variable = pod_combat_weapon_energy_on_damage
			pod_combat_result_code_indicates_hp_damage_trigger = yes
			has_variable = pod_combat_actor_char
		}
		# Scope-switch first: the splat-agnostic pool is the CURRENT scope's,
		# so a bare call would refill the PC on every AI swing (§3's trap).
		# COUNT is therefore read in CHARACTER scope and must be absolute.
		var:pod_combat_actor_char = { POD_splat_agnostic_energy_gain = { COUNT = scope:combat_host.var:pod_combat_weapon_energy_on_damage } }
		# Share the §9b lift so the existing energy-gain log segment renders it.
		# Add rather than overwrite: a move can fire an energy_gain bonus AND
		# carry a feeding weapon, and the row should show the whole gain.
		# Both amounts are scaled to the actor's resource, so the sum is too.
		pod_combat_scale_energy_to_splat_effect = { CHARACTER = var:pod_combat_actor_char COUNT = scope:combat_host.var:pod_combat_weapon_energy_on_damage }
		if = {
			limit = { has_variable = pod_combat_last_energy_gain }
			change_variable = { name = pod_combat_last_energy_gain add = scope:pod_combat_scaled_energy }
		}
		else = {
			set_variable = { name = pod_combat_last_energy_gain value = scope:pod_combat_scaled_energy }
		}
	}

	# ── 10. Morale heal ──────────────────────────────────────────
	if = {
		limit = { $STORY$ = { has_variable = morale_heal } }
		if = {
			limit = { $STORY$ = { var:targeting = flag:team_ally } }
			pod_combat_apply_team_ally_morale_heal_effect = { STORY = $STORY$ }
		}
		else_if = {
			# single_ally / single_pc - see §9's branch.
			limit = { $STORY$ = { OR = { var:targeting = flag:single_ally  var:targeting = flag:single_pc } } }
			pod_combat_restore_morale_effect = { SIDE = target AMOUNT = $STORY$.var:morale_heal }
		}
		else = {
			pod_combat_restore_morale_effect = { SIDE = actor AMOUNT = $STORY$.var:morale_heal }
		}
	}

	# ── 12. Non-tier presence flags ──────────────────────────────
	# Like tier buffs, defensive stances belong to the selected beneficiary on
	# single_ally moves and to the actor otherwise.
	if = {
		limit = { $STORY$ = { var:targeting = flag:single_ally } }
		pod_combat_apply_workspace_stances_effect = { STORY = $STORY$ SIDE = target }
	}
	else = {
		pod_combat_apply_workspace_stances_effect = { STORY = $STORY$ SIDE = actor }
	}

	# Bonus stance for the actor when the move's condition held (set §2b). The
	# stance is encoded in bonus_effect (no value), so bonus_value stays unused.
	if = {
		limit = { has_variable = pod_combat_bonus_active  $STORY$ = { var:bonus_effect = flag:stance_guard } }
		set_variable = { name = pod_combat_actor_is_guarding value = yes }
	}
	if = {
		limit = { has_variable = pod_combat_bonus_active  $STORY$ = { var:bonus_effect = flag:stance_dodge } }
		set_variable = { name = pod_combat_actor_is_dodging value = yes }
	}
	if = {
		limit = { has_variable = pod_combat_bonus_active  $STORY$ = { var:bonus_effect = flag:stance_stealth } }
		pod_combat_apply_stealth_workspace_effect = { SIDE = actor }
	}

	# ── 12b. AoE buff broadcast ──────────────────────────────────
	# team_ally: extend §5c/§12's actor buffs (tiers + stance flags) to ALL ally
	# slots, plus the soak bridge var on each ally's char so POD_soak_chance picks
	# it up immediately rather than only on their next turn.
	if = {
		limit = { $STORY$ = { var:targeting = flag:team_ally } }
		pod_combat_apply_team_ally_buffs_effect = { STORY = $STORY$ }
	}

	# ── 12c. Marker character modifier ───────────────────────────
	# Stamps a persistent marker on the actor that LATER moves read through the
	# has_modifier condition predicate. Deliberately after §2b, so a move never
	# satisfies its own condition; the payoff is cross-move.
	# The switch exists because CK3 needs a literal for add_character_modifier -
	# marker_modifier holds a flag, which cannot be interpolated into the key.
	# Mirrors the generated pod_combat_cond_has_modifier_trigger; a new key needs
	# a case in both, plus removal in the two teardown effects and slot init.
	# days = 30 is only a leak backstop (a fight resolves within the same day);
	# combat teardown is what normally strips it.
	if = {
		limit = {
			$STORY$ = { has_variable = marker_modifier }
			has_variable = pod_combat_actor_char
		}
		switch = {
			trigger = $STORY$.var:marker_modifier
			flag:pod_combat_anima_flare = {
				var:pod_combat_actor_char = {
					add_character_modifier = {
						modifier = pod_combat_anima_flare
						days     = 30
					}
				}
			}
		}
	}

	# ── 13. Retaliation stances ──────────────────────────────────
	# Arms a reaction that fires in §15 next turn when the actor is attacked.
	#   retaliate_if_hit - fires on contact (hp_damage/guard_blocked/soaked).
	#                         Thorns: any attack reaching the defender counters,
	#                         even if soak/guard absorbed it.
	#   retaliate_if_missed - fires on evasion (missed/dodged/stealthed). Riposte.
	#   retaliate_if_attacked - fires on ANY resolved attack, hit or miss (and on
	#                         immunity, which belongs to neither partition). One
	#                         charge: a specific stance above answers first and
	#                         holds this one. See pod_combat_check_retaliation_effect.
	# Amount fallback, agg marking, and kind seeding: see
	# pod_combat_arm_retaliation_effect (the canonical contract).
	if = {
		limit = { $STORY$ = { has_variable = retaliate_if_hit } }
		pod_combat_arm_retaliation_effect = { STORY = $STORY$ WHEN = hit }
	}
	if = {
		limit = { $STORY$ = { has_variable = retaliate_if_missed } }
		pod_combat_arm_retaliation_effect = { STORY = $STORY$ WHEN = miss }
	}
	if = {
		limit = { $STORY$ = { has_variable = retaliate_if_attacked } }
		pod_combat_arm_retaliation_effect = { STORY = $STORY$ WHEN = any }
	}

	# ── 13b. Damage-over-Time application ─────────────────────────
	# Seed a burn on the target's slot; it ticks at each round end
	# (pod_combat_tick_dots_effect in group_end_round). Single-target for v1 - an
	# AoE DoT would fold into the §5b/AoE phase. Refresh and kind-precedence
	# rules: see pod_combat_seed_dot_on_slot_effect (the canonical contract).
	# After the damage phases so the move's own direct hit (if any) lands first.
	if = {
		limit = {
			$STORY$ = { has_variable = dot_rounds }
			has_variable = pod_combat_target_char
			# Don't seed a burn the delivery never landed (dodged/stealthed/
			# missed). NOT(miss) - not the positive contact predicate - so a landed
			# hp0 MORALE DoT (outcome=morale_damage, no attack_result) still applies.
			NOT = { pod_combat_result_code_indicates_miss_trigger = yes }
		}
		every_in_list = {
			variable = pod_combat_active_slots
			# Skip a target the same move just killed/downed (no affliction row on a corpse).
			limit = { var:slot_char = { this = scope:combat_host.var:pod_combat_target_char }  NOT = { pod_combat_slot_should_be_marked_ko_trigger = yes } }
			pod_combat_seed_dot_on_slot_effect = { STORY = $STORY$ AMOUNT = $STORY$.var:dot_amount ROUNDS = $STORY$.var:dot_rounds CASTER = scope:combat_host.var:pod_combat_actor_char }
			# Application log: morale is explicit; HP stays class-neutral because
			# the wound class resolves per victim when each later tick lands.
			if = {
				limit = { $STORY$ = { has_variable = dot_morale } }
				pod_combat_push_dot_applied_log_effect = { MESSAGE_KEY = dot_applied_morale }
			}
			else = {
				pod_combat_push_dot_applied_log_effect = { MESSAGE_KEY = dot_applied }
			}
		}
	}


	# ── 13b-bonus. Conditional bonus DoT ─────────────────────────
	# A burn applied to the target when the move's bonus condition held (Section
	# 2b). bonus_value = per-tick amount; fixed pod_combat_bonus_dot_rounds duration
	# (the bonus carries one scalar). Same target slot + kind seeding as 13b; the
	# tick is typed from the damage class, the application log uses the generic burn row.
	if = {
		limit = {
			has_variable = pod_combat_bonus_active
			$STORY$ = { var:bonus_effect = flag:dot }
			has_variable = pod_combat_target_char
			# Same delivery gate as §13b (see there).
			NOT = { pod_combat_result_code_indicates_miss_trigger = yes }
		}
		every_in_list = {
			variable = pod_combat_active_slots
			limit = { var:slot_char = { this = scope:combat_host.var:pod_combat_target_char }  NOT = { pod_combat_slot_should_be_marked_ko_trigger = yes } }
			pod_combat_seed_dot_on_slot_effect = { STORY = $STORY$ AMOUNT = $STORY$.var:bonus_value ROUNDS = pod_combat_bonus_dot_rounds CASTER = scope:combat_host.var:pod_combat_actor_char }
			pod_combat_push_dot_applied_log_effect = { MESSAGE_KEY = dot_applied }
		}
	}
	# ── 13b-bonus-morale. Conditional bonus morale DoT ───────────
	# As 13b-bonus, but the burn erodes will rather than flesh. The kind force
	# lives inside pod_combat_seed_dot_on_slot_effect (a $STORY$ read, ordered
	# after the weapon retype like the base dot_morale flag) rather than a slot
	# write here, which the seed effect's contract rules out. The generator
	# forbids pairing this with a base DoT, so the two never race for the one
	# slot_dot_* slot.
	if = {
		limit = {
			has_variable = pod_combat_bonus_active
			$STORY$ = { var:bonus_effect = flag:morale_dot }
			has_variable = pod_combat_target_char
			# Same delivery gate as §13b (see there).
			NOT = { pod_combat_result_code_indicates_miss_trigger = yes }
		}
		every_in_list = {
			variable = pod_combat_active_slots
			limit = { var:slot_char = { this = scope:combat_host.var:pod_combat_target_char }  NOT = { pod_combat_slot_should_be_marked_ko_trigger = yes } }
			pod_combat_seed_dot_on_slot_effect = { STORY = $STORY$ AMOUNT = $STORY$.var:bonus_value ROUNDS = pod_combat_bonus_dot_rounds CASTER = scope:combat_host.var:pod_combat_actor_char }
			pod_combat_push_dot_applied_log_effect = { MESSAGE_KEY = dot_applied_morale }
		}
	}
	# ── 13c. Mind control application ───────────────────────────
	# Dispatched on the mind_control move flag: marks the target slot
	# controlled, stealing their next turn (pod_combat_execute_ai_turn_effect
	# inverts their target selection once, then clears the marker). Players
	# are immune; PoD's mental-defense trait flags gate the rest (the
	# defender's strong_mental_defenses loses to the caster's bypass flag).
	# Single-target only: an AoE mind-control move would affect the pre-AoE
	# target alone. A blocked cast posts a visible resist row.
	if = {
		limit = {
			$STORY$ = { has_variable = mind_control }
			has_variable = pod_combat_target_char
			NOT = { has_variable = pod_combat_control_delivery_failed }
			# Morale-delivered mind control only lands when that morale attack was
			# not evaded. Pure mind-control rows use the control-contact tripwire.
			OR = {
				$STORY$ = { NOT = { has_variable = morale_damage } }
				has_variable = pod_combat_last_morale_damage_amount
			}
		}
		if = {
			limit = {
				var:pod_combat_target_char = { is_ai = yes }
				trigger_if = {
					limit = { var:pod_combat_target_char = { POD_has_strong_mental_defenses_trigger = yes } }
					var:pod_combat_actor_char = { can_bypass_mental_defenses_trigger = yes }
				}
			}
			every_in_list = {
				variable = pod_combat_active_slots
				limit = { var:slot_char = { this = scope:combat_host.var:pod_combat_target_char } }
				set_variable = { name = slot_controlled value = 1 }
				var:slot_char = { save_scope_as = mind_log_victim }
				pod_combat_push_mind_log_effect = { MESSAGE_KEY = mind_seized }
			}
		}
		else = {
			var:pod_combat_target_char = { save_scope_as = mind_log_victim }
			pod_combat_push_mind_log_effect = { MESSAGE_KEY = mind_resisted }
		}
	}

	# ── 14. Parametric non-tier flags ────────────────────────────
	if = {
		limit = { $STORY$ = { has_variable = ap_gain } }
		change_variable = { name = pod_combat_current_ap add = $STORY$.var:ap_gain }
	}

	# Bonus AP refund when the condition held (set §2b). The generator caps
	# bonus_value below ap_cost, so the move still costs at least 1 AP net.
	# The transient lift is what makes the refund visible in the log: the
	# segment is outcome-style, so it must render only when the bonus actually
	# fired, never off the move's declared bonus_effect.
	if = {
		limit = { has_variable = pod_combat_bonus_active  $STORY$ = { var:bonus_effect = flag:ap_refund } }
		change_variable = { name = pod_combat_current_ap add = $STORY$.var:bonus_value }
		set_variable = { name = pod_combat_last_ap_refund value = $STORY$.var:bonus_value }
	}
	# AP-drain / init-debuff write the target workspace mirrors; save_target
	# pushes them to the slot. Consumed at different times:
	#   * slot_ap_drain - target's NEXT TURN start (pod_combat_apply_ap_drain_effect).
	#   * slot_init_bonus - ROUND END, read off the slot directly by
	#     pod_combat_calculate_initiative_effect (outside the workspace lifecycle).
	# Effective AP drain = base ap_drain (if any) + bonus ap_drain (cond_mode=bonus,
	# bonus_effect=ap_drain, condition held at §2b). Summed FIRST so the take-max
	# below weighs this move's full drain, not the base alone - a bonus added after
	# the take-max could bleed onto another attacker's larger pending value.
	if = {
		limit = {
			NOT = { has_variable = pod_combat_control_delivery_failed }
			OR = {
				$STORY$ = { has_variable = ap_drain }
				AND = {
					has_variable = pod_combat_bonus_active
					$STORY$ = { var:bonus_effect = flag:ap_drain }
				}
			}
		}
		save_scope_value_as = {
			name = effective_ap_drain
			value = {
				value = 0
				if = { limit = { $STORY$ = { has_variable = ap_drain } } add = $STORY$.var:ap_drain }
				if = {
					limit = { has_variable = pod_combat_bonus_active  $STORY$ = { var:bonus_effect = flag:ap_drain } }
					add = $STORY$.var:bonus_value
				}
			}
		}
		# Take-max, never overwrite: slot_ap_drain round-trips through this mirror, so a
		# teammate's small drain landing after a full drain (ap_drain_6 = the stun) must
		# not shrink the pending value.
		if = {
			limit = { NOT = { has_variable = pod_combat_target_ap_drain } }
			set_variable = { name = pod_combat_target_ap_drain value = scope:effective_ap_drain }
		}
		else_if = {
			limit = { var:pod_combat_target_ap_drain < scope:effective_ap_drain }
			set_variable = { name = pod_combat_target_ap_drain value = scope:effective_ap_drain }
		}
		# The refund lift's outcome-style twin, carrying the EFFECTIVE total so the
		# log segment's stun arm can label a base+bonus stack the base alone would
		# read as a partial daze. Must stay inside this block's control-failure
		# limit: a resisted pure-control move drains nothing and must log nothing.
		if = {
			limit = { scope:effective_ap_drain > 0 }
			set_variable = { name = pod_combat_last_ap_drain value = scope:effective_ap_drain }
		}
	}
	if = {
		limit = {
			$STORY$ = { has_variable = initiative_debuff }
			NOT = { has_variable = pod_combat_control_delivery_failed }
		}
		save_scope_value_as = {
			name = init_debuff_neg
			value = {
				value = $STORY$.var:initiative_debuff
				multiply = -1
			}
		}
		# Stacking, not overwrite: multiple debuffs landing on one target within a
		# round (multiple attackers, or one actor's multi-action turn) accumulate
		# before the round-end initiative fold, matching the documented stacked-debuff
		# design (COMBAT_REFERENCE initiative section). Numbers, so the slot_init_bonus
		# round-trip carries no flag degradation.
		if = {
			limit = { NOT = { has_variable = pod_combat_target_init_bonus } }
			set_variable = { name = pod_combat_target_init_bonus value = scope:init_debuff_neg }
		}
		else = {
			change_variable = { name = pod_combat_target_init_bonus add = scope:init_debuff_neg }
		}
	}

	# ── 14b. Summon a combatant ──────────────────────────────────
	# Placed after the damage/AoE phases so a same-cast AoE never hits the just-summoned
	# unit, and before retaliation so it can't counter on its summon turn (it acts NEXT
	# round; the add sets slot_has_acted = 1). Glue is hidden_effect-wrapped (tooltip-build).
	if = {
		limit = { $STORY$ = { has_variable = summons_combatant } }
		pod_combat_summon_combatant_glue_effect = { MOVE_STORY = $STORY$ }
	}

	# ── 15. Retaliation check ────────────────────────────────────
	# Fire any retaliation the target armed on a previous turn.
	# Skipped for team AoE: retaliation can only fire on one workspace target, so
	# firing it would arbitrarily single out the pre-AoE target.
	if = {
		limit = {
			$STORY$ = { has_variable = hp_damage }
			$STORY$ = { NOR = { var:targeting = flag:team_enemy  var:targeting = flag:team_all } }
		}
		pod_combat_check_retaliation_effect = { TARGET_SIDE = target ACTOR_SIDE = actor }
	}

	# ── 16. Push combat log entry ────────────────────────────────
	# AoE-aware gate: AoE iterators push per-target rows and set the
	# pod_combat_aoe_phase_pushed_rows tripwire. When set, skip the wrap-up row
	# UNLESS actor-side state fired that the per-target rows missed (self-heal /
	# self-morale-heal / retal overlay) - then the wrap-up carries just those
	# riders. Non-AoE paths never set the tripwire, so they always push.
	if = {
		limit = {
			OR = {
				NOT = { has_variable = pod_combat_aoe_phase_pushed_rows }
				has_variable = pod_combat_last_heal_amount
				has_variable = pod_combat_last_morale_heal_amount
				has_variable = pod_combat_last_retal_fired
				# self_damage recoil is applied once to the actor, not per AoE
				# target, so the per-target rows never carry it - emit the wrap-up
				# row to surface it (else an AoE+self_damage move's recoil is silent).
				has_variable = pod_combat_last_self_damage
				# Energy bonuses apply once at §9b, also after the per-target loop.
				has_variable = pod_combat_last_energy_gain
				has_variable = pod_combat_last_energy_drain
				# Same for the §14 AP refund - without this an AoE move's refund
				# would land after the last per-target row and never be shown.
				has_variable = pod_combat_last_ap_refund
				# The §14 AP drain lands there too. Inert while every drain move is
				# single-enemy (those never set the tripwire), so this only matters
				# the day one is authored as AoE.
				has_variable = pod_combat_last_ap_drain
			}
		}
		pod_combat_push_log_effect = { MOVE_STORY = $STORY$ }
	}

	# ── 17. Bonus teardown ───────────────────────────────────────
	# Consumed by the bonus phases above; clear so it never leaks into a nested
	# reaction execution or the next move (remove on unset = safe no-op).
	# Revert the §5d hit bonus first: it mutated the live workspace mirror,
	# which only rebuilds on a recalc, so without this it kept boosting every
	# later attack this turn. The recalc restores the exact tier+stealth value.
	if = {
		limit = {
			has_variable = pod_combat_bonus_active
			has_variable = pod_combat_current_hit_chance
			$STORY$ = { var:bonus_effect = flag:hit }
		}
		pod_combat_recalc_hit_actor_effect = yes
	}
	remove_variable = pod_combat_bonus_active
	remove_variable = pod_combat_bonus_damage
	remove_variable = pod_combat_bonus_aggravated
	pod_combat_clear_actor_weapon_profile_effect = yes
	# Belt-and-braces: the force-agg and weapon-attack markers are removed at
	# the dispatch site; this covers any future early-exit path. The is_alive
	# guard matters in a fatality fight - a §15 counter can kill the actor
	# synchronously, and remove_variable on a dead char ref errors
	# (a dead actor's stray marker dies with the character anyway).
	var:pod_combat_actor_char ?= {
		if = {
			limit = { is_alive = yes }
			remove_variable = pod_combat_force_agg
			remove_variable = pod_combat_weapon_attack
			remove_variable = pod_combat_weapon_soak_pen
		}
	}
}


######################################################
###### HP DAMAGE DISPATCH (single target)
######################################################
# Two-stage dispatch:
#   1. pod_combat_apply_single_hp_damage_effect - maps the move's damage
#      class (var:damage_type) to a PoD death_reason suffix (REASON) and
#      handles the change_type bonus, then forwards.
#   2. pod_combat_dispatch_damage_pipeline_effect - runs the unified
#      pod_combat_resolve_contact_effect (capability-flag-driven) and
#      fires pod_combat_apply_damage_payload_effect on hit.

pod_combat_apply_single_hp_damage_effect = {
	# $STORY$ = move story
	# $TARGET_SIDE$ = "target" (the workspace side that takes the damage)
	#
	# Single-axis dispatch on the move's damage classification (see
	# POD_combat_move_set_damage_class): an element rides its REASON into
	# PoD core so immunities / downgrades / splat multipliers apply per
	# target; `aggravated` forces duel_aggravated; superficial = duel.
	# The switch's kind table is mirrored by the change_class and weapon-retype
	# switches just below it, the retaliation dispatcher
	# (pod_combat_deal_typed_retal_damage_effect) and the DoT tick switch
	# (pod_combat_deal_dot_tick_from_host_effect) - keep the five in step.
	#
	# The change_class bonus (move story bonus_param) converts a superficial
	# move's damage to a named element for one action when its condition held.
	# It outranks the weapon retype and, like it, declines against a victim
	# immune to the converted element rather than dealing nothing.
	#
	# The change_type bonus (host pod_combat_bonus_aggravated):
	#   - on a superficial move it upgrades the REASON to duel_aggravated;
	#   - on an elemental move the element must survive (immunity checks
	#     key off it), so the ACTOR gets a transient pod_combat_force_agg
	#     marker the core classifier reads via scope:wound_attacker (the
	#     devpod delta in POD_health_track_triggers.txt) and the elemental
	#     REASON rides through, force-classified aggravated.
	#
	# A uses_weapon move adds a second, actor-side source of classification:
	# the equipped weapon's retype (host pod_combat_weapon_class, seeded in
	# §2c) and the wielded-artifact aggravation list. The retype is only ever
	# seeded over a superficial move, so it cannot collide with the branches
	# above.
	#
	# Save the move as scope:current_move so the resolver (and the whole dispatch
	# chain below it) reads capability flags without re-threading $STORY$.
	$STORY$ = { save_scope_as = current_move }

	# An ARMED uses_weapon move marks the actor for the duration of this one
	# dispatch, which is what lets POD_applies_superficial_as_aggravated_damage's
	# wielded-artifact list promote the wound. Recoil, counters and DoT ticks
	# reach the same classifier and deliberately do NOT carry the marker, so a
	# bane sword no longer aggravates everything its bearer does.
	# The armed test matters because that list is the loose one: its arms scan
	# any equipped artifact with no slot or can_benefit filter (see
	# pod_combat_weapon_triggers.txt), so without it a bare-handed swing would
	# still be aggravated by a bane trinket the bearer cannot even wield.
	# The swing test runs inside scope:combat_host because the latch the
	# conditional arm reads is a host var and this effect runs per victim.
	if = {
		limit = {
			scope:combat_host = { pod_combat_move_swings_weapon_trigger = { STORY = scope:current_move } }
			scope:combat_host.var:pod_combat_actor_char ?= { pod_combat_actor_is_armed_trigger = yes }
		}
		scope:combat_host.var:pod_combat_actor_char ?= { set_variable = pod_combat_weapon_attack }
		# The soak-shred magnitude rides the SAME window as the marker, and for
		# the same reason: PoD's soak roll runs inside the wound call and can
		# only see the attacker, so the number has to sit on the actor. Riding
		# the marker's lifetime exactly is load-bearing - a counter's victim IS
		# this actor, so a leaked penalty would shred their own armour.
		if = {
			limit = { scope:combat_host = { has_variable = pod_combat_weapon_soak_penalty } }
			scope:combat_host.var:pod_combat_actor_char ?= {
				set_variable = { name = pod_combat_weapon_soak_pen value = scope:combat_host.var:pod_combat_weapon_soak_penalty }
			}
		}
	}

	if = {
		limit = {
			OR = {
				$STORY$ = { has_variable = damage_type var:damage_type = flag:aggravated }
				AND = {
					scope:combat_host = { has_variable = pod_combat_bonus_aggravated }
					$STORY$ = { NOT = { pod_combat_move_class_is_elemental_trigger = yes } }
					# A retyped move is elemental in all but name: its element has
					# to ride the REASON so immunities still key off it.
					scope:combat_host = { NOT = { has_variable = pod_combat_weapon_class } }
				}
			}
		}
		pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = duel_aggravated }
	}
	else = {
		if = {
			limit = {
				scope:combat_host = { has_variable = pod_combat_bonus_aggravated }
				OR = {
					$STORY$ = { pod_combat_move_class_is_elemental_trigger = yes }
					scope:combat_host = { has_variable = pod_combat_weapon_class }
				}
			}
			scope:combat_host.var:pod_combat_actor_char ?= { set_variable = pod_combat_force_agg }
		}
		# change_class: the fired condition converts this action's damage to the
		# element the move names in bonus_param. Read in place off the move
		# story - copying a flag into a host var degrades it to a plain value.
		# Ordered above the weapon retype because a granted element is this
		# action's identity; the generator also forbids the pairing, so this
		# ordering is only the runtime tiebreak if invalid data slips through.
		# Per victim, same waste rule as the retype: a conversion the target is
		# IMMUNE to falls through to the authored (superficial) switch rather
		# than dealing nothing, so the bonus never makes the move worse.
		if = {
			limit = {
				scope:combat_host = { has_variable = pod_combat_bonus_active }
				scope:current_move = {
					# Guarded chain, not a bare AND: boolean blocks do not
					# short-circuit and almost no move declares a bonus.
					trigger_if = {
						limit = { has_variable = bonus_effect  has_variable = bonus_param }
						var:bonus_effect = flag:change_class
						NOT = {
							pod_combat_stamped_type_is_wasted_trigger = {
								KIND_VAR = bonus_param
								VICTIM   = scope:combat_host.var:pod_combat_$TARGET_SIDE$_char
							}
						}
					}
					trigger_else = { always = no }
				}
			}
			switch = {
				trigger = scope:current_move.var:bonus_param
				flag:fire      = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = fire } }
				flag:cold      = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = cold } }
				flag:lightning = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = lightning } }
				flag:sun       = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = sun } }
				flag:silver    = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = silver } }
				flag:gold      = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = gold } }
				flag:holy      = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = true_faith } }
				flag:balefire  = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = balefire } }
				flag:poison    = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = poison } }
				flag:cold_iron = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = cold_iron } }
				fallback = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = duel } }
			}
		}
		# The weapon retype only ever holds a class when the move's own class is
		# superficial (seeded in §2c), so this never fights an authored element.
		# Evaluated per victim: a retype the target is IMMUNE to would deal
		# nothing at all, so it falls back to the move's own damage rather than
		# leaving the bearer worse off than unarmed.
		else_if = {
			limit = {
				scope:combat_host = {
					has_variable = pod_combat_weapon_class
					NOT = {
						pod_combat_weapon_retype_is_wasted_trigger = { VICTIM = var:pod_combat_$TARGET_SIDE$_char }
					}
				}
			}
			switch = {
				trigger = scope:combat_host.var:pod_combat_weapon_class
				flag:fire      = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = fire } }
				flag:cold      = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = cold } }
				flag:lightning = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = lightning } }
				flag:sun       = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = sun } }
				flag:silver    = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = silver } }
				flag:gold      = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = gold } }
				flag:holy      = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = true_faith } }
				flag:balefire  = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = balefire } }
				flag:poison    = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = poison } }
				flag:cold_iron = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = cold_iron } }
				fallback = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = duel } }
			}
		}
		else = {
			switch = {
				trigger = scope:current_move.var:damage_type
				flag:fire      = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = fire } }
				flag:cold      = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = cold } }
				flag:lightning = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = lightning } }
				flag:sun       = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = sun } }
				flag:silver    = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = silver } }
				flag:gold      = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = gold } }
				# holy keys PoD's aggravated promotion against true_faith.
				flag:holy      = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = true_faith } }
				flag:balefire  = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = balefire } }
				flag:poison    = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = poison } }
				flag:cold_iron = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = cold_iron } }
				# superficial (the hp default) or unset -> generic physical.
				fallback = { pod_combat_dispatch_damage_pipeline_effect = { STORY = $STORY$ TARGET_SIDE = $TARGET_SIDE$ REASON = duel } }
			}
		}
		# Classification is synchronous inside the wound call, so the
		# marker can drop immediately.
		scope:combat_host.var:pod_combat_actor_char ?= { remove_variable = pod_combat_force_agg }
	}
	# Same reasoning for the weapon marker, but it is set before the
	# aggravated branch so it has to drop outside the if/else.
	scope:combat_host.var:pod_combat_actor_char ?= {
		remove_variable = pod_combat_weapon_attack
		remove_variable = pod_combat_weapon_soak_pen
	}
}

pod_combat_dispatch_damage_pipeline_effect = {
	# $STORY$ = move story
	# $TARGET_SIDE$ = "actor" or "target"
	# $REASON$ = PoD death_reason suffix (passed through to the payload)
	# Runs the resolver, fires the payload on hit, consumes actor stealth.
	# Relies on scope:current_move (saved in pod_combat_apply_single_hp_damage_effect).
	pod_combat_resolve_contact_effect = {
		TARGET_SIDE = $TARGET_SIDE$
		ACTOR_SIDE  = actor
	}
	if = {
		limit = { scope:contact_result = 0 }
		save_scope_value_as = {
			name = effective_hp_damage
			value = {
				value = $STORY$.var:hp_damage
				if = {
					limit = { scope:combat_host = { has_variable = pod_combat_bonus_damage } }
					add = scope:combat_host.var:pod_combat_bonus_damage
				}
				# The equipped weapon's own bite, on every target a uses_weapon
				# move reaches (single and AoE both funnel through here).
				if = {
					limit = { scope:combat_host = { has_variable = pod_combat_weapon_bonus_damage } }
					add = scope:combat_host.var:pod_combat_weapon_bonus_damage
				}
			}
		}
		pod_combat_apply_damage_payload_effect = {
			TARGET_SIDE = $TARGET_SIDE$
			AMOUNT      = scope:effective_hp_damage
			REASON      = $REASON$
		}
	}
	# Consumed regardless of contact outcome - committing the action breaks
	# stealth even on a miss. The roll-based team_enemy HP AoE sets a host
	# tripwire so the per-target consume is deferred to ONE post-loop consume,
	# letting every swept target keep the stealth accuracy; single-target moves
	# (flag unset) consume here exactly as before.
	if = {
		limit = { scope:combat_host = { NOT = { has_variable = pod_combat_aoe_suppress_stealth_consume } } }
		pod_combat_consume_actor_stealth_effect = { ACTOR_SIDE = actor }
	}
}


######################################################
###### AOE DISPATCH (team_enemy + team_ally)
######################################################
# Apply a move across a whole team. enemy/ally resolves RELATIVE to the actor
# (slot combat_team vs actor's) via the two shared filter triggers
# (pod_combat_aoe_team_{enemy,ally}_filter_trigger - see their header). Damage
# dispatchers also run pod_combat_aoe_check_target_alive_effect after each
# save_target (KO clauses centralised there).
#
# ── Scope handling ──────────────────────────────────────────────
# every_in_list switches the body to the iterated slot story, so each dispatcher
# opens `save_scope_as = host` to keep host vars reachable; `scope:host = { }`
# re-enters host for workspace/character helpers. Two names for the host char:
# scope:combat_host (saved at §0, long-lived for nested damage routing) vs
# scope:host (iterator-local, survives the scope switch even when combat_host
# isn't the outer scope - e.g. push_log reached mid-switch).
#
# Damage/heal AoEs use the workspace round-trip; buff/debuff AoEs write slot
# vars directly (no roll/death check). Rationale: COMBAT_REFERENCE.info §7.


# One log row per AoE target. Bumps aoe_iter_burst_index (index 0 = first row
# of the move's burst, which gates GUI buff-rider visibility so move-level
# tier/stance lines render once, not per row), pushes the row, clears transient.
#
# Caller contract: from an every_in_list = pod_combat_active_slots body that has
# already done `save_scope_as = host`. Param STORY -> forwarded to push_log.
pod_combat_aoe_push_per_target_log_effect = {
	scope:host = {
		if = {
			limit = { has_variable = pod_combat_aoe_phase_pushed_rows }
			change_variable = { name = pod_combat_aoe_iter_burst_index add = 1 }
		}
		else = {
			set_variable = { name = pod_combat_aoe_iter_burst_index value = 0 }
		}
		set_variable = { name = pod_combat_aoe_phase_pushed_rows value = 1 }
		pod_combat_push_log_effect = { MOVE_STORY = $STORY$ }
		pod_combat_clear_per_target_state_effect = yes
	}
}


pod_combat_apply_team_enemy_hp_damage_effect = {
	# $STORY$ = move story
	# Pushes one log row per target so the log reflects each enemy's outcome,
	# not just the last. Canonical AoE per-iteration sequence (other AoE
	# dispatchers reference this one):
	#   1. save_target_to_story - persist per-target workspace state to the slot.
	#   2. aoe_check_target_alive - flip slot_alive on KO / morale-break.
	#   3. (inside push helper) set pod_combat_aoe_phase_pushed_rows so §16 skips
	#      its wrap-up row unless actor-side state fired.
	#   4. push_log - the per-target row.
	#   5. clear per-target last_* (keeps the pushed-rows flag).
	save_scope_as = host
	# Burst-level stealth. Suppress the per-target consume during the loop so
	# EVERY target's hit recalc keeps the stealth bonus (not just the first swept),
	# then consume ONCE after. Read by pod_combat_dispatch_damage_pipeline_effect.
	scope:combat_host = { set_variable = pod_combat_aoe_suppress_stealth_consume }
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			pod_combat_slot_is_alive_trigger = yes
			pod_combat_aoe_team_enemy_filter_trigger = yes
		}
		pod_combat_load_target_from_story_effect = yes
		scope:host = {
			pod_combat_apply_single_hp_damage_effect = { STORY = $STORY$ TARGET_SIDE = target }
			# Drain self-heal survivor: the per-target push helper clears the host
			# last_* result pair, so the §9 on_hp_damage rider can't see it after the
			# loop. Latch a surviving flag (excluded from the per-target clear) so the
			# actor heals once if ANY enemy took HP damage. Reuses the rider's own
			# predicate to stay in lockstep. Set BEFORE the push helper clears.
			if = {
				limit = { pod_combat_result_code_indicates_hp_damage_trigger = yes }
				set_variable = pod_combat_aoe_any_hp_damage
			}
		}
		pod_combat_save_target_to_story_effect = yes
		pod_combat_aoe_check_target_alive_effect = yes
		pod_combat_aoe_push_per_target_log_effect = { STORY = $STORY$ }
	}
	# End the suppress window, then consume the actor's stealth ONCE for the
	# whole burst (idempotent breaker; no-op if unstealthed). Persisted because the
	# target-only reload below leaves the live actor workspace intact. Consume
	# BEFORE the reload so its recalc reflects the stealth-off actor hit chance.
	scope:combat_host = { remove_variable = pod_combat_aoe_suppress_stealth_consume }
	pod_combat_consume_actor_stealth_effect = { ACTOR_SIDE = actor }
	scope:host = { pod_combat_reload_selected_target_effect = yes }
}

pod_combat_apply_team_all_hp_damage_effect = {
	# $STORY$ = move story
	# The team_enemy sequence (see pod_combat_apply_team_enemy_hp_damage_effect for
	# the canonical per-iteration pattern and the burst-stealth window) run over
	# both teams, so the actor's own side takes the hit too. A KO mid-sweep does
	# not stop the loop - victory is only scored once the whole move completes.
	#
	# The CASTER is hit as well, but after the sweep and through the actor
	# workspace rather than from inside the loop. Sweeping its own slot would put
	# the same combatant in both workspace sides, and the end-of-move actor flush
	# rewrites every mirrored slot field from the actor workspace - so the loop's
	# slot-side writes for the caster would be silently reverted (morale most
	# visibly; HP survives only because it lives on PoD's wound track, not the
	# slot). Routing it through the actor side keeps one authoritative copy.
	save_scope_as = host
	scope:combat_host = { set_variable = pod_combat_aoe_suppress_stealth_consume }
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			pod_combat_slot_is_alive_trigger = yes
			pod_combat_aoe_team_all_sweep_filter_trigger = yes
		}
		pod_combat_load_target_from_story_effect = yes
		scope:host = {
			pod_combat_apply_single_hp_damage_effect = { STORY = $STORY$ TARGET_SIDE = target }
			if = {
				limit = { pod_combat_result_code_indicates_hp_damage_trigger = yes }
				set_variable = pod_combat_aoe_any_hp_damage
			}
		}
		pod_combat_save_target_to_story_effect = yes
		pod_combat_aoe_check_target_alive_effect = yes
		pod_combat_aoe_push_per_target_log_effect = { STORY = $STORY$ }
	}
	# The caster's own share. Bypass-guard and no hit roll - you neither block
	# nor miss yourself - but PoD still soaks and classifies it, so a caster
	# immune to its own element shrugs the blast off. The kind seeder is the
	# literal-flag path (a var-to-var copy would degrade the flag), and it always
	# writes, defaulting to flag:duel, so the switch never reads an unset var.
	pod_combat_seed_kind_from_class_effect = { STORY = $STORY$ KIND_VAR = pod_combat_team_all_self_kind }
	# Same swing, same instant: a weapon retype covers the caster's own share too.
	pod_combat_override_kind_from_weapon_effect = { KIND_VAR = pod_combat_team_all_self_kind }
	# As does a granted change_class conversion, which outranks it. No waste
	# guard here, deliberately, matching the retype above: a caster immune to
	# its own blast shrugging it off is a benefit, not a wasted conversion.
	pod_combat_override_kind_from_bonus_class_effect = { STORY = $STORY$ KIND_VAR = pod_combat_team_all_self_kind }
	pod_combat_deal_typed_retal_damage_effect = {
		TARGET_SIDE = actor
		AMOUNT      = $STORY$.var:hp_damage
		KIND_VAR    = pod_combat_team_all_self_kind
	}
	remove_variable = pod_combat_team_all_self_kind
	# Surface it through the existing recoil rider, which is what makes the §16
	# wrap-up row fire for an AoE. Additive: a team_all move may also carry a
	# self_damage rider, and the two would otherwise overwrite each other.
	if = {
		limit = { scope:damage_dealt > 0 }
		if = {
			limit = { has_variable = pod_combat_last_self_damage }
			change_variable = { name = pod_combat_last_self_damage add = scope:damage_dealt }
		}
		else = {
			set_variable = { name = pod_combat_last_self_damage value = scope:damage_dealt }
		}
	}
	scope:combat_host = { remove_variable = pod_combat_aoe_suppress_stealth_consume }
	pod_combat_consume_actor_stealth_effect = { ACTOR_SIDE = actor }
	scope:host = { pod_combat_reload_selected_target_effect = yes }
}

# Arm one retaliation stance on the actor's workspace. $STORY$ = move story;
# $WHEN$ = hit | miss | any (the pod_combat_actor_retaliate_<WHEN>* var family).
# Take-strongest compares within one family only, so an `any` charge never
# displaces - nor is displaced by - an armed hit/miss charge.
# Amount fallback: retaliate_damage -> hp_damage -> 1, staged into
# pod_combat_retal_arm_candidate so the take-strongest test and the write agree
# by construction, then boosted by a fired retaliate_boost bonus. Kind
# precedence: morale > retaliate_agg > the move's damage class element >
# a granted change_class element > the live weapon retype > plain duel.
# The extra _morale presence flag feeds the stance tooltip (a bare
# flag-compare on the slot-story custom-loc root never matches, so the
# dispatcher reads this).
pod_combat_arm_retaliation_effect = {
	# Stage the candidate amount ONCE, then compare and write from it. The
	# fallback used to be spelled out twice (in the take-strongest test and
	# again in the writes); staging is what keeps the retaliate_boost term
	# below in both, so a boosted charge cannot lose a comparison it should win.
	if = {
		limit = { $STORY$ = { has_variable = retaliate_damage } }
		set_variable = { name = pod_combat_retal_arm_candidate value = $STORY$.var:retaliate_damage }
	}
	else_if = {
		limit = { $STORY$ = { has_variable = hp_damage } }
		set_variable = { name = pod_combat_retal_arm_candidate value = $STORY$.var:hp_damage }
	}
	else = {
		# Covers a move arming with neither var: without this the workspace var
		# stays unset and the §15 has_variable gate silently swallows the reaction.
		set_variable = { name = pod_combat_retal_arm_candidate value = 1 }
	}
	# retaliate_boost: the fired condition makes the counter this move arms hit
	# harder. Folded in before the comparison, so it is the boosted number that
	# contends for the slot. A move arming several charges boosts each of them
	# (this effect restages per WHEN).
	if = {
		limit = {
			has_variable = pod_combat_bonus_active
			$STORY$ = { var:bonus_effect = flag:retaliate_boost }
		}
		change_variable = { name = pod_combat_retal_arm_candidate add = $STORY$.var:bonus_value }
	}
	# Take-strongest: arm only when unarmed, or the incoming counter is strictly
	# stronger by amount. Equal or weaker keeps the incumbent, so the
	# payload-redundancy gate can admit a move whose retaliation rider is
	# redundant without DOWNGRADING an armed stronger counter. Kind is not
	# compared (amount-only).
	if = {
		limit = {
			trigger_if = {
				limit = { has_variable = pod_combat_actor_retaliate_$WHEN$ }
				var:pod_combat_actor_retaliate_$WHEN$ < var:pod_combat_retal_arm_candidate
			}
			trigger_else = { always = yes }
		}
		# On replace, clear stale kind fields first: the kind chain below has no
		# final else, so a morale/agg counter replaced by a plain one would keep
		# the stale _kind / _morale presence flags.
		remove_variable = pod_combat_actor_retaliate_$WHEN$_kind
		remove_variable = pod_combat_actor_retaliate_$WHEN$_kind_from_weapon
		remove_variable = pod_combat_actor_retaliate_$WHEN$_morale
		set_variable = { name = pod_combat_actor_retaliate_$WHEN$ value = var:pod_combat_retal_arm_candidate }
		if = {
			limit = { $STORY$ = { has_variable = retaliate_morale } }
			set_variable = { name = pod_combat_actor_retaliate_$WHEN$_kind value = flag:morale }
			set_variable = { name = pod_combat_actor_retaliate_$WHEN$_morale value = yes }
		}
		else_if = {
			limit = { $STORY$ = { has_variable = retaliate_agg } }
			set_variable = { name = pod_combat_actor_retaliate_$WHEN$_kind value = flag:aggravated }
		}
		else_if = {
			limit = { $STORY$ = { pod_combat_move_class_is_elemental_trigger = yes } }
			# Literal-flags-only seeding
			pod_combat_seed_kind_from_class_effect = { STORY = $STORY$ KIND_VAR = pod_combat_actor_retaliate_$WHEN$_kind }
		}
		else_if = {
			# A change_class conversion the condition granted carries into the
			# counter it armed. Ranks below an authored class (which the
			# generator anyway forbids pairing with it) and above the weapon.
			limit = {
				has_variable = pod_combat_bonus_active
				$STORY$ = { var:bonus_effect = flag:change_class }
			}
			pod_combat_override_kind_from_bonus_class_effect = { STORY = $STORY$ KIND_VAR = pod_combat_actor_retaliate_$WHEN$_kind }
			# Same provenance marker as the weapon arm below: the name is
			# historical, what it means downstream is "transient-source kind, so
			# stand down against an immune victim rather than counter for zero".
			set_variable = { name = pod_combat_actor_retaliate_$WHEN$_kind_from_weapon value = yes }
		}
		else_if = {
			# A counter armed while holding a typed weapon strikes back with it.
			# Ranks below an authored class, which is the move's own identity.
			limit = { scope:combat_host ?= { has_variable = pod_combat_weapon_class } }
			pod_combat_override_kind_from_weapon_effect = { KIND_VAR = pod_combat_actor_retaliate_$WHEN$_kind }
			# Provenance, for the fire-time waste guard: only a WEAPON-derived kind
			# stands down against an immune victim. An authored elemental counter
			# keeps its identity and may legitimately read immune. The victim is
			# unknowable until the counter fires, so the test cannot run here; this
			# marker rides the slot round-trip with the other stance fields.
			set_variable = { name = pod_combat_actor_retaliate_$WHEN$_kind_from_weapon value = yes }
		}
	}
	remove_variable = pod_combat_retal_arm_candidate
}

######################################################
###### AOE BUFF DISPATCH (team_ally / team_enemy)
######################################################
# Broadcast tier buffs / stance flags across a team. Unlike damage/heal AoE,
# these skip the workspace - they write per-slot tier/flag vars + (for soak) the
# character bridge var directly, so all teammates carry the buff. Slot soak/hit
# display lags until the ally/enemy next loads workspace, but the soak bridge var
# is live immediately so mid-combat damage rolls pick it up.
# Also hosts two shared write helpers used well beyond AoE:
# pod_combat_seed_dot_on_slot_effect (§13b DoT seeding) and
# pod_combat_upgrade_tier_var_effect (the canonical upgrade-only tier write).

# Seed/refresh a DoT on the CURRENT slot story (re-apply overwrites amount /
# rounds / kind - no stacking). Kind precedence: dot_morale > the move's
# damage class > plain duel (superficial). Also stages the dot_applied_*
# log scopes - the CALLER pushes its own application row (the class-based
# dispatch in §13b, the generic row in §13b-bonus).
# That dispatch must read ONLY $STORY$ vars: `this` slot vars are unreliable
# after the victim save_scope_as switch here (iteration-scope corruption).
# $AMOUNT$ / $ROUNDS$ = per-tick amount / tick count.
pod_combat_seed_dot_on_slot_effect = {
	set_variable = { name = slot_dot_amount value = $AMOUNT$ }
	set_variable = { name = slot_dot_rounds value = $ROUNDS$ }
	# Store the applying character so the round-end tick credits the burn
	# to its caster (killer attribution / attacker-keyed aggravated), not the PC
	# host. Overwrites on re-apply like the other slot_dot_* vars (no stacking).
	set_variable = { name = slot_dot_source value = $CASTER$ }
	# Saved ahead of the kind seeding (rather than beside the other
	# dot_applied_* scopes below) so the waste guard can hand the victim to a
	# host-scope trigger.
	var:slot_char = { save_scope_as = dot_applied_victim }
	# Literal-flags-only seeding: the kind IS the damage class (aggravated
	# ticks as duel_aggravated; superficial/absent falls to flag:duel).
	pod_combat_seed_kind_from_class_effect = { STORY = $STORY$ KIND_VAR = slot_dot_kind }
	# A weapon retype rides onto the burn it lit, same as the direct hit -
	# including its waste guard. A burn this victim is immune to ticks zero for
	# its whole duration, which is strictly worse than the authored kind the
	# retype replaced. Unlike the counter path the victim is known here (the
	# seed runs in the actor's own dispatch), so the LIVE profile trigger is the
	# right one.
	if = {
		limit = {
			scope:combat_host ?= {
				NOT = { pod_combat_weapon_retype_is_wasted_trigger = { VICTIM = scope:dot_applied_victim } }
			}
		}
		pod_combat_override_kind_from_weapon_effect = { KIND_VAR = slot_dot_kind }
	}
	# A fired change_class conversion outranks the weapon retype (it is the
	# move's granted identity for this action), so it lands after it. Same
	# waste rule: a burn the victim is immune to would tick zero for its whole
	# duration. Invoked in $STORY$ scope because bonus_param lives on the move.
	if = {
		limit = {
			$STORY$ = {
				NOT = { pod_combat_stamped_type_is_wasted_trigger = { KIND_VAR = bonus_param VICTIM = scope:dot_applied_victim } }
			}
		}
		pod_combat_override_kind_from_bonus_class_effect = { STORY = $STORY$ KIND_VAR = slot_dot_kind }
	}
	# Morale wins over the damage class (matches the counter-damage precedence).
	# The morale_dot bonus forces it the same way the static flag does; a
	# $STORY$ read is the one channel this effect's contract allows here.
	if = {
		limit = {
			$STORY$ = {
				trigger_if = { limit = { has_variable = dot_morale }  always = yes }
				trigger_else_if = {
					limit = { has_variable = bonus_effect }
					var:bonus_effect = flag:morale_dot
				}
				trigger_else = { always = no }
			}
		}
		set_variable = { name = slot_dot_kind value = flag:morale }
	}
	save_scope_value_as = { name = dot_applied_amount value = $AMOUNT$ }
	save_scope_value_as = { name = dot_applied_rounds value = $ROUNDS$ }
}

# CANONICAL upgrade-only tier write, shared by §5c's workspace writes, both
# team appliers' slot writes, and every char-bridge mirror: fires only on
# first application or a strictly-higher tier, so a low-tier move can't
# silently DOWNGRADE a stacked tier-3 effect. Selection filters already
# reject downgrades; this guards bypass paths (console, test actions).
# $STORY$ = move story; $SRC$ = the move's tier var; $DST$ = the tier var on
# the CURRENT scope (host workspace, slot story, or character mirror).
pod_combat_upgrade_tier_var_effect = {
	if = {
		limit = {
			$STORY$ = { has_variable = $SRC$ }
			OR = {
				NOT = { has_variable = $DST$ }
				var:$DST$ < $STORY$.var:$SRC$
			}
		}
		set_variable = { name = $DST$ value = $STORY$.var:$SRC$ }
	}
}

# Apply every move-authored BUFF tier to one loaded workspace side. This keeps

		limit = { has_variable = pod_combat_$SIDE$_char }

			pod_combat_upgrade_tier_var_effect = { STORY = $STORY$ SRC = soak_buff_round_tier DST = pod_combat_tier_soak_buff_round }

			$STORY$ = {
				OR = {
					has_variable = soak_buff_round_tier
					has_variable = soak_buff_combat_tier
				}
			}
		}

				}
			}
		}

	}
}

		}
		# Tier hit buffs: slot var only (hit chance is workspace-local; soak
		# tiers go to the char bridge below).
		pod_combat_upgrade_tier_var_effect = { STORY = $STORY$ SRC = hit_buff_round_tier DST = slot_hit_buff_round_tier }
		pod_combat_upgrade_tier_var_effect = { STORY = $STORY$ SRC = hit_buff_combat_tier DST = slot_hit_buff_combat_tier }
		# Stance flags broadcast to each ally.
		if = {
			limit = { $STORY$ = { has_variable = set_guard } }
			set_variable = { name = slot_is_guarding  value = yes }
		}
		if = {
			limit = { $STORY$ = { has_variable = dodge } }
			set_variable = { name = slot_is_dodging   value = yes }
		}
		if = {
			limit = { $STORY$ = { has_variable = stealth } }
			# Idempotent slot-stealth applicator (flag + +30 hit_chance).
			pod_combat_apply_stealth_slot_effect = yes
		}
		# Char-scope soak tiers LAST - a single scope switch after all the
		# slot-local writes above, so no inline slot read/write ever follows a switch
		# (the every_in_list iteration-scope-corruption trap). The upgrade guard
		# matters per-ally at write time: the selection gate only checks the
		# actor's tier, not each iterated ally's. POD_soak_chance reads these.
		var:slot_char = {
			pod_combat_upgrade_tier_var_effect = { STORY = $STORY$ SRC = soak_buff_round_tier DST = pod_combat_tier_soak_buff_round }
			pod_combat_upgrade_tier_var_effect = { STORY = $STORY$ SRC = soak_buff_combat_tier DST = pod_combat_tier_soak_buff_combat }
		}
	}
}

pod_combat_apply_team_enemy_debuffs_effect = {
	# $STORY$ = move story
	# Mirrors pod_combat_apply_team_ally_buffs_effect but for the enemy-target
	# side: debuff tier vars + bypass/remove flag clearances.
	save_scope_as = host
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			pod_combat_slot_is_alive_trigger = yes
			pod_combat_aoe_team_enemy_filter_trigger = yes
		}
		# Upgrade-only - see pod_combat_apply_team_ally_buffs_effect.
		pod_combat_upgrade_tier_var_effect = { STORY = $STORY$ SRC = hit_debuff_round_tier DST = slot_hit_debuff_round_tier }
		pod_combat_upgrade_tier_var_effect = { STORY = $STORY$ SRC = hit_debuff_combat_tier DST = slot_hit_debuff_combat_tier }
		# Bypass / remove flags - strip the target's stances.
		if = {
			limit = { $STORY$ = { has_variable = bypass_dodge } }
			remove_variable = slot_is_dodging
		}
		if = {
			limit = { $STORY$ = { has_variable = bypass_stealth } }
			# Idempotent slot-stealth teardown - drops slot_is_stealthed and
			# recalcs so the +30 (folded into the recalc in
			# pod_combat_tier_effects.txt) drops with the flag.
			pod_combat_break_stealth_slot_effect = yes
		}
		if = {
			limit = { $STORY$ = { has_variable = remove_guard } }
			remove_variable = slot_is_guarding
		}
		# Char-scope soak debuffs LAST (see pod_combat_apply_team_ally_buffs_effect)
		# - one scope switch after all slot-local writes.
		var:slot_char = {
			pod_combat_upgrade_tier_var_effect = { STORY = $STORY$ SRC = soak_debuff_round_tier DST = pod_combat_tier_soak_debuff_round }
			pod_combat_upgrade_tier_var_effect = { STORY = $STORY$ SRC = soak_debuff_combat_tier DST = pod_combat_tier_soak_debuff_combat }
		}
	}
}

# team_all only: the caster's share of a broadcast debuff, applied to the actor
# workspace after the sweep (same routing rationale as the HP applier).
# Tiers only. bypass_dodge / bypass_stealth describe how the ATTACK meets each
# victim's defenses, not a debuff the blast inflicts, so self-applying them is
# meaningless - and breaking the caster's own stealth here, in the pre-damage
# phase, would also rob the burst of the stealth accuracy it is entitled to.
pod_combat_apply_team_all_self_debuffs_effect = {
	pod_combat_apply_workspace_debuff_tiers_effect = { STORY = $STORY$ SIDE = actor }
}
