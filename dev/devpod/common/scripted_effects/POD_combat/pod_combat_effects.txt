# ─── INITIALIZATION ────────────────────────────────────────────────

pod_combat_initialize_effect = {
	# Thin wrapper for simple 1v1 combat — PC vs a single target, no allies
	# or knights. Delegates to pod_combat_initialize_group_effect with MODE=0,
	# which populates slot 0 (PC) and slot 5 (opponent) and leaves the rest
	# empty. The unified initiative + workspace flow handles it from there.
	# $OPPONENT$ = the target character
	pod_combat_initialize_group_effect = { OPPONENT = $OPPONENT$ MODE = 0 }
}

# ─── TURN MANAGEMENT ──────────────────────────────────────────────

pod_combat_end_player_turn_effect = {
	# Player turn is over. Always process the remaining actors in initiative
	# order via the group turn loop — 1v1 and NvN share the same code path.
	pod_combat_process_group_turns_effect = yes
}

# ─── RESOURCE MANIPULATION ────────────────────────────────────────

pod_combat_spend_ap_effect = {
	# $AMOUNT$ = AP to spend
	change_variable = {
		name = pod_combat_current_ap
		add = {
			value = $AMOUNT$
			multiply = -1
		}
	}
}

pod_combat_check_retaliation_effect = {
	# Called after every damage-dealing move resolves. Checks whether the
	# target (opponent) has a retaliation stance and whether the trigger
	# condition was met:
	#   retaliate_if_hit  — fires when the attack dealt damage (last_damage > 0)
	#   retaliate_if_missed — fires when the attack missed (last_was_missed = yes)
	#
	# Retaliation damage inherits the damage_source and damage_type from
	# the skill that set up the stance:
	#   - Superficial: goes through the actor's soak check (can be soaked)
	#   - Aggravated: soak at half effectiveness (-50%), tracked as aggravated damage
	# The _agg companion variable (1 = aggravated, absent = superficial)
	# is set by the generator alongside the retaliation amount.
	#
	# $TARGET_SIDE$ = "opponent" (the side that has the retaliation stance)
	# $ACTOR_SIDE$  = "player"  (the side that takes retaliation damage)

	# Retaliate on hit
	if = {
		limit = {
			has_variable = pod_combat_$TARGET_SIDE$_retaliate_hit
			has_variable = pod_combat_last_damage
			var:pod_combat_last_damage > 0
		}
		# Soak check — aggravated retaliates at half soak, superficial at full
		save_scope_value_as = {
			name = retaliate_soak
			value = var:pod_combat_$ACTOR_SIDE$_soak
		}
		if = {
			limit = { has_variable = pod_combat_$TARGET_SIDE$_retaliate_hit_agg }
			save_scope_value_as = {
				name = retaliate_soak
				value = { value = var:pod_combat_$ACTOR_SIDE$_soak divide = 2 floor = yes }
			}
		}
		random_list = {
			0 = {
				modifier = { add = scope:retaliate_soak }
				set_variable = { name = pod_combat_last_was_retaliate_soaked value = yes }
			}
			100 = {
				modifier = { add = { value = scope:retaliate_soak multiply = -1 } }
				save_scope_value_as = {
					name = retaliate_hp
					value = {
						value = var:pod_combat_$ACTOR_SIDE$_hp
						subtract = var:pod_combat_$TARGET_SIDE$_retaliate_hit
						min = 0
					}
				}
				set_variable = { name = pod_combat_$ACTOR_SIDE$_hp value = scope:retaliate_hp }
				if = {
					limit = { has_variable = pod_combat_$TARGET_SIDE$_retaliate_hit_agg }
					change_variable = { name = pod_combat_$ACTOR_SIDE$_agg_damage add = var:pod_combat_$TARGET_SIDE$_retaliate_hit }
				}
				set_variable = { name = pod_combat_last_was_retaliated value = yes }
			}
		}
		remove_variable = pod_combat_$TARGET_SIDE$_retaliate_hit
		remove_variable = pod_combat_$TARGET_SIDE$_retaliate_hit_agg
	}

	# Retaliate on miss
	if = {
		limit = {
			has_variable = pod_combat_$TARGET_SIDE$_retaliate_miss
			has_variable = pod_combat_last_was_missed
		}
		# Soak check — aggravated at half, superficial at full
		save_scope_value_as = {
			name = retaliate_soak
			value = var:pod_combat_$ACTOR_SIDE$_soak
		}
		if = {
			limit = { has_variable = pod_combat_$TARGET_SIDE$_retaliate_miss_agg }
			save_scope_value_as = {
				name = retaliate_soak
				value = { value = var:pod_combat_$ACTOR_SIDE$_soak divide = 2 floor = yes }
			}
		}
		random_list = {
			0 = {
				modifier = { add = scope:retaliate_soak }
				set_variable = { name = pod_combat_last_was_retaliate_soaked value = yes }
			}
			100 = {
				modifier = { add = { value = scope:retaliate_soak multiply = -1 } }
				save_scope_value_as = {
					name = retaliate_hp
					value = {
						value = var:pod_combat_$ACTOR_SIDE$_hp
						subtract = var:pod_combat_$TARGET_SIDE$_retaliate_miss
						min = 0
					}
				}
				set_variable = { name = pod_combat_$ACTOR_SIDE$_hp value = scope:retaliate_hp }
				if = {
					limit = { has_variable = pod_combat_$TARGET_SIDE$_retaliate_miss_agg }
					change_variable = { name = pod_combat_$ACTOR_SIDE$_agg_damage add = var:pod_combat_$TARGET_SIDE$_retaliate_miss }
				}
				set_variable = { name = pod_combat_last_was_retaliated value = yes }
			}
		}
		remove_variable = pod_combat_$TARGET_SIDE$_retaliate_miss
		remove_variable = pod_combat_$TARGET_SIDE$_retaliate_miss_agg
	}
}

pod_combat_apply_ap_drain_effect = {
	# Apply any AP drain debuff from the previous round to the current
	# actor. Reads pod_combat_player_ap_drain from the workspace (loaded
	# from pod_combat_c{N}_ap_drain). Clears the drain after application
	# so it only fires once.
	if = {
		limit = { has_variable = pod_combat_player_ap_drain }
		change_variable = {
			name = pod_combat_current_ap
			add = {
				value = var:pod_combat_player_ap_drain
				multiply = -1
			}
		}
		# Clamp to minimum 1 AP so the actor can always do something
		if = {
			limit = { var:pod_combat_current_ap < 1 }
			set_variable = { name = pod_combat_current_ap value = 1 }
		}
		remove_variable = pod_combat_player_ap_drain
	}
}

pod_combat_deal_hp_damage_effect = {
	# $TARGET_SIDE$ = "player" or "opponent" (who takes damage)
	# $AMOUNT$ = damage amount
	# NOTE: pod_combat_last_was_blocked and pod_combat_last_was_soaked must be
	#        cleared by the MOVE effect before calling this, NOT here — because
	#        Reckless Charge calls this twice and the 2nd call would erase the 1st.

	# Check if target is guarding and not feinted
	if = {
		limit = {
			has_variable = pod_combat_$TARGET_SIDE$_is_guarding
			NOT = { has_variable = pod_combat_$TARGET_SIDE$_feinted }
		}
		# Guard blocks the attack — consume the guard. The log-routing
		# step uses pod_combat_last_was_blocked (not a flag) to pick the
		# "Blocked!" message, so we no longer set pod_combat_last_action.
		remove_variable = pod_combat_$TARGET_SIDE$_is_guarding
		set_variable = { name = pod_combat_last_was_blocked value = yes }
		set_variable = { name = pod_combat_last_damage value = 0 }
	}
	else = {
		# ── Soak check ──
		# Read target's soak chance into a scope for the random_list
		save_scope_value_as = {
			name = target_soak
			value = var:pod_combat_$TARGET_SIDE$_soak
		}

		random_list = {
			# Soaked — damage negated
			0 = {
				modifier = { add = scope:target_soak }
				set_variable = { name = pod_combat_last_was_soaked value = yes }
				set_variable = { name = pod_combat_last_damage value = 0 }
			}
			# Not soaked — apply full damage
			100 = {
				modifier = {
					add = {
						value = scope:target_soak
						multiply = -1
					}
				}

				# Apply HP damage (clamped to 0)
				save_scope_value_as = {
					name = new_hp
					value = {
						value = var:pod_combat_$TARGET_SIDE$_hp
						subtract = $AMOUNT$
						min = 0
					}
				}
				set_variable = { name = pod_combat_$TARGET_SIDE$_hp value = scope:new_hp }

				# Apply morale spillover (HP damage * 0.5, floored)
				save_scope_value_as = {
					name = morale_spill
					value = {
						value = $AMOUNT$
						multiply = pod_combat_morale_from_hp_ratio
						floor = yes
					}
				}
				if = {
					limit = { scope:morale_spill > 0 }
					save_scope_value_as = {
						name = new_morale
						value = {
							value = var:pod_combat_$TARGET_SIDE$_morale
							subtract = scope:morale_spill
							min = 0
						}
					}
					set_variable = { name = pod_combat_$TARGET_SIDE$_morale value = scope:new_morale }
				}

				set_variable = { name = pod_combat_last_damage value = $AMOUNT$ }
				# Victory check runs at the end of pod_combat_save_workspace_effect
				# once the slot data is flushed. No in-pipeline check needed here —
				# combat-end detection is handled at the move-complete boundary.
			}
		}
	}
}

# ── Actor stealth consumption ──────────────────────────────────
# Stealth persists until the stealthed character attacks OR is attacked.
# When an attack is attacked, the damage pipeline consumes the target's
# stealth flag via the $TARGET_SIDE$ check at the top of each damage
# effect. When the stealthed character acts (attacks), we consume it
# here — called at the END of each damage effect so the +30 hit bonus
# still applies to this attack, then gets reverted for future actions.
# The actor side is always "player" (workspace convention).
pod_combat_consume_actor_stealth_effect = {
	if = {
		limit = { has_variable = pod_combat_player_is_stealthed }
		remove_variable = pod_combat_player_is_stealthed
		# Revert the +30 hit chance bonus stealth granted
		save_scope_value_as = {
			name = new_hit_after_actor_stealth
			value = { value = var:pod_combat_player_hit_chance subtract = 30 min = 20 max = 95 }
		}
		set_variable = { name = pod_combat_player_hit_chance value = scope:new_hit_after_actor_stealth }
		# Keep current_hit_chance in sync in case the damage pipeline
		# already copied it elsewhere in the same turn.
		set_variable = { name = pod_combat_current_hit_chance value = scope:new_hit_after_actor_stealth }
	}
}

pod_combat_deal_physical_hp_damage_effect = {
	# $TARGET_SIDE$ = "player" or "opponent" (who takes damage)
	# $AMOUNT$ = damage amount
	# Physical/Superficial: spirit gate → stealth → dodge → hit chance → guard → soak → damage

	# ── Spirit-damage gate ──
	# Physical attacks pass harmlessly through an incorporeal spirit target
	# unless the actor has the supernatural means to strike them. Mirrors
	# PoD's POD_damage_spirit_custom_trigger. Stealth and Dodge stances
	# are preserved on a pass-through — there was nothing to evade.
	if = {
		limit = {
			has_variable = pod_combat_$TARGET_SIDE$_incorporeal_spirit
			NOT = { has_variable = pod_combat_player_can_damage_spirit }
		}
		set_variable = { name = pod_combat_last_was_missed value = yes }
		set_variable = { name = pod_combat_last_damage value = 0 }
	}
	# Stealth then dodge — each negates the entire attack; stealth is consumed first so a stealthed dodger keeps their dodge
	else_if = {
		limit = { has_variable = pod_combat_$TARGET_SIDE$_is_stealthed }
		remove_variable = pod_combat_$TARGET_SIDE$_is_stealthed
		# Revert the +30 hit chance bonus stealth granted now that stealth is consumed
		save_scope_value_as = {
			name = new_hit_after_stealth
			value = { value = var:pod_combat_$TARGET_SIDE$_hit_chance subtract = 30 min = 20 max = 95 }
		}
		set_variable = { name = pod_combat_$TARGET_SIDE$_hit_chance value = scope:new_hit_after_stealth }
		set_variable = { name = pod_combat_last_was_dodged value = yes }
		set_variable = { name = pod_combat_last_damage value = 0 }
	}
	else_if = {
		limit = { has_variable = pod_combat_$TARGET_SIDE$_is_dodging }
		remove_variable = pod_combat_$TARGET_SIDE$_is_dodging
		set_variable = { name = pod_combat_last_was_dodged value = yes }
		set_variable = { name = pod_combat_last_damage value = 0 }
	}
	else = {
		save_scope_value_as = {
			name = hit_chance
			value = var:pod_combat_current_hit_chance
		}

		random_list = {
			# Miss — attack didn't connect
			0 = {
				modifier = {
					add = {
						value = 100
						subtract = scope:hit_chance
					}
				}
				set_variable = { name = pod_combat_last_was_missed value = yes }
				set_variable = { name = pod_combat_last_damage value = 0 }
			}
			# Hit — proceed with guard/soak/damage checks
			0 = {
				modifier = { add = scope:hit_chance }
				pod_combat_deal_hp_damage_effect = { TARGET_SIDE = $TARGET_SIDE$ AMOUNT = $AMOUNT$ }
			}
		}
	}
	# Consume the actor's stealth (if any) now that the attack has resolved
	pod_combat_consume_actor_stealth_effect = yes
}

pod_combat_deal_morale_damage_effect = {

		set_variable = { name = pod_combat_last_was_dodged value = yes }
		set_variable = { name = pod_combat_last_damage value = 0 }
	}
	else_if = {
		limit = { has_variable = pod_combat_$TARGET_SIDE$_is_dodging }
		remove_variable = pod_combat_$TARGET_SIDE$_is_dodging
		set_variable = { name = pod_combat_last_was_dodged value = yes }
		set_variable = { name = pod_combat_last_damage value = 0 }
	}
	else = {

		# Victory check runs at the end of pod_combat_save_workspace_effect.
	}
}

pod_combat_restore_morale_effect = {
	# $SIDE$ = "player" or "opponent"
	# $AMOUNT$ = morale to restore
	save_scope_value_as = {
		name = new_morale
		value = {
			value = var:pod_combat_$SIDE$_morale
			add = $AMOUNT$
			max = pod_combat_max_morale
		}
	}
	set_variable = { name = pod_combat_$SIDE$_morale value = scope:new_morale }
	set_variable = { name = pod_combat_last_damage value = $AMOUNT$ }
	set_variable = { name = pod_combat_last_was_morale_heal value = yes }
}

# ─── VICTORY CHECK ────────────────────────────────────────────────
# Victory detection is now handled exclusively by
# pod_combat_check_group_victory_effect, called once at the end of
# pod_combat_save_workspace_effect when slot data is flushed and the
# per-slot _alive flags are accurate. The old in-pipeline
# pod_combat_check_end_effect wrapper has been retired — calling it
# during the damage pipeline read stale slot variables and never
# detected kills, so it was dead code that only burned compute.

# ─── AUTO-RESOLVE ──────────────────────────────────────────────────

pod_combat_auto_resolve_effect = {
	# Save PC scope alias to avoid bare `root` references in set_variable
	save_scope_as = combat_pc

	# Simple prowess comparison with random factor
	save_scope_value_as = {
		name = player_score
		value = {
			value = prowess
			add = {
				integer_range = { min = 0 max = prowess }
			}
		}
	}
	var:pod_combat_opponent = {
		save_scope_value_as = {
			name = opponent_score
			value = {
				value = prowess
				add = {
					integer_range = { min = 0 max = prowess }
				}
			}
		}
	}

	if = {
		limit = { scope:player_score >= scope:opponent_score }
		set_variable = { name = pod_combat_winner value = scope:combat_pc }
		set_variable = { name = pod_combat_loser value = var:pod_combat_opponent }
	}
	else = {
		set_variable = { name = pod_combat_winner value = var:pod_combat_opponent }
		set_variable = { name = pod_combat_loser value = scope:combat_pc }
	}

	# Set final HP/Morale for display
	set_variable = { name = pod_combat_opponent_morale value = 0 }
	set_variable = { name = pod_combat_is_over value = yes }
	# Combat window stays open — player clicks "End Combat" to dismiss
}

# Each entry stores: msg ID (1-57), actor slot (0-9), target slot (0-9, -1 = none), damage, result flag.
#
# Message IDs assigned in CSV order — see pod_combat_skills_master.csv
# Result codes: 0 = normal, 1 = guard-blocked, 2 = soaked, 3 = missed, 4 = countered, 5 = dodged

# ─── HEALTH TRACK WRITE-BACK ──────────────────────────────────────
# Apply combat damage to each combatant's actual PoD health track.
# Called at end of each round and at combat end, BEFORE cleanup.
# Splits damage into aggravated (tracked by agg_damage counter) and superficial.
# Aggravated: halved soak, applied to POD_health_aggravated_damage_num.
# Superficial: remainder, applied to POD_health_damage_num.
# Aggravated can push out superficial if the track is full.
#
# Unified flow: all combats (including 1v1) use the group writeback, which
# iterates every populated slot. A 1v1 is just two slots (PC + primary
# opponent) and the same loop handles it correctly.

pod_combat_apply_health_track_effect = {
	# Fatality mode determines how (or if) combat damage is written back
	# to each combatant's PoD health track.
	#
	#   practice — sparring only; no wounds written back at all.
	#   no       — non-lethal; wounds written but track is capped so
	#              no one's track is completely full (at least 1 empty box).
	#   anything else (possible/default/always) — normal writeback.

	if = {
		limit = {
			has_variable = pod_combat_sc_fatality
			var:pod_combat_sc_fatality = flag:practice
		}
		# Practice duel — intentionally skip writeback.
		# Combat damage stays in the minigame state but is never applied
		# to the characters' actual PoD health tracks.
	}
	else = {
		# Write combat damage to PoD health tracks
		pod_combat_group_health_writeback_effect = yes

		# Non-fatal duels: ensure no combatant's track is completely full
		if = {
			limit = {
				has_variable = pod_combat_sc_fatality
				var:pod_combat_sc_fatality = flag:no
			}
			pod_combat_cap_nonfatal_slot_effect = { SLOT = 0 }
			pod_combat_cap_nonfatal_slot_effect = { SLOT = 1 }
			pod_combat_cap_nonfatal_slot_effect = { SLOT = 2 }
			pod_combat_cap_nonfatal_slot_effect = { SLOT = 3 }
			pod_combat_cap_nonfatal_slot_effect = { SLOT = 4 }
			pod_combat_cap_nonfatal_slot_effect = { SLOT = 5 }
			pod_combat_cap_nonfatal_slot_effect = { SLOT = 6 }
			pod_combat_cap_nonfatal_slot_effect = { SLOT = 7 }
			pod_combat_cap_nonfatal_slot_effect = { SLOT = 8 }
			pod_combat_cap_nonfatal_slot_effect = { SLOT = 9 }
		}
	}
}

pod_combat_cap_nonfatal_slot_effect = {
	# After health writeback, ensure combatant in slot $SLOT$ still has
	# at least 1 empty health box. If their track is completely full,
	# reduce superficial damage by 1 (or aggravated if no superficial).
	if = {
		limit = { has_variable = pod_combat_c$SLOT$_char }
		var:pod_combat_c$SLOT$_char = {
			save_scope_value_as = {
				name = nf_empty
				value = POD_empty_damage_boxes
			}
		}
		if = {
			limit = { scope:nf_empty <= 0 }
			var:pod_combat_c$SLOT$_char = {
				if = {
					limit = {
						has_variable = POD_health_damage_num
						var:POD_health_damage_num >= 1
					}
					change_variable = { name = POD_health_damage_num add = -1 }
				}
				else_if = {
					limit = {
						has_variable = POD_health_aggravated_damage_num
						var:POD_health_aggravated_damage_num >= 1
					}
					change_variable = { name = POD_health_aggravated_damage_num add = -1 }
				}
				POD_update_wound_trait_xp = yes
			}
		}
	}
}

# ─── SUPERNATURAL DAMAGE EFFECTS ─────────────────────────────────

pod_combat_deal_supernatural_hp_damage_effect = {
	# $TARGET_SIDE$ = "player" or "opponent"
	# $AMOUNT$ = damage amount
	# Supernatural/Superficial: auto-hit, no guard, dodge → soak → damage
	# NOTE: Caller must clear pod_combat_last_was_blocked/soaked/missed first!

	# Stealth then dodge — each negates the entire attack; stealth is consumed first so a stealthed dodger keeps their dodge
	if = {
		limit = { has_variable = pod_combat_$TARGET_SIDE$_is_stealthed }
		remove_variable = pod_combat_$TARGET_SIDE$_is_stealthed
		# Revert the +30 hit chance bonus stealth granted now that stealth is consumed
		save_scope_value_as = {
			name = new_hit_after_stealth
			value = { value = var:pod_combat_$TARGET_SIDE$_hit_chance subtract = 30 min = 20 max = 95 }
		}
		set_variable = { name = pod_combat_$TARGET_SIDE$_hit_chance value = scope:new_hit_after_stealth }
		set_variable = { name = pod_combat_last_was_dodged value = yes }
		set_variable = { name = pod_combat_last_damage value = 0 }
	}
	else_if = {
		limit = { has_variable = pod_combat_$TARGET_SIDE$_is_dodging }
		remove_variable = pod_combat_$TARGET_SIDE$_is_dodging
		set_variable = { name = pod_combat_last_was_dodged value = yes }
		set_variable = { name = pod_combat_last_damage value = 0 }
	}
	else = {
		save_scope_value_as = {
			name = target_soak
			value = var:pod_combat_$TARGET_SIDE$_soak
		}

		random_list = {
			# Soaked — damage negated
			0 = {
				modifier = { add = scope:target_soak }
				set_variable = { name = pod_combat_last_was_soaked value = yes }
				set_variable = { name = pod_combat_last_damage value = 0 }
			}
			# Not soaked — apply damage
			100 = {
				modifier = {
					add = {
						value = scope:target_soak
						multiply = -1
					}
				}

				save_scope_value_as = {
					name = new_hp
					value = {
						value = var:pod_combat_$TARGET_SIDE$_hp
						subtract = $AMOUNT$
						min = 0
					}
				}
				set_variable = { name = pod_combat_$TARGET_SIDE$_hp value = scope:new_hp }

				# Morale spillover
				save_scope_value_as = {
					name = morale_spill
					value = {
						value = $AMOUNT$
						multiply = pod_combat_morale_from_hp_ratio
						floor = yes
					}
				}
				if = {
					limit = { scope:morale_spill > 0 }
					save_scope_value_as = {
						name = new_morale
						value = {
							value = var:pod_combat_$TARGET_SIDE$_morale
							subtract = scope:morale_spill
							min = 0
						}
					}
					set_variable = { name = pod_combat_$TARGET_SIDE$_morale value = scope:new_morale }
				}

				set_variable = { name = pod_combat_last_damage value = $AMOUNT$ }
			}
		}
	}
	# Consume the actor's stealth (if any) now that the attack has resolved
	pod_combat_consume_actor_stealth_effect = yes
}

pod_combat_deal_physical_bypass_guard_hp_damage_effect = {
	# Physical bypass guard: spirit gate → stealth → dodge → hit chance → soak → damage (skips guard)
	# Used by Lightning Strike, etc.

	# ── Spirit-damage gate ──
	# See pod_combat_deal_physical_hp_damage_effect for details. Even
	# bypass-guard Physical attacks can't reach an incorporeal spirit
	# without POD_can_damage_spirit_trigger-equivalent capability.
	if = {
		limit = {
			has_variable = pod_combat_$TARGET_SIDE$_incorporeal_spirit
			NOT = { has_variable = pod_combat_player_can_damage_spirit }
		}
		set_variable = { name = pod_combat_last_was_missed value = yes }
		set_variable = { name = pod_combat_last_damage value = 0 }
	}
	# Stealth then dodge — each negates the entire attack; stealth is consumed first so a stealthed dodger keeps their dodge
	else_if = {
		limit = { has_variable = pod_combat_$TARGET_SIDE$_is_stealthed }
		remove_variable = pod_combat_$TARGET_SIDE$_is_stealthed
		# Revert the +30 hit chance bonus stealth granted now that stealth is consumed
		save_scope_value_as = {
			name = new_hit_after_stealth
			value = { value = var:pod_combat_$TARGET_SIDE$_hit_chance subtract = 30 min = 20 max = 95 }
		}
		set_variable = { name = pod_combat_$TARGET_SIDE$_hit_chance value = scope:new_hit_after_stealth }
		set_variable = { name = pod_combat_last_was_dodged value = yes }
		set_variable = { name = pod_combat_last_damage value = 0 }
	}
	else_if = {
		limit = { has_variable = pod_combat_$TARGET_SIDE$_is_dodging }
		remove_variable = pod_combat_$TARGET_SIDE$_is_dodging
		set_variable = { name = pod_combat_last_was_dodged value = yes }
		set_variable = { name = pod_combat_last_damage value = 0 }
	}
	else = {
		save_scope_value_as = {
			name = hit_chance
			value = var:pod_combat_current_hit_chance
		}

		random_list = {
			# Miss
			0 = {
				modifier = {
					add = {
						value = 100
						subtract = scope:hit_chance
					}
				}
				set_variable = { name = pod_combat_last_was_missed value = yes }
				set_variable = { name = pod_combat_last_damage value = 0 }
			}
			# Hit — skip guard, straight to soak/damage
			0 = {
				modifier = { add = scope:hit_chance }
				# Delegate to the soak-only path (reuse supernatural for soak-only check)
				save_scope_value_as = {
					name = target_soak
					value = var:pod_combat_$TARGET_SIDE$_soak
				}
				random_list = {
					0 = {
						modifier = { add = scope:target_soak }
						set_variable = { name = pod_combat_last_was_soaked value = yes }
						set_variable = { name = pod_combat_last_damage value = 0 }
					}
					100 = {
						modifier = { add = { value = scope:target_soak multiply = -1 } }
						save_scope_value_as = {
							name = new_hp
							value = { value = var:pod_combat_$TARGET_SIDE$_hp subtract = $AMOUNT$ min = 0 }
						}
						set_variable = { name = pod_combat_$TARGET_SIDE$_hp value = scope:new_hp }
						save_scope_value_as = {
							name = morale_spill
							value = { value = $AMOUNT$ multiply = pod_combat_morale_from_hp_ratio floor = yes }
						}
						if = {
							limit = { scope:morale_spill > 0 }
							save_scope_value_as = {
								name = new_morale
								value = { value = var:pod_combat_$TARGET_SIDE$_morale subtract = scope:morale_spill min = 0 }
							}
							set_variable = { name = pod_combat_$TARGET_SIDE$_morale value = scope:new_morale }
						}
						set_variable = { name = pod_combat_last_damage value = $AMOUNT$ }
					}
				}
			}
		}
	}
	# Consume the actor's stealth (if any) now that the attack has resolved
	pod_combat_consume_actor_stealth_effect = yes
}

# ─── AGGRAVATED DAMAGE EFFECTS ───────────────────────────────────
# Aggravated damage halves the target's soak chance, making it much harder to resist.
# Also tracks aggravated damage separately for PoD health write-back.

pod_combat_deal_physical_aggravated_hp_damage_effect = {
	# Physical/Aggravated: spirit gate → stealth → dodge → hit chance → guard → soak (halved) → damage
	# $TARGET_SIDE$ = "player" or "opponent"
	# $AMOUNT$ = damage amount

	# ── Spirit-damage gate ──
	# See pod_combat_deal_physical_hp_damage_effect for details. Aggravated
	# physical damage (claws, bites, silver, wooden stakes) still counts as
	# Physical for spirit-immunity purposes — an incorporeal target ignores
	# it unless the actor can damage spirits.
	if = {
		limit = {
			has_variable = pod_combat_$TARGET_SIDE$_incorporeal_spirit
			NOT = { has_variable = pod_combat_player_can_damage_spirit }
		}
		set_variable = { name = pod_combat_last_was_missed value = yes }
		set_variable = { name = pod_combat_last_damage value = 0 }
	}
	else_if = {
		limit = { has_variable = pod_combat_$TARGET_SIDE$_is_stealthed }
		remove_variable = pod_combat_$TARGET_SIDE$_is_stealthed
		# Revert the +30 hit chance bonus stealth granted now that stealth is consumed
		save_scope_value_as = {
			name = new_hit_after_stealth
			value = { value = var:pod_combat_$TARGET_SIDE$_hit_chance subtract = 30 min = 20 max = 95 }
		}
		set_variable = { name = pod_combat_$TARGET_SIDE$_hit_chance value = scope:new_hit_after_stealth }
		set_variable = { name = pod_combat_last_was_dodged value = yes }
		set_variable = { name = pod_combat_last_damage value = 0 }
	}
	else_if = {
		limit = { has_variable = pod_combat_$TARGET_SIDE$_is_dodging }
		remove_variable = pod_combat_$TARGET_SIDE$_is_dodging
		set_variable = { name = pod_combat_last_was_dodged value = yes }
		set_variable = { name = pod_combat_last_damage value = 0 }
	}
	else = {
		save_scope_value_as = {
			name = hit_chance
			value = var:pod_combat_current_hit_chance
		}

		random_list = {
			0 = {
				modifier = { add = { value = 100 subtract = scope:hit_chance } }
				set_variable = { name = pod_combat_last_was_missed value = yes }
				set_variable = { name = pod_combat_last_damage value = 0 }
			}
			0 = {
				modifier = { add = scope:hit_chance }
				# Guard check
				if = {
					limit = {
						has_variable = pod_combat_$TARGET_SIDE$_is_guarding
						NOT = { has_variable = pod_combat_$TARGET_SIDE$_feinted }
					}
					remove_variable = pod_combat_$TARGET_SIDE$_is_guarding
					set_variable = { name = pod_combat_last_was_blocked value = yes }
					set_variable = { name = pod_combat_last_damage value = 0 }
				}
				else = {
					# Aggravated soak check — soak at half effectiveness.
					# Aggravated damage is harder to resist but not impossible.
					save_scope_value_as = {
						name = agg_soak
						value = { value = var:pod_combat_$TARGET_SIDE$_soak divide = 2 floor = yes }
					}
					random_list = {
						0 = {
							modifier = { add = scope:agg_soak }
							set_variable = { name = pod_combat_last_was_soaked value = yes }
							set_variable = { name = pod_combat_last_damage value = 0 }
						}
						100 = {
							modifier = { add = { value = scope:agg_soak multiply = -1 } }
							save_scope_value_as = {
								name = new_hp
								value = { value = var:pod_combat_$TARGET_SIDE$_hp subtract = $AMOUNT$ min = 0 }
							}
							set_variable = { name = pod_combat_$TARGET_SIDE$_hp value = scope:new_hp }
							change_variable = { name = pod_combat_$TARGET_SIDE$_agg_damage add = $AMOUNT$ }
							save_scope_value_as = {
								name = morale_spill
								value = { value = $AMOUNT$ multiply = pod_combat_morale_from_hp_ratio floor = yes }
							}
							if = {
								limit = { scope:morale_spill > 0 }
								save_scope_value_as = {
									name = new_morale
									value = { value = var:pod_combat_$TARGET_SIDE$_morale subtract = scope:morale_spill min = 0 }
								}
								set_variable = { name = pod_combat_$TARGET_SIDE$_morale value = scope:new_morale }
							}
							set_variable = { name = pod_combat_last_damage value = $AMOUNT$ }
						}
					}
				}
			}
		}
	}
	# Consume the actor's stealth (if any) now that the attack has resolved
	pod_combat_consume_actor_stealth_effect = yes
}

# ─── ELEMENTAL / METAL / HOLY DAMAGE WRAPPERS ───────────────────────
# Fire, Sunlight, Silver, Gold, and Holy are "promoted" damage sources.
# Mechanically they reuse the Physical pipeline (hit → guard → soak → damage),
# but against vulnerable creature types they are ALWAYS aggravated:
#   Fire     — aggravated vs vampires
#   Sunlight — aggravated vs vampires
#   Silver   — aggravated vs werewolves and Bastet
#   Gold     — aggravated vs Mokole
#   Holy     — aggravated vs vampires, demons, Kuei-jin, and wraiths
# Non-vulnerable targets take the skill's baseline damage_type from CSV
# (these wrappers only cover the baseline-Superficial case; baseline-Aggravated
# skills route straight to pod_combat_deal_physical_aggravated_hp_damage_effect
# in the Python generator since they're already aggravated for everyone).
#
# All wrappers assume $TARGET_SIDE$ = opponent in practice — they look up the
# target's character reference via var:pod_combat_opponent. They accept the
# TARGET_SIDE parameter for symmetry with the rest of the damage pipeline.

pod_combat_deal_fire_hp_damage_effect = {
	# $TARGET_SIDE$ = "player" or "opponent" (always "opponent" in practice)
	# $AMOUNT$ = damage amount
	# Fire: aggravated vs vampires, else physical superficial.
	if = {
		limit = {
			exists = var:pod_combat_opponent
			var:pod_combat_opponent = { POD_is_vampire_trigger = yes }
		}
		pod_combat_deal_physical_aggravated_hp_damage_effect = { TARGET_SIDE = $TARGET_SIDE$ AMOUNT = $AMOUNT$ }
	}
	else = {
		pod_combat_deal_physical_hp_damage_effect = { TARGET_SIDE = $TARGET_SIDE$ AMOUNT = $AMOUNT$ }
	}
}

# ─── HEALING EFFECTS ────────────────────────────────────────────────
# Three variants so the Python generator can route based on damage_type:
#   Superficial — mundane/quick healing (bandages, Regeneration, feeding).
#                 Cannot heal through aggravated wounds; HP is capped at
#                 (max_hp - agg_damage) so aggravated damage remains a
#                 lasting wound until it is specifically cured.
#   Aggravated  — rare / potent healing (Obeah Mend, Fortitude high-level,
#                 soul-drain). Restores HP AND reduces the agg_damage counter
#                 by the amount healed, so aggravated wounds are properly
#                 cleared for write-back to the real PoD health track.
#   Legacy      — pod_combat_restore_hp_effect. Unconditional heal capped at
#                 max_hp with no agg_damage interaction. Used when a skill
#                 leaves damage_type blank (no type distinction).

pod_combat_restore_hp_superficial_effect = {
	# $SIDE$ = "player" or "opponent"
	# $AMOUNT$ = superficial HP to restore
	# Ceiling: max_hp minus current aggravated damage, so aggravated wounds
	# cannot be healed by superficial means.
	save_scope_value_as = {
		name = sup_ceiling
		value = {
			value = var:pod_combat_$SIDE$_max_hp
			subtract = var:pod_combat_$SIDE$_agg_damage
		}
	}
	save_scope_value_as = {
		name = new_hp
		value = {
			value = var:pod_combat_$SIDE$_hp
			add = $AMOUNT$
			max = scope:sup_ceiling
		}
	}
	set_variable = { name = pod_combat_$SIDE$_hp value = scope:new_hp }
	set_variable = { name = pod_combat_last_damage value = $AMOUNT$ }
	set_variable = { name = pod_combat_last_was_heal value = yes }
}

pod_combat_restore_hp_aggravated_effect = {
	# $SIDE$ = "player" or "opponent"
	# $AMOUNT$ = aggravated HP to restore
	# Reduces the agg_damage counter by up to $AMOUNT$ (bounded by the current
	# counter so we don't go negative), then adds $AMOUNT$ to HP capped at
	# max_hp. This keeps the health write-back consistent: if we cured 2
	# aggravated wounds mid-combat, those 2 wounds won't reappear on the PoD
	# health track when combat ends.
	save_scope_value_as = {
		name = agg_heal
		value = {
			value = $AMOUNT$
			max = var:pod_combat_$SIDE$_agg_damage
		}
	}
	change_variable = {
		name = pod_combat_$SIDE$_agg_damage
		add = {
			value = scope:agg_heal
			multiply = -1
		}
	}
	save_scope_value_as = {
		name = new_hp
		value = {
			value = var:pod_combat_$SIDE$_hp
			add = $AMOUNT$
			max = var:pod_combat_$SIDE$_max_hp
		}
	}
	set_variable = { name = pod_combat_$SIDE$_hp value = scope:new_hp }
	set_variable = { name = pod_combat_last_damage value = $AMOUNT$ }
	set_variable = { name = pod_combat_last_was_heal value = yes }
}

pod_combat_restore_hp_effect = {
	# $SIDE$ = "player" or "opponent"
	# $AMOUNT$ = HP to restore
	# Legacy unconditional heal: no distinction between superficial and
	# aggravated. Used by skills that leave damage_type blank.
	save_scope_value_as = {
		name = new_hp
		value = {
			value = var:pod_combat_$SIDE$_hp
			add = $AMOUNT$
			max = var:pod_combat_$SIDE$_max_hp
		}
	}
	set_variable = { name = pod_combat_$SIDE$_hp value = scope:new_hp }
	set_variable = { name = pod_combat_last_damage value = $AMOUNT$ }
	set_variable = { name = pod_combat_last_was_heal value = yes }
}

# ─── END COMBAT (INLINE) ──────────────────────────────────────────
# Replaces the old pod_combat.0090 event. Runs inline from the victory
# check, auto-resolve, or forfeit path — no event popup, no scope
# boundary. For duel-driven combat (pod_combat_sc_driven), the
# dispatcher reconstructs PoD scopes and fires the follow-up event.
# For standalone/debug combat, shows a result toast.

pod_combat_end_combat_effect = {
	# Re-entry guard: the skill button's GUI execute and the "End Combat"
	# button both call this when pod_combat_is_over is set. The first call
	# removes pod_combat_active (SC-driven) or runs full cleanup (standalone),
	# so subsequent calls must be no-ops.
	if = {
		limit = { NOT = { has_variable = pod_combat_active } }
		# Already ended — nothing to do
	}
	else = {

	# Write combat damage back to the PoD health track.
	pod_combat_apply_health_track_effect = yes

	# If launched via configure_start_single_combat_effect, fire the
	# post-combat dispatch event immediately. The event rebuilds all
	# scopes it needs from stashed character variables on the PC, so it
	# doesn't depend on the current scope chain. (An earlier version fired
	# with days = 1 to escape the GUI execute context entirely; no longer
	# needed once scope reconstruction was moved into the event itself.)
	#
	# The combat window closes immediately (pod_combat_active removed).
	# The stash variables persist on the PC until the event fires the
	# follow-up and cleans them up.
	if = {
		limit = { has_variable = pod_combat_sc_driven }
		# Close the combat window immediately but keep pod_combat_is_over
		# as a re-entry guard (the victory check gates on NOT is_over).
		# pod_combat.9999 handles full cleanup after firing the follow-up.
		remove_variable = pod_combat_active
		trigger_event = { id = pod_combat.9999 }
	}

	# For standalone/debug combat (no duel override), show a toast
	# and clean up immediately (no scope bridge needed).
	else = {
		if = {
			limit = {
				has_variable = pod_combat_winner
				var:pod_combat_winner ?= root
			}
			send_interface_toast = {
				title = pod_combat_result_victory_toast
				left_icon = root
			}
		}
		else = {
			send_interface_toast = {
				title = pod_combat_result_defeat_toast
				left_icon = root
			}
		}
		pod_combat_cleanup_effect = yes
	}

	} # end re-entry guard
}
