# Damage helpers, soak, retaliation dispatcher, push_log, and the
# unified contact resolver. Companion file to pod_combat_move_effects.txt
# (which dispatches into here); both are called from POD_execute_combat_move.
#
# Result-code map (pod_combat_last_result_code), sentinel/priority
# semantics, the self-damage snapshot/restore pattern, and the
# reaction-system architecture all live in COMBAT_REFERENCE.info §4 and §6.

pod_combat_set_result_code_effect = {
	# $CODE$ = 0..12 — overwrites the current value.
	# Use for primary HP-event outcomes (damage / blocked / soaked /
	# missed / dodged / stealthed) that are mutually exclusive within
	# the damage pipeline.
	set_variable = { name = pod_combat_last_result_code value = $CODE$ }
}

pod_combat_set_result_code_if_unset_effect = {
	# $CODE$ = 0..12
	# Sets the code only when no primary code has fired
	# (pod_combat_last_result_code is still -1 / unset). Used by the
	# morale-damage and heal pipelines so a "damage + heal" or
	# "damage + morale" hybrid keeps the damage code as primary; the
	# morale / heal amounts ride along as separate amount fields.
	if = {
		limit = {
			OR = {
				NOT = { has_variable = pod_combat_last_result_code }
				var:pod_combat_last_result_code < 0
			}
		}
		set_variable = { name = pod_combat_last_result_code value = $CODE$ }
	}
}

pod_combat_promote_result_code_retaliation_effect = {
	# Retaliation fires after the primary damage outcome resolves.
	# Promote 0 (damage) → 6 (damage + retaliation triggered) and
	# 2 (soaked) → 7 (damage + retaliation soaked). Other codes
	# (missed / blocked / etc.) shouldn't reach here because
	# pod_combat_check_retaliation_effect gates retaliation on
	# pod_combat_last_damage > 0 OR pod_combat_last_result_code = 3
	# (missed) — but be defensive and only promote if the current
	# code is 0 or 2.
	if = {
		limit = {
			has_variable = pod_combat_last_result_code
			var:pod_combat_last_result_code = 0
		}
		set_variable = { name = pod_combat_last_result_code value = 6 }
	}
	else_if = {
		limit = {
			has_variable = pod_combat_last_result_code
			var:pod_combat_last_result_code = 2
		}
		set_variable = { name = pod_combat_last_result_code value = 7 }
	}
}


# ─── RESOURCE MANIPULATION ────────────────────────────────────────

pod_combat_spend_ap_effect = {
	# $AMOUNT$ = AP to spend
	change_variable = {
		name = pod_combat_current_ap
		add = {
			value = $AMOUNT$
			multiply = -1
		}
	}
}

# ── REACTION PHASE ─────────────────────────────────────────────
# §15 of POD_execute_combat_move. The dispatcher
# pod_combat_check_retaliation_effect walks the workspace
# pod_combat_$TARGET_SIDE$_retaliate_* slots and fires each one
# whose §6 outcome matched its trigger condition. For the system
# overview, the recipe to add a new reaction class, and the
# damage_type inheritance rules see COMBAT_REFERENCE.info §6.
#
# Test note: retaliation has not been well tested. Treat it as
# reference rather than hardened.

# Generic reaction-firing helper. Called once per reaction slot pair
# from pod_combat_check_retaliation_effect. Encapsulates the limit-
# then-fire-then-clear pattern so each retaliation type is one block.
pod_combat_apply_reaction_if_triggered_effect = {
	# $TRIGGER_VAR$       = workspace var holding the reaction amount
	#                       (presence = reaction is armed; value = damage)
	# $AGG_FLAG_VAR$      = workspace var whose presence marks the
	#                       reaction as aggravated (absent = superficial)
	# $CONDITION_TRIGGER$ = name of the result-code partition trigger
	#                       to gate on (pod_combat_result_code_indicates_*)
	# $ACTOR_SIDE$        = side taking the reaction damage (the side
	#                       that just attacked)
	# $RETALIATOR_SIDE$   = side dishing out the reaction (the side
	#                       that was just attacked / had the reaction
	#                       armed)
	if = {
		limit = {
			has_variable = $TRIGGER_VAR$
			$CONDITION_TRIGGER$ = yes
		}
		pod_combat_apply_retaliation_helper = {
			ACTOR_SIDE      = $ACTOR_SIDE$
			RETALIATOR_SIDE = $RETALIATOR_SIDE$
			AMOUNT_VAR      = $TRIGGER_VAR$
			AGG_FLAG_VAR    = $AGG_FLAG_VAR$
		}
		remove_variable = $TRIGGER_VAR$
		remove_variable = $AGG_FLAG_VAR$
	}
}

pod_combat_check_retaliation_effect = {
	# Reaction phase — see the REACTION PHASE block comment above.
	#
	# Retaliation damage inherits the damage_source and damage_type
	# from the skill that armed the stance:
	#   - Superficial: goes through the actor's soak check (can be soaked)
	#   - Aggravated: can not be soaked, tracked as aggravated damage
	# The _agg companion variable (presence = aggravated, absent =
	# superficial) is set on the workspace alongside the amount in §13.
	#
	# $TARGET_SIDE$ = "opponent" — side that has the retaliation stance
	# $ACTOR_SIDE$  = "player"   — side that takes retaliation damage
	#
	# Both reactions use the shared payload helper, which preserves
	# the original attack's pod_combat_last_damage across the inner
	# damage call and promotes the result_code from 0 / 2 → 6 / 7
	# (damage / soaked → "+ retaliation" overlay) for the log line.

	# Retaliate on contact — fires whenever the original attack made
	# physical contact with the target, regardless of whether wounds
	# went through (codes 0 / 1 / 2 / 6 / 7: HP damage, guard-blocked,
	# soaked, and the retaliation overlays). This matches the natural
	# semantics of thorns / spite stances: the razors / spines / spiteful
	# aura punish anyone who reaches you, even if your armor or guard
	# absorbed their blow. Evasion outcomes (miss / dodge / stealth)
	# stay excluded — there was no contact to retaliate against.
	#
	# Why _contact_trigger and not _hp_damage_trigger: a literal "fires
	# on wound" reading would exclude soaked and blocked outcomes, but
	# armor-style retaliation should still bite back when the attacker
	# physically struck even if the wound failed to land.
	pod_combat_apply_reaction_if_triggered_effect = {
		TRIGGER_VAR       = pod_combat_$TARGET_SIDE$_retaliate_hit
		AGG_FLAG_VAR      = pod_combat_$TARGET_SIDE$_retaliate_hit_agg
		CONDITION_TRIGGER = pod_combat_result_code_indicates_contact_trigger
		ACTOR_SIDE        = $ACTOR_SIDE$
		RETALIATOR_SIDE   = $TARGET_SIDE$
	}

	# Retaliate on evasion — fires on miss / dodge / stealth (codes
	# 3 / 5 / 12). All three are semantically the same outcome (the
	# attack didn't land), so a "lash out when evaded" stance fires
	# for any of them.
	pod_combat_apply_reaction_if_triggered_effect = {
		TRIGGER_VAR       = pod_combat_$TARGET_SIDE$_retaliate_miss
		AGG_FLAG_VAR      = pod_combat_$TARGET_SIDE$_retaliate_miss_agg
		CONDITION_TRIGGER = pod_combat_result_code_indicates_miss_trigger
		ACTOR_SIDE        = $ACTOR_SIDE$
		RETALIATOR_SIDE   = $TARGET_SIDE$
	}
}

pod_combat_apply_retaliation_helper = {
	# $ACTOR_SIDE$      = "player" or "opponent" — side taking retaliation
	#                     (the original attacker, who is now on the
	#                     receiving end of the counter-strike).
	# $RETALIATOR_SIDE$ = "player" or "opponent" — side dishing out the
	#                     retaliation (the original attack's target).
	#                     Gates the retaliation through the
	#                     spirit-damage check: if the retaliation target
	#                     is incorporeal, the retaliator must have
	#                     pod_combat_$RETALIATOR_SIDE$_can_damage_spirit
	#                     for the counter-hit to land.
	# $AMOUNT_VAR$  = workspace variable name holding retaliation damage
	#                 (e.g. pod_combat_target_retaliate_hit)
	# $AGG_FLAG_VAR$ = workspace variable name whose presence means the
	#                  retaliation is aggravated (absent = superficial)
	#
	# Saves the original attack's last_damage so the helper's overwrite
	# doesn't clobber the main-attack log line. Uses the bypass_guard
	# helper because retaliation ignores the actor's guard stance (by
	# design — guard is for incoming attacks, not counter strikes).

	# ── Spirit-damage gate ──
	# Mirrors the gate in pod_combat_resolve_contact_effect: if the
	# retaliation target is an incorporeal spirit, the counter-strike
	# only lands when the retaliator has the spirit-damage capability.
	# Otherwise the retaliation no-ops (no damage, no flag flip) so the
	# main-attack log line is preserved as-is.
	#
	# Retaliation skips the contact resolver entirely (it ignores
	# stance / guard / hit roll — it's a reflex, not a deliberate
	# attack), so we hand-roll just the spirit gate here. The rest of
	# the pre-damage decision tree doesn't apply.
	if = {
		limit = {
			has_variable = pod_combat_$ACTOR_SIDE$_incorporeal_spirit
			NOT = { has_variable = pod_combat_$RETALIATOR_SIDE$_can_damage_spirit }
		}
		# No-op: retaliation passes through harmlessly.
	}
	else = {
		save_scope_value_as = {
			name  = retaliation_pre_last_damage
			value = var:pod_combat_last_damage
		}

		# Pre-resolve var:$AMOUNT_VAR$ at THIS scope (the combat host,
		# where pod_combat_$TARGET_SIDE$_retaliate_hit / _miss actually
		# live) and pass the integer through the chain as a saved scope
		# value. The naive AMOUNT = var:$AMOUNT_VAR$ pattern silently
		# misroutes: macro substitution is textual, so by the time the
		# engine reaches `value = $COUNT$` inside PoD's
		# POD_increase_wounds_effect, the text has become
		# `value = var:pod_combat_target_retaliate_hit` — and that
		# wound effect runs inside `var:pod_combat_$TARGET_SIDE$_char =
		# { ... }`, so the var lookup hits the TARGET character (who
		# doesn't carry workspace state) and PoD logs "var not set" +
		# treats the value as `none`. Saving the int up front routes
		# around that — scope values resolve through the effect chain
		# without being re-evaluated against the runtime scope.
		save_scope_value_as = {
			name  = retaliation_amount
			value = var:$AMOUNT_VAR$
		}

		if = {
			limit = { has_variable = $AGG_FLAG_VAR$ }
			pod_combat_deal_hp_damage_bypass_guard_effect = {
				TARGET_SIDE = $ACTOR_SIDE$
				AMOUNT      = scope:retaliation_amount
				REASON      = duel_aggravated
			}
		}
		else = {
			pod_combat_deal_hp_damage_bypass_guard_effect = {
				TARGET_SIDE = $ACTOR_SIDE$
				AMOUNT      = scope:retaliation_amount
				REASON      = duel
			}
		}

		# Stash the retaliation's actual landed damage (post-soak)
		# BEFORE we restore last_damage. Without this, when the original
		# attack was guard-blocked (rcode 1) or soaked (rcode 2), the
		# log entry's `damage` field carries the original's value (= 0)
		# and the rcode 6 dispatch falls through to the em-dash fallback
		# (both rcode 6 branches gate on `var:damage > 0`). Retaliation
		# fires regularly from those original outcomes (per the
		# _contact_trigger gate above), so the missing field surfaces.
		# push_log writes this onto entry.var:retal_damage; the loc
		# dispatch consults it for a "Counter-attack -X HP" fallback
		# row when entry.damage is 0.
		set_variable = {
			name  = pod_combat_last_retal_damage
			value = var:pod_combat_last_damage
		}

		# Promote the inner-call result_code to the retaliation
		# variant: 0 (damage landed) → 6 (damage + retaliation
		# triggered), 2 (soaked) → 7 (damage + retaliation soaked).
		# pod_combat_promote_result_code_retaliation_effect is
		# defensive — it only promotes 0/2, leaving anything else
		# untouched.
		pod_combat_promote_result_code_retaliation_effect = yes

		# Restore the original attack's last_damage so the log line for the
		# main attack still reads correctly. Retaliation's own damage
		# is captured in pod_combat_last_retal_damage above.
		set_variable = { name = pod_combat_last_damage value = scope:retaliation_pre_last_damage }
	}
}

pod_combat_apply_ap_drain_effect = {
	# Apply any AP drain debuff from the previous round to the current
	# actor. Reads pod_combat_actor_ap_drain from the workspace (mirrored
	# from the actor slot story's slot_ap_drain by
	# pod_combat_load_actor_from_story_effect). Clears the drain after
	# application so it only fires once.
	if = {
		limit = { has_variable = pod_combat_actor_ap_drain }
		change_variable = {
			name = pod_combat_current_ap
			add = {
				value = var:pod_combat_actor_ap_drain
				multiply = -1
			}
		}
		# Clamp to minimum 1 AP so the actor can always do something
		if = {
			limit = { var:pod_combat_current_ap < 1 }
			set_variable = { name = pod_combat_current_ap value = 1 }
		}
		remove_variable = pod_combat_actor_ap_drain
	}
}

######################################################
###### POD WOUND-SYSTEM INTEGRATION HELPERS
######################################################
# Shim tht route combat damage through PoD's wound pipeline
# (POD_increase_wounds_hostile_effect → POD_apply_wound_boxes →
# POD_health_death_check_effect). Existing pipeline handles:
#   - Soak roll (POD_soak_chance, single knob on the character)
#   - Splat-weakness multiplier (POD_wound_multiplier_from_splat_weaknesses)
#   - Aggravated classification (POD_damage_type_is_aggravated — the
#     death_reason flag picks the branch)
#   - Damage cap (POD_damage_cap_per_death_reason)
#   - Immunity protection short-circuits
#   - Wound-box application (superficial + aggravated variables with
#     overflow rules) and wound trait XP updates
#   - Inline death check — fires PoD's death on-action if the damage
#     fills the aggravated track past threshold
#   - Interface message for "took damage" — fires on every wound write,
#     so during combat it produces one toast per landed hit. Suppressing
#     these toasts would require a custom POD_increase_wounds_* path that
#     skips the interface call; for now we accept the toast spam as the
#     cost of routing through PoD's standard pipeline.
#
# Combat retains: hit roll, stealth consumption, dodge consumption,
# guard consumption, spirit-damage gate, and morale spillover (morale is
# a minigame-only concept with no PoD equivalent).
#
# Workspace state (HP, morale, agg_damage) is re-derived from PoD
# variables after each hit so mid-combat queries stay consistent.

pod_combat_route_to_pod_damage = {
	# $TARGET_SIDE$ = "player" or "opponent"
	# $REASON$ = death reason suffix (without the "death_" prefix) — e.g.
	#            duel, duel_aggravated, fire, sun, silver, gold, true_faith.
	#            POD_increase_wounds_effect builds the flag internally.
	# $COUNT$ = raw damage (PoD will multiply by splat weakness + soak-roll).
	#
	# Sets scope:damage_dealt for downstream logic (0 if fully soaked /
	# immune, remaining-HP-worth if the blow killed). The result_code
	# write happens internally — the helper sets pod_combat_last_result_code
	# directly (2 = soaked, 0 = damage, retained per-call) without
	# requiring callers to pre-clear anything. The §2 reset at the top
	# of POD_execute_combat_move is the only place that initialises the
	# code per-move.

	# Practice mode: sparring duel, no wounds actually inflicted. Skip
	# the entire PoD wound-pipeline call so nothing hits the char's
	# POD_health_damage_num / _aggravated_damage_num. Morale damage
	# (tracked separately on the workspace) still applies — a practice
	# duel can still break someone's composure even if it can't mark
	# their body. Returns scope:damage_dealt=0 so the caller's morale
	# spillover / retaliation / log-line logic all cleanly short-circuit.
	if = {
		limit = {
			has_variable = pod_combat_sc_context
			var:pod_combat_sc_context.var:fatality = flag:practice
		}
		save_scope_value_as = { name = damage_dealt value = 0 }
		set_variable = { name = pod_combat_last_damage value = 0 }
	}
	else = {

	# Snapshot pre-damage state. POD_total_damage_boxes and its sub-SVs
	# gate on is_alive=yes, so post-damage reads would return 0 for a
	# killed target and the delta calc would look like "fully soaked"
	# (scope:post_total_wounds = 0 < pre = nonzero → min=0 result).
	# We snapshot track_length too so the death branch below can
	# reconstruct the wound state without re-reading dead-char SVs.
	#
	# We snapshot superficial and aggravated counts SEPARATELY so the
	# damage_dealt calc below correctly handles PoD's aggravated push-
	# out: when aggravated damage lands on an already-wounded target,
	# POD_apply_wound_boxes converts existing superficial wounds into
	# aggravated wounds (e.g. 5 sup + 0 agg → 2 sup + 3 agg on a 3-agg
	# hit). Total boxes (POD_total_damage_boxes) stays at 5 both before
	# and after, so a naive post_total - pre_total delta reads as 0
	# and falsely classifies the hit as "soaked" (code 2). Tracking
	# the per-track delta and clamping each at 0 mirrors PoD's own
	# `total_wounds_gained` semantics (see POD_apply_wound_boxes).
	var:pod_combat_$TARGET_SIDE$_char = {
		save_scope_value_as = { name = pre_total_wounds       value = POD_total_damage_boxes }
		save_scope_value_as = { name = pre_superficial_wounds value = POD_superficial_damage_boxes }
		save_scope_value_as = { name = pre_aggravated_wounds  value = POD_aggravated_damage_boxes }
		save_scope_value_as = { name = pre_empty_boxes        value = POD_empty_damage_boxes }
		save_scope_value_as = { name = pre_track_length       value = POD_health_track_length }
	}
	# Combat host (PC) holds workspace; scope:combat_host must be
	# saved upstream by POD_execute_combat_move (single point of
	# truth — see line ~44 of pod_combat_move_effects.txt). 
	#
	# The deeper call chain (configure_start_group_combat_effect →
	# pod_combat_initialize_effect → process_pre_pc_turns →
	# execute_ai_turn → ai_take_turn → ai_try_action →
	# ai_select_and_execute → POD_execute_combat_move →
	# POD_combat_apply_single_hp_damage → POD_combat_dispatch_damage_pipeline
	# → pod_combat_apply_damage_payload_effect →
	# pod_combat_deal_hp_damage_bypass_guard_effect → here) walks
	# through enough effect dispatch that the engine treats root as
	# unset by the time the local fallback ran, which then blew up
	# the chained POD_increase_wounds_hostile_effect call below
	# that reads scope:combat_host. Every caller of this effect
	# threads through POD_execute_combat_move, so the upstream save
	# is guaranteed and the local fallback was redundant.
	var:pod_combat_$TARGET_SIDE$_char = {
		POD_increase_wounds_hostile_effect = {
			ATTACKER = scope:combat_host.var:pod_combat_actor_char
			REASON   = $REASON$
			COUNT    = $COUNT$
		}
	}

	# Branch on whether PoD killed the target. POD_apply_wound_boxes
	# calls POD_health_death_check_effect inline; a fatal blow fires
	# pod_death and is_alive flips to no before this line resumes.
	if = {
		limit = { var:pod_combat_$TARGET_SIDE$_char = { is_alive = no } }
		# Target died. Dead-char wound SVs are zeroed by their is_alive
		# limits, so we use the pre-snapshot to approximate damage_dealt:
		# whatever empty HP they had before the blow counts as the
		# damage that killed them.
		save_scope_value_as = {
			name = damage_dealt
			value = scope:pre_empty_boxes
		}
		set_variable = { name = pod_combat_last_damage value = scope:damage_dealt }
		pod_combat_set_result_code_effect = { CODE = 0 }
	}
	else = {
		# Target still alive — read fresh post-damage wound state.
		# Superficial / aggravated tracked separately so the push-out
		# case (snapshot comment above) reads damage_dealt > 0 from
		# the aggravated delta even when total boxes are unchanged.
		var:pod_combat_$TARGET_SIDE$_char = {
			save_scope_value_as = { name = post_superficial_wounds value = POD_superficial_damage_boxes }
			save_scope_value_as = { name = post_aggravated_wounds  value = POD_aggravated_damage_boxes }
		}
		save_scope_value_as = {
			name = superficial_added
			value = {
				value    = scope:post_superficial_wounds
				subtract = scope:pre_superficial_wounds
				min      = 0
			}
		}
		save_scope_value_as = {
			name = aggravated_added
			value = {
				value    = scope:post_aggravated_wounds
				subtract = scope:pre_aggravated_wounds
				min      = 0
			}
		}
		# damage_dealt = real wound boxes + aggravated conversions.
		# Mirrors PoD's own total_wounds_gained calc (see
		# POD_apply_wound_boxes): max(0, sup_delta) + agg_delta.
		# A push-out case (5 sup → 2 sup + 3 agg) reads 0 + 3 = 3,
		# correctly logging the hit as damage instead of as soaked.
		save_scope_value_as = {
			name = damage_dealt
			value = {
				value = scope:superficial_added
				add   = scope:aggravated_added
			}
		}

		if = {
			limit = { scope:damage_dealt = 0 }
			# Fully soaked or immune: treat as "soaked" (code 2) for log.
			pod_combat_set_result_code_effect = { CODE = 2 }
			set_variable = { name = pod_combat_last_damage value = 0 }
		}
		else = {
			# Char's PoD wound vars are now the source of truth; no
			# workspace HP copy to update. SV accessors
			# (pod_combat_$SIDE$_hp_value etc.) derive from the char.
			set_variable = { name = pod_combat_last_damage value = scope:damage_dealt }
			pod_combat_set_result_code_effect = { CODE = 0 }

			# Morale spillover scales off actual damage (post-soak/multiplier),
			# not the raw AMOUNT passed in. The spillover also bleeds into
			# Mental Strain on the target, matching direct morale damage's
			# stress application (see pod_combat_deal_morale_damage_effect).
			save_scope_value_as = {
				name = morale_spill
				value = {
					value = scope:damage_dealt
					multiply = pod_combat_morale_from_hp_ratio
					floor = yes
				}
			}
			if = {
				limit = { scope:morale_spill > 0 }
				# Cap the spillover at current morale so mental-strain
				# stress doesn't over-count past a morale-broken target.
				# See pod_combat_deal_morale_damage_effect for the
				save_scope_value_as = {
					name = morale_actually_spilled
					value = {
						value = scope:morale_spill
						max   = var:pod_combat_$TARGET_SIDE$_morale
					}
				}
				save_scope_value_as = {
					name = new_target_morale
					value = {
						value = var:pod_combat_$TARGET_SIDE$_morale
						subtract = scope:morale_spill
						min = 0
					}
				}
				set_variable = { name = pod_combat_$TARGET_SIDE$_morale value = scope:new_target_morale }
				# Mental Strain on the spillover delta.
				if = {
					limit = { scope:morale_actually_spilled > 0 }
					save_scope_value_as = {
						name = mental_strain_from_spillover
						value = {
							value = scope:morale_actually_spilled
							multiply = pod_combat_morale_stress_multiplier
						}
					}
					var:pod_combat_$TARGET_SIDE$_char = {
						POD_static_stress_gain = { COUNT = scope:mental_strain_from_spillover }
					}
				}
			}
		}
	}
	}  # end else (non-practice path)
}


pod_combat_deal_hp_damage_bypass_guard_effect = {
	# $TARGET_SIDE$ = "player" or "opponent" (who takes damage)
	# $AMOUNT$ = damage amount (raw, pre-soak, pre-multiplier)
	# $REASON$ = death reason suffix — see pod_combat_route_to_pod_damage.
	#
	# Bare PoD damage write — no guard / stance / spirit / hit-roll
	# logic. The contact resolver (pod_combat_resolve_contact_effect)
	# already handled all of those before the damage payload fires, so
	# this helper just hands the raw COUNT to PoD's wound pipeline.
	# Also called directly by the self-damage and retaliation paths,
	# both of which intentionally bypass the contact resolver (self-
	# damage targets the actor, retaliation is a reflex strike).
	#
	# pod_combat_last_result_code is initialised once per move in §2
	# of POD_execute_combat_move; this helper (via route_to_pod_damage)
	# overwrites it with the per-call outcome.
	pod_combat_route_to_pod_damage = {
		TARGET_SIDE = $TARGET_SIDE$
		REASON      = $REASON$
		COUNT       = $AMOUNT$
	}
}

# ── Actor stealth consumption ──────────────────────────────────
# Stealth persists until the stealthed character attacks OR is attacked.
# When the stealthed character is attacked, the damage pipeline consumes
# the target's stealth flag via the $TARGET_SIDE$ check at the top of
# each damage effect. When the stealthed character acts (attacks), we
# consume it here — called at the END of each damage effect so the +30
# hit bonus still applies to this attack, then gets reverted for future
# actions.
#
# $ACTOR_SIDE$ = "player" or "opponent" — the side that just attacked.
# Forward attacks pass ACTOR_SIDE=actor (workspace convention has
# the actor on the player side). Retaliation calls have the
# retaliator on the opponent side, so retaliation paths that route
# through this helper would pass ACTOR_SIDE=target. Today
# pod_combat_apply_retaliation_helper goes through the bare
# pod_combat_deal_hp_damage_bypass_guard_effect (which doesn't call
# this stealth-consume), so retaliation doesn't currently consume
# the retaliator's stealth — by design (retaliation is a reflex,
# not a deliberate attack that breaks stealth).
pod_combat_consume_actor_stealth_effect = {
	if = {
		limit = { has_variable = pod_combat_$ACTOR_SIDE$_is_stealthed }
		remove_variable = pod_combat_$ACTOR_SIDE$_is_stealthed
		# Revert the +30 hit chance bonus stealth granted
		save_scope_value_as = {
			name = new_hit_after_actor_stealth
			value = { value = var:pod_combat_$ACTOR_SIDE$_hit_chance subtract = 30 min = 20 max = 95 }
		}
		set_variable = { name = pod_combat_$ACTOR_SIDE$_hit_chance value = scope:new_hit_after_actor_stealth }
		# Keep current_hit_chance in sync in case the damage pipeline
		# already copied it elsewhere in the same turn.
		set_variable = { name = pod_combat_current_hit_chance value = scope:new_hit_after_actor_stealth }
	}
}

# ── Target stance consumption ──────────────────────────────────
# Stealth → dodge consumption on the target side, used by every
# damage helper as a pre-check before damage application. Stealth
# is consumed FIRST so a stealthed-and-dodging target keeps their
# dodge for the next attack (they "used" stealth for this one).
#
# $TARGET_SIDE$ = "player" or "opponent" — the side being attacked.
#
# Saves scope:stance_consumed for callers to branch on:
#   0 = no stance fired (caller proceeds to hit roll / damage)
#   1 = stealth consumed (caller skips its else-branch entirely)
#   2 = dodge consumed (same)
# Both stance-fired branches set pod_combat_last_result_code (12 for
# stealth, 5 for dodge) and pod_combat_last_damage = 0 internally so
# callers don't need to reset those.
#
# Centralised here so stealth / dodge semantics live in one place. The
# HP-damage path calls this exclusively through pod_combat_resolve_contact_effect;
# the morale path still calls it inline (its auto-hit / no-guard
# semantics don't justify the resolver overhead).
pod_combat_consume_target_stance_effect = {
	if = {
		limit = { has_variable = pod_combat_$TARGET_SIDE$_is_stealthed }
		remove_variable = pod_combat_$TARGET_SIDE$_is_stealthed
		# Revert the +30 hit chance bonus stealth granted now that
		# stealth is consumed.
		save_scope_value_as = {
			name = new_hit_after_stealth
			value = { value = var:pod_combat_$TARGET_SIDE$_hit_chance subtract = 30 min = 20 max = 95 }
		}
		set_variable = { name = pod_combat_$TARGET_SIDE$_hit_chance value = scope:new_hit_after_stealth }
		# Code 12 = stealth-triggered auto-miss. Distinct from dodge
		# (code 5) so the log line links to the Stealth concept rather
		# than the Dodge one. Same auto-miss mechanic, different cause.
		pod_combat_set_result_code_effect = { CODE = 12 }
		set_variable = { name = pod_combat_last_damage value = 0 }
		save_scope_value_as = { name = stance_consumed value = 1 }
	}
	else_if = {
		limit = { has_variable = pod_combat_$TARGET_SIDE$_is_dodging }
		remove_variable = pod_combat_$TARGET_SIDE$_is_dodging
		pod_combat_set_result_code_effect = { CODE = 5 }
		set_variable = { name = pod_combat_last_damage value = 0 }
		save_scope_value_as = { name = stance_consumed value = 2 }
	}
	else = {
		save_scope_value_as = { name = stance_consumed value = 0 }
	}
}

# ── UNIFIED CONTACT RESOLVER ──────────────────────────────────────
# Single resolver that walks the pre-damage decision tree (spirit gate
# → stance consume → hit roll → guard check) using capability flags
# from the executing move's story. One pipeline serves every move's
# pre-damage steps; per-source damage application happens after.
#
# Capability flags (set on scope:current_move at move-init time, see
# resolve_capability_flags in generate_combat_files.py):
#   checks_spirit_gate — physical attacks pass through incorporeal
#                        spirits unless actor has can_damage_spirit.
#   checks_stance      — stealth → dodge consumption gates damage.
#                        Default-on for every attack_type.
#   requires_hit_roll  — roll var:pod_combat_current_hit_chance vs 100.
#                        Default-on for physical, off for supernatural/mental.
#   checks_guard       — consume target's pod_combat_$TARGET_SIDE$_is_guarding
#                        and block damage. Default-on for physical only.
#
# Outputs:
#   scope:contact_result enum (consumed by callers to decide whether
#   to fire damage payload):
#     0 = hit            — proceed to pod_combat_apply_damage_payload_effect
#     1 = blocked        — guard consumed, no damage
#     2 = miss           — hit roll failed
#     3 = dodged         — dodge stance consumed
#     4 = stealthed      — stealth stance consumed
#     6 = spirit_immune  — gate denied
#   pod_combat_last_result_code — written for each non-hit outcome
#                        (1 / 3 / 5 / 12; spirit-immune logs as miss).
#                        Hit (contact_result = 0) leaves the code
#                        for the payload to write (code 0).
#   pod_combat_last_damage = 0 — set on every non-hit outcome so
#                        push_log doesn't read stale damage.
pod_combat_resolve_contact_effect = {
	# $TARGET_SIDE$ = "player" or "opponent" — who takes the attack.
	# $ACTOR_SIDE$  = "player" or "opponent" — who is attacking.
	# Caller must have saved the move story as scope:current_move so
	# the capability-flag reads resolve.

	# 1. Spirit gate (capability-driven)
	if = {
		limit = {
			scope:current_move = { has_variable = checks_spirit_gate }
			has_variable = pod_combat_$TARGET_SIDE$_incorporeal_spirit
			NOT = { has_variable = pod_combat_$ACTOR_SIDE$_can_damage_spirit }
		}
		# Spirit gate denies the attack — log as missed (no dedicated
		# spirit-immune log code; functionally equivalent to a miss).
		pod_combat_set_result_code_effect = { CODE = 3 }
		set_variable = { name = pod_combat_last_damage value = 0 }
		save_scope_value_as = { name = contact_result value = 6 }
	}
	else = {
		# 2. Stance consume (capability-driven)
		if = {
			limit = { scope:current_move = { has_variable = checks_stance } }
			pod_combat_consume_target_stance_effect = { TARGET_SIDE = $TARGET_SIDE$ }
			# consume_target_stance writes scope:stance_consumed (0/1/2)
			# and sets the matching result_code (12 stealth / 5 dodge).
		}
		else = {
			save_scope_value_as = { name = stance_consumed value = 0 }
		}

		if = {
			limit = { scope:stance_consumed = 1 }
			save_scope_value_as = { name = contact_result value = 4 }
		}
		else_if = {
			limit = { scope:stance_consumed = 2 }
			save_scope_value_as = { name = contact_result value = 3 }
		}
		else = {
			# 3. Hit roll (capability-driven). No flag = auto-hit.
			if = {
				limit = { scope:current_move = { has_variable = requires_hit_roll } }
				save_scope_value_as = {
					name  = hit_chance
					value = var:pod_combat_current_hit_chance
				}
				random_list = {
					# Miss
					0 = {
						modifier = {
							add = {
								value    = 100
								subtract = scope:hit_chance
							}
						}
						pod_combat_set_result_code_effect = { CODE = 3 }
						set_variable = { name = pod_combat_last_damage value = 0 }
						save_scope_value_as = { name = contact_result value = 2 }
					}
					# Hit — roll succeeded, fall through to guard check
					0 = {
						modifier = { add = scope:hit_chance }
						pod_combat_check_guard_effect = { TARGET_SIDE = $TARGET_SIDE$ }
					}
				}
			}
			else = {
				# Auto-hit — skip the roll, still check guard if applicable.
				pod_combat_check_guard_effect = { TARGET_SIDE = $TARGET_SIDE$ }
			}
		}
	}
}

# ── GUARD CHECK SUBROUTINE ─────────────────────────────────────────
# Extracted so both the hit-roll-success branch and the auto-hit
# branch in pod_combat_resolve_contact_effect can share the same
# guard-handling code without duplication.
pod_combat_check_guard_effect = {
	# $TARGET_SIDE$ = "player" or "opponent"
	# Reads scope:current_move.var:checks_guard. Writes scope:contact_result
	# (0 = hit, 1 = blocked) and the matching result_code.
	if = {
		limit = {
			scope:current_move = { has_variable = checks_guard }
			has_variable = pod_combat_$TARGET_SIDE$_is_guarding
		}
		remove_variable = pod_combat_$TARGET_SIDE$_is_guarding
		pod_combat_set_result_code_effect = { CODE = 1 }
		set_variable = { name = pod_combat_last_damage value = 0 }
		save_scope_value_as = { name = contact_result value = 1 }
	}
	else = {
		save_scope_value_as = { name = contact_result value = 0 }
	}
}

# ── DAMAGE PAYLOAD ─────────────────────────────────────────────────
# Fires only when scope:contact_result = 0 (hit). The contact resolver
# already handled spirit gate, stance, hit roll, and guard, so the
# payload is just the PoD damage call. Routes through bypass-guard
# helper since the resolver explicitly handles guard above.
pod_combat_apply_damage_payload_effect = {
	# $TARGET_SIDE$ = "player" or "opponent"
	# $AMOUNT$      = damage amount (raw, pre-soak)
	# $REASON$      = PoD death_reason suffix
	pod_combat_deal_hp_damage_bypass_guard_effect = {
		TARGET_SIDE = $TARGET_SIDE$
		AMOUNT      = $AMOUNT$
		REASON      = $REASON$
	}
}


pod_combat_deal_morale_damage_effect = {
	# $TARGET_SIDE$ = "player" or "opponent"
	# $AMOUNT$ = morale damage (bypasses HP entirely)
	# Mental damage: auto-hit, no guard, no soak, but dodge still works.
	# On top of reducing the workspace morale bar, it also applies
	# Mental Strain to the target character at a configurable morale:stress
	# ratio (pod_combat_morale_stress_multiplier). Keeps the combat
	# minigame's psychological attrition in lockstep with PoD's stress
	# track so the target feels the pressure outside of combat too.

	# Stealth → dodge consume on the target. Sets scope:stance_consumed
	# (1 = stealth fired, 2 = dodge fired, 0 = neither).
	pod_combat_consume_target_stance_effect = { TARGET_SIDE = $TARGET_SIDE$ }
	if = {
		limit = { scope:stance_consumed = 0 }
		# Cap the workspace morale delta at the smaller of (requested
		# amount, current morale) so mental-strain stress doesn't
		# over-count past a near-broken target — e.g. 1 morale
		# remaining + 5 morale damage → 1 actual damage, 5 stress not
		# 25.
		save_scope_value_as = {
			name = morale_actually_lost
			value = {
				value = $AMOUNT$
				max   = var:pod_combat_$TARGET_SIDE$_morale
			}
		}
		save_scope_value_as = {
			name = new_morale
			value = {
				value = var:pod_combat_$TARGET_SIDE$_morale
				subtract = $AMOUNT$
				min = 0
			}
		}
		set_variable = { name = pod_combat_$TARGET_SIDE$_morale value = scope:new_morale }
		# Capture the morale-damage amount in its own variable so we
		# don't clobber pod_combat_last_damage (which holds the HP
		# damage amount on damage+morale-damage moves).
		set_variable = { name = pod_combat_last_morale_damage_amount value = $AMOUNT$ }
		# Code 9 = morale-only outcome. Only set if no primary HP-damage
		# code has fired — a damage+morale move keeps the HP code as
		# primary; the morale amount rides along as a separate field.
		pod_combat_set_result_code_if_unset_effect = { CODE = 9 }

		# Mirror the morale loss onto the target as Mental Strain.
		# Only applies when the workspace char alias is populated (it
		# always is during active combat — the aliases are written in
		# the workspace-load effects — but guard for safety in case an
		# odd path calls this effect outside combat).
		if = {
			limit = {
				has_variable = pod_combat_$TARGET_SIDE$_char
				scope:morale_actually_lost > 0
			}
			save_scope_value_as = {
				name = mental_strain_from_morale
				value = {
					value = scope:morale_actually_lost
					multiply = pod_combat_morale_stress_multiplier
				}
			}
			var:pod_combat_$TARGET_SIDE$_char = {
				POD_static_stress_gain = { COUNT = scope:mental_strain_from_morale }
			}
		}
		# Victory check runs at the end of pod_combat_save_workspace_effect.
	}
}

pod_combat_restore_morale_effect = {
	# $SIDE$ = "player" or "opponent"
	# $AMOUNT$ = morale to restore
	save_scope_value_as = {
		name = new_morale
		value = {
			value = var:pod_combat_$SIDE$_morale
			add = $AMOUNT$
			max = pod_combat_max_morale
		}
	}
	set_variable = { name = pod_combat_$SIDE$_morale value = scope:new_morale }
	# Parallel-track in its own variable so it doesn't clobber the
	# HP damage amount captured in pod_combat_last_damage.
	set_variable = { name = pod_combat_last_morale_heal_amount value = $AMOUNT$ }
	# Code 11 = morale heal. Only set if no primary code has fired —
	# damage+morale-heal hybrids keep the damage code as primary.
	pod_combat_set_result_code_if_unset_effect = { CODE = 11 }
}

# ─── AUTO-RESOLVE ──────────────────────────────────────────────────

pod_combat_auto_resolve_effect = {
	# Save PC scope alias to avoid bare `root` references in set_variable
	save_scope_as = combat_pc

	# Simple prowess comparison with random factor
	save_scope_value_as = {
		name = player_score
		value = {
			value = prowess
			add = {
				integer_range = { min = 0 max = prowess }
			}
		}
	}
	var:pod_combat_opponent = {
		save_scope_value_as = {
			name = opponent_score
			value = {
				value = prowess
				add = {
					integer_range = { min = 0 max = prowess }
				}
			}
		}
	}

	if = {
		limit = { scope:player_score >= scope:opponent_score }
		set_variable = { name = pod_combat_winner value = scope:combat_pc }
		set_variable = { name = pod_combat_loser value = var:pod_combat_opponent }
	}
	else = {
		set_variable = { name = pod_combat_winner value = var:pod_combat_opponent }
		set_variable = { name = pod_combat_loser value = scope:combat_pc }
	}

	# Set final HP/Morale for display
	set_variable = { name = pod_combat_target_morale value = 0 }
	set_variable = { name = pod_combat_is_over value = yes }
	# Combat window stays open — player clicks "End Combat" to dismiss
}

# ─── COMBAT LOG ────────────────────────────────────────────────────
# Rolling log of combat actions, stored as POD_combat_log_entry
# story_cycles in the PC's pod_combat_log variable list. Each entry
# captures a snapshot of one resolved move:
#
#   move        — story scope of the executed move. The GUI reads
#                 the move's name, attack_type, damage_source,
#                 targeting, and stance/buff/debuff flags directly
#                 from this scope — no per-entry duplication needed.
#   actor       — character ref of the combatant who acted
#   target      — character ref of the targeted combatant (optional;
#                 self-targeting and team_* moves may skip)
#   damage      — int amount of HP / morale change resolved
#   result_code — 0..12, see PODCombatLogResultText custom_loc for
#                 the full meaning table
#
# The list is appended-to and never trimmed during combat; cleanup at
# combat end (pod_combat_cleanup_effect) ends_story on every entry and
# clears the list. Each combat starts fresh.
#
# Render path: a single datamodel template in pod_combat_window.gui
# iterates pod_combat_log. There is no per-entry GUI block — every
# row is rendered by the same template reading the entry's vars
# directly via Scope.Var('move').Story.MakeScope etc.

pod_combat_push_log_effect = {
	# Build a new POD_combat_log_entry story holding this turn's
	# resolved-action snapshot and append it to the PC's
	# pod_combat_log variable list. The GUI's datamodel iterates
	# this list and reads each entry's vars directly.
	#
	# Parameters:
	#   $MOVE_STORY$ — the move story_cycle being logged. Threaded
	#                  explicitly from POD_execute_combat_move §16
	#                  (instead of via a host-var handoff) so push_log
	#                  has no implicit shared-state dependency on its
	#                  caller. A future caller (e.g. per-target AoE
	#                  log entries) can pass any move story without
	#                  touching combat_host's vars.
	#
	# Inputs read off the host's per-turn state:
	#   var:pod_combat_actor_char              — actor character ref
	#   var:pod_combat_target_char           — target character ref (optional)
	#   var:pod_combat_last_damage             — HP damage rolled amount
	#   var:pod_combat_last_heal_amount        — HP heal applied amount
	#   var:pod_combat_last_morale_damage_amount  — morale damage amount
	#   var:pod_combat_last_morale_heal_amount    — morale heal amount
	#   var:pod_combat_last_retal_damage       — retaliation damage (when armed)
	#   var:pod_combat_last_result_code        — single-int outcome 0..12.
	#       Damage helpers, retaliation, morale and heal pipelines all
	#       write here directly; push_log reads it unconditionally and
	#       falls back to 8 (utility) only when no helper wrote a value.
	#       Mapping lives in COMBAT_REFERENCE.info §4.
	#
	# Result code map (must stay in sync with PODCombatLogResultText
	# customizable_loc and pod_combat_log_result_text_* loc keys):
	#   0  — HP damage (-X [wound-box])
	#   1  — guard-blocked
	#   2  — soaked
	#   3  — missed
	#   5  — dodged
	#   6  — HP damage + retaliation triggered (-X [box] retaliated)
	#   7  — HP damage + retaliation soaked (-X [box] counter soaked)
	#   8  — utility move applied (no damage)
	#   9  — morale damage only (-X Morale)
	#  10  — HP healed (+X [box])
	#  11  — morale healed (+X Morale)
	#  12  — stealth-triggered auto-miss (separate from rcode 5 dodged
	#         so the log line links to the Stealth concept, not Dodge)
	#
	# Code 4 is reserved (originally "countered") — never written,
	# never dispatched, no loc. Kept reserved so a future use can
	# claim the slot without renumbering the rest.
	#
	# Story is owned by Charon (POD_wraith_01) — the same off-map anchor
	# used by combat-move stories. Owning all log entries on a guaranteed-
	# alive character avoids "owner died" cleanup races, and matches the
	# pattern in POD_init_combat_move. The PC-side pod_combat_log list
	# just holds references.

	# Capture the PC scope explicitly. By the time push_log fires, the
	# current scope IS the combat host (PC) — for SGUI turns root=PC,
	# and the AI-turn chain switches into PC scope (scope:combat_host)
	# before calling POD_execute_combat_move. Saving here makes the
	# rest of this effect independent of `root` validity, which the
	# AI-turn call chain doesn't always preserve.
	save_scope_as = combat_host

	character:POD_wraith_01 = {
		create_story = {
			type = POD_combat_log_entry
			save_scope_as = new_log_entry
		}
	}
	scope:new_log_entry = {
		# Move story comes through directly as the MOVE_STORY parameter
		# — no host-var handoff. Caller-passes-explicit keeps the
		# contract local and lets push_log fire from any path without
		# requiring a host-var pre-stamp.
		set_variable = { name = move value = $MOVE_STORY$ }
		if = {
			limit = { scope:combat_host = { has_variable = pod_combat_actor_char } }
			set_variable = { name = actor value = scope:combat_host.var:pod_combat_actor_char }
		}
		if = {
			limit = { scope:combat_host = { has_variable = pod_combat_target_char } }
			set_variable = { name = target value = scope:combat_host.var:pod_combat_target_char }
		}
		if = {
			limit = { scope:combat_host = { has_variable = pod_combat_last_damage } }
			set_variable = { name = damage value = scope:combat_host.var:pod_combat_last_damage }
		}
		else = {
			set_variable = { name = damage value = 0 }
		}
		# Parallel amount fields — heal / morale damage / morale heal
		# now write to their own variables instead of clobbering
		# pod_combat_last_damage, so damage+heal / damage+morale moves
		# preserve the HP-damage value above while still carrying the
		# secondary amount through to the log entry.
		if = {
			limit = { scope:combat_host = { has_variable = pod_combat_last_heal_amount } }
			set_variable = { name = heal_amount value = scope:combat_host.var:pod_combat_last_heal_amount }
		}
		if = {
			limit = { scope:combat_host = { has_variable = pod_combat_last_morale_damage_amount } }
			set_variable = { name = morale_damage_amount value = scope:combat_host.var:pod_combat_last_morale_damage_amount }
		}
		if = {
			limit = { scope:combat_host = { has_variable = pod_combat_last_morale_heal_amount } }
			set_variable = { name = morale_heal_amount value = scope:combat_host.var:pod_combat_last_morale_heal_amount }
		}
		# Retaliation damage — captured by pod_combat_apply_retaliation_helper
		# right after the inner damage call but BEFORE the helper
		# restores pod_combat_last_damage to the original attack's
		# value. Lets the loc dispatch render the counter-attack's hit
		# even when the original attack was guard-blocked / soaked
		# (entry.damage = 0 in that case).
		if = {
			limit = { scope:combat_host = { has_variable = pod_combat_last_retal_damage } }
			set_variable = { name = retal_damage value = scope:combat_host.var:pod_combat_last_retal_damage }
		}
		# Result code resolution. Damage helpers and heal pipelines
		# write pod_combat_last_result_code directly during move
		# execution (see the result-code helpers at the top of this
		# file); push_log just reads the final value here. The
		# helpers handle priority via their write order:
		#   - HP damage helpers SET the code unconditionally (one
		#     branch per outcome, mutually exclusive within a single
		#     call).
		#   - Retaliation PROMOTES 0→6 / 2→7 if it fires.
		#   - Morale damage / heal / morale-heal use the
		#     "set if unset" helper so they only become primary when
		#     no HP-damage code has fired.
		# The fallback to code 8 (utility) covers moves that didn't
		# write any code — pure stance / buff / debuff / AP-mod moves.
		if = {
			limit = {
				scope:combat_host = {
					has_variable = pod_combat_last_result_code
					var:pod_combat_last_result_code >= 0
				}
			}
			set_variable = { name = result_code value = scope:combat_host.var:pod_combat_last_result_code }
		}
		else = {
			# No primary code fired — utility outcome.
			set_variable = { name = result_code value = 8 }
		}
	}
	# Insert at the FRONT of pod_combat_log so the GUI shows newest
	# entries at the top. CK3 variable lists only support append, so
	# we rebuild via a temp list:
	#   1. Snapshot the existing list (already in newest-first order
	#      from prior pushes, so iteration is newest -> oldest).
	#   2. Clear the main list.
	#   3. Add the NEW entry to main (now [new]).
	#   4. Re-append every snapshotted item (newest -> oldest order),
	#      yielding main = [new, prev_newest, ..., oldest].
	# Cleared entirely at combat end via pod_combat_cleanup_effect.
	#
	# Scope handling: every_in_list switches `this` to each iterated
	# item. To add the item to combat_host's list (not to the item's
	# own list), the inner block explicitly scope-switches back to
	# combat_host, with `prev` referring to the iterated item.
	scope:combat_host = {
		# Step 1: snapshot existing list to temp (preserves newest-first order)
		every_in_list = {
			variable = pod_combat_log
			scope:combat_host = {
				add_to_variable_list = { name = pod_combat_log_rebuild target = prev }
			}
		}
		# Step 2 + 3: clear main, add new entry first
		clear_variable_list = pod_combat_log
		add_to_variable_list = { name = pod_combat_log target = scope:new_log_entry }
		# Step 4: re-append snapshot in original order
		every_in_list = {
			variable = pod_combat_log_rebuild
			scope:combat_host = {
				add_to_variable_list = { name = pod_combat_log target = prev }
			}
		}
		# Step 5: clean up temp
		clear_variable_list = pod_combat_log_rebuild
	}
}

# ─── POST-COMBAT HEALTH FINALIZATION ─────────────────────────────
# Despite the historical name, this effect does NOT apply combat damage
# — wounds are written directly to each combatant's PoD wound track
# during move execution (the damage helpers in pod_combat_effects.txt
# call POD_increase_wounds_* immediately, not deferred to end-of-combat).
# This effect runs once at combat end (BEFORE cleanup) and only handles
# fatality-mode finalization:
#   - fatality = no  →  cap every populated slot's wound track to leave
#                       at least one empty box (so the duel doesn't kill
#                       anyone), then strip the pod_combat_nonfatal_active
#                       flag so future damage outside this combat can
#                       still kill normally.
#   - fatality = always / default / practice  →  no-op (combat damage
#                       has already persisted, and these modes either
#                       want kills to land or skipped HP writes entirely).

pod_combat_apply_nonfatal_effect = {
	# Damage is applied to each combatant's PoD wound vars
	# (POD_health_damage_num / POD_health_aggravated_damage_num) live
	# during move execution — this effect does NOT apply combat damage.
	# Its name is a fossil from an earlier deferred-writeback model
	#
	# Fatality modes:
	#   practice — no HP writes at all during combat (the damage +
	#              heal helpers short-circuit when the context story's
	#              var:fatality = flag:practice, so only morale state
	#              changes). Nothing to restore / cap post-combat.
	#   no       — PoD's inline death check is gated during combat by
	#              the pod_combat_nonfatal_active character flag (see
	#              common/scripted_triggers/POD_combat/pod_combat_death_gate_overrides.txt).
	#              Wounds still accumulate but nobody dies. At combat
	#              end we cap every combatant's wound track to leave at
	#              least one empty box, THEN strip the flag so the
	#              character can die from wounds normally in the
	#              future (just not from this duel).
	#   default  — no-op (damage already persisted via tight integration).

	if = {
		limit = {
			has_variable = pod_combat_sc_context
			var:pod_combat_sc_context.var:fatality = flag:no
		}
		# Iterate every slot story on the host. Inside the iteration body
		# the current scope is the slot story, so `has_variable = slot_char`
		# checks the slot story's vars and `var:slot_char` switches into
		# the participant. Ensures combatant in each filled slot keeps at
		# least one empty health box; if their track is full, peel one
		# superficial (preferred) or aggravated wound. Damage was applied
		# live during the fight; this is just the nonfatal cap pass.
		every_in_list = {
			variable = pod_combat_slots
			limit = { has_variable = slot_char }
			var:slot_char = {
				save_scope_value_as = {
					name  = nf_empty
					value = POD_empty_damage_boxes
				}
				if = {
					limit = { scope:nf_empty <= 0 }
					if = {
						limit = {
							has_variable = POD_health_damage_num
							var:POD_health_damage_num >= 1
						}
						change_variable = { name = POD_health_damage_num add = -1 }
					}
					else_if = {
						limit = {
							has_variable = POD_health_aggravated_damage_num
							var:POD_health_aggravated_damage_num >= 1
						}
						change_variable = { name = POD_health_aggravated_damage_num add = -1 }
					}
					POD_update_wound_trait_xp = yes
				}
			}
		}
		# Strip the nonfatal flag AFTER the cap — order matters.
		pod_combat_clear_nonfatal_flag_effect = yes
	}
}

# ─── AGGRAVATED DAMAGE EFFECTS ───────────────────────────────────
# Aggravated wounds route to PoD's POD_health_aggravated_damage_num
# via the same live-application path the superficial pipeline uses
# (POD_increase_wounds_hostile_effect → POD_apply_wound_boxes →
# the agg/sup split inside that helper).
#
# Soakability: aggravated damage IS soakable, but only by characters
# with the specific modifiers PoD's POD_can_soak_aggravated_damage
# trigger lists (Mokole undying body, Demon infernal fortitude,
# Dragon's ichor temp, Autumn's armor). For unflagged targets, the
# soak roll is skipped because POD_damage_type_is_soakable returns
# no for aggravated death reasons. Combat rcode 2 ("soaked") will
# fire for those modifier-equipped targets when the soak roll wins,
# matching the superficial soaked semantics.
#
# Push-out interaction: aggravated damage on an already-wounded target
# converts existing superficial wounds to aggravated rather than
# adding to total boxes. POD_total_damage_boxes can stay flat across
# the call, so pod_combat_route_to_pod_damage tracks superficial and
# aggravated deltas separately to avoid mistakenly classifying a
# push-out as "soaked" — see the snapshot block in that effect for
# the canonical writeup.

# ─── ELEMENTAL / SOURCE DISPATCH (lives in pod_combat_move_effects.txt) ──
# POD_combat_dispatch_by_source is a flat flag → REASON map; PoD's
# wound system handles splat-weakness multipliers, aggravated
# classification, and immunity off the REASON suffix. The combat layer
# just supplies the flag.
#
# To trace the live path: pod_combat_move_effects.txt's
# POD_combat_apply_single_hp_damage → POD_combat_dispatch_by_source
# → POD_combat_dispatch_damage_pipeline → pod_combat_resolve_contact_effect
# → pod_combat_apply_damage_payload_effect → pod_combat_deal_hp_damage_bypass_guard_effect
# → pod_combat_route_to_pod_damage → POD_increase_wounds_hostile_effect.

# ─── HEALING EFFECTS ────────────────────────────────────────────────
# Two variants the move dispatcher (pod_combat_move_effects.txt's
# hp_heal block) routes to based on the move's heal_type variable:
#   Superficial — mundane/quick healing (bandages, Regeneration, feeding).
#                 Cannot heal through aggravated wounds; HP is capped at
#                 (max_hp - agg_damage) so aggravated damage remains a
#                 lasting wound until it is specifically cured. Default
#                 when heal_type is blank or unset.
#   Aggravated  — rare / potent healing. Restores HP AND reduces the agg_damage 
#                 counter by the amount healed, so aggravated wounds are properly
#                 cleared for write-back to the real PoD health track.
#                 Routed when heal_type = flag:aggravated.

# Heal effects route through PoD's heal API so superficial / aggravated
# box counts stay coherent with the wound system. The new helper mirrors
# the damage-side pattern: scope-switch to the target, call PoD, re-
# derive workspace HP from the updated state.

pod_combat_route_to_pod_heal = {
	# $SIDE$ = "player" or "opponent"
	# $AGGRAVATED$ = yes / no — yes calls POD_heal_aggravated_wound_effect
	#                (heals aggravated boxes; superficial overflow can
	#                convert into superficial per PoD rules), no calls
	#                POD_heal_superficial_wound_effect (heals superficial
	#                only, cannot touch aggravated damage).
	# $AMOUNT$ = boxes to heal (matches the scale combat uses for HP —
	#            combat HP is wound-track-aligned, so 1 HP = 1 box).
	#
	# Sets scope:heal_applied so downstream logic can read the actual
	# amount healed (bounded by current wound counts).

	# Practice mode: no wounds were inflicted on the way in, so there's
	# nothing to heal. Skip the PoD heal call entirely and return
	# scope:heal_applied = 0. Keeps log / tooltip flags consistent with
	# the damage-side practice guard.
	if = {
		limit = {
			has_variable = pod_combat_sc_context
			var:pod_combat_sc_context.var:fatality = flag:practice
		}
		save_scope_value_as = { name = heal_applied value = 0 }
		# Parallel-track the heal amount in its own variable. Practice
		# mode heals nothing — last_heal_amount = 0 lets push_log /
		# the result-text dispatcher render "—" instead of "+0 HP".
		set_variable = { name = pod_combat_last_heal_amount value = 0 }
		# Code 10 = HP heal. Only set if no primary code has fired —
		# damage+heal moves keep the damage code; the heal amount
		# rides along as the heal_amount field.
		pod_combat_set_result_code_if_unset_effect = { CODE = 10 }
	}
	else = {
		var:pod_combat_$SIDE$_char = {
			save_scope_value_as = { name = pre_total_wounds value = POD_total_damage_boxes }
			if = {
				limit = { always = $AGGRAVATED$ }
				POD_heal_aggravated_wound_effect = { COUNT = $AMOUNT$ }
			}
			else = {
				POD_heal_superficial_wound_effect = { COUNT = $AMOUNT$ }
			}
			save_scope_value_as = { name = post_total_wounds value = POD_total_damage_boxes }
		}
		save_scope_value_as = {
			name = heal_applied
			value = {
				value = scope:pre_total_wounds
				subtract = scope:post_total_wounds
				min = 0
			}
		}
		# No workspace HP / agg to update — pod_combat_$SIDE$_hp_value and
		# pod_combat_$SIDE$_agg_damage_value SVs derive from the char on read.
		# Parallel-track the actual healed amount in its own variable so
		# damage+heal moves preserve the HP-damage value in
		# pod_combat_last_damage for the log line.
		set_variable = { name = pod_combat_last_heal_amount value = scope:heal_applied }
		# Code 10 = HP heal. Only set if no primary code has fired —
		# damage+heal moves keep the damage code; the heal amount
		# rides along as the heal_amount field.
		pod_combat_set_result_code_if_unset_effect = { CODE = 10 }
	}
}

pod_combat_restore_hp_superficial_effect = {
	# $SIDE$ = "player" or "opponent"
	# $AMOUNT$ = superficial HP to restore
	# Heals only superficial wound boxes — aggravated damage is untouched.
	# PoD's POD_heal_damage clamps to current superficial count, so
	# over-healing past max_hp is handled inside the wound system.
	pod_combat_route_to_pod_heal = {
		SIDE       = $SIDE$
		AGGRAVATED = no
		AMOUNT     = $AMOUNT$
	}
}

pod_combat_restore_hp_aggravated_effect = {
	# $SIDE$ = "player" or "opponent"
	# $AMOUNT$ = aggravated HP to restore
	# Heals aggravated wound boxes first (rare/potent healing — Obeah
	# Mend, Fortitude high-tier, soul-drain). PoD's heal logic lets
	# aggravated healing "push out" into superficial if agg is empty.
	pod_combat_route_to_pod_heal = {
		SIDE       = $SIDE$
		AGGRAVATED = yes
		AMOUNT     = $AMOUNT$
	}
}

# ─── END COMBAT (INLINE) ──────────────────────────────────────────
# Runs inline from the victory check, auto-resolve, or forfeit path
# no event popup, no scope boundary. For duel-driven combat
# (signaled by pod_combat_sc_context pointing to a POD_combat_sc_context
# story), the dispatcher reconstructs PoD scopes from the story and
# fires the follow-up event.
# For standalone/debug combat, shows a result toast.

pod_combat_end_combat_effect = {
	# Re-entry guard: the skill button's GUI execute and the "End Combat"
	# button both call this when pod_combat_is_over is set. The first call
	# removes pod_combat_active (SC-driven) or runs full cleanup (standalone),
	# so subsequent calls must be no-ops.
	if = {
		limit = { NOT = { has_variable = pod_combat_active } }
		# Already ended — nothing to do
	}
	else = {

	# Post-combat finalization — caps wound tracks for nonfatal duels
	# and strips the nonfatal-active flag. Damage itself was already
	# applied during move execution; this is purely fatality-mode
	# bookkeeping. See the header comment on the effect for details.
	pod_combat_apply_nonfatal_effect = yes

	# Fire the duel-end on_action — our formal extension point for
	# combat conclusion, parallel to PoD's POD_on_combat_end_winner /
	# POD_on_combat_end_loser for army combat. Handlers run with slot
	# vars + workspace still intact so they can iterate combatants.
	# Current responsibility: strip discipline battle modifiers from
	# all combatants (moved out of pod_combat_cleanup_effect so
	# submods can extend without touching our cleanup path). See
	# pod_combat_on_actions.txt for the handler definition.
	#
	# Saved scopes pod_combat_winner / pod_combat_loser are set first
	# so the duel-end effect (and any submod that overrides it) has
	# named access to the outcome characters.
	if = {
		limit = { has_variable = pod_combat_winner }
		var:pod_combat_winner = { save_scope_as = pod_combat_winner }
	}
	if = {
		limit = { has_variable = pod_combat_loser }
		var:pod_combat_loser = { save_scope_as = pod_combat_loser }
	}
	# pod_combat_duel_end_effect is a scripted-effect equivalent of an
	# on_action handler. Inlining as a scripted effect preserves all the
	# cleanup behavior.
	pod_combat_duel_end_effect = yes

	# If launched via configure_start_single_combat_effect, fire the
	# post-combat dispatch event immediately. The event reads the
	# POD_combat_sc_context story through the PC's pod_combat_sc_context
	# pointer var, restores all scopes from the story, applies death,
	# fires the follow-up event, then ends the story + removes the
	# pointer in cleanup.
	#
	# The combat window closes immediately (pod_combat_active removed).
	# The context story persists on Charon until pod_combat.9999 fires
	# the follow-up and cleanup ends it.
	if = {
		limit = { has_variable = pod_combat_sc_context }
		# Close the combat window immediately but keep pod_combat_is_over
		# as a re-entry guard (the victory check gates on NOT is_over).
		# pod_combat.9999 handles full cleanup after firing the follow-up.
		remove_variable = pod_combat_active
		trigger_event = { id = pod_combat.9999 }
	}

	# For standalone/debug combat (no duel override), show a toast
	# and clean up immediately (no scope bridge needed).
	else = {
		if = {
			limit = {
				has_variable = pod_combat_winner
				var:pod_combat_winner ?= root
			}
			send_interface_toast = {
				title = pod_combat_result_victory_toast
				left_icon = root
			}
		}
		else = {
			send_interface_toast = {
				title = pod_combat_result_defeat_toast
				left_icon = root
			}
		}
		pod_combat_cleanup_effect = yes
	}

	} # end re-entry guard
}

# ─── DUEL-END EFFECT ────────────────────────────────────────────────
# Scripted-effect equivalent of an on_action handler — runs once at
# combat end (called by pod_combat_end_combat_effect BEFORE
# pod_combat_cleanup_effect strips the slot vars). Two jobs per
# combatant slot:
#   1. Strip discipline battle modifiers added by
#      POD_{enemy,ally}_knight_set_discipline_battle_effect at init,
#      so discipline bonuses don't leak outside of the duel.
#   2. Remove the pod_combat_duel_active state marker so other
#      systems (feeding, schemes, travel) stop deferring on this
#      character. The modifier has a safety-net 30-day expiry, but
#      the explicit removal keeps state clean.
#
# Submods wanting to extend can override this scripted effect via
# load order, or wrap it with their own pre/post effects.

pod_combat_duel_end_effect = {
	# Iterate every slot story registered to the host's pod_combat_slots
	# list. Inside the iteration body, current scope IS the slot story,
	# so `has_variable = slot_char` checks the slot story's vars and
	# `var:slot_char` switches into the participant's character scope
	# without the host's per-slot pointer var. The list is populated by
	# pod_combat_create_one_slot_story_effect and cleared by
	# pod_combat_clear_slot_stories_effect — empty during tooltip-build,
	# so this iteration is automatically a no-op there.
	every_in_list = {
		variable = pod_combat_slots
		limit = { has_variable = slot_char }
		var:slot_char = {
			POD_clear_discipline_battle_effect = yes
			remove_character_modifier = pod_combat_duel_active
		}
	}

	# Also remove the nonfatal-mode flag from every combatant. This is
	# the gate on POD's death pipeline during combat; it needs to come
	# off so future fatal damage (outside the minigame) resolves
	# normally. Extends to all slots even if they never took damage —
	# the per-slot remove is cheap and idempotent.
	pod_combat_clear_nonfatal_flag_effect = yes
}

# ─── CLEANUP ───────────────────────────────────────────────────────

pod_combat_cleanup_effect = {
	# Release all tier variables (both round and combat duration) across
	# all 10 slots. The underlying stat vars (soak, hit_chance) are removed
	# by pod_combat_group_cleanup_effect separately.
	pod_combat_clear_combat_tiers_effect = yes

	# Discipline battle modifiers and the duel-active marker are
	# cleared upstream by pod_combat_duel_end_effect, called inline
	# from pod_combat_end_combat_effect BEFORE this cleanup runs.
	# (We had wanted fire_on_action for built-in extensibility, but
	# the targeted engine version doesn't recognize that effect — see
	# the comment on the inline call site above. Submods that want to
	# extend can override pod_combat_duel_end_effect via load order.)

	# Remove all combat variables from the player character
	remove_variable = pod_combat_active
	remove_variable = pod_combat_is_over
	# ── Single-combat caller-context teardown ──
	# All sc_* state (initiator/attacker/defender refs, rule flags,
	# follow-up event id, parent-event alias snapshots) lives on a
	# POD_combat_sc_context story_cycle anchored on Charon. End the
	# story to atomically drop every variable on it, then remove the
	# PC-side pointer. This block runs harmlessly for standalone/debug
	# combat (no context story exists, the if-guard short-circuits).
	#
	# Adding a new aliased scope to the launcher's whitelist no longer
	# requires a matching remove_variable here — the story takes care
	# of itself. See pod_combat_sc_stash_parent_aliases_effect in
	# z_Combat-PoD_override_single_combat_effects.txt for the whitelist.
	if = {
		limit = { has_variable = pod_combat_sc_context }
		var:pod_combat_sc_context = { end_story = yes }
	}
	remove_variable = pod_combat_sc_context
	remove_variable = pod_combat_opponent
	remove_variable = pod_combat_actor_char
	remove_variable = pod_combat_target_char
	remove_variable = pod_combat_round
	remove_variable = pod_combat_current_ap
	remove_variable = pod_combat_current_turn_side
	remove_variable = pod_combat_actor_soak
	remove_variable = pod_combat_actor_morale
	remove_variable = pod_combat_target_soak
	remove_variable = pod_combat_target_morale
	remove_variable = pod_combat_actor_is_guarding
	remove_variable = pod_combat_target_is_guarding
	remove_variable = pod_combat_winner
	remove_variable = pod_combat_loser
	remove_variable = pod_combat_last_damage
	# pod_combat_ai_last_move / _prev_move / _prev2_move are NOT
	# host-side vars — the selector writes them on the AI character
	# (scope:ai_actor.var:...). They're cleared per-fight by
	# pod_combat_group_cleanup_effect (combat exit) and
	# pod_combat_init_slot_from_character_effect (combat entry).
	remove_variable = pod_combat_ai_saved_enemy_target
	remove_variable = pod_combat_ai_last_swap_target
	remove_variable = pod_combat_last_result_code
	remove_variable = pod_combat_last_heal_amount
	remove_variable = pod_combat_last_morale_damage_amount
	remove_variable = pod_combat_last_morale_heal_amount
	remove_variable = pod_combat_last_retal_damage
	# Damage system tracking
	remove_variable = pod_combat_actor_base_soak
	remove_variable = pod_combat_target_base_soak
	remove_variable = pod_combat_actor_base_hit_chance
	remove_variable = pod_combat_target_base_hit_chance
	remove_variable = pod_combat_actor_energy_used
	remove_variable = pod_combat_target_energy_used
	remove_variable = pod_combat_actor_is_dodging
	remove_variable = pod_combat_target_is_dodging
	remove_variable = pod_combat_actor_is_stealthed
	remove_variable = pod_combat_target_is_stealthed
	# Per-slot stance / init_bonus / ap_drain / retaliate vars live on
	# the slot stories; pod_combat_clear_slot_stories_effect end_story-s
	# every slot story at combat end, taking those vars with them. We
	# only need to wipe the workspace mirrors here.
	remove_variable = pod_combat_actor_init_bonus
	remove_variable = pod_combat_target_init_bonus
	remove_variable = pod_combat_actor_ap_drain
	remove_variable = pod_combat_target_ap_drain
	remove_variable = pod_combat_actor_retaliate_hit
	remove_variable = pod_combat_target_retaliate_hit
	remove_variable = pod_combat_actor_retaliate_miss
	remove_variable = pod_combat_target_retaliate_miss
	remove_variable = pod_combat_actor_retaliate_hit_agg
	remove_variable = pod_combat_target_retaliate_hit_agg
	remove_variable = pod_combat_actor_retaliate_miss_agg
	remove_variable = pod_combat_target_retaliate_miss_agg
	remove_variable = pod_combat_actor_prowess
	remove_variable = pod_combat_target_prowess
	remove_variable = pod_combat_actor_hit_chance
	remove_variable = pod_combat_target_hit_chance
	remove_variable = pod_combat_current_hit_chance
	# Combat log story_cycle entries — end_story on every entry, then
	# clear the list. Also clear the rebuild scratch list (used by
	# push_log_effect to maintain newest-first ordering); defensive in
	# case a push errored mid-rebuild and left items orphaned.
	every_in_list = {
		variable = pod_combat_log
		end_story = yes
	}
	clear_variable_list = pod_combat_log
	clear_variable_list = pod_combat_log_rebuild
	# Clean up group combat slot variables
	pod_combat_group_cleanup_effect = yes
}
