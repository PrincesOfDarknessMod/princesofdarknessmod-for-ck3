# Damage helpers, soak, retaliation dispatcher, push_log, and the
# unified contact resolver. Companion to pod_combat_move_effects.txt;
# both are called from POD_execute_combat_move. Three-axis result
# model, sentinel/priority semantics, and the reaction architecture

# ─── ATTACK RESULT WRITERS ──────────────────────────────────────────

pod_combat_set_attack_result_effect = {
	# $CODE$ = attack_result flag: hp_damage | soaked | missed | dodged
	#          | stealthed | guard_blocked.
	# Writes outcome=attack + attack_result in lockstep so readers can
	# gate on outcome=attack to know attack_result is meaningful.
	set_variable = { name = pod_combat_last_outcome       value = flag:attack }
	set_variable = { name = pod_combat_last_attack_result value = flag:$CODE$ }
}

# ─── OUTCOME WRITERS (if-unset) ─────────────────────────────────────
# Rider paths (morale damage, HP heal, morale heal): a combined move
# (damage+heal, damage+morale) keeps the attack outcome as primary.

pod_combat_set_outcome_if_unset_effect = {
	# $CODE$ = outcome flag: heal | morale_damage | morale_healed.
	#          (attack is owned by pod_combat_set_attack_result_effect.)
	if = {
		limit = { NOT = { has_variable = pod_combat_last_outcome } }
		set_variable = { name = pod_combat_last_outcome value = flag:$CODE$ }
	}
}

# ─── RETAL FIRED OVERLAY ────────────────────────────────────────────
# Independent of outcome/attack_result: a missed main attack with a
# landed retal sets attack_result=missed AND retal_fired=1, and the
# log dispatcher's retal rider renders both axes.
pod_combat_set_retal_fired_effect = {
	set_variable = { name = pod_combat_last_retal_fired value = 1 }
}

pod_combat_clear_last_result_vars_effect = {

	remove_variable = pod_combat_last_retal_fired
	remove_variable = pod_combat_last_retal_agg

	remove_variable = pod_combat_last_retal_source
	remove_variable = pod_combat_last_heal_amount
	remove_variable = pod_combat_last_morale_damage_amount
	remove_variable = pod_combat_last_morale_heal_amount
	remove_variable = pod_combat_last_retal_damage
	remove_variable = pod_combat_last_damage
	remove_variable = pod_combat_last_self_damage
	remove_variable = pod_combat_aoe_phase_pushed_rows
	remove_variable = pod_combat_aoe_iter_burst_index
}

# ─── RESOURCE MANIPULATION ────────────────────────────────────────

pod_combat_spend_ap_effect = {
	# $AMOUNT$ = AP to spend
	change_variable = {
		name = pod_combat_current_ap
		add = {
			value = $AMOUNT$
			multiply = -1
		}
	}
}

# ── REACTION PHASE ─────────────────────────────────────────────
# §15 of POD_execute_combat_move. The dispatcher
# pod_combat_check_retaliation_effect walks the workspace
# pod_combat_$TARGET_SIDE$_retaliate_* slots and fires each one
# whose §6 outcome matched its trigger condition. For the system
# overview, the recipe to add a new reaction class, and the
# damage_type inheritance rules see COMBAT_REFERENCE.info §6.

# Generic reaction-firing helper. Called once per reaction slot pair
# from pod_combat_check_retaliation_effect. Encapsulates the limit-
# then-fire-then-clear pattern so each retaliation type is one block.
pod_combat_apply_reaction_if_triggered_effect = {
	# $TRIGGER_VAR$       = workspace var holding the reaction amount
	#                       (presence = armed; value = damage)
	# $AGG_FLAG_VAR$      = presence marks the reaction aggravated
	# $KIND_VAR$          = flag-valued: a damage source (fire/cold/...) or `morale`
	# $CONDITION_TRIGGER$ = result-code partition trigger to gate on
	#                       (pod_combat_result_code_indicates_*)
	# $ACTOR_SIDE$        = side taking the reaction damage (just attacked)
	# $RETALIATOR_SIDE$   = side dishing out the reaction (was attacked)
	if = {
		limit = {
			has_variable = $TRIGGER_VAR$
			$CONDITION_TRIGGER$ = yes
		}
		pod_combat_apply_retaliation_helper = {
			ACTOR_SIDE      = $ACTOR_SIDE$
			RETALIATOR_SIDE = $RETALIATOR_SIDE$
			AMOUNT_VAR      = $TRIGGER_VAR$
			AGG_FLAG_VAR    = $AGG_FLAG_VAR$
			KIND_VAR        = $KIND_VAR$
		}
		remove_variable = $TRIGGER_VAR$
		remove_variable = $AGG_FLAG_VAR$
		remove_variable = $KIND_VAR$
	}
}

pod_combat_check_retaliation_effect = {
	# Reaction phase - damage_type inheritance / soak / host-var
	# contract live on pod_combat_apply_retaliation_helper below.
	#
	# $TARGET_SIDE$ = side holding the retaliation stance (almost
	#                 always "target": the AI/PC attacks, target retaliates).
	# $ACTOR_SIDE$  = side that takes retaliation damage (original attacker).

	# Retaliate on contact - any physical contact (hp_damage /
	# guard_blocked / soaked); evasion (missed / dodged / stealthed)
	# stays excluded so thorns/spite only punish those who reach you.
	pod_combat_apply_reaction_if_triggered_effect = {
		TRIGGER_VAR       = pod_combat_$TARGET_SIDE$_retaliate_hit
		AGG_FLAG_VAR      = pod_combat_$TARGET_SIDE$_retaliate_hit_agg
		KIND_VAR          = pod_combat_$TARGET_SIDE$_retaliate_hit_kind
		CONDITION_TRIGGER = pod_combat_result_code_indicates_contact_trigger
		ACTOR_SIDE        = $ACTOR_SIDE$
		RETALIATOR_SIDE   = $TARGET_SIDE$
	}

	# Retaliate on evasion - fires on missed / dodged / stealthed.
	pod_combat_apply_reaction_if_triggered_effect = {
		TRIGGER_VAR       = pod_combat_$TARGET_SIDE$_retaliate_miss
		AGG_FLAG_VAR      = pod_combat_$TARGET_SIDE$_retaliate_miss_agg
		KIND_VAR          = pod_combat_$TARGET_SIDE$_retaliate_miss_kind
		CONDITION_TRIGGER = pod_combat_result_code_indicates_miss_trigger
		ACTOR_SIDE        = $ACTOR_SIDE$
		RETALIATOR_SIDE   = $TARGET_SIDE$
	}
}

pod_combat_apply_retaliation_helper = {
	# $ACTOR_SIDE$      = side taking retaliation (original attacker).
	# $RETALIATOR_SIDE$ = side dishing out retaliation (original target).
	# $AMOUNT_VAR$      = workspace var holding retaliation damage.
	# $AGG_FLAG_VAR$    = workspace var whose presence marks aggravated.
	# $KIND_VAR$        = flag-valued workspace var: a damage source (fire/cold/...) or `morale`.
	#
	# bypass_guard: retaliation ignores the actor's guard (guard is for
	# incoming attacks, not counter strikes). Its scope-export return
	# contract leaves the main attack's outcome / attack_result vars
	# untouched.

	# ── Spirit-damage gate ──
	# Retaliation skips the contact resolver (reflex, not a deliberate
	# attack), so hand-roll just the spirit gate: a physical counter
	# against an incorporeal spirit only lands with the spirit-damage
	# capability, else no-op so the main-attack log line is preserved.
	if = {
		limit = {
			has_variable = pod_combat_$ACTOR_SIDE$_incorporeal_spirit
			NOT = { has_variable = pod_combat_$RETALIATOR_SIDE$_can_damage_spirit }
		}
		# No-op: retaliation passes through harmlessly.
	}
	else = {
		save_scope_value_as = {
			name  = retaliation_amount
			value = var:$AMOUNT_VAR$
		}

		# Dispatch on the counter kind: morale > aggravated > source > plain.
		# pod_combat_last_retal_damage is captured per-branch because HP must
		# record the post-soak amount while morale records the armed amount.
		if = {
			# Morale counter - erodes will, not flesh (also bleeds onto PoD stress).
			limit = { has_variable = $KIND_VAR$  var:$KIND_VAR$ = flag:morale }
			pod_combat_deal_morale_damage_effect = {
				TARGET_SIDE = $ACTOR_SIDE$
				AMOUNT      = scope:retaliation_amount
			}
			set_variable = { name = pod_combat_last_retal_morale value = yes }
			set_variable = { name = pod_combat_last_retal_damage value = scope:retaliation_amount }
		}
		else_if = {
			limit = { has_variable = $AGG_FLAG_VAR$ }
			pod_combat_deal_hp_damage_bypass_guard_effect = {
				TARGET_SIDE = $ACTOR_SIDE$
				AMOUNT      = scope:retaliation_amount
				REASON      = duel_aggravated
			}
			# Aggravated wound-box icon on the retal rider (entry.var:retal_agg).
			set_variable = { name = pod_combat_last_retal_agg value = yes }
			set_variable = { name = pod_combat_last_retal_damage value = scope:damage_dealt }
		}
		else_if = {
			# Sourced counter (fire/cold/...) - routes through PoD's weakness pipeline.
			limit = { has_variable = $KIND_VAR$ }
			pod_combat_deal_typed_retal_damage_effect = {
				TARGET_SIDE = $ACTOR_SIDE$
				AMOUNT      = scope:retaliation_amount
				KIND_VAR    = $KIND_VAR$
			}
			set_variable = { name = pod_combat_last_retal_source value = var:$KIND_VAR$ }
			set_variable = { name = pod_combat_last_retal_damage value = scope:damage_dealt }
		}
		else = {
			pod_combat_deal_hp_damage_bypass_guard_effect = {
				TARGET_SIDE = $ACTOR_SIDE$
				AMOUNT      = scope:retaliation_amount
				REASON      = duel
			}
			set_variable = { name = pod_combat_last_retal_damage value = scope:damage_dealt }
		}

		# Set the retal_fired overlay (independent of outcome /
		# attack_result) - PODCombatLogRetalRider reads it + retal_damage
		# to emit the counter line alongside the main outcome.
		pod_combat_set_retal_fired_effect = yes
	}
}

pod_combat_deal_typed_retal_damage_effect = {
	# Sourced retaliation: map the $KIND_VAR$ source flag to a PoD death_reason and
	# deal it bypass-guard (the reflex path). Mirrors POD_combat_dispatch_by_source
	# (identity in 7 of 8; holy -> true_faith). Aggravated-ness comes from the source's
	# PoD weakness mapping, so no separate agg flag is needed here.
	# $TARGET_SIDE$ = who takes the counter. $AMOUNT$ = pre-soak amount.
	switch = {
		trigger = var:$KIND_VAR$
		flag:fire = {
			pod_combat_deal_hp_damage_bypass_guard_effect = { TARGET_SIDE = $TARGET_SIDE$ AMOUNT = $AMOUNT$ REASON = fire } }
		flag:cold = {
			pod_combat_deal_hp_damage_bypass_guard_effect = { TARGET_SIDE = $TARGET_SIDE$ AMOUNT = $AMOUNT$ REASON = cold } }
		flag:lightning = {
			pod_combat_deal_hp_damage_bypass_guard_effect = { TARGET_SIDE = $TARGET_SIDE$ AMOUNT = $AMOUNT$ REASON = lightning } }
		flag:sun = {
			pod_combat_deal_hp_damage_bypass_guard_effect = { TARGET_SIDE = $TARGET_SIDE$ AMOUNT = $AMOUNT$ REASON = sun } }
		flag:silver = {
			pod_combat_deal_hp_damage_bypass_guard_effect = { TARGET_SIDE = $TARGET_SIDE$ AMOUNT = $AMOUNT$ REASON = silver } }
		flag:gold = {
			pod_combat_deal_hp_damage_bypass_guard_effect = { TARGET_SIDE = $TARGET_SIDE$ AMOUNT = $AMOUNT$ REASON = gold } }
		flag:holy = {
			pod_combat_deal_hp_damage_bypass_guard_effect = { TARGET_SIDE = $TARGET_SIDE$ AMOUNT = $AMOUNT$ REASON = true_faith } }
		flag:balefire = {
			pod_combat_deal_hp_damage_bypass_guard_effect = { TARGET_SIDE = $TARGET_SIDE$ AMOUNT = $AMOUNT$ REASON = balefire } }
		fallback = {
			pod_combat_deal_hp_damage_bypass_guard_effect = { TARGET_SIDE = $TARGET_SIDE$ AMOUNT = $AMOUNT$ REASON = duel }
		}
	}
}

# Unsoakable DoT wound, applied in the victim character's scope. Mirrors
# POD_increase_wounds_effect's non-soaked branch (final_damage = COUNT * splat-weakness,
# capped; box apply + vanilla treatment + balefire stacks). Only the soak random_list is
# dropped - the round-end burn is defined as unsoakable; hard immunity and clay-man
# protection still block.
# Caller contract: victim scope, reached from HOST scope via var:pod_combat_target_char (the
# proven route_to_pod_damage shape).
pod_combat_apply_dot_wound_effect = {
	$ATTACKER$ = { save_scope_as = wound_attacker }
	save_scope_value_as = { name = death_reason value = flag:death_$REASON$ }
	save_scope_value_as = {
		name = final_damage
		value = { value = $COUNT$  multiply = POD_wound_multiplier_from_splat_weaknesses  max = POD_damage_cap_per_death_reason  ceiling = yes }
	}
	POD_set_scope_value_from_trigger = { NAME = is_aggravated TRIGGER = "POD_damage_type_is_aggravated = { DEATH_REASON = scope:death_reason }" }
	if = {
		limit = { POD_has_immunity_to_damage_type = { DEATH_REASON = scope:death_reason } }
		# Hard immunity blocks the tick entirely; only the soak roll is bypassed.
	}
	else_if = {
		limit = { POD_has_clay_man_protection = yes }
		POD_trigger_clay_man_protection = yes
	}
	else = {
		POD_apply_wound_boxes = { COUNT = scope:final_damage AGGRAVATED = scope:is_aggravated REASON = $REASON$ }
		POD_vanilla_wound_treatment = { REASON = $REASON$ }
		if = {
			limit = { scope:death_reason = flag:death_balefire }
			while = { count = scope:final_damage  POD_add_modifier_and_track_stacks = { TYPE = character MODIFIER = balefire_health_modifier } }
		}
	}
}

# Morale DoT tick, slot scope. Erodes slot_morale and mirrors PoD Mental Strain. Reads the
# amount from scope:dot_tick_amount (saved by the tick loop BEFORE expiry can remove the
# slot var on the final tick).
pod_combat_deal_slot_dot_morale_tick_effect = {
	save_scope_value_as = { name = dot_morale_lost value = { value = scope:dot_tick_amount  max = var:slot_morale } }
	set_variable = { name = slot_morale value = { value = var:slot_morale  subtract = scope:dot_tick_amount  min = 0 } }
	if = {
		limit = { scope:dot_morale_lost > 0 }
		var:slot_char = { POD_static_stress_gain = { COUNT = { value = scope:dot_morale_lost  multiply = pod_combat_morale_stress_multiplier } } }
	}
}

# HP DoT tick wound dispatch, HOST scope. The tick loop loads the burning slot into the
# TARGET workspace first (the proven AoE per-target shape: wounds are dealt from host scope
# via the workspace, never from slot scope - slot-scope wound attempts silently no-op in the
# round-end loop). The switch dispatches on the literal-seeded kind through the saved slot
# OBJECT (scope:dot_slot) because a flag lifted via save_scope_value_as degrades to 'value'
# type and flag compares then TYPE-ERROR; the fallback covers the untyped (kind-less) tick.
# The amount (a number) rides safely in scope:dot_tick_amount.
pod_combat_deal_dot_tick_from_host_effect = {
	switch = {
		trigger = scope:dot_slot.var:slot_dot_kind
		flag:fire      = { var:pod_combat_target_char = { pod_combat_apply_dot_wound_effect = { ATTACKER = scope:combat_host REASON = fire COUNT = scope:dot_tick_amount } } }
		flag:cold      = { var:pod_combat_target_char = { pod_combat_apply_dot_wound_effect = { ATTACKER = scope:combat_host REASON = cold COUNT = scope:dot_tick_amount } } }
		flag:lightning = { var:pod_combat_target_char = { pod_combat_apply_dot_wound_effect = { ATTACKER = scope:combat_host REASON = lightning COUNT = scope:dot_tick_amount } } }
		flag:sun       = { var:pod_combat_target_char = { pod_combat_apply_dot_wound_effect = { ATTACKER = scope:combat_host REASON = sun COUNT = scope:dot_tick_amount } } }
		flag:silver    = { var:pod_combat_target_char = { pod_combat_apply_dot_wound_effect = { ATTACKER = scope:combat_host REASON = silver COUNT = scope:dot_tick_amount } } }
		flag:gold      = { var:pod_combat_target_char = { pod_combat_apply_dot_wound_effect = { ATTACKER = scope:combat_host REASON = gold COUNT = scope:dot_tick_amount } } }
		flag:holy      = { var:pod_combat_target_char = { pod_combat_apply_dot_wound_effect = { ATTACKER = scope:combat_host REASON = true_faith COUNT = scope:dot_tick_amount } } }
		flag:balefire  = { var:pod_combat_target_char = { pod_combat_apply_dot_wound_effect = { ATTACKER = scope:combat_host REASON = balefire COUNT = scope:dot_tick_amount } } }
		fallback = {
			# Untyped tick: honor the move's damage_type the same way §6 does for direct hits
			# (duel_aggravated REASON), read off the slot's always-set 0/1 marker.
			if = {
				limit = { scope:dot_slot = { has_variable = slot_dot_aggravated  var:slot_dot_aggravated = 1 } }
				var:pod_combat_target_char = { pod_combat_apply_dot_wound_effect = { ATTACKER = scope:combat_host REASON = duel_aggravated COUNT = scope:dot_tick_amount } }
			}
			else = {
				var:pod_combat_target_char = { pod_combat_apply_dot_wound_effect = { ATTACKER = scope:combat_host REASON = duel COUNT = scope:dot_tick_amount } }
			}
		}
	}
}

# DoT expiry, slot scope. Called as the tick loop's LAST statement (a contained call, safe after
# phase B's inline switches). Kept separate from the round decrement so slot_dot_kind survives
# until the host wound dispatch has read it.
pod_combat_expire_dot_if_done_effect = {
	if = {
		limit = { var:slot_dot_rounds <= 0 }
		remove_variable = slot_dot_amount
		remove_variable = slot_dot_rounds
		remove_variable = slot_dot_kind
		remove_variable = slot_dot_aggravated
	}
}

# Mind-control log row. Reads scope:mind_log_victim (the dominated or
# resisting character) + scope:combat_host (round). MESSAGE_KEY = mind_seized /
# mind_resisted / mind_betrayal / mind_fizzle.
pod_combat_push_mind_log_effect = {
	hidden_effect_new_object = {
		character:POD_wraith_01 = {
			create_story = {
				type = POD_combat_log_entry
				save_scope_as = new_log_entry
			}
		}
		scope:new_log_entry = {
			set_variable = { name = system_message value = flag:$MESSAGE_KEY$ }
			set_variable = { name = actor          value = scope:mind_log_victim }
			set_variable = { name = actor_side     value = -1 }
			if = {
				limit = { scope:combat_host = { has_variable = pod_combat_round } }
				set_variable = { name = round value = scope:combat_host.var:pod_combat_round }
			}
		}
		pod_combat_prepend_capped_log_entry_effect = yes
	}
}

pod_combat_apply_ap_drain_effect = {
	# Apply last round's AP-drain debuff to the current
	# actor, then clear it so it fires once. Reads the workspace
	# mirror of the actor slot's slot_ap_drain. Runs in HOST scope
	# at every turn-start site, right after current_ap is set to base.
	if = {
		limit = { has_variable = pod_combat_actor_ap_drain }
		save_scope_as = combat_host
		save_scope_value_as = { name = daze_log_amount value = var:pod_combat_actor_ap_drain }
		var:pod_combat_actor_char = { save_scope_as = daze_log_victim }
		change_variable = {
			name = pod_combat_current_ap
			add = {
				value = var:pod_combat_actor_ap_drain
				multiply = -1
			}
		}
		# Floor 0, not 1: a full drain (>= base AP, authored as ap_drain_6) IS the
		# stun primitive - no move costs 0 AP, so a fully drained combatant cannot
		# act and loses the turn. Partial drains still leave a partial turn.
		if = {
			limit = { var:pod_combat_current_ap < 0 }
			set_variable = { name = pod_combat_current_ap value = 0 }
		}
		remove_variable = pod_combat_actor_ap_drain
		# Consumption log row: full drain reads as a stun, partial as dazed.
		if = {
			limit = { var:pod_combat_current_ap <= 0 }
			pod_combat_push_ap_drain_log_effect = { MESSAGE_KEY = stunned }
		}
		else = {
			pod_combat_push_ap_drain_log_effect = { MESSAGE_KEY = dazed }
		}
	}
}

######################################################
###### POD WOUND-SYSTEM INTEGRATION HELPERS
######################################################
# Shims that route combat damage through PoD's wound pipeline
# (POD_increase_wounds_hostile_effect → POD_apply_wound_boxes →
# POD_health_death_check_effect), which owns soak / splat-weakness /
# aggravated classification / cap / immunity / wound-box overflow / the
# inline death check. Combat keeps only what PoD has no concept of: hit
# roll, stance consumption, spirit gate, morale spillover.
#
# Tradeoff: PoD fires a "took damage" toast on every wound write - one
# per landed hit. We accept the toast spam as the cost of routing
# through PoD's standard pipeline (suppressing would need a custom
# POD_increase_wounds_* path that skips the interface call).

pod_combat_route_to_pod_damage = {
	# $TARGET_SIDE$ = "actor" or "target"
	# $REASON$ = death_reason suffix (no "death_" prefix). Routed values
	#            map lives in POD_combat_dispatch_by_source
	#            (pod_combat_move_effects.txt).
	# $COUNT$ = raw damage (PoD multiplies by splat weakness + soak roll).
	#
	# Return contract - scope values only, NO host-var side effects:
	#   scope:damage_dealt       = post-soak/multiplier amount (0 if fully
	#                              soaked or practice mode)
	#   scope:attack_result_flag = flag:hp_damage / soaked / utility
	#                              (utility = practice sentinel; callers
	#                              always see exactly one of the three)
	# Each caller decides whether to lift these into pod_combat_last_*;
	# see the lift sites in pod_combat_apply_damage_payload_effect (main
	# attack), §8 self-damage (discards), and the retaliation helper.

	# Practice mode: sparring duel - skip the wound pipeline entirely
	# (morale still applies, tracked on the workspace). Returns
	# damage_dealt=0 + flag:utility so morale spillover / retaliation /
	# log-line logic all short-circuit to the utility fallback row.
	if = {
		limit = {
			has_variable = pod_combat_sc_context
			var:pod_combat_sc_context.var:fatality = flag:practice
		}
		save_scope_value_as = { name = damage_dealt       value = 0 }
		save_scope_value_as = { name = attack_result_flag value = flag:utility }
	}
	else = {

	# Snapshot superficial + aggravated counts SEPARATELY (the wound SVs
	# gate on is_alive, so post reads are 0 on a kill). Separate tracks
	# are required for PoD's aggravated push-out: agg damage on an
	# already-wounded target converts existing superficial into
	# aggravated (5 sup → 2 sup + 3 agg on a 3-agg hit), so total boxes
	# are unchanged and a naive post_total - pre_total delta reads as 0,
	# falsely classifying the hit as soaked. Per-track delta clamped at 0
	# mirrors PoD's total_wounds_gained (see POD_apply_wound_boxes).
	# pre_empty_boxes is held for the death-branch overkill clamp.
	var:pod_combat_$TARGET_SIDE$_char = {
		save_scope_value_as = { name = pre_superficial_wounds value = POD_superficial_damage_boxes }
		save_scope_value_as = { name = pre_aggravated_wounds  value = POD_aggravated_damage_boxes }
		save_scope_value_as = { name = pre_empty_boxes        value = POD_empty_damage_boxes }
	}
	# scope:combat_host (read below for ATTACKER) is saved upstream by
	# POD_execute_combat_move - root isn't reliable this deep in the
	# AI-turn dispatch chain.
	var:pod_combat_$TARGET_SIDE$_char = {
		POD_increase_wounds_hostile_effect = {
			ATTACKER = scope:combat_host.var:pod_combat_actor_char
			REASON   = $REASON$
			COUNT    = $COUNT$
		}
	}

	# Branch on whether PoD killed the target. POD_apply_wound_boxes
	# calls POD_health_death_check_effect inline; a fatal blow fires
	# pod_death and is_alive flips to no before this line resumes.
	if = {
		limit = { var:pod_combat_$TARGET_SIDE$_char = { is_alive = no } }
		# Dead-char wound SVs read 0 (is_alive limit), so approximate
		# damage_dealt as the pre-blow empty boxes.
		save_scope_value_as = {
			name = damage_dealt
			value = scope:pre_empty_boxes
		}
		save_scope_value_as = { name = attack_result_flag value = flag:hp_damage }
	}
	else = {
		# Target still alive - read fresh post-damage wound state.
		var:pod_combat_$TARGET_SIDE$_char = {
			save_scope_value_as = { name = post_superficial_wounds value = POD_superficial_damage_boxes }
			save_scope_value_as = { name = post_aggravated_wounds  value = POD_aggravated_damage_boxes }
		}
		save_scope_value_as = {
			name = superficial_added
			value = {
				value    = scope:post_superficial_wounds
				subtract = scope:pre_superficial_wounds
				min      = 0
			}
		}
		save_scope_value_as = {
			name = aggravated_added
			value = {
				value    = scope:post_aggravated_wounds
				subtract = scope:pre_aggravated_wounds
				min      = 0
			}
		}
		# max(0, sup_delta) + agg_delta - see push-out note above.
		save_scope_value_as = {
			name = damage_dealt
			value = {
				value = scope:superficial_added
				add   = scope:aggravated_added
			}
		}

		if = {
			limit = { scope:damage_dealt = 0 }
			# Fully soaked or immune.
			save_scope_value_as = { name = attack_result_flag value = flag:soaked }
		}
		else = {
			# Live damage landed. Workspace HP SVs derive from the char,
			# so no workspace update needed here.
			save_scope_value_as = { name = attack_result_flag value = flag:hp_damage }

			# Morale spillover scales off actual (post-soak) damage and
			# also bleeds into Mental Strain, matching direct morale
			# damage (see pod_combat_deal_morale_damage_effect).
			save_scope_value_as = {
				name = morale_spill
				value = {
					value = scope:damage_dealt
					multiply = pod_combat_morale_from_hp_ratio
					floor = yes
				}
			}
			if = {
				limit = {
					scope:morale_spill > 0
					# Gate on >0 so the stress / SV chain below doesn't
					# run for zero net delta on an already-broken target.
					var:pod_combat_$TARGET_SIDE$_morale > 0
				}
				# Cap at current morale so stress doesn't over-count past
				# a broken target (as pod_combat_deal_morale_damage_effect).
				save_scope_value_as = {
					name = morale_actually_spilled
					value = {
						value = scope:morale_spill
						max   = var:pod_combat_$TARGET_SIDE$_morale
					}
				}
				save_scope_value_as = {
					name = new_target_morale
					value = {
						value = var:pod_combat_$TARGET_SIDE$_morale
						subtract = scope:morale_spill
						min = 0
					}
				}
				set_variable = { name = pod_combat_$TARGET_SIDE$_morale value = scope:new_target_morale }
				# Mental Strain on the spillover delta.
				if = {
					limit = { scope:morale_actually_spilled > 0 }
					save_scope_value_as = {
						name = mental_strain_from_spillover
						value = {
							value = scope:morale_actually_spilled
							multiply = pod_combat_morale_stress_multiplier
						}
					}
					var:pod_combat_$TARGET_SIDE$_char = {
						POD_static_stress_gain = { COUNT = scope:mental_strain_from_spillover }
					}
				}
			}
		}
	}
	}  # end else (non-practice path)
}


pod_combat_deal_hp_damage_bypass_guard_effect = {
	# $TARGET_SIDE$ = "actor" or "target" - who takes damage.
	# $AMOUNT$ = damage amount (raw, pre-soak, pre-multiplier)
	# $REASON$ = death reason suffix - see pod_combat_route_to_pod_damage.
	#
	# Bare PoD damage write - no guard / stance / spirit / hit-roll.
	# The contact resolver handled those before the payload; self-damage
	# and retaliation call this directly to skip the resolver on purpose.
	# Return contract inherited from pod_combat_route_to_pod_damage.
	pod_combat_route_to_pod_damage = {
		TARGET_SIDE = $TARGET_SIDE$
		REASON      = $REASON$
		COUNT       = $AMOUNT$
	}
}

# ── Actor stealth consumption ──────────────────────────────────
# Consumes the attacker's stealth after it acts. Called at the END of
# the damage effect so the stealth hit bonus still applies to THIS
# attack, then reverts for future actions.
#
# $ACTOR_SIDE$ = "actor" or "target" - workspace side that just attacked.
# Retaliation deliberately does NOT route through here (it goes via the
# bare bypass_guard helper) - a reflex strike doesn't break stealth.
pod_combat_consume_actor_stealth_effect = {
	pod_combat_break_stealth_workspace_effect = { SIDE = $ACTOR_SIDE$ }
}

# ── Shared workspace-stealth breaker ──────────────────────────────
# Removes the *_is_stealthed flag AND reverts the stealth hit-chance
# bonus (pod_combat_stealth_hit_bonus) on the same workspace side.
# Idempotent (no-op when unset).
#
#   SIDE - "actor" or "target". Also parametrically picks the matching
#          pod_combat_recalc_hit_$SIDE$_effect; the actor recalc syncs
#          pod_combat_current_hit_chance for free, the target recalc
#          correctly doesn't (current is an actor-only mirror).
pod_combat_break_stealth_workspace_effect = {
	if = {
		limit = { has_variable = pod_combat_$SIDE$_is_stealthed }
		remove_variable = pod_combat_$SIDE$_is_stealthed
		pod_combat_recalc_hit_$SIDE$_effect = yes
	}
}

# ── Slot-scope stealth helpers ────────────────────────────────────
# Slot-scope mirrors of the workspace stealth applicator/breaker
# above - same flag+hit_chance invariant, applied to slot_X vars
# instead of workspace pod_combat_$SIDE$_*. Used by AoE broadcasts
# iterating pod_combat_active_slots; caller must be in slot-story scope.

pod_combat_apply_stealth_slot_effect = {
	if = {
		limit = { NOT = { has_variable = slot_is_stealthed } }
		set_variable = { name = slot_is_stealthed value = yes }
		pod_combat_recalc_slot_stats_via_story_effect = yes
	}
}

# Workspace mirror of pod_combat_apply_stealth_slot_effect: an
# idempotent no-op when already stealthed. The stealth hit bonus is
# applied inside the hit recalc, keyed off the stealth flag.
pod_combat_apply_stealth_workspace_effect = {
	if = {
		limit = { NOT = { has_variable = pod_combat_$SIDE$_is_stealthed } }
		set_variable = { name = pod_combat_$SIDE$_is_stealthed value = yes }
		pod_combat_recalc_hit_$SIDE$_effect = yes
	}
}

pod_combat_break_stealth_slot_effect = {
	if = {
		limit = { has_variable = slot_is_stealthed }
		remove_variable = slot_is_stealthed
		# Recalc rebuilds slot_hit_chance without the stealth-bonus branch.
		pod_combat_recalc_slot_stats_via_story_effect = yes
	}
}

# ── Target stance consumption ──────────────────────────────────
# Stealth consumed FIRST so a stealthed-and-dodging target keeps its
# dodge for the next attack. The HP path reaches this via the contact
# resolver; the morale path calls it inline.
#
# $TARGET_SIDE$ = "actor" or "target" - the side being attacked.
# Saves scope:stance_consumed: 0 = none, 1 = stealth, 2 = dodge.
# Both stance-fired branches set the outcome+attack_result pair and
# pod_combat_last_damage = 0 internally, so callers needn't reset those.
pod_combat_consume_target_stance_effect = {
	if = {
		limit = { has_variable = pod_combat_$TARGET_SIDE$_is_stealthed }
		pod_combat_break_stealth_workspace_effect = { SIDE = $TARGET_SIDE$ }
		# flag:stealthed vs flag:dodged: same auto-miss, but links the
		# log line to the Stealth concept rather than Dodge.
		pod_combat_set_attack_result_effect = { CODE = stealthed }
		set_variable = { name = pod_combat_last_damage value = 0 }
		save_scope_value_as = { name = stance_consumed value = 1 }
	}
	else_if = {
		limit = { has_variable = pod_combat_$TARGET_SIDE$_is_dodging }
		remove_variable = pod_combat_$TARGET_SIDE$_is_dodging
		pod_combat_set_attack_result_effect = { CODE = dodged }
		set_variable = { name = pod_combat_last_damage value = 0 }
		save_scope_value_as = { name = stance_consumed value = 2 }
	}
	else = {
		save_scope_value_as = { name = stance_consumed value = 0 }
	}
}

# ── UNIFIED CONTACT RESOLVER ──────────────────────────────────────
# Walks the pre-damage decision tree (spirit gate → stance consume →
# hit roll → guard check) off the move story's capability flags.
#
# Capability flags (set on scope:current_move at move-init time):
#   checks_spirit_gate - physical attacks pass through incorporeal
#                        spirits unless actor has can_damage_spirit.
#   checks_stance      - stealth → dodge consumption gates damage.
#   requires_hit_roll  - roll pod_combat_current_hit_chance vs 100;
#                        off for supernatural/mental (auto-hit).
#   checks_guard       - consume target's _is_guarding and block.
#
# Outputs scope:contact_result - integers LOCAL to this resolver,
# NOT the result-flag values above:
#     0 = hit (proceed to payload)   1 = blocked   2 = miss
#     3 = dodged   4 = stealthed   6 = spirit_immune
# Non-hit branches set the outcome+attack_result pair (spirit-immune
# logs as missed) and pod_combat_last_damage = 0; hit leaves the result
# vars for the payload to write.
pod_combat_resolve_contact_effect = {
	# $TARGET_SIDE$ = "actor" or "target" - who takes the attack.
	# $ACTOR_SIDE$  = "actor" or "target" - who is attacking.
	# Caller must have saved the move story as scope:current_move so
	# the capability-flag reads resolve.

	# 1. Spirit gate (capability-driven)
	if = {
		limit = {
			scope:current_move = { has_variable = checks_spirit_gate }
			has_variable = pod_combat_$TARGET_SIDE$_incorporeal_spirit
			NOT = { has_variable = pod_combat_$ACTOR_SIDE$_can_damage_spirit }
		}
		# Spirit gate denies the attack - logs as missed
		# (no spirit-immune log code).
		pod_combat_set_attack_result_effect = { CODE = missed }
		set_variable = { name = pod_combat_last_damage value = 0 }
		save_scope_value_as = { name = contact_result value = 6 }
	}
	else = {
		# 2. Stance consume (capability-driven)
		if = {
			limit = { scope:current_move = { has_variable = checks_stance } }
			pod_combat_consume_target_stance_effect = { TARGET_SIDE = $TARGET_SIDE$ }
		}
		else = {
			save_scope_value_as = { name = stance_consumed value = 0 }
		}

		if = {
			limit = { scope:stance_consumed = 1 }
			save_scope_value_as = { name = contact_result value = 4 }
		}
		else_if = {
			limit = { scope:stance_consumed = 2 }
			save_scope_value_as = { name = contact_result value = 3 }
		}
		else = {
			# 3. Hit roll (capability-driven). No flag = auto-hit.
			if = {
				limit = { scope:current_move = { has_variable = requires_hit_roll } }
				save_scope_value_as = {
					name  = hit_chance
					value = var:pod_combat_current_hit_chance
				}
				random_list = {
					# Miss
					0 = {
						modifier = {
							add = {
								value    = 100
								subtract = scope:hit_chance
							}
						}
						pod_combat_set_attack_result_effect = { CODE = missed }
						set_variable = { name = pod_combat_last_damage value = 0 }
						save_scope_value_as = { name = contact_result value = 2 }
					}
					# Hit - roll succeeded, fall through to guard check
					0 = {
						modifier = { add = scope:hit_chance }
						pod_combat_check_guard_effect = { TARGET_SIDE = $TARGET_SIDE$ }
					}
				}
			}
			else = {
				# Auto-hit - skip the roll, still check guard if applicable.
				pod_combat_check_guard_effect = { TARGET_SIDE = $TARGET_SIDE$ }
			}
		}
	}
}

# ── GUARD CHECK SUBROUTINE ─────────────────────────────────────────
# Shared by the resolver's hit-roll-success and auto-hit branches.
pod_combat_check_guard_effect = {
	# $TARGET_SIDE$ = "actor" or "target"
	# Reads scope:current_move.var:checks_guard. Writes scope:contact_result
	# (0 = hit, 1 = blocked) and sets attack_result=guard_blocked on block
	# (the hit branch is set by the damage payload).
	if = {
		limit = {
			scope:current_move = { has_variable = checks_guard }
			has_variable = pod_combat_$TARGET_SIDE$_is_guarding
		}
		remove_variable = pod_combat_$TARGET_SIDE$_is_guarding
		pod_combat_set_attack_result_effect = { CODE = guard_blocked }
		set_variable = { name = pod_combat_last_damage value = 0 }
		save_scope_value_as = { name = contact_result value = 1 }
	}
	else = {
		save_scope_value_as = { name = contact_result value = 0 }
	}
}

# ── DAMAGE PAYLOAD ─────────────────────────────────────────────────
# Fires only on contact_result = 0 (hit); the resolver handled the rest.
# Owns the log entry, so it lifts the route helper's scope exports into
# pod_combat_last_damage + the outcome/attack_result pair (other lift
# sites are listed on pod_combat_route_to_pod_damage).
pod_combat_apply_damage_payload_effect = {
	# $TARGET_SIDE$ = "actor" or "target"
	# $AMOUNT$      = damage amount (raw, pre-soak)
	# $REASON$      = PoD death_reason suffix
	pod_combat_deal_hp_damage_bypass_guard_effect = {
		TARGET_SIDE = $TARGET_SIDE$
		AMOUNT      = $AMOUNT$
		REASON      = $REASON$
	}
	# Practice-mode flag:utility = no real attack: don't promote to
	# outcome=attack - leave the outcome vars for
	# the if-unset writers (morale/heal) and the utility fallback.
	if = {
		limit = { NOT = { scope:attack_result_flag = flag:utility } }
		set_variable = { name = pod_combat_last_damage         value = scope:damage_dealt }
		set_variable = { name = pod_combat_last_outcome       value = flag:attack }
		set_variable = { name = pod_combat_last_attack_result value = scope:attack_result_flag }
	}
}


pod_combat_deal_morale_damage_effect = {
	# $TARGET_SIDE$ = "actor" or "target"
	# $AMOUNT$ = morale damage (bypasses HP entirely)
	# Auto-hit, no guard, no soak - but stealth/dodge still block (gated
	# on stance_consumed = 0 below). Also mirrors the loss onto PoD's
	# stress track as Mental Strain (pod_combat_morale_stress_multiplier)
	# so the pressure persists outside combat.
	pod_combat_consume_target_stance_effect = { TARGET_SIDE = $TARGET_SIDE$ }
	if = {
		limit = { scope:stance_consumed = 0 }
		# Cap at current morale so stress doesn't over-count past a
		# near-broken target (1 left + 5 dmg → 1 actual, not 5 stress).
		save_scope_value_as = {
			name = morale_actually_lost
			value = {
				value = $AMOUNT$
				max   = var:pod_combat_$TARGET_SIDE$_morale
			}
		}
		save_scope_value_as = {
			name = new_morale
			value = {
				value = var:pod_combat_$TARGET_SIDE$_morale
				subtract = $AMOUNT$
				min = 0
			}
		}
		set_variable = { name = pod_combat_$TARGET_SIDE$_morale value = scope:new_morale }
		# Own var (pod_combat_last_damage holds HP on damage+morale moves).
		set_variable = { name = pod_combat_last_morale_damage_amount value = scope:morale_actually_lost }
		pod_combat_set_outcome_if_unset_effect = { CODE = morale_damage }

		# Mirror the loss onto the target as Mental Strain. The char alias
		# is always populated mid-combat; guard for off-combat callers.
		if = {
			limit = {
				has_variable = pod_combat_$TARGET_SIDE$_char
				scope:morale_actually_lost > 0
			}
			save_scope_value_as = {
				name = mental_strain_from_morale
				value = {
					value = scope:morale_actually_lost
					multiply = pod_combat_morale_stress_multiplier
				}
			}
			var:pod_combat_$TARGET_SIDE$_char = {
				POD_static_stress_gain = { COUNT = scope:mental_strain_from_morale }
			}
		}
		# Victory check runs at the end of pod_combat_save_workspace_effect.
	}
}

pod_combat_restore_morale_effect = {
	# $SIDE$ = "actor" or "target"
	# $AMOUNT$ = morale to restore (requested; the logged amount below is
	#            the ACTUAL delta after the max-morale cap)
	save_scope_value_as = {
		name = old_morale
		value = var:pod_combat_$SIDE$_morale
	}
	save_scope_value_as = {
		name = new_morale
		value = {
			value = scope:old_morale
			add = $AMOUNT$
			max = pod_combat_max_morale
		}
	}
	set_variable = { name = pod_combat_$SIDE$_morale value = scope:new_morale }
	# Own var (keeps pod_combat_last_damage for HP). Log the actual
	# capped delta, not requested $AMOUNT$.
	set_variable = {
		name = pod_combat_last_morale_heal_amount
		value = { value = scope:new_morale subtract = scope:old_morale }
	}
	pod_combat_set_outcome_if_unset_effect = { CODE = morale_healed }
}

# ─── AUTO-RESOLVE ──────────────────────────────────────────────────

pod_combat_auto_resolve_effect = {
	# Save PC scope alias to avoid bare `root` references in set_variable.
	save_scope_as = combat_pc

	# slot_prowess is captured at slot init (which runs AFTER discipline
	# battle modifiers apply), so it already reflects the in-combat
	# prowess for each combatant. Reading the slot var avoids a per-
	# iteration scope flip into var:slot_char.
	#
	# Winner / loser are the TEAM LEADERS (player/0 / enemy/0), matching
	# the contract pod_combat_check_group_victory_effect already uses.
	# The post-combat dispatcher routes off these two character refs.

	# Player team prowess sum (alive only). PREV walks back to the host
	# scope to accumulate the slot-scope var:slot_prowess.
	save_scope_value_as = { name = player_prowess_sum value = 0 }
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			pod_combat_slot_is_player_team_trigger = yes
			pod_combat_slot_is_alive_trigger = yes
		}
		save_temporary_scope_value_as = { name = slot_prowess_tmp value = var:slot_prowess }
		PREV = {
			save_scope_value_as = {
				name = player_prowess_sum
				value = {
					value = scope:player_prowess_sum
					add = scope:slot_prowess_tmp
				}
			}
		}
	}

	# Enemy team prowess sum (enemy team, alive only)
	save_scope_value_as = { name = enemy_prowess_sum value = 0 }
	every_in_list = {
		variable = pod_combat_active_slots
		limit = {
			pod_combat_slot_is_enemy_trigger = yes
			pod_combat_slot_is_alive_trigger = yes
		}
		save_temporary_scope_value_as = { name = slot_prowess_tmp value = var:slot_prowess }
		PREV = {
			save_scope_value_as = {
				name = enemy_prowess_sum
				value = {
					value = scope:enemy_prowess_sum
					add = scope:slot_prowess_tmp
				}
			}
		}
	}

	# Apply per-team random jitter - score = sum + rand(0..sum).
	# integer_range accepts a dynamic max.
	save_scope_value_as = {
		name = player_score
		value = {
			value = scope:player_prowess_sum
			add = { integer_range = { min = 0 max = scope:player_prowess_sum } }
		}
	}
	save_scope_value_as = {
		name = enemy_score
		value = {
			value = scope:enemy_prowess_sum
			add = { integer_range = { min = 0 max = scope:enemy_prowess_sum } }
		}
	}

	# Resolve. Winner/loser are the team leaders so the post-combat
	# dispatcher routes off pod_combat_winner / pod_combat_loser as in
	# a normal victory. The enemy-leader guards are defensive against
	# corrupt state (enemy/0 is always populated at init).
	if = {
		limit = { scope:player_score >= scope:enemy_score }
		set_variable = { name = pod_combat_winner value = scope:combat_pc }
		if = {
			limit = { has_variable = pod_combat_enemy_leader_slot  var:pod_combat_enemy_leader_slot = { has_variable = slot_char } }
			set_variable = { name = pod_combat_loser value = var:pod_combat_enemy_leader_slot.var:slot_char }
		}
	}
	else = {
		if = {
			limit = { has_variable = pod_combat_enemy_leader_slot  var:pod_combat_enemy_leader_slot = { has_variable = slot_char } }
			set_variable = { name = pod_combat_winner value = var:pod_combat_enemy_leader_slot.var:slot_char }
		}
		set_variable = { name = pod_combat_loser value = scope:combat_pc }
	}

	# Zero the displayed target's morale so the loser portrait reads
	# defeated (touches only the workspace's selected_target).
	set_variable = { name = pod_combat_target_morale value = 0 }
	set_variable = { name = pod_combat_is_over value = yes }
	# System log: combat ends. Outcome flag picks victory vs
	# defeat from whichever side won the prowess-sum roll above.
	if = {
		limit = { var:pod_combat_winner = scope:combat_pc }
		pod_combat_push_system_log_effect = { MESSAGE_KEY = combat_ended_victory }
	}
	else = {
		pod_combat_push_system_log_effect = { MESSAGE_KEY = combat_ended_defeat }
	}
	# Combat window stays open - player clicks "End Combat" to dismiss.
}

# ─── COMBAT LOG ────────────────────────────────────────────────────
# Rolling log of combat actions, stored as POD_combat_log_entry
# story_cycles in the PC's pod_combat_log variable list. Each entry
# captures a snapshot of one resolved move:
#
#   move                  - story scope of the executed move. The GUI
#                            reads name, capability flags, damage_source /
#                            _type, targeting, and stance/buff/debuff flags
#                            off this scope.
#   actor / target        - character refs (target optional: self / team_*
#                            moves may skip)
#   damage                - int HP damage (morale uses morale_damage_amount)
#   heal_amount           - int HP healed when the move heals
#   morale_damage_amount  - int morale loss when the move drops morale
#   morale_heal_amount    - int morale restored when the move heals it
#   retal_damage          - int damage from a triggered retaliation
#                            (captured pre-restore so the counter renders
#                            even when the original was blocked / soaked)
#   outcome / attack_result / retal_fired
#                         - three-axis flag model (see COMBAT_REFERENCE.info
#                            §4; dispatch tables in pod_combat_log_custom_loc.txt).
#
# Entries are PREPENDED - push_log rebuilds the list to insert each new
# row at the FRONT so the GUI datamodel renders newest-first without
# any sort logic. A single datamodel template in pod_combat_window.gui
# renders every row, reading the entry's vars directly.

# Shared prepend-and-cap helper for both push_log and push_system_log.
# Inserts scope:new_log_entry at the FRONT of the host's pod_combat_log
# variable list, then trims oldest entries past pod_combat_log_max_entries.
# Trimmed entries are end_story'd so the underlying POD_combat_log_entry
# story_cycle doesn't leak on Charon for the rest of the session.
#
# Caller contract:
#   - scope:combat_host points at the combat host (the PC).
#   - scope:new_log_entry points at the freshly-created log entry story.
# Parameterless - purely scope-driven.
pod_combat_prepend_capped_log_entry_effect = {
	scope:combat_host = {
		every_in_list = {
			variable = pod_combat_log
			scope:combat_host = {
				add_to_variable_list = { name = pod_combat_log_rebuild target = prev }
			}
		}
		clear_variable_list = pod_combat_log
		add_to_variable_list = { name = pod_combat_log target = scope:new_log_entry }
		set_variable = { name = pod_combat_log_append_count value = 1 }
		every_in_list = {
			variable = pod_combat_log_rebuild
			scope:combat_host = {
				if = {
					limit = { var:pod_combat_log_append_count < pod_combat_log_max_entries }
					add_to_variable_list = { name = pod_combat_log target = prev }
					change_variable = { name = pod_combat_log_append_count add = 1 }
				}
				else = {
					prev = { end_story = yes }
				}
			}
		}
		clear_variable_list = pod_combat_log_rebuild
		remove_variable = pod_combat_log_append_count
	}
}

pod_combat_push_log_effect = {
	# Build a new POD_combat_log_entry story holding this turn's
	# resolved-action snapshot and append it to the PC's
	# pod_combat_log variable list. The GUI's datamodel iterates
	# this list and reads each entry's vars directly.
	#
	#   $MOVE_STORY$ - the move story_cycle being logged. Threaded
	#                  explicitly (not via a host-var handoff) so push_log
	#                  carries no shared-state dependency on its caller.
	#
	# Reads the host's per-turn pod_combat_last_* transients (damage,
	# heal_amount, morale_damage/heal_amount, retal_damage) plus the
	# actor/target char refs onto matching entry vars; each field is
	# catalogued in the section header above. The outcome / attack_result
	# / retal_fired three-axis model is in COMBAT_REFERENCE.info §4 -
	# adding a flag is a coordinated edit across §4, PODCombatLogResultText,
	# the pod_combat_log_result_text_* loc keys, and the partition triggers
	# in pod_combat_move_execution_triggers.txt.
	#
	# Entries owned by Charon (POD_wraith_01), a guaranteed-alive anchor,
	# to dodge "owner died" cleanup races. The PC's pod_combat_log just
	# holds references.

	# Save the host scope: the AI-turn call chain doesn't reliably
	# preserve `root`, so the rest of this effect reads scope:combat_host.
	save_scope_as = combat_host

	character:POD_wraith_01 = {
		create_story = {
			type = POD_combat_log_entry
			save_scope_as = new_log_entry
		}
	}
	scope:new_log_entry = {
		# entry.var:move is a LIVE story reference, not a per-row copy: the
		# dispatcher reads damage_source / type / targeting / tier flags off
		# the underlying POD_combat_move. Safe because POD_reload_combat_moves
		# never fires mid-combat, and cleanup ends every entry before any
		# reload could occur.
		set_variable = { name = move value = $MOVE_STORY$ }
		if = {
			limit = { scope:combat_host = { has_variable = pod_combat_actor_char } }
			set_variable = { name = actor value = scope:combat_host.var:pod_combat_actor_char }
		}
		if = {
			limit = { scope:combat_host = { has_variable = pod_combat_target_char } }
			set_variable = { name = target value = scope:combat_host.var:pod_combat_target_char }
		}
		# Actor side snapshot for the GUI's ally/enemy distinction.
		# Team CODE (not slot index): 0 = player, 1 = enemy, -1 = system.
		# Precomputed so border colour / actor tint is a single Var read.
		# ALWAYS set: the -1 sentinel routes system rows to neutral
		# rendering (the GUI tint gates on actor_side = 0 / = 1).
		if = {
			limit = { scope:combat_host = { has_variable = pod_combat_current_actor_team } }
			if = {
				limit = { scope:combat_host = { var:pod_combat_current_actor_team = flag:enemy } }
				set_variable = { name = actor_side value = 1 }
			}
			else = {
				set_variable = { name = actor_side value = 0 }
			}
		}
		else = {
			set_variable = { name = actor_side value = -1 }
		}
		# AoE burst index (set only inside an AoE iterator): 0 = first row,
		# 1+ = subsequent. The GUI's buff-rider gate uses it to emit
		# move-level stance / tier-buff lines once per burst. Absent on
		# non-AoE / §16 wrap-up rows so the rider renders normally.
		if = {
			limit = { scope:combat_host = { has_variable = pod_combat_aoe_iter_burst_index } }
			set_variable = { name = aoe_burst_index value = scope:combat_host.var:pod_combat_aoe_iter_burst_index }
		}
		if = {
			limit = { scope:combat_host = { has_variable = pod_combat_last_damage } }
			set_variable = { name = damage value = scope:combat_host.var:pod_combat_last_damage }
		}
		else = {
			set_variable = { name = damage value = 0 }
		}
		# heal / morale-damage / morale-heal ride their own vars (not
		# pod_combat_last_damage) so hybrid moves keep both amounts.
		if = {
			limit = { scope:combat_host = { has_variable = pod_combat_last_heal_amount } }
			set_variable = { name = heal_amount value = scope:combat_host.var:pod_combat_last_heal_amount }
		}
		if = {
			limit = { scope:combat_host = { has_variable = pod_combat_last_morale_damage_amount } }
			set_variable = { name = morale_damage_amount value = scope:combat_host.var:pod_combat_last_morale_damage_amount }
		}
		if = {
			limit = { scope:combat_host = { has_variable = pod_combat_last_morale_heal_amount } }
			set_variable = { name = morale_heal_amount value = scope:combat_host.var:pod_combat_last_morale_heal_amount }
		}
		if = {
			limit = { scope:combat_host = { has_variable = pod_combat_last_self_damage } }
			set_variable = { name = self_damage value = scope:combat_host.var:pod_combat_last_self_damage }
		}
		# Retaliation damage, in its own var so the counter renders even
		# when the original attack was blocked / soaked (entry.damage = 0).
		if = {
			limit = { scope:combat_host = { has_variable = pod_combat_last_retal_damage } }
			set_variable = { name = retal_damage value = scope:combat_host.var:pod_combat_last_retal_damage }
		}
		# Three-axis result capture (model in the header / COMBAT_REFERENCE
		# §4). Outcome falls back to flag:utility for moves that wrote no
		# flag (pure stance / buff / AP-mod). attack_result and retal_fired
		# are only mirrored when their host vars are present.
		if = {
			limit = {
				scope:combat_host = { has_variable = pod_combat_last_outcome }
			}
			set_variable = { name = outcome value = scope:combat_host.var:pod_combat_last_outcome }
		}
		else = {
			# No primary outcome fired - utility move.
			set_variable = { name = outcome value = flag:utility }
		}
		if = {
			limit = {
				scope:combat_host = { has_variable = pod_combat_last_attack_result }
			}
			set_variable = { name = attack_result value = scope:combat_host.var:pod_combat_last_attack_result }
		}
		if = {
			limit = {
				scope:combat_host = { has_variable = pod_combat_last_retal_fired }
			}
			set_variable = { name = retal_fired value = 1 }
		}
		# Aggravated marker for the retal overlay; PODCombatLogRetalRider
		# swaps the wound-box icon on it. Absent = superficial.
		if = {
			limit = {
				scope:combat_host = { has_variable = pod_combat_last_retal_agg }
			}
			set_variable = { name = retal_agg value = 1 }
		}
		# Morale marker: the counter eroded will, not flesh. PODCombatLogRetalRider
		# renders the counter as morale instead of an HP wound-box line.
		if = {
			limit = {
				scope:combat_host = { has_variable = pod_combat_last_retal_morale }
			}
			set_variable = { name = retal_morale value = 1 }
		}
		# Source marker (fire/sun/...): stored as the flag so the rider GetFlagName's it.
		if = {
			limit = {
				scope:combat_host = { has_variable = pod_combat_last_retal_source }
			}
			set_variable = { name = retal_source value = scope:combat_host.var:pod_combat_last_retal_source }
		}
	}
	pod_combat_prepend_capped_log_entry_effect = yes

	scope:combat_host = {
		# Clear the per-action transients now they're captured (also done
		# at §2 of the next move; this keeps them null between moves).
		# Must use the per-target variant: it preserves the AoE tracking
		# pair (pushed_rows + iter_burst_index) across the per-target
		# pushes in one AoE iteration. A full clear here would reset
		# burst_index every row and let §16's gate fire a stray wrap-up.
		pod_combat_clear_per_target_state_effect = yes
	}
}

# ─── SYSTEM LOG ENTRY ────────────────────────────────────────────
# Move-less log entry carrying only a system_message flag, for
# lifecycle bookends (combat start, victory, defeat). The GUI renders
# these via PODCombatLogSystemMessageText; actor / target / move columns
# collapse via their IsSet gates since nothing is captured on them.
# Caller scope: PC; entry owned by Charon, as regular entries.
pod_combat_push_system_log_effect = {
	# $MESSAGE_KEY$ = flag name (combat_started / combat_ended_victory
	#                / combat_ended_defeat / round_begins). Each one
	#                needs a matching pod_combat_log_system_$MESSAGE_KEY$
	#                loc key and a dispatcher branch in
	#                PODCombatLogSystemMessageText.
	save_scope_as = combat_host

	# hidden_effect_new_object: the tooltip-preview pass skips create_story,
	# which would leave scope:new_log_entry unset and crash the downstream
	# read with "Undefined event target 'new_log_entry'".
	hidden_effect_new_object = {
		character:POD_wraith_01 = {
			create_story = {
				type = POD_combat_log_entry
				save_scope_as = new_log_entry
			}
		}
		scope:new_log_entry = {
			set_variable = { name = system_message value = flag:$MESSAGE_KEY$ }
			# -1 actor sentinel (as regular push_log) so the GUI's side tint
			# collapses cleanly for system rows.
			set_variable = { name = actor_side  value = -1 }
			# Round snapshot on every system entry, for later "row in round Y"
			# display without re-deriving from log position.
			if = {
				limit = { scope:combat_host = { has_variable = pod_combat_round } }
				set_variable = { name = round value = scope:combat_host.var:pod_combat_round }
			}
		}
		pod_combat_prepend_capped_log_entry_effect = yes
	}
}

	# also stamps entry.var:actor with $CHARACTER$ so the loc key can name the
	# combatant (e.g. "<name> joins the fray"). $MESSAGE_KEY$ picks the dispatcher

	hidden_effect_new_object = {

			# -1 keeps the row in the neutral system lane; the joiner is named in the
			# text, not via the GUI's team-coloured actor column.
			set_variable = { name = actor_side value = -1 }
			if = {

	}

# ─── POST-COMBAT HEALTH FINALIZATION ─────────────────────────────
# Post-combat nonfatal-mode finalizer. Damage is applied LIVE during
# move execution (the damage helpers in pod_combat_effects.txt call
# POD_increase_wounds_* immediately); this effect runs once at combat
# end, BEFORE cleanup, and only handles fatality-mode postprocessing:
#   - fatality = no  →  cap every populated slot's wound track to leave
#                       one empty box, then strip pod_combat_nonfatal_active.
#   - fatality = always / default / practice  →  no-op.

pod_combat_apply_nonfatal_effect = {
	# Only fatality = no acts. During the fight its inline death check is
	# gated by the pod_combat_nonfatal_active flag (see POD_combat/
	# pod_combat_death_gate_overrides.txt), so wounds accumulate but
	# nobody dies; here we cap the track THEN strip the flag (order
	# matters - see below). practice writes no HP at all; default lets
	# damage persist as-is.

	if = {
		limit = {
			has_variable = pod_combat_sc_context
			var:pod_combat_sc_context.var:fatality = flag:no
		}
		# Cap pass: keep one empty box per combatant, peeling a superficial
		# (preferred) or aggravated wound if the track is full.
		every_in_list = {
			variable = pod_combat_active_slots
			var:slot_char = {
				save_scope_value_as = {
					name  = nf_empty
					value = POD_empty_damage_boxes
				}
				if = {
					limit = { scope:nf_empty <= 0 }
					if = {
						limit = {
							has_variable = POD_health_damage_num
							var:POD_health_damage_num >= 1
						}
						change_variable = { name = POD_health_damage_num add = -1 }
					}
					else_if = {
						limit = {
							has_variable = POD_health_aggravated_damage_num
							var:POD_health_aggravated_damage_num >= 1
						}
						change_variable = { name = POD_health_aggravated_damage_num add = -1 }
					}
					POD_update_wound_trait_xp = yes
				}
			}
		}
		# Strip the nonfatal flag AFTER the cap - order matters.
		pod_combat_clear_nonfatal_flag_effect = yes
	}
}

# ─── HEALING EFFECTS ────────────────────────────────────────────────
# Mirror the damage side: scope-switch to the target, call PoD's heal
# effect, re-derive workspace HP from the wound state.

pod_combat_route_to_pod_heal = {
	# $SIDE$ = "actor" or "target"
	# $AGGRAVATED$ = yes  → POD_heal_aggravated_wound_effect (agg boxes;
	#                       can push out into superficial)
	#                no   → POD_heal_superficial_wound_effect (superficial only)
	# $AMOUNT$ = boxes to heal (1 HP = 1 box).
	# Sets scope:heal_applied = actual amount healed (clamped to wounds).

	# Practice mode inflicted no wounds: nothing to heal, return 0.
	if = {
		limit = {
			has_variable = pod_combat_sc_context
			var:pod_combat_sc_context.var:fatality = flag:practice
		}
		save_scope_value_as = { name = heal_applied value = 0 }
		# 0 lets the dispatcher render "-" instead of "+0 HP".
		set_variable = { name = pod_combat_last_heal_amount value = 0 }
		pod_combat_set_outcome_if_unset_effect = { CODE = heal }
	}
	else = {
		var:pod_combat_$SIDE$_char = {
			save_scope_value_as = { name = pre_total_wounds value = POD_total_damage_boxes }
			if = {
				limit = { always = $AGGRAVATED$ }
				POD_heal_aggravated_wound_effect = { COUNT = $AMOUNT$ }
			}
			else = {
				POD_heal_superficial_wound_effect = { COUNT = $AMOUNT$ }
			}
			save_scope_value_as = { name = post_total_wounds value = POD_total_damage_boxes }
		}
		save_scope_value_as = {
			name = heal_applied
			value = {
				value = scope:pre_total_wounds
				subtract = scope:post_total_wounds
				min = 0
			}
		}
		# Workspace HP SVs derive from the char on read, so no update here.
		# Own var keeps pod_combat_last_damage intact on damage+heal moves.
		set_variable = { name = pod_combat_last_heal_amount value = scope:heal_applied }
		pod_combat_set_outcome_if_unset_effect = { CODE = heal }
	}
}

pod_combat_restore_hp_superficial_effect = {
	# $SIDE$ = "actor" or "target"
	# $AMOUNT$ = superficial HP to restore (aggravated untouched; PoD
	#            clamps to current superficial count).
	pod_combat_route_to_pod_heal = {
		SIDE       = $SIDE$
		AGGRAVATED = no
		AMOUNT     = $AMOUNT$
	}
}

pod_combat_restore_hp_aggravated_effect = {
	# $SIDE$ = "actor" or "target"
	# $AMOUNT$ = aggravated HP to restore (rare/potent: Obeah Mend,
	#            high-tier Fortitude, soul-drain).
	pod_combat_route_to_pod_heal = {
		SIDE       = $SIDE$
		AGGRAVATED = yes
		AMOUNT     = $AMOUNT$
	}
}

# ─── END COMBAT (INLINE) ──────────────────────────────────────────
# Runs inline from the victory check, auto-resolve, or forfeit path.
# Duel-driven combat (pod_combat_sc_context present) hands off to the
# pod_combat.9999 dispatcher; standalone/debug combat shows a toast.

pod_combat_end_combat_effect = {
	# Re-entry guard: the skill button's GUI execute and the "End Combat"
	# button both call this when pod_combat_is_over is set. The first call
	# removes pod_combat_active (SC-driven) or runs full cleanup (standalone),
	# so subsequent calls must be no-ops.
	if = {
		limit = { NOT = { has_variable = pod_combat_active } }
		# Already ended - nothing to do
	}
	else = {

	# Nonfatal cap + flag strip (header explains the mode handling).
	pod_combat_apply_nonfatal_effect = yes

	# Save winner/loser scopes first so the duel-end effect (and any
	# submod override of it) has named access to the outcome characters.
	if = {
		limit = { has_variable = pod_combat_winner }
		var:pod_combat_winner = { save_scope_as = pod_combat_winner }
	}
	if = {
		limit = { has_variable = pod_combat_loser }
		var:pod_combat_loser = { save_scope_as = pod_combat_loser }
	}
	# Duel-end extension point - runs with slot vars still intact. See
	# its definition below for the submod-override contract.
	pod_combat_duel_end_effect = yes

	# Duel-launched combat: fire the post-combat dispatch (pod_combat.9999)
	# which restores scopes from the context story, applies death, fires
	# the follow-up, then ends the story in cleanup.
	if = {
		limit = { has_variable = pod_combat_sc_context }
		# Close the combat window immediately but keep pod_combat_is_over
		# as a re-entry guard (the victory check gates on NOT is_over).
		# pod_combat.9999 handles full cleanup after firing the follow-up.
		remove_variable = pod_combat_active
		trigger_event = { id = pod_combat.9999 }
	}

	# For standalone/debug combat (no duel override), show a toast
	# and clean up immediately (no scope bridge needed).
	else = {
		if = {
			limit = {
				has_variable = pod_combat_winner
				var:pod_combat_winner ?= root
			}
			send_interface_toast = {
				title = pod_combat_result_victory_toast
				left_icon = root
			}
		}
		else = {
			send_interface_toast = {
				title = pod_combat_result_defeat_toast
				left_icon = root
			}
		}
		pod_combat_cleanup_effect = yes
	}

	} # end re-entry guard
}

# ─── DUEL-END EFFECT ────────────────────────────────────────────────
# Scripted-effect equivalent of an on_action handler, runs once at
# combat end (called by pod_combat_end_combat_effect BEFORE
# pod_combat_cleanup_effect strips the slot vars). Two jobs per
# combatant slot:
#   1. Strip discipline battle modifiers added by
#      POD_{enemy,ally}_knight_set_discipline_battle_effect at init,
#      so discipline bonuses don't leak outside of the duel.
#   2. Remove the pod_combat_duel_active state marker so other
#      systems (feeding, schemes, travel) stop deferring on this
#      character. The modifier has a safety-net 30-day expiry, but
#      the explicit removal keeps state clean.
#
# Submods wanting to extend can override this scripted effect via
# load order, or wrap it with their own pre/post effects.

pod_combat_duel_end_effect = {
	# Filled-only list, empty during tooltip-build (auto no-op there).
	every_in_list = {
		variable = pod_combat_active_slots
		var:slot_char = {
			POD_clear_discipline_battle_effect = yes
			remove_character_modifier = pod_combat_duel_active
		}
	}

	# Don't strip pod_combat_nonfatal_active here: pod_combat_apply_nonfatal_effect
	# owns it and must cap THEN strip (a strip before the cap would let
	# the inline death check fire during the cap pass).
}

# ─── CLEANUP ───────────────────────────────────────────────────────

pod_combat_cleanup_effect = {
	# Tier vars across all slots (stat vars go via group_cleanup below).
	pod_combat_clear_combat_tiers_effect = yes

	# (Discipline modifiers + duel-active marker already stripped by
	# pod_combat_duel_end_effect, which runs before this.)

	# Remove all combat variables from the player character
	remove_variable = pod_combat_active
	remove_variable = pod_combat_is_over
	# ── Single-combat caller-context teardown ──
	# All sc_* state lives on the POD_combat_sc_context story; end_story
	# drops it atomically (so new aliases need no matching remove here),
	# then clear the PC-side pointer. No-op for standalone/debug.
	if = {
		limit = { has_variable = pod_combat_sc_context }
		var:pod_combat_sc_context = { end_story = yes }
	}
	remove_variable = pod_combat_sc_context
	remove_variable = pod_combat_actor_char
	remove_variable = pod_combat_target_char
	remove_variable = pod_combat_round
	remove_variable = pod_combat_current_ap
	remove_variable = pod_combat_actor_soak
	remove_variable = pod_combat_actor_morale
	remove_variable = pod_combat_target_soak
	remove_variable = pod_combat_target_morale
	remove_variable = pod_combat_actor_is_guarding
	remove_variable = pod_combat_target_is_guarding
	remove_variable = pod_combat_winner
	remove_variable = pod_combat_loser
	# (ai_last_move / _prev_move / _prev2_move live on the AI char, not
	# the host; cleared by group_cleanup + slot-init.)
	remove_variable = pod_combat_ai_saved_enemy_target_team
	remove_variable = pod_combat_ai_saved_enemy_target_position
	# Per-action result transients via the shared helper (new
	# pod_combat_last_* fields update only the helper, not this site).
	pod_combat_clear_last_result_vars_effect = yes
	# Damage system tracking
	remove_variable = pod_combat_actor_base_soak
	remove_variable = pod_combat_target_base_soak
	remove_variable = pod_combat_actor_base_hit_chance
	remove_variable = pod_combat_target_base_hit_chance
	remove_variable = pod_combat_actor_energy_used
	remove_variable = pod_combat_target_energy_used
	remove_variable = pod_combat_actor_is_dodging
	remove_variable = pod_combat_target_is_dodging
	remove_variable = pod_combat_actor_is_stealthed
	remove_variable = pod_combat_target_is_stealthed
	# Workspace mirrors only - the slot-side originals die with their
	# slot stories (pod_combat_clear_slot_stories_effect).
	remove_variable = pod_combat_actor_ap_drain
	remove_variable = pod_combat_target_ap_drain
	remove_variable = pod_combat_actor_init_bonus
	remove_variable = pod_combat_target_init_bonus
	remove_variable = pod_combat_actor_retaliate_hit
	remove_variable = pod_combat_target_retaliate_hit
	remove_variable = pod_combat_actor_retaliate_miss
	remove_variable = pod_combat_target_retaliate_miss
	remove_variable = pod_combat_actor_retaliate_hit_agg
	remove_variable = pod_combat_target_retaliate_hit_agg
	remove_variable = pod_combat_actor_retaliate_miss_agg
	remove_variable = pod_combat_target_retaliate_miss_agg
	remove_variable = pod_combat_actor_prowess
	remove_variable = pod_combat_target_prowess
	remove_variable = pod_combat_actor_hit_chance
	remove_variable = pod_combat_target_hit_chance
	remove_variable = pod_combat_current_hit_chance
	# Log entries: end_story each, clear the list, and clear the rebuild
	# scratch list (defensive against a push that errored mid-rebuild).
	every_in_list = {
		variable = pod_combat_log
		end_story = yes
	}
	clear_variable_list = pod_combat_log
	clear_variable_list = pod_combat_log_rebuild
	# Clean up group combat slot variables
	pod_combat_group_cleanup_effect = yes
}
