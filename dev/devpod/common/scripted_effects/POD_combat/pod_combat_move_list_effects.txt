# Writes the body of POD_initialize_combat_moves, which
# runs once at on_game_start (see
# pod_combat_on_actions.txt) and registers every move
# story_cycle on Charon. Save-version mismatch is handled
# separately by POD_reload_combat_moves, which fires
# whenever POD_combat_version differs from
# POD_current_combat_version.
#
# Peer files:
#   pod_combat_move_init_effects.txt  — setters + constructor + reload
#   pod_combat_move_effects.txt       — execution engine

# ══════════════════════════════════════════════════════════
# TIER CONSTANTS
# ══════════════════════════════════════════════════════════
# Rebalance-in-one-place macros. Every move's numeric cost /
# damage / healing references one of these, so changing the
# constant's value rebalances every move using it.
# Outlier numeric values that don't fit any tier print as
# raw integers.

POD_initialize_combat_moves = {
	# Record the version — POD_reload_combat_moves checks this
	# against the current POD_current_combat_version to detect
	# save-data mismatch and rebuild the global catalogue.
	set_global_variable = {
		name  = POD_combat_version
		value = POD_current_combat_version
	}

	# Suppress 'flag used but never set' warnings 
	# for flags referenced in AI-dispatch limits.
	# This branch never runs at load-time (always = no).
	hidden_effect = {
		if = {
			limit = { always = no }
			character:POD_wraith_01 = {
				set_variable = { name = POD_combat_flag_sink value = flag:wounded }
				set_variable = { name = POD_combat_flag_sink value = flag:healthy }
				set_variable = { name = POD_combat_flag_sink value = flag:round_robin }
				set_variable = { name = POD_combat_flag_sink value = flag:threat }
				set_variable = { name = POD_combat_flag_sink value = flag:single_enemy }
				set_variable = { name = POD_combat_flag_sink value = flag:single_ally }
				set_variable = { name = POD_combat_flag_sink value = flag:single_pc }
				set_variable = { name = POD_combat_flag_sink value = flag:self }
				set_variable = { name = POD_combat_flag_sink value = flag:team_enemy }
				set_variable = { name = POD_combat_flag_sink value = flag:team_ally }
				set_variable = { name = POD_combat_flag_sink value = flag:fire }
				set_variable = { name = POD_combat_flag_sink value = flag:cold }
				set_variable = { name = POD_combat_flag_sink value = flag:lightning }
				set_variable = { name = POD_combat_flag_sink value = flag:sun }
				set_variable = { name = POD_combat_flag_sink value = flag:silver }
				set_variable = { name = POD_combat_flag_sink value = flag:gold }
				set_variable = { name = POD_combat_flag_sink value = flag:holy }
				set_variable = { name = POD_combat_flag_sink value = flag:balefire }
				set_variable = { name = POD_combat_flag_sink value = flag:superficial }
				set_variable = { name = POD_combat_flag_sink value = flag:aggravated }
				set_variable = { name = POD_combat_flag_sink value = flag:ATK }
				set_variable = { name = POD_combat_flag_sink value = flag:DEF }
				set_variable = { name = POD_combat_flag_sink value = flag:CTL }
				set_variable = { name = POD_combat_flag_sink value = flag:PWR }
				set_variable = { name = POD_combat_flag_sink value = flag:always }
				set_variable = { name = POD_combat_flag_sink value = flag:on_hp_damage }
				set_variable = { name = POD_combat_flag_sink value = flag:on_contact }
				set_variable = { name = POD_combat_flag_sink value = flag:on_miss }
				remove_variable = POD_combat_flag_sink
			}
		}
	}

	# Suppress 'variable used but never set' warnings for
	# every optional story variable the execution engine
	# or AI dispatcher reads via has_variable. Covers the
	# set of attributes the modular setters can write.
	hidden_effect = {
		if = {
			limit = { always = no }
			character:POD_wraith_01 = {
				set_variable = ap_cost
				set_variable = energy_cost
				set_variable = gold_cost
				set_variable = stress_cost
				set_variable = piety_cost
				set_variable = prestige_cost
				set_variable = hp_damage
				set_variable = morale_damage
				set_variable = self_damage
				set_variable = hp_heal
				set_variable = morale_heal
				set_variable = heal_type
				set_variable = damage_source
				set_variable = damage_type
				set_variable = soak_buff_round_tier
				set_variable = soak_buff_combat_tier
				set_variable = soak_debuff_round_tier
				set_variable = soak_debuff_combat_tier
				set_variable = hit_buff_round_tier
				set_variable = hit_buff_combat_tier
				set_variable = hit_debuff_round_tier
				set_variable = hit_debuff_combat_tier
				set_variable = set_guard
				set_variable = dodge
				set_variable = stealth
				set_variable = remove_guard
				set_variable = bypass_dodge
				set_variable = bypass_stealth
				set_variable = bypass_guard
				set_variable = retaliate_if_hit
				set_variable = retaliate_if_missed
				set_variable = retaliate_agg
				set_variable = is_starter_move
				set_variable = drain
				set_variable = ap_gain
				set_variable = ap_drain
				set_variable = initiative_debuff
				set_variable = retaliate_damage
				set_variable = target_priority
				set_variable = checks_spirit_gate
				set_variable = requires_hit_roll
				set_variable = checks_guard
				set_variable = checks_stance
				set_variable = heal_condition
				set_variable = morale_damage_condition
				set_variable = self_damage_condition
			}
		}
	}

	# Suppress 'set but never used' on story / log-entry variables
	# whose consumers the analyzer can't follow:
	#   result_code — read by PODCombatLogResultText custom_loc.
	#   follow_up   — read by pod_combat.9999 via
	#                 trigger_event saved_event_id.
	# Reading them here in a never-run branch counts as use
	# from the analyzer's perspective.
	hidden_effect = {
		if = {
			limit = { always = no }
			character:POD_wraith_01 = {
				if = { limit = { has_variable = result_code             } }
				if = { limit = { has_variable = follow_up               } }
			}
		}
	}

	# Suppress 'event target is used but never set' for the temp
	# scopes the customizable_localization log-row dispatcher reads
	# (entry, move, rcode). They're set via save_temporary_scope_as
	# inside loc-render-time evaluation — outside any
	# scripted-effect, so the analyzer never sees the writes.
	hidden_effect = {
		if = {
			limit = { always = no }
			save_scope_as = entry
			save_scope_as = move
			save_scope_as = rcode
		}
	}
