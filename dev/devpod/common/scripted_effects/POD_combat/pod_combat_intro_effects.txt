# First-fight editor and deferred launch effects. Deferral is armed by
# pod_combat_sc_intro_or_launch_effect; see COMBAT_REFERENCE.info section 10.

# PC scope, called by Configure Combat Skills and the intro review option.
# Callers control tooltip visibility. Preserve the initialization order below.
pod_combat_open_loadout_editor_effect = {
	if = {
		limit = {
			OR = {
				NOT = { exists = global_var:POD_combat_version }
				global_var:POD_combat_version != POD_current_combat_version
			}
			# Defer while any player is mid-fight (see the duel-create twin).
			NOT = { pod_combat_any_fight_active_trigger = yes }
		}
		POD_reload_combat_moves = yes
	}

	# Seed default loadout if empty (first time opening, new save,
	# or just after version reload which cleared the list).
	if = {
		limit = { pod_combat_no_moves_equipped_trigger = yes }
		POD_apply_default_combat_loadout = yes
	}

	set_variable = { name = pod_combat_config_open value = yes }

	# Bake unlocks after catalogue reload and loadout seeding.
	pod_combat_bake_pc_catalogue_unlocks_effect = yes

	# Seed the loadout-count display cache so the pip strips read cheap
	# vars while the editor is open.
	pod_combat_refresh_loadout_counts_effect = yes

	# Rebuild after unlocks: the known-only sheet filter reads the baked list.
	if = {
		limit = { has_variable = pod_combat_sheet_view }
		pod_combat_sheet_rebuild_effect = yes
	}
}

# Deferred launch arm; only FATALITY varies.
# OUTPUT_EVENT must be a literal placeholder; restore the real event ID after launch.
# Fixed duels bypass deferral; deferred locale defaults to battlefield.
pod_combat_intro_launch_arm_effect = {
	pod_combat_sc_launch_minigame_effect = {
		OPPONENT     = scope:pod_combat_intro_opponent
		OUTPUT_EVENT = pod_combat.0299
		FATALITY     = $FATALITY$
		FIXED        = no
		LOCALE       = battlefield
	}
}

# PC scope, called by hidden carrier pod_combat.0202 from either Begin option.
# The carrier avoids walking the duel build during option tooltip preview.
# Keep this in the intro event chain: it preserves caller scopes for the shell.
# A scripted-GUI close action would have no inherited scope table.
pod_combat_intro_begin_fight_effect = {
	hidden_effect = {
		if = {
			limit = { pod_combat_intro_launch_valid_trigger = yes }
			# A vanilla duel during deferral may have cleared these markers; restore them before launch.
			scope:sc_attacker = { set_variable = { name = engaged_in_single_combat value = yes } }
			scope:sc_defender = { set_variable = { name = engaged_in_single_combat value = yes } }

			# Use a combat-specific alias so the shell cannot overwrite a caller's generic opponent scope.
			if = {
				limit = { this = scope:sc_attacker }
				scope:sc_defender = { save_scope_as = pod_combat_intro_opponent }
			}
			else = {
				scope:sc_attacker = { save_scope_as = pod_combat_intro_opponent }
			}

			# Pass literal fatality tokens to preserve flag type during deferred launch.
			# Keep arms aligned with POD_combat_restore_sc_rule_flags_effect in pod_combat_sc_context_effects.txt.
			if = {
				limit = { scope:fatality = flag:always }
				pod_combat_intro_launch_arm_effect = { FATALITY = always }
			}
			else_if = {
				limit = { scope:fatality = flag:default }
				pod_combat_intro_launch_arm_effect = { FATALITY = default }
			}
			else_if = {
				limit = { scope:fatality = flag:possible }
				pod_combat_intro_launch_arm_effect = { FATALITY = possible }
			}
			else_if = {
				limit = { scope:fatality = flag:practice }
				pod_combat_intro_launch_arm_effect = { FATALITY = practice }
			}
			else = {
				pod_combat_intro_launch_arm_effect = { FATALITY = no }
			}

			# Replace the placeholder before .9999 reads follow_up. Event-ID scope values preserve their type.
			var:pod_combat_session ?= {
				set_variable = { name = follow_up value = scope:follow_up_event }
			}

			remove_variable = pod_combat_intro_pending
		}
		else = {
			# Revalidation can fail while the intro waits. Release only living, unseated principals:
			# dead-character mutation errors, and a newer fight owns its participants' markers.
			if = {
				limit = {
					exists = scope:sc_attacker
					scope:sc_attacker = {
						is_alive = yes
						NOT = { pod_combat_is_seated_trigger = yes }
					}
				}
				scope:sc_attacker = { remove_variable = engaged_in_single_combat }
			}
			if = {
				limit = {
					exists = scope:sc_defender
					scope:sc_defender = {
						is_alive = yes
						NOT = { pod_combat_is_seated_trigger = yes }
					}
				}
				scope:sc_defender = { remove_variable = engaged_in_single_combat }
			}

			remove_variable = pod_combat_intro_pending

			# trigger_event no-ops on a dead recipient, matching Part 2's own
			# unguarded fire.
			scope:sc_initiator ?= {
				trigger_event = { saved_event_id = scope:invalidation_event }
			}
		}
	}
}
