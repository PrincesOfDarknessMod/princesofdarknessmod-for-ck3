POD_diablerie_V5_16gen_raise_effect = {		
		
		if = {
			limit = {
				is_vampgeneration16_trigger = yes
				scope:victim = { is_vampgeneration16_trigger = no }
			}
			
			if = {
				limit = {
					has_trait = vampgeneration16
				}
				remove_trait = vampgeneration16
			}

			if = {
				limit = {
					any_secret = { secret_type = secret_vampgeneration16 }
				}
				random_secret = {
					limit = { secret_type = secret_vampgeneration16 }
					remove_secret = yes
				}	
			}
			
			add_secret = {
				type = secret_vampgeneration15
			}	
		}
}

POD_diablerie_piety_loss_effect = {	

	if = {
		limit = {
			faith = {
				has_doctrine_parameter = sanctioned_diableries
			}
			scope:victim = {	
				OR = {
					has_trait = craven
					has_trait = arbitrary
					faith = {
						faith_hostility_level = {
						target = root.faith
						value = 3
						}
					}	
					}	
				}
			}
		add_piety = minor_piety_loss
	}

	else_if = {
		limit = {
			faith = {
				has_doctrine_parameter = sanctioned_diableries
			}
			scope:victim = {	
				OR = {
					has_trait = just
					has_trait = brave
					faith = {
						faith_hostility_level = {
						target = root.faith
						value = 0
						}
					}	
					}	
				}
			}
		add_piety = massive_piety_loss
	}		
	
	else_if = {
		limit = {
			faith = {
				has_doctrine_parameter = vampirediablerie_illegal
			}
		}
		add_piety = massive_piety_loss
	}
	
	else_if = {
		limit = {
			faith = {
				has_doctrine_parameter = vampirediablerie_shunned
			}
		}
		add_piety = massive_piety_loss
	}	
	
	else_if = {
		limit = {
			faith = {
				has_doctrine_parameter = vampirediablerie_ignorant
			}
		}
		add_piety = massive_piety_loss
	}		

	else_if = {
		limit = {
			faith = {
				has_doctrine_parameter = vampirediablerie_accepted
			}
		}
		add_piety = major_piety_loss
	}
	
}
