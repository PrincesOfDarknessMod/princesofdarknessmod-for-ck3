# Gratiano content: Abyss Mysticism effects
######################################

# Dispatch before POD_end_journey_effect; delayed botch events cannot read journey vars.
POD_gratiano_abyss_miscast_effect = {
	if = {
		limit = { var:current_journey.var:ritual = flag:one_with_the_dark }
		trigger_event = { id = POD_gratiano.0315 days = 1 }
	}
	else_if = {
		limit = { var:current_journey.var:ritual = flag:summoning_the_blatherskites }
		trigger_event = { id = POD_gratiano.0316 days = 1 }
	}
	else_if = {
		limit = { var:current_journey.var:ritual = flag:boukephos_gateway }
		trigger_event = { id = POD_gratiano.0317 days = 1 }
	}
	else = {
		random_list = {
			25 = { trigger_event = { id = POD_gratiano.0310 days = 1 } }
			25 = { trigger_event = { id = POD_gratiano.0311 days = 1 } }
			25 = { trigger_event = { id = POD_gratiano.0312 days = 1 } }
			25 = { trigger_event = { id = POD_gratiano.0313 days = 1 } }
		}
	}
}

POD_gratiano_abyss_touched_threshold_sync_effect = {
	if = {
		limit = {
			has_trait_xp = { trait = abyss_touched value >= 50 }
			NOT = { has_character_flag = POD_portrait_abyss_eyes_flag }
		}
		add_character_flag = POD_portrait_abyss_eyes_flag
	}
}

POD_gratiano_abyss_first_delve_milestone_effect = {
	if = {
		limit = { NOT = { has_character_flag = POD_gratiano_milestone_first_delve } }
		add_character_flag = POD_gratiano_milestone_first_delve
		trigger_event = { id = POD_gratiano.0340 days = 1 }
	}
}

######################################
# Wounded character scope, called from the sun-immunity branch.
# The triggering blow remains absorbed.
######################################
POD_gratiano_syllable_sun_touched_effect = {
	if = {
		limit = {
			has_character_modifier = syllable_of_unbeing_modifier
			NOT = { has_character_flag = POD_gratiano_syllable_failing }
		}
		add_character_flag = {
			flag = POD_gratiano_syllable_failing
			days = 45
		}
		# remove + add: re-adding a held modifier is not documented to reset its duration
		remove_character_modifier = syllable_of_unbeing_modifier
		add_character_modifier = {
			modifier = syllable_of_unbeing_modifier
			days = 30
		}
		save_scope_as = gratiano_syllable_holder
		send_interface_toast = {
			title = POD_gratiano_syllable_failing_toast
			left_icon = scope:gratiano_syllable_holder
			custom_tooltip = POD_gratiano_syllable_failing_tt
		}
		trigger_event = { id = POD_gratiano.0330 days = 40 }
	}
}
