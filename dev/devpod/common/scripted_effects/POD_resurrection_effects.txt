pod_death = {
	if = {
		limit = {
			OR = {
				NOT = { exists = this }
				is_alive = no 
			}
		}
		# Do nothing
	}
	else = {
		if = {
			limit = { 
				has_character_modifier = life_warden_modifier 
				has_relation_life_warden = $KILLER$
			}
			remove_relation_life_warden = $KILLER$
		}
		if = {
			limit = { has_POD_resurrection_trigger = yes }
			pod_resurrection_effect = { KILLER = $KILLER$ }
		}
		else = {
			show_as_tooltip = {
				death = {
					death_reason = $DEATH_REASON$
					killer = $KILLER$
				}
			}
			hidden_effect = {
				if = {
					limit = { 
						can_convert_to_wraith_trigger = yes
						any_player = { has_character_modifier = soul_beacon_modifier }
					}
					pod_soul_beacon_effect = yes
				}
				random_list = {
					1 = {
						trigger = {
							can_convert_to_wraith_trigger = yes
						}
						pod_return_from_death_effect = yes
						modifier = {
							factor = 100
							has_trait = dhampyr
						}
					}
					1000 = {
						trigger = {
							#NOT = { has_perk = inurement_perk }
							NOT = { exists = var:marked_for_return }
						}
						death = {
							death_reason = $DEATH_REASON$
							killer = $KILLER$
						}
					}
				}
			}
		}
	}
}

pod_death_no_killer = {
	if = {
		limit = {
			OR = {
				NOT = { exists = this }
				is_alive = no 
			}
		}
		# Do nothing
	}
	else = {
		pod_set_death_reason_flag_effect = { DEATH_REASON = $DEATH_REASON$ }
		if = {
			limit = { 
				has_POD_resurrection_trigger = yes
				NOT = { has_character_flag = death_suicide }
			}
			pod_resurrection_from_accident_effect = yes
		}
		else = {
			show_as_tooltip = { death = { death_reason = $DEATH_REASON$ } }
			hidden_effect = {
				if = {
					limit = { 
						any_player = { has_character_modifier = soul_beacon_modifier }
						can_convert_to_wraith_trigger = yes
					}
					pod_soul_beacon_effect = yes
				}
				random_list = {
					1 = {
						trigger = { can_convert_to_wraith_trigger = yes }
						
						pod_return_from_death_effect = yes
						modifier = {
							factor = 100
							has_trait = dhampyr
						}
					}
					1000 = {
						trigger = { NOT = { exists = var:marked_for_return } }
						death = {
							death_reason = $DEATH_REASON$
						}
					}
				}
			}
		}
	}
}

pod_return_from_death_effect = {
	if  = {
		limit = { 
			is_ruler = yes 
			is_ai = yes
		}
		depose = yes
	}
	if = {
		limit = { is_imprisoned = yes }
		release_from_prison = yes
	}
	if = {
		limit = { exists = var:marked_for_return }
		var:marked_for_return = {
			save_scope_as = necromancer
		}
	}
	POD_become_wraith_effect = yes
	if = {
		limit = { exists = scope:necromancer }
		if = {
			limit = { NOT = { location = scope:necromancer.location } } 
			set_employer = scope:necromancer
			return_to_court = yes
			add_character_flag = {
				flag = blocked_from_leaving
				years = 5
			}
		}
		save_scope_as = wraith
		scope:necromancer = {
			send_interface_message = {
				title = marked_wraith_has_arrive
				left_icon = scope:wraith
				right_icon = scope:necromancer
				if = {
					limit = { has_trait = mummy }
					add_hekau_necromancy_lifestyle_xp = medium_lifestyle_xp
				}
			}
			trigger_event = POD_wraith_binding.1000
		}
		remove_variable = marked_for_return
	}
}

pod_resurrection_effect = {
	save_scope_as = resurrected_victim 
	add_opinion = {
		target = $KILLER$
		modifier = POD_killed_me_opinion
	}
	hidden_effect = {
		if = {
			limit = { is_imprisoned = yes }
			release_from_prison = yes
		}
		every_targeting_scheme = { end_scheme = yes }
		remove_character_modifier = prison_of_the_mind
		remove_character_modifier = eye_of_sakhmet_modifier
		remove_character_modifier = thin_blood_perk_modifier
		remove_trait = demonic_plague 
		remove_trait = demonic_plague_non_infectious
		remove_trait = demonic_plague_non_lethal 
		remove_trait = demonic_plague_non_lethal_and_infectious
		remove_trait = sickly
		remove_trait = infirm
		remove_trait = incapable
		remove_trait = impotent
		remove_stress_trait_effect = yes
		remove_injury_trait_effect = yes
		remove_scar_trait_effect = yes
		remove_disease_trait_effect = yes
		if = {
			limit = { has_POD_shapeshifted_form_trigger = yes }
			POD_shapeshift_revert_all_effect = yes
		}
			
		if = {
			limit = { is_ai = no }
			set_variable = {
				name = my_killer
				value = $KILLER$
			}
		}
	
		$KILLER$ = { 
			send_interface_toast = {
				title = victim_has_resurrected_toast
				left_icon = $KILLER$
				right_icon = scope:resurrected_victim

				show_as_tooltip = { scope:resurrected_victim = { pod_resurrection_cost_effect = yes } }
			}
		}
		scope:resurrected_victim = { 
			send_interface_toast = {
				title = you_have_resurrected_toast
				left_icon = this
				right_icon = $KILLER$

				show_as_tooltip = { pod_resurrection_cost_effect = yes }
			}
		}
		# Send a message to all other players.
		every_player = {
			limit = {
				NOR = {
					this = scope:resurrected_victim
					this = $KILLER$
				}
				has_any_relation_trigger = { CHARACTER = scope:resurrected_victim }
			}
			send_interface_toast = {
				title = victim_has_resurrected_toast
				left_icon = scope:resurrected_victim
				right_icon = $KILLER$

				show_as_tooltip = { scope:resurrected_victim = { pod_resurrection_cost_effect = yes } }
			}
		}
		pod_resurrection_travel_effect = yes
	}
	pod_resurrection_cost_effect = yes
}

pod_resurrection_cost_effect = {
	if = {
		limit = { has_character_modifier = extra_life_4_modifier }
		remove_character_modifier = extra_life_4_modifier
		add_character_modifier = extra_life_3_modifier
	}
	else_if = {
		limit = { has_character_modifier = extra_life_3_modifier }
		remove_character_modifier = extra_life_3_modifier
		add_character_modifier = extra_life_2_modifier
	}
	else_if = {
		limit = { has_character_modifier = extra_life_2_modifier }
		remove_character_modifier = extra_life_2_modifier
		add_character_modifier = extra_life_1_modifier
	}
	else_if = {
		limit = { has_character_modifier = extra_life_1_modifier }
		remove_character_modifier = extra_life_1_modifier
	}
	else_if = {
		limit = { 
			has_trait = spectre 
		}
		trigger_event = {
			id = POD_umbra_lost.100
			days = 1
		}
	}
	else_if = {
		limit = { 
			has_trait = wraith 
			NOT = { has_trait = ferryman }
			NOT = { has_trait = shadow }
		}
		POD_become_spectre_effect = yes
	}
	else_if = {
		limit = { has_trait = demon }
		if = {
			limit = { has_trait = truefallen }
			POD_next_demon_vessel_effect = yes
		}
		if = {
			limit = { has_variable = stone_prison }
			var:stone_prison = { add_durability = -50 }
		}
		if = {
			limit = { has_variable = earth_bound }
			var:earth_bound = { change_county_control = -50 }
		}
	}
	else_if = {
		limit = { has_perk = ashes_of_the_phoenix_perk }
		remove_perk = ashes_of_the_phoenix_perk
	}
	else_if = {
		limit = { has_perk = sleep_of_the_hero_perk }
		remove_perk = sleep_of_the_hero_perk
		add_trait = torpor
		depose = yes
	}
	else_if = {
		limit = { has_perk = rebirthing_perk }
		if = { 
			limit = { has_trait = shapeshifter }
			POD_rage_lose_the_wolf_effect = yes
			POD_kinfolk_initialization_effect = yes
			POD_remove_wrong_supernatural_traits_effect = yes

				id = POD_werewolves_maintenance.2
				years = { 13 18 }
			}
			remove_character_flag = has_auspice_renunciation
		}
		remove_perk = rebirthing_perk
		add_trait = reincarnation
		refund_all_perks = yes
		set_age = 1
		change_current_weight = { value = current_weight multiply = -1 }
		if = { 
			limit = { has_character_flag = rebirthing_perk_delay }
			add_character_modifier = health_gain_6_modifier
			remove_character_flag = rebirthing_perk_delay
		}
	}
	else_if = {
		limit = { has_perk = draught_of_phoenix_blood_perk }
		if = { 
			limit = { has_trait = vampgeneration }
			change_trait_rank = {
				trait = vampgeneration
				rank = 1
				max = 16
			}
		}
		refund_all_perks = yes
	}
	else_if = {
		limit = { 
			num_of_relation_life_warden > 0
		}
		hidden_effect = { 
			random_relation = {
				type = life_warden
				send_interface_message = {
					type = event_generic_bad
					title = resurrection_cost_notification
					left_icon = this
					right_icon = prev

					add_piety =  massive_piety_loss
				}
			}

			increase_wounds_no_death_effect = { REASON = treatment }
			increase_wounds_no_death_effect = { REASON = treatment }
		}
	}
	
	if = {
		limit = { has_trait = asekh_sen }
		POD_become_hideous_effect = yes
	}
}

pod_soul_beacon_effect = {
	save_scope_as = interest
	random_player = {
		limit = {
			NOT = { this = scope:interest }
			OR = {
				AND = {
					has_character_flag = flag_soul_beacon_diplomacy
					scope:interest = { diplomacy > 25 }
				}
				AND = {
					has_character_flag = flag_soul_beacon_martial
					scope:interest = { martial > 25 }
				}
				AND = {
					has_character_flag = flag_soul_beacon_stewardship
					scope:interest = { stewardship > 25 }
				}
				AND = {
					has_character_flag = flag_soul_beacon_intrigue
					scope:interest = { intrigue > 25 }
				}
				AND = {
					has_character_flag = flag_soul_beacon_learning
					scope:interest = { learning > 25 }
				}
				AND = {
					has_character_flag = flag_soul_beacon_artifact
					scope:interest = { 
						any_character_artifact = {
							OR = {
								rarity = illustrious
								rarity = famed
							}
						}
					}
				}
				AND = {
					has_character_flag = flag_soul_beacon_gold
					scope:interest = { gold > 1500 }
				}
			}
		}
		set_variable = {
			name = marked_for_return
			value = scope:interest
			days = 10
		}
	}
}
