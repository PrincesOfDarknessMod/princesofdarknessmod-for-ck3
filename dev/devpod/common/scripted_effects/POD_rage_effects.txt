POD_rage_increase_effect = {
	if = {
		limit = {
			POD_splat_has_rage_trigger = yes
			NOT = { stress_level >= 3 }
			NOT = { has_character_modifier = POD_rage_5_modifier }	
		}
		custom_tooltip = POD_rage_increase_effect.tt

			if = {
				limit = {
					has_character_modifier = POD_rage_4_modifier
				}
					hidden_effect = {
						remove_character_modifier = POD_rage_4_modifier
					}
					add_character_modifier = {
						modifier = POD_rage_5_modifier						
					}
			}

			if = {
				limit = {
					has_character_modifier = POD_rage_3_modifier
				}
					hidden_effect = {
						remove_character_modifier = POD_rage_3_modifier
					}
					add_character_modifier = {
						modifier = POD_rage_4_modifier					
					}
			}

			if = {
				limit = {
					has_character_modifier = POD_rage_2_modifier
				}
					hidden_effect = {
						remove_character_modifier = POD_rage_2_modifier
					}
					add_character_modifier = {
						modifier = POD_rage_3_modifier						
					}
			}

			if = {
				limit = {
					has_character_modifier = POD_rage_1_modifier
				}
					hidden_effect = {
						remove_character_modifier = POD_rage_1_modifier
					}
					add_character_modifier = {
						modifier = POD_rage_2_modifier						
					}
			}

			if = {
				limit = {
					has_character_modifier = POD_rage_0_modifier

						remove_character_modifier = POD_rage_0_modifier

	}
}

POD_revert_to_breed_form_effect = {
	if = {
		limit = {
			has_trait = homid_born
			OR = {
				has_character_modifier = POD_werewolf_form_crinos_modifier
				has_character_modifier = POD_werewolf_form_glabro_modifier
				has_character_modifier = POD_werewolf_form_hispo_modifier
				has_character_modifier = POD_werewolf_form_lupus_modifier
			}
		}	
		custom_tooltip = POD_revert_to_breed_form_effect.tt
		POD_shapeshift_remove_werewolf_form_effect = yes
		hidden_effect = {
			send_interface_toast = {
				title = POD_revert_to_breed_form_effect.desc		
			}
		}
	}

	if = {
		limit = {
			has_trait = lupus_born
			NOT = { has_character_modifier = POD_werewolf_form_lupus_modifier }
		}
		custom_tooltip = POD_revert_to_breed_form_effect.tt
		POD_shapeshift_remove_werewolf_form_effect = yes
		add_character_modifier = {
			modifier = POD_werewolf_form_lupus_modifier
		}
		hidden_effect = {
			send_interface_toast = {
				title = POD_revert_to_breed_form_effect.desc		
			}
		}
	}

	if = {
		limit = {
			has_trait = metis
			NOT = { has_character_modifier = POD_werewolf_form_crinos_modifier }
		}
		custom_tooltip = POD_revert_to_breed_form_effect.tt
		POD_shapeshift_remove_werewolf_form_effect = yes
		add_character_modifier = {
			modifier = POD_werewolf_form_crinos_modifier
		}
		hidden_effect = {
			send_interface_toast = {
				title = POD_revert_to_breed_form_effect.desc		
			}
		}
	}
}
					



POD_rage_lose_the_wolf_effect = {
	if = {
		limit = {
			POD_splat_has_rage_trigger = yes
		}
		
		custom_tooltip = POD_rage_lose_the_wolf_effect.tt
		
		if = { 
			limit = {
				POD_is_werewolf_trigger = yes
			}
			remove_trait = werewolf
			add_character_flag = werewolf_breed
			add_trait = kinfolk	
			add_trait = gnosis
		}

		if = { 
			limit = {
				POD_is_ajaba_trigger = yes
			}
			remove_trait = ajaba
			add_character_flag = ajaba_breed
			add_trait = kinfolk
			add_trait = gnosis
		}	
		
		if = { 
			limit = {
				POD_is_bastet_trigger = yes
			}
			remove_trait = bastet
			add_character_flag = bastet_breed
			add_trait = kinfolk
			add_trait = gnosis
		}
		
		POD_remove_rage_modifiers_effect = yes
	}
}
