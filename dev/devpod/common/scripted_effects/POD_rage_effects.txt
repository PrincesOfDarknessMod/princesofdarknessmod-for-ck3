POD_rage_increase_effect = {

		}

POD_rage_lose_the_wolf_effect = {
	if = {
		limit = {
			POD_splat_has_rage_trigger = yes
		}
		
		custom_tooltip = POD_rage_lose_the_wolf_effect.tt
		
		if = { 
			limit = {
				POD_is_werewolf_trigger = yes
			}
			remove_trait = werewolf
			add_character_flag = werewolf_breed
			POD_shapeshift_remove_werewolf_form_effect = yes	
		}

		if = { 
			limit = {
				POD_is_ajaba_trigger = yes
			}
			remove_trait = ajaba
			add_character_flag = ajaba_breed
			POD_shapeshift_remove_ajaba_form_effect = yes
		}	
		
		if = { 
			limit = {
				POD_is_bastet_trigger = yes
			}
			remove_trait = bastet
			add_character_flag = bastet_breed
			POD_shapeshift_remove_bastet_form_effect = yes
		}
		
		add_trait = kinfolk	
		add_trait = gnosis
		POD_remove_rage_modifiers_effect = yes
	}
}
