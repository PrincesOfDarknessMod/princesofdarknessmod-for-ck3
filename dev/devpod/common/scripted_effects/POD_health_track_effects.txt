POD_increase_wounds_hostile_effect = {
	$ATTACKER$ = { save_scope_as = wound_attacker }
	POD_increase_wounds_effect = {
		REASON = $REASON$
		COUNT = $COUNT$
	}
	if = {
		limit = { POD_reflect_incoming_damage_as_superficial_trigger = yes }
		scope:wound_attacker ?= {
			POD_increase_wounds_effect = {
				REASON = wounds
				COUNT = $COUNT$
			}
		}
	}
}

POD_increase_wounds_effect = {
	save_scope_value_as = {
		name  = death_reason
		value = flag:death_$REASON$
	}
	save_scope_value_as = {
		name  = final_damage
		value = {
			value = $COUNT$
			multiply = POD_wound_multiplier_from_splat_weaknesses
			max = POD_damage_cap_per_death_reason
			ceiling = yes
		}
	}
	# Unconditional soak-count export for the combat log's partial-soak
	# note. Every invocation starts at 0 (the immunity / clay-man /
	# unsoakable paths never touch it); the per-point roll below overwrites it.
	# Without this, an AoE loop's later immune target would read the PREVIOUS
	# target's count - saved scopes persist for the whole effect chain.
	save_scope_value_as = { name = soaked_count value = 0 }
	POD_set_scope_value_from_trigger = {
		NAME = is_soakable
		TRIGGER = "POD_damage_type_is_soakable = { DEATH_REASON = scope:death_reason }"
	}
	POD_set_scope_value_from_trigger = {
		NAME = is_aggravated
		TRIGGER = "POD_damage_type_is_aggravated = { DEATH_REASON = scope:death_reason }"
	}
	# ehh
	# comment out this line if you feel like checking all the vanilla death_reasons from the error log
	POD_error_suppression_unset_flag = { FLAG = $REASON$ }
	if = {
		limit = {
			POD_has_immunity_to_damage_type = { DEATH_REASON = scope:death_reason }
		}
		send_interface_message = {
			type  = POD_damage_avoided
			title = POD_damage_immunity_title
			custom_tooltip = POD_damage_immunity_tt
			right_icon = this
		}
	}
	else_if = { 
		limit = { POD_has_clay_man_protection = yes }
		POD_trigger_clay_man_protection = yes
	}
	else = {
		# Snapshot the soak chance once; the loop below rolls it per point of final_damage.
		save_scope_value_as = { name = soak_pct value = POD_soak_chance }

		# an armour-shredding weapon lowers the soak of the packet
		# it is delivering. Applied HERE and not inside
		# POD_soak_chance because that script value is asked "how soaky is this
		# character" with no attacker in hand - the slot card, the AI hoist, the
		# character sidebar and the maim roll all evaluate it, and a term keyed
		# on scope:wound_attacker would leak into every one of them. This is the
		# single point where "this incoming blow, from this attacker" exists.
		# Sitting above the preview keeps the tooltip honest about the real odds.
		#
		# scope:combat_host narrows it to a live minigame dispatch. The
		# nested trigger_if is mandatory: neither AND nor OR short-circuits, so a
		# bare var read would fire on attackers that carry no penalty.
		# min = 0 is required - nothing downstream re-clamps soak_pct.
		if = {
			limit = {
				exists = scope:combat_host
				exists = scope:wound_attacker
				scope:wound_attacker = {
					trigger_if = {
						limit = { has_variable = pod_combat_weapon_soak_pen }
						var:pod_combat_weapon_soak_pen > 0
					}
					trigger_else = { always = no }
				}
			}
			save_scope_value_as = {
				name  = soak_pct
				value = {
					value    = scope:soak_pct
					subtract = scope:wound_attacker.var:pod_combat_weapon_soak_pen
					min      = 0
				}
			}
		}
		# The same weapon outside the minigame. There is no per-dispatch marker
		# out here, so the weapon-shaped death reason is what stands in for "this
		# wound was actually struck with the thing" - which is also what stops a
		# shredding sword from thinning armour against a poisoning. The weapon is
		# read straight off the attacker rather than from a transient.
		#
		# The combat exclusion is NOT redundant with the branch above: in a fight
		# the marker is only set for uses_weapon moves, so a bite or a mind blast
		# fails that branch and would otherwise fall through to here, where
		# REASON = duel is itself weapon-shaped. Recoil would shred the actor's
		# own armour the same way.
		else_if = {
			limit = {
				NOT = { exists = scope:combat_host }
				exists = scope:wound_attacker
				# Same reason as the classifier arm: the saved attacker is never
				# cleared, so refuse the self case rather than let a character's
				# own weapon thin their own armour against someone else's blow.
				NOT = { this = scope:wound_attacker }
				POD_is_death_by_weapon = { DEATH_REASON = scope:death_reason }
				scope:wound_attacker = {
					any_equipped_character_artifact = {
						artifact_slot_type = primary_armament
						prev = { can_benefit_from_artifact = prev }
						has_variable = artifact_weapon_soak_penalty
					}
				}
			}
			scope:wound_attacker = {
				random_equipped_character_artifact = {
					limit = {
						artifact_slot_type = primary_armament
						prev = { can_benefit_from_artifact = prev }
						has_variable = artifact_weapon_soak_penalty
					}
					save_scope_as = pod_weapon_wound
				}
			}
			save_scope_value_as = {
				name  = soak_pct
				value = {
					value    = scope:soak_pct
					subtract = scope:pod_weapon_wound.var:artifact_weapon_soak_penalty
					min      = 0
				}
			}
		}

		# Visible damage preview for caller tooltips (event options, interaction accept buttons).
		if = {
			limit = {
				scope:is_soakable = yes
				scope:soak_pct > 0
			}
			if = {
				limit = { scope:is_aggravated = yes }
				custom_tooltip = pod_wound_preview_aggravated_tt
			}
			else = {
				custom_tooltip = pod_wound_preview_superficial_tt
			}
		}
		else_if = {
			limit = { scope:is_soakable = yes }
			if = {
				limit = { scope:is_aggravated = yes }
				custom_tooltip = pod_wound_preview_aggravated_nosoak_tt
			}
			else = {
				custom_tooltip = pod_wound_preview_superficial_nosoak_tt
			}
		}
		else = {
			if = {
				limit = { scope:is_aggravated = yes }
				custom_tooltip = pod_wound_preview_aggravated_unsoakable_tt
			}
			else = {
				custom_tooltip = pod_wound_preview_superficial_unsoakable_tt
			}
		}

		hidden_effect = {
			if = {
				limit = {
					scope:is_soakable = yes
					scope:soak_pct > 0
				}
				set_variable = { name = pod_soak_hits value = 0 }
				while = {
					count = scope:final_damage
					random = {
						chance = scope:soak_pct
						change_variable = { name = pod_soak_hits add = 1 }
					}
				}
				# Derive results and drop the counter BEFORE applying wounds, so the death
				# check inside POD_apply_wound_boxes can't leave a var-mutation on a dead ref.
				save_scope_value_as = { name = soaked_count  value = pod_soak_hits_value }
				save_scope_value_as = {
					name  = landed_damage
					value = { value = scope:final_damage subtract = pod_soak_hits_value min = 0 }
				}
				remove_variable = pod_soak_hits
			}
			else = {
				save_scope_value_as = { name = landed_damage value = scope:final_damage }
			}

			# Apply the landed points once (equivalent box / aggravated / death-check math
			# to base PoD's single application, just with the post-soak count).
			if = {
				limit = { scope:landed_damage > 0 }
				POD_apply_wound_boxes = {
					COUNT = scope:landed_damage
					AGGRAVATED = scope:is_aggravated
					REASON = $REASON$
				}
				POD_vanilla_wound_treatment = {
					REASON = $REASON$
				}
				if = {
					# Combat Mini-game practice-spar addition: no lasting balefire stacks in practice.
					limit = {
						scope:death_reason = flag:death_balefire
						NOT = { has_character_flag = pod_combat_practice_active }
					}
					while = {
						count = scope:landed_damage
						POD_add_modifier_and_track_stacks = {
							TYPE = character
							MODIFIER = balefire_health_modifier
						}
					}
				}
			}

			# One "soaked" notification when any point was avoided (loc shows "X of N").
			if = {
				limit = { scope:soaked_count > 0 }
				send_interface_message = {
					type  = POD_damage_avoided
					title = POD_damage_soaked_title
					custom_tooltip = POD_damage_soaked_tt
					right_icon = this
					POD_soak_aftereffects = yes
				}
			}
		}
	}
}

	POD_remove_character_modifier = { MODIFIER = POD_bottomless_depths_defense_modifier }

# simpler effect since both wound types work the same for the purposes of impairments
POD_convert_aggravated_to_superficial_wound = {
	save_scope_value_as = {
		name  = conversion
		value = {
			value = $COUNT$
			max = POD_aggravated_damage_boxes
		}
	}
	change_variable = {
		name = POD_health_aggravated_damage_num
		subtract = scope:conversion
	}
	change_variable = {
		name = POD_health_damage_num
		add = scope:conversion
	}
	POD_remove_modifier_and_track_stacks = {
		TYPE = character
		MODIFIER = balefire_health_modifier
	}
}

# need to pass TRAIT and REASON
# (the REASON is used to determine whether the damage can be soaked)
POD_add_maim_trait = {
	save_scope_value_as = {
		name  = death_reason
		value = flag:death_$REASON$
	}
	POD_set_scope_value_from_trigger = {
		NAME = is_soakable
		TRIGGER = "POD_damage_type_is_soakable = { DEATH_REASON = scope:death_reason }"
	}
	POD_set_scope_value_from_trigger = {
		NAME = is_aggravated
		TRIGGER = "POD_damage_type_is_aggravated = { DEATH_REASON = scope:death_reason }"
	}
	# ehh
	# comment out this line if you feel like checking all the vanilla death_reasons from the error log
	POD_error_suppression_unset_flag = { FLAG = $REASON$ }
	if = {
		limit = {
			POD_has_immunity_to_damage_type = { DEATH_REASON = scope:death_reason }
		}
		send_interface_message = {
			type  = POD_damage_avoided
			title = POD_damage_immunity_title
			custom_tooltip = POD_damage_immunity_tt
			right_icon = this
		}
	}
	else_if = { 
		limit = { POD_has_clay_man_protection = yes }
		POD_trigger_clay_man_protection = yes
	}
	else = {
		random_list = {
			0 = {
				trigger = { scope:is_soakable = yes }
				modifier = { add = POD_soak_chance }
				send_interface_message = {
					type  = POD_damage_avoided
					title = POD_damage_soaked_title
					custom_tooltip = POD_maim_soaked_tt
					right_icon = this
					POD_soak_aftereffects = yes
				}
			}
			100 = {
				modifier = {
					scope:is_soakable = yes
					add = {
						value = POD_soak_chance
						multiply = -1
					}
				}
				add_trait = $TRAIT$
			}
		}
	}
}

# need to pass COUNT, AGGRAVATED, and REASON
POD_apply_wound_boxes = {
	save_scope_value_as = {
		name = superficial_damage_added
		value = {
			if = {
				limit = { $AGGRAVATED$ = yes }
				value = 0
			}
			else = {
				value = $COUNT$
				max = POD_empty_damage_boxes
			}
		}
	}
	# overflow of normal wounds turns into aggravated wounds
	save_scope_value_as = {
		name = aggravated_damage_added
		value = {
			value = $COUNT$
			subtract = scope:superficial_damage_added
		}
	}
	# aggravated wounds can "push out" existing normal wounds
	if = {
		limit = {
			#$AGGRAVATED$ = yes
			$COUNT$ > POD_empty_damage_boxes
		}
		save_scope_value_as = {
			name = superficial_damage_added
			value = {
				value = scope:superficial_damage_added
				subtract = {
					value = $COUNT$
					subtract = POD_empty_damage_boxes
				}
			}
		}
	}
	
	# saving these as scope values first so they can be used later
	save_scope_value_as = {
		name = superficial_damage_boxes
		value = {
			value = POD_superficial_damage_boxes
			add = scope:superficial_damage_added
			min = 0
		}
	}
	save_scope_value_as = {
		name = aggravated_damage_boxes
		value = {
			value = POD_aggravated_damage_boxes
			add = scope:aggravated_damage_added
			max = POD_health_track_length
		}
	}
	save_scope_value_as = {
		name = total_wounds_gained
		value = {
			value = scope:superficial_damage_added
			min = 0
			add = scope:aggravated_damage_added
		}
	}
	
	# change wound variables
	set_variable = { 
		name = POD_health_damage_num
		value = scope:superficial_damage_boxes
	}
	set_variable = { 
		name = POD_health_aggravated_damage_num
		value = scope:aggravated_damage_boxes
	}
	
	# tooltips with interface message
	if = {
		limit = { scope:total_wounds_gained > 0 }
		
		send_interface_message = {
			type  = POD_damage_taken
			title = POD_damage_taken_title
			right_icon = this
			
			if = {
				limit = { scope:superficial_damage_added > 0 }
				custom_description = {
					text = POD_change_superficial_damage_desc
					value = scope:superficial_damage_added
				}
			}
			if = {
				limit = { scope:aggravated_damage_added > 0 }
				custom_description = {
					text = POD_change_aggravated_damage_desc
					value = scope:aggravated_damage_added
				}
			}
			
			#POD_side_effects_from_gaining_wounds = { COUNT = scope:total_wounds_gained } #disabled since the only effect there was removed for now
			POD_update_wound_trait_xp = yes
		}
	}
	
	POD_health_death_check_effect = {
		TOTAL_SUPERFICIAL_DAMAGE = scope:superficial_damage_boxes
		TOTAL_AGGRAVATED_DAMAGE = scope:aggravated_damage_boxes
		REASON = $REASON$
	}
}

POD_heal_damage = {
	save_scope_value_as = {
		name = aggravated_damage_healed
		value = {
			if = {
				limit = { $AGGRAVATED$ = no }
				value = 0
			}
			else = {
				value = $COUNT$
				max = POD_aggravated_damage_boxes
				multiply = -1
			}
		}
	}
	# aggravated wound healing also heals superficial wounds if there's "overflow"
	save_scope_value_as = {
		name = superficial_damage_healed
		value = {
			value = $COUNT$
			add = scope:aggravated_damage_healed
			max = POD_superficial_damage_boxes
			multiply = -1
		}
	}
	
	# tooltips
	if = {
		limit = { scope:superficial_damage_healed < 0 }
		custom_description = {
			text = POD_change_superficial_damage_desc
			value = scope:superficial_damage_healed
			set_variable = { 
				name = POD_health_damage_num
				value = { 
					value = POD_superficial_damage_boxes 
					add = scope:superficial_damage_healed
				}
			}
		}
	}
	if = {
		limit = { scope:aggravated_damage_healed < 0 }
		custom_description = {
			text = POD_change_aggravated_damage_desc
			value = scope:aggravated_damage_healed
			set_variable = { 
				name = POD_health_aggravated_damage_num
				value = { 
					value = POD_aggravated_damage_boxes 
					add = scope:aggravated_damage_healed
				}
			}
		}
	}
	hidden_effect = {
		POD_update_wound_trait_xp = yes
		if = {
			limit = { $AGGRAVATED$ = yes }
			POD_remove_modifier_and_track_stacks = {
				TYPE = character
				MODIFIER = balefire_health_modifier
			}
		}
	}
}

POD_update_wound_trait_xp = {
	if = {
		limit = { POD_has_wrong_wound_trait = yes }
		hidden_effect = { POD_trait_injector_wound = { EFFECT = POD_remove_trait } }
		if = {
			limit = { POD_health_track_percentage >= POD_wound_3_xp_threshold }
			add_trait = wounded_3
		}
		else_if = {
			limit = { POD_health_track_percentage >= POD_wound_2_xp_threshold }
			add_trait = wounded_2
		}
		else_if = {
			limit = { POD_health_track_percentage >= POD_wound_1_xp_threshold }
			add_trait = wounded_1
		}
		else_if = {
			limit = { POD_health_track_percentage >= 1 }
			add_trait = POD_wounded
		}
		else = {
			remove_variable = POD_health_damage_num
			remove_variable = POD_health_aggravated_damage_num
		}
	}
	else = {
		hidden_effect = {
			POD_trait_injector_wound_with_value = {
				EFFECT = POD_add_trait_xp
				VALUE  = -100
			}
		}
	}
	hidden_effect = {
		POD_trait_injector_wound_with_value = {
			EFFECT = POD_add_trait_xp
			VALUE  = POD_health_track_percentage
		}
		force_character_skill_recalculation = yes
	}
}

POD_health_death_check_effect = {
	if = {
		limit = {
			POD_health_track_length <= {
				value = $TOTAL_SUPERFICIAL_DAMAGE$
				add = $TOTAL_AGGRAVATED_DAMAGE$
			}
			$TOTAL_AGGRAVATED_DAMAGE$ >= POD_aggravated_damage_threshold_for_death
			POD_can_die_from_wounds = yes
		}
		if = { 
			limit = { 
				is_immortal = yes 
				POD_has_guaranteed_resurrection = no
			}
			POD_long_depose_effect = yes 
		}
		else_if = {
			limit = { exists = scope:wound_attacker }
			pod_death = {
				killer = scope:wound_attacker

			}
		}
		else = {
			pod_death_no_killer = {
				death_reason = death_$REASON$
			}
		}
	}
}

					save_temporary_scope_as = clay_man
					set_artifact_name = artifact_clay_man_amulet_name
				}
			}
		}
	}
