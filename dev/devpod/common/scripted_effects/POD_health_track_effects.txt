POD_increase_wounds_hostile_effect = {
	$ATTACKER$ = { save_scope_as = wound_attacker }
	POD_increase_wounds_effect = {
		REASON = $REASON$
		COUNT = $COUNT$
	}
}

POD_increase_wounds_effect = {
	save_scope_value_as = {
		name  = death_reason
		value = flag:death_$REASON$
	}
	save_scope_value_as = {
		name  = final_damage
		value = {
			value = $COUNT$
			multiply = POD_wound_multiplier_from_splat_weaknesses
			max = POD_damage_cap_per_death_reason
			ceiling = yes
		}
	}
	POD_set_scope_value_from_trigger = {
		NAME = is_soakable
		TRIGGER = "POD_damage_type_is_soakable = { DEATH_REASON = scope:death_reason }"
	}
	POD_set_scope_value_from_trigger = {
		NAME = is_aggravated
		TRIGGER = "POD_damage_type_is_aggravated = { DEATH_REASON = scope:death_reason }"
	}
	# ehh
	# comment out this line if you feel like checking all the vanilla death_reasons from the error log
	POD_error_suppression_unset_flag = { FLAG = $REASON$ }
	if = {
		limit = {
			POD_has_immunity_to_damage_type = { DEATH_REASON = scope:death_reason }
		}
		send_interface_message = {
			type  = POD_damage_avoided
			title = POD_damage_immunity_title
			custom_tooltip = POD_damage_immunity_tt
			right_icon = this
		}
	}
	else_if = { 
		limit = { POD_has_clay_man_protection = yes }
		POD_trigger_clay_man_protection = yes
	}
	else = {
		hidden_effect = {
			# Snapshot the soak chance once, then roll it per point of final_damage.
			save_scope_value_as = { name = soak_pct value = POD_soak_chance }
			if = {
				limit = {
					scope:is_soakable = yes
					scope:soak_pct > 0
				}
				set_variable = { name = podsoak_hits value = 0 }
				while = {
					count = scope:final_damage
					random = {
						chance = scope:soak_pct
						change_variable = { name = podsoak_hits add = 1 }
					}
				}
				# Derive results and drop the counter BEFORE applying wounds, so the death
				# check inside POD_apply_wound_boxes can't leave a var-mutation on a dead ref.
				save_scope_value_as = { name = soaked_count  value = pod_soak_hits_value }
				save_scope_value_as = {
					name  = landed_damage
					value = { value = scope:final_damage subtract = pod_soak_hits_value min = 0 }
				}
				remove_variable = podsoak_hits
			}
			else = {
				save_scope_value_as = { name = soaked_count  value = 0 }
				save_scope_value_as = { name = landed_damage value = scope:final_damage }
			}

			# Apply the landed points once (equivalent box / aggravated / death-check math
			# to base PoD's single application, just with the post-soak count).
			if = {
				limit = { scope:landed_damage > 0 }
				POD_apply_wound_boxes = {
					COUNT = scope:landed_damage
					AGGRAVATED = scope:is_aggravated
					REASON = $REASON$
				}
				POD_vanilla_wound_treatment = {
					REASON = $REASON$
				}
				if = {
					limit = { scope:death_reason = flag:death_balefire }
					while = {
						count = scope:landed_damage
						POD_add_modifier_and_track_stacks = {
							TYPE = character
							MODIFIER = balefire_health_modifier
						}
					}
				}
			}

			# One "soaked" notification when any point was avoided (loc shows "X of N").
			if = {
				limit = { scope:soaked_count > 0 }
				send_interface_message = {
					type  = POD_damage_avoided
					title = POD_damage_soaked_title
					custom_tooltip = POD_damage_soaked_tt
					right_icon = this
				}
			}
		}
	}
}

		name  = death_reason
		value = flag:death_$REASON$
	}
	POD_set_scope_value_from_trigger = {
		NAME = is_soakable
		TRIGGER = "POD_damage_type_is_soakable = { DEATH_REASON = scope:death_reason }"
	}
	POD_set_scope_value_from_trigger = {
		NAME = is_aggravated
		TRIGGER = "POD_damage_type_is_aggravated = { DEATH_REASON = scope:death_reason }"
	}
	# ehh
	# comment out this line if you feel like checking all the vanilla death_reasons from the error log
	POD_error_suppression_unset_flag = { FLAG = $REASON$ }
	if = {
		limit = {
			POD_has_immunity_to_damage_type = { DEATH_REASON = scope:death_reason }
		}
		send_interface_message = {
			type  = POD_damage_avoided
			title = POD_damage_immunity_title
			custom_tooltip = POD_damage_immunity_tt
			right_icon = this
		}
	}
	else_if = { 
		limit = { POD_has_clay_man_protection = yes }
		POD_trigger_clay_man_protection = yes
	}
	else = {
		random_list = {
			0 = {
				trigger = { scope:is_soakable = yes }
				modifier = { add = POD_soak_chance }
				send_interface_message = {
					type  = POD_damage_avoided
					title = POD_damage_soaked_title
					custom_tooltip = POD_maim_soaked_tt
					right_icon = this
				}
			}
			100 = {
				modifier = {
					scope:is_soakable = yes
					add = {
						value = POD_soak_chance
						multiply = -1
					}
				}
				add_trait = $TRAIT$
			}
		}
	}
}

# need to pass COUNT, AGGRAVATED, and REASON
POD_apply_wound_boxes = {
	save_scope_value_as = {
		name = superficial_damage_added
		value = {
			if = {
				limit = { $AGGRAVATED$ = yes }
				value = 0
			}
			else = {
				value = $COUNT$
				max = POD_empty_damage_boxes
			}
		}
	}
	# overflow of normal wounds turns into aggravated wounds
	save_scope_value_as = {
		name = aggravated_damage_added
		value = {
			value = $COUNT$
			subtract = scope:superficial_damage_added
		}
	}
	# aggravated wounds can "push out" existing normal wounds
	if = {
		limit = {
			#$AGGRAVATED$ = yes
			$COUNT$ > POD_empty_damage_boxes
		}
		save_scope_value_as = {
			name = superficial_damage_added
			value = {
				value = scope:superficial_damage_added
				subtract = scope:aggravated_damage_added
			}
		}
	}
	
	# saving these as scope values first so they can be used later
	save_scope_value_as = {
		name = superficial_damage_boxes
		value = {
			value = POD_superficial_damage_boxes
			add = scope:superficial_damage_added
			min = 0
		}
	}
	save_scope_value_as = {
		name = aggravated_damage_boxes
		value = {
			value = POD_aggravated_damage_boxes
			add = scope:aggravated_damage_added
			max = POD_health_track_length
		}
	}
	save_scope_value_as = {
		name = total_wounds_gained
		value = {
			value = scope:superficial_damage_added
			min = 0
			add = scope:aggravated_damage_added
		}
	}
	
	# change wound variables
	set_variable = { 
		name = POD_health_damage_num
		value = scope:superficial_damage_boxes
	}
	set_variable = { 
		name = POD_health_aggravated_damage_num
		value = scope:aggravated_damage_boxes
	}
	
	# tooltips with interface message
	if = {
		limit = { scope:total_wounds_gained > 0 }
		
		send_interface_message = {
			type  = POD_damage_taken
			title = POD_damage_taken_title
			right_icon = this
			
			if = {
				limit = { scope:superficial_damage_added > 0 }
				custom_description = {
					text = POD_change_superficial_damage_desc
					value = scope:superficial_damage_added
				}
			}
			if = {
				limit = { scope:aggravated_damage_added > 0 }
				custom_description = {
					text = POD_change_aggravated_damage_desc
					value = scope:aggravated_damage_added
				}
			}
			
			#POD_side_effects_from_gaining_wounds = { COUNT = scope:total_wounds_gained } #disabled since the only effect there was removed for now
			POD_update_wound_trait_xp = yes
		}
	}
	
	POD_health_death_check_effect = {
		TOTAL_SUPERFICIAL_DAMAGE = scope:superficial_damage_boxes
		TOTAL_AGGRAVATED_DAMAGE = scope:aggravated_damage_boxes
		REASON = $REASON$
	}
}

POD_heal_damage = {
	save_scope_value_as = {
		name = aggravated_damage_healed
		value = {
			if = {
				limit = { $AGGRAVATED$ = no }
				value = 0
			}
			else = {
				value = $COUNT$
				max = POD_aggravated_damage_boxes
				multiply = -1
			}
		}
	}
	# aggravated wound healing also heals superficial wounds if there's "overflow"
	save_scope_value_as = {
		name = superficial_damage_healed
		value = {
			value = $COUNT$
			add = scope:aggravated_damage_healed
			max = POD_superficial_damage_boxes
			multiply = -1
		}
	}
	
	# tooltips
	if = {
		limit = { scope:superficial_damage_healed < 0 }
		custom_description = {
			text = POD_change_superficial_damage_desc
			value = scope:superficial_damage_healed
			set_variable = { 
				name = POD_health_damage_num
				value = { 
					value = POD_superficial_damage_boxes 
					add = scope:superficial_damage_healed
				}
			}
		}
	}
	if = {
		limit = { scope:aggravated_damage_healed < 0 }
		custom_description = {
			text = POD_change_aggravated_damage_desc
			value = scope:aggravated_damage_healed
			set_variable = { 
				name = POD_health_aggravated_damage_num
				value = { 
					value = POD_aggravated_damage_boxes 
					add = scope:aggravated_damage_healed
				}
			}
		}
	}
	hidden_effect = {
		POD_update_wound_trait_xp = yes
		if = {
			limit = { $AGGRAVATED$ = yes }
			POD_remove_modifier_and_track_stacks = {
				TYPE = character
				MODIFIER = balefire_health_modifier
			}
		}
	}
}

POD_update_wound_trait_xp = {
	if = {
		limit = { POD_has_wrong_wound_trait = yes }
		hidden_effect = { POD_trait_injector_wound = { EFFECT = POD_remove_trait } }
		if = {
			limit = { POD_health_track_percentage >= POD_wound_3_xp_threshold }
			add_trait = wounded_3
		}
		else_if = {
			limit = { POD_health_track_percentage >= POD_wound_2_xp_threshold }
			add_trait = wounded_2
		}
		else_if = {
			limit = { POD_health_track_percentage >= POD_wound_1_xp_threshold }
			add_trait = wounded_1
		}
		else_if = {
			limit = { POD_health_track_percentage >= 1 }
			add_trait = POD_wounded
		}
		else = {
			remove_variable = POD_health_damage_num
			remove_variable = POD_health_aggravated_damage_num
		}
	}
	else = {
		hidden_effect = {
			POD_trait_injector_wound_with_value = {
				EFFECT = POD_add_trait_xp
				VALUE  = -100
			}
		}
	}
	hidden_effect = {
		POD_trait_injector_wound_with_value = {
			EFFECT = POD_add_trait_xp
			VALUE  = POD_health_track_percentage
		}
		force_character_skill_recalculation = yes
	}
}

POD_health_death_check_effect = {
	if = {
		limit = {
			POD_health_track_length <= {
				value = $TOTAL_SUPERFICIAL_DAMAGE$
				add = $TOTAL_AGGRAVATED_DAMAGE$
			}
			$TOTAL_AGGRAVATED_DAMAGE$ >= POD_aggravated_damage_threshold_for_death
			POD_can_die_from_wounds = yes
		}
		if = { 
			limit = { 
				is_immortal = yes 
				POD_has_guaranteed_resurrection = no
			}
			POD_long_depose_effect = yes 
		}
		else_if = {
			limit = { exists = scope:wound_attacker }
			pod_death = {
				killer = scope:wound_attacker
				death_reason = death_$REASON$
			}
		}
		else = {
			pod_death_no_killer = {
				death_reason = death_$REASON$
			}
		}
	}
}

		}
	}

# based on the effects in common\scripted_effects\20_health_effects.txt
# e.g. increase_wounds_no_death_effect
POD_vanilla_wound_treatment = {
	if = { #Wounds from treatments give no infection and no additional treatment
		limit = { NOT = { flag:death_$REASON$ = flag:death_treatment } }

		#CHANCE OF INFECTION
		hidden_effect = {
			random = {
				chance = 10
				trigger_event = {
					id = health.0201
					days = { 30 60 }
				}
			}
		}

		#HANDLE TREATMENT
		if = { #To send notification message or trigger the right event
			limit = {
				is_immortal = no
				OR = {
					has_trait = POD_wounded
					has_trait = wounded_1
				}
				court_physician_available_trigger = yes
			}
			save_scope_as = sick_character
			safe_wound_treatment_effect = yes #Wounded 1 only gives a "result of treatment" notification" if you have a physician, i.e. you get no events about it.
		}
		else_if = {
			limit = {
				has_trait = wounded_2
				has_recent_wound_treatment_trigger = no
			}
			#A "real" event if you're not receiving treatment
			#Nothing happens if you're already being treated for wounds
			trigger_event = {
				id = health.0102
				days = { 2 3 }
			}
		}
		else_if = {
			limit = {
				has_trait = wounded_3
				has_recent_wound_treatment_trigger = no
			}
			#A "real" event if you're not receiving treatment
			#Nothing happens if you're already being treated for wounds
			trigger_event = {
				id = health.0104
				days = { 2 3 }
			}

			#Epilepsy risk
			epilepsy_brain_trauma_risk_effect = { CHANCE = 5 }
		}
	}
	POD_error_suppression_unused_flag = { FLAG = $REASON$ }
}
