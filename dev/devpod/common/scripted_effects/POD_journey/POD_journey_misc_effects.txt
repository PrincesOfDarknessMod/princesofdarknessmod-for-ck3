POD_hunt_in_capital_create_list_effect = {
	if = {
		limit = { has_variable_list = hunt_in_capital_types }
		clear_variable_list = hunt_in_capital_types
	}
	
	add_to_variable_list = { name = hunt_in_capital_types target = flag:pod_peasants }
	add_to_variable_list = { name = hunt_in_capital_types target = flag:pod_diplomats }
	add_to_variable_list = { name = hunt_in_capital_types target = flag:pod_warriors }
	add_to_variable_list = { name = hunt_in_capital_types target = flag:pod_merchants }
	add_to_variable_list = { name = hunt_in_capital_types target = flag:pod_scoundrels }
	add_to_variable_list = { name = hunt_in_capital_types target = flag:pod_clergy }
	if = {
		limit = {
			OR = {
				has_perk = scry_the_soul_perk
				has_trait = auspexadvanced
			}
		}
		add_to_variable_list = { name = hunt_in_capital_types target = flag:choleric }
		add_to_variable_list = { name = hunt_in_capital_types target = flag:melancholic }
		add_to_variable_list = { name = hunt_in_capital_types target = flag:phlegmatic }
		add_to_variable_list = { name = hunt_in_capital_types target = flag:sanguine }
	}
	if = {
		limit = { dynasty ?= { has_dynasty_perk = ashen_nobles_legacy_4 } }
		add_to_variable_list = { name = hunt_in_capital_types target = flag:ashen_nobles }
	}
	if = {
		limit = { dynasty ?= { has_dynasty_perk = ashen_knights_legacy_4 } }
		add_to_variable_list = { name = hunt_in_capital_types target = flag:ashen_knights }
	}
	if = {
		limit = { dynasty ?= { has_dynasty_perk = ashen_seneschals_legacy_4 } }
		add_to_variable_list = { name = hunt_in_capital_types target = flag:ashen_seneschals }
	}
	if = {
		limit = { dynasty ?= { has_dynasty_perk = ashen_thieves_legacy_4 } }
		add_to_variable_list = { name = hunt_in_capital_types target = flag:ashen_thieves }
	}
	if = {
		limit = { dynasty ?= { has_dynasty_perk = ashen_cults_legacy_4 } }
		add_to_variable_list = { name = hunt_in_capital_types target = flag:ashen_cults }
	}
	
	if = {
		limit = { NOT = { has_variable = hunt_in_capital_selection } }
		set_variable = { name = hunt_in_capital_selection value = flag:pod_peasants }
	}
}

POD_hunt_in_capital_cost_effect = {
	switch = {
		trigger = var:hunt_in_capital_selection
		
		flag:pod_peasants = { }
		flag:pod_scoundrels = {
			send_interface_toast = {
				title = POD_find_mortals_interaction_scoundrels_toast
				left_icon = root
				random_list = {
					25 = { remove_short_term_gold = { value = minor_gold_value    multiply = journey_cost_mult } }
					25 = { add_prestige           = { value = minor_prestige_loss multiply = journey_cost_mult } }
					25 = { add_piety              = { value = minor_piety_loss    multiply = journey_cost_mult } }
				}
			}
		}
		flag:pod_diplomats = { add_prestige           = { value = minor_prestige_loss multiply = journey_cost_mult } }
		flag:pod_warriors  = { add_prestige           = { value = minor_prestige_loss multiply = journey_cost_mult } }
		flag:pod_merchants = { remove_short_term_gold = { value = minor_gold_value    multiply = journey_cost_mult } }
		flag:pod_clergy    = { add_piety              = { value = minor_piety_loss    multiply = journey_cost_mult } }
		flag:choleric    = { POD_hunger_rouse_check_effect = yes }
		flag:melancholic = { POD_hunger_rouse_check_effect = yes }
		flag:phlegmatic  = { POD_hunger_rouse_check_effect = yes }
		flag:sanguine    = { POD_hunger_rouse_check_effect = yes }
		flag:ashen_nobles     = { add_prestige           = { value = medium_prestige_loss multiply = journey_cost_mult } }

					target = scope:mortal_ashen_noble
				}
			}
		}

			while = {
				count = 3

					location = root.capital_province
					template = POD_mortal_ashen_knight_character
					save_scope_as = mortal_ashen_knight
				}
				scope:mortal_ashen_knight = {
					POD_hunt_in_capital_character_setup_effect = yes
				}
				add_to_variable_list = {
					name = hunt_in_capital_mortals
					target = scope:mortal_ashen_knight
				}
			}
		}
		
		flag:ashen_seneschals = {
			while = {
				count = 3
				
				create_character = {
					location = root.capital_province
					template = POD_mortal_ashen_seneschal_character
					save_scope_as = mortal_ashen_seneschal

				scope:mortal_ashen_seneschal = {

				}
			}
		}
		
		flag:ashen_thieves = {
			while = {
				count = 3

				count = 3

					location = root.capital_province
					template = POD_mortal_ashen_cult_character

				add_to_variable_list = {
					name = hunt_in_capital_mortals
					target = scope:mortal_ashen_cultist
				}
			}
		}
	}
	
	if = {
		limit = {
			any_directly_owned_province = {
				has_building_with_flag = grand_city_commoners
			}
		}
		every_in_list = {
			variable = hunt_in_capital_mortals
			POD_resonance_increase_intensity_effect = yes
		}
	}
	
	# If Ventrue, select the best mortal who fits the feeding restrictions
	if = {
		limit = { is_ventrue_trigger = yes }
		ordered_in_list = {
			variable = hunt_in_capital_mortals
			
			limit = {
				ventrue_weakness_compatible_trigger = {
					ACTOR = root
					RECIPIENT = this
				}
			}
			
			order_by = resonance_rating_value
			position = 0
			
			prev = {
				set_variable = {
					name = hunt_in_capital_selected_mortal
					value = prev
				}
			}
		}
	}
	
	# Otherwise, select the best mortal
	if = {
		limit = { NOT = { has_variable = hunt_in_capital_selected_mortal } }
		ordered_in_list = {
			variable = hunt_in_capital_mortals
			order_by = resonance_rating_value
			position = 0
			
			prev = {
				set_variable = {
					name = hunt_in_capital_selected_mortal
					value = prev
				}
			}
		}
	}
}
