POD_numina_sorcery_perk_point_check = {
	if = { limit = { has_variable = POD_numina_sorcery_path_num } 
		save_scope_value_as = { name = path_num value = var:POD_numina_sorcery_path_num }
		custom_description_no_bullet = { 
			text = POD_numina_sorcery_perk_point_check_2 
			value = scope:path_num
			save_scope_value_as = { name = path_num2 value = { value = var:POD_numina_sorcery_path_num multiply = -1 add = 1 } }
			add_sorcery_lifestyle_perk_points = scope:path_num2
		}
		change_variable = { name = POD_numina_sorcery_path_num add = 1 } 
		save_scope_value_as = { name = path_num value = var:POD_numina_sorcery_path_num }
		if = { limit = { POD_has_enough_perk_points_for_new_path = no } custom_description_no_bullet = { text = POD_numina_sorcery_perk_point_check value = scope:path_num } }
	}
	else = { 
		if = { limit = { has_game_rule = pod_sorcery_additional_cost_per_path_0 } }
		else_if = { limit = { has_game_rule = pod_sorcery_additional_cost_per_path_1 } set_variable = { name = POD_numina_sorcery_path_num value = 2 } }
		else = { set_variable = { name = POD_numina_sorcery_path_num value = 3 } }
	}
}
pod_sorcery_6th_perk_check = {
	if = { 
		limit = { has_game_rule = pod_sorcery_6th_perk_requirements_disabled } 
		custom_description_no_bullet = { text = numina_disabled_6th_perk } 
	}
	else = {
		if = { 
			limit = { 
				has_game_rule = pod_sorcery_6th_perk_requirements_unlockable 
				NOT = { has_character_flag = numina_$PATH$_6_perk_flag_requirement } 
			} 
			custom_description_no_bullet = { text = numina_$PATH$_6_perk_flag_requirement } 
		}
		if = { 
			limit = { NOT = { has_perk = numina_$PATH$_5_perk } } 
			custom_description_no_bullet = { text = numina_require_5th_perk } 
		}
	}
}

POD_sorcerer_embrace_effect = {
	if = {
		limit = {
			has_character_flag = sorcerer_embrace
			has_POD_vamp_blood_trigger = yes
		}
		remove_character_flag = sorcerer_embrace
		if = {
			limit = { NOT = { has_trait = bloodsorcerydiscipline } }
			add_trait = bloodsorcery
		}
		if = {
			limit = { has_any_bloodmagic_path = no }
			add_character_flag = has_thaumaturgy_bloodmagic_flag
		}
	}
}
