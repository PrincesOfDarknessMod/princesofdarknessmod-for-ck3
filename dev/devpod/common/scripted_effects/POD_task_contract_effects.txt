POD_save_task_contract_for_event_widget = {
	set_variable = {
		name  = task_contract_widget
		value = scope:task_contract
	}
}

POD_cleanup_task_contract_event_widget = {
	remove_variable = task_contract_widget
}

# character scope
POD_setup_task_contract_travel_plan_with_custom_travel_start_event = {
	# Edge case resolution
	# (you can abandon the contract while the event window is open)
	if = {
		limit = { any_character_active_contract = { scope:task_contract ?= this } }
		custom_tooltip = POD_accept_contract_tt
		if = {
			limit = { scope:task_contract.task_contract_location = prev.location }
			trigger_event = $ARRIVAL_EVENT$
		}
		else_if = {
			limit = {
				is_landless_adventurer = yes
				has_variable = journey_moves_camp
			}
			start_travel_plan = {
				destination = scope:task_contract.task_contract_location
				on_arrival_event = $ARRIVAL_EVENT$
				on_start_event = $TRAVEL_START_EVENT$
				on_travel_planner_cancel_event = POD_task_contracts.2
				travel_with_domicile = yes
				return_trip = no
				on_arrival_destinations = last
			}
		}
		else = {
			start_travel_plan = {
				destination = scope:task_contract.task_contract_location
				on_arrival_event = $ARRIVAL_EVENT$
				on_start_event = $TRAVEL_START_EVENT$
				on_travel_planner_cancel_event = POD_task_contracts.2
			}
		}
	}
	else = {
		trigger_event = POD_task_contracts.2
	}
}

# character scope
POD_setup_task_contract_travel_plan = {
	POD_setup_task_contract_travel_plan_with_custom_travel_start_event = {
		ARRIVAL_EVENT = $ARRIVAL_EVENT$
		TRAVEL_START_EVENT = POD_task_contracts.4
	}
}

POD_pause_task_contract_travel_plan = {
	hidden_effect = {
		scope:task_contract ?= {
			var:travel_plan ?= {
				pause_travel_plan = yes
			}
		}
	}
}

# task_contract scope
POD_setup_timekeeping_vars_for_contract = {
	set_variable = {
		name = end_date
		value = {
			value = current_date
			add = {
				value = $YEARS_REMAINING$
				multiply = 365
			}
		}
	}
	set_variable = {
		name  = time_remaining
		value = yes
		days = POD_days_remaining_for_task_contract
	}
	trigger_event = {
		id   = POD_task_contracts.1
		days = POD_days_remaining_for_task_contract
	}
}

# task_contract scope
POD_recreate_contract_with_timekeeping_vars = {
	hidden_effect_new_object = {
		task_contract_taker = {
			if = {
				limit = { prev = { exists = task_contract_employer } }
				create_task_contract = {
					task_contract_type = $CONTRACT_TYPE$
					task_contract_tier = prev.task_contract_tier
					location = prev.task_contract_location
					task_contract_employer = prev.task_contract_employer
					save_scope_as = $SAVE_AS$
				}
			}
			else = {
				create_task_contract = {
					task_contract_type = $CONTRACT_TYPE$
					task_contract_tier = prev.task_contract_tier
					location = prev.task_contract_location
					save_scope_as = $SAVE_AS$
				}
			}
			scope:$SAVE_AS$ = {
				set_variable = {
					name  = end_date
					value = root.var:end_date
				}
				set_variable = {
					name  = time_remaining
					value = yes
					# needs to be a script value since inline values don't work here
					days  = POD_days_remaining_for_task_contract
				}
				trigger_event = {
					id   = POD_task_contracts.1
					days = POD_days_remaining_for_task_contract
				}
			}
		}
	}
}
