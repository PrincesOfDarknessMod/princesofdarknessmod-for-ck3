POD_umbra_expedition_randomize_effect = {
	if = {
		limit = { stress_level >= 3 }
		random_list = {
			25 = { set_variable = { name = $ADVTURE$ value = flag:trial } }
			25 = { set_variable = { name = $ADVTURE$ value = flag:fight } }
			40 = { set_variable = { name = $ADVTURE$ value = flag:elite } }
			0 = { set_variable = { name = $ADVTURE$ value = flag:chest } }
			10 = { set_variable = { name = $ADVTURE$ value = flag:event } }
		}
	}
	else_if = {
		limit = { stress_level >= 2 }
		random_list = {
			30 = { set_variable = { name = $ADVTURE$ value = flag:trial } }
			30 = { set_variable = { name = $ADVTURE$ value = flag:fight } }
			30 = { set_variable = { name = $ADVTURE$ value = flag:elite } }
			0 = { set_variable = { name = $ADVTURE$ value = flag:chest } }
			10 = { set_variable = { name = $ADVTURE$ value = flag:event } }
		}
	}
	else_if = {
		limit = { stress_level >= 1 }
		random_list = {
			35 = { set_variable = { name = $ADVTURE$ value = flag:trial } }
			35 = { set_variable = { name = $ADVTURE$ value = flag:fight } }
			20 = { set_variable = { name = $ADVTURE$ value = flag:elite } }
			0 = { set_variable = { name = $ADVTURE$ value = flag:chest } }
			10 = { set_variable = { name = $ADVTURE$ value = flag:event } }
		}
	}
	else = {
		random_list = {
			40 = { set_variable = { name = $ADVTURE$ value = flag:trial } }
			40 = { set_variable = { name = $ADVTURE$ value = flag:fight } }
			10 = { set_variable = { name = $ADVTURE$ value = flag:elite } }
			0 = { set_variable = { name = $ADVTURE$ value = flag:chest } }
			10 = { set_variable = { name = $ADVTURE$ value = flag:event } }
		}
	}
}

		limit = { NOT = { any_memory = { has_memory_type = umbra_expedition_memory } } }

		if = {
			limit = { NOT = { var:spirit_location ?= flag:earth } }

			}

		}
	}
}

POD_umbra_artifact_init_effect = {
	if = {
		limit = {
			OR = {
				NOT = { exists = scope:artifact_flag_1 } 
				scope:artifact_flag_1 = flag:no
			}
		}
		save_scope_value_as = {
			name = artifact_flag_1
			value = flag:$ARTIFACT$
		}
	}
	else_if = {
		limit = {
			OR = {
				NOT = { exists = scope:artifact_flag_2 } 
				scope:artifact_flag_2 = flag:no
			}
		}
		save_scope_value_as = {
			name = artifact_flag_2
			value = flag:$ARTIFACT$
		}
	}
	else = {
		save_scope_value_as = {
			name = artifact_flag_3
			value = flag:$ARTIFACT$
		}
	}
}

# Discovery
POD_umbra_set_expedition_discovery_pos_effect = {
	save_temporary_scope_as = this_character
	set_variable = { name = umbra_discovery_type value = flag:$TYPE$ years = umbra_discovery_duration } 
	if = {
		limit = { NOT = { scope:player = this } }
		scope:player = {
			send_interface_message = {
				type = event_generic_good_with_text
				title = umbra_discovery.title
				desc = umbra_discovery.$TYPE$.desc
				left_icon = scope:player
				right_icon = scope:this_character
				
				scope:this_character = {
					create_character_memory = {
						type = umbra_discovery_memory
					}
				}
			}
			add_to_variable_list = { name = umbra_discoveries_list target = scope:this_character.var:umbra_stationed_location } 
		}
	}
	else = {
		send_interface_message = {
			type = event_generic_good_with_text
			title = umbra_discovery.title
			desc = umbra_discovery.$TYPE$.desc
			right_icon = this
			
			create_character_memory = {
				type = umbra_discovery_memory
			}
		}
		add_to_variable_list = { name = umbra_discoveries_list target = var:umbra_stationed_location } 
	}
}

POD_umbra_remove_expedition_discovery_effect = {
	if = {
		limit = { var:umbra_stationed_location ?= $LOCATION$ }
		POD_umbra_remove_discovery_effect = { TARGET = root }
	}
	else = {
		every_relation = {
			type = umbra_explorer
			limit = { var:umbra_stationed_location ?= $LOCATION$ }
			save_temporary_scope_as = explorer
			POD_umbra_remove_discovery_effect = { TARGET = scope:explorer }
		}
	}
	remove_list_variable = { name = umbra_discoveries_list target = $LOCATION$ }
}

POD_umbra_remove_discovery_effect = {
	every_memory = {
		limit = { has_memory_type = umbra_discovery_memory }
		$TARGET$ = { destroy_character_memory = prev }
	}
	remove_variable = umbra_discovery_type
}

POD_umbra_swap_discovery_effect = {
	if = {
		limit = { 
			$CHARACTER1$ = { has_variable = umbra_discovery_type }
			$CHARACTER2$ = { has_variable = umbra_discovery_type }
		}
		save_scope_value_as = {
			name = saved_discovery
			value = $CHARACTER1$.var:umbra_discovery_type
		}
		$CHARACTER1$ = { 
			every_memory = {
				limit = { has_memory_type = umbra_discovery_memory }
				$CHARACTER1$ = { destroy_character_memory = prev }
			}
			create_character_memory = { type = umbra_discovery_memory }
			set_variable = { name = umbra_discovery_type value = $CHARACTER2$.var:umbra_discovery_type years = umbra_discovery_duration }
		}
		$CHARACTER2$ = { 
			every_memory = {
				limit = { has_memory_type = umbra_discovery_memory }
				$CHARACTER2$ = { destroy_character_memory = prev }
			}
			create_character_memory = { type = umbra_discovery_memory }
			set_variable = { name = umbra_discovery_type value = scope:saved_discovery years = umbra_discovery_duration }
		}
	}
	else_if = {
		limit = { $CHARACTER1$ = { has_variable = umbra_discovery_type } }
		$CHARACTER2$ = { 
			create_character_memory = { type = umbra_discovery_memory }
			set_variable = { name = umbra_discovery_type value = $CHARACTER1$.var:umbra_discovery_type }
		}
		$CHARACTER1$ = { POD_umbra_remove_discovery_effect = { TARGET = $CHARACTER1$ } }
	}
	else_if = {
		limit = { $CHARACTER2$ = { has_variable = umbra_discovery_type } }
		$CHARACTER1$ = { 
			create_character_memory = { type = umbra_discovery_memory }
			set_variable = { name = umbra_discovery_type value = $CHARACTER2$.var:umbra_discovery_type }
		}
		$CHARACTER2$ = { POD_umbra_remove_discovery_effect = { TARGET = $CHARACTER2$ } }
	}
}

### Dragon ###
start_dragon_story_cycle_effect = {
	show_as_tooltip = {
		add_character_modifier = {
			modifier = dragon_story_modifier
		}
	}
	hidden_effect = {
		create_story = POD_story_pet_dragon
	}
}

#Assign the gender of the dragon
assign_dragon_gender_effect = {
	save_temporary_scope_value_as = {
		name = gender
		value = flag:$GENDER$
	}
	if = {
		limit = { scope:gender = flag:random }
		hidden_effect = {
			random_list = {
				50 = {
					set_variable = {
						name = dragon_gender
						value = flag:female
					}
				}
				50 = {
					set_variable = {
						name = dragon_gender
						value = flag:male
					}
				}
			}
		}
	}
	else_if = {
		limit = {
			OR = {
				scope:gender = flag:male
				scope:gender = flag:female
			}
		}
		set_variable = {
			name = dragon_gender
			value = flag:$GENDER$
		}
	}
}

#Remove everything related to the story
remove_dragon_story_modifiers_effect = {
	if = {
		limit = {
			has_character_modifier = dragon_story_modifier
		}
		remove_character_modifier = dragon_story_modifier
	}
	
	prev.var:artifact_dragon ?= { destroy_artifact = this }
}

remove_dragon_name_effect = {
	remove_variable = story_cycle_dragon_name
	remove_localized_text = story_cycle_dragon_name
}
