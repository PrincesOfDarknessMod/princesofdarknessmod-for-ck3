add_quint = {
	if = {
		limit = {
			has_variable = quint_amount
		}
		change_variable = {
			name = quint_amount
			add = $AMOUNT$
		}
		clamp_variable = {
			name = quint_amount
			max = max_quint
			min = 0
		}
	}
	else = {
		set_variable = { name = quint_amount value = $AMOUNT$ }
	}
}

subtract_quint = {
	set_local_variable = { name = minus_amount value = {value = $AMOUNT$  multiply = -1} }
	add_quint = {AMOUNT = local_var:minus_amount}
}
