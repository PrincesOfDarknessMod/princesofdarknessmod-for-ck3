
# TYPE - Enum(forge,brew,enhance)
# VALUE - Int
POD_give_one_time_boost_to = {
	save_scope_value_as = { name = type_loca value = flag:$TYPE$ }
	save_scope_value_as = { name = value_loca value = $VALUE$ }
	custom_tooltip = {
		text = POD_give_one_time_boost_to
		#value = $VALUE$
		if = { limit = { has_variable = POD_one_time_$TYPE$_boost } change_variable = { name = POD_one_time_$TYPE$_boost add = $VALUE$ } }
		else = { set_variable = { name = POD_one_time_$TYPE$_boost value = $VALUE$ } }
		if = { limit = { has_character_modifier = POD_one_time_$TYPE$_boost } remove_character_modifier = POD_one_time_$TYPE$_boost }
		add_character_modifier = POD_one_time_$TYPE$_boost
	}
}

POD_increase_durability_effect = {
	custom_description = {
		text = POD_increase_durability_effect
		object = THIS
		value = $VALUE$
		set_max_durability = { value = { value = artifact_max_durability add = $VALUE$ } }
	}
}
POD_decrease_durability_effect = {
	if = {
		limit = { has_variable = no_crafting_durability_cost }
		custom_tooltip = {
			text = crafting_no_durability_loss_tt
			remove_variable = no_crafting_durability_cost
		}
	}
	else = {
		custom_description = {
			text = POD_decrease_durability_effect
			object = THIS
			value = $VALUE$
			set_max_durability = { value = { value = artifact_max_durability subtract = $VALUE$ } }
		}
	}
}
