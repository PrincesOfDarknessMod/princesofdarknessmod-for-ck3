POD_selfless_invasion_victory_effect = {
	POD_clear_character_select_vars = yes
	save_scope_as = invasion_leader
	
	# Save target
	ordered_in_list = {
		list = target_titles
		order_by = tier
		save_scope_as = target
	}
	
	# Demons can nominate their cultists
	# TODO: turn newly generated sub-barons into your cultists too
	if = {
		limit = { POD_is_demon_trigger = yes }
		POD_save_tooltip = {
			LOC = POD_selfless_invasion_victory_demon
		}
		every_relation = {
			type = worshipping_cultist
			limit = {
				POD_is_valid_beneficiary_for_selfless_invasion = yes
			}
			scope:invasion_leader = {
				add_to_variable_list = {
					name   = event_character_list_presorted
					target = prev
				}
			}
		}
	}
	# Mummies can nominate their neteru
	else_if = {
		limit = { POD_is_mummy_trigger = yes }
		POD_save_tooltip = {
			LOC = POD_selfless_invasion_victory_mummy
		}
		every_relation = {
			type = worshipping_neteru
			limit = {
				POD_is_valid_beneficiary_for_selfless_invasion = yes
			}
			scope:invasion_leader = {
				add_to_variable_list = {
					name   = event_character_list_presorted
					target = prev
				}
			}
		}
	}
	# Hunters can nominate their entire house
	else_if = {
		limit = { POD_is_supehunter_trigger = yes }
		POD_save_tooltip = {
			LOC = POD_selfless_invasion_victory_hunter
		}
		house ?= {
			every_house_member = {
				limit = {
					NOT = { this = scope:attacker }
					POD_is_valid_beneficiary_for_selfless_invasion = yes
					is_supehunter_trigger = yes # Mortals can't hold titles
				}
				scope:invasion_leader = {
					add_to_variable_list = {
						name   = event_character_list_presorted
						target = prev
					}
				}
			}
		}
	}
	# Everyone else can nominate close family members
	# TODO: checks for vampires, fera, fae, KJ
	else = {
		POD_save_tooltip = {
			LOC = POD_selfless_invasion_victory_fallback
		}
		every_close_family_member = {
			limit = {
				POD_is_valid_beneficiary_for_selfless_invasion = yes
			}
			scope:invasion_leader = {
				add_to_variable_list = {
					name   = event_character_list_presorted
					target = prev
				}
			}
		}
	}
	
	custom_tooltip = POD_saved_tooltip
	
	## Ambassador can always give lands to their patron
	if = {
		limit = {
			any_relation = {
				type = ashen_ambassadorial_patron
				count >= 1
			}
		}
		custom_tooltip = POD_selfless_invasion_victory_ambassador
		every_relation = {
			type = ashen_ambassadorial_patron
			limit = {
				POD_is_valid_beneficiary_for_selfless_invasion = yes		
			}
			scope:invasion_leader = {
				add_to_variable_list = {
					name   = event_character_list_presorted
					target = prev
				}
			}
		}
	}
	
	POD_sort_event_character_list = yes
	
	custom_tooltip = POD_selfless_invasion_victory_patron
	
	trigger_event = POD_war.1
}

POD_sort_character_list = {
	clear_variable_list = $SORTED_LIST$
	remove_variable = $SELECTED_CHAR_VAR$
	save_scope_as = sort_root
	
	# list_size or inline complex triggers don't seem to work for this, so let's do it the hard way
	save_scope_value_as = {
		name = character_list_size
		value = {
			every_in_list = {
				variable = $PRESORT_LIST$
				add = 1
			}
		}
	}
	
	ordered_in_list = {
		variable = $PRESORT_LIST$
		# TODO: ability to sort in the GUI
		order_by = sum_of_all_skills_value
		max = scope:character_list_size
		check_range_bounds = no
		
		limit = {
			trigger_if = {
				limit = { prev = { has_variable = customsort_pinned_only } }
				any_pinning_character = { this = scope:sort_root }
			}
			trigger_else = { always = yes }
		}
		
		prev = {
			add_to_variable_list = {
				name = $SORTED_LIST$
				target = prev
			}
		}
	}
	ordered_in_list = {
		variable = $SORTED_LIST$
		prev = {
			set_variable = {
				name = $SELECTED_CHAR_VAR$
				value = prev
			}
		}
	}
}

POD_conquest_title_change = {
	create_title_and_vassal_change = {
		type = conquest
		add_claim_on_loss = yes
		save_scope_as = title_change
	}
	
	every_in_list = {
		list = target_titles
		# Compile a list of all counties in the kingdom we will take from the defender.
		every_in_de_jure_hierarchy = {
			limit = {
				# Current holder must be a (sub)vassal of the defender (no poaching titles from 3rd parties!)
				exists = holder
				holder = {
					OR = {
						this = scope:defender
						target_is_liege_or_above = scope:defender
					}
				}
				trigger_if = {
					limit = { tier = tier_barony }
					is_capital_barony = no # error suppression
				}
			}
			add_to_list = titles_taken
		}
		#add_to_list = titles_taken
	}

	every_in_list = {
		list = titles_taken
		change_title_holder = {
			holder = scope:attacker
			change = scope:title_change
			take_baronies = yes
		}
	}
	
	if = {
		limit = {
			scope:attacker = {
				is_landed = no 
			}
		}
		scope:attacker = {
			becomes_independent = {
				change = scope:title_change
			}
		}
	}
	
	resolve_title_and_vassal_change = scope:title_change
}

# target_titles list needs to exist in TopScope (e.g. after winning a war)
POD_give_titles_to_beneficiary = {
	$BENEFICIARY$ = { save_scope_as = attacker }
	
	send_interface_message = {
		type = send_interface_message
		title = POD_gave_titles_to_beneficiary.t
		left_icon = scope:attacker
		custom_description = {
			text = POD_give_titles_to_beneficiary
			subject = scope:attacker
			object = scope:target
			POD_conquest_title_change = yes
		}
	}
	
	### Set a new capital.
	#### We prefer the de jure capital of the other.
	if = {
		limit = {
			scope:target.title_capital_county = { is_in_list = target_titles }
			scope:attacker = { is_landed = no }
		}
		scope:target.title_capital_county = {
			save_temporary_scope_as = new_capital
			scope:attacker = { set_realm_capital = scope:new_capital }
		}
	}
	### Otherwise, tribals go by pure levy count.
	else_if = {
		limit = {
			scope:attacker = { government_has_flag = government_is_tribal }
			scope:attacker = { is_landed = no }
		}
		ordered_in_list = {
			list = target_titles
			limit = { tier = tier_county }
			order_by = total_county_levies
			save_temporary_scope_as = new_capital
			scope:attacker = { set_realm_capital = scope:new_capital }
		}
	}
	### Generally everyone prefers the highest development.
	else_if = {
		limit = {
			scope:attacker = { is_landed = no }
		}
		ordered_in_list = {
			list = target_titles
			limit = { tier = tier_county }
			order_by = { add = development_level }
			save_temporary_scope_as = new_capital
			scope:attacker = { set_realm_capital = scope:new_capital }
		}
	}
	else = {}
	
	hidden_effect = {
		scope:new_capital ?= {
			set_county_faith = scope:attacker.faith
		}
	}
	
	add_favour_hook_if_possible_simple_effect = {
		TARGET = scope:attacker
	}
	reverse_add_opinion = {
		modifier = helped_me_against_tyrant_opinion
		opinion = 65
		years = 50
		target = scope:attacker
	}
	if = {
		limit = { NOT = { has_contact = scope:attacker } }
		add_contact = scope:attacker
	}
	
	hidden_effect = {
		scope:attacker = {
			trigger_event = fp1_varangian_adventures.0001 # control boost
		}
	}
	
	# Other effects: Ashen Ambassador XP
	if = {
		limit = {
			scope:attacker = { has_relation_ashen_ambassadorial_patron = prev }
			has_trait = ashen_adventurer
		}
		add_trait_xp = {
			trait = ashen_adventurer
			track = ashen_ambassador
			value = {
				value = scope:target.tier
				# 6 for a county, 9 for a duchy, 12 for a kingdom
				multiply = 3
			}
		}
	}
	
	# Other effects: Curia Support event
	if = {
		limit = {
			faith = faith:shadowinquisition
			NOT = { has_trait = excommunicated }
		}
		if = {
			limit = { scope:target.tier >= tier_kingdom }
			trigger_event = POD_war.4
		}
		else_if = {
			limit = { scope:target.tier = tier_duchy }
			trigger_event = POD_war.3
		}
		else_if = {
			limit = { scope:target.tier = tier_county }
			trigger_event = POD_war.2
		}
	}
}

###Bandaid because our AIs will never call to arms and we can't figure out why

POD_war_ai_ally_bandaid_effect = {
	scope:war = { save_temporary_scope_as = target }
	$PARTICIPANT$ = {
		save_temporary_scope_as = actor
		every_ally = {
			limit = {
				is_ai = yes
				prev = {
					is_character_interaction_valid = {
						recipient = prev
						interaction = call_ally_interaction
					}
				}

				save_temporary_scope_as = recipient

				#################################################################################
				####### has_valid_target_showing_failures_only from call_ally_interaction #######
				#################################################################################

				# since is_character_interaction_potentially_accepted doesn't work with interactions with a scope:target

				scope:target = {
					is_war_leader = scope:actor
				}
		
				scope:actor = {
					trigger_if = {
						limit = {
							scope:target = {
								OR = {
									using_cb = undirected_great_holy_war
									using_cb = directed_great_holy_war
								}
							}
						}
		
						trigger_if = {
							limit = {
								this = scope:target.casus_belli.primary_attacker
							}
							#if it's a religious war the recipient has to be of the same faith
							custom_description = {
								text = is_of_wrong_faith_for_holy_war
								subject = scope:actor
								object = scope:recipient
								faith = scope:recipient.faith
							}
						}
						trigger_else = {
							custom_description = {
								text = is_of_attacker_faith_in_holy_war
								subject = scope:actor
								object = scope:recipient
								scope:target = { NOT = { primary_attacker.faith = scope:recipient.faith } }
							}
						}
					}
				}
		
		
				joiner_not_already_in_another_war_with_any_target_war_participants_trigger = {
					WARRIOR = scope:actor
					JOINER = scope:recipient
				}
				
				########################################################
				####### can_be_picked from call_ally_interaction #######
				########################################################

				can_join_war_liege_vassal_check_trigger = { WARRIOR = scope:actor JOINER = scope:recipient }

				scope:target = {
					is_war_leader = scope:actor

					# not already in target war
					custom_description = {
						text = join_war_interaction_already_in_target_war
						NOR = {
							any_war_attacker = { this = scope:recipient }
							any_war_defender = { this = scope:recipient }
						}
					}

					# Diarchs can only be called to internal wars.
					custom_tooltip = {
						text = call_ally_interaction.tt.diarchs_called_to_internal_wars
						trigger_if = {
							limit = { scope:recipient ?= scope:actor.diarch }
							primary_defender = scope:actor
							diarch_callable_in_internal_war_trigger = yes
						}
					}
				}

				#############################################################
				####### ai_accept blockers from call_ally_interaction #######
				#############################################################

				trigger_if = { # Refuse call to conflicts of little benefit
					limit = { scope:target = { using_cb = fp2_border_raid } }
					scope:recipient = { # Unless they like going on raids
						OR = {
							has_trait = viking
							has_trait = reaver
						}
					}
				}

				trigger_if = {  # Refuse call against Heir
					limit = { exists = scope:recipient.player_heir }
					trigger_if = {
						limit = { scope:target.casus_belli.primary_defender = scope:actor }
						NOT = { scope:target.casus_belli.primary_attacker = scope:recipient.player_heir }
					}
					trigger_else = {
						NOT = { scope:target.casus_belli.primary_defender = scope:recipient.player_heir }
					}
				}
				
				trigger_if = { # Refuse call against Spouse
					limit = { scope:target.casus_belli.primary_defender = scope:actor }
					NOT = {
						any_spouse = {
							this = scope:target.casus_belli.primary_attacker
						}
					}
				}
				trigger_else = {
					NOT = {
						any_spouse = {
							this = scope:target.casus_belli.primary_defender
						}
					}
				}
			}
			
			call_ally_interaction_effect = yes
		}
	}
}
