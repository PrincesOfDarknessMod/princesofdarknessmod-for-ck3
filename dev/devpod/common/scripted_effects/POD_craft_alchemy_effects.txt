POD_create_alchemy_artifact_effect = {
	save_scope_as = owner
	set_artifact_rarity_common = yes
		
	create_artifact = {
		name = $ALCHEMY$_modifier
		description = $ALCHEMY$_modifier_desc
		visuals = $VISUAL$
		type = miscellaneous
		wealth = scope:wealth
		quality = scope:quality
		modifier = $ALCHEMY$_modifier
		template = alchemy_artifact_template
		generate_history = no
		save_scope_as = newly_created_artifact
	}
	hidden_effect_new_object = { 
		scope:newly_created_artifact = {
			set_variable = { name = alchemy value = flag:$ALCHEMY$ }
			set_max_durability = alchemy_duration_value
			flag_as_trash_artifact = yes
			if = {
				limit = { flag:$POISON$ = flag:yes }
				set_variable = poison
			}
		}
	}
}
