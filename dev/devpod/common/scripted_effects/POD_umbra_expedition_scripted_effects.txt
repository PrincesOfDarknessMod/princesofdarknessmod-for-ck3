POD_umbra_prowess_challenge_effect = {
	save_scope_as = duelist
	if = {
		limit = {
			has_variable = coterie_spirit_travel
		}
		POD_force_save_coterie_duelists = yes
		save_scope_value_as = {
			name = duelist
			value = scope:prowess_duelist
		}
	}
	
	# 
	if = {
		limit = {
			umbra_difficulty >= 5
		}
		POD_umbra_prowess_challenge_scopes = {
			DUEL_CHALLENGE_VALUE = POD_prowess_very_high_skill_rating
			PRESTIGE_GAIN = medium_dynasty_prestige_gain
			WOUND_MULTIPLIER = 2
			STRESS_GAIN = medium_stress_gain
			PIETY_LOSS = minor_piety_loss
		}
	}
	else_if = {
		limit = {
			umbra_difficulty >= 4
		}
		POD_umbra_prowess_challenge_scopes = {
			DUEL_CHALLENGE_VALUE = POD_prowess_high_skill_rating
			PRESTIGE_GAIN = medium_dynasty_prestige_gain
			WOUND_MULTIPLIER = 2
			STRESS_GAIN = medium_stress_gain
			PIETY_LOSS = minor_piety_loss
		}
	}
	else_if = {
		limit = {
			umbra_difficulty >= 3
		}
		POD_umbra_prowess_challenge_scopes = {
			DUEL_CHALLENGE_VALUE = POD_prowess_decent_skill_rating
			PRESTIGE_GAIN = minor_dynasty_prestige_gain
			WOUND_MULTIPLIER = 1
			STRESS_GAIN = minor_stress_gain
			PIETY_LOSS = minor_piety_loss
		}
	}
	else_if = {
		limit = {
			umbra_difficulty >= 2
		}
		POD_umbra_prowess_challenge_scopes = {
			DUEL_CHALLENGE_VALUE = POD_prowess_medium_skill_rating
			PRESTIGE_GAIN = minor_dynasty_prestige_gain
			WOUND_MULTIPLIER = 1
			STRESS_GAIN = minor_stress_gain
			PIETY_LOSS = minor_piety_loss
		}
	}
	else_if = {
		limit = {
			umbra_difficulty >= 1
		}
		POD_umbra_prowess_challenge_scopes = {
			DUEL_CHALLENGE_VALUE = POD_prowess_mediocre_skill_rating
			PRESTIGE_GAIN = miniscule_dynasty_prestige_gain
			WOUND_MULTIPLIER = 1
			STRESS_GAIN = minor_stress_gain
			PIETY_LOSS = minor_piety_loss
		}
	}
	else = {
		POD_umbra_prowess_challenge_scopes = {
			DUEL_CHALLENGE_VALUE = POD_prowess_low_skill_rating
			PRESTIGE_GAIN = miniscule_dynasty_prestige_gain
			WOUND_MULTIPLIER = 1
			STRESS_GAIN = minor_stress_gain
			PIETY_LOSS = minor_piety_loss
		}
	}

	
	save_temporary_scope_as = player
	scope:player = {
		duel = {
			desc = POD_umbra_prowess.desc
			skill = prowess
			value = scope:duel_challenge_value
		
			25 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = 0.5
				}
				desc = POD_umbra_prowess.success
				scope:player.dynasty = { add_dynasty_prestige =  scope:prestige_gain }
				years_from_game_start_difficulty_modifier = yes
			}
			25 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = -0.5
				}
				desc = POD_umbra_prowess.fail
				if = {
					limit = { has_variable = physical_spirit_travel }
					increase_wounds_multiple_effect = { REASON = fight AMOUNT = scope:wound_multiplier }
				}
				else = {
					add_stress = scope:stress_gain
				}
				scope:player = { add_piety = scope:piety_loss }
			}
		}
	}
}
