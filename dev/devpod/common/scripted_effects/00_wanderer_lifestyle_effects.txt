		visiting_poi_variable_effect = { PROVINCE = province:1611 }
		visiting_poi_variable_effect = { PROVINCE = province:1609 }
	}
	else_if = { # Danevirke
		limit = {
			OR = {
				scope:province = province:62
				scope:province = province:84
			}
		}
		visiting_poi_variable_effect = { PROVINCE = province:62 }
		visiting_poi_variable_effect = { PROVINCE = province:84 }
	}
	else_if = { # Great Wall of Gorgan
		limit = {
			OR = {
				scope:province = province:4031
				scope:province = province:4030
			}
		}
		visiting_poi_variable_effect = { PROVINCE = province:4031 }
		visiting_poi_variable_effect = { PROVINCE = province:4030 }
	}
	else = {
		visiting_poi_variable_effect = { PROVINCE = scope:province }
	}

		limit = {
			rite = { rite_has_parameter = tenet_scholasticism_muslim_hebrew_greek_poi }

				OR = {
					religion = religion:judaism_religion
					religion = religion:islam_religion
					religion = religion:zoroastrianism_religion
				}
			}
		}
		add_piety = minor_piety_gain
		add_learning_lifestyle_xp = 25
		change_spiritual_fulfillment = minor_spiritual_fulfillment_value
	}

	if = {
		limit = {
			culture = { has_cultural_parameter = stewardship_gain_from_foreign_poi }
			NOT = { culture = scope:province.culture }
		}
		add_stewardship_lifestyle_xp = 25
	}

	megaliths_building_poi_double_experience_effect = yes
}

voyager_souvenirs_aplenty_effect = {
	if = {
		limit = {
			has_perk = souvenirs_aplenty_perk
			NOT = { government_has_flag = government_is_merchant }
			NOT = { location.barony ?= root.capital_barony }
			location.county.prosperity_level > root.capital_county.prosperity_level
		}
		if = {
			limit = {
				has_variable = souvenirs_aplenty_var
			}
			change_variable = {
				name = souvenirs_aplenty_var
				add = {
					add = voyager_souvenirs_aplenty_script_value
					max = voyager_souvenirs_aplenty_max_value
				}
			}
		}
		else = {
			set_variable = {
				name = souvenirs_aplenty_var
				value = voyager_souvenirs_aplenty_script_value
			}
		}
	}
}

merchant_souvenirs_aplenty_travel_effect = {
	if = {
		limit = {
			government_has_flag = government_is_merchant
			has_perk = souvenirs_aplenty_perk
			exists = location.barony.trade_market
		}
		location.barony.trade_market = { save_scope_as = merchant_keepsakes_market }
		if = {
			limit = {
				any_trade_route = {
					any_trade_market_on_route = {
						this = scope:merchant_keepsakes_market
					}
				}
				OR = {
					NOT = { has_variable_list = merchant_keepsakes_markets }
					NOT = {
						is_target_in_variable_list = {
							name = merchant_keepsakes_markets
							target = scope:merchant_keepsakes_market
						}
					}
				}
			}
			add_to_variable_list = {
				name = merchant_keepsakes_markets
				target = scope:merchant_keepsakes_market
			}
		}
	}
}

merchant_souvenirs_aplenty_clear_effect = {
	if = {
		limit = { has_variable_list = merchant_keepsakes_markets }
		clear_variable_list = merchant_keepsakes_markets
	}
	if = {
		limit = { has_variable = merchant_keepsakes_prestige_value }
		remove_variable = merchant_keepsakes_prestige_value
	}
}

voyager_souvenirs_aplenty_completion_effect = {
	if = {
		limit = {
			has_perk = souvenirs_aplenty_perk
			NOT = { government_has_flag = government_is_merchant }
			has_variable = souvenirs_aplenty_var
		}
		send_interface_toast = {
			type = event_generic_good
			title = souvenirs_aplenty_perk_title_message
			
			root.capital_county = {
				change_prosperity_progress_with_overflow = root.var:souvenirs_aplenty_var
			}
		}
		remove_variable = souvenirs_aplenty_var
	}
	if = {
		limit = {
			government_has_flag = government_is_merchant
			has_variable_list = merchant_keepsakes_markets
		}
		if = {
			limit = { has_perk = souvenirs_aplenty_perk }
			set_variable = {
				name = merchant_keepsakes_prestige_value
				value = 0
			}
			every_in_list = {
				variable = merchant_keepsakes_markets
				root = {
					change_variable = {
						name = merchant_keepsakes_prestige_value
						add = 25
					}
				}
			}
			send_interface_toast = {
				type = event_generic_good
				title = souvenirs_aplenty_perk_title_message

				add_prestige = {
					value = var:merchant_keepsakes_prestige_value
				}
			}
		}
		merchant_souvenirs_aplenty_clear_effect = yes
	}
}

voyager_gracious_host_impeccable_guest_effect = {
	if = {
		limit = {
			has_perk = gracious_host_impeccable_guest_perk
			NOT = { government_has_flag = government_is_merchant }
			root.location.barony.holder.capital_barony ?= root.location.barony
			this != root.location.barony.holder
		}
		root.location.barony.holder = {
			add_opinion = {
				modifier = impeccable_guest_opinion
				opinion = 5
				target = root
			}
		}
	}
	if = {
		limit = {
			exists = root.location.barony.holder
			this != root.location.barony.holder
			root.location.barony.holder = {
				has_perk = gracious_host_impeccable_guest_perk
			}
		}
		add_opinion = {
			modifier = gracious_host_opinion
			opinion = 5
			target = root.location.barony.holder
		}
	}
	if = {
		limit = {
			government_has_flag = government_is_merchant
			has_perk = gracious_host_impeccable_guest_perk
			location.barony.trade_market ?= {}
		}
		location.barony.trade_market = { save_scope_as = polished_introductions_market }
		scope:polished_introductions_market = {
			if = {
				limit = {
					exists = trade_market_owner
					trade_market_owner != root
					trade_market_owner = {
						NOT = {
							has_opinion_modifier = {
								target = root
								modifier = merchant_polished_introductions_opinion
							}
						}
					}
				}
				trade_market_owner = {
					add_opinion = {
						target = root
						modifier = merchant_polished_introductions_opinion
						years = 5
					}
				}
			}
			every_character_in_market = {
				limit = {
					government_has_flag = government_is_merchant
					this != root
					NOT = {
						has_opinion_modifier = {
							target = root
							modifier = merchant_polished_introductions_opinion
						}
					}
				}
				add_opinion = {
					target = root
					modifier = merchant_polished_introductions_opinion
					years = 5
				}
			}
		}
	}
}

charting_the_realm_perk_effect = {
	if = {
		limit = {
			has_perk = realm_charts_perk
			OR = {
				NOT = { has_variable_list = mapmaking_location_list }
				NOT = {
					is_target_in_variable_list = {
						name = mapmaking_location_list
						target = location
	 				}
				}
			}
			exists = location.county
			location = { has_holding = yes }
			OR = { #We save domain provinces, top lieges capital or direct vassals capitals at count tier or over - always by link comparisons
				AND = {
					location.county.holder = root
					location.county != location.county.holder.capital_county
				}
				AND = {
					top_liege != this
					location.county.holder != root
					root.top_liege = location.county.holder.top_liege
					location = location.county.holder.top_liege.capital_province
				}
				AND = {
					location.county.holder != root
					location.county.holder.liege = root
					location = location.county.holder.capital_province
				}
			}
		}
		add_to_variable_list = {
			name = mapmaking_location_list
			target = location
		}
	}
}

surveyor_no_stone_unturned_perk_effect = {
	if = {
		limit = {
			OR = {
				AND = {
					has_perk = no_stone_unturned_perk
					government_has_flag = government_is_settled
				}
				AND = {
					has_perk = know_your_land_know_your_people_perk
					government_has_flag = government_is_nomadic
				}
			}
			this = scope:host
			exists = location.county
			location.county.holder = root
		}
		if = {
			limit = {
				location.county = {
					NOR = {
						has_county_modifier = no_stone_unturned_perk_modifier
						has_county_modifier = no_stone_unturned_boosted_perk_modifier
					}
				}
			}
			if = {
				limit = {
					has_perk = personal_touch_perk #Increased effect
				}
				send_interface_toast = {
					type = event_generic_good
					title = no_stone_unturned_perk_modifier

						add_county_modifier = {
							modifier = no_stone_unturned_boosted_perk_modifier
							years = 10
						}
						if = {
							limit = {
								scope:host = { government_has_flag = government_is_nomadic }
							}
							change_county_fertility = 15
						}
						else = {
							change_prosperity_progress_with_overflow = 15
						}
					}
				}
			}
			else = { #Standard Effect
				send_interface_toast = {
					type = event_generic_good
					title = no_stone_unturned_perk_modifier
					
					location.county = {
						add_county_modifier = {
							modifier = no_stone_unturned_perk_modifier
							years = 10
						}
						if = {
							limit = {
								scope:host = { government_has_flag = government_is_nomadic }

							change_county_fertility = 10
						}
						else = {
							change_prosperity_progress_with_overflow = 5
						}
					}
				}
			}
		}
		else = {
			location.county = { save_scope_as = perk_county }
			send_interface_toast = {
				type = event_generic_good
				title = no_stone_unturned_perk_cooldown_title_message
				custom_description_no_bullet = {
					text = 	no_stone_unturned_perk_cooldown_message
				}
			}
		}
	}
	if = {
		limit = {
			has_perk = local_inspection_perk
			NOT = { has_perk = no_stone_unturned_perk }
			government_has_flag = government_is_settled
			this = scope:host
			exists = location.county
			location.county.holder = root
			location.county = {
				NOT = { has_county_modifier = local_inspection_perk_modifier }
			}
		}
		send_interface_toast = {
			type = event_generic_good
			title = local_inspection_perk_modifier
			location.county = {
				add_county_modifier = {
					modifier = local_inspection_perk_modifier
					years = 5
				}
			}
		}
	}
}
