POD_become_spirit_effect = {
	save_temporary_scope_value_as = {
		name = saved_immortal_age
		value = POD_visual_age
	}
	hidden_effect = {
		if = {
			limit = { has_trait = shapeshifter }
			POD_rage_lose_the_wolf_effect = yes
		}
		POD_remove_splat_traits_effect = yes
	}
	add_trait = spirit
	if = {
		limit = { exists = scope:saved_immortal_age }
		hidden_effect = { set_immortal_age = scope:saved_immortal_age }
	}
	hidden_effect = {
		POD_remove_wrong_supernatural_traits_effect = yes
		POD_remove_wrong_supernatural_modifiers_effect = yes
		POD_remove_supernatural_relation_effect = yes
		POD_abandon_splat_journeys_effect = yes
		
		if = {
			limit = { is_pregnant = yes } 
			end_pregnancy = yes
			pregnancy_cleanup_effect = yes
		}
		if = {
			limit = { has_trait = drunkard }
			remove_trait = drunkard
		}
		if = {
			limit = { has_trait = hashishiyah }
			remove_trait = hashishiyah
		}
		if = {
			limit = { has_trait = rakish }
			remove_trait = rakish
		}
		if = {
			limit = { has_trait = flagellant }
			remove_trait = flagellant
		}
		if = {
			limit = { has_trait = comfort_eater }
			remove_trait = comfort_eater
		}
		if = {
			limit = { has_trait = inappetetic }
			remove_trait = inappetetic
		}
		if = {
			limit = { has_trait = athletic }
			remove_trait = athletic
		}
	}
}

POD_become_drone_effect = {
	save_temporary_scope_value_as = {
		name = saved_immortal_age
		value = POD_visual_age
	}
	add_trait_force_tooltip = drone
	if = {
		limit = { exists = scope:saved_immortal_age }
		hidden_effect = { set_immortal_age = scope:saved_immortal_age }
	}
	add_trait = possessed_genetic
	if = {
		limit = { POD_is_weaver_servant_trigger = no }
		POD_become_weaver_servant_effect = yes
		add_piety = massive_piety_loss
	}
}

possessed_faith_conversion_suicide_effect = {
	if = {
		limit = {
			OR = {
				AND = {
					has_trait = kami
					POD_is_wyrm_servant_trigger = yes
				}
				AND = {
					has_trait = drone
					POD_is_wyld_servant_trigger = yes
				}
				AND = {
					has_trait = gorgon
					POD_is_weaver_servant_trigger = yes
				}
				AND = {
					has_trait = fomori
					POD_is_gaia_servant_trigger = yes
				}
			}
		}
		death = { death_reason = death_annihilated }
	}
}
