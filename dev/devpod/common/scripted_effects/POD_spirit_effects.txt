POD_spirit_initialization_effect = {
	POD_remove_disease_traits_effect = yes
	POD_remove_supernatural_disease_traits_effect = yes
	add_character_flag = immune_to_disease
	force_character_skill_recalculation = yes

	POD_spirit_assign_rank_effect = yes
	POD_assign_spirit_language_effect = yes
	if = {
		limit = { faith = { is_spirit_faith_trigger = no } }
		if = {
			limit = { religion = { is_in_family = rf_gaia } }
			set_character_faith = faith:velvet_brood
		}
		else = {
			set_character_faith = faith:wyrmbound
		}
	}
	if = {
		limit = { has_variable = fera_pre_ancestor_rank }
		remove_variable = fera_pre_ancestor_rank
	}
}

POD_assign_spirit_language_effect = {
	if = {
		limit = { POD_spirit_uses_harmony_language_trigger = yes }
		learn_language = language_harmonys_melody
	}
	else_if = {
		limit = { POD_spirit_uses_conflict_language_trigger = yes }
		learn_language = language_conflicts_drum
	}
	else_if = {
		limit = { POD_spirit_uses_artisan_language_trigger = yes }
		learn_language = language_artisans_anthem
	}
	else_if = {
		limit = { POD_spirit_uses_sedition_language_trigger = yes }
		learn_language = language_seditions_lullaby
	}
	else_if = {
		limit = { POD_spirit_uses_wisdom_language_trigger = yes }
		learn_language = language_wisdoms_hymn
	}
	else_if = {
		limit = { POD_spirit_uses_chaos_language_trigger = yes }
		learn_language = language_chaos_threnody
	}
	else_if = {
		limit = { 
			POD_spirit_uses_ancestor_language_trigger = yes
			POD_has_spirit_ancestor_language_trigger = no
		}
		switch = {
			trigger = has_character_flag

			spirit_werewolf = { learn_language = language_ancestor_werewolf }
			spirit_bastet = { learn_language = language_ancestor_bastet }
			spirit_mokole = { learn_language = language_ancestor_mokole }
		}
	}
}

#The actual spawning. You should very rarely, if ever use this directly. Use the effects above instead
POD_spawn_spirit_effect = {
	if = {
		limit = { NOT = { exists = scope:spawning_character } }
		save_scope_as = spawning_character
	}
	create_character = { 
		location = this.location
		template = $TEMPLATE$ 
	}
	send_interface_message = {
		type = POD_spirit_message
		desc = POD_spawn_spirit_effect_desc
		left_icon = scope:spawned_spirit
	}
}

#For easily removing a spirit that was spawned for a single event
POD_despawn_spirit_effect = {
	hidden_effect = {
		scope:spawned_spirit = { pod_death_no_killer = { DEATH_REASON = death_disappearance } }
	}
}




POD_become_spirit_effect = {
	save_temporary_scope_value_as = { name = saved_immortal_age value = POD_visual_age }
	hidden_effect = { POD_remove_splat_traits_effect = yes }
	add_trait = spirit
	POD_depose_effect = yes
	hidden_effect = {
		set_immortal_age = scope:saved_immortal_age
		POD_remove_wrong_supernatural_traits_on_existing_character = yes
		POD_remove_supernatural_relation_effect = yes
		POD_remove_energy_modifiers_effect = yes
		POD_remove_spouse_relation_effect = yes
		POD_end_pregnancy = yes
		POD_remove_physical_coping_strategies = yes
		POD_side_effects_from_changing_splats = yes
		POD_spirit_initialization_effect = yes
	}
}

POD_become_ancestor_spirit_effect = {
	save_temporary_scope_value_as = { name = saved_immortal_age value = POD_visual_age }
	hidden_effect = { 
		switch = {
			trigger = has_trait

			werewolf = { add_character_flag = spirit_werewolf }
			bastet = { add_character_flag = spirit_bastet }
			mokole = { add_character_flag = spirit_mokole }

			fallback = { add_character_flag = spirit_werewolf }
		}
		set_variable = { name = fera_pre_ancestor_rank value = POD_fera_rank_value }
		POD_remove_splat_traits_effect = yes 
	}
	add_trait = spirit
	POD_depose_effect = yes
	hidden_effect = {
		set_immortal_age = scope:saved_immortal_age
		POD_remove_wrong_supernatural_traits_on_existing_character = yes
		POD_remove_supernatural_relation_effect = yes
		POD_remove_energy_modifiers_effect = yes
		POD_remove_spouse_relation_effect = yes
		POD_end_pregnancy = yes
		POD_remove_physical_coping_strategies = yes
		POD_side_effects_from_changing_splats = yes
		POD_spirit_initialization_effect = yes
	}
}
