POD_become_drone_effect = {
	save_temporary_scope_value_as = {
		name = saved_immortal_age
		value = POD_visual_age
	}
	add_trait_force_tooltip = drone
	if = {
		limit = { exists = scope:saved_immortal_age }
		hidden_effect = { set_immortal_age = scope:saved_immortal_age }
	}
	add_trait = possessed_genetic
	if = {
		limit = { POD_is_weaver_servant_trigger = no }
		POD_become_weaver_servant_effect = yes
		add_piety = massive_piety_loss
	}
}

possessed_faith_conversion_suicide_effect = {
	if = {
		limit = {
			OR = {
				AND = {
					has_trait = kami
					OR = {
						faith = { has_doctrine = doctrine_triat_alignment_wyrm }
						faith = { has_doctrine = wyrm_tainted_doctrine }
					}
				}
				AND = {
					has_trait = drone
					faith = { has_doctrine = doctrine_triat_alignment_wyld }
				}
				AND = {
					has_trait = gorgon
					faith = { has_doctrine = doctrine_triat_alignment_weaver }
				}
				AND = {
					has_trait = fomori
					religion = { is_in_family = rf_gaia }
					NOT = { faith = { has_doctrine = doctrine_triat_alignment_wyrm } }
				}
			}
		}
		pod_death_no_killer = {
			DEATH_REASON = death_suicide
		}
	}
}
