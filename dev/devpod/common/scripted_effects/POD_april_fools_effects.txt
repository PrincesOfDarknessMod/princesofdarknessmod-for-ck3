POD_turn_into_wetrius = {
	copy_inheritable_appearance_from = character:505084
	POD_make_character_wet = yes
	change_first_name = "Wetrius"
	add_character_flag = is_wetrius
	add_character_flag = POD_shaders_ignore_clothing
}

POD_wetriuspocalypse = {
	hidden_effect = {
		every_independent_ruler = {
			limit = {
				#highest_held_title_tier >= tier_county
				is_landed = yes
				is_ai = yes
			}
			POD_turn_into_wetrius = yes
		}
		character:505084 = {
			change_first_name = "Wetrius_Prime"
			POD_become_scripted_conqueror = yes
		}
		set_global_variable = wetrius_spawned
	}
}
