	save_temporary_scope_as = predator
	if = {
		limit = { #Not already a predator
			NOR = {
				has_trait = predatortype_consensualist
				any_secret = { secret_type = secret_predatortype_consensualist }
			}
		}
		if = {
			limit = {
				trait_is_shunned_or_criminal_in_my_or_lieges_faith_trigger = { TRAIT = predatortype_consensualist GENDER_CHARACTER = scope:predator }
			}
			add_secret = {
				type = secret_predatortype_consensualist
			}
			# random_secret = {

		}
	}
