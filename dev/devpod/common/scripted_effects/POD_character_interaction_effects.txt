POD_teleport_army_interaction_effect = {
	scope:actor = {
		custom_description = {
			text = POD_teleport_army_tt
			object = scope:recipient.location.barony
			commanding_army = {
				set_army_location = scope:recipient.location
			}
		}
	}
}

POD_teleport_travel_interaction_effect = {
	scope:actor = {
		custom_description = {
			text = POD_teleport_travel_tt
			object = scope:actor.current_travel_plan.next_destination_province.barony
			current_travel_plan ?= {
				# temporarily disable points of interest so they don't get collected on the way
				### common\travel\point_of_interest_types\travel_point_of_interest_types.txt
				every_future_path_location = {
					limit = { NOT = { this = prev.next_destination_province } }
					if = {
						limit = { has_travel_point_of_interest = poi_capitals }
						remove_travel_point_of_interest = poi_capitals
						trigger_event = { id = POD_POI_maintenance.1 days = 1 }
					}
					if = {
						limit = { has_travel_point_of_interest = poi_special_buildings_martial }
						remove_travel_point_of_interest = poi_special_buildings_martial
						trigger_event = { id = POD_POI_maintenance.2 days = 1 }
					}
					if = {
						limit = { has_travel_point_of_interest = poi_special_buildings_learning }
						remove_travel_point_of_interest = poi_special_buildings_learning
						trigger_event = { id = POD_POI_maintenance.3 days = 1 }
					}
					if = {
						limit = { has_travel_point_of_interest = poi_special_buildings_religious }
						remove_travel_point_of_interest = poi_special_buildings_religious
						trigger_event = { id = POD_POI_maintenance.4 days = 1 }
					}
					if = {
						limit = { has_travel_point_of_interest = poi_special_buildings_diplomatic }
						remove_travel_point_of_interest = poi_special_buildings_diplomatic
						trigger_event = { id = POD_POI_maintenance.5 days = 1 }
					}
					if = {
						limit = { has_travel_point_of_interest = poi_grand_city }
						remove_travel_point_of_interest = poi_grand_city
						trigger_event = { id = POD_POI_maintenance.6 days = 1 }
					}
					if = {
						limit = { has_travel_point_of_interest = poi_special_buildings_wonder }
						remove_travel_point_of_interest = poi_special_buildings_wonder
						trigger_event = { id = POD_POI_maintenance.7 days = 1 }
					}
					if = {
						limit = { has_travel_point_of_interest = poi_special_buildings_economic }
						remove_travel_point_of_interest = poi_special_buildings_economic
						trigger_event = { id = POD_POI_maintenance.8 days = 1 }
					}
					if = {
						limit = { has_travel_point_of_interest = poi_mausoleum_at_halicarnassus }
						remove_travel_point_of_interest = poi_mausoleum_at_halicarnassus

					if = {

					}

						limit = { has_travel_point_of_interest = poi_natural_feature }
						remove_travel_point_of_interest = poi_natural_feature
						trigger_event = { id = POD_POI_maintenance.11 days = 1 }
					}
				}
				add_destination_progress = { years = 9999 }
			}
		}
	}
