# to be used with POD_building_injector_haven (character scope)
POD_save_haven_building_macro = {
	if = {
		limit = { domicile = { has_domicile_building_or_higher = $BUILDING$ } }
		add_to_variable_list = {
			name = saved_haven_buildings
			target = flag:$BUILDING$
		}
	}
}

# to be used with POD_building_injector_haven (character scope)
POD_restore_saved_haven_buildings_macro = {
	if = {
		limit = {
			is_target_in_variable_list = {
				name = saved_haven_buildings
				target = flag:$BUILDING$
			}
			domicile = { NOT = { has_domicile_building = $BUILDING$ } }
		}
		domicile = { add_domicile_building = $BUILDING$ }
	}
}

# to be used with POD_building_injector_haven (character scope)
POD_demolish_incompatible_haven_buildings_macro = {
	domicile ?= {
		if = {
			limit = { has_domicile_building = $BUILDING$ }
			# this effect gives you a refund. because i'm nice
			# (there's also remove_domicile_building_no_refund but all that would do is add micro)
			remove_domicile_building = $BUILDING$
		}
	}
}

POD_save_haven_buildings = {
	if = {
		limit = { exists = domicile }
		POD_building_injector_haven = {
			EFFECT = POD_save_haven_building_macro
			HAS_REQUIRED_SPLAT = yes
			CHECK_MAIN_BUILDING = yes
		}
	}
}

POD_restore_saved_haven_buildings = {
	hidden_effect = {
		if = {
			limit = {
				exists = domicile
				has_variable_list = saved_haven_buildings
			}
			set_variable = recreating_haven
			POD_building_injector_haven = {
				EFFECT = POD_restore_saved_haven_buildings_macro
				HAS_REQUIRED_SPLAT = yes
				CHECK_MAIN_BUILDING = yes
			}
			remove_variable = recreating_haven
			clear_variable_list = saved_haven_buildings
			POD_save_haven_buildings = yes
		}
	}
}

POD_demolish_incompatible_haven_buildings = {
	hidden_effect = {
		if = {
			limit = { exists = domicile }
			set_variable = recreating_haven
			POD_building_injector_haven = {
				EFFECT = POD_demolish_incompatible_haven_buildings_macro
				HAS_REQUIRED_SPLAT = no
				CHECK_MAIN_BUILDING = no
			}
			remove_variable = recreating_haven
		}
	}
}
