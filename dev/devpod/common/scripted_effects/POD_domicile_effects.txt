###### General effects

POD_create_haven_landed_domicile = {
	if = {
		limit = {
			is_playable_character = yes
			is_landed = yes
			NOT = { exists = domicile }
			# barons should be covered under is_playable_character but apparently they aren't
			highest_held_title_tier > tier_barony
		}
		if = {
			limit = { has_government = traditions_government }
			POD_create_domicile_title = { GOVERNMENT = traditions_government }
		}
		else_if = {
			limit = { has_government = august_court_government }
			POD_create_domicile_title = { GOVERNMENT = august_court_government }
		}
		else_if = {
			limit = { has_government = inquisition_government }
			POD_create_domicile_title = { GOVERNMENT = inquisition_government }
		}
		else_if = {
			limit = { has_government = cell_government }
			POD_create_domicile_title = { GOVERNMENT = cell_government }
		}
		else_if = {
			limit = { has_government = cult_government }
			POD_create_domicile_title = { GOVERNMENT = cult_government }
		}
		else_if = {
			limit = {
				has_government = fera_tribe_government
				POD_is_mokole_trigger = yes
			}
			POD_create_domicile_title = { GOVERNMENT = fera_tribe_government }
		}
		# this works for the purposes of assigning domiciles to landed wraiths
		# but it won't let them keep domicile buildings on the landed <-> landless transition
		# (until wraith domiciles get reworked)
		else_if = {
			limit = { has_government = dark_kingdom_government }
			POD_create_domicile_title = { GOVERNMENT = dark_kingdom_government }
		}
	}
}

POD_create_domicile_title = {
	save_scope_as = holder
	create_nomad_title = {
		name = POD_domicile_title
		holder = scope:holder
		# no scopes allowed here, only literals. boooo
		government = $GOVERNMENT$
		save_scope_as = new_domicile_title
	}
	scope:new_domicile_title = {
		set_variable = {
			name  = haven_government
			value = government_type:$GOVERNMENT$
		}
	}
}

POD_recreate_domicile_on_government_change = {
	every_held_title = {
		limit = { POD_title_is_haven_landed_domicile = yes }
		prev = { destroy_title = prev }
	}
	POD_create_haven_landed_domicile = yes
	POD_restore_saved_haven_buildings = yes
}

###### Macro effects

# to be used with POD_building_injector_haven (character scope)
POD_save_haven_building_macro = {
	if = {
		limit = { domicile = { has_domicile_building_or_higher = $BUILDING$ } }
		add_to_variable_list = {
			name = saved_haven_buildings
			target = flag:$BUILDING$
		}
	}
}

# to be used with POD_building_injector_haven (character scope)
POD_restore_saved_haven_buildings_macro = {
	if = {
		limit = {
			is_target_in_variable_list = {
				name = saved_haven_buildings
				target = flag:$BUILDING$
			}
			domicile = { NOT = { has_domicile_building_or_higher = $BUILDING$ } }
		}
		domicile = { add_domicile_building = $BUILDING$ }
	}
}

# to be used with POD_building_injector_haven (character scope)
POD_demolish_incompatible_haven_buildings_macro = {
	domicile ?= {
		if = {
			limit = { has_domicile_building = $BUILDING$ }
			# this effect gives you a refund. because i'm nice
			# (there's also remove_domicile_building_no_refund but all that would do is add micro)
			remove_domicile_building = $BUILDING$
		}
	}
}

POD_save_haven_buildings = {
	if = {
		limit = { exists = domicile }
		POD_building_injector_haven = {
			EFFECT = POD_save_haven_building_macro
			HAS_REQUIRED_SPLAT = yes
			CHECK_MAIN_BUILDING = yes
		}
	}
}

POD_restore_saved_haven_buildings = {
	hidden_effect = {
		if = {
			limit = {
				exists = domicile
				has_variable_list = saved_haven_buildings
			}
			set_variable = recreating_haven
			POD_building_injector_haven = {
				EFFECT = POD_restore_saved_haven_buildings_macro
				HAS_REQUIRED_SPLAT = yes
				CHECK_MAIN_BUILDING = yes
			}
			remove_variable = recreating_haven
			clear_variable_list = saved_haven_buildings
			remove_variable = inheriting_player_haven
			POD_save_haven_buildings = yes
		}
	}
}

POD_demolish_incompatible_haven_buildings = {
	hidden_effect = {
		if = {
			limit = { exists = domicile }
			set_variable = recreating_haven
			POD_building_injector_haven = {
				EFFECT = POD_demolish_incompatible_haven_buildings_macro
				HAS_REQUIRED_SPLAT = no
				CHECK_MAIN_BUILDING = no
			}
			remove_variable = recreating_haven
		}
	}
}
