add_children_to_list = {
	save_scope_as = current
	if = {
		limit = { is_alive = yes }
		$CHARACTER$ = {
			add_to_variable_list = {
				name = bloodline
				target = scope:current
			}
		}
	}
	every_child = {
		even_if_dead = yes
		add_children_to_list = { CHARACTER = $CHARACTER$ }
	}
}

POD_initiate_blood_sorcery_exp_effect = {
	if = {
		limit = { NOT = { has_character_flag = got_BS_exp } }
		if = {
			limit = { has_trait = bloodsorceryadvanced }
			set_variable = { name = blood_sorcery_points value = 600}
		}
		else = {
			set_variable = { name = blood_sorcery_points value = 100}
		}
		add_character_flag = got_BS_exp
	}
}

POD_dominate_effect = {
	if = {
		limit = {
			has_strong_hook = $TARGET$
		}
		add_hook = { # refresh cooldown
			target = $TARGET$
			type = conditioning_hook
			days = dominate_duraction
		}
	}
	else_if = {
		limit = {
			has_hook_of_type = {
				target = $TARGET$
				type = rationalize_hook
			}
		}
		remove_hook = {
			target = $TARGET$
			type = rationalize_hook
		}	
		add_hook = {
			target = $TARGET$
			type = conditioning_hook
			days = dominate_duraction
		}
	}
	else_if = {
		limit = {
			has_hook_of_type = {
				target = $TARGET$
				type = mesmerize_hook
			}
		}
		remove_hook = {
			target = $TARGET$
			type = mesmerize_hook
		}	
		add_hook = {
			target = $TARGET$
			type = rationalize_hook
			days = dominate_duraction
		}
	}
	else = {
		remove_hook = { target = $TARGET$ }
		add_hook = {
			target = $TARGET$
			type = mesmerize_hook
			days = dominate_duraction
		}
	}
	if = {
		limit = { NOT = { has_relation_dominate_target = $TARGET$ } }
		set_relation_dominate_target = $TARGET$
	}
	$TARGET$ = { save_scope_as = dominate_target }
	trigger_event = { id = POD_mesmerize.700 days = dominate_duraction }
}

POD_learn_cappadocian_necromancy_effect = {
	if = {
		limit = { has_cappadocian_necromancy_trigger = no }
		custom_description = {
			text = POD_learn_cappadocian_necromancy
			subject = this
			add_character_flag = has_cappadocian_necromancy_flag
		}
	}
}
