add_children_to_list = {
	save_scope_as = current
	if = {
		limit = { is_alive = yes }
		$CHARACTER$ = {
			add_to_variable_list = {
				name = bloodline
				target = scope:current
			}
		}
	}
	every_child = {
		even_if_dead = yes
		add_children_to_list = { CHARACTER = $CHARACTER$ }
	}
}

POD_initiate_blood_sorcery_exp_effect = {
	if = {
		limit = { NOT = { has_character_flag = got_BS_exp } }
		if = {
			limit = { has_trait = bloodsorceryadvanced }
			set_variable = { name = blood_sorcery_points value = 600}
		}
		else = {
			set_variable = { name = blood_sorcery_points value = 100}
		}
		add_character_flag = got_BS_exp
	}
}
