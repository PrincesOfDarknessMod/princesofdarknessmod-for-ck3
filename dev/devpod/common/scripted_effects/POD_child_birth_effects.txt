# need to pass MOTHER and REAL_FATHER
POD_supernatural_trait_assignment_on_birth = {
	if = {
		limit = {
			NOT = { 
				$MOTHER$ ?= { has_character_modifier = warded_womb_modifier }
				$REAL_FATHER$ ?= { has_character_modifier = warded_womb_modifier }
			}
		}
		
		# Fae/Changeling/Kinain
		if = {
			limit = {
				$MOTHER$ ?= { has_trait = fae }
				$REAL_FATHER$ ?= { has_trait = fae }
				trigger_if = {
					limit = { $MOTHER$ ?= { is_ai = no } }
					$MOTHER$ = { NOT = { has_trait = fera } }
				}
				trigger_if = {
					limit = { $REAL_FATHER$ ?= { is_ai = no } }
					$REAL_FATHER$ = { NOT = { has_trait = fera } }
				}
			}
			add_trait = fae
			POD_fae_initialization_effect = yes
			add_character_flag = had_POD_character_initialization
		}
		else_if = {
			limit = {
				OR = {
					$MOTHER$ ?= { 
						OR = { 
							has_trait = fae	
							has_trait = changeling
							has_trait = kinain
						}
					}
					$REAL_FATHER$ ?= { 
						OR = { 
							has_trait = fae	
							has_trait = changeling
							has_trait = kinain
						}
					}
				}
				trigger_if = {
					limit = { $MOTHER$ ?= { is_ai = no } }
					$MOTHER$ = { NOT = { has_trait = fera } }
				}
				trigger_if = {
					limit = { $REAL_FATHER$ ?= { is_ai = no } }
					$REAL_FATHER$ = { NOT = { has_trait = fera } }
				}
			}
			add_trait = kinain
			POD_kinain_initialization_effect = yes
			add_character_flag = had_POD_character_initialization
			
			random = {
				chance = 10
				
				modifier = {
					add = 75
					$MOTHER$ ?= { has_trait = fae }
				}
				modifier = {
					add = 15
					$MOTHER$ ?= { has_trait = changeling }
				}
				
				modifier = {
					add = 75
					$REAL_FATHER$ ?= { has_trait = fae }
				}
				modifier = {
					add = 15
					$REAL_FATHER$ ?= { has_trait = changeling }
				}
				
				modifier = {
					add = 10
					$REAL_FATHER$ ?= {
						is_ai = yes
						is_landed = yes
					}
				}
				modifier = {
					add = 10
					$MOTHER$ ?= {
						is_ai = yes
						is_landed = yes
					}
				}
				
				trigger_event = { 			
					id = POD_fae_maintenance.1
					years = { 13 18 }
				}
			}
		}
		#Kinfolk/First Change	
		if = {
			limit = {
				OR = {
					$MOTHER$ ?= { 
						OR = {
							has_trait = kinfolk
							has_trait = fera
						}
					}
					$REAL_FATHER$ ?= { 
						OR = {
							has_trait = kinfolk
							has_trait = fera
						}
					}
				}
			}
			
			add_trait = kinfolk
			POD_kinfolk_initialization_effect = yes
			add_character_flag = had_POD_character_initialization
			
			random = {
				chance = 0
				modifier = { add = fera_shapeshifter_chance }

				add_character_flag = fera_first_change
			}

		}		
		# Revenant
		if = {
			limit = {
				OR = {
					$MOTHER$ ?= {
						OR = {
							AND = {
								POD_is_ghoul_trigger = yes
								NOT = { has_character_flag = recent_ghoulify }
							}
							POD_is_revenant_trigger = yes
						}
					}
					$REAL_FATHER$ ?= {
						OR = {
							AND = {
								POD_is_ghoul_trigger = yes
								NOT = { has_character_flag = recent_ghoulify }
							}
							POD_is_revenant_trigger = yes
						}
					}
				}
			}
			
			random = {
				chance = 0
				
				modifier = {
					add = 5
					$MOTHER$ ?= { POD_is_zadruga_trigger = yes }
				}
				modifier = {
					add = 5
					$REAL_FATHER$ ?= { POD_is_zadruga_trigger = yes }
				}
				modifier = {
					add = 5
					$MOTHER$ ?= { POD_is_szlachta_trigger = yes }	
				}
				modifier = {
					add = 5
					$REAL_FATHER$ ?= { POD_is_szlachta_trigger= yes }	
				}
				modifier = {
					add = 45
					$MOTHER$ ?= { POD_is_thinblooded_trigger = yes }
				}
				modifier = {
					add = 45
					$REAL_FATHER$ ?= { POD_is_thinblooded_trigger = yes }	
				}			
				modifier = {
					add = 40
					$MOTHER$ ?= { POD_is_revenant_trigger = yes }	
				}
				modifier = {
					add = 40
					$REAL_FATHER$ ?= { POD_is_revenant_trigger = yes }	
				}
				modifier = {
					add = 20
					$MOTHER$ ?= { 
						POD_is_ghoul_trigger = yes
						NOT = { has_character_flag = recent_ghoulify }
					}
				}
				modifier = {
					add = 20
					$REAL_FATHER$ ?= { 
						POD_is_ghoul_trigger = yes
						NOT = { has_character_flag = recent_ghoulify }
					}
				}
				
				add_trait = revenant
				
				POD_character_revenant_discipline_initialization_effect = yes
				
				if = {
					limit = {
						is_a_cultist = no
						POD_is_supehunter_trigger = no
						NOT = { faith.religion = { is_in_family = rf_kindred } }
					}
					POD_character_revenant_faith_initialization_effect = yes
				}
				POD_character_masquerade_initialization_effect = yes
				if = {
					limit = { POD_has_any_blood_hunger_modifier = no }
					add_character_modifier = bloodhunger0_modifier
				}
				if = {
					limit = { has_any_blood_potency_modifier = no }
					add_character_modifier = bloodpotency0_modifier
				}
				
				add_character_flag = had_POD_character_initialization
			}	
		}
		# Laham/Nephilim
		if = {
			limit = {
				OR = {
					$MOTHER$ ?= { has_trait = demon }
					$REAL_FATHER$ ?= { has_trait = demon }
				}
			}
			
			add_trait = laham
			add_character_flag = had_POD_character_initialization
		}
		if = {
			limit = {
				OR = {
					$MOTHER$ ?= { domicile ?= { has_domicile_parameter = unlock_nephilim_parameter } }
					$REAL_FATHER$ ?= { domicile ?= { has_domicile_parameter = unlock_nephilim_parameter } }
					$MOTHER$ ?= { this = character:795166 }
				}				
			}			
			add_trait = nephilim
			POD_nephilim_initialization_effect = yes
			add_character_flag = had_POD_character_initialization
			if = {
				limit = {
					OR = {
						$MOTHER$ ?= { domicile ?= { has_domicile_parameter = improved_nephilim_parameters } }
						$REAL_FATHER$ ?= { domicile ?= { has_domicile_parameter = improved_nephilim_parameters } }
					}
					NOT = {
						has_trait = physique_good
					}
				}			
				random_list = {
					50 = { add_trait = physique_good_1 }
					30 = { add_trait = physique_good_2 }
					20 = { add_trait = physique_good_3 }
				}
			}
		}
		if = {
			limit = {
				has_trait = laham
				has_trait = nephilim
			}
			remove_trait = laham
		}
		#Dhampyr ovewrite all 
		if = {
			limit = {
				OR = {
					$MOTHER$ ?= { 
						has_trait = kueijin
					}
					$REAL_FATHER$ ?= {
						has_trait = kueijin
					}
					#objective for dhampyr
					AND = {
						is_target_in_global_variable_list = {
							name = unavailable_unique_decisions
							target = flag:golden_children_decision
						}						
						OR = {
						$MOTHER$ ?= { 
							has_trait = dhampyr
						}
						$REAL_FATHER$ ?= {
							has_trait = dhampyr
						}
						}
					}
				}
			}
			if = {
				limit = {
					has_POD_race_trigger = yes
					POD_splat_mixing_haight_mode = no
				}
				POD_remove_splat_traits_effect = yes
				POD_remove_supernatural_basic_modifiers_effect = yes
				POD_remove_wrong_supernatural_traits_on_new_character = yes
				#if = {
				#	limit = {
				#		$MOTHER$ ?= { 
				#			is_ai = no 
				#			NOT = { any_child = { has_trait = dhampyr } }
				#		}
				#	}
				#	save_scope_as = dhampyr_child
				#	$MOTHER$ = { 
				#		trigger_event = podkueijin.260
				#	}
				#}
				#if = {
				#	limit = {
				#		father ?= { 
				#			is_ai = no 
				#			NOT = { any_child = { has_trait = dhampyr } }
				#		}
				#	}
				#	save_scope_as = dhampyr_child
				#	father = { 
				#		trigger_event = podkueijin.260
				#	}
				#}
			}
			add_trait = dhampyr
			POD_dhampyr_initialization_effect = yes
			add_character_flag = had_POD_character_initialization
		}	
		# Numina talent
		if = {
			limit = {
				OR = {
					has_main_supernatural_race_trigger = no
					POD_numina_on_birth_unrestricted = yes
				}
			}
			random = {
				chance = 1
				add_trait = psychic
			}
		}
		# Dhampir or Mortal
		if = {
			limit = {
				OR = {
					has_POD_race_trigger = no
					POD_splat_mixing_haight_mode = yes
				}
				OR = {
					$MOTHER$ ?= { 
						has_trait = vampire
						NOT = { has_character_flag = recent_embraced }
					}
					$REAL_FATHER$ ?= {
						has_trait = vampire
						NOT = { has_character_flag = recent_embraced }
					}
				}
			}
			add_trait = dhampir
			add_character_flag = had_POD_character_initialization
			
			if = {
				limit = { POD_has_any_blood_hunger_modifier = no }
				add_character_modifier = bloodhunger0_modifier
			}
			if = {
				limit = { has_any_blood_potency_modifier = no }
				add_character_modifier = bloodpotency0_modifier
			}
		}
		
		# band-aid for making sure the character only has one energy type
		POD_change_to_appropriate_energy_type = yes
	}
}
