POD_willpower_rouse_check_effect = {
	custom_tooltip = POD_willpower_rouse_effect.tt

	hidden_effect = {
		random = {
			chance = 50
			send_interface_toast = {
				title = POD_willpower_rouse
				left_icon = this
				POD_static_stress_gain = { COUNT = minor_stress_impact_gain }
			}
		}
	}
}

POD_willpower_regain_check_effect = {
	custom_tooltip = POD_willpower_regain_effect.tt

	hidden_effect = {
		random = {
			chance = 50
			send_interface_toast = {
				title = POD_willpower_regain
				left_icon = this
				add_stress = minor_stress_impact_loss
			}
		}
	}
}
POD_static_stress_gain = {
	custom_description = {
		text = "POD_static_stress_gain"
		value = $COUNT$
		if = { limit = { NOT = { $COUNT$ = 0 } } 
			add_character_flag = static_stress_gain_flag
			save_scope_value_as = { name = saved_stress  value = stress }
			set_variable = { name = saved_stress  value = stress }
			set_variable = { name = saved_stress_level  value = stress_level }
			
			add_stress = 100
			
			while = {
				limit = { stress = scope:saved_stress }
				add_character_modifier = POD_static_stress_gain_1_modifier
				force_character_skill_recalculation = yes
				add_stress = 100
			}
			add_stress = -600
			while = {
				limit = { stress > 0 }
				add_character_modifier = POD_static_stress_gain_2_modifier
				force_character_skill_recalculation = yes
				add_stress = -600
			}      
			add_stress = 100
			while = {
				limit = { NOT = { stress = 100 } }
				if = { limit = { stress > 100 } add_character_modifier = POD_static_stress_gain_3_modifier }
				else = { add_character_modifier = POD_static_stress_gain_35_modifier }

				force_character_skill_recalculation = yes
				add_stress = -600
				add_stress = 100
			}
			add_stress = -600
			add_stress = 100
			add_stress = -50
			while = {
				limit = { NOT = { stress = 50 } }
				if = { limit = { stress < 50 } add_character_modifier = POD_static_stress_gain_4_modifier }
				else = { add_character_modifier = POD_static_stress_gain_45_modifier }
				force_character_skill_recalculation = yes

				add_stress = -600
				add_stress = 100
				add_stress = -50
			}
			add_stress = -600
			add_stress = scope:saved_stress
			remove_character_flag = static_stress_gain_flag
			remove_variable = saved_stress
			remove_variable = saved_stress_level
			add_stress = $COUNT$
			remove_all_character_modifier_instances = POD_static_stress_gain_1_modifier
			remove_all_character_modifier_instances = POD_static_stress_gain_2_modifier
			remove_all_character_modifier_instances = POD_static_stress_gain_3_modifier
			remove_all_character_modifier_instances = POD_static_stress_gain_35_modifier
			remove_all_character_modifier_instances = POD_static_stress_gain_4_modifier
			remove_all_character_modifier_instances = POD_static_stress_gain_45_modifier
			force_character_skill_recalculation = yes
		}
	}
}
