
POD_character_initialization_for_werewolf_assign_auspice_effect = {
	if = {
		limit = {
			POD_is_werewolf_trigger = yes
			POD_has_auspice_trigger = no
		}	
		
		# Assign Auspice at Day of Month
		if = {
			limit = { current_day <= 3 }
			add_trait = ragabash
		}
		else_if = {
			limit = { current_day <= 9 }
			add_trait = theurge
		}
		else_if = {
			limit = { current_day <= 15 }
			add_trait = philodox
		}
		else_if = {
			limit = { current_day <= 21 }
			add_trait = galliard
		}
		else_if = {
			limit = { current_day <= 27 }
			add_trait = ahroun
		}
		else = {
			add_trait = ragabash
		}
	}
}
