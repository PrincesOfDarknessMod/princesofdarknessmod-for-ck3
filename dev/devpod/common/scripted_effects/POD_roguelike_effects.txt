POD_select_roguelike_character_effect = {
	hidden_effect = {
		random_ruler = {
			limit = { POD_is_valid_roguelike_ruler_trigger = yes }
			save_scope_as = new_character
		}
	
		set_player_character = scope:new_character
	
		scope:new_character = {
			trigger_event = podintroductions.7001
		}
	}
}

POD_roguelike_switch_to_randomized_vampire_effect = {
	hidden_effect = {
		random_ruler = {
			limit = {
				has_trait = vampire
				POD_is_valid_roguelike_ruler_trigger = yes
			}
			save_scope_as = character_to_replace
		}
	
		# Select random religion
		random_religion_global = {
			limit = {
				is_in_family = rf_kindred
				NOT = { this = religion:modernanarch_religion }
				NOT = { this = religion:camarilla_religion    }
				NOT = { this = religion:sabbat_religion       }
			}
			random_faith = {
				save_scope_as = randomized_faith
			}
		}
	
		# Select random culture
		random_culture_global = {
			save_scope_as = randomized_culture
		}
		
		create_character = {
			template = POD_roguelike_base_template
			location = scope:character_to_replace.location
			faith    = scope:randomized_faith
			culture  = scope:randomized_culture
	
			save_scope_as = new_character
		}
	
		hidden_effect_new_object = {
			scope:new_character = {
				add_trait = vampire
				set_age = 100
				set_immortal_age = { 18 50 }
	
				# always 8th generation
				add_trait = vampgeneration8
	
				# random clan
				random_trait = {
					limit = {
						has_trait_flag = vamp_clan
						NOT = { has_trait_flag = exclude_from_randomizer }
					}
					prev = { add_trait = prev }
				}
	
				# 3 random disciplines
				while = {
					count = 3
					random_trait_in_category = {
						category = lifestyle
						limit = {
							has_trait_flag = discipline
							NOT = { has_trait_flag = discipline_advanced     }
							NOT = { has_trait_flag = exclude_from_randomizer }
							NOT = { prev = { has_trait = prev } }
						}
						prev = { add_trait = prev }
					}
				}
	
				if = {
					limit = {
						has_trait = bloodsorcery
						has_any_bloodmagic_path = no
					}
					random_list = {
						1 = { add_character_flag = has_koldunic_bloodmagic_flag    }
						1 = { add_character_flag = has_sadhana_bloodmagic_flag     }
						1 = { add_character_flag = has_akhu_bloodmagic_flag        }
						1 = { add_character_flag = has_dur_an_ki_bloodmagic_flag   }
						1 = { add_character_flag = has_thaumaturgy_bloodmagic_flag }
					}
				}
	
				if = {
					limit = {
						has_trait = necromancy
						has_any_necromancy_path = no
					}
					random_list = {
						1 = { add_character_flag = has_giovanni_necromancy_flag    }
						1 = { add_character_flag = has_cappadocian_necromancy_flag }
						1 = { add_character_flag = has_harbinger_necromancy_flag   }
						1 = { add_character_flag = has_lamia_necromancy_flag       }
						1 = { add_character_flag = has_impundulu_necromancy_flag   }
						1 = { add_character_flag = has_nagaraja_necromancy_flag    }
					}
				}
				
				# Make one of the disciplines advanced
				random_list = {
					1 = { trigger = { has_trait = necromancy } assign_melancholy_necromancy_effect = yes }
					1 = {
						trigger = { has_trait = quietus }
						random_list = {
							1 = { assign_sorcerer_quietus_effect = yes }
							1 = { assign_warrior_quietus_effect  = yes }
							1 = { assign_vizier_quietus_effect   = yes }
						}
					}
					1 = {
						trigger = { has_trait = valeren }
						random_list = {
							1 = { assign_healer_valeren_effect  = yes }
							1 = { assign_warrior_valeren_effect = yes }
							1 = { assign_watcher_valeren_effect = yes }
						}
					}
					1 = { trigger = { has_trait = abombwe       } assign_abombwe_effect         = yes }
					1 = { trigger = { has_trait = animalism     } assign_animalism_effect       = yes }
					1 = { trigger = { has_trait = auspex        } assign_auspex_effect          = yes }
					1 = { trigger = { has_trait = bardo         } assign_bardo_effect           = yes }
					1 = { trigger = { has_trait = bloodsorcery  } assign_bloodsorcery_effect    = yes }
					1 = { trigger = { has_trait = celerity      } assign_celerity_effect        = yes }
					1 = { trigger = { has_trait = chimerstry    } assign_chimerstry_effect      = yes }
					1 = { trigger = { has_trait = daimonion     } assign_daimonion_effect       = yes }
					1 = { trigger = { has_trait = dementation   } assign_dementation_effect     = yes }
					1 = { trigger = { has_trait = dominate      } assign_dominate_effect        = yes }
					1 = { trigger = { has_trait = fortitude     } assign_fortitude_effect       = yes }
					1 = { trigger = { has_trait = melpominee    } assign_melpominee_effect      = yes }
					1 = { trigger = { has_trait = mytherceria   } assign_mytherceria_effect     = yes }
					1 = { trigger = { has_trait = obfuscate     } assign_obfuscate_effect       = yes }
					1 = { trigger = { has_trait = obtenebration } assign_obtenebration_effect   = yes }
					1 = { trigger = { has_trait = potence       } assign_potence_effect         = yes }
					1 = { trigger = { has_trait = presence      } assign_presence_effect        = yes }
					1 = { trigger = { has_trait = protean       } assign_protean_effect         = yes }
					1 = { trigger = { has_trait = serpentis     } assign_serpentis_effect       = yes }
					1 = { trigger = { has_trait = spiritus      } assign_animal_spiritus_effect = yes }
					1 = { trigger = { has_trait = temporis      } assign_temporis_effect        = yes }
					
					1 = { trigger = { has_trait = thinbloodalchemy } assign_thinbloodalchemy_effect = yes }
				}
	
				# Necessary stuff
				POD_remove_wrong_supernatural_traits_effect = yes
				POD_set_resonance_effect = yes
				POD_character_vampire_clan_special_effect = yes
				remove_disease_trait_effect = yes
				add_character_flag = immune_to_disease
				POD_character_initialization_random_background_effect = yes
	
				# Random predator type
				POD_character_vampire_predatortype_initialization_effect = yes
	
				# Random prey exclusion for Ventrue (age or education only)
				if = {
					limit = { is_ventrue_trigger = yes }
					random_list = {
						1 = { add_secret = { type = secret_prey_young_men         } }
						1 = { add_secret = { type = secret_prey_young_women       } }
						1 = { add_secret = { type = secret_prey_middle_aged_men   } }
						1 = { add_secret = { type = secret_prey_middle_aged_women } }
						1 = { add_secret = { type = secret_prey_old_men           } }
						1 = { add_secret = { type = secret_prey_old_women         } }
						1 = { add_secret = { type = secret_prey_diplomacy         } }
						1 = { add_secret = { type = secret_prey_martial           } }
						1 = { add_secret = { type = secret_prey_stewardship       } }
						1 = { add_secret = { type = secret_prey_intrigue          } }
						1 = { add_secret = { type = secret_prey_learning          } }
					}
				}
	
				# Modifiers
				POD_character_vampire_bloodpotency_initialization_effect = yes
				if = {
					limit = { POD_has_infinite_blood_trigger = yes }
					add_character_modifier = bloodhunger0_modifier
				}
				else = {
					add_character_modifier = bloodhunger1_modifier
				}
				if = {
					limit = { POD_splat_has_masquerade_trigger = yes }
					if = {
						limit = { minimum_masquerade_exposure = 1 }
						add_character_modifier = masquerade1_modifier
					}
					else = {
						add_character_modifier = masquerade0_modifier
					}
				}
			}
			
			# Give title to new character
			create_title_and_vassal_change = {
				type = usurped
				save_scope_as = change
				add_claim_on_loss = no
			}
			scope:character_to_replace.primary_title = {
				change_title_holder_include_vassals = {
					holder = scope:new_character
					change = scope:change
				}
			}
			resolve_title_and_vassal_change = scope:change
	
			scope:new_character = {
				# Starting gold and piety (based on previous character)
				add_gold     = scope:character_to_replace.gold
				add_prestige = scope:character_to_replace.prestige
				add_piety    = scope:character_to_replace.piety
			}
	
			# Switch to new character
			set_player_character = scope:new_character
	
			# Set up curses
			scope:new_character = {
				trigger_event = podintroductions.7001
			}
		}
	}
}

POD_major_objective_finished_effect = {
	if = {
		limit = { POD_has_roguelike_curse = yes }
		show_as_tooltip = { POD_lift_roguelike_curses_effect = yes }
		trigger_event = pod1230objectives.7000
	}
}

POD_lift_roguelike_curses_effect = {
	custom_tooltip = pod1230objectives.7000.tt

	if = {
		limit = { has_character_flag = POD_roguelike_all_curses }
		remove_character_flag = POD_roguelike_all_curses
		add_character_flag = POD_lifted_all_roguelike_curses
	}
	else_if = {
		limit = { has_character_flag = POD_roguelike_cursed }
		remove_character_flag = POD_roguelike_cursed
		add_character_flag = POD_lifted_roguelike_curse
	}

	if = {
		limit = { has_character_modifier = POD_roguelike_no_wealth }
		remove_character_modifier = POD_roguelike_no_wealth
	}
	if = {
		limit = { has_character_modifier = POD_roguelike_no_love }
		remove_character_modifier = POD_roguelike_no_love
	}
	if = {
		limit = { has_character_modifier = POD_roguelike_no_rest }
		remove_character_modifier = POD_roguelike_no_rest
	}
	if = {
		limit = { has_character_modifier = POD_roguelike_no_belongings }
		remove_character_modifier = POD_roguelike_no_belongings
	}
	if = {
		limit = { has_character_modifier = POD_roguelike_no_curiosity }
		remove_character_modifier = POD_roguelike_no_curiosity
	}
	if = {
		limit = { has_character_modifier = POD_roguelike_no_guile }
		remove_character_modifier = POD_roguelike_no_guile
	}
	if = {
		limit = { has_character_modifier = POD_roguelike_no_demands }
		remove_character_modifier = POD_roguelike_no_demands
	}
}

POD_roguelike_ensure_gameover_effect = {
	hidden_effect = {
		dynasty = {
			every_dynasty_member = {
				death = { death_reason = death_disappearance }
			}
		}
	}
}

POD_roguelike_ensure_gameover_on_death_effect = {
	save_scope_as = roguelike_player
	hidden_effect = {
		dynasty = {
			every_dynasty_member = {
				limit = { NOT = { this = scope:roguelike_player } }
				death = { death_reason = death_disappearance }
			}
		}
	}
}
