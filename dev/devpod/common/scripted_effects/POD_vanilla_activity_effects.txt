POD_pilgrimage_on_start = {
	scope:activity ?= {
		if = {
			limit = { scope:host = { has_character_modifier = allahs_blessing_character_modifier } }
			scope:activity = { activity_special_type_progression_variable = { NUM = 20 } }
		}
	}
}

POD_pilgrimage_on_enter_travel_state = {
	scope:activity ?= {
		scope:host = {
			if = {
				limit = {
					has_character_modifier = allahs_blessing_character_modifier
					current_travel_plan = { NOT = { has_travel_plan_modifier = allahs_blessing_travel_modifier } }
				}
				current_travel_plan = { add_travel_plan_modifier = allahs_blessing_travel_modifier }
			}
		}
	}
}
