POD_investiture_perk_cost_effect = {
	hidden_effect = {
		save_scope_value_as = {
			name = cost
			value = {
				value = $VALUE$
				add = -1
				multiply = -1
			}
		}
		add_investiture_lifestyle_perk_points = scope:cost
	}
	custom_description_no_bullet = { text = investiture_cost_effect value = $VALUE$ }
}

POD_bind_demon_effect = {
	$DEMON$ = { save_scope_as = demon }
	hidden_effect = {
		save_scope_value_as = {
			name = perk_points_gained
			value = {
				add = {
					value = scope:demon.sum_of_all_skills_value
					divide = 15
				}
				add = {
					value = scope:demon.prowess
					divide = 20
				}
				floor = yes
			}
		}
		add_investiture_lifestyle_perk_points = scope:perk_points_gained
		if = {
			limit = { has_relation_worshipped_demon = scope:demon }
			remove_relation_worshipped_demon = scope:demon
			if = { limit = { has_trait = mortal can_hold_title_as_mortal_trigger = no } POD_depose_effect = yes }
		}
		set_relation_bound_demon = scope:demon
	}
	custom_tooltip = POD_bind_demon_effect
}
