	}

POD_create_pact_effect = {
	create_story = {
		type = POD_story_pact
		save_scope_as = pact
	}
	##############################
	$OBLIGATION$ = { save_scope_as = true_obligation }
	save_scope_value_as = { name = true_obligation_level value = $OBLIGATION_LEVEL$ }
	##############################
	save_scope_value_as = { name = true_frequency value = $FREQUENCY$ }
	if = {
		limit = { POD_pact_obligation_is_prohibitory_trigger = { OBLIGATION = scope:true_obligation } }
		save_scope_value_as = { name = true_frequency value = 4 }
		custom_tooltip = POD_create_pact_effect_forced_frequency
		save_scope_as = ignore_frequency_tooltip
	}
	##############################
	$CONSEQUENCE$ = { save_scope_as = true_consequence }
	if = {
		limit = { POD_has_pact_with_consequence_trigger = { CONSEQUENCE = flag:life } }
		save_scope_value_as = { name = true_consequence value = flag:soul }
		custom_tooltip = POD_create_pact_effect_forced_consequence
	}
	save_scope_value_as = { 
		name = true_consequence_level
		value = {
			value = $CONSEQUENCE_LEVEL$
			if = {
				limit = { POD_has_pact_with_consequence_trigger = { CONSEQUENCE = flag:life } }
				value = 4
			}
		}
	}
	##############################
	save_scope_value_as = {
		name = perk_points_gained
		value = {
			value = scope:true_obligation_level
			add = scope:true_consequence_level
			subtract = { value = scope:true_frequency divide = 4 }
			min = 0
		}
	}
	add_investiture_lifestyle_perk_points = scope:perk_points_gained
	##############################
	$DEMON$ = { save_scope_as = demon }
	custom_tooltip = POD_create_pact_effect
	if = { 
		limit = { NOT = { exists = scope:ignore_obligation_tooltip } } 
		custom_tooltip = POD_create_pact_effect_obligation
		save_scope_as = ignore_obligation_tooltip
	}
	if = { 
		limit = { NOT = { exists = scope:ignore_frequency_tooltip } } 
		custom_tooltip = POD_create_pact_effect_frequency
		save_scope_as = ignore_frequency_tooltip
	}
	if = { 
		limit = { NOT = { exists = scope:ignore_consequence_tooltip } } 
		custom_tooltip = POD_create_pact_effect_consequence
		show_as_tooltip = {
			POD_pact_consequence_effect = { 
				THRALL = THIS 
				CONSEQUENCE = scope:true_consequence
			}
		}
		save_scope_as = ignore_consequence_tooltip
	}
	##############################
	hidden_effect_new_object = {	
		scope:pact ?= {
			set_variable = { name = obligation value = scope:true_obligation }
			set_variable = { name = frequency value = scope:true_frequency }
			set_variable = { name = consequence value = scope:true_consequence }
			set_variable = { name = perk_points_gained value = scope:perk_points_gained }
			set_variable = { name = demon value = scope:demon }
			set_variable = { name = completed value = no }
		}
		add_to_variable_list = {
			name = demonic_pacts
			target = scope:pact
		}
	}
}

POD_pact_complete_obligation_effect = {
	if = {
		limit = {
			any_in_list = {
				variable = demonic_pacts
				var:obligation = flag:$OBLIGATION$
				var:completed = no
				save_temporary_scope_as = demonic_pact
			}
		}
		if = {
			limit = { POD_pact_obligation_is_prohibitory_trigger = { OBLIGATION = flag:$OBLIGATION$ } }
			POD_pact_consequence_effect = {
				THRALL = THIS
				CONSEQUENCE = scope:demonic_pact.var:consequence
			}
		}
		else = {
			custom_tooltip = POD_pact_complete_obligation_effect
			scope:demonic_pact ?= { set_variable = { name = completed value = yes } }
			if = {
				limit = {
					any_in_list = {
						variable = demonic_pacts
						var:obligation = flag:$OBLIGATION$
						var:completed != yes
						NOT = { THIS = scope:demonic_pact }
					}
				}
				custom_tooltip = POD_pact_complete_obligation_effect_additional
			}
		}
	}
}
POD_pact_consequence_effect = {
	$CONSEQUENCE$ = { save_scope_as = consequence }
	$THRALL$ = { 
		send_interface_message = {
			title = POD_infernalism_events.1.consequence
			type = event_generic_bad
			switch = {
				trigger = scope:consequence
				flag:will = { add_stress = 100 }
				flag:health = { 
					POD_increase_wounds_effect = {
						REASON = consequence_of_pact
						COUNT = 2
					}
				}
				flag:knowledge = { 
					add_character_modifier = {
						modifier = POD_pact_consequence_knowledge_modifier
						years = 20
					}
				}
				flag:soul = { 
					add_character_modifier = POD_pact_consequence_soul_modifier
					
				}
				flag:life = { 
					death = { death_reason = death_consequence_of_pact } 
				}
			}
		}
	}
}
