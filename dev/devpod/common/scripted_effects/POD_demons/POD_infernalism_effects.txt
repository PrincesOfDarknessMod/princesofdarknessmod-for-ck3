# based on give_witch_secret_or_trait_effect
POD_give_and_expose_witch_trait_effect = {
	save_temporary_scope_as = witch_character
	if = {
		limit = {
			NOT = { has_trait = witch }
			POD_splat_can_be_infernalist = yes
		}
		add_trait_force_tooltip = witch
		
		if = {
			limit = { any_secret = { type = secret_witch } }
			hidden_effect = {
				every_secret = {
					type = secret_witch
					remove_secret = yes
				}
			}
		}
		else = {
			POD_side_effects_from_becoming_infernalist = yes
		}
		POD_side_effects_from_becoming_public_infernalist = yes
		
		# If you have a legend maybe we change it?
		if = {
			limit = {
				has_legend_chapter_trigger = { CHAPTER = opening }
			}
			random = {
				chance = 10
				send_interface_toast = {
					title = legend_chapter_change_toast
					left_icon = root
					promoted_legend = {
						set_legend_chapter = {
							name = opening
							localization_key = legend_chapter_opening_witch
						}
					}
				}
			}
		}
	}
}

POD_bind_demon = {
	$DEMON$ = { save_scope_as = demon }
	$ARTIFACT$ = { save_scope_as = artifact }
	hidden_effect = {
		scope:demon = {
			save_scope_value_as = {
				name = perk_points_gained
				value = POD_diabolist_perks_gained
			}
			set_variable = {
				name = diabolist_perks_gained
				value = scope:perk_points_gained
			}
			set_variable = {
				name = bound_demon_artifact
				value = scope:artifact
			}
			add_to_variable_list = {
				name = diabolists_with_investiture
				target = prev
			}
		}
		scope:artifact = {
			POD_clear_bound_demon_artifact_variables = yes
			set_variable = {
				name = artifact_bound_demon
				value = scope:demon
			}
			hidden_effect = {
				while = {
					count = scope:perk_points_gained
					add_artifact_modifier = bound_demon_artifact_modifier
				}
			}
		}
		add_investiture_lifestyle_perk_points = scope:perk_points_gained
		add_character_flag = access_to_investiture
		save_scope_as = force_witch_secret
		give_witch_secret_or_trait_effect = yes
		if = {
			limit = { has_relation_worshipped_demon = scope:demon }
			remove_relation_worshipped_demon = scope:demon
			if = {
				limit = {
					has_trait = mortal
					can_hold_title_as_mortal_trigger = no
				}
				POD_depose_effect = yes
			}
		}
		set_relation_bound_demon = scope:demon
	}
	custom_tooltip = POD_bind_demon_effect
}

POD_clear_demon_binding = {
	save_scope_as = demon
	every_in_list = {
		variable = diabolists_with_investiture
		limit = {
			exists = this
			is_alive = yes
		}
		trigger_event = POD_demon_maintenance.3
	}

	var:bound_demon_artifact ?= {
		POD_clear_bound_demon_artifact_variables = yes
	}
	remove_variable = bound_demon_artifact
}

		}
	}
}

POD_create_pact_from_stipulation_stories = {
	if = {
		limit = {
			exists = $FREQUENCY_STORY$
			NOT = {
				POD_pact_obligation_is_prohibitory_trigger = {
					OBLIGATION = $OBLIGATION_STORY$.var:id
				}
			}
		}
		POD_get_pact_frequency_from_id = {
			ID = $FREQUENCY_STORY$.var:id
			SAVE_AS = frequency
		}
		save_scope_value_as = {
			name  = frequency_level
			value = $FREQUENCY_STORY$.var:level
		}
	}
	else = {
		save_scope_value_as = {
			name  = frequency
			value = 4
		}
		save_scope_value_as = {
			name  = frequency_level
			value = -1
		}
	}
	save_scope_value_as = {
		name = perks_gained
		value = {
			add = $OBLIGATION_STORY$.var:level
			add = $CONSEQUENCE_STORY$.var:level
			add = scope:frequency_level
			min = 0
		}
	}
	POD_create_pact = {
		DEMON        = $DEMON$
		OBLIGATION   = $OBLIGATION_STORY$.var:id
		FREQUENCY    = scope:frequency
		CONSEQUENCE  = $CONSEQUENCE_STORY$.var:id
		PERKS_GAINED = scope:perks_gained
	}
}

POD_create_pact = {
	$DEMON$ = { save_scope_as = demon }
	$OBLIGATION$  = { save_scope_as = obligation_flag }
	$CONSEQUENCE$ = { save_scope_as = consequence_flag }
	save_scope_value_as = {
		name  = frequency
		value = $FREQUENCY$
	}
	save_scope_value_as = {
		name  = perk_points_gained
		value = $PERKS_GAINED$
	}
	
	add_investiture_lifestyle_perk_points = scope:perk_points_gained
	
	custom_tooltip = POD_create_pact_effect_obligation
	if = {
		limit = { NOT = { POD_pact_obligation_is_prohibitory_trigger = { OBLIGATION = scope:obligation_flag } } }
		custom_tooltip = POD_create_pact_effect_frequency
	}
	custom_tooltip = POD_create_pact_effect_consequence
	show_as_tooltip = {
		POD_pact_consequence_from_fine_print = {
			THRALL = THIS
			CONSEQUENCE = scope:consequence_flag
		}
	}
	
	hidden_effect_new_object = {
		create_story = {
			type = POD_story_pact
			save_scope_as = pact
		}
		scope:pact ?= {
			set_variable = { name = obligation value = scope:obligation_flag }
			set_variable = { name = frequency value = scope:frequency }
			set_variable = { name = consequence value = scope:consequence_flag }
			set_variable = { name = perk_points_gained value = scope:perk_points_gained }
			set_variable = { name = demon value = scope:demon }
			set_variable = { name = completed value = no }
		}
		add_to_variable_list = {
			name = demonic_pacts
			target = scope:pact
		}
	}
}

POD_pact_complete_obligation_with_demon_effect = {
	if = {
		limit = {
			any_in_list = {
				variable = demonic_pacts
				var:obligation = flag:$OBLIGATION$
				var:completed = no
				save_temporary_scope_as = demonic_pact
			}
		}
		custom_tooltip = POD_pact_complete_obligation_effect
		scope:demonic_pact ?= {
			set_variable = { name = completed value = yes }
			if = {
				limit = { var:demon ?= { is_alive = yes } }
				var:demon = {
					$DEMON_EFFECT$
				}
			}
		}
		if = {
			limit = {
				any_in_list = {
					variable = demonic_pacts
					var:obligation = flag:$OBLIGATION$
					var:completed != yes
					this != scope:demonic_pact
				}
			}
			custom_tooltip = POD_pact_complete_obligation_effect_additional
		}
	}
}

POD_pact_consequence_from_fine_print = {
	$CONSEQUENCE$ = { save_scope_as = consequence }
	$THRALL$ = { 
		send_interface_message = {
			title = game_concept_demonic_pact_the_fine_print
			type = event_generic_bad
			switch = {
				trigger = scope:consequence
				flag:will = { add_stress = 100 }
				flag:health = { 
					POD_increase_wounds_effect = {
						REASON = consequence_of_pact
						COUNT = POD_double_wound_damage
					}
				}
				flag:knowledge = { 
					add_character_modifier = {
						modifier = POD_pact_consequence_knowledge_modifier
						years = 20
					}
				}
				flag:soul = { 
					add_character_modifier = POD_pact_consequence_soul_modifier
				}
				flag:life = { 
					death = { death_reason = death_consequence_of_pact } 
				}
			}
		}
	}
}

# arguments: ID, SAVE_AS
POD_get_pact_frequency_from_id = {
	save_scope_value_as = {
		name = $SAVE_AS$
		value = {
			switch = {
				trigger = $ID$
				flag:4_years  = { value = 4 }
				flag:8_years  = { value = 8 }
				flag:12_years = { value = 12 }
				flag:16_years = { value = 16 }
				flag:20_years = { value = 20 }
				fallback      = { value = 4 }
			}
		}
	}
}

POD_pact_gui_select_stipulation = {

		save_scope_as = pact_stipulation
	}

		name = saved_level
		value = scope:pact_stipulation.var:level
	}
	switch = {
		trigger = scope:pact_stipulation.var:type
		flag:obligation = { 
			set_variable = { name = selected_obligation value = scope:pact_stipulation }

					text = POD_pact_select_detail_obligation_prohibitory_effect

				}
			}

		flag:frequency = {

			set_variable = { name = selected_consequence value = scope:pact_stipulation }
