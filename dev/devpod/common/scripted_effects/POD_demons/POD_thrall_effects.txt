POD_cult_leader_pulse_effect = {
	if = { 
		limit = { var:cult_leader_cooldown ?= 0 } 
		set_variable = {
			name = cult_leader_cooldown
			value = cult_leader_court_position_level
		}
		if = { 
			limit = { 
				any_courtier = { 
					count >= 1 
					NOT = { has_relation_rival = ROOT }

						target = ROOT

					give_infernalist_secret_or_trait_effect = yes
					give_cultist_secret_or_trait_effect = yes
				}
			}
		}
		else_if = {
			limit = {
				any_relation = {
					type = demonic_thrall
					count >= 1

						NOT = { has_relation_nemesis = ROOT }
						NOT = { has_relation_worshipped_demon = ROOT }
						opinion = {
							target = ROOT
							value >= 50
						}

						POD_hostile_mental_interaction_is_shown_custom = {
							ACTOR = ROOT
							RECIPIENT = scope:cultist
						}
					}
				}
			}
			send_interface_message = {
				type = POD_event_demon
				title = POD_set_cultist_interaction
				desc = POD_set_cultist_interaction_effect
				left_icon = scope:cultist

				scope:cultist = {
					set_relation_worshipped_demon = ROOT
					if = { limit = { has_trait = supehunter } ROOT = { save_scope_as = vampiresire } POD_hunter_enthrall_effect = yes }
					give_infernalist_secret_or_trait_effect = yes
					give_cultist_secret_or_trait_effect = yes
				}
			}
		}
		else_if = {
			limit = {
				any_relation = {
					type = demonic_thrall
					count >= 1
					is_landed = yes
					any_vassal = {
						count >= 1 
						NOT = { has_relation_rival = ROOT }
						NOT = { has_relation_nemesis = ROOT }
						NOT = { has_relation_worshipped_demon = ROOT }

							target = ROOT
							value >= 50
						}

					}

			}

				type = POD_event_demon
				title = POD_set_cultist_interaction
				desc = POD_set_cultist_interaction_effect
				left_icon = scope:cultist

				scope:cultist = {
					set_relation_worshipped_demon = ROOT
					if = { limit = { has_trait = supehunter } ROOT = { save_scope_as = vampiresire } POD_hunter_enthrall_effect = yes }

					give_cultist_secret_or_trait_effect = yes
				}
			}
		}

			name = cult_leader_cooldown

		}
	}
	else = {
		set_variable = {
			name = cult_leader_cooldown
			value = cult_leader_court_position_level
		}
	}
}

POD_hunter_enthrall_effect = {
	remove_trait = supehunter
	POD_hunterembrace_effect = yes

		limit = { has_trait = shih }
		POD_become_kueijin_effect = yes

	}
	else_if = {
		limit = { has_POD_race_trigger = no }
		add_trait_force_tooltip = mortal
		if = {
			limit = { can_hold_title_as_mortal_trigger = no }

		}
	}

		modifier = POD_turned_me_opinion
		target = scope:vampiresire
	}
}

give_demonology_secret_or_trait_effect = {
	save_temporary_scope_as = demonologist
	if = {
		limit = { #Not already a demonologist
			NOR = {
				POD_splat_can_be_infernalist = no
				has_trait = demonology
				any_secret = { secret_type = secret_demonology }
			}
		}
		if = {
			limit = {
				trait_is_shunned_or_criminal_in_my_or_lieges_faith_trigger = { TRAIT = demonology GENDER_CHARACTER = scope:demonologist }
			}
			add_secret = {
				type = secret_demonology
			}
			set_variable = {
				name  = secret_demonologist_xp
				value = 0
			}
			set_variable = {
				name  = secret_thrall_xp
				value = 0
			}
			set_variable = {
				name  = secret_diabolist_xp
				value = 0
			}
		}
		else = {
			add_trait = demonology
		}
	}
}

give_infernalist_secret_or_trait_effect = {
	give_demonology_secret_or_trait_effect = yes
	if = {
		limit = {
			POD_has_true_faith_trigger = yes
			POD_splat_mixing_haight_mode = no
		}
		POD_lose_true_faith_effect = yes
		add_stress = massive_stress_impact_gain
		add_trait = depressed_1
	}
	save_temporary_scope_as = infernalist
	if = {
		limit = { #Not already a infernalist
			NOR = {
				POD_splat_can_be_infernalist = no
				has_trait = infernalist
				any_secret = { secret_type = secret_infernalist }
			}
		}
		
		if = {
			limit = {
				trait_is_shunned_or_criminal_in_my_or_lieges_faith_trigger = { TRAIT = infernalist GENDER_CHARACTER = scope:infernalist }
			}
			add_secret = {
				type = secret_infernalist
			}
			# random_secret = {
				# limit = { secret_type = secret_infernalist }
				# save_scope_as = infernalist_secret
			# }
		}
		else = {
			add_trait = infernalist
			if = {
				limit = {
					has_trait = supehunter
					POD_splat_mixing_haight_mode = no
				}
				POD_hunter_enthrall_effect = yes
			}
		}
		
		if = {
			limit = { POD_middle_kingdom_splat_trigger = yes } 
			POD_assing_age_modifier_effect = yes
		}
	}
}

add_demonologist_xp = {
	if = {
		limit = { has_trait = demonology }
		add_trait_xp = {
			trait = demonology
			track = demonologist
			value = $VALUE$
		}
	}
	else_if = {
		limit = {
			any_secret = { secret_type = secret_demonology }
		}
		if = {
			limit = { NOT = { has_variable = secret_demonologist_xp } }
			set_variable = {
				name  = secret_demonologist_xp
				value = 0
			}
		}
		custom_description = {
			text = POD_add_demonologist_xp
			subject = this
			value = $VALUE$
			set_variable = {
				name  = secret_demonologist_xp
				value = {
					value = var:secret_demonologist_xp
					add = $VALUE$
					min = 0
					max = 100
				}
			}
		}
	}
}

POD_yearly_pulse_infernalist_xp_effect = {
	if = {
		limit = {
			NOT = { has_trait = demon }
			OR = {
				is_a_cultist = yes
				is_a_diabolist = yes
			}
		}
		set_local_variable = { name = demonic_investiture_gain value = 0 }
		set_local_variable = { name = diabolist_xp value = 0 }
		set_local_variable = { name = thrall_xp value = 0 }
		every_relation = {
			type = bound_demon
			change_local_variable = { name = diabolist_xp add = 1 }
		}
		every_relation = {
			type = worshipped_demon
			save_temporary_opinion_value_as = { name = target_opinion target = root }
			change_local_variable = { name = demonic_investiture_gain add = scope:target_opinion }
			change_local_variable = { name = thrall_xp add = 1 }
		}
		send_interface_message = {
			type = event_generic_good
			title = POD_investiture_maintenance
			custom_tooltip = POD_investiture_maintenance.tt
			add_investiture_lifestyle_xp = local_var:demonic_investiture_gain
			add_diabolist_xp = { VALUE = local_var:diabolist_xp }
			add_thrall_xp = { VALUE = local_var:thrall_xp }
		}
	}
}

POD_reveal_demon_worship = {
	add_character_flag = obviously_demonic
	every_secret = {
		limit = { secret_type = secret_demonology }
		expose_secret = prev
	}
	every_secret = {
		limit = { secret_type = secret_infernalist }
		expose_secret = prev
	}
	every_secret = {
		limit = { secret_type = secret_cultist }
		expose_secret = prev
	}
}

POD_hide_demon_worship = {
	remove_character_flag = obviously_demonic
	if = {
		limit = { has_trait = infernalist }
		remove_trait = infernalist
		add_secret = { TYPE = secret_infernalist }
	}
	if = {
		limit = { has_trait = cultist }
		remove_trait = cultist
		add_secret = { TYPE = secret_cultist }
	}
}
