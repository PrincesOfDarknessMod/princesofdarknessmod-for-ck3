POD_cult_leader_pulse_effect = {
	save_scope_as = demon
	if = {
		limit = { var:cult_leader_cooldown ?= 0 } 
		set_variable = {
			name = cult_leader_cooldown
			value = cult_leader_court_position_level
		}
		if = {
			limit = {
				any_vassal = {
					POD_can_be_converted_by_cult_leader = { DEMON = scope:demon }
				} 
			}
			ordered_vassal = {
				order_by = "opinion(scope:demon)"
				limit = { POD_can_be_converted_by_cult_leader = { DEMON = scope:demon } }
				save_scope_as = new_cultist
			}
		}
		else_if = {
			limit = {
				any_courtier = {
					POD_can_be_converted_by_cult_leader = { DEMON = scope:demon }
				}
			}
			ordered_courtier = {
				order_by = "opinion(scope:demon)"
				limit = { POD_can_be_converted_by_cult_leader = { DEMON = scope:demon } }
				save_scope_as = new_cultist
			}
		}
		else_if = {
			limit = {
				any_relation = {
					type = demonic_thrall
					is_landed = yes
					any_vassal = {
						count >= 1
						POD_can_be_converted_by_cult_leader = { DEMON = scope:demon }
					}
				}
			}
			ordered_relation = {
				type = demonic_thrall
				order_by = "opinion(scope:demon)"
				limit = { is_landed = yes }
				ordered_vassal = {
					order_by = "opinion(scope:demon)"
					limit = { POD_can_be_converted_by_cult_leader = { DEMON = scope:demon } }
					save_scope_as = new_cultist
				}
			}
		}
		else_if = {
			limit = {
				any_relation = {
					type = demonic_thrall
					is_landed = yes
					any_courtier = {
						POD_can_be_converted_by_cult_leader = { DEMON = scope:demon }
					}
				}
			}
			ordered_relation = {
				type = demonic_thrall
				order_by = "opinion(scope:demon)"
				limit = { is_landed = yes }
				ordered_courtier = {
					order_by = "opinion(scope:demon)"
					limit = { POD_can_be_converted_by_cult_leader = { DEMON = scope:demon } }
					save_scope_as = new_cultist
				}
			}
		}
		if = {
			limit = { exists = scope:new_cultist }
			send_interface_message = {
				type = POD_event_demon
				title = POD_set_cultist_interaction
				desc = POD_set_cultist_interaction_effect
				left_icon = scope:new_cultist

				scope:new_cultist = {
					set_relation_worshipped_demon = scope:demon
					if = {
						limit = { has_trait = supehunter }
						scope:demon = { save_scope_as = vampiresire }
						POD_hunter_enthrall_effect = yes
					}
					give_witch_secret_or_trait_effect = yes
				}
			}
			clear_saved_scope = new_cultist
		}
	}
	else_if = { 
		limit = { has_variable = cult_leader_cooldown }
		change_variable = {
			name = cult_leader_cooldown
			add = -1
		}
	}
	else = {
		set_variable = {
			name = cult_leader_cooldown
			value = {
				value = cult_leader_court_position_level
				subtract = 1 # prevent off-by-one error in first month
			}
		}
	}
}

POD_hunter_enthrall_effect = {
	remove_trait = supehunter
	POD_hunterembrace_effect = yes	
	if = {
		limit = { has_trait = shih }
		POD_become_kueijin_effect = yes
		set_character_faith = faith:wayofthetenthousandscreams
	}
	else_if = {
		limit = { has_POD_race_trigger = no }
		add_trait_force_tooltip = mortal
		if = {
			limit = { can_hold_title_as_mortal_trigger = no }
			POD_depose_effect = yes
		}
	}
	remove_opinion= {
		modifier = POD_turned_me_opinion
		target = scope:vampiresire
	}
}
