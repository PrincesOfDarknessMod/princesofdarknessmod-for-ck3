POD_become_demon_effect = {
	save_temporary_scope_value_as = {
		name = saved_immortal_age
		value = POD_visual_age
	}
	hidden_effect = { POD_remove_splat_traits_effect = yes }
	add_trait_force_tooltip = demon
	if = {
		limit = { exists = scope:saved_immortal_age }
		hidden_effect = { set_immortal_age = scope:saved_immortal_age }
	}
	hidden_effect = { 
		POD_remove_wrong_supernatural_traits_effect = yes
		POD_remove_wrong_supernatural_modifiers_effect = yes
		POD_remove_supernatural_relation_effect = yes
		POD_side_effects_from_changing_splats = yes
		if = {
			limit = { is_pregnant = yes } 
			end_pregnancy = yes
			pregnancy_cleanup_effect = yes
		}
		if = {
			limit = { has_trait = drunkard }
			remove_trait = drunkard
		}
		if = {
			limit = { has_trait = hashishiyah }
			remove_trait = hashishiyah
		}
		if = {
			limit = { has_trait = rakish }
			remove_trait = rakish
		}
		if = {
			limit = { has_trait = flagellant }
			remove_trait = flagellant
		}
		if = {
			limit = { has_trait = comfort_eater }
			remove_trait = comfort_eater
		}
		if = {
			limit = { has_trait = inappetetic }
			remove_trait = inappetetic
		}
		if = {
			limit = { has_trait = athletic }
			remove_trait = athletic
		}
		remove_trait = demonology
	}
			
	POD_demon_initialization_effect = yes
	
	if = {
		limit = { is_ai = no }
		trigger_event = podintroductions.2402
	}
}

		limit = { has_variable = soultaker }
		custom_description = {
			text = POD_store_energy
			subject = $EXECUTIONER$
			value = 10
			set_variable = { name = energy_stored value = 10 }
		}
		if = {
			limit = { $VICTIM$ = { has_trait = demon } }

				chance = var:soultaker_betray

					custom_tooltip = soultaker_betray
					death = {
						death_reason = death_soultaker
						killer = $VICTIM$
					}
				}
			}
			change_variable = { name = soultaker_betray add = { 1 5 } }
		}
		else = {
			add_artifact_modifier = artifact_monthly_lifestyle_xp_1_modifier
			$EXECUTIONER$ = { add_character_modifier = soultaker_stress_gain_modifier }
			# add_artifact_modifier = artifact_monthly_piety_negative_1_modifier
		}
	}
}

	hidden_effect = { POD_remove_resolve_modifiers_effect = yes }
	save_scope_value_as = { name = calc value = { value = $RESOLVE$ max = 10 min = 0 } }
	switch = {
		trigger = scope:calc
		0 = { add_character_modifier = resolve0_modifier }
		1 = { add_character_modifier = resolve1_modifier }
		2 = { add_character_modifier = resolve2_modifier }
		3 = { add_character_modifier = resolve3_modifier }
		4 = { add_character_modifier = resolve4_modifier }

		6 = { add_character_modifier = resolve6_modifier }
		7 = { add_character_modifier = resolve7_modifier }
		8 = { add_character_modifier = resolve8_modifier }
		9 = { add_character_modifier = resolve9_modifier }

POD_resolve_change_effect = {
	save_scope_value_as = { name = resolve_calc value = { value = resolve_level add = $COUNT$ } }
	POD_set_resolve_to_effect = { RESOLVE = scope:resolve_calc }
}

POD_resolve_cost_effect = {
	if = {
		limit = {
			any_character_artifact = {
				has_variable = energy_stored
				var:energy_stored > 0
			}
		}
		ordered_character_artifact = {
			limit = {
				has_variable = energy_stored 
				var:energy_stored > 0
			}
			order_by = {
				value = 0
				add = var:energy_stored
			}
			position = 0
			POD_energy_vessel_decrease_multiple_effect = { COUNT = $COUNT$ }
		}
	}
	else = {
		POD_resolve_change_effect = { COUNT = -$COUNT$ }
	}
}

POD_resolve_income_effect = {
	if = {
		limit = {
			OR = {
				resolve_difference >= $COUNT$
				NOT = {
					any_character_artifact = {
						NOT = { has_variable = temporary_vessel }
						has_variable = energy_stored
						vessel_empty_energy > 0
					}
				}
			}
		}
		POD_resolve_change_effect = { COUNT = $COUNT$ }
	}
	else = {
		save_scope_value_as = {
			name = excess_resolve
			value = {
				value = $COUNT$
				subtract = resolve_difference
			}
		}
		POD_resolve_change_effect = { COUNT = resolve_difference }
		ordered_character_artifact = {
			limit = {
				NOT = { has_variable = temporary_vessel }
				has_variable = energy_stored 
				vessel_empty_energy > 0
			}
			order_by = {
				value = 0
				add = vessel_empty_energy
			}
			position = 0
			POD_energy_vessel_increase_multiple_effect = { COUNT = scope:excess_resolve }
		}
	}
}

POD_arcana_perk_check_effect = {
	save_scope_value_as = { name = calc value = { value = $LEVEL$ subtract = 1 multiply = -1 } }
	if = { 
		limit = { NOT = { POD_have_enough_arcana_perk_points = { LEVEL = $LEVEL$ } } } 
		custom_description_no_bullet = { text = POD_arcana_perk_check_effect_perk value = $LEVEL$ }
		hidden_effect = { add_arcana_lifestyle_perk_points = scope:calc }
	}
	else = {
		custom_description_no_bullet = { 
			text = POD_arcana_perk_check_effect_perk_full 
			value = $LEVEL$
			add_arcana_lifestyle_perk_points = scope:calc
		}
	}
}
