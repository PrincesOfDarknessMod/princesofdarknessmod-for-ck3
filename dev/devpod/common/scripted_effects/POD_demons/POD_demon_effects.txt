POD_set_resolve_to_effect = {
	hidden_effect = { POD_remove_resolve_modifiers_effect = yes }

}

			add_arcana_lifestyle_perk_points = scope:calc
		}
	}
}

POD_landless_demon_gain_knowledge = {
	if = {
		limit = {
			POD_splat_can_gain_demon_knowledge = yes
			POD_landless_demon_clues < POD_landless_demon_clues_for_3rd_unlock
		}
		
		# for the custom_tooltip
		save_scope_value_as = {
			name = landless_demon_clues_gained
			value = $COUNT$
		}
		custom_tooltip = {
			text = landless_demon_gain_clues_tt
			set_variable = {
				name = landless_demon_clues
				value = {
					value = POD_landless_demon_clues
					add = scope:landless_demon_clues_gained
					min = 0
					max = POD_landless_demon_clues_for_3rd_unlock
				}
			}
		}
		if = {
			limit = { POD_landless_demon_clues >= POD_landless_demon_clues_for_1st_unlock }
			trigger_event = POD_demon_flavor.6
		}
		if = {
			limit = { POD_landless_demon_clues >= POD_landless_demon_clues_for_2nd_unlock }
			trigger_event = POD_demon_flavor.7
		}
		if = {
			limit = { POD_landless_demon_clues >= POD_landless_demon_clues_for_3rd_unlock }
			trigger_event = POD_demon_flavor.8
		}
	}
}
