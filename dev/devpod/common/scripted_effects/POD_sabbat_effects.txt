	religion:sabbat_religion = { add_to_variable_list = { name = cardinals target = prev } }

}

	religion:sabbat_religion = { remove_list_variable = { name = cardinals target = prev } }
}

	religion:sabbat_religion = { remove_list_variable = { name = cardinals target = prev } }
}
