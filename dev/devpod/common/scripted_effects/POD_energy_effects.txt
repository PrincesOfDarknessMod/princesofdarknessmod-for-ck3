##### Naming scheme:

# (energy)_gain: Gain energy for your "mana bar" only
# (energy)_loss: Lose energy from your "mana bar" only
# (energy)_assign_value: Set your "mana bar" to a specific value
# (energy)_refill: Gain energy for your "mana bar", with side effects if you gain more energy than your "mana bar" can store
# (energy)_drain: Lose energy from your "mana bar", with side effects if you lose more energy than you currently have
# (energy)_income: Gain energy for your "mana bar", the excess gets stored in energy vessels, further excess causes side effects
# (energy)_cost: Lose energy from energy vessels first, then spend any remainder from your "mana bar", further spending causes side effects

# For 99% of cases, (energy)_income and (energy)_cost is all you need.


##### Priority list:

# 1. Hunger (Blood)
# 2. Angst
# 3. Pathos
# 4. Resolve
# 5. Gnosis
# 6. Glamour
# 7. Dread
# 8. Fallback (static_stress)

# A character will use the highest-priority type of energy whose requirements they fulfill.

# If you change that priority list (or add new types of energy), also change all the places where it's used:
# - In this file
#   - POD_splat_agnostic_energy_gain
#   - POD_splat_agnostic_energy_loss
#   - POD_splat_agnostic_energy_assign_value
#   - POD_fully_refill_energy
#   - POD_initialize_energy_modifier
# - In common\scripted_triggers\POD_energy_triggers.txt
#   - POD_has_enough_energy
#	- also, all the priority triggers (e.g. POD_splat_has_hunger_exclusively)
# - In common\script_values\POD_energy_values.txt
#   - maximum_splat_energy
#   - maximum_splat_energy_reserves
#   - splat_energy_points_value
# - In common\customizable_localization\00_POD_misc_custom_loc.txt
#   - PODEnergySplat

POD_scale_energy_count_to = {
	save_scope_value_as = {
		name = $SAVE_AS$
		value = {
			value = $COUNT$
			multiply = $SPLAT_MAXIMUM_ENERGY$
			divide = maximum_splat_agnostic_energy
			ceiling = yes
		}
	}
}

# Gain energy for your "mana bar", with side effects if you gain more energy than your "mana bar" can store.
# Uses splat agnostic values (COUNT from 0-10)
POD_splat_agnostic_energy_refill = {
	save_scope_value_as = {
		name = capped_energy_gain
		value = {
			value = $COUNT$
			max = splat_agnostic_energy_deficit
		}
	}
	save_scope_value_as = {
		name = energy_overflow
		value = {
			value = $COUNT$
			subtract = scope:capped_energy_gain
		}
	}
	POD_splat_agnostic_energy_gain = { COUNT = scope:capped_energy_gain }
	if = {
		limit = { scope:energy_overflow > 0 }
		POD_side_effects_from_splat_agnostic_energy_overflow = {
			COUNT = scope:energy_overflow
		}
	}
}

# Lose energy from your "mana bar", with side effects if you lose more energy than you currently have.
# Uses splat agnostic values (COUNT from 0-10)
POD_splat_agnostic_energy_drain = {
	save_scope_value_as = {
		name = capped_energy_loss
		value = {
			value = $COUNT$
			max = splat_agnostic_energy_points_value
		}
	}
	save_scope_value_as = {
		name = energy_overdraw
		value = {
			value = $COUNT$
			subtract = scope:capped_energy_loss
		}
	}
	POD_splat_agnostic_energy_loss = { COUNT = scope:capped_energy_loss }
	if = {
		limit = { scope:energy_overdraw > 0 }
		POD_side_effects_from_splat_agnostic_energy_overdraw = {
			COUNT = scope:energy_overdraw
		}
	}
}

# What happens if you gain more energy than you can store?
POD_side_effects_from_splat_agnostic_energy_overflow = {
	# Only Glamour has side effects for now
	if = {
		limit = {
			POD_splat_has_glamour_exclusively = yes
		}
		add_stress = {
			value = $COUNT$
			# 5 stress loss per splat-agnostic energy point (so, 10 per glamour point)
			multiply = -5
		}
	}
}

# What happens if you spend more energy than you have?
POD_side_effects_from_splat_agnostic_energy_overdraw = {
	POD_increase_wounds_effect = {
		COUNT  = $COUNT$
		REASON = drained_of_quintessence
	}
}

POD_splat_agnostic_energy_drain_hostile = {
	# scope:attacker is used in POD_increase_wounds_effect -> POD_health_death_check_effect
	$ATTACKER$ = { save_scope_as = attacker }
	POD_splat_agnostic_energy_drain = { COUNT = $COUNT$ }
}

# Gain energy for your "mana bar". The excess gets stored in energy vessels, further excess causes side effects
POD_energy_income = {
	save_scope_value_as = {
		name = splat_agnostic_energy_income
		value = {
			value = $COUNT$
			multiply = maximum_splat_agnostic_energy
			divide = $SPLAT_MAXIMUM_ENERGY$
		}
	}
	save_scope_value_as = {
		name = splat_agnostic_vessel_income
		value = {
			value = scope:splat_agnostic_energy_income
			subtract = {
				value = scope:splat_agnostic_energy_income
				max = splat_agnostic_energy_deficit
			}
			max = total_storable_vessel_empty_energy
		}
	}
	save_scope_value_as = {
		name = splat_agnostic_energy_refill
		value = {
			value = scope:splat_agnostic_energy_income
			subtract = scope:splat_agnostic_vessel_income
		}
	}
	if = {
		limit = { scope:splat_agnostic_energy_refill > 0 }
		POD_splat_agnostic_energy_refill = {
			COUNT = scope:splat_agnostic_energy_refill
		}
	}
	if = {
		limit = { scope:splat_agnostic_vessel_income > 0 }
		POD_store_splat_agnostic_energy_in_vessels = {
			AMOUNT = scope:splat_agnostic_vessel_income
		}
	}
}

# Lose energy from energy vessels first, then spend any remainder from your "mana bar". Further spending causes side effects
POD_energy_cost = {
	save_scope_value_as = {
		name = splat_agnostic_energy_cost
		value = {
			value = $COUNT$
			multiply = maximum_splat_agnostic_energy
			divide = $SPLAT_MAXIMUM_ENERGY$
		}
	}
	save_scope_value_as = {
		name = splat_agnostic_vessel_cost
		value = {
			value = total_usable_vessel_energy
			max = scope:splat_agnostic_energy_cost
		}
	}
	save_scope_value_as = {
		name = splat_agnostic_energy_drain
		value = {
			value = scope:splat_agnostic_energy_cost
			subtract = scope:splat_agnostic_vessel_cost
		}
	}
	if = {
		limit = { scope:splat_agnostic_vessel_cost > 0 }
		POD_spend_splat_agnostic_energy_from_vessels = {
			AMOUNT = scope:splat_agnostic_vessel_cost
		}
	}
	if = {
		limit = { scope:splat_agnostic_energy_drain > 0 }
		POD_splat_agnostic_energy_drain = {
			COUNT = scope:splat_agnostic_energy_drain
		}
	}
}

POD_splat_agnostic_energy_income = {
	POD_energy_income = {
		COUNT = $COUNT$
		SPLAT_MAXIMUM_ENERGY = maximum_splat_agnostic_energy
	}
}

POD_splat_agnostic_energy_cost = {
	POD_energy_cost = {
		COUNT = $COUNT$
		SPLAT_MAXIMUM_ENERGY = maximum_splat_agnostic_energy
	}
}

POD_suck_splat_agnostic_energy_from = {
	$TARGET$ = {
		save_scope_value_as = {
			name = remaining_energy
			value = {
				value = $COUNT$
				max = splat_agnostic_energy_points_value
			}
		}
		POD_splat_agnostic_energy_drain_hostile = {
			ATTACKER = prev
			COUNT = scope:remaining_energy
		}
	}
	POD_splat_agnostic_energy_income = { COUNT = scope:remaining_energy }
}

POD_spend_splat_agnostic_energy_from_vessels = {
	save_scope_value_as = {
		name = amount_to_spend
		value = $AMOUNT$
	}
	# make sure not to count "half" energy points
	# (e.g. 1 energy point would be 0.5 blood for a vampire)
	save_scope_value_as = {
		name = minimum_vessel_energy
		value = POD_full_energy_point_for_vessel
	}
	ordered_character_artifact = {
		max = 100
		check_range_bounds = no
		limit = {
			has_variable = energy_stored
			prev = { can_benefit_from_artifact = prev }
			# only artifacts that have energy left
			var:energy_stored >= scope:minimum_vessel_energy
		}
		order_by = {
			value = var:energy_stored
			if = {
				limit = { has_variable = temporary_vessel }
				add = -1000 # use temporary vessels last
			}
		}
		if = {
			limit = { scope:amount_to_spend > 0 }
			save_scope_value_as = {
				name = current_vessel_deficit
				value = {
					value = scope:amount_to_spend
					# "half" energy points get cleared from vessels
					# so no modulo here
					max = var:energy_stored
				}
			}
			prev = {
				POD_spend_splat_agnostic_energy_from_specific_vessel = {
					ARTIFACT = prev
					AMOUNT   = scope:current_vessel_deficit
				}
			}
			save_scope_value_as = {
				name = amount_to_spend
				value = {
					value = scope:amount_to_spend
					subtract = {
						value = scope:current_vessel_deficit
						# "half" energy points don't count against the full amount you need to spend
						subtract = {
							value = scope:current_vessel_deficit
							modulo = scope:minimum_vessel_energy
						}
					}
				}
			}
		}
	}
}

POD_spend_splat_agnostic_energy_from_specific_vessel = {
	# save values for tooltip
	save_scope_as = this_character
	$ARTIFACT$ = { save_scope_as = this_artifact }
	save_scope_value_as = {
		name  = spent_energy_amount
		value = $AMOUNT$
	}
	custom_tooltip = POD_extract_energy_desc
	
	$ARTIFACT$ = {
		if = {
			limit = {
				has_variable = temporary_vessel
				$AMOUNT$ >= var:energy_stored
			}
			POD_destroy_artifact_side_effect = yes
			destroy_artifact = this
		}
		else = {
			set_variable = {
				name = energy_stored
				value = {
					value = var:energy_stored
					subtract = $AMOUNT$
					min = 0
				}
			}
			hidden_effect = {
				POD_energy_vessel_calc_modifiers_effect = yes
			}
		}
	}
}


POD_store_splat_agnostic_energy_in_vessels = {
	save_scope_value_as = {
		name = amount_to_store
		value = $AMOUNT$
	}
	ordered_character_artifact = {
		max = 100
		check_range_bounds = no
		limit = {
			has_variable = energy_stored
			NOT = { has_variable = temporary_vessel }
			prev = { can_benefit_from_artifact = prev }
			# only artifacts that have energy left
			storable_vessel_empty_energy > 0
		}
		order_by = vessel_empty_energy
		if = {
			limit = {
				scope:amount_to_store > 0
				storable_vessel_empty_energy > 0
			}
			save_scope_value_as = {
				name = current_vessel_excess
				value = {
					value = scope:amount_to_store
					# "half" energy points in vessels get filled
					max = storable_vessel_empty_energy
				}
			}
			prev = {
				POD_store_splat_agnostic_energy_in_specific_vessel = {
					ARTIFACT = prev
					AMOUNT   = scope:current_vessel_excess
				}
			}
			save_scope_value_as = {
				name = amount_to_store
				value = {
					value = scope:amount_to_store
					subtract = scope:current_vessel_excess
				}
			}
		}
	}
}

POD_store_splat_agnostic_energy_in_specific_vessel = {
	# save values for tooltip
	save_scope_as = this_character
	$ARTIFACT$ = {
		save_scope_as = this_artifact
		save_scope_value_as = {
			name  = stored_energy_amount
			value = {
				value = $AMOUNT$
				max = storable_vessel_empty_energy
			}
		}
	}
	custom_tooltip = POD_store_energy_desc
	
	$ARTIFACT$ = {
		set_variable = {
			name = energy_stored
			value = {
				value = var:energy_stored
				add = $AMOUNT$
				max = var:max_energy_stored
			}
		}
		hidden_effect = {
			POD_energy_vessel_calc_modifiers_effect = yes
		}
	}
}

POD_energy_vessel_calc_modifiers_effect = {
	if = {
		limit = { has_artifact_modifier = energy_vessel_empty_modifier }
		remove_artifact_modifier = energy_vessel_empty_modifier
	}
	while = {
		limit = { has_artifact_modifier = energy_vessel_10percent_modifier }
		remove_artifact_modifier = energy_vessel_10percent_modifier
	}
	while = {
		limit = { has_artifact_modifier = energy_vessel_1percent_modifier }
		remove_artifact_modifier = energy_vessel_1percent_modifier
	}
	if = {
		limit = {
			has_variable = energy_stored
			has_variable = max_energy_stored
		}
		save_scope_value_as = {
			name  = percentage_filled
			value = {
				value = var:energy_stored
				divide = var:max_energy_stored
				multiply = 100
				floor = yes
			}
		}
		save_scope_value_as = {
			name  = 10percent_stacks
			value = {
				value = scope:percentage_filled
				divide = 10
				floor = yes
			}
		}
		save_scope_value_as = {
			name  = 1percent_stacks
			value = {
				value = scope:percentage_filled
				modulo = 10
			}
		}
		
		add_artifact_modifier = energy_vessel_empty_modifier
		while = {
			count = scope:10percent_stacks
			add_artifact_modifier = energy_vessel_10percent_modifier
		}
		while = {
			count = scope:1percent_stacks
			add_artifact_modifier = energy_vessel_1percent_modifier
		}
	}
}

POD_energy_vessel_add_full_energy_bar_as_max = {
	POD_energy_vessel_add_max_effect = {
		COUNT = maximum_splat_agnostic_energy
	}
}

POD_energy_vessel_add_max_effect = {
	if = {
		limit = { NOT = { has_variable = max_energy_stored } }
		set_variable = {
			name = max_energy_stored
			value = 0
		}
	}
	if = {
		limit = { NOT = { has_variable = energy_stored } }
		set_variable = {
			name = energy_stored
			value = 0
		}
	}
	# safeguard to make sure that no "half" energy points ever get added
	save_scope_value_as = {
		name  = added_energy
		value = {
			value = $COUNT$
			add = {
				value = $COUNT$
				modulo = 2
			}
		}
	}
	custom_description = {
		text = POD_energy_vessel_add_max_effect
		value = scope:added_energy
		change_variable = {
			name = max_energy_stored
			add = scope:added_energy
		}
		POD_energy_vessel_calc_modifiers_effect = yes
	}
}

POD_temporary_energy_vessel_add_max_effect = {

	set_variable = { name = max_energy_stored   value = $COUNT$ }
	set_variable = { name = energy_stored       value = $COUNT$ }
	custom_description = {
		text = POD_energy_vessel_add_max_effect
		value = $COUNT$
		POD_energy_vessel_calc_modifiers_effect = yes
	}
