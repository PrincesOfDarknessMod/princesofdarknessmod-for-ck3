POD_umbra_send_effect = {
	if = {
		limit = { is_ai = yes }
		set_variable = { name = spirit_location value = $LOCATION$ }
	}
	else = {
		POD_umbra_location_initiate_effect = { LOCATION = $LOCATION$ }
	}
}

POD_return_perk_effect = {
	if = { 
		limit = { 
			capital_province ?= {
				OR = {
					geographical_region = world_europe
					geographical_region = world_asia_minor
				}
			}
		}
		add_to_variable_list = { name = underworld_realms_list target = flag:dark_kingdom_of_iron }
	}
	else_if = { 
		limit = { capital_province ?= { geographical_region = world_africa } }
		add_to_variable_list = { name = underworld_realms_list target = flag:dark_kingdom_of_ivory }
	}
	else_if = { 
		limit = { 
			capital_province ?= {
				OR = {
					geographical_region = world_tibet
					geographical_region = world_steppe
				}
			}
		}
		add_to_variable_list = { name = underworld_realms_list target = flag:dark_kingdom_of_jade }
	}
	else_if = { 
		limit = { capital_province ?= { geographical_region = world_india } }
		add_to_variable_list = { name = underworld_realms_list target = flag:swar }
	}
	else_if = { 
		limit = { capital_province ?= { geographical_region = world_burma } }
		add_to_variable_list = { name = underworld_realms_list target = flag:night_markets }
	}
	else_if = { 
		limit = { 
			capital_province ?= {
				OR = {
					geographical_region = world_middle_east
					geographical_region = world_middle_east_persia
				}
			} 
		}
		add_to_variable_list = { name = underworld_realms_list target = flag:mantaqat_khayal }
	}
	else = { 
		add_to_variable_list = { name = underworld_realms_list target = flag:dark_kingdom_of_clay }
		add_to_variable_list = { name = underworld_realms_list target = flag:dark_kingdom_of_obsidian }
	}
}
