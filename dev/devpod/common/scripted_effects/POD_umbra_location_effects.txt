POD_umbra_send_effect = {
	if = {
		limit = { has_variable = spirit_location } 
		POD_change_umbra_modifier_effect = { START = var:spirit_location END = $LOCATION$ }
	}
	else = {
		POD_initiate_umbra_modifier_effect = { LOCATION = $LOCATION$ }
	}
	
	set_variable = { name = spirit_location value = $LOCATION$ }
	
	if = {
		limit = { 
			has_variable = spirit_destination 
		}
		remove_variable = spirit_destination
	}
}

POD_return_perk_effect = {
	if = { 
		limit = { 
			capital_province ?= {
				OR = {
					geographical_region = world_europe
					geographical_region = world_asia_minor
				}
			}
		}
		POD_create_umbra_realm_effect = { NAME = flag:dark_kingdom_of_iron }
	}
	else_if = { 
		limit = { capital_province ?= { geographical_region = world_africa } }
		POD_create_umbra_realm_effect = { NAME = flag:dark_kingdom_of_ivory }
	}
	else_if = { 
		limit = { 
			capital_province ?= {
				OR = {
					geographical_region = world_tibet
					geographical_region = world_steppe
				}
			}
		}
		POD_create_umbra_realm_effect = { NAME = flag:dark_kingdom_of_jade }
	}
	else_if = { 
		limit = { capital_province ?= { geographical_region = world_india } }
		POD_create_umbra_realm_effect = { NAME = flag:swar }
	}
	else_if = { 
		limit = { capital_province ?= { geographical_region = world_burma } }
		POD_create_umbra_realm_effect = { NAME = flag:night_markets }
	}
	else_if = { 
		limit = { 
			capital_province ?= {
				OR = {
					geographical_region = world_middle_east
					geographical_region = world_middle_east_persia
				}
			} 
		}
		POD_create_umbra_realm_effect = { NAME = flag:mantaqat_khayal }
	}
	else = { 
		POD_create_umbra_realm_effect = { NAME = flag:dark_kingdom_of_clay }
		POD_create_umbra_realm_effect = { NAME = flag:dark_kingdom_of_obsidian }
	}
}
