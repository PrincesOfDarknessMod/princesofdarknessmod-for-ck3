
POD_adopt_correct_government_effect = {
	if = {
		limit = { POD_splat_is_valid_for_litany_government = yes }
		POD_set_litany_government_effect = yes
	}
	else_if = {
		limit = { POD_splat_is_valid_for_infernal_court_government = yes }
		change_government = infernal_court_government
	}
	else_if = {
		limit = { POD_splat_is_valid_for_cult_government = yes }
		change_government = cult_government
	}
	else_if = {
		limit = { POD_splat_is_valid_for_inquisition_government = yes }
		change_government = inquisition_government
	}
	else_if = {
		limit = { POD_splat_is_valid_for_cell_government = yes }
		change_government = cell_government
	}
	else_if = {
		limit = { POD_splat_is_valid_for_traditions_government = yes }
		change_government = traditions_government
	}
	else_if = {
		limit = { POD_splat_is_valid_for_august_court_government = yes }
		change_government = august_court_government
	}
	else_if = {
		limit = { POD_splat_is_valid_for_household_government = yes }
		change_government = household_government
	}
	else_if = {
		limit = { POD_splat_is_valid_for_dark_kingdom_government = yes }
		change_government = dark_kingdom_government
	}
	# restore buildings from domicile
	POD_restore_saved_haven_buildings = yes
}

POD_set_litany_government_effect = {
	change_government = litany_government
	save_temporary_scope_as = new_owner
	every_held_title = {
		limit = { tier <= 2 }
		title_province = {
			if = {
				limit = { 
					NOT = { has_holding_type = tribal_holding }
					province_owner = scope:new_owner
				}
				set_local_variable = {
					name = pod_building_level
					value = combined_building_level
					days = 1
				}
				set_holding_type = tribal_holding
				while = {
					limit = { local_var:pod_building_level > 1 }
					generate_building = yes
					change_local_variable = {
						name = pod_building_level
						subtract = 1
					}
				}
				remove_local_variable = pod_building_level
			}
		}
	}
}
