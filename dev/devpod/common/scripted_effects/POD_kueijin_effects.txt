POD_enlightenment_duel_effect = {
	duel = {
		skill = $SKILL$
		value = $VALUE$
		10 = {
			compare_modifier = { value = scope:duel_value multiplier = 3.5 min = -9 }
			modifier = {
				has_variable = additional_enlightenment_chance
				add = var:additional_enlightenment_chance
			}
			show_as_tooltip = { POD_advance_dharma_effect = yes }
			trigger_event = pod1230objectives_kj.2
			POD_change_wheel_of_ages_progress = { VALUE = -50 }
			remove_variable = additional_enlightenment_chance
		}
		50 = {
			compare_modifier = { value = scope:duel_value multiplier = -3.5 min = -49 }
			show_as_tooltip = { 
				add_trait_xp = { trait = dharma9 value = -50 } 
				POD_static_stress_gain = { COUNT = 150 }
			}
			if = { limit = { has_variable = additional_enlightenment_chance } change_variable = { name = additional_enlightenment_chance add = 10 } }
			else = { set_variable = { name = additional_enlightenment_chance value = 10 } }
			trigger_event = pod1230objectives_kj.3
		}
	}
}
POD_change_wheel_of_ages_progress = {
	if = { limit = { NOT = { exists = global_var:wheel_of_ages } } set_global_variable = { name = wheel_of_ages value = 4 } }
	custom_description = {
		text = POD_change_wheel_of_ages_progress
		value = $VALUE$
		change_global_variable = {
			name = wheel_of_ages_progress
			add = $VALUE$
		}
		if = { limit = { wheel_of_ages_progress >= 100 } POD_change_wheel_of_ages = { VALUE = 1 } }
		if = { limit = { wheel_of_ages_progress <= 0 } POD_change_wheel_of_ages = { VALUE = -1 } }
	}
}
POD_change_wheel_of_ages = {
	set_global_variable = {
		name = wheel_of_ages_progress
		value = 50
	}
	change_global_variable = {
		name = wheel_of_ages
		add = $VALUE$
	}
	every_living_character = {
		limit = { POD_middle_kingdom_splat_trigger = yes }
		switch = {
			trigger = wheel_of_ages
			2 = { trigger_event = POD_age.1 }
			3 = { trigger_event = POD_age.2 }
			4 = { trigger_event = POD_age.3 }
			5 = { trigger_event = POD_age.4 }
			6 = { trigger_event = POD_age.5 }
		}
	}
}
POD_assing_age_modifier_effect = {
	hidden_effect = {
		if = { limit = { has_character_modifier = joyous_age_modifier } remove_character_modifier = joyous_age_modifier }
		if = { limit = { has_character_modifier = godbodies_modifier } remove_character_modifier = godbodies_modifier }
		if = { limit = { has_character_modifier = akuma_debuff_modifier } remove_character_modifier = akuma_debuff_modifier }
		if = { limit = { has_character_modifier = akuma_buff_modifier } remove_character_modifier = akuma_buff_modifier }
		if = { limit = { has_character_modifier = akuma_huge_buff_modifier } remove_character_modifier = akuma_huge_buff_modifier }
		if = { limit = { has_character_modifier = yama_debuff_modifier } remove_character_modifier = yama_debuff_modifier }
		if = { limit = { has_character_modifier = yama_buff_modifier } remove_character_modifier = yama_buff_modifier }
		if = { limit = { has_character_modifier = yama_huge_buff_modifier } remove_character_modifier = yama_huge_buff_modifier }
		if = { limit = { has_character_modifier = kj_harder_dharma_modifier } remove_character_modifier = kj_harder_dharma_modifier }
		if = { limit = { has_character_modifier = kj_imposible_dharma_modifier } remove_character_modifier = kj_imposible_dharma_modifier }
	}
	switch = {
		trigger = wheel_of_ages
		3 = {
			if = { limit = { is_akuma_trigger = yes } 
				add_character_modifier = akuma_debuff_modifier
			}
			if = { limit = { has_character_modifier = yama_king_modifier } 
				add_character_modifier = yama_debuff_modifier
			}
			if = { limit = { has_trait = dharma } 
				add_character_modifier = godbodies_modifier
			}
			add_character_modifier = joyous_age_modifier
		}
		4 = {
			if = { limit = { has_trait = dharma } 
				add_character_modifier = godbodies_modifier
			}
		}
		5 = {
			if = { limit = { is_akuma_trigger = yes } 
				add_character_modifier = akuma_buff_modifier
			}
			else = {
				add_character_modifier = kj_harder_dharma_modifier
			}
			if = { limit = { has_character_modifier = yama_king_modifier } 
				add_character_modifier = yama_buff_modifier
			}
		}
		6 = {
			if = { limit = { is_akuma_trigger = yes } 
				add_character_modifier = akuma_huge_buff_modifier
				add_character_modifier = kj_harder_dharma_modifier
			}
			else = {
				add_character_modifier = kj_imposible_dharma_modifier
			}
			if = { limit = { has_character_modifier = yama_king_modifier } 
				add_character_modifier = yama_huge_buff_modifier
			}
		}
	}
