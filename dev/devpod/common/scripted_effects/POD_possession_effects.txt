POD_wraith_possession_remove_atribute_gain_effect = {
	if = {
		limit = { has_character_modifier = wraith_possession_diplomacy_gain_modifier }
		while = {
			limit = { has_character_modifier = wraith_possession_diplomacy_gain_modifier }
			remove_character_modifier = wraith_possession_diplomacy_gain_modifier
		}
	}
	if = {
		limit = { has_character_modifier = wraith_possession_martial_gain_modifier }
		while = {
			limit = { has_character_modifier = wraith_possession_martial_gain_modifier }
			remove_character_modifier = wraith_possession_martial_gain_modifier
		}
	}
	if = {
		limit = { has_character_modifier = wraith_possession_stewardship_gain_modifier }
		while = {
			limit = { has_character_modifier = wraith_possession_stewardship_gain_modifier }
			remove_character_modifier = wraith_possession_stewardship_gain_modifier
		}
	}
	if = {
		limit = { has_character_modifier = wraith_possession_intrigue_gain_modifier }
		while = {
			limit = { has_character_modifier = wraith_possession_intrigue_gain_modifier }
			remove_character_modifier = wraith_possession_intrigue_gain_modifier
		}
	}
	if = {
		limit = { has_character_modifier = wraith_possession_learning_gain_modifier }

			limit = { has_character_modifier = wraith_possession_learning_gain_modifier }
			remove_character_modifier = wraith_possession_learning_gain_modifier

POD_wraith_complete_possession_effect = {
	$PUPPET$ = {
		POD_wraith_possession_remove_atribute_gain_effect = yes
	}
	$WRAITH$ = {

		add_hook = {
			target = $PUPPET$
			type = possessed_hook
		}
		reverse_add_opinion = {
			modifier = POD_complete_possession_opinion
			target = $PUPPET$
		}
	}
	$WRAITH$ = {
		hidden_effect = {
			POD_get_base_prowess_effect = yes
			POD_get_base_martial_effect = yes
			POD_get_base_intrigue_effect = yes
			POD_get_base_learning_effect = yes
			POD_get_base_diplomacy_effect = yes
			POD_get_base_stewardship_effect = yes
		}
	}
	$PUPPET$ = {
		while = {
			count = $WRAITH$.half_diplomacy
			add_character_modifier = wraith_possession_diplomacy_gain_modifier
		}
		while = {
			count = $WRAITH$.half_stewardship
			add_character_modifier = wraith_possession_stewardship_gain_modifier
		}
		while = {
			count = $WRAITH$.half_martial
			add_character_modifier = wraith_possession_martial_gain_modifier
		}
		while = {
			count = $WRAITH$.half_intrigue
			add_character_modifier = wraith_possession_intrigue_gain_modifier
		}
		while = {
			count = $WRAITH$.half_learning
			add_character_modifier = wraith_possession_learning_gain_modifier
		}
		while = {
			count = $WRAITH$.half_prowess
			add_character_modifier = wraith_possession_prowess_gain_modifier
		}

		POD_remove_personality_traits_effect = yes
		POD_copy_personality_effect = { TARGET = $WRAITH$ }
	}
}

POD_wraith_increase_possession = {
	if = {
		limit = { NOT = { $PUPPET$ = { has_relation_possessesing_wraith = $WRAITH$ } } }
		$PUPPET$ = { set_relation_possessesing_wraith = $WRAITH$ }
		POD_wraith_weak_possession_effect = { WRAITH = $WRAITH$ PUPPET = $PUPPET$ }
		$WRAITH$ = { add_character_modifier = wraith_possesing_modifier }
	}
	else_if = {
		limit = {
			$WRAITH$ = {
				has_hook_of_type  = {
					target = $PUPPET$
					type = possessed_weak_hook
				}
			}
		}
		POD_wraith_possession_effect = { WRAITH = $WRAITH$ PUPPET = $PUPPET$}

	}
	else = {
		POD_wraith_complete_possession_effect = { WRAITH = $WRAITH$ PUPPET = $PUPPET$}
	}
}

POD_wraith_end_possession = {
	$WRAITH$ = { save_temporary_scope_as = wraith }
	$PUPPET$ = { save_temporary_scope_as = puppet }

	scope:puppet = {
		POD_wraith_possession_remove_atribute_gain_effect = yes
	}

	scope:wraith = {
		scope:puppet = {
			remove_opinion = {
				target = prev
				modifier = POD_weak_possession_opinion

			}
			remove_opinion = {
				target = prev
				modifier = POD_possession_opinion

			}
			remove_opinion = {
				target = prev
				modifier = POD_complete_possession_opinion

			}
			remove_relation_possessesing_wraith = prev
		}
		remove_hook = {
			type = possessed_hook
			target = scope:puppet
		}
		remove_hook = {
			type = possessed_weak_hook
			target = scope:puppet
		}
		remove_character_modifier = wraith_possesing_modifier
	}
}
