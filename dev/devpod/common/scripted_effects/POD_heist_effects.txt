
POD_heist_setup_effect = {
	hidden_effect = {
		$COURT$ = { save_scope_as = court_owner }
		create_story = {
			type = POD_story_heist
			save_scope_as = $SAVE_AS$
		}
		hidden_effect_new_object = {
			scope:$SAVE_AS$ = {
				POD_heist_select_court_owner = { COURT_CHAR = scope:court_owner }

				$MAPGEN$ = yes
				
				random_in_list = {
					variable = nodes
					limit = { POD_heist_is_node_potential_start = yes }
					prev = {
						set_variable = {
							name  = current_location
							value = prev
						}
					}
				}

				POD_save_event_as_var = {
					VAR = ending_event
					ID  = $ENDING$
				}

				if = {
					limit = { prev = { POD_has_scouted_heist_trigger = yes } }
					every_in_list = {
						variable = nodes
						set_variable = {
							name  = explored
							value = yes
						}
					}
				}
				
				POD_heist_refresh = yes
			}
			
			set_variable = {
				name  = current_heist
				value = scope:$SAVE_AS$
			}
		}
	}
}

POD_heist_select_court_owner = {
	if = {
		limit = {
			$COURT_CHAR$ = {
				is_playable_character = yes
				highest_held_title_tier >= tier_county
			}
		}
		set_variable = {
			name  = court_owner
			value = $COURT_CHAR$
		}
	}
	else_if = {
		limit = {
			$COURT_CHAR$ = {
				exists = liege_or_court_owner
				NOT = { liege_or_court_owner = prev.story_owner } # don't heist yourself if targeting your courtiers
			}
		}
		set_variable = {
			name  = court_owner
			value = $COURT_CHAR$.liege_or_court_owner
		}
	}
	else = {
		set_variable = {
			name  = court_owner
			value = $COURT_CHAR$
		}
	}
}

POD_heist_select_map = {
	if = {
		limit = { var:court_owner = { POD_is_infernalist_trigger = yes } }
		POD_heist_create_pentagram_map = yes
	}
	else_if = {
		limit = { var:court_owner = { has_variable = haven_map } }

		switch = {
			trigger = var:court_owner.var:haven_map

			flag:map_star    = { POD_heist_create_star_map    = yes }
			flag:map_tilted  = { POD_heist_create_tilted_map  = yes }
			flag:map_cross   = { POD_heist_create_cross_map   = yes }
			flag:map_compass = { POD_heist_create_compass_map = yes }
		}
	}
	else = {
		random_list = {
			1 = { POD_heist_set_haven_map = { MAP = star } }
			1 = { POD_heist_set_haven_map = { MAP = tilted } }
			1 = { POD_heist_set_haven_map = { MAP = cross } }
			1 = { POD_heist_set_haven_map = { MAP = compass } }
		}
	}
}

	var:court_owner = { set_variable = { name = haven_map  value = flag:map_$MAP$ } }
}

POD_heist_refresh = {
	hidden_effect = {
		every_in_list = {
			variable = nodes
	
			if = {
				limit = {
					OR = {
						this = prev.var:current_location
						POD_heist_is_node_connected_trigger = { TARGET = prev.var:current_location }
					}
					var:explored = no
				}
				set_variable = {
					name  = explored
					value = yes
				}
			}
		}
		every_in_list = {
			variable = lines
			
			if = {
				limit = {
					OR = {
						var:start_node = prev.var:current_location
						var:end_node   = prev.var:current_location
					}
				}
				set_variable = {
					name  = available
					value = yes
				}
				if = {
					limit = { var:end_node = prev.var:current_location }
					# Make lines radiate outwards from current location
					POD_heist_reverse_line = yes
				}
			}
			else = {
				set_variable = {
					name  = available
					value = no
				}
			}
		}
	}
}

POD_heist_raise_suspicion_effect = {
	if = {
		limit = { var:current_heist ?= { POD_heist_uses_suspicion_trigger = yes } }

		save_scope_value_as = {
			name = new_suspicion
			value = {
				value = var:current_heist.var:suspicion
				add = $VALUE$
				max = suspicion_max
			}
		}

		custom_description = {
			text  = POD_heist_raise_suspicion_desc
			value = $VALUE$
			var:current_heist ?= {
				set_variable = {
					name  = suspicion
					value = scope:new_suspicion
				}
			}
		}

		if = {
			limit = {
				scope:new_suspicion >= suspicion_max
			}
			custom_description_no_bullet = {
				text  = POD_heist_maximum_suspicion_desc

				POD_heist_trigger_suspicion_failure_event = yes
			}
		}
		POD_heist_check_suspicion_threshold = { NUM = 3 }
		POD_heist_check_suspicion_threshold = { NUM = 2 }
		POD_heist_check_suspicion_threshold = { NUM = 1 }
	}
}

POD_heist_check_suspicion_threshold = {
	if = {
		limit = {
			scope:new_suspicion >= suspicion_threshold_$NUM$
			var:current_heist ?= { NOT = { has_variable = triggered_suspicion_event_$NUM$ } }
		}
		custom_description_no_bullet = {
			text  = POD_heist_trigger_suspicion_event_desc
			value = suspicion_threshold_$NUM$

			POD_heist_pause_for_suspicion_event = yes
			trigger_event = { saved_event_id = var:current_heist.var:suspicion_event_$NUM$ }
			var:current_heist = { set_variable = triggered_suspicion_event_$NUM$ }
		}
	}
}

# currently valid resources: gold, prestige, piety
POD_heist_track_resource_delta = {
	# save as scope value first so that script_values can be used regardless of the scope change
	if = {
		limit = { NOT = { exists = scope:$RESOURCE$_delta } }
		save_scope_value_as = {
			name  = $RESOURCE$_delta
			value = $AMOUNT$
		}
	}
	var:current_heist ?= {
		change_variable = {
			name = $RESOURCE$_delta
			add  = scope:$RESOURCE$_delta
		}
	}
}
POD_heist_add_and_track_resource = {
	add_$RESOURCE$ = $AMOUNT$
	POD_heist_track_$RESOURCE$_delta = { AMOUNT = $AMOUNT$ }
}

POD_heist_track_gold_delta     = { POD_heist_track_resource_delta = { RESOURCE = gold     AMOUNT = $AMOUNT$ } }
POD_heist_track_prestige_delta = { POD_heist_track_resource_delta = { RESOURCE = prestige AMOUNT = $AMOUNT$ } }
POD_heist_track_piety_delta    = { POD_heist_track_resource_delta = { RESOURCE = piety    AMOUNT = $AMOUNT$ } }

POD_heist_add_and_track_gold = {
	if = {
		limit = { $AMOUNT$ > 0 }
		add_short_term_gold = $AMOUNT$
	}
	else = {
		remove_short_term_gold = { value = 0 subtract = $AMOUNT$ }
	}
	POD_heist_track_gold_delta = { AMOUNT = $AMOUNT$ }
}
POD_heist_steal_and_track_gold = {

		save_scope_value_as = {
			name  = gold_delta

		pay_short_term_gold = {
			target = prev
			gold = $AMOUNT$
		}
	}
	POD_heist_track_gold_delta = { AMOUNT = $AMOUNT$ }
	var:current_heist ?= {
		change_variable = {
			name = stolen_gold
			add  = scope:gold_delta
		}
	}
}

POD_heist_add_key_to_loot = {
	if = {
		limit = {
			scope:story ?= {
				NOT = { has_variable = found_key }
			}
		}

				}

				name  = keys_found
				value = scope:keys_found
			}

			else = {
				custom_description = {
					text  = POD_heist_found_key_desc
					value = scope:keys_found
				}
			}
		}

	else = {

	}
}

POD_heist_increase_infiltration_skill = {
	if = {
		limit = {
			NAND = {
				has_variable = infiltration_experience_modifier_stacks
				var:infiltration_experience_modifier_stacks >= infiltration_experience_max_stacks
			}
		}

		save_scope_value_as = {
			name = infiltration_experience
			value = {
				if = {
					limit = { has_variable = infiltration_experience }
					value = var:infiltration_experience
				}
				else = {
					value = 0
				}
				add = 1
			}
		}

		if = {
			limit = { scope:infiltration_experience >= heists_needed_for_level_up }

			POD_add_modifier_and_track_stacks = {
				TYPE = character
				MODIFIER = infiltration_experience_modifier
			}
			remove_variable = infiltration_experience
		}
		else = {
			set_variable = {
				name  = infiltration_experience
				value = scope:infiltration_experience
			}
			save_scope_value_as = {
				name = heists_till_level_up
				value = {
					value = heists_needed_for_level_up
					subtract = scope:infiltration_experience
				}
			}
			custom_description = {
				text  = POD_heist_levelup_hint_desc
				value = scope:heists_till_level_up
			}
		}
	}
}

POD_heist_populate_default_nodes = {
	if = {
		limit = { POD_heist_is_hostile_trigger = { HEIST = this } }
		
		every_in_list = {
			variable = nodes
			limit = { NOT = { has_variable = type } }
			random_list = {
				3 = { POD_heist_set_node_type = { TYPE = martial     ICON = martial } }
				3 = { POD_heist_set_node_type = { TYPE = intrigue    ICON = intrigue } }
				3 = { POD_heist_set_node_type = { TYPE = learning    ICON = learning } }
				3 = { POD_heist_set_node_type = { TYPE = prowess     ICON = prowess } }
	
				1 = { POD_heist_set_node_type = { TYPE = fight_elite ICON = fight_elite } }
				
				1 = {
					trigger = { NOT = { exists = scope:scramble } }
					POD_heist_set_node_type = { TYPE = empty ICON = empty }
				}
				1 = {
					trigger = { NOT = { exists = scope:scramble } }
					POD_heist_set_node_type = { TYPE = scramble ICON = scramble }
				}
			}
		}
	}
	else = {
		every_in_list = {
			variable = nodes
			limit = { NOT = { has_variable = type } }
			random_list = {
				3 = { POD_heist_set_node_type = { TYPE = diplomacy   ICON = diplomacy } }
				3 = { POD_heist_set_node_type = { TYPE = stewardship ICON = stewardship } }
				3 = { POD_heist_set_node_type = { TYPE = intrigue    ICON = intrigue } }
				3 = { POD_heist_set_node_type = { TYPE = learning    ICON = learning } }
				3 = { POD_heist_set_node_type = { TYPE = prowess     ICON = prowess } }
	
				1 = {
					trigger = { NOT = { exists = scope:scramble } }
					POD_heist_set_node_type = { TYPE = empty ICON = empty }
				}
				1 = {
					trigger = { NOT = { exists = scope:scramble } }
					POD_heist_set_node_type = { TYPE = scramble ICON = scramble }
				}
				
				0 = { POD_heist_set_node_type = { TYPE = event       ICON = event } } # TODO
			}
		}
	}
}

POD_heist_initialize_artifact_heist = {
	$ARTIFACT$ = { save_temporary_scope_as = target_artifact }

	if = {
		limit = { scope:target_artifact = { POD_heist_artifact_should_be_double_locked = yes } }

		save_scope_value_as = {
			name  = keys_required
			value = 2
		}
		custom_tooltip = POD_heist_interaction_two_keys
	}
	else = {
		save_scope_value_as = {
			name  = keys_required
			value = 1
		}
		custom_tooltip = POD_heist_interaction_one_key
	}

	hidden_effect_new_object = {
		$HEIST$ ?= {
			set_variable = {
				name  = target_artifact
				value = scope:target_artifact
			}
			set_variable = {
				name  = gold_owner
				value = var:court_owner
			}

			random_in_list = {
				variable = nodes
				limit = { POD_heist_is_node_potential_objective = yes }
				POD_heist_set_node_type = { TYPE = artifact ICON = artifact }
			}

			set_variable = {
				name  = keys_required
				value = scope:keys_required
			}

			if = {
				limit = { scope:keys_required = 2 }

				while = {
					count = 2
					random_in_list = {
						variable = nodes
						limit = { POD_heist_is_node_uninitialized_side_objective = yes }
						POD_heist_set_node_type = { TYPE = key_half ICON = key }
					}
				}
			}
			else = {
				random_in_list = {
					variable = nodes
					limit = { POD_heist_is_node_uninitialized_side_objective = yes }
					POD_heist_set_node_type = { TYPE = key ICON = key }
				}
			}
			
			set_variable = {
				name  = keys_found
				value = 0
			}

			# Side exits after keys so they don't get overwritten
			every_in_list = {
				variable = nodes
				limit = { POD_heist_is_node_potential_exit = yes }
				POD_heist_set_node_type = { TYPE = exit ICON = exit }
			}

			if = {
				limit = { POD_heist_is_hostile_trigger = { HEIST = this } }

				# generate a map first
				random_in_list = {
					variable = nodes
					limit = { POD_heist_is_node_uninitialized_side_objective = yes }
					POD_heist_set_node_type = { TYPE = map ICON = map }
				}

				if = {
					limit = { var:court_owner = { has_prisoners = yes } }
					random_in_list = {
						variable = nodes
						limit = { POD_heist_is_node_uninitialized_side_objective = yes }
						POD_heist_set_node_type = { TYPE = prisoners ICON = prisoners }
					}
				}

				every_in_list = {
					variable = nodes
					limit = { POD_heist_is_node_uninitialized_side_objective = yes }
					random_list = {
						1 = { POD_heist_set_node_type = { TYPE = money  ICON = money } }
						1 = { POD_heist_set_node_type = { TYPE = gossip ICON = gossip } }
						1 = {
							trigger = { $HEIST$.var:court_owner = { number_of_knights >= 1 } }
							POD_heist_set_node_type = { TYPE = fight_elite ICON = fight_elite }
						}
					}
				}
			}
			else = {
				every_in_list = {
					variable = nodes
					limit = { POD_heist_is_node_uninitialized_side_objective = yes }
					random_list = {
						1 = { POD_heist_set_node_type = { TYPE = money    ICON = money } }
						1 = { POD_heist_set_node_type = { TYPE = fame     ICON = fame } }
						1 = { POD_heist_set_node_type = { TYPE = gossip   ICON = gossip } }
						1 = { POD_heist_set_node_type = { TYPE = destress ICON = destress } }
					}
				}
			}
			
			POD_heist_populate_default_nodes = yes
		}
	}
}
