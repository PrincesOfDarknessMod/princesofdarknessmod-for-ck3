
POD_heist_setup_effect = {
	hidden_effect = {
		character:POD_wraith_01 = { # it's ya boi
			create_story = {
				type = POD_story_heist
				save_scope_as = $SAVE_AS$
			}
		}
		if = {
			limit = { exists = scope:$SAVE_AS$ } # it's ya error suppression
	
			scope:$SAVE_AS$ = {
				POD_heist_create_$MAP$_map = yes
				
				random_in_list = {
					variable = nodes
					limit = { POD_heist_is_node_potential_start_trigger = yes }
					prev = {
						set_variable = {
							name  = current_location
							value = prev
						}
					}
				}

				POD_save_event_as_var = {
					VAR = ending_event
					ID  = $ENDING$
				}

				if = {
					limit = { prev = { POD_has_scouted_heist_trigger = yes } }
					every_in_list = {
						variable = nodes
						set_variable = {
							name  = explored
							value = yes
						}
					}
				}
				
				POD_heist_refresh = yes
			}
			
			# using this until I find a way to access saved story scopes from a TopScope
			set_variable = {
				name  = current_heist
				value = scope:heist
			}
		}
	}
}

	character:POD_wraith_01 = { # it's ya boi
		create_story = {
			type = POD_story_heist_node
			save_scope_as = $SAVE_AS$
		}
	}
	if = {
		limit = { exists = scope:$SAVE_AS$ } # it's ya error suppression

		scope:$SAVE_AS$ = {
			set_variable = {
				name  = coord_x
				value = $X$
			}
			set_variable = {
				name  = coord_y
				value = $Y$
			}
		}
		
		add_to_variable_list = {
			name   = nodes
			target = scope:$SAVE_AS$
		}
	}
}

				}
			}
		}
		
		add_to_variable_list = {

			target = scope:temp_line

POD_heist_refresh = {
	hidden_effect = {
		every_in_list = {
			variable = nodes
	
			if = {
				limit = {
					OR = {
						this = prev.var:current_location
						POD_heist_is_node_connected_trigger = { TARGET = prev.var:current_location }
					}
					var:explored = no
				}
				set_variable = {
					name  = explored
					value = yes
				}
			}
		}
		every_in_list = {
			variable = lines
			
			if = {
				limit = {
					OR = {
						var:start_node = prev.var:current_location
						var:end_node   = prev.var:current_location
					}
				}
				set_variable = {
					name  = available
					value = yes
				}
				if = {
					limit = { var:end_node = prev.var:current_location }
					# Make lines radiate outwards from current location
					POD_heist_reverse_line = yes
				}
			}
			else = {
				set_variable = {
					name  = available
					value = no
				}
			}
		}
	}
}

POD_heist_raise_suspicion_effect = {
	if = {
		limit = { var:current_heist ?= { POD_heist_uses_suspicion_trigger = yes } }

		save_scope_value_as = {
			name = new_suspicion
			value = {
				value = var:current_heist.var:suspicion
				add = $VALUE$
				max = suspicion_max
			}
		}

		custom_description = {
			text  = POD_heist_raise_suspicion_desc
			value = $VALUE$
			var:current_heist ?= {
				set_variable = {
					name  = suspicion
					value = scope:new_suspicion
				}
			}
		}

		if = {
			limit = {
				scope:new_suspicion >= suspicion_max
			}
			custom_description_no_bullet = {
				text  = POD_heist_maximum_suspicion_desc

				POD_heist_pause_for_event = yes
				trigger_event = { saved_event_id = var:current_heist.var:suspicion_event_4 }
			}
		}
		POD_heist_check_suspicion_threshold = { NUM = 3 }
		POD_heist_check_suspicion_threshold = { NUM = 2 }
		POD_heist_check_suspicion_threshold = { NUM = 1 }
	}
}

POD_heist_check_suspicion_threshold = {
	if = {
		limit = {
			scope:new_suspicion >= suspicion_threshold_$NUM$
			var:current_heist ?= { NOT = { has_variable = triggered_suspicion_event_$NUM$ } }
		}
		custom_description_no_bullet = {
			text  = POD_heist_trigger_suspicion_event_desc
			value = suspicion_threshold_$NUM$

			POD_heist_pause_for_event = yes
			trigger_event = { saved_event_id = var:current_heist.var:suspicion_event_$NUM$ }
			var:current_heist = { set_variable = triggered_suspicion_event_$NUM$ }
		}
	}
}

# currently valid resources: gold, prestige, piety
POD_heist_track_resource_delta = {
	# save as scope value first so that script_values can be used regardless of the scope change
	if = {
		limit = { NOT = { exists = scope:$RESOURCE$_delta } }
		save_scope_value_as = {
			name  = $RESOURCE$_delta
			value = $AMOUNT$
		}
	}
	var:current_heist ?= {
		change_variable = {
			name = $RESOURCE$_delta
			add  = scope:$RESOURCE$_delta
		}
	}
}
POD_heist_add_and_track_resource = {
	add_$RESOURCE$ = $AMOUNT$
	POD_heist_track_$RESOURCE$_delta = { AMOUNT = $AMOUNT$ }
}

POD_heist_track_gold_delta     = { POD_heist_track_resource_delta = { RESOURCE = gold     AMOUNT = $AMOUNT$ } }
POD_heist_track_prestige_delta = { POD_heist_track_resource_delta = { RESOURCE = prestige AMOUNT = $AMOUNT$ } }
POD_heist_track_piety_delta    = { POD_heist_track_resource_delta = { RESOURCE = piety    AMOUNT = $AMOUNT$ } }

POD_heist_add_and_track_gold = {
	if = {
		limit = { $AMOUNT$ > 0 }
		add_short_term_gold = $AMOUNT$
	}
	else = {
		remove_short_term_gold = { value = 0 subtract = $AMOUNT$ }
	}
	POD_heist_track_gold_delta = { AMOUNT = $AMOUNT$ }
}
POD_heist_steal_and_track_gold = {
	var:current_heist.var:court_owner ?= {
		save_scope_value_as = {
			name  = gold_delta
			value = $AMOUNT$
		}
		pay_short_term_gold = {
			target = prev
			gold = $AMOUNT$
		}
	}
	POD_heist_track_gold_delta = { AMOUNT = $AMOUNT$ }
	var:current_heist ?= {
		change_variable = {
			name = stolen_gold
			add  = scope:gold_delta
		}
	}
}
POD_heist_add_and_track_prestige = {

}
POD_heist_add_and_track_piety = {

	POD_heist_track_piety_delta = { AMOUNT = $AMOUNT$ }
}

	scope:$STAT$_duelist = {
		duel = {
			skill = $STAT$
			value = $VALUE$
			$WEIGHT$ = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = 0.5
				}
				desc = POD_heist_effect_node_$STAT$_success
			}
			$WEIGHT$ = {
				desc = POD_heist_effect_node_$STAT$_fail
				root = { POD_heist_raise_suspicion_effect = { VALUE = minor_suspicion_value } }
			}
			$WEIGHT$ = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = -0.5
				}
				desc = POD_heist_effect_node_$STAT$_critfail
				root = { POD_heist_raise_suspicion_effect = { VALUE = major_suspicion_value } }
				$ONFAIL$
			}
		}
	}
}

POD_heist_add_artifact_to_loot = {
	if = {
		limit = {
			var:current_heist ?= {
				NOT = {
					is_target_in_variable_list = {
						name   = loot_artifacts
						target = var:target_artifact
					}
				}
			}
		}
		
		POD_commit_theft_effect = yes

		POD_heist_raise_suspicion_effect = { VALUE = massive_suspicion_value }

		if = {
			# Don't take the artifact if the Infiltration would fail upon entering the node
			limit = { scope:new_suspicion < suspicion_max }

			play_sound_effect = "event:/DLC/EP1/SFX/UI/Notifications/inspiration_project_complete"

			send_interface_toast = {
				title = POD_heist_gain_artifact_toast
				left_icon = var:current_heist.var:target_artifact

				custom_description = { text = POD_heist_gain_artifact_desc }
				custom_tooltip = POD_heist_gain_artifact_hint
			}
			hidden_effect = {
				var:current_heist = {
					add_to_variable_list = {
						name   = loot_artifacts
						target = var:target_artifact
					}
				}
			}
		}
	}
}

POD_heist_display_expenditures = {
	custom_tooltip = POD_heist_resource_deltas_tt
	show_as_tooltip = {
		if = {
			limit = { var:current_heist.var:gold_delta > 0 }
			add_short_term_gold = var:current_heist.var:gold_delta
		}
		else_if = {
			limit = { var:current_heist.var:gold_delta < 0 }
			remove_short_term_gold = { value = 0 subtract = var:current_heist.var:gold_delta }
		}
		if = {
			limit = { NOT = { var:current_heist.var:prestige_delta = 0 } }
			add_prestige = var:current_heist.var:prestige_delta
		}
		if = {
			limit = { NOT = { var:current_heist.var:piety_delta = 0 } }
			add_piety = var:current_heist.var:piety_delta
		}
	}
}

POD_heist_initialize_artifact_heist = {
	$ARTIFACT$ = { save_temporary_scope_as = target_artifact }
	hidden_effect = {
		$HEIST$ ?= {
			set_variable = {
				name  = target_artifact
				value = scope:target_artifact
			}
			set_variable = {
				name  = court_owner
				value = scope:target_artifact.artifact_owner ################# TODO: differentiate between landed/unlanded
			}

			random_in_list = {
				variable = nodes
				limit = { POD_heist_is_node_potential_objective_trigger = yes }
				POD_heist_set_node_type = { TYPE = artifact ICON = artifact }
			}

			every_in_list = {
				variable = nodes
				limit = {
					POD_heist_is_node_side_objective_trigger = yes
					NOT = { has_variable = type }
				}
				random_list = {
					1 = { POD_heist_set_node_type = { TYPE = money    ICON = money } }
					1 = { POD_heist_set_node_type = { TYPE = fame     ICON = fame } }
					1 = { POD_heist_set_node_type = { TYPE = gossip   ICON = gossip } }
					1 = { POD_heist_set_node_type = { TYPE = destress ICON = destress } }
				}
			}
			
			POD_heist_populate_default_nodes = yes
		}
	}
}
