POD_heist_setup_effect = {
	hidden_effect = {
		character:POD_wraith_01 = { # it's ya boi
			create_story = {
				type = POD_story_heist
				save_scope_as = $SAVE_AS$
			}
		}
		if = {
			limit = { exists = scope:$SAVE_AS$ } # it's ya error suppression
	
			scope:$SAVE_AS$ = {
				POD_heist_create_$MAP$_map = yes
	
				random_in_list = {
					variable = nodes
					limit = { var:type ?= flag:node_exit }
					prev = {
						set_variable = {
							name  = current_location
							value = prev
						}
					}
				}
	
				POD_heist_refresh = yes
			}
	
			# using this until i find a way to access saved story scopes from a TopScope
			set_variable = {
				name  = current_heist
				value = scope:heist
			}
		}
	}
}

POD_heist_cleanup_effect = {
	hidden_effect = {
		var:current_heist ?= { end_story = yes }
		remove_variable = current_heist
	}
}

POD_heist_create_node = {
	character:POD_wraith_01 = { # it's ya boi
		create_story = {
			type = POD_story_heist_node
			save_scope_as = $SAVE_AS$
		}
	}
	if = {
		limit = { exists = scope:$SAVE_AS$ } # it's ya error suppression

		scope:$SAVE_AS$ = {
			set_variable = {
				name  = coord_x
				value = $X$
			}
			set_variable = {
				name  = coord_y
				value = $Y$
			}
		}
		
		add_to_variable_list = {
			name   = nodes
			target = scope:$SAVE_AS$
		}
	}
}

# POD_heist_create_line = { NODE1 = scope:node1 NODE2 = scope:node2 }
POD_heist_create_line = {
	character:POD_wraith_01 = { # it's ya boi
		create_story = {
			type = POD_story_heist_line
			save_scope_as = temp_line
		}
	}
	if = {
		limit = { exists = scope:temp_line } # it's ya error suppression

		scope:temp_line = {
			set_variable = {
				name  = start_node
				value = $NODE1$
			}
			set_variable = {
				name  = end_node
				value = $NODE2$
			}
		
			$NODE1$ = {
				add_to_variable_list = {
					name   = lines
					target = prev
				}
			}
			$NODE2$ = {
				add_to_variable_list = {
					name   = lines
					target = prev
				}
			}
		}
		
		add_to_variable_list = {
			name   = lines
			target = scope:temp_line
		}
	}
}

POD_heist_set_node_type = {
	set_variable = {
		name  = type
		value = flag:node_$TYPE$
	}
}

POD_heist_reverse_line = {
	var:start_node = { save_temporary_scope_as = previous_start_node }
	set_variable = {
		name  = start_node
		value = var:end_node
	}
	set_variable = {
		name  = end_node
		value = scope:previous_start_node
	}
}

POD_heist_refresh = {
	hidden_effect = {
		every_in_list = {
			variable = nodes
	
			if = {
				limit = {
					OR = {
						this = prev.var:current_location
						POD_heist_is_node_connected_trigger = { TARGET = prev.var:current_location }
					}
					var:explored = no
				}
				set_variable = {
					name  = explored
					value = yes
				}
			}
		}
		every_in_list = {
			variable = lines
			
			if = {
				limit = {
					OR = {
						var:start_node = prev.var:current_location
						var:end_node   = prev.var:current_location
					}
				}
				set_variable = {
					name  = status
					value = flag:available
				}
				if = {
					limit = { var:end_node = prev.var:current_location }
					# Make lines radiate outwards from current location
					POD_heist_reverse_line = yes
				}
			}
			else = {
				set_variable = {
					name  = status
					value = flag:unavailable
				}
			}
		}
	}
}
