POD_heist_setup_effect = {
	hidden_effect = {
		character:POD_wraith_01 = { # it's ya boi
			create_story = {
				type = POD_story_heist
				save_scope_as = $SAVE_AS$
			}
		}
		if = {
			limit = { exists = scope:$SAVE_AS$ } # it's ya error suppression
	
			scope:$SAVE_AS$ = {
				POD_heist_create_$MAP$_map = yes
				
				random_in_list = {
					variable = nodes
					limit = { POD_heist_is_node_potential_start_trigger = yes }
					prev = {
						set_variable = {
							name  = current_location
							value = prev
						}
					}
				}

				POD_save_event_as_var = {
					VAR = ending_event
					ID  = $ENDING$
				}
				
				POD_heist_refresh = yes
			}
			
			# using this until I find a way to access saved story scopes from a TopScope
			set_variable = {
				name  = current_heist
				value = scope:heist
			}
		}
	}
}

POD_heist_create_node = {
	character:POD_wraith_01 = { # it's ya boi
		create_story = {
			type = POD_story_heist_node
			save_scope_as = $SAVE_AS$
		}
	}
	if = {
		limit = { exists = scope:$SAVE_AS$ } # it's ya error suppression

		scope:$SAVE_AS$ = {
			set_variable = {
				name  = coord_x
				value = $X$
			}
			set_variable = {
				name  = coord_y
				value = $Y$
			}
		}
		
		add_to_variable_list = {
			name   = nodes
			target = scope:$SAVE_AS$
		}
	}
}

POD_heist_create_line = {
	character:POD_wraith_01 = { # it's ya boi
		create_story = {
			type = POD_story_heist_line
			save_scope_as = temp_line
		}
	}
	if = {
		limit = { exists = scope:temp_line } # it's ya error suppression

		scope:temp_line = {
			set_variable = {
				name  = start_node
				value = $NODE1$
			}
			set_variable = {
				name  = end_node
				value = $NODE2$
			}
		
			$NODE1$ = {
				add_to_variable_list = {
					name   = lines
					target = prev
				}
			}
			$NODE2$ = {
				add_to_variable_list = {
					name   = lines
					target = prev
				}
			}
		}
		
		add_to_variable_list = {
			name   = lines
			target = scope:temp_line
		}
	}
}

POD_heist_reverse_line = {
	var:start_node = { save_temporary_scope_as = previous_start_node }
	set_variable = {
		name  = start_node
		value = var:end_node
	}
	set_variable = {

	}
}

POD_heist_refresh = {
	hidden_effect = {
		every_in_list = {
			variable = nodes
	
			if = {
				limit = {
					OR = {
						this = prev.var:current_location
						POD_heist_is_node_connected_trigger = { TARGET = prev.var:current_location }
					}
					var:explored = no
				}
				set_variable = {
					name  = explored
					value = yes
				}
			}
		}
		every_in_list = {
			variable = lines
			
			if = {
				limit = {
					OR = {
						var:start_node = prev.var:current_location
						var:end_node   = prev.var:current_location
					}
				}
				set_variable = {
					name  = available
					value = yes
				}
				if = {
					limit = { var:end_node = prev.var:current_location }
					# Make lines radiate outwards from current location
					POD_heist_reverse_line = yes
				}
			}
			else = {
				set_variable = {
					name  = available
					value = no
				}
			}
		}
	}
}

POD_heist_raise_suspicion_effect = {
	if = {
		limit = { var:current_heist ?= { POD_heist_uses_suspicion_trigger = yes } }

		save_scope_value_as = {
			name = new_suspicion
			value = {
				value = var:current_heist.var:suspicion
				add = $VALUE$
				max = suspicion_max
			}
		}

		custom_description = {
			text  = POD_heist_raise_suspicion_desc
			value = $VALUE$
			var:current_heist ?= {
				set_variable = {
					name  = suspicion
					value = scope:new_suspicion
				}
			}
		}

		if = {
			limit = {
				scope:new_suspicion >= suspicion_max
			}
			custom_description_no_bullet = {
				text  = POD_heist_maximum_suspicion_desc

				POD_heist_pause_for_event = yes
				#POD_trigger_event_from_var = { VAR = var:current_heist.var:suspicion_event_4 }
				trigger_event = { saved_event_id = var:current_heist.var:suspicion_event_4 }
			}
		}
		POD_heist_check_suspicion_threshold = { NUM = 3 }
		POD_heist_check_suspicion_threshold = { NUM = 2 }
		POD_heist_check_suspicion_threshold = { NUM = 1 }
	}
}

POD_heist_check_suspicion_threshold = {
	if = {
		limit = {
			scope:new_suspicion >= suspicion_threshold_$NUM$
			var:current_heist ?= { NOT = { has_variable = triggered_suspicion_event_$NUM$ } }
		}
		custom_description_no_bullet = {
			text  = POD_heist_trigger_suspicion_event_desc
			value = suspicion_threshold_$NUM$

			POD_heist_pause_for_event = yes
			#POD_trigger_event_from_var = { VAR = var:current_heist.var:suspicion_event_$NUM$ }
			trigger_event = { saved_event_id = var:current_heist.var:suspicion_event_$NUM$ }
			var:current_heist = { set_variable = triggered_suspicion_event_$NUM$ }
		}
	}
}

POD_heist_pause_for_event = {
	var:current_heist ?= {
		if = {
			limit = { NOT = { has_variable = paused_for_event } }
			set_variable = paused_for_event
		}
	}
}

POD_heist_resume_after_event = {
	var:current_heist ?= {
		if = {
			limit = { has_variable = paused_for_event }
			remove_variable = paused_for_event
		}
	}
}

POD_heist_add_artifact_to_loot = {
	if = {
		limit = {
			var:current_heist ?= {
				NOT = {
					is_target_in_variable_list = {
						name   = loot_artifacts
						target = var:target_artifact
					}
				}
			}
		}
		
		POD_commit_theft_effect = yes

		POD_heist_raise_suspicion_effect = { VALUE = major_suspicion_value }

		if = {
			# Don't take the artifact if the Infiltration would fail upon entering the node
			limit = { scope:new_suspicion < suspicion_max }

			custom_description = { text = POD_heist_gain_artifact_desc }
			custom_tooltip = POD_heist_gain_artifact_hint
			hidden_effect = {
				var:current_heist ?= {
					add_to_variable_list = {
						name   = loot_artifacts
						target = var:target_artifact
					}
				}
			}
			POD_heist_pause_for_event = yes
			trigger_event = POD_heists.50
		}
	}
}

POD_heist_gain_loot_effect = {
	save_temporary_scope_as = loot_receiver
	var:current_heist ?= {
		every_in_list = {
			variable  = loot_artifacts
			set_owner = scope:loot_receiver
		}
	}
}

POD_heist_initialize_artifact_heist = {
	$ARTIFACT$ = { save_temporary_scope_as = target_artifact }
	hidden_effect = {
		$HEIST$ ?= {
			set_variable = {
				name  = target_artifact
				value = scope:target_artifact
			}
			set_variable = {
				name  = court_owner
				value = scope:target_artifact.artifact_owner ################# TODO: differentiate between landed/unlanded
			}

			random_in_list = {
				variable = nodes
				limit = {
					POD_heist_is_node_potential_objective_trigger = yes
				}
				POD_heist_set_node_type = { TYPE = artifact ICON = artifact }
			}

			every_in_list = {
				variable = nodes
				limit = {
					POD_heist_is_node_side_objective_trigger = yes
					NOT = { has_variable = type }
				}
				random_list = {
					1 = { POD_heist_set_node_type = { TYPE = money ICON = money } }
				}
			}
		
			every_in_list = {
				variable = nodes
				limit = { NOT = { has_variable = type } }
				random_list = {
					1 = { POD_heist_set_node_type = { TYPE = event       ICON = event } }
					1 = { POD_heist_set_node_type = { TYPE = diplomacy   ICON = diplomacy } }
					1 = { POD_heist_set_node_type = { TYPE = intrigue    ICON = intrigue } }
					1 = { POD_heist_set_node_type = { TYPE = learning    ICON = learning } }
					1 = { POD_heist_set_node_type = { TYPE = fight       ICON = fight } }
					0 = { POD_heist_set_node_type = { TYPE = fight_elite ICON = fight_elite } }
					1 = { POD_heist_set_node_type = { TYPE = bribe       ICON = bribe } }
				}
			}
		}
	}
