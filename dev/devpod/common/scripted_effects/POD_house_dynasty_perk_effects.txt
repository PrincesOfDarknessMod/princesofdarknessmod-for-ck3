meritocracy_2_effect_a = {
	if = {
		limit = {
			has_trait = lifestyle_blademaster
			has_trait_xp = { trait = lifestyle_blademaster value = 100 }
			NOT = { has_trait = berserker }
		}
		add_trait = berserker
	}
	
	else_if = {
		limit = { 
			has_trait = lifestyle_blademaster 
			NOT = { has_trait_xp = { trait = lifestyle_blademaster value = 100 } }
		}
		add_trait_xp = { trait = lifestyle_blademaster value = 50 }
	}
		
	else = {
		limit = { NOT = { has_trait = lifestyle_blademaster } }
		add_trait = lifestyle_blademaster
	}
}

meritocracy_5_effect = {
	if = {
		limit = { 
			OR = { 
				POD_is_infernalist_trigger = no
				POD_splat_mixing_haight_mode = yes
			}
		}
		if = {
			limit = { POD_has_true_faith_trigger = no }
			random = {
				chance = 5
				add_trait = truefaith
			}
		}
		else_if = {
			limit = { has_trait = truefaith }
			random = {
				chance = 1
				assign_truefaith_effect = yes
			}
		}
	}
}
