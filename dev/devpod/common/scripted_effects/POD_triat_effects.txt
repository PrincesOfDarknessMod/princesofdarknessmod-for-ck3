POD_become_gaia_servant_effect = {
	if = {
		limit = { POD_is_gaia_servant_trigger = no }

			limit = {

				}

		}
	}
}

POD_become_weaver_servant_effect = {
	if = {
		limit = { POD_is_weaver_servant_trigger = no }
		if = {
			limit = { POD_has_inheritable_fera_type = { TRAIT = trait:werewolf } }

		}
	}
}

POD_become_wyrm_servant_effect = {
	if = {
		limit = { POD_is_wyrm_servant_trigger = no }

		if = {
			limit = { has_POD_vamp_blood_trigger = yes }
			set_character_faith = faith:viatantabus
		}

		#Fera
		if = {
			limit = {
				OR = {
					has_trait = kinfolk
					has_trait = fera 
				}
			}
			POD_fera_fall_to_wyrm_effect = yes
		}
		#Mummies
		if = {
			limit = {
				OR = {
					is_amenti_trigger = yes
					is_east_mummy_trigger = yes
				}
			}
			set_character_faith = faith:faces_of_apophis
			POD_remove_mummy_traits_effect = yes
			add_trait = banemummy
		}
		#Wraiths
		if = {
			limit = { has_trait = wraith }
			POD_become_spectre_effect = yes
			add_trait_xp = {
				trait = spectre
				value = 25
			}
		}
		#Hunters
		if = {
			limit = { has_trait = supehunter }
			POD_hunter_enthrall_effect = yes
		}
		#KJ
		if = {
			limit = { has_trait = kueijin }
			set_character_faith = faith:wayofthetenthousandscreams
		}

		add_character_modifier = { modifier = wyrm_new_servant_modifier years = 25 }
	}
	
}
