
POD_adopt_correct_religion_effect = {
	if = {
		limit = {
			exists = mother
			pod_convert_trigger = { CHARACTER = this FAITH = mother.faith }
		}
		set_character_faith = mother.faith
	}	
	else_if = {
		limit = {
			exists = father
			pod_convert_trigger = { CHARACTER =this FAITH = father.faith }
		}
		set_character_faith = father.faith
	}
	else_if = {
		limit = {
			exists = liege_or_court_owner
			pod_convert_trigger = { CHARACTER =this FAITH = liege_or_court_owner.faith }
		}
		set_character_faith = liege_or_court_owner.faith
	}
	else_if = {
		limit = { is_a_cultist = yes }
		random_relation = {
			type = worshipped_demon
			save_temporary_scope_as = demon_temp
		}
		set_character_faith = scope:demon_temp.faith
	}
	else = {
		random_list = {
			10 = { 
				trigger = { 
					OR = {
						has_trait = vampire
						has_trait = dhampir
					}
				}
				POD_character_vampire_faith_initialization_effect = yes
			}
			10 = { 
				trigger = { POD_is_any_kind_of_ghoul_trigger = yes }
				POD_character_revenant_faith_initialization_effect = yes
			}
			10 = { 
				trigger = { 
					has_POD_kj_blood_trigger = yes 
					NOT = { has_trait = hun_and_po }
				}
				POD_character_kueijin_faith_initialization_effect = yes
			}
			10 = { 
				trigger = { has_trait = supehunter }
				POD_character_hunter_faith_initialization_effect = yes
			}
			10 = { 
				trigger = { 
					OR = {
						has_trait = fera 
						has_trait = kinfolk
					}
				}
				POD_character_shapeshifter_faith_initialization_effect = yes
			}
			10 = { 
				trigger = { has_POD_fae_blood_trigger = yes }
				POD_character_fae_faith_initiation_effect = yes
			}
			10 = { 
				trigger = { has_trait = mummy }
				POD_character_mummy_faith_initiation_effect = yes
			}
			10 = { 
				trigger = { has_trait = demon }
				POD_character_demon_faith_initiation_effect = yes
			}
			10 = { 
				trigger = { has_trait = wraith }
				POD_character_wraith_faith_initiation_effect = yes
			}
		}
	}
	POD_adopt_faith_side_effect = yes
}

POD_adopt_faith_side_effect = {
	POD_side_effects_from_changing_splats = yes
	if = {
		limit = {
			has_trait = supehunter 
			has_main_supernatural_race_trigger = yes
			NOT = { faith = faith:sanbenito }
		}
		remove_trait = supehunter 
		POD_adopt_correct_religion_effect = yes
		POD_adopt_correct_government_effect = yes # No need check illegal... you can't choose gov in RD
		POD_remove_wrong_supernatural_traits_effect = yes
	}
	if = {
		limit = { has_trait = mummy }
		mummy_faith_conversion_change_trait_effect = yes
	}
	if = {
		limit = { has_vampire_blood_race_trigger = yes }
		POD_character_vampire_infernalist_initialization_effect = yes
	}
	if = {
		limit = { has_trait = truepossessed }
		possessed_faith_conversion_suicide_effect = yes
	}
}
