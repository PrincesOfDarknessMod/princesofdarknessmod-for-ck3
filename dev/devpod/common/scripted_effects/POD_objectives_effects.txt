# Impundulu objectives

### Apply Covenant Renewed (objective branch A) rewards
### Called from the objective completion event
POD_impundulu_apply_covenant_renewed_effect = {
	dynasty ?= {
		add_dynasty_modifier = covenant_of_life_and_death_modifier
	}
	add_prestige = major_prestige_value
	add_piety = major_piety_value
}

### Apply Ishologu Rite (objective branch B) rewards
### Called from the objective completion event
POD_impundulu_apply_ishologu_rite_effect = {
	add_character_modifier = ishologu_modifier
	add_prestige = massive_prestige_value
	dynasty = {
		add_dynasty_prestige = major_dynasty_prestige_value
	}
}
