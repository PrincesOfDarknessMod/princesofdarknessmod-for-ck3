POD_initialize_keyboard_letter = {
	set_variable = {
		name = POD_qwerty_keyboard_letter_$L$
		value = flag:$L$
	}
	add_to_variable_list = {
		name = POD_qwerty_keyboard_row_$ROW$
		target = var:POD_qwerty_keyboard_letter_$L$
	}
	POD_error_suppression_unused_flag = { FLAG = $L$ }
}
