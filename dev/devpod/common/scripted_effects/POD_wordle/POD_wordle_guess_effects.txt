POD_save_wordle_guess = {
	switch = {
		trigger = exists
		var:wordle_guess_6 = {}
		var:wordle_guess_5 = { POD_set_wordle_guess_value = { N = 6 } }
		var:wordle_guess_4 = { POD_set_wordle_guess_value = { N = 5 } }
		var:wordle_guess_3 = { POD_set_wordle_guess_value = { N = 4 } }
		var:wordle_guess_2 = { POD_set_wordle_guess_value = { N = 3 } }
		var:wordle_guess_1 = { POD_set_wordle_guess_value = { N = 2 } }
		fallback           = { POD_set_wordle_guess_value = { N = 1 } }
	}
}

POD_set_wordle_guess_value = {
	set_variable = { 
		name = wordle_guess_$N$ 
		value = var:wordle_guess
	}
}
