POD_create_wordle_guess_keyboard = {
	create_story = { type = POD_story_wordle_guess save_scope_as = wordle_guess }
	var:wordle_minigame ?= {		
		scope:wordle_guess ?= {
			set_variable = { 
				name = wordle_guess_letter_1
				value = $PLAYER$.var:keyboard_entry_1
			}
			set_variable = { 
				name = wordle_guess_letter_2
				value = $PLAYER$.var:keyboard_entry_2
			}
			set_variable = { 
				name = wordle_guess_letter_3
				value = $PLAYER$.var:keyboard_entry_3
			}
			set_variable = { 
				name = wordle_guess_letter_4
				value = $PLAYER$.var:keyboard_entry_4
			}
			set_variable = { 
				name = wordle_guess_letter_5
				value = $PLAYER$.var:keyboard_entry_5
			}

			# Error suppression
			if = { limit = { exists = var:wordle_guess_letter_1 } }
			if = { limit = { exists = var:wordle_guess_letter_2 } }
			if = { limit = { exists = var:wordle_guess_letter_3 } }
			if = { limit = { exists = var:wordle_guess_letter_4 } }
			if = { limit = { exists = var:wordle_guess_letter_5 } }
		}
		POD_save_wordle_guess = yes
	}

	$PLAYER$ = {
		set_variable = { name = keyboard_entry_1 value = flag:__ }
		set_variable = { name = keyboard_entry_2 value = flag:__ }
		set_variable = { name = keyboard_entry_3 value = flag:__ }
		set_variable = { name = keyboard_entry_4 value = flag:__ }
		set_variable = { name = keyboard_entry_5 value = flag:__ }
	}
}

POD_save_wordle_guess = {
	switch = {
		trigger = exists
		var:wordle_guess_6 = {}
		var:wordle_guess_5 = { POD_set_wordle_guess_value = { N = 6 } }
		var:wordle_guess_4 = { POD_set_wordle_guess_value = { N = 5 } }
		var:wordle_guess_3 = { POD_set_wordle_guess_value = { N = 4 } }
		var:wordle_guess_2 = { POD_set_wordle_guess_value = { N = 3 } }
		var:wordle_guess_1 = { POD_set_wordle_guess_value = { N = 2 } }
		fallback           = { POD_set_wordle_guess_value = { N = 1 } }
	}
}

POD_set_wordle_guess_value = {
	set_variable = { 
		name = wordle_guess_$N$ 
		value = scope:wordle_guess
	}
	add_to_variable_list = {
		name = wordle_guesses_list
		target = var:wordle_guess_$N$ 
	}
}

POD_wordle_correct_guess_effect = {
	if = {
		limit = {
			var:wordle_minigame = { var:wordle_dictionary = flag:spell_of_life }
		}
		send_interface_toast = {
			title = POD_mummy_spell_of_life_journeys.20.effect.success
			left_icon = root
			POD_add_spell_of_life_xp = {
				VALUE = 5
			}
		}
	}
	else_if = {
		limit = {
			var:wordle_minigame = { var:wordle_dictionary = flag:elixir_of_eternal_life }
		}
		send_interface_toast = {
			title = POD_mummy_elixir_of_eternal_life_journeys.21.effect.success
			left_icon = root
			POD_add_elixir_of_eternal_life_xp = {
				VALUE = 5
			}
		}
	}
}

POD_wordle_incorrect_guess_effect = {
	if = {
		limit = {
			var:wordle_minigame = { var:wordle_dictionary = flag:spell_of_life }
		}
		if = {
			limit = { POD_wordle_guess_count >= 6 }
			send_interface_toast = {
				title = POD_mummy_spell_of_life_journeys.20.effect.failure
				left_icon = root
				custom_tooltip = { text = POD_mummy_spell_of_life_journeys.20.effect.failure.details }
			}
		}
		else = {
			trigger_event = POD_mummy_spell_of_life_journeys.20
		}
	}
	else_if = {
		limit = {
			var:wordle_minigame = { var:wordle_dictionary = flag:elixir_of_eternal_life }
		}
		if = {
			limit = { POD_wordle_guess_count >= 6 }
			send_interface_toast = {
				title = POD_mummy_elixir_of_eternal_life_journeys.21.effect.failure
				left_icon = root
				custom_tooltip = { text = POD_mummy_elixir_of_eternal_life_journeys.21.effect.failure.details }
			}
		}
		else = {
			trigger_event = POD_mummy_elixir_of_eternal_life_journeys.21
		}
	}
}
