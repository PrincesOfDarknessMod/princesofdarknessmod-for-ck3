POD_wordle_start_game = {
	set_variable = { name = hunt_in_capital_selection value = flag:A }

	var:wordle_minigame ?= { end_story = yes }
	create_story = { type = POD_story_wordle_minigame save_scope_as = wordle_minigame }
	set_variable = {
		name = wordle_minigame
		value = scope:wordle_minigame
	}

	create_story = { type = POD_story_wordle_guess save_scope_as = wordle_guess }

	var:wordle_minigame ?= {
		POD_wordle_set_solution_word = {L1 = H L2 = O L3 = U L4 = S L5 = E} # HOUSE

		set_variable = {
			name = wordle_guess_1
			value = scope:wordle_guess
		}
		var:wordle_guess_1 ?= {
			set_variable = { 
				name = wordle_guess_letter_1
				value = flag:H
			}
			set_variable = { 
				name = wordle_guess_letter_2
				value = flag:H
			}
			set_variable = { 
				name = wordle_guess_letter_3
				value = flag:E
			}
			set_variable = { 
				name = wordle_guess_letter_4
				value = flag:D
			}
			set_variable = { 
				name = wordle_guess_letter_5
				value = flag:E
			}
		}
	}
}


POD_wordle_set_solution_word = {
	POD_set_wordle_solution_letter = {N = 1 L = $L1$}
	POD_set_wordle_solution_letter = {N = 2 L = $L2$}
	POD_set_wordle_solution_letter = {N = 3 L = $L3$}
	POD_set_wordle_solution_letter = {N = 4 L = $L4$}
	POD_set_wordle_solution_letter = {N = 5 L = $L5$}
	set_variable = {
		name = wordle_solution
		value = flag:$L1$$L2$$L3$$L4$$L5$
	}
}

POD_set_wordle_solution_letter = {
	set_variable = {
		name = solution_letter_$N$
		value = flag:$L$
	}
	add_to_variable_list = {
		name = wordle_solution_letters
		target = flag:$L$
	}
}
