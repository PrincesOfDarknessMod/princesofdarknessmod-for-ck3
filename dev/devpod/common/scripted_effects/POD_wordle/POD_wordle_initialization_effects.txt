POD_wordle_start_game = {
	var:wordle_minigame ?= { end_story = yes }
	create_story = { type = POD_story_wordle_minigame save_scope_as = wordle_minigame }
	set_variable = {
		name = wordle_minigame
		value = scope:wordle_minigame
	}

	POD_wordle_set_solution_word = { MINIGAME = var:wordle_minigame }
}


POD_wordle_set_solution_word = {
	POD_initialize_wordle_dictionary_spell_of_life = yes
	# Select random solution from dictionary
	random_in_list = {
		variable = wordle_dictionary
		save_scope_as = selected_wordle_solution
	}
	
	$MINIGAME$ ?= {
		set_variable = {
			name = wordle_solution
			value = scope:selected_wordle_solution
		}

		POD_set_wordle_solution_letter = { N = 1 L = var:wordle_solution.var:wordle_letter_1 }
		POD_set_wordle_solution_letter = { N = 2 L = var:wordle_solution.var:wordle_letter_2 }
		POD_set_wordle_solution_letter = { N = 3 L = var:wordle_solution.var:wordle_letter_3 }
		POD_set_wordle_solution_letter = { N = 4 L = var:wordle_solution.var:wordle_letter_4 }
		POD_set_wordle_solution_letter = { N = 5 L = var:wordle_solution.var:wordle_letter_5 }
		set_variable = {
			name = wordle_solution_word
			value = var:wordle_solution.var:solution_word
		}
		if = { limit = { exists = var:wordle_solution_word } }  # Error suppression
	}

	# With it done, clean out the dictionary
	every_in_list = {
		variable = wordle_dictionary
		end_story = yes
	}
	clear_variable_list = wordle_dictionary
}

POD_set_wordle_solution_letter = {
	set_variable = {
		name = solution_letter_$N$
		value = $L$
	}
	add_to_variable_list = {
		name = wordle_solution_letters
		target = $L$
	}
}

POD_initialize_wordle_word = {
	create_story = { type = POD_story_wordle_solution save_scope_as = wordle_solution }
	scope:wordle_solution ?= {
		POD_initialize_wordle_letter = {N = 1 L = $L1$}
		POD_initialize_wordle_letter = {N = 2 L = $L2$}
		POD_initialize_wordle_letter = {N = 3 L = $L3$}
		POD_initialize_wordle_letter = {N = 4 L = $L4$}
		POD_initialize_wordle_letter = {N = 5 L = $L5$}
		set_variable = {
			name = solution_word
			value = flag:$L1$$L2$$L3$$L4$$L5$
		}
		POD_error_suppression_unused_flag = { FLAG = $L1$$L2$$L3$$L4$$L5$ }
	}
	add_to_variable_list = {
		name = wordle_dictionary
		target = scope:wordle_solution
	}
}

POD_initialize_wordle_letter = {
	set_variable = {
		name = wordle_letter_$N$
		value = flag:$L$
	}
	add_to_variable_list = {
		name = wordle_letters
		target = flag:$L$
	}
	if = { limit = { exists = var:wordle_letters } }  # Error suppression
}
