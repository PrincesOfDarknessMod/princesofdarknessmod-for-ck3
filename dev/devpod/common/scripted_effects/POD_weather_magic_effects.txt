POD_weather_magic_specific_effect = {
	every_held_title = {
		limit = {
			tier = tier_county
		}
		county = {
			POD_weather_magic_effect = yes
		}
	}
	if = {
		limit = {
			is_ai = no
		}
		send_interface_toast = {
			title = POD_weather_magic_from_enemy_toast
			left_icon = scope:recipient
			right_icon = scope:actor
			custom_tooltip = POD_weather_magic_from_enemy.tt
		}
	}
	if = {
		limit = {
			NOT = { this = scope:actor }
		}
		add_opinion = {
			target = scope:actor
			modifier = POD_cast_evil_weather_magic_to_my_realm
		}
	}
}

POD_weather_magic_effect = {
	if = {
		limit = { scope:flow = yes }
		add_county_modifier = {
			modifier = weather_magic_flow_modifier
			years = 5
		}

		change_development_level = -1
	}

			modifier = weather_magic_meteor_shower_modifier

		}
		change_county_control = -20
