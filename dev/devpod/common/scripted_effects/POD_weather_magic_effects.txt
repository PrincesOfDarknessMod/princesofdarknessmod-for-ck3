POD_weather_magic_target_effect = {
	hidden_effect = { every_sub_realm_county = { POD_weather_magic_apply_$TYPE$_effect = yes } }
	save_scope_value_as = { name = type_loca value = flag:type_loca_$TYPE$ }
	primary_title = { save_scope_as = target_title custom_tooltip = POD_weather_magic_general_effect }
	custom_tooltip = POD_weather_magic_general_effect_ex
	show_as_tooltip = { primary_title.title_capital_county = { POD_weather_magic_apply_$TYPE$_effect = yes } }
}
# SCOPE = TARGET LOCATION
# SIZE = COUNTY, DUCHY, KINGDOM, EMPIRE, CONTINENT
# TYPE
POD_weather_magic_general_effect = {
	save_scope_value_as = { name = size_loca value = flag:size_loca_$SIZE$ }
	save_scope_value_as = { name = type_loca value = flag:type_loca_$TYPE$ }
	$SIZE$ = {
		save_scope_as = target_title
		custom_tooltip = { text = POD_weather_magic_general_effect every_de_jure_county = { POD_weather_magic_apply_$TYPE$_effect = yes } }
	}
	custom_tooltip = POD_weather_magic_general_effect_ex
	show_as_tooltip = { county = { POD_weather_magic_apply_$TYPE$_effect = yes } }
}
