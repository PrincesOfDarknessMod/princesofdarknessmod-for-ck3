# The actual effect for running weather magic. character scope (to apply opinion modifiers)
# All the target counties first need to be saved in a list called weather_targets (use one of the effects below for this)
# and the type of weather needs to be saved in a scope value called weather_type (e.g. flag:pod_weather_flood)
# (why not use macro arguments for those? long story, it's ck3 jank)
# arguments:
# MONTHS: the duration of county modifiers (if applicable). if 0, modifiers are permanent
POD_weather_magic_effect = {
	save_scope_as = weather_user
	save_scope_value_as = {
		name  = num_of_targets
		value = list_size:weather_targets
	}
	hidden_effect = {
		every_in_list = {
			list = weather_targets
			POD_apply_weather_effects = { MONTHS = $MONTHS$ }
		}
	}
	ordered_in_list = {
		list = weather_targets
		custom_tooltip = POD_weather_magic_general_effect_ex
		show_as_tooltip = {
			POD_apply_weather_effects = { MONTHS = $MONTHS$ }
		}
	}
	hidden_effect = {
		if = {
			limit = { scope:is_hostile_weather_magic ?= yes }
			POD_clear_list = { LIST = weather_rulers }
			every_in_list = {
				list = weather_targets
				holder = {
					if = {
						limit = { NOT = { this = scope:weather_user } }
						# adding these to a list to prevent duplicates
						add_to_list = weather_rulers
					}
				}
			}
			every_in_list = {
				list = weather_rulers
				send_interface_toast = {
					title = POD_weather_magic_from_enemy_toast
					left_icon = this
					right_icon = scope:weather_user
					custom_tooltip = POD_weather_magic_from_enemy.tt
					add_opinion = {
						target = scope:weather_user
						modifier = POD_casted_harmful_weather_magic
					}
				}
			}
		}
	}
}

# all-in-one macro since these effects are often used together
POD_cast_specific_weather_magic_on_ruler = {
	save_scope_value_as = {
		name  = weather_type
		value = $TYPE$
	}
	POD_setup_weather_magic_counties_from_ruler = {
		RULER = $RULER$
	}
	POD_weather_magic_effect = {
		MONTHS = POD_weather_default_duration
	}
}

# saves all DE JURE counties of a title as targets for weather magic (in the weather_targets list)
# also displays an appropriate tooltip
# arguments:
# TITLE: the de jure title being targeted
POD_setup_weather_magic_counties_from_title = {
	POD_setup_weather_magic_hostility = yes
	POD_clear_weather_magic_targets = yes
	$TITLE$ = {
		save_scope_as = target_title
		every_de_jure_county = {
			POD_add_county_to_weather_magic_targets = yes
		}
	}
	custom_tooltip = POD_weather_magic_general_effect
}

# saves all counties of a geographical region as targets for weather magic (in the weather_targets list)
# also displays an appropriate tooltip
# arguments:
# REGION: the geographical region being targeted
POD_setup_weather_magic_counties_from_region = {
	POD_setup_weather_magic_hostility = yes
	POD_clear_weather_magic_targets = yes
	save_scope_value_as = { name = continent_loca value = flag:$REGION$ }
	every_county_in_region = {
		region = $REGION$
		POD_add_county_to_weather_magic_targets = yes
	}
	custom_tooltip = POD_weather_magic_continent_effect
	POD_error_suppression_unused_flag = { FLAG = $REGION$ }
}

# wrapper for the above effect, selecting the appropriate geographical region from a province
# arguments:
# PROVINCE: the province whose region is being selected
POD_setup_weather_magic_counties_from_province_region = {
	$PROVINCE$ = {
		POD_region_injector = {
			EFFECT = POD_setup_weather_magic_counties_from_region
		}
	}
}

# saves all DE FACTO counties of a ruler's realm as targets for weather magic (in the weather_targets list)
# also displays an appropriate tooltip
# arguments:
# RULER: the ruler being targeted
POD_setup_weather_magic_counties_from_ruler = {
	POD_setup_weather_magic_hostility = yes
	POD_clear_weather_magic_targets = yes
	$RULER$ = {
		primary_title = { save_scope_as = target_title } # only for localization purposes
		every_sub_realm_county = {
			POD_add_county_to_weather_magic_targets = yes
		}
	}
	custom_tooltip = POD_weather_magic_general_effect
}

# if the weather magic is beneficial, it'll go through even if the county's owner is immune to weather magic
# scope:weather_type needs to exist on the TopScope
POD_setup_weather_magic_hostility = {
	if = {
		limit = {
			# add more beneficial weather magic types here if necessary
			OR = {
				scope:weather_type = flag:pod_weather_cycle_of_ra
				scope:weather_type = flag:pod_weather_fertile_land
				scope:weather_type = flag:pod_weather_abundant_flora
				scope:weather_type = flag:pod_weather_calm_animals
				scope:weather_type = flag:pod_weather_nourishing_tide
			}
		}
		save_scope_value_as = {
			name  = is_hostile_weather_magic
			value = no
		}
	}
	else = {
		save_scope_value_as = {
			name  = is_hostile_weather_magic
			value = yes
		}
	}
}

POD_clear_weather_magic_targets = {
	POD_clear_list = { LIST = weather_targets }
}

POD_add_county_to_weather_magic_targets = {
	if = {
		limit = {
			trigger_if = {
				limit = { always = scope:is_hostile_weather_magic }
				POD_title_is_immune_to_weather_magic = no
			}
		}
		add_to_list = weather_targets
	}
}
