POD_set_grand_city_location = {
	add_to_global_variable_map = {
		name  = grand_cities
		key   = $TYPE$
		value = $PROVINCE$
	}
	$PROVINCE$ = {
		set_variable = {
			name  = grand_city_type
			value = $TYPE$
		}
	}
}

POD_initialize_grand_cities_on_game_start = {
	# Baghdad setup
	if = {
		limit = {
			POD_save_grand_city_province = {
				TYPE = flag:baghdad
				SAVE_AS = baghdad_province
			}
		}
		scope:baghdad_province = {
			if = {
				limit = {
					game_start_date >= 1258.1.1
				}
				add_province_modifier = POD_baghdad_sacked_rebuilt_modifier
			}
			else = {
				add_province_modifier = POD_baghdad_complacency_modifier
			}
		}
	}
}

POD_initialize_grand_cities_on_game_start_after_lobby = {
	every_key_in_global_variable_map = {
		variable = grand_cities
		"global_variable_map(grand_cities|this)" = {
			POD_reset_grand_city_in_province = yes
		}
	}
}

POD_reset_grand_city_in_province = {
	switch = {
		trigger = var:grand_city_type
		
		flag:rome = {
			if = {
				limit = { POD_rome_has_wrong_holder = yes }
				POD_reset_rome = yes
			}
		}
		flag:constantinople = {
			if = {
				limit = { POD_constantinople_has_wrong_holder = yes }
				POD_reset_constantinople = yes
			}
		}
		flag:paris = {
			if = {
				limit = { POD_paris_has_wrong_holder = yes }
				POD_reset_paris = yes
			}
		}
		flag:cairo = {
			if = {
				limit = { POD_cairo_has_wrong_holder = yes }
				POD_reset_cairo = yes
			}
		}
		flag:jerusalem = {
			if = {
				limit = { POD_jerusalem_has_wrong_holder = yes }
				POD_reset_jerusalem = yes
			}
		}
		flag:baghdad = {
			if = {
				limit = { POD_baghdad_has_wrong_holder = yes }
				POD_reset_baghdad = yes
			}
		}
		flag:bjarkarey = {
			if = {
				limit = { POD_bjarkarey_has_wrong_holder = yes }
				POD_reset_bjarkarey = yes
			}
		}
		flag:mogadishu = {
			if = {
				limit = { POD_mogadishu_has_wrong_holder = yes }
				POD_reset_mogadishu = yes
			}
		}
	}
}

POD_reset_grand_cities_on_splat_change = {
	every_held_title = {
		limit = {
			tier <= tier_county
			title_province ?= { POD_is_grand_city_location = yes }
		}
		title_province = { POD_reset_grand_city_in_province = yes }
	}
}

POD_clear_grand_city_districts = {
	remove_special_district_slot = yes
	every_built_regular_district = {
		include_inactive = yes
		limit = { POD_is_grand_city_regular_district = yes }
		remove_district = yes
	}
}

POD_reset_rome = {
	if = {
		limit = {
			POD_save_grand_city_province = {
				TYPE = flag:rome
				SAVE_AS = rome_province
			}
		}
		scope:rome_province = {
			county = {
				if = {
					limit = { has_county_modifier = san_michele_modifier }
					remove_county_modifier = san_michele_modifier
				}
			}
			
			POD_clear_grand_city_districts = yes
			
			if = {
				limit = { POD_building_requirement_roma_aeterna = yes }
				POD_build_roma_aeterna_in_province = yes
			}
			else_if = {
				limit = { POD_building_requirement_st_peters_city = yes }
				POD_build_st_peters_city_in_province = yes
			}
			else_if = {
				limit = { POD_building_requirement_eternal_rome = yes }
				POD_build_eternal_rome_in_province = yes
			}
			else = {
				POD_build_tribal_rome_in_province = yes
			}
		}
	}
}

	if = {
		limit = {

				SAVE_AS = constantinople_province

			else_if = {
				limit = { POD_building_requirement_rebuild_constantinople = yes }
				POD_build_rebuild_constantinople_in_province = yes
			}
			else_if = {
				limit = { POD_building_requirement_reform_constantinople = yes }
				POD_build_reform_constantinople_in_province = yes
			}
			else = {
				POD_build_tribal_constantinople_in_province = yes
			}
		}
	}
}

# province scope
POD_constantinople_gain_plunder_reward = {
	if = {
		limit = {
			OR = {
				NOT = { exists = var:plundered_$BUILDING$_level }
				var:plundered_$BUILDING$_level < $LEVEL$
			}
		}
		barony.holder = {
			$REWARD$
		}
		set_variable = {
			name  = plundered_$BUILDING$_level
			value = $LEVEL$
		}
	}
	if = {
		limit = { $LEVEL$ >= 6 }
		trigger_event = POD_maintenance.23
	}
}

POD_constantinople_reset_plundered_building_to_previous_max = {
	if = {
		limit = { NOT = { exists = var:plundered_$BUILDING$_level } }
		set_variable = {
			name  = plundered_$BUILDING$_level
			value = 1
		}
	}
	random_built_regular_district = {
		limit = { district_type = district_type:POD_plundered_constantinople_district }
		add_building = constantinople_plundered_$BUILDING$_01
		if = {
			limit = { prev.var:plundered_$BUILDING$_level >= 2 }
			add_building = constantinople_plundered_$BUILDING$_02
			if = {
				limit = { prev.var:plundered_$BUILDING$_level >= 3 }
				add_building = constantinople_plundered_$BUILDING$_03
				if = {
					limit = { prev.var:plundered_$BUILDING$_level >= 4 }
					add_building = constantinople_plundered_$BUILDING$_04
					if = {
						limit = { prev.var:plundered_$BUILDING$_level >= 5 }
						add_building = constantinople_plundered_$BUILDING$_05
						if = {
							limit = { prev.var:plundered_$BUILDING$_level >= 6 }
							add_building = constantinople_plundered_$BUILDING$_06
						}
					}
				}
			}
		}
	}
}

			}
			else_if = {
				limit = { POD_building_requirement_jewish_jerusalem = yes }

			}
			else_if = {
				limit = { POD_building_requirement_divided_jerusalem = yes }

			}
			else = {
				POD_build_tribal_jerusalem_in_province = yes

POD_add_jerusalem_influence_percentage = {
	POD_save_jerusalem_influence = yes

	POD_save_jerusalem_influence_ranks = {
		CHRISTIAN_SAVE_AS = old_christian_rank
		MUSLIM_SAVE_AS    = old_muslim_rank
		JEWISH_SAVE_AS    = old_jewish_rank
	}

	save_scope_value_as = {
		name = influence_type
		value = flag:$INFLUENCE_LOC$
	}
	POD_error_suppression_unused_flag = { FLAG = $INFLUENCE_LOC$ }

	save_scope_value_as = {
		name  = total_pool
		value = POD_jerusalem_max_total_influence
	}
	# Apply influence gain mult
	if = {
		limit = { scope:influence_type = flag:POD_christian_influence }
		save_scope_value_as = {
			name = influence_gain_mult
			value = POD_jerusalem_christian_influence_gain_mult
		}
	}
	if = {
		limit = { scope:influence_type = flag:POD_muslim_influence }
		save_scope_value_as = {
			name = influence_gain_mult
			value = POD_jerusalem_muslim_influence_gain_mult
		}
	}
	if = {
		limit = { scope:influence_type = flag:POD_jewish_influence }
		save_scope_value_as = {
			name = influence_gain_mult
			value = POD_jerusalem_jewish_influence_gain_mult
		}
	}
	# Enforce the max and min so no single influence exceeds 100 or falls below 0
	if = {
		limit = { $VALUE$ >= 0 }
		save_scope_value_as = {
			name = max_positive_headroom
			value = {
				value = scope:total_pool
				subtract = scope:$VAR$
				subtract = POD_jerusalem_min_influence # Ensures other factions don't drop below the minimum required remaining pool
				min = 0
			}
		}
		save_scope_value_as = {
			name  = value_to_add
			value = {
				value = $VALUE$
				multiply = scope:influence_gain_mult
				max = scope:max_positive_headroom
				min = 0
			}
		}
	}
	else = {
		save_scope_value_as = {
			name = max_negative_headroom
			value = {
				value = 0
				subtract = {
					value = scope:$VAR$
					subtract = POD_jerusalem_min_influence
					min = 0  # prevents negative before I even subtract
				}
			}
		}
		# No modifier to control influence_loss_mult so no multiply param
		save_scope_value_as = {
			name  = value_to_add
			value = {
				value = $VALUE$
				max = 0
				min = scope:max_negative_headroom
			}
		}
	}
	# Calculate pools for proportional reduction
	save_scope_value_as = {
		name = old_remaining_total_pool
		value = {
			value = scope:total_pool
			subtract = scope:$VAR$
			min = 0.00001  # prevent division by 0
		}
	}
	save_scope_value_as = {
		name = new_remaining_total_pool
		value = {
			value = scope:total_pool
			subtract = scope:$VAR$
			subtract = scope:value_to_add
		}
	}

	# Increase VAR by VALUE
	save_scope_value_as = {
		name = $VAR$
		value = {
			value = scope:$VAR$
			add = scope:value_to_add
			min = 0
		}
	}

	# Proportional reduction for non-targets
	if = {
		limit = { NOT = { scope:influence_type = flag:POD_christian_influence } }
		save_scope_value_as = {
			name = christian_influence
			value = {
				value = scope:christian_influence
				multiply = scope:new_remaining_total_pool
				divide = scope:old_remaining_total_pool
				floor = yes
				min = 0
			}
		}
		save_scope_value_as = {
			name = christian_loss
			value = {
				value = scope:christian_influence
				multiply = scope:new_remaining_total_pool
				modulo = scope:old_remaining_total_pool
			}
		}
	}
	if = {
		limit = { NOT = { scope:influence_type = flag:POD_muslim_influence } }
		save_scope_value_as = {
			name = muslim_influence
			value = {
				value = scope:muslim_influence
				multiply = scope:new_remaining_total_pool
				divide = scope:old_remaining_total_pool
				floor = yes
				min = 0
			}
		}
		save_scope_value_as = {
			name = muslim_loss
			value = {
				value = scope:muslim_influence
				multiply = scope:new_remaining_total_pool
				modulo = scope:old_remaining_total_pool
			}
		}
	}
	if = {
		limit = { NOT = { scope:influence_type = flag:POD_jewish_influence } }
		save_scope_value_as = {
			name = jewish_influence
			value = {
				value = scope:jewish_influence
				multiply = scope:new_remaining_total_pool
				divide = scope:old_remaining_total_pool
				floor = yes
				min = 0
			}
		}
		save_scope_value_as = {
			name = jewish_loss
			value = {
				value = scope:jewish_influence
				multiply = scope:new_remaining_total_pool
				modulo = scope:old_remaining_total_pool
			}
		}
	}
	
	# Correct for rounding errors by applying remainder to whichever faction suffered the least fractional loss during rounding
	# Least fractional loss so that non-target faction with largest current influence suffers the greatest decline and experiences the greatest increase
	save_scope_value_as = {
		name = rounding_remainder
		value = {
			value = scope:total_pool
			subtract = scope:christian_influence
			subtract = scope:muslim_influence
			subtract = scope:jewish_influence
			min = 0
		}
	}
	if = {
		limit = { scope:rounding_remainder > 0 }

		if = {
			limit = { scope:influence_type = flag:POD_christian_influence }
			if = {
				limit = { scope:muslim_loss <= scope:jewish_loss }
				save_scope_value_as = {
					name = muslim_influence
					value = { value = scope:muslim_influence add = scope:rounding_remainder }
				}
			}
			else = {
				save_scope_value_as = {
					name = jewish_influence
					value = { value = scope:jewish_influence add = scope:rounding_remainder }
				}
			}
		}

		if = {
			limit = { scope:influence_type = flag:POD_muslim_influence }
			if = {
				limit = { scope:christian_loss <= scope:jewish_loss }
				save_scope_value_as = {
					name = christian_influence
					value = { value = scope:christian_influence add = scope:rounding_remainder }
				}
			}
			else = {
				save_scope_value_as = {
					name = jewish_influence
					value = { value = scope:jewish_influence add = scope:rounding_remainder }
				}
			}
		}

		if = {
			limit = { scope:influence_type = flag:POD_jewish_influence }
			if = {
				limit = { scope:christian_loss <= scope:muslim_loss }
				save_scope_value_as = {
					name = christian_influence
					value = { value = scope:christian_influence add = scope:rounding_remainder }
				}
			}
			else = {
				save_scope_value_as = {
					name = muslim_influence
					value = { value = scope:muslim_influence add = scope:rounding_remainder }
				}
			}
		}
	}


	# Where var:christian_influence, var:muslim_influence, and var:jewish_influence are set
	POD_save_jerusalem_influence_ranks = {
		CHRISTIAN_SAVE_AS = new_christian_rank
		MUSLIM_SAVE_AS    = new_muslim_rank
		JEWISH_SAVE_AS    = new_jewish_rank
	}
	title:c_jerusalem = {
		set_variable = {
			name  = christian_influence
			value = scope:christian_influence
		}
		set_variable = {
			name  = muslim_influence
			value = scope:muslim_influence
		}
		set_variable = {
			name  = jewish_influence
			value = scope:jewish_influence
		}
	}

	# Influence gain/loss tooltip
	save_scope_value_as = {
		name  = influence_delta
		value = scope:value_to_add
	}
	if = {
		limit = { scope:value_to_add >= 0 }
		custom_tooltip = jerusalem_influence_gain
	}
	else = {
		custom_tooltip = jerusalem_influence_loss
	}

	# Reapply/recalculate the province modifiers
	POD_reapply_jerusalem_modifiers = {
		OLD_CHRISTIAN_RANK = scope:old_christian_rank
		OLD_MUSLIM_RANK    = scope:old_muslim_rank
		OLD_JEWISH_RANK    = scope:old_jewish_rank
		NEW_CHRISTIAN_RANK = scope:new_christian_rank
		NEW_MUSLIM_RANK    = scope:new_muslim_rank
		NEW_JEWISH_RANK    = scope:new_jewish_rank
	}
}

POD_reset_baghdad = {
	if = {
		limit = {
			POD_save_grand_city_province = {
				TYPE = flag:baghdad
				SAVE_AS = baghdad_province
			}
		}
		scope:baghdad_province = {
			POD_clear_grand_city_districts = yes

			if = {
				limit = { POD_building_requirement_looted_baghdad = yes }
				POD_build_looted_baghdad_in_province = yes
			}
			else_if = {
				limit = { POD_building_requirement_the_shadowed_pheonix_ctesiphon = yes }
				POD_build_the_shadowed_pheonix_ctesiphon_in_province = yes
			}
			else_if = {
				limit = { POD_building_requirement_jewel_of_allah_baghdad = yes }
				POD_build_jewel_of_allah_baghdad_in_province = yes
			}
			else_if = {
				limit = { POD_building_requirement_sinful_baghdad = yes }
				POD_build_baghdad_sinful_in_province = yes
			}
			else_if = {
				limit = { POD_building_requirement_rebuilt_baghdad = yes }
				POD_build_rebuilt_baghdad_in_province = yes
			}
			else = {
				POD_build_tribal_baghdad_in_province = yes
			}
		}
	}
}

POD_baghdad_mongol_sack_effect = {
	if = {
		limit = {
			POD_save_grand_city_province = {
				TYPE = flag:baghdad
				SAVE_AS = baghdad_province
			}
		}
		scope:baghdad_province = {
			if = {
				limit = { has_province_modifier = POD_baghdad_complacency_modifier }
				remove_province_modifier = POD_baghdad_complacency_modifier
			}
			add_province_modifier = POD_baghdad_looted_1_modifier
			if = {
				limit = { has_building_or_higher = prince_city_01 }
				duchy_capital_district = {
					random_district_building = {
						limit = { is_building_or_higher = prince_city_01 }
						remove_this_building = yes
					}
					add_building = prince_city_01
				}
			}
			county = {
				change_county_control = -100
				change_prosperity_level = -50
			}
		}
	}
	trigger_event = { id = podgrandcity.202 years = 5 }
	custom_tooltip = {
		text = podgrandcity.201.tooltip.a
		POD_reset_baghdad = yes
	}
}

POD_reset_bjarkarey = {
	if = {
		limit = {
			POD_save_grand_city_province = {
				TYPE = flag:bjarkarey
				SAVE_AS = bjarkarey_province
			}
		}
		scope:bjarkarey_province = {
			POD_clear_grand_city_districts = yes

			if = {
				limit = { POD_building_requirement_hollr_bjarkarey = yes }
				POD_build_hollr_bjarkarey_in_province = yes
			}
			else_if = {
				limit = { POD_building_requirement_hvitakristr_bjarkarey = yes }
				POD_build_hvitakristr_bjarkarey_in_province = yes
			}
			else_if = {
				limit = { POD_building_requirement_taema_bjarkarey = yes }
				POD_build_taema_bjarkarey_in_province = yes
			}
			else = {
				POD_build_tribal_bjarkarey_in_province = yes
			}
		}
	}
}

POD_reset_mogadishu = {
	if = {
		limit = {
			POD_save_grand_city_province = {
				TYPE = flag:mogadishu
				SAVE_AS = mogadishu_province
			}
		}
		scope:mogadishu_province = {
			POD_clear_grand_city_districts = yes
			
			POD_clear_mogadishu_values = yes
			
			if = {
				limit = { POD_building_requirement_mogadishu_sarapion_pearl = yes }
				POD_build_sarapion_pearl_in_province = yes
			}
			else_if = {
				limit = { POD_building_requirement_mogadishu_caines_pearl = yes }
				POD_build_caines_pearl_in_province = yes
			}
			else_if = {
				limit = { POD_building_requirement_mogadishu_zanj = yes }
				POD_build_zanj_in_province = yes
			}
			else = {
				POD_build_tribal_mogadishu_in_province = yes
			}
		}
	}
}
