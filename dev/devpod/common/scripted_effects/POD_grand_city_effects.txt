POD_add_building_and_expand_slots = {
	if = {
		limit = { has_free_building_slot = no }
		save_scope_value_as = {
			name  = added_building_slots
			value = 0
		}
		while = {
			limit = { has_free_building_slot = no }
			add_province_modifier = extra_building_slot
			POD_increment_scope_value = { SCOPE = added_building_slots }
		}
		add_building = $BUILDING$
		while = {
			count = scope:added_building_slots
			remove_province_modifier = extra_building_slot
		}
	}
	else = {
		add_building = $BUILDING$
	}
}

# province scope
POD_constantinople_gain_plunder_reward = {
	if = {
		limit = {
			OR = {
				NOT = { exists = var:plundered_$BUILDING$_level }
				var:plundered_$BUILDING$_level < $LEVEL$
			}
		}
		barony.holder = {
			$REWARD$
		}
		set_variable = {
			name  = plundered_$BUILDING$_level
			value = $LEVEL$
		}
	}
	if = {
		limit = { $LEVEL$ >= 6 }
		barony.holder = { trigger_event = POD_maintenance.23 }
	}
}

POD_constantinople_reset_plundered_building_to_previous_max = {
	if = {
		limit = { NOT = { exists = var:plundered_$BUILDING$_level } }
		set_variable = {
			name  = plundered_$BUILDING$_level
			value = 1
		}
	}
	add_building = constantinople_plundered_$BUILDING$_01
	if = {
		limit = { var:plundered_$BUILDING$_level >= 2 }
		add_building = constantinople_plundered_$BUILDING$_02
		if = {
			limit = { var:plundered_$BUILDING$_level >= 3 }
			add_building = constantinople_plundered_$BUILDING$_03
			if = {
				limit = { var:plundered_$BUILDING$_level >= 4 }
				add_building = constantinople_plundered_$BUILDING$_04
				if = {
					limit = { var:plundered_$BUILDING$_level >= 5 }
					add_building = constantinople_plundered_$BUILDING$_05
					if = {
						limit = { var:plundered_$BUILDING$_level >= 6 }
						add_building = constantinople_plundered_$BUILDING$_06
					}
				}
			}
		}
	}
}

POD_add_jerusalem_influence_percentage = {
	POD_save_jerusalem_influence = yes

	POD_save_jerusalem_influence_ranks = {
		CHRISTIAN_SAVE_AS = old_christian_rank
		MUSLIM_SAVE_AS    = old_muslim_rank
		JEWISH_SAVE_AS    = old_jewish_rank
	}

	save_scope_value_as = {
		name = influence_type
		value = flag:$INFLUENCE_LOC$
	}
	POD_error_suppression_unused_flag = { FLAG = $INFLUENCE_LOC$ }

	save_scope_value_as = {
		name  = total_pool
		value = POD_jerusalem_max_total_influence
	}
	# Apply influence gain mult
	if = {
		limit = { scope:influence_type = flag:POD_christian_influence }
		save_scope_value_as = {
			name = influence_gain_mult
			value = POD_jerusalem_christian_influence_gain_mult
		}
	}
	if = {
		limit = { scope:influence_type = flag:POD_muslim_influence }
		save_scope_value_as = {
			name = influence_gain_mult
			value = POD_jerusalem_muslim_influence_gain_mult
		}
	}
	if = {
		limit = { scope:influence_type = flag:POD_jewish_influence }
		save_scope_value_as = {
			name = influence_gain_mult
			value = POD_jerusalem_jewish_influence_gain_mult
		}
	}
	# Enforce the max and min so no single influence exceeds 100 or falls below 0
	if = {
		limit = { $VALUE$ >= 0 }
		save_scope_value_as = {
			name = max_positive_headroom
			value = {
				value = scope:total_pool
				subtract = scope:$VAR$
				subtract = POD_jerusalem_min_influence # Ensures other factions don't drop below the minimum required remaining pool
				min = 0
			}
		}
		save_scope_value_as = {
			name  = value_to_add
			value = {
				value = $VALUE$
				multiply = scope:influence_gain_mult
				max = scope:max_positive_headroom
				min = 0
			}
		}
	}
	else = {
		save_scope_value_as = {
			name = max_negative_headroom
			value = {
				value = 0
				subtract = {
					value = scope:$VAR$
					subtract = POD_jerusalem_min_influence
					min = 0  # prevents negative before I even subtract
				}
			}
		}
		# No modifier to control influence_loss_mult so no multiply param
		save_scope_value_as = {
			name  = value_to_add
			value = {
				value = $VALUE$
				max = 0
				min = scope:max_negative_headroom
			}
		}
	}
	# Calculate pools for proportional reduction
	save_scope_value_as = {
		name = old_remaining_total_pool
		value = {
			value = scope:total_pool
			subtract = scope:$VAR$
			min = 0.00001  # prevent division by 0
		}
	}
	save_scope_value_as = {
		name = new_remaining_total_pool
		value = {
			value = scope:total_pool
			subtract = scope:$VAR$
			subtract = scope:value_to_add
		}
	}

	# Increase VAR by VALUE
	save_scope_value_as = {
		name = $VAR$
		value = {
			value = scope:$VAR$
			add = scope:value_to_add
			min = 0
		}
	}

	# Proportional reduction for non-targets
	if = {
		limit = { NOT = { scope:influence_type = flag:POD_christian_influence } }
		save_scope_value_as = {
			name = christian_influence
			value = {
				value = scope:christian_influence
				multiply = scope:new_remaining_total_pool
				divide = scope:old_remaining_total_pool
				floor = yes
				min = 0
			}
		}
		save_scope_value_as = {
			name = christian_loss
			value = {
				value = scope:christian_influence
				multiply = scope:new_remaining_total_pool
				modulo = scope:old_remaining_total_pool
			}
		}
	}
	if = {
		limit = { NOT = { scope:influence_type = flag:POD_muslim_influence } }
		save_scope_value_as = {
			name = muslim_influence
			value = {
				value = scope:muslim_influence
				multiply = scope:new_remaining_total_pool
				divide = scope:old_remaining_total_pool
				floor = yes
				min = 0
			}
		}
		save_scope_value_as = {
			name = muslim_loss
			value = {
				value = scope:muslim_influence
				multiply = scope:new_remaining_total_pool
				modulo = scope:old_remaining_total_pool
			}
		}
	}
	if = {
		limit = { NOT = { scope:influence_type = flag:POD_jewish_influence } }
		save_scope_value_as = {
			name = jewish_influence
			value = {
				value = scope:jewish_influence
				multiply = scope:new_remaining_total_pool
				divide = scope:old_remaining_total_pool
				floor = yes
				min = 0
			}
		}
		save_scope_value_as = {
			name = jewish_loss
			value = {
				value = scope:jewish_influence
				multiply = scope:new_remaining_total_pool
				modulo = scope:old_remaining_total_pool
			}
		}
	}
	
	# Correct for rounding errors by applying remainder to whichever faction suffered the least fractional loss during rounding
	# Least fractional loss so that non-target faction with largest current influence suffers the greatest decline and experiences the greatest increase
	save_scope_value_as = {
		name = rounding_remainder
		value = {
			value = scope:total_pool
			subtract = scope:christian_influence
			subtract = scope:muslim_influence
			subtract = scope:jewish_influence
			min = 0
		}
	}
	if = {
		limit = { scope:rounding_remainder > 0 }

		if = {
			limit = { scope:influence_type = flag:POD_christian_influence }
			if = {
				limit = { scope:muslim_loss <= scope:jewish_loss }
				save_scope_value_as = {
					name = muslim_influence
					value = { value = scope:muslim_influence add = scope:rounding_remainder }
				}
			}
			else = {
				save_scope_value_as = {
					name = jewish_influence
					value = { value = scope:jewish_influence add = scope:rounding_remainder }
				}
			}
		}

		if = {
			limit = { scope:influence_type = flag:POD_muslim_influence }
			if = {
				limit = { scope:christian_loss <= scope:jewish_loss }
				save_scope_value_as = {
					name = christian_influence
					value = { value = scope:christian_influence add = scope:rounding_remainder }
				}
			}
			else = {
				save_scope_value_as = {
					name = jewish_influence
					value = { value = scope:jewish_influence add = scope:rounding_remainder }
				}
			}
		}

		if = {
			limit = { scope:influence_type = flag:POD_jewish_influence }
			if = {
				limit = { scope:christian_loss <= scope:muslim_loss }
				save_scope_value_as = {
					name = christian_influence
					value = { value = scope:christian_influence add = scope:rounding_remainder }
				}
			}
			else = {
				save_scope_value_as = {
					name = muslim_influence
					value = { value = scope:muslim_influence add = scope:rounding_remainder }
				}
			}
		}
	}


	# Where var:christian_influence, var:muslim_influence, and var:jewish_influence are set
	POD_save_jerusalem_influence_ranks = {
		CHRISTIAN_SAVE_AS = new_christian_rank
		MUSLIM_SAVE_AS    = new_muslim_rank
		JEWISH_SAVE_AS    = new_jewish_rank
	}
	title:c_jerusalem = {
		set_variable = {
			name  = christian_influence
			value = scope:christian_influence
		}
		set_variable = {
			name  = muslim_influence
			value = scope:muslim_influence
		}
		set_variable = {
			name  = jewish_influence
			value = scope:jewish_influence
		}
	}

	# Influence gain/loss tooltip
	save_scope_value_as = {
		name  = influence_delta
		value = scope:value_to_add
	}
	if = {
		limit = { scope:value_to_add >= 0 }
		custom_tooltip = jerusalem_influence_gain
	}
	else = {
		custom_tooltip = jerusalem_influence_loss
	}

	# Reapply/recalculate the province modifiers
	POD_reapply_jerusalem_modifiers = {
		OLD_CHRISTIAN_RANK = scope:old_christian_rank
		OLD_MUSLIM_RANK    = scope:old_muslim_rank
		OLD_JEWISH_RANK    = scope:old_jewish_rank
		NEW_CHRISTIAN_RANK = scope:new_christian_rank
		NEW_MUSLIM_RANK    = scope:new_muslim_rank
		NEW_JEWISH_RANK    = scope:new_jewish_rank
	}
}

		title_province = {
			set_holding_type = tribal_holding
			set_holding_type = grand_city_holding

			# the set_holding_type tribal->grand_city cycle above, which wipes the
			# regular sub-buildings. Mirrors POD_reset_rome_effect. All three versions'
			# specials are cleared so a switch from any version to any other is clean.

			POD_remove_building_effect = { BUILDING = bjarkarey_hvitakristr_02 }
			POD_remove_building_effect = { BUILDING = bjarkarey_hvitakristr_03 }

			POD_remove_building_effect = { BUILDING = bjarkarey_taema_01 }
			POD_remove_building_effect = { BUILDING = bjarkarey_taema_02 }
			POD_remove_building_effect = { BUILDING = bjarkarey_taema_03 }

			if = {
