POD_grand_city_initialization_effect = {
	if = {
		limit = { rome_has_wrong_holder_trigger = yes }
		POD_reset_rome_effect = yes
	}
	if = {
		limit = { constantinople_has_wrong_holder_trigger = yes }
		POD_reset_constantinople_effect = yes
	}
}

POD_reset_grand_city_effect = {
	set_holding_type = grand_city_holding
	if = {
		limit = { barony = title:b_roma }
		POD_reset_rome_effect = yes
	}
	if = {
		limit = { barony = title:b_constantinople }
		POD_reset_constantinople_effect = yes
	}
}

POD_reset_grand_cities_on_splat_change = {
	# Only Rome is determined by your splat
	if = {
		limit = { has_title = title:c_roma }
		POD_reset_rome_effect = yes
	}
}

POD_reset_rome_effect = {
	title:b_roma = {
		county = {
			if = {
				limit = { has_county_modifier = san_michele_modifier }
				remove_county_modifier = san_michele_modifier
			}
		}
		title_province = {
			set_holding_type = tribal_holding
			set_holding_type = grand_city_holding
			
			if = {
				limit = { building_requirement_roma_aeterna = yes }
				POD_build_roma_aeterna_effect = yes
			}
			else_if = {
				limit = { building_requirement_st_peters_city = yes }
				POD_build_st_peters_city_effect = yes
			}
			else_if = {
				limit = { building_requirement_eternal_rome = yes }
				POD_build_eternal_rome_effect = yes
			}
			else = {
				POD_build_tribal_rome_effect = yes
			}
		}
	}
}

POD_build_roma_aeterna_effect = {
	title:b_roma.title_province = {
		add_building = rome_roma_aeterna_01
	}
}

POD_build_st_peters_city_effect = {
	title:b_roma.title_province = {
		add_building = rome_st_peters_city_01
		if = {
			# give the AI the buildings immediately because they're slow at decisions
			limit = { barony_controller = { is_ai = yes } }
			POD_build_san_michele_effect = yes
		}
	}
}

POD_build_eternal_rome_effect = {
	title:b_roma.title_province = {
		add_building = rome_eternal_rome_01
		add_building = rome_colosseum_01
		add_building = rome_hadrianstomb_01
		add_building = rome_aqueduct_01
		#add_building = rome_pasquino_01 # wasn't unearthed until the 15th century
	}
}

# Triggered via decision, not automatically
POD_build_san_michele_effect = {
	title:b_roma = {
		county = {
			if = {
				limit = { NOT = { has_county_modifier = san_michele_modifier } }
				add_county_modifier = san_michele_modifier
			}
		}
		title_province = {
			# Build rank 1 buildings (minor bonuses)
			add_building = rome_san_michele_library_01
			add_building = rome_san_michele_aedificium_01
			add_building = rome_san_michele_reliquary_01
			
			if = {
				limit = { has_free_building_slot = no } 
				add_province_modifier = extra_building_slot
				add_building = rome_san_michele_basilica_01
				remove_province_modifier = extra_building_slot
			}
			else = {
				add_building = rome_san_michele_basilica_01
			}
			
			if = {
				limit = { has_free_building_slot = no } 
				add_province_modifier = extra_building_slot
				add_building = rome_san_michele_cloisters_01
				remove_province_modifier = extra_building_slot
			}
			else = {
				add_building = rome_san_michele_cloisters_01
			}
		}
	}
}

POD_build_tribal_rome_effect = {
	title:b_roma.title_province = {
		set_holding_type = tribal_holding
		add_building = rome_tiber_01
	}
}

POD_reset_constantinople_effect = {
	title:b_constantinople = {
		title_province = {
			set_holding_type = tribal_holding
			set_holding_type = grand_city_holding
			
			if = {
				limit = { building_plundered_constantinople = yes }
				POD_build_plundered_constantinople_effect = yes
			}
			else_if = {
				limit = { building_rebuild_constantinople = yes }
				POD_build_rebuild_constantinople_effect = yes
			}
			else_if = {
				limit = { building_reform_constantinople = yes }
				POD_build_reform_constantinople_effect = yes
			}
			else = {
				POD_build_tribal_constantinople_effect = yes
			}
		}
	}
}

POD_build_plundered_constantinople_effect = {
	title:b_constantinople.title_province = {
		if = { limit = { NOT = { has_building_or_higher = constantinople_plundered_constantinople_01 } } add_building = constantinople_plundered_constantinople_01 }
		add_building = constantinople_plundered_great_palace_01
		add_building = constantinople_plundered_the_hippodrome_01
		add_building = constantinople_plundered_harbor_of_sophia_01
		add_building = constantinople_plundered_the_great_bazar_01
		add_building = constantinople_plundered_hagia_sophia_01
		add_building = constantinople_plundered_monastery_of_st_myrelaion_01
		add_building = constantinople_plundered_seraglio_lighthouse_01
		add_building = constantinople_plundered_aqueduct_of_valens_01
	}
}

POD_build_reform_constantinople_effect = {

			add_building = constantinople_reform_great_palace_01
			add_building = constantinople_reform_the_hippodrome_01
			add_building = constantinople_reform_harbor_of_sophia_01
			add_building = constantinople_reform_the_great_bazar_01
			add_building = constantinople_reform_hagia_sophia_01
			add_building = constantinople_reform_monastery_of_st_myrelaion_01
			add_building = constantinople_reform_seraglio_lighthouse_01
			add_building = constantinople_reform_aqueduct_of_valens_01
		}

		}
	}
}
