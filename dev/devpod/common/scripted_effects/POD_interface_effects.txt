		name  = cooldown_loc
		value = flag:$LOC$
	}
	if = {
		limit = { scope:cooldown_loc = flag:$LOC$ } # error suppression
		trigger_event = {
			id = POD_interface.1
			$COOLDOWN$
		}
	}
}

# save the last created artifact as a link
# the link can be used in datafunctions with the macro GetPODArtifact(link)
# or in script via POD_character_owns_linked_artifact
POD_save_new_artifact_as_link = {
	character:POD_wraith_01 ?= {
		set_variable = {
			name = $LINK$
			value = scope:newly_created_artifact
		}
		# error suppression
		if = {
			limit = {
				always = no
				has_variable = $LINK$
			}
		}
	}
}

# character scope (the character who'll see the selection event)
# arguments:
# - TARGET (the character whose secrets to save)
# - FILTER_UNKNOWN (yes/no, yes means that only secrets known to the first character are saved)
POD_save_secrets_to_list_for_event = {
	save_scope_as = secret_root
	$TARGET$ = {
		every_secret = {
			limit = {
				trigger_if = {
					limit = { always = $FILTER_UNKNOWN$ }
					is_known_by = scope:secret_root
				}
				trigger_else = { always = yes }
			}
			scope:secret_root = {
				add_to_variable_list = {
					name = event_secrets
					target = prev
				}
			}
		}
	}
	ordered_in_list = {
		variable = event_secrets
		prev = {
			set_variable = {
				name = event_secret_selection
				value = prev
			}
		}
	}
}
