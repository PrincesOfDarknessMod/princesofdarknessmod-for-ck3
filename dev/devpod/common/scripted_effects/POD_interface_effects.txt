# macro with 2 arguments
# LOC: localization key for the content of the interface message
# COOLDOWN: cooldown value, needs to be passed in quotation marks. e.g.: COOLDOWN = "years = 5"
POD_queue_cooldown_message = {
	save_scope_value_as = {
		name  = cooldown_loc
		value = flag:$LOC$
	}
	if = {
		limit = { scope:cooldown_loc = flag:$LOC$ } # error suppression
		trigger_event = {
			id = POD_interface.1
			$COOLDOWN$
		}
	}
}

# used by common\scripted_guis\POD_portrait_shader_guis.txt
POD_toggle_portrait_shader_var = {
	if = {
		limit = { has_variable = $VAR$ }
		remove_variable = $VAR$
	}
	else = {
		set_variable = {
			name  = $VAR$
			value = 1.0 # maximum value that shaders can read
		}
	}
}

# used by common\scripted_guis\POD_portrait_shader_guis.txt
# need to pass scope:delta to the SGUI
POD_edit_portrait_shader_var = {
	if = {
		limit = { NOT = { has_variable = $VAR$ } }
		set_variable = { name = $VAR$ value = 0.5 }
	}
	set_variable = {
		name = $VAR$
		value = {
			value = var:$VAR$
			add = scope:delta
			min = 0.004 # minimum value (not zero so that the decal doesn't disappear)
			max = 1.0   # maximum value that shaders can read
		}
	}
}

POD_toggle_clothing_shader_var = {

}

POD_edit_clothing_shader_var = {

POD_backup_all_house_colors = {
	POD_clothing_color_injector = { EFFECT = POD_backup_house_color }
}

POD_backup_house_color = {
	if = {
		limit = { house ?= { has_variable = $VAR$ } }
		set_variable = {
			name  = $VAR$_backup
			value = house.var:$VAR$
		}
	}
}
