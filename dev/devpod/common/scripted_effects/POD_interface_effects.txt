# macro with 2 arguments
# LOC: localization key for the content of the interface message
# COOLDOWN: cooldown value, needs to be passed in quotation marks. e.g.: COOLDOWN = "years = 5"
POD_queue_cooldown_message = {
	save_scope_value_as = {
		name  = cooldown_loc
		value = flag:$LOC$
	}
	if = {
		limit = { scope:cooldown_loc = flag:$LOC$ } # error suppression
		trigger_event = {
			id = POD_interface.1
			$COOLDOWN$
		}
	}
}

POD_clear_resource_hint_vars_effect = {
	# Blood
	remove_variable = hunt_in_capital_story
	remove_variable = suggested_herdmember
	# Resolve
	remove_variable = suggested_revelation_target
}

POD_save_potential_mortal_herd_effect = {
	ordered_courtier_or_guest = {
		limit = {
			prev = {
				is_character_interaction_valid = {
					recipient = prev
					interaction = POD_herdify_mortal_interaction
				}
			}
		}
		order_by = resonance_rating_value
		position = 0
		save_scope_as = potential_mortal_herd
	}
}

POD_save_potential_dreamer_effect = {
	ordered_courtier_or_guest = {
		limit = {
			prev = {
				is_character_interaction_valid = {
					recipient = prev
					interaction = POD_musing_mortal_interaction
				}
			}

		}

	}
}

POD_save_potential_host_effect = {
	ordered_relation = {
		type = worshipping_cultist
		limit = { 
			can_have_children = yes
			trigger_if = {
				limit = { root = { is_great_demon_trigger = yes } }
				POD_can_be_possessed_trigger = yes
			}
			trigger_else = {
				can_be_demonic_possesed = yes
			}
			NOT = { has_character_modifier = excluded_from_possession_modifier }
			is_ai = yes
		}
		alternative_limit = { 
			trigger_if = {
				limit = { root = { is_great_demon_trigger = yes } }
				POD_can_be_possessed_trigger = yes
			}
			trigger_else = {
				can_be_demonic_possesed = yes
			}
			NOT = { has_character_modifier = excluded_from_possession_modifier }
			is_ai = yes
		}
		order_by = sum_of_all_skills_value
		position = 0
		save_scope_as = potential_host
	}
}
