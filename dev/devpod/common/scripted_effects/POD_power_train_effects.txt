
POD_masstrain_discipline_effect = {
	while = {
		count = $LIFESTYLE$_perk_half_points
		custom_tooltip = {
			text = POD_train_$DISCIPLINE$
			random_in_list = {
				list = disciplinetrainee
				limit = { has_$DISCIPLINE$_trigger = no }
				add_opinion = {
					target = prev
					modifier = mentored_me_opinion
				}
				add_trait = $DISCIPLINE$
				prev = { subtract_thousand_xp_effect = { LIFESTYLE = $LIFESTYLE$ } }
			}
		}
		POD_side_effects_from_mentoring_effect = yes
	}
}

POD_masstrain_tree_effect = {
	while = {
		count = $LIFESTYLE$_perk_points
		custom_tooltip = {
			text = POD_train_$TREE$
			random_in_list = {
				list = disciplinetrainee
				limit = { perks_in_tree = { tree = $TREE$ value < 5 } }
				add_opinion = {
					target = prev
					modifier = mentored_me_opinion
				}
				assign_$TREE$_effect = yes
				prev = { add_$LIFESTYLE$_perk_points = -1 }
			}
		}
		POD_side_effects_from_mentoring_effect = yes
	}
}

subtract_thousand_xp_effect = {
	if = { limit = { NOT = { has_perk = cycle_of_the_midgard_serpent_perk } } # is there better way? of course. Will i do it better way? No way. 
		if = {
			limit = { $LIFESTYLE$_perk_points > 0 }
			add_$LIFESTYLE$_perk_points = -1
			add_$LIFESTYLE$_xp = 1000
		} 
		else = {
			add_$LIFESTYLE$_xp = -1000
		}
	}
	else = {
		custom_tooltip = cycle_of_the_midgard_serpent_perk_tt
	}
}

POD_side_effects_from_mentoring_effect = {
	
	if = { 
		limit = { POD_should_keep_track_of_taught_perk_points = yes }
		POD_set_or_change_variable = { VAR = taught_perk_points VAL = 1 }
	}
}
