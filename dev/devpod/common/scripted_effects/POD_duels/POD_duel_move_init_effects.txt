POD_init_duel_move = {
	# stories need to be initialized on a character, so let's use charon.
	# since he's immortal, off-map, untargetable, and alive in every bookmark
	character:POD_wraith_01 = {
		create_story = {
			type = POD_duel_move
			save_scope_as = new_duel_move
		}
		
		scope:new_duel_move = {
			set_variable = {
				name = decision
				value = decision:$DECISION$
			}
			set_variable = {
				name = category
				value = flag:$CATEGORY$
			}
			set_variable = {
				name = ap_cost
				value = $AP_COST$
			}
			# single target assumed to be the default
			set_variable = {
				name = target
				value = flag:single_enemy
			}
			# add other mandatory modules if necessary
		}
		
		# add the story_cycle to the global list of all duel moves
		add_to_global_variable_list = {
			name = POD_duel_moves
			target = scope:new_duel_move
		}
		
		# we also save each individual duel move as a variable on charon.
		# this will help us later to auto-generate tooltips for them
		set_variable = {
			name  = $DECISION$
			value = scope:new_duel_move
		}
		
		# this does nothing, it's just to stop the error log from complaining.
		# can be removed once those variables are used in script
		if = {
			limit = {
				always = no
				POD_can_execute_duel_move = { STORY = scope:new_duel_move }
				has_variable = category
				has_variable = $DECISION$
				has_variable = damage_type
				has_variable = healing_type
			}
			POD_error_suppression_unused_flag = { FLAG = single_enemy }
			POD_error_suppression_unused_flag = { FLAG = $CATEGORY$ }
			POD_error_suppression_unused_flag = { FLAG = superficial }
			POD_error_suppression_unused_flag = { FLAG = aggravated }
		}
	}
}

# run when loading an old save
POD_reload_duel_moves = {
	every_player = {
		clear_variable_list = POD_duel_moves
	}
	every_in_global_list = {
		variable = POD_duel_moves
		end_story = yes
	}
	clear_global_variable_list = POD_duel_moves
	POD_initialize_duel_moves = yes
}


POD_duel_move_set_target = {
	set_variable = {
		name = target
		value = flag:$TYPE$
	}
	POD_error_suppression_unused_flag = { FLAG = $TYPE$ }
}
