# these values are reader_export macros.
# sort of like constants in other programming languages.
# if all the values are in one place, it makes rebalancing them easier.
#
# let's say, for example, you want to try making AP more granular.
# so, doubling both the max AP and the AP costs of every duel move.
# this would be very tedious if there are 400+ duel moves.
# but if the values are named constants, you can just double all the constants.

@ap_cost_tiny		= 1
@ap_cost_minor		= 2
@ap_cost_medium		= 3
@ap_cost_major		= 4
@ap_cost_massive	= 5
@ap_cost_maximum	= 6

# energy values are splat-agnostic so they can be used by anyone
# they assume a "mana bar" where the maximum is 10 (e.g. demon resolve)
# other values are scaled accordingly
# e.g. 4 "energy" is equivalent to 2 hunger or 40 mental strain
@energy_cost_medium	= 2
@energy_cost_major	= 4

@piety_cost_minor	= 50
@piety_cost_medium	= 100
@piety_cost_major	= 250

@damage_minor	= 1
@damage_medium	= 2
@damage_major	= 3
@damage_massive	= 4

@healing_minor		= 1
@healing_medium		= 2
@healing_major		= 3
@healing_massive	= 4


# runs at the start of the game and puts all duel moves in a global list
POD_initialize_duel_moves = {

	# if you change or add duel moves, increment the script_value POD_current_duel_version.
	# if script_value doesn't match the global variable that was set at the start of the game,
	# that means the player loaded an old save.
	# in which case, we can reload the global variable list.
	set_global_variable = {
		name  = POD_duel_version
		value = POD_current_duel_version
	}

####### Basic

	# the principle here is that only the necessary attributes get set.
	# all duel moves need:
	# - a decision, which contains loc, an illustration, and splat triggers
	# - a category (currently unused but can be used to filter moves)
	# - an AP cost
	# add other attributes in case they're mandatory.
	# though, if an attribute is mandatory but often has the same value,
	# you can initialize it with a default value in POD_init_duel_move.
	POD_init_duel_move = {
		DECISION = POD_duel_move_strike
		CATEGORY = POD_basic
		AP_COST  = @ap_cost_medium
	}
	scope:new_duel_move = {
		# optional attributes get set here
		POD_duel_move_set_damage = {
			VALUE = @damage_medium
		}
	}

	POD_init_duel_move = {
		DECISION = POD_duel_move_heavy_blow
		CATEGORY = POD_basic
		AP_COST  = @ap_cost_massive
	}
	scope:new_duel_move = {
		POD_duel_move_set_damage = {
			VALUE = @damage_massive
		}
	}

	POD_init_duel_move = {
		DECISION = POD_duel_move_quick_jab
		CATEGORY = POD_basic
		AP_COST  = @ap_cost_minor
	}
	scope:new_duel_move = {
		POD_duel_move_set_damage = {
			VALUE = @damage_minor
		}
	}

####### Trait

	# this one's just a random example to demonstrate the piety cost attribute
	# and a more complex trigger from the decision
	POD_init_duel_move = {
		DECISION = POD_duel_move_zornhau
		CATEGORY = POD_trait
		AP_COST  = @ap_cost_medium
	}
	scope:new_duel_move = {
		POD_duel_move_set_damage = {
			VALUE = @damage_major
		}
		POD_duel_move_set_piety_cost = {
			VALUE = @piety_cost_minor
		}
	}

####### Hunter
	
	POD_init_duel_move = {
		DECISION = POD_duel_move_torch_sweep
		CATEGORY = POD_hunter
		AP_COST  = @ap_cost_massive
	}
	scope:new_duel_move = {
		POD_duel_move_set_target_aoe_enemy = yes
		POD_duel_move_set_damage = {
			VALUE = @damage_medium
		}
		# to simplify the demo script i didn't use damage types
		# other than superficial/aggravated
		# but you could add other flags and display them the same way
		POD_duel_move_set_damage_type_aggravated = yes
		POD_duel_move_set_undodgeable = yes
	}

####### Vampire
	
	POD_init_duel_move = {
		DECISION = POD_duel_move_blood_heal
		CATEGORY = POD_vampire
		AP_COST  = @ap_cost_minor
	}
	scope:new_duel_move = {
		POD_duel_move_set_target_self = yes
		POD_duel_move_set_energy_cost = {
			VALUE = @energy_cost_medium
		}
		POD_duel_move_set_healing = {
			VALUE = @healing_medium
		}
	}
}
