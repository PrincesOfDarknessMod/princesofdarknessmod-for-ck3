	if = {

						}
					}
					yang_points_value < 7
				}

						OR = {
							has_character_modifier = yinchi5_modifier 
							has_character_modifier = yinchi4_modifier
						}
					}
					yang_points_value < 5
				}

					limit = { 
						OR = {
							has_character_modifier = yinchi3_modifier 
							has_character_modifier = yinchi2_modifier
						}
					}
					yang_points_value < 3
				}
				trigger_else = {
					OR = {
						has_character_modifier = yinchi10_modifier

					}
				}
			}
		}

	}

POD_absorb_chi_from_flesh_or_blood_effect = {
	$MURDERER$ = { save_temporary_scope_as = bloodloss_attacker }
	if = {
		limit = { $MURDERER$.yin_difference >= 1 }
		if = {
			limit = { $MURDERER$.yin_difference > $VICTIM$.lifeforce_points_value }
			$MURDERER$ = { POD_yinchi_increase_multiple_final_effect = { COUNT = $VICTIM$.lifeforce_points_value } }
			$VICTIM$ = { POD_power_cost_bloodloss_effect = { COUNT = lifeforce_points_value } }
		}
		else = {
			$MURDERER$ = { POD_yinchi_increase_multiple_final_effect = { COUNT = yin_difference } }
			$VICTIM$ = { POD_power_cost_bloodloss_effect = { COUNT = $MURDERER$.yin_difference } }
		}
	}
	if = {
		limit = { $MURDERER$.yang_difference >= 1 }
		if = {
			limit = { $MURDERER$.yang_difference > $VICTIM$.lifeforce_points_value }
			$MURDERER$ = { POD_yangchi_increase_multiple_final_effect = { COUNT = $VICTIM$.lifeforce_points_value } }
			$VICTIM$ = { POD_power_cost_bloodloss_effect = { COUNT = lifeforce_points_value } }
		}
		else = {
			$MURDERER$ = { POD_yangchi_increase_multiple_final_effect = { COUNT = yang_difference } }
			$VICTIM$ = { POD_power_cost_bloodloss_effect = { COUNT = $MURDERER$.yang_difference } }
		}
	}
}

			$VICTIM$ = { POD_universal_yin_energy_decrease_multiple_effect = { COUNT = 10 } }
		}
	}
	else = {
		$MURDERER$ = { POD_yinchi_increase_multiple_final_effect = { COUNT = yin_difference } }
		$VICTIM$ = { POD_universal_yin_energy_decrease_multiple_effect = { COUNT = $MURDERER$.yin_difference } }
	}
}

POD_chi_cost_effect = {
	if = {
		limit = { POD_has_infinite_chi_trigger = yes }
	}
	else_if = {
		limit = {
			any_character_artifact = {
				has_variable = energy_stored
				var:energy_stored > 0
			}
		}
		ordered_character_artifact = {
			limit = {
				has_variable = energy_stored 
				var:energy_stored > 0
			}
			order_by = {
				value = 0
				add = var:energy_stored
			}
			position = 0
			POD_energy_vessel_decrease_multiple_effect = { COUNT = $COUNT$ }
		}
	}
	else = {
		POD_chi_decrease_multiple_effect = { COUNT = $COUNT$ }
	}
}

POD_chi_income_effect = {
	if = {
		limit = {
			OR = {
				chi_difference >= $COUNT$
				NOT = {
					any_character_artifact = {
						NOT = { has_variable = temporary_vessel }
						has_variable = energy_stored
						vessel_empty_energy > 0
					}
				}
			}
		}
		POD_chi_increase_multiple_effect = { COUNT = $COUNT$ }
	}
	else = {
		save_scope_value_as = {
			name = excess_chi
			value = {
				value = $COUNT$
				subtract = chi_difference
			}
		}
		POD_chi_increase_multiple_effect = { COUNT = chi_difference }
		ordered_character_artifact = {
			limit = {
				NOT = { has_variable = temporary_vessel }
				has_variable = energy_stored 
				vessel_empty_energy > 0
			}
			order_by = {
				value = 0
				add = vessel_empty_energy
			}
			position = 0
			POD_energy_vessel_increase_multiple_effect = { COUNT = scope:excess_chi }
		}
	}
}
