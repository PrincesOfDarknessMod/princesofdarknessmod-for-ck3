POD_herdify_effect = {
	save_scope_as = herd_owner
	add_hook = {
		type   = herd_hook
		target = $TARGET$
	}
	$TARGET$ = {
		save_scope_as = herdmember
		POD_set_resonance_effect = yes
		set_relation_herd_owner  = scope:herd_owner

		if = {
			limit = { NOT = { is_courtier_of = scope:herd_owner } }
			scope:herd_owner = { add_courtier = prev }
			hidden_effect = {
				return_to_court = yes
			}
		}
		add_character_flag = blocked_from_leaving
		
		# Herd automation event
		trigger_event = POD_herd_automation.100
	}
}

POD_remove_herd_effect = {
	if = {
		limit = { $TARGET$ = { POD_is_mortal_trigger = yes } }
		POD_minimal_risk_of_increasing_masquerade_exposure = yes
	}
	POD_remove_herd_no_masquerade_effect = { TARGET = $TARGET$ }
}

POD_remove_herd_no_masquerade_effect = {
	remove_hook = {
		target = $TARGET$
		type   = herd_hook
	}
	$TARGET$ = {
		remove_relation_herd_owner = prev
		remove_character_flag = blocked_from_leaving
	}
}
