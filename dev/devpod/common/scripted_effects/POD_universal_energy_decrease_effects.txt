POD_quinary_change_power_cost_change_effect = {
	if = {
		limit = {
			OR = {
				$COUNT$ = 1
				$COUNT$ = 2
			}
		}
		POD_$TYPE$_multiple_effect = { COUNT = 1 }
	}
	else_if = {
		limit = {
			OR = {
				$COUNT$ = 3
				$COUNT$ = 4
			}
		}
		POD_$TYPE$_multiple_effect = { COUNT = 2 }
	}
	else_if = {
		limit = {
			OR = {
				$COUNT$ = 5
				$COUNT$ = 6
			}
		}
		POD_$TYPE$_multiple_effect = { COUNT = 3 }
	}
	else_if = {
		limit = {
			OR = {
				$COUNT$ = 7
				$COUNT$ = 8
			}
		}
		POD_$TYPE$_multiple_effect = { COUNT = 4 }
	}
	else_if = {
		limit = {
			OR = {
				$COUNT$ = 9
				$COUNT$ = 10
			}
		}
		POD_$TYPE$_multiple_effect = { COUNT = 5 }
	}
}

POD_pay_travel_cost_effect = {
	custom_tooltip = POD_pay_travel_cost_tooltip
	if = {
		limit = {
			exists = capital_province
		}
		capital_province = {
			save_scope_as = start_province
		}
	}
	else = {
		location = {
			save_scope_as = start_province
		}
	}
		
	if = {
		limit = {
			$DESTINATION$ = {
				title_capital_county = {
					squared_distance = {
						target = scope:start_province
						value <= short_pilgrimage_max_length
					}
				}
			}
		}
		remove_short_term_gold = pilgrimage_minimum_cost
	}
	else_if = {
		limit = {
			$DESTINATION$ = {
				title_capital_county = {
					squared_distance = {
						target = scope:start_province
						value <= medium_pilgrimage_max_length
					}
				}
			}
		}
		remove_short_term_gold = pilgrimage_medium_extra_cost
	}
	else_if = {
		limit = {
			$DESTINATION$ = {
				title_capital_county = {
					squared_distance = {
						target = scope:start_province
						value <= long_pilgrimage_max_length
					}
				}
			}
		}
		remove_short_term_gold = pilgrimage_long_extra_cost
	}
	else = {
		remove_short_term_gold = pilgrimage_very_long_extra_cost
	}
}
