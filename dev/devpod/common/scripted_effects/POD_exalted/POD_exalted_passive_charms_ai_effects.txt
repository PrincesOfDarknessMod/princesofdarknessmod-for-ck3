POD_select_desired_charm_loadout_effect = {
	POD_passive_charm_injector = { EFFECT = POD_deactivate_charm_effect }
	if = {
		limit = { is_at_war = yes }
		POD_passive_charm_ai_war_injector = { EFFECT = POD_activate_charm_effect }
	}
	else_if = {
		random_list = {
			10 = {
				modifier = {
					OR = {
						any_scheme = { is_hostile = yes }
						any_targeting_scheme = { 
							hostile_scheme_trigger = yes
							is_scheme_exposed = yes 
						}
					}
					add = 90
				}
				POD_passive_charm_ai_scheme_injector = { EFFECT = POD_activate_charm_effect }
			}
			50 = {
				POD_passive_charm_ai_standby_injector = { EFFECT = POD_activate_charm_effect }
			}
		}
	}
}

POD_deactivate_charm_effect = {
	if = {
		limit = { has_perk = $CHARM$ }
		remove_perk = $CHARM$
		POD_passive_charm_update_side_effect = yes
	}
	if = {
		limit = {
			flag:$TYPE$ = flag:$TYPE$
			has_perk = $PERK$
			flag:$FLAG$ = flag:$FLAG$
		}
	}
}
POD_activate_charm_effect = {
	if = {
		limit = { NOT = { has_perk = $CHARM$ } }
		add_perk = $CHARM$
	}
	if = {
		limit = {
			flag:$TYPE$ = flag:$TYPE$
			has_perk = $PERK$
			flag:$FLAG$ = flag:$FLAG$
		}
	}
}
