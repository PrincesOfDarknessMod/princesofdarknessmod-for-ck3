POD_activate_demesne_effect = {
	save_scope_value_as = {
		name = scaling
		value = POD_demesne_activation_cost
	}
	add_character_modifier = POD_exalted_activated_demesne_modifier
	custom_tooltip = {
		text = POD_activate_demesne_effect
		POD_demesne_activation_injector = { EFFECT = POD_activate_demesne_macro }
		POD_refresh_coterie_modifiers_for_all_members = yes
	}
}
POD_activate_demesne_macro = {
	if = {
		limit = { domicile ?= { has_domicile_building = $BUILDING$ } }
		add_character_modifier = $MODIFIER$
	}
}

POD_deactivate_demesne_effect = {
	if = {
		limit = { has_character_modifier = POD_exalted_activated_demesne_modifier }
		remove_character_modifier = POD_exalted_activated_demesne_modifier
	}

	custom_tooltip = {
		text = POD_deactivate_demesne_effect
		POD_demesne_activation_injector = { EFFECT = POD_deactivate_demesne_macro }
		POD_refresh_coterie_modifiers_for_all_members = yes
	}
}
POD_deactivate_demesne_macro = {
	if = {
		limit = { has_character_modifier = $MODIFIER$ }
		remove_character_modifier = $MODIFIER$
	}
	if = {
		limit = { domicile ?= { has_domicile_building = $BUILDING$ } }
	}
}

POD_reapply_demesne_effect = {
	if = {
		limit = { has_character_modifier = POD_exalted_activated_demesne_modifier }
		POD_deactivate_demesne_effect = yes
		POD_activate_demesne_effect = yes
	}
}
POD_reapply_demesne_domicile_effect = {
	scope:owner ?= { POD_reapply_demesne_effect = yes }
}


POD_reapply_family_library_effect = {
	save_scope_value_as = {
		name = exalted_family_library_level
		value = {
			value = 0
			if = {
				limit = {
					OR = {
						AND = {
							has_character_modifier = POD_exalted_activated_demesne_modifier
							domicile ?= { has_domicile_building_or_higher = terrestrial_exalted_family_library_02 }
						}
						any_relation = {
							type = coterie_member
							has_character_modifier = POD_exalted_activated_demesne_modifier
							domicile ?= { has_domicile_building_or_higher = terrestrial_exalted_family_library_02 }
						}
					}
				}
				value = 1
			}
			if = {
				limit = {
					OR = {
						AND = {
							has_character_modifier = POD_exalted_activated_demesne_modifier
							domicile ?= { has_domicile_building_or_higher = terrestrial_exalted_family_library_04 }
						}
						any_relation = {
							type = coterie_member
							has_character_modifier = POD_exalted_activated_demesne_modifier
							domicile ?= { has_domicile_building_or_higher = terrestrial_exalted_family_library_04 }
						}
					}
				}
				value = 2
			}
			if = {
				limit = {
					OR = {
						AND = {
							has_character_modifier = POD_exalted_activated_demesne_modifier
							domicile ?= { has_domicile_building_or_higher = terrestrial_exalted_family_library_06 }
						}
						any_relation = {
							type = coterie_member
							has_character_modifier = POD_exalted_activated_demesne_modifier
							domicile ?= { has_domicile_building_or_higher = terrestrial_exalted_family_library_06 }
						}
					}
				}
				value = 3
			}
		}
	}
	POD_remove_character_modifier = { MODIFIER = terrestrial_exalted_family_library_1_modifier }
	POD_remove_character_modifier = { MODIFIER = terrestrial_exalted_family_library_2_modifier }
	POD_remove_character_modifier = { MODIFIER = terrestrial_exalted_family_library_3_modifier }
	switch = {
		trigger = scope:exalted_family_library_level
		1 = { add_character_modifier = terrestrial_exalted_family_library_1_modifier }
		2 = { add_character_modifier = terrestrial_exalted_family_library_2_modifier }
		3 = { add_character_modifier = terrestrial_exalted_family_library_3_modifier }
	}
}
