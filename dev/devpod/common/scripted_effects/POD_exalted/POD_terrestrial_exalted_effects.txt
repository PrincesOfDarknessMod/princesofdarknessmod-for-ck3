		1 = {  
			modifier = { 
				OR = {
					dynasty ?= dynasty:POD_exalted_dalca
					dynasty ?= dynasty:POD_exalted_barkhad

		}
		1 = {  
			modifier = { dynasty ?= dynasty:POD_exalted_burano add = 2 }
			set_character_faith = faith:the_dragon_cult
		}
		1 = {  
			modifier = { dynasty ?= dynasty:POD_exalted_barkhad add = 2 }
			set_character_faith = faith:preservationists
		}
		1 = { 
			modifier = { dynasty ?= dynasty:POD_exalted_ophris add = 2 }
			set_character_faith = faith:realmists
		}
	}
}
