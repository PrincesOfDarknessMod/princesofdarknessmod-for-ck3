
POD_set_essence_to_effect = {
	hidden_effect = { POD_remove_essence_modifiers_effect = yes }
	save_scope_value_as = {
		name = calc
		value = {
			value = $ESSENCE$
			floor = yes
			min = 0
			max = maximum_essence_proper # corresponds to maximum_essence script_value
		}
	}
	switch = {
		trigger = scope:calc
		0 = { add_character_modifier = POD_essence0_modifier }
		1 = { add_character_modifier = POD_essence1_modifier }
		2 = { add_character_modifier = POD_essence2_modifier }
		3 = { add_character_modifier = POD_essence3_modifier }
		4 = { add_character_modifier = POD_essence4_modifier }
		5 = { add_character_modifier = POD_essence5_modifier }
		6 = { add_character_modifier = POD_essence6_modifier }
		7 = { add_character_modifier = POD_essence7_modifier }
		8 = { add_character_modifier = POD_essence8_modifier }
		9 = { add_character_modifier = POD_essence9_modifier }
		10 = { add_character_modifier = POD_essence10_modifier }
	}
	if = {
		limit = { scope:calc = 0 }
		custom_tooltip = {
			text = POD_deactivate_all_passive_charms
			refund_perks = passive_charms_lifestyle
			add_passive_charms_lifestyle_perk_points = -1000
		}
	}
}

POD_essence_change_effect = {
	if = {
		limit = { NOT = { $COUNT$ = 0 } }
		custom_description = {
			text = POD_essence_change_desc
			value = $COUNT$
			subject = this
		}
		save_scope_value_as = {
			name = essence_calc
			value = {
				value = essence_level
				add = $COUNT$
			}
		}
		POD_set_essence_to_effect = { ESSENCE = scope:essence_calc }
	}
}

POD_essence_increase_effect = {
	POD_essence_change_effect = { COUNT = $COUNT$ }
}

POD_essence_decrease_effect = {
	save_scope_value_as = {
		name = essence_cost
		value = {
			value = 0
			subtract = $COUNT$
		}
	}
	POD_essence_change_effect = { COUNT = scope:essence_cost }
}

POD_essence_cost_effect = {
	POD_energy_cost = {
		COUNT = $COUNT$
		SPLAT_MAXIMUM_ENERGY = maximum_essence
	}
}

POD_essence_income_effect = {
	POD_energy_income = {
		COUNT = $COUNT$
		SPLAT_MAXIMUM_ENERGY = maximum_essence
	}
}

POD_essence_fluctuation_pulse_effect = {
	if = {
		limit = { POD_essence_fluctuation > 0 }
		if = {
			limit = { POD_essence_fluctuation_certain_number > 0 }
			POD_essence_income_effect = { COUNT = POD_essence_fluctuation_certain_number }
		}
		random = {
			chance = POD_essence_fluctuation_singular_chance
			POD_essence_income_effect = { COUNT = 1 }
		}
	}
	else = {
		if = {
			limit = { POD_essence_fluctuation_certain_number > 0 }
			POD_essence_cost_effect = { COUNT = POD_essence_fluctuation_certain_number }
		}
		random = {
			chance = POD_essence_fluctuation_singular_chance
			POD_essence_cost_effect = { COUNT = 1 }
		}
	}
	if = {
		limit = { has_character_modifier = POD_essence0_modifier }
		send_interface_message = {
			type = event_generic_bad_text
			title = POD_deactivate_all_passive_charms
			desc = POD_deactivate_all_passive_charms
		}
	}
}

POD_set_essence_rating_to_effect = {
	switch = {
		trigger = has_trait
		exalted_terrestrial = { trait:exalted_terrestrial = { save_scope_as = exalted } }
	}
	save_scope_value_as = {
		name = calc_xp
		value = {
			value = $VALUE$
			multiply = 25
			add = {
				value = POD_exalted_rating
				multiply = -25
			}
		}
	}
	add_trait_xp = {
		trait = scope:exalted
		value = scope:calc_xp
	}
}

POD_add_essence_rating_effect = {
	save_scope_value_as = {
		name = new_level
		value = {
			value = $VALUE$
			add = POD_exalted_rating
		}
	}
	POD_set_essence_rating_to_effect = { VALUE = scope:new_level }
}
