# HOW TO MAKE NEW PASSIVE CHARMS
# Every passive charm must be present in `POD_passive_charm_injector` to tell game of it's existence.
# Additionally, every passive charm needs a `scripted_value` named using template `POD_passive_charm_$FLAG$_cost`
# Then to actually make it cost something, go into `POD_essence_fluctuation_charms` and add your charm.

POD_passive_charms_close_effect = {
	every_in_list = {
		variable = possible_charms
		end_story = yes
	}
	clear_variable_list = possible_charms
}

POD_passive_charms_innit_effect = {
	if = {
		limit = {
			OR = {
				has_perk = $PERK$
				has_character_flag = POD_passive_charm_$FLAG$
			}
		}
		create_story = {
			type = POD_story_passive_charm_gui
			save_scope_as = story
		}
		scope:story ?= {
			set_variable = { name = flag_type value = flag:$TYPE$ }
			set_variable = { name = flag_name value = flag:$FLAG$ }
			set_variable = { name = charm_perk value = flag:$CHARM$ }
			set_variable = { name = cost value = PREV.POD_passive_charm_$FLAG$_cost }
			
			if = {
				limit = {
					var:flag_type ?= flag:$TYPE$
					var:flag_name ?= flag:$FLAG$
					var:charm_perk ?= flag:$CHARM$
					var:cost ?= PREV.POD_passive_charm_$FLAG$_cost
				}
			}
		}

		add_to_variable_list = {
			name = possible_charms
			target = scope:story
		}
		if = {
			limit = { has_perk = $CHARM$ }
			scope:story ?= { set_variable = { name = activated value = flag:yes } }
		}
		else = {
			scope:story ?= { set_variable = { name = activated value = flag:no } }
		}
	}
	if = {
		limit = { always = no }
		add_character_flag = POD_passive_charm_$FLAG$
	}
}

POD_passive_charms_update_effect = {
	save_scope_value_as = {
		name = charm
		value = flag:$CHARM$
	}
	if = {
		limit = {
			any_in_list = {
				variable = possible_charms
				var:flag_name = flag:$FLAG$
				var:activated = flag:yes
			}
			NOT = { has_perk = $CHARM$ }
			flag:$TYPE$ = flag:$TYPE$
		}
		custom_tooltip = {
			text = POD_passive_charms_update_activate_effect
			ROOT = { add_perk = $CHARM$ }
		}
	}
	else_if = {
		limit = {
			any_in_list = {
				variable = possible_charms
				var:flag_name = flag:$FLAG$
				var:activated = flag:no
			}
			has_perk = $CHARM$
		}
		custom_tooltip = {
			text = POD_passive_charms_update_deactivate_effect
			ROOT = { remove_perk = $CHARM$ }
			POD_passive_charm_update_side_effect = yes
		}
	}
	if = { limit = { has_perk = $PERK$ } }
}

POD_passive_charm_list_refresh_effect = {
	clear_variable_list = charm_list
	ordered_in_list = {
		variable = possible_charms
		limit = {
			trigger_if = {
				limit = { PREV.var:POD_passive_charms_type_filter != flag:any }
				var:flag_type = PREV.var:POD_passive_charms_type_filter
			}
			trigger_if = {
				limit = { PREV.var:POD_passive_charms_active_filter != flag:any }
				var:activated = PREV.var:POD_passive_charms_active_filter
			}
		}
		order_by = {
			value = 0
			switch = {
				trigger = PREV.var:POD_passive_charms_sort_by
				flag:default = {
					switch = {
						trigger = var:flag_type
						flag:terrestrial_air = { add = 5 }
						flag:terrestrial_earth = { add = 4 }
						flag:terrestrial_fire = { add = 3 }
						flag:terrestrial_water = { add = 2 }
						flag:terrestrial_wood = { add = 1 }
					}
				}
				flag:cost = { add = { value = var:cost multiply = -1 } }
				flag:active = {
					if = { limit = { var:activated = flag:yes } add = 1 }
				}
			}
			if = {
				limit = { PREV = { has_variable = POD_passive_charms_sort_by_reverse } }
				multiply = -1
			}
		}
		max = 1000
		check_range_bounds = no
		save_scope_as = charm
		PREV = {
			add_to_variable_list = {
				name = charm_list
				target = scope:charm
			}
		}
	}
}

# Effect for any side buffs charms give, and which should be gone upon losing it.
POD_passive_charm_update_side_effect = {
	if = {
		limit = { 
			has_perk = charm_excellence_of_the_raging_storm_extra_perk
			NOT = { has_perk = charm_excellence_of_the_raging_storm_perk }
		}
		remove_perk = charm_excellence_of_the_raging_storm_extra_perk
	}
	if = {
		limit = { 
			has_perk = charm_excellence_of_the_eternal_mountain_extra_perk
			NOT = { has_perk = charm_excellence_of_the_eternal_mountain_perk }
		}
		remove_perk = charm_excellence_of_the_eternal_mountain_extra_perk
	}
	if = {
		limit = { 
			has_perk = charm_excellence_of_roaring_flame_extra_perk
			NOT = { has_perk = charm_excellence_of_roaring_flame_perk }
		}
		remove_perk = charm_excellence_of_roaring_flame_extra_perk
	}
	if = {
		limit = { 
			has_perk = charm_excellence_of_the_crashing_wave_extra_perk
			NOT = { has_perk = charm_excellence_of_the_crashing_wave_perk }
		}
		remove_perk = charm_excellence_of_the_crashing_wave_extra_perk
	}
	if = {
		limit = { 
			has_perk = charm_excellence_of_the_verdant_forest_extra_perk
			NOT = { has_perk = charm_excellence_of_the_verdant_forest_perk }
		}
		remove_perk = charm_excellence_of_the_verdant_forest_extra_perk
	}
}

# WHAT EVERYTHING DOES.
# Type is a flag used for filtering and sorting.
# Perk is used as the default way to mark passive charm as available, if player has it, passive charm is shown
# Charm is the real passive charm which have all the effects, and towards which everything about active passive charms is targeted
# Flag is used both as indicator for GUI, link towards cost(scripted_value) and emergency unlock.
# Emergency unlock checks for flag POD_passive_charm_$FLAG$.

# Macro if passive charm is made using template exalted_$NAME$_perk for Unlocking, and charm_$NAME$_perk for charm proper.
POD_passive_charm_injector_macro = {
	$EFFECT$ = {
		TYPE = $TYPE$
		PERK = exalted_$NAME$_perk
		CHARM = charm_$NAME$_perk
		FLAG = $NAME$
	}
}
