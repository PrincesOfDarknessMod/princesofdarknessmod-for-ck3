POD_create_ancient_sorcery_artifact_effect = {
	$OWNER$ = { save_scope_as = owner }
	set_artifact_rarity_famed = yes

	# Create the artifact
	create_artifact = {
		decaying = no
		name = artifact_ancient_sorcery_tome_name
		description = artifact_ancient_sorcery_tome_description
		type = journal
		wealth = scope:wealth
		quality = scope:quality
		history = {
			type = created_before_history
		}
		template = ancient_sorcery_tome_template
		visuals = book_POD_black
		modifier = artifact_ancient_sorcery_modifier
		save_scope_as = newly_created_artifact
	}

	scope:newly_created_artifact ?= {
		set_variable = historical_unique_artifact
		set_variable = { name = artifact_ancient_sorcery value = $SORCERY$ }
	}
}

POD_grant_character_full_ancient_sorcery_effect = {
	POD_create_ancient_sorcery_artifact_effect = { OWNER = THIS SORCERY = flag:$SORCERY$ }
	add_perk = $SORCERY$_perk
}

POD_learn_ancient_sorcery_from_artifact_effect = {
	$ARTIFACT$ = { save_scope_as = artifact }
	POD_ancient_sorcery_flag_and_perk_effect_injector = { EFFECT = POD_learn_ancient_sorcery_from_artifact_macro }
}

POD_learn_ancient_sorcery_from_artifact_macro = {
	if = {
		limit = { scope:artifact.var:artifact_ancient_sorcery ?= flag:$FLAG$ NOT = { has_perk = $PERK$ } }
		add_perk = $PERK$
	}
}

POD_grant_province_ancient_tome_effect = {
	if = {
		limit = { flag:$SORCERY$ = flag:any }
		while = {
			limit = { NOT = { exists = scope:sorcery } }
			POD_ancient_sorcery_flag_and_perk_effect_injector = { EFFECT = POD_grant_province_ancient_tome_macro }
		}
	}
	else = { save_scope_value_as = { name = sorcery value = flag:$SORCERY$ } }
	if = {
		limit = { flag:$TYPE$ = flag:any }
		random_list = {
			1 = { save_scope_value_as = { name = type value = flag:noble } }
			1 = { save_scope_value_as = { name = type value = flag:ruins } }
			1 = { save_scope_value_as = { name = type value = flag:vampire } }
			1 = { save_scope_value_as = { name = type value = flag:garou } }
			# 1 = { save_scope_value_as = { name = type value = flag:mage } }
		}
	}
	else = { save_scope_value_as = { name = type value = flag:$TYPE$ } }

	character:POD_wraith_01 = {
		create_story = {
			type = POD_story_cycle_ancient_tome_location
			save_scope_as = story
		}
		scope:story ?= {
			set_variable = { name = province_ancient_sorcery value = scope:sorcery }
			set_variable = { name = ancient_sorcery_type value = scope:type }
		}
	}
	hidden_effect_new_object = {
		add_to_variable_list = {
			name = POD_ancient_tomes_list
			target = scope:story
		}
	}
	add_province_modifier = POD_ancient_repository_modifier
}

POD_grant_province_ancient_tome_macro = {
	if = {
		limit = { flag:$FLAG$ = flag:$FLAG$ }
		random = {
			chance = 5
			save_scope_value_as = { name = sorcery value = flag:$FLAG$ }
		}
	}
	if = {
		limit = { character:POD_wraith_01 = { has_perk = $PERK$ } }
	}
}

POD_take_ancient_tome_from_province_effect = {
	$PROVINCE$ = { save_scope_as = province }
	POD_copy_ancient_tome_from_province_effect = { PROVINCE = scope:province TYPE = $TYPE$ }
	scope:province = {
		remove_list_variable = {
			name = POD_ancient_tomes_list
			target = scope:tome_story
		}
		if = {
			limit = { NOT = { has_variable_list = POD_ancient_tomes_list } }
			remove_province_modifier = POD_ancient_repository_modifier
		}
	}
	scope:tome_story ?= {
		end_story = yes
	}
	custom_tooltip = POD_take_ancient_tome_from_province_effect
}
POD_copy_ancient_tome_from_province_effect = {
	$PROVINCE$ = { save_scope_as = province }
	save_scope_as = character
	scope:province = {
		random_in_list = {
			variable = POD_ancient_tomes_list
			limit = { 
				save_temporary_scope_as = tome_story
				var:ancient_sorcery_type = flag:$TYPE$
				scope:character = {
					NOT = { POD_knows_ancient_sorcery_trigger = { SORCERY = scope:tome_story.var:province_ancient_sorcery } }
				}
			}
			save_scope_as = tome_story
		}
		if = {
			limit = { NOT = { exists = scope:tome_story } }
			random_in_list = {
				variable = POD_ancient_tomes_list
				limit = {
					var:ancient_sorcery_type = flag:$TYPE$
				}
				save_scope_as = tome_story
			}
		}
	}
	POD_create_ancient_sorcery_artifact_effect = { OWNER = THIS SORCERY = scope:tome_story.var:province_ancient_sorcery }
}

POD_ancient_sorcery_ceremony_effect = {
	save_scope_value_as = {
		name = essence_value
		value = $SORCERY_COST$
	}
	custom_tooltip = {
		text = POD_ancient_sorcery_ceremony_start_effect
		POD_start_targeted_journey_effect = {
			TYPE	= exalted_ancient_sorcery_casting
			MONTHLY = exalted_ancient_sorcery_casting
			TARGET  = $TARGET$
		}
		var:current_journey ?= {
			set_variable = targets_location
			set_variable = { name = sorcery value = flag:$SORCERY$ }
			set_variable = { name = sorcery_monthly_progress value = $SORCERY_SPEED$ }
			set_variable = { name = sorcery_essence_cost value = $SORCERY_COST$ }
		}
	}

	$TARGET$ = { custom_tooltip = POD_ancient_sorcery_ceremony_tooltip_effect }
	show_as_tooltip = { $EFFECT$ = { TARGET = $TARGET$ } }
}
