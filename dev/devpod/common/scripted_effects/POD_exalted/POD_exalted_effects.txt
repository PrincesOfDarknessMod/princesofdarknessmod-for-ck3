POD_charms_perk_cost_effect = {
	hidden_effect = {
		save_scope_value_as = {
			name = cost
			value = {
				value = $VALUE$
				add = -1
				multiply = -1
			}
		}
		add_charms_lifestyle_perk_points = scope:cost
	}
	custom_description_no_bullet = {
		text = additional_perk_cost_effect
		value = $VALUE$
	}
}

POD_deduct_extra_charm_perk_cost_effect = {
	save_scope_value_as = {
		name = charm
		value = flag:$CHARM$
	}
	save_scope_value_as = {
		name = level
		value = $LEVEL$
	}
	save_scope_value_as = {
		name = extra_perk_points_deducted
		value = {
			value = POD_charm_perk_cost
			add = -1
			multiply = -1
		}
	}
	if = {
		limit = { scope:extra_perk_points_deducted < 0 }
		custom_tooltip = POD_charm_perk_cost
		hidden_effect = { add_charms_lifestyle_perk_points = scope:extra_perk_points_deducted }
	}
}

POD_recalculate_exalted_council_effect = {
	save_scope_value_as = { name = highest_diplomacy value = -1 }
	save_scope_value_as = { name = highest_martial value = -1 }
	save_scope_value_as = { name = highest_stewardship value = -1 }
	save_scope_value_as = { name = highest_intrigue value = -1 }
	save_scope_value_as = { name = highest_learning value = -1 }
	title:e_exalted_hierarchy.holder ?= {
		every_powerful_vassal = {
			switch = {
				trigger = highest_skill
				diplomacy = {
					save_scope_value_as = {
						name = highest_diplomacy
						value = { value = scope:highest_diplomacy add = diplomacy }
					}
				}
				martial = {
					save_scope_value_as = {
						name = highest_martial
						value = { value = scope:highest_martial add = martial }
					}
				}
				stewardship = {
					save_scope_value_as = {
						name = highest_stewardship
						value = { value = scope:highest_stewardship add = stewardship }
					}
				}
				intrigue = {
					save_scope_value_as = {
						name = highest_intrigue
						value = { value = scope:highest_intrigue add = intrigue }
					}
				}
				learning = {
					save_scope_value_as = {
						name = highest_learning
						value = { value = scope:highest_learning add = learning }
					}
				}
			}
		}
	}
	POD_recalculate_exalted_council_macro = { SKILL = diplomacy }
	POD_recalculate_exalted_council_macro = { SKILL = martial }
	POD_recalculate_exalted_council_macro = { SKILL = stewardship }
	POD_recalculate_exalted_council_macro = { SKILL = intrigue }
	POD_recalculate_exalted_council_macro = { SKILL = learning }
}
POD_recalculate_exalted_council_macro = { # Remember everything starts at -1
	if = {
		limit = { scope:highest_$SKILL$ <= -1 }
		faith:the_enduring ?= {
			if = { 
				limit = { has_doctrine = special_exalted_council_weak_$SKILL$_doctrine } 
				remove_doctrine = special_exalted_council_weak_$SKILL$_doctrine
			}
			if = { 
				limit = { has_doctrine = special_exalted_council_normal_$SKILL$_doctrine } 
				remove_doctrine = special_exalted_council_normal_$SKILL$_doctrine
			}
			if = { 
				limit = { has_doctrine = special_exalted_council_high_$SKILL$_doctrine } 
				remove_doctrine = special_exalted_council_high_$SKILL$_doctrine
			}
		}
	}
	else_if = {
		limit = { scope:highest_$SKILL$ <= 4 }
		faith:the_enduring ?= {
			if = { 
				limit = { has_doctrine = special_exalted_council_normal_$SKILL$_doctrine } 
				remove_doctrine = special_exalted_council_normal_$SKILL$_doctrine
			}
			if = { 
				limit = { has_doctrine = special_exalted_council_high_$SKILL$_doctrine } 
				remove_doctrine = special_exalted_council_high_$SKILL$_doctrine
			}
			add_doctrine = special_exalted_council_weak_$SKILL$_doctrine
		}
	}
	else_if = {
		limit = { scope:highest_$SKILL$ <= 14 }
		faith:the_enduring ?= {
			if = { 
				limit = { has_doctrine = special_exalted_council_weak_$SKILL$_doctrine } 
				remove_doctrine = special_exalted_council_weak_$SKILL$_doctrine
			}
			if = { 
				limit = { has_doctrine = special_exalted_council_high_$SKILL$_doctrine } 
				remove_doctrine = special_exalted_council_high_$SKILL$_doctrine
			}
			add_doctrine = special_exalted_council_normal_$SKILL$_doctrine
		}
	}
	else_if = {
		limit = { scope:highest_$SKILL$ >= 15 }
		faith:the_enduring ?= {
			if = { 
				limit = { has_doctrine = special_exalted_council_weak_$SKILL$_doctrine } 
				remove_doctrine = special_exalted_council_weak_$SKILL$_doctrine
			}
			if = { 
				limit = { has_doctrine = special_exalted_council_normal_$SKILL$_doctrine } 
				remove_doctrine = special_exalted_council_normal_$SKILL$_doctrine
			}
			add_doctrine = special_exalted_council_high_$SKILL$_doctrine
		}
	}
}
