POD_character_vampire_infernalist_initialization_effect = {
	if = {
		limit = {
			faith = { is_demon_worshipper_religion_trigger = yes }
		}
		give_infernalist_secret_or_trait_effect = yes
	}
}

POD_character_initialization_random_background_effect = {
	# this should result for 30,000 characters to have 3,000 with a background: 300 diablerists, 300 infernalists, 300 unbondable, etc...
	
	random = {
		chance = 10
		
		random_list = {
			5 = {
				trigger = {
					is_diablerist_trigger = no
					is_kueijin_trigger = no 
				}
				add_secret = {
					type = secret_diablerist 
				}
			}
			5 = {
				trigger = { is_infernalist_trigger = no }
				give_infernalist_secret_or_trait_effect = yes
			}
			10 = {
				trigger = { is_unbondable_trigger = no }
				give_unbondable_secret_or_trait_effect = yes
			}			
			10 = {
				trigger = {
					is_oathbreaker_trigger = no
				}
				add_trait = oathbreaker
			}
			10 = {
				trigger = { is_malkavian_trigger = no }
				POD_character_vampire_madness_initialization_effect = yes
			}
			
			5 = {
				trigger = { 
					OR = {
						is_toreador_trigger = yes
						is_malkavian_trigger = yes
						has_faith = faith:viaeinherjar
						has_faith = faith:viaarchangeli
					}
					NOT = { has_trait = lifestyle_poet }
				}
				add_trait = lifestyle_poet
			}
			20 = {
				# Wealth
				random_list = {
					50 = { add_gold = medium_gold_value }
					40 = { add_gold = major_gold_value }
					10 = { add_gold = massive_gold_value }
				}
			}			
			20 = {
				# piety
				random_list = {
					40 = { add_piety = medium_piety_loss }
					40 = { add_piety = medium_piety_loss }
					10 = { add_piety = major_piety_gain }
					10 = { add_piety = major_piety_loss }
				}
			}			
		}
	}
}


give_unbondable_secret_or_trait_effect = {
	save_temporary_scope_as = unbondable
	if = {
		limit = { #Not already a unbondable
			NOR = {
				has_trait = unbondable
				any_secret = { secret_type = secret_unbondable }
			}
		}
		if = {
			limit = {
				trait_is_shunned_or_criminal_in_my_or_lieges_faith_trigger = { TRAIT = unbondable GENDER_CHARACTER = scope:unbondable }
			}
			add_secret = {
				type = secret_unbondable
			}
			# random_secret = {
				# limit = { secret_type = secret_unbondable }
				# save_scope_as = unbondable_secret
			# }
		}
		else = {
			add_trait = unbondable
		}
	}
}
