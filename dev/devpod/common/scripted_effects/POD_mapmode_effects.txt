POD_mapmode_reset_title_name_effect = {
	if = {
		limit = { has_global_variable_list = map_label_changed_list }
		every_in_global_list = {
			variable = map_label_changed_list
			reset_title_name = yes
			save_scope_as = title
			holder = { trigger_event = POD_title_flavor.0002 }
		}
		clear_global_variable_list = map_label_changed_list
	}
}

POD_mapmode_splatmode_observer_effect = {
	POD_mapmode_reset_title_name_effect = yes # So observer can reset
	POD_mapmode_splatmode_effect = yes
}

POD_mapmode_splatmode_effect = {
	every_barony = {
		if = {
			limit = {
				county.holder = { 
					is_forsaken_or_gantenbein_trigger = yes
				}
			}
			set_color_from_title = title:k_calomenasforsaken
		}
		else_if = {
			limit = {
				county.holder = { 
					OR = {
						has_trait = supehunter
						is_child_of_supehunter_trigger = yes
					}
				}
			}
			set_color_from_title = title:e_shadow_inquisition
		}
		else_if = {
			limit = {
				county.holder = { 
					OR = { 
						has_trait = vampire
						has_trait = dhampir
					}
				}
			}
			set_color_from_title = title:e_color_vampire
		}
		else_if = {
			limit = {
				county.holder = { has_trait = wraith }
			}
			set_color_from_title = title:d_adrar
		}
		else_if = {
			limit = {
				county.holder = { has_trait = fae }
			}
			set_color_from_title = title:e_autumn_court
		}
		else_if = {
			limit = {
				county.holder = { has_trait = mage }
			}
			set_color_from_title = title:e_color_mage
		}
		else_if = {
			limit = {
				county.holder = { has_POD_kj_blood_trigger = yes }
			}
			set_color_from_title = title:e_color_kueijin
		}
		else_if = {
			limit = {
				county.holder = { 
					OR = { 
						has_trait = fera
						AND = {
							has_trait = kinfolk
							OR = { 
								religion = { is_in_family = rf_gaia } 
								religion = { is_in_family = rf_wyrm } 
							}
						}
					}
				}
			}
			set_color_from_title = title:e_color_fera
		}
		else_if = {
			limit = {
				county.holder = { 
					OR = {
						has_trait = mummy
						trigger_if = { 
							limit = { has_trait = mortal }
							is_a_neteru = yes
						}
					}
				}
			}
			set_color_from_title = title:e_color_mummy
		}
		else_if = {
			limit = {
				county.holder = { 
					OR = {
						has_trait = demon
						has_trait = laham
						has_trait = nephilim
					}
				}
			}
			set_color_from_title = title:e_voivodate
		}
		else_if = {
			limit = {
				county.holder = { 
					OR = {
						has_trait = changeling
						has_trait = kinain
					}
					religion = { is_in_family = rf_fae }
				}
			}
			set_color_from_title = title:e_autumn_court
		}
		else_if = {
			limit = {
				county.holder = {
					POD_is_any_kind_of_ghoul_trigger = yes
					religion = { is_in_family = rf_kindred }
				}
			}
			set_color_from_title = title:d_sjaelland
		}
		else_if = {
			limit = {
				county.holder = { 
					religion = { is_in_family = rf_demons }
				}
			}
			set_color_from_title = title:e_color_infernalist
		}
		else = {
			set_color_from_title = title:k_angara
		}
	}
}
