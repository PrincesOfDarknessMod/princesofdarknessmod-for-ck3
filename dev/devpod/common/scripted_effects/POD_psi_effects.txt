POD_summon_animal_effect = {
	if = { limit = { $ANIMAL$ ?= flag:cat } start_cat_story_cycle_effect = yes }
	if = { limit = { $ANIMAL$ ?= flag:dog } start_dog_story_cycle_effect = yes }
	if = { limit = { $ANIMAL$ ?= flag:horse } start_warhorse_story_cycle_effect = yes }
	if = { limit = { $ANIMAL$ ?= flag:wolf } add_character_modifier = { modifier = exotic_pet_wolf years = 10 } create_artifact_wolf_effect = { OWNER = this } }
	if = { limit = { $ANIMAL$ ?= flag:bear } add_character_modifier = { modifier = exotic_pet_bear years = 10 } create_artifact_bear_effect = { OWNER = this } }
	if = { limit = { $ANIMAL$ ?= flag:lion } add_character_modifier = { modifier = exotic_pet_lion years = 10 } create_artifact_lion_effect = { OWNER = this } }
	if = { limit = { $ANIMAL$ ?= flag:tiger } add_character_modifier = { modifier = exotic_pet_tiger years = 10 } create_artifact_tiger_effect = { OWNER = this } }
	if = { limit = { $ANIMAL$ ?= flag:rat } add_character_modifier = { modifier = prison_pet_rat_modifier years = 10 } }
	if = { limit = { $ANIMAL$ ?= flag:cockroach } add_character_modifier = { modifier = prison_pet_cockroach_modifier years = 10 } }
}
