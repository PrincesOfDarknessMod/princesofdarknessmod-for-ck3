POD_static_stress_gain = {
    custom_description = {
        text = "POD_static_stress_gain"
        value = $COUNT$
        add_character_flag = static_stress_gain_flag
        save_scope_value_as = { name = saved_stress  value = stress }
        add_stress = -400
        while = {
            limit = { stress >= 1 }
            add_character_modifier = POD_static_stress_gain_2_modifier
            force_character_skill_recalculation = yes
            add_stress = -400
        }
        add_stress = 90
        while = {
            limit = { stress = 0 }
            add_character_modifier = POD_static_stress_gain_1_modifier
            force_character_skill_recalculation = yes
            add_stress = 90
        }

        save_scope_value_as = { name = stress_diffrence value = { value = 90 subtract = stress } }
        add_stress = -1000
        if = { limit = { NOT = { scope:stress_diffrence = 0 scope:stress_diffrence = 300 } } 
            save_scope_value_as = { name = stress_modifier_percent value = { value = 1 subtract = { value = scope:stress_diffrence divide = 90 } } }
            if = { limit = { NOT = { scope:stress_modifier_percent = 0 } } 
                add_stress = { value = scope:saved_stress divide = scope:stress_modifier_percent }
                remove_character_flag = static_stress_gain_flag
                add_stress = { value = $COUNT$ divide = scope:stress_modifier_percent }
            }
            remove_character_flag = static_stress_gain_flag
        }
        else = { 
            add_stress = scope:saved_stress
            remove_character_flag = static_stress_gain_flag
            add_stress = $COUNT$
        }
        remove_all_character_modifier_instances = POD_static_stress_gain_1_modifier
        remove_all_character_modifier_instances = POD_static_stress_gain_2_modifier
    }
}
