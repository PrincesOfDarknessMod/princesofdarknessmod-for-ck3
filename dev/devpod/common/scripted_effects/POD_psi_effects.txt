POD_static_stress_gain = {
    custom_description = {
        text = "POD_static_stress_gain"
        value = $COUNT$
        if = { limit = { NOT = { $COUNT$ = 0 } } 
            add_character_flag = static_stress_gain_flag
            save_scope_value_as = { name = saved_stress  value = stress }
            save_scope_value_as = { name = saved_stress_level  value = stress_level }
            
            add_stress = 100
            
            while = {
                limit = { stress = scope:saved_stress }
                add_character_modifier = POD_static_stress_gain_1_modifier
                force_character_skill_recalculation = yes
                add_stress = 100
            }
            save_scope_value_as = { name = save_stress value = stress }
            add_stress = -400
            while = {
                limit = { stress > 0 }
                add_character_modifier = POD_static_stress_gain_2_modifier
                force_character_skill_recalculation = yes
                add_stress = -400
            }       
            add_stress = 50
            save_scope_value_as = { name = add_percentage value = { value = stress divide = 50 } }
            if = { limit = { NOT = { scope:add_percentage = 0 } } 
                add_stress = -40
                if = { limit = {NOT = { stress = 0 } } 
                    save_scope_value_as = { name = remove_percentage value = { value = stress divide = { value = { value = scope:add_percentage multiply = 50 } subtract = 40 } } }
                }
                else = {
                    save_scope_value_as = { name = remove_percentage value = 0 }
                }
                add_stress = -500

                add_stress = { value = scope:saved_stress divide = scope:add_percentage }
                remove_character_flag = static_stress_gain_flag
                if = { limit = { $COUNT$ < 0 NOT = { scope:remove_percentage = 0 } } add_stress = { value = $COUNT$ divide = scope:remove_percentage } }
                else = { add_stress = { value = $COUNT$ divide = scope:add_percentage } }
                
                # save_scope_value_as = { name = expected_number value = { value = $COUNT$ add = scope:saved_stress } }

                # if = { limit = { scope:expected_number >= 400 } save_scope_value_as = { name = expected_number value = 400 } }
                # if = { limit = { scope:expected_number < 0 } save_scope_value_as = { name = expected_number value = 0 } }
                # while = { 
                #     count = 100
                #     limit = { NOT = { stress = scope:expected_number } } 
                #     if = { limit = { stress > scope:expected_number } add_stress = -1 }
                #     else_if = { limit = { stress < scope:expected_number } add_stress = 1 } 
                # }
            }
            remove_all_character_modifier_instances = POD_static_stress_gain_1_modifier
            remove_all_character_modifier_instances = POD_static_stress_gain_2_modifier
            force_character_skill_recalculation = yes
        }
    }
}
