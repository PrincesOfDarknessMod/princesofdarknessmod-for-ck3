POD_become_kueijin_effect = {
	save_temporary_scope_value_as = {
		name = saved_immortal_age
		value = POD_visual_age
	}
	hidden_effect = { POD_remove_splat_traits_effect = yes }
	add_trait_force_tooltip = kueijin
	if = {
		limit = { exists = scope:saved_immortal_age }
		hidden_effect = { set_immortal_age = scope:saved_immortal_age }
	}
	hidden_effect = {
		POD_remove_wrong_supernatural_traits_effect = yes
		POD_remove_wrong_supernatural_modifiers_effect = yes
		POD_remove_supernatural_relation_effect = yes
		POD_side_effects_from_changing_splats = yes
	}
	add_trait = dharma0
	hidden_effect = {
		POD_kueijin_initialization_effect = yes
	}
	if = {
		limit = { is_ai = no }
		trigger_event = podintroductions.5102
	}
}

POD_kj_ai_adoption_effect = {
	if = {
		limit = {
			$PARENT$ = {
				is_ai = yes
				has_trait = kueijin
				dynasty = dynasty:10000022
				age > prev.age
				trigger_if = {
					limit = { highest_held_title_tier >= tier_county }
					any_child = { is_alive = yes count < 2 }
				}
				trigger_else = {
					any_child = { is_alive = yes count < 1 }
				}
				POD_is_clone_trigger = no
			}
		}
		POD_set_parent_effect = { PARENT = $PARENT$ }
	}
}

	if = {
		limit = {
			is_ai = no
		}
		if = {
			limit = { has_perk = $PERK$ }
			hidden_effect = {
				if = {
					limit = { $LEVEL$ <= 4 }
					POD_require_dharma_tooltip_effect = { LEVEL = $LEVEL$ }
				}
				else = {
					if = {
						limit = { dharma_level < shinta_mastery_dharma_requirement }
						POD_require_dharma_tooltip_effect = { LEVEL = shinta_mastery_dharma_requirement }
					}
				}
				switch = {
					trigger = flag:$TYPE$
					fallback = {
						POD_increase_wounds_effect = {
							REASON = self_mutilation
							COUNT = POD_self_mutilation_base_damage
						}
					}
				}
			}
		}
		else = {
			if = {
				limit = { $LEVEL$ <= 4 }
				POD_require_dharma_tooltip_effect = { LEVEL = $LEVEL$ }
			}
			else = {
				if = {
					limit = { dharma_level < shinta_mastery_dharma_requirement }
					POD_require_dharma_tooltip_effect = { LEVEL = shinta_mastery_dharma_requirement }
				}
			}
			switch = {
				trigger = flag:$TYPE$
				fallback = {
					POD_increase_wounds_effect = {
						REASON = self_mutilation
						COUNT = POD_self_mutilation_base_damage
					}
				}
			}
			
		}
	}
}

POD_shintai_of_devotion_refresh_coterie_effect = {
	if = {
		limit = {
			any_relation = {
				type = coterie_member
				has_perk = shintai_touch_of_the_asura_perk
			}
		}
		add_character_modifier = shintai_touch_of_the_asura_modifier
	}
	else = {
		POD_remove_character_modifier = { MODIFIER = shintai_touch_of_the_asura_modifier }
	}
	every_relation = {
		type = coterie_member
		limit = {
			has_perk = shintai_the_shintai_of_devotion_perk
			NOT = {
				has_hook_of_type = {
					type = eternal_loyalty_hook
					target = prev
				}
			}
		}
		add_hook = {
			type = eternal_loyalty_hook
			target = prev
		}
	}
}

POD_clear_shintai_of_devotion_hooks = {
	every_relation = {
		type = coterie_member
		limit = {
			has_perk = shintai_the_shintai_of_devotion_perk
			has_hook_of_type = {
				type = eternal_loyalty_hook
				target = prev
			}
		}
		remove_hook = {
			type = eternal_loyalty_hook
			target = prev
		}
	}
}
