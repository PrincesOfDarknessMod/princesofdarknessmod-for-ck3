POD_become_kueijin_effect = {
	save_temporary_scope_value_as = {
		name = saved_immortal_age
		value = POD_visual_age
	}
	hidden_effect = { POD_remove_splat_traits_effect = yes }
	add_trait_force_tooltip = kueijin
	if = {
		limit = { exists = scope:saved_immortal_age }
		hidden_effect = { set_immortal_age = scope:saved_immortal_age }
	}
	hidden_effect = {
		POD_remove_wrong_supernatural_traits_effect = yes
		POD_remove_wrong_supernatural_modifiers_effect = yes
		POD_remove_supernatural_relation_effect = yes
		POD_side_effects_from_changing_splats = yes
	}
	add_trait = dharma0
	POD_kueijin_initialization_effect = yes
	if = {
		limit = { is_ai = no }
		trigger_event = podintroductions.5102
	}
}

POD_dhampyr_initialization_effect = {
	if = {
		limit = {
			is_a_cultist = no
			POD_is_supehunter_trigger = no
			NOT = { religion = { is_in_family = rf_kueijin } }
		}
		POD_character_kueijin_faith_initialization_effect = yes
	}
	if = {
		limit = {
			NOT = { has_character_modifier = joss_modifier }
		}
		add_character_modifier = joss_modifier
	}
}

POD_kueijin_initialization_effect = {
	if = {
		limit = {
			has_dharma_trigger = no
		}
		POD_set_dharma_effect = yes
	}
	if = {
		limit = {
			is_a_cultist = no
			POD_is_supehunter_trigger = no
			NOT = { religion = { is_in_family = rf_kueijin } }
		}
		POD_character_kueijin_faith_initialization_effect = yes
	}
	if = {
		limit = { # Only count+ can buy perk
			highest_held_title_tier >= tier_county
		}
		if = {
			limit = {
				demon_arts_lifestyle_perks = 0
				demon_arts_lifestyle_perk_points = 0
			}
			add_demon_arts_lifestyle_perk_points = 1 # All characters learn at least one Demon Art following the Second Breath;
		}
	}
}

POD_kj_ai_adoption_effect = {
	if = {
		limit = { 
			$PARENT$  = {
				is_ai = yes
				has_trait = kueijin
				dynasty = dynasty:10000022
				age > prev.age
				trigger_if = {
					limit = { highest_held_title_tier >= tier_county }
					any_child = { is_alive = yes count < 2 }
				}
				trigger_else = {
					any_child = { is_alive = yes count < 1 }
				}
				POD_is_clone_trigger = no
			}
		}
		POD_set_parent_effect = { PARENT = $PARENT$ }
	}
}

POD_set_dharma_effect = {
	if = {
		limit = {		
			age >= 1000
		}
		add_trait = dharma9
	}
	else_if = {
		limit = {		
			age >= 500
		}
		add_trait = dharma8
	}
	else_if = {
		limit = {		
			age >= 250
		}
		add_trait = dharma7
	}
	else_if = {
		limit = {		
			age >= 100
		}
		add_trait = dharma6
	}
	else = {
		add_trait = dharma5
	}
}

POD_give_dharma_xp = {
	switch = {
		trigger = has_trait
		dharma1 = { 
			add_trait_xp = { trait = dharma1 value = $VALUE$ } 
			if = { limit = { has_trait_xp = { trait = dharma1 value >= 5 } } POD_advance_dharma = yes }
		}
		dharma2 = { 
			add_trait_xp = { trait = dharma2 value = $VALUE$ } 
			if = { limit = { has_trait_xp = { trait = dharma2 value >= 10 } } POD_advance_dharma = yes }
		}
		dharma3 = { 
			add_trait_xp = { trait = dharma3 value = $VALUE$ } 
			if = { limit = { has_trait_xp = { trait = dharma3 value >= 15 } } POD_advance_dharma = yes }
		}
		dharma4 = {

}
