POD_become_kueijin_effect = {
	save_temporary_scope_value_as = {
		name = saved_immortal_age
		value = POD_visual_age
	}
	hidden_effect = { POD_remove_splat_traits_effect = yes }
	add_trait_force_tooltip = kueijin
	if = {
		limit = { exists = scope:saved_immortal_age }
		hidden_effect = { set_immortal_age = scope:saved_immortal_age }
	}
	hidden_effect = {
		POD_remove_wrong_supernatural_traits_effect = yes
		POD_remove_wrong_supernatural_modifiers_effect = yes
		POD_remove_supernatural_relation_effect = yes
		POD_side_effects_from_changing_splats = yes
	}
	add_trait = dharma0
	POD_kueijin_initialization_effect = yes
	if = {
		limit = { is_ai = no }
		trigger_event = podintroductions.5102
	}
}

	}
	if = {
		limit = {
			NOT = { has_character_modifier = joss_modifier }
		}
		add_character_modifier = joss_modifier
	}
}

POD_kj_ai_adoption_effect = {
	if = {
		limit = {
			$PARENT$  = {
				is_ai = yes
				has_trait = kueijin
				dynasty = dynasty:10000022
				age > prev.age
				trigger_if = {
					limit = { highest_held_title_tier >= tier_county }
					any_child = { is_alive = yes count < 2 }
				}
				trigger_else = {
					any_child = { is_alive = yes count < 1 }
				}
				POD_is_clone_trigger = no
			}
		}
		POD_set_parent_effect = { PARENT = $PARENT$ }
	}
}

POD_kj_discipline_cost_effect = {
	save_scope_value_as = {
		name = vessel_cost
		value = {
			value = total_stored_energy
			max = $COUNT$
		}
	}
	save_scope_value_as = {
		name = energy_overflow
		value = {
			value = $COUNT$
			subtract = scope:vessel_cost
		}
	}
	if = {
		limit = { scope:vessel_cost > 0 }
		POD_spend_energy_from_vessels = { AMOUNT = scope:vessel_cost }
	}
	if = {
		limit = { scope:energy_overflow > 0 }
	 	POD_static_stress_cost_effect = { COUNT = scope:energy_overflow }
	}
}
