POD_change_dharma_xp_effect = {
	switch = {
		trigger = has_trait
		dharma1 = { trait:dharma1 = { save_scope_as = dharma } }
		dharma2 = { trait:dharma2 = { save_scope_as = dharma } }
		dharma3 = { trait:dharma3 = { save_scope_as = dharma } }
		dharma4 = { trait:dharma4 = { save_scope_as = dharma } }
		dharma5 = { trait:dharma5 = { save_scope_as = dharma } }
		dharma6 = { trait:dharma6 = { save_scope_as = dharma } }
		dharma7 = { trait:dharma7 = { save_scope_as = dharma } }
		dharma8 = { trait:dharma8 = { save_scope_as = dharma } }
		dharma9 = { trait:dharma9 = { save_scope_as = dharma } }
	}
	if = {
		limit = { exists = scope:dharma }
		
		add_trait_xp = {
			trait = scope:dharma
			value = $VALUE$
		}
		
		if = {
			limit = {
				has_trait_xp = {
					trait = scope:dharma
					value >= POD_kueijin_dharma_to_advance
				}
			}
			POD_advance_dharma_effect = yes
		}
		else_if = {
			limit = {
				$VALUE$ < 0
				has_trait_xp = {
					trait = scope:dharma
					value <= 0
				}
			}
			POD_decrease_dharma_effect = yes
			hidden_effect = {
				if = { 
					limit = { has_trait = dharma9 } 
					add_trait_xp = {
						trait = dharma9
						value = 50
					}
				}
			}
			if = { 
				limit = { is_ai = yes } 
				save_scope_value_as = {
					name = pity_dharma_xp
					value = {
						value = POD_kueijin_dharma_diao_value
						max = {
							value = POD_kueijin_dharma_to_advance
							subtract = 1
						}
					}
				}
				switch = {
					trigger = has_trait
					dharma1 = { trait:dharma1 = { save_scope_as = dharma } }
					dharma2 = { trait:dharma2 = { save_scope_as = dharma } }
					dharma3 = { trait:dharma3 = { save_scope_as = dharma } }
					dharma4 = { trait:dharma4 = { save_scope_as = dharma } }
					dharma5 = { trait:dharma5 = { save_scope_as = dharma } }
					dharma6 = { trait:dharma6 = { save_scope_as = dharma } }
					dharma7 = { trait:dharma7 = { save_scope_as = dharma } }
					dharma8 = { trait:dharma8 = { save_scope_as = dharma } }
					dharma9 = { trait:dharma9 = { save_scope_as = dharma } }
				}
				add_trait_xp = {
					trait = scope:dharma
					value = scope:pity_dharma_xp
				}
			}
		}
	}
}

POD_give_dharma_xp_effect = {

}

	}

POD_enlightenment_duel_effect = {
	duel = {
		skill = $SKILL$
		value = $VALUE$
		10 = {
			compare_modifier = { value = scope:duel_value multiplier = 3.5 min = -9 }
			modifier = {
				has_variable = additional_enlightenment_chance
				add = var:additional_enlightenment_chance
			}
			show_as_tooltip = { POD_advance_dharma_effect = yes }
			trigger_event = pod1230objectives_kj.2
			#POD_change_wheel_of_ages_progress = { VALUE = -50 }
			remove_variable = additional_enlightenment_chance
		}
		50 = {
			compare_modifier = { value = scope:duel_value multiplier = -3.5 min = -49 }
			show_as_tooltip = {
				add_trait_xp = { trait = dharma9 value = -50 }
				POD_static_stress_gain = { COUNT = 150 }
			}
			if = { 
				limit = { has_variable = additional_enlightenment_chance } 
				change_variable = { name = additional_enlightenment_chance add = 10 } 
			}
			else = { 
				set_variable = { name = additional_enlightenment_chance value = 10 } 
			}
			trigger_event = pod1230objectives_kj.3
		}
	}
}
