POD_static_dread_gain = {
	if = {
		limit = { NOT = { $COUNT$ = 0 } } 
		custom_description = {
			text = add_dread # reuse vanilla loc
			value = $COUNT$

			if = {
				limit = { $COUNT$ > 0 }
				save_scope_value_as = {
					name = gain_mult
					value = { value = "this.modifier_value(dread_gain_mult)" }
				}
				add_character_modifier = POD_static_dread_gain
			}
			else = {
				save_scope_value_as = {
					name = loss_mult
					value = { value = "this.modifier_value(dread_loss_mult)" }
				}
				add_character_modifier = POD_static_dread_loss
			}
			force_character_skill_recalculation = yes
			add_dread = $COUNT$
			remove_character_modifier = POD_static_dread_gain
			remove_character_modifier = POD_static_dread_loss
			force_character_skill_recalculation = yes
		}
	}
}
