POD_static_dread_gain = {
	if = {
		limit = { NOT = { $COUNT$ = 0 } } 
		custom_description = {
			text = add_dread # reuse vanilla loc
			value = $COUNT$
			save_scope_value_as = { name = saved_dread  value = dread }
			set_variable = { name = saved_dread  value = dread days = 1 }
			
			add_dread = -50
			while = {
				limit = { dread > 0 }
				add_character_modifier = POD_static_dread_gain_2_modifier
				force_character_skill_recalculation = yes
				add_dread = -50
			}
			add_dread = 50
			
			while = {
				limit = { dread = scope:saved_dread }
				add_character_modifier = POD_static_dread_gain_1_modifier
				force_character_skill_recalculation = yes
				add_dread = 50
			}
			add_dread = -50
			while = {
				limit = { dread > 0 }
				add_character_modifier = POD_static_dread_gain_2_modifier
				force_character_skill_recalculation = yes
				add_dread = -50
			}      
			add_dread = 25
			while = {
				limit = { NOT = { dread = 25 } }
				if = { limit = { dread > 25 } add_character_modifier = POD_static_dread_gain_3_modifier }
				else = { add_character_modifier = POD_static_dread_gain_35_modifier }

				force_character_skill_recalculation = yes
				add_dread = -100
				add_dread = 25
			}
			add_dread = -600
			add_dread = 50
			add_dread = -25
			while = {
				limit = { NOT = { dread = 25 } }
				if = { limit = { dread < 25 } add_character_modifier = POD_static_dread_gain_4_modifier }
				else = { add_character_modifier = POD_static_dread_gain_45_modifier }
				force_character_skill_recalculation = yes

				add_dread = -600
				add_dread = 50
				add_dread = -25
			}
			add_dread = -600
			add_dread = scope:saved_dread
			remove_variable = saved_dread
			add_dread = $COUNT$
			remove_all_character_modifier_instances = POD_static_dread_gain_1_modifier
			remove_all_character_modifier_instances = POD_static_dread_gain_2_modifier
			remove_all_character_modifier_instances = POD_static_dread_gain_3_modifier
			remove_all_character_modifier_instances = POD_static_dread_gain_35_modifier
			remove_all_character_modifier_instances = POD_static_dread_gain_4_modifier
			remove_all_character_modifier_instances = POD_static_dread_gain_45_modifier
			force_character_skill_recalculation = yes
		}
	}
}

POD_static_stress_gain = {
	if = {
		limit = { NOT = { $COUNT$ = 0 } } 
		custom_description = {
			text = "POD_static_stress_gain"
			value = $COUNT$

			save_scope_value_as = {
				name = desired_stress
				value = { value = stress add = $COUNT$ }
			}
			save_scope_value_as = { name = saved_stress_level value = stress_level }
			POD_set_stress_effect = { STRESS = scope:desired_stress }
			if = {
				limit = { stress_level > scope:saved_stress_level }
				if = {
					limit = { stress_level = 1 }
					trigger_event = { on_action = on_stress_level_1 }
				}
				else_if = {
					limit = { stress_level = 2 }
					trigger_event = { on_action = on_stress_level_2 }
				}
				else_if = {
					limit = { stress_level = 3 }
					trigger_event = { on_action = on_stress_level_3 }
				}
				else_if = {
					limit = { stress_level = 4 }
					trigger_event = { on_action = on_stress_level_4 }
				}
			}
			else_if = {
				limit = { stress_level < scope:saved_stress_level }
				trigger_event = { on_action = on_stress_level_reduced }
			}
		}
	}
}

POD_set_stress_effect = {
	hidden_effect = {
		add_character_flag = static_stress_gain_flag # disables stress level on_actions

		save_scope_value_as = {
			name = desired_stress
			value = { value = $STRESS$ min = 0 max = 399 }
		}

		add_character_modifier = POD_mental_strain_loss
		scriptedtests_recalculate_character_modifier = yes
		add_stress = -1000

		if = {
			limit = { scope:desired_stress > 0 }
			
			save_scope_value_as = {
				name  = stress_gain_mult_delta
				value = 0
			}
			add_stress = scope:desired_stress
			
			while = {
				limit = { stress != scope:desired_stress }

				save_scope_value_as = {
					name = stress_gain_mult_delta
					value = {
						value = stress
						divide = scope:desired_stress
						add = -1
						multiply = -1
						add = scope:stress_gain_mult_delta
					}
				}
				
				remove_character_modifier = POD_mental_strain_gain
				add_character_modifier = POD_mental_strain_gain
				scriptedtests_recalculate_character_modifier = yes

				add_stress = -1000
				add_stress = scope:desired_stress
			}
		}
		
		remove_character_modifier = POD_mental_strain_loss
		remove_character_modifier = POD_mental_strain_gain
		remove_character_flag = static_stress_gain_flag
	}
}
