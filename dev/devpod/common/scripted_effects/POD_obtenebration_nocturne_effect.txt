nocturne_power_effect = {
    save_scope_as = victim
    random_list = {
	    5 = {
		    if = {
			    limit = {
				    $NOCTURNE_USER$ = {
				        NOT = { has_perk = the_darkness_within_perk }
						NOT = { has_trait = obtenebrationadvanced }
					}
				}
			    add_character_modifier = nocturne_drained_stamina_modifier
                $NOCTURNE_USER$ = {
                    add_piety = -10
				}				
			}
			else_if = {
			    limit = {
				    $NOCTURNE_USER$ = {
					    AND = {
						    OR = {
							    has_trait = obtenebrationadvanced
								has_perk = the_darkness_within_perk
							}
							NOT = { has_perk = devour_all_hope_perk }
						}
					}
				}
				POD_hunger_feeding_bloodloss_effect = {
			        VICTIM = scope:victim
			        COUNT = 1
			        VAMPIRE = $NOCTURNE_USER$
				}
				$NOCTURNE_USER$ = {
			        POD_hunger_feeding_sip_from_victim_effect = { VICTIM = scope:victim }
			    }
			    add_character_modifier = darkness_within_drained_stamina_modifier
				$NOCTURNE_USER$ = {
                    add_piety = -15
				}	
			}
			else_if = {
			    limit = {
				    $NOCTURNE_USER$ = {
				        has_perk = devour_all_hope_perk
					}
				}
				POD_hunger_feeding_bloodloss_effect = {
			        VICTIM = scope:victim
			        COUNT = 2
			        VAMPIRE = $NOCTURNE_USER$
				}
				$NOCTURNE_USER$ = {
			        POD_hunger_feeding_drink_from_victim_effect = { VICTIM = scope:victim }
			    }
			    add_character_modifier = devour_all_hope_drained_stamina_modifier
				$NOCTURNE_USER$ = {
                    add_piety = -25
				}	
			}
			
			
			modifier = {
			    add = 10
				stewardship < 5
			}
			modifier = {
			    add = 10
				stewardship < 10
			}
			modifier = {
			    add = 10
			    OR = {
				    has_perk = the_darkness_within_perk
					has_trait = obtenebrationadvanced
				}
			}
			modifier = {
			    add = 15
				has_perk = devour_all_hope_perk
			}
		}
		10 = {
		    
			modifier = {
			    add = 10
				stewardship > 15
			}
			modifier = {
			    add = 10
				stewardship > 20
			}
            modifier = {
                add = 5
				has_trait = fortitude
			}
 			modifier = {
                add = 20
				has_trait = fortitudeadvanced
			}
			modifier = {
			    add = 10
				has_trait = truefaith
			}
			modifier = {
			    add = 20
				has_trait = beatific
			}
			modifier = {
			    add = 5
				has_trait = obtenebration
			}
			modifier = {
			    add = 10
				has_trait = obtenebrationadvanced
			}
		}
	}
	
	$NOCTURNE_USER$ = {
		if = {
			limit = {
				NOT = { has_character_flag = POD_used_disciplines_in_battle_flag }
			}
			add_character_flag = POD_used_disciplines_in_battle_flag
		}
	}
}
