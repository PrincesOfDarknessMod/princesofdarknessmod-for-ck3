POD_remove_character_modifier = {
	if = {
		limit = { has_character_modifier = $MODIFIER$ }
		remove_character_modifier = $MODIFIER$
	}
}

POD_remove_all_character_modifier_instances = {
	if = {
		limit = { has_character_modifier = $MODIFIER$ }
		remove_all_character_modifier_instances = $MODIFIER$
	}
}

POD_remove_county_modifier = {
	if = {
		limit = { has_county_modifier = $MODIFIER$ }
		remove_county_modifier = $MODIFIER$
	}
}

POD_remove_house_modifier = {
	if = {
		limit = { has_house_modifier = $MODIFIER$ }
		remove_house_modifier = $MODIFIER$
	}
}

POD_remove_character_flag = {
	remove_character_flag = $FLAG$
}

POD_remove_trait = {
	remove_trait = $TRAIT$
}

POD_remove_secret = {
	random_secret = {
		limit = { secret_type = $SECRET$ }
		remove_secret = yes
	}
}

POD_add_trait_xp = {
	if = {
		limit = { has_trait = $TRAIT$ }
		add_trait_xp = {
			trait = $TRAIT$
			value = $VALUE$
		}
	}
}

# house scope
POD_create_historical_feud_with_house = {
	set_house_relation_level_effect = {
		HOUSE = $HOUSE$
		REASON = historical
		LEVEL = feud
	}
}
