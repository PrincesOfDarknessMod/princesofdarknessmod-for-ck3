POD_become_vampire_effect = {
	if = {
		limit = { 
			OR = { 
				is_immortal = yes 
				has_POD_longevity_trigger = yes
			}
		}
		set_variable = {
			name = saved_immortal_age
			value = effective_age
		}
	}
	POD_embrace_remove_incompatible_trait_effect = yes
	POD_remove_child_relation_effect = yes
	add_trait_force_tooltip = vampire
	if = {
		limit = { has_variable = saved_immortal_age }
		set_immortal_age = var:saved_immortal_age
		remove_variable = saved_immortal_age
	}
}
