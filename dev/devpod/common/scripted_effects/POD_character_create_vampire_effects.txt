POD_character_create_vampirechilde_effect = {

	#Gender
	random_list = {
		50 = { #Female
			dummy_female = { save_temporary_scope_as = vampire_gender }
		}
		50 = { #Male
			dummy_male = { save_temporary_scope_as = vampire_gender }
		}
	}
	
	if = {
		limit = {
			scope:vampiresire = {
				is_female = yes
				OR = {
					is_lamia_trigger = yes
					is_ahrimane_trigger = yes
				}
			}
		}
		dummy_female = { save_temporary_scope_as = vampire_gender }
	}
	
	scope:vampiresire = {
		if = {
			limit = { is_female = yes }
			create_character = {
				save_temporary_scope_as = created_vampirechilde
				mother = scope:vampiresire
				age = { 18 60 }
				location = scope:vampiresire.capital_province
				dynasty_house = scope:vampiresire.house
				culture = scope:vampiresire.capital_province.culture
				faith = scope:vampiresire.faith
				gender = scope:vampire_gender
				trait = vampire
				random_traits = yes
			}
		}
		else = {
			create_character = {
				save_temporary_scope_as = created_vampirechilde
				father = scope:vampiresire
				age = { 18 60 }
				location = scope:vampiresire.capital_province
				dynasty_house = scope:vampiresire.house
				culture = scope:vampiresire.capital_province.culture
				faith = scope:vampiresire.faith
				gender = scope:vampire_gender
				trait = vampire
				random_traits = yes
			}			
		}
	}
	
	scope:created_vampirechilde = {
		POD_remove_all_terrible_traits_effect = yes
		POD_character_vampire_inherit_generation_effect = yes
		POD_character_vampire_inherit_clan_effect = yes
		POD_character_vampire_clan_traits_effect = yes
		random = {
			chance = 30
			POD_character_vampire_faith_initialization_effect = yes
		}
		POD_character_vampire_clan_special_effect = yes
		POD_character_vampire_bloodpotency_initialization_effect = yes
		POD_character_vampire_bloodhunger_initialization_effect = yes
		POD_character_vampire_discipline_initialization_effect = yes
		#POD_character_vampire_predatortype_initialization_effect = yes
		POD_character_masquerade_initialization_effect = yes
		# if = {
			# limit = { 
				# is_ventrue_trigger = yes 
			# }
			# POD_character_ventrue_prey_exclusion_initialization_effect = yes
		# }						
		#set_house = scope:vampiresire.house
		POD_character_initialization_random_background_effect = yes
		#POD_clear_resonance_effect = yes
		POD_set_resonance_effect = yes
		
		POD_commander_trait_initialization_effect = yes
		POD_salubri_caste_new_vampire_initialization_effect = yes
		
		add_character_flag = {
			flag = immune_to_disease
		}					
		
		if = {
			limit = {
				house.house_head ?= { is_amenti_trigger = yes }
			}
			set_relation_worshipped_reborn = scope:child.house.house_head
		}
			
		if = {
			limit = { scope:vampiresire = { is_a_thrall = yes } }
			scope:vampiresire = {
				every_relation = {
					type = demonic_master
					limit = {
						OR = {
							is_ai = yes 
							has_character_flag = auto_thrall_child
						}
					}
					if = {
						limit = { is_ai = no }
						add_prestige = minor_prestige_loss
					}
					scope:created_vampirechilde = { set_relation_demonic_master = prev }
				}
			}
		}
		
		add_character_flag = {
			flag = had_POD_character_initialization
		}
	}
	
	clear_saved_scope = vampire_gender
}

	}

shifter_embrace_effect = {
	add_trait = abomination
	if = {
		limit = {
			OR = {
				has_trait = werewolf 
				has_trait = ajaba
			}
		}
		POD_rage_lose_the_wolf_effect = yes
	}
	else_if = {
		limit = {
			OR = {
				has_trait = bastet # TODO:trigger death without gnosis
				has_trait = gurahl
			}
		}
		POD_rage_lose_the_wolf_effect = yes
		remove_trait = gnosis
		POD_remove_gnosis_modifiers_effect = yes
	}
	else_if = {
		limit = {
			OR = {
				has_trait = mokole # TODO:trigger death with frenzy
				has_trait = rokea
			}
		}
		POD_rage_lose_the_wolf_effect = yes
		POD_become_wight_effect = yes
	}
	else_if = {
		limit = { has_trait = corax }
		death = { death_reason = death_sun }
		if = {
			limit = { 
				exists = scope:vampiresire 
				scope:vampiresire = { POD_sunny_again_trigger = no }
			}
			scope:vampiresire = { pod_death_no_killer = { DEATH_REASON = death_sun } }
		}
	}
	else_if = {
		limit = { has_trait = kitsune }
		death = { death_reason = death_fire }
		if = {
			limit = { 
				exists = scope:vampiresire 
				scope:vampiresire = { POD_immune_to_fire_trigger = no }
			}
			scope:vampiresire = { pod_death_no_killer = { DEATH_REASON = death_fire } }
		}
	}
	else_if = {
		limit = { has_trait = ratkin }
		death = { death_reason = death_old_age }
	}
	else = {
		death = { death_reason = death_failed_embrace }
	}
}
