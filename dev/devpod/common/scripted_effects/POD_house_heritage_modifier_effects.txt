POD_add_house_heritage_modifier_effect = {

	#Baali
	if = {
		limit = { scope:house_head = { is_baali_trigger = yes } }
		#if = {
		#	limit = { scope:house_head = { is_ennamaru_trigger = yes } }
		#	house = { add_house_modifier = { modifier = heritage_ennamaru_modifier } }
		#}
		house = { add_house_modifier = { modifier = heritage_baali_modifier } }
	}

	#Brujah
	else_if = {
		limit = { scope:house_head = { is_brujah_trigger = yes } }
		if = {
			limit = { scope:house_head = { is_truebrujah_trigger = yes } }
			house = { add_house_modifier = { modifier = heritage_truebrujah_modifier } }
		}
		else = {
			house = { add_house_modifier = { modifier = heritage_brujah_modifier } }
		}
	}

	#Cappadocian
	else_if = {
		limit = {
			AND = {
				scope:house_head = { is_cappadocian_trigger = yes }
				scope:house_head = { is_giovanni_trigger = no }
				scope:house_head = { is_lamia_trigger = no }
				scope:house_head = { is_harbingersofskulls_trigger = no }
			}
		}
		house = { add_house_modifier = { modifier = heritage_cappadocian_modifier } }
	}

	#Gangrel
	else_if = {
		limit = { scope:house_head = { is_gangrel_trigger = yes } }
		if = {
			limit = { scope:house_head = { is_ahrimane_trigger = yes } }
			house = { add_house_modifier = { modifier = heritage_ahrimane_modifier } }
		}
		else_if = {
			limit = { scope:house_head = { is_akunanse_trigger = yes } }
			house = { add_house_modifier = { modifier = heritage_akunanse_modifier } }
		}
		else_if = {
			limit = { scope:house_head = { is_greek_gangrel_trigger = yes } }
			house = { add_house_modifier = { modifier = heritage_greek_gangrel_modifier } }
		}
		else = {
			house = { add_house_modifier = { modifier = heritage_gangrel_modifier } }
		}
	}

	#Giovanni
	else_if = {
		limit = { scope:house_head = { is_giovanni_trigger = yes } }
		house = { add_house_modifier = { modifier = heritage_giovanni_modifier } }
	}
	
	#Harbingers of Skulls
	else_if = {
		limit = { scope:house_head = { is_harbingersofskulls_trigger = yes } }
		house = { add_house_modifier = { modifier = heritage_harbingersofskulls_modifier } }
	}

	#Lamia
	else_if = {
		limit = { scope:house_head = { is_lamia_trigger = yes } }
		house = { add_house_modifier = { modifier = heritage_lamia_modifier } }
	}

	#Setite
	else_if = {
		limit = { scope:house_head = { is_setite_trigger = yes } }
		if = {
			limit = { scope:house_head = { is_warriors_of_glycon_trigger = yes } }
			house = { add_house_modifier = { modifier = heritage_warriors_of_glycon_modifier } }
		}
		else_if = {
			limit = { scope:house_head = { is_witches_of_echidna_trigger = yes } }
			house = { add_house_modifier = { modifier = heritage_witches_of_echidna_modifier } }
		}
		else = {
			house = { add_house_modifier = { modifier = heritage_setite_modifier } }
		}
	}

	#Toreador
	else_if = {
		limit = { scope:house_head = { is_toreador_trigger = yes } }
		if = {
			limit = { scope:house_head = { is_volgirre_trigger = yes } }
			house = { add_house_modifier = { modifier = heritage_volgirre_modifier } }
		}
		else = {
			house = { add_house_modifier = { modifier = heritage_toreador_modifier } }
		}
	}
	
	#Nosferatu
	else_if = {
		limit = { scope:house_head = { is_nosferatu_trigger = yes } }
		#if = {
		#	limit = { scope:house_head = { is_mnemachien_trigger = yes } }
		#	house = { add_house_modifier = { modifier = heritage_mnemachien_modifier } }
		#}
		house = { add_house_modifier = { modifier = heritage_nosferatu_modifier } }
	}
	
	#Ravnos
	if = {
		limit = { scope:house_head = { is_ravnos_trigger = yes } }
		#if = {
		#	limit = { scope:house_head = { is_erk_trigger = yes } }
		#	house = { add_house_modifier = { modifier = heritage_erk_modifier } }
		#}
		house = { add_house_modifier = { modifier = heritage_ravnos_modifier } }
	}
}

POD_remove_house_heritage_modifier_effect = {
	house = {
		POD_remove_house_modifier = { MODIFIER = heritage_ahrimane_modifier }
		POD_remove_house_modifier = { MODIFIER = heritage_akunanse_modifier }
		POD_remove_house_modifier = { MODIFIER = heritage_brujah_modifier }
		POD_remove_house_modifier = { MODIFIER = heritage_cappadocian_modifier }
		POD_remove_house_modifier = { MODIFIER = heritage_gangrel_modifier }
		POD_remove_house_modifier = { MODIFIER = heritage_giovanni_modifier }
		POD_remove_house_modifier = { MODIFIER = heritage_greek_gangrel_modifier }
		POD_remove_house_modifier = { MODIFIER = heritage_harbingersofskulls_modifier }
		POD_remove_house_modifier = { MODIFIER = heritage_lamia_modifier }
		POD_remove_house_modifier = { MODIFIER = heritage_setite_modifier }
		POD_remove_house_modifier = { MODIFIER = heritage_toreador_modifier }
		POD_remove_house_modifier = { MODIFIER = heritage_truebrujah_modifier }
		POD_remove_house_modifier = { MODIFIER = heritage_volgirre_modifier }
		POD_remove_house_modifier = { MODIFIER = heritage_warriors_of_glycon_modifier }
		POD_remove_house_modifier = { MODIFIER = heritage_witches_of_echidna_modifier }
		POD_remove_house_modifier = { MODIFIER = heritage_ravnos_modifier }
		POD_remove_house_modifier = { MODIFIER = heritage_nosferatu_modifier }

	}
}
