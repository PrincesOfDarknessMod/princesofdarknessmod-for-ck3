POD_add_house_heritage_modifier_effect = {
	
	#Ahrimane
	if = {
		limit = { house.house_head = { is_ahrimane_trigger = yes } }
		house = { add_house_modifier = { modifier = heritage_ahrimane_modifier } }
	}

	#Akunanse
	if = {
		limit = { house.house_head = { is_akunanse_trigger = yes } }
		house = { add_house_modifier = { modifier = heritage_akunanse_modifier } }
	}

	#Banu Haqim
	if = {
		limit = { house.house_head = { is_assamite_trigger = yes } }
		house = { add_house_modifier = { modifier = heritage_banu_haqim_modifier } }
	}

	#Baali
	if = {
		limit = { house.house_head = { is_baali_trigger = yes } }
		house = { add_house_modifier = { modifier = heritage_baali_modifier } }
	}

	#Brujah
	if = {
		limit = {
			AND = {
				house.house_head = { is_brujah_trigger = yes }
				house.house_head = { is_truebrujah_trigger = no }
			}
		}
		house = { add_house_modifier = { modifier = heritage_brujah_modifier } }
	}

	#Cappadocian
	if = {
		limit = {
			AND = {
				house.house_head = { is_cappadocian_trigger = yes }
				house.house_head = { is_giovanni_trigger = no }
				house.house_head = { is_lamia_trigger = no }
			}
		}
		house = { add_house_modifier = { modifier = heritage_cappadocian_modifier } }
	}

	#Gangrel
	if = {
		limit = {
			AND = {
				house.house_head = { is_gangrel_trigger = yes }
				house.house_head = { is_ahrimane_trigger = no }
				house.house_head = { is_akunanse_trigger = no }
				house.house_head = { is_greek_gangrel_trigger = no }
			}
		}
		house = { add_house_modifier = { modifier = heritage_gangrel_modifier } }
	}

	#Giovanni
	if = {
		limit = { house.house_head = { is_giovanni_trigger = yes } }
		house = { add_house_modifier = { modifier = heritage_giovanni_modifier } }
	}

	#Greek Gangrel
	if = {
		limit = { house.house_head = { is_greek_gangrel_trigger = yes } }
		house = { add_house_modifier = { modifier = heritage_greek_gangrel_modifier } }
	}

	#Impundulu
	if = {
		limit = { house.house_head = { is_impundulu_trigger = yes } }
		house = { add_house_modifier = { modifier = heritage_impundulu_modifier } }
	}

	#Kiasyd
	if = {
		limit = { house.house_head = { is_kiasyd_trigger = yes } }
		house = { add_house_modifier = { modifier = heritage_kiasyd_modifier } }
	}

	#Lamia
	if = {
		limit = { house.house_head = { is_lamia_trigger = yes } }
		house = { add_house_modifier = { modifier = heritage_lamia_modifier } }
	}

	#Lasombra
	if = {
		limit = { house.house_head = { is_lasombra_trigger = yes } }
		house = { add_house_modifier = { modifier = heritage_lasombra_modifier } }
	}

	#Maeghar
	if = {
		limit = { house.house_head = { is_maeghar_trigger = yes } }
		house = { add_house_modifier = { modifier = heritage_maeghar_modifier } }
	}

	#Malkavian
	if = {
		limit = { house.house_head = { is_malkavian_trigger = yes } }
		house = { add_house_modifier = { modifier = heritage_malkavian_modifier } }
	}

	#Nagaraja
	if = {
		limit = { house.house_head = { is_nagaraja_trigger = yes } }
		house = { add_house_modifier = { modifier = heritage_nagaraja_modifier } }
	}

	#Nosferatu
	if = {
		limit = { house.house_head = { is_nosferatu_trigger = yes } }
		house = { add_house_modifier = { modifier = heritage_nosferatu_modifier } }
	}

	#Ravnos
	if = {
		limit = { house.house_head = { is_ravnos_trigger = yes } }
		house = { add_house_modifier = { modifier = heritage_ravnos_modifier } }
	}

	#Salubri
	if = {
		limit = { house.house_head = { is_salubri_trigger = yes } }
		house = { add_house_modifier = { modifier = heritage_salubri_modifier } }
	}

	#Setite
	if = {
		limit = {
			AND = {
				house.house_head = { is_setite_trigger = yes }
				house.house_head = { is_warriors_of_glycon_trigger = no }
				house.house_head = { is_witches_of_echidna_trigger = no }
			}
		}
		house = { add_house_modifier = { modifier = heritage_setite_modifier } }
	}

	#Toreador
	if = {
		limit = { house.house_head = { is_toreador_trigger = yes } }
		house = { add_house_modifier = { modifier = heritage_toreador_modifier } }
	}

	#Tremere
	if = {
		limit = { house.house_head = { is_tremere_trigger = yes } }
		house = { add_house_modifier = { modifier = heritage_tremere_modifier } }
	}

	#Tzimisce
	if = {
		limit = { house.house_head = { is_tzimisce_trigger = yes } }
		house = { add_house_modifier = { modifier = heritage_tzimisce_modifier } }
	}

	#Ventrue
	if = {
		limit = { house.house_head = { is_ventrue_trigger = yes } }
		house = { add_house_modifier = { modifier = heritage_ventrue_modifier } }
	}

	#Volgirre
	if = {
		limit = { house.house_head = { is_volgirre_trigger = yes } }
		house = { add_house_modifier = { modifier = heritage_volgirre_modifier } }
	}

	#Warriors of Glycon
	if = {
		limit = {
			AND = {
				house.house_head = { is_setite_trigger = yes }
				house.house_head = { is_warriors_of_glycon_trigger = yes }
				house.house_head = { is_witches_of_echidna_trigger = no }
			}
		}
		house = { add_house_modifier = { modifier = heritage_warriors_of_glycon_modifier } }
	}

	#Witches of Echidna
	if = {
		limit = {
			AND = {
				house.house_head = { is_setite_trigger = yes }
				house.house_head = { is_warriors_of_glycon_trigger = no }
				house.house_head = { is_witches_of_echidna_trigger = yes }
			}
		}
		house = { add_house_modifier = { modifier = heritage_witches_of_echidna_modifier } }
	}
}
