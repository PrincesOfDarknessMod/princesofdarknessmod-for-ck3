POD_become_wraith_effect = {
	if = {
		limit = { 
			OR = { 
				is_immortal = yes 
				has_POD_longevity_trigger = yes
			}
		}
		set_local_variable = {
			name = saved_immortal_age
			value = effective_age
		}
	}
	hidden_effect = {
		if = {
			limit = { has_trait = lament }
			add_character_flag = {
				flag = has_lament
				days = 10
			}
		}
		if = {
			limit = { is_pregnant = yes } 
			end_pregnancy = yes
			pregnancy_cleanup_effect = yes
		}
		if = {
			limit = { has_trait = drunkard }
			remove_trait = drunkard
		}
		if = {
			limit = { has_trait = hashishiyah }
			remove_trait = hashishiyah
		}
		if = {
			limit = { has_trait = rakish }
			remove_trait = rakish
		}
		if = {
			limit = { has_trait = flagellant }
			remove_trait = flagellant
		}
		if = {
			limit = { has_trait = comfort_eater }
			remove_trait = comfort_eater
		}
		if = {
			limit = { has_trait = inappetetic }
			remove_trait = inappetetic
		}
		if = {
			limit = { has_trait = athletic }
			remove_trait = athletic
		}
	}
	hidden_effect = {
		POD_remove_all_supernatural_atribute_effect = yes
		POD_remove_supernatural_relation_effect = yes
	}
	
	add_trait = wraith
	
	if = {
		limit = { 
			OR = { 
				has_character_flag = has_lament
				AND = { 
					exists = scope:necromancer
					scope:necromancer = {
						OR = {
							has_perk = necromancy_possession_perk 
							has_perk = ebon_binding_perk
						}
					} 
				}
			} 
		}
		add_trait = lament
	}
	else = {
		add_character_modifier = enfant_modifier
		trigger_event = {
			id = POD_wraith_maintenance.20
			days = { 7 30 }
		}
	}
	POD_wraith_initialization_effect = yes
	POD_arcanoi_initiation_effect = yes
	if = {
		limit = { exists = local_var:saved_immortal_age }
		set_immortal_age = local_var:saved_immortal_age
	}
}

POD_become_spectre_effect = {
	if = {
		limit = {
			has_trait = risen 
		}
		remove_trait = risen
		hidden_effect = { 
			POD_remove_conduit_modifiers_effect = yes
			POD_remove_conduit_fetter_effect = yes
		}
	}
	if = {
		limit = {
			has_trait = lament 
		}
		remove_trait = lament
		hidden_effect = { 
			refund_perks = horror_lifestyle
			add_horror_lifestyle_perk_points = -100
		}
	}
	add_trait_force_tooltip = spectre
	if = {
		limit = { is_landed = yes }
		change_government = hive_mind_government
	}
	else = { move_to_pool = yes }
	hidden_effect = { 
		refund_perks = thorn_lifestyle
		add_thorn_lifestyle_perk_points = -100
		trigger_event = POD_wraith_maintenance.2
		save_temporary_scope_as = spectre
		every_relation = {
			type = possessed_target
			POD_wraith_end_possession = {
				WRAITH = scope:spectre
				PUPPET = this
			}
		}
		every_relation = {
			type = necromancer
			POD_unbind_wraith_effect = {
				WRAITH = scope:spectre
				NECROMANCER = this
			}
		}
		every_relation = {
			type = separated_shadow_servant
			limit = { is_alive = yes }
			death = {
			    death_reason = death_disappearance
			}
		}
	}
}


POD_wraith_initialization_effect = {
	if = { 
		limit = { 
			OR = { 
				exists = scope:character_being_changed
				is_from_ruler_designer = yes
			}
		}
		POD_wraith_chargen_starting_xp_effect = yes
	}
	if = {
		limit = {
			is_landed = yes
			has_wraith_government = no
		}
		POD_character_wraith_government_initiation_effect = yes
	}
	if = {
		limit = {
			NOT = { religion = { is_in_family = rf_wraiths } }
		}
		POD_character_wraith_faith_initiation_effect = yes
	}
	if = {
		limit = {
			has_pathos_trigger = no
		}
		add_character_modifier = pathos5_modifier
	}
	if = {
		limit = {
			NOT = { has_character_flag = immune_to_disease }
		}
		add_character_flag = immune_to_disease
	}
	if = {
		limit = { 
			NOT = { has_trait = risen }
			NOT = { has_trait = lament }
			NOT = { has_character_modifier = enfant_modifier }
			NOT = { has_variable = spirit_location }
		}
		POD_send_to_dark_kingdom_effect = yes
	}
	if = {
		limit = { is_ai = no }
		if = {
			limit = { NOT = { exists = var:passion_count } }
			set_variable = {
				name = passion_count
				value = 10
			}
		}
	}
}

				}

						geographical_region = world_steppe
					}
				}

POD_pathos_increase_effect = {
	if = {
		limit = {has_character_modifier = pathos0_modifier}
		remove_character_modifier = pathos0_modifier
		add_character_modifier = pathos1_modifier
	}
	else_if = {
		limit = {has_character_modifier = pathos1_modifier}
		remove_character_modifier = pathos1_modifier
		add_character_modifier = pathos2_modifier
	}
	else_if = {
		limit = {has_character_modifier = pathos2_modifier}
		remove_character_modifier = pathos2_modifier
		add_character_modifier = pathos3_modifier
	}
	else_if = {
		limit = {has_character_modifier = pathos3_modifier}
		remove_character_modifier = pathos3_modifier
		add_character_modifier = pathos4_modifier
	}
	else_if = {
		limit = {has_character_modifier = pathos4_modifier}
		remove_character_modifier = pathos4_modifier
		add_character_modifier = pathos5_modifier
	}
	else = {
		random_character_artifact = {
			limit = {
				trigger_if = {
					limit = { NOT = { has_variable = greater_sekhem_vessel } }
					has_variable = stored_sekhem
					var:stored_sekhem < 5
				}
				trigger_else = {
					has_variable = stored_sekhem
					var:stored_sekhem < 20
				}
			}
			change_variable = { name = stored_sekhem add = 1 }
			set_vessel_artifact_name_effect = yes
		}
	}
}

POD_bind_wraith_effect = {

	
	if = {
		limit = {
			$NECROMANCER$ = {
				has_hook_of_type = {
					target = $WRAITH$
					type = necromancy_bind_hook
				}	
			}
		}
		$NECROMANCER$ = {
			
			add_hook = {
				target = $WRAITH$
				type = necromancy_bind_strong_hook
			}
			reverse_add_opinion = {
				modifier = POD_binding_strong_opinion
				target = $WRAITH$
			}
		}
		$WRAITH$ = {
			remove_opinion = {
				target = $NECROMANCER$
				modifier = POD_binding_opinion
			}
		}
		
	}
	else_if = {
		limit = {
			$NECROMANCER$ = {
				has_hook_of_type = {
					target = $WRAITH$
					type = necromancy_bind_strong_hook
				}	
			}
		}
		$NECROMANCER$ = {
			remove_hook = {
				target = $WRAITH$
				type = necromancy_bind_strong_hook
			}	
			add_hook = {
				target = $WRAITH$
				type = necromancy_bind_permanent_hook
			}
			reverse_add_opinion = {
				modifier = POD_binding_permanent_opinion
				target = $WRAITH$
			}
		}
		$WRAITH$ = {
			remove_opinion = {
				target = $NECROMANCER$
				modifier = POD_binding_strong_opinion
			}
		}
		
	
	}
	else = {
		$NECROMANCER$ = {
			add_hook = {
				target = $WRAITH$
				type = necromancy_bind_hook
			}
			reverse_add_opinion = {
				modifier = POD_binding_opinion
				target = $WRAITH$
			}
		}
	}
	$NECROMANCER$ = {
		set_relation_bound_wraith = $WRAITH$
	}

	$WRAITH$ = {
		if = {
			limit = {
				is_in_umbra_trigger = yes
			}
			remove_character_modifier = traveling_underworld_modifier
			remove_variable = spirit_location
		}
	}
}


POD_unbind_wraith_effect = {
	$NECROMANCER$ = { save_scope_as = necromancer }
	$WRAITH$ = { save_scope_as = wraith }
	
	if = {
		limit = {
			scope:necromancer = {
				has_hook_of_type = {
					target = scope:wraith
					type = necromancy_bind_strong_hook
				}	
			}
		}
		scope:necromancer = {
			remove_hook = {
				target = scope:wraith
				type = necromancy_bind_strong_hook
			}
		}
		scope:wraith = {
			remove_opinion = {
				target = scope:necromancer
				modifier = POD_binding_strong_opinion
			}
		}
	}
	else_if = {
		limit = {
			scope:necromancer = {
				has_hook_of_type = {
					target = scope:wraith
					type = necromancy_bind_permanent_hook
				}	
			}
		}
		scope:necromancer = {
			remove_hook = {
				target = scope:wraith
				type = necromancy_bind_permanent_hook
			}
		}
		scope:wraith = {
			remove_opinion = {
				target = scope:necromancer
				modifier = POD_binding_permanent_opinion
			}
		}
	}
	else = {
		scope:necromancer = {
			remove_hook = {
				target = scope:wraith
				type = necromancy_bind_hook
			}
		}
		scope:wraith = {
			remove_opinion = {
				modifier = POD_binding_opinion
				target = scope:necromancer
			}
		}
	}
	scope:necromancer = {
		remove_relation_bound_wraith = scope:wraith
	}
}

POD_wraith_weak_possession_effect = {
	$WRAITH$ = {
		add_hook = {
			target = $PUPPET$
			type = possessed_weak_hook
		}
		reverse_add_opinion = {
			modifier = POD_weak_possession_opinion
			target = $PUPPET$
		}
		if = {
			limit = { num_of_relation_necromancer >= 1 }
			every_relation = {
				type = necromancer
				send_interface_toast = {
					title = POD_wraith_possession_toast_title
					custom_tooltip = POD_wraith_possession_toast_text
					left_icon = scope:actor
					right_icon = scope:recipient
					add_hook = {
						target = $PUPPET$
						type = necromancy_bind_hook
					}
					reverse_add_opinion = {
						modifier = POD_possessed_by_bound_ghost
						target = $PUPPET$
					}
				}
			}
		}
	}
	$PUPPET$ = {
		while = {
			count = $WRAITH$.half_prowess
			add_character_modifier = wraith_possession_prowess_gain_modifier
		}
		set_variable = {name = wraith value = $WRAITH$}
		add_character_flag = { 
			flag = possessed
			days = 3650
		}
		trigger_event = {
			id = POD_wraith_possession.20
			days = 3650
		}
	}
	

}

POD_wraith_end_possession = {
	$WRAITH$ = { save_scope_as = wraith }
	$PUPPET$ = { save_scope_as = puppet }

	scope:puppet = { 
		POD_wraith_possession_remove_atribute_gain_effect = yes
	}
	
	scope:wraith = {
		scope:puppet = {
			remove_opinion = {
				target = prev
				modifier = POD_weak_possession_opinion
			
			}
			remove_opinion = {
				target = prev
				modifier = POD_possession_opinion
			
			}
			remove_opinion = {
				target = prev
				modifier = POD_complete_possession_opinion
			
			}
			remove_relation_possessesing_wraith = prev
			remove_variable = wraith
			remove_character_flag = possessed
		}
		remove_hook = {
			type = possessed_hook
			target = scope:puppet
		}
		remove_hook = {
			type = possessed_weak_hook
			target = scope:puppet
		}
		if = {					  
			limit = { num_of_relation_necromancer >= 1 }
			every_relation = {
				type = necromancer
				remove_hook = {
					target = scope:puppet
					type = necromancy_bind_permanent_hook
				}
				remove_hook = {
					target = scope:puppet
					type = necromancy_bind_strong_hook
				}
				remove_hook = {
					target = scope:puppet
					type = necromancy_bind_hook
				}
				save_scope_as = necromancer
				scope:puppet = {
					remove_opinion = {
						target = scope:necromancer
						modifier = POD_possessed_by_bound_ghost
					}
				}
			}
		}
	
	}

}

		type = $TYPE$
		modifier = artifact_stress_gain_1_penalty_modifier
		save_scope_as = newly_created_artifact
		wealth = scope:wealth
		quality = scope:quality
		#decaying = no
	}
	scope:newly_created_artifact = {
		if = {
			limit = {
				scope:owner = { has_royal_court = yes }
			}

		}
		add_scaled_artifact_modifier_minor_prestige_effect = yes
		if = {
			limit = {

		}
		else_if = {
			limit = {
				rarity = famed
			}
			add_artifact_modifier = artifact_dread_baseline_add_1_modifier
		}
		add_soulforged_artifact_modifier_effect = { KILLED = scope:killed }

get_soulforged_artifact_quality_effect = {
	hidden_effect = {
		random_list = {
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 0.5
				} 
			}
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 1
				} 
			}
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 2
				} 
			}
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 2.5
				} 
			}
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 3
				} 
			}
		}
	}
	
	# Calculate the final quality of the artifact
	save_scope_value_as = {
		name = quality
		value = {
			value = 0
			
			scope:killed = {
				# Bonus quality from high skills
				add = {	
					if = {
						limit = { scope:armor = yes }
						value = armor_inspiration_average_skill_value
					}
					else_if = {
						limit = { scope:weapon = yes }
						value = weapon_inspiration_average_skill_value
					}
					else_if = {
						limit = { scope:shinies = yes }
						value = smith_inspiration_average_skill_value
					}
					else_if = {
						limit = { scope:book = yes }
						value = book_inspiration_average_skill_value
					}
					else_if = {
						limit = { scope:furniture = yes }
						value = weaver_inspiration_average_skill_value
					}
					else_if = {
						limit = { scope:trinket = yes }
						value = artisan_inspiration_average_skill_value
					}
					multiply = {
						value = quality_bonus_per_skill_level_value
						add = scope:random_inspired_skill_quality_bonus
					}
				}

				add = {	
					if = {
						limit = {
							OR = {
								has_trait = education_martial_4
								has_trait = education_diplomacy_4
								has_trait = education_learning_4
								has_trait = education_stewardship_4
								has_trait = education_intrigue_4
								has_trait = education_martial_prowess_4
							}
						}
						value = 20
					}
					else_if = {
						limit = {
							OR = {
								has_trait = education_martial_3
								has_trait = education_diplomacy_3
								has_trait = education_learning_3
								has_trait = education_stewardship_3
								has_trait = education_intrigue_3
								has_trait = education_martial_prowess_3
							}
						}
						value = 15
					}
					else_if = {
						limit = {
							OR = {
								has_trait = education_martial_2
								has_trait = education_diplomacy_2
								has_trait = education_learning_2
								has_trait = education_stewardship_2
								has_trait = education_intrigue_2
								has_trait = education_martial_prowess_2
							}
						}
						value = 10
					}
					else = {
						value = 5
					}
				}
				
				if = {
					limit = { scope:creator = { has_trait = necromancyadvanced } }
					add = 10
				}
				else_if = {
					limit = { scope:creator = { has_trait = necromancy } }
					add = 5
				}
				
				if = {
					limit = { scope:creator = { has_trait = oblivionadvanced } }
					add = 10
				}
				else_if = {
					limit = { scope:creator = { has_trait = oblivion } }
					add = 5
				}
				
				if = {
					limit = { scope:creator = { has_perk = forge_affinity_perk } }
					add = 20
				}
			}
			
			# Can't have negative quality
			min = 1
		}
	}
}

	else_if = {

	else_if = {

	}
	remove_list_variable = { name = owned_fetter_list target = $FETTER$ }
}

POD_rank_down_fetter_effect = {
	$FETTER$ = { change_variable = { name = fetter_value subtract = 1 } }
	if = {
		limit = { $FETTER$ = { var:fetter_value = 0 } }
		remove_list_variable = { name = owned_fetter_list target = $FETTER$ }
