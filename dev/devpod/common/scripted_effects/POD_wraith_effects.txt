POD_become_wraith_effect = {
	save_temporary_scope_value_as = {
		name = saved_immortal_age
		value = POD_visual_age
	}
	hidden_effect = { POD_remove_splat_traits_effect = yes }
	add_trait = wraith
	if = {
		limit = { exists = scope:saved_immortal_age }
		hidden_effect = { set_immortal_age = scope:saved_immortal_age }
	}
	hidden_effect = {
		POD_remove_wrong_supernatural_traits_effect = yes
		POD_remove_wrong_supernatural_modifiers_effect = yes
		POD_remove_supernatural_relation_effect = yes
		POD_side_effects_from_changing_splats = yes
		
		if = {
			limit = { is_pregnant = yes } 
			end_pregnancy = yes
			pregnancy_cleanup_effect = yes
		}
		if = {
			limit = { has_trait = drunkard }
			remove_trait = drunkard
		}
		if = {
			limit = { has_trait = hashishiyah }
			remove_trait = hashishiyah
		}
		if = {
			limit = { has_trait = rakish }
			remove_trait = rakish
		}
		if = {
			limit = { has_trait = flagellant }
			remove_trait = flagellant
		}
		if = {
			limit = { has_trait = comfort_eater }
			remove_trait = comfort_eater
		}
		if = {
			limit = { has_trait = inappetetic }
			remove_trait = inappetetic
		}
		if = {
			limit = { has_trait = athletic }
			remove_trait = athletic
		}
	}
	
	if = {
		limit = {
			scope:necromancer ?= {
				OR = {
					has_perk = ns_daemonic_possession_perk 
					has_perk = ebon_binding_perk
				} 
			}
		}
		add_trait = lament
		set_character_faith = faith:drop_outs
	}
	else_if = {
		limit = { has_trait = lament }
		set_character_faith = faith:drop_outs
	}
	else = {
		add_character_modifier = caul_modifier
		trigger_event = {
			id = POD_wraith_maintenance.20
			days = { 7 30 }
		}
	}
	POD_wraith_initialization_effect = yes
	# POD_arcanoi_initiation_effect = yes
	if = {
		limit = { is_ai = no }
		trigger_event = podintroductions.2302
	}
}

POD_become_spectre_effect = {
	if = {
		limit = {
			has_trait = risen 
		}
		remove_trait = risen
		POD_remove_vampire_risen_discipline_traits_effect = yes
		hidden_effect = { 
			POD_remove_conduit_modifiers_effect = yes
			POD_remove_conduit_fetter_effect = yes
		}
	}
	if = {
		limit = {
			has_trait = lament 
		}
		remove_trait = lament
		hidden_effect = { 
			refund_perks = horror_lifestyle
			add_horror_lifestyle_perk_points = -100
		}

	if = {
		limit = {
			has_pathos_trigger = yes
		}
		POD_remove_pathos_modifiers_effect = yes
	}

		limit = {
			has_angst_trigger = no

	add_trait_xp = {
		trait = wraith

		refund_perks = thorn_lifestyle
		add_thorn_lifestyle_perk_points = -100

		every_relation = {
			type = possessed_target
			POD_wraith_end_possession = {
				WRAITH = scope:spectre
				PUPPET = this
			}
		}
		every_relation = {
			type = separated_shadow_servant
			limit = { is_alive = yes }
			death = {
			    death_reason = death_disappearance
			}
		}
	}
	if = {
		limit = { has_variable = passion_count }
		remove_variable = passion_count	
		remove_character_flag = passion_find_killer_flag
		remove_character_flag = passion_amass_wealth_flag
		remove_character_flag = passion_become_exalted_flag
		remove_character_flag = passion_become_paragon_flag
		remove_character_flag = passion_create_a_treasury_flag
		remove_character_flag = passion_build_a_great_work_flag
		remove_character_flag = passion_make_a_friend_flag
		remove_character_flag = passion_protect_family_flag
		remove_character_flag = passion_kill_rival_flag
	}
	POD_character_spectre_faith_initiation_effect = yes
	give_infernalist_secret_or_trait_effect = yes
}

POD_pathos_income_effect = {
	if = {
		limit = {
			OR = {
				pathos_difference > $COUNT$
				NOT = {
					any_character_artifact = {
						NOT = { has_variable = temporary_vessel }
						has_variable = stored_sekhem
						vessel_difference > 0
					}
				}
			}
		}
		POD_pathos_increase_multiple_effect = { COUNT = $COUNT$ }
	}
	else = {
		save_scope_value_as = {
			name = extra_pathos
			value = {
				value = $COUNT$
				subtract = pathos_difference
			}
		}
		POD_pathos_increase_multiple_effect = { COUNT = pathos_difference }
		ordered_character_artifact = {
			limit = {
				NOT = { has_variable = temporary_vessel }
				has_variable = stored_sekhem 
				vessel_difference > 0
			}
			order_by = {
				value = 0
				add = vessel_difference
			}
			position = 0
			POD_pathos_vessel_increase_multiple_effect = { COUNT = scope:extra_pathos OWNER = prev }
		}
	}
}

POD_pathos_vessel_increase_multiple_effect = {
	if = {
		limit = { vessel_difference >= $COUNT$ }
		custom_description = {
			text = POD_store_pathos
			subject = $OWNER$
			value = $COUNT$
			change_variable = { name = stored_sekhem add = $COUNT$ }
		}
	}
	else = {
		save_scope_value_as = {
			name = ex_stress_loss
			value = {
				value = $COUNT$
				subtract = vessel_difference
				multiply = -15
			}
		}
		custom_description = {
			text = POD_store_pathos
			subject = $OWNER$
			value = vessel_difference
			change_variable = { name = stored_sekhem add = vessel_difference }
		}

POD_angst_income_effect = {
	if = {
		limit = {
			OR = {
				angst_difference > $COUNT$
				NOT = {
					any_character_artifact = {
						NOT = { has_variable = temporary_vessel }
						has_variable = stored_sekhem
						vessel_difference > 0
					}
				}
			}
		}
		POD_angst_increase_multiple_effect = { COUNT = $COUNT$ }
	}
	else = {
		save_scope_value_as = {
			name = extra_angst
			value = {
				value = $COUNT$
				subtract = angst_difference
			}
		}
		POD_angst_increase_multiple_effect = { COUNT = angst_difference }
		ordered_character_artifact = {
			limit = {
				NOT = { has_variable = temporary_vessel }
				has_variable = stored_sekhem 
				vessel_difference > 0
			}
			order_by = {
				value = 0
				add = vessel_difference
			}
			position = 0
			POD_angst_vessel_increase_multiple_effect = { COUNT = scope:extra_angst OWNER = prev }
		}
	}
}

POD_recalculate_fetter_points = {
	hidden_effect = {
		add_trait_xp = {
			trait = wraith
			track = fetter
			value = -100
		}
		every_in_list = {
			variable = owned_fetter_list
			prev = {
				add_trait_xp = {
					trait = wraith
					track = fetter

						value = prev.var:level
						multiply = 10
					}
				}
			}
		}
	}
}

POD_assign_fetter_points = {
	random_in_list = {
		variable = owned_fetter_list
		limit = { var:type = flag:$TYPE$ }
		set_variable = {
			name = level
			value = $LEVEL$
		}
		save_scope_as = fetter
	}
	hidden_effect = {
		if = { limit = { POD_has_fetter_of_type = { TYPE = $TYPE$ } }
			POD_fetter_modifier_from_type = { TYPE = flag:$TYPE$ ACTION = remove }
			POD_recalculate_fetter_points = yes
			POD_fetter_modifier_from_type = { TYPE = flag:$TYPE$ ACTION = add }
		}
	}	
}

# OBJECT = Scope
POD_make_object_fetter_type = {
	random_in_list = {
		variable = owned_fetter_list
		limit = { var:type = flag:$TYPE$ }
		if = { limit = { has_variable = object } 
			var:object = { remove_variable = fetter_owner remove_variable = fetter_type remove_variable = fetter_level } 
		}
		set_variable = {
			name = object
			value = $OBJECT$
		}
		save_scope_as = fetter
	}
	$OBJECT$ = {
		set_variable = { name = fetter_owner value = PREV }
		set_variable = { name = fetter_type value = flag:$TYPE$ }
		set_variable = { name = fetter_level value = scope:fetter.var:level }
	}
}

	custom_tooltip = {
		text = POD_destroy_fetter_text
		save_scope_as = fetter
		var:fetter_owner = {
			hidden_effect = {
				POD_fetter_modifier_from_type = { TYPE = scope:fetter.var:fetter_type ACTION = remove }
				every_in_list = {
					variable = owned_fetter_list
					limit = { var:type = scope:fetter.var:fetter_type }
					PREV = { remove_list_variable = { name = owned_fetter_list target = PREV } }
				}
				POD_recalculate_fetter_points = yes
			}
		}
		if = { limit = { exists = learning } remove_relation_fetter = var:fetter_owner }
		remove_variable = fetter_owner 
		remove_variable = fetter_type 
		remove_variable = fetter_level 
	}
}

POD_wraith_increase_possession = {
	if = {
		limit = { NOT = { $PUPPET$ = { has_relation_possessesing_wraith = $WRAITH$ } } }
		$PUPPET$ = { set_relation_possessesing_wraith = $WRAITH$ }
		POD_wraith_weak_possession_effect = { WRAITH = $WRAITH$ PUPPET = $PUPPET$ }
		$WRAITH$ = { add_character_modifier = wraith_possesing_modifier }
	}
	else_if = {
		limit = {
			$WRAITH$ = {
				has_hook_of_type  = {
					target = $PUPPET$
					type = possessed_weak_hook
				}
			}
		}
		POD_wraith_possession_effect = { WRAITH = $WRAITH$ PUPPET = $PUPPET$}
	
	}
	else = {
		POD_wraith_complete_possession_effect = { WRAITH = $WRAITH$ PUPPET = $PUPPET$}
	}
}

POD_bind_wraith_effect = {
	if = {
		limit = {
			$NECROMANCER$ = {
				has_hook_of_type = {
					target = $WRAITH$
					type = necromancy_bind_permanent_hook
				}	
			}
		}
	}
	else_if = {
		limit = {
			$NECROMANCER$ = {
				has_hook_of_type = {
					target = $WRAITH$
					type = necromancy_bind_strong_hook
				}	
			}
		}
		$NECROMANCER$ = {
			remove_hook = {
				target = $WRAITH$
				type = necromancy_bind_strong_hook
			}	
			add_hook = {
				target = $WRAITH$
				type = necromancy_bind_permanent_hook
			}
			reverse_add_opinion = {
				modifier = POD_binding_permanent_opinion
				target = $WRAITH$
			}
		}
		$WRAITH$ = {
			remove_opinion = {
				target = $NECROMANCER$
				modifier = POD_binding_strong_opinion
			}
		}
	}
	else_if = {
		limit = {
			$NECROMANCER$ = {
				has_hook_of_type = {
					target = $WRAITH$
					type = necromancy_bind_hook
				}	
			}
		}
		$NECROMANCER$ = {
			remove_hook = {
				target = $WRAITH$
				type = necromancy_bind_hook
			}	
			add_hook = {
				target = $WRAITH$
				type = necromancy_bind_strong_hook
			}
			reverse_add_opinion = {
				modifier = POD_binding_strong_opinion
				target = $WRAITH$
			}
		}
		$WRAITH$ = {
			remove_opinion = {
				target = $NECROMANCER$
				modifier = POD_binding_opinion
			}
		}
	}
	else = {
		$NECROMANCER$ = {
			add_hook = {
				target = $WRAITH$
				type = necromancy_bind_hook
			}
			reverse_add_opinion = {
				modifier = POD_binding_opinion
				target = $WRAITH$
			}
		}
	}
	$NECROMANCER$ = {
		if = {
			limit = { NOT = { has_relation_bound_wraith = $WRAITH$ } }
			set_relation_bound_wraith = $WRAITH$
		}
	}
	$WRAITH$ = {
		if = {
			limit = {
				is_in_umbra_trigger = yes
			}
			remove_character_modifier = traveling_underworld_modifier
			remove_variable = spirit_location
		}
	}
}

POD_unbind_wraith_effect = {
	$NECROMANCER$ = { save_temporary_scope_as = necromancer }
	$WRAITH$ = { save_temporary_scope_as = wraith }
	
	if = {
		limit = {
			scope:necromancer = {
				has_hook_of_type = {
					target = scope:wraith
					type = necromancy_bind_permanent_hook
				}	
			}
		}
		scope:necromancer = {
			remove_hook = {
				target = scope:wraith
				type = necromancy_bind_permanent_hook
			}
		}
		scope:wraith = {
			remove_opinion = {
				target = scope:necromancer
				modifier = POD_binding_permanent_opinion
			}
		}
	}
	else_if = {
		limit = {
			scope:necromancer = {
				has_hook_of_type = {
					target = scope:wraith
					type = necromancy_bind_strong_hook
				}	
			}
		}
		scope:necromancer = {
			remove_hook = {
				target = scope:wraith
				type = necromancy_bind_strong_hook
			}
		}
		scope:wraith = {
			remove_opinion = {
				target = scope:necromancer
				modifier = POD_binding_strong_opinion
			}
		}
	}
	else = {
		scope:necromancer = {
			remove_hook = {
				target = scope:wraith
				type = necromancy_bind_hook
			}
		}
		scope:wraith = {
			remove_opinion = {
				modifier = POD_binding_opinion
				target = scope:necromancer
			}
		}
	}
	scope:necromancer = {
		remove_relation_bound_wraith = scope:wraith
	}
