# POD_create_node_map
# 	SAVE_AS = saved scope name
POD_create_node_map = {
	character:POD_wraith_01 = {
		create_story = {
			type = POD_story_node_map
			save_scope_as = $SAVE_AS$
		}
	}
}

# POD_create_node
#	MAP = a node map story
#	COORD_X = x coordinate
#	COORD_Y = y coordinate
#	SAVE_AS = saved scope name
POD_create_node = {
	character:POD_wraith_01 = {
		create_story = {
			type = POD_story_node
			save_scope_as = $SAVE_AS$
		}
	}
	scope:$SAVE_AS$ = {
		set_variable = {
			name = coord_x
			value = $COORD_X$
		}
		set_variable = {
			name = coord_y
			value = $COORD_Y$
		}
	}
	$MAP$ = {
		add_to_variable_list = {
			name = nodes
			target = scope:$SAVE_AS$
		}
	}
}

# POD_create_line
#	MAP = a node map story
#	START = a node story
#	END = a node story
#	SAVE_AS = saved scope name
POD_create_line = {
	character:POD_wraith_01 = {
		create_story = {
			type = POD_story_line
			save_scope_as = $SAVE_AS$

			set_variable = {
				name = start
				value = $START$
			}

			set_variable = {
				name = end
				value = $END$
			}
		}
	}

	$START$ = {
		add_to_variable_list = {
			name = lines
			value = scope:$SAVE_AS$
		}
	}

	$END$ = {
		add_to_variable_list = {
			name = lines
			value = scope:$SAVE_AS$
		}
	}

	$MAP$ = {
		add_to_variable_list = {
			name = lines
			target = scope:$SAVE_AS$
		}
	}
}
