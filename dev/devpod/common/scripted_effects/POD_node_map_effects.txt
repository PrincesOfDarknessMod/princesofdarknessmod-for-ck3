# POD_coord_squared_distance
#	COORD1_X = x coordinate of first point
#	COORD1_Y = y coordinate of first point
#	COORD2_X = x coordinate of second point
#	COORD2_Y = y coordinate of second point
#	SAVE_AS = saved scope name
POD_coord_squared_distance = {
	save_scope_value_as = {
		name = $SAVE_AS$
		value = {
			add = {
				value = $COORD1_X$
				subtract = $COORD2_X$
				multiply = {
					value = $COORD1_X$
					subtract = $COORD2_X$
				}
			}
			add = {
				value = $COORD1_Y$
				subtract = $COORD2_Y$
				multiply = {
					value = $COORD1_Y$
					subtract = $COORD2_Y$
				}
			}
		}
	}
}

# POD_node_squared_distance
#	NODE1 = first node
#	NODE2 = second node
#	SAVE_AS = saved scope name
POD_node_squared_distance = {
	POD_coord_squared_distance = {
		COORD1_X = $NODE1$.var:coord_x
		COORD1_Y = $NODE1$.var:coord_y
		COORD2_X = $NODE2$.var:coord_x
		COORD2_Y = $NODE2$.var:coord_y
		SAVE_AS = $SAVE_AS$
	}
}

# to cleanly delete nodes from a node_map
# meaning this also deletes dangling lines and removes them from variable lists
POD_delete_nodes_that_fulfill_condition = {
	if = {
		limit = { has_local_variable_list = nodes_to_delete }
		clear_local_variable_list = nodes_to_delete
	}
	$MAP$ = {
		every_in_list = {
			variable = nodes
			limit = {
				$CONDITION$
			}
			every_in_list = {
				variable = lines
				if = {
					limit = { NOT = { var:start_node = prev } }
					var:start_node = {
						remove_list_variable = {
							name   = lines
							target = prev
						}
					}
				}
				if = {
					limit = { NOT = { var:end_node = prev } }
					var:end_node = {
						remove_list_variable = {
							name   = lines
							target = prev
						}
					}
				}
				end_story = yes
			}
			add_to_local_variable_list = {
				name   = nodes_to_delete
				target = this
			}
		}
	}
	every_in_local_list = {
		variable = nodes_to_delete
		$MAP$ = {
			remove_list_variable = {
				name   = nodes
				target = prev
			}
		}
		end_story = yes
	}
}

# Quadrant-based means:
#	- The node_map is divided into equally-sized grid cells
#	- Each cell contains one node
#	- Nodes have a "row" and "column" variable denoting the position of their grid cell
#	- The node_map has "quadrant_size_x" and "quadrant_size_y" variables


# TODO: More efficient to get by index but cut down on number of stories by doing it this way
# Need to see if performance is bad here
POD_get_node_by_row_and_column = {
	if = {
		limit = { exists = scope:$SAVE_AS$ }
		clear_saved_scope = $SAVE_AS$
	}
	$MAP$ = {
		every_in_list = {
			variable = nodes
			limit = {
				NOT = {
					exists = scope:$SAVE_AS$
				}
				var:row ?= $ROW$
				var:column ?= $COLUMN$
			}
			save_scope_as = $SAVE_AS$
		}
	}
}


# POD_create_random_node_in_quadrant_with_padding
#	PADDING = internal padding of the node (to prevent it from being right on the edges of the quadrant)
#	ROW = the quadrant row
#	COLUMN = the quadrant column
#	MAP = the node map
POD_create_random_node_in_quadrant_with_padding = {
	save_scope_value_as = {
		name = coord_x
		value = {
			value = {
				0
				1
			}
			multiply = {
				value = $MAP$.var:quadrant_size_x
				subtract = {
					value = $PADDING$
					multiply = 2
				}
			}
			add = $PADDING$
			# offset by current column
			add = {
				value = $MAP$.var:quadrant_size_x
				multiply = $COLUMN$
			}
		}
	}
	save_scope_value_as = {
		name = coord_y
		value = {
			value = { 0 1 }
			multiply = {
				value = $MAP$.var:quadrant_size_y
				subtract = {
					value = $PADDING$
					multiply = 2
				}
			}
			add = $PADDING$
			# offset by current row
			add = {
				value = $MAP$.var:quadrant_size_y
				multiply = $ROW$
			}
		}
	}
	POD_create_node = {
		MAP = $MAP$
		COORD_X = scope:coord_x
		COORD_Y = scope:coord_y
		SAVE_AS = created_node
	}
	scope:created_node = {
		set_variable = {
			name = row
			value = $ROW$
		}
		set_variable = {
			name = column
			value = $COLUMN$
		}
	}
}

# args: MAP, MOVE_X, MOVE_Y
POD_move_all_nodes_to_next_quadrants = {
	$MAP$ = {
		every_in_list = {
			variable = nodes
			
			change_variable = {
				name = column
				add = $MOVE_X$
			}
			change_variable = {
				name = row
				add = $MOVE_Y$
			}
			change_variable = {
				name = coord_x
				add = {
					value = $MOVE_X$
					multiply = prev.var:quadrant_size_x
				}
			}
			change_variable = {
				name = coord_y
				add = {
					value = $MOVE_Y$
					multiply = prev.var:quadrant_size_y
				}
			}
		}
	}
}

# completes a delaunay triangulation on a node_map that doesn't have any lines yet, and saves the resulting mesh as lines
# need to pass $MAP$ (a node_map)
# also $MAP_WIDTH$ and $MAP_HEIGHT$ (move these to variables in the node_map?)
#	https://www.gorillasun.de/blog/bowyer-watson-algorithm-for-delaunay-triangulation/
#
# NOTE: out of the box, this algorithm makes adjacency errors
# on wide aspect ratios, some nodes will be "skipped over" vertically
# on tall aspect ratios, the skips happen horizontally
# on square maps, everything seems to work fine
# my mathbrain is too smooth to figure out the real reason for this so i just added a height multiplier
# so that the map gets treated as square even when it isn't
#
# ck3 doesn't have a sqrt function (and even if it did, it'd be slow) so this uses squared distances
# maybe that's the reason
#
POD_delaunay_triangulation = {
	hidden_effect_new_object = {
		if = {
			limit = { has_local_variable_list = triangles }
			clear_local_variable_list = triangles
		}
		save_scope_value_as = {
			name  = height_mult
			value = {
				value = $MAP_WIDTH$
				divide = $MAP_HEIGHT$
			}
		}
		# create a triangle whose circumcircle encompasses all nodes
		POD_create_super_triangle = {
			MAP_WIDTH   = $MAP_WIDTH$
			MAP_HEIGHT  = $MAP_HEIGHT$
			HEIGHT_MULT = scope:height_mult
			SAVE_AS     = super_triangle
		}
		scope:super_triangle = {
			set_variable = super_triangle
		}
		add_to_local_variable_list = {
			name   = triangles
			target = scope:super_triangle
		}
		$MAP$ = {
			every_in_list = {
				variable = nodes
				# we create the dummy node after triangulation is finished so checking this isn't necessary
				#limit = { NOT = { has_variable = dummy_node } }
				save_scope_as = this_node
				# the big iteration function, creating a list of triangles
				POD_delaunay_add_node = {
					NODE = scope:this_node
					HEIGHT_MULT = scope:height_mult
					TRIANGLES = triangles
				}
			}
		}
		# # remove triangles that share edges with super triangle
		every_in_local_list = {
			variable = triangles
			limit = {
				NOR = {
					var:node1 = scope:super_triangle.var:node1
					var:node1 = scope:super_triangle.var:node2
					var:node1 = scope:super_triangle.var:node3
					var:node2 = scope:super_triangle.var:node1
					var:node2 = scope:super_triangle.var:node2
					var:node2 = scope:super_triangle.var:node3
					var:node3 = scope:super_triangle.var:node1
					var:node3 = scope:super_triangle.var:node2
					var:node3 = scope:super_triangle.var:node3
				}
			}
			add_to_local_variable_list = {
				name   = output_triangles
				target = this
			}
		}
		# convert remaining triangles into lines
		every_in_local_list = {
			variable = output_triangles
			
			save_scope_as = this_triangle
			
			POD_add_triangle_line_to_node_map_if_unique = {
				TRIANGLE       = scope:this_triangle
				START_NODE_VAR = var:node1
				END_NODE_VAR   = var:node2
				MAP            = $MAP$
			}
			POD_add_triangle_line_to_node_map_if_unique = {
				TRIANGLE       = scope:this_triangle
				START_NODE_VAR = var:node2
				END_NODE_VAR   = var:node3
				MAP            = $MAP$
			}
			POD_add_triangle_line_to_node_map_if_unique = {
				TRIANGLE       = scope:this_triangle
				START_NODE_VAR = var:node3
				END_NODE_VAR   = var:node1
				MAP            = $MAP$
			}
		}
		scope:super_triangle = {
			var:node1 = { end_story = yes }
			var:node2 = { end_story = yes }
			var:node3 = { end_story = yes }
			end_story = yes
		}
		every_in_local_list = {
			variable = triangles
			end_story = yes
		}
	}
}


# POD_delaunay_add_node
#	NODE = a node
#	TRIANGLES = a local variable list containing POD_story_triangle story_cycles
#	HEIGHT_MULT = aspect ratio (because the algorithm makes mistakes if the map isn't square and idk why)
POD_delaunay_add_node = {
	if = {
		limit = { has_local_variable_list = new_triangles }
		clear_local_variable_list = new_triangles
	}
	if = {
		limit = { has_local_variable_list = lines_to_check }
		clear_local_variable_list = lines_to_check
	}
	save_scope_value_as = {
		name = node_y_scaled
		value = {
			value = $NODE$.var:coord_y
			multiply = $HEIGHT_MULT$
		}
	}
	every_in_local_list = {
		variable = $TRIANGLES$
		
		# get the node's squared distance from the circumcenter
		POD_coord_squared_distance = {
			COORD1_X = var:circumcenter_x
			COORD1_Y = var:circumcenter_y
			COORD2_X = $NODE$.var:coord_x
			COORD2_Y = scope:node_y_scaled
			SAVE_AS = node_distance
		}
		
		if = {
			# check whether the node we're checking is in this triangle's circumcircle
			# if it is, the triangle is invalid and gets discarded
			limit = { scope:node_distance <= var:squared_radius }
			
			save_scope_as = this_triangle
			
			POD_create_line_without_map = {
				START   = scope:this_triangle.var:node1
				END     = scope:this_triangle.var:node2
				SAVE_AS = line1
			}
			POD_create_line_without_map = {
				START   = scope:this_triangle.var:node2
				END     = scope:this_triangle.var:node3
				SAVE_AS = line2
			}
			POD_create_line_without_map = {
				START   = scope:this_triangle.var:node3
				END     = scope:this_triangle.var:node1
				SAVE_AS = line3
			}
			
			hidden_effect_new_object = {
				add_to_local_variable_list = {
					name   = lines_to_check
					target = scope:line1
				}
				add_to_local_variable_list = {
					name   = lines_to_check
					target = scope:line2
				}
				add_to_local_variable_list = {
					name   = lines_to_check
					target = scope:line3
				}
			}
			
			if = {
				limit = { NOT = { has_variable = super_triangle } }
				end_story = yes
			}
		}
		# if the node is outside the circumcircle, keep the triangle
		else = {
			add_to_local_variable_list = {
				name   = new_triangles
				target = this
			}
		}
	}
	# copy variable list over so the invalid triangles aren't kept
	# this step could be skipped if it's possible to delete entries from a list you're iterating over
	# but it probably isn't
	clear_local_variable_list = $TRIANGLES$
	every_in_local_list = {
		variable = new_triangles
		add_to_local_variable_list = {
			name   = $TRIANGLES$
			target = this
		}
	}
	
	POD_get_unique_lines_from_local_list = {
		INPUT_LIST  = lines_to_check
		OUTPUT_LIST = unique_lines
	}
	
	every_in_local_list = {
		variable = unique_lines
		
		save_scope_as = this_line
		
		POD_create_triangle_from_existing_nodes = {
			NODE1 = $NODE$
			NODE2 = scope:this_line.var:start_node
			NODE3 = scope:this_line.var:end_node
			SAVE_AS = triangle
		}
		
		hidden_effect_new_object = {
			POD_save_height_adjusted_circumcenter_on_triangle = {
				TRIANGLE    = scope:triangle
				HEIGHT_MULT = $HEIGHT_MULT$
			}
			add_to_local_variable_list = {
				name   = $TRIANGLES$
				target = scope:triangle
			}
		}
	}
	every_in_local_list = {
		variable = lines_to_check
		end_story = yes
	}
	clear_local_variable_list = lines_to_check
	clear_local_variable_list = unique_lines
}


POD_get_unique_lines_from_local_list = {
	if = {
		limit = { has_local_variable_list = $OUTPUT_LIST$ }
		clear_local_variable_list = $OUTPUT_LIST$
	}
	clear_saved_scope = is_not_unique
	every_in_local_list = {
		variable = $INPUT_LIST$
		
		every_in_local_list = {
			variable = $INPUT_LIST$
			limit = {
				NOT = { exists = scope:is_not_unique } # to early out of the loop
				NOT = { this = prev }
				OR = {
					AND = {
						var:start_node = prev.var:start_node
						var:end_node   = prev.var:end_node
					}
					AND = {
						var:start_node = prev.var:end_node
						var:end_node   = prev.var:start_node
					}
				}
			}
			save_scope_as = is_not_unique
		}
		if = {
			limit = { NOT = { exists = scope:is_not_unique } }
			add_to_local_variable_list = {
				name   = $OUTPUT_LIST$
				target = this
			}
		}
		clear_saved_scope = is_not_unique
	}
}

# like the previous effect but adding a height multiplier so we can pretend the map is square
POD_get_height_adjusted_circumcenter_from_triangle = {
	$TRIANGLE$ = {
		save_scope_value_as = {
			name  = coord1_y_scaled

			}
		}
		save_scope_value_as = {
			name  = coord2_y_scaled
			value = {
				value = var:node2.var:coord_y
				multiply = $HEIGHT_MULT$
			}
		}
		save_scope_value_as = {
			name  = coord3_y_scaled
			value = {
				value = var:node3.var:coord_y
				multiply = $HEIGHT_MULT$
			}
		}
		POD_get_circumcenter_from_3_coords = {

			COORD1_Y = scope:coord1_y_scaled

			COORD2_Y = scope:coord2_y_scaled
			COORD3_X = var:node3.var:coord_x
			COORD3_Y = scope:coord3_y_scaled

			SAVE_AS_Y = $SAVE_AS_Y$
		}
	}

# pass coordinates: COORD1_X, COORD1_Y, COORD2_X, COORD2_Y, COORD3_X, COORD3_Y
# SAVE_AS_X and SAVE_AS_Y to get the return values
#	https://stackoverflow.com/questions/56224824/how-do-i-find-the-circumcenter-of-the-triangle-using-python-without-external-lib
#	https://en.wikipedia.org/wiki/Circumcircle#Circumcenter_coordinates
POD_get_circumcenter_from_3_coords = {
	# d = 2 * (ax * (by - cy) + bx * (cy - ay) + cx * (ay - by))
	save_scope_value_as = {
		name = d
		value = {
			value = 2
			multiply = {
				value = {
					value = $COORD1_X$
					multiply = {
						value = $COORD2_Y$
						subtract = $COORD3_Y$
					}
				}
				add = {
					value = $COORD2_X$
					multiply = {
						value = $COORD3_Y$
						subtract = $COORD1_Y$
					}
				}
				add = {
					value = $COORD3_X$
					multiply = {
						value = $COORD1_Y$
						subtract = $COORD2_Y$
					}
				}
			}
		}
	}
	# (ax * ax + ay * ay)
	save_scope_value_as = {
		name = a
		value = {
			value = {
				value = $COORD1_X$
				multiply = $COORD1_X$
			}
			add = {
				value = $COORD1_Y$
				multiply = $COORD1_Y$
			}
		}
	}
	# (bx * bx + by * by)
	save_scope_value_as = {
		name = b
		value = {
			value = {
				value = $COORD2_X$
				multiply = $COORD2_X$
			}
			add = {
				value = $COORD2_Y$
				multiply = $COORD2_Y$
			}
		}
	}
	# (cx * cx + cy * cy)
	save_scope_value_as = {
		name = c
		value = {
			value = {
				value = $COORD3_X$
				multiply = $COORD3_X$
			}
			add = {
				value = $COORD3_Y$
				multiply = $COORD3_Y$
			}
		}
	}
	# ux = ((ax * ax + ay * ay) * (by - cy) + (bx * bx + by * by) * (cy - ay) + (cx * cx + cy * cy) * (ay - by)) / d
	save_scope_value_as = {
		name = $SAVE_AS_X$
		value = {
			value = {
				value = {
					value = scope:a # (ax * ax + ay * ay)
					multiply = {
						value = $COORD2_Y$
						subtract = $COORD3_Y$
					}
				}
				add = {
					value = scope:b # (bx * bx + by * by)
					multiply = {
						value = $COORD3_Y$
						subtract = $COORD1_Y$
					}
				}
				add = {
					value = scope:c # (cx * cx + cy * cy)
					multiply = {
						value = $COORD1_Y$
						subtract = $COORD2_Y$
					}
				}
			}
			divide = scope:d
		}
	}
	# uy = ((ax * ax + ay * ay) * (cx - bx) + (bx * bx + by * by) * (ax - cx) + (cx * cx + cy * cy) * (bx - ax)) / d
	save_scope_value_as = {
		name = $SAVE_AS_Y$
		value = {
			value = {
				value = {
					value = scope:a # (ax * ax + ay * ay)
					multiply = {
						value = $COORD3_X$
						subtract = $COORD2_X$
					}
				}
				add = {
					value = scope:b # (bx * bx + by * by)
					multiply = {
						value = $COORD1_X$
						subtract = $COORD3_X$
					}
				}
				add = {
					value = scope:c # (cx * cx + cy * cy)
					multiply = {
						value = $COORD2_X$
						subtract = $COORD1_X$
					}
				}
			}
			divide = scope:d
		}
	}
}

# to convert triangles into lines after triangulation is complete
POD_add_triangle_line_to_node_map_if_unique = {
	if = {
		limit = {
			$MAP$ = {
				NOT = {
					any_in_list = {
						variable = lines
						$TRIANGLE$ = {
							OR = {
								AND = {
									prev.var:start_node = $START_NODE_VAR$
									prev.var:end_node   = $END_NODE_VAR$
								}
								AND = {
									prev.var:start_node = $END_NODE_VAR$
									prev.var:end_node   = $START_NODE_VAR$
								}
							}
						}
					}
				}
			}
		}
		$TRIANGLE$ = {
			$START_NODE_VAR$ = { save_scope_as = start_node }
			$END_NODE_VAR$   = { save_scope_as = end_node }
			POD_create_line = {
				MAP = $MAP$
				START = scope:start_node
				END = scope:end_node
				SAVE_AS = line
			}
		}
	}
}
