POD_remove_incompatible_building_effect = { # Scope province
	if = {
		limit = {
			OR = {
				trigger_if = {
					limit = { has_holding_type = tribal_holding }
					root = { NOT = { government_has_flag = government_is_tribal } }
				}
				trigger_else = { root = { government_has_flag = government_is_tribal } }
				POD_has_wrong_type_of_building_trigger = yes 
			}
		}
		set_local_variable = {
			name = pod_building_level
			value = combined_building_level
			days = 1
		}
		POD_remove_common_building_effect = yes
		while = {
			limit = { local_var:pod_building_level > 1 }
			generate_building = yes
			change_local_variable = {
				name = pod_building_level
				subtract = 1
			}
		}
		remove_local_variable = pod_building_level
		if = {
			limit = { is_county_capital = yes }
			POD_remove_incompatible_ducal_building_effect = yes 
		}
	}
}

POD_remove_ducal_building_effect = {
	if = {
		limit = { has_building_or_higher = $BUILDING$ }
		if = {
			limit = { NOT = { has_building = $BUILDING$ } } # Replace higher level buidling
			if = {
				limit = { has_free_building_slot = no } 
				add_province_modifier = extra_building_slot
				add_building = $BUILDING$
				remove_province_modifier = extra_building_slot
			}
			else = { add_building = $BUILDING$ }
		}
		remove_building = $BUILDING$
	}
}

POD_replace_tribal_holding_effect = {
	if = {
		limit = { is_grand_city_location_trigger = yes }
		POD_reset_grand_city_effect = yes
	}
	else_if = {
		limit = { NOT = { county = { any_county_province = { has_holding_type = castle_holding } } } }
		set_holding_type = castle_holding 
	}
	else_if = {
		limit = { NOT = { county = { any_county_province = { has_holding_type = city_holding } } } }
		set_holding_type = city_holding 
	}
	else_if = {
		limit = { NOT = { county = { any_county_province = { has_holding_type = church_holding } } } }
		set_holding_type = church_holding 
	}
	else = {
		random_list = {
			10 = { set_holding_type = castle_holding }
			1 = { set_holding_type = city_holding }
			1 = { set_holding_type = church_holding }
		}
	}
}
