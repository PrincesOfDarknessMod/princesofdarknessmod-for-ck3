POD_remove_common_building_effect = {
	if = {
		limit = { has_holding_type = castle_holding }
		if = {
			limit = {
				province_owner = { has_government = litany_government }
			}
			set_holding_type = tribal_holding
		}
		else_if = {
			limit = {
				has_building = castle_04
			}
			set_holding_type = tribal_holding
			set_holding_type = castle_holding
			add_building = castle_04
		}
		else_if = {
			limit = {
				has_building = castle_03
			}
			set_holding_type = tribal_holding
			set_holding_type = castle_holding
			add_building = castle_03
		}
		else_if = {
			limit = {
				has_building = castle_02
			}
			set_holding_type = tribal_holding
			set_holding_type = castle_holding
			add_building = castle_02
		}
		else = {
			set_holding_type = tribal_holding
			set_holding_type = castle_holding
		}
	}
	else_if = {
		limit = { has_holding_type = city_holding }
		if = {
			limit = {
				province_owner = { has_government = litany_government }
			}
			set_holding_type = tribal_holding
		}
		else_if = {
			limit = {
				has_building = city_04
			}
			set_holding_type = tribal_holding
			set_holding_type = city_holding
			add_building = city_04
		}
		else_if = {
			limit = {
				has_building = city_03
			}
			set_holding_type = tribal_holding
			set_holding_type = city_holding
			add_building = city_03
		}
		else_if = {
			limit = {
				has_building = city_02
			}
			set_holding_type = tribal_holding
			set_holding_type = city_holding
			add_building = city_02
		}
		else = {
			set_holding_type = tribal_holding
			set_holding_type = city_holding
		}
	}
	else_if = {
		limit = { has_holding_type = church_holding }
		if = {
			limit = {
				province_owner = { has_government = litany_government }
			}
			set_holding_type = tribal_holding
		}
		else_if = {
			limit = {
				has_building = temple_04
			}
			set_holding_type = tribal_holding
			set_holding_type = church_holding
			add_building = temple_04
		}
		else_if = {
			limit = {
				has_building = temple_03
			}
			set_holding_type = tribal_holding
			set_holding_type = church_holding
			add_building = temple_03
		}
		else_if = {
			limit = {
				has_building = temple_02
			}
			set_holding_type = tribal_holding
			set_holding_type = church_holding
			add_building = temple_02
		}
		else = {
			set_holding_type = tribal_holding
			set_holding_type = church_holding
		}
	}
	else = {
		# limit = { has_holding_type = tribal_holding }
		if = {
			limit = {
				province_owner = { 
					NOT = { has_government = litany_government } 
					NOT = { has_government = tribal_government }
				}
			}
			random_list = {
				10 = { set_holding_type = castle_holding }
				1 = { set_holding_type = city_holding }
				1 = { set_holding_type = church_holding }
			}
		}
		else_if = {
			limit = {
				has_building = tribe_02
			}
			set_holding_type = castle_holding
			set_holding_type = tribal_holding
			add_building = tribe_02
		}
		else = {
			set_holding_type = castle_holding
			set_holding_type = tribal_holding
		}
	}
}

		limit = { county = title:c_upland }
