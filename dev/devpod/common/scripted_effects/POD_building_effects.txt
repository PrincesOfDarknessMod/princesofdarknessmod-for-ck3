POD_replace_tribal_holding_effect = {
	if = {
		limit = { is_grand_city_location_trigger = yes }
		POD_reset_grand_city_effect = yes
	}
	else_if = {
		limit = { NOT = { county = { any_county_province = { has_holding_type = castle_holding } } } }
		set_holding_type = castle_holding
	}
	else_if = {
		limit = { NOT = { county = { any_county_province = { has_holding_type = city_holding } } } }
		set_holding_type = city_holding
	}
	else_if = {
		limit = { NOT = { county = { any_county_province = { has_holding_type = church_holding } } } }
		set_holding_type = church_holding
	}
	else = {
		random_list = {
			10 = { set_holding_type = castle_holding }
			1 = { set_holding_type = city_holding }
			1 = { set_holding_type = church_holding }
		}
	}
}

POD_replace_special_building_effect = {
	if = {
		limit = {
			this = province:1862
		}
		if = {
			limit = {
				has_building_or_higher = barcelona_library_01
				root = { has_POD_fae_blood_trigger = yes }
			}
			add_building = bullring_01
		}
		if = {
			limit = {
				has_building_or_higher = bullring_01
				root = { has_POD_fae_blood_trigger = no }
			}
			add_building = barcelona_library_01
		}
	}
	if = {
		limit = {
			this = province:1569
		}
		if = {
			limit = {
				has_building_or_higher = castle_avalon_01
				root = { has_POD_fae_blood_trigger = yes }
			}
			add_building = chaw_gully_mine_01
		}
		if = {
			limit = {
				has_building_or_higher = chaw_gully_mine_01
				root = { has_trait = vampire }
			}
			add_building = castle_avalon_01
		}
	}
}

	if = {
		limit = { has_building = $BUILDING$ }
