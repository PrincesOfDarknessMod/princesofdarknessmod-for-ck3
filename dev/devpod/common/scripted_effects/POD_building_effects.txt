POD_remove_incompatible_building_effect = { # Scope province
	if = {
		limit = {
			OR = {
				trigger_if = {
					limit = { has_holding_type = tribal_holding }
					root = { NOT = { government_has_flag = government_is_tribal } }
				}
				trigger_else = { root = { government_has_flag = government_is_tribal } }
				POD_has_wrong_type_of_building_trigger = yes 
			}
		}
		set_local_variable = {
			name = pod_building_level
			value = combined_building_level
			days = 1
		}
		if = {
			limit = { is_county_capital = yes }
			POD_remove_incompatible_ducal_building_effect = yes
		}
		POD_remove_common_building_effect = yes
		if = {
			limit = { POD_should_generate_buildings_trigger = yes }
			while = {
				limit = { local_var:pod_building_level > 1 }
				generate_building = yes
				change_local_variable = {
					name = pod_building_level
					subtract = 2
				}
			}
		}
		remove_local_variable = pod_building_level
	}
}

POD_remove_incompatible_ducal_building_effect = {
	# Princedom
	if = {
		limit = {
			root = { NOT = { has_trait = vampire } }
			county.previous_holder ?= { has_trait = vampire }
		}
		POD_remove_ducal_building_effect = { BUILDING = prince_city MAX = 9 }
	}
	# Werewolves
	if = {
		limit = {
			root = { 
				NOT = { has_trait = shapeshifter } 
				NOT = { has_trait = kinfolk } 
				NOT = { has_trait = spirit } 
			}
			county.previous_holder ?= { 
				OR = { 
					has_trait = shapeshifter 
					has_trait = kinfolk
					has_trait = spirit
				}
			}
		}
		POD_remove_ducal_building_effect = { BUILDING = caern_rage MAX = 5 }
		POD_remove_ducal_building_effect = { BUILDING = caern_will MAX = 5 }
		POD_remove_ducal_building_effect = { BUILDING = caern_enigmas MAX = 5 }
		POD_remove_ducal_building_effect = { BUILDING = caern_leadership MAX = 5 }
		POD_remove_ducal_building_effect = { BUILDING = caern_wisdom MAX = 5 }
	}	
	# Ghoul
	if = {
		limit = {
			root = { 
				OR = { 
					POD_is_any_kind_of_ghoul_trigger = no 
					is_forsaken_or_gantenbein_trigger = yes
				}
			}
			county.previous_holder ?= { 
				POD_is_any_kind_of_ghoul_trigger = yes
				is_forsaken_or_gantenbein_trigger = no
			}
		}
		POD_remove_ducal_building_effect = { BUILDING = family_rights MAX = 4 }
	}
	# Gantenbein
	if = {
		limit = {
			root = { POD_is_gantenbein_trigger = no }
			county.previous_holder ?= { POD_is_gantenbein_trigger = yes }
		}
		POD_remove_ducal_building_effect = { BUILDING = heretic_hall MAX = 4 }
	}
	# Inquisitors
	if = {
		limit = {
			root = { building_requirement_inquisitor = no }
			county.previous_holder ?= { building_requirement_inquisitor = yes}
		}
		POD_remove_ducal_building_effect = { BUILDING = inquisition_armory MAX = 4 }
		POD_remove_ducal_building_effect = { BUILDING = inquisition_keep MAX = 4 }
		POD_remove_ducal_building_effect = { BUILDING = inquisition_church MAX = 4 }
		POD_remove_ducal_building_effect = { BUILDING = inquisition_office MAX = 4 }
		POD_remove_ducal_building_effect = { BUILDING = inquisition_palace MAX = 4 }
	}
	# Hunters
	if = {
		limit = {
			root = { building_requirement_hunter = no }
			county.previous_holder ?= { building_requirement_hunter = yes }
		}
		POD_remove_ducal_building_effect = { BUILDING = hunter_ducal MAX = 5 }
	}	
	# KJ
	if = {
		limit = {
			root = { POD_middle_kingdom_splat_trigger = no }
			county.previous_holder ?= { POD_middle_kingdom_splat_trigger = yes }
		}
		POD_remove_ducal_building_effect = { BUILDING = dragon_nest MAX = 3 }
	}
	# Mummy
	if = {
		limit = {
			root = { 
				NOT = { has_trait = mummy }
				is_a_neteru = no
			}
			county.previous_holder ?= { 
				OR = { 
					has_trait = mummy
					is_a_neteru = yes
				} 
			}
		}
		POD_remove_ducal_building_effect = { BUILDING = web_of_faith MAX = 8 }
		POD_remove_ducal_building_effect = { BUILDING = occult_affiliation MAX = 8 }
		POD_remove_ducal_building_effect = { BUILDING = maats_haven MAX = 8 }
		POD_remove_ducal_building_effect = { BUILDING = cult_of_amkhat MAX = 8 }
	}
	# Demon
	if = {
		limit = {
			root = {
				building_requirement_demon = no
			}
			county.previous_holder ?= {
				building_requirement_demon = yes
			}
		}
		POD_remove_ducal_building_effect = { BUILDING = bastion MAX = 6 }
	}
	# Wraith
	if = {
		limit = {
			root = { 
				NOT = { has_trait = wraith } 
			}
			county.previous_holder ?= { 
				has_trait = wraith
			}
		}
		POD_remove_ducal_building_effect = { BUILDING = citadel MAX = 6 }
	}
	# Fae
	if = {
		limit = {
			root = { 
				NOR = {
					has_trait = fae 
					has_trait = changeling
					has_trait = kinain
				}
			}
			county.previous_holder ?= { 
				OR = {
					has_trait = fae 
					has_trait = changeling
					has_trait = kinain
				}
			}
		}
		POD_remove_ducal_building_effect = { BUILDING = land_trod MAX = 3 }
		POD_remove_ducal_building_effect = { BUILDING = water_trod MAX = 3 }
		POD_remove_ducal_building_effect = { BUILDING = sun_trod MAX = 3 }
		POD_remove_ducal_building_effect = { BUILDING = moon_trod MAX = 3 }
		POD_remove_ducal_building_effect = { BUILDING = chimerical_trod MAX = 3 }
		POD_remove_ducal_building_effect = { BUILDING = nightmare_trod MAX = 3 }
		POD_remove_ducal_building_effect = { BUILDING = wild_portal MAX = 3 }
	}
}

POD_add_next_building_tier_effect = {
	if = {
		limit = { has_free_building_slot = no } 
		add_province_modifier = extra_building_slot
		if = {
			limit = { has_free_building_slot = yes } 
			POD_add_next_building_tier_max_$MAX$_effect = { BUILDING = $BUILDING$ }
		}
		remove_province_modifier = extra_building_slot
	}
	else = {
		POD_add_next_building_tier_max_$MAX$_effect = { BUILDING = $BUILDING$ }
	}
}

	}
