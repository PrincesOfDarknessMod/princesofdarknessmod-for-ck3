POD_remove_incompatible_ducal_building_effect = {
	# Princedom
	if = {
		limit = {
			province_owner = { NOT = { has_trait = vampire } }
			exists = scope:previous_holder
			scope:previous_holder = { has_trait = vampire }
		}
		POD_remove_building_effect = { BUILDING = prince_city_00 }
	}
	# Werewolves
	if = {
		limit = {
			province_owner = { 
				NOT = { has_trait = werewolf } 
				NOT = { has_trait = kinfolk } 
			}
			exists = scope:previous_holder
			scope:previous_holder = { 
				OR = { 
					has_trait = werewolf 
					has_trait = kinfolk
				}
			}
		}
		POD_remove_building_effect = { BUILDING = caern_rage_01 }
		POD_remove_building_effect = { BUILDING = caern_will_01 }
		POD_remove_building_effect = { BUILDING = caern_enigmas_01 }
		POD_remove_building_effect = { BUILDING = caern_leadership_01 }
		POD_remove_building_effect = { BUILDING = caern_wisdom_01 }
	}	
	# Ghoul
	if = {
		limit = {
			province_owner = { 
				OR = { 
					POD_is_any_kind_of_ghoul_trigger = no 
					is_forsaken_or_gantenbein_trigger = yes
				}
			}
			exists = scope:previous_holder
			scope:previous_holder = { 
				POD_is_any_kind_of_ghoul_trigger = yes
				is_forsaken_or_gantenbein_trigger = no
			}
		}
		POD_remove_building_effect = { BUILDING = family_rights_01 }
	}
	# Gantenbein
	if = {
		limit = {
			province_owner = { POD_is_gantenbein_trigger = no }
			exists = scope:previous_holder
			scope:previous_holder = { POD_is_gantenbein_trigger = yes }
		}
		POD_remove_building_effect = { BUILDING = heretic_hall_01 }
	}
	# Inquisitors
	if = {
		limit = {
			province_owner = { 
				NOT = { has_trait = supehunter }
				is_mortal_child_of_supehunter_trigger = no
			}
			exists = scope:previous_holder
			scope:previous_holder = { 
				OR = { 
					has_trait = supehunter 
					is_mortal_child_of_supehunter_trigger = yes
				} 
			}
		}
		POD_remove_building_effect = { BUILDING = inquisition_armory_01 }
		POD_remove_building_effect = { BUILDING = inquisition_keep_01 }
		POD_remove_building_effect = { BUILDING = inquisition_office_01 }
		POD_remove_building_effect = { BUILDING = inquisition_palace_01 }
	}
	# KJ
	if = {
		limit = {
			province_owner = { 
				NOR = { 
					has_trait = kueijin
					has_trait = dhampyr
					is_east_mummy_trigger = yes
				} 
			}
			exists = scope:previous_holder
			scope:previous_holder = { 
				OR = { 
					has_trait = kueijin
					has_trait = dhampyr
					is_east_mummy_trigger = yes
				} 
			}
		}
		POD_remove_building_effect = { BUILDING = dragon_nest_01 }
	}
	# Mummy
	if = {
		limit = {
			province_owner = { 
				NOT = { has_trait = mummy }
				is_a_neteru = no
			}
			exists = scope:previous_holder
			scope:previous_holder = { 
				OR = { 
					has_trait = mummy
					is_a_neteru = yes
				} 
			}
		}
		POD_remove_building_effect = { BUILDING = web_of_faith_01 }
		POD_remove_building_effect = { BUILDING = occult_affiliation_01 }
		POD_remove_building_effect = { BUILDING = maats_haven_01 }
	}
	# Demon
	if = {
		limit = {
			province_owner = { 
				NOT = { has_trait = demon }
				NOT = { has_trait = laham }
				NOT = { has_trait = nephilim }
				is_a_thrall = no
			}
			exists = scope:previous_holder
			scope:previous_holder = { 
				OR = { 
					has_trait = demon
					has_trait = laham
					has_trait = nephilim
					is_a_thrall = yes
				} 
			}
		}
		POD_remove_building_effect = { BUILDING = bastion_01 }
	}
	# Wraith
	if = {
		limit = {
			province_owner = { 
				NOT = { has_trait = wraith } 
			}
			exists = scope:previous_holder
			scope:previous_holder = { 
				has_trait = wraith
			}
		}
		POD_remove_building_effect = { BUILDING = necropolis_01 }
	}
}

POD_remove_building_effect = {
	if = {
		limit = { has_building_or_higher = $BUILDING$ }
		if = {
			limit = { NOT = { has_building = $BUILDING$ } } # Replace higher level buidling
			if = {
				limit = { has_free_building_slot = no } 
				add_building_slot = 1
				add_building = $BUILDING$
				add_building_slot = -1
			}
			else = { add_building = $BUILDING$ }
		}
		remove_building = $BUILDING$
	}
}

			}

					1 = { set_holding_type = city_holding }

			add_building = tribe_02
		}
		else = {

			set_holding_type = tribal_holding
		}
	}
