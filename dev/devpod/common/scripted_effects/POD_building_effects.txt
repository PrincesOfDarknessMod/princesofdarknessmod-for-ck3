# Province scope
POD_set_holding_type = {
	if = {
		limit = { NOT = { has_holding_type = $TYPE$ } }
		set_holding_type = $TYPE$
		POD_initialize_regular_districts = yes
	}
}

# Province scope
POD_initialize_regular_districts = {
	if = {
		limit = {
			NOT = { province_has_district = POD_wod_district }
			NOT = { has_holding_type = grand_city_holding }
		}
		add_regular_district = POD_wod_district
	}
}

# Province scope
POD_remove_incompatible_buildings_in_province = {
	barony_controller = { save_scope_as = province_holder }
	
	if = {
		limit = { POD_is_grand_city_location = yes }
		POD_reset_grand_city_in_province = yes
	}
	else_if = {
		limit = { root = { government_has_flag = government_is_tribal } }
		POD_set_holding_type = { TYPE = tribal_holding }
	}
	else_if = {
		limit = { has_holding_type = tribal_holding }
		POD_replace_tribal_holding_in_province = yes
	}
	
	POD_remove_incompatible_duchy_building_in_province = yes
	
	every_built_regular_district = {
		include_inactive = yes
		limit = { district_type = district_type:POD_wod_district }
		every_district_building = {
			limit = {
				is_primary_building = no
				building_type = {
					POD_building_type_is_incompatible = {
						HOLDER = scope:province_holder
					}
				}
			}
			remove_this_building = yes
		}
		else_if = {
			limit = {
				barony = title:b_mogadishu
				mogadishu_has_wrong_holder_trigger = yes
			}
			POD_reset_mogadishu_effect = yes
		}
	}
}

POD_replace_tribal_holding_in_province = {
	if = {
		limit = { POD_is_grand_city_location = yes }
		POD_reset_grand_city_in_province = yes
	}
	else = {
		if = {
			limit = { NOT = { county = { any_county_province = { has_holding_type = castle_holding } } } }
			set_holding_type = castle_holding
		}
		else_if = {
			limit = { NOT = { county = { any_county_province = { has_holding_type = city_holding } } } }
			set_holding_type = city_holding
		}
		else_if = {
			limit = { NOT = { county = { any_county_province = { has_holding_type = church_holding } } } }
			set_holding_type = church_holding
		}
		else = {
			random_list = {
				10 = { set_holding_type = castle_holding }
				1 = { set_holding_type = city_holding }
				1 = { set_holding_type = church_holding }
			}
		}
		POD_initialize_regular_districts = yes
	}
}

# Province scope
# assumes that scope:province_holder is on the TopScope
POD_remove_duchy_building_and_replace_if_ai = {
	if = {
		limit = { has_building_or_higher = $BUILDING$ }
		save_scope_value_as = {
			name  = new_max_level
			value = POD_duchy_building_level
		}
		duchy_capital_district = {
			random_district_building = {
				limit = { is_building_or_higher = $BUILDING$ }
				remove_this_building = yes
			}
		}
		if = {
			limit = { scope:province_holder = { is_ai = yes } }
			duchy_capital_district = { save_scope_as = target_district }
			while = {
				count = scope:new_max_level
				build_or_upgrade_random_district_building = {
					district = scope:target_district
				}
			}
		}
	}
}
