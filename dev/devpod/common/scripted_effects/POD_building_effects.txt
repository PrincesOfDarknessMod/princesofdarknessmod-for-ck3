POD_remove_incompatible_ducal_building_effect = {
	# Princedom
	if = {
		limit = {
			province_owner = { NOT = { has_trait = vampire } }
			scope:previous_holder ?= { has_trait = vampire }
		}
		POD_remove_ducal_building_effect = { BUILDING = prince_city_00 }
	}
	# Werewolves
	if = {
		limit = {
			province_owner = { 
				NOT = { has_trait = shapeshifter } 
				NOT = { has_trait = kinfolk } 
			}
			scope:previous_holder ?= { 
				OR = { 
					has_trait = shapeshifter 
					has_trait = kinfolk
				}
			}
		}
		POD_remove_ducal_building_effect = { BUILDING = caern_rage_01 }
		POD_remove_ducal_building_effect = { BUILDING = caern_will_01 }
		POD_remove_ducal_building_effect = { BUILDING = caern_enigmas_01 }
		POD_remove_ducal_building_effect = { BUILDING = caern_leadership_01 }
		POD_remove_ducal_building_effect = { BUILDING = caern_wisdom_01 }
	}	
	# Ghoul
	if = {
		limit = {
			province_owner = { 
				OR = { 
					POD_is_any_kind_of_ghoul_trigger = no 
					is_forsaken_or_gantenbein_trigger = yes
				}
			}
			scope:previous_holder ?= { 
				POD_is_any_kind_of_ghoul_trigger = yes
				is_forsaken_or_gantenbein_trigger = no
			}
		}
		POD_remove_ducal_building_effect = { BUILDING = family_rights_01 }
	}
	# Gantenbein
	if = {
		limit = {
			province_owner = { POD_is_gantenbein_trigger = no }
			scope:previous_holder ?= { POD_is_gantenbein_trigger = yes }
		}
		POD_remove_ducal_building_effect = { BUILDING = heretic_hall_01 }
	}
	# Inquisitors
	if = {
		limit = {
			province_owner = { building_requirement_inquisitor = no }
			scope:previous_holder ?= { building_requirement_inquisitor = yes}
		}
		POD_remove_ducal_building_effect = { BUILDING = inquisition_armory_01 }
		POD_remove_ducal_building_effect = { BUILDING = inquisition_keep_01 }
		POD_remove_ducal_building_effect = { BUILDING = inquisition_church_01 }
		POD_remove_ducal_building_effect = { BUILDING = inquisition_office_01 }
		POD_remove_ducal_building_effect = { BUILDING = inquisition_palace_01 }
	}
	# Hunters
	if = {
		limit = {
			province_owner = { building_requirement_hunter = no }
			scope:previous_holder ?= { building_requirement_hunter = yes}
		}
		POD_remove_ducal_building_effect = { BUILDING = hunter_ducal_01  }
	}	
	# KJ
	if = {
		limit = {
			province_owner = { POD_chi_user_trigger = no }
			scope:previous_holder ?= { POD_chi_user_trigger = yes }
		}
		POD_remove_ducal_building_effect = { BUILDING = dragon_nest_01 }
	}
	# Mummy
	if = {
		limit = {
			province_owner = { 
				NOT = { has_trait = mummy }
				is_a_neteru = no
			}
			scope:previous_holder ?= { 
				OR = { 
					has_trait = mummy
					is_a_neteru = yes
				} 
			}
		}
		POD_remove_ducal_building_effect = { BUILDING = web_of_faith_01 }
		POD_remove_ducal_building_effect = { BUILDING = occult_affiliation_01 }
		POD_remove_ducal_building_effect = { BUILDING = maats_haven_01 }
	}
	# Demon
	if = {
		limit = {
			province_owner = {
				is_demonic_trigger = no
			}
			scope:previous_holder ?= {
				is_demonic_trigger = yes
			}
		}
		POD_remove_ducal_building_effect = { BUILDING = bastion_01 }
	}
	# Wraith
	if = {
		limit = {
			province_owner = { 
				NOT = { has_trait = wraith } 
			}
			scope:previous_holder ?= { 
				has_trait = wraith
			}
		}
		POD_remove_ducal_building_effect = { BUILDING = necropolis_01 }
	}
	# Fae
	if = {
		limit = {
			province_owner = { 
				NOR = {
					has_trait = fae 
					has_trait = changeling
					has_trait = kinain
				}
			}
			scope:previous_holder ?= { 
				OR = {
					has_trait = fae 
					has_trait = changeling
					has_trait = kinain
				}
			}
		}
		POD_remove_ducal_building_effect = { BUILDING = land_trod_01 }
		POD_remove_ducal_building_effect = { BUILDING = water_trod_01 }
		POD_remove_ducal_building_effect = { BUILDING = sun_trod_01 }
		POD_remove_ducal_building_effect = { BUILDING = moon_trod_01 }
		POD_remove_ducal_building_effect = { BUILDING = chimerical_trod_01 }
		POD_remove_ducal_building_effect = { BUILDING = nightmare_trod_01 }
	}
}

POD_remove_ducal_building_effect = {
	if = {
		limit = { has_building_or_higher = $BUILDING$ }
		if = {
			limit = { NOT = { has_building = $BUILDING$ } } # Replace higher level buidling
			if = {
				limit = { has_free_building_slot = no } 
				add_province_modifier = extra_building_slot
				add_building = $BUILDING$
				remove_province_modifier = extra_building_slot
			}
			else = { add_building = $BUILDING$ }
		}
		remove_building = $BUILDING$
	}
}

POD_replace_tribal_holding_effect = {
	if = {
		limit = { is_grand_city_location_trigger = yes }
		POD_reset_grand_city_effect = yes
	}
	else_if = {
		limit = { NOT = { county = { any_county_province = { has_holding_type = castle_holding } } } }
		set_holding_type = castle_holding 
	}
	else_if = {
		limit = { NOT = { county = { any_county_province = { has_holding_type = city_holding } } } }
		set_holding_type = city_holding 
	}
	else_if = {
		limit = { NOT = { county = { any_county_province = { has_holding_type = church_holding } } } }
		set_holding_type = church_holding 
	}
	else = {
		random_list = {
			10 = { set_holding_type = castle_holding }
			1 = { set_holding_type = city_holding }
			1 = { set_holding_type = church_holding }
		}
	}
}
