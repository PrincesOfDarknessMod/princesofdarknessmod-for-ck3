POD_crafter_xp_gain = {

		random = {
			chance = 50
			add_trait_xp = { trait = pod_crafter track = $TRACK$ value = 1 }
		}
	}

		}
	}
}

POD_crafting_create_minigame = {
	save_scope_as = owner # For Reforging Purpose
	var:crafting_minigame ?= { end_story = yes }
	character:POD_wraith_01 = {
		create_story = { type = POD_story_node_map save_scope_as = crafting_minigame }
                save_scope_as = owner
	        set_artifact_rarity_common = yes
                create_artifact = {
                        name = dummy_artifact_loca  description = dummy_artifact_loca
                        type = armor_plate visuals = armor
                        wealth = 1 quality = 1
                        modifier = artifact_placeholder_modifier
                        generate_history = no
                        save_scope_as = newly_created_artifact
                }
                scope:newly_created_artifact = { remove_artifact_modifier = artifact_placeholder_modifier }
	}
	scope:crafting_minigame ?= {
		set_variable = { name = crafter value = PREV.var:craft_creator }
		set_variable = { name = width value = 700 }
		set_variable = { name = height value = 500 }
		set_variable = { name = size value = var:crafter.crafting_minigame_size }
		set_variable = { name = padding value = 18 }
		set_variable = { name = node_chance value = 75 }
                set_variable = { name = dummy_artifact value = scope:newly_created_artifact }
	}
	set_variable = {
		name = crafting_minigame
		value = scope:crafting_minigame
	}
}

POD_crafting_minigame_generate_map = {
	POD_create_node_map = { SAVE_AS = node_map }
	scope:node_map ?= {
		set_variable = {
			name = quadrant_size_x
			value = { value = $WIDTH$ divide = $SIZE$ }
		}
		set_variable = {
			name = quadrant_size_y
			value = { value = $HEIGHT$ divide = $SIZE$ }
		}
		POD_umbra_adventure_generate_grid_lines = {
			WIDTH = $WIDTH$
			HEIGHT = $HEIGHT$
			COLUMNS = $SIZE$
			ROWS = $SIZE$
			MAP = scope:node_map
		}
	}
	save_scope_value_as = { name = idx value = 0 }
	save_scope_value_as = { name = row_idx value = 0 }
	while = {
		limit = { scope:row_idx < $SIZE$ }
		save_scope_value_as = { name = column_idx value = 0 }
		while = {
			limit = { scope:column_idx < $SIZE$ }
			random = {
				chance = $NODE_CHANCE$
				POD_create_random_node_in_quadrant_with_padding = {
					PADDING = $PADDING$
					COLUMN = scope:column_idx
					ROW = scope:row_idx
					MAP = scope:node_map
				}
                                scope:created_node = { POD_crafting_generate_$TYPE$_nodes = yes }
			}
			POD_increment_scope_value = { SCOPE = column_idx }
		}
		POD_increment_scope_value = { SCOPE = row_idx }
	}
}

POD_crafting_create_forge = {
	var:crafting_minigame ?= { save_scope_as = crafting_minigame }
	POD_crafting_minigame_generate_map = {
                WIDTH = scope:crafting_minigame.var:width
                HEIGHT = scope:crafting_minigame.var:height
                SIZE = scope:crafting_minigame.var:size
                PADDING = scope:crafting_minigame.var:padding
                NODE_CHANCE = scope:crafting_minigame.var:node_chance
		TYPE = forge
	}
	scope:node_map ?= {
		while = {
			count = 3
			random_in_list = {
				variable = nodes
				random_list = {
					1 = { set_variable = { name = node_type value = flag:starting_node_amulet } }
					1 = { set_variable = { name = node_type value = flag:starting_node_crown } }
					1 = { set_variable = { name = node_type value = flag:starting_node_weapon } }
					1 = { set_variable = { name = node_type value = flag:starting_node_regalia } }
					1 = { set_variable = { name = node_type value = flag:starting_node_armor } }
				}
			}
		}
	}
	var:crafting_minigame ?= {
		set_variable = { name = type value = flag:forge }
		set_variable = { name = node_map value = scope:node_map }
		var:dummy_artifact ?= { add_artifact_modifier = artifact_monthly_prestige_penalty_1_modifier }
		var:crafter ?= {
			if = { limit = { has_perk = ward_of_permanence_perk } var:dummy_artifact ?= { set_max_durability = { value = { value = artifact_max_durability add = 20 } } } }
			if = { limit = { has_perk = lesser_seal_of_maat_perk } var:dummy_artifact ?= { set_max_durability = { value = { value = artifact_max_durability add = 10 } } } }
			if = { limit = { has_perk = seal_of_maat_perk } var:dummy_artifact ?= { set_max_durability = { value = { value = artifact_max_durability add = 10 } } } }
			if = { limit = { has_perk = greater_seal_of_maat_perk } var:dummy_artifact ?= { set_max_durability = { value = { value = artifact_max_durability add = 10 } } } }
		}
	}
}

POD_crafting_create_enhance = {
	var:crafting_minigame ?= { save_scope_as = crafting_minigame }
	POD_crafting_minigame_generate_map = {
                WIDTH = scope:crafting_minigame.var:width
                HEIGHT = scope:crafting_minigame.var:height
                SIZE = scope:crafting_minigame.var:size
                PADDING = scope:crafting_minigame.var:padding
                NODE_CHANCE = scope:crafting_minigame.var:node_chance
		TYPE = enhance
	}
	var:crafting_minigame ?= {
		if = { limit = { PREV = { has_variable = crafting_artifact_target } } 
			set_variable = { name = artifact_target value = PREV.var:crafting_artifact_target } 
			var:dummy_artifact = {
				set_artifact_rarity_famed = yes 
			}
			var:artifact_target = {
				switch = {
					trigger = artifact_slot_type
					helmet = { PREV.var:dummy_artifact ?= { reforge_artifact = { type = helmet visuals = crown } } }
					primary_armament = { PREV.var:dummy_artifact ?= { reforge_artifact = { type = sword visuals = sword } } }
					regalia = { PREV.var:dummy_artifact ?= { reforge_artifact = { type = regalia visuals = regalia } } }
					armor = { PREV.var:dummy_artifact ?= { reforge_artifact = { type = armor_plate visuals = armor } } }
					miscellaneous = { PREV.var:dummy_artifact ?= { reforge_artifact = { type = miscellaneous visuals = necklace } } }
					fallback = { PREV.var:dummy_artifact ?= { reforge_artifact = { type = journal visuals = necklace } } } #Court, Mount, Etc.
				}
			}
		}
		set_variable = { name = type value = flag:enhance }
		set_variable = { name = node_map value = scope:node_map }
	}
}
