POD_crafting_minigame_create = {
        if = {
		limit = {
			has_variable = crafting_minigame
		}
		var:crafting_minigame = {
			end_story = yes
		}
	}
	character:POD_wraith_01 = {
		create_story = {
			type = POD_story_node_map
			save_scope_as = crafting_minigame
		}
                save_scope_as = owner
	        set_artifact_rarity_common = yes
                create_artifact = {
                        name = dummy_artifact
                        description = dummy_artifact_desc
                        type = armor_plate
                        visuals = armor
                        wealth = 1
                        quality = 1
                        modifier = artifact_placeholder_modifier
                        generate_history = no
                        save_scope_as = newly_created_artifact
                }
                scope:newly_created_artifact = { remove_artifact_modifier = artifact_placeholder_modifier }
	}
	scope:crafting_minigame = {
		set_variable = {
			name = width
			value = 700
		}
		set_variable = {
			name = height
			value = 500
		}
		set_variable = {
			name = columns
			value = 4
		}
		set_variable = {
			name = rows
			value = 4
		}
		set_variable = {
			name = padding
			value = 18
		}
		set_variable = {
			name = node_chance
			value = 75
		}
                set_variable = {
                        name = dummy_artifact
                        value = scope:newly_created_artifact
                }
	}
	set_variable = {
		name = crafting_minigame
		value = scope:crafting_minigame
	}
	scope:crafting_minigame.var:padding = {
		debug_log_scopes = no
	}
	POD_crafting_minigame_generate_map = {
                WIDTH = scope:crafting_minigame.var:width
                HEIGHT = scope:crafting_minigame.var:height
                COLUMNS = scope:crafting_minigame.var:columns
                ROWS = scope:crafting_minigame.var:rows
                PADDING = scope:crafting_minigame.var:padding
                NODE_CHANCE = scope:crafting_minigame.var:node_chance
                SAVE_AS = node_map
	}

	scope:crafting_minigame = {
		set_variable = {
			name = node_map
			value = scope:node_map
		}
		if = { limit = { PREV = { has_variable = target_artifact } } set_variable = { name = target_artifact value = PREV.var:target_artifact } }
	}
}
POD_crafting_minigame_generate_map = {
	POD_create_node_map = {
		SAVE_AS = $SAVE_AS$
	}
	scope:$SAVE_AS$ = {
		save_scope_as = node_map
		set_variable = {
			name = quadrant_size_x
			value = {
				value = $WIDTH$
				divide = $COLUMNS$
			}
		}
		set_variable = {
			name = quadrant_size_y
			value = {
				value = $HEIGHT$
				divide = $ROWS$
			}
		}
	}
	save_scope_value_as = {
		name = idx
		value = 0
	}
	save_scope_value_as = {
		name = row_idx
		value = 0
	}
	while = {
		limit = {
			scope:row_idx < $ROWS$
		}
		save_scope_value_as = {
			name = column_idx
			value = 0
		}
		while = {
			limit = {
				scope:column_idx < $COLUMNS$
			}
			random = {
				chance = $NODE_CHANCE$
				POD_create_random_node_in_quadrant_with_padding = {
					PADDING = $PADDING$
					COLUMN = scope:column_idx
					ROW = scope:row_idx
					MAP = scope:node_map
				}
                                scope:created_node = {
                                        POD_crafting_minigame_generate_node = yes
                                        set_variable = { name = index value = scope:idx }
                                }
			}
			POD_increment_scope_value = {
				SCOPE = column_idx
			}
		}
		POD_increment_scope_value = {
			SCOPE = row_idx
		}
	}
	scope:node_map = {
		while = {
			count = 3
			limit = { NOT = { PREV = { has_variable = target_artifact } } }
			random_in_list = {
				variable = nodes
				POD_crafting_minigame_generate_strating_node = yes
			}
		}
	}
        POD_umbra_adventure_generate_grid_lines = {
		WIDTH = $WIDTH$
		HEIGHT = $HEIGHT$
		COLUMNS = $COLUMNS$
		ROWS = $ROWS$
		MAP = scope:node_map
	}
}
POD_crafting_minigame_generate_node = {
	random_list = {
		1 = { set_variable = { name = type value = flag:node_positive_prestige } }
		1 = { trigger = { PREV = { NOT = { has_variable = target_artifact } } } set_variable = { name = type value = flag:node_positive_durability } }
		1 = { set_variable = { name = type value = flag:node_positive_prowess } }

		1 = { set_variable = { name = type value = flag:node_negative_prestige } }
		1 = { trigger = { PREV = { NOT = { has_variable = target_artifact } } } set_variable = { name = type value = flag:node_negative_durability } }
		1 = { set_variable = { name = type value = flag:node_negative_prowess } }
	}
}
POD_crafting_minigame_generate_strating_node = {
	random_list = {
		1 = { set_variable = { name = type value = flag:node_amulet } }
		1 = { set_variable = { name = type value = flag:node_crown } }
		1 = { set_variable = { name = type value = flag:node_weapon } }
		1 = { set_variable = { name = type value = flag:node_regalia } }
		1 = { set_variable = { name = type value = flag:node_armor } }
		#1 = { set_variable = { name = type value = flag:node_throne } }
	}
}
POD_crafting_minigame_apply_modifier_from_node = {
        switch = {
                trigger = scope:story.var:type
                flag:node_positive_prestige = { var:dummy_artifact = { add_artifact_modifier = artifact_placeholder_modifier } }
                flag:node_positive_durability = { 
			var:dummy_artifact = { 
				set_max_durability = {
					value = {
						value = artifact_max_durability
						add = 20
					}
				} 
			} 
		}
                flag:node_positive_prowess = { var:dummy_artifact = { add_artifact_modifier = artifact_prowess_1_modifier } }
                
		flag:node_negative_prestige = { var:dummy_artifact = { add_artifact_modifier = artifact_monthly_prestige_penalty_modifier } }
                flag:node_negative_durability = { 
			var:dummy_artifact = { 
				set_max_durability = {
					value = {
						value = artifact_max_durability
						add = -20
					}
				} 
			} 
		}
                flag:node_negative_prowess = { var:dummy_artifact = { add_artifact_modifier = artifact_prowess_1_negative_modifier } }
        }
}
