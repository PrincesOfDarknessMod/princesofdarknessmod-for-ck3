POD_shifter_inherit_breed_effect = {
	random_list = {
		50 = {
			trigger = {
				OR = {
					mother ?= { has_trait = homid_born } 
					real_father ?= { has_trait = homid_born }
				}
			}
			add_trait = homid_born
		}
		50 = {
			trigger = {
				OR = {
					mother ?= { has_trait = beast_born } 
					real_father ?= { has_trait = beast_born }
				}
			}
			add_trait = beast_born
		}
		50 = {
			trigger = {
				OR = {
					mother ?= { 
						has_trait = metis
						has_trait_rank = {
							trait = shapeshifter
							character = prev
						}
					}
					real_father ?= { 
						has_trait = metis 
						has_trait_rank = {
							trait = shapeshifter
							character = prev
						}
					}
				}
			}
			add_trait = metis
		}
	}
	if = {
		limit = { POD_has_breed_trigger = no }
		add_trait = homid_born
	}
}
