
POD_shih_initialization_effect = {
	if = {
		limit = {
			NOT = { has_trait = shih }
		}
		add_trait = shih
	}
	if = {
		limit = {
			is_a_cultist = no
			NOT = { religion = { is_in_family = rf_hunters } }
		}
		set_character_faith = faith:shih_faith
	}
	if = {
		limit = {
			has_chi_trigger = no
		}
		add_character_modifier = yinvirtue5_modifier
		add_character_modifier = yangvirtue5_modifier
		POD_set_chi_effect = yes
	}
}

POD_remove_shih_atribute_effect = {

	POD_clean_chivirtue_effect = yes
}
