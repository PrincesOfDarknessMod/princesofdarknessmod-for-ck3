
POD_create_coterie_effect = {
	create_story = {
		type = POD_story_coterie
		save_scope_as = coterie
	}
	
	# if-statement necessary so error log doesn't complain
	if = {
		limit = { exists = scope:coterie }
		
		scope:coterie = {
			set_variable = {
				name = root_char
				value = prev
			}
		}
	
		POD_join_coterie_effect = yes
	}
}

# need to pass coterie as a scope
POD_join_coterie_effect = {
	save_scope_as = new_coterie_member
	
	scope:coterie = {
		every_in_list = {
			variable = members
			set_relation_coterie_member = scope:new_coterie_member
			send_interface_toast = {
				title = POD_new_coterie_member_title
				left_icon = scope:new_coterie_member
				custom_tooltip = POD_new_coterie_member_tt
			}
		}
		add_to_variable_list = {
			name = members
			target = prev
		}
	}
	
	set_variable = {
		name = coterie
		value = scope:coterie
	}
}

POD_leave_coterie_effect = {
	save_scope_as = old_coterie_member
	
	every_relation = {
		type = coterie_member
		send_interface_toast = {
			title = POD_coterie_member_left_title
			left_icon = scope:old_coterie_member
			custom_tooltip = POD_coterie_member_left_tt
		}
		prev = {
			remove_relation_coterie_member = prev
		}
	}
	var:coterie = {
		remove_list_variable = {
			name = members
			target = prev
		}
	}
	remove_variable = coterie
}

POD_disband_coterie_effect = {
	var:coterie = {
		end_story = yes
	}
}

# Run this before every "skill check" during Journey events
POD_save_coterie_duelists = {
	hidden_effect = {
		if = {
			limit = {
				has_variable = coterie
				has_variable = traveling_with_coterie
			}
			save_temporary_scope_as = coterie_leader
			var:coterie = {
				ordered_in_list = {
					variable = members
					limit = {
						OR = {
							current_travel_plan ?= { this = scope:coterie_leader.current_travel_plan }
							NOT = { exists = scope:coterie_leader.current_travel_plan }
						}
					}
					order_by = diplomacy
					position = 0
					save_scope_as = diplomacy_duelist
				}
				ordered_in_list = {
					variable = members
					limit = {
						OR = {
							current_travel_plan ?= { this = scope:coterie_leader.current_travel_plan }
							NOT = { exists = scope:coterie_leader.current_travel_plan }
						}
					}
					order_by = martial
					position = 0
					save_scope_as = martial_duelist
				}
				ordered_in_list = {
					variable = members
					limit = {
						OR = {
							current_travel_plan ?= { this = scope:coterie_leader.current_travel_plan }
							NOT = { exists = scope:coterie_leader.current_travel_plan }
						}
					}
					order_by = stewardship
					position = 0
					save_scope_as = stewardship_duelist
				}
				ordered_in_list = {
					variable = members
					limit = {
						OR = {
							current_travel_plan ?= { this = scope:coterie_leader.current_travel_plan }
							NOT = { exists = scope:coterie_leader.current_travel_plan }
						}
					}
					order_by = intrigue
					position = 0
					save_scope_as = intrigue_duelist
				}
				ordered_in_list = {
					variable = members
					limit = {
						OR = {
							current_travel_plan ?= { this = scope:coterie_leader.current_travel_plan }
							NOT = { exists = scope:coterie_leader.current_travel_plan }
						}
					}
					order_by = learning
					position = 0
					save_scope_as = learning_duelist
				}
				ordered_in_list = {
					variable = members
					limit = {
						OR = {
							current_travel_plan ?= { this = scope:coterie_leader.current_travel_plan }
							NOT = { exists = scope:coterie_leader.current_travel_plan }
						}
					}
					order_by = prowess
					position = 0
					save_scope_as = prowess_duelist
				}
			}
		}
		else = {
			save_scope_as = diplomacy_duelist
			save_scope_as = martial_duelist
			save_scope_as = stewardship_duelist
			save_scope_as = intrigue_duelist
			save_scope_as = learning_duelist
			save_scope_as = prowess_duelist
		}
	}
}
