POD_chi_decrease_effect = {
	if = {
		limit = {
			OR = {
				trigger_if = {
					limit = { 
						OR = {
							has_character_modifier = yinchi9_modifier 
							has_character_modifier = yinchi8_modifier
						}
					}
					yang_points_value < 9
				}
				trigger_else_if = {
					limit = { 
						OR = {
							has_character_modifier = yinchi7_modifier 
							has_character_modifier = yinchi6_modifier
						}
					}
					yang_points_value < 7
				}
				trigger_else_if = {
					limit = { 
						OR = {
							has_character_modifier = yinchi5_modifier 
							has_character_modifier = yinchi4_modifier
						}
					}
					yang_points_value < 5
				}
				trigger_else_if = {
					limit = { 
						OR = {
							has_character_modifier = yinchi3_modifier 
							has_character_modifier = yinchi2_modifier
						}
					}
					yang_points_value < 3
				}
				trigger_else = {
					OR = {
						has_character_modifier = yinchi10_modifier
						has_character_modifier = yangchi0_modifier
					}
				}
			}
		}
		POD_yinchi_decrease_effect = yes
	}
	else = {

	}

POD_absorb_chi_from_flesh_or_blood_effect = {
	if = {
		limit = { scope:actor.yin_difference >= 1 }
		if = {
			limit = { scope:actor.yang_difference > scope:recipient.lifeforce_points_value }

			scope:actor = { POD_yinchi_increase_multiple_final_effect = { COUNT = yin_difference } }
