POD_judges_words_of_healing_effect = {
	hidden_effect = {
		POD_remove_character_modifier = { MODIFIER = POD_judge_ties_that_bind_best_friend_modifier }
		POD_remove_character_modifier = { MODIFIER = POD_judge_ties_that_bind_friend_modifier }
		if = {
			limit = { has_perk = endowments_judge_sephardim_4_perk }

			if = {
				limit = { any_relation = { type = best_friend } }
				add_character_modifier = POD_judge_ties_that_bind_best_friend_modifier
			}
			else_if = {
				limit = { any_relation = { type = friend } }
				add_character_modifier = POD_judge_ties_that_bind_friend_modifier
			}
		}
	}
}

POD_create_appropriate_akritai_relic_artifact_from_character_effect = {

	$OWNER$ = { save_scope_as = owner }
	get_artifact_quality_effect = yes
	get_artifact_wealth_effect = yes
	$OWNER$ = {
		random_list = {
			20 = {
				trigger = {

			15 = {

			}
			15 = {

				}
				create_akritai_relic_artifact_effect = { NAME = rib TYPE = pedestal VISUAL = reliquary }
			}
			15 = {
				trigger = {
					scope:creator = { has_corpse_trigger = yes }
				}
				create_akritai_relic_artifact_effect = { NAME = hair TYPE = pedestal VISUAL = reliquary }
			}
			10 = {
				create_akritai_relic_artifact_effect = { NAME = vestment TYPE = pedestal VISUAL = reliquary }
			}
			10 = {

			10 = {

				add_scaled_artifact_modifier_devotion_effect = yes
				add_scaled_artifact_modifier_grandeur_small_effect = yes
				remove_artifact_modifier = artifact_placeholder_modifier
			}
		}
	}
}

create_akritai_relic_artifact_effect = {
	create_artifact = {	
		name = pam_saint_relic_$NAME$_name
		description = pam_saint_relic_$NAME$_desc
		visuals = $VISUAL$
		type = $TYPE$
		creator = scope:creator
		template = akritai_saint_relic_template
		modifier = artifact_placeholder_modifier
		save_scope_as = newly_created_artifact
		wealth = scope:wealth
		quality = scope:quality
	}
}
