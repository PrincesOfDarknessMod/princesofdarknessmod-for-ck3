	hidden_effect = {

		POD_remove_character_modifier = { MODIFIER = POD_judge_ties_that_bind_friend_modifier }

			limit = { has_perk = endowments_judge_sephardim_4_perk }

			if = {
				limit = { any_relation = { type = best_friend } }
				add_character_modifier = POD_judge_ties_that_bind_best_friend_modifier
			}
			else_if = {
				limit = { any_relation = { type = friend } }
				add_character_modifier = POD_judge_ties_that_bind_friend_modifier
			}
		}
	}
}

create_akritai_relic_artifact_effect = {
	create_artifact = {	
		name = pam_saint_relic_$NAME$_name
		description = pam_saint_relic_$NAME$_desc
		visuals = $VISUAL$
		type = $TYPE$
		creator = scope:creator
		template = akritai_saint_relic_template
		modifier = artifact_placeholder_modifier
		save_scope_as = newly_created_artifact
		wealth = scope:wealth
		quality = scope:quality
	}
}

POD_akritai_reduce_patience = {
	save_scope_value_as = {
		name = reduced_patience
		value = $VAL$
	}
	save_scope_value_as = {
		name = actual_change
		value = {
			value = $VAL$
			multiply = -1
		}
	}
	custom_tooltip = {
		text = POD_akritai_reduce_patience_tt
		POD_set_or_change_variable = { VAR = akritai_patience VAL = scope:actual_change }
	}
	hidden_effect = {
		if = {
			limit = { var:akritai_patience < 0 }
			set_variable = { name = akritai_patience value = 0 }
		}
		POD_story_akritai_pulled_in_two_directions_resync_effect = yes
	}
}

# Force-update story's vars to match char vars
# Also check if Patience has hit 0
POD_story_akritai_pulled_in_two_directions_resync_effect = {
	save_scope_as = story_owner
	every_owned_story = {
		limit = { story_type = POD_story_akritai_pulled_in_two_directions }
		# Only mirror while the story is unresolved
		if = {
			limit = { scope:story_owner = { POD_story_akritai_pulled_in_two_directions_resolved_trigger = no } }
			# Update the variables to update the visible counters
			if = {
				limit = { scope:story_owner = { has_variable = akritai_influence_balance } }
				set_variable = {
					name = akritai_influence_balance
					value = scope:story_owner.var:akritai_influence_balance
				}
			}
			if = {
				limit = { scope:story_owner = { has_variable = akritai_patience } }
				set_variable = {
					name = akritai_patience
					value = scope:story_owner.var:akritai_patience
				}
			}

			#If 0 Patience, resolve the story
			if = {
				limit = { var:akritai_patience ?= 0 }

				# Russian
				if = {
					limit = { var:akritai_influence_balance > 25 }
					set_global_variable = {
						name = pod_akritai_chosen_faction
						value = flag:russian
					}
					scope:story_owner = {
						add_character_modifier = POD_story_akritai_pulled_in_two_directions_russian_support_modifier
						# Remove true faith as you lose your holy focus in favor of more political influence, though I won't remove beatific as it's way more special
						if = {
							limit = { has_trait = truefaith }
							remove_trait = truefaith
						}
						# Swap holy sites for Russian ones
						faith:akritai = {
							add_holy_site = {
								target = holy_site:kiev
								actor = scope:story_owner
							}
							add_holy_site = {
								target = holy_site:moskva
								actor = scope:story_owner
							}
							add_holy_site = {
								target = holy_site:novgorod
								actor = scope:story_owner
							}
							add_holy_site = {
								target = holy_site:saran
								actor = scope:story_owner
							}
							add_holy_site = {
								target = holy_site:minsk
								actor = scope:story_owner
							}

							hidden_effect = {
								remove_holy_site = {
									target = holy_site:jerusalem
									actor = scope:story_owner
								}
								remove_holy_site = {
									target = holy_site:constantinople
									actor = scope:story_owner
								}
								remove_holy_site = {
									target = holy_site:rome
									actor = scope:story_owner
								}
								remove_holy_site = {
									target = holy_site:alexandria
									actor = scope:story_owner
								}
								remove_holy_site = {
									target = holy_site:antioch
									actor = scope:story_owner
								}
							}
						}
					}
					
					if = {
						limit = { var:akritai_influence_balance >= 90 }
						scope:story_owner = { add_character_modifier = POD_story_akritai_pulled_in_two_directions_super_russian_modifier }
					}
				}

				# Independent
				else_if = {
					limit = {
						var:akritai_influence_balance <= 25
						var:akritai_influence_balance >= -25
					}
					set_global_variable = {
						name = pod_akritai_chosen_faction
						value = flag:independent
					}
					scope:story_owner = { add_character_modifier = POD_story_akritai_pulled_in_two_directions_independent_support_modifier }

					if = {
						limit = {
							var:akritai_influence_balance <= 5
							var:akritai_influence_balance >= -5
						}
						scope:story_owner = { add_character_modifier = POD_story_akritai_pulled_in_two_directions_super_independent_modifier }
					}
				}

				# Byzantine
				else = {
					set_global_variable = {
						name = pod_akritai_chosen_faction
						value = flag:byzantine
					}
					# Remove true faith as you lose your holy focus in favor of more political influence, though I won't remove beatific as it's way more special
					scope:story_owner = {
						add_character_modifier = POD_story_akritai_pulled_in_two_directions_byzantine_support_modifier
						if = {
							limit = { has_trait = truefaith }
							remove_trait = truefaith
						}
					}
					
					if = {
						limit = { var:akritai_influence_balance <= -90 }
						scope:story_owner = { add_character_modifier = POD_story_akritai_pulled_in_two_directions_super_byzantine_modifier }
					}
				}


				# Hide the counters
				if = { limit = { has_variable = akritai_influence_balance } remove_variable = akritai_influence_balance }
				if = { limit = { has_variable = akritai_patience } remove_variable = akritai_patience }
			}
		}
	}
}
