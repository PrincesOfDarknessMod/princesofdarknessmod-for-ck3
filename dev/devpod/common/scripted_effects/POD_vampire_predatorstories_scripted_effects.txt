POD_ventrue_friendly_gender_effect = {
	random_list = {
		50 = { #Female
			modifier = {
				factor = 0
				OR = {
					any_secret = { secret_type = secret_prey_young_men } 
					any_secret = { secret_type = secret_prey_middle_aged_men }
					any_secret = { secret_type = secret_prey_old_men }
				}
			}	
			dummy_female = { save_scope_as = ventrue_friendly_gender }
		}
		50 = { #Male
			modifier = {
				factor = 0
				OR = {
					any_secret = { secret_type = secret_prey_young_women }
					any_secret = { secret_type = secret_prey_middle_aged_women }
					any_secret = { secret_type = secret_prey_old_women }
				}
			}
			dummy_male = { save_scope_as = ventrue_friendly_gender }	
		}
	}
}
