# Ran before supernatural perks, modifiers etc are applied.
# For newly generated characters (e.g. random barons, ruler designer)
POD_remove_wrong_supernatural_traits_on_new_character = {
	# Remove Wraith traits first so the Risen trait won't interfere with Discipline removal
	if = {
		limit = { POD_is_wraith_trigger = no }
		POD_remove_wraith_traits_effect = yes
		POD_remove_wraith_arcanoi_traits_effect = yes
	}
	
	if = {
		limit = { has_POD_vamp_blood_trigger = no }
		if = {
			limit = { has_trait = risen }
			POD_remove_vampire_non_risen_discipline_traits_effect = yes
		}
		else = {
			POD_remove_vampire_discipline_traits_effect = yes
		}
	}
	
	if = {
		limit = { POD_is_vampire_trigger = no }
		POD_remove_vampire_generation_traits_effect = yes
		POD_remove_vampire_clan_traits_effect = yes
		POD_remove_vampire_misc_traits_effect = yes
		POD_remove_vampire_predatortype_traits_effect = yes
	}
	
	if = {
		limit = {
			has_POD_vamp_blood_trigger = no
			POD_can_participate_in_kj_politics = no
		}
		remove_trait = ashen_adventurer
	}
	
	if = {
		limit = { POD_splat_has_dharma_trigger = no }		
		POD_clean_dharma_effect = yes
	}
	
	if = {
		limit = { POD_splat_has_access_to_chi_arts = no }
		POD_trait_injector_chi_arts = { EFFECT = POD_remove_trait }
	}
	
	if = {
		limit = { POD_is_supehunter_trigger = no }
		POD_remove_hunter_traits_effect = yes
		remove_trait = shih
	}
	
	if = {
		limit = { POD_is_mummy_trigger = no }
		POD_remove_mummy_traits_effect = yes
	}
	
	if = {
		limit = { POD_is_demon_trigger = no }
		POD_remove_demon_traits_effect = yes
	}
	
	if = {
		limit = {
			NOR = {
				has_trait = abomination
				has_trait = fera
			}
		}
		POD_remove_breed_traits_effect = yes
		POD_remove_lunar_auspice_traits_effect = yes
		POD_remove_pryio_traits_effect = yes
		POD_remove_solar_auspice_traits_effect = yes
	}
	if = {
		limit = { NOT = { has_trait = kinfolk } }
		remove_trait = gnosis
	}
	if = {
		limit = { 
			POD_is_fae_trigger = no 
			POD_is_changeling_trigger = no
		}
		POD_remove_fae_kith_traits_effect = yes
	}
	
	if = {
		limit = { has_POD_fae_blood_trigger = no }
		POD_remove_fae_art_traits_effect = yes
		POD_remove_fae_realm_traits_effect = yes
	}
	
	if = {
		limit = { POD_can_be_possessed_basic_trigger = no }
		remove_trait = fomori
		remove_trait = drone
		remove_trait = gorgon
		if = {
			limit = { has_trait_with_flag = is_spirit }
			remove_trait = kami
		}
	}
	
	if = {
		limit = {
			OR = {
				has_trait = theurgist
				is_sorcerer_trigger = yes
			}
			POD_splat_is_valid_for_sorcery = no
		}
		# blood sorcery assignment handled in POD_sorcerer_embrace_effect
		add_character_flag = {
			flag = sorcerer_embrace
			days = 1
		}
		remove_trait = theurgist
		remove_trait = sorcerer
		random_secret = {
			limit = { secret_type = secret_sorcerer }
			remove_secret = yes
		}
		if = {
			limit = { has_POD_vamp_blood_trigger = yes }
			add_blood_magic_lifestyle_perk_points = {
				value = POD_numina_sorcery_perk_amount
				add   = POD_numina_theurgy_perk_amount
			}
		}
		POD_remove_all_perks_in_lifestyle = {
			LIFESTYLE = sorcery_lifestyle
		}
		POD_remove_all_perks_in_lifestyle = {
			LIFESTYLE = theurgy_lifestyle
		}
		remove_variable = POD_numina_sorcery_additional_perk_cost
	}
	
	if = {
		limit = { POD_splat_is_valid_for_psychic = no }
		remove_trait = psychic
	}
	
	if = {
		limit = { POD_splat_can_be_infernalist = no }
		remove_trait = witch
	}
}

# For characters who've already received their splat initialization from before.
# Usually for player characters who switch their splat.
# More expensive in terms of performance. Use sparingly.
POD_remove_wrong_supernatural_traits_on_existing_character = {
	POD_remove_wrong_supernatural_traits_on_new_character = yes
	
	##### Energy maintenance
	
	hidden_effect = {
		POD_change_to_appropriate_energy_type = yes
	}
	
	##### Traits
	
	if = {
		limit = {
			has_trait = wight
			POD_splat_can_be_wight = no
		}
		remove_trait = wight
	}
	
	##### Modifiers
	
	if = {
		limit = { POD_splat_has_bloodpotency_trigger = no }
		POD_remove_bloodpotency_modifiers_effect = yes
	}
	if = {
		limit = { POD_splat_has_masquerade_trigger = no }
		POD_remove_masquerade_modifiers_effect = yes
	}
	if = {
		limit = { POD_is_vampire_trigger = no }
		POD_remove_vampire_clan_modifiers_effect = yes
	}
	if = {
		limit = { POD_splat_has_wheel_of_ages_modifiers = no }
		POD_remove_wheel_of_ages_modifiers = yes
	}
	if = {
		limit = { POD_is_wraith_trigger = no }
		POD_remove_conduit_modifiers_effect = yes
		POD_clear_wraith_support_modifier = yes
		POD_side_effects_if_no_longer_wraith = yes
	}
	if = {
		limit = { POD_is_pooka_trigger = no }
		POD_remove_pooka_affinity_modifiers_effect = yes
	}
	
	##### Lifestyles
	
	# Vampire lifestyles
	
	if = {
		limit = {
			POD_splat_has_access_to_disciplines = no
			POD_splat_has_access_to_risen_disciplines = no
		}
		POD_remove_all_perks_in_lifestyle = { LIFESTYLE = animal_lifestyle }
		POD_remove_all_perks_in_lifestyle = { LIFESTYLE = choleric_lifestyle }
		POD_remove_all_perks_in_lifestyle = { LIFESTYLE = melancholy_lifestyle }
		POD_remove_all_perks_in_lifestyle = { LIFESTYLE = phlegmatic_lifestyle }
		POD_remove_all_perks_in_lifestyle = { LIFESTYLE = sanguine_lifestyle }
		POD_remove_all_perks_in_lifestyle = { LIFESTYLE = blood_magic_lifestyle }
		POD_remove_all_perks_in_lifestyle = { LIFESTYLE = necro_magic_lifestyle }
		
		POD_remove_vampire_bloodsorcery_traditions_effect = yes
		POD_remove_vampire_necromancy_traditions_effect = yes
		
		remove_character_flag = POD_ghoul_shouldnt_get_new_disciplines
		
		remove_character_flag = duplicate_warrior_valeren_cost_flag
		remove_character_flag = duplicate_watcher_valeren_cost_flag
		remove_character_flag = duplicate_healer_valeren_cost_flag
		remove_character_flag = triplicate_warrior_valeren_cost_flag
		remove_character_flag = triplicate_watcher_valeren_cost_flag
		remove_character_flag = triplicate_healer_valeren_cost_flag
		
		POD_flag_injector_aspect_of_the_beast = {
			EFFECT = POD_remove_character_flag
		}
		POD_modifier_injector_spirit_beast = {
			EFFECT = POD_remove_character_modifier
		}
	}
	else = {
		POD_lose_inaccessible_discipline_perks_effect = yes
	}
	
	if = {
		limit = { POD_splat_can_be_fleshcrafted = no }
		remove_trait = fleshcrafted
		POD_remove_all_perks_in_lifestyle = { LIFESTYLE = fleshcrafting_lifestyle }
		# spirit lobe maintenance
		every_relation = {
			type = umbra_employer
			trigger_event = { id = POD_umbra_expedition.7000 days = 1 }
		}
	}

	
	# Kuei-Jin lifestyles
	
	if = {
		limit = { POD_splat_has_access_to_demon_arts = no }
		POD_remove_all_perks_in_lifestyle = { LIFESTYLE = demon_arts_lifestyle }
		POD_modifier_injector_demon_arts = {
			EFFECT = POD_remove_all_character_modifier_instances_and_tracked_stacks
		}
		POD_flag_injector_demon_regalia = { EFFECT = POD_remove_character_flag }
	}
	
	if = {
		limit = { POD_splat_has_access_to_soul_arts = no }
		POD_remove_all_perks_in_lifestyle = { LIFESTYLE = soul_arts_lifestyle }
	}
	
	if = {
		limit = { POD_splat_has_access_to_chi_arts = no }
		POD_remove_all_perks_in_lifestyle = { LIFESTYLE = chi_arts_lifestyle }
	}
	
	if = {
		limit = { POD_splat_has_access_to_shintai = no }
		POD_remove_all_perks_in_lifestyle = { LIFESTYLE = shintai_lifestyle }
	}
	
	### TODO: Fera lifestyles
	
	# Mummy lifestyles
	
	if = {
		limit = { POD_splat_has_access_to_hekau = no }
		POD_remove_all_perks_in_lifestyle = { LIFESTYLE = hekau_alchemy_lifestyle }
		POD_remove_all_perks_in_lifestyle = { LIFESTYLE = hekau_amulets_lifestyle }
		POD_remove_all_perks_in_lifestyle = { LIFESTYLE = hekau_celestial_lifestyle }
		POD_remove_all_perks_in_lifestyle = { LIFESTYLE = hekau_effigy_lifestyle }
	}
	
	if = {
		limit = { POD_splat_has_access_to_hekau_necromancy = no }
		POD_remove_all_perks_in_lifestyle = { LIFESTYLE = hekau_necromancy_lifestyle }
	}
	
	if = {
		limit = { POD_splat_has_access_to_hekau_nomenclature = no }
		POD_remove_all_perks_in_lifestyle = { LIFESTYLE = hekau_nomenclature_lifestyle }
	}
	
	if = {
		limit = { POD_splat_has_access_to_tan = no }
		POD_remove_all_perks_in_lifestyle = { LIFESTYLE = tan_lifestyle }
	}
	
	# Demonic lifestyles
	
	if = {
		limit = { POD_splat_has_access_to_arcana = no }
		POD_remove_all_perks_in_lifestyle = { LIFESTYLE = arcana_lifestyle }
	}
	
	if = {
		limit = { POD_splat_has_access_to_lores = no }
		POD_remove_all_perks_in_lifestyle = { LIFESTYLE = lore_lifestyle }
	}
	
	if = {
		limit = { POD_splat_can_be_infernalist = no }
		POD_clear_investiture_lifestyle = yes
	}
	
	# Fae lifestyles
	
	if = {
		limit = { POD_splat_has_access_to_fae_arts = no }
		POD_remove_all_perks_in_lifestyle = { LIFESTYLE = realm_lifestyle }
		POD_remove_all_perks_in_lifestyle = { LIFESTYLE = common_arts_lifestyle }
		POD_remove_all_perks_in_lifestyle = { LIFESTYLE = rare_arts_lifestyle }
	}
	
	if = {
		limit = { POD_splat_has_access_to_kith_arts = no }
		POD_remove_all_perks_in_lifestyle = { LIFESTYLE = kith_arts_lifestyle }
	}
	
	if = {
		limit = { POD_splat_has_access_to_slivers = no }
		POD_remove_all_perks_in_lifestyle = { LIFESTYLE = slivers_lifestyle }
	}
	
	if = {
		limit = { POD_splat_has_access_to_wu_tan = no }
		POD_remove_all_perks_in_lifestyle = { LIFESTYLE = wu_tan_lifestyle }
	}
	
	# Wraith lifestyles
	
	if = {
		limit = { POD_splat_has_access_to_arcanoi = no }
		POD_remove_all_perks_in_lifestyle = { LIFESTYLE = arcanoi_lifestyle }
		POD_remove_all_perks_in_lifestyle = { LIFESTYLE = jade_arcanoi_lifestyle }
		#POD_remove_all_perks_in_lifestyle = { LIFESTYLE = dark_arcanoi_lifestyle }
		#POD_remove_all_perks_in_lifestyle = { LIFESTYLE = misc_arcanoi_lifestyle }
		POD_remove_all_perks_in_lifestyle = { LIFESTYLE = thorn_lifestyle }
	}
	
	# Numina lifestyles
	
	if = {
		limit = { NOT = { has_trait = psychic } }
		POD_remove_all_perks_in_lifestyle = {
			LIFESTYLE = psi_lifestyle
		}
	}
	
	if = {
		limit = { NOT = { has_trait = truepossessed } }
		POD_remove_all_perks_in_lifestyle = { LIFESTYLE = possessedpower_lifestyle }
	}
	
	##### Variables
	
	if = {
		limit = { POD_should_have_fera_type_variable = no }
		remove_variable = fera_type
	}

	if = {
		limit = { NOT = { has_trait = ghoul } }
		POD_clear_bound_ghoul_variables = yes
		remove_character_flag = recent_ghoulify
	}
	
	##### Flags
	
	if = {
		limit = { POD_should_have_disease_immunity = no }
		remove_character_flag = immune_to_disease
	}
	
	##### Relationhips
	
	if = {
		limit = { POD_is_vampire_trigger = no }
		every_relation = {
			type = herd_member
			prev = { POD_remove_herd_no_masquerade_effect = { TARGET = prev } }
		}
	}
	
	if = {
		limit = { POD_splat_can_create_ghouls_naturally = no }
		every_relation = {
			type = ghoul_thrall
			POD_clear_bound_ghoul_variables_and_relations = yes
		}
	}
	
	if = {
		limit = { POD_splat_has_access_to_arcana = no }
		every_relation = {
			type = worshipping_cultist
			prev = {
				if = {
					limit = { has_relation_demonic_thrall = prev }
					remove_relation_demonic_thrall = prev
				}
				remove_relation_worshipping_cultist = prev
			}
		}
	}
}

POD_remove_child_relation_effect = {
	if = {
		limit = {
			any_child = {
				even_if_dead = yes
				count >= 1
			}
		}
		create_character = {
			location = this.capital_province
			age = 1200
			culture = this.culture
			faith = this.faith
			dynasty = none
			gender = this
			random_traits = no
			save_temporary_scope_as = created_badparent
		}
								
		if = {
			limit = { exists = father }
			scope:created_badparent = { set_father = prev.father }
		}				
		if = {
			limit = { exists = mother }
			scope:created_badparent = { set_mother = prev.mother }
		}			
		if = {
			limit = { exists = house }
			scope:created_badparent = { set_house = prev.house }
		}
		
		if = {
			limit = { is_female = yes }
			every_child = {
				even_if_dead = yes
				set_mother = scope:created_badparent
			}
		}
		else = {
			every_child = {
				even_if_dead = yes
				set_father = scope:created_badparent
			}
		}
		
		scope:created_badparent = {
			death = { death_reason = death_disappearance }
		}
	}
	if = {
		limit = {
			is_female = no
			any_consort = { is_pregnant = yes }
		}
		create_character = {
			location = this.capital_province
			age = 40
			culture = this.culture
			faith = this.faith
			dynasty = none
			gender = male
			trait = mortal
			random_traits = no
			save_temporary_scope_as = assumed_father
		}
		every_consort = {
			limit = { is_pregnant = yes }
			set_pregnancy_assumed_father = scope:assumed_father
		}
		scope:assumed_father = {
			death = { death_reason = death_disappearance }
		}
	}
}
