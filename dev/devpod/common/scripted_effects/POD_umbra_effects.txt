POD_move_at_local_spirit_list_effect = {
	if = {
		limit = { 
			trigger_if = {
				limit = { 
					has_character_modifier = traveling_spirit_world_modifier
					$PLAYER$ = { has_character_modifier = traveling_spirit_world_modifier }
				}
				has_variable = spirit_location
				$PLAYER$ = { has_variable = spirit_location }
				var:spirit_location = $PLAYER$.var:spirit_location
			}
			trigger_else = {
				NOT = { has_character_modifier = traveling_spirit_world_modifier }
				$PLAYER$ = { NOT = { has_character_modifier = traveling_spirit_world_modifier } }
			}
		}
		$PLAYER$ = {
			add_to_variable_list = {
				name = spirit_list
				target = prev
			}
		}
	}
	if = {
		limit = { NOT = { has_relation_necromancer = $PLAYER$ } }
		move_to_pool = yes
	}
}

create_artifact_reweaved_artifact_effect = { 
	$OWNER$ = { save_scope_as = owner }
			
	$CREATOR$ = { save_scope_as = creator }

	scope:owner = {
		if = {
			limit = { scope:weapon = yes }
			create_artifact_weapon_effect = {
				OWNER = scope:owner
				CREATOR = scope:creator

			}
		}
		else_if = {
			limit = { scope:armor = yes }
			create_artifact_armor_effect = {
				OWNER = scope:owner
				CREATOR = scope:creator
				SET_ARMOR_TYPE = flag:no
			}
		}
		else_if = {
			limit = { scope:shinies = yes }
			random_list = {

						OWNER = scope:owner
						SMITH = scope:owner
					}
				}
				10 = {
					create_artifact_brooch_effect = {
						OWNER = scope:owner
						SMITH = scope:owner
					}
				}
			}
		}
		else_if = {
			limit = { scope:book = yes }
			create_artifact_book_effect = {
				OWNER = scope:owner
				CREATOR = scope:creator
				SET_SUBJECT = flag:no
				SET_TOPIC = flag:no
			}
		}
		else_if = {
			limit = { scope:furniture = yes }
			random_list = {
				10 = {
					create_artifact_throne_effect = {
						OWNER = scope:owner
						MAKER = scope:owner
					}
				}
				10 = {
					create_artifact_tapestry_effect = {
						OWNER = scope:owner
						WEAVER = scope:owner
					}
				}
				10 = {
					create_artifact_box_small_effect = {
						OWNER = scope:owner
						CARPENTER = scope:owner
					}
				}
				10 = {
					create_artifact_box_medium_effect = {
						OWNER = scope:owner
						CARPENTER = scope:owner
					}
				}
				10 = {
					create_artifact_goblet_effect = {
						OWNER = scope:owner
						SMITH = scope:owner
					}
				}
				10 = {
					create_artifact_animal_hide_big_effect = {
						OWNER = scope:owner
						HUNTER = scope:owner
					}
				}
				10 = {
					create_artifact_animal_hide_medium_effect = {
						OWNER = scope:owner
						HUNTER = scope:owner
					}
				}
				10 = {
					create_artifact_animal_hide_small_effect = {
						OWNER = scope:owner
						HUNTER = scope:owner
					}
				}
				10 = {
					create_artifact_wall_banner_effect = {
						OWNER = scope:owner
						CREATOR = scope:owner
						TARGET = scope:owner.primary_title
					}
				}
				10 = {
					create_artifact_wall_skull_effect = {
						OWNER = scope:owner
						HUNTER = scope:owner
					}
				}
				10 = {
					create_artifact_sculpture_effect = {
						OWNER = scope:owner
						SCULPTER = scope:owner
					}
				}
				10 = {
					create_artifact_wall_shield_effect = {
						OWNER = scope:owner
						TARGET = scope:owner.house
						CREATOR = scope:owner
					}
				}
			}
		}
		else_if = {
			limit = { scope:trinket = yes }
			create_artifact = {
				name = cuckoldry_psalter_name
				description = cuckoldry_psalter_description
				type = miscellaneous
				visuals = ring
				modifier = artifact_placeholder_modifier
			}
		}
		
		hidden_effect = {
			scope:newly_created_artifact = { clear_artifact_modifiers = yes }
			if = {
				limit = { scope:owner = scope:creator }
				every_character_artifact = {
					limit = {
						has_variable = artifact_pieces
					}
					if = {
						limit = { 
							rarity = illustrious 
							scope:owner = { has_perk = princely_restoration_perk }
						}
						scope:owner = { add_piety = major_piety_loss }
						scope:newly_created_artifact = { copy_artifact_modifiers = prev }
						destroy_artifact = this
					}
					else_if = {
						limit = { 
							rarity = famed 
							scope:owner = { has_perk = greater_restoration_perk }
						}
						scope:owner = { add_piety = medium_piety_loss }
						scope:newly_created_artifact = { copy_artifact_modifiers = prev }
						destroy_artifact = this
					}
					else_if = {
						limit = { 
							rarity = masterwork 
							scope:owner = { has_perk = major_restoration_perk }
						}
						scope:owner = { add_piety = minor_piety_loss }
						scope:newly_created_artifact = { copy_artifact_modifiers = prev }
						destroy_artifact = this
					}
					else = {
						scope:owner = { add_piety = miniscule_piety_loss }
						scope:newly_created_artifact = { copy_artifact_modifiers = prev }
						destroy_artifact = this
					}
				}
			}
			else = {
				every_equipped_character_artifact = {
					limit = {
						artifact_slot_type = miscellaneous
						has_variable = artifact_pieces
					}
					scope:newly_created_artifact = { copy_artifact_modifiers = prev }
					if = {
						limit = { rarity = common }
						scope:owner = { add_piety = miniscule_piety_loss }
					}
					else_if = {
						limit = { rarity = masterwork }
						scope:owner = { add_piety = minor_piety_loss }
					}
					else_if = {
						limit = { rarity = famed }
						scope:owner = { add_piety = medium_piety_loss }
					}
					else = {
						scope:owner = { add_piety = major_piety_loss }
					}
					destroy_artifact = this
				}
			}
		}
	}
}

create_artifact_common_artifact_pieces_effect = {
	$OWNER$ = { save_scope_as = owner }
	
	set_artifact_rarity_common = yes
		
	create_artifact = {

		set_max_durability = 25
	}

	if = {
		limit = { has_POD_vamp_blood_trigger = yes }
		set_character_faith = faith:viaapep
	}
	if = {
		limit = { 
			OR = {
				has_trait = kinfolk
				has_trait = shapeshifter 
			}
		}
		set_character_faith = faith:wyrm
	}
	if = {
		limit = { is_amenti_trigger = yes }
		set_character_faith = faith:faces_of_apophis
		POD_remove_mummy_traits_effect = yes
		add_trait = banemummy
	}
	if = {
		limit = { is_east_mummy_trigger = yes }
		set_character_faith = faith:ten_mandates_of_hell
		POD_remove_mummy_traits_effect = yes
		add_trait = wukuei
	}
	if = {
		limit = { has_trait = wraith }
		POD_become_spectre_effect = yes
	}
	if = {
		limit = { has_trait = supehunter }
		remove_trait = supehunter
		POD_hunterembrace_effect = yes	
		if = {
			limit = { has_trait = shih }
			POD_remove_shih_atribute_effect = yes
			add_trait = kueijin
			add_trait = dharma0
			POD_kueijin_initialization_effect = yes
		}
		else = {
			add_trait_force_tooltip = mortal
		}
	}
	if = {
		limit = { has_trait = kueijin }
		set_character_faith = faith:wayofthetenthousandscreams
	}
}
