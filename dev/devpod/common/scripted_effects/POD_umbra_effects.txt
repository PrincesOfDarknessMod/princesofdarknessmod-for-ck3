POD_set_local_spirit_list_effect = {
	initiate_grace_effect = yes
	remove_variable = umbra_ruler
	remove_variable = local_umbra_explorer
	switch = {
		trigger = var:spirit_location
		flag:dark_kingdom_of_iron = { set_variable = { name = umbra_ruler value = character:POD_wraith_01 } }
		flag:dark_kingdom_of_jade = { set_variable = { name = umbra_ruler value = character:POD_wraith_03 } }
	}
	if = {
		limit = { 
			has_variable = umbra_stationed_location 
			var:umbra_stationed_location = var:spirit_location
		} 
		set_variable = { name = local_umbra_explorer value = this }
	}
	else = {
		random_relation = {
			type = umbra_explorer
			limit = { var:umbra_stationed_location = root.var:spirit_location }
			root = { set_variable = { name = local_umbra_explorer value = prev } }
		}
	}
}

POD_umbra_choose_effct = {
	set_variable = { name = spirit_destination value = flag:$LOCATION$ }
	trigger_event = { on_action = spirit_travel_movement_start_pulse }
	
	custom_description_no_bullet = { text = $LOCATION$_desc }
	if = {
		limit = { var:spirit_location = flag:$LOCATION$ }
		custom_description_no_bullet = { text = UMBRA_HEAD }
	}
	else = {
		custom_tooltip = line_break
		show_as_tooltip = { POD_change_umbra_modifier_effect = { START = var:spirit_location END = flag:$LOCATION$ } }
	}
	custom_tooltip = $LOCATION$_0_tt
	if = {
		limit = { is_target_in_variable_list = { name = umbra_level_5_list target = flag:$LOCATION$ }  }
		custom_tooltip = $LOCATION$_1_tt
	}
	if = {
		limit = { is_target_in_variable_list = { name = umbra_level_5_list target = flag:$LOCATION$ }  }
		custom_tooltip = $LOCATION$_2_tt
	}
	if = {
		limit = { is_target_in_variable_list = { name = umbra_level_5_list target = flag:$LOCATION$ }  }
		custom_tooltip = $LOCATION$_3_tt
	}
	if = {
		limit = { is_target_in_variable_list = { name = umbra_level_5_list target = flag:$LOCATION$ }  }
		custom_tooltip = $LOCATION$_4_tt
	}
	if = {
		limit = { is_target_in_variable_list = { name = umbra_level_5_list target = flag:$LOCATION$ }  }
		custom_tooltip = $LOCATION$_5_tt
	}
}


# Exploration
POD_change_umbra_progress_effect = {
	$VICTIM$ = { change_variable = { name = umbra_exploration_progress add = umbra_aptitude } }
	if = {
		limit = { 
			$VICTIM$ = { var:umbra_exploration_progress >= umbra_stationed_location_size }
			NOT = { is_target_in_variable_list = { name = umbra_level_5_list target = $VICTIM$.var:umbra_stationed_location } }
		}
		$VICTIM$ = { set_variable = { name = umbra_exploration_progress value = 0 } }
		if = {
			limit = { is_target_in_variable_list = { name = umbra_level_4_list target = $VICTIM$.var:umbra_stationed_location } }
			send_interface_toast = {
				title = umbra_level_upgrade_tt
				left_icon = root
				right_icon = $VICTIM$
				
				POD_gain_umbra_level_5_effect = { LOCATION = $VICTIM$.var:umbra_stationed_location }
			}
		}
		else_if = {
			limit = { is_target_in_variable_list = { name = umbra_level_3_list target = $VICTIM$.var:umbra_stationed_location } }
			send_interface_toast = {
				title = umbra_level_upgrade_tt
				left_icon = root
				right_icon = $VICTIM$
				
				POD_gain_umbra_level_4_effect = { LOCATION = $VICTIM$.var:umbra_stationed_location }
			}
		}
		else_if = {
			limit = { is_target_in_variable_list = { name = umbra_level_2_list target = $VICTIM$.var:umbra_stationed_location } }
			send_interface_toast = {
				title = umbra_level_upgrade_tt
				left_icon = root
				right_icon = $VICTIM$
				
				POD_gain_umbra_level_3_effect = { LOCATION = $VICTIM$.var:umbra_stationed_location }
			}
		}
		else_if = {
			limit = { is_target_in_variable_list = { name = umbra_level_1_list target = $VICTIM$.var:umbra_stationed_location } }
			send_interface_toast = {
				title = umbra_level_upgrade_tt
				left_icon = root
				right_icon = $VICTIM$
				
				POD_gain_umbra_level_2_effect = { LOCATION = $VICTIM$.var:umbra_stationed_location }
			}
		}
		else = {
			send_interface_toast = {
				title = umbra_level_upgrade_tt
				left_icon = root
				right_icon = $VICTIM$
				
				POD_gain_umbra_level_1_effect = { LOCATION = $VICTIM$.var:umbra_stationed_location }
			}
		}
	}
}

POD_remove_umbra_explorer_effect = {
	if = {
		limit = { this = $TARGET$ } 
		remove_variable = umbra_stationed_location
		remove_variable = umbra_exploration_progress
	}
	else = {
		remove_relation_umbra_explorer = $TARGET$
		$TARGET$ = { 
			remove_variable = umbra_stationed_location 
			remove_variable = umbra_exploration_progress
		}
	}
}

# Reweave Artifact
create_artifact_reweaved_artifact_effect = { 
	$OWNER$ = { save_scope_as = owner }
			
	$CREATOR$ = { save_scope_as = creator }

	scope:owner = {
		if = {
			limit = { scope:weapon = yes }
			create_artifact_weapon_effect = {
				OWNER = scope:owner
				CREATOR = scope:creator
				SET_WEAPON_TYPE = flag:no
			}
		}
		else_if = {
			limit = { scope:armor = yes }
			create_artifact_armor_effect = {
				OWNER = scope:owner
				CREATOR = scope:creator
				SET_ARMOR_TYPE = flag:no
			}
		}
		else_if = {
			limit = { scope:shinies = yes }
			random_list = {
				10 = {
					create_artifact_regalia_effect = {
						OWNER = scope:owner
						SMITH = scope:owner
					}
				}
				10 = {
					create_artifact_crown_effect = {
						OWNER = scope:owner
						SMITH = scope:owner
					}
				}
				10 = {
					create_artifact_brooch_effect = {
						OWNER = scope:owner
						SMITH = scope:owner
					}
				}
			}
		}
		else_if = {
			limit = { scope:book = yes }
			create_artifact_book_effect = {
				OWNER = scope:owner
				CREATOR = scope:creator
				SET_SUBJECT = flag:no
				SET_TOPIC = flag:no
			}
		}
		else_if = {
			limit = { scope:furniture = yes }
			random_list = {
				10 = {
					create_artifact_throne_effect = {
						OWNER = scope:owner
						MAKER = scope:owner
					}
				}
				10 = {
					create_artifact_tapestry_effect = {
						OWNER = scope:owner
						WEAVER = scope:owner
					}
				}
				10 = {
					create_artifact_box_small_effect = {
						OWNER = scope:owner
						CARPENTER = scope:owner
					}
				}
				10 = {
					create_artifact_box_medium_effect = {
						OWNER = scope:owner
						CARPENTER = scope:owner
					}
				}
				10 = {
					create_artifact_goblet_effect = {
						OWNER = scope:owner
						SMITH = scope:owner
					}
				}
				10 = {
					create_artifact_wall_banner_effect = {
						OWNER = scope:owner
						CREATOR = scope:owner
						TARGET = scope:owner.primary_title
					}
				}
				10 = {
					create_artifact_wall_skull_effect = {
						OWNER = scope:owner
						HUNTER = scope:owner
						LEGENDARY = no
						ANIMAL = flag:none
					}
				}
				10 = {
					create_artifact_sculpture_effect = {
						OWNER = scope:owner
						SCULPTER = scope:owner
					}
				}
				10 = {
					create_artifact_wall_shield_effect = {
						OWNER = scope:owner
						TARGET = scope:owner.house
						CREATOR = scope:owner
					}
				}
			}
		}
		else_if = {
			limit = { scope:trinket = yes }
			create_artifact = {
				name = cuckoldry_psalter_name
				description = cuckoldry_psalter_description
				type = miscellaneous
				visuals = ring
				modifier = artifact_placeholder_modifier
			}
		}
		
		hidden_effect = {
			scope:newly_created_artifact = { clear_artifact_modifiers = yes }
			if = {
				limit = { scope:owner = scope:creator }
				every_character_artifact = {
					limit = {
						has_variable = artifact_pieces
					}
					if = {
						limit = { 
							rarity = illustrious 
							scope:owner = { has_perk = princely_restoration_perk }
						}
						scope:owner = { add_piety = major_piety_loss }
						scope:newly_created_artifact = { copy_artifact_modifiers = prev }
						destroy_artifact = this
					}
					else_if = {
						limit = { 
							rarity = famed 
							scope:owner = { has_perk = greater_restoration_perk }
						}
						scope:owner = { add_piety = medium_piety_loss }
						scope:newly_created_artifact = { copy_artifact_modifiers = prev }
						destroy_artifact = this
					}
					else_if = {
						limit = { 
							rarity = masterwork 
							scope:owner = { has_perk = major_restoration_perk }
						}
						scope:owner = { add_piety = minor_piety_loss }
						scope:newly_created_artifact = { copy_artifact_modifiers = prev }
						destroy_artifact = this
					}
					else = {
						scope:owner = { add_piety = miniscule_piety_loss }
						scope:newly_created_artifact = { copy_artifact_modifiers = prev }
						destroy_artifact = this
					}
				}
			}
			else = {
				every_equipped_character_artifact = {
					limit = {
						artifact_slot_type = miscellaneous
						has_variable = artifact_pieces
					}
					scope:newly_created_artifact = { copy_artifact_modifiers = prev }
					if = {
						limit = { rarity = common }
						scope:owner = { add_piety = miniscule_piety_loss }
					}
					else_if = {
						limit = { rarity = masterwork }
						scope:owner = { add_piety = minor_piety_loss }
					}
					else_if = {
						limit = { rarity = famed }
						scope:owner = { add_piety = medium_piety_loss }
					}
					else = {
						scope:owner = { add_piety = major_piety_loss }
					}
					destroy_artifact = this
				}
			}
		}
	}
}

pay_for_spirit_travel_effect = {
	if = {
		limit = { has_trait = kueijin }
		if = {
			limit = {
				has_variable = spirit_destination
				var:spirit_destination = flag:shadowlands 
			}
			if = {
				limit = { NOT = { has_perk = semblance_of_the_ebon_dragon_perk } }
				send_interface_toast = {
					type = event_generic_bad
					title = paying_for_spirit_travel_toast
					left_icon = root
					custom_tooltip = paying_for_spirit_travel_notification_yin

					remove_short_term_gold = passing_gauntlet_cost
				}
			}
		}
		else = {
			if = {
				limit = { NOT = { has_perk = semblance_of_the_scalet_queen_perk } }
				send_interface_toast = {
					type = event_generic_bad
					title = paying_for_spirit_travel_toast
					left_icon = root
					custom_tooltip = paying_for_spirit_travel_notification_yang

					remove_short_term_gold = passing_gauntlet_cost
				}
			}
		}
	}
	else_if = {
		limit = { has_trait = shapeshifter }
		if = {
			limit = {
				OR = {
					NOT = { has_variable = spirit_location }
					NOT = { var:spirit_location = flag:custom_realm }
				}
			}
			send_interface_toast = {
				type = event_generic_neutral
				title = stepping_sideways_toast
				left_icon = root

				custom_tooltip = stepping_sideways_tooltip
				POD_gnosis_decrease_multiple_effect = { COUNT = gauntlet_level }
			}
		}
	}
	else = {
		send_interface_toast = {
			type = event_generic_neutral
			title = passing_gauntlet_toast
			left_icon = root
			 
			custom_tooltip = passing_gauntlet_tooltip
			remove_short_term_gold = passing_gauntlet_cost
		}
	}
}

POD_add_umbra_modifier_effect = {
	if = {
		limit = { in_astral_reaches_trigger = { LOCATION = $LOCATION$ } }
		add_character_modifier = traveling_astral_reaches_modifier
	}
	else_if = {
		limit = { in_spirit_wilds_trigger = { LOCATION = $LOCATION$ } }
		add_character_modifier = traveling_spirit_wilds_modifier
	}
	else_if = {
		limit = { in_underworld_trigger = { LOCATION = $LOCATION$ } }
		add_character_modifier = traveling_underworld_modifier
	}
	else_if = {
		limit = { in_dreaming_trigger = { LOCATION = $LOCATION$ } }
		add_character_modifier = traveling_dreaming_modifier
	}
	else_if = {
		limit = { in_deep_umbra_trigger = { LOCATION = $LOCATION$ } }
		add_character_modifier = traveling_deep_umbra_modifier
	}
}

POD_change_umbra_modifier_effect = {
	if = {
		limit = { in_astral_reaches_trigger = { LOCATION = $START$ } }
		if = {
			limit = { NOT = { in_astral_reaches_trigger = { LOCATION = $END$ } } }
			remove_character_modifier = traveling_astral_reaches_modifier
			POD_add_umbra_modifier_effect = { LOCATION = $END$ }
		}
	}
	else_if = {
		limit = { in_spirit_wilds_trigger = { LOCATION = $START$ } }
		if = {
			limit = { NOT = { in_spirit_wilds_trigger = { LOCATION = $END$ } } }
			remove_character_modifier = traveling_spirit_wilds_modifier
			POD_add_umbra_modifier_effect = { LOCATION = $END$ }
		}
	}
	else_if = {
		limit = { in_underworld_trigger = { LOCATION = $START$ } }
		if = {
			limit = { NOT = { in_underworld_trigger = { LOCATION = $END$ } } }
			remove_character_modifier = traveling_underworld_modifier
			POD_add_umbra_modifier_effect = { LOCATION = $END$ }
		}
	}
	else_if = {
		limit = { in_dreaming_trigger = { LOCATION = $START$ } }
		if = {
			limit = { NOT = { in_dreaming_trigger = { LOCATION = $END$ } } }
			remove_character_modifier = traveling_dreaming_modifier
			POD_add_umbra_modifier_effect = { LOCATION = $END$ }
		}
	}
	else_if = {
		limit = { in_deep_umbra_trigger = { LOCATION = $START$ } }
		if = {
			limit = { NOT = { in_deep_umbra_trigger = { LOCATION = $END$ } } }
			remove_character_modifier = traveling_deep_umbra_modifier
			POD_add_umbra_modifier_effect = { LOCATION = $END$ }
		}
	}
	else_if = {
		limit = { in_periphery_trigger = { LOCATION = $START$ } }
		if = {
			limit = { NOT = { in_periphery_trigger = { LOCATION = $END$ } } }
			send_interface_message = {
				type = character_arrived
				title = spirit_travel_arrived_title
				desc = spirit_travel_arrived_desc
				right_icon = root
				
				pay_for_spirit_travel_effect = yes
				POD_add_umbra_modifier_effect = { LOCATION = $END$ }
				if = {
					limit = { 
						has_POD_shapeshifted_form_trigger = yes 
						NOT = { has_perk = aspect_of_duat_perk }
					}
					POD_shapeshift_revert_effect = yes
				}
			}
		}
	}
	else = {
		POD_add_umbra_modifier_effect = { LOCATION = $END$ }
	}
}
