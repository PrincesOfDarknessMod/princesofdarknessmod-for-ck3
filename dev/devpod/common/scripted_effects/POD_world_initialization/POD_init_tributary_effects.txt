POD_setup_tributaries_effect = {
	if = {
    	limit = { POD_bookmark_is_war_of_princes = yes } 
       	POD_vda_tributary_setup_effect = yes
    }
	else_if = {
    	limit = { POD_bookmark_is_battle_royale = yes } 
       	POD_wod_tributary_setup_effect = yes
    }
	else_if = {
    	limit = { POD_bookmark_is_dark_ages_fae = yes } 
       	POD_fae_tributary_setup_effect = yes
    }		
}

POD_setup_tributary = {
	$TRIBUTARY$ = {
		break_subject_contract_and_establish_tributary_effect = {
			SUZERAIN = $SUZERAIN$
			TRIBUTARY = $TRIBUTARY$
		}
	}
}
