POD_start_canon_diarchy = {
	if = {
		limit = {
			$LIEGE$ ?= { is_landed = yes }
			$DIARCH$ ?= { is_alive = yes }
		}
		$DIARCH$ = {
			add_character_flag = has_had_diarch_intro_event_flag
		}
		$LIEGE$ = {
			designate_diarch = $DIARCH$
			try_start_diarchy = pod_power_struggle
			set_diarchy_swing = $SWING$
		}
	}
}
