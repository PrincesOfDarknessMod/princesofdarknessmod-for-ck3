POD_change_umbra_progress_effect = {
	$VICTIM$ = { change_variable = { name = umbra_exploration_progress add = umbra_aptitude } }
	if = {
		limit = { 
			$VICTIM$ = { var:umbra_exploration_progress >= umbra_stationed_location_size }
		}
		$VICTIM$ = { set_variable = { name = umbra_exploration_progress value = 0 } }
		$VICTIM$ = { POD_umbra_set_expedition_discovery_effect = yes }
	}
}

POD_gain_umbra_progress_effect = {
	$VICTIM$ = { change_variable = { name = umbra_exploration_progress add = $VALUE$ } }
	if = {
		limit = { 
			$VICTIM$ = { var:umbra_exploration_progress >= umbra_stationed_location_size }
		}
		$VICTIM$ = { set_variable = { name = umbra_exploration_progress value = 0 } }
		$VICTIM$ = { POD_umbra_set_expedition_discovery_effect = yes }
	}
}

POD_lose_umbra_progress_effect = {
	var:local_umbra_explorer = { change_variable = { name = umbra_exploration_progress subtract = $VALUE$ } }
	if = {
		limit = { var:local_umbra_explorer = { var:umbra_exploration_progress < 0 } }
		var:local_umbra_explorer = { set_variable = { name = umbra_exploration_progress value = 0 } }
	}
}

POD_inherit_umbra_level_effect = {
	save_temporary_scope_as = heir
	$TARGET$ = {
		every_owned_story = {
			limit = { story_type = POD_story_umbra_gui }
			make_story_owner = scope:heir
			scope:heir = {
				add_to_variable_list = {
					name = known_umbra_realms_list
					target = prev
				}
			}
			if = {
				limit = { has_variable = explorer }
				var:explorer = { save_scope_as = this_explorer }
				$TARGET$ = { POD_dismiss_umbra_explorer_effect = { TARGET = scope:this_explorer } }
			}
		}
	}
}
