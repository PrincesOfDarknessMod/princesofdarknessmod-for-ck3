pay_for_spirit_travel_effect = {
	if = {
		limit = {
			in_underworld_trigger = { LOCATION = $END$ }
		}
		if = {
			limit = { has_trait = kueijin }
			if = {
				limit = { NOT = { has_perk = semblance_of_the_ebon_dragon_perk } }
				send_interface_message = {
					type = POD_pay_for_spirit_travel
					title = paying_for_spirit_travel_toast
					left_icon = root
					desc = paying_for_spirit_travel_notification_yin

					remove_short_term_gold = passing_shroud_cost
				}
			}
		}
		else_if = {
			limit = { has_trait = fera }
			if = {
				limit = { passing_shroud_gnosis_cost > 0 }
				send_interface_message = {
					type = POD_pay_for_spirit_travel
					title = stepping_sideways_toast
					left_icon = root

					desc = stepping_sideways_tooltip
					POD_gnosis_decrease_multiple_effect = { COUNT = passing_shroud_gnosis_cost }
				}
			}
		}
		else = {
			send_interface_message = {
				type = POD_pay_for_spirit_travel
				title = passing_shroud_toast
				left_icon = root
				 
				desc = passing_shroud_tooltip
				remove_short_term_gold = passing_shroud_cost
			}
		}
	}
	else = {
		if = {
			limit = { has_perk = numina_ephemera_6_perk }
		}
		else_if = {
			limit = { has_trait = kueijin }
			if = {
				limit = { NOT = { has_perk = semblance_of_the_scalet_queen_perk } }
				send_interface_message = {
					type = POD_pay_for_spirit_travel
					title = paying_for_spirit_travel_toast
					left_icon = root
					desc = paying_for_spirit_travel_notification_yang

					remove_short_term_gold = passing_gauntlet_cost
				}
			}
		}
		else_if = {
			limit = { has_trait = fera }
			if = {
				limit = { passing_gauntlet_gnosis_cost > 0 }
				send_interface_message = {
					type = POD_pay_for_spirit_travel
					title = stepping_sideways_toast
					left_icon = root

					desc = stepping_sideways_tooltip
					POD_gnosis_decrease_multiple_effect = { COUNT = passing_gauntlet_gnosis_cost }
				}
			}
		}
		else = {
			send_interface_message = {
				type = POD_pay_for_spirit_travel
				title = passing_gauntlet_toast
				left_icon = root
				 
				desc = passing_gauntlet_tooltip
				remove_short_term_gold = passing_gauntlet_cost
			}
