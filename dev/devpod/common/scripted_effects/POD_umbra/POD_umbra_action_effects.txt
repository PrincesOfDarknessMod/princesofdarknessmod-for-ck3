umbra_train_effect = {
	if = {
		limit = { $LEVEL$ = 7 }
		POD_umbra_fallback_extremely_high_effect = { SKILL = $SKILL$ }
	}
	else_if = {
		limit = { $LEVEL$ = 6 }
		POD_umbra_fallback_very_high_effect = { SKILL = $SKILL$ }
	}
	else_if = {
		limit = { $LEVEL$ = 5 }
		POD_umbra_fallback_high_effect = { SKILL = $SKILL$ }
	}
	else_if = {
		limit = { $LEVEL$ = 4 }
		POD_umbra_fallback_decent_effect = { SKILL = $SKILL$ }
	}
	else_if = {
		limit = { $LEVEL$ = 3 }
		POD_umbra_fallback_medium_effect = { SKILL = $SKILL$ }
	}
	else_if = {
		limit = { $LEVEL$ = 2 }
		POD_umbra_fallback_mediocre_effect = { SKILL = $SKILL$ }
	}
	else = {
		POD_umbra_fallback_low_effect = { SKILL = $SKILL$ }
	}
}

POD_umbra_fallback_low_effect = {
	duel = {
		skill = $SKILL$
		value = low_skill_rating
	
		25 = {
			compare_modifier = {
				value = scope:duel_value
				multiplier = 0.5
			}
			add_$SKILL$_lifestyle_xp = 10
		}
		25 = {
			compare_modifier = {
				value = scope:duel_value
				multiplier = -0.5
			}
		}
	}
}
