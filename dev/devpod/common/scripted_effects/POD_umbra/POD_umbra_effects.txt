POD_create_umbra_realm_effect = {
	if = {
		limit = { 
			NOT = { POD_umbra_realm_known_trigger = { NAME = flag:$REALM$ } }
		}
		custom_tooltip = {
			text = umbra_realm_$REALM$
			POD_create_umbra_location = { NAME = flag:$REALM$ }
		}
	}
}

POD_assign_umbra_explorer_effect = {
	if = {
		limit = { this = $TARGET$ }
		if = {
			limit = { has_variable = umbra_stationed_realm }
			POD_remove_umbra_explorer_effect = { TARGET = $TARGET$ }
		}
	}
	else_if = { 
		limit = { has_relation_umbra_explorer = $TARGET$ }
		POD_remove_umbra_explorer_effect = { TARGET = $TARGET$ }
	}
	else = {
		set_relation_umbra_explorer = $TARGET$
	}
	$TARGET$ = {
		set_variable = { name = umbra_exploration_progress value = $VALUE$ }
		set_variable = { name = umbra_stationed_realm value = root.var:local_umbra_realm }
	}
	root.var:local_umbra_realm = { set_variable = { name = explorer value = $TARGET$ } }
	set_variable = { name = local_umbra_explorer value = $TARGET$ }
}

POD_swap_umbra_explorer_effect = {
	var:local_umbra_explorer = {
		save_scope_value_as = {
			name = saved_progress
			value = var:umbra_exploration_progress
		}
	}
	if = { 
		limit = { $TARGET$ = { has_variable = umbra_stationed_realm } }
		$TARGET$.var:umbra_stationed_realm = { set_variable = { name = explorer value = root.var:local_umbra_explorer } }
		var:local_umbra_explorer = {
			set_variable = { name = umbra_exploration_progress value = $TARGET$.var:umbra_exploration_progress }
			set_variable = { name = umbra_stationed_realm value = $TARGET$.var:umbra_stationed_realm }
		}
	}
	else = {
		POD_remove_umbra_explorer_effect = { TARGET = var:local_umbra_explorer }
	}
	POD_assign_umbra_explorer_effect = { TARGET = $TARGET$ VALUE = scope:saved_progress }
}

POD_remove_umbra_explorer_effect = {
	if = {
		limit = { NOT = { this = $TARGET$ } }
		remove_relation_umbra_explorer = $TARGET$
	}
	$TARGET$ = {
		remove_variable = umbra_exploration_progress
		var:umbra_stationed_realm = { remove_variable = explorer }
		remove_variable = umbra_stationed_realm
	}
}

POD_dismiss_umbra_explorer_effect = {
	$TARGET$ = { POD_umbra_remove_discovery_effect = yes }
	POD_remove_umbra_explorer_effect = { TARGET = $TARGET$ }
}
