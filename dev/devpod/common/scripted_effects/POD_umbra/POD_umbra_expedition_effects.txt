POD_umbra_expedition_return_effect = {
	if = {
		limit = { has_variable = umbra_origin_location }
		if = {
			limit = { 
				NOT = { var:spirit_location = var:umbra_origin_location } 
				NOT = { in_shadow_realm_trigger = { LOCATION = var:spirit_location } }
			}
			set_variable = { name = spirit_destination value = var:umbra_origin_location }
			show_as_tooltip = { POD_umbra_travel_effect = { LOCATION = var:umbra_origin_location } }
			if = {
				limit = { POD_umbra_bypass_trigger = { LOCATION = var:umbra_origin_location } }
				trigger_event = { on_action = spirit_travel_movement_end_pulse }
			}
			else = {
				trigger_event = { on_action = spirit_travel_movement_start_pulse }
			}
		}
		remove_variable = umbra_origin_location
	}
	else = {
		POD_remove_umbra_modifier_effect = yes
		remove_variable = spirit_location
	}
}

POD_umbra_expedition_cooldown_effect = {
	if = {
		limit = { NOT = { any_memory = { has_memory_type = umbra_expedition_memory } } }
		custom_tooltip = line_break
		if = {
			limit = { NOT = { var:spirit_location ?= flag:earth } }
			custom_description_no_bullet = {
				text = umbra_cooldown_desc
			}
		}
		create_character_memory = {
			type = umbra_expedition_memory
		}
	}
}

POD_umbra_artifact_init_effect = {
	if = {
		limit = {
			OR = {
				NOT = { exists = scope:artifact_flag_1 } 
				scope:artifact_flag_1 = flag:no
			}
		}
		save_scope_value_as = {
			name = artifact_flag_1
			value = flag:$ARTIFACT$
		}
	}
	else_if = {
		limit = {
			OR = {
				NOT = { exists = scope:artifact_flag_2 } 
				scope:artifact_flag_2 = flag:no
			}
		}
		save_scope_value_as = {
			name = artifact_flag_2
			value = flag:$ARTIFACT$
		}
	}
	else = {
		save_scope_value_as = {
			name = artifact_flag_3
			value = flag:$ARTIFACT$
		}
	}
}

# Startup
POD_umbra_expedition_on_start = {
	if = {
		limit = {
			NOT = {
				exists = var:umbra_map
			}
		}
		POD_create_umbra_expedition = yes
	}
}

POD_umbra_expedition_cleanup_effect = {
	hidden_effect = {
		if = {
			limit = { has_variable = umbra_map }
			var:umbra_map ?= {
				end_story = yes
			}
			remove_variable = umbra_map
		}
	}
}
