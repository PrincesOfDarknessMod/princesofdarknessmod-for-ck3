# POD_umbra_adventure_get_neighbors
#	NODE = the current node
#	MAP = the node map
#	SAVE_AS = name of the saved local variable list
POD_umbra_adventure_get_neighbors = {
	if = {
		limit = {
			has_local_variable_list = $SAVE_AS$
		}
		clear_local_variable_list = $SAVE_AS$
	}
	$NODE$ = {
		save_scope_as = current_node
	}
	save_scope_value_as = {
		name = row_idx
		value = {
			value = scope:current_node.var:row
			subtract = 1
			min = 0
		}
	}
	save_scope_value_as = {
		name = row_max
		value = {
			value = scope:current_node.var:row
			add = 2
		}
	}
	save_scope_value_as = {
		name = column_max
		value = {
			value = scope:current_node.var:column
			add = 2
		}
	}
	while = {
		limit = {
			scope:row_idx < scope:row_max
		}
		save_scope_value_as = {
			name = column_idx
			value = {
				value = scope:current_node.var:column
				subtract = 1
				min = 0
			}
		}
		while = {
			limit = {
				scope:column_idx < scope:column_max
			}
			POD_get_node_by_row_and_column = {
				MAP = $MAP$
				ROW = scope:row_idx
				COLUMN = scope:column_idx
				SAVE_AS = neighbor_node
			}
			if = {
				limit = {
					AND = {
						exists = scope:neighbor_node
						scope:neighbor_node != scope:current_node
					}
				}
				add_to_local_variable_list = {
					name = $SAVE_AS$
					target = scope:neighbor_node
				}
			}
			POD_increment_scope_value = {
				SCOPE = column_idx
			}
		}
		POD_increment_scope_value = {
			SCOPE = row_idx
		}
	}
}

# POD_umbra_adventure_generate_neighbor_lines
#	MAP = the node map
#	CHANCE = likelihood of generating a line
POD_umbra_adventure_generate_neighbor_lines = {
	$MAP$ = {
		every_in_list = {
			variable = nodes
			save_scope_as = current_node
			POD_umbra_adventure_get_neighbors = {
				NODE = scope:current_node
				MAP = $MAP$
				SAVE_AS = neighbors
			}
			every_in_local_list = {
				variable = neighbors
				# prevent dupes
				limit = { NOT = { POD_line_exists_between_nodes = { NODE1 = this NODE2 = scope:current_node} } }
				clear_saved_scope = neighbor
				random = {
					chance = $CHANCE$
					save_scope_as = neighbor
					POD_create_line = {
						MAP = scope:node_map
						START = scope:current_node
						END = scope:neighbor
						SAVE_AS = created_line
					}
				}
			}
		}
	}
}

# POD_umbra_adventure_generate_spanning_tree
#	MAP = the node map
POD_umbra_adventure_generate_spanning_tree = {
	$MAP$ = {
		ordered_in_list = {
			variable = nodes
			position = 0
			set_variable = {
				name = spanned
				value = yes
			}
			add_to_temporary_list = spanned_nodes
		}
		while = {
			limit = {
				any_in_list = {
					variable = nodes
					NOT = { var:spanned ?= yes }
				}
			 }
			clear_saved_scope = best_distance
			# Pathfinding
			# Note that this implementation is exponential due to difficulties in implementing a more efficient
			# aglorithm with the limited data structures available to us. This will be fine for small graphs,
			# but for extremely large graphs find a better approach.
			# Step 1: Find the optimal lowest weight edge to the spanned nodes
			every_in_list = {
				list = spanned_nodes

				save_scope_as = spanned_node
				$MAP$ = {
					ordered_in_list = {
						variable = nodes
						limit = {
							NOT = {
								var:spanned ?= yes
							}
						}
						# squared distance
						order_by = {
							value = 0
							add = {
								value = scope:spanned_node.var:coord_x
								subtract = var:coord_x
								multiply = {
									value = scope:spanned_node.var:coord_x
									subtract = var:coord_x
								}
							}
							add = {
								value = scope:spanned_node.var:coord_y
								subtract = var:coord_y
								multiply = {
									value = scope:spanned_node.var:coord_y
									subtract = var:coord_y
								}
							}
							# ascending
							multiply = -1
						}
						position = 0
						save_scope_as = candidate_node
					}
				}
				# Step 2: Recalculate squared distance for spanned_node -> candidate
				POD_node_squared_distance = { NODE1 = scope:spanned_node NODE2 = scope:candidate_node SAVE_AS = spanned_node_to_candidate_distance }
				# Step 3: Pick this edge as a candidate if it is an improvement on the current best distance
				if = {
					limit = { 
						OR = {
							NOT = { exists = scope:best_distance }
							scope:spanned_node_to_candidate_distance < scope:best_distance
						}
					}
					save_scope_value_as = {
						name = best_distance
						value = scope:spanned_node_to_candidate_distance
					}
					scope:spanned_node = {
						save_scope_as = start_node
					}
					scope:candidate_node = {
						save_scope_as = end_node
					}
				}
			}
			POD_create_line = {
				MAP = $MAP$
				START = scope:start_node
				END = scope:end_node
				SAVE_AS = created_line
			}
			scope:end_node = {
				set_variable = {
					name = spanned
					value = yes
				}
				add_to_temporary_list = spanned_nodes
			}
		}
	}
}

# POD_umbra_adventure_generate_grid_lines
#	WIDTH = width of the entire map
#	HEIGHT = height of the entire map
#	COLUMNS = number of columns to subdivide the map
#	ROWS = number of rows to subdivide the map
#	MAP = the node map
POD_umbra_adventure_generate_grid_lines = {
	save_scope_value_as = {
		name = column_idx
		value = 0
	}
	save_scope_value_as = {
		name = row_idx
		value = 0
	}
	save_scope_value_as = {
		name = quadrant_size_x
		value = {
			value = $WIDTH$
			divide = $COLUMNS$
		}
	}
	save_scope_value_as = {
		name = quadrant_size_y
		value = {
			value = $HEIGHT$
			divide = $ROWS$
		}
	}
	$MAP$ = {
		while = {
			limit = {
				scope:column_idx <= $COLUMNS$
			}
			save_scope_value_as = {
				name = column_x
				value = {
					value = scope:column_idx
					multiply = scope:quadrant_size_x
				}
			}
			save_scope_value_as = {
				name = end_y
				value = {
					value = $ROWS$
					multiply = scope:quadrant_size_y
				}
			}
			POD_create_grid_line = {
				MAP = $MAP$
				START_X = scope:column_x
				START_Y = 0
				END_X = scope:column_x
				END_Y = scope:end_y
				SAVE_AS = grid_line
			}
			POD_increment_scope_value = {
				SCOPE = column_idx
			}
		}
		while = {
			limit = {
				scope:row_idx <= $ROWS$
			}
			save_scope_value_as = {
				name = row_y
				value = {
					value = scope:row_idx
					multiply = scope:quadrant_size_y
				}
			}
			save_scope_value_as = {
				name = end_x
				value = {
					value = $COLUMNS$
					multiply = scope:quadrant_size_x
				}
			}
			POD_create_grid_line = {
				MAP = $MAP$
				START_X = 0
				START_Y = scope:row_y
				END_X = scope:end_x
				END_Y = scope:row_y
				SAVE_AS = grid_line
			}
			POD_increment_scope_value = {
				SCOPE = row_idx
			}
		}
	}
}

# Uses FTLs map generation algorithm
# Algorithm:
#	The map is broken into quandrants of equal size (by the given row and column size)
# 	Each quadrant has a 50% chance at a node, where x = random(0,quadrant_size_x-padding*2) + padding and y = random(0,quadrant_size_y-padding*2)
#	Run pathfinding to generate a minimum spanning tree to ensure no node is orphaned (guarantees a single path to visit all nodes)
#	Each quadrant with a node gets a line between itself and any adjacent quadrant with a node (ignoring duplicates)
# POD_umbra_adventure_generate_map
#	WIDTH = width of the entire map
#	HEIGHT = height of the entire map
#	COLUMNS = number of columns to subdivide the map
#	ROWS = number of rows to subdivide the map
#	PADDING = internal padding of the node (to prevent it from being right on the edges of the quadrant)
#	NODE_CHANCE = likelihood of generating a node in each quadrant
#	NEIGHBOR_CHANCE = likelihood of creating a line between nodes in neighboring quadrants
#	SAVE_AS = saved scope of the created map
POD_umbra_adventure_generate_map = {
	POD_create_node_map = {
		SAVE_AS = $SAVE_AS$
	}
	scope:$SAVE_AS$ = {
		save_scope_as = node_map
		set_variable = {
			name = quadrant_size_x
			value = {
				value = $WIDTH$
				divide = $COLUMNS$
			}
		}
		set_variable = {
			name = quadrant_size_y
			value = {
				value = $HEIGHT$
				divide = $ROWS$
			}
		}
	}
	save_scope_value_as = {
		name = row_idx
		value = 0
	}
	while = {
		limit = {
			scope:row_idx < $ROWS$
		}
		save_scope_value_as = {
			name = column_idx
			value = 0
		}
		while = {
			limit = {
				scope:column_idx < $COLUMNS$
			}
			random = {
				chance = $NODE_CHANCE$
				POD_create_random_node_in_quadrant_with_padding = {
					PADDING = $PADDING$
					COLUMN = scope:column_idx
					ROW = scope:row_idx
					MAP = scope:node_map
				}
			}
			POD_increment_scope_value = {
				SCOPE = column_idx
			}
		}
		POD_increment_scope_value = {
			SCOPE = row_idx
		}
	}
	POD_umbra_adventure_generate_spanning_tree = {
		MAP = scope:node_map
	}
	POD_umbra_adventure_generate_neighbor_lines = {
		MAP = scope:node_map
		CHANCE = $NEIGHBOR_CHANCE$
	}
	POD_umbra_adventure_generate_grid_lines = {
		WIDTH = $WIDTH$
		HEIGHT = $HEIGHT$
		COLUMNS = $COLUMNS$
		ROWS = $ROWS$
		MAP = scope:node_map
	}
}

POD_umbra_adventure_add_stackable_item = {
	ordered_in_list = {
		variable = inventory
		position = 0
		limit = {
			var:type ?= $ITEM$.var:type
		}
		save_scope_as = stackable_item
	}
	if = {
		limit = {
			exists = scope:stackable_item
		}
		scope:stackable_item = {
			set_variable = {
				name = quantity
				value = {
					value = scope:stackable_item.var:quantity
					add = $ITEM$.var:quantity
				}
			}
		}
		$ITEM$ = {
			end_story = yes
		}
	}
	else = {
		add_to_variable_list = {
			name = inventory
			target = $ITEM$
		}
	}
}

POD_umbra_adventure_add_to_inventory = {
	var:umbra_adventure = {
		save_scope_as = umbra_adventure
	}
	$ITEM$ = {
		save_scope_as = inventory_item
	}
	if = {
		limit = {
			scope:umbra_adventure = {
				any_in_list = {
					variable = inventory
					count < 8
				}
			}
		}
		# handle stackables
		if = {
			limit = {
				scope:inventory_item = {
					has_variable = quantity
				}
			}
			scope:umbra_adventure = {
				POD_umbra_adventure_add_stackable_item = {
					ITEM = scope:inventory_item
				}
			}
		}
		else = {
			scope:umbra_adventure = {
				add_to_variable_list = {
					name = inventory
					target = scope:inventory_item
				}
			}
		}
	}
	else = {
		# cleanup the inventory item
		scope:inventory_item = {
			end_story = yes
		}
	}
}
