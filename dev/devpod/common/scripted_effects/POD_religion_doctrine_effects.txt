POD_remove_enlightenment_virtues_doctrine_effect = {
	if = {
		limit = { has_doctrine = doctrine_enlightenment_virtues_conscience_and_selfcontrol }
		remove_doctrine = doctrine_enlightenment_virtues_conscience_and_selfcontrol 
	}
	if = {
		limit = { has_doctrine = doctrine_enlightenment_virtues_conviction_and_selfcontrol }
		remove_doctrine = doctrine_enlightenment_virtues_conviction_and_selfcontrol 
	}
	if = {
		limit = { has_doctrine = doctrine_enlightenment_virtues_conviction_and_instinct }
		remove_doctrine = doctrine_enlightenment_virtues_conviction_and_instinct 
	}
	if = {
		limit = { has_doctrine = doctrine_enlightenment_virtues_conscience_and_instinct }
		remove_doctrine = doctrine_enlightenment_virtues_conscience_and_instinct 
	}
}

	if = {
		limit = { has_variable = doctrine_enlightenment_virtues }

POD_remove_crime_doctrine_effect = {
	if = {
		limit = { has_doctrine = $DOCTRINE$_crime }
		remove_doctrine = $DOCTRINE$_crime 
	}
	if = {
		limit = { has_doctrine = $DOCTRINE$_shunned }
		remove_doctrine = $DOCTRINE$_shunned 
	}
	if = {
		limit = { has_doctrine = $DOCTRINE$_accepted }
		remove_doctrine = $DOCTRINE$_accepted 
	}
}

POD_change_crime_doctrine_effect = {
	if = {
		limit = { has_variable = $DOCTRINE$ }
		POD_remove_crime_doctrine_effect = { DOCTRINE = $DOCTRINE$ }
		if = {
			limit = { var:$DOCTRINE$ = flag:$DOCTRINE$_crime }
			add_doctrine = $DOCTRINE$_crime
		}
		if = {
			limit = { var:$DOCTRINE$ = flag:$DOCTRINE$_shunned }
			add_doctrine = $DOCTRINE$_shunned
		}
		if = {
			limit = { var:$DOCTRINE$ = flag:$DOCTRINE$_accepted }
			add_doctrine = $DOCTRINE$_accepted
		}
		remove_variable = $DOCTRINE$
	}
}

POD_make_crime_doctrine_accepted_effect = {
	if = {
		limit = { has_doctrine = $DOCTRINE$_evil }
		remove_doctrine = $DOCTRINE$_crime 
	}
	else_if = {
		limit = { has_doctrine = $DOCTRINE$_crime }
		remove_doctrine = $DOCTRINE$_crime 
	}
	else_if = {
		limit = { has_doctrine = $DOCTRINE$_shunned }
		remove_doctrine = $DOCTRINE$_shunned 
	}
	if = {
		limit = { NOT = { has_doctrine = $DOCTRINE$_accepted } }
		add_doctrine = $DOCTRINE$_accepted 
	}
}

POD_make_crime_doctrine_accepted_noevil_effect = {

	}
	if = {

		add_doctrine = $DOCTRINE$_accepted 
	}
}
