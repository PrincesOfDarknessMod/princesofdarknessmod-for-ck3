# currently valid resources: gold, prestige, piety
POD_heist_track_resource_delta = {
	# save as scope value first so that script_values can be used regardless of the scope change
	if = {
		limit = { NOT = { exists = scope:$RESOURCE$_delta } }
		save_scope_value_as = {
			name  = $RESOURCE$_delta
			value = $AMOUNT$
		}
	}
	var:current_heist ?= {
		change_variable = {
			name = $RESOURCE$_delta
			add  = scope:$RESOURCE$_delta
		}
	}
}
POD_heist_add_and_track_resource = {
	add_$RESOURCE$ = $AMOUNT$
	POD_heist_track_$RESOURCE$_delta = { AMOUNT = $AMOUNT$ }
}

POD_heist_track_gold_delta     = { POD_heist_track_resource_delta = { RESOURCE = gold     AMOUNT = $AMOUNT$ } }
POD_heist_track_prestige_delta = { POD_heist_track_resource_delta = { RESOURCE = prestige AMOUNT = $AMOUNT$ } }
POD_heist_track_piety_delta    = { POD_heist_track_resource_delta = { RESOURCE = piety    AMOUNT = $AMOUNT$ } }

POD_heist_add_and_track_gold = {
	if = {
		limit = { $AMOUNT$ > 0 }
		add_short_term_gold = $AMOUNT$
	}
	else = {
		remove_short_term_gold = { value = 0 subtract = $AMOUNT$ }
	}
	POD_heist_track_gold_delta = { AMOUNT = $AMOUNT$ }
}
POD_heist_steal_and_track_gold = {
	var:current_heist.var:gold_owner ?= {
		save_scope_value_as = {
			name  = gold_delta
			value = $AMOUNT$
		}
		pay_short_term_gold = {
			target = prev
			gold = $AMOUNT$
		}
	}
	POD_heist_track_gold_delta = { AMOUNT = $AMOUNT$ }
	var:current_heist ?= {
		change_variable = {
			name = stolen_gold
			add  = scope:gold_delta
		}
	}
}
POD_heist_add_and_track_prestige = {
	add_prestige = $AMOUNT$
	POD_heist_track_prestige_delta = { AMOUNT = $AMOUNT$ }
}
POD_heist_add_and_track_piety = {
	add_piety = $AMOUNT$
	POD_heist_track_piety_delta = { AMOUNT = $AMOUNT$ }
}

POD_heist_gain_wound_with_toast = {
	send_interface_toast = {
		title = POD_heist_wounded_toast
		left_icon = root
		increase_wounds_effect = { REASON = wounds }
	}
}

POD_heist_gain_stress_with_toast = {
	send_interface_toast = {
		title = POD_heist_stress_toast
		left_icon = root
		add_stress = heist_node_stress_gain
	}
}

POD_heist_pause_for_node_event = {
	var:current_heist ?= {
		if = {
			limit = { NOT = { has_variable = paused_for_node_event } }
			set_variable = paused_for_node_event
		}
	}
}
POD_heist_resume_after_node_event = {
	var:current_heist ?= {
		if = {
			limit = { has_variable = paused_for_node_event }
			remove_variable = paused_for_node_event
		}
	}
}

POD_heist_stat_duel_effect = {
	scope:$STAT$_duelist = {
		duel = {
			skill = $STAT$
			value = $VALUE$
			$WEIGHT$ = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = 0.5
				}
				desc = POD_heist_effect_node_$STAT$_success
			}
			$WEIGHT$ = {
				desc = POD_heist_effect_node_$STAT$_fail
				root = { POD_heist_raise_suspicion_effect = { VALUE = minor_suspicion_value } }
			}
			$WEIGHT$ = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = -0.5
				}
				desc = POD_heist_effect_node_$STAT$_critfail
				root = { POD_heist_raise_suspicion_effect = { VALUE = major_suspicion_value } }
				$ONFAIL$
			}
		}
		POD_clear_onetime_buffs = yes
	}
}

POD_heist_scramble_node_effect = {
	POD_heist_raise_suspicion_effect = { VALUE = minor_suspicion_value }
	custom_tooltip = POD_heist_effect_node_scramble
	hidden_effect = {
		save_scope_value_as = {
			name  = currently_scrambling
			value = yes
		}
		var:current_heist ?= {
			every_in_list = {
				variable = nodes
				limit = {
					NOR = {
						has_variable = potential_start
						has_variable = potential_exit
						has_variable = potential_objective
						has_variable = side_objective
						var:type = flag:node_empty
						var:type = flag:node_scramble
					}
				}
				POD_heist_clear_node_type = yes
				set_variable = {
					name  = explored
					value = no
				}
			}
			POD_heist_populate_obstacle_nodes = yes
		}
	}
}

POD_heist_map_node_effect = {
	if = {
		limit = {
			var:current_heist ?= {
				any_in_list = {
					variable = nodes
					var:explored = no
				}
			}
		}
		custom_tooltip = {
			text = POD_heist_effect_node_map
			
			POD_heist_raise_suspicion_effect = { VALUE = major_suspicion_value }
			var:current_heist = {
				every_in_list = {
					variable = nodes
					set_variable = {
						name  = explored
						value = yes
					}
				}
			}
		}
	}
	else = {
		custom_tooltip = NO_EFFECT
	}
}

POD_heist_fight_elite_node_effect = {
	custom_tooltip = {
		# The tooltip intentionally doesn't tell you who you're going to fight. Live dangerously
		text = POD_heist_effect_node_fight_elite

		var:current_heist.var:court_owner ?= {
			ordered_knight = {
				order_by = prowess
				save_scope_as = knight_to_duel
			}
		}

		if = {
			limit = { exists = scope:knight_to_duel }
			
			POD_heist_pause_for_node_event = yes

			# This is where I'd put group combat mechanics. IF I HAD ANY
			configure_start_single_combat_effect = {
				SC_INITIATOR = this
				SC_ATTACKER = this
				SC_DEFENDER = scope:knight_to_duel
				FATALITY = no
				FIXED = no
				LOCALE = corridor_night
				OUTPUT_EVENT = POD_heists.201
				INVALIDATION_EVENT = fp1_tbc.0102
			}
		}
		else = {
			send_interface_toast = {
				# Fallback. Shouldn't happen in normal gameplay
				title = POD_heist_fight_elite_error_toast
			}
			var:current_heist.var:current_location ?= {
				POD_heist_clear_node_type = yes
				POD_heist_set_node_type_prowess = yes
			}
		}
	}
}

POD_heist_add_key_to_loot = {
	if = {
		limit = {
			scope:story ?= {
				NOT = { has_variable = found_key }
			}
		}
		scope:story = { set_variable = found_key }

		var:current_heist = {
			save_scope_value_as = { # saving as scope value since variables don't work for future tooltipping
				name = keys_found
				value = {
					value = var:keys_found
					add = 1
				}
			}
			set_variable = {
				name  = keys_found
				value = scope:keys_found
			}
		}

		send_interface_toast = {
			title = POD_heist_gain_key_toast

			if = {
				limit = { scope:keys_found >= var:current_heist.var:keys_required }
				custom_description = { text = POD_heist_found_all_keys_desc }
			}
			else = {
				custom_description = {
					text  = POD_heist_found_key_desc
					value = scope:keys_found
				}
			}
		}
	}
	else = {
		custom_tooltip = POD_heist_node_key_empty
	}
}

POD_heist_add_artifact_to_loot = {
	if = {
		limit = { POD_heist_found_all_keys_trigger = no }
		custom_tooltip = POD_heist_vault_no_key
	}
	else_if = {
		limit = {
			var:current_heist ?= {
				NOT = {
					is_target_in_variable_list = {
						name   = loot_artifacts
						target = var:target_artifact
					}
				}
			}
		}
		
		POD_commit_theft_effect = yes

		POD_heist_raise_suspicion_effect = { VALUE = extreme_suspicion_value }

		if = {
			# Don't take the artifact if the Infiltration would fail upon entering the node
			limit = { scope:new_suspicion < suspicion_max }

			play_sound_effect = "event:/DLC/EP1/SFX/UI/Notifications/inspiration_project_complete"

			send_interface_toast = {
				title = POD_heist_gain_artifact_toast
				left_icon = var:current_heist.var:target_artifact

				custom_description = { text = POD_heist_gain_artifact_desc }
				custom_tooltip = POD_heist_gain_artifact_hint
			}
			hidden_effect = {
				var:current_heist = {
					add_to_variable_list = {
						name   = loot_artifacts
						target = var:target_artifact
					}
				}
			}
		}
	}
}
