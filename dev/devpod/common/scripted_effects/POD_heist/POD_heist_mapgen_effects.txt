# story_cycle scope (the heist)
POD_heist_populate_nodes = {
	POD_heist_populate_side_objective_nodes = yes
	POD_heist_populate_obstacle_nodes = yes
}

# story_cycle scope (the heist)
POD_heist_populate_side_objective_nodes = {
	if = {
		limit = { POD_heist_has_potential_node_type = { TYPE = map } }
		random_in_list = {
			variable = nodes
			limit = { POD_heist_is_node_uninitialized_side_objective = yes }
			POD_heist_set_node_type_map = yes
		}
	}

	if = {
		limit = {
			POD_heist_has_potential_node_type = { TYPE = prisoners }
			var:court_owner = { has_prisoners = yes }
		}
		random_in_list = {
			variable = nodes
			limit = { POD_heist_is_node_uninitialized_side_objective = yes }
			POD_heist_set_node_type_prisoners = yes
		}
	}

	every_in_list = {
		variable = nodes
		limit = { POD_heist_is_node_uninitialized_side_objective = yes }
		random_list = {
			1 = {
				trigger = { prev = { POD_heist_has_potential_node_type = { TYPE = money } } }
				POD_heist_set_node_type_money = yes
			}
			1 = {
				trigger = { prev = { POD_heist_has_potential_node_type = { TYPE = fame } } }
				POD_heist_set_node_type_fame = yes
			}
			1 = {
				trigger = { prev = { POD_heist_has_potential_node_type = { TYPE = destress } } }
				POD_heist_set_node_type_destress = yes
			}
			1 = {
				trigger = { prev = { POD_heist_has_potential_node_type = { TYPE = gossip } } }
				POD_heist_set_node_type_gossip = yes
			}
			1 = {
				trigger = {
					prev = { POD_heist_has_potential_node_type = { TYPE = fight_elite } }
					prev.var:court_owner = { number_of_knights >= 1 }
				}
				POD_heist_set_node_type_fight_elite = yes
			}
		}
	}
}

# story_cycle scope (the heist)
POD_heist_populate_obstacle_nodes = {
	every_in_list = {
		variable = nodes
		limit = { NOT = { has_variable = type } }
		random_list = {
			3 = {
				trigger = { prev = { POD_heist_has_potential_node_type = { TYPE = diplomacy } } }
				POD_heist_set_node_type_diplomacy = yes
			}
			3 = {
				trigger = { prev = { POD_heist_has_potential_node_type = { TYPE = martial } } }
				POD_heist_set_node_type_martial = yes
			}
			3 = {
				trigger = { prev = { POD_heist_has_potential_node_type = { TYPE = stewardship } } }
				POD_heist_set_node_type_stewardship = yes
			}
			3 = {
				trigger = { prev = { POD_heist_has_potential_node_type = { TYPE = intrigue } } }
				POD_heist_set_node_type_intrigue = yes
			}
			3 = {
				trigger = { prev = { POD_heist_has_potential_node_type = { TYPE = learning } } }
				POD_heist_set_node_type_learning = yes
			}
			3 = {
				trigger = { prev = { POD_heist_has_potential_node_type = { TYPE = prowess } } }
				POD_heist_set_node_type_prowess = yes
			}

			1 = {
				trigger = { prev = { POD_heist_has_potential_node_type = { TYPE = fight_elite } } }
				POD_heist_set_node_type_fight_elite = yes
			}
			
			1 = {
				trigger = { NOT = { exists = scope:currently_scrambling } }
				POD_heist_set_node_type_empty = yes
			}
			1 = {
				trigger = { NOT = { exists = scope:currently_scrambling } }
				POD_heist_set_node_type_scramble = yes
			}
			
			1 = {
				trigger = { always = no } # unused
				POD_heist_set_node_type_event = yes
			}
		}
	}
}

# story_cycle scope (the heist)
POD_heist_add_default_potential_obstacle_node_types = {
	POD_heist_add_to_potential_node_types = { TYPE = intrigue }
	POD_heist_add_to_potential_node_types = { TYPE = learning }
	POD_heist_add_to_potential_node_types = { TYPE = prowess }
	
	if = {
		limit = { POD_heist_is_hostile_trigger = { HEIST = this } }
		
		POD_heist_add_to_potential_node_types = { TYPE = martial }
		POD_heist_add_to_potential_node_types = { TYPE = fight_elite }
	}
	else = {
		POD_heist_add_to_potential_node_types = { TYPE = diplomacy }
		POD_heist_add_to_potential_node_types = { TYPE = stewardship }
	}
}

POD_heist_initialize_artifact_heist = {
	$ARTIFACT$ = { save_temporary_scope_as = target_artifact }

	if = {
		limit = { scope:target_artifact = { POD_heist_artifact_should_be_double_locked = yes } }

		save_scope_value_as = {
			name  = keys_required
			value = 2
		}
		custom_tooltip = POD_heist_interaction_two_keys
	}
	else = {
		save_scope_value_as = {
			name  = keys_required
			value = 1
		}
		custom_tooltip = POD_heist_interaction_one_key
	}

	hidden_effect_new_object = {
		$HEIST$ ?= {
			set_variable = {
				name  = target_artifact
				value = scope:target_artifact
			}
			set_variable = {
				name  = gold_owner
				value = var:court_owner
			}

			random_in_list = {
				variable = nodes
				limit = { POD_heist_is_node_potential_objective = yes }
				POD_heist_set_node_type_artifact = yes
			}

			set_variable = {
				name  = keys_required
				value = scope:keys_required
			}

			if = {
				limit = { scope:keys_required = 2 }

				while = {
					count = 2
					random_in_list = {
						variable = nodes
						limit = { POD_heist_is_node_uninitialized_side_objective = yes }
						POD_heist_set_node_type_key_half = yes
					}
				}
			}
			else = {
				random_in_list = {
					variable = nodes
					limit = { POD_heist_is_node_uninitialized_side_objective = yes }
					POD_heist_set_node_type_key = yes
				}
			}
			
			set_variable = {
				name  = keys_found
				value = 0
			}

			# Side exits after keys so they don't get overwritten
			every_in_list = {
				variable = nodes
				limit = { POD_heist_is_node_potential_exit = yes }
				POD_heist_set_node_type_exit = yes
			}
			
			POD_heist_add_default_potential_obstacle_node_types = yes
			
			POD_heist_add_to_potential_node_types = { TYPE = money }
			POD_heist_add_to_potential_node_types = { TYPE = gossip }

			if = {
				limit = { POD_heist_is_hostile_trigger = { HEIST = this } }
				
				POD_heist_add_to_potential_node_types = { TYPE = map }
				POD_heist_add_to_potential_node_types = { TYPE = prisoners }
				POD_heist_add_to_potential_node_types = { TYPE = fight_elite }
			}
			else = {
				POD_heist_add_to_potential_node_types = { TYPE = fame }
				POD_heist_add_to_potential_node_types = { TYPE = destress }
			}
			
			POD_heist_populate_nodes = yes
		}
	}
}
