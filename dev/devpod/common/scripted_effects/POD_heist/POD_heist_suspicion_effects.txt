POD_heist_setup_suspicion_effect = {
	custom_tooltip = POD_heist_setup_suspicion_tt
	$HEIST$ ?= {
		set_variable = {
			name  = suspicion
			value = prev.starting_suspicion
		}
		POD_save_event_as_var = {
			VAR = suspicion_event_1
			ID  = $EVENT1$
		}
		POD_save_event_as_var = {
			VAR = suspicion_event_2
			ID  = $EVENT2$
		}
		POD_save_event_as_var = {
			VAR = suspicion_event_3
			ID  = $EVENT3$
		}
		POD_save_event_as_var = {
			VAR = suspicion_event_4
			ID  = $EVENT4$
		}
	}
}

POD_heist_trigger_suspicion_failure_event = {
	POD_heist_pause_for_suspicion_event = yes
	show_as_tooltip = {
		death = {
			death_reason = death_execution
		}
	}
	trigger_event = { saved_event_id = var:current_heist.var:suspicion_event_4 }
}

POD_heist_raise_suspicion_effect = {
	if = {
		limit = { var:current_heist ?= { POD_heist_uses_suspicion_trigger = yes } }

		save_scope_value_as = {
			name = new_suspicion
			value = {
				value = var:current_heist.var:suspicion
				add = $VALUE$
				max = suspicion_max
			}
		}

		custom_description = {
			text  = POD_heist_raise_suspicion_desc
			value = $VALUE$
			var:current_heist ?= {
				set_variable = {
					name  = suspicion
					value = scope:new_suspicion
				}
			}
		}

		if = {
			limit = {
				scope:new_suspicion >= suspicion_max
			}
			custom_description_no_bullet = {
				text  = POD_heist_maximum_suspicion_desc

				POD_heist_trigger_suspicion_failure_event = yes
			}
		}
		POD_heist_check_suspicion_threshold = { NUM = 3 }
		POD_heist_check_suspicion_threshold = { NUM = 2 }
		POD_heist_check_suspicion_threshold = { NUM = 1 }
	}
}

POD_heist_check_suspicion_threshold = {
	if = {
		limit = {
			scope:new_suspicion >= suspicion_threshold_$NUM$
			var:current_heist ?= { NOT = { has_variable = triggered_suspicion_event_$NUM$ } }
		}
		custom_description_no_bullet = {
			text  = POD_heist_trigger_suspicion_event_desc
			value = suspicion_threshold_$NUM$

			POD_heist_pause_for_suspicion_event = yes
			trigger_event = { saved_event_id = var:current_heist.var:suspicion_event_$NUM$ }
			var:current_heist = { set_variable = triggered_suspicion_event_$NUM$ }
		}
	}
}

POD_heist_pause_for_suspicion_event = {
	var:current_heist ?= {
		if = {
			limit = { NOT = { has_variable = paused_for_suspicion_event } }
			set_variable = paused_for_suspicion_event
		}
	}
}
POD_heist_resume_after_suspicion_event = {
	var:current_heist ?= {
		if = {
			limit = { has_variable = paused_for_suspicion_event }
			remove_variable = paused_for_suspicion_event
		}
	}
}
