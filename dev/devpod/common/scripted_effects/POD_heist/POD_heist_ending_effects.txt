POD_close_heist = {
	hidden_effect = {
		set_variable = hide_heist_window
	}
}

POD_heist_cleanup_effect = {
	hidden_effect = {
		var:current_heist ?= { end_story = yes }
		remove_variable = current_heist
		remove_variable = hide_heist_window
	}
}

POD_heist_trigger_ending_event = {
	POD_heist_pause_for_suspicion_event = yes
	trigger_event = { saved_event_id = var:current_heist.var:ending_event }
}


POD_heist_gain_loot_effect = {
	save_temporary_scope_as = loot_receiver
	var:current_heist ?= {
		every_in_list = {
			variable  = loot_artifacts
			set_owner = scope:loot_receiver
		}
	}
	if = {
		limit = {
			is_ai = no
			var:current_heist ?= {
				var:stolen_gold >= 200
			}
		}
		set_global_variable = pod_achiev_heisttutorial # What's Yours Is Mine
	}
}

POD_heist_give_stolen_gold_back = {
	if = {
		limit = { var:current_heist = { exists = var:gold_owner } }
		pay_short_term_gold = {
			target = var:current_heist.var:gold_owner
			gold = var:current_heist.var:stolen_gold
		}
	}
}

POD_heist_get_imprisoned_if_possible = {
	if = {
		limit = {
			$IMPRISONER$ = { is_ruler = yes }
			POD_damage_spirit_custom_trigger = {
				RECIPIENT = this
				ACTOR = $IMPRISONER$
			}
		}
		rightfully_imprison_character_less_verbose_effect = {
			TARGET = this
			IMPRISONER = $IMPRISONER$
		}
	}
}

POD_heist_increase_infiltration_skill = {
	if = {
		limit = {
			NAND = {
				has_variable = infiltration_experience_modifier_stacks
				var:infiltration_experience_modifier_stacks >= infiltration_experience_max_stacks
			}
		}

		save_scope_value_as = {
			name = infiltration_experience
			value = {
				if = {
					limit = { has_variable = infiltration_experience }
					value = var:infiltration_experience
				}
				else = {
					value = 0
				}
				add = 1
			}
		}

		if = {
			limit = { scope:infiltration_experience >= heists_needed_for_level_up }

			POD_add_modifier_and_track_stacks = {
				TYPE = character
				MODIFIER = infiltration_experience_modifier
			}
			remove_variable = infiltration_experience
		}
		else = {
			set_variable = {
				name  = infiltration_experience
				value = scope:infiltration_experience
			}
			save_scope_value_as = {
				name = heists_till_level_up
				value = {
					value = heists_needed_for_level_up
					subtract = scope:infiltration_experience
				}
			}
			custom_description = {
				text  = POD_heist_levelup_hint_desc
				value = scope:heists_till_level_up
			}
		}
	}
}

POD_heist_display_expenditures = {
	custom_tooltip = POD_heist_resource_deltas_tt
	show_as_tooltip = {
		if = {
			limit = { var:current_heist.var:gold_delta > 0 }
			add_short_term_gold = var:current_heist.var:gold_delta
		}
		else_if = {
			limit = { var:current_heist.var:gold_delta < 0 }
			remove_short_term_gold = { value = 0 subtract = var:current_heist.var:gold_delta }
		}
		if = {
			limit = { NOT = { var:current_heist.var:prestige_delta = 0 } }
			add_prestige = var:current_heist.var:prestige_delta
		}
		if = {
			limit = { NOT = { var:current_heist.var:piety_delta = 0 } }
			add_piety = var:current_heist.var:piety_delta
		}
	}
}

POD_secondary_effects_when_dying_during_heist = {
	POD_clear_onetime_buffs = yes
	if = {
		limit = { POD_respawns_outside_heist_on_death_trigger = yes }

		hidden_effect = {
			POD_end_journey_effect = yes
			POD_heist_give_stolen_gold_back = yes
		}
		trigger_event = POD_heists.6001
		POD_heist_cleanup_effect = yes
	}
	else_if = {
		limit = { POD_has_guaranteed_resurrection = no }

		hidden_effect = { POD_heist_give_stolen_gold_back = yes }
		POD_heist_cleanup_effect = yes
	}
}

POD_die_during_heist_effect = {
	POD_secondary_effects_when_dying_during_heist = yes
	pod_death = {

	}
}

	every_relation = {
		type = coterie_member
		limit = {
			POD_is_in_same_journey_trigger = { TARGET = root }
		}
		pod_death = {
			death_reason = $DEATH_REASON$
			killer = $KILLER$
		}
	}
}

			death_reason = $DEATH_REASON$
		}
	}
}
