POD_heist_set_node_type = {
	set_variable = {
		name  = type
		value = flag:node_$TYPE$
	}
	set_variable = {
		name  = icon
		value = flag:$ICON$
	}
	POD_error_suppression_unused_flag = { FLAG = $ICON$ }
}
POD_heist_clear_node_type = {
	remove_variable = type
	remove_variable = icon
	clear_variable_list = node_interactions
}

POD_heist_flag_node_as_potential_start = {
	set_variable = potential_start
}

POD_heist_flag_node_as_potential_exit = {
	set_variable = potential_exit
}

POD_heist_flag_node_as_potential_objective = {
	set_variable = potential_objective
}

POD_heist_flag_node_as_side_objective = {
	set_variable = side_objective
}

POD_heist_add_node_interaction = {
	add_to_variable_list = {
		name   = node_interactions
		target = flag:$FLAG$
	}
	POD_error_suppression_unused_flag = { FLAG = $FLAG$ }
}
