
# Effects in this file copy various things from scope:cloned_character into current scope

#POD_remove_vanilla_traits_effect - removes vanilla not physical traits
#POD_remove_physical_traits_effect - removes vanilla congenital traits and various traits affecting physical condition in one way or another
#POD_remove_vanilla_secrets_effect - removes vanilla secrets
#POD_remove_important_supernatural_modifiers_effect - removes only important POD modifiers, such as blood potency, hunger, masquerade exposure and wraith potency
#POD_remove_supernatural_modifiers_effect - removes POD modifiers, but not those considered important
#POD_remove_supernatural_secrets_effect - removes POD secrets
#POD_copy_vanilla_traits_effect - copies vanilla not physical traits
#POD_copy_physical_traits_effect - copies vanilla congenital traits and various traits affecting physical condition in one way or another
#POD_copy_supernatural_traits_effect - copies POD traits
#POD_copy_important_supernatural_modifiers_effect - copies only important POD modifiers, such as blood potency, hunger, masquerade exposure and wraith potency
#POD_copy_supernatural_modifiers_effect - copies all POD modifiers, but not those considered important
#POD_copy_vanilla_secrets_effect - copies vanilla secrets
#POD_copy_supernatural_secrets_effect - copies POD secrets
#POD_clone_replace_in_family_effect - replaces scope:cloned_character in a family tree
#POD_copy_physical_appearance_effect - copies physical appearance, right now immortal age is set to 30 by default
#POD_steal_titles_effect - current character takes all titles from scope:cloned_character
#POD_copy_base_skills_effect - copies base skills, should be used before traits, modifiers and perks are copied
#POD_copy_scripted_hairstyle_effect - copies scripted hairstyle from scope:cloned_character to current scope
#POD_copy_scripted_beard_effect - copies scripted beard from scope:cloned_character to current scope
#POD_copy_supernatural_lifestyle_perks_effect - copies POD lifestyle perks
#POD_copy_vanilla_lifestyle_perks_effect - copies vanilla lifestyle perks


POD_create_shadow_twin_effect = {
    save_scope_as = cloned_character
     
    create_character = {
        save_scope_as = shadow_twin
	    age = scope:cloned_character.age
		faith = scope:cloned_character.faith
		gender = scope:cloned_character
		random_traits = no
		dynasty = none
		employer = scope:cloned_character
	}
    if = {
	    limit = {
		    exists = scope:cloned_character
			exists = scope:shadow_twin
		}
		scope:shadow_twin = {
			POD_copy_base_skills_effect = {
			    CLONED_CHAR = scope:cloned_character
				CLONE = scope:shadow_twin
			}
	        copy_traits = scope:cloned_character
		    POD_copy_important_supernatural_modifiers_effect = yes
            POD_copy_supernatural_secrets_effect = yes
		    POD_copy_physical_appearance_effect = yes
		    POD_copy_vanilla_lifestyle_perks_effect = yes
			POD_copy_supernatural_lifestyle_perks_effect = yes
			add_character_modifier = shadow_twin_modifier
			give_nickname = nick_shadow_twin
			set_relation_shadow_twin_master = scope:cloned_character
			add_character_flag = blocked_from_leaving
	    }
		scope:cloned_character = {
		    add_hook = {
			    target = scope:shadow_twin
				type = shadow_twin_bind_hook
			}
		}
	}
    
}

#Replace in family tree
POD_clone_replace_in_family_effect = {
    
	set_house = scope:cloned_character.house
	
	scope:cloned_character = {
	    if = {
		    limit = {
			    is_female = no
			}
			every_child = {
			    even_if_dead = yes
			    set_father = scope:clone
		    }
		}
		else_if = {
		    limit = {
			    is_female = yes
			}
			every_child = {
			    even_if_dead = yes
			    set_mother = scope:clone
			}
		}
	}
	if = {
	    limit = {
		    exists = scope:cloned_character.father
		}
		set_father = scope:cloned_character.father
		create_character = {
		       save_scope_as = new_father
			   location = scope:cloned_character.capital_province
			   culture = scope:cloned_character.culture
			   faith = scope:cloned_character.faith
			   age = 3000
			   gender = male
			   trait = vampire
			   random_traits = yes
		}
		
		
		scope:cloned_character = {
		    set_father = scope:new_father
			set_house = scope:new_father.house
		}
		scope:new_father = {
		    death = { death_reason = death_disappearance }
		}
	}
	else_if = {
	    limit = {
		    exists = scope:cloned_character.mother
		}
		set_mother = scope:cloned_character.mother
		create_character = {
		       save_scope_as = new_mother
			   location = scope:cloned_character.capital_province
			   culture = scope:cloned_character.culture
			   faith = scope:cloned_character.faith
			   age = 3000
			   gender = female
			   trait = vampire
			   random_traits = yes
		}
		scope:cloned_character = {
		    set_father = scope:new_mother
			set_house = scope:new_mother.house
		}
		scope:new_mother = {
		    death = { death_reason = death_disappearance }
		}
	}
}

#Copy base skills
POD_copy_base_skills_effect = {
	hidden_effect = {
		$CLONED_CHAR$ = { 
			every_equipped_character_artifact = { 
				add_to_temporary_list = unequipped_artifact_list
				unequip_artifact_from_owner = yes 
			}
			force_character_skill_recalculation = yes
			save_scope_as = cloned_character 
		}
		create_character = {
			save_scope_as = dummyclone
			age = 16
			intrigue = 0	
			prowess = 0	
			martial = 0
			stewardship = 0
			diplomacy = 0
			learning = 0
			faith = $CLONED_CHAR$.faith
			culture = $CLONED_CHAR$.culture
			gender = female
			random_traits = no
			dynasty = none
			employer = $CLONED_CHAR$
		}
		scope:dummyclone = { copy_traits = $CLONED_CHAR$ }
		scope:dummyclone = { POD_copy_supernatural_lifestyle_perks_effect = yes }
		scope:dummyclone = { POD_copy_vanilla_lifestyle_perks_effect = yes }
		scope:dummyclone = { force_character_skill_recalculation = yes }
	}
    add_martial_skill = { 
	    value = $CLONED_CHAR$.martial subtract = $CLONE$.martial subtract = scope:dummyclone.martial
	}
	add_intrigue_skill = { 
	    value = $CLONED_CHAR$.intrigue subtract = $CLONE$.intrigue subtract = scope:dummyclone.intrigue
	}
	add_learning_skill = { 
	    value = $CLONED_CHAR$.learning subtract = $CLONE$.learning subtract = scope:dummyclone.learning
	}
	add_diplomacy_skill = { 
	    value = $CLONED_CHAR$.diplomacy subtract = $CLONE$.diplomacy subtract = scope:dummyclone.diplomacy
	}
	add_stewardship_skill = { 
	    value = $CLONED_CHAR$.stewardship subtract = $CLONE$.stewardship subtract = scope:dummyclone.stewardship
	}
	add_prowess_skill = { 
	    value = $CLONED_CHAR$.prowess subtract = $CLONE$.prowess subtract = scope:dummyclone.prowess
	}
	hidden_effect = { 
		scope:dummyclone = { death = { death_reason = death_vanished } } 
		$CLONED_CHAR$ = { 
	        every_in_list = {
	        	list = unequipped_artifact_list
	        	equip_artifact_to_owner_replace = yes
	        }
		}
	}
}
