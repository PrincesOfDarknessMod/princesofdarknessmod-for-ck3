

court_ulama_study_theology = {
	if = {
		limit = { NOT = { has_perk = theologian_perk } }
		scope:liege = {
			send_interface_message = {
				type = event_generic_good
				title = court_ulama_study_theology_title
				right_icon = prev

				prev = {
					if = {
						limit = { 
							has_perk = defender_of_the_faith_perk 
						}
						add_perk = theologian_perk
					}
					else_if = {
						limit = {
							has_perk = prophet_perk
							has_perk = radiant_perk
							NOT = { has_perk = defender_of_the_faith_perk }
						}
						add_perk = defender_of_the_faith_perk
					}
					#left side
					else_if = {
						limit = { 
							has_perk = religious_icon_perk
							NOT = { has_perk = prophet_perk }
						}
						add_perk = prophet_perk
					}
					else_if = {
						limit = { 
							has_perk = zealous_proselytizer_perk
							NOT = { has_perk = religious_icon_perk }
						}
						add_perk = religious_icon_perk
					}
					else_if = {
						limit = { 
							has_perk = faithful_perk
							NOT = { has_perk = zealous_proselytizer_perk }
						}
						add_perk = zealous_proselytizer_perk
					}
					#right side
					else_if = {
						limit = { 
							has_perk = church_and_state_perk
							NOT = { has_perk = radiant_perk } 
						}
						add_perk = radiant_perk
					}
					else_if = {
						limit = { 
							has_perk = clerical_justifications_perk
							NOT = { has_perk = church_and_state_perk }
						}
						add_perk = church_and_state_perk
					}
					else_if = {
						limit = { 
							has_perk = faithful_perk
							NOT = { has_perk = clerical_justifications_perk }
						}
						add_perk = clerical_justifications_perk
					}
					else = {
						add_perk = faithful_perk
					}
				}
			}
		}
	}
}

court_ulama_study_scholarship = {
	if = {
		limit = { NOT = { has_perk = scholar_perk } }
		scope:liege = {
			send_interface_message = {
				type = event_generic_good
				title = court_ulama_study_theology_title
				right_icon = prev

				prev = {
					if = {
						limit = { 
							has_perk = sanctioned_loopholes_perk
						}
						add_perk = scholar_perk
					}
					else_if = {
						limit = { 
							has_perk = learn_on_the_job_perk
							NOT = { has_perk = sanctioned_loopholes_perk }
						}
						add_perk = sanctioned_loopholes_perk
					}
					else_if = {
						limit = { 
							has_perk = apostate_perk
							has_perk = scholarly_circles_perk
							NOT = { has_perk = learn_on_the_job_perk }
						}
						add_perk = learn_on_the_job_perk
					}
					#left side
					else_if = {
						limit = { 
							has_perk = open_minded_perk
							NOT = { has_perk = apostate_perk }
						}
						add_perk = apostate_perk
					}
					else_if = {
						limit = { 
							has_perk = pedagogy_perk
							NOT = { has_perk = open_minded_perk }
						}
						add_perk = open_minded_perk
					}
					else_if = {
						limit = { 
							NOT = { has_perk = pedagogy_perk }
							NOT = { has_perk = pedagogy_perk }
						}
						add_perk = pedagogy_perk
					}
					#right side
					else_if = {
						limit = { 
							has_perk = planned_cultivation_perk
							NOT = { has_perk = scholarly_circles_perk }
						}
						add_perk = scholarly_circles_perk
					}
					else_if = {
						limit = { 
							has_perk = scientific_perk
							NOT = { has_perk = planned_cultivation_perk }
						}
						add_perk = planned_cultivation_perk
					}
					else = {
						add_perk = scientific_perk
					}
				}
			}
		}
	}
}


court_mullah_zealots_cairo_effect = {
	scope:liege = {
		send_interface_message = {
			type = event_generic_good
			title = court_mullah_zealots_cairo_effect_title
			right_icon = prev

			if = {
				limit = { has_trait = cynical }
				remove_trait = cynical
			}
			else_if = {
				limit = { any_courtier = { has_trait = cynical } }
				random_courtier = {
					limit = { has_trait = cynical }
					remove_trait = cynical
				}
			}
			else = {
				random_vassal = {
					limit = { has_trait = cynical }
					remove_trait = cynical
				}
			}
		}
	}
}
