

court_ulama_study_theology = {
	if = {
		limit = { NOT = { has_perk = theologian_perk } }
		scope:liege = {
			send_interface_message = {
				type = event_generic_good
				title = court_ulama_study_theology_title
				right_icon = prev

				prev = {
					POD_gain_next_perk_in_lifestyle = {
						LIFESTYLE = theology
					}
				}
			}
		}
	}
}

court_mullah_zealots_cairo_effect = {
	scope:liege = {
		if = {
			limit = { has_trait = cynical }
			save_scope_as = cynic
		}
		if = {
			limit = { NOT = { exists = scope:cynic } }
			random_courtier = {
				limit = { has_trait = cynical }
				save_scope_as = cynic
			}
		}
		if = {
			limit = { NOT = { exists = scope:cynic } }
			random_vassal = {
				limit = { has_trait = cynical }
				save_scope_as = cynic
			}
		}
		
		# only show message if a character was found
		if = {
			limit = { exists = scope:cynic }
			send_interface_message = {
				type = event_generic_good
				title = court_mullah_zealots_cairo_effect_title
				right_icon = prev
	
				scope:cynic = { remove_trait = cynical }
			}
		}
	}
}
