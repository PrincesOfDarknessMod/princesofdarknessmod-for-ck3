######################################
# Story cycle sync
######################################
#
# Pushes St James current character variables (paranoia_or_trust) 
#into the matching story-scope variables on the story.
# Called from every event that mutates those
# character variables (podintroductions.216 , decision events).
#
POD_hunters_st_james_story_effect = {
	every_owned_story = {
		limit = { story_type = POD_hunters_story_cycle_st_james }
		# Only mirror while the arc is unresolved. After a path is chosen the
		# resolution / oscillation effects own the counters directly (and retire
		# some of them), so the generic mirror must not re-create or clobber them.
		if = {
			if = {
				limit = { prev = { has_variable = paranoia_or_trust } }
				set_variable = {
					name = paranoia_or_trust
					value = prev.var:paranoia_or_trust
				}
			}
		}
	}
}
