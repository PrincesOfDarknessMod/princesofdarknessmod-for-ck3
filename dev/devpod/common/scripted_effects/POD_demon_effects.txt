POD_demon_initialization_effect = {
	if = {
		limit = { has_character_modifier = unknown_eminence_modifier }
		add_character_modifier = unknown_eminence_modifier	
	}
	if = {
		limit = { NOT = { religion = { is_in_family = rf_demons } } }
		POD_character_demon_faith_initiation_effect = yes
	}
	POD_character_resolve_initiation_effect = yes
}

POD_character_demon_faith_initiation_effect = {
	religion:demons_religion = {
		random_faith = {
			save_scope_as = this_faith
		}
	}
	set_character_faith = scope:this_faith
}

POD_character_resolve_initiation_effect = {
	if = {
		limit = { has_resolve_trigger = no }
		if = {
			limit = { has_trait = demon }
			add_character_modifier = resolve3_modifier
		}
		else = {
			add_character_modifier = resolve1_modifier
		}
	}
}

POD_get_demon_vessel_effect = {
	ordered_relation = {
		type = possessed_target
		order_by = sum_of_all_skills_value
		position = 0
		save_scope_as = vessel
	}
	if = {
		limit = {
			NOT = { exists = scope:vessel }
			has_perk = cerement_of_walking_dust_perk
		}
		ordered_in_list = {
			variable = corpse_list
			order_by = sum_of_all_skills_value
			position = 0
			save_scope_as = vessel
		}
	}
}

POD_next_demon_vessel_effect = {
	if = {
		limit = { NOT = { has_variable = host } }
		trigger_event = POD_demon_maintenance.20
	}
	else = {
		if = {
			limit = { var:host = { is_alive = no } }
			send_interface_message = {
				type = event_generic_bad
				title = host_death_notification_title
				left_icon = root
				right_icon = var:host
				var:host = {
					add_trait_force_tooltip = wounded_3
					remove_list_variable = { name = corpse_list target = this }
				}
				show_as_tooltip = { POD_remove_host_leech_modifier_effect = yes }
			}
		}
		else = {
			send_interface_message = {
				type = event_generic_bad
				title = host_death_notification_title
				left_icon = root
				right_icon = var:host
				var:host = { death = { death_reason = death_suicide } }
				show_as_tooltip = { POD_remove_host_leech_modifier_effect = yes }
			}
		}
		POD_new_demon_vessel_effect = yes
	}
}

POD_new_demon_vessel_effect = {
	remove_variable = host
	POD_remove_host_leech_modifier_effect = yes
	POD_get_demon_vessel_effect = yes
	if = {
		limit = { NOT = { exists = scope:vessel } }
		trigger_event = POD_demon_maintenance.20
	}
	else = {
		POD_first_demon_vessel_effect = { HOST = scope:vessel }
	}
}

POD_first_demon_vessel_effect = {
	if = {
		limit = {
			NOR = {
				has_character_modifier = host_leech_modifier_terrible
				has_character_modifier = host_leech_modifier_medium
				has_character_modifier = host_leech_modifier_good
			}
		}
		send_interface_message = {
			type = event_generic_good
			title = POD_host_possession
			left_icon = $HOST$
			
			set_variable = { name = host value = $HOST$ }
			POD_resolve_decrease_multiple_effect = { COUNT = 5 }
			if = {
				limit = {
					$HOST$ = { sum_of_all_skills_value <= 20 } 	
				}
				add_character_modifier = host_leech_modifier_terrible				
			}
			else_if = {
				limit = {
					$HOST$ = { sum_of_all_skills_value <= 40 } 	
				}
				add_character_modifier = host_leech_modifier_medium				
			}
			else = {
				add_character_modifier = host_leech_modifier_good				
			}
			
			$HOST$ = { 
				add_trait = possessed_genetic	
				copy_inheritable_appearance_from = prev
			}
		}
	}
}

POD_remove_host_leech_modifier_effect = {
	if = {
		limit = {
			has_character_modifier = host_leech_modifier_terrible
		}
		remove_character_modifier = host_leech_modifier_terrible
	}
	else_if = {
		limit = {
			has_character_modifier = host_leech_modifier_medium
		}
		remove_character_modifier = host_leech_modifier_medium
	}
	else_if = {
		limit = {
			has_character_modifier = host_leech_modifier_good
		}
		remove_character_modifier = host_leech_modifier_good
	}	
}
