POD_demon_initialization_effect = {
	if = { 
		limit = { 
			OR = { 
				exists = scope:character_being_changed
				is_from_ruler_designer = yes
			}
		}
		POD_demon_chargen_starting_xp_effect = yes
	}
	else = { 
		if = {
			limit = { NOT = { religion = { is_in_family = rf_demons } } }
			POD_character_demon_faith_initiation_effect = yes
		}
	}
	if = {
		limit = { is_demonologist_trigger = no }
		give_demonology_secret_or_trait_effect = yes
	}
	if = {
		limit = {
			has_demonologist_xp = {
				VALUE = 0
				COMPARATOR = "="
			}
		}
		add_demonologist_xp = {
			VALUE = 100
		}
		add_trait_xp = { trait = demon value = 100 } # Modern Fallen could start lower
	}
	POD_character_resolve_initiation_effect = yes
	POD_character_demon_lifestyle_initialization_effect = yes
}

POD_demon_house_initialization_effect = {
	if = {
		limit = {
			exists = house
			house.house_head = { has_trait = demon }
			NOT = { house.house_head = this }
		}
		create_cadet_branch = yes
		if = { 
			limit = { exists = scope:coa_target }
			house = {
				set_coa = scope:coa_target
				set_house_name_from_house = scope:coa_target
			}
		}
	}
	if = {
		limit = { 
			OR = { 
				NOT = { exists = dynasty } 
				dynasty = { NOT = { has_dynasty_perk = eminence_legacy_1 } } 
			}
		}
		add_character_modifier = unknown_eminence_modifier
	}
	else_if = { 
		limit = { dynasty = { NOT = { has_dynasty_perk = eminence_legacy_2 } } }
		add_character_modifier = known_eminence_modifier
	}
	else_if = { 
		limit = { dynasty = { NOT = { has_dynasty_perk = eminence_legacy_3 } } }
		add_character_modifier = respected_eminence_modifier
	}
	else_if = { 
		limit = { dynasty = { NOT = { has_dynasty_perk = eminence_legacy_4 } } }
		add_character_modifier = honored_eminence_modifier
	}
	else_if = { 
		limit = { dynasty = { NOT = { has_dynasty_perk = eminence_legacy_5 } } }
		add_character_modifier = powerful_eminence_modifier
	}
	else = { 
		add_character_modifier = exalted_eminence_modifier
	}
}

POD_resolve_vessel_decrease_multiple_effect = {
	if = {
		limit = { var:stored_sekhem >= $COUNT$ }
		custom_description = {
			text = POD_extract_resolve
			subject = $OWNER$
			value = $COUNT$
			change_variable = { name = stored_sekhem subtract = $COUNT$ }
		}
	}
	else = {
		save_scope_value_as = {
			name = excess_resolve
			value = {
				value = $COUNT$
				subtract = var:stored_sekhem
			}
		}
		custom_description = {
			text = POD_extract_resolve
			subject = $OWNER$
			value = var:stored_sekhem
			change_variable = { name = stored_sekhem subtract = var:stored_sekhem }
		}
		$OWNER$ = {
			POD_resolve_decrease_multiple_effect = { COUNT = scope:excess_resolve }
		}
	}
	if = {
		limit = { NOT = { has_variable = temporary_vessel } }
		set_vessel_artifact_name_effect = yes
	}
	if = {
		limit = { 
			OR = {
				has_variable = temporary_vessel
				var:stored_sekhem <= 0 
			}
			NOT = { has_variable = permanent_vessel }
		}
		save_scope_as = vessel
		$OWNER$ = {
			send_interface_toast = {
				right_icon = scope:vessel
				title = resolve_vessel_damaged
				scope:vessel = { degrade_vessel_durability_effect = yes }
			}
		}
	}
}

POD_resolve_cost_effect = {
	if = {
		limit = {
			any_character_artifact = {
				has_variable = stored_sekhem
				var:stored_sekhem > 0
			}
		}
		ordered_character_artifact = {
			limit = {
				has_variable = stored_sekhem 
				var:stored_sekhem > 0
			}
			order_by = {
				value = 0
				add = var:stored_sekhem
			}
			position = 0
			POD_resolve_vessel_decrease_multiple_effect = { COUNT = $COUNT$ OWNER = prev }
		}
	}
	else = {
		POD_resolve_decrease_multiple_effect = { COUNT = $COUNT$ }
	}
}

POD_resolve_income_effect = {
	if = {
		limit = {
			OR = {
				resolve_difference > $COUNT$
				NOT = {
					any_character_artifact = {
						NOT = { has_variable = temporary_vessel }
						has_variable = stored_sekhem
						vessel_difference > 0
					}
				}
			}
		}
		POD_resolve_increase_multiple_effect = { COUNT = $COUNT$ }
	}
	else = {
		save_scope_value_as = {
			name = excess_resolve
			value = {
				value = $COUNT$
				subtract = resolve_difference
			}
		}
		POD_resolve_increase_multiple_effect = { COUNT = resolve_difference }
		ordered_character_artifact = {
			limit = {
				NOT = { has_variable = temporary_vessel }
				has_variable = stored_sekhem 
				vessel_difference > 0
			}
			order_by = {
				value = 0
				add = vessel_difference
			}
			position = 0
			POD_resolve_vessel_increase_multiple_effect = { COUNT = scope:excess_resolve OWNER = prev }
		}
	}
}

	save_scope_as = owner
	save_scope_value_as = { name = quality value = 1 }
	get_artifact_wealth_effect = yes
	
	create_artifact = {
		name = artifact_reliquarie_name
		creator = scope:owner
		description = artifact_reliquarie_description
		visuals = $VISUALS$
		type = $TYPE$
		modifier = artifact_placeholder_modifier
		save_scope_as = newly_created_artifact
		wealth = scope:wealth
		quality = scope:quality
		#decaying = no
	}
	hidden_effect_new_object = { 
		scope:newly_created_artifact = {
			remove_artifact_modifier = artifact_placeholder_modifier
			set_variable = { name = reliquarie value = scope:owner }
			set_variable = possessed_artifact 
			add_soulforged_artifact_modifier_effect = { KILLED = scope:owner }
		}
	}
	set_variable = { name = stone_prison value = scope:newly_created_artifact }
}

		set_relation_possessed_target = $HOST$
	}

		type = event_generic_good
		title = POD_host_possession
		left_icon = $HOST$
		
		set_variable = { name = host value = $HOST$ }
		$HOST$ = { 
			add_character_flag = {
				flag = recent_host
				days = 280
			}
		}

		POD_remove_host_leech_modifier_effect = yes
		
		if = {
			limit = {
				$HOST$ = { has_trait = automaton } 	
			}
		}
		else_if = {
			limit = {
				$HOST$ = { sum_of_all_skills_value <= 20 } 	
			}
			add_character_modifier = host_leech_modifier_terrible				
		}
		else_if = {
			limit = {
				$HOST$ = { sum_of_all_skills_value <= 40 } 	
			}
			add_character_modifier = host_leech_modifier_medium				
		}
		else = {
			add_character_modifier = host_leech_modifier_good				
		}

			add_trait = possessed_genetic	
			copy_inheritable_appearance_from = prev
			add_character_modifier = possessed_demonic_modifier
		}
	}
}
