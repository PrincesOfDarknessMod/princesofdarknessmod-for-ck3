#Umbra
create_artifact_klaive_effect = {
	# Get the character the artifact is being made for.
	$OWNER$ = { save_scope_as = owner }
	# Not really used, but if we don't set the scopes we get errors in the feature selection
	set_artifact_rarity_common = yes

	# Create the artifact
	create_artifact = {
		decaying = no	
		name = artifact_klaive_name
		description = artifact_klaive_description
		#category = inventory
		type = dagger
		template = fetish_template
		visuals = dagger_POD_klaive_any
		modifier = klaive_modifier	
		wealth = scope:wealth
		quality = scope:quality
		history = {
			type = created_before_history
		}	
		save_scope_as = newly_created_artifact
	}
	scope:newly_created_artifact = {
		set_variable = possessed_artifact 
		set_variable = { name = historical_unique_artifact value = yes }
		set_variable = klaive
		#save_scope_as = epic
		save_scope_value_as = {
			name = klaive
			value = yes
		}
	}	
}

	$OWNER$ = { save_scope_as = owner }
	# Not really used, but if we don't set the scopes we get errors in the feature selection
	set_artifact_rarity_famed = yes

	# Create the artifact
	create_artifact = {
		decaying = no	
		name = artifact_pain_eagle_name
		description = artifact_pain_eagle_description
		#category = inventory
		type = hammer
		template = fetish_template	
		visuals = hammer
		modifier = pain_eagle_modifier	
		wealth = scope:wealth
		quality = scope:quality
		history = {
			type = created_before_history
		}	
		save_scope_as = newly_created_artifact
	}
	scope:newly_created_artifact = {
		set_variable = possessed_artifact 
		set_variable = { name = historical_unique_artifact value = yes }
		set_variable = pain_eagle
		#save_scope_as = epic
		save_scope_value_as = {
			name = pain_eagle
			value = yes
		}
	}	
}
