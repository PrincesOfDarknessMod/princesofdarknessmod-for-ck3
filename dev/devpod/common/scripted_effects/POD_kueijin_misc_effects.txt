POD_soulbond_effect = {
	$MASTER$ = {
		if = {
			limit = {
				has_hook_of_type = {
					target = $VICTIM$
					type = strong_soul_shackles_hook
				}
			}
			send_interface_toast = {
				title = POD_shared_loca_success
				left_icon = $VICTIM$
				right_icon = $MASTER$
				
				custom_tooltip = POD_soulbond_prisoner_interaction_growing
				# This Spirit Bond overrides all other forms of loyalty, even a previously established Blood Bond.
				remove_hook = { target = $VICTIM$ }
				add_hook = {
					target = $VICTIM$
					type = soul_bond_hook
				}
				POD_chi_cost_effect = { COUNT = 3 }
			}
		}
		else_if = {
			limit = {
				OR = {
					has_perk = divine_fetters_perk
					has_hook_of_type = {
						target = $VICTIM$
						type = weak_soul_shackles_hook
					}
				}
			}
			send_interface_toast = {
				title = POD_shared_loca_success
				left_icon = $VICTIM$
				right_icon = $MASTER$
				
				custom_tooltip = POD_soulbond_prisoner_interaction_growing
				remove_hook = { target = $VICTIM$ }
				add_hook = {
					target = $VICTIM$
					type = strong_soul_shackles_hook
				}
				POD_chi_cost_effect = { COUNT = 2 }
			}
		}
		else = {
			send_interface_toast = {
				title = POD_shared_loca_success
				left_icon = $VICTIM$
				right_icon = $MASTER$
				
				custom_tooltip = POD_soulbond_prisoner_interaction_growing
				if = {
					limit = { has_hook = $VICTIM$ }
					remove_hook = { target = $VICTIM$ }
				}
				add_hook = {
					target = $VICTIM$
					type = weak_soul_shackles_hook
				}
				POD_chi_cost_effect = { COUNT = 1 }
			}
		}
		POD_willpower_rouse_check_effect = yes 
	}
}
