POD_set_military_order_precept = {
	set_variable = {
		name  = military_order_precept
		value = $PRECEPT$
	}
	set_variable = {
		name  = military_order_rank
		value = 1
	}
}

POD_gain_ashen_grandmaster_xp_from_war = {
	if = {
		limit = {
			has_trait = ashen_adventurer
			is_landless_adventurer = yes
			POD_is_ashen_grandmaster = yes
			# Don't double-dip if this war was part of a contract
			any_character_active_contract = {
				count = 0
				OR = {
					AND = {
						has_task_contract_type = laamp_raid_contract
						var:task_contract_war ?= scope:war
					}
					AND = {
						has_task_contract_type = laamp_join_war_contract
						var:task_contract_war ?= scope:war
					}
					AND = {
						has_task_contract_type = laamp_help_claimant_contract
						task_contract_employer = scope:war.primary_attacker
					}
					AND = {
						has_task_contract_type = laamp_join_faction_contract
						task_contract_employer = {
							is_participant_in_war = scope:war
						}
					}
				}
			}
		}
		add_trait_xp = {
			trait = ashen_adventurer
			track = ashen_grandmaster
			value = 2
		}
	}
}
