# TODO: More efficient to get by index but cut down on number of stories by doing it this way
# Need to see if performance is bad here
POD_umbra_adventure_get_node_by_row_and_column = {
	clear_saved_scope = $SAVE_AS$
	$MAP$ = {
		every_in_list = {
			variable = nodes
			limit = {
				NOT = {
					exists = scope:$SAVE_AS$
				}
				var:row ?= $ROW$
				var:column ?= $COLUMN$
			}
			save_scope_as = $SAVE_AS$
		}
	}
}

# POD_umbra_adventure_create_node
#	WIDTH = width of the entire map
#	HEIGHT = height of the entire map
#	ROWS = number of rows to subdivide the map
#	COLUMNS = number of columns to subdivide the map
#	PADDING = internal padding of the node (to prevent it from being right on the edges of the quadrant)
#	ROW = the quadrant row
#	COLUMN = the quadrant column
#	MAP = the node map
POD_umbra_adventure_create_node = {
	save_scope_value_as = {
		name = quadrant_size_x
		value = {
			value = $WIDTH$
			divide = $COLUMNS$
		}
	}
	save_scope_value_as = {
		name = quadrant_size_y
		value = {
			value = $HEIGHT$
			divide = $ROWS$
		}
	}
	save_scope_value_as = {
		name = coord_x
		value = {
			value = {
				0
				1
			}
			multiply = {
				value = scope:quadrant_size_x
				subtract = {
					value = $PADDING$
					multiply = 2
				}
			}
			add = $PADDING$
			# offset by current column
			add = {
				value = scope:quadrant_size_x
				multiply = $COLUMN$
			}
		}
	}
	save_scope_value_as = {
		name = coord_y
		value = {
			value = { 0 1 }
			multiply = {
				value = scope:quadrant_size_x
				subtract = {
					value = $PADDING$
					multiply = 2
				}
			}
			add = $PADDING$
			# offset by current row
			add = {
				value = scope:quadrant_size_y
				multiply = $ROW$
			}
		}
	}
	POD_create_node = {
		MAP = $MAP$
		COORD_X = scope:coord_x
		COORD_Y = scope:coord_y
		SAVE_AS = created_node
	}
	scope:created_node = {
		set_variable = {
			name = row
			value = $ROW$
		}
		set_variable = {
			name = column
			value = $COLUMN$
		}
	}
}

# POD_umbra_adventure_get_neighbors
#	NODE = the current node
#	MAP = the node map
#	ROWS = the number of rows in the map
#	COLUMNS = the number of columns in the map
#	SAVE_AS = name of the saved list (NOT a variable list)
POD_umbra_adventure_get_neighbors = {
	saved_scope_value_as = {
		name = current_node
		value = $NODE$
	}
	save_scope_value_as = {
		name = row_idx
		value = {
			value = scope:current_node.var:row
			subtract = 1
			min = 0
		}
	}
	save_scope_value_as = {
		name = row_max
		value = {
			value = scope:current_node.var:row
			add = 2
			max = $ROWS$
		}
	}
	save_scope_value_as = {
		name = column_max
		value = {
			value = scope:current_node.var:column
			add = 2
			max = $COLUMNS$
		}
	}
	while = {
		limit = {
			scope:row_idx < scope:row_max
		}
		save_scope_value_as = {
			name = column_idx
			value = {
				value = scope:current_node.var:column
				subtract = 1
				min = 0
			}
		}
		while = {
			limit = {
				scope:column_idx < scope:column_max
			}
			POD_umbra_adventure_get_node_by_row_and_column = {
				MAP = $MAP$
				ROW = scope:row_idx
				COLUMN = scope:column_idx
				SAVE_AS = neighbor_node
			}
			if = {
				limit = {
					AND = {
						exists = scope:neighbor_node
						scope:neighbor_node != scope:current_node
					}
				}
				scope:neighbor_node = {
					add_to_list = $SAVE_AS$
				}
			}
			POD_increment_scope_value = {
				SCOPE = column_idx
			}
		}
		POD_increment_scope_value = {
			SCOPE = row_idx
		}
	}
}

# POD_umbra_adventure_generate_grid_lines
#	MAP = the node map
POD_umbra_adventure_generate_spanning_tree = {
	save_scope_value_as = {
		name = idx
		value = 0
	}
	$MAP$ = {
		ordered_in_list = {
			variable = nodes
			position = 0
			set_variable = {
				name = spanned
				value = yes
			}
			add_to_temporary_list = spanned_nodes
		}
		while = {
			limit = { variable_list_size = { name = nodes value > scope:idx } }
			clear_saved_scope = best_distance
			# Prims Algorithm
			# Note that this implementation is very likely exponential, which is fine for small, sparse graphs
			# but which also scales horribly for large dense graphs, use carefully
			# Find the optimal lowest weight edge to the spanned nodes
			every_in_list = {
				list = spanned_nodes

				save_scope_as = spanned_node
				$MAP$ = {
					ordered_in_list = {
						variable = nodes
						limit = {
							NOT = {
								var:spanned ?= yes
							}
						}
						# squared distance
						order_by = {
							value = 0
							add = {
								value = scope:spanned_node.var:coord_x
								subtract = var:coord_x
								multiply = {
									value = scope:spanned_node.var:coord_x
									subtract = var:coord_x
								}
							}
							add = {
								value = scope:spanned_node.var:coord_y
								subtract = var:coord_y
								multiply = {
									value = scope:spanned_node.var:coord_y
									subtract = var:coord_y
								}
							}
							# ascending
							multiply = -1
						}
						position = 0
						save_scope_as = candidate_node
					}
				}
				# Recalculate squared distance for spanned_node -> candidate
				POD_node_squared_distance = { NODE1 = scope:spanned_node NODE2 = scope:candidate_node SAVE_AS = spanned_node_to_candidate_distance }
				if = {
					limit = { 
						OR = {
							NOT = { exists = scope:best_distance }
							scope:spanned_node_to_candidate_distance < scope:best_distance
						}
					}
					save_scope_value_as = {
						name = best_distance
						value = scope:spanned_node_to_candidate_distance
					}
					scope:spanned_node = {
						save_scope_as = start_node
					}
					scope:candidate_node = {
						save_scope_as = end_node
					}
				}
			}
			POD_create_line = {
				MAP = $MAP$
				START = scope:start_node
				END = scope:end_node
				SAVE_AS = created_line
			}
			scope:end_node = {
				set_variable = {
					name = spanned
					value = yes
				}
				add_to_temporary_list = spanned_nodes
			}			
			POD_increment_scope_value = {
				SCOPE = idx
			}
		}
	}
}

# POD_umbra_adventure_generate_grid_lines
#	WIDTH = width of the entire map
#	HEIGHT = height of the entire map
#	ROWS = number of rows to subdivide the map
#	COLUMNS = number of columns to subdivide the map
#	MAP = the node map
POD_umbra_adventure_generate_grid_lines = {
	save_scope_value_as = {
		name = row_idx
		value = 0
	}
	save_scope_value_as = {
		name = column_idx
		value = 0
	}
	save_scope_value_as = {
		name = quadrant_size_x
		value = {
			value = $WIDTH$
			divide = $COLUMNS$
		}
	}
	save_scope_value_as = {
		name = quadrant_size_y
		value = {
			value = $HEIGHT$
			divide = $ROWS$
		}
	}
	$MAP$ = {
		while = {
			limit = {
				scope:column_idx <= $COLUMNS$
			}
			save_scope_value_as = {
				name = column_x
				value = {
					value = scope:column_idx
					multiply = scope:quadrant_size_x
				}
			}
			save_scope_value_as = {
				name = end_y
				value = {
					value = $ROWS$
					multiply = scope:quadrant_size_y
				}
			}
			POD_create_grid_line = {
				MAP = $MAP$
				START_X = scope:column_x
				START_Y = 0
				END_X = scope:column_x
				END_Y = scope:end_y
				SAVE_AS = grid_line
			}
			POD_increment_scope_value = {
				SCOPE = column_idx
			}
		}
		while = {
			limit = {
				scope:row_idx <= $ROWS$
			}
			save_scope_value_as = {
				name = row_y
				value = {
					value = scope:row_idx
					multiply = scope:quadrant_size_y
				}
			}
			save_scope_value_as = {
				name = end_x
				value = {
					value = $COLUMNS$
					multiply = scope:quadrant_size_x
				}
			}
			POD_create_grid_line = {
				MAP = $MAP$
				START_X = 0
				START_Y = scope:row_y
				END_X = scope:end_x
				END_Y = scope:row_y
				SAVE_AS = grid_line
			}
			POD_increment_scope_value = {
				SCOPE = row_idx
			}
		}
	}
}

# Uses FTLs map generation algorithm
# Algorithm:
#	The map is broken into quandrants of equal size (by the given row and column size)
# 	Each quadrant has a 50% chance at a node, where x = random(0,quadrant_size_x-padding*2) + padding and y = random(0,quadrant_size_y-padding*2)
#	Each quadrant with a node gets a line between itself and any adjacent quadrant with a node
#	Orphan nodes are then connected to their 2 nearest neighbors (TBD on this step, might be distance based if you get a second line)
# POD_umbra_adventure_generate_map
#	WIDTH = width of the entire map
#	HEIGHT = height of the entire map
#	ROWS = number of rows to subdivide the map
#	COLUMNS = number of columns to subdivide the map
#	PADDING = internal padding of the node (to prevent it from being right on the edges of the quadrant)
#	SAVE_AS = saved scope of the created map
POD_umbra_adventure_generate_map = {
	POD_create_node_map = {
		SAVE_AS = $SAVE_AS$
	}
	scope:$SAVE_AS$ = {
		save_scope_as = node_map
	}
	save_scope_value_as = {
		name = row_idx
		value = 0
	}
	while = {
		limit = {
			scope:row_idx < $ROWS$
		}
		save_scope_value_as = {
			name = column_idx
			value = 0
		}
		while = {
			limit = {
				scope:column_idx < $COLUMNS$
			}
			random_list = {
				50 = {
					POD_umbra_adventure_create_node = {
						WIDTH = $WIDTH$
						HEIGHT = $HEIGHT$
						ROWS = $ROWS$
						COLUMNS = $COLUMNS$
						PADDING = $PADDING$
						ROW = scope:row_idx
						COLUMN = scope:column_idx
						MAP = scope:node_map
					}
				}
				50 = {
					# no-op
				}
			}
			POD_increment_scope_value = {
				SCOPE = column_idx
			}
		}
		POD_increment_scope_value = {
			SCOPE = row_idx
		}
	}

	POD_umbra_adventure_generate_spanning_tree = {

}

POD_umbra_adventure_add_stackable_item = {
	ordered_in_list = {
		variable = inventory
		position = 0
		limit = {
			var:type ?= $ITEM$.var:type
		}
		save_scope_as = stackable_item
	}
	if = {
		limit = {
			exists = scope:stackable_item
		}
		scope:stackable_item = {
			set_variable = {
				name = quantity
				value = {
					value = scope:stackable_item.var:quantity
					add = $ITEM$.var:quantity
				}
			}
		}
		$ITEM$ = {
			end_story = yes
		}
	}
	else = {
		add_to_variable_list = {
			name = inventory
			target = $ITEM$
		}
	}
}

POD_umbra_adventure_add_to_inventory = {
	var:umbra_adventure = {
		save_scope_as = umbra_adventure
	}
	$ITEM$ = {
		save_scope_as = inventory_item
	}
	if = {
		limit = {
			scope:umbra_adventure = {
				any_in_list = {
					variable = inventory
					count < 8
				}
			}
		}
		# handle stackables
		if = {
			limit = {
				scope:inventory_item = {
					has_variable = quantity
				}
			}
			scope:umbra_adventure = {
				POD_umbra_adventure_add_stackable_item = {
					ITEM = scope:inventory_item
				}
			}
		}
		else = {
			scope:umbra_adventure = {
				add_to_variable_list = {
					name = inventory
					target = scope:inventory_item
				}
			}
		}
	}
	else = {
		# cleanup the inventory item
		scope:inventory_item = {
			end_story = yes
		}
	}
}
