POD_umbra_adventure_add_to_inventory = {
	var:umbra_adventure = {
		save_scope_as = umbra_adventure
	}
	scope:umbra_adventure = {
		ordered_in_list = {
			variable = inventory
			position = 7
			save_scope_as = last_item
		}
	}
	if = {
		limit = {
			AND = {
				scope:last_item = {
					NOT = {
						has_variable = type
					}
				}
				$ITEM$ = {
					has_variable = type
				}
			}
		}
		if = {
			limit = {
				$ITEM$ = {
					has_variable = quantity
				}
			}
			ordered_in_list = {
				variable = inventory
				position = 0
				limit = {
					var:type ?= $ITEM$.var:type
				}
				save_scope_as = stackable_item
			}
		}
		if = {
			limit = {
				exists = scope:stackable_item
			}
			scope:stackable_item = {
				set_variable = {
					name = quantity
					value = {
						value = scope:stackable_item.var:quantity
						add = $ITEM$.var:quantity
					}
				}
			}
			$ITEM$ = {
				end_story = yes
			}
		}
		else = {
			var:umbra_adventure = {
				remove_list_variable = {
					name = inventory
					target = scope:last_item
				}
				scope:last_item = {
					end_story = yes
				}
				add_to_variable_list = {
					name = inventory
					target = $ITEM$
				}
			}
		}
	}
	scope:umbra_adventure = {
		ordered_in_list = {
			variable = inventory
			max = variable_list_size:inventory
			order_by = {
				value = 0
				if = {
					limit = {
						has_variable = type
					}
					add = 1
				}
			}
			add_to_list = temp_list
		}
		clear_variable_list = inventory
		every_in_list = {
			list = temp_list
			save_scope_as = item
			scope:umbra_adventure = {
				add_to_variable_list = {
					name = inventory
					target = scope:item
				}
			}
		}
	}
}
