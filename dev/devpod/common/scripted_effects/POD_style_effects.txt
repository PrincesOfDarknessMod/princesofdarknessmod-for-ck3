# Values of 0 stop the decal from showing up, so this is necessary
@ZERO = 0.004

# Color schemes can be set on character or house scopes
# Character colors override house colors
POD_initialize_colorschemes = {
	house:house_kaleva ?= {
		POD_set_colorscheme_santa = yes
	}
}

POD_set_colorscheme_santa = {
	set_variable = { name = POD_portrait_clothing_1_r value = 0.7 }
	set_variable = { name = POD_portrait_clothing_1_g value = 0.05 }
	set_variable = { name = POD_portrait_clothing_1_b value = 0.05 }

	set_variable = { name = POD_portrait_clothing_2_r value = 0.7 }
	set_variable = { name = POD_portrait_clothing_2_g value = 0.7 }
	set_variable = { name = POD_portrait_clothing_2_b value = 0.7 }

	set_variable = { name = POD_portrait_clothing_3_r value = 1.0 }
	set_variable = { name = POD_portrait_clothing_3_g value = 1.0 }
	set_variable = { name = POD_portrait_clothing_3_b value = 1.0 }

	set_variable = { name = POD_portrait_clothing_4_r value = @ZERO }
	set_variable = { name = POD_portrait_clothing_4_g value = 0.2 }
	set_variable = { name = POD_portrait_clothing_4_b value = 0.1 }
	
	set_variable = { name = POD_portrait_clothing_5_r value = 0.7 }
	set_variable = { name = POD_portrait_clothing_5_g value = 0.05 }
	set_variable = { name = POD_portrait_clothing_5_b value = 0.05 }
}
