
# character scope, called in monthly pulse
POD_select_progress_value_effect = {
	switch = {
		trigger = var:current_journey.var:progress_value
		
		flag:five_months       = { POD_journey_progress = { AMOUNT = five_months_progress } }
		flag:ten_months        = { POD_journey_progress = { AMOUNT = ten_months_progress } }
		flag:twelve_months     = { POD_journey_progress = { AMOUNT = twelve_months_progress } }
		flag:five_years        = { POD_journey_progress = { AMOUNT = five_years_progress } }
		flag:ten_years         = { POD_journey_progress = { AMOUNT = ten_years_progress } }
		
		flag:seek_knowledge    = { POD_journey_progress = { AMOUNT = seek_knowledge_progress } }
		
		flag:find_artifact     = { POD_journey_progress = { AMOUNT = find_artifact_progress } }
		
		flag:wake_from_torpor  = { POD_journey_progress = { AMOUNT = wake_from_torpor_progress } }
		flag:assault_lair      = { POD_journey_progress = { AMOUNT = assault_lair_progress } }
		
		flag:werewolf_training = { POD_journey_progress = { AMOUNT = werewolf_training_progress } }
		
		flag:instant           = { POD_journey_progress = { AMOUNT = instant_progress } }
		
		fallback               = { POD_journey_progress = { AMOUNT = ten_months_progress } }
	}
}
