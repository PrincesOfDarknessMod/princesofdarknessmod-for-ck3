POD_character_vampire_bloodpotency_initialization_effect = {
	if = {
		 limit = {
			is_vampgeneration1_trigger = yes
		}
		add_character_modifier = {
			modifier = bloodpotency10_modifier 
		}		
	}
	else_if = {
		 limit = {
			is_vampgeneration2_trigger = yes
		}
		add_character_modifier = {
			modifier = bloodpotency9_modifier 
		}		
	}
	else_if = {
		 limit = {
			is_vampgeneration3_trigger = yes
		}
		add_character_modifier = {
			modifier = bloodpotency8_modifier 
			}		
	}
	else_if = {
		 limit = {
			is_vampgeneration4_trigger = yes
		}
		add_character_modifier = {
			modifier = bloodpotency5_modifier 
		}		
	}
	else_if = {
		 limit = {
			is_vampgeneration5_trigger = yes
		}
		add_character_modifier = {
			modifier = bloodpotency4_modifier 
		}		
	}
	else_if = {
		limit = {
			has_trait_rank = {
				trait = vampgeneration
				rank < 8
			}
		}
		add_character_modifier = {
			modifier = bloodpotency3_modifier 
		}		
	}
	else_if = {
		limit = {
			has_trait_rank = {
				trait = vampgeneration
				rank < 10
			}
		}
		add_character_modifier = {
			modifier = bloodpotency2_modifier 
		}		
	}
	else_if = {
		 limit = {
			has_trait_rank = {
				trait = vampgeneration
				rank < 14
			}
		}
		add_character_modifier = {
			modifier = bloodpotency1_modifier 
		}		
	}
	else_if = {
		limit = {
			has_POD_vampire_generation_trigger = yes
		}
		add_character_modifier = {
			modifier = bloodpotency0_modifier 
		}		
	}
}
