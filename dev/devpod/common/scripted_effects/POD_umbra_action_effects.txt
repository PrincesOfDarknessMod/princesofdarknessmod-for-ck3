# Possession
POD_become_wyrm_servant_effect = {
	if = {
		limit = { has_POD_vamp_blood_trigger = yes }
		set_character_faith = faith:viaapep
	}
	if = {
		limit = { 
			OR = {
				has_trait = kinfolk
				has_trait = shapeshifter 
			}
		}
		set_character_faith = faith:wyrm
	}
	if = {
		limit = { is_amenti_trigger = yes }
		set_character_faith = faith:faces_of_apophis
		POD_remove_mummy_traits_effect = yes
		add_trait = banemummy
	}
	if = {
		limit = { is_east_mummy_trigger = yes }
		set_character_faith = faith:ten_mandates_of_hell
		POD_remove_mummy_traits_effect = yes
		add_trait = wukuei
	}
	if = {
		limit = { has_trait = wraith }
		POD_become_spectre_effect = yes
	}
	if = {
		limit = { has_trait = supehunter }
		POD_hunter_enthrall_effect = yes
	}
	if = {
		limit = { has_trait = kueijin }
		set_character_faith = faith:wayofthetenthousandscreams
	}
}

POD_umbra_fallback_low_effect = {
	duel = {
		skill = $SKILL$
		value = low_skill_rating
	
		25 = {
			compare_modifier = {
				value = scope:duel_value
				multiplier = 0.5
			}
			send_interface_message = {
				type = event_generic_good_with_text
				title = POD_umbra_$SKILL$.success
				desc = POD_umbra_$SKILL$.desc
				right_icon = root
				
				add_$SKILL$_lifestyle_xp = miniscule_lifestyle_xp
			}
		}
		25 = {
			compare_modifier = {
				value = scope:duel_value
				multiplier = -0.5
			}
			send_interface_message = {
				type = event_generic_bad_with_text
				title = POD_umbra_$SKILL$.fail
				desc = POD_umbra_$SKILL$.desc
				right_icon = root
				
				add_stress = major_stress_gain
				add_piety = minor_piety_loss
			}
		}
	}
}
