POD_werewolf_advance_rank_effect = { 
	if = {
		limit = { 
			NOR = { 
				has_trait_xp = { trait = werewolf value = 19 }
				has_trait_xp = { trait = werewolf value = 39 }
				has_trait_xp = { trait = werewolf value = 59 }
				has_trait_xp = { trait = werewolf value = 79 }
				has_trait_xp = { trait = werewolf value = 99 }
			}
		}
		add_trait_xp = { trait = werewolf value = 1 }
	}
}

POD_gain_wisdom_xp_effect = {
	if = {
		limit = { faith = { has_doctrine = special_doctrine_beast_courts } }
		POD_shifter_advance_rank_effect = yes
	}
	else = {
		switch = {
			trigger = has_trait
			werewolf = {
				if = {
					limit = { 
						OR = {
							has_trait = galliard
							has_trait = philodox
							has_trait = theurge
							has_trait = ragabash
						}
					}
					POD_werewolf_advance_rank_effect = yes
				}
			}
		}
	}
}
