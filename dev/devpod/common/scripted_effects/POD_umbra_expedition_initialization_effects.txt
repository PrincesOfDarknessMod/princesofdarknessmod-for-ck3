POD_increment_value = {
	save_scope_value_as = {
		name = $SCOPE$
		value = {
			value = scope:$SCOPE$
			add = 1
		}
	}
}

POD_create_umbra_node = {
	character:POD_wraith_01 = {
		create_story = {
			type = POD_story_umbra_node
			save_scope_as = $SAVE_AS$
		}
	}

	hidden_effect_new_object = {
		scope:$SAVE_AS$ = {
			set_variable = {
				name  = row
				value = $ROW$
			}
			set_variable = {
				name  = COLUMN
				value = $COLUMN$
			}
		}
		
		add_to_variable_list = {
			name   = nodes
			target = scope:$SAVE_AS$
		}
	}
}


# POD_initialize_umbra_expedition = { ROWS = x COLUMNS = y }
# Columns should be odd in order to center the first and last nodes
# Columns MUST be at least 3 otherwise pathing will break
# Rows should be at least 3 to be useful
POD_initialize_umbra_expedition = {
	if = {
		limit = { NOT = { exists = var:umbra_map } } 
		create_story = {
			type = POD_story_umbra_map
			save_scope_as = umbra_map

			set_variable = {
				name = rows
				value = $ROWS$
			}

			set_variable = {
				name = columns
				value = $COLUMNS$
			}
		}

		
		save_scope_value_as  = {
			name = idx_i
			value = 0
		}

		save_scope_value_as = {
			name = idx_j
			value = 0
		}

		while = {
			limit = { scope:idx_i < $ROWS$ }

			while = {
				limit = { scope:idx_j < $COLUMNS$ }

				POD_increment_value = { SCOPE = idx_j }

				scope:umbra_map = {
					POD_create_umbra_node = { SAVE_AS = current_node ROW = idx_i COLUMN = idx_j }
				}
			}

			# reset j
			save_scope_value_as = {
				name = idx_j
				value = 0
			}

			POD_increment_value = { SCOPE = idx_i }
		}

		# Start node
		save_scope_value_as = {
			name = start_idx
			value = {
				value = $COLUMNS$
				divide = 2
				floor = yes
			}
		}
		POD_at_index = { LIST = scope:umbra_map.var:nodes INDEX = scope:start_idx SAVE_AS = start_node EFFECT = "set_variable = { name = include value = yes } set_variable = { name = type value = flag:start }" }


		# End node
		# start_idx * (ROWS - 1)
		save_scope_value_as = {
			name = end_idx
			value = {
				value = scope:start_idx
				multiply = {
					value = $ROWS$
					subtract = 1
				}
			}
		}
		POD_at_index = { LIST = scope:umbra_map.var:nodes INDEX = scope:end_idx SAVE_AS = end_idx EFFECT = "set_variable = { name = include value = yes } set_variable = { name = type value = flag:end }" }


		set_variable = {
			name = umbra_map
			value = scope:umbra_map
		}
	}
}

POD_pick_next_room = {
	save_temporary_scope_value_as = {
		name = previous_room_row
		value = {
			value = $PREVIOUS_ROOM_IDX$
			divide = scope:umbra_map.var:columns
			floor = yes
		}
	}

	save_temporary_scope_value_as = {
		name = previous_room_column
		value = {
			value = $PREVIOUS_ROOM_IDX$
			modulo = scope:umbra_map.var:columns
		}
	}

	save_temporary_scope_value_as = {
		name = first_room_idx
		value = {
			value = scope:previous_room_column
			subtract = 1
			max = 0
			# column position
			# add (previous row number + 1) * cols to get actual position
			add = {
				value = {
					value = scope:previous_room_row
				}
				add = 1
				multiply = scope:umbra_map.var:columns
			}
		}
	}

	save_temporary_scope_value_as = {
		name = first_room_idx
		value = {
			value = scope:previous_room_column
			add = 1
			min = {
				value = scope:umbra_map.var:columns
				subtract = 1
			}
			# column position
			# add (previous row number + 1) * cols to get actual position
			add = {
				value = {
					value = scope:previous_room_row
				}
				add = 1
				multiply = scope_umbra_map.var:columns
			}
		}
	}

	save_scope_value_as = {
		name = next_room
		value = {
			integer_range = {
				min = scope:first_room_idx
				max = scope:last_room_idx
			}
		}
	}
}
