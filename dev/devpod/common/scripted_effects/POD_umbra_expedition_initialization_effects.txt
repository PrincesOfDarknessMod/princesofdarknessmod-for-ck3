# POD_at_index = { LIST = list INDEX = 2 SAVE_AS = foo }
POD_at_index = {
	ordered_in_list = {
		variable = $LIST$
		position = $INDEX$
		save_scope_as = $SAVE_AS$
	}
}

# Inclusive range(0,3) = list of 0,1,2,3
# Stored in a local variable list named $SAVE_AS$
POD_create_range = {
	save_scope_value_as = {
		name = range_idx
		value = $START$
	}
	while = {
		limit = {
			scope:range_idx <= $END$
		}
		add_to_local_variable_list = {
			name = $SAVE_AS$
			target = scope:range_idx
		}
		POD_increment_scope_value = {
			SCOPE = range_idx
		}
	}
}

POD_create_umbra_node = {
	character:POD_wraith_01 = {
		create_story = {
			type = POD_story_umbra_node
			save_scope_as = $SAVE_AS$
		}
	}
	hidden_effect_new_object = {
		scope:$SAVE_AS$ = {
			set_variable = {
				name = coord_y
				value = $ROW$
			}
			set_variable = {
				name = coord_x
				value = $COLUMN$
			}
			set_variable = {
				name = index
				value = $INDEX$
			}
		}
		add_to_variable_list = {
			name = nodes
			target = scope:$SAVE_AS$
		}
	}
}

POD_create_umbra_line = {
	if = {
		# skip duplicate 
		limit = {
			trigger_if = {
				limit = { has_variable_list = lines }
				NOT = {
					any_in_list = {
						variable = lines
						AND = {
							var:start_node = $NODE1$
							var:end_node = $NODE2$
						}
					}
				}
			}
		}
		character:POD_wraith_01 = {
			create_story = {
				type = POD_story_umbra_line
				save_scope_as = line
			}
		}
		hidden_effect_new_object = {
			scope:line = {
				set_variable = {
					name = start_node
					value = $NODE1$
				}
				set_variable = {
					name = end_node
					value = $NODE2$
				}
			}
			$NODE1$ = {
				set_variable = {
					name = include
					value = yes
				}
				add_to_variable_list = {
					name = lines
					target = scope:line
				}
			}
			$NODE2$ = {
				set_variable = {
					name = include
					value = yes
				}
				add_to_variable_list = {
					name = lines
					target = scope:line
				}
			}
			add_to_variable_list = {
				name = lines
				target = scope:line
			}
		}
	}
}

# POD_umbra_x_from_index = { INDEX = 5 SAVE_AS = foo }
POD_umbra_x_from_index = {
	save_scope_value_as = {
		name = $SAVE_AS$
		value = {
			value = $INDEX$
			modulo = scope:umbra_map.var:columns
		}
	}
}

# POD_umbra_y_from_index = { INDEX = 5 SAVE_AS = foo }
POD_umbra_y_from_index = {
	save_scope_value_as = {
		name = $SAVE_AS$
		value = {
			value = $INDEX$
			divide = scope:umbra_map.var:columns
			floor = yes
		}
	}
}

# POD_initialize_umbra_expedition = { ROWS = x COLUMNS = y }
# Columns should be odd in order to center the first and last nodes
# Columns MUST be at least 3 otherwise pathing will break
# Rows should be at least 3 to be useful
POD_initialize_umbra_expedition = {
	# Start node
	save_scope_value_as = {
		name = start_idx
		value = {
			value = $COLUMNS$
			divide = 2
			floor = yes
		}
	}
	# End node
	# start_idx + COLS * (ROWS - 1)
	save_scope_value_as = {
		name = end_idx
		value = {
			value = scope:start_idx
			add = {
				value = $COLUMNS$
				multiply = {
					value = $ROWS$
					subtract = 1
				}
			}
		}
	}
	create_story = {
		type = POD_story_umbra_map
		save_scope_as = umbra_map
	}
	scope:umbra_map = {
		set_variable = {
			name = rows
			value = $ROWS$
		}
		set_variable = {
			name = columns
			value = $COLUMNS$
		}
		set_variable = {
			name = start_idx
			value = scope:start_idx
		}
		set_variable = {
			name = end_idx
			value = scope:end_idx
		}
	}
	save_scope_value_as = {
		name = idx_i
		value = 0
	}
	save_scope_value_as = {
		name = idx_j
		value = 0
	}
	while = {
		limit = {
			scope:idx_i < $ROWS$
		}
		while = {
			limit = {
				scope:idx_j < $COLUMNS$
			}
			save_scope_value_as = {
				name = current_index
				value = {
					value = scope:idx_j
					add = {
						value = $COLUMNS$
						multiply = scope:idx_i
					}
				}
			}
			scope:umbra_map = {
				POD_create_umbra_node = {
					SAVE_AS = current_node
					ROW = scope:idx_i
					COLUMN = scope:idx_j
					INDEX = scope:current_index
				}
			}
			POD_increment_scope_value = {
				SCOPE = idx_j
			}
		}
		# reset j
		save_scope_value_as = {
			name = idx_j
			value = 0
		}
		POD_increment_scope_value = {
			SCOPE = idx_i
		}
	}
	scope:umbra_map = {
		POD_at_index = {
			LIST = nodes
			INDEX = scope:start_idx
			SAVE_AS = start_node
		}
		POD_at_index = {
			LIST = nodes
			INDEX = scope:end_idx
			SAVE_AS = end_node
		}
	}
	scope:start_node = {
		set_variable = {
			name = include
			value = yes
		}
		set_variable = {
			name = type
			value = flag:node_start
		}
	}
	scope:end_node = {
		set_variable = {
			name = include
			value = yes
		}
		set_variable = {
			name = type
			value = flag:node_end
		}
	}
	scope:umbra_map = {
		set_variable = {
			name = start_node
			value = scope:start_node
		}
		set_variable = {
			name = end_node
			value = scope:end_node
		}
	}
}

POD_pick_next_room = {
	POD_umbra_x_from_index = { INDEX = $PREVIOUS_ROOM_IDX$ SAVE_AS = previous_room_x }
	POD_umbra_y_from_index = { INDEX = $PREVIOUS_ROOM_IDX$ SAVE_AS = previous_room_y }

	#
	save_scope_value_as = {
		name = first_room_idx
		value = {
			value = scope:umbra_map.var:columns
			multiply = {
				value = scope:previous_room_y
				add = 1
			}
			add = {
				value = scope:previous_room_x
				subtract = 1
				min = 0
			}
		}
	}
	# adjust first_room_idx to account for crossing paths
	POD_umbra_x_from_index = { INDEX = scope:first_room_idx SAVE_AS = first_room_x }
	save_scope_value_as = {
		name = neighbor_below
		value = {
			value = scope:previous_room_idx
			add = scope:umbra_map.var:columns
		}
	}
	save_scope_value_as = {
		name = neighbor_left
		value = {
			value = scope:previous_room_idx
			subtract = 1
			min = {
				value = scope:previous_room_y
				multiply = scope:umbra_map.var:columns
			}
		}
	}
	if = {
		limit = {
			AND = {
				# Must be actually leftward
				scope:first_room_x < scope:previous_room_x
				scope:umbra_map = {
					any_in_list = {
						variable = lines
						AND = {
							# No path may go from the neighbor to the left of the previous room choice to below the previous room choice
							# As this would cross over the possible path *to* the room choice itself
							var:start_node.var:index = scope:neighbor_left
							var:end_node.var:index = scope:neighbor_below
						}
					}
				}
			}
		}
		save_scope_value_as = {
			name = first_room_offset
			value = yes
		}
		save_scope_value_as = {
			name = first_room_idx
			value = scope:neighbor_below
		}
	}
	
	#
	save_scope_value_as = {
		name = last_room_idx
		value = {
			value = scope:umbra_map.var:columns
			multiply = {
				value = scope:previous_room_y
				add = 1
			}
			add = {
				value = scope:previous_room_x
				add = 1
				max = {
					value = scope:umbra_map.var:columns
					subtract = 1
				}
			}
		}
	}
	# adjust last_room_idx to account for crossing paths
	POD_umbra_x_from_index = { INDEX = scope:last_room_idx SAVE_AS = last_room_x }
	save_scope_value_as = {
		name = neighbor_below
		value = {
			value = scope:previous_room_idx
			add = scope:umbra_map.var:columns
		}
	}
	save_scope_value_as = {
		name = neighbor_right
		value = {
			value = scope:previous_room_idx
			add = 1
			max = {
				value = scope:previous_room_y
				add = 1
				multiply = scope:umbra_map.var:columns
			}
		}
	}
	if = {
		limit = {
			AND = {
				# Must be actually rightward
				scope:last_room_x > scope:previous_room_x
				scope:umbra_map = {
					any_in_list = {
						variable = lines
						AND = {
							# No path may go from the neighbor to the left of the previous room choice to below the previous room choice
							# As this would cross over the possible path *to* the room choice itself
							var:start_node.var:index = scope:neighbor_right
							var:end_node.var:index = scope:neighbor_below
						}
					}
				}
			}
		}
		save_scope_value_as = {
			name = last_room_offset
			value = yes
		}
		save_scope_value_as = {
			name = last_room_idx
			value = scope:neighbor_below
		}
	}

	save_scope_value_as = {
		name = $SAVE_AS$
		value = {
			integer_range = {
				min = scope:first_room_idx
				max = scope:last_room_idx
			}
		}
	}
}

# https://www.reddit.com/r/slaythespire/comments/ndqweh/i_have_reverseengineered_map_generation_algorithm/
# https://kosgames.com/slay-the-spire-map-generation-guide-26769/
POD_create_umbra_expedition_lines = {
	while = {
		count = $PASSES$
		# Always start at the first room
		scope:umbra_map.var:start_idx = {
			save_scope_as = next_room_idx
		}
		# Generate the path
		# Iterate from 0 to rows-1 and pick the next room. Penultimate room always connects to the end
		save_scope_value_as = {
			name = idx_row
			value = 0
		}
		while = {
			# TODO: count for some reason does not work with math AND scope/vars
			count = {
				value = $ROWS$
				# ignore last room and indexed by 0
				subtract = 2
			}
			scope:next_room_idx = {
				save_scope_as = previous_room_idx
			}
			POD_pick_next_room = {
				PREVIOUS_ROOM_IDX = scope:previous_room_idx
				SAVE_AS = next_room_idx
			}
			scope:umbra_map = {
				POD_at_index = {
					LIST = nodes
					INDEX = scope:previous_room_idx
					SAVE_AS = previous_node
				}
				POD_at_index = {
					LIST = nodes
					INDEX = scope:next_room_idx
					SAVE_AS = next_node
				}
				POD_create_umbra_line = {
					NODE1 = scope:previous_node
					NODE2 = scope:next_node
				}
			}
			scope:next_node = {
				POD_assign_umbra_room_type = yes
			}
		}
		scope:umbra_map = {
			POD_create_umbra_line = {
				NODE1 = scope:next_node
				NODE2 = scope:umbra_map.var:end_node
			}
		}
	}
}

# Generates a 5x8 expedition
POD_create_umbra_expedition = {
	if = {
		limit = {
			exists = var:umbra_map
		}
		var:umbra_map = {
			end_story = yes
		}
	}
	hidden_effect_new_object = {
		POD_initialize_umbra_expedition = {
			COLUMNS = 5
			ROWS = 8
		}
		POD_create_umbra_expedition_lines = {
			PASSES = 6
			ROWS = 8
		}
		scope:umbra_map = {
			set_variable = {
				name = current_location
				value = var:start_node
			}
		}
	}
	set_variable = {
		name = umbra_map
		value = scope:umbra_map
	}
}
