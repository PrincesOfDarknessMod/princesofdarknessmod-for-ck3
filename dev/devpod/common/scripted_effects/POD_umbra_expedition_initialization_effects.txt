POD_increment_value = {
	save_scope_value_as = {
		name = $SAVE_AS$
		value = {
			value = scope:$SAVE_AS$
			add = 1
		}
	}
}

# POD_at_index = { LIST = list INDEX = 2 SAVE_AS = foo }
POD_at_index = {
	ordered_in_list = {
		variable = $LIST$
		position = $INDEX$
		save_scope_as = $SAVE_AS$
	}
}

# Inclusive range(0,3) = list of 0,1,2,3
# Stored in a local variable list named $SAVE_AS$
POD_create_range = {
	save_scope_value_as = {
		name = idx
		value = $START$
	}
	while = {
		limit = {
			scope:idx <= $END$
		}
		add_to_local_variable_list = {
			name = $SAVE_AS$
			target = scope:idx
		}
		POD_increment_value = {
			SAVE_AS = idx
		}
	}
}

# POD_initialize_umbra_expedition = { ROWS = x COLUMNS = y }
# Columns should be odd in order to center the first and last nodes
# Columns MUST be at least 3 otherwise pathing will break
# Rows should be at least 3 to be useful
POD_initialize_umbra_expedition = {
	# Start node
	save_scope_value_as = {
		name = start_idx
		value = {
			value = $COLUMNS$
			divide = 2
			floor = yes
		}
	}
	# End node
	# start_idx + COLS * (ROWS - 1)
	save_scope_value_as = {
		name = end_idx
		value = {
			value = scope:start_idx
			add = {
				value = $COLUMNS$
				multiply = {
					value = $ROWS$
					subtract = 1
				}
			}
		}
	}
	create_story = {
		type = POD_story_umbra_map
		save_scope_as = umbra_map
	}
	scope:umbra_map = {
		set_variable = {
			name = rows
			value = $ROWS$
		}
		set_variable = {
			name = columns
			value = $COLUMNS$
		}
		set_variable = {
			name = start_idx
			value = scope:start_idx
		}
		set_variable = {
			name = end_idx
			value = scope:end_idx
		}
	}
	save_scope_value_as = {
		name = idx_i
		value = 0
	}
	save_scope_value_as = {
		name = idx_j
		value = 0
	}
	while = {
		limit = {
			scope:idx_i < $ROWS$
		}
		while = {
			limit = {
				scope:idx_j < $COLUMNS$
			}
			save_scope_value_as = {
				name = current_index
				value = {
					value = scope:idx_j
					add = {
						value = $COLUMNS$
						multiply = scope:idx_i
					}
				}
			}
			scope:umbra_map = {
				POD_create_umbra_node = {
					SAVE_AS = current_node
					ROW = scope:idx_i
					COLUMN = scope:idx_j
					INDEX = scope:current_index
				}
			}
			POD_increment_value = {
				SAVE_AS = idx_j
			}
		}
		# reset j
		save_scope_value_as = {
			name = idx_j
			value = 0
		}
		POD_increment_value = {
			SAVE_AS = idx_i
		}
	}
	scope:umbra_map = {
		POD_at_index = {
			LIST = nodes
			INDEX = scope:start_idx
			SAVE_AS = start_node
		}
		POD_at_index = {
			LIST = nodes
			INDEX = scope:end_idx
			SAVE_AS = end_node
		}
	}
	scope:start_node = {
		set_variable = {
			name = include
			value = yes
		}
		set_variable = {
			name = type
			value = flag:start
		}
	}
	scope:end_node = {
		set_variable = {
			name = include
			value = yes
		}
		set_variable = {
			name = type
			value = flag:end
		}
	}
	scope:umbra_map = {
		set_variable = {
			name = start_node
			value = scope:start_node
		}
		set_variable = {
			name = end_node
			value = scope:end_node
		}
	}
}

POD_get_filtered_rooms = {
	POD_create_range = {
		START = scope:first_room_idx
		END = scope:last_room_idx
		SAVE_AS = available_rooms
	}
	ordered_in_local_list = {
		variable = available_rooms
		limit = {
			save_temporary_scope_as = available_room
			save_temporary_scope_value_as = {
				name = available_room_row
				value = {
					value = this
					divide = scope:umbra_map.var:columns
					floor = yes
				}
			}
			save_temporary_scope_value_as = {
				name = available_room_column
				value = {
					value = this
					modulo = scope:umbra_map.var:columns
				}
			}
			save_temporary_scope_value_as = {
				name = available_room_left_neighbor
				value = {
					value = scope:available_room
					subtract = 1
				}
			}
			save_temporary_scope_value_as = {
				name = available_room_right_neighbor
				value = {
					value = scope:available_room
					add = 1
				}
			}
			save_temporary_scope_value_as = {
				name = available_room_neighbor_below
				value = {
					value = scope:available_room
					subtract = scope:umbra_map.var:columns
				}
			}
			# Two filtered conditions to prevent crossing paths
			# leftward path to the first_room
			trigger_if = {
				limit = {
					# Must be an actual leftward path
					# Make sure that available_room_column < previous_room_column
					scope:available_room_column < scope:previous_room_column
				}
				# Make sure that available_room - total_columns has no line to available_room + 1
				scope:umbra_map = {
					NOT = {
						any_in_list = {
							variable = lines
							AND = {
								var:start_node.var:index = scope:available_room_neighbor_below
								# No path may go from below the room choice and to the right of the room choice
								# As this would cross over the possible path *to* the room choice itself
								var:end_node.var:index = scope:available_room_right_neighbor
							}
						}
					}
				}
			}
			# rightward path to the last room
			trigger_if = {
				limit = {
					# Must be an actual leftward path
					# Make sure that available_room_column < previous_room_column
					scope:available_room_column > scope:previous_room_column
				}
				# Make sure that available_room - total_columns has no line to available_room - 1
				scope:umbra_map = {
					NOT = {
						any_in_list = {
							variable = lines
							AND = {
								var:start_node.var:index = scope:available_room_neighbor_below
								# No path may go from below the room choice and to the left of the room choice
								# As this would cross over the possible path *to* the room choice itself
								var:end_node.var:index = scope:available_room_left_neighbor
							}
						}
					}
				}
			}
		}
		add_to_local_variable_list = {
			name = $SAVE_AS$
			target = this
		}
	}
}

POD_pick_next_room = {
	save_scope_value_as = {
		name = previous_room_row
		value = {
			value = $PREVIOUS_ROOM_IDX$
			divide = scope:umbra_map.var:columns
			floor = yes
		}
	}
	save_scope_value_as = {
		name = previous_room_column
		value = {
			value = $PREVIOUS_ROOM_IDX$
			modulo = scope:umbra_map.var:columns
		}
	}
	save_scope_value_as = {
		name = first_room_idx
		value = {
			value = scope:umbra_map.var:columns
			multiply = {
				value = scope:previous_room_row
				add = 1
			}
			add = {
				value = scope:previous_room_column
				subtract = 1
				min = 0
			}
		}
	}
	save_scope_value_as = {
		name = last_room_idx
		value = {
			value = scope:umbra_map.var:columns
			multiply = {
				value = scope:previous_room_row
				add = 1
			}
			add = {
				value = scope:previous_room_column
				add = 1
				max = {
					value = scope:umbra_map.var:columns
					subtract = 1
				}
			}
		}
	}
	POD_get_filtered_rooms = { SAVE_AS = room_choices }
	random_in_local_list = {
		variable = room_choices
		save_scope_as = $SAVE_AS$
	}
}

# https://www.reddit.com/r/slaythespire/comments/ndqweh/i_have_reverseengineered_map_generation_algorithm/
# https://kosgames.com/slay-the-spire-map-generation-guide-26769/
POD_create_umbra_expedition_lines = {
	while = {
		count = $PASSES$
		# Always start at the first room
		scope:umbra_map.var:start_idx = {
			save_scope_as = next_room_idx
		}
		# Generate the path
		# Iterate from 0 to rows-1 and pick the next room. Penultimate room always connects to the end
		save_scope_value_as = {
			name = idx_row
			value = 0
		}
		while = {
			# TODO: count for some reason does not work with math AND scope/vars
			count = {
				value = $ROWS$
				# ignore last room and indexed by 0
				subtract = 2
			}
			scope:next_room_idx = {
				save_scope_as = previous_room_idx
			}
			POD_pick_next_room = {
				PREVIOUS_ROOM_IDX = scope:previous_room_idx
				SAVE_AS = next_room_idx
			}
			scope:umbra_map = {
				POD_at_index = {
					LIST = nodes
					INDEX = scope:previous_room_idx
					SAVE_AS = previous_node
				}
				POD_at_index = {
					LIST = nodes
					INDEX = scope:next_room_idx
					SAVE_AS = next_node
				}
				POD_create_umbra_line = {
					NODE1 = scope:previous_node
					NODE2 = scope:next_node
				}
			}
			scope:next_node = {
				POD_assign_umbra_room_type = yes
			}
		}
		scope:umbra_map = {
			POD_create_umbra_line = {
				NODE1 = scope:next_node
				NODE2 = scope:umbra_map.var:end_node
			}
		}
	}
}

# Generates a 7x15 expedition
POD_create_umbra_expedition = {
	if = {
		limit = {
			exists = var:umbra_map
		}
		var:umbra_map = {
			end_story = yes
		}
	}
	hidden_effect_new_object = {
		POD_initialize_umbra_expedition = {
			COLUMNS = 5
			ROWS = 8
		}
		POD_create_umbra_expedition_lines = {
			PASSES = 6
			ROWS = 8
		}
		scope:umbra_map = {
			set_variable = {
				name = current_location
				value = var:start_node
			}
		}
	}
	set_variable = {
		name = umbra_map
		value = scope:umbra_map
	}
}
