
POD_enhance_initiate_ingredient_effect = {
	POD_ingredient_clear_effect = yes
	add_to_variable_list = { name = ingredient_list target = flag:water }
	add_to_variable_list = { name = ingredient_list target = flag:fuel }
	add_to_variable_list = { name = ingredient_list target = flag:mineral }
	switch = {
		trigger = $RECIPE$
		flag:aid_from_spirits = {
			if = {
				limit = { 
					character:POD_wraith_01 = {
						OR = {
							NOT = { has_variable = character_material }
							var:character_material = { is_alive = no }
						}
					}
				}
				POD_enhance_generate_artifact_spirit_effect = yes
			}
		}
		# flag:sword_of_the_righteous = {
			# add_to_variable_list = { name = ingredient_list target = flag:vampire_blood }
		# }
		flag:inspire_life = {
			if = {
				limit = { 
					character:POD_wraith_01 = {
						OR = {
							NOT = { has_variable = character_material }
							var:character_material = { is_alive = no }
						}
					}
				}
				POD_enhance_generate_artifact_spirit_effect = yes
			}
		}
		flag:strengthen = {
			add_to_variable_list = { name = ingredient_list target = flag:illustrious_artifact_piece }
			add_to_variable_list = { name = alt_ingredient_list target = flag:famed_artifact_piece }
			add_to_variable_list = { name = alt_ingredient_list target = flag:masterwork_artifact_piece }
			add_to_variable_list = { name = alt_ingredient_list target = flag:common_artifact_piece }
		}
	}
	if = {
		limit = { is_target_in_variable_list = { name = ingredient_list target = flag:water } }
		add_to_variable_list = { name = alt_ingredient_list target = flag:exotic_water }
		add_to_variable_list = { name = alt_ingredient_list target = flag:magic_water }
	}
	if = {
		limit = { is_target_in_variable_list = { name = ingredient_list target = flag:fuel } }
		add_to_variable_list = { name = alt_ingredient_list target = flag:exotic_fuel }
		add_to_variable_list = { name = alt_ingredient_list target = flag:magic_fuel }
	}
	if = {
		limit = { is_target_in_variable_list = { name = ingredient_list target = flag:mineral } }
		add_to_variable_list = { name = alt_ingredient_list target = flag:exotic_mineral }
		add_to_variable_list = { name = alt_ingredient_list target = flag:magic_mineral }
	}
}

					}

			}
			flag:sword_of_the_righteous = {
				var:target_artifact = { 
					add_artifact_modifier = artifact_prowess_5_modifier
					set_variable = enchanted_artifact
				}
			}
			flag:POD_rune = {
				var:target_artifact = { 
					add_enhance_artifact_modifier_effect = yes
					set_variable = enhanced_artifact
				}
			}
			flag:channel_yin = {
				var:target_artifact = { 
					add_yin_artifact_modifier_effect = yes
					set_variable = channelled_artifact
				}
			}
			flag:inspire_life = {

				}
			}

					set_variable = enchanted_artifact
				}
			}
			flag:blessing_the_blade = {
				var:target_artifact = { 
					add_artifact_modifier = artifact_prowess_5_modifier
					set_variable = blessed_artifact
				}
			}
			flag:sanctify = {

						var:target_artifact = { 
							add_soulforged_artifact_modifier_effect = { KILLED = scope:killed }
							set_variable = possessed_artifact

						hidden_effect = {
							scope:killed = { death = { death_reason = death_soulforging } }
						}

					add_artifact_modifier = artifact_tyranny_gain_penalty_1_modifier
					set_variable = possessed_artifact
				}
			}
			flag:bound_spirit = {
				if = {
					limit = { has_variable = character_material }
					var:character_material = { save_scope_as = killed }
					var:target_artifact = { 
						set_variable = { name = fetish value = scope:killed } 
						set_variable = possessed_artifact 
						add_soulforged_artifact_modifier_effect = { KILLED = scope:killed }
						save_scope_as = this_artifact
						set_artifact_name = artifact_new_fetish_name
						set_artifact_description = artifact_new_fetish_desc
					}
					scope:killed = {
						set_variable = { name = stone_prison value = scope:target }
						if = {
							limit = { NOT = { this = root } }
							add_opinion = {
								target = root
								modifier = ask_spirit_opinion
								opinion = -25
							}

					}

				if = {
					limit = { is_target_in_variable_list = { name = ingredient_list target = flag:masterwork_artifact_piece } }

					var:target_artifact = { copy_artifact_modifiers = global_var:masterwork_artifact_to_create }

					limit = { is_target_in_variable_list = { name = ingredient_list target = flag:illustrious_artifact_piece } }

					var:target_artifact = { copy_artifact_modifiers = global_var:illustrious_artifact_to_create }
				}
				var:target_artifact = { 
					set_variable = strengthened_artifact
				}

			flag:empower_jade = {
				if = {
					limit = { has_variable = character_material }
					var:character_material = { save_scope_as = killed }
					var:target_artifact = { 
						add_soulforged_artifact_modifier_effect = { KILLED = scope:killed }
						set_variable = possessed_artifact
					}
					execute_opinion_effect = { VICTIM = root.var:character_material EXECUTIONER = root }
					var:character_material = {
						death = {
							death_reason = death_soulforging
							killer = root
						}
					}
					remove_variable = character_material
				}
			}
			flag:rework = {
				if = {
					limit = { has_variable = artifact_material }
					var:target_artifact = { 
						var:target_artifact = { copy_artifact_modifiers = root.var:artifact_material }
						set_variable = reworked_artifact
					}
					var:artifact_material = {
						destroy_artifact = this
					}
					remove_variable = artifact_material
				}
			}
			flag:imbue_object = {
				if = {
					limit = { has_variable = character_material }
					var:character_material = { 
						save_scope_as = killed 
						if = { 
							limit = { has_trait = demon }
							POD_free_from_abyss_effect = yes
							POD_remove_stone_prison_effect = yes
							POD_remove_earth_bound_effect = yes
							POD_become_earthbound_effect = yes
							var:target_artifact = { set_variable = { name = reliquarie value = scope:killed } }
						}
						else_if = { 
							limit = { has_trait = spirit }
							POD_depossess_fetish_effect = yes
							var:target_artifact = { set_variable = { name = fetish value = scope:killed } }
						}
						set_variable = { name = stone_prison value = root.var:target_artifact }
					}
					var:target_artifact = {
						add_soulforged_artifact_modifier_effect = { KILLED = scope:killed }
						set_variable = possessed_artifact
					}
					execute_opinion_effect = { VICTIM = root.var:character_material EXECUTIONER = root }
					var:character_material = {
						pod_death = {
							DEATH_REASON = death_soulforging
							KILLER = root
						}
					}
					remove_variable = character_material
				}
			}
			flag:enchantment = {
				var:target_artifact = { 
					POD_enchantment_effect = yes
				}
			}
			flag:conforma = {
				var:target_artifact = { 
					add_enhance_artifact_modifier_effect = yes
					if = { limit = { prev = { has_perk = scinde_perk } } add_enhance_artifact_modifier_effect = yes }
					set_variable = enhanced_artifact
				}
			}
		}
		POD_achiev_mummytutorial_check = yes
	}
	if = {
		limit = { has_perk = create_object_perk }
		random = {
			chance = 5
			add_gold = ingredient_total_cost
		}
	}
	var:target_artifact = { 
		add_durability = {
			value = artifact_max_durability
			multiply = -0.3
		}
	}
}

POD_enchantment_effect = {
	if = { limit = { NOT = { has_artifact_modifier = sorcery_enchantment_modifier } } add_artifact_modifier = sorcery_enchantment_modifier }
	prev = { save_scope_as = checker }
	if = { limit = { prev = { has_perk = enchantment_2_perk } }
		if = { limit = { has_artifact_modifier = sorcery_enchantment_diplomacy_modifier } remove_artifact_modifier = sorcery_enchantment_diplomacy_modifier }
		if = { limit = { has_artifact_modifier = sorcery_enchantment_martial_modifier } remove_artifact_modifier = sorcery_enchantment_martial_modifier }
		if = { limit = { has_artifact_modifier = sorcery_enchantment_stewardship_modifier } remove_artifact_modifier = sorcery_enchantment_stewardship_modifier }
		if = { limit = { has_artifact_modifier = sorcery_enchantment_intrigue_modifier } remove_artifact_modifier = sorcery_enchantment_intrigue_modifier }
		if = { limit = { has_artifact_modifier = sorcery_enchantment_learning_modifier } remove_artifact_modifier = sorcery_enchantment_learning_modifier }
		while = { 
			count = sorcery_enchantment_level
			random_list = {
				1 = {
					add_artifact_modifier = sorcery_enchantment_diplomacy_modifier
				}
				1 = {
					add_artifact_modifier = sorcery_enchantment_martial_modifier
				}
				1 = {
					add_artifact_modifier = sorcery_enchantment_stewardship_modifier
				}
				1 = {
					add_artifact_modifier = sorcery_enchantment_intrigue_modifier
				}
				1 = {
					add_artifact_modifier = sorcery_enchantment_learning_modifier
				}
			}
		}
	}
	if = { limit = { OR = { artifact_slot_type = armor artifact_slot_type = primary_armament } } 
		if = { limit = { has_artifact_modifier = sorcery_enchantment_1_prowess_modifier } remove_artifact_modifier = sorcery_enchantment_1_prowess_modifier }
		if = { limit = { has_artifact_modifier = sorcery_enchantment_3_prowess_modifier } remove_artifact_modifier = sorcery_enchantment_3_prowess_modifier }
		if = { limit = { has_artifact_modifier = sorcery_enchantment_5_prowess_modifier } remove_artifact_modifier = sorcery_enchantment_5_prowess_modifier }
	
		if = { limit = { prev = { has_perk = enchantment_5_perk } } add_artifact_modifier = sorcery_enchantment_5_prowess_modifier }
		else_if = { limit = { prev = { has_perk = enchantment_3_perk } } add_artifact_modifier = sorcery_enchantment_3_prowess_modifier }
		else_if = { limit = { prev = { has_perk = enchantment_1_perk } } add_artifact_modifier = sorcery_enchantment_1_prowess_modifier }
	}
	else_if = { limit = { OR = { artifact_slot_type = wall_big artifact_slot_type = wall_small } } 
		if = { limit = { has_artifact_modifier = sorcery_enchantment_1_renown_modifier } remove_artifact_modifier = sorcery_enchantment_1_renown_modifier }
		if = { limit = { has_artifact_modifier = sorcery_enchantment_3_renown_modifier } remove_artifact_modifier = sorcery_enchantment_3_renown_modifier }
		if = { limit = { has_artifact_modifier = sorcery_enchantment_5_renown_modifier } remove_artifact_modifier = sorcery_enchantment_5_renown_modifier }
		if = { limit = { has_artifact_modifier = sorcery_enchantment_1_prestige_modifier } remove_artifact_modifier = sorcery_enchantment_1_prestige_modifier }
		if = { limit = { has_artifact_modifier = sorcery_enchantment_3_prestige_modifier } remove_artifact_modifier = sorcery_enchantment_3_prestige_modifier }
		if = { limit = { has_artifact_modifier = sorcery_enchantment_5_prestige_modifier } remove_artifact_modifier = sorcery_enchantment_5_prestige_modifier }
		random_list = {
			1 = {
				if = { limit = { prev = { has_perk = enchantment_5_perk } } add_artifact_modifier = sorcery_enchantment_5_renown_modifier }
				else_if = { limit = { prev = { has_perk = enchantment_3_perk } } add_artifact_modifier = sorcery_enchantment_3_renown_modifier }
				else_if = { limit = { prev = { has_perk = enchantment_1_perk } } add_artifact_modifier = sorcery_enchantment_1_renown_modifier }
			}
			1 = {
				if = { limit = { prev = { has_perk = enchantment_5_perk } } add_artifact_modifier = sorcery_enchantment_5_prestige_modifier }
				else_if = { limit = { prev = { has_perk = enchantment_3_perk } } add_artifact_modifier = sorcery_enchantment_3_prestige_modifier }
				else_if = { limit = { prev = { has_perk = enchantment_1_perk } } add_artifact_modifier = sorcery_enchantment_1_prestige_modifier }
			}
		}
	}
	else_if = { limit = { OR = { artifact_slot_type = throne artifact_slot_type = sculpture artifact_slot_type = pedestal } } 
		if = { limit = { has_artifact_modifier = sorcery_enchantment_1_court_modifier } remove_artifact_modifier = sorcery_enchantment_1_court_modifier }
		if = { limit = { has_artifact_modifier = sorcery_enchantment_3_court_modifier } remove_artifact_modifier = sorcery_enchantment_3_court_modifier }
		if = { limit = { has_artifact_modifier = sorcery_enchantment_5_court_modifier } remove_artifact_modifier = sorcery_enchantment_5_court_modifier }
