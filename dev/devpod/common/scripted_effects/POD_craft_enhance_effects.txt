
POD_enhance_initiate_ingredient_effect = {
	set_variable = { name = current_recipe value = flag:no } # For error.log
	POD_ingredient_clear_effect = yes
	add_to_variable_list = { name = ingredient_list target = flag:water }
	add_to_variable_list = { name = ingredient_list target = flag:fuel }
	add_to_variable_list = { name = ingredient_list target = flag:mineral }
	switch = {
		trigger = $METHOD$
		flag:aid_from_spirits = {
			if = {
				limit = { 
					character:POD_wraith_01 = {
						OR = {
							NOT = { has_variable = character_material }
							var:character_material = { is_alive = no }
						}
					}
				}
				POD_enhance_generate_artifact_spirit_effect = yes
			}
		}
		flag:inspire_life = {
			if = {
				limit = { 
					character:POD_wraith_01 = {
						OR = {
							NOT = { has_variable = character_material }
							var:character_material = { is_alive = no }
						}
					}
				}
				POD_enhance_generate_artifact_spirit_effect = yes
			}
		}
	}
	if = {
		limit = { is_target_in_variable_list = { name = ingredient_list target = flag:water } }
		add_to_variable_list = { name = alt_ingredient_list target = flag:exotic_water }
		add_to_variable_list = { name = alt_ingredient_list target = flag:magic_water }
	}
	if = {
		limit = { is_target_in_variable_list = { name = ingredient_list target = flag:fuel } }
		add_to_variable_list = { name = alt_ingredient_list target = flag:exotic_fuel }
		add_to_variable_list = { name = alt_ingredient_list target = flag:magic_fuel }
	}
	if = {
		limit = { is_target_in_variable_list = { name = ingredient_list target = flag:mineral } }
		add_to_variable_list = { name = alt_ingredient_list target = flag:exotic_mineral }
		add_to_variable_list = { name = alt_ingredient_list target = flag:magic_mineral }
	}
}

POD_enhance_generate_artifact_spirit_effect = {
	create_character = {
		employer = character:POD_wraith_01
		template = POD_enemy_spirit_character
		save_scope_as = spirit
		after_creation = { character:POD_wraith_01 = { set_variable = { name = character_material value = prev } } }
	}
}

POD_enhance_create_effect = {
	random = {
		chance = enhance_success_chance
		switch = {
			trigger = var:current_method
			flag:aid_from_spirits = {
				if = {
					limit = { character:POD_wraith_01 = { has_variable = character_material } }
					character:POD_wraith_01.var:character_material = { save_scope_as = killed }
					var:target_artifact = { 
						add_soulforged_artifact_modifier_effect = { KILLED = scope:killed }
						set_variable = possessed_artifact
					}
					hidden_effect = {
						scope:killed = { death = { death_reason = death_soulforging } }
					}
				}
				POD_enhance_generate_artifact_spirit_effect = yes
			}
			flag:sword_of_the_righteous = {
				var:target_artifact = { 
					add_artifact_modifier = artifact_prowess_5_modifier
					set_variable = enchanted_artifact
				}
			}
			flag:channel_yin = {
				var:target_artifact = { 
					add_yin_artifact_modifier_effect = yes
					set_variable = enchanted_artifact
				}
			}
			flag:inspire_life = {
				if = {
					limit = { character:POD_wraith_01 = { has_variable = character_material } }
					character:POD_wraith_01.var:character_material = { save_scope_as = killed }
					var:target_artifact = { 
						add_soulforged_artifact_modifier_effect = { KILLED = scope:killed }
						set_variable = possessed_artifact
					}
					hidden_effect = {
						scope:killed = { death = { death_reason = death_soulforging } }
					}
				}
				POD_enhance_generate_artifact_spirit_effect = yes
			}
			flag:sorcerers_blade = {
				var:target_artifact = { 
					add_artifact_modifier = artifact_prowess_5_modifier
					set_variable = enchanted_artifact
				}
			}
			flag:sanctify = {
				var:target_artifact = { 
					add_artifact_modifier = artifact_prowess_4_modifier
					add_artifact_modifier = artifact_enemy_hard_casualty_modifier_4_modifier
					set_variable = blessed_artifact
				}
			}
			flag:impress = {
				if = {
					limit = { has_variable = character_material }
					var:character_material = { save_scope_as = killed }
					var:target_artifact = { 
						add_soulforged_artifact_modifier_effect = { KILLED = scope:killed }
						set_variable = possessed_artifact
					}
					execute_opinion_effect = { VICTIM = root.var:character_material EXECUTIONER = root }
					var:character_material = {
						death = {
							death_reason = death_soulforging
							killer = root
						}
					}
					remove_variable = character_material
				}
			}
			flag:empower_jade = {
				if = {
					limit = { has_variable = character_material }
					var:character_material = { save_scope_as = killed }
					var:target_artifact = { 
						add_soulforged_artifact_modifier_effect = { KILLED = scope:killed }
						set_variable = possessed_artifact
					}
					execute_opinion_effect = { VICTIM = root.var:character_material EXECUTIONER = root }
					var:character_material = {
						death = {
							death_reason = death_soulforging
							killer = root
						}
					}
					remove_variable = character_material
				}
			}
		}
	}
}

		}
