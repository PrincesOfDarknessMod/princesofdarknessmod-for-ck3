POD_smith_create_effect = {
	if = {
		limit = { is_target_in_variable_list = { name = alt_ingredient_list target = flag:no_inertion } }
		custom_tooltip = inertion_quality_tooltip
	}
	else = {
		custom_tooltip = smith_quality_tooltip
	}
	remove_short_term_gold = ingredient_total_cost
	POD_cost_sekhem_effect = yes
	POD_cost_sekhem_vessel_effect = yes
	POD_cost_chattel_effect = { EFFIGY = artifact_lesser_chattel }
	POD_cost_chattel_effect = { EFFIGY = artifact_complex_chattel }
	POD_cost_chattel_effect = { EFFIGY = artifact_superior_chattel }
	POD_cost_creature_effect = { EFFIGY = artifact_simple_creature }
	POD_cost_creature_effect = { EFFIGY = artifact_lesser_creature }
	POD_cost_creature_effect = { EFFIGY = artifact_major_creature }
	POD_cost_creature_effect = { EFFIGY = artifact_superior_creature }
	POD_cost_creature_effect = { EFFIGY = artifact_princely_creature }
	POD_cost_creature_effect = { EFFIGY = artifact_viziers_beast }
	POD_cost_creature_effect = { EFFIGY = artifact_kingly_beast }
	POD_cost_material_effect = { ALCHEMY = exotic_water }
	POD_cost_material_effect = { ALCHEMY = exotic_fuel }
	POD_cost_material_effect = { ALCHEMY = exotic_mineral }
	POD_cost_material_effect = { ALCHEMY = magic_water }
	POD_cost_material_effect = { ALCHEMY = magic_fuel }
	POD_cost_material_effect = { ALCHEMY = magic_mineral }
	POD_smith_create_artifact_effect = { RECIPE = var:current_recipe }
	show_as_tooltip = {
		if = {
			limit = { rarity_to_create >= 180 }
			global_var:illustrious_artifact_to_create = { add_amulet_artifact_modifier_effect = { RECIPE = root.var:current_recipe } }
		}
		else_if = {
			limit = { rarity_to_create >= 130 }
			global_var:famed_artifact_to_create = { add_amulet_artifact_modifier_effect = { RECIPE = root.var:current_recipe } }
		}
		else_if = {
			limit = { rarity_to_create >= 50 }
			global_var:masterwork_artifact_to_create = { add_amulet_artifact_modifier_effect = { RECIPE = root.var:current_recipe } }
		}
		else = {
			global_var:common_artifact_to_create = { add_amulet_artifact_modifier_effect = { RECIPE = root.var:current_recipe } }
		}
	}
	if = {
		limit = { has_perk = create_object_perk }
		random = {
			chance = 5
			add_gold = ingredient_total_cost
		}
	}
	POD_achiev_mummytutorial_check = yes
}

get_amulet_artifact_quality_effect = {
	save_scope_value_as = {
		name = quality
		value = {
			value = amulets_craft_outcome
			min = 1
		}
	}
}

get_effigy_artifact_quality_effect = {
	save_scope_value_as = {
		name = quality
		value = {
			value = effigy_craft_outcome
			min = 1
		}
	}
}

get_sekhem_vessel_material_quality_effect = {
	ordered_character_artifact = {
		limit = { has_variable = sekhem_vessel }
		order_by = artifact_rarity
		position = 0
		POD_copy_artifact_rarity_effect = { TARGET = this }
	}
}

get_chattel_material_quality_effect = {
	ordered_character_artifact = {
		limit = { 
			var:chattel ?= flag:$EFFIGY$ 
			var:active_effigy = no
		}
		alternative_limit = { var:chattel ?= flag:$EFFIGY$ }
		order_by = artifact_rarity
		position = 0
		POD_copy_artifact_rarity_effect = { TARGET = this }
	}
}

create_artifact_ward_effect = {
	get_amulet_artifact_quality_effect = yes
	get_artifact_wealth_effect = yes

	create_artifact = {	
		name = $NAME$
		creator = scope:creator
		description = artifact_ward_description
		visuals = brooch
		type = miscellaneous
		modifier = artifact_placeholder_modifier
		wealth = scope:wealth
		quality = scope:quality
		save_scope_as = newly_created_artifact
	}

	hidden_effect_new_object = {
		scope:newly_created_artifact = {
