POD_smith_initiate_effect = {
	set_variable = { name = current_recipe value = scope:recipe }
	POD_ingredient_clear_effect = yes
	add_to_variable_list = { name = ingredient_list target = flag:water }
	add_to_variable_list = { name = ingredient_list target = flag:fuel }
	add_to_variable_list = { name = ingredient_list target = flag:mineral }
	switch = {
		trigger = scope:recipe
		flag:artifact_empower = {
			add_to_variable_list = { name = ingredient_list target = flag:common_artifact_piece }
			add_to_variable_list = { name = alt_ingredient_list target = flag:masterwork_artifact_piece }
			add_to_variable_list = { name = alt_ingredient_list target = flag:famed_artifact_piece }
		}
		flag:artifact_shatter = {
			add_to_variable_list = { name = ingredient_list target = flag:illustrious_artifact_piece }
			add_to_variable_list = { name = alt_ingredient_list target = flag:masterwork_artifact_piece }
			add_to_variable_list = { name = alt_ingredient_list target = flag:famed_artifact_piece }
		}
		fallback = {
			if = {
				limit = { is_target_in_variable_list = { name = ingredient_list target = flag:water } }
				add_to_variable_list = { name = alt_ingredient_list target = flag:exotic_water }
				add_to_variable_list = { name = alt_ingredient_list target = flag:magic_water }
			}
			if = {
				limit = { is_target_in_variable_list = { name = ingredient_list target = flag:fuel } }
				add_to_variable_list = { name = alt_ingredient_list target = flag:exotic_fuel }
				add_to_variable_list = { name = alt_ingredient_list target = flag:magic_fuel }
			}
			if = {
				limit = { is_target_in_variable_list = { name = ingredient_list target = flag:mineral } }
				add_to_variable_list = { name = alt_ingredient_list target = flag:exotic_mineral }
				add_to_variable_list = { name = alt_ingredient_list target = flag:magic_mineral }
			}
		}
	}
}

POD_smith_create_effect = {
	if = {
		limit = { smith_time > 0 }
		custom_tooltip = {
			text = smith_time_tt
			set_variable = { name = smith_cooldown days = smith_time }
		}
	}
	custom_tooltip = smith_quality_tooltip
	remove_short_term_gold = ingredient_total_cost
	POD_smith_create_artifact_effect = { RECIPE = var:current_recipe }
	show_as_tooltip = {
		if = {
			limit = { rarity_to_create >= 180 }
			global_var:illustrious_artifact_to_create = { add_amulet_artifact_modifier_effect = { RECIPE = root.var:current_recipe } }
		}
		else_if = {
			limit = { rarity_to_create >= 130 }
			global_var:famed_artifact_to_create = { add_amulet_artifact_modifier_effect = { RECIPE = root.var:current_recipe } }
		}
		else_if = {
			limit = { rarity_to_create >= 50 }
			global_var:masterwork_artifact_to_create = { add_amulet_artifact_modifier_effect = { RECIPE = root.var:current_recipe } }
		}
		else = {
			global_var:common_artifact_to_create = { add_amulet_artifact_modifier_effect = { RECIPE = root.var:current_recipe } }
		}
	}
}

get_amulet_artifact_quality_effect = {
	save_scope_value_as = {
		name = quality
		value = {
			value = amulets_craft_outcome
			min = 1
		}
	}
}

create_artifact_amulet_effect = {
	get_amulet_artifact_quality_effect = yes
	get_artifact_wealth_effect = yes

	create_artifact = {	
		name = current_forge_name
		creator = scope:creator
		description = current_forge_description
		visuals = necklace
		type = miscellaneous
		modifier = artifact_placeholder_modifier
		wealth = scope:wealth
		quality = scope:quality
		save_scope_as = newly_created_artifact
	}

	hidden_effect_new_object = {
		scope:newly_created_artifact = {
			remove_artifact_modifier = artifact_placeholder_modifier
			set_max_durability = amulet_durability_value
			set_variable = { name = amulet value = root.var:current_recipe }
			add_amulet_artifact_modifier_effect = { RECIPE = root.var:current_recipe }
		}
	}
}
