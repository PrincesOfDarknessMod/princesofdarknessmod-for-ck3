change_hand_resistance_effect = {
	change_variable = {
		name = hand_resistance
		add = hand_resistance_value
	}
	if = {
		limit = { var:hand_resistance < 0 }
		set_variable = {
			name = hand_resistance
			value = 0
		}
	}
	else_if = {
		limit = { var:hand_resistance > 100 }
		set_variable = {
			name = hand_resistance
			value = { 0 75 }
		}
		hand_exposed_effect = yes
	}
}

refresh_hand_modifier_effect = {
	switch = {
		trigger = var:hand_status
		3 = {
			remove_hand_modifier_effect = yes
			set_hand_modifier_3_effect = yes 
		}
		2 = {
			remove_hand_modifier_effect = yes
			set_hand_modifier_2_effect = yes 
		}
		1 = {
			remove_hand_modifier_effect = yes
			set_hand_modifier_1_effect = yes 
		}
		fallback = { remove_hand_modifier_effect = yes }
	}
}

		name = hand_status

		years = 20
	}
	switch = {
		trigger = this
		religion:sabbat_religion = {
			religion:sabbat_religion.var:regent ?= {
