POD_character_new_dynasty_effect = {
	if = { 
		limit = { NOT = { exists = dynasty } }
		POD_character_force_new_dynasty_effect = yes
	}
}

POD_character_new_dynasty_based_on_existing_house_effect = {
	house ?= { save_scope_as = old_house }
	POD_character_force_new_dynasty_effect = yes
	hidden_effect_new_object = {
		if = {
			limit = { exists = scope:old_house }
			dynasty ?= {
				set_coa = scope:old_house
				#set_house_name_from_house = scope:old_house
			}
			house ?= {
				set_coa = scope:old_house
				set_house_name_from_house = scope:old_house
				set_variable = {
					name = color_root
					value = this
				}
				POD_apply_saved_clothing_colors = {
					VAR = scope:old_house
				}
				remove_variable = color_root
			}
		}
	}
}

POD_character_force_new_dynasty_effect = {
	save_scope_as = characterneedsnewdynasty
	scope:characterneedsnewdynasty = {	
		if = {
			limit = { NOT = { exists = location } }
			
			random_independent_ruler = {
				limit = {
					exists = capital_province
					capital_province = { local_pool_is_full_trigger = no }
				}
				alternative_limit = { exists = capital_province }
				capital_province = { save_scope_as = destination }
			}
		}
		else = {
			scope:characterneedsnewdynasty.location = { save_scope_as = destination }
		}
		
		create_character = {
			save_temporary_scope_as = dummynewdynast
			location = scope:destination
			culture = scope:characterneedsnewdynasty.culture
			faith = scope:characterneedsnewdynasty.faith
			gender = male
			random_traits = yes
			dynasty = generate
		}
		
		hidden_effect_new_object = {
			if = {
				limit = { POD_is_vampire_trigger = yes }
				random_character_trait = {
					limit = { has_trait_flag = vamp_clan }
					save_scope_as = dynast_clan
				}
				scope:dummynewdynast = {
					add_trait = vampire
					add_trait = scope:dynast_clan
				}
			}
			else_if = {
				limit = { POD_is_supehunter_trigger = yes }
				scope:dummynewdynast = { add_trait = supehunter }
			}
			else_if = {
				limit = { has_trait = kinfolk }
				scope:dummynewdynast = { add_trait = kinfolk }
			}
			else_if = {
				limit = { has_trait = ghoul }
				scope:dummynewdynast = { add_trait = ghoul }
			}
			else = {
				scope:dummynewdynast = { add_trait = mortal }
			}
			
			scope:characterneedsnewdynasty = {
				set_house = scope:dummynewdynast.house
			}
			scope:dummynewdynast = {
				#set_house = house:house_ald
				if = {
					limit = { is_alive = yes }
					death = { death_reason = death_disappearance }
				}
			}
		}
	}
}
