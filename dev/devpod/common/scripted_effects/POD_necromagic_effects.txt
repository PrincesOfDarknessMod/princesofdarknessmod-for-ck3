POD_necromagic_ritual_start = {
	custom_tooltip = {
		text = POD_START_CASTING_RITUAL
		POD_start_targeted_journey_effect = {
			TYPE	= ritual_necromagic_casting
			MONTHLY = ritual_necromagic_casting
			TARGET  = scope:target
		}
		var:current_journey ?= {
			set_variable = { name = ritual value = flag:$RITUAL$ }
			set_variable = { name = ritual_monthly_progress value = $RITUAL_SPEED$ }
		}
	}
	

	custom_tooltip = POD_bloodmagic_ritual_start_tt
	show_as_tooltip = { POD_$RITUAL$_cast_effect = { TARGET = scope:target } }
}

POD_necromagic_ritual_location_start = {
	custom_tooltip = {
		text = POD_START_CASTING_RITUAL
		POD_start_targeted_journey_effect = {
			TYPE	= ritual_necromagic_casting

}

POD_the_ferrymans_recall_cast_effect = {
	add_piety = -500
	POD_static_stress_gain = { COUNT = 75 }
	POD_discipline_cost_effect = { COUNT = ns_ritual_the_ferrymans_recall_cost }

	POD_lose_golconda_in_storyteller_mode_effect = yes
	if = {
		limit = {
			faith = {
				OR = {
					has_doctrine_parameter = vampirism_illegal
					has_doctrine_parameter = hurt_mortal_lose_piety
				}
			}
		}
		add_piety = massive_piety_loss
	}
	else_if = {
		limit = {
			faith = {
				has_doctrine_parameter = vampirism_shunned
			}
		}
		add_piety = major_piety_loss
	}
	else = {
		add_piety = medium_piety_loss
	}
	$TARGET$ = {
		POD_remove_wraith_traits_effect = yes
		POD_remove_wraith_arcanoi_traits_effect = yes
		POD_remove_splat_traits_effect = yes
		save_scope_as = created_vampirechilde
	}
	save_scope_as = vampiresire
	POD_player_character_create_vampirechilde_effect = yes
	
	if = {
		limit = { POD_has_a_beast_trigger = yes }
		custom_tooltip = POD_embrace.100.option.a.tt.beast
	}
}
