# used by common\scripted_guis\POD_portrait_shader_guis.txt
POD_toggle_portrait_shader_var = {
	if = {
		limit = { has_variable = $VAR$ }
		remove_variable = $VAR$
	}
	else = {
		set_variable = {
			name  = $VAR$
			value = 1.0 # maximum value that shaders can read
		}
	}
}

# used by common\scripted_guis\POD_portrait_shader_guis.txt
# need to pass scope:delta to the SGUI
POD_edit_portrait_shader_var = {
	if = {
		limit = { NOT = { has_variable = $VAR$ } }
		set_variable = { name = $VAR$ value = 0.5 }
	}
	set_variable = {
		name = $VAR$
		value = {
			value = var:$VAR$
			add = scope:delta
			min = @ZERO
			max = 1.0
		}
	}
}

		if = {
			limit = { has_variable = $VAR$ }

			remove_variable = $VAR$
		}
		create_story = {
			type = POD_story_colors
			save_scope_as = saved_color_scheme
		}
	}
	hidden_effect_new_object = {
		scope:player = {
			set_variable = {

			}
		}
		POD_clothing_color_injector = {
			EFFECT = POD_backup_clothing_color
			TARGET_SCOPE = scope:saved_color_scheme
		}
	}
}

		create_story = {
			type = POD_story_colors
			save_scope_as = $SAVE_AS$
		}
	}
	hidden_effect_new_object = {
		scope:$SAVE_AS$ = {
			# TODO: doesn't work if scope:target is someone other than the player
			if = {
				limit = { prev.var:color_root ?= { has_variable = POD_portrait_clothing_$CHANNEL$_r } }
				set_variable = {
					name  = POD_portrait_clothing_r
					value = prev.var:color_root.var:POD_portrait_clothing_$CHANNEL$_r
				}
			}
			if = {
				limit = { prev.var:color_root ?= { has_variable = POD_portrait_clothing_$CHANNEL$_g } }
				set_variable = {
					name  = POD_portrait_clothing_g
					value = prev.var:color_root.var:POD_portrait_clothing_$CHANNEL$_g
				}
			}
			if = {
				limit = { prev.var:color_root ?= { has_variable = POD_portrait_clothing_$CHANNEL$_b } }
				set_variable = {
					name  = POD_portrait_clothing_b
					value = prev.var:color_root.var:POD_portrait_clothing_$CHANNEL$_b
				}
			}
		}
	}
}
