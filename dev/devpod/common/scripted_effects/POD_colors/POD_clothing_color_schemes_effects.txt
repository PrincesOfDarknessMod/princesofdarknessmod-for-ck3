# Values of 0 stop the decal from showing up, so this is necessary
@ZERO = 0.004

# Layout
#
# Channel 1 is the primary fabric color of an outfit.
#
# Monk cassocks use channel 2. So if you want those cassocks black, use:
#	POD_set_clothing_color = {
#		CHANNEL = 2
#		R = 0.02
#		G = 0.02
#		B = 0.02
#	}
#
# Leather on Byzantine armor also uses channel 2. Here is a brown:
#	POD_set_clothing_color = {
#		CHANNEL = 2
#		R = 0.6
#		G = 0.23
#		B = @ZERO
#	}
#
# Material under the Byzantine scale armor uses Channel 3.
#
# The color of chainmail is channel 4. To keep that looking shiny, use:
#	POD_set_clothing_color = {
#		CHANNEL = 4
#		R = 0.95
#		G = 0.95
#		B = 0.95
#	}
#
# The color of alternating scales on the Byzantine armor is on channel 5. To keep that looking shiny, use:
#	POD_set_clothing_color = {
#		CHANNEL = 5
#		R = 0.95
#		G = 0.95
#		B = 0.95
#	}
#
