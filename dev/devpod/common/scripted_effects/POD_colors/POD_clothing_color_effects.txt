# Values of 0 stop the decal from showing up, so this is necessary
@ZERO = 0.004

# Generic macro (edit this in case Paradox adds a native implementation of scriptable clothing colors)
POD_set_clothing_color = {
	set_variable = { name = POD_portrait_clothing_$CHANNEL$_r value = $R$ }
	set_variable = { name = POD_portrait_clothing_$CHANNEL$_g value = $G$ }
	set_variable = { name = POD_portrait_clothing_$CHANNEL$_b value = $B$ }
}

# Lets make a common palette of colors
# Tincture Colors according to https://www.crwflags.com/fotw/flags/vxt-h1.html
# Black/Sable: (0,0,0) = RGB decimal of 255 (0, 0, 0)
# White: (255,255,255) = RGB decimal of 255 (1.0, 1.0 , 1.0)
# Red/Gules: RGB (255,0,0) = RGB decimal of 255 (1.0, 0, 0)
# Blue/Azure: RGB(0,0,255) = RGB decimal of 255 (0, 0, 1.0)
# Sky Blue/Bleu Celest: RGB(150,200,250) = RGB decimal of 255 (0.59, 0.78, 0.98)
# Silver/Argent: RGB (240,240,240) = RGB decimal of 255 ( 0.94, 0.94, 0.94)
# Gold/Or: RGB (255,240,240) = RGB decimal of 255 (1.0, 0.94 , 0.94)
# Green/Vert: RGB (0,150,0) = RGB decimal of 255 (0, 0.59, 0)
# Purple/Purpure: RGB (170,0,170) = RGB decimal of 255 (0.67, 0, 0.67)
# Mulberry/Murrey: RGB (140,0,75) = RGB decimal of 255 (0.55, 0, 0.29)
# Blood Red/Sanguine: RGB (190,0,0) = RGB decimal of 255 (0.75, 0, 0)
# Tawny/Tenné: RGB (250,150,0) = RGB decimal of 255 (0.98, 0.59, 0)
