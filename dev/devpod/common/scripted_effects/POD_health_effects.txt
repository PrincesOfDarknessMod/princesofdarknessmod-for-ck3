POD_remove_major_injury_traits_effect = { # Disfigured and scar halded other wise
	if = {
		limit = {
			has_trait = maimed
			NOT = { has_character_flag = POD_permanent_maimed }
		}
		remove_trait = maimed
	}

	if = {
		limit = {
			has_trait = one_legged
			NOT = { has_character_flag = POD_permanent_one_legged }
		}
		remove_trait = one_legged
	}

	if = {
		limit = {
			has_trait = eunuch_1
			NOT = { has_character_flag = POD_permanent_eunuch }
		}
		remove_trait = eunuch_1
	}

	if = {
		limit = {
			has_trait = one_eyed
			NOT = { has_character_flag = POD_permanent_one_eyed }
		}
		remove_trait = one_eyed
	}

	if = {
		limit = {
			has_trait = blind
			NOT = { has_character_flag = POD_permanent_blind }
		}
		remove_trait = blind
		add_trait = one_eyed
	}

	POD_remove_character_modifier = { MODIFIER = infected_wound_modifier }
	POD_remove_character_modifier = { MODIFIER = gangrene_modifier }
}

	if = {

no_blood_left_effect = {
	if = {
		limit = { # No double death
			OR = {
				exists = scope:execution_exsanguination
				exists = scope:execution_diablerie
			}
		}
	}
	else_if = {
		limit = {
			OR = {
				has_trait = vampire
				has_trait = kueijin
				has_trait = wraith
				has_trait = spirit
			}
		}
		add_trait = torpor
		POD_depose_effect = yes
	}
	else = {
		if = {
			limit = { exists = scope:bloodloss_attacker }
			pod_death = {
				DEATH_REASON = death_bloodloss
				KILLER = scope:bloodloss_attacker
			}
		}
		else = {
			pod_death_no_killer = {
				DEATH_REASON = death_bloodloss
			}
		}
	}
}
