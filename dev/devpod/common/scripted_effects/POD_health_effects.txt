# PoD Health System

POD_apply_normal_damage_effect = {
	save_scope_value_as = { name = amount value = { value = $COUNT$ max = POD_health_damage_rest } }
	custom_description = {
		text = POD_apply_normal_damage_effect
		value = scope:amount
		set_variable = { 
			name = POD_health_damage_num
			value = { 
				value = POD_health_normal_damage 
				add = scope:amount
			}
		}
	}
	POD_update_damage_effect = yes
}

POD_apply_aggravated_damage_effect = {
	save_scope_value_as = { name = amount value = { value = $COUNT$ max = { value = POD_health_damage_limit subtract = POD_health_aggravated_damage } } }
	custom_description = {
		text = POD_apply_aggravated_damage_effect
		value = scope:amount
		set_variable = { 
			name = POD_health_aggravated_damage_num
			value = { 
				value = POD_health_aggravated_damage 
				add = scope:amount
			}
		}
		while = { 
			limit = { POD_health_damage_rest < 0 }  
			set_variable = { 
				name = POD_health_damage_num
				value = { 
					value = POD_health_normal_damage 
					add = -1
				}
			}
		}
	}
	POD_update_damage_effect = yes
}

POD_update_damage_effect = {
	hidden_effect = {
		remove_trait = POD_wounded
		remove_trait = wounded_1
		remove_trait = wounded_2
		remove_trait = wounded_3
		if = { limit = { POD_health_value >= 80 }
			add_trait = wounded_3
			add_trait_xp = { trait = wounded_3 value = POD_health_value }
		}
		else_if = { limit = { POD_health_value >= 40 }
			add_trait = wounded_2
			add_trait_xp = { trait = wounded_2 value = POD_health_value }
		}
		else_if = { limit = { POD_health_value >= 20 }
			add_trait = wounded_1
			add_trait_xp = { trait = wounded_1 value = POD_health_value }
		}
		else_if = { limit = { POD_health_value >= 1 }
			add_trait = POD_wounded
			add_trait_xp = { trait = POD_wounded value = POD_health_value }
		}

		if = { limit = { var:POD_health_damage_num ?= 0 } remove_variable = POD_health_damage_num }
		if = { limit = { var:POD_health_aggravated_damage_num ?= 0 } remove_variable = POD_health_aggravated_damage_num }
	}
}

POD_health_death_check_effect = {
	show_as_tooltip = {
		if = { limit = { POD_can_die_from_wounds = yes POD_health_damage_rest_tt <= 0 } 
			if = { 
				limit = { 
					is_immortal = yes 
					has_POD_resurrection_trigger = no 
					NOT = {
						trigger_if = { limit  = { flag:$TYPE$ = flag:normal } POD_health_aggravated_damage >= POD_health_damage_limit_half }
						trigger_else = { POD_health_aggravated_damage_tt >= POD_health_damage_limit_half }
					}
				} 
				POD_long_depose_effect = yes 
			}
			else_if = { limit = { exists = scope:attacker } pod_death = { KILLER = scope:attacker DEATH_REASON = death_$REASON$ } }
			else = { pod_death_no_killer = { DEATH_REASON = death_$REASON$ } }
		}
	}
	hidden_effect = {
		if = { limit = { POD_can_die_from_wounds = yes POD_health_damage_rest <= 0 } 
			if = { 
				limit = { 
					is_immortal = yes 
					has_POD_resurrection_trigger = no 
					NOT = {
						trigger_if = { limit  = { flag:$TYPE$ = flag:normal } POD_health_aggravated_damage >= POD_health_damage_limit_half }
						trigger_else = { POD_health_aggravated_damage_tt >= POD_health_damage_limit_half }
					}
				} 
				POD_long_depose_effect = yes 
			}
			else_if = { limit = { exists = scope:attacker } pod_death = { KILLER = scope:attacker DEATH_REASON = death_$REASON$ } }
			else = { pod_death_no_killer = { DEATH_REASON = death_$REASON$ } }
		}
	}
}

POD_increase_wounds_effect = {
	save_temporary_scope_value_as = {
		name = reason_type
		value = flag:$REASON$ 
	}
	if = { 
		limit = { any_equipped_character_artifact = { has_variable = clay_man_protection } }
		send_interface_toast = {
			title = clay_man_amulet_soaked_damage
			left_icon = root
			custom_tooltip = clay_man_amulet_soaked_damage_ct
			random_equipped_character_artifact = { 
				limit = { has_variable = clay_man_protection }
				change_variable = {
					name = clay_man_protection
					add = -1
				}
				if = {
					limit = { var:clay_man_protection <= 0 }
					destroy_artifact = this
				}
				else = {
					save_temporary_scope_as = clay_man
					set_artifact_name = artifact_clay_man_amulet_name
				}
			}
		}
	}
	else_if = {
		limit = { scope:reason_type = flag:fire POD_immune_to_fire_trigger = yes }
	}
	else_if = {
		limit = { NOT = { POD_soak_chance = 0 } $SOAKABLE$ = yes }
		random_list = {
			0 = {
				modifier = { add = POD_soak_chance }
			}
			100 = {
				modifier = { add = { value = POD_soak_chance multiply = -1 } }
				POD_apply_$TYPE$_damage_effect = { COUNT = $COUNT$ }
				POD_health_death_check_effect = { REASON = $REASON$ TYPE = $TYPE$ }
			}
		}
	}
	else_if = {
		limit = { 
			scope:reason_type = flag:sun
			OR = {
				has_trait = setite 
				is_bushi_trigger = yes
			}
			POD_suppress_clan_curse_trigger = no 
		}
		save_scope_value_as = { name = sun_curse value = { value = $COUNT$ multiply = 2 } }
		POD_apply_$TYPE$_damage_effect = { COUNT = scope:sun_curse }
		POD_health_death_check_effect = { REASON = $REASON$ TYPE = $TYPE$ }
	}
	else_if = {
		limit = { 
			POD_death_fire_trigger = { DEATH_REASON = scope:reason_type }
			OR = {
				is_bushi_trigger = yes
			}
			POD_suppress_clan_curse_trigger = no 
		}
		save_scope_value_as = { name = fire_curse value = { value = $COUNT$ multiply = 2 } }
		POD_apply_$TYPE$_damage_effect = { COUNT = scope:fire_curse }
		POD_health_death_check_effect = { REASON = $REASON$ TYPE = $TYPE$ }
	}
	else = {
		POD_apply_$TYPE$_damage_effect = { COUNT = $COUNT$ }
		POD_health_death_check_effect = { REASON = $REASON$ TYPE = $TYPE$ }
	}
	
	if = { 
		limit = { has_perk = pain_strength_perk }
		while = { 
			count = $COUNT$
			add_character_modifier = { 
				modifier = pain_strength_perk_modifier
				years = 5
			}
		}
	}

	# vanilla
	if = { #Wounds from treatments give no infection and no additional treatment
		limit = { NOT = { scope:reason_type = flag:treatment } }

		#CHANCE OF INFECTION
		hidden_effect = {
			random = {
				chance = 10
				trigger_event = {
					id = health.0201
					days = { 30 60 }
				}
			}
		}

		#HANDLE TREATMENT
		if = { #To send notification message or trigger the right event
			limit = {
				OR = {
					has_trait = POD_wounded
					has_trait = wounded_1
				}
				court_physician_available_trigger = yes
			}
			save_scope_as = sick_character
			safe_wound_treatment_effect = yes #Wounded 1 only gives a "result of treatment" notification" if you have a physician, i.e. you get no events about it.
		}
		else_if = {
			limit = {
				has_trait = wounded_2
				has_recent_wound_treatment_trigger = no
			}
			#A "real" event if you're not receiving treatment
			#Nothing happens if you're already being treated for wounds
			trigger_event = {
				id = health.0102
				days = { 2 3 }
			}
		}
		else_if = {
			limit = {
				has_trait = wounded_3
				has_recent_wound_treatment_trigger = no
			}
			#A "real" event if you're not receiving treatment
			#Nothing happens if you're already being treated for wounds
			trigger_event = {
				id = health.0104
				days = { 2 3 }
			}

			#Epilepsy risk
			epilepsy_brain_trauma_risk_effect = { CHANCE = 5 }
		}
	}
	if = { limit = { exists = flag:$REASON$ } }
}
