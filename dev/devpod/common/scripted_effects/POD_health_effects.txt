no_health_left_effect = {
	if = {
		limit = {
			has_character_flag = POD_in_execution_flag
		}
		remove_character_flag = POD_in_execution_flag
	}
	else_if = {
		limit = {
			OR = {
				has_trait = vampire
				has_trait = kueijin
				has_trait = wraith
			}
		}
		add_trait = torpor
		if = {
			limit = { is_landed = yes }
			depose = yes
		}
	}
	else = {
		pod_death_without_killer_effect = {
			DEATH_REASON = death_lost_all_life_force
		}
	}
}
