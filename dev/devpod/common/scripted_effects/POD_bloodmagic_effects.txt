POD_bloodmagic_ritual_start = {
	custom_tooltip = {
		text = POD_START_CASTING_RITUAL
		POD_start_targeted_journey_effect = {
			TYPE	= ritual_blood_sorcery_casting
			MONTHLY = ritual_blood_sorcery_casting
			TARGET  = scope:target
		}
		var:current_journey ?= {
			set_variable = { name = ritual value = flag:$RITUAL$ }
			set_variable = { name = ritual_monthly_progress value = $RITUAL_SPEED$ }
		}
	}
	

	custom_tooltip = POD_bloodmagic_ritual_start_tt
	show_as_tooltip = { POD_$RITUAL$_cast_effect = { TARGET = scope:target } }
}

# Abyss Mysticism content: The Abyssal Blade (T2).
POD_abyssal_blade_cast_effect = {
	POD_static_stress_gain = { COUNT = bs_abyss_ritual_T2_cost }
	$TARGET$ = {
		random_equipped_character_artifact = {
			# refuse already-steeped steel: a weapon swap between ritual start and completion
			# must not re-halve a claimed blade (the SGUI gate is not re-checked at fire time)
			limit = {
				artifact_slot_type = primary_armament
				NOT = { has_artifact_modifier = abyssal_blade_modifier }
			}
			save_scope_as = abyss_mysticism_abyssal_blade
			set_max_durability = {
				value = artifact_max_durability
				divide = 2
			}
			# the engine does not auto-clamp current durability when max drops (M0-7): land it on the new max
			if = {
				limit = { artifact_durability > artifact_max_durability }
				add_durability = {
					value = artifact_max_durability
					subtract = artifact_durability
				}
			}
		}
		if = {
			# exists also covers a blade shattered by the clamp itself (max 1 halves to 0)
			limit = { exists = scope:abyss_mysticism_abyssal_blade }
			# enchanting pattern (POD_crafting_core_effects.txt): the boon lives ON the artifact
			scope:abyss_mysticism_abyssal_blade = {
				add_artifact_modifier = abyssal_blade_modifier
			}
		}
		else = {
			custom_tooltip = POD_abyss_mysticism_abyssal_blade_no_weapon_tt
		}
	}
}

# Abyss Mysticism content: One with the Dark (T3).
POD_one_with_the_dark_cast_effect = {
	POD_static_stress_gain = { COUNT = bs_abyss_ritual_T3_cost }
	$TARGET$ = { save_scope_as = abyss_mysticism_dark_read_target }
	if = {
		limit = {
			scope:abyss_mysticism_dark_read_target = {
				any_secret = { NOT = { is_known_by = root } }
			}
		}
		custom_tooltip = POD_abyss_mysticism_one_with_the_dark_reveal_tt
		hidden_effect = {
			scope:abyss_mysticism_dark_read_target = {
				random_secret = {
					limit = { NOT = { is_known_by = root } }
					reveal_to = root
				}
				random = {
					chance = 50
					if = {
						limit = { any_secret = { NOT = { is_known_by = root } } }
						random_secret = {
							limit = { NOT = { is_known_by = root } }
							reveal_to = root
						}
						# nested 50% of 50% = the specified 25% overall for a third
						random = {
							chance = 50
							if = {
								limit = { any_secret = { NOT = { is_known_by = root } } }
								random_secret = {
									limit = { NOT = { is_known_by = root } }
									reveal_to = root
								}
							}
						}
					}
				}
			}
		}
	}
	else = {
		# nothing hidden to read: the dark hands back leverage and technique instead
		custom_tooltip = POD_abyss_mysticism_one_with_the_dark_fallback_tt
		hidden_effect = {
			add_hook_if_possible_default_length_effect = {
				TARGET = scope:abyss_mysticism_dark_read_target
				TYPE = manipulation_hook
			}
			add_intrigue_lifestyle_xp = 100	# [TUNE]
		}
	}
	if = {
		limit = { NOT = { has_faith = faith:roadoftheabyss } }
		add_piety = -150
	}
}

POD_claiming_the_dark_cast_effect = {
	POD_static_stress_gain = { COUNT = bs_abyss_ritual_T4_cost }
	$TARGET$ = {
		# One claim, sized by mastery: the deepest tier the caster holds decides the
		# single modifier (no stacking). The release decision strips whichever is worn.
		if = {
			limit = { has_perk = ahrimans_demesne_perk }
			add_character_modifier = void_augmented_elder_modifier
		}
		else_if = {
			limit = { has_trait = obtenebrationadvanced }
			add_character_modifier = void_augmented_advanced_modifier
		}
		else = {
			add_character_modifier = void_augmented_modifier
		}
		# The claiming pours attunement into the track (+25; the dispatcher trickle adds
		# the ritual tier besides, +4 for this T4 rite). The 99-clamp holds level 3 behind path mastery: without
		# level 5 the chunk lands only while xp stays short of the last threshold.
		if = {
			limit = { NOT = { has_trait = abyss_touched } }
			add_trait = abyss_touched
		}
		if = {
			limit = {
				OR = {
					has_5_level_of_abyssal_mysticism_trigger = yes
					NOT = { has_trait_xp = { trait = abyss_touched value >= 75 } }
				}
			}
			add_trait_xp = {
				trait = abyss_touched
				value = 25
			}
		}
		POD_abyss_mysticism_abyss_touched_threshold_sync_effect = yes
	}
}
