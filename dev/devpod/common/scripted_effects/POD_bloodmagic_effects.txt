POD_bloodmagic_ritual_start = {
	custom_tooltip = {
		text = POD_START_CASTING_RITUAL
		POD_start_targeted_journey_effect = {
			TYPE	= ritual_blood_sorcery_casting
			MONTHLY = ritual_blood_sorcery_casting
			TARGET  = scope:target
		}
		var:current_journey ?= {
			set_variable = { name = ritual value = flag:$RITUAL$ }
			set_variable = { name = ritual_monthly_progress value = $RITUAL_HARDNESS$ }
		}
	}
	

	custom_tooltip = POD_bloodmagic_ritual_start_tt
	show_as_tooltip = { POD_$RITUAL$_cast_effect = { TARGET = scope:target } }
}

POD_bloodmagic_ritual_location_start = {
	custom_tooltip = {
		text = POD_START_CASTING_RITUAL
		POD_start_targeted_journey_effect = {
			TYPE	= ritual_blood_sorcery_casting
			MONTHLY = ritual_blood_sorcery_casting
			TARGET  = scope:target
		}
		var:current_journey ?= {
			set_variable = { name = ritual value = flag:$RITUAL$ }
			set_variable = { name = ritual_monthly_progress value = $RITUAL_HARDNESS$ }
			set_variable = { name = destination value = $LOCATION$ }
		}
	}
	

	custom_tooltip = POD_bloodmagic_ritual_start_tt
	show_as_tooltip = { POD_$RITUAL$_cast_effect = { TARGET = scope:target } }
}

	POD_static_stress_gain = { COUNT = 25 }
	$TARGET$ = {
		location = {
			duchy = {
				every_de_jure_county = {
					add_county_modifier = {
						modifier = ritual_cry_that_slays_light_modifier
						years = 20
					} 
					holder = { 
						if = { 
							limit = { NOT = { is_in_list = county_owners } } 
							add_to_list = county_owners 
						}
					}
				}
			}
		}
	}
	every_in_list = {
		list = county_owners
		if = {
			limit = {
				has_relation_rival = ROOT
				NOT = { has_relation_nemesis = ROOT }
				can_set_relation_nemesis_trigger = { CHARACTER = ROOT }
			}
			set_relation_nemesis = {
				reason = nemesis_downgrade_from_rival
				copy_reason = rival
				target = ROOT
			}
		}
		else_if = {
			limit = {
				NOT = { has_relation_rival = ROOT }
			}
			set_relation_rival = {
				target = ROOT
				reason = rival_grave_insult
			}
		}
	}
}
