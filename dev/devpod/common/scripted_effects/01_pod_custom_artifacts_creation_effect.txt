create_ananata_artifact_effect = {
	get_ananata_artifact_quality_effect = yes
	get_artifact_wealth_effect = yes
	
	create_artifact = {
		name = artifact_ananata_name
		creator = scope:creator
		description = artifact_ananata_description
		visuals = pocket_pouch
		type = miscellaneous
		modifier = artifact_scheme_resistance_add_2_modifier
		save_scope_as = newly_created_artifact
		wealth = scope:wealth
		quality = scope:quality
		#decaying = no
	}
	scope:newly_created_artifact = {
		if = {
			limit = {
				rarity = illustrious
			}
			add_artifact_modifier = artifact_character_travel_speed_4_modifier
			add_artifact_modifier = artifact_movement_speed_4_modifier
			add_artifact_modifier = artifact_embarkation_cost_mult_4_modifier
			add_artifact_modifier = artifact_no_water_crossing_penalty_modifier
			add_artifact_modifier = artifact_no_disembark_penalty_modifier
		}
		else_if = {
			limit = {
				rarity = famed
			}
			add_artifact_modifier = artifact_character_travel_speed_3_modifier
			add_artifact_modifier = artifact_movement_speed_3_modifier
			add_artifact_modifier = artifact_embarkation_cost_mult_3_modifier
		}
		else_if = {
			limit = {
				rarity = masterwork
			}
			add_artifact_modifier = artifact_character_travel_speed_2_modifier
			add_artifact_modifier = artifact_movement_speed_2_modifier
			add_artifact_modifier = artifact_embarkation_cost_mult_2_modifier
		}
		else_if = {
			limit = {
				rarity = common
			}
			add_artifact_modifier = artifact_character_travel_speed_1_modifier
			add_artifact_modifier = artifact_movement_speed_1_modifier
			add_artifact_modifier = artifact_embarkation_cost_mult_1_modifier
		}
		set_variable = ananata
	}
}

get_ananata_artifact_quality_effect = {
	hidden_effect = {
		random_list = {
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 0.5
				} 
			}
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 1
				} 
			}
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 2
				} 
			}
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 2.5
				} 
			}
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 3
				} 
			}
		}
	}
	
	# Calculate the final quality of the artifact
	save_scope_value_as = {
		name = quality
		value = {
			value = 0
			
			scope:creator = {
				# Bonus quality from high skills
				add = {	
					value = artisan_inspiration_average_skill_value
					multiply = {
						value = quality_bonus_per_skill_level_value
						add = scope:random_inspired_skill_quality_bonus
					}
				}
					

				add = {	
					if = {
						limit = { scope:monumental_gold = yes }
						value = 40
					}
					else_if = {
						limit = { scope:massive_gold = yes }
						value = 30
					}
					else_if = {
						limit = { scope:major_gold = yes }
						value = 20
					}
					else = {
						value = 10
					}
				}
			}
			
			# Can't have negative quality
			min = 1
		}
	}
}
