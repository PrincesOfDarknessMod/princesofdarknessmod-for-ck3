add_artifact_modifier_by_rarity_effect = {
	if = {
		limit = { rarity = illustrious }
		add_artifact_modifier = artifact_$MODIFIER$_4_modifier
	}
	else_if = {
		limit = { rarity = famed }
		add_artifact_modifier = artifact_$MODIFIER$_3_modifier
	}
	else_if = {
		limit = { rarity = masterwork }
		add_artifact_modifier = artifact_$MODIFIER$_2_modifier
	}
	else = {
		add_artifact_modifier = artifact_$MODIFIER$_1_modifier
	}
}

get_ananata_artifact_quality_effect = {
	hidden_effect = {
		random_list = {
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 0.5
				} 
			}
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 1
				} 
			}
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 2
				} 
			}
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 2.5
				} 
			}
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 3
				} 
			}
		}
	}
	
	# Calculate the final quality of the artifact
	save_scope_value_as = {
		name = quality
		value = {
			value = 0
			
			scope:creator = {
				# Bonus quality from high skills
				add = {	
					value = artisan_inspiration_average_skill_value
					multiply = {
						value = quality_bonus_per_skill_level_value
						add = scope:random_inspired_skill_quality_bonus
					}
				}
					

				add = {	
					if = {
						limit = { scope:monumental_gold = yes }
						value = 40
					}
					else_if = {
						limit = { scope:massive_gold = yes }
						value = 30
					}
					else_if = {
						limit = { scope:major_gold = yes }
						value = 20
					}
					else = {
						value = 10
					}
				}
			}
			
			# Can't have negative quality
			min = 1
		}
	}
}
