create_soulforged_artifact_effect = {
	get_soulforged_artifact_quality_effect = yes
	get_artifact_wealth_effect = yes
	
	create_artifact = {
		name = artifact_soulforged_artifact_name
		creator = scope:creator
		description = artifact_soulforged_artifact_description
		visuals = $VISUALS$
		type = $TYPE$
		modifier = artifact_stress_gain_1_penalty_modifier
		save_scope_as = newly_created_artifact
		wealth = scope:wealth
		quality = scope:quality
		#decaying = no
	}
	scope:newly_created_artifact = {
		if = {
			limit = {
				scope:owner = { has_royal_court = yes }
			}
			add_scaled_artifact_modifier_grandeur_small_effect = yes
		}
		add_scaled_artifact_modifier_minor_prestige_effect = yes
		if = {
			limit = {
				rarity = illustrious
			}
			add_artifact_modifier = artifact_dread_baseline_add_2_modifier
		}
		else_if = {
			limit = {
				rarity = famed
			}
			add_artifact_modifier = artifact_dread_baseline_add_1_modifier
		}
		add_soulforged_artifact_modifier_effect = { KILLED = scope:killed }
		set_variable = possessed_artifact
	}
}

get_soulforged_artifact_quality_effect = {
	hidden_effect = {
		save_scope_value_as = {
			name = random_quality_bonus
			value = 0
		}
		save_scope_value_as = {
			name = random_inspired_skill_quality_bonus
			value = 0
		}
		
		if = {
			limit = { employs_court_position = artificer_journeymen_court_position }
			random_court_position_holder = {
				type = artificer_journeymen_court_position
				save_scope_as = artificer_journeymen
			}
			if = {
				limit = { scope:artificer_journeymen.aptitude:artificer_journeymen_court_position = 0 }
				save_scope_value_as = {
					name = random_quality_bonus
					value = 4
				} 
			}
			else_if = {
				limit = { scope:artificer_journeymen.aptitude:artificer_journeymen_court_position = 1 }
				save_scope_value_as = {
					name = random_quality_bonus
					value = 8
				} 
			}
			else_if = {
				limit = { scope:artificer_journeymen.aptitude:artificer_journeymen_court_position = 2 }
				save_scope_value_as = {
					name = random_quality_bonus
					value = 12
				} 
			}
			else_if = {
				limit = { scope:artificer_journeymen.aptitude:artificer_journeymen_court_position = 3 }
				save_scope_value_as = {
					name = random_quality_bonus
					value = 16
				} 
			}
			else_if = {
				limit = { scope:artificer_journeymen.aptitude:artificer_journeymen_court_position = 4 }
				save_scope_value_as = {
					name = random_quality_bonus
					value = 20
				} 
			}
		}

		random_list = {
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 0.5
				} 
			}
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 1
				} 
			}
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 2
				} 
			}
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 2.5
				} 
			}
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 3
				} 
			}
		}
	}
	
	# Calculate the final quality of the artifact
	save_scope_value_as = {
		name = quality
		value = {
			value = scope:random_quality_bonus
			
			scope:killed = {
				# Bonus quality from high skills
				add = {	
					if = {
						limit = { scope:armor = yes }
						value = armor_inspiration_average_skill_value
					}
					else_if = {
						limit = { scope:weapon = yes }
						value = weapon_inspiration_average_skill_value
					}
					else_if = {
						limit = { scope:shinies = yes }
						value = smith_inspiration_average_skill_value
					}
					else_if = {
						limit = { scope:book = yes }
						value = book_inspiration_average_skill_value
					}
					else_if = {
						limit = { scope:furniture = yes }
						value = weaver_inspiration_average_skill_value
					}
					else_if = {
						limit = { scope:trinket = yes }
						value = artisan_inspiration_average_skill_value
					}
					multiply = {
						value = quality_bonus_per_skill_level_value
						add = scope:random_inspired_skill_quality_bonus
					}
				}

				add = {	
					if = {
						limit = {
							OR = {
								has_trait = education_martial_5
								has_trait = education_diplomacy_5
								has_trait = education_learning_5
								has_trait = education_stewardship_5
								has_trait = education_intrigue_5
								has_trait = education_martial_prowess_5
							}
						}
						value = 25
					}
					else_if = {
						limit = {
							OR = {
								has_trait = education_martial_4
								has_trait = education_diplomacy_4
								has_trait = education_learning_4
								has_trait = education_stewardship_4
								has_trait = education_intrigue_4
								has_trait = education_martial_prowess_4
							}
						}
						value = 20
					}
					else_if = {
						limit = {
							OR = {
								has_trait = education_martial_3
								has_trait = education_diplomacy_3
								has_trait = education_learning_3
								has_trait = education_stewardship_3
								has_trait = education_intrigue_3
								has_trait = education_martial_prowess_3
							}
						}
						value = 15
					}
					else_if = {
						limit = {
							OR = {
								has_trait = education_martial_2
								has_trait = education_diplomacy_2
								has_trait = education_learning_2
								has_trait = education_stewardship_2
								has_trait = education_intrigue_2
								has_trait = education_martial_prowess_2
							}
						}
						value = 10
					}
					else = {
						value = 5
					}
				}
				
				if = {
					limit = { scope:creator = { has_trait = necromancyadvanced } }
					add = 10
				}
				else_if = {
					limit = { scope:creator = { has_trait = necromancy } }
					add = 5
				}
				
				if = {
					limit = { scope:creator = { has_trait = oblivionadvanced } }
					add = 10
				}
				else_if = {
					limit = { scope:creator = { has_trait = oblivion } }
					add = 5
				}
				
				if = {
					limit = { scope:creator = { has_perk = forge_affinity_perk } }
					add = 20
				}
				else_if = {
					limit = { scope:artificer_journeymen ?= { has_perk = forge_affinity_perk } }
					add = 20
				}
			}
			
			# Can't have negative quality
			min = 1
		}
	}
}

create_fleshcrafted_artifact_effect = {
	get_fleshcrafted_artifact_quality_effect = yes
	get_artifact_wealth_effect = yes
	
	create_artifact = {
		name = artifact_fleshcrafted_artifact_name
		creator = scope:creator
		description = artifact_fleshcrafted_artifact_description
		visuals = $VISUALS$
		type = $TYPE$
		modifier = artifact_general_opinion_penalty_2_modifier
		modifier = artifact_roadofmetamorphosis_religion_opinion_modifier
		modifier = artifact_tyranny_gain_penalty_1_modifier
		save_scope_as = newly_created_artifact
		wealth = scope:wealth
		quality = scope:quality
		#decaying = no
	}
	scope:newly_created_artifact = {
		if = {
			limit = {
				scope:owner = { has_royal_court = yes }
			}
			add_scaled_artifact_modifier_grandeur_small_effect = yes
		}
		add_scaled_artifact_modifier_minor_prestige_effect = yes
		if = {
			limit = {
				rarity = illustrious
			}
			add_artifact_modifier = artifact_dread_baseline_add_2_modifier
		}
		else_if = {
			limit = {
				rarity = famed
			}
			add_artifact_modifier = artifact_dread_baseline_add_1_modifier
		}
		if = {
			limit = { rarity = illustrious }
			add_fleshcrafted_artifact_modifier_effect = { 
				KILLED = scope:killed 
				VALUE = 6
			}
		}
		else_if = {
			limit = { rarity = famed }
			add_fleshcrafted_artifact_modifier_effect = { 
				KILLED = scope:killed 
				VALUE = 5
			}
		}
		else_if = {
			limit = { rarity = masterwork }
			add_fleshcrafted_artifact_modifier_effect = { 
				KILLED = scope:killed 
				VALUE = 4
			}
		}
		else = {
			add_fleshcrafted_artifact_modifier_effect = { 
				KILLED = scope:killed 
				VALUE = 3
			}
		}
	}

get_fleshcrafted_artifact_quality_effect = {
	hidden_effect = {
		random_list = {
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 0.5
				} 
			}
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 1
				} 
			}
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 2
				} 
			}
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 2.5
				} 
			}
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 3
				} 
			}
		}
	}
	
	# Calculate the final quality of the artifact
	save_scope_value_as = {
		name = quality
		value = {
			value = 0
			
			scope:killed = {
				# Bonus quality from high skills
				add = {	
					if = {
						limit = { scope:armor = yes }
						value = armor_inspiration_average_skill_value
					}
					else_if = {
						limit = { scope:weapon = yes }
						value = weapon_inspiration_average_skill_value
					}
					else_if = {
						limit = { scope:shinies = yes }
						value = smith_inspiration_average_skill_value
					}
					else_if = {
						limit = { scope:book = yes }
						value = book_inspiration_average_skill_value
					}
					else_if = {
						limit = { scope:furniture = yes }
						value = weaver_inspiration_average_skill_value
					}
					else_if = {
						limit = { scope:trinket = yes }
						value = artisan_inspiration_average_skill_value
					}
					else_if = {
						limit = { scope:mount = yes }
						value = adventure_inspiration_average_skill_value
					}
					multiply = {
						value = quality_bonus_per_skill_level_value
						add = scope:random_inspired_skill_quality_bonus
					}
				}

				# Bonus quality from high skills
				add = {	
					if = {
						limit = { scope:armor = yes }
						value = armor_inspiration_average_skill_value
					}
					else_if = {
						limit = { scope:weapon = yes }
						value = weapon_inspiration_average_skill_value
					}
					else_if = {
						limit = { scope:shinies = yes }
						value = smith_inspiration_average_skill_value
					}
					else_if = {
						limit = { scope:book = yes }
						value = book_inspiration_average_skill_value
					}
					else_if = {
						limit = { scope:furniture = yes }
						value = weaver_inspiration_average_skill_value
					}
					else_if = {
						limit = { scope:trinket = yes }
						value = artisan_inspiration_average_skill_value
					}
					else_if = {
						limit = { scope:mount = yes }
						value = adventure_inspiration_average_skill_value
					}
					multiply = {
						value = quality_bonus_per_skill_level_value
						add = scope:random_inspired_skill_quality_bonus
					}
				}

				add = {	
					if = {
						limit = {
							has_trait = physique_good_3
						}
						value = 20
					}
					else_if = {
						limit = {
							has_trait = physique_good_2
						}
						value = 15
					}
					else_if = {
						limit = {
							has_trait = physique_good_1
						}
						value = 5
					}
					else = {
						value = 5
					}
				}
				
				if = {
					limit = { has_trait = strong }
					add = 20
				}
				
				if = {
					limit = { 
						scope:creator = { has_trait = tzimisce }
					}
					add = 20
				}
			}
			
			# Can't have negative quality
			min = 1
		}
	}
}

	hidden_effect = {
		random_list = {
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 0.5
				} 
			}
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 1
				} 
			}
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 2
				} 
			}
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 2.5
				} 
			}
			20 = {
				save_scope_value_as = {
					name = random_inspired_skill_quality_bonus
					value = 3
				} 
			}
		}
	}
	
	# Calculate the final quality of the artifact
	save_scope_value_as = {
		name = quality
		value = {
			value = 0
			
			scope:killed = {
				# Bonus quality from high skills
				add = {	
					if = {
						limit = { scope:armor = yes }
						value = armor_inspiration_average_skill_value
					}
					else_if = {
						limit = { scope:weapon = yes }
						value = weapon_inspiration_average_skill_value
					}
					else_if = {
						limit = { scope:shinies = yes }
						value = smith_inspiration_average_skill_value
					}
					else_if = {
						limit = { scope:book = yes }
						value = book_inspiration_average_skill_value
					}
					else_if = {
						limit = { scope:furniture = yes }
						value = weaver_inspiration_average_skill_value
					}
					else_if = {
						limit = { scope:trinket = yes }
						value = artisan_inspiration_average_skill_value
					}
					multiply = {
						value = quality_bonus_per_skill_level_value
						add = scope:random_inspired_skill_quality_bonus
					}
				}

				add = {	
					value = dread
					multiply = 0.20
					floor = yes
				}
				
				if = {
					limit = { 
						scope:creator = {
							has_demonologist_xp = {
								VALUE = 100
								COMPARATOR = ">="
							}
						}
					}
					add = 20
				}
				else_if = {
					limit = { 
						scope:creator = {
							has_demonologist_xp = {
								VALUE = 50
								COMPARATOR = ">="
							}
						}
					}
					add = 10
				}
				
				if = {
					limit = { 
						scope:creator = {
							has_diabolist_xp = {
								VALUE = 100
								COMPARATOR = ">="
							}
						}
					}
					add = 20
				}
				else_if = {
					limit = { 
						scope:creator = {
							has_diabolist_xp = {
								VALUE = 50
								COMPARATOR = ">="
							}
						}
					}
					add = 10
				}
				
				if = {
					limit = { 
						scope:creator = {
							has_thrall_xp = {
								VALUE = 100
								COMPARATOR = ">="
							}
						}
					}
					add = 20
				}
				else_if = {
					limit = { 
						scope:creator = {
							has_thrall_xp = {
								VALUE = 50
								COMPARATOR = ">="
							}
						}

					add = 10
				}
				
				if = {
					limit = {

						limit = { scope:massive_gold = yes }
						value = 30
					}
					else_if = {
						limit = { scope:major_gold = yes }
						value = 20
					}
					else = {
						value = 10
					}
				}
			}
			
			# Can't have negative quality
			min = 1
		}
	}
}

# Reweave Artifact
create_artifact_common_artifact_pieces_effect = {
	$OWNER$ = { save_scope_as = owner }
	
	set_artifact_rarity_common = yes
		
	create_artifact = {
		name = artifact_artifact_pieces_name
		description = artifact_artifact_pieces_description
		type = miscellaneous
		visuals = small_box
		wealth = scope:wealth
		quality = scope:quality		
		modifier = artifact_placeholder_modifier
		generate_history = no
		save_scope_as = newly_created_artifact
	}
	hidden_effect_new_object = { 
		scope:newly_created_artifact = {
			remove_artifact_modifier = artifact_placeholder_modifier
			set_variable = artifact_pieces
			add_artifact_pieces_modifier_effect = yes
			set_max_durability = 25 
			flag_as_trash_artifact = yes
		}
	}
}

create_artifact_restored_artifact_pieces_effect = {
	$OWNER$ = { save_scope_as = owner }
	
	get_restored_artifact_quality_effect = yes
		
	create_artifact = {
		name = artifact_artifact_pieces_name
		description = artifact_artifact_pieces_description
		creator = scope:creator
		type = miscellaneous
		visuals = small_box
		wealth = scope:wealth
		quality = scope:quality	
		modifier = artifact_placeholder_modifier
		save_scope_as = newly_created_artifact
	}
	hidden_effect_new_object = {
		scope:newly_created_artifact = {
			remove_artifact_modifier = artifact_placeholder_modifier
			set_variable = {
				name = restored_artifact_piece_level
				value = scope:owner.restored_artifact_piece_level_value
			}
			set_artifact_name = artifact_restored_artifact_pieces_name
			scope:owner = { copy_artifact_pieces_modifier_effect = { ARTIFACT = scope:newly_created_artifact } }
			set_max_durability = 25
			set_variable = artifact_pieces
			flag_as_trash_artifact = yes
		}
	}
}

get_restored_artifact_quality_effect = {
	if = {
		limit = { restored_artifact_quality = 5 }
		set_artifact_rarity_illustrious = yes
	}
	else_if = {
		limit = { restored_artifact_quality = 4 }
		set_artifact_rarity_famed = yes
	}
	else_if = {
		limit = { restored_artifact_quality = 3 }
		set_artifact_rarity_masterwork = yes
	}
	else = {
		set_artifact_rarity_common = yes
	}
}

copy_artifact_pieces_modifier_effect = {
	$ARTIFACT$ = { save_temporary_scope_as = this_artifact }
	every_equipped_character_artifact = {
		limit = {
			artifact_slot_type = miscellaneous
			has_variable = artifact_pieces
		}
		scope:this_artifact = { copy_artifact_modifiers = prev }
		destroy_artifact = this
	}
}
