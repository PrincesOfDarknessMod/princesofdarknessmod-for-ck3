POD_gnosis_increase_effect = {
	if = {
		limit = {
			POD_splat_has_gnosis_trigger = yes
			NOT = { has_character_modifier = POD_gnosis_5_modifier }	
		}
		custom_tooltip = POD_gnosis_increase_effect.tt
		switch = {
			trigger = has_character_modifier
			POD_gnosis_0_modifier = {
				hidden_effect = { remove_character_modifier = POD_gnosis_0_modifier }
				add_character_modifier = POD_gnosis_1_modifier
			}
			POD_gnosis_1_modifier = {
				hidden_effect = { remove_character_modifier = POD_gnosis_1_modifier }
				add_character_modifier = POD_gnosis_2_modifier
			}
			POD_gnosis_2_modifier = {
				hidden_effect = { remove_character_modifier = POD_gnosis_2_modifier }
				add_character_modifier = POD_gnosis_3_modifier
			}
			POD_gnosis_3_modifier = {
				if = {
					limit = { POD_is_kinfolk_trigger = no }
					hidden_effect = { remove_character_modifier = POD_gnosis_3_modifier }
					add_character_modifier = POD_gnosis_4_modifier
				}
			}
			POD_gnosis_4_modifier = {
				if = {
					limit = { POD_is_kinfolk_trigger = no }
					hidden_effect = { remove_character_modifier = POD_gnosis_4_modifier }
					add_character_modifier = POD_gnosis_5_modifier
				}
			}
			fallback = { POD_assign_initial_gnosis_effect = yes }
		}
	}
}

POD_gnosis_decrease_effect = {
	if = {
		limit = {
			POD_splat_has_gnosis_trigger = yes
			gnosis_level >= 1
		}
		custom_tooltip = POD_gnosis_decrease_effect.tt
		switch = {
			trigger = has_character_modifier
			POD_gnosis_1_modifier = {
				hidden_effect = { remove_character_modifier = POD_gnosis_1_modifier }
				add_character_modifier = POD_gnosis_0_modifier
			}
			POD_gnosis_2_modifier = {
				hidden_effect = { remove_character_modifier = POD_gnosis_2_modifier }
				add_character_modifier = POD_gnosis_1_modifier
			}
			POD_gnosis_3_modifier = {
				hidden_effect = { remove_character_modifier = POD_gnosis_3_modifier }
				add_character_modifier = POD_gnosis_2_modifier
			}
			POD_gnosis_4_modifier = {
				hidden_effect = { remove_character_modifier = POD_gnosis_4_modifier }
				add_character_modifier = POD_gnosis_3_modifier
			}
			POD_gnosis_5_modifier = {
				hidden_effect = { remove_character_modifier = POD_gnosis_5_modifier }
				add_character_modifier = POD_gnosis_4_modifier
			}
			fallback = { POD_assign_initial_gnosis_effect = yes }
		}
	}
	else_if = {
		limit = { 
			POD_splat_has_gnosis_trigger = yes
			gnosis_level < 1
		}
		custom_tooltip = POD_gnosis_decrease_effect2.tt
		add_stress = major_stress_gain
	}
}

POD_gnosis_regen_effect = {
	if = {
		limit = {
			POD_splat_has_gnosis_trigger = yes
			trigger_if = {
				limit = { POD_is_kinfolk_trigger = yes }
				gnosis_level < 3
			}
			trigger_else = {
				NOT = { has_character_modifier = POD_gnosis_5_modifier }
				gnosis_level < 5
			}
		}
		
		random = {
			chance = gnosis_regen_chance
			POD_gnosis_increase_effect = yes
		}
	}
}

POD_assign_initial_gnosis_effect = {
	if = {
		limit = {
			POD_has_gnosis_modifier_trigger = no
			POD_splat_has_gnosis_trigger = yes
		}

		if = { 
			limit = {
				POD_is_kinfolk_trigger = yes
			}
			add_character_modifier = {
				modifier = POD_gnosis_0_modifier			
			}
		}
		else_if = { 
			limit = {
				has_trait = homid_born
			}
			add_character_modifier = {
				modifier = POD_gnosis_1_modifier			
			}
		}
		else_if = {
			limit = {
				has_trait = beast_born
			}
			add_character_modifier = {
				modifier = POD_gnosis_3_modifier			
			}
		}
		else = { 
			add_character_modifier = POD_gnosis_5_modifier
		}
	}
}

POD_gift_cost_effect = {
	if = {
		limit = {
			any_character_artifact = {
				has_variable = energy_stored
				var:energy_stored > 0
			}
		}
		ordered_character_artifact = {
			limit = {
				has_variable = energy_stored 
				var:energy_stored > 0
			}
			order_by = {
				value = 0
				add = var:energy_stored
			}
			position = 0
			POD_energy_vessel_decrease_multiple_effect = { COUNT = $COUNT$ }
		}
	}
	else = {
		POD_gnosis_decrease_multiple_effect = { COUNT = $COUNT$ }
	}
}

POD_gnosis_income_effect = {
	if = {
		limit = {
			OR = {
				gnosis_difference >= $COUNT$
				NOT = {
					any_character_artifact = {
						NOT = { has_variable = temporary_vessel }
						has_variable = energy_stored
						vessel_empty_energy > 0
					}
				}
			}
		}
		POD_gnosis_increase_multiple_effect = { COUNT = $COUNT$ }
	}
	else = {
		save_scope_value_as = {
			name = excess_gnosis
			value = {
				value = $COUNT$
				subtract = gnosis_difference
			}
		}
		POD_gnosis_increase_multiple_effect = { COUNT = gnosis_difference }
		ordered_character_artifact = {
			limit = {
				NOT = { has_variable = temporary_vessel }
				has_variable = energy_stored 
				vessel_empty_energy > 0
			}
			order_by = {
				value = 0
				add = vessel_empty_energy
			}
			position = 0
			POD_energy_vessel_increase_multiple_effect = { COUNT = scope:excess_gnosis }
		}
	}
}
