POD_set_gnosis_to_effect = {
	hidden_effect = { POD_remove_gnosis_modifiers_effect = yes }
	save_scope_value_as = {
		name = clamp
		value = {
			value = $GNOSIS$
			floor = yes
			min = 0
			max = 5 # corresponds to maximum_gnosis script_value
		}
	}
	switch = {
		trigger = scope:clamp
		0 = { add_character_modifier = POD_gnosis_0_modifier }
		1 = { add_character_modifier = POD_gnosis_1_modifier }
		2 = { add_character_modifier = POD_gnosis_2_modifier }
		3 = { add_character_modifier = POD_gnosis_3_modifier }
		4 = { add_character_modifier = POD_gnosis_4_modifier }
		5 = { add_character_modifier = POD_gnosis_5_modifier }
	}
}

POD_gnosis_change_effect = {
	if = {
		limit = { NOT = { $COUNT$ = 0 } }
		custom_description = {
			text = POD_gnosis_change_desc
			value = $COUNT$
			subject = this
		}
		save_scope_value_as = {
			name = new_gnosis
			value = {
				value = gnosis_level
				add = $COUNT$
			}
		}
		POD_set_gnosis_to_effect = { GNOSIS = scope:new_gnosis }
	}
}

POD_gnosis_increase_effect = {
	POD_gnosis_change_effect = { COUNT = $COUNT$ }
}

POD_gnosis_decrease_effect = {
	save_scope_value_as = {
		name = gnosis_cost
		value = {
			value = 0
			subtract = $COUNT$
		}
	}
	POD_gnosis_change_effect = { COUNT = scope:gnosis_cost }
}

POD_gift_cost_effect = {
	POD_energy_cost = {
		COUNT = $COUNT$
		SPLAT_MAXIMUM_ENERGY = maximum_gnosis
	}
}

POD_gift_cost_with_decimals_effect = {
    save_scope_value_as = {
        name = cost_floor
        value = {
            value = $COUNT$
            floor = yes
        }
    }
    save_scope_value_as = {
        name = cost_ceiling
        value = {
            value = $COUNT$
            ceiling = yes
        }
    }
    save_scope_value_as = {
        name = cost_ceiling_chance
        value = {
            value = $COUNT$
            modulo = 1
            multiply = 100
        }
    }

    random_list = {
        100 = {
            modifier = { add = { value = scope:cost_ceiling_chance multiply = -1 } }
            POD_gift_cost_effect = { COUNT = scope:cost_floor }
        }
        0 = {
            modifier = { add = scope:cost_ceiling_chance }
            POD_gift_cost_effect = { COUNT = scope:cost_ceiling }
        }
    }
}

POD_gnosis_income_effect = {
	POD_energy_income = {
		COUNT = $COUNT$
		SPLAT_MAXIMUM_ENERGY = maximum_gnosis
	}
}


POD_gnosis_regen_effect = {
	random = {
		chance = gnosis_regen_chance
		POD_gnosis_income_effect = { COUNT = 1 }
	}
}

POD_assign_initial_gnosis_effect = {
	if = {
		limit = { POD_is_kinfolk_trigger = yes }
		add_character_modifier = POD_gnosis_0_modifier
	}
	else_if = {
		limit = { has_trait = homid_born }
		add_character_modifier = POD_gnosis_1_modifier
	}
	else_if = {
		limit = { has_trait = beast_born }
		add_character_modifier = POD_gnosis_3_modifier
	}
	else = {
		add_character_modifier = POD_gnosis_5_modifier
	}
}
