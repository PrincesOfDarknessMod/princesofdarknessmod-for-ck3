


POD_assign_mokole_stream_effect = {
	if = {
		limit = { POD_mokole_has_stream_trigger = no }
		if = {
			limit = {
				is_lowborn = no
				dynasty = { has_variable = mokole_stream }
			}
			dynasty.var:mokole_stream = { save_scope_as = stream }
			POD_assign_mokole_stream_custom_effect = { STREAM = scope:stream }
		}
		else_if = {
			limit = { liege ?= { POD_mokole_has_stream_trigger = yes } }
			liege.dynasty.var:mokole_stream = { save_scope_as = stream }
			POD_assign_mokole_stream_custom_effect = { STREAM = scope:stream }
		}
		else = {
			#TODO Mokole, fine tune depending on region/faith
			random_list = {
				25 = { POD_assign_mokole_stream_custom_effect = { STREAM = mokole_mbembe } }
				25 = { POD_assign_mokole_stream_custom_effect = { STREAM = gumagan } }
				25 = { POD_assign_mokole_stream_custom_effect = { STREAM = makara } }
				25 = { POD_assign_mokole_stream_custom_effect = { STREAM = zhong_lung } }
			}
		}
	}
}


POD_assign_mokole_stream_custom_effect = {
	if = {
		limit = { 
			is_lowborn = no 
			dynasty = { NOT = { has_variable = mokole_stream } }
		}
		dynasty = { set_variable = { name = mokole_stream value = flag:$STREAM$ } }
	}
	else_if = {
		limit = { NOT = { has_variable = mokole_stream } }
		set_variable = { name = mokole_stream value = flag:$STREAM$ }
	}
}


POD_mokole_shapeshift_suchid_effect = {
	switch = {
		trigger = var:suchid_form

		flag:champsa = { add_character_modifier = POD_mokole_form_suchid_champsa_modifier }
		flag:karna = { add_character_modifier = POD_mokole_form_suchid_karna_modifier }
		flag:ora = { add_character_modifier = POD_mokole_form_suchid_ora_modifier }
		flag:gharial = { add_character_modifier = POD_mokole_form_suchid_gharial_modifier }
		flag:makara = { add_character_modifier = POD_mokole_form_suchid_makara_modifier }
		flag:naja = { add_character_modifier = POD_mokole_form_suchid_naja_modifier }
		flag:lung = { add_character_modifier = POD_mokole_form_suchid_lung_modifier }
		flag:suchus = { add_character_modifier = POD_mokole_form_suchid_suchus_modifier }
	}
}

POD_mokole_remove_shapeshift_suchid_effect = {
	POD_remove_character_modifier = { MODIFIER = POD_mokole_form_suchid_champsa_modifier }
	POD_remove_character_modifier = { MODIFIER = POD_mokole_form_suchid_karna_modifier }
	POD_remove_character_modifier = { MODIFIER = POD_mokole_form_suchid_ora_modifier }
	POD_remove_character_modifier = { MODIFIER = POD_mokole_form_suchid_gharial_modifier }
	POD_remove_character_modifier = { MODIFIER = POD_mokole_form_suchid_makara_modifier }
	POD_remove_character_modifier = { MODIFIER = POD_mokole_form_suchid_naja_modifier }

	POD_remove_character_modifier = { MODIFIER = POD_mokole_form_suchid_suchus_modifier }
}
