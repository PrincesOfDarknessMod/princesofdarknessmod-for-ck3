


POD_assign_mokole_stream_effect = {
	if = {
		limit = { POD_mokole_has_stream_trigger = no }
		if = {
			limit = { 
				is_lowborn = no
				is_independent_ruler = no
				liege = { POD_mokole_has_stream_trigger = yes }
			}
			liege.dynasty.var:mokole_stream = { save_scope_as = dynasty_stream }
			POD_assign_mokole_stream_from_dynasty_effect = { TARGET = scope:dynasty_stream }
		}
		else = {
			#TODO Mokole, fine tune depending on region/faith
			random_list = {
				25 = { POD_assign_mokole_stream_custom_effect = { STREAM = mokole_mbembe } }
				25 = { POD_assign_mokole_stream_custom_effect = { STREAM = gumagan } }
				25 = { POD_assign_mokole_stream_custom_effect = { STREAM = makara } }
				25 = { POD_assign_mokole_stream_custom_effect = { STREAM = zhong_lung } }
			}
		}
	}
}

POD_assign_mokole_stream_from_dynasty_effect = {
	if = {
		limit = {
			is_lowborn = no 
			dynasty = { NOT = { has_variable = mokole_stream } }
		}
		dynasty = { set_variable = { name = mokole_stream value = $TARGET$ } }
	}
	else_if = { 
		limit = { NOT = { has_variable = mokole_stream } }
		set_variable = { name = mokole_stream value = $TARGET$ }
	}
}


POD_assign_mokole_stream_custom_effect = {
	if = {
		limit = { 
			is_lowborn = no 
			dynasty = { NOT = { has_variable = mokole_stream } }
		}
		dynasty = { set_variable = { name = mokole_stream value = flag:$STREAM$ } }
	}
	else_if = {
		limit = { NOT = { has_variable = mokole_stream } }
		set_variable = { name = mokole_stream value = flag:$STREAM$ }
	}
}

POD_mokole_birth_effects = {
	if = {
		limit = {
			OR = {
				real_father ?= { POD_is_mokole_trigger = yes }
				mother ?= { POD_is_mokole_trigger = yes }
			}
		}

		if = {
			limit = { dynasty ?= { has_dynasty_perk = mokole_mbembe_legacy_3 } }
			add_martial_skill = 2
		}
	}
}

POD_mokole_spawn_dream_hunter_effect = {
	hidden_effect_new_object = {
		save_scope_value_as = { name = spirit_spawn_notification value = no }

		POD_spawn_hostile_spirit_effect = yes
		scope:spawned_spirit = {
			add_diplomacy_skill = 2
			add_martial_skill = 2
			add_stewardship_skill = 2
			add_intrigue_skill = 2
			add_learning_skill = 2
			add_prowess_skill = 10


			##perks here


			if = {
				limit = { prowess_diff = { target = scope:spawning_character value <= -20 } }
				add_prowess_skill = 10
			}

			POD_spirit_assign_rank_effect = yes
			save_scope_as = dream_hunter
		}
	}
}

POD_mokole_dream_hunter_damage_effect = {
	save_scope_as = mokole
	POD_increase_wounds_hostile_effect = { ATTACKER = scope:dream_hunter COUNT = mokole_mnesis_dream_hunter_damage_value REASON = fight }
	if = {
		limit = { mokole_mnesis_age_value = 4 }
		add_prestige = -200
		add_piety = -300
	}
	else_if = {
		limit = { mokole_mnesis_age_value = 3 }
		add_prestige = -100
		add_piety = -200
	}
	else_if = {
		limit = { mokole_mnesis_age_value = 1 }
		add_piety = -100
	}
}

POD_mokole_bloody_mindedness_effect = {
	POD_remove_character_modifier = { MODIFIER = POD_mokole_bloody_mindedness_nemesis_modifier }
	POD_remove_character_modifier = { MODIFIER = POD_mokole_bloody_mindedness_rival_modifier }
	if = {
		limit = { has_perk = bloody_mindedness_perk }

		if = {
			limit = { num_of_relation_nemesis >= 1 }
			add_character_modifier = POD_mokole_bloody_mindedness_nemesis_modifier
		}
		else_if = {
			limit = { num_of_relation_rival >= 1 }
			add_character_modifier = POD_mokole_bloody_mindedness_rival_modifier
		}
	}
}

	}
}

POD_mokole_army_of_the_ancestors_effect = {
	if = {
		limit = {
			has_perk = army_of_the_ancestors_perk
			NOT = { has_variable_list = army_of_the_ancestors_knights }
		}
		while = {
			count = 3

			POD_spawn_spirit_effect = { TEMPLATE = spirit_mokole_eclipsed_perk_template }

			add_to_variable_list = { name = army_of_the_ancestors_knights target = scope:spawned_spirit }
		}
	}
}

POD_mokole_army_of_the_ancestors_despawn = {
	if = {
		limit = { has_variable_list = army_of_the_ancestors_knights is_at_war = no }
		every_in_list = {
			variable = army_of_the_ancestors_knights

			death = { death_reason = death_disappearance }
			remove_from_list = army_of_the_ancestors_knights
		}
	}
}
