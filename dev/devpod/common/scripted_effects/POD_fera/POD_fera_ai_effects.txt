POD_fera_ai_spirit_quest_effect = {
	#Try to get a new Fetish
	if = {
		limit = { 
			any_character_artifact = {
				count < 5
				has_variable = fetish_artifact
			}
		}

		random_list = {
			#Common
			60 = {
				trigger = { POD_has_all_common_quest_artifacts = no }
				POD_fera_spirit_quest_common_fetish = yes
			}

			#Masterwork
			37 = {
				trigger = { POD_has_all_masterwork_quest_artifacts = no }
				POD_fera_spirit_quest_masterwork_fetish = yes
			}

			#Famed
			3 = {
				trigger = { POD_has_all_famed_quest_artifacts = no }
				POD_fera_spirit_quest_famed_fetish = yes
			}

			#AI don't get illustrious
		}
		scope:newly_created_artifact ?= { set_variable = ai_quest_fetish }
	}

	#Cheap simulation of doing a higher lvl spirit quest
	if = {
		limit = { POD_fera_uses_renown_wisdom = yes }
		random = {
			chance = 25
			POD_fera_add_static_renown_effect = { VALUE = 1 } #This uses static renown to not double stack with the script values
		}
	}

	#Get Misc Rewards
	POD_fera_spirit_quest_rewards_effect = yes
}
