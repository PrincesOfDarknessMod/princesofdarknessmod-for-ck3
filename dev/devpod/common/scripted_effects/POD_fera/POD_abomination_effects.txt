

#For abominations without a sire
POD_become_abomination_effect = {
	if = {
		limit = { POD_fera_can_become_abomination = yes }
		save_temporary_scope_value_as = { name = saved_immortal_age value = POD_visual_age }
		POD_embrace_remove_incompatible_trait_effect = yes
		POD_save_energy_level_before_splat_change = yes
		POD_save_fera_shape_effect = yes
		POD_save_fera_rank_effect = yes
		POD_remove_fera_on_embrace_effect = yes
		add_trait_force_tooltip = abomination
		
		hidden_effect = {
			POD_remove_wrong_supernatural_traits_on_existing_character = yes
			POD_remove_supernatural_relation_effect = yes
			POD_side_effects_from_changing_splats = yes

			POD_remove_vampire_generation_traits_effect = yes
			POD_remove_vampire_clan_traits_effect = yes
		}


		add_trait_force_tooltip = $CLAN$
		add_trait_force_tooltip = $GENERATION$

		POD_abomination_initialization_effect = yes
	}
}

#Needs scope:vampiresire
fera_embrace_effect = {
	if = {
		limit = { POD_fera_can_become_abomination = yes }
		save_temporary_scope_value_as = { name = saved_immortal_age value = POD_visual_age }
		POD_embrace_remove_incompatible_trait_effect = yes
		POD_save_energy_level_before_splat_change = yes
		POD_save_fera_shape_effect = yes
		POD_save_fera_rank_effect = yes
		POD_remove_fera_on_embrace_effect = yes
		add_trait_force_tooltip = abomination
		
		hidden_effect = {
			POD_remove_wrong_supernatural_traits_on_existing_character = yes
			POD_remove_supernatural_relation_effect = yes
			POD_side_effects_from_changing_splats = yes

			POD_remove_vampire_generation_traits_effect = yes
			POD_remove_vampire_clan_traits_effect = yes
		}


		POD_remove_parent_relation_effect = yes
		POD_remove_child_relation_effect = yes
		POD_set_parent_effect = { PARENT = scope:vampiresire }

		######TODO Abominations: Temporary
		set_house = scope:vampiresire.house
		if = { #TODO Abominations: remove
			limit = { scope:vampiresire = { religion = { is_in_family = rf_kindred } } }
			set_character_faith = scope:vampiresire.faith
		}
		else = { POD_character_vampire_faith_initialization_effect = yes }
		######


		POD_character_vampire_inherit_generation_effect = yes
		POD_character_vampire_inherit_clan_effect = yes
		POD_character_vampire_clan_special_effect = yes

		POD_abomination_initialization_effect = yes
	}
}
