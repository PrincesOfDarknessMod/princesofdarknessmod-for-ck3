


#####How to use Fera Renown:#####
##The main thing you should be concerned about is the very first effect "POD_fera_add_renown_effect" 
##Its used like so: POD_fera_add_renown_effect = { TYPE = X }  where is the renown type 
##To see which renown types exist or modify which fera type uses which renown, consult "POD_fera_renown_triggers" in the scripted_triggers/POD_fera folder
##To modify how many points you get, consult "POD_fera_renown_values" in the script_values/POD_fera folder
##If you wish to add more sources of a certain renown type, please also add it to the rank up decision localization in "POD_decisions_fera_l_english"
##
##Example usage: POD_fera_add_renown_effect = { TYPE = ferocity }
##Adding this to an objective decision would add ferocity renown/trait exp to the character if it is a Fera and uses the ferocity renown type
##You do not need to make any additional checks before using the effect, the effect itself does all the checking for you
##
##If you wish to search for sources of Glory renown within the files, you would search for "POD_fera_add_renown_effect = { TYPE = glory }"
##
##Notes:
##Apis, Camazotz, and Grondr are extinct. However, they are included in the effect for completion's sake. Whether those fera splats will ever be playable is a different story.



POD_fera_add_renown_effect = {
	if = {
		limit = { 
			POD_fera_uses_renown_$TYPE$ = yes
			exists = var:fera_rank
			var:fera_rank < 6
		}
		switch = {
			trigger = has_trait

			werewolf = { POD_fera_add_renown_werewolf_effect = { TYPE = $TYPE$ } }
			ajaba = { POD_fera_add_renown_ajaba_effect = { TYPE = $TYPE$ } }
			ananasi = { POD_fera_add_renown_ananasi_effect = { TYPE = $TYPE$ } }
			apis = { POD_fera_add_renown_apis_effect = { TYPE = $TYPE$ } }
			bastet = { POD_fera_add_renown_bastet_effect = { TYPE = $TYPE$ } }
			camazotz = { POD_fera_add_renown_camazotz_effect = { TYPE = $TYPE$ } }
			corax = { POD_fera_add_renown_corax_effect = { TYPE = $TYPE$ } }
			grondr = { POD_fera_add_renown_grondr_effect = { TYPE = $TYPE$ } }
			gurahl = { POD_fera_add_renown_gurahl_effect = { TYPE = $TYPE$ } }
			kitsune = { POD_fera_add_renown_kitsune_effect = { TYPE = $TYPE$ } }
			mokole = { POD_fera_add_renown_mokole_effect = { TYPE = $TYPE$ } }
			nagah = { POD_fera_add_renown_nagah_effect = { TYPE = $TYPE$ } }
			nuwisha = { POD_fera_add_renown_nuwisha_effect = { TYPE = $TYPE$ } }
			ratkin = { POD_fera_add_renown_ratkin_effect = { TYPE = $TYPE$ } }
			rokea = { POD_fera_add_renown_rokea_effect = { TYPE = $TYPE$ } }
		}
		POD_achiev_feratutorial_check = yes
	}
}

###These are intentionally kept seperate due to possible deviations when the other fera types are properly implemented(like nagah)###
POD_fera_add_renown_werewolf_effect = {
	if = {
		limit = { is_ai = yes }
		add_trait_xp = { trait = werewolf value = fera_renown_$TYPE$_value }
		POD_fera_rank_initialization_effect = yes
	}
	else = {
		save_scope_value_as = { name  = xp_gain value = fera_renown_$TYPE$_value }
		if = {
			limit = { scope:xp_gain > POD_xp_to_next_fera_rank_minus_1 }
			save_scope_value_as = {
				name  = xp_gain
				value = POD_xp_to_next_fera_rank_minus_1
			}
			add_character_flag = fera_rank_up_ready
		}
		add_trait_xp = { trait = werewolf value = scope:xp_gain }
	}
}
