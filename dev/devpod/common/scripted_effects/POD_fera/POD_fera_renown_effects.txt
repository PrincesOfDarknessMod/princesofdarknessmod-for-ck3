


#####How to use Fera Renown:#####
##The main thing you should be concerned about is the very first effect "POD_fera_add_renown_effect"
##Its used like so: POD_fera_add_renown_effect = { TYPE = X }  where X is the renown type
##To see which renown types exist or modify which fera type uses which renown, consult "POD_fera_renown_triggers" in the scripted_triggers/POD_fera folder
##To modify how many points you get, consult "POD_fera_renown_values" in the script_values/POD_fera folder
##If you wish to add more sources of a certain renown type, please also add it to the rank up decision localization in "POD_decisions_fera_l_english"
##
##Example usage: POD_fera_add_renown_effect = { TYPE = ferocity }
##Adding this to an objective decision would add ferocity renown/trait exp to the character if it is a Fera and uses the ferocity renown type
##You do not need to make any additional checks before using the effect, the effect itself does all the checking for you
##
##If you wish to search for sources of Glory renown within the files, you would search for "POD_fera_add_renown_effect = { TYPE = glory }"
##
##Notes:
##Apis, Camazotz, and Grondr are extinct. However, they are included in the effect for completion's sake. Whether those fera splats will ever be playable is a different story.



POD_fera_add_renown_effect = {
	if = {
		limit = {
			POD_fera_uses_renown_$TYPE$ = yes
			exists = var:fera_rank
			var:fera_rank < 6
		}

		if = {
			limit = { is_ai = yes }

			switch = {
				trigger = has_trait

				werewolf = { add_trait_xp = { trait = werewolf value = fera_renown_$TYPE$_value } }
				ajaba = { add_trait_xp = { trait = ajaba value = fera_renown_$TYPE$_value } }
				ananasi = { add_trait_xp = { trait = ananasi value = fera_renown_$TYPE$_value } }
				apis = { add_trait_xp = { trait = apis value = fera_renown_$TYPE$_value } }
				bastet = { add_trait_xp = { trait = bastet value = fera_renown_$TYPE$_value } }
				camazotz = { add_trait_xp = { trait = camazotz value = fera_renown_$TYPE$_value } }
				corax = { add_trait_xp = { trait = corax value = fera_renown_$TYPE$_value } }
				grondr = { add_trait_xp = { trait = grondr value = fera_renown_$TYPE$_value } }
				gurahl = { add_trait_xp = { trait = gurahl value = fera_renown_$TYPE$_value } }
				kitsune = { add_trait_xp = { trait = kitsune value = fera_renown_$TYPE$_value } }
				mokole = { add_trait_xp = { trait = mokole value = fera_renown_$TYPE$_value } }
				nagah = { add_trait_xp = { trait = nagah value = fera_renown_$TYPE$_value } }
				nuwisha = { add_trait_xp = { trait = nuwisha value = fera_renown_$TYPE$_value } }
				ratkin = { add_trait_xp = { trait = ratkin value = fera_renown_$TYPE$_value } }
				rokea = { add_trait_xp = { trait = rokea value = fera_renown_$TYPE$_value } }
			}
			POD_fera_rank_initialization_effect = yes
		}

		else = {
			save_scope_value_as = { name  = xp_gain value = fera_renown_$TYPE$_value }
			if = {
				limit = { scope:xp_gain > POD_xp_to_next_fera_rank_minus_1 }
				save_scope_value_as = {
					name  = xp_gain
					value = POD_xp_to_next_fera_rank_minus_1
				}
				add_character_flag = fera_rank_up_ready
			}

			switch = {
				trigger = has_trait

				werewolf = { add_trait_xp = { trait = werewolf value = scope:xp_gain } }
				ajaba = { add_trait_xp = { trait = ajaba value = scope:xp_gain } }
				ananasi = { add_trait_xp = { trait = ananasi value = scope:xp_gain } }
				apis = { add_trait_xp = { trait = apis value = scope:xp_gain } }
				bastet = { add_trait_xp = { trait = bastet value = scope:xp_gain } }
				camazotz = { add_trait_xp = { trait = camazotz value = scope:xp_gain } }
				corax = { add_trait_xp = { trait = corax value = scope:xp_gain } }
				grondr = { add_trait_xp = { trait = grondr value = scope:xp_gain } }
				gurahl = { add_trait_xp = { trait = gurahl value = scope:xp_gain } }
				kitsune = { add_trait_xp = { trait = kitsune value = scope:xp_gain } }
				mokole = { add_trait_xp = { trait = mokole value = scope:xp_gain } }
				nagah = { add_trait_xp = { trait = nagah value = scope:xp_gain } }
				nuwisha = { add_trait_xp = { trait = nuwisha value = scope:xp_gain } }
				ratkin = { add_trait_xp = { trait = ratkin value = scope:xp_gain } }
				rokea = { add_trait_xp = { trait = rokea value = scope:xp_gain } }
			}

			POD_achiev_feratutorial_check = yes
		}

		#POD_fera_renown_secondary_effects = yes
	}
}
