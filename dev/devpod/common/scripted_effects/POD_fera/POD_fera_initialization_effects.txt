POD_werewolf_initialization_effect = {
	if = {
		limit = {
			is_a_cultist = no
			POD_is_supehunter_trigger = no
			NOR = {
				religion = { is_in_family = rf_wyrm }
				religion = { is_in_family = rf_gaia }
			}
		}
		POD_character_werewolf_faith_initialization_effect = yes
	}
	
	POD_character_initialization_for_werewolf_assign_auspice_effect = yes			
	POD_character_initialization_for_werewolf_assign_breed_effect = yes	
	POD_assign_initial_gnosis_effect = yes
	
	if = {
		limit = { NOT = { has_character_flag = immune_to_disease } }
		remove_disease_trait_effect = yes
		add_character_flag = immune_to_disease	
	}
	if = {
		limit = { is_landed = yes }
		POD_set_litany_government_effect = yes
	}
}

POD_kinfolk_initialization_effect = {
	if = {
		limit = {
			OR = {
				religion = { is_in_family = rf_wyrm }
				religion = { is_in_family = rf_gaia }
			}
		}
		random = {
			chance = 10
			add_trait = gnosis
			POD_assign_initial_gnosis_effect = yes
			POD_resonance_increase_intensity_effect = yes
		}
	}
	if = {
		limit = { NOT = { exists = var:fera_type } }
		random_list = {
			1 = {
				trigger = { real_mother ?= { POD_has_any_inheritable_fera_type = yes } }
				modifier = {
					# make sure a player's children inherit the player's changing breed
					real_father ?= {
						is_ai = no
						POD_has_any_inheritable_fera_type = yes
					}
					factor = 0
				}
				real_mother = { POD_save_fera_type_as = { SAVE_AS = fera_type } }
				POD_set_fera_type = { TRAIT = scope:fera_type }
			}
			1 = {
				trigger = { real_father ?= { POD_has_any_inheritable_fera_type = yes } }
				modifier = {
					# make sure a player's children inherit the player's changing breed
					real_mother ?= {
						is_ai = no
						POD_has_any_inheritable_fera_type = yes
					}
					factor = 0
				}
				real_father = { POD_save_fera_type_as = { SAVE_AS = fera_type } }
				POD_set_fera_type = { TRAIT = scope:fera_type }
			}
		}
	}
	if = {
		limit = {
			is_adult = yes
			NOT = { exists = var:fera_type }
			liege_or_court_owner ?= { POD_has_any_inheritable_fera_type = yes }
		}
		liege_or_court_owner = { POD_save_fera_type_as = { SAVE_AS = fera_type } }
		POD_set_fera_type = { TRAIT = scope:fera_type }
	}
	if = {
		limit = { NOT = { exists = var:fera_type } }
		if = {
			limit = {
				OR = {
					real_mother ?= { has_character_modifier = hathors_blessing_perk_modifier }
					real_father ?= { has_character_modifier = hathors_blessing_perk_modifier }
				}
			}
			POD_set_fera_type = { TRAIT = trait:apis }
		}
		else = {
			POD_set_fera_type = { TRAIT = trait:werewolf }
		}
	}
}

POD_save_fera_type_as = {
	random_character_trait = {
		limit = { has_trait_flag = fera_type }
		save_scope_as = $SAVE_AS$
	}
	if = {
		limit = { NOT = { exists = scope:$SAVE_AS$ } }
		var:fera_type ?= { save_scope_as = $SAVE_AS$ }
	}
}

POD_set_fera_type = {
	set_variable = {
		name  = fera_type
		value = $TRAIT$
	}
}

POD_save_shifter_shape_effect = {
	random_character_trait = {
		limit = { has_trait_flag = fera_type }
		prev = { POD_set_fera_type = { TRAIT = prev } }
	}
}
