POD_save_fera_type_as = {
	random_character_trait = {
		limit = { has_trait_flag = fera_type }
		save_scope_as = $SAVE_AS$
	}
	if = {
		limit = { NOT = { exists = scope:$SAVE_AS$ } }
		var:fera_type ?= { save_scope_as = $SAVE_AS$ }
	}
}

POD_set_fera_type = {
	set_variable = {
		name  = fera_type
		value = $TRAIT$
	}
}

POD_save_fera_shape_effect = {
	random_character_trait = {
		limit = { has_trait_flag = fera_type }
		prev = { POD_set_fera_type = { TRAIT = prev } }
	}
}


POD_remove_fera_effect = {
	if = {
		limit = { has_trait = fera }

		switch = {
			trigger = has_trait

			werewolf = { remove_trait = werewolf }
			ajaba = { remove_trait = ajaba }
			ananasi = { remove_trait = ananasi }
			apis = { remove_trait = apis }
			bastet = { remove_trait = bastet }
			camazotz = { remove_trait = camazotz }
			corax = { remove_trait = corax }
			grondr = { remove_trait = grondr }
			gurahl = { remove_trait = gurahl }
			kitsune = { remove_trait = kitsune }
			mokole = { remove_trait = mokole }
			nagah = { remove_trait = nagah }
			nuwisha = { remove_trait = nuwisha }
			ratkin = { remove_trait = ratkin }
			rokea = { remove_trait = rokea }
		}
		POD_shapeshift_revert_all_effect = yes
		
		add_trait = kinfolk
		add_trait = gnosis
	}
}

POD_werewolf_initialization_effect = {
	#Faith
	if = {
		limit = {
			is_a_cultist = no
			POD_is_supehunter_trigger = no
			NOR = {
				religion = { is_in_family = rf_wyrm }
				religion = { is_in_family = rf_gaia }
			}
		}
		POD_character_werewolf_faith_initialization_effect = yes
	}
	#Breed
	if = {
		limit = { POD_fera_has_breed_trigger = no }	
		
		# Assign Breed at Random 
		random_list = {
			10 = { 
				modifier = {
					religion = religion:redtalon_religion
					add = 75
				}
				add_trait = beast_born
			}
			10 = { 
				modifier = {
					NOT = { religion = religion:redtalon_religion }
					add = 75
				}
				add_trait = homid_born
			} 
		}
	}
	#Auspice
	if = {
		limit = { POD_fera_has_lunar_auspice_trigger = no }	
		
		# Assign Auspice at Day of Month
		if = {
			limit = { current_day <= 3 }
			add_trait = ragabash
		}
		else_if = {
			limit = { current_day <= 9 }
			add_trait = theurge
		}
		else_if = {
			limit = { current_day <= 15 }
			add_trait = philodox
		}
		else_if = {
			limit = { current_day <= 21 }
			add_trait = galliard
		}
		else_if = {
			limit = { current_day <= 27 }
			add_trait = ahroun
		}
		else = {
			add_trait = ragabash
		}
	}
}

POD_kinfolk_initialization_effect = {
	if = {
		limit = {
			OR = {
				religion = { is_in_family = rf_wyrm }
				religion = { is_in_family = rf_gaia }
			}
		}
		random = {
			chance = 10
			add_trait = gnosis
			POD_assign_initial_gnosis_effect = yes
			POD_resonance_increase_intensity_effect = yes
		}
	}
	if = {
		limit = { NOT = { exists = var:fera_type } }
		random_list = {
			1 = {
				trigger = { real_mother ?= { POD_has_any_inheritable_fera_type = yes } }
				modifier = {
					# make sure a player's children inherit the player's changing breed
					real_father ?= {
						is_ai = no
						POD_has_any_inheritable_fera_type = yes
					}
					factor = 0
				}
				real_mother = { POD_save_fera_type_as = { SAVE_AS = fera_type } }
				POD_set_fera_type = { TRAIT = scope:fera_type }
			}
			1 = {
				trigger = { real_father ?= { POD_has_any_inheritable_fera_type = yes } }
				modifier = {
					# make sure a player's children inherit the player's changing breed
					real_mother ?= {
						is_ai = no
						POD_has_any_inheritable_fera_type = yes
					}
					factor = 0
				}
				real_father = { POD_save_fera_type_as = { SAVE_AS = fera_type } }
				POD_set_fera_type = { TRAIT = scope:fera_type }
			}
		}
	}
	if = {
		limit = {
			is_adult = yes
			NOT = { exists = var:fera_type }
			liege_or_court_owner ?= { POD_has_any_inheritable_fera_type = yes }
		}
		liege_or_court_owner = { POD_save_fera_type_as = { SAVE_AS = fera_type } }
		POD_set_fera_type = { TRAIT = scope:fera_type }
	}
}
