POD_save_fera_type_as = {
	random_character_trait = {
		limit = { has_trait_flag = fera_type }
		save_scope_as = $SAVE_AS$
	}
	if = {
		limit = { NOT = { exists = scope:$SAVE_AS$ } }
		var:fera_type ?= { save_scope_as = $SAVE_AS$ }
	}
}

POD_set_fera_type = {
	set_variable = {
		name  = fera_type
		value = $TRAIT$
	}
}

POD_save_fera_shape_effect = {
	random_character_trait = {
		limit = { has_trait_flag = fera_type }
		prev = { POD_set_fera_type = { TRAIT = prev } }
	}
}

POD_remove_fera_on_embrace_effect = {
	if = {
		limit = { has_trait = fera }

		POD_trait_injector_fera = { EFFECT = POD_remove_trait }
		
		POD_shapeshift_revert_all_effect = yes
	}
}

POD_revert_fera_to_kinfolk_effect = {
	if = {
		limit = { has_trait = fera }

		POD_trait_injector_fera = { EFFECT = POD_remove_trait }
		
		POD_shapeshift_revert_all_effect = yes
		
		add_trait = kinfolk
		add_trait = gnosis
		
		POD_remove_wrong_supernatural_traits_on_existing_character = yes
	}
}


POD_fera_pick_pack_totem = {
	add_character_modifier = $MODIFIER$
	hidden_effect = {
		every_relation = {
			type = coterie_member
			add_character_modifier = $MODIFIER$
		}
		var:coterie = {
			set_variable = {
				name = coterie_totem
				value = flag:$MODIFIER$
			}
		}
	}
}

POD_fera_add_pack_totem_modifiers_coterie_effect = {
	if = {
		limit = { POD_fera_coterie_has_pack_totem = yes }
		POD_modifier_injector_coterie_totem = { EFFECT = POD_apply_pack_totem_from_coterie_flag }
	}
}

POD_apply_pack_totem_from_coterie_flag = {
	if = {
		limit = { var:coterie.var:coterie_totem = flag:$MODIFIER$ }
		add_character_modifier = $MODIFIER$
	}
}


POD_fera_initialization_effect = {
	switch = {
		trigger = has_trait
		werewolf = { POD_werewolf_initialization_effect = yes }
		#ajaba = { POD_ajaba_initialization_effect = yes }
		#ananasi = { POD_ananasi_initialization_effect = yes }
		#apis = { POD_apis_initialization_effect = yes }
		bastet = { POD_bastet_initialization_effect = yes }
		#camazotz = { POD_camazotz_initialization_effect = yes }
		#corax = { POD_corax_initialization_effect = yes }
		#grondr = { POD_grondr_initialization_effect = yes }
		#gurahl = { POD_gurahl_initialization_effect = yes }
		#kitsune = { POD_kitsune_initialization_effect = yes }
		#mokole = { POD_mokole_initialization_effect = yes }
		#nagah = { POD_nagah_initialization_effect = yes }
		#nuwisha = { POD_nuwisha_initialization_effect = yes }
		#ratkin = { POD_ratkin_initialization_effect = yes }
		#rokea = { POD_rokea_initialization_effect = yes }
	}

	POD_initialize_energy_modifier = yes
	if = {
		limit = { NOT = { has_character_flag = immune_to_disease } }
		POD_remove_disease_traits_effect = yes
		add_character_flag = immune_to_disease
	}
	if = {
		limit = { is_landed = yes }
		POD_set_fera_tribe_government_effect = yes
	}
	POD_fera_assign_ancestral_language_effect = yes
}

POD_werewolf_initialization_effect = {
	#Faith
	if = {
		limit = {
			top_liege ?= { has_trait = shemsuheru }
			has_relation_worshipped_reborn = top_liege
		}
		set_character_faith = faith:silentstrider
	}
	else_if = {
		limit = {
			is_a_cultist = no
			POD_is_supehunter_trigger = no
			NOR = {
				religion = { is_in_family = rf_wyrm }
				religion = { is_in_family = rf_gaia }
			}
		}
		POD_character_werewolf_faith_initialization_effect = yes
	}
	#Breed
	if = {
		limit = { POD_fera_has_breed_trigger = no }
		
		# Assign Breed at Random
		random_list = {
			10 = {
				modifier = {
					religion = religion:redtalon_religion
					add = 75
				}
				add_trait = beast_born
			}
			10 = {
				modifier = {
					NOT = { religion = religion:redtalon_religion }
					add = 75
				}
				add_trait = homid_born
			}
		}
	}
	#Auspice
	if = {
		limit = { POD_fera_has_lunar_auspice_trigger = no }
		
		# Assign Auspice at Day of Month
		if = {
			limit = { current_day <= 3 }
			add_trait = ragabash
		}
		else_if = {
			limit = { current_day <= 9 }
			add_trait = theurge
		}
		else_if = {
			limit = { current_day <= 15 }
			add_trait = philodox
		}
		else_if = {
			limit = { current_day <= 21 }
			add_trait = galliard
		}
		else_if = {
			limit = { current_day <= 27 }
			add_trait = ahroun
		}
		else = {
			add_trait = ragabash
		}
	}
}

POD_kinfolk_initialization_effect = {
	if = {
		limit = {
			OR = {
				religion = { is_in_family = rf_wyrm }
				religion = { is_in_family = rf_gaia }
			}
		}
		random = {
			chance = 10
			add_trait = gnosis
			POD_initialize_energy_modifier = yes
			POD_resonance_increase_intensity_effect = yes
		}
	}
	if = {
		limit = { NOT = { exists = var:fera_type } }
		random_list = {
			1 = {
				trigger = { real_mother ?= { POD_has_any_inheritable_fera_type = yes } }
				modifier = {
					# make sure a player's children inherit the player's changing breed
					real_father ?= {
						is_ai = no
						POD_has_any_inheritable_fera_type = yes
					}
					factor = 0
				}
				real_mother = { POD_save_fera_type_as = { SAVE_AS = fera_type } }
				POD_set_fera_type = { TRAIT = scope:fera_type }
			}
			1 = {
				trigger = { real_father ?= { POD_has_any_inheritable_fera_type = yes } }
				modifier = {
					# make sure a player's children inherit the player's changing breed
					real_mother ?= {
						is_ai = no
						POD_has_any_inheritable_fera_type = yes
					}
					factor = 0
				}
				real_father = { POD_save_fera_type_as = { SAVE_AS = fera_type } }
				POD_set_fera_type = { TRAIT = scope:fera_type }
			}
		}
	}
	if = {
		limit = {
			is_adult = yes
			NOT = { exists = var:fera_type }
			liege_or_court_owner ?= { POD_has_any_inheritable_fera_type = yes }
		}
		liege_or_court_owner = { POD_save_fera_type_as = { SAVE_AS = fera_type } }
		POD_set_fera_type = { TRAIT = scope:fera_type }
	}
	# safeguard: become garou kinfolk by default
	if = {
		limit = { NOT = { exists = var:fera_type } }
		POD_set_fera_type = { TRAIT = trait:werewolf }
	}
}

POD_fera_assign_ancestral_language_effect = {
	switch = {
		trigger = has_trait

		werewolf = { learn_language = language_ancestral_werewolf }
		bastet = { learn_language = language_ancestral_bastet }
		mokole = { learn_language = language_ancestral_mokole }

		fallback = {}
	}
}
