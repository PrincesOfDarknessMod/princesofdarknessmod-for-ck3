
#scoped to the character that would be affected
POD_fera_delirium_effect = {
	if = {
		limit = { $FERA$ = { POD_inflicts_delirium_trigger = yes } }

		if = {
			limit = {
				OR = {
					$FERA$ = { POD_bypass_immunity_to_the_delirium_trigger = yes }
					POD_immunity_to_the_delirium_trigger = no
				}
				NOT = { has_trait = fera }
			}

			if = {
				limit = { $FERA$ = { POD_inflicts_full_delirium_trigger = yes } }
				add_character_modifier = { modifier = POD_delirium_modifier years = 2 }
			}
			else_if = {
				limit = { $FERA$ = { POD_inflicts_reduced_delirium_trigger = yes } }
				add_character_modifier = { modifier = POD_reduced_delirium_modifier years = 2 }
			}
		}
	}
}


POD_fera_pick_pack_totem = {
	add_character_modifier = $MODIFIER$
	hidden_effect = {
		every_relation = {
			type = coterie_member
			add_character_modifier = $MODIFIER$
		}
		var:coterie = {
			set_variable = {
				name = coterie_totem
				value = flag:$MODIFIER$
			}
		}
	}
}

POD_fera_add_pack_totem_modifiers_coterie_effect = {
	if = {
		limit = { POD_fera_coterie_has_pack_totem = yes }
		POD_modifier_injector_coterie_totem = { EFFECT = POD_apply_pack_totem_from_coterie_flag }
	}
}

POD_apply_pack_totem_from_coterie_flag = {
	if = {
		limit = { var:coterie.var:coterie_totem = flag:$MODIFIER$ }
		add_character_modifier = $MODIFIER$
	}
}

POD_fera_remove_pack_totem = {
	if = {
		limit = { POD_fera_coterie_has_pack_totem = yes }
		custom_tooltip = {
			text = POD_fera_remove_pack_totem_tooltip
			var:coterie = {
				every_in_list = {
					variable = members
					POD_modifier_injector_coterie_totem = { EFFECT = POD_remove_character_modifier }
				}
				remove_variable = coterie_totem
			}
		}
	}
}
