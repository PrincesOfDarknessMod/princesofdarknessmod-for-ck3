


POD_fera_pick_pack_totem = {
	add_character_modifier = $MODIFIER$
	hidden_effect = {
		every_relation = {
			type = coterie_member
			add_character_modifier = $MODIFIER$
		}
		var:coterie = {
			set_variable = {
				name = coterie_totem
				value = flag:$MODIFIER$
			}
		}
	}
}

POD_fera_add_pack_totem_modifiers_coterie_effect = {
	if = {
		limit = { POD_fera_coterie_has_pack_totem = yes }
		POD_modifier_injector_coterie_totem = { EFFECT = POD_apply_pack_totem_from_coterie_flag }
	}
}

POD_apply_pack_totem_from_coterie_flag = {
	if = {
		limit = { var:coterie.var:coterie_totem = flag:$MODIFIER$ }
		add_character_modifier = $MODIFIER$
	}
}

POD_fera_remove_pack_totem = {
	if = {
		limit = { POD_fera_coterie_has_pack_totem = yes }
		custom_tooltip = {
			text = POD_fera_remove_pack_totem_tooltip
			var:coterie = {
				every_in_list = {
					variable = members
					POD_modifier_injector_coterie_totem = { EFFECT = POD_remove_character_modifier }
				}
				remove_variable = coterie_totem
			}
		}
	}
}

# For POD_release_disease_interaction: release a disease to the target's court
# (needs scope:recipient on the TopScope)
POD_judgment_of_pestilence_release_basic_disease = {
	if = {
		limit = { has_trait = $TRAIT$ }
		recover_from_disease_effect = { DISEASE = $TRAIT$ }
		scope:recipient = {
			if = {
				limit = { NOT = { has_character_flag = immune_to_disease } }
				add_trait = $TRAIT$
			}
			every_courtier_or_guest = {
				limit = { NOT = { has_character_flag = immune_to_disease } }
				add_trait = $TRAIT$
			}
			add_opinion = {
				target = prev
				modifier = hurt_opinion
				opinion = $OPINION$
			}
		}
	}
}

# For POD_release_disease_interaction: release a disease as an epidemic
# (needs scope:recipient on the TopScope)
POD_judgment_of_pestilence_release_epidemic = {
	if = {
		limit = { has_trait = $TRAIT$ }
		recover_from_disease_effect = { DISEASE = $TRAIT$ }
		scope:recipient = {
			capital_province = {
				create_epidemic_outbreak = {
					type = $TRAIT$
					intensity = minor
				}
			}
			add_opinion = {
				target = prev
				modifier = hurt_opinion
				opinion = $OPINION$
			}
		}
	}
}
