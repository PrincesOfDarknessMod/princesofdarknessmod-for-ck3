POD_change_wyrm_taint_effect = {
    if = {
        limit = { POD_splat_can_have_wyrm_taint = yes }
        save_scope_value_as = {
            name = new_wyrm_taint
            value = {
                value = wyrm_taint_value
                add = $COUNT$
                min = 0
                max = 100
            }
        }
        custom_description = {
            text = POD_change_wyrm_taint
            value = $COUNT$
            set_variable = {
                name = wyrm_taint
                value = scope:new_wyrm_taint
            }
        }
		if = {
			limit = { $COUNT$ > 0 }
			send_interface_message = {
				type = POD_fera_bad
				title = POD_wyrm_taint_increasing

				custom_description = {
					text = POD_change_wyrm_taint
					value = $COUNT$
				}
			}
		}
        if = {
            limit = { scope:new_wyrm_taint >= 100 }
            trigger_event = POD_wyrm.101
        }
    }
}

POD_yearly_wyrm_taint_effect = {
	POD_change_wyrm_taint_effect = { COUNT = wyrm_taint_yearly_change_value }
}


POD_fera_fall_to_wyrm_effect = {
	if = {
		limit = {
			NOT = { religion = { is_in_family = rf_wyrm } }
			OR = {
				has_trait = fera
				has_trait = kinfolk
			}
		}
		POD_fera_remove_planetary_gifts_effect = yes
		if = {
			limit = {
				POD_fera_coterie_has_gaian_totem = yes
				OR = {
					is_ai = no
					POD_has_coterie_players_trigger = no
				}
			}
			POD_fera_remove_pack_totem = yes
		}

		if = {
			limit = { POD_has_inheritable_fera_type = { TRAIT = trait:werewolf } }
			if = {
				limit = { has_religion = religion:bonegnawer_religion }
				set_character_faith = faith:maneaters
			}
			else_if = {
				limit = { has_faith = faith:hakken }
				set_character_faith = faith:ebon_centipede
			}
			else_if = {
				limit = { has_religion = religion:shadowlord_religion }
				random_list = {
					50 = { set_character_faith = faith:masks }
					50 = { set_character_faith = faith:societyofnidhogg }
				}
			}
			else_if = {
				limit = { has_religion = religion:silentstrider_religion }
				set_character_faith = faith:eatersofthedead
			}
			else_if = {
				limit = { has_religion = religion:stargazer_religion }
				random_list = {
					50 = { set_character_faith = faith:metasticbirth }
					50 = { set_character_faith = faith:ouroboroans }
				}
			}
			else_if = {
				limit = { has_religion = religion:warderofmen_religion }
				set_character_faith = faith:boli_zouhisze
			}
			else = { set_character_faith = faith:blackspiraldancer }
		}
		else_if = {
			limit = { POD_has_inheritable_fera_type = { TRAIT = trait:bastet } }
			set_character_faith = faith:hellcats
		}
		else_if = {
			limit = { POD_has_inheritable_fera_type = { TRAIT = trait:mokole } }
			set_character_faith = faith:mnetics
		}
		#Fallback
		else = { set_character_faith = faith:blackspiraldancer }
	}
}
