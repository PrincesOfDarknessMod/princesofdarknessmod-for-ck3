
POD_create_lunar_auspice_story_effect = {
	if = {
		limit = { NOT = { any_owned_story ?= { has_variable = auspice_story } } }
		switch = {
			trigger = has_trait

			ahroun = { create_story = POD_story_lunar_auspice_ahroun }
			galliard = { create_story = POD_story_lunar_auspice_galliard }
			philodox = { create_story = POD_story_lunar_auspice_philodox }
			ragabash = { create_story = POD_story_lunar_auspice_ragabash }
			theurge = { create_story = POD_story_lunar_auspice_theurge }

			fallback = {}
		}
	}
}

POD_auspice_reward_check = {
	if = {
		limit = {
			POD_lunar_auspice_trait_xp >= 100
			POD_is_werewolf_trigger = yes
			NOR = {
				has_perk = unstoppable_warrior_perk
				has_perk = storyteller_perk
				has_perk = break_the_bonds_perk
				has_perk = firebringer_perk
				has_perk = invoked_presence_perk
			}
		}
		switch = {
			trigger = has_trait

			ahroun = { trigger_event = POD_werewolves.109 }
			galliard = { trigger_event = POD_werewolves.119 }
			philodox = { trigger_event = POD_werewolves.129 }
			ragabash = { trigger_event = POD_werewolves.139 }
			theurge = { trigger_event = POD_werewolves.149 }
		}
	}
}

POD_galliard_song_setup_effect = {
	scope:actor = {
		POD_add_modifier_and_track_stacks = { TYPE = character MODIFIER = galliard_song_upkeep_modifier }
		if = {
			limit = { any_owned_story = { type = POD_story_lunar_auspice_galliard } }
			random_owned_story = { type = POD_story_lunar_auspice_galliard add_to_variable_list = { name = galliard_song_targets target = scope:recipient } }
		}
	}

	save_scope_value_as = { name = song_duration value = POD_galliard_song_duration }


	if = {
		limit = { scope:primal_song_development ?= yes }
		POD_add_flag_to_list = { LIST = galliard_effects  FLAG = flag:primal_song_development }
	}
	if = {
		limit = { scope:primal_song_spirit ?= yes }
		POD_add_flag_to_list = { LIST = galliard_effects  FLAG = flag:primal_song_spirit }
	}
	if = {
		limit = { scope:song_valor_heal ?= yes }
		POD_add_flag_to_list = { LIST = galliard_effects  FLAG = flag:song_valor_heal }
	}
	if = {
		limit = { scope:song_valor_prowess ?= yes }
		POD_add_flag_to_list = { LIST = galliard_effects  FLAG = flag:song_valor_prowess }
	}
	if = {
		limit = { scope:song_rage_irrational ?= yes }
		POD_add_flag_to_list = { LIST = galliard_effects  FLAG = flag:song_rage_irrational }
	}
	if = {
		limit = { scope:song_rage_countermeasures ?= yes }
		POD_add_flag_to_list = { LIST = galliard_effects  FLAG = flag:song_rage_countermeasures }
	}
	if = {
		limit = { scope:song_siren_opinion ?= yes }
		POD_add_flag_to_list = { LIST = galliard_effects  FLAG = flag:song_siren_opinion }
	}
	if = {
		limit = { scope:song_siren_hook ?= yes }
		POD_add_flag_to_list = { LIST = galliard_effects  FLAG = flag:song_siren_hook }
	}
	if = {
		limit = { scope:song_mockery_opinion ?= yes }
		POD_add_flag_to_list = { LIST = galliard_effects  FLAG = flag:song_mockery_opinion }
	}
	if = {
		limit = { scope:song_mockery_prestige ?= yes }
		POD_add_flag_to_list = { LIST = galliard_effects  FLAG = flag:song_mockery_prestige }
	}
	if = {
		limit = { scope:song_serenity_stress ?= yes }
		POD_add_flag_to_list = { LIST = galliard_effects  FLAG = flag:song_serenity_stress }
	}
	if = {
		limit = { scope:song_serenity_war ?= yes }
		POD_add_flag_to_list = { LIST = galliard_effects  FLAG = flag:song_serenity_war }
	}
	if = {
		limit = { scope:song_storm_damage ?= yes }
		POD_add_flag_to_list = { LIST = galliard_effects  FLAG = flag:song_storm_damage }
	}
	if = {
		limit = { scope:song_storm_movement ?= yes }
		POD_add_flag_to_list = { LIST = galliard_effects  FLAG = flag:song_storm_movement }
	}

	add_character_modifier = { modifier = galliard_song_timer_modifier years = POD_galliard_song_duration_effect_value }
	
	custom_tooltip = POD_primal_song_interaction_tt
	
	if = {
		limit = { list_size:galliard_effects > 0 }
		show_as_tooltip = {
			POD_galliard_song_effect_list = yes
		}
		trigger_event = { id = POD_werewolves.110 years = POD_galliard_song_duration_effect_value }
	}
	else = {
		custom_tooltip = NO_EFFECT
	}
}


POD_galliard_song_trigger_effect = {
	scope:recipient = {
		POD_remove_character_modifier = { MODIFIER = galliard_song_development_decline_modifier }
		POD_remove_character_modifier = { MODIFIER = galliard_song_prowess_modifier }
		POD_remove_character_modifier = { MODIFIER = galliard_song_irrational_modifier }
		POD_remove_character_modifier = { MODIFIER = galliard_song_irrational_storyteller_modifier }
		POD_remove_character_modifier = { MODIFIER = galliard_song_countermeasure_modifier }
		POD_remove_character_modifier = { MODIFIER = galliard_song_countermeasure_storyteller_modifier }
		POD_remove_character_modifier = { MODIFIER = galliard_song_vassal_opinion_modifier }
		POD_remove_character_modifier = { MODIFIER = galliard_song_no_war_modifier }
		POD_remove_character_modifier = { MODIFIER = galliard_song_no_war_storyteller_modifier }
		POD_remove_character_modifier = { MODIFIER = galliard_song_speed_modifier }
		if = {
			limit = { has_opinion_modifier = { modifier = galliard_song_of_the_siren_opinion target = scope:actor } }
			remove_opinion = { modifier = galliard_song_of_the_siren_opinion target = scope:actor }
		}
	}
	scope:actor = {
		POD_remove_modifier_and_track_stacks = { TYPE = character MODIFIER = galliard_song_upkeep_modifier }
		if = {
			limit = { any_owned_story = { type = POD_story_lunar_auspice_galliard } }
			random_owned_story = { type = POD_story_lunar_auspice_galliard remove_list_variable = { name = galliard_song_targets target = scope:recipient } }
		}

		#seperate message for xp
		if = {
			limit = { has_trait_xp = { trait = galliard value < 100 } }
			save_scope_value_as = {
				name = galliard_xp
				value = {
					value = 0
					add = list_size:galliard_effects
					divide = 2
					min = 1
				}
			}
			send_interface_message = {
				type = POD_werewolf_message_good
				title = lunar_auspice_xp_gain_message
				right_icon = scope:actor
				
				add_trait_xp = { trait = galliard value = scope:galliard_xp }
			}
		}
	}

	#Notify the user if they're not the target
	if = {
		limit = { scope:actor != scope:recipient }
		scope:actor = {
			send_interface_message = {
				type = POD_werewolf_message_neutral
				title = galliard_song_galliard_notification
				right_icon = scope:recipient
				
				show_as_tooltip = { scope:recipient = { POD_galliard_song_effect_list = yes } }
			}
		}
	}

	#do the actual effects and notification
	scope:recipient = {
		send_interface_message = {
			type = POD_werewolf_message_neutral
			title = galliard_song_galliard_notification
			right_icon = scope:recipient
			
			POD_galliard_song_effect_list = yes
		}
	}
}

POD_galliard_song_effect_list = {
	every_in_list = {
		list = galliard_effects

		switch = {
			trigger = this

			flag:primal_song_development = {
				if = {
					limit = { scope:actor = { has_perk = storyteller_perk } }
					scope:recipient = { add_character_modifier = { modifier = galliard_song_development_decline_modifier years = 10 } }
				}
				scope:recipient = { add_character_modifier = { modifier = galliard_song_development_decline_modifier years = 10 } }
			}

			flag:primal_song_spirit = {
				save_scope_value_as = { name = is_summoned_spirit value = yes }
				if = {
					limit = { scope:actor = { has_perk = storyteller_perk } }
					scope:recipient = { POD_spawn_random_spirit_effect = yes }
				}
				scope:recipient = { POD_spawn_random_spirit_effect = yes }
			}

			flag:song_valor_heal = {
				if = {
					limit = { scope:actor = { has_perk = storyteller_perk } }
					scope:recipient = { POD_heal_aggravated_wound_effect = { COUNT = 2 } }
				}
				scope:recipient = { POD_heal_aggravated_wound_effect = { COUNT = 2 } }
			}

			flag:song_valor_prowess = {
				if = {
					limit = { scope:actor = { has_perk = storyteller_perk } }
					scope:recipient = { add_character_modifier = { modifier = galliard_song_prowess_modifier years = 10 } }
				}
				scope:recipient = { add_character_modifier = { modifier = galliard_song_prowess_modifier years = 10 } }
			}

			flag:song_rage_irrational = {
				if = {
					limit = { scope:actor = { has_perk = storyteller_perk } }
					scope:recipient = { add_character_modifier = { modifier = galliard_song_irrational_storyteller_modifier years = 10 } }
				}
				scope:recipient = { add_character_modifier = { modifier = galliard_song_irrational_modifier years = 10 } }
			}

			flag:song_rage_countermeasures = {
				if = {
					limit = { scope:actor = { has_perk = storyteller_perk } }
					scope:recipient = { add_character_modifier = { modifier = galliard_song_countermeasure_storyteller_modifier years = 10 } }
				}
				scope:recipient = { add_character_modifier = { modifier = galliard_song_countermeasure_modifier years = 10 } }
			}

			flag:song_siren_opinion = {
				if = {
					limit = { scope:actor = { has_perk = storyteller_perk } }
					scope:recipient = { add_opinion = { modifier = galliard_song_of_the_siren_opinion target = scope:actor } }
				}
				scope:recipient = { add_opinion = { modifier = galliard_song_of_the_siren_opinion target = scope:actor } }
			}

			flag:song_siren_hook = {
				if = {
					limit = { scope:actor = { has_perk = storyteller_perk } }
					scope:actor = { add_hook = { type = song_of_the_siren_strong_hook target = scope:recipient } }
				}
				else = { scope:actor = { add_hook = { type = song_of_the_siren_weak_hook target = scope:recipient } } }
			}

			flag:song_mockery_opinion = {
				if = {
					limit = { scope:actor = { has_perk = storyteller_perk } }
					scope:recipient = { add_character_modifier = { modifier = galliard_song_vassal_opinion_modifier years = 10 } }
				}
				scope:recipient = { add_character_modifier = { modifier = galliard_song_vassal_opinion_modifier years = 10 } }
			}

			flag:song_mockery_prestige = {
				#script value handles storyteller perk
				scope:recipient = { add_prestige = { value = POD_galliard_song_mockery_prestige_gain multiply = -1 } }
				scope:actor = { add_prestige = POD_galliard_song_mockery_prestige_gain }
			}

			flag:song_serenity_stress = {
				if = {
					limit = { scope:actor = { has_perk = storyteller_perk } }
					scope:recipient = { POD_static_stress_gain = { COUNT = -100 } }
				}
				else = { scope:recipient = { POD_static_stress_gain = { COUNT = -50 } } }
			}
			
			flag:song_serenity_war = {
				if = {
					limit = { scope:actor = { has_perk = storyteller_perk } }
					scope:recipient = { add_character_modifier = { modifier = galliard_song_no_war_storyteller_modifier years = 10 } }
				}
				scope:recipient = { add_character_modifier = { modifier = galliard_song_no_war_modifier years = 10 } }

				if = {
					limit = { scope:recipient = { is_at_war_as_attacker = yes } }
					scope:recipient = {
						every_character_war = {
							limit = { scope:recipient = { is_attacker_in_war = prev } }
							end_war = white_peace
						}
					}
				}
			}

			flag:song_storm_damage = {
				if = {
					limit = { scope:actor = { has_perk = storyteller_perk } }
					scope:recipient = { POD_increase_wounds_hostile_effect = { ATTACKER = scope:actor COUNT = POD_double_lightning_damage REASON = lightning } }
				}
				else = { scope:recipient = { POD_increase_wounds_hostile_effect = { ATTACKER = scope:actor COUNT = POD_base_lightning_damage REASON = lightning } } }
			}

			flag:song_storm_movement = {
				if = {
					limit = { scope:actor = { has_perk = storyteller_perk } }
					scope:recipient = { add_character_modifier = { modifier = galliard_song_speed_modifier years = 10 } }
				}
				scope:recipient = { add_character_modifier = { modifier = galliard_song_speed_modifier years = 10 } }
			}
		}
	}
}

						education_learning = { scope:ragabash = { add_learning_lifestyle_perk_points = 1 } }

			}


			if = {
