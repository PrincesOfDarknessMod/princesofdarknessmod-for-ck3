
POD_fera_pack_teaching_effect = {
	if = {
		limit = { NOT = { scope:pack_$ATTRIBUTE$_teacher = THIS } }

		if = {
			limit = {
				$ATTRIBUTE$_diff = {
					target = scope:pack_$ATTRIBUTE$_teacher
					value <= -7
				}
			}
			add_$ATTRIBUTE$_skill = 3
		}
		else_if = {
			limit = {
				$ATTRIBUTE$_diff = {
					target = scope:pack_$ATTRIBUTE$_teacher
					value <= -4
				}
			}
			add_$ATTRIBUTE$_skill = 2
		}
		else_if = {
			limit = {
				$ATTRIBUTE$_diff = {
					target = scope:pack_$ATTRIBUTE$_teacher
					value <= -1
				}
			}
			add_$ATTRIBUTE$_skill = 1
		}
	}
}
