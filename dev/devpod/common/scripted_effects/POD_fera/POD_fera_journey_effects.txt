
POD_fera_pack_teaching_effect = {
	if = {
		limit = { NOT = { scope:pack_$ATTRIBUTE$_teacher = THIS } }

		if = {
			limit = {
				$ATTRIBUTE$_diff = {
					target = scope:pack_$ATTRIBUTE$_teacher
					value <= -7
				}
			}
			add_$ATTRIBUTE$_skill = 3
		}
		else_if = {
			limit = {
				$ATTRIBUTE$_diff = {
					target = scope:pack_$ATTRIBUTE$_teacher
					value <= -4
				}
			}
			add_$ATTRIBUTE$_skill = 2
		}
		else_if = {
			limit = {
				$ATTRIBUTE$_diff = {
					target = scope:pack_$ATTRIBUTE$_teacher
					value <= -1
				}
			}
			add_$ATTRIBUTE$_skill = 1
		}
	}
}

POD_fera_pack_philodox_effect = {
	if = {
		limit = { var:coterie ?= { any_in_list = { variable = members has_trait = philodox } } }
		var:coterie ?= {
			ordered_in_list = {
				variable = members
				order_by = {
					value = stewardship
					if = {
						limit = { is_ai = no }
						add = 100
					}
					if = {
						limit = { has_trait = werewolf }
						add = 20
					}
				}
				max = 1
				save_scope_as = pack_philodox
			}
		}
		scope:pack_philodox = {
			if = {
				limit = { has_trait_xp = { trait = philodox value < 100 } }
				add_trait_xp = { trait = philodox value = POD_philodox_xp_gain }
				POD_auspice_reward_check = yes
			}
			
			
			#Perk effects
			if = {
				limit = { has_perk = building_a_legend_perk }
				if = {
					limit = { is_ai = no }
					custom_tooltip = POD_fera_pack_philodox_effect_tooltip
				}
				else = { custom_tooltip = POD_fera_pack_philodox_effect_tooltip2 }
				var:coterie ?= {
					every_in_list = {
						variable = members
						limit = {
							has_trait = lunar_auspice
							NOT = { scope:pack_philodox = this }
						}
						POD_trait_injector_lunar_auspice_with_value = { EFFECT = POD_add_trait_xp VALUE = POD_philodox_building_legend_xp }
					}
				}

				#This could be written better but this is the best format for the tooltip to be readable
				if = {
					limit = { has_perk = bonds_that_tie_us_perk }
					
					custom_tooltip = {
						text = POD_fera_pack_philodox_effect_tooltip_philodox_auspice
						var:coterie ?= { every_in_list = { variable = members add_character_modifier = { modifier = philodox_pack_auspice_philodox_modifier years = 3 } } }
					}
					if = {
						limit = { var:coterie ?= { any_in_list = { variable = members has_trait = ahroun } } }
						custom_tooltip = {
							text = POD_fera_pack_philodox_effect_tooltip_ahroun_auspice
							var:coterie ?= { every_in_list = { variable = members add_character_modifier = { modifier = philodox_pack_auspice_ahroun_modifier years = 3 } } }
						}
					}
					if = {
						limit = { var:coterie ?= { any_in_list = { variable = members has_trait = galliard } } }
						custom_tooltip = {
							text = POD_fera_pack_philodox_effect_tooltip_galliard_auspice
							var:coterie ?= { every_in_list = { variable = members add_character_modifier = { modifier = philodox_pack_auspice_galliard_modifier years = 3 } } }
						}
					}
					if = {
						limit = { var:coterie ?= { any_in_list = { variable = members has_trait = ragabash } } }
						custom_tooltip = {
							text = POD_fera_pack_philodox_effect_tooltip_ragabash_auspice
							var:coterie ?= { every_in_list = { variable = members add_character_modifier = { modifier = philodox_pack_auspice_ragabash_modifier years = 3 } } }
						}
					}
					if = {
						limit = { var:coterie ?= { any_in_list = { variable = members has_trait = theurge } } }
						custom_tooltip = {
							text = POD_fera_pack_philodox_effect_tooltip_theurge_auspice
							var:coterie ?= { every_in_list = { variable = members add_character_modifier = { modifier = philodox_pack_auspice_theurge_modifier years = 3 } } }
						}
					}
				}

				if = {
					limit = { has_perk = rally_to_the_cause_perk }
					var:coterie ?= {
						every_in_list = {
							variable = members
							add_stress = medium_stress_loss
						}
					}
				}

				if = {
					limit = { has_perk = honour_bound_perk }
					var:coterie ?= {
						every_in_list = {
							variable = members
							limit = { has_trait = fera }
							add_prestige = POD_philodox_honour_bound_prestige
							if = {
								limit = { is_ruler = yes is_landed = yes }
								add_legitimacy = POD_philodox_honour_bound_legitimacy
							}
						}
					}
				}

				if = {
					limit = {
						has_perk = wisdom_of_the_ancient_ways_perk
						POD_fera_coterie_has_pack_totem = yes
					}
					every_in_list = {
						variable = members
						add_character_modifier = { modifier = philodox_ancient_ways_modifier years = 3 }
						if = {
							limit = { is_ai = no }
							POD_gnosis_income_effect = { COUNT = 2 }
						}
					}
				}
			}
		}
	}
}


#General rewards for finishing any spirit quest
POD_fera_spirit_quest_rewards_effect = {
	POD_fera_add_renown_effect = { TYPE = wisdom }

	if = {
		limit = {
			has_perk = placation_perk
			num_of_relation_spirit_companion > 0
		}
		save_scope_value_as = {
			name = placation_xp
			value = {
				value = fera_spirit_quest_difficulty
				add = { 1 3 }
			}
		}
		every_relation = {
			type = spirit_companion
			limit = { has_trait_xp = { trait = spirit value < 100 } }
			add_trait_xp = { trait = spirit value = scope:placation_xp }
		}
	}
}
