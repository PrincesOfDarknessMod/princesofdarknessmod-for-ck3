POD_character_initialization_for_entomos_assign_breed_effect = {
	if = {
		limit = {
			POD_is_entomos_trigger = yes
			POD_has_entomos_breed_trigger = no
		}	
		
		# Assign Breed at Random 
		random_list = {
			30 = { add_trait = macro_born }
			30 = { add_trait = micro_born }
			30 = { add_trait = shifter_born }
			# 0 = { add_trait = titan_born }
			10 = { add_trait = homid_born } 
		}
	}
}

POD_character_initialization_for_werewolf_assign_breed_effect = {
	if = {
		limit = {
			POD_is_werewolf_trigger = yes
			POD_has_breed_trigger = no
		}	
		
		# Assign Breed at Random 
		random_list = {
			10 = { 
				modifier = {
					religion = religion:redtalon_religion
					add = 75
				}
				add_trait = beast_born
			}
			10 = { 
				modifier = {
					NOT = { religion = religion:redtalon_religion }
					add = 75
				}
				add_trait = homid_born
			} 
		}
	}
}

POD_shifter_inherit_breed_effect = {
	random_list = {
		50 = {
			trigger = {
				OR = {
					mother ?= { has_trait = homid_born } 
					real_father ?= { has_trait = homid_born }
				}
			}
			add_trait = homid_born
		}
		50 = {
			trigger = {
				OR = {
					mother ?= { has_trait = beast_born } 
					real_father ?= { has_trait = beast_born }
				}
			}
			add_trait = beast_born
		}
	}
	if = {
		limit = { POD_has_breed_trigger = no }
		add_trait = homid_born
	}
}
