POD_shapeshift_remove_ananasi_form_effect = {
	if = {
		limit = { has_character_modifier = POD_ananasi_form_warform_modifier }
		remove_character_modifier = POD_ananasi_form_warform_modifier
	}	
	if = {
		limit = { has_character_modifier = POD_ananasi_form_macroform_modifier }
		remove_character_modifier = POD_ananasi_form_macroform_modifier
	}	
	if = {
		limit = { has_character_modifier = POD_ananasi_form_microform_modifier }
		remove_character_modifier = POD_ananasi_form_microform_modifier
	}	
	if = {
		limit = { has_character_modifier = POD_ananasi_form_swarmform_modifier }
		remove_character_modifier = POD_ananasi_form_swarmform_modifier

	if = {
		limit = { has_character_modifier = POD_ananasi_form_titaniform_modifier }
		remove_character_modifier = POD_ananasi_form_titaniform_modifier
	}
}
