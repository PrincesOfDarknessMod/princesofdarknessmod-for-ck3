POD_fae_initialization_effect = {
	if = {
		limit = { has_kith_trigger = no }
		hidden_effect = { POD_character_fae_kith_initiation_effect = yes }
	}
	if = { 
		limit = { 
			OR = { 
				exists = scope:character_being_changed
				is_from_ruler_designer = yes
			}
		}
		POD_fae_chargen_starting_xp_effect = yes
	}
	else = {
		if = {
			limit = { NOT = { religion = { is_in_family = rf_fae } } }
			POD_character_fae_faith_initiation_effect = yes
		}
	}
	if = {
		limit = { has_glamour_trigger = no }
		add_character_modifier = glamour5_modifier
	}
	if = {
		limit = { NOT = { has_character_flag = immune_to_disease } }
		add_character_flag = immune_to_disease
	}
}

POD_copy_artifact_effect = {
	save_scope_as = owner
	if = {
		limit = { scope:target = { rarity = illustrious } }
		set_artifact_rarity_illustrious = yes
	}
	else_if = {
		limit = { scope:target = { rarity = famed } }
		set_artifact_rarity_famed = yes
	}
	else_if = {
		limit = { scope:target = { rarity = masterwork } }
		set_artifact_rarity_masterwork = yes
	}
	else = {
		set_artifact_rarity_common = yes
	}
	
	POD_copy_artifact_pre_effect = yes
	
	scope:newly_created_artifact = {
		hidden_effect = {
			set_variable = {
				name = active_effigy
				value = yes
			}
			remove_artifact_modifier = artifact_placeholder_modifier
			copy_artifact_modifiers = scope:target
		}
	}
}
