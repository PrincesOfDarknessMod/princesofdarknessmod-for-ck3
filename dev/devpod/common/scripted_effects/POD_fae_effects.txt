POD_fae_initialization_effect = {
	if = { 
		limit = { 
			OR = { 
				exists = scope:character_being_changed
				is_from_ruler_designer = yes
			}
		}
		POD_fae_chargen_starting_xp_effect = yes
	}
	else_if = {
		limit = { has_kith_trigger = no }
		random_list = {
			50 = {
				trigger = { 
					exists = mother
					mother = {
						OR = {
							has_trait = fae
							has_trait = changeling
						}
					}
				}
				POD_custom_copy_kith_traits_effect = { ORIGIN = mother }
			}
			50 = {
				trigger = { 
					exists = real_father
					real_father = {
						OR = {
							has_trait = fae
							has_trait = changeling
						}
					}
				}
				POD_custom_copy_kith_traits_effect = { ORIGIN = real_father }
			}
		}
	}
	if = {
		limit = { has_kith_trigger = no }
		hidden_effect = { POD_character_fae_kith_initiation_effect = yes }
	}
	if = {
		limit = { has_trait = pooka }
		POD_character_pooka_initiation_effect = yes
	}
	if = {
		limit = { NOT = { religion = { is_in_family = rf_fae } } }
		if = { # Maybe Ward?
			limit = { 
				exists = mother 
				mother.religion = { is_in_family = rf_fae }
			}
			set_character_faith = mother.faith
		}
		else_if = {
			limit = { 
				exists = father 
				father.religion = { is_in_family = rf_fae }
			}
			set_character_faith = father.faith
		}
		else = {
			POD_character_fae_faith_initiation_effect = yes
		}
	}
	if = {
		limit = { has_trait = hsien }
		if = {
			limit = { has_chi_trigger = no }
			if = {
				limit = { 
					OR = { 
						has_trait = chujung
						has_trait = nyan
					}
				}
				add_character_modifier = yinvirtue3_modifier
				add_character_modifier = yangvirtue7_modifier
			}
			else_if = {
				limit = { 
					OR = { 
						has_trait = suijen
						has_trait = hengpo
					}
				}
				add_character_modifier = yinvirtue7_modifier
				add_character_modifier = yangvirtue3_modifier
			}
			else_if = {
				limit = { 
					OR = { 
						has_trait = komuko
						has_trait = hanumen
					}
				}
				add_character_modifier = yinvirtue5_modifier
				add_character_modifier = yangvirtue5_modifier
			}
			else_if = {
				limit = { 
					OR = { 
						has_trait = chuihyu
						has_trait = tanuki
					}
				}
				random_list = { 
					50 = {
						add_character_modifier = yinvirtue7_modifier
						add_character_modifier = yangvirtue3_modifier
					}
					50 = {
						add_character_modifier = yinvirtue5_modifier
						add_character_modifier = yangvirtue5_modifier
					}
				}
			}
			else = {
				random_list = { 
					50 = {
						add_character_modifier = yinvirtue3_modifier
						add_character_modifier = yangvirtue7_modifier
					}
					50 = {
						add_character_modifier = yinvirtue5_modifier
						add_character_modifier = yangvirtue5_modifier
					}
				}
			}
			POD_set_chi_effect = yes
		}
	}
	else = {
		if = {
			limit = { has_glamour_trigger = no }
			add_character_modifier = glamour5_modifier
		}
	}
	if = {
		limit = { NOT = { has_character_flag = immune_to_disease } }
		add_character_flag = immune_to_disease
	}
	if = {
		limit = {
			religion = religion:spring_court_religion
			NOT = { knows_language = language_spring }
		}
		learn_language = language_spring
	}
	if = {
		limit = {
			religion = religion:summer_court_religion
			NOT = { knows_language = language_summer }
		}
		learn_language = language_summer
	}
	if = {
		limit = {
			religion = religion:autumn_court_religion
			NOT = { knows_language = language_autumn }
		}
		learn_language = language_autumn
	}
	if = {
		limit = {
			religion = religion:winter_court_religion
			NOT = { knows_language = language_winter }
		}
		learn_language = language_winter
	}
	if = {
		limit = {
			has_title = title:d_crystal_fortress
			NOT = { knows_language = language_ancient_arcadian }
		}
		learn_language = language_ancient_arcadian
	}
}

POD_cantrip_cost_effect = {
	if = {
		limit = {
			any_character_artifact = {
				has_variable = stored_sekhem
				var:stored_sekhem > 0
			}
		}
		ordered_character_artifact = {
			limit = {
				has_variable = stored_sekhem 
				var:stored_sekhem > 0
			}
			order_by = {
				value = 0
				add = var:stored_sekhem
			}
			max = 1
			POD_glamour_vessel_decrease_multiple_effect = { COUNT = $COUNT$ OWNER = prev }
		}
	}
	else = {
		POD_glamour_or_chi_decrease_multiple_effect = { COUNT = $COUNT$ }
	}
}

POD_glamour_or_chi_vessel_decrease_multiple_effect = {
	if = {
		limit = { $OWNER$ = { has_trait = hsien } }
		POD_chi_vessel_decrease_multiple_effect = { COUNT = $COUNT$ OWNER = $OWNER$ }
	}
	else = {
		POD_glamour_vessel_decrease_multiple_effect = { COUNT = $COUNT$ OWNER = $OWNER$ }
	}
}

POD_glamour_vessel_decrease_multiple_effect = {
	if = {
		limit = { var:stored_sekhem >= $COUNT$ }
		custom_description = {
			text = POD_extract_glamour
			subject = $OWNER$
			value = $COUNT$
			change_variable = { name = stored_sekhem subtract = $COUNT$ }
		}
	}
	else = {
		save_scope_value_as = {
			name = excess_glamour
			value = {
				value = $COUNT$
				subtract = var:stored_sekhem
			}
		}
		custom_description = {
			text = POD_extract_glamour
			subject = $OWNER$
			value = var:stored_sekhem
			change_variable = { name = stored_sekhem subtract = var:stored_sekhem }
		}
		$OWNER$ = {
			POD_glamour_decrease_multiple_effect = { COUNT = scope:excess_glamour }
		}
	}
	if = {
		limit = { NOT = { has_variable = temporary_vessel } }
		set_vessel_artifact_name_effect = yes
	}
	if = {
		limit = {
			OR = {
				has_variable = temporary_vessel
				var:stored_sekhem <= 0 
			}
			NOT = { has_variable = permanent_vessel }
		}
		$OWNER$ = {
			send_interface_toast = {
				right_icon = prev
				title = sekhem_vessel_damaged
				prev = { degrade_vessel_durability_effect = yes }
			}
		}
	}
}

POD_glamour_increase_effect = {
	if = {
		limit = {has_character_modifier = glamour0_modifier}
		remove_character_modifier = glamour0_modifier
		add_character_modifier = glamour1_modifier
	}
	else_if = {
		limit = {has_character_modifier = glamour1_modifier}
		remove_character_modifier = glamour1_modifier
		add_character_modifier = glamour2_modifier
	}
	else_if = {
		limit = {has_character_modifier = glamour2_modifier}
		remove_character_modifier = glamour2_modifier
		add_character_modifier = glamour3_modifier
	}
	else_if = {
		limit = {has_character_modifier = glamour3_modifier}
		remove_character_modifier = glamour3_modifier
		add_character_modifier = glamour4_modifier
	}
	else_if = {
		limit = {has_character_modifier = glamour4_modifier}
		remove_character_modifier = glamour4_modifier
		add_character_modifier = glamour5_modifier
	}
	else = {
		random_character_artifact = {
			limit = {
				trigger_if = {
					limit = { NOT = { has_variable = greater_sekhem_vessel } }
					has_variable = stored_sekhem
					var:stored_sekhem < 5
				}
				trigger_else = {
					has_variable = stored_sekhem
					var:stored_sekhem < 20
				}
			}
			change_variable = { name = stored_sekhem add = 1 }
			set_vessel_artifact_name_effect = yes
		}
	}
}
