POD_mummy_initialization_effect = {
	if = {
		limit = { has_mummy_trait_trigger = no }
		hidden_effect = { add_trait = shemsuheru }
	}
	if = { 
		limit = { 
			OR = { 
				exists = scope:character_being_changed
				is_from_ruler_designer = yes
			}
		}
		POD_change_government_if_landed = { GOVERNMENT = cult_government }
		POD_character_mummy_dynasty_initiation_effect = yes
		#POD_mummy_random_perk_initialization_for_noble_effect = yes
	}
	else = {
		if = {
			limit = {
				house.house_head ?= { is_amenti_trigger = yes } # housemember are neteru 
				NOT = { house.house_head = this }
			}
			POD_create_cadet_branch_effect = yes
		}
	}
	if = {
		limit = { NOT = { religion = { is_in_family = rf_mummies } } }
		POD_character_mummy_faith_initiation_effect = yes
	}
	if = {
		limit = { is_east_mummy_trigger = yes }
		if = {
			limit = { has_chi_trigger = no }
			add_character_modifier = yangvirtue10_modifier
			add_character_modifier = yangchi10_modifier
		}
	}
	else = {
		if = {
			limit = { has_sekhem_trigger = no }
			POD_set_sekhem_effect = yes
		}
	}
	if = {
		limit = { is_ai = no }
		if = {
			limit = { 
				faith = faith:faces_of_apophis 
				NOT = { exists = var:aggravated_natural_hazard_count }
			}
		}
		if = {
			limit = { 
				OR = {
					faith = faith:ishmaelite
					faith = faith:cabiri
				} 
				NOT = { exists = var:stopped_natural_hazard_count }
			}
			set_variable = { name = stopped_natural_hazard_count value = 0 }
		}
	}
	add_to_global_variable_list = {
		name = mummy_list
		target = this
	}
}

POD_character_mummy_dynasty_initiation_effect = {
	house = { save_scope_as = coa_target }
	if = {
		limit = { 
			has_trait = shemsuheru 
			faith = faith:code_of_horus
			NOT = { dynasty = dynasty:POD_mummy_dynasty_horus }
		}
		set_house = character:792593.house
	}
	else_if = {
		limit = { 
			has_trait = shemsuheru 
			faith = faith:ishmaelite
			NOT = { dynasty = dynasty:POD_mummy_dynasty_ishmaelites }
		}
		set_house = character:792607.house
	}
	else_if = {
		limit = { 
			has_trait = banemummy 
			NOT = { dynasty = dynasty:POD_mummy_dynasty_apepnu }
		}
		set_house = character:792611.house
	}
	else_if = {
		limit = { 
			has_trait = cabiri 
			NOT = { dynasty = dynasty:POD_mummy_dynasty_cabiri }
		}
		set_house = character:792600.house
	}
	else_if = {
		limit = { 
			is_east_mummy_trigger = yes
			NOT = { dynasty = dynasty:POD_mummy_dynasty_wufeng }
			NOT = { dynasty = dynasty:POD_mummy_dynasty_xianlung }
		}
		trigger_event = POD_chargen.1801
	}
	if = {
		limit = {
			house.house_head = { is_amenti_trigger = yes } # housemember are neteru 
			NOT = { house.house_head = this }
		}
		POD_create_cadet_branch_effect = yes
	}
}

		limit = {

			$LIEGE$ = { 
				OR = { 
					has_trait = banemummy
					has_trait = wukuei
				}
			} 
		}
		POD_become_wyrm_servant_effect = yes
	}
	else_if = {
		limit = {
			exists = $LIEGE$
			OR = {
				$LIEGE$.faith = { has_doctrine = mummy_special_kingdom_hades_doctrine }
				$LIEGE$.faith = { has_doctrine = mummy_special_kingdom_hellheim_doctrine }
				$LIEGE$.faith = { has_doctrine = mummy_special_kingdom_yomi_doctrine }
				$LIEGE$.faith = { has_doctrine = mummy_special_kingdom_hell_doctrine }
			}
		}
		set_character_faith = $LIEGE$.faith
	}
	else_if = {
		limit = {
			exists = $LIEGE$
			
			$LIEGE$.faith = faith:cabiri

		}
		set_character_faith = $LIEGE$.faith
	}	

	else = {
		if = {
			limit = { has_POD_vamp_blood_trigger = yes }
			set_character_faith = faith:childofosiris
		}
		if = {
			limit = { 
				OR = { 
					has_trait = werewolf
					has_trait = kinfolk
				}
			}
			set_character_faith = faith:silentstrider
		}
		if = {
			limit = { 
				has_trait = supehunter
				NOT  = { has_trait = shih }
			}
			set_character_faith = faith:ikhwan_al_safa
		}
	}
}

POD_hekau_cost_effect = {
	# COUNT can be a scriptvalue, and we're passing it in a different
	# context than we received it, so save its value here.
	save_temporary_scope_value_as = {
		name = hekau_cost_count
		value = $COUNT$
	}
	if = {
		limit = {
			any_character_artifact = {
				has_variable = energy_stored
				var:energy_stored > 0
			}
		}
		ordered_character_artifact = {
			limit = {
				has_variable = energy_stored 
				var:energy_stored > 0
			}
			order_by = {
				value = 0
				add = var:energy_stored
			}
			position = 0
			POD_energy_vessel_decrease_multiple_effect = { COUNT = scope:hekau_cost_count }
		}
	}
	else = {
		POD_sekhem_or_chi_decrease_multiple_effect = { COUNT = scope:hekau_cost_count }
	}
}

POD_sekhem_income_effect = {
	if = {
		limit = {
			OR = {
				sekhem_difference >= $COUNT$
				NOT = {
					any_character_artifact = {
						NOT = { has_variable = temporary_vessel }
						has_variable = energy_stored
						vessel_empty_energy > 0
					}
				}
			}
		}
		POD_sekhem_increase_multiple_effect = { COUNT = $COUNT$ }
	}
	else = {
		save_scope_value_as = {
			name = excess_sekhem
			value = {
				value = $COUNT$
				subtract = sekhem_difference
			}
		}
		POD_sekhem_increase_multiple_effect = { COUNT = sekhem_difference }
		ordered_character_artifact = {
			limit = {
				NOT = { has_variable = temporary_vessel }
				has_variable = energy_stored
				vessel_empty_energy > 0
			}
			order_by = {
				value = 0
				add = vessel_empty_energy
			}
			position = 0
			POD_energy_vessel_increase_multiple_effect = { COUNT = scope:excess_sekhem }
		}
	}
}

POD_weather_magic_specific_effect = {
	every_held_title = {
		limit = {
			tier = tier_county
		}
		county = {
			POD_weather_magic_effect = yes
		}
	}
	if = {
		limit = {
			is_ai = no
		}
		send_interface_toast = {
			title = POD_weather_magic_from_enemy_toast
			left_icon = scope:recipient
			right_icon = scope:actor
			custom_tooltip = POD_weather_magic_from_enemy.tt
		}
	}
	if = {
		limit = {
			NOT = { this = scope:actor }
		}
		add_opinion = {
			target = scope:actor
			modifier = POD_cast_evil_weather_magic_to_my_realm
		}
	}
}

POD_discover_true_name_effect = {
	if = {
		limit = {
			OR = {
				$MASTER$ = {
					has_hook_of_type = {
						target = $VICTIM$
						type = half_true_name_hook
					}
				}
				$VICTIM$ = { has_supernatural_race_trigger = no }
			}
		}
		$MASTER$ = {
			custom_tooltip = POD_discover_true_name_growing
			remove_hook = { target = $VICTIM$ }
			add_hook = {
				target = $VICTIM$
				type = true_name_hook
			}
		}
	}
	else_if = {
		limit = {
			OR = {
				$MASTER$ = {
					has_hook_of_type = {
						target = $VICTIM$
						type = partial_true_name_hook
					}
				}
				$VICTIM$ = { has_main_supernatural_race_trigger = no }
			}
		}
		$MASTER$ = {
			custom_tooltip = POD_discover_true_name_growing
			remove_hook = { target = $VICTIM$ }
			add_hook = {
				target = $VICTIM$
				type = half_true_name_hook
			}
		}
	}
	else = {
		$MASTER$ = {
			custom_tooltip = POD_discover_true_name_growing
			if = {
				limit = { has_hook = $VICTIM$ }
				remove_hook = { target = $VICTIM$ }
			}
			add_hook = {
				target = $VICTIM$
				type = partial_true_name_hook
			}
		}
	}
}

		value = POD_visual_age

		limit = { has_trait = mortal }
		remove_trait = mortal	
	}
	if = { 
		limit = {
			has_trait = supehunter 
			NOT = { has_character_flag = chargen_changing_splat_flag }
			scope:domitor ?= { POD_is_supehunter_trigger = no }
		}
		scope:domitor = { save_scope_as = vampiresire }
		remove_trait = supehunter
		POD_hunterembrace_effect = yes
	}
	if = {
		limit = { POD_splat_mixing_haight_mode = no }
		if = {
			limit = { has_trait = kinfolk }
			remove_trait = kinfolk
			if = {
				limit = { has_trait = gnosis }
				remove_trait = gnosis
			}
		}
		if = {
			limit = { has_trait = kinain }
			remove_trait = kinain
		}
	}
	if = {
		limit = { has_trait = changeling }
		add_trait = lunatic_genetic # Dauntain
	}
