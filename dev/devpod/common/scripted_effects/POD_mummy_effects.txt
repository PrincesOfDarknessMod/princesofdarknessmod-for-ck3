POD_mummy_initialization_effect = {
	if = {
		limit = { has_mummy_trait_trigger = no }
		hidden_effect = { add_trait = shemsuheru }
	}
	if = { 
		limit = { 
			OR = { 
				exists = scope:character_being_changed
				is_from_ruler_designer = yes
			}
		}
		change_government = cult_government
		POD_character_mummy_dynasty_initiation_effect = yes
		POD_mummy_random_perk_initialization_for_noble_effect = yes
	}
	else = {
		if = {
			limit = {
				house.house_head ?= { has_trait = mummy }
				NOT = { house.house_head = this }
			}
			POD_create_cadet_branch_effect = yes
		}
	}
	if = {
		limit = { NOT = { religion = { is_in_family = rf_mummies } } }
		POD_character_mummy_faith_initiation_effect = yes
	}
	if = {
		limit = { is_east_mummy_trigger = yes }
		if = {
			limit = { has_chi_trigger = no }
			add_character_modifier = yangvirtue10_modifier
			add_character_modifier = yangchi10_modifier
		}
	}
	else = {
		if = {
			limit = { has_sekhem_trigger = no }
			POD_set_sekhem_effect = yes
		}
	}
	if = {
		limit = {
			has_alchemy_craft_slot_trigger = no
		}
		POD_set_alchemy_craft_slot_effect = yes
	}
	if = {
		limit = { is_ai = no }
		if = {
			limit = { 
				faith = faith:faces_of_apophis 
				NOT = { exists = var:aggravated_natural_hazard_count }
			}
		}
		if = {
			limit = { 
				OR = {
					faith = faith:ishmaelite
					faith = faith:cabiri
				} 
				NOT = { exists = var:stopped_natural_hazard_count }
			}
			set_variable = { name = stopped_natural_hazard_count value = 0 }
		}
	}
	add_to_global_variable_list = {
		name = mummy_list
		target = this
	}
}

POD_character_mummy_dynasty_initiation_effect = {
	house = { save_scope_as = coa_target }
	if = {
		limit = { 
			has_trait = shemsuheru 
			faith = faith:code_of_horus
			NOT = { dynasty = dynasty:POD_mummy_dynasty_horus }
		}
		set_house = character:792593.house
	}
	else_if = {
		limit = { 
			has_trait = shemsuheru 
			faith = faith:ishmaelite
			NOT = { dynasty = dynasty:POD_mummy_dynasty_ishmaelites }
		}
		set_house = character:792607.house
	}
	else_if = {
		limit = { 
			has_trait = banemummy 
			NOT = { dynasty = dynasty:POD_mummy_dynasty_apepnu }
		}
		set_house = character:792611.house
	}
	else_if = {
		limit = { 
			has_trait = cabiri 
			NOT = { dynasty = dynasty:POD_mummy_dynasty_cabiri }
		}
		set_house = character:792600.house
	}
	else_if = {
		limit = { 
			is_east_mummy_trigger = yes
			NOT = { dynasty = dynasty:POD_mummy_dynasty_wufeng }
			NOT = { dynasty = dynasty:POD_mummy_dynasty_xianlung }
		}
		trigger_event = POD_chargen.1801
	}
	if = {
		limit = {
			house.house_head = { has_trait = mummy }
			NOT = { house.house_head = this }
		}
		POD_create_cadet_branch_effect = yes
	}
}

POD_hekau_cost_effect = {
	# COUNT can be a scriptvalue, and we're passing it in a different
	# context than we received it, so save its value here.
	save_temporary_scope_value_as = {
		name = hekau_cost_count
		value = $COUNT$
	}
	if = {
		limit = {
			any_character_artifact = {
				has_variable = stored_sekhem
				var:stored_sekhem > 0
			}
		}
		ordered_character_artifact = {
			limit = {
				has_variable = stored_sekhem 
				var:stored_sekhem > 0
			}
			order_by = {
				value = 0
				add = var:stored_sekhem
			}
			max = 1
			POD_sekhem_or_chi_vessel_decrease_multiple_effect = { COUNT = scope:hekau_cost_count OWNER = prev }
		}
	}
	else = {
		POD_sekhem_or_chi_decrease_multiple_effect = { COUNT = scope:hekau_cost_count }
	}
}

	pod_mummy_amulets_craft_wealth_effect = yes

		description = placeholder #properly set later
		visuals = necklace

		quality = scope:quality
		save_scope_as = newly_created_artifact

pod_mummy_amulets_craft_quality_effect = {

	# Calculate the final quality of the artifact
	save_scope_value_as = {
		name = quality
		value = {
			value = 0

			scope:owner = {
				add = {	
					value = learning
					multiply = {
						value = var:amulets_craft_outcome
					}
				}
			}

			# Can't have negative quality
			min = 1
		}
	}
}

pod_mummy_amulets_craft_wealth_effect = {
	save_scope_value_as = {
		name = wealth
		value = {
			if = {
				# If our artifact already has a quality, set base wealth equal to 1/4th quality value.
				limit = { exists = scope:quality }
				add = scope:quality
				multiply = 0.25
				round = yes

			}
			else = {
				add = 10 # Otherwise, set base wealth value to 10.
			}

			if = {
				limit = {
					exists = scope:owner
					scope:owner.highest_held_title_tier >= 0
				}
				add = {
					# Add 10 wealth per character tier (up to a max of 50 wealth for emperors)
					value = scope:owner.highest_held_title_tier
					multiply = 10
				}

				# Add up to 25 wealth, scaling with capital's development
				if = {
					limit = {
						exists = scope:owner.capital_county
					}
					scope:owner.capital_county = {
						add = {
							value = development_level
							multiply = 0.25
						}
					}
				}
			}
		}
	}
}

POD_discover_true_name_effect = {
	if = {
		limit = {
			OR = {
				$MASTER$ = {
					has_hook_of_type = {
						target = $VICTIM$
						type = half_true_name_hook
					}
				}
				$VICTIM$ = { has_supernatural_race_trigger = no }
			}
		}
		$MASTER$ = {
			custom_tooltip = POD_discover_true_name_growing
			remove_hook = { target = $VICTIM$ }
			add_hook = {
				target = $VICTIM$
				type = true_name_hook
			}
		}
	}
	else_if = {
		limit = {
			OR = {
				$MASTER$ = {
					has_hook_of_type = {
						target = $VICTIM$
						type = partial_true_name_hook
					}
				}
				$VICTIM$ = { has_trait = ghoul }
				$VICTIM$ = { has_trait = revenant }
			}
		}
		$MASTER$ = {
			custom_tooltip = POD_discover_true_name_growing
			remove_hook = { target = $VICTIM$ }
			add_hook = {
				target = $VICTIM$
				type = half_true_name_hook
			}
		}
	}
	else = {
		$MASTER$ = {
			custom_tooltip = POD_discover_true_name_growing
			if = {
				limit = { has_hook = $VICTIM$ }
				remove_hook = { target = $VICTIM$ }
			}
			add_hook = {
				target = $VICTIM$
				type = partial_true_name_hook
			}
		}
	}
}
