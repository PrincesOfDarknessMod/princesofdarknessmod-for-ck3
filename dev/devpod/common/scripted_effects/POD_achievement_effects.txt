POD_achiev_coterietutorial_check = {
	if = {
		limit = {
			var:coterie ?= {
				has_variable = canon_coterie # backwards compatibility
				var:canon_coterie = no
				any_in_list = {
					variable = members
					is_ai = no
				}
			}
			POD_coterie_at_max_size_trigger = yes
		}
		set_global_variable = pod_achiev_coterietutorial # Achievement: By Our Powers Combined...
	}
}

				}
			}
		}
		set_global_variable = pod_achiev_helena2 # Achievement: A Fitting Start for this Fanfiction
	}
}

POD_achiev_nomnom_initialize = {
	if = {
		limit = {
			is_ai = no
			POD_is_vampire_trigger = yes
			has_trait = vampgeneration
			has_trait_rank = {
				trait = vampgeneration
				rank >= 13
			}
		}
		add_character_flag = valid_for_achiev_nomnom
	}
}

POD_achiev_nomnom_check = {
	if = {
		limit = {
			is_ai = no
			has_character_flag = valid_for_achiev_nomnom
			POD_is_vampire_trigger = yes
			has_trait = vampgeneration
			has_trait_rank = {
				trait = vampgeneration
				rank <= 4
			}
		}
		set_global_variable = pod_achiev_nomnom # Achievement: Ate My Way Up The Ladder
	}
}

	}
}

POD_achiev_redlist_eligibility_check = {
	if = {
		limit = {
			POD_provoke_vatican_completed_trigger = yes
			NOT = { has_game_rule = age_of_inquisition_difficulty_hard }
		}
		# for naughty players who switch game rules while inquisitors are spawning
		set_global_variable = pod_achiev_redlist_unavailable
	}
}

POD_achiev_redlist_check = {
	if = {
		limit = {

			has_game_rule = age_of_inquisition_difficulty_hard
			NOT = { has_global_variable = pod_achiev_redlist_unavailable }
		}
		set_global_variable = pod_achiev_redlist # Achievement: Bragging Rights
	}
}
