POD_achievement_monthly_check = {
	# don't need to check for is_ai = no since that's already in the event trigger
	if = {
		limit = {
			has_trait = bastet
			has_faith = faith:ceilican
			has_title = title:e_avalon
			character:505071 = {
				OR = {
					is_alive = no
					has_trait = torpor
				}
			}
		}
		set_global_variable = pod_achievement_34 # Achievement: Catching the Red Dot
	}
}

POD_achievement_24_check = {
	if = {
		limit = {
			OR = {
				root = { is_ai = no }
				scope:target = { is_ai = no }
			}
			# need to check twice because relationship on_actions aren't commutative. paradox pls
			OR = {
				AND = {
					root = character:790179 # Helena
					scope:target = character:790309 # Menele
				}
				AND = {
					root = character:790309 # Menele
					scope:target = character:790179 # Helena
				}
			}
		}
		set_global_variable = pod_achievement_24 # Achievement: A Fitting Start for this Fanfiction
	}
}
