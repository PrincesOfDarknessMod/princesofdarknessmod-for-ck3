	if = {
		limit = { has_$TYPE$_modifier = $MODIFIER$ }
		remove_$TYPE$_modifier = $MODIFIER$
		hidden_effect = {
			if = {
				limit = { has_variable = $MODIFIER$_stacks }
				change_variable = {
					name = $MODIFIER$_stacks

				}

					remove_variable = $MODIFIER$_stacks
				}
			}
		}
	}
}

# trigger an event with a delay (like trigger_event)
# but also clears all saved scopes in the process
#
# necessary since clear_saved_scope doesn't work in future events of a chain
# which causes bugs if e.g. an event chain has multiple single combat duels
#
# need to pass ID and DAYS
# only works for character scopes
POD_clear_topscope_and_trigger_event = {
	hidden_effect = {
		POD_save_event_as_var = {
			VAR = jank_event_id
			ID = $ID$
		}
		set_variable = {
			name  = jank_event_delay
			value = $DAYS$
		}
		run_interaction = {
			interaction = POD_trigger_delayed_event_interaction
			actor = this
			recipient = this
			execute_threshold = decline
		}
	}
}

	}
}

POD_while_accepting_decimals = {
	save_scope_value_as = { name = value_decimal value = { value = $VALUE$ modulo = 1 } }
	save_scope_value_as = { name = value_full value = { value = $VALUE$ subtract = scope:value_decimal } }
	if = { limit = { $VALUE$ = 1 }
		$EFFECT$
	}
	if = { limit = { $VALUE$ > 1 }
		while = {
			count = scope:value_full
			$EFFECT$
		}
	}
	if = { limit = { NOT = { scope:value_decimal = 0 } }
		random = {
			chance = { value = scope:value_decimal multiply = 100 }
			$EFFECT$
		}
	}
}
