POD_disappearance_effect = {
	trigger_event = {
		id = POD_disappearance.1
		days = 1
	}
}

POD_release_effect = {
	if = {
		limit = { is_imprisoned = yes }
		add_character_flag = {
			flag = escaped_from_prison_flag
			days = 1
		}
		add_character_flag = { #To make sure you don't get spammed about the release
			flag = block_for_prison_release_notification
			days = 1
		}
		release_from_prison = yes
		if = {
			limit = { is_ruler = no }
			move_to_pool = yes
		}
	}
}

POD_depose_effect = {
	if = {
		limit = { is_imprisoned = yes }
		add_character_flag = {
			flag = escaped_from_prison_flag
			days = 1
		}
		add_character_flag = { #To make sure you don't get spammed about the release
			flag = block_for_prison_release_notification
			days = 1
		}
		release_from_prison = yes
	}
	if = {
		limit = { is_ruler = yes }
		depose = yes
	}
	move_to_pool = yes
}

POD_run_effect_if_alive = {
	if = {
		limit = { is_alive = yes }
		$EFFECT$ = yes
	}
}

POD_add_modifier_and_track_stacks = {
	add_$TYPE$_modifier = $MODIFIER$
	hidden_effect = {
		if = {
			limit = { has_variable = $MODIFIER$_stacks }
			change_variable = {
				name = $MODIFIER$_stacks
				add  = 1
			}
		}
		else = {
			set_variable = {
				name  = $MODIFIER$_stacks
				value = 1
			}
		}
	}
}

POD_save_event_as_var = {
	save_scope_value_as = {
		name  = $VAR$
		value = event_id:$ID$
	}
	set_variable = {
		name  = $VAR$
		value = scope:$VAR$
	}
}

# saves the scoped character's prowess value as scope:real_prowess (even beyond 100)
POD_save_real_prowess = {
	hidden_effect = {
		save_scope_value_as = {
			name  = real_prowess
			value = prowess
		}
		if = {
			limit = { prowess >= 100 }
			add_character_modifier = save_real_prowess_negate_modifier
			POD_save_real_prowess_iteration = yes
			while = {
				limit = { prowess >= 100 }
				POD_save_real_prowess_iteration = yes
			}
			remove_character_modifier = save_real_prowess_negate_modifier
			remove_all_character_modifier_instances = save_real_prowess_modifier
			force_character_skill_recalculation = yes
		}
	}
}
POD_save_real_prowess_iteration = {
	add_character_modifier = save_real_prowess_modifier
	force_character_skill_recalculation = yes
	save_scope_value_as = {
		name  = real_prowess
		value = {
			value = scope:real_prowess
			add   = prowess
		}
	}
}

POD_vassals_imperial_up_effect = {
	
	every_vassal_or_below = {
		limit = {
			POD_imperial_governement_trigger = no
		}
			if = {
				limit = {
					has_vamp_government = yes
				}
			change_government = imperial_unity_government
			}
			
			else_if = {
				limit = {
					has_government = litany_government
				}
			change_government = imperial_tribal_government
			}
			
			else = {
				change_government = imperial_no_unity_government
			}
	}
	
}

POD_imperial_up_effect = {
	if = {
		limit = {
			has_vamp_government = yes
		}
		change_government = imperial_unity_government
	}
	
	else_if = {
		limit = {
			has_government = litany_government
		}
		change_government = imperial_tribal_government
	}
	
	else_if  = {
		limit = {
			OR = {
				is_mummy_trigger = yes
				AND = {
					is_supehunter_trigger = yes
					NOT = { has_trait = shih }
					has_faith = faith:shadowinquisition
				}
				is_demon_trigger = yes
			}				
		}
		change_government = imperial_no_unity_government
	}
	
	else = {}
	
	if = {
		limit = {
			NOT = { has_realm_law = crown_authority_3 }
		}
		add_realm_law = crown_authority_3			
	}
}

POD_imperial_down_effect = {
	if = {
		limit = {
			is_vampire_trigger = yes
		}
		change_government = traditions_government
	}
	
	else_if = {
		limit = {
			OR = {
				is_mummy_trigger = yes
				is_a_neteru = yes
			}
		}
		change_government = cult_government
	}
	
	else_if  = {
		limit = {
			is_supehunter_trigger = yes
			NOT = { has_trait = shih }
			has_faith = faith:shadowinquisition		
		}
		change_government = inquisition_government
	}

	else_if  = {
		limit = {
			is_demonic_trigger = yes
		}
		change_government = infernal_court_government
	}
	
	else_if  = {
		limit = {
			has_trait = shapeshifter			
		}
		change_government = litany_government
	}
}

###Bandaid because our AIs will never call to arms and we can't figure out why
# macro, PARTICIPANT needs to be either "attacker" or "defender"

POD_war_ai_ally_bandaid_effect = {
	$PARTICIPANT$ = { save_temporary_scope_as = actor }
	scope:war = { save_temporary_scope_as = target }
	every_ally = {
		limit = {
			is_ai = yes
			prev = {
				is_character_interaction_valid = {
					recipient = prev
					interaction = call_ally_interaction
				}
			}
			save_temporary_scope_as = recipient

			### has_valid_target_showing_failures_only from call_ally_interaction
			# since is_character_interaction_potentially_accepted doesn't work with interactions with a scope:target

			scope:target = {
				is_war_leader = scope:actor
			}

			trigger_if = {
				limit = {
					scope:target = {
						OR = {
							using_cb = undirected_great_holy_war
							using_cb = directed_great_holy_war
						}
					}
				}

				trigger_if = {
					limit = {
						this = scope:target.casus_belli.primary_attacker
					}
					#if it's a religious war the recipient has to be of the same faith
					faith = scope:recipient.faith
				}
				trigger_else = {
					scope:target = { NOT = { primary_attacker.faith = scope:recipient.faith } }
				}
			}

			joiner_not_already_in_another_war_with_any_target_war_participants_trigger = {
				WARRIOR = scope:actor
				JOINER = scope:recipient
			}

			### ai_accept blockers from call_ally_interaction

			trigger_if = { # Refuse call to conflicts of little benefit
				limit = { scope:target = { using_cb = fp2_border_raid } }
				scope:recipient = { # Unless they like going on raids
					OR = {
						has_trait = viking
						has_trait = reaver
					}
				}
			}

			trigger_if = {  # Refuse call against Heir
				limit = { exists = scope:recipient.player_heir }
				trigger_if = {
					limit = { scope:target.casus_belli.primary_defender = scope:actor }
					NOT = { scope:target.casus_belli.primary_attacker = scope:recipient.player_heir }
				}
				trigger_else = {
					NOT = { scope:target.casus_belli.primary_defender = scope:recipient.player_heir }
				}
			}
			
			trigger_if = { # Refuse call against Spouse
				limit = { scope:target.casus_belli.primary_defender = scope:actor }
				NOT = {
					any_spouse = {
						this = scope:target.casus_belli.primary_attacker
					}
				}
			}
			trigger_else = {
				NOT = {
					any_spouse = {
						this = scope:target.casus_belli.primary_defender
					}
				}
			}
		}
		
		call_ally_interaction_effect = yes
	}
}
