POD_increment_scope_value = {
	save_scope_value_as = {
		name = $SCOPE$
		value = {
			value = scope:$SCOPE$
			add = 1
		}
	}
}

POD_toggle_variable = {
	if = {
		limit = { has_variable = $VAR$ }
		remove_variable = $VAR$
	}
	else = {
		set_variable = $VAR$
	}
}

POD_remove_character_modifier = {
	if = {
		limit = { has_character_modifier = $MODIFIER$ }
		remove_character_modifier = $MODIFIER$
	}
}

POD_disappearance_effect = {
	trigger_event = {
		id = POD_disappearance.1
		days = 1
	}
}

POD_release_effect = {
	if = {
		limit = { is_imprisoned = yes }
		add_character_flag = {
			flag = escaped_from_prison_flag
			days = 1
		}
		add_character_flag = { #To make sure you don't get spammed about the release
			flag = block_for_prison_release_notification
			days = 1
		}
		release_from_prison = yes
		if = {
			limit = { is_ruler = no }
			move_to_pool = yes
		}
	}
}

POD_depose_effect = {
	if = {
		limit = { is_ruler = yes }
		depose_effect = { DEPOSER = root }
	}
	if = {
		limit = { is_imprisoned = no }
		move_to_pool = yes
	}
}

POD_long_depose_effect = {
	add_trait = torpor
	POD_depose_effect = yes
	random_landed_ruler = { save_scope_as = new_owner }
	every_character_artifact = { 
		set_owner = {
			target = scope:new_owner
			history = {
				type = inherited
				recipient = scope:new_owner
			}
		}
	}
}

# used when becoming an adventurer
POD_transfer_all_titles_to_character = {
	$CHARACTER$ = { save_scope_as = inheritor }
	
	hidden_effect = {
		# Transfer landed titles
		every_held_title = {
			limit = {
				is_landless_type_title = no
				tier > tier_barony
			}
			add_to_list = target_titles
		}
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = change
			add_claim_on_loss = no
		}
		hidden_effect = {
			every_in_list = {
				list = target_titles
				change_title_holder_include_vassals = {
					holder = scope:inheritor
					change = scope:change
				}
			}
			resolve_title_and_vassal_change = scope:change
		}
	}
}

POD_run_effect_if_alive = {
	if = {
		limit = { is_alive = yes }
		$EFFECT$ = yes
	}
}

POD_add_modifier_and_track_stacks = {
	add_$TYPE$_modifier = $MODIFIER$
	hidden_effect = {
		if = {
			limit = { has_variable = $MODIFIER$_stacks }
			change_variable = {
				name = $MODIFIER$_stacks
				add  = 1
			}
		}
		else = {
			set_variable = {
				name  = $MODIFIER$_stacks
				value = 1
			}
		}
	}
}

# to make sure existing portrait flags don't get overwritten or removed afterwards
# use both in the same event (immediate + after blocks)
POD_apply_temporary_portrait_flag = {
	if = {
		limit = { NOT = { has_character_flag = $FLAG$ } }
		add_character_flag = $FLAG$
		save_scope_value_as = {
			name  = $FLAG$
			value = yes
		}
	}
}
POD_remove_temporary_portrait_flag = {
	if = {
		limit = { exists = scope:$FLAG$ }
		remove_character_flag = $FLAG$
	}
}

# need to pass SUCCESS_EVENT and FAILURE_EVENT as arguments
POD_finish_complex_scheme = {
	save_scope_as = scheme
	scheme_owner = { save_scope_as = owner }
	save_scope_value_as = {
		name = follow_up_event
		value = event_id:POD_maintenance.1000
	}
	save_scope_value_as = {
		name = pod_scheme_success_event
		value = event_id:$SUCCESS_EVENT$
	}
	save_scope_value_as = {
		name = pod_scheme_failure_event
		value = event_id:$FAILURE_EVENT$
	}
	if = {
		limit = {
			NOT = { exists = scope:suppress_next_event }
		}
		scheme_owner = { trigger_event = scheme_critical_moments.0002 }
	}
}

# saves the scoped character's prowess value as scope:real_prowess (even beyond 100)
POD_save_real_prowess = {
	hidden_effect = {
		save_scope_value_as = {
			name  = real_prowess
			value = prowess
		}
		if = {
			limit = { prowess >= 100 }
			add_character_modifier = save_real_prowess_negate_modifier
			POD_save_real_prowess_iteration = yes
			while = {
				limit = { prowess >= 100 }
				POD_save_real_prowess_iteration = yes
			}
			remove_character_modifier = save_real_prowess_negate_modifier
			remove_all_character_modifier_instances = save_real_prowess_modifier
			force_character_skill_recalculation = yes
		}
	}
}
POD_save_real_prowess_iteration = {
	add_character_modifier = save_real_prowess_modifier
	force_character_skill_recalculation = yes
	save_scope_value_as = {
		name  = real_prowess
		value = {
			value = scope:real_prowess
			add   = prowess
		}
	}
}

###Bandaid because our AIs will never call to arms and we can't figure out why

POD_war_ai_ally_bandaid_effect = {
	scope:war = { save_temporary_scope_as = target }
	$PARTICIPANT$ = {
		save_temporary_scope_as = actor
		every_ally = {
			limit = {
				is_ai = yes
				prev = {
					is_character_interaction_valid = {
						recipient = prev
						interaction = call_ally_interaction
					}
				}

				save_temporary_scope_as = recipient

				#################################################################################
				####### has_valid_target_showing_failures_only from call_ally_interaction #######
				#################################################################################

				# since is_character_interaction_potentially_accepted doesn't work with interactions with a scope:target

				scope:target = {
					is_war_leader = scope:actor
				}
		
				scope:actor = {
					trigger_if = {
						limit = {
							scope:target = {
								OR = {
									using_cb = undirected_great_holy_war
									using_cb = directed_great_holy_war
								}
							}
						}
		
						trigger_if = {
							limit = {
								this = scope:target.casus_belli.primary_attacker
							}
							#if it's a religious war the recipient has to be of the same faith
							custom_description = {
								text = is_of_wrong_faith_for_holy_war
								subject = scope:actor
								object = scope:recipient
								faith = scope:recipient.faith
							}
						}
						trigger_else = {
							custom_description = {
								text = is_of_attacker_faith_in_holy_war
								subject = scope:actor
								object = scope:recipient
								scope:target = { NOT = { primary_attacker.faith = scope:recipient.faith } }
							}
						}
					}
				}
		
		
				joiner_not_already_in_another_war_with_any_target_war_participants_trigger = {
					WARRIOR = scope:actor
					JOINER = scope:recipient
				}
				
				########################################################
				####### can_be_picked from call_ally_interaction #######
				########################################################

				can_join_war_liege_vassal_check_trigger = { WARRIOR = scope:actor JOINER = scope:recipient }

				scope:target = {
					is_war_leader = scope:actor

					# not already in target war
					custom_description = {
						text = join_war_interaction_already_in_target_war
						NOR = {
							any_war_attacker = { this = scope:recipient }
							any_war_defender = { this = scope:recipient }
						}
					}

					# Diarchs can only be called to internal wars.
					custom_tooltip = {
						text = call_ally_interaction.tt.diarchs_called_to_internal_wars
						trigger_if = {
							limit = { scope:recipient ?= scope:actor.diarch }
							primary_defender = scope:actor
							diarch_callable_in_internal_war_trigger = yes
						}
					}
				}

				#############################################################
				####### ai_accept blockers from call_ally_interaction #######
				#############################################################

				trigger_if = { # Refuse call to conflicts of little benefit
					limit = { scope:target = { using_cb = fp2_border_raid } }
					scope:recipient = { # Unless they like going on raids
						OR = {
							has_trait = viking
							has_trait = reaver
						}
					}
				}

				trigger_if = {  # Refuse call against Heir
					limit = { exists = scope:recipient.player_heir }
					trigger_if = {
						limit = { scope:target.casus_belli.primary_defender = scope:actor }
						NOT = { scope:target.casus_belli.primary_attacker = scope:recipient.player_heir }
					}
					trigger_else = {
						NOT = { scope:target.casus_belli.primary_defender = scope:recipient.player_heir }
					}
				}
				
				trigger_if = { # Refuse call against Spouse
					limit = { scope:target.casus_belli.primary_defender = scope:actor }
					NOT = {
						any_spouse = {
							this = scope:target.casus_belli.primary_attacker
						}
					}
				}
				trigger_else = {
					NOT = {
						any_spouse = {
							this = scope:target.casus_belli.primary_defender
						}
					}
				}
			}
			
			call_ally_interaction_effect = yes
		}
	}
}
