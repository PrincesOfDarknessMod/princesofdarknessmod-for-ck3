POD_add_character_modifier_and_track_stacks = {
	add_character_modifier = $MODIFIER$
	hidden_effect = {
		if = {
			limit = { has_variable = $MODIFIER$_stacks }
			change_variable = {
				name = $MODIFIER$_stacks
				add  = 1
			}
		}
		else = {
			set_variable = {
				name  = $MODIFIER$_stacks
				value = 1
			}
		}
	}
}

POD_vassals_imperial_up_effect = {
	
	every_vassal_or_below = {
		limit = {
			POD_imperial_governement_trigger = no
		}
			if = {
				limit = {
					has_vamp_government = yes
				}
			change_government = imperial_unity_government
			}
			
			else_if = {
				limit = {
					has_government = litany_government
				}
			change_government = imperial_tribal_government
			}
			
			else = {
				change_government = imperial_no_unity_government
			}
	}
	
}

POD_imperial_up_effect = {
	if = {
		limit = {
			has_vamp_government = yes
		}
	change_government = imperial_unity_government
	}
	
	else_if = {
		limit = {
			has_government = litany_government
		}
	change_government = imperial_tribal_government
	}
	
	else_if  = {
		limit = {
		OR = {
			is_mummy_trigger = yes
			AND = {
			is_supehunter_trigger = yes
			NOT = { has_trait = shih }
			has_faith = faith:shadowinquisition
			}
			is_demon_trigger = yes
		}				
		}
		change_government = imperial_no_unity_government
	}
	
	else = {}
}

POD_imperial_down_effect = {
	if = {
		limit = {
			is_vampire_trigger = yes
		}
	change_government = traditions_government
	}
	
	else_if = {
		limit = {
			OR = {
			is_mummy_trigger = yes
			is_a_neteru = yes
			}
		}
	change_government = cult_government
	}
	
	else_if  = {
		limit = {
		is_supehunter_trigger = yes
		NOT = { has_trait = shih }
		has_faith = faith:shadowinquisition		
		}
	change_government = inquisition_government
	}

	else_if  = {
		limit = {
		OR = {
			has_trait = demon 
			has_trait = laham
			has_trait = nephilim
			is_a_thrall = yes
		}			
		}
	change_government = infernal_court_government
	}
	
	else_if  = {
		limit = {
		has_trait = shapeshifter			
		}
	change_government = litany_government
	}
}
