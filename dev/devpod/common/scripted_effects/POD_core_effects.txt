POD_increment_scope_value = {
	save_scope_value_as = {
		name = $SCOPE$
		value = {
			value = scope:$SCOPE$
			add = 1
		}
	}
}

POD_toggle_variable = {
	if = {
		limit = { has_variable = $VAR$ }
		remove_variable = $VAR$
	}
	else = {
		set_variable = $VAR$
	}
}

POD_remove_character_modifier = {
	if = {
		limit = { has_character_modifier = $MODIFIER$ }
		remove_character_modifier = $MODIFIER$
	}
}

# save a localization key to be used later
# e.g. with custom_tooltip = POD_saved_tooltip
POD_save_tooltip = {
	save_scope_value_as = {
		name = saved_tooltip
		value = flag:$LOC$
	}
	# error suppression
	if = {
		limit = { flag:$LOC$ != flag:$LOC$ }
	}
}

POD_disappearance_effect = {
	trigger_event = {
		id = POD_disappearance.1
		days = 1
	}
}

POD_release_effect = {
	if = {
		limit = { is_imprisoned = yes }
		add_character_flag = {
			flag = escaped_from_prison_flag
			days = 1
		}
		add_character_flag = { #To make sure you don't get spammed about the release
			flag = block_for_prison_release_notification
			days = 1
		}
		release_from_prison = yes
		if = {
			limit = { is_ruler = no }
			move_to_pool = yes
		}
	}
}

POD_depose_effect = {
	if = {
		limit = { is_ruler = yes }
		depose_effect = { DEPOSER = root }
	}
	if = {
		limit = { is_imprisoned = no }
		move_to_pool = yes
	}
}

# used when becoming an adventurer
POD_transfer_all_titles_to_character = {
	$CHARACTER$ = { save_scope_as = inheritor }
	
	hidden_effect = {
		# Transfer landed titles
		every_held_title = {
			limit = {
				is_landless_type_title = no
				tier > tier_barony
			}
			add_to_list = target_titles
		}
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = change
			add_claim_on_loss = no
		}
		hidden_effect = {
			every_in_list = {
				list = target_titles
				change_title_holder_include_vassals = {
					holder = scope:inheritor
					change = scope:change
				}
			}
			resolve_title_and_vassal_change = scope:change
		}
	}
}

POD_add_modifier_and_track_stacks = {
	add_$TYPE$_modifier = $MODIFIER$
	hidden_effect = {
		if = {
			limit = { has_variable = $MODIFIER$_stacks }
			change_variable = {
				name = $MODIFIER$_stacks
				add  = 1
			}
		}
		else = {
			set_variable = {
				name  = $MODIFIER$_stacks
				value = 1
			}
		}
	}
}

# to make sure existing portrait flags don't get overwritten or removed afterwards
# use both in the same event (immediate + after blocks)
POD_apply_temporary_portrait_flag = {
	if = {
		limit = { NOT = { has_character_flag = $FLAG$ } }
		add_character_flag = $FLAG$
		save_scope_value_as = {
			name  = $FLAG$
			value = yes
		}
	}
}
POD_remove_temporary_portrait_flag = {
	if = {
		limit = { exists = scope:$FLAG$ }
		remove_character_flag = $FLAG$
	}
}

# need to pass SUCCESS_EVENT and FAILURE_EVENT as arguments
POD_finish_complex_scheme = {
	save_scope_as = scheme
	scheme_owner = { save_scope_as = owner }
	save_scope_value_as = {
		name = follow_up_event
		value = event_id:POD_maintenance.1000
	}
	save_scope_value_as = {
		name = pod_scheme_success_event
		value = event_id:$SUCCESS_EVENT$
	}
	save_scope_value_as = {
		name = pod_scheme_failure_event
		value = event_id:$FAILURE_EVENT$
	}
	if = {
		limit = {
			NOT = { exists = scope:suppress_next_event }
		}
		scheme_owner = { trigger_event = scheme_critical_moments.0002 }
	}
}

# saves the scoped character's prowess value as scope:real_prowess (even beyond 100)
POD_save_real_prowess = {
	hidden_effect = {
		save_scope_value_as = {
			name  = real_prowess
			value = prowess
		}
		if = {
			limit = { prowess >= 100 }
			add_character_modifier = save_real_prowess_negate_modifier
			POD_save_real_prowess_iteration = yes
			while = {
				limit = { prowess >= 100 }
				POD_save_real_prowess_iteration = yes
			}
			remove_character_modifier = save_real_prowess_negate_modifier
			remove_all_character_modifier_instances = save_real_prowess_modifier
			force_character_skill_recalculation = yes
		}
	}
}
POD_save_real_prowess_iteration = {
	add_character_modifier = save_real_prowess_modifier
	force_character_skill_recalculation = yes
	save_scope_value_as = {
		name  = real_prowess
		value = {
			value = scope:real_prowess
			add   = prowess
		}
	}
}

POD_energy_vessel_increase_multiple_effect = {
	if = {
		limit = { var:max_energy_stored >= $COUNT$ }
		custom_description = {
			text = POD_store_energy
			value = $COUNT$
			change_variable = { name = energy_stored add = $COUNT$ }
		}
	}
	else = {
		custom_description = {
			text = POD_store_energy
			value = var:max_energy_stored
			change_variable = { name = energy_stored add = var:max_energy_stored }
		}
	}
	hidden_effect = {
		POD_energy_vessel_calc_modifiers_effect = yes
	}
}

POD_energy_vessel_decrease_multiple_effect = {
	if = {
		limit = { var:energy_stored >= $COUNT$ }
		custom_description = {
			text = POD_extract_energy
			value = $COUNT$
			change_variable = { name = energy_stored subtract = $COUNT$ }
		}
	}
	else = {
		save_scope_value_as = {
			name = excess_energy
			value = {
				value = $COUNT$
				subtract = var:energy_stored
			}
		}
		custom_description = {
			text = POD_extract_energy
			value = var:energy_stored
			change_variable = { name = energy_stored subtract = var:energy_stored }
		}
		# $OWNER$ = { Infinite Loop. I don't care if unbalance it.
		# 	POD_universal_energy_decrease_multiple_effect = { COUNT = scope:excess_energy }
		# }
	}
	hidden_effect = {
		POD_energy_vessel_calc_modifiers_effect = yes
	}
}

POD_energy_vessel_calc_modifiers_effect = {
	while = {
		limit = { has_artifact_modifier = energy_vessel_energy_point_modifier }
		remove_artifact_modifier = energy_vessel_energy_point_modifier
	}
	if = { limit = { has_variable = energy_stored } 
		while = {
			count = var:energy_stored
			add_artifact_modifier = energy_vessel_energy_point_modifier
		}
	}
}

POD_energy_vessel_add_max_effect = {
	if = { limit = { NOT = { has_variable = max_energy_stored } } set_variable = { name = max_energy_stored value = 0 } }
	if = { limit = { NOT = { has_variable = energy_stored } } set_variable = { name = energy_stored value = 0 } }
	custom_description = {
		text = POD_energy_vessel_add_max_effect
		value = $COUNT$
		change_variable = { name = max_energy_stored add = $COUNT$ }
		while = {
			count = $COUNT$
			add_artifact_modifier = energy_vessel_modifier
		}
	}
}
