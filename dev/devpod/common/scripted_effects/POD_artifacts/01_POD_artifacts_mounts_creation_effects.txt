create_artifact_warhorse_effect = {
	# Get the character the artifact is being made for.
	$OWNER$ = { save_scope_as = owner }
	# Not really used, but if we don't set the scopes we get errors in the feature selection
	set_artifact_rarity_common = yes
	
	# Create the artifact
	create_artifact = {
		name = artifact_warhorse_name
		description = artifact_warhorse_description
		type = animal
		visuals = animal_trinket
		wealth = scope:wealth
		quality = scope:quality
		modifier = artifact_character_travel_speed_1_modifier
		template = animal_template
		decaying = no # Handled by remove_horse_story_modifiers_effect
		save_scope_as = newly_created_artifact
	}

	scope:newly_created_artifact = {
		 set_variable = { name = pet_artifact value = flag:horse }
	}
}

create_artifact_archator_effect = {
	$OWNER$ = { save_scope_as = owner }
	set_artifact_rarity_common = yes
	create_artifact = {
		name = artifact_archator_name
		description = artifact_archator_description
		type = animal
		template = animal_template
		visuals = animal_trinket
		modifier = archator_modifier
		max_durability = 10
		wealth = scope:wealth
		quality = scope:quality
		history = {
			type = created
			actor = scope:owner
			recipient = scope:owner
			location = scope:owner.location
		}
		save_scope_as = newly_created_artifact
	}
	hidden_effect_new_object = {
		scope:newly_created_artifact = {
			set_variable = historical_unique_artifact
			set_variable = archator
			set_variable = mokole_dinosaur
			set_variable = summoned_artifact
		}
	}
}
