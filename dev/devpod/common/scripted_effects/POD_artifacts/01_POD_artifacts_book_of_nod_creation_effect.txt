# POD Book of Nod Artifacts Creation #

create_artifact_book_of_nod_effect = {
	# Get the character the artifact is being made for.
	$OWNER$ = { save_scope_as = owner }
	# Not really used, but if we don't set the scopes we get errors in the feature selection
	set_artifact_rarity_illustrious = yes
	
	# Create the artifact
	create_artifact = {
		name = artifact_book_of_nod_name
		description = artifact_book_of_nod_description
		type = $TYPE$	
		template = book_of_nod_template
		visuals = book_POD_book_of_nod 
		wealth = scope:wealth
		quality = scope:quality			
		history = {
			type = created_before_history
		}
		modifier =  artifact_POD_book_of_nod_modifier  #artifact_learning_1_modifier
		save_scope_as = newly_created_artifact
		decaying = no		
	}

	scope:newly_created_artifact = {
		set_variable = historical_unique_artifact
		set_variable = book_of_nod
		save_scope_as = epic
		save_scope_value_as = {
			name = book_of_nod
			value = yes
		}
	}	
}

	random_list = {
		10 = { 
			send_interface_toast = {
				title = POD_artifacts_book_of_nod.reading.gain
				left_icon = root
				add_piety = miniscule_piety_value 
			}
		}
		10 = {
			send_interface_toast = {
				title = POD_artifacts_book_of_nod.reading.loss
				left_icon = root
				add_stress = miniscule_stress_impact_gain
			}
		}
		80 = { }
	}
}

POD_reading_tale_of_the_first_city_effect = {
	random_list = {
		10 = {
			if = {
				limit = {
					NOT = { knows_language_of_culture = culture:enochian }
					POD_has_complete_book_of_nod_trigger = yes
				}
				send_interface_toast = {
					title = POD_artifacts_book_of_nod.reading.learn.enochian
					left_icon = root
					learn_language_of_culture = culture:enochian
				}
			}
		}
		10 = { 
			send_interface_toast = {
				title = POD_artifacts_book_of_nod.reading.gain
				left_icon = root
				add_learning_lifestyle_xp = miniscule_lifestyle_experience 
			}
		}
		10 = {
			send_interface_toast = {
				title = POD_artifacts_book_of_nod.reading.loss
				left_icon = root
				add_stress = miniscule_stress_impact_gain
			}
		}
		70 = { }
	}
}
