#Everyone - Common
create_artifact_lepers_rags_effect = {
	$OWNER$ = { save_scope_as = owner }
	set_artifact_rarity_common = yes

	create_artifact = {
		name = artifact_lepers_rags_name
		description = artifact_lepers_rags_description
		type = miscellaneous
		template = fetish_template
		visuals = small_box
		modifier = lepers_rags_modifier
		wealth = scope:wealth
		quality = scope:quality
		history = {
			type = created_before_history
		}
		save_scope_as = newly_created_artifact
	}
	hidden_effect_new_object = {
		scope:newly_created_artifact = {
			set_variable = possessed_artifact
			set_variable = fetish_artifact
			set_variable = lepers_rags
			set_variable = do_not_destroy_artifact 
		}
	}
}
