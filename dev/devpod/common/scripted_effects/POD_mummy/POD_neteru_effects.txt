neteru_faith_initiation_effect = {
	if = {
		limit = { 
			exists = $LIEGE$
			$LIEGE$ ?= { 
				OR = { 
					has_trait = banemummy
					has_trait = wukuei
				}
			} 
		}
		POD_become_wyrm_servant_effect = yes
	}
	else_if = {
		limit = { 
			exists = $LIEGE$ 
			$LIEGE$ = { 
				OR = {
					has_trait = cabiri 
					has_trait = wutian 
				} 
			}
		}
		set_character_faith = $LIEGE$.faith
	}
	else = {
		if = {
			limit = { has_POD_vamp_blood_trigger = yes }
			set_character_faith = faith:childofosiris
		}
		else_if = {
			limit = { 
				OR = { 
					has_trait = werewolf
					has_trait = kinfolk
				}
			}
			set_character_faith = faith:silentstrider
		}
		else_if = {
			limit = { 
				has_trait = supehunter
				NOT  = { has_trait = shih }
			}
			set_character_faith = faith:ikhwan_al_safa
		}
		else = {
			set_character_faith = $LIEGE$.faith
		}
	}
}

neteru_legacies_initiation_effect = {
	if = {
		limit = { any_relation = { type = worshipped_reborn dynasty = { has_dynasty_perk = mortuary_cult_mummy_legacy_5 } } }
		add_character_modifier = POD_neteru_legacy_5_modifier
	}
	else_if = {
		limit = { any_relation = { type = worshipped_reborn dynasty = { has_dynasty_perk = mortuary_cult_mummy_legacy_4 } } }
		add_character_modifier = POD_neteru_legacy_4_modifier
	}
	else_if = {
		limit = { any_relation = { type = worshipped_reborn dynasty = { has_dynasty_perk = mortuary_cult_mummy_legacy_3 } } }
		add_character_modifier = POD_neteru_legacy_3_modifier
	}
	else_if = {
		limit = { any_relation = { type = worshipped_reborn dynasty = { has_dynasty_perk = mortuary_cult_mummy_legacy_2 } } }
		add_character_modifier = POD_neteru_legacy_2_modifier
	}
	else_if = {
		limit = { any_relation = { type = worshipped_reborn dynasty = { has_dynasty_perk = mortuary_cult_mummy_legacy_1 } } }
		add_character_modifier = POD_neteru_legacy_1_modifier
	}
}

neteru_legacies_removal_effect = {
	if = {
		limit = { has_character_modifier = POD_neteru_legacy_5_modifier }
		remove_character_modifier = POD_neteru_legacy_5_modifier
	}
	if = {
		limit = { has_character_modifier = POD_neteru_legacy_4_modifier }
		remove_character_modifier = POD_neteru_legacy_4_modifier
	}
	if = {
		limit = { has_character_modifier = POD_neteru_legacy_3_modifier }
		remove_character_modifier = POD_neteru_legacy_3_modifier
	}
	if = {
		limit = { has_character_modifier = POD_neteru_legacy_2_modifier }
		remove_character_modifier = POD_neteru_legacy_2_modifier
	}
	if = {
		limit = { has_character_modifier = POD_neteru_legacy_1_modifier }
		remove_character_modifier = POD_neteru_legacy_1_modifier
	}
}
