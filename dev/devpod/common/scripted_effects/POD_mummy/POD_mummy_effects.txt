POD_become_mummy_effect = {
	save_temporary_scope_value_as = {
		name = saved_immortal_age
		value = POD_visual_age
	}
	hidden_effect = { POD_remove_splat_traits_effect = yes }
	add_trait_force_tooltip = mummy
	if = {
		limit = { exists = scope:saved_immortal_age }
		hidden_effect = { set_immortal_age = scope:saved_immortal_age }
	}
	hidden_effect = {
		POD_remove_wrong_supernatural_traits_on_existing_character = yes
		POD_remove_supernatural_relation_effect = yes
		POD_remove_physical_coping_strategies = yes

		POD_side_effects_from_changing_splats = yes
	}

	POD_mummy_initialization_effect = yes

	if = {
		limit = { is_ai = no }
		trigger_event = podintroductions.2202
	}
}

POD_mummy_initialization_effect = {
	if = {
		limit = {
			OR = {
				exists = scope:character_being_changed
				is_from_ruler_designer = yes
			}
		}
		POD_change_government_if_landed = { GOVERNMENT = cult_government }
		add_trait_xp = { trait = mummy value = 50 }
	}
	hidden_effect = {
		if = {
			limit = {
				NOR = {
					religion = { is_in_family = rf_mummies }
					POD_has_valid_splat_agnostic_faith = yes
				}
			}
			POD_character_mummy_faith_initiation_effect = yes
		}
	}
	if = { 
		limit = { 
			NOT = { has_trait = integrity }
			POD_should_have_integrity_trigger = yes
		} 
		POD_mummy_initialization_integrity_effect = yes 
	}
}

POD_mummy_spell_of_life_creation_effect = {
	POD_error_suppression_unset_flag = { FLAG = $DYNASTY$ }

	if = {
		limit = { flag:$DYNASTY$ = flag:asekh_sen }
		add_opinion = {
			target = scope:actor
			modifier = gift_opinion
			opinion = 35
		}
		random_list = {
			# You are more likely to make Asekh-sen as you master the SoL & bane mummies always make them
			50 = {
				modifier = {
					add = {
						value = "scope:actor.has_trait_xp(eternal_rebirth_seeker|spell_of_life)"
						subtract = 5
					}
				}
				modifier = {
					scope:actor = { has_trait = banemummy }
					add = 500
				}
				POD_mummy_initialization_asekh_sen_effect = yes
				if = {
					limit = { scope:actor = { has_religion = religion:amenti_religion } }
					set_character_faith = scope:actor.faith
				}
				else = {
					set_character_faith = faith:faces_of_apophis
				}
			}
			50 = {
				modifier = {
					add = {
						value = "scope:actor.has_trait_xp(eternal_rebirth_seeker|spell_of_life)"
						subtract = 5
						multiply = -1
					}
				}
				modifier = {
					scope:actor = { has_trait = banemummy }
					add = -500
				}
				# Borrowed from the alchemy potion that turns you into a vampire, with a few tweaks
				scope:actor = { save_scope_as = vampiresire }
				scope:recipient = { save_scope_as = created_vampirechilde }
				POD_become_vampire_effect = yes
                if = {
					limit = { is_alive = yes }

					add_trait_force_tooltip = caitiff
					give_witch_secret_or_trait_effect = yes
					
					hidden_effect = {
						if = {
							limit = { scope:actor = { NOT = { religion = { is_in_family = rf_kindred } } } }
							scope:actor = {
								switch = {
									trigger = has_faith
									faith:code_of_horus    = { scope:recipient = { set_character_faith = faith:childofosiris } }
									faith:ishmaelite       = { scope:recipient = { set_character_faith = faith:viahumanitatis } }
									# chosen for similar doctrines & virtues, cabiri focus on scholarly pursuits, and that mummies would know about those 3 primordial spirits
									faith:cabiri           = { scope:recipient = { set_character_faith = faith:viaprimordia } }
									faith:faces_of_apophis = { scope:recipient = { set_character_faith = faith:viaapep } }
									fallback 	     = { 
										switch = {
											trigger = has_doctrine
											special_doctrine_osirian_league               = { scope:recipient = { set_character_faith = faith:childofosiris } }
											special_doctrine_set_pillars_kindred         = { scope:recipient = { set_character_faith = faith:viaapep } }
											special_doctrine_set_corrupted_kindred       = { scope:recipient = { set_character_faith = faith:viaapep } }
											special_doctrine_allied_to_apophis_hostility = { scope:recipient = { set_character_faith = faith:viaapep } }
											fallback                                     = { scope:recipient = { set_character_faith = faith:viahumanitatis } }
										}
									}
								}
							}
						}
						else = {
							set_character_faith = scope:actor.faith
						}
						POD_character_vampire_bloodpotency_initialization_effect = yes
						POD_character_masquerade_initialization_effect = yes
						POD_character_vampire_discipline_initialization_effect = yes

						random_list = {
							30 = { add_trait = vampgeneration13 }
							30 = { add_trait = vampgeneration12 }
							20 = { add_trait = vampgeneration11 }
							10 = { add_trait = vampgeneration10 }
							5 = { add_trait = vampgeneration9 }
							5 = { add_trait = vampgeneration8 } 
						}
						POD_achiev_nomnom_initialize = yes

						if = {
							limit = { is_landed = yes }
							change_government = traditions_government
						}
						POD_remove_disease_traits_effect = yes
						add_character_flag = immune_to_disease
					}
                }
			}
		}
	}
	else = {
		remove_relation_worshipped_reborn = scope:actor
		add_opinion = {
			modifier = POD_spell_of_life_opinion
			target = scope:actor
		}

		POD_become_mummy_effect = yes
		POD_mummy_initialization_$DYNASTY$_effect = yes
		set_house = scope:actor.house

		if = {
			limit = { scope:actor = { has_religion = religion:amenti_religion } }
			set_character_faith = scope:actor.faith
		}
	}

	# If you don't create a true mummy and are shemsu-heru/ishmaelite, you lose a piety level for making an abomination
	if = {
		limit = { 
			NOT = { flag:shemsu_heru = flag:$DYNASTY$ } 
			scope:actor = { has_trait = shemsuheru }
		}
		scope:actor = { add_piety_level = -1 }
	}
}

		value = POD_visual_age
	}

		limit = { has_trait = mortal }
		remove_trait = mortal
	}
	if = {
		limit = {
			has_trait = supehunter
			NOT = { has_character_flag = chargen_changing_splat_flag }
			scope:domitor ?= { POD_is_supehunter_trigger = no }
		}
		scope:domitor = { save_scope_as = vampiresire }
		remove_trait = supehunter
		POD_hunterembrace_effect = yes
	}

		limit = { POD_splat_mixing_haight_mode = no }
		if = {
			limit = { has_trait = kinfolk }
			remove_trait = kinfolk
			if = {
				limit = { has_trait = gnosis }
				remove_trait = gnosis
			}
		}
		if = {
			limit = { has_trait = kinain }
			remove_trait = kinain
		}
	}
	if = {
		limit = { has_trait = changeling }
		add_trait = lunatic_genetic # Dauntain
	}
	add_trait_force_tooltip = asekh_sen
	if = {
		limit = { exists = scope:saved_immortal_age }
		hidden_effect = { set_immortal_age = scope:saved_immortal_age }
	}
	hidden_effect = {
		POD_remove_wrong_supernatural_traits_on_existing_character = yes
		# POD_remove_supernatural_relation_effect = yes
		POD_side_effects_from_changing_splats = yes
	}
	add_character_modifier = extra_life_4_modifier
	give_witch_secret_or_trait_effect = yes
	hidden_effect = {
		POD_asekh_sen_bane_infusion_lotto_effect = yes
		POD_asekh_sen_bane_infusion_lotto_effect = yes
		POD_asekh_sen_bane_infusion_lotto_effect = yes
		POD_asekh_sen_bane_infusion_lotto_effect = yes
		POD_asekh_sen_bane_infusion_lotto_effect = yes
		POD_asekh_sen_bane_infusion_lotto_effect = yes

	}

# Used to initialize the trait and ignore the various safety measures elsewhere
POD_initialize_spell_of_life_to_x_level = {
	hidden_effect = {
		# Add the trait and XP, just in case they don't already have it
		add_trait = eternal_rebirth_seeker
		add_trait_xp = {
			trait = eternal_rebirth_seeker
			track = spell_of_life
			value = $VALUE$ 
		}

		# They don't get any Phase 3 flags since the player gave it to them
		# Any Stage 1-specific flags must be manually set for individual characters as deemed relevant

		# Set up the stage flags to move the process along to the right spot
		# Especially important if character already has some XP
		if = {
			limit = {
				has_trait_xp = {
					trait = eternal_rebirth_seeker
					track = spell_of_life
					value >= 90
				}
			}
			add_character_flag = spell_of_life_stage3_done
		}

		# Set up the proper subsplat based on current subsplat
		if = {
			limit = {
				has_trait_xp = {
					trait = eternal_rebirth_seeker
					track = spell_of_life
					value = 100
				}
			}
			# Fallback only triggered by people like Kemintiri or by debug interactions
			switch = {
				trigger = has_trait
				shemsuheru           = { 
					if = {
						limit = { NOT = { has_faith = faith:ishmaelite } }
						add_character_flag = spell_of_life_shemsu_heru 
					}
					else             = { add_character_flag = spell_of_life_ishmaelite }
				}
				cabiri               = { add_character_flag = spell_of_life_cabiri }
				custom_mummy_dynasty = { add_character_flag = spell_of_life_custom_mummy_dynasty }
				fallback 			 = { 
					switch = {
						trigger = has_doctrine
						special_doctrine_osirian_league               = { add_character_flag = spell_of_life_shemsu_heru  }
						special_doctrine_set_pillars_kindred         = { add_character_flag = spell_of_life_custom_mummy_dynasty }
						special_doctrine_set_corrupted_kindred       = { add_character_flag = spell_of_life_custom_mummy_dynasty }
						special_doctrine_allied_to_apophis_hostility = { add_character_flag = spell_of_life_custom_mummy_dynasty }
						fallback                                     = { add_character_flag = spell_of_life_ishmaelite }
					}
				}
			}
		}
	}
}

POD_add_spell_of_life_xp = {
	# Add XP for each stage with proper caps
	add_trait = eternal_rebirth_seeker
	if = {
		limit = {
			has_trait_xp = {
				trait = eternal_rebirth_seeker
				track = spell_of_life
				value < 30
			}
		}

		POD_add_spell_of_life_xp_with_cap = {
			VALUE = $VALUE$ 
			CAP = 30
		}
	}
	else_if = {
		limit = {
			has_trait_xp = {
				trait = eternal_rebirth_seeker
				track = spell_of_life
				value < 90
			}
			has_trait_xp = {
				trait = eternal_rebirth_seeker
				track = spell_of_life
				value >= 30
			}
		}

		POD_add_spell_of_life_xp_with_cap = {
			VALUE = $VALUE$ 
			CAP = 90
		}

		# Horus grants a +5 bonus
		if = {
			limit = { has_character_flag = spell_of_life_stage1_horus_phase3_decision }
			custom_tooltip = spell_of_life_stage1_horus_phase3_bonus_xp
			hidden_effect = {
				POD_add_spell_of_life_xp_with_cap = {
					VALUE = 5
					CAP = 90
				}
			}
		}
	}
	else_if = {
		limit = {
			has_trait_xp = {
				trait = eternal_rebirth_seeker
				track = spell_of_life
				value < 100
			}
			has_trait_xp = {
				trait = eternal_rebirth_seeker
				track = spell_of_life
				value >= 90
			}
		}

		POD_add_spell_of_life_xp_with_cap = {
			VALUE = $VALUE$ 
			CAP = 100
		}
	}

	# Trigger Stage 3:
	# If Phase 3 decision not taken, check if given character alive. If alive, they do it.
	# If dead, get random ruler if possible. Otherwise, skip the event entirely.
	if = {
		limit = {
			has_trait_xp = {
				trait = eternal_rebirth_seeker
				track = spell_of_life
				value >= 90
			}
			NOT = { has_character_flag = spell_of_life_stage3_done }
		}
		add_character_flag = spell_of_life_stage3_in_progress

		if = {
			limit = {
				NOT = { has_character_flag = spell_of_life_stage1_horus_phase3_decision }
			}

			if = {
				limit = {
					NOR = { 
						POD_character_is_defeated = { CHARACTER = character:792593 } 
						has_relation_coterie_member = character:792593
						this = character:792593
					}
				}

				trigger_event = {
					id = POD_mummy_spell_of_life.30
					days = 5
				}
			}
			else_if = {
				limit = {
					any_ruler = {
						count >= 1
						faith = faith:code_of_horus
					}
				}

				trigger_event = {
					id = POD_mummy_spell_of_life.31
					days = 5
				}
			}
		}
		if = {
			limit = {
				NOT = { has_character_flag = spell_of_life_stage1_cabirus_phase3_decision }
			}

			if = {
				limit = {
					NOR = { 
						POD_character_is_defeated = { CHARACTER = character:790363 } 
						has_relation_coterie_member = character:790363
						this = character:790363
					}
				}

				trigger_event = {
					id = POD_mummy_spell_of_life.32
					days = 10
				}
			}
			else_if = {
				limit = {
					any_ruler = {
						count >= 1
						religion = religion:codeoftremere_religion
					}
				}

				trigger_event = {
					id = POD_mummy_spell_of_life.33
					days = 10
				}
			}
		}
		if = {
			limit = {
				NOT = { has_character_flag = spell_of_life_stage1_kemintiri_phase3_decision }
			}

			if = {
				limit = {
					NOR = { 
						POD_character_is_defeated = { CHARACTER = character:790512 } 
						has_relation_coterie_member = character:790512
						this = character:790512
					}
				}

				trigger_event = {
					id = POD_mummy_spell_of_life.34
					days = 15
				}
			}
			else_if = {
				limit = {
					any_ruler = {
						count >= 1
						faith = faith:childofosiris
					}
				}

				trigger_event = {
					id = POD_mummy_spell_of_life.35
					days = 15
				}
			}
		}
		if = {
			limit = {
				NOT = { has_character_flag = spell_of_life_stage1_setite_phase3_decision }
			}

			if = {
				limit = {
					NOR = { 
						POD_character_is_defeated = { CHARACTER = character:790988 } 
						has_relation_coterie_member = character:790988
						this = character:790988
					}
				}

				trigger_event = {
					id = POD_mummy_spell_of_life.36
					days = 20
				}
			}
			else_if = {
				limit = {
					any_ruler = {
						count >= 1
						OR = {
							has_trait = setite
							has_trait = banemummy
						}
					}
				}

				trigger_event = {
					id = POD_mummy_spell_of_life.37
					days = 20
				}
			}
		}

		trigger_event = {
			id = POD_mummy_spell_of_life.38
			days = 21
		}
	}

	# Set up stuff for Stage 4
	if = {
		limit = {
			OR = {
				has_trait_xp = {
					trait = eternal_rebirth_seeker
					track = spell_of_life
					value = 100
				}
				# Necessary to make tooltips appear properly
				AND = {
					has_trait_xp = {
						trait = eternal_rebirth_seeker
						track = spell_of_life
						value = 90
					}
					$VALUE$ >= 10
				}
			}
		}

		if = {
			limit = {
				has_character_flag = spell_of_life_stage1_horus_phase3_decision
			}
			add_character_flag = spell_of_life_shemsu_heru
			custom_tooltip = POD_mummy_spell_of_life_shemsuheru
		}
		else_if = {
			limit = {
				has_character_flag = spell_of_life_stage1_kemintiri_phase3_decision
			}
			add_character_flag = spell_of_life_ishmaelite
			custom_tooltip = POD_mummy_spell_of_life_ishmaelite
		}
		else_if = {
			limit = {
				has_character_flag = spell_of_life_stage1_cabirus_phase3_decision
			}
			add_character_flag = spell_of_life_cabiri
