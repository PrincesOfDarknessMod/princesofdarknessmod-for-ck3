POD_become_mummy_effect = {
	save_temporary_scope_value_as = {
		name = saved_immortal_age
		value = POD_visual_age
	}
	hidden_effect = { POD_remove_splat_traits_effect = yes }
	add_trait_force_tooltip = mummy
	if = {
		limit = { exists = scope:saved_immortal_age }
		hidden_effect = { set_immortal_age = scope:saved_immortal_age }
	}
	hidden_effect = {
		POD_remove_wrong_supernatural_traits_effect = yes
		POD_remove_wrong_supernatural_modifiers_effect = yes
		POD_remove_supernatural_relation_effect = yes
		POD_remove_physical_coping_strategies = yes

		POD_side_effects_from_changing_splats = yes
	}

	POD_mummy_initialization_effect = yes

	if = {
		limit = { is_ai = no }
		trigger_event = podintroductions.2202
	}
}

POD_mummy_initialization_effect = {
	if = {
		limit = {
			OR = {
				exists = scope:character_being_changed
				is_from_ruler_designer = yes
			}
		}
		POD_change_government_if_landed = { GOVERNMENT = cult_government }
	}
	hidden_effect = {
		if = { limit = { NOT = { religion = { is_in_family = rf_mummies } } } POD_character_mummy_faith_initiation_effect = yes }
	}
	POD_mummy_memory_change_effect = { COUNT = 50 }
	if = { limit = { POD_should_have_integrity_trigger = yes } POD_mummy_initialization_integrity_effect = yes }
}

POD_mummy_initialization_integrity_effect = {
	add_trait = integrity
	POD_mummy_integrity_change_effect = { COUNT = 50 }
}

POD_character_mummy_faith_initiation_effect = {
	if = {
		limit = { has_trait = wutian NOT = { faith = faith:ten_mandates_of_heaven } }
		POD_mummy_initialization_wutian_effect = yes
	}
	else_if = {
		limit = { has_trait = wukuei NOT = { faith = faith:ten_mandates_of_hell } }
		POD_mummy_initialization_wukuei_effect = yes
	}
	else_if = {
		limit = { has_trait = cabiri NOT = { faith = faith:cabiri } }
		POD_mummy_initialization_cabiri_effect = yes
	}
	else_if = {
		limit = { has_trait = banemummy NOT = { faith = faith:faces_of_apophis } }
		POD_mummy_initialization_banemummy_effect = yes
	}
	else_if = {
		limit = { has_trait = shemsuheru NOT = { faith = faith:code_of_horus } }
		POD_mummy_initialization_shemsu_heru_effect = yes
	}
	else = {
		POD_mummy_initialization_ishmaelite_effect = yes
	}
}

POD_hekau_cost_effect = {
	# COUNT can be a scriptvalue, and we're passing it in a different
	# context than we received it, so save its value here.
	save_temporary_scope_value_as = {
		name = hekau_cost_count
		value = $COUNT$
	}
	POD_universal_energy_decrease_multiple_effect = { COUNT = scope:hekau_cost_count }
}

POD_become_asekh_sen_effect = {
	save_temporary_scope_value_as = {
		name = saved_immortal_age
		value = POD_visual_age
	}
	if = {
		limit = { has_trait = mortal }
		remove_trait = mortal
	}
	if = {
		limit = {
			has_trait = supehunter
			NOT = { has_character_flag = chargen_changing_splat_flag }
			scope:domitor ?= { POD_is_supehunter_trigger = no }
		}
		scope:domitor = { save_scope_as = vampiresire }
		remove_trait = supehunter
		POD_hunterembrace_effect = yes
	}
	if = {
		limit = { POD_splat_mixing_haight_mode = no }
		if = {
			limit = { has_trait = kinfolk }
			remove_trait = kinfolk
			if = {
				limit = { has_trait = gnosis }
				remove_trait = gnosis
			}
		}
		if = {
			limit = { has_trait = kinain }
			remove_trait = kinain
		}
	}
	if = {
		limit = { has_trait = changeling }
		add_trait = lunatic_genetic # Dauntain
	}
	add_trait_force_tooltip = asekh_sen
	if = {
		limit = { exists = scope:saved_immortal_age }
		hidden_effect = { set_immortal_age = scope:saved_immortal_age }
	}
	hidden_effect = {
		POD_remove_wrong_supernatural_traits_effect = yes
		POD_remove_wrong_supernatural_modifiers_effect = yes
		# POD_remove_supernatural_relation_effect = yes
		POD_side_effects_from_changing_splats = yes
	}
	add_character_modifier = extra_life_4_modifier
	give_infernalist_secret_or_trait_effect = yes
}

POD_get_hekau_level_effect = {
	if = { limit = { NOT = { has_perk = $PERK$ } } 
		if = { limit = { NOT = { POD_can_use_or_get_hekau_level_trigger = { LEVEL = $LEVEL$ } } }
			custom_description_no_bullet = {
				text = POD_get_hekau_level_effect
				value = $LEVEL$
			}
		}
		POD_mummy_memory_change_effect = { COUNT = -$LEVEL$ }
		if = { limit = { has_trait = integrity } POD_mummy_integrity_change_effect = { COUNT = 2 } }
	}
}

POD_mummy_memory_change_effect = {
	if = { limit = { $COUNT$ > 0 } 
		custom_description = {
			text = POD_mummy_memory_increase_effect
			value = $COUNT$
			add_trait_xp = { trait = mummy value = $COUNT$ }
		}
	}
	else = {
		custom_description = {
			text = POD_mummy_memory_decrease_effect
			value = $COUNT$
			add_trait_xp = { trait = mummy value = $COUNT$ }
		}
	}
}
