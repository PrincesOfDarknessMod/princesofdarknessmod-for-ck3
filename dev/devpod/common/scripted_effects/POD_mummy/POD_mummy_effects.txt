POD_mummy_initialization_effect = {
	if = {
		limit = {
			OR = {
				exists = scope:character_being_changed
				is_from_ruler_designer = yes
			}

	hidden_effect = {
		if = {
			limit = {
				NOR = {

		limit = { 
			NOT = { has_trait = integrity }
			POD_should_have_integrity_trigger = yes
		} 
		POD_mummy_initialization_integrity_effect = yes 
	}
}

POD_get_hekau_level_effect = {
	if = {
		limit = { NOT = { has_perk = $PERK$ } }
		if = {
			limit = { NOT = { POD_can_use_or_get_hekau_level_trigger = { LEVEL = $LEVEL$ } } }
			custom_description_no_bullet = {
				text = POD_get_hekau_level_effect
				value = $LEVEL$
			}
		}
		if = { limit = { has_trait = mummy } add_trait_xp = { trait = mummy value = -$LEVEL$ } }
		POD_gain_integrity = { VALUE = 2 }
	}
	else = {
		hidden_effect = {
			if = {
				limit = { NOT = { POD_can_use_or_get_hekau_level_trigger = { LEVEL = $LEVEL$ } } }
				custom_description_no_bullet = {
					text = POD_get_hekau_level_effect
					value = $LEVEL$
				}
			}
			if = { limit = { has_trait = mummy } add_trait_xp = { trait = mummy value = -$LEVEL$ } }
			POD_gain_integrity = { VALUE = 2 }
		}
	}
}
