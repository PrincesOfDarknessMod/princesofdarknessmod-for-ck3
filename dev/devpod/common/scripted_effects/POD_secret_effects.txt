
POD_give_massregnant_secret_effect = {
	save_temporary_scope_as = massregnant_character
	if = {
		limit = { #Not already a massregnant
			NOR = {
				has_trait = massregnant
				any_secret = { secret_type = secret_massregnant }
			}	
		}
		if = {
			limit = {
				trait_is_shunned_or_criminal_in_my_or_lieges_rite_trigger = { TRAIT = trait:massregnant GENDER_CHARACTER = scope:massregnant_character }
			}
			add_secret = {
				type = secret_massregnant
			}
		}
		else = {
			add_trait = massregnant
		}
	}	
}

give_sorcerer_secret_or_trait_effect = {
	save_temporary_scope_as = sorcerer
	if = {
		limit = {
			is_sorcerer_trigger = no
		}
		if = {
			limit = {
				trait_is_shunned_or_criminal_in_my_or_lieges_rite_trigger = { TRAIT = trait:sorcerer GENDER_CHARACTER = scope:sorcerer }
			}
			add_secret = {
				type = secret_sorcerer
			}
		}
		else = {
			add_trait = sorcerer
		}
	}
}
