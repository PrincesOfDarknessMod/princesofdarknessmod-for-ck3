POD_add_necromancy_modifier_effect = {
	if = { 
		limit = { scope:secondary_recipient.POD_necromancy_$SKILL$ >= POD_necromancy_high_threshold } 
		add_character_modifier = POD_send_bound_wraith_interaction_$TYPE$4_modifier 
		$WRAITH$ = { set_variable = POD_send_bound_wraith_interaction_$TYPE$4_modifier }
	}
	else_if = { 
		limit = { scope:secondary_recipient.POD_necromancy_$SKILL$ >= POD_necromancy_medium_threshold } 
		add_character_modifier = POD_send_bound_wraith_interaction_$TYPE$3_modifier 
		$WRAITH$ = { set_variable = POD_send_bound_wraith_interaction_$TYPE$3_modifier }
	}
	else_if = { 
		limit = { scope:secondary_recipient.POD_necromancy_$SKILL$ >= POD_necromancy_low_threshold } 
		add_character_modifier = POD_send_bound_wraith_interaction_$TYPE$2_modifier 
		$WRAITH$ = { set_variable = POD_send_bound_wraith_interaction_$TYPE$2_modifier }
	}
	else = { 
		add_character_modifier = POD_send_bound_wraith_interaction_$TYPE$1_modifier 
		$WRAITH$ = { set_variable = POD_send_bound_wraith_interaction_$TYPE$1_modifier }
	}
}
