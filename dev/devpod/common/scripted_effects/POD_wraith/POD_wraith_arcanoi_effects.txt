# need to pass WRAITH, TARGET, and ARCANOS
POD_start_wraith_support = {
	$TARGET$ = { save_scope_as = wraith_target }
	$WRAITH$ = {
		if = {
			limit = {
				NAND = {
					var:POD_send_wraith_target  ?= scope:wraith_target
					var:POD_send_wraith_arcanos ?= $ARCANOS$
				}
			}
			POD_clear_wraith_support_modifier = yes
		}
	}
	POD_apply_wraith_support_modifier = {
		WRAITH  = $WRAITH$
		TARGET  = $TARGET$
		ARCANOS = $ARCANOS$
	}
}

# scoped to the wraith
POD_clear_wraith_support_modifier = {
	if = {
		limit = {
			has_variable = POD_send_wraith_target
			has_variable = POD_send_wraith_arcanos
		}
		
		var:POD_send_wraith_target  = { save_scope_as = cleared_target }
		var:POD_send_wraith_arcanos = { save_scope_as = arcanos }
		
		scope:cleared_target ?= {
			if = {
				limit = { is_alive = yes }
				
				# moving these to a (non-variable) list makes for better tooltips
				every_in_list = {
					variable = supporting_wraiths
					limit = { var:POD_send_wraith_arcanos ?= scope:arcanos }
					add_to_list = supporting_wraiths
				}
				prev = { remove_from_list = supporting_wraiths }
				remove_list_variable = {
					name   = supporting_wraiths
					target = prev
				}
				
				if = {
					limit = { any_in_list = { list = supporting_wraiths } }
					POD_necromancy_modifier_injector = {
						ARCANOS = scope:arcanos
						EFFECT  = POD_reapply_scaling_character_modifier
					}
				}
				else = {
					POD_necromancy_modifier_injector = {
						ARCANOS = scope:arcanos
						EFFECT  = POD_remove_character_modifier
					}
				}
			}
		}
		
		remove_variable = POD_send_wraith_target
		remove_variable = POD_send_wraith_arcanos
	}
}

# need to pass WRAITH, TARGET, ARCANOS
POD_apply_wraith_support_modifier = {
	$WRAITH$  = { save_scope_as = wraith }
	$TARGET$  = { save_scope_as = wraith_target }
	$ARCANOS$ = { save_scope_as = arcanos }
	scope:wraith = {
		set_variable = {
			name  = POD_send_wraith_target
			value = scope:wraith_target
		}
		set_variable = {
			name  = POD_send_wraith_arcanos
			value = $ARCANOS$
		}
	}
	# safeguard
	POD_clear_list = { LIST = supporting_wraiths }
	scope:wraith_target = {
		every_in_list = {
			variable = supporting_wraiths
			limit = { var:POD_send_wraith_arcanos ?= scope:arcanos }
			add_to_list = supporting_wraiths
		}
		scope:wraith = { add_to_list = supporting_wraiths }
		add_to_variable_list = {
			name   = supporting_wraiths
			target = scope:wraith
		}
		POD_necromancy_modifier_injector = {
			ARCANOS = $ARCANOS$
			EFFECT  = POD_reapply_scaling_character_modifier
		}
	}
}

POD_necromancy_modifier_injector = {
	switch = {
		trigger = $ARCANOS$
		trait:arcanos_argos			= { $EFFECT$ = { MODIFIER = POD_send_bound_wraith_interaction_argos_modifier } }
		trait:arcanos_castigate		= { $EFFECT$ = { MODIFIER = POD_send_bound_wraith_interaction_castigate_modifier } }
		trait:arcanos_embody		= { $EFFECT$ = { MODIFIER = POD_send_bound_wraith_interaction_embody_modifier } }
		trait:arcanos_fatalism		= { $EFFECT$ = { MODIFIER = POD_send_bound_wraith_interaction_fatalism_modifier } }
		trait:arcanos_flux			= { $EFFECT$ = { MODIFIER = POD_send_bound_wraith_interaction_flux_modifier } }
		trait:arcanos_inhabit		= { $EFFECT$ = { MODIFIER = POD_send_bound_wraith_interaction_inhabit_modifier } }
		trait:arcanos_intimation	= { $EFFECT$ = { MODIFIER = POD_send_bound_wraith_interaction_intimation_modifier } }
		trait:arcanos_lifeweb		= { $EFFECT$ = { MODIFIER = POD_send_bound_wraith_interaction_lifeweb_modifier } }
		trait:arcanos_moliate		= { $EFFECT$ = { MODIFIER = POD_send_bound_wraith_interaction_moliate_modifier } }
		trait:arcanos_outrage		= { $EFFECT$ = { MODIFIER = POD_send_bound_wraith_interaction_outrage_modifier } }
		trait:arcanos_puppetry		= { $EFFECT$ = { MODIFIER = POD_send_bound_wraith_interaction_puppetry_modifier } }
		trait:arcanos_usury			= { $EFFECT$ = { MODIFIER = POD_send_bound_wraith_interaction_usury_modifier } }
		# no modifiers from keening, mnemonysis, pandemonium, and phantasm
		fallback = {}
	}
