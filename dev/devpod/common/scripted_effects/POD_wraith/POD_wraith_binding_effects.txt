POD_give_hook_based_on_bind_level_effect = {
	if = { 
		limit = { 
			$NECROMANCER$ = { NOT = { has_hook_of_type = { TYPE = necromancy_bind_permanent_hook target = PREV } } }
			fetter_bind_level >= 10
		}
		$NECROMANCER$ = {
			remove_hook = {
				target = PREV
				type = necromancy_bind_strong_hook
			}	
			add_hook = {
				target = PREV
				type = necromancy_bind_permanent_hook
			}
		}
	}
	else_if = { 
		limit = { 
			$NECROMANCER$ = { NOT = { has_hook_of_type = { TYPE = necromancy_bind_strong_hook target = PREV } } }
			fetter_bind_level >= 5 
		}
		$NECROMANCER$ = {
			remove_hook = {
				target = PREV
				type = necromancy_bind_hook
			}	
			add_hook = {
				target = PREV
				type = necromancy_bind_strong_hook
			}
		}
	}
	else_if = { 
		limit = { $NECROMANCER$ = { NOT = { has_hook_of_type = { TYPE = necromancy_bind_hook target = PREV } } } }
		$NECROMANCER$ = {
			add_hook = {
				target = PREV
				type = necromancy_bind_hook
			}
		}
	}
}

POD_unbind_wraith_effect = {
	$NECROMANCER$ = { 
		remove_hook = {
			target = PREV
			type = necromancy_bind_strong_hook
		}	
		remove_hook = {
			target = PREV
			type = necromancy_bind_hook
		}	
		remove_hook = {
			target = PREV
			type = necromancy_bind_hook
		}	
		remove_relation_bound_wraith = scope:wraith
	}
	$WRAITH$ = { 
		POD_destroy_fetter_of_type = { TYPE = bind }
	}
}
