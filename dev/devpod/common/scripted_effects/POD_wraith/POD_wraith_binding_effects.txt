POD_give_hook_based_on_bind_level_effect = {
	$NECROMANCER$ = { save_scope_as = necromancer }
	if = { 
		limit = { 
			scope:necromancer = { NOT = { has_hook_of_type = { TYPE = necromancy_bind_permanent_hook target = PREV } } }
			fetter_bind_level >= 10
		}
		scope:necromancer = {
			remove_hook = {
				target = PREV
				type = necromancy_bind_strong_hook
			}	
			add_hook = {
				target = PREV
				type = necromancy_bind_permanent_hook
			}
		}
	}
	else_if = { 
		limit = { 
			scope:necromancer = { NOT = { has_hook_of_type = { TYPE = necromancy_bind_strong_hook target = PREV } } }
			fetter_bind_level >= 5 
		}
		scope:necromancer = {
			remove_hook = {
				target = PREV
				type = necromancy_bind_hook
			}	
			add_hook = {
				target = PREV
				type = necromancy_bind_strong_hook
			}
		}
	}
	else_if = { 
		limit = { scope:necromancer = { NOT = { has_hook_of_type = { TYPE = necromancy_bind_hook target = PREV } } } }
		scope:necromancer = {
			add_hook = {
				target = PREV
				type = necromancy_bind_hook
			}
		}
	}
}

POD_unbind_wraith_effect = {
	$NECROMANCER$ = { save_scope_as = necromancer }
	$WRAITH$ = { save_scope_as = wraith }
	scope:necromancer = { 
		remove_hook = {
			target = scope:wraith
			type = necromancy_bind_strong_hook
		}	
		remove_hook = {
			target = scope:wraith
			type = necromancy_bind_hook
		}	
		remove_hook = {
			target = scope:wraith
			type = necromancy_bind_hook
		}	
		remove_relation_bound_wraith = scope:wraith
	}
	scope:wraith = {
		POD_destroy_fetter_of_type = { TYPE = bind }
		# Hideout Upgrades — remove domicile wraith enhancement modifiers
		POD_remove_character_modifier = { MODIFIER = POD_domicile_wraith_modifier_t1 }
		POD_remove_character_modifier = { MODIFIER = POD_domicile_wraith_modifier_t2 }
		POD_remove_character_modifier = { MODIFIER = POD_domicile_wraith_modifier_t3 }
	}
}
