		limit = { rarity_to_create >= 180 }
		change_variable = {
			name = illustrious_to_create

		}

			add = 1
		}
	}
}

		text = remove_famed_artifact_pieces
		value = $VALUE$
		if = {
			limit = { has_variable = famed_artifact_piece_reserve }
			change_variable = {
				name = famed_artifact_piece_reserve
				subtract = $VALUE$
			}
		}
	}
}
