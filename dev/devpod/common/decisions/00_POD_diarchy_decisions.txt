POD_start_powerstruggle_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_knight_kneeling.dds"
	}


	desc = POD_start_powerstruggle_decision_desc
	decision_group_type = pod_political	
	
	ai_check_interval = 60
	
	cooldown = { years = 10 }

	is_shown = {
		highest_held_title_tier >= tier_county		
		is_independent_ruler = no
        liege ?= { has_active_diarchy = no }
		is_powerful_vassal = yes
		liege.realm_size > 5
	}
		
	is_valid = {
		custom_tooltip = {
			text = "diarchy_prestige_tt"
			
			prestige > liege.prestige
		}
		custom_tooltip = {
			text = "diarchy_military_tt"
			
			max_military_strength > liege.max_military_strength
		}
	}
	
	is_valid_showing_failures_only = {
		is_at_war = no
		is_imprisoned = no
		is_adult = yes
	}
	
	effect = {
		custom_tooltip = POD_start_powerstruggle_decision_effect_tooltip
		
		save_scope_as = potential_diarch

		liege ?= {
		trigger_event = POD_diarchy_events.0002	
		}
	}

	ai_potential = {
		years_from_game_start > 10 #Gives the setup room to breathe
		NOT = { likes_character_trigger = { CHARACTER = liege } } #Don't do it to someone you like
	}

	ai_will_do = {
		base = 100
	}
}	

POD_end_powerstruggle_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_knight_kneeling.dds"
	}


	desc = POD_end_powerstruggle_decision_desc
	decision_group_type = pod_political
	ai_check_interval = 0
	
	cost = { prestige = 100 }

	is_shown = {
		is_diarch_of_target = liege
	}
		
	is_valid = {
		is_at_war = no
		is_imprisoned = no
	}
	
	effect = {
		liege ?= {
			end_diarchy = yes	
		}
	}

	ai_will_do = {
		base = 0
	}
}
