
change_house_colors_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_spend_money.dds"
	}

	decision_group_type = pod_customization
	ai_check_interval = 0

	widget = {
		gui = "decision_view_widget_option_list_generic"
		controller = decision_option_list_controller
		decision_to_second_step_button = "NEXT_LABEL"
		show_from_start = yes

		item = {
			value = POD_colors_personal
			current_description = change_house_colors_decision_desc
			localization = change_house_colors_option_personal
			icon = "gfx/interface/icons/character_interactions/icon_personal.dds"
			ai_chance = { value = 100 }
		}
		item = {
			value = POD_colors_coterie
			is_valid = {
				POD_has_coterie_trigger = yes
			}
			current_description = change_house_colors_decision_desc
			localization = change_house_colors_option_coterie
			icon = "gfx/interface/icons/character_interactions/council.dds"
			ai_chance = { value = 0 }
		}
		item = {
			value = POD_colors_house
			is_shown = {
				exists = house
			}
			is_valid = {
				is_house_head = yes
			}
			current_description = change_house_colors_decision_desc
			localization = change_house_colors_option_house
			icon = "gfx/interface/icons/character_interactions/icon_ingratiate_family.dds"
			ai_chance = { value = 0 }
		}
	}
	
	is_shown = {
		always = yes
	}
	
	desc = change_house_colors_decision_desc

	effect = {
		custom_tooltip = change_house_colors_decision_effect_tooltip
		if = {
			limit = { scope:POD_colors_coterie = yes }
			custom_tooltip = change_house_colors_decision_effect_tooltip_coterie
		}
		if = {
			limit = { scope:POD_colors_house = yes }
			custom_tooltip = change_house_colors_decision_effect_tooltip_house
		}
		custom_tooltip = change_house_colors_decision_effect_tooltip_3

		save_scope_as = target
		scope:target = {
			switch = {
				trigger = yes
				scope:POD_colors_personal = {
					# Variables are for the event widget, scopes are for the color display
					set_variable = {
						name  = color_root
						value = this
					}
					save_scope_as = color_root
				}
				scope:POD_colors_coterie = {
					set_variable = {
						name  = color_root
						value = var:coterie
					}
					var:coterie = { save_scope_as = color_root }
					random_relation = {
						type = coterie_member
						save_scope_as = secondary_target
					}
				}
				scope:POD_colors_house = {
					set_variable = {
						name  = color_root
						value = house
					}
					house = { save_scope_as = color_root }
					house = {
						random_house_member = {
							limit = { NOT = { this = scope:target } }
							save_scope_as = secondary_target
						}
					}
				}
			}
		}
		custom_tooltip = POD_current_clothing_colors
		trigger_event = POD_style.1
	}

	ai_will_do = {
		base = 0
	}
}
