		remove_character_modifier = dah_modifier
		
		remove_character_flag = lost_dharma_2
		remove_character_flag = lost_dharma_3

		remove_character_flag = lost_dharma_5
		remove_character_flag = lost_dharma_6
		remove_character_flag = lost_dharma_7
		remove_character_flag = lost_dharma_8
		remove_character_flag = lost_dharma_9

			rank = 1

			limit = {

					trait = education_martial

				}

		}
	}

		always = yes
	}

	ai_will_do = {
		base = 100
	}
}

become_enlightened_decision = {
	title = become_enlightened_decision
	selection_tooltip = become_enlightened_decision_tooltip
	picture = "gfx/interface/illustrations/decisions/decision_personal_religious.dds"
	desc = become_enlightened_decision_desc
	confirm_text = become_enlightened_decision_confirm
	
	major = yes

	ai_check_interval = 0

	is_shown = {
		is_ai = no
		is_ruler = yes
		is_landed = yes
		is_kueijin_trigger = yes
		has_trait = dharma9
	}
	
	is_valid = {
		trigger_if = {
			limit = { faith = faith:deviltiger }
			custom_description = {
				text = deviltiger_become_enlightened_decision_trigger
				exists = var:pod_killed_person_count
				var:pod_killed_person_count >= 500
			}
		}
		trigger_else_if = {
			limit = { faith = faith:thrashingdragon }
			custom_tooltip = {
				text = become_enlightened_decision_thrashingdragon_tt
				has_character_modifier = knowledge_of_thousand_cranes_mother_modifier
			}
		}
		trigger_else_if = {
			limit = { faith = faith:songoftheshadow }
			custom_tooltip = {
				text = become_enlightened_decision_songoftheshadow_tt
				has_character_modifier = wisdom_of_black_pearl_mistress_modifier
			}
		}
		trigger_else_if = {
			limit = { faith = faith:resplendentcrane }
			has_government = august_court_government
			custom_tooltip = {
				text = become_enlightened_decision_resplendentcrane_tt
				capital_county.de_jure_liege.de_jure_liege.de_jure_liege = {
					any_de_jure_county = {
						count = all
						faith = root.faith
					}
				}
			}
		}
		trigger_else_if = {
			limit = { faith = faith:thousandwhispers }
			custom_description = {
				text = thousandwhispers_become_enlightened_decision_trigger
				exists = var:pod_killed_dhampyrchildren_count
				var:pod_killed_dhampyrchildren_count >= 23
			}
		}
		trigger_else = {
			OR = { 
				faith = faith:deviltiger
				faith = faith:thrashingdragon
				faith = faith:songoftheshadow
				faith = faith:resplendentcrane
				faith = faith:thousandwhispers
			}
		}
		OR = {
			piety >= 100000
			has_character_modifier = dah_modifier
		}
	}
	
	effect = {
		if = {
			limit = {
				has_character_modifier = dah_modifier
			}
			remove_character_modifier = dah_modifier
		}
		else = {
			add_piety = -100000
		}
		remove_trait = dharma9
		add_trait_force_tooltip = dharma10
		trigger_event = podkueijin.230
	}	
	
	ai_potential = {
		always = no
	}
}
