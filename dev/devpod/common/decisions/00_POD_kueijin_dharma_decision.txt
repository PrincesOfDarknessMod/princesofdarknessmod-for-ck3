become_enlightened_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}
	
	decision_group_type = major

	ai_check_interval = 0

	is_shown = {
		is_landed = yes
		is_kueijin_trigger = yes
		has_trait = dharma9
		religion = { is_in_family = rf_kueijin }
	}
	
	is_valid = {
		trigger_if = {
			limit = { has_global_variable = wan_xian_wan_kuei }
		}
		trigger_else = {
			trigger_if = {
				limit = { religion = religion:deviltigers_religion }
				custom_tooltip = {
					text = deviltiger_become_enlightened_decision_trigger
					POD_killed_character_count >= 500
				}
			}
			trigger_else_if = {
				limit = { religion = religion:thrashingdragons_religion }
				custom_tooltip = {
					text = become_enlightened_decision_thrashingdragon_tt
					has_character_modifier = knowledge_of_thousand_cranes_mother_modifier
				}
			}
			trigger_else_if = {
				limit = { religion = religion:boneflowers_religion }
				custom_tooltip = {
					text = become_enlightened_decision_songoftheshadow_tt
					has_character_modifier = wisdom_of_black_pearl_mistress_modifier
				}
			}
			trigger_else_if = {
				limit = { religion = religion:resplendentcranes_religion }
				custom_tooltip = {
					text = become_enlightened_decision_resplendentcrane_tt2
					has_government = august_court_government
				}
				custom_tooltip = {
					text = become_enlightened_decision_resplendentcrane_tt
					capital_county.de_jure_liege.de_jure_liege.de_jure_liege = {
						any_de_jure_county = {
							count = all
							faith = root.faith
						}
					}
				}
			}
			trigger_else_if = {
				limit = { religion = religion:thousandwhispers_religion }
				custom_tooltip = {
					text = thousandwhispers_become_enlightened_decision_trigger
					POD_killed_dhampyrchildren_count >= 23
				}
			}
			trigger_else = {
				days_of_continuous_peace >= 50000
			}
			OR = {
				piety >= 100000
				has_trait_xp = {
					trait = dharma9
					track = dharma9
					value >= 100
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		faith = { has_doctrine = special_doctrine_against_to_yomi_wan_hostility }
		title:e_demon_warrior_empire = { is_title_created = no }
		is_akuma_trigger = no
	}
	
	effect = {
		POD_major_objective_finished_effect = yes
		if = {
			limit = {
				piety < 100000
				has_character_modifier = dah_modifier
			}
			remove_character_modifier = dah_modifier
		}
		remove_trait = dharma9
		add_trait_force_tooltip = dharma10
		trigger_event = podkueijin.230
	}	
	
	ai_potential = {
		always = no
	}
}
