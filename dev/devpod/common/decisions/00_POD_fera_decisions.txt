hengeyokai_advance_rank_decision = {
	picture = "gfx/interface/illustrations/decisions/decision_personal_religious.dds"
	
	major = yes

	ai_check_interval = 60
	
	cooldown = { years = 5 }

	is_shown = {
		is_landed = yes
		has_trait = shapeshifter
		faith = { has_doctrine = special_doctrine_beast_courts }
		has_rank_xp = {
			COMPARATOR = "<"
			VALUE = 100
		}
	}
	
	is_valid = {
		trigger_if = {
			limit = { has_trait = kitsune }
			trigger_if = {
				limit = { has_trait_xp = { trait = kitsune value < 12 } }
				has_trait_xp = { trait = kitsune value >= 11 }
				piety_level >= 1
				legitimacy_level >= 1
				prestige_level >= 1
			}
			trigger_else_if = {
				limit = { has_trait_xp = { trait = kitsune value < 25 } }
				has_trait_xp = { trait = kitsune value >= 24 }
				piety_level >= 2
				legitimacy_level >= 2
				prestige_level >= 2
			}
			trigger_else_if = {
				limit = { has_trait_xp = { trait = kitsune value < 37 } }
				has_trait_xp = { trait = kitsune value >= 36 }
				piety_level >= 3
				legitimacy_level >= 3
				prestige_level >= 3
			}
			trigger_else_if = {
				limit = { has_trait_xp = { trait = kitsune value < 50 } }
				has_trait_xp = { trait = kitsune value >= 49 }
				piety_level >= 4
				legitimacy_level >= 4
				prestige_level >= 4
			}
			trigger_else_if = {
				limit = { has_trait_xp = { trait = kitsune value < 63 } }
				has_trait_xp = { trait = kitsune value >= 62 }
				piety_level >= 5
				legitimacy_level >= 5
				prestige_level >= 5
			}
			trigger_else_if = {
				limit = { has_trait_xp = { trait = kitsune value < 75 } }
				has_trait_xp = { trait = kitsune value >= 74 }
				piety_level >= 5
				legitimacy_level >= 5
				prestige_level >= 5
			}
			trigger_else_if = {
				limit = { has_trait_xp = { trait = kitsune value < 88 } }
				has_trait_xp = { trait = kitsune value >= 87 }
				piety_level >= 5
				legitimacy_level >= 5
				prestige_level >= 5
			}
			trigger_else = {
				has_trait_xp = { trait = kitsune value >= 99 }
				piety_level >= 5
				legitimacy_level >= 5
				prestige_level >= 5
			}
		}
		trigger_else = {
			trigger_if = {
				limit = { has_rank_xp = { COMPARATOR = "<" VALUE = 20 } }
				has_rank_xp = { COMPARATOR = ">=" VALUE = 19 }
				piety_level >= 1
				legitimacy_level >= 1
				prestige_level >= 1
			}
			trigger_else_if = {
				limit = { has_rank_xp = { COMPARATOR = "<" VALUE = 40 } }
				has_rank_xp = { COMPARATOR = ">=" VALUE = 39 }
				piety_level >= 2
				legitimacy_level >= 2
				prestige_level >= 2
			}
			trigger_else_if = {
				limit = { has_rank_xp = { COMPARATOR = "<" VALUE = 60 } }
				has_rank_xp = { COMPARATOR = ">=" VALUE = 59 }
				piety_level >= 3
				legitimacy_level >= 3
				prestige_level >= 3
			}
			trigger_else_if = {
				limit = { has_rank_xp = { COMPARATOR = "<" VALUE = 80 } }
				has_rank_xp = { COMPARATOR = ">=" VALUE = 79 }
				piety_level >= 4
				legitimacy_level >= 4
				prestige_level >= 4
			}
			trigger_else = {
				has_rank_xp = { COMPARATOR = ">=" VALUE = 99 }
				piety_level >= 5
				legitimacy_level >= 5
				prestige_level >= 5
			}
		}
	}
	
	effect = {
		switch = {
			trigger = has_trait
			werewolf = { add_trait_xp = { trait = werewolf value = 1 } }
			ajaba = { add_trait_xp = { trait = ajaba value = 1 } }
			ananasi = { add_trait_xp = { trait = ananasi value = 1 } }
			bastet = { add_trait_xp = { trait = bastet value = 1 } }
			corax = { add_trait_xp = { trait = corax value = 1 } }
			gurahl = { add_trait_xp = { trait = gurahl value = 1 } }
			kitsune = { add_trait_xp = { trait = kitsune value = 1 } }
			mokole = { add_trait_xp = { trait = mokole value = 1 } }
			nagah = { add_trait_xp = { trait = nagah value = 1 } }
			nuwisha = { add_trait_xp = { trait = nuwisha value = 1 } }
			ratkin = { add_trait_xp = { trait = ratkin value = 1 } }
			rokea = { add_trait_xp = { trait = rokea value = 1 } }
			apis = { add_trait_xp = { trait = rokea value = 1 } }
			camazotz = { add_trait_xp = { trait = camazotz value = 1 } }
			grondr = { add_trait_xp = { trait = grondr value = 1 } }
		}
		add_piety = minor_piety_gain
		add_legitimacy = minor_legitimacy_gain
		add_prestige = minor_prestige_gain
	}	
	
	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}

changing_moon_decision = {
	picture = "gfx/interface/illustrations/decisions/decision_personal_religious.dds"
	
	cost = { gold = { value = minor_gold_value } }
	cooldown = { years = 20 }

	is_shown = {
		is_landed = yes
		POD_is_gurahl_trigger = yes
		NOT = { has_trait = philodox }
	}
		
	is_valid = {	
		NOT = { has_trait = philodox } # tooltip only	
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
		is_at_war = no
	}
		
	effect = {
		if = {
			limit = { has_trait = ragabash }
			remove_trait = ragabash
			add_trait = ahroun
		}
		else_if = {
			limit = { has_trait = ahroun }
			remove_trait = ahroun
			add_trait = galliard
		}
		else_if = {
			limit = { has_trait = galliard }
			remove_trait = galliard
			add_trait = theurge
		}
		else = {
			remove_trait = theurge
			add_trait = philodox
		}
	}
	
	ai_check_interval = 360

	ai_potential = {
		years_from_game_start > 20
	}

	ai_will_do = {
		base = 100
	}
}
