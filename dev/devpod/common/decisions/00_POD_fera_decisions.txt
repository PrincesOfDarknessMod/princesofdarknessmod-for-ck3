
changing_moon_decision = {
	picture = "gfx/interface/illustrations/decisions/decision_personal_religious.dds"
	
	cost = { gold = { value = minor_gold_value } }
	cooldown = { years = 20 }

	is_shown = {
		is_landed = yes
		POD_is_gurahl_trigger = yes
		NOT = { has_trait = philodox }
	}
		
	is_valid = {	
		NOT = { has_trait = philodox } # tooltip only	
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
		is_at_war = no
	}
		
	effect = {
		if = {
			limit = { has_trait = ragabash }
			remove_trait = ragabash
			add_trait = ahroun
		}
		else_if = {
			limit = { has_trait = ahroun }
			remove_trait = ahroun
			add_trait = galliard
		}
		else_if = {
			limit = { has_trait = galliard }
			remove_trait = galliard
			add_trait = theurge
		}
		else = {
			remove_trait = theurge
			add_trait = philodox
		}
	}
	
	ai_check_interval = 360

	ai_potential = {
		years_from_game_start > 20
	}

	ai_will_do = {
		base = 100
	}
}
