						}
						trigger_else = {
							always = yes
						}
					}
				}
			}
		}
		#if medium, gain strong
		trigger_else_if = {
			limit = {
				has_character_flag = nergalite_jerusalem_sleeper_medium
			}
			highest_held_title_tier >= tier_kingdom
			completely_controls = title:c_jerusalem.kingdom

					any_in_de_jure_hierarchy = {

						}

							always = yes
						}
					}
				}
			}
		}

			limit = {
				has_character_flag = nergalite_jerusalem_sleeper_strong

				title:c_jerusalem.empire = {
					any_in_de_jure_hierarchy = {
						count = all
						trigger_if = {
							limit =  { tier = tier_county }
							development_level > 5
						}
						trigger_else = {
							always = yes
						}
					}
				}
			}
		}
		#if no flag, gain weak
		trigger_else = {
			highest_held_title_tier >= tier_county
			completely_controls = title:c_jerusalem
			title:c_jerusalem = {
				development_level > 5
			}
		}
	}
	
	effect = {
		if = {
			limit = {
				has_character_flag = nergalite_jerusalem_sleeper_strong
			}
			remove_character_flag = nergalite_jerusalem_sleeper_strong
			add_character_flag = nergalite_jerusalem_sleeper_awake
