### Decision 1: "Invoke the Authority of the Living" ###
# Requires Child of Aye (Tier 3)

pod_laibon_invoke_authority_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_king_of_kings.dds"
	}

	decision_group_type = pod_splat
	ai_check_interval = 120

	desc = pod_laibon_invoke_authority_decision_desc

	is_shown = {
		is_laibon_vampire = yes
		#has_aye_tier_3 = yes
	}

	is_valid = {
		has_aye_tier_3 = yes
		is_landed = yes
	}

	cost = {
		prestige = major_prestige_value
	}

	cooldown = { years = 10 }

	effect = {
		# +10 popular opinion and +10% control growth for 5 years
		add_character_modifier = {
			modifier = invoke_authority_of_the_living_modifier
			years = 5
		}

		# All Laibon vassals gain +5 opinion for 5 years
		every_vassal = {
			limit = {
				religion = religion:roadoflaibon_religion
			}
			add_character_modifier = {
				modifier = laibon_vassal_authority_modifier
				years = 5
			}
		}
	}

	ai_will_do = {
		base = 50
	}
}

### Decision 2: "Commune with the Ancestors" ###
# Requires Child of Orun (Tier 3)

pod_laibon_commune_ancestors_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_POD_quietus.dds"
	}

	decision_group_type = pod_splat
	ai_check_interval = 120

	desc = pod_laibon_commune_ancestors_decision_desc

	is_shown = {
		is_laibon_vampire = yes
		#has_orun_tier_3 = yes
	}

	is_valid = {
		has_orun_tier_3 = yes
	}

	cost = {
		piety = medium_piety_value
	}

	cooldown = { years = 10 }

	effect = {
		# Reveal a secret in your realm
		custom_tooltip = pod_laibon_commune_reveal_secret_tt
		hidden_effect = {
			save_scope_as = actor
			random_vassal_or_below = {
				limit = {  
					any_secret = { 
						NOT = { is_known_by = scope:actor } 
						}
				}
				random_secret = {
					limit = { NOT = { is_known_by = scope:actor } }
					reveal_to = scope:actor
				}
			}
		}

		# +5 prowess and scheme resistance for 5 years
		add_character_modifier = {
			modifier = commune_with_ancestors_modifier
			years = 5
		}
	}

	ai_will_do = {
		base = 50
	}
}
