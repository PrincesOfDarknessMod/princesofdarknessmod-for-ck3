
POD_demon_demand_kill_pure_decision = {
    picture = "gfx/interface/illustrations/decisions/decision_misc.dds"

    ai_check_interval = 0
	
    is_shown = {
		has_character_flag = demand_kill_pure_flag
	}

	is_valid = {
		custom_tooltip = {
			text = POD_demon_demand_kill_pure_decision_condition
			has_character_flag = demand_kill_pure_flag
		}
	}

	is_valid_showing_failures_only = {
		is_available = yes
	}

    effect = {
		remove_character_flag = demand_kill_pure_flag
		remove_character_flag = finish_kill_pure_flag
		
		add_investiture_lifestyle_xp = 100
	}

	ai_will_do = {
		base = 0
	}
}
