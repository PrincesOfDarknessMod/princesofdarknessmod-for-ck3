POD_demon_demand_prepare_bodies_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}
	decision_group_type = pod_quest
    ai_check_interval = 0
	
    is_shown = {
		has_character_flag = demand_prepare_bodies_flag
	}

	is_valid = {
		custom_tooltip = {
			text = POD_demon_demand_prepare_bodies_decision_condition
			any_prisoner = { 
				count >= 10
				has_trait = mortal
			}
		}
	}

	is_valid_showing_failures_only = {
		is_available = yes
	}

    effect = {
		remove_character_flag = demand_prepare_bodies_flag
		ordered_prisoner = {
			limit = { has_trait = mortal }
			order_by = age
			max = 10
			check_range_bounds = no
			death = {
	        	death_reason = death_sacrificed_to_gods
				killer = root
	        }
		}

		add_investiture_lifestyle_xp = 100
	}

	ai_will_do = {
		base = 0
	}
}

POD_demon_demand_bring_infernalist_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}
	decision_group_type = pod_quest
    ai_check_interval = 0
	
    is_shown = {
		has_character_flag = demand_bring_infernalist_flag
	}

	is_valid = {
		custom_tooltip = {
			text = POD_demon_demand_bring_infernalist_decision_condition
			var:to_infernalize = {
				OR = {
					is_alive = no
					opinion = {
						target = root
						value > 0
					}
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		is_available = yes
	}

    effect = {
		remove_character_flag = demand_bring_infernalist_flag
		
		if = {
			limit = { var:to_infernalize = { is_alive = yes } }
			set_character_faith = root.faith
		}
		remove_variable = to_infernalize
		
		add_investiture_lifestyle_xp = 100
	}

	ai_will_do = {
		base = 0
	}
}
