	}

	decision_group_type = pod_splat

	desc = sacrifice_perk_from_obtenebration_decision_desc

	ai_check_interval = 0
	
	cooldown = { years = 5 }
	
	is_shown = {
		has_trait = vampire
		NOT = { has_perk = abyssal_mysticism_level_5_perk }
		has_trait = obtenebrationadvanced
		
	}

	is_valid_showing_failures_only = {
		has_abyssal_mysticism_bloodmagic_trigger = yes

		else_if = { limit = { has_perk = abyssal_mysticism_level_3_perk } assign_abyssal_mysticism_effect = { LEVEL = 4 } }
		else_if = { limit = { has_perk = abyssal_mysticism_level_2_perk } assign_abyssal_mysticism_effect = { LEVEL = 3 } }
		else_if = { limit = { has_perk = abyssal_mysticism_level_1_perk } assign_abyssal_mysticism_effect = { LEVEL = 2 } }
		else = { assign_abyssal_mysticism_effect = { LEVEL = 1 } }
	}
