		calc_true_if = {
			amount >= 5

			completely_controls = title:c_baghdad
			has_trait = scholar
			has_trait = sorcerer
			has_trait = lifestyle_mystic
			has_trait_xp = {
				trait = lifestyle_mystic
				value >= 100
			}
			custom_tooltip = {
				text = judge_contact_magi_objective_decision_know_sorcerer
				OR = {
					any_relation = {
						type = coterie_member
						has_trait = sorcerer
					}
					any_relation = {
						type = friend
						has_trait = sorcerer
					}
					any_relation = {
						type = best_friend
						has_trait = sorcerer
					}
					any_relation = {
						type = lover
						has_trait = sorcerer
					}
					any_relation = {
						type = soulmate
						has_trait = sorcerer
					}
				}
			}
		}
	}
	
	is_valid_showing_failures_only = {
		is_capable_adult = yes
		is_imprisoned = no
		is_independent_ruler = yes
		is_at_war = no
	}

	effect = {
		POD_major_objective_finished_effect = yes
		show_as_tooltip = {
			if = {
				limit = { POD_splat_is_valid_for_sorcery = yes }
				faith:judge = {
					add_doctrine = special_doctrine_sorcerer_ordination
				}
				if = {
					limit = { NOT = { faith = faith:judge } }
					faith = {
						add_doctrine = special_doctrine_sorcerer_ordination
					}
				}
				add_trait = sorcerer
			}
			faith = {
				add_doctrine = doctrine_magicraft_accepted
			}
			POD_add_free_dynasty_perk = { PERK = hunter_judges_legacy_1 }
		}
		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:judge_contact_magi_objective_decision
		}
		trigger_event = pod1230objectives_judges.1

		# changed this to judge-only since sargon shards are for a judge objective
		if = {
			limit = { faith = faith:judge }
			custom_tooltip = quest_for_sargon_codex_objective_decision_at_effect1
		}
