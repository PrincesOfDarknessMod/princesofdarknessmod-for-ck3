# Adopt Republican Ideals/Adopt Republic Government
gdp_adopt_republic_government_decision = {
	title = gdp_adopt_republic_government_decision
	desc = gdp_adopt_republic_government_decision_desc
	selection_tooltip = gdp_adopt_republic_government_decision_tooltip
	picture = {
		reference = "gfx/interface/illustrations/event_story/gdp_republican.dds"
	}

	sort_order = 40
	decision_group_type = realm

	cost = {
		prestige = {
			value = monumental_prestige_value
			if = {
				limit = { primary_title.tier >= tier_kingdom }
				multiply = 4
			}
			else_if = {
				limit = { primary_title.tier = tier_duchy }
				multiply = 2
			}
		}
		gold = {
			value = 200
			if = {
				limit = { primary_title.tier >= tier_kingdom }
				multiply = 8
			}
			else_if = {
				limit = { primary_title.tier = tier_duchy }
				multiply = 4
			}
		}
	}

	ai_check_interval_by_tier = {
		barony = 0
		county = 240
		duchy = 0
		kingdom = 0
		empire = 0
		hegemony = 0
	}

	is_shown = {
		always = no
		has_gdp_dlc_trigger = yes
		# You are not already a republic
		NOR = {
			government_has_flag = government_is_republic
			government_has_flag = government_is_tribal
			government_has_flag = government_has_herd
		}
		# Your realm isn't already too large, at which point we hide the decision altogether
		sub_realm_size < 75
		# You own land
		is_landed = yes
	}

	is_valid = {
		save_temporary_scope_as = founding_ruler
		prestige_level >= 3
		capital_county = {
			popular_opinion >= 20
		}
		sub_realm_size < 25 # Large realm should just become admin instead
		capital_province = {
			province_size_level >= 3
		}
		OR = {
			is_independent_ruler = yes
			trigger_if = {
				limit = {
					has_variable = has_liege_permission_to_found_republic
					var:has_liege_permission_to_found_republic = liege
				}
				custom_tooltip = {
					text = liege_has_granted_city_charter_duration
					has_variable = has_liege_permission_to_found_republic
					var:has_liege_permission_to_found_republic = liege
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = liege_has_granted_city_charter
					has_variable = has_liege_permission_to_found_republic
					var:has_liege_permission_to_found_republic = liege
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		is_available_quick = {
			adult = yes
			at_war = no
			imprisoned = no
			incapable = no
		}
	}

	effect = {
		show_as_tooltip = { # Actual effect is applied in follow-up event
			gdp_found_a_republic_effect = yes
		}
		trigger_event = republic_event.0001
	}

	ai_potential = {
		# AI rulers will only attempt to become republics if their capital is in an acceptable location
		capital_province ?= {
			geographical_region = world_europe
		}
		# And is of a culture we would expect to form republics
		culture ?= {
			OR = {
				# German cities
				has_cultural_pillar = heritage_central_germanic
				# Visby
				has_cultural_pillar = heritage_north_germanic
				# Italian city states
				has_cultural_tradition = tradition_republican_legacy
			}
		}
	}

	ai_will_do = {
		base = 100
	}
}

				calc_true_if = {
					amount >= 10
					has_innovation = innovation_motte
					has_innovation = innovation_catapult
					has_innovation = innovation_barracks
					has_innovation = innovation_mustering_grounds
					has_innovation = innovation_bannus
					has_innovation = innovation_quilted_armor
					has_innovation = innovation_pattern_welding
					has_innovation = innovation_gavelkind
					has_innovation = innovation_crop_rotation
					has_innovation = innovation_city_planning
					has_innovation = innovation_casus_belli
					has_innovation = innovation_plenary_assemblies
					has_innovation = innovation_ledger
					has_innovation = innovation_prosperity_01
					has_innovation = innovation_currency_01
				}
			}
		}
		OR = {
			is_independent_ruler = yes
			trigger_if = {
				limit = {
					has_variable = has_liege_permission_to_found_republic
					var:has_liege_permission_to_found_republic = liege
				}
				custom_tooltip = {
					text = liege_has_granted_city_charter_duration
					has_variable = has_liege_permission_to_found_republic
					var:has_liege_permission_to_found_republic = liege
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = liege_has_granted_city_charter
					has_variable = has_liege_permission_to_found_republic
					var:has_liege_permission_to_found_republic = liege
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		custom_tooltip = {
			text = permafrost_feudal_ban
			NOT = { culture = { has_cultural_parameter = permafrost_cannot_become_feudal } }
		}
		is_available_quick = {
			adult = yes
			at_war = no
			imprisoned = no
			incapable = no
		}
	}

	effect = {
		show_as_tooltip = { # Actual effect is applied in follow-up event
			custom_tooltip = {
				text = gdp_tribal_capital_becomes_city_tt
			}
			custom_tooltip = {
				text = gdp_other_tribal_holdings_become_castles_tt
			}
			if = {
				limit = {
					any_vassal_or_below = {
						government_has_flag = government_is_tribal
					}
				}
				custom_tooltip = {
					text = gdp_tribal_vassals_will_feudalize_tt
				}
			}
			if = {
				limit = {
					any_sub_realm_county = {
						OR = {
							has_county_modifier = mpo_siberian_permafrost_modifier
							has_county_modifier = mpo_siberian_permafrost_modifier_bad
						}
					}
				}
				custom_tooltip = {
					text = all_permafrost_will_clear
				}
			}
			gdp_found_a_republic_effect = yes
		}
		trigger_event = republic_event.0002
	}

	ai_potential = {
		is_independent_ruler = yes
		NOT = { mpo_has_gok_mongol_empire_trigger = yes }
		capital_province ?= {
			is_coastal = yes
			geographical_region = world_europe
		}
		culture ?= {
			OR = {
				has_cultural_pillar = heritage_central_germanic
				has_cultural_pillar = heritage_north_germanic
				has_cultural_pillar = heritage_east_slavic
				has_cultural_tradition = tradition_republican_legacy
			}
		}
	}

	ai_will_do = {
		base = 100
	}
}

	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}
	decision_group_type = realm
	sort_order = 40
	desc = convert_to_feudalism_from_republic_decision_desc

	is_shown = {
		government_has_flag = government_is_republic
		highest_liege_matching_rule:uses_assembly = this
	}

	is_valid = {
		is_independent_ruler = yes
		has_realm_law = republican_despot_succession_law
		prestige_level >= { # Depends on tier
			value = primary_title.tier
			subtract = 1
			min = 2
		}
	}

	cost = {
		prestige = 1000
		treasury = {
			value = 1000
			culture = {
				if = {
					limit = { has_cultural_era_or_later = culture_era_late_medieval }
					multiply = 1.75
				}
				else_if = {
					limit = { has_cultural_era_or_later = culture_era_high_medieval }
					multiply = 1.25
				}
				else_if = {
					limit = { has_cultural_era_or_later = culture_era_early_medieval }
					multiply = 1
				}
				else = {
					multiply = 0.75
				}
			}
		}
	}

	effect = {
		convert_to_feudalism_from_republic_effect = yes
		save_scope_as = actor
		trigger_event = ep3_decisions_event.4070
		hidden_effect = {
			every_vassal = {
				trigger_event = {
					id = ep3_decisions_event.4075
					days = 3
				}
			}
		}
	}

	ai_check_interval = 0

	ai_potential = {
		always = no
	}
}

				if = {
					limit = {
						trade_post_count > 0
					}

							barony = {
								trade_market = {
									if = {
										limit = {
											NOT = { is_in_list = item_list }
										}
										add_to_list = item_list
									}
								}
							}
						}
					}
				}
				else = { # We add the capital's market as a fallback, so that the effects remains visible even without trade posts
					capital_barony.trade_market = { add_to_list = item_list }
				}
			}
		}

		default_item = {
			scope:actor = {
				ordered_owned_trade_post_district = {
					limit = {
						district_province.barony.trade_market = {
							gdp_can_leverage_trade_posts_in_market_trigger = { CHARACTER = scope:actor }

					}

		}

		ai_select_item = {
			scope:actor = {
				if = {
					limit = {
						any_owned_trade_post_district = {
							district_province.barony = {
								this = scope:actor.capital_barony
								trade_market = {
									gdp_can_leverage_trade_posts_in_market_trigger = { CHARACTER = scope:actor }
								}
							}
						}
					}

				}
				else = {
					random_owned_trade_post_district = {
						limit = {
							district_province.barony.trade_market = {
								gdp_can_leverage_trade_posts_in_market_trigger = { CHARACTER = scope:actor }
							}
						}
						district_province.barony.trade_market = { save_scope_as = selected_item }
					}
				}
			}
		}
	}

	}
