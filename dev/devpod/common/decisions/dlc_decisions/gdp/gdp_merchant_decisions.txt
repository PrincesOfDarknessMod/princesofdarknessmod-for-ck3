### Become Merchant ###
become_merchant_decision = {
	picture = {
		reference = "gfx/interface/illustrations/event_story/gdp_decision_become_merchant.dds"
	}
	sort_order = 41
	decision_group_type = realm

	ai_check_interval = 0

	desc = become_merchant_decision_desc
	selection_tooltip = become_merchant_decision_tooltip

	is_shown = {
		always = no
		# DLC check
		has_gdp_dlc_trigger = yes
		# House check
		exists = house.house_head
		OR = {
			government_has_flag = government_is_landless_adventurer
			government_has_flag = government_is_republic
			is_landed = yes
			AND = {
				government_has_mechanic = administrative
				top_liege = this
				is_landed = no
			}
		}
		# Relevant cultures/faiths
		OR = {
			government_has_flag = government_is_landless_adventurer
			culture = { has_cultural_parameter = unlock_voluntary_mercantilism }
			AND = {
				government_has_mechanic = administrative
				top_liege = this
				is_landed = no
			}
		}
	}

	is_valid = {
		trigger_if = { # Triggers for non-laamps
			limit = {
				NOT = { government_has_flag = government_is_landless_adventurer }
			}
			gdp_is_valid_for_merchant_trigger = yes
			# Is not HOF
			custom_tooltip = {
				text = become_landless_adventurer_decision_hof_tt
				faith.religious_head_title.holder != root
			}
			# Counts and dukes only
			custom_tooltip = {
				text = become_landless_adventurer_decision_tier_tt
				highest_held_title_tier >= tier_county
				highest_held_title_tier <= tier_duchy
			}
			# Not too big
			custom_tooltip = {
				text = become_landless_adventurer_decision_domain_size_tt
				domain_size <= 3
			}
			# Must be a house head
			custom_tooltip = {
				text = become_merchant_decision_house_head_tt
				is_house_head = yes
			}
			# Relevant cultures/faiths
			trigger_if = {
				limit = {
					NOT = { government_allows = profit_trade_routes }
				}
				trigger_if = {
					limit = {
						OR = {
							is_ai = yes
							NOT = { government_has_mechanic = administrative }
							top_liege != this
							is_landed = yes
						}
					}
					culture = { has_cultural_parameter = unlock_voluntary_mercantilism }
				}
			}
		}
		trigger_else_if = {
			limit = {
				exists = var:gdp_recently_became_adventurer_cooldown
			}
			custom_tooltip = {
				text = gdp_recently_became_adventurer_decision_cooldown_tt
				NOT = { exists = var:gdp_recently_became_adventurer_cooldown }
				government_has_flag = government_is_landless_adventurer
			}
		}
		trigger_else = { # trigger for laamps
			custom_tooltip = {
				text = landless_government_tt
				government_has_flag = government_is_landless_adventurer
			}
		}
	}

	is_valid_showing_failures_only = {
		is_at_war = no
		is_alive = yes
		is_available_adult = yes
		has_contagious_deadly_disease_trigger = no
	}

	effect = {
		save_scope_as = merchant
		custom_tooltip = become_merchant_tt
		# A camp has no realm to pass on, so it gets no inheritor
		if = {
			limit = {
				NOT = { government_has_flag = government_is_landless_adventurer }
			}
			if = {
				limit = {
					player_heir ?= {
						this != root
						is_adult = yes
					}
				}
				player_heir = {
					save_scope_as = laamp_heir
					save_scope_as = laamp_inheritor
					custom_tooltip = become_landless_adventurer_decision_title_scope_tt
				}
			}
			else_if = {
				limit = { top_liege != this }
				liege = {
					save_scope_as = laamp_liege
					save_scope_as = laamp_inheritor
					custom_tooltip = become_landless_adventurer_decision_title_scope_tt
				}
			}
			else = {
				custom_tooltip = become_landless_adventurer_decision_title_created_tt
				if = {
					limit = {
						NOT = { exists = scope:laamp_inheritor }
					}
					hidden_effect_new_object = {
						create_character = {
							template = laamp_inheritor_template
							culture = root.capital_county.culture
							faith = root.capital_county.faith
							dynasty = generate
							location = root.capital_province
							after_creation = {
								save_scope_as = laamp_inheritor
								add_character_flag = new_laamp_inheritor
							}
						}
					}
				}
			}
		}
		show_as_tooltip = {
			add_character_modifier = {
				modifier = gdp_voluntary_merchant_modifier
				years = 5
			}
		}
		# Save scopes for loc
		primary_title = { save_scope_as = old_primary_title }
		capital_province = { save_scope_as = old_capital_province }
		if = {
			limit = { top_liege != this }
			liege = { save_scope_as = old_liege }
		}
		trigger_event = { id = gdp_merchant.0001 }
		# Debug tracking
		if = {
			limit = {
				is_ai = yes
				debug_only = yes
			}
			if = {
				limit = { exists = global_var:ai_voluntary_merchant_count }
				change_global_variable = {
					name = ai_voluntary_merchant_count
					add = 1
				}
			}
			else = {
				set_global_variable = {
					name = ai_voluntary_merchant_count
					value = 1
				}
			}
			debug_log = "AI became merchant voluntarily"
			debug_log_scopes = yes
		}
		custom_tooltip = lose_all_prestige_tt
		custom_tooltip = lose_all_prestige_level_tt
		show_as_tooltip = {
			stress_impact = {
				craven = minor_stress_impact_gain
				content = minor_stress_impact_gain
				lazy = minor_stress_impact_gain
			}
		}
	}

	ai_potential = {
		is_valid_for_laampdom = yes
	}

	ai_will_do = {
		base = -75
		modifier = { add = merchant_chance_score_value }
	}
}

### Abuse Charter Authority ###
abuse_charter_authority_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/fractious.dds" }
	decision_group_type = merchant_major
	sort_order = 80

	cooldown = { years = 5 }

	ai_check_interval_by_tier = {
		barony = 0
		county = 120
		duchy = 0
		kingdom = 0
		empire = 0
		hegemony = 0
	}

	is_shown = {
		has_gdp_dlc_trigger = yes
		merchant_has_any_charter_trigger = yes
	}

	is_valid = {
		prestige_level >= 2
		# Must have a Charter from Patron or liege-or-above
		merchant_has_usable_charter_trigger = {
			ONLY_EQUIPPED = yes
			INCLUDE_FORGERIES = yes
		}
		# Must not be blocked
		trigger_if = {
			limit = { has_character_flag = no_more_charter_abuse }
			custom_tooltip = {
				text = merchant_charter_no_more_abuse
				NOT = { has_character_flag = no_more_charter_abuse }
			}
		}
	}

	is_valid_showing_failures_only = {
		is_available_quick = {
			adult = yes
			incapable = no
			imprisoned = no
			at_war = no
		}
		# Domicile county not at war
		custom_tooltip = {
			text = merchant_charter_domicile_location_peace
			domicile.domicile_location.county.holder = { is_at_war = no }
		}
		# Have not intervened here recently
		custom_tooltip = {
			text = merchant_charter_intervention_county_cooldown
			domicile.domicile_location.county = {
				valid_charter_abuse_market_trigger = { MERCHANT = root }
			}
		}
	}

	cost = {
		prestige = {
			value = 10
			if = {
				limit = { exists = domicile.domicile_location.county }
				multiply = {
					value = domicile.domicile_location.county.prosperity_level
					add = domicile.domicile_location.province_size_level
				}
			}
			min = 50
			max = 500
		}
	}

	effect = {
		add_to_variable_list = {
			name = charter_intervention_county_cooldown
			target = domicile.domicile_location.county
			years = charter_intervention_county_cooldown_years_value
		}
		add_dealbreaker_secret_effect = yes
		merchant_charter_authority_scope_effect = {
			PATRON = root.suzerain
			MERCHANT = root
		}
		show_as_tooltip = {
			merchant_charter_abuse_counter_effect = {
				PATRON = scope:patron
				MERCHANT = root
			}
			domicile.domicile_location.county ?= {
				merchant_charter_abuse_authority_county_effect = {
					MERCHANT = root
					CUT = flag:no
				}
			}
		}
		trigger_event = gdp_decision.1020
	}

	ai_will_do = {
		base = -20
		modifier = {
			add = -15
			suzerain ?= { has_character_flag = merchant_charter_abuse_punished }
		}
		modifier = {
			add = 10
			has_trait = greedy
		}
		modifier = {
			add = 10
			has_trait = wrathful
		}
		modifier = {
			add = 10
			has_trait = ambitious
		}
		modifier = {
			add = 10
			has_trait = callous
		}
		modifier = {
			add = -10
			has_trait = just
		}
	}
}

### Invite Merchants ###
invite_merchants_decision = {
	picture = { reference = "gfx/interface/illustrations/event_story/gdp_decision_become_merchant.dds" }
	sort_order = 99

	desc = invite_merchants_decision_desc
	selection_tooltip = invite_merchants_decision_tooltip

	cooldown = { years = 5 }

	cost = {
		gold = { value = major_gold_value }
	}

	ai_check_interval = 0

	is_shown = {
		always = no
		# Landed
		is_landed = yes
	}

	is_valid = {
		custom_tooltip = {
			text = invite_merchants_decision_capital_full_tt
			gdp_capital_province_merchant_count_value < invite_merchants_max_capital_merchants_value
		}
	}

	is_valid_showing_failures_only = {
		is_available_quick = {
			adult = yes
			at_war = no
			imprisoned = no
			incapable = no
		}
	}

	effect = {
		save_scope_as = new_patron
		custom_tooltip = invite_merchants_decision_effect_1
		custom_tooltip = {
			text = invite_merchants_decision_effect_2
			every_in_global_list = {
				variable = merchants_tally
				holder ?= {
					if = {
						limit = {
							# AI Patrons only generate new merchants
							scope:new_patron = { is_ai = no }
							# Standard checks
							government_has_flag = government_is_merchant
							exists = domicile
							exists = suzerain
							is_available_quick = {
								adult = yes
								at_war = no
								imprisoned = no
								incapable = no
							}
							NOR = {
								# Not already one your Beneficiary
								suzerain = scope:new_patron
								# Has not moved to any Patron recently
								has_character_flag = entice_merchant_cooldown
								# Not in queue to be asked to move
								has_character_flag = entice_merchant_blocker
								has_character_flag = entice_merchant_invited
								# Has refused all future invitations
								has_character_flag = entice_merchant_permanent_refusal
								# Has not been offered by same Patron recently
								is_target_in_variable_list = {
									name = entice_merchant_cooldown
									target = scope:new_patron
								}
							}
							# Does not have any Monopolies
							character_has_trade_monopoly = no
							# Does not have a usable charter
							NOT = {
								merchant_has_usable_charter_trigger = {
									ONLY_EQUIPPED = no
									INCLUDE_FORGERIES = no
								}
							}
							# Patron is not cooler than you
							suzerain.highest_held_title_tier <= root.highest_held_title_tier
							# Not invested in Patron
							NOT = { has_hook = suzerain }
							# Prevent poaching player Patron's AI merchants
							trigger_if = {
								limit = { is_ai = yes }
								suzerain = { is_ai = yes }
							}
							# Not allied/married to Patron
							NOT = { is_allied_to = scope:new_patron }
							# Does not love Patron
							NOT = {
								has_any_good_relationship_with_character_trigger = { CHARACTER = scope:new_patron }
							}
						}
						if = {
							limit = { suzerain.top_liege ?= scope:new_patron.top_liege }
							add_to_temporary_list = same_realm_merchants
						}
						else_if = {
							limit = {
								in_diplomatic_range = scope:new_patron
							}
							add_to_temporary_list = nearby_merchants
						}
					}
				}
			}
			# How many merchants this use may invite in total, moved and generated
			save_scope_value_as = {
				name = entice_merchant_budget
				value = {
					value = invite_merchants_max_capital_merchants_value
					subtract = gdp_capital_province_merchant_count_value
					max = invite_merchants_max_per_use_value
					min = 0
				}
			}
			# How many max merchants to move from nearby or same realm
			save_scope_value_as = {
				name = max_moving_merchants
				value = {
					value = highest_held_title_tier
					subtract = 2
					min = 1
					multiply = 2
					max = scope:entice_merchant_budget
				}
			}
			# How many merchants to move from nearby or same realm
			save_scope_value_as = {
				name = moving_merchants
				value = {
					value = scope:max_moving_merchants
					divide = 2
					floor = yes
					min = 1
					max = scope:entice_merchant_budget
				}
			}
			# Move Same-Realm Merchants
			ordered_in_list = {
				list = same_realm_merchants
				order_by = entice_merchants_order_value
				max = scope:moving_merchants
				check_range_bounds = no
				add_character_flag = {
					flag = entice_merchant_blocker
					months = 2
				}
				trigger_event = {
					id = gdp_decision.3000
					days = { 7 56 }
				}
				add_to_list = moving_merchants
			}
			# Move Nearby Merchants
			save_scope_value_as = {
				name = nearby_moving_merchants
				value = {
					value = scope:entice_merchant_budget
					every_in_list = {
						list = moving_merchants
						subtract = 1
					}
					max = scope:moving_merchants
					min = 0
				}
			}
			ordered_in_list = {
				list = nearby_merchants
				order_by = entice_merchants_order_value
				max = scope:nearby_moving_merchants
				check_range_bounds = no
				add_character_flag = {
					flag = entice_merchant_blocker
					months = 2
				}
				trigger_event = {
					id = gdp_decision.3000
					days = { 7 56 }
				}
				add_to_list = moving_merchants
			}
			# Generate merchants as a fallback
			save_scope_value_as = {
				name = unused_entice_merchant_budget
				value = {
					value = scope:entice_merchant_budget
					every_in_list = {
						list = moving_merchants
						subtract = 1
					}
					min = 0
				}
			}
			save_scope_value_as = {
				name = remaining_moving_merchants # how many fallback merchants to generate
				value = {
					value = scope:max_moving_merchants
					subtract = scope:moving_merchants
					min = 0
					max = 2
					# With a budget of 1 the formula above is 0, so generate one if nobody could be moved
