POD_wutian_change_house_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_POD_bardo.dds"
	}

	ai_check_interval = 0
	decision_group_type = pod_splat

	is_shown = {
		has_trait = wutian
	}

	is_valid = {
		has_trait_xp = {
			trait = integrity
			value >= 2
		}
		trigger_if = {
			limit = { dynasty = dynasty:POD_mummy_dynasty_xianlung }
			custom_tooltip = {
				text = POD_wutian_change_house_decision_requirement_xianlung
				POD_tan_lifestyle_feng_tan_tree_perk_amount > POD_tan_lifestyle_lung_tan_tree_perk_amount
			}
		}
		trigger_else = {
			custom_tooltip = {
				text = POD_wutian_change_house_decision_requirement_wufeng
				POD_tan_lifestyle_lung_tan_tree_perk_amount > POD_tan_lifestyle_feng_tan_tree_perk_amount
			}
		}
	}

	cooldown = { years = 5 }

	cost = { 
		piety = 500 
	}

	effect = {
		POD_static_stress_gain = {
			COUNT = 15
		}
		add_trait_xp = {
			trait = integrity
			value = -2
		}
		if = {
			limit = { dynasty = dynasty:POD_mummy_dynasty_xianlung }
			custom_tooltip = POD_wutian_change_house_decision_effect_wufeng
		}
		else = {
			custom_tooltip = POD_wutian_change_house_decision_effect_xianlung
		}
		trigger_event = podmummy.6000
	}

	ai_will_do = {
		base = 0
	}
}
