POD_wutian_change_house_decision = {

		reference = "gfx/interface/illustrations/decisions/decision_POD_bardo.dds"
	}

	ai_check_interval = 0
	decision_group_type = pod_splat

	is_shown = {
		has_trait = wutian
	}

	is_valid = {
		has_trait_xp = {
			trait = integrity
			value >= 2
		}
		OR = {
			custom_tooltip = {
				text = POD_wutian_change_house_decision_requirement_artifact
				any_character_artifact = {
					has_variable = flowers_scarlet_ebon
				}
			}
			trigger_if = {
				limit = { dynasty = dynasty:POD_mummy_dynasty_xianlung }
				custom_tooltip = {
					text = POD_wutian_change_house_decision_requirement_xianlung
					POD_tan_lifestyle_feng_tan_tree_perk_amount > POD_tan_lifestyle_lung_tan_tree_perk_amount
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = POD_wutian_change_house_decision_requirement_wufeng
					POD_tan_lifestyle_lung_tan_tree_perk_amount > POD_tan_lifestyle_feng_tan_tree_perk_amount
				}
			}
		}
	}

	cooldown = { years = 5 }

	cost = { 
		piety = 500 
	}

	effect = {
		POD_static_stress_gain = {
			COUNT = 15
		}
		add_trait_xp = {
			trait = integrity
			value = -2
		}
		if = {
			limit = { dynasty = dynasty:POD_mummy_dynasty_xianlung }
			custom_tooltip = POD_wutian_change_house_decision_effect_wufeng
		}
		else = {
			custom_tooltip = POD_wutian_change_house_decision_effect_xianlung
		}
		trigger_event = podmummy.6000
	}

	ai_will_do = {
		base = 0
	}
}

POD_spell_of_life_conversion = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_the_truth_about_brujah.dds"  #TODO: swap image
	}
	desc = POD_spell_of_life_conversion_desc
	selection_tooltip = POD_spell_of_life_conversion_tooltip
	decision_group_type = pod_splat
	ai_check_interval = 120
	
	cooldown = { years = 10 }

	is_shown = {
		is_landed = yes
		realm_county_to_convert_value >= 5  # must have a decent number of counties to convert
		has_trait = eternal_rebirth_seeker
		has_trait_xp = {
			trait = eternal_rebirth_seeker
			track = spell_of_life
			value = 100
		}
		NOT = { has_character_flag = cant_use_spell_of_life }
	}

	is_valid = {
		is_independent_ruler = yes
		trigger_if = {
			limit = { has_trait = integrity }
			has_trait_xp = { trait = integrity value >= 5 }
		}
	}

	cost = {		
		piety = {
			value = realm_county_to_convert_value
			multiply = 50
		}
	}

	effect = {
		save_temporary_scope_as = liege
		save_temporary_scope_value_as = {
			name = half_of_converted_counties
			value = {
				value = realm_county_to_convert_value
				divide = 2
				floor = yes
			}
		}
		# Add counties being converted
		every_sub_realm_county = {
			limit = {
				NOT = { faith = scope:liege.faith }
			}
			add_to_list = war_supported_counties_list
		}

		POD_gain_integrity_inverted_for_evil = { VALUE = -5 }  # evil loves destroying minds with the umbra
		
		custom_tooltip = {
			text = POD_spell_of_life_conversion_effect2
			change_global_variable = { 
				name = mummy_harm_to_reality
				add = scope:half_of_converted_counties
			}
		}

		custom_tooltip = {
			text = POD_spell_of_life_conversion_effect1
			every_in_list = {
				list = war_supported_counties_list
				set_county_faith = scope:liege.faith
			}
		}

		# Uses neteru_faith_initiation_effect to apply correct faith for given splat
		# Event prevents lag
		custom_tooltip = {
			text = POD_spell_of_life_conversion_effect4
			trigger_event = podmummy.4000
		}		
		
		hidden_effect = {
			root.capital_county = { save_scope_as = target }
			create_character = {
				location = root.location
				template = POD_enemy_spectre_character
				save_scope_as = attacker
			}
			scope:attacker = {
				# Initialize splat & make them a better commander
				add_martial_skill = 10
				add_prowess_skill = 30
				add_trait = aggressive_attacker
				add_trait = reckless
				add_trait = peasant_leader
				set_variable = { name = spirit_location value = flag:earth }  # prevents POD_send_to_dark_kingdom_effect from being called
				POD_become_spectre_effect = yes
		  		POD_arcanoi_initiation_effect = yes

				# Initialize as if faction head
				if = {
					limit = {
						NOT = { exists = scope:attacker.primary_title }
					}
					# Create a new title for the peasant.
					create_dynamic_title = {
						tier = duchy
						name = FORMER_NOMAD_KINGDOM_NAME
						adj = FORMER_NOMAD_KINGDOM_NAME_ADJ
					}
					create_title_and_vassal_change = {
						type = created
						save_scope_as = change
						add_claim_on_loss = no
					}
					scope:new_title = {
						set_capital_county = scope:target
						set_landless_title = yes
						set_destroy_on_succession = yes
						set_delete_on_destroy = yes
						set_no_automatic_claims = yes
						set_definitive_form = yes
						set_can_be_named_after_dynasty = no
						set_can_use_nomadic_naming = no
						change_title_holder = {
							holder = scope:attacker
							change = scope:change
						}
					}
					resolve_title_and_vassal_change = scope:change
					scope:attacker = {
						set_variable = {
							name = peasant_title
							value = scope:new_title
						}
					}
					scope:new_title = { generate_coa = factions }
				}
				# Add all converted counties to target list
				scope:liege = {
					every_in_list = {
						list = war_supported_counties_list
						scope:attacker = {
							add_to_variable_list = {
								name = war_supported_counties_list
								target = prev
							}
						}
					}
				}

				# Start war
				# CB chosen as works for landless and ends when no troops remaining
				start_war = {
					casus_belli = ep3_laamp_peasant_war
					target = root
				}
				# Should only have one war
				random_character_war = { save_temporary_scope_as = war }

				# Spawn army at capital to split player attention
				# Not supposed to be too hard, just risky if done at a bad time
				spawn_army = {
					men_at_arms = {
						type = hungry_shade
						stacks = scope:half_of_converted_counties
					}
					men_at_arms = {
						type = spectres
						stacks = {
							value = scope:half_of_converted_counties
							multiply = 2
						}
					}
					location = root.capital_province
					war = scope:war
					uses_supply = no
					inheritable = no
					war_keep_on_attacker_victory = yes
					name = summoned_army
				}
			}
		}
	}
}
