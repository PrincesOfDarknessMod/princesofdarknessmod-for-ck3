animal_perk_removal_decision = {
	picture = "gfx/interface/illustrations/decisions/channel_animal.dds"

    desc = animal_perk_removal_decision_desc

	is_shown = {
    has_POD_vamp_blood_trigger = yes
	animal_lifestyle_perk_points > 0
	}

    effect = {
	custom_tooltip = {
		text = animal_perk_removal_decision_custom
		add_animal_lifestyle_perk_points = -1
	}
	add_character_modifier = {
		modifier = excessive_animal_modifier
		years = 5
			}
    }
	
	ai_check_interval = 0
}
