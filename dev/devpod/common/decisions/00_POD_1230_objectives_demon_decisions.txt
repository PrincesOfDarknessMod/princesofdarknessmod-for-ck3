			if = {

				house = { add_house_modifier = eternal_avarice_modifier }
			}
			if = {
				limit = { faith = { has_doctrine = special_doctrine_vices_envy } }
				house = { add_house_modifier = eternal_envy_modifier }
			}
			if = {
				limit = { faith = { has_doctrine = special_doctrine_vices_gluttony } }
				house = { add_house_modifier = eternal_gluttony_modifier }
			}
			if = {
				limit = { faith = { has_doctrine = special_doctrine_vices_lust } }
				house = { add_house_modifier = eternal_lust_modifier }
			}
			if = {
				limit = { faith = { has_doctrine = special_doctrine_vices_pride } }
				house = { add_house_modifier = eternal_pride_modifier }
			}
			if = {
				limit = { faith = { has_doctrine = special_doctrine_vices_sloth } }
				house = { add_house_modifier = eternal_sloth_modifier }
			}
			if = {
				limit = { faith = { has_doctrine = special_doctrine_vices_wrath } }
				house = { add_house_modifier = eternal_wrath_modifier }
			}
		}
		custom_tooltip = POD_demon_gather_the_faithful_decision_unlock_sin_trait_vassalization
		custom_tooltip = POD_demon_gather_the_faithful_decision_next_objective

		add_to_global_variable_list = {
			name = unavailable_unique_decisions

	}
