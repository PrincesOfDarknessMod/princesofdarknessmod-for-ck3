				count >= 1
			}

			limit = {
				any_independent_ruler = {
					PoD_is_member_of_seasonal_courts = yes
					OR = {
						has_relation_rival = ROOT
						has_relation_nemesis = ROOT
					}
					opinion = {
						target = ROOT
						value <= 50
					}
					save_temporary_scope_as = example
				}
			}
			custom_tooltip = {
				text = POD_war_of_seasons_peace_victory_objective_custom2_ex
				always = no
			}
		}
		trigger_else = {
			custom_tooltip = {

			}
		}

POD_form_united_seasons_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_POD_faejourney_realms.dds"
	}

	decision_group_type = major

	ai_check_interval = 0

	desc = POD_form_united_seasons_decision_desc

	is_shown = {
		is_target_in_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:POD_war_of_seasons_peace_victory_objective
		}
		title:h_united_seasons ?= { is_title_created = no }
		is_ai = no
		is_landed = yes
	}

	is_valid = {
		is_independent_ruler = yes
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_at_war = no
	}

	effect = {
		hidden_effect = {
			every_held_title = { #Should shift all dejure of all Empires owned at the time.
				limit = {
					tier = tier_empire
				}
				every_in_de_jure_hierarchy = {
					limit = {
						tier = tier_kingdom
					}
					set_de_jure_liege_title = title:h_united_seasons

				}
			}
			every_held_title = { #Completely Controlled Kingdoms as well.
				limit = {
					tier = tier_kingdom
					root = { completely_controls = prev }
				}
				set_de_jure_liege_title = title:h_united_seasons
			}
		}
		create_title_and_vassal_change = {
			type = created
			save_scope_as = change
			add_claim_on_loss = no
		}
		title:h_united_seasons = {
			change_title_holder = {
				holder = root
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change
		hidden_effect = { set_primary_title_to = title:h_united_seasons }
		every_held_title = {
			limit = {
				tier = tier_hegemony
				NOT = { this = title:h_united_seasons }
			}
			root = { destroy_title = prev }
		}
	}

	ai_potential = {
		NOT = { any_player = { has_POD_fae_blood_trigger = yes } }
	}

	ai_will_do = {
		base = 0
	}
}
