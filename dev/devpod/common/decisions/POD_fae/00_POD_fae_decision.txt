POD_fable_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}
	decision_group_type = pod_splat
	ai_check_interval = 0

	is_shown = {
		is_ai = no
		is_landless_adventurer = yes
		POD_can_participate_in_fae_politics = yes
		NOR = {
			has_trait_xp = { trait = fables track = fable_of_herald value = 100 }
			has_trait_xp = { trait = fables track = fable_of_marshal value = 100 }
			has_trait_xp = { trait = fables track = fable_of_steward value = 100 }
			has_trait_xp = { trait = fables track = fable_of_shadow value = 100 }
			has_trait_xp = { trait = fables track = fable_of_sage value = 100 }
		}
		trigger_if = {
			limit = { has_trait = fables }
			has_variable = chosen_fable
		}
	}

	is_valid_showing_failures_only = {
		is_available = yes
		trigger_if = {
			limit = { has_variable = chosen_fable }
			switch = {
				trigger = var:chosen_fable
				flag:fable_of_herald = {
					trigger_if = {
						limit = { has_trait_xp = { trait = fables track = fable_of_herald value < 25 } }
						POD_has_completed_diplomacy_contracts_trigger = { VALUE = 5 }
						prestige_level >= 2
					}
					trigger_else_if = {
						limit = { has_trait_xp = { trait = fables track = fable_of_herald value < 50 } }
						POD_has_completed_diplomacy_contracts_trigger = { VALUE = 10 }
						prestige_level >= 3
					}
					trigger_else_if = {
						limit = { has_trait_xp = { trait = fables track = fable_of_herald value < 75 } }
						POD_has_completed_diplomacy_contracts_trigger = { VALUE = 15 }
						prestige_level >= 4
					}
					trigger_else = {
						limit = { has_trait_xp = { trait = fables track = fable_of_herald value < 100 } }
						POD_has_completed_diplomacy_contracts_trigger = { VALUE = 20 }
						prestige_level >= 5
					}
				}
				flag:fable_of_marshal = {
					trigger_if = {
						limit = { has_trait_xp = { trait = fables track = fable_of_marshal value < 25 } }
						POD_has_completed_martial_contracts_trigger = { VALUE = 5 }
						POD_has_won_battles_as_commander_trigger = { VALUE = 20 }
					}
					trigger_else_if = {
						limit = { has_trait_xp = { trait = fables track = fable_of_marshal value < 50 } }
						POD_has_completed_martial_contracts_trigger = { VALUE = 10 }
						POD_has_won_battles_as_commander_trigger = { VALUE = 40 }
					}
					trigger_else_if = {
						limit = { has_trait_xp = { trait = fables track = fable_of_marshal value < 75 } }
						POD_has_completed_martial_contracts_trigger = { VALUE = 15 }
						POD_has_won_battles_as_commander_trigger = { VALUE = 60 }
					}
					trigger_else = {
						limit = { has_trait_xp = { trait = fables track = fable_of_marshal value < 100 } }
						POD_has_completed_martial_contracts_trigger = { VALUE = 20 }
						POD_has_won_battles_as_commander_trigger = { VALUE = 80 }
					}
				}
				flag:fable_of_steward = {
					trigger_if = {
						limit = { has_trait_xp = { trait = fables track = fable_of_steward value < 25 } }
						POD_has_completed_stewardship_contracts_trigger = { VALUE = 5 }
						gold >= 500
					}
					trigger_else_if = {
						limit = { has_trait_xp = { trait = fables track = fable_of_steward value < 50 } }
						POD_has_completed_stewardship_contracts_trigger = { VALUE = 10 }
						gold >= 1000
					}
					trigger_else_if = {
						limit = { has_trait_xp = { trait = fables track = fable_of_steward value < 75 } }
						POD_has_completed_stewardship_contracts_trigger = { VALUE = 15 }
						gold >= 1500
					}
					trigger_else = {
						limit = { has_trait_xp = { trait = fables track = fable_of_steward value < 100 } }
						POD_has_completed_stewardship_contracts_trigger = { VALUE = 20 }
						gold >= 2000
					}
				}
				flag:fable_of_shadow = {
					trigger_if = {
						limit = { has_trait_xp = { trait = fables track = fable_of_shadow value < 25 } }
						POD_has_completed_intrigue_contracts_trigger = { VALUE = 5 }
						POD_has_completed_hostile_scheme_trigger = { VALUE = 2 }
					}
					trigger_else_if = {
						limit = { has_trait_xp = { trait = fables track = fable_of_shadow value < 50 } }
						POD_has_completed_intrigue_contracts_trigger = { VALUE = 10 }
						POD_has_completed_hostile_scheme_trigger = { VALUE = 4 }
					}
					trigger_else_if = {
						limit = { has_trait_xp = { trait = fables track = fable_of_shadow value < 75 } }
						POD_has_completed_intrigue_contracts_trigger = { VALUE = 15 }
						POD_has_completed_hostile_scheme_trigger = { VALUE = 6 }
					}
					trigger_else = {
						limit = { has_trait_xp = { trait = fables track = fable_of_shadow value < 100 } }
						POD_has_completed_intrigue_contracts_trigger = { VALUE = 20 }
						POD_has_completed_hostile_scheme_trigger = { VALUE = 8 }
					}
				}
				flag:fable_of_sage = {
					trigger_if = {
						limit = { has_trait_xp = { trait = fables track = fable_of_sage value < 25 } }
						POD_has_completed_learning_contracts_trigger = { VALUE = 5 }
						POD_has_taught_perk_points_trigger = { VALUE = 5 }
					}
					trigger_else_if = {
						limit = { has_trait_xp = { trait = fables track = fable_of_sage value < 50 } }
						POD_has_completed_learning_contracts_trigger = { VALUE = 10 }
						POD_has_taught_perk_points_trigger = { VALUE = 10 }
					}
					trigger_else_if = {
						limit = { has_trait_xp = { trait = fables track = fable_of_sage value < 75 } }
						POD_has_completed_learning_contracts_trigger = { VALUE = 15 }
						POD_has_taught_perk_points_trigger = { VALUE = 15 }
					}
					trigger_else = {
						limit = { has_trait_xp = { trait = fables track = fable_of_sage value < 100 } }
						POD_has_completed_learning_contracts_trigger = { VALUE = 20 }
						POD_has_taught_perk_points_trigger = { VALUE = 20 }
					}
				}
			}
		}
	}

	effect = {
		if = {
			limit = { has_variable = chosen_fable }
			custom_tooltip = POD_fable_decision_advance
			switch = {
				trigger = var:chosen_fable
				flag:fable_of_herald = {
					add_trait_xp = {
						trait = fables
						track = fable_of_herald
						value = 25
					}
				}
				flag:fable_of_marshal = {
					add_trait_xp = {
						trait = fables
						track = fable_of_marshal
						value = 25
					}
				}
				flag:fable_of_steward = {
					add_trait_xp = {
						trait = fables
						track = fable_of_steward
						value = 25
					}
				}
				flag:fable_of_shadow = {
					add_trait_xp = {
						trait = fables
						track = fable_of_shadow
						value = 25
					}
				}
				flag:fable_of_sage = {
					add_trait_xp = {
						trait = fables
						track = fable_of_sage
						value = 25
					}
				}
			}
		}
		else = {
			custom_tooltip = POD_fable_decision_chose_fable
			random_list = {
				1 = {
					show_chance = no
					desc = POD_fae_maintenance.40.a
					custom_tooltip = POD_fae_maintenance.40.a.tt
				}
				1 = {
					show_chance = no
					desc = POD_fae_maintenance.40.b
					custom_tooltip = POD_fae_maintenance.40.b.tt
				}
				1 = {
					show_chance = no
					desc = POD_fae_maintenance.40.c
					custom_tooltip = POD_fae_maintenance.40.c.tt
				}
				1 = {
					show_chance = no
					desc = POD_fae_maintenance.40.d
					custom_tooltip = POD_fae_maintenance.40.d.tt
				}
				1 = {
					show_chance = no
					desc = POD_fae_maintenance.40.e
					custom_tooltip = POD_fae_maintenance.40.e.tt
				}
			}
			trigger_event = POD_fae_maintenance.40
		}
	}

	ai_will_do = {
		base = 0
	}
}
