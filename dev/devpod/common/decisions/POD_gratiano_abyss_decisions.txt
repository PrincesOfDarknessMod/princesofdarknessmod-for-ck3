delve_rituals_of_darkness_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_political_sin.dds"
	}

	decision_group_type = pod_splat

	desc = delve_rituals_of_darkness_decision_desc

	is_shown = {
		has_abyssal_mysticism_bloodmagic_trigger = yes
	}

	is_valid_showing_failures_only = {
		NOT = { has_trait = torpor }
		custom_tooltip = {
			text = delve_rituals_cooldown_tt
			NOT = { has_character_flag = delve_rituals_cooldown }
		}
		custom_tooltip = {
			text = delve_rituals_none_learnable_tt
			POD_gratiano_any_learnable_abyss_ritual_trigger = yes
		}
	}

	effect = {
		custom_tooltip = delve_rituals_of_darkness_decision_effect_tt
		hidden_effect = {
			if = {
				limit = {
					domicile ?= {
						OR = {
							has_domicile_parameter = domicile_lasombra_abyss_ritual_cost_bonus_t5
							has_domicile_parameter = domicile_lasombra_abyss_ritual_cost_bonus_t6
						}
					}
				}
				add_character_flag = { flag = delve_rituals_cooldown days = 1825 }
			}
			else_if = {
				limit = {
					domicile ?= {
						OR = {
							has_domicile_parameter = domicile_lasombra_abyss_ritual_cost_bonus_t3
							has_domicile_parameter = domicile_lasombra_abyss_ritual_cost_bonus_t4
						}
					}
				}
				add_character_flag = { flag = delve_rituals_cooldown days = 2555 }
			}
			else_if = {
				limit = {
					domicile ?= {
						OR = {
							has_domicile_parameter = domicile_lasombra_abyss_ritual_cost_bonus_t1
							has_domicile_parameter = domicile_lasombra_abyss_ritual_cost_bonus_t2
						}
					}
				}
				add_character_flag = { flag = delve_rituals_cooldown days = 3285 }
			}
			else = {
				add_character_flag = { flag = delve_rituals_cooldown days = 3650 }
			}
			trigger_event = POD_gratiano.0300
		}
	}

	# Player-only ritual acquisition.
	ai_check_interval = 0
	ai_potential = { always = no }
	ai_will_do = { base = 0 }
}
