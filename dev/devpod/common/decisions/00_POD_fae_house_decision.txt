sidhe_house_dominance_beaumayn_objective_decision = {

		has_POD_fae_blood_trigger = yes
		exists = dynasty

				name = unavailable_unique_decisions

			}
		}
	}

	is_valid = {
		is_dynast = yes
		dynasty:POD_fae_dynasty_beaumayn = { 
			dynasty_prestige_level >= max_dynasty_prestige_level 
		}
		trigger_if = {
			limit = { house = house:house_brandell }
			custom_description = {
				text = deverra_custom_decision # Re-use Loca
				object = character:POD_fae_character_54
				character:POD_fae_character_54 = { is_alive = no }
			}
		}
		trigger_else_if = {
			limit = { house = house:house_markesh }
			custom_description = {
				text = deverra_custom_decision # Re-use Loca
				object = character:POD_fae_character_53
				character:POD_fae_character_53 = { is_alive = no }
			}
		}
		trigger_else = {
			OR = {
				custom_description = {
					text = deverra_custom_decision # Re-use Loca
					object = character:POD_fae_character_53
					character:POD_fae_character_53 = { is_alive = no }
				}
				custom_description = {
					text = deverra_custom_decision # Re-use Loca
					object = character:POD_fae_character_54
					character:POD_fae_character_54 = { is_alive = no }
				}
			}
		}
	}
	
	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_at_war = no
	}

	effect = {
		show_as_tooltip = {
			add_prestige = massive_prestige_value
			dynasty = { add_dynasty_prestige = excessive_dynasty_prestige_value }
			if = {
				limit = { house = house:house_brandell }
				dynasty = { add_dynasty_modifier = brandell_victory_modifier }
			}
			else_if = {
				limit = { house = house:house_markesh }
				dynasty = { add_dynasty_modifier = markesh_victory_modifier }
			}
			else = {
				random_list = {
					50 = {
						show_chance = no
						desc = pod1230objectives_fae.160.a
						dynasty = { add_dynasty_modifier = brandell_victory_modifier }
					}
					50 = {
						show_chance = no
						desc = pod1230objectives_fae.160.b
						dynasty = { add_dynasty_modifier = markesh_victory_modifier }
					}
				}
				
			}
		}
		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:sidhe_house_dominance_beaumayn_objective_decision
		}
		trigger_event = pod1230objectives_fae.160
	}
	
	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}
