# Entry point for the loadout editor. The effect:
#   1. Detects a save-data version mismatch (the move dataset changed
#      since this save was last loaded) and rebuilds the global catalogue
#      via POD_reload_combat_moves so story references stay consistent.
#   2. Seeds the default loadout if the character has nothing equipped.
#   3. Flags the GUI window open.
#
# is_valid blocks the decision while combat is active
# (NOT = pod_combat_active) — players can't reconfigure mid-fight.
# A player who tries gets a silently disabled decision; the gate
# lives here rather than as a custom_tooltip-triggered failure
# message because mid-combat the loadout window itself is closed
# and a re-open attempt should just be a no-op rather than a popup.
pod_combat_configure_skills = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_personal_religious.dds"
	}
	ai_check_interval = 0
	desc = pod_combat_configure_skills_desc

	is_shown = {
		is_ai = no
		has_game_rule = pod_combat_enabled
	}

	is_valid = {
		NOT = { has_variable = pod_combat_active }
	}

	effect = {
		hidden_effect = {
			# Save-compat: if the script-side version constant doesn't match
			# the global variable set during init, rebuild every story and
			# re-seed starter loadouts. POD_reload_combat_moves handles the
			# heavy lifting (see pod_combat_move_init_effects.txt).
			if = {
				limit = {
					OR = {
						NOT = { exists = global_var:POD_combat_version }
						global_var:POD_combat_version != POD_current_combat_version
					}
				}
				POD_reload_combat_moves = yes
			}

			# Seed default loadout if empty (first time opening, new save,
			# or just after version reload which cleared the list).
			if = {
				limit = { pod_combat_no_moves_equipped_trigger = yes }
				POD_apply_default_combat_loadout = yes
			}

			set_variable = { name = pod_combat_config_open value = yes }

			# Bake every move's unlock onto pc_can_use for the cheap per-frame
			# editor-pool check Runs after the version-reload + default-seed
			# so it reflects the current catalogue. 
			pod_combat_bake_pc_catalogue_unlocks_effect = yes

			# Seed the loadout-count display cache so the pip strips read cheap
			# vars while the editor is open.
			pod_combat_refresh_loadout_counts_effect = yes
		}
	}
}
