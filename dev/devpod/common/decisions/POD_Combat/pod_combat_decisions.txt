# Entry point for the loadout editor. Brings the catalogue and the PC's
# loadout up to date, then sets the window-open flag the skills window's
# `visible` binding watches; the ordering constraints are commented in
# the effect body.
#
# is_valid blocks the decision while combat is active
# (NOT = pod_combat_active), so players can't reconfigure mid-fight.
# A silently disabled decision rather than a custom_tooltip-triggered
# failure message: mid-combat the loadout window itself is closed, so a
# re-open attempt should be a no-op rather than a popup.
pod_combat_configure_skills = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/ep3_decision_varangian_guard.dds"
	}
	decision_group_type = pod_splat
	sort_order = 1000
	
	ai_check_interval = 0
	
	desc = pod_combat_configure_skills_desc

	is_shown = {
		is_ai = no
		has_game_rule = pod_combat_enabled
	}

	is_valid_showing_failures_only = {
		NOT = { pod_combat_is_active_trigger = yes }
	}

	effect = {
		hidden_effect = {
			# Save-compat: if the script-side version constant doesn't match
			# the global variable set during init, rebuild every story and
			# re-seed starter loadouts. POD_reload_combat_moves handles the
			# heavy lifting (see pod_combat_move_init_effects.txt).
			if = {
				limit = {
					OR = {
						NOT = { exists = global_var:POD_combat_version }
						global_var:POD_combat_version != POD_current_combat_version
					}
					# Defer while any player is mid-fight (see the duel-create twin).
					NOT = { pod_combat_any_fight_active_trigger = yes }
				}
				POD_reload_combat_moves = yes
			}

			# Seed default loadout if empty (first time opening, new save,
			# or just after version reload which cleared the list).
			if = {
				limit = { pod_combat_no_moves_equipped_trigger = yes }
				POD_apply_default_combat_loadout = yes
			}

			set_variable = { name = pod_combat_config_open value = yes }

			# Build the PC's pod_combat_unlocked_moves list for the cheap per-frame
			# editor-pool check. Runs after the version-reload + default-seed
			# so it reflects the current catalogue.
			pod_combat_bake_pc_catalogue_unlocks_effect = yes

			# Seed the loadout-count display cache so the pip strips read cheap
			# vars while the editor is open.
			pod_combat_refresh_loadout_counts_effect = yes

			# The sheet scratch list was cleared at last close; if the pool
			# was left in sheet view, rebuild it now (after the unlock bake
			# above - the known-only filter tests pod_combat_unlocked_moves membership).
			if = {
				limit = { has_variable = pod_combat_sheet_view }
				pod_combat_sheet_rebuild_effect = yes
			}
		}
	}
}
