POD_torture_all_prisoners_decision = {
	picture = "gfx/interface/illustrations/decisions/decision_physician.dds"

	desc = POD_torture_all_prisoners_decision_desc

	is_shown = {
		any_prisoner = {
			NOT = { has_character_modifier = recently_tortured }
			count > 1
		}
		POD_is_not_trapped_in_umbra_trigger = yes
	}

	effect = {
		custom_tooltip = POD_torture_all_prisoners_decision_effect

		if = {
			limit = { mass_torture_piety_gain != 0 }
			custom_tooltip = POD_torture_all_prisoners_decision_effect_piety
		}

		if = {
			limit = { has_trait = sadistic }
			custom_tooltip = POD_torture_all_prisoners_decision_effect_stress_sadistic
		}
		if = {
			limit = { has_trait = zealous }
			custom_tooltip = POD_torture_all_prisoners_decision_effect_stress_zealous
		}
		if = {
			limit = {
				OR = {
					has_trait = compassionate
					has_trait = craven
					has_trait = just
					has_trait = forgiving
					has_trait = generous
					has_trait = trusting
					has_trait = content
				}
			}
			custom_tooltip = POD_torture_all_prisoners_decision_effect_stress_gain
		}

		hidden_effect = {
			save_scope_as = actor
			save_scope_value_as = { name = torture_method value = flag:whip }
			every_prisoner = {
				limit = {
					NOT = { has_character_modifier = recently_tortured }
					NOT = { has_character_flag = is_being_tortured }
				}
				save_scope_as = recipient

				torture_interaction_actor_effect = yes
				torturer_interaction_recipient_effect = yes
				house_feud_torture_event_effect = yes

				prev = {
					POD_secondary_effects_from_torturing = yes
					
					if = {
						limit = { has_perk = dark_insights_perk }
						dark_insights_skill_gain_roll_effect = yes
					}
				}
			}
		}
	}
	
	ai_check_interval = 0
}
