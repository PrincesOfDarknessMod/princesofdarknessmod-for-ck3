
POD_nushi_duty_punish_impious_decision = {
    picture = "gfx/interface/illustrations/decisions/decision_misc.dds"

    ai_check_interval = 0
	
    is_shown = {
		house ?= { has_nushi_modifier_trigger = yes }
		has_character_flag = duty_punish_impious_flag
	}

	is_valid = {
		custom_tooltip = {
			text = POD_nushi_duty_punish_impious_decision_condition
			has_character_flag = finish_punish_impious_flag
		}
	}

    effect = {
		remove_variable = impious
		remove_character_flag = duty_punish_impious_flag
		remove_character_flag = finish_punish_impious_flag
		
		POD_kj_advance_dharma_effect = yes
	}

	ai_will_do = {
		base = 0
	}
}
