witching_lands = {
	picture = {
		reference = "gfx/interface/illustrations/event_scenes/forestgraveyard.dds"
	}

	desc = witching_lands_desc
	decision_group_type = major

	ai_check_interval = 120

	is_shown = {
		is_landed = yes
		OR = {
		house = house:house_lhiannan
		faith = faith:pathoftheeightfoldwheel
		faith = faith:baobhansith
		}
		is_vampire_trigger = yes	
		NOT = {  #Can be redone if the Title is destroyed again.
			exists = title:e_lhiannan.holder
		}
	}

	is_valid = {	
		prestige >= 1000
		realm_size >= 50
		completely_controls = title:k_ireland	
		completely_controls = title:k_wales
	}

	is_valid_showing_failures_only = {
		is_capable_adult = yes
		is_imprisoned = no
		is_independent_ruler = yes
		is_at_war = no
	}

	effect = {
	custom_tooltip = witching_lands_effect_tooltip

	save_scope_as = scoped_ruler
		create_title_and_vassal_change = {
			type = created
			save_scope_as = change
			add_claim_on_loss = no
		}

	title:e_lhiannan = {
		change_title_holder = {
			holder = scope:scoped_ruler
			change = scope:change
		}
	}
	resolve_title_and_vassal_change = scope:change
	hidden_effect = {
		set_primary_title_to = title:e_lhiannan
	}
	}
	
	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}
