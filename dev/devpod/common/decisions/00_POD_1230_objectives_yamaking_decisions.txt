				domicile ?= { has_domicile_building_or_higher = yamaking_hells_of_shadows_06 }
				domicile ?= { has_domicile_building_or_higher = yamaking_hells_of_shadows_06_defiled }
				domicile ?= { has_domicile_building_or_higher = yamaking_hells_of_rage_06 }
			}
		}
		POD_combined_defiled_quality_equal_or_higher_trigger = { AMOUNT = yamaking_combined_defiled_quality_requirement }

			add_prestige_level = 1
		}

		add_to_variable_list = {
			name = unavailable_unique_decisions
