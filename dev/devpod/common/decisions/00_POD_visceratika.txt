visceratika_gargoyle_decision = {
	picture = "gfx/interface/illustrations/decisions/decision_the_will_of_tremere.dds"
	major = yes
	ai_check_interval = 120

    desc = visceratika_gargoyle_decision_desc

	is_shown = {
		has_trait = gargoyle
		NOT = { has_trait = visceratika }
	 	NOT = { has_trait = visceratikaadvanced }
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:visceratika_decision
			}
		}
	}

	is_valid = {
		is_independent_ruler = yes
		highest_held_title_tier >= tier_kingdom
	}
	
    effect = {
		every_living_character = {
			limit = { is_gargoyle_trigger = yes }
			add_trait = visceratika
		}
		dynasty = {
			add_dynasty_prestige = 2500
		}
		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:visceratika_decision
		}
    }
	
	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}
