POD_kueijin_become_enlightened_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_demon_redemption.dds" }

	decision_group_type = major

	ai_check_interval = 0

	is_shown = {
		has_trait = dharma
	}

	is_valid = {
		OR = {
			AND = {
				trigger_if = {
					limit = { has_global_variable = wan_xian_wan_kuei }
				}
				trigger_else_if = { limit = { religion = religion:boneflowers_religion }
					num_of_relation_bound_wraith >= 15
				}
				trigger_else_if = { limit = { religion = religion:deviltigers_religion }
					custom_tooltip = {
						text = POD_kueijin_become_enlightened_decision_deviltigers_religion
						has_variable = deviltigers_religion_torture_count
						var:deviltigers_religion_torture_count >= 150
					}
				}
				trigger_else_if = { limit = { religion = religion:resplendentcranes_religion }
					realm_size >= 150
				}
				trigger_else_if = { limit = { religion = religion:thousandwhispers_religion }
					custom_tooltip = {
						text = POD_kueijin_become_enlightened_decision_thousandwhispers_religion
						has_variable = thousandwhispers_religion_relation_lost_count
						var:thousandwhispers_religion_relation_lost_count >= 50
					}

				}
				trigger_else_if = { limit = { religion = religion:thrashingdragons_religion }
					custom_tooltip = {
						text = POD_kueijin_become_enlightened_decision_thrashingdragons_religion
						relation_num >= 15
					}

				}
				trigger_else_if = { limit = { faith = faith:flameoftherisingphoenix }
					custom_tooltip = {
						text = POD_kueijin_become_enlightened_decision_flameoftherisingphoenix
						has_variable = flameoftherisingphoenix_feast_count
						var:flameoftherisingphoenix_feast_count >= 25
					}

				}
				trigger_else_if = { limit = { faith = faith:faceofthegods }
					custom_tooltip = {
						text = POD_kueijin_become_enlightened_decision_faceofthegods
						has_variable = faceofthegods_win_wars_count
						var:faceofthegods_win_wars_count >= 25
					}

				}
				trigger_else_if = { limit = { faith = faith:inwardway }
					custom_tooltip = {
						text = POD_kueijin_become_enlightened_decision_inwardway
						has_variable = inwardway_meditate_count
						var:inwardway_meditate_count >= 25
					}

				}
				trigger_else_if = { limit = { faith = faith:wayofthetenthousandscreams }
					custom_tooltip = {
						text = POD_kueijin_become_enlightened_decision_wayofthetenthousandscreams
						POD_killed_character_count >= 500
					}

				}
				trigger_else = {
					learning >= 75
				}
			}
			learning >= 75
		}
	}

	is_valid_showing_failures_only = {
		has_trait = dharma9
		has_trait_xp = { trait = dharma9 value >= 100}
		is_akuma_trigger = no
	}

	effect = {
		POD_major_objective_finished_effect = yes
		remove_trait = dharma9
		add_trait_force_tooltip = dharma10
		#trigger_event = podkueijin.231
	}

	ai_potential = {
		always = no
	}
}

POD_yama_wealth_and_influence_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}
	decision_group_type = pod_splat
	ai_check_interval = 240

	is_shown = {
		has_perk = favour_of_the_yama_kings_perk
	}

	widget = {
		gui = "decision_view_widget_option_list_generic"
		controller = decision_option_list_controller
		decision_to_second_step_button = "NEXT_LABEL"
		show_from_start = yes

		item = {
			value = wealth_and_influence_multiple_points
			is_valid = {
				demon_arts_lifestyle_perk_points >= 2
			}
			current_description = wealth_and_influence_multiple_points_tooltip
			localization = wealth_and_influence_multiple_points_name
			icon = "gfx/interface/icons/character_interactions/POD_demon_interaction.dds"
			ai_chance = {
				value = 0
			}
		}
		item = {
			value = wealth_and_influence_single_point
			current_description = wealth_and_influence_single_point_tooltip
			localization = wealth_and_influence_single_point_name
			icon = "gfx/interface/icons/character_interactions/POD_demon_interaction.dds"
			is_default = yes
			ai_chance = {
				value = 100
			}
		}
	}

	is_valid = {
		demon_arts_lifestyle_perk_points >= 1
	}

	effect = {
		if = {
			limit = { scope:wealth_and_influence_single_point = yes }
			add_demon_arts_lifestyle_perk_points = -1
			add_gold = minor_gold_value
			add_prestige = medium_prestige_gain
		}
		else = {
			while = {
				count = demon_arts_lifestyle_perk_points
				add_demon_arts_lifestyle_perk_points = -1
				add_gold = minor_gold_value
				add_prestige = medium_prestige_gain
			}
		}
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}
