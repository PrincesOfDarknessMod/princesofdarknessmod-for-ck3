POD_possessed_maeljin_incarna_decision = {
	picture = {
		reference = "gfx/interface/illustrations/event_scenes/pod_wyrm_reaches.dds"
	}

	decision_group_type = major

	ai_check_interval = 0
	
	is_shown = {
		has_trait = fomori # TODO POSSESSED: other possessed
		POD_has_access_to_fomori_breed_perks = yes
	}

	is_valid = {
		trigger_if = {
			limit = { has_perk = possessed_enticer_perk }
			has_perk = possessed_urge_of_desire_perk
			save_temporary_scope_value_as = {
				name = fomori_lover_requirement
				value = 20
			}
			save_temporary_scope_value_as = {
				name = fomori_lover_count
				value = {
					every_relation = {
						type = lover
						limit = { has_trait = fomori }
						add = 1
					}
				}
			}
			custom_tooltip = {
				text = POD_enticer_ascension_requirement_fomori
				scope:fomori_lover_count >= scope:fomori_lover_requirement
			}
			save_temporary_scope_value_as = {
				name = king_lover_requirement
				value = 10
			}
			save_temporary_scope_value_as = {
				name = king_lover_count
				value = {
					every_relation = {
						type = lover
						limit = { highest_held_title_tier >= tier_kingdom }
						add = 1
					}
				}
			}
			custom_tooltip = {
				text = POD_enticer_ascension_requirement_king
				scope:fomori_lover_count >= scope:fomori_lover_requirement
			}
			save_temporary_scope_value_as = {
				name = war_requirement
				value = 3
			}
			custom_tooltip = {
				text = POD_enticer_ascension_requirement_wars
				POD_wars_won_without_raising_armies >= scope:war_requirement
			}
		}
		trigger_else = {
			OR = {
				has_perk = possessed_urge_of_desire_perk
				has_perk = possessed_urge_of_consumption_perk
				has_perk = possessed_urge_of_cruelty_perk
				has_perk = possessed_urge_of_violence_perk
				has_perk = possessed_urge_of_power_perk
			}
			custom_tooltip = {
				text = POD_possessed_maeljin_incarna_decision_requirement_base
				always = no
			}
		}
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_at_war = no
	}

	effect = {
		POD_major_objective_finished_effect = yes
		add_trait = immortal
		add_character_modifier = POD_maeljin_incarna_modifier
	}
	
	ai_potential = {
		always = no
	}

	ai_will_do = {
		base = 0
	}
}

		switch = {

				if = {

				}
			}
