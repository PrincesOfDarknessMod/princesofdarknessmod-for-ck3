POD_demon_unearthbound_decision = {
    picture = "gfx/interface/illustrations/decisions/decision_earthbound.dds"

    ai_check_interval = 0
	
	cost = {
		gold = {
			value = unearthbound_gold_cost
			# AI
			if = {
				limit = { is_ai = yes }
				multiply = 0
			}
		}
	}
	
    is_shown = {
		is_infernalist_trigger = no
		any_held_title = { has_variable = reliquarie }
	}

	is_valid = {
	}

    effect = {
		show_as_tooltip = {
			add_piety = massive_piety_gain
					
			random_held_title = {
				limit = { has_variable = reliquarie }
				
				every_de_jure_county = { 
					remove_county_modifier = earthbound_modifier 
					remove_county_modifier = earthbound_negative_modifier 
				}
				var:reliquarie = {
					remove_trait = earthbound
				}
			}
		}
		trigger_event = POD_demon_maintenance.30
	}

	ai_will_do = {
		base = 0
	}
}

POD_demon_earthbound_duchy_decision = {
    picture = "gfx/interface/illustrations/decisions/decision_earthbound.dds"

    ai_check_interval = 0
	
	major = yes
	
	cost = {
		gold = 1000
		piety = 1000
		prestige = 1000
	}
	
    is_shown = {
		is_ai = no
		has_variable = earth_bound
		var:earth_bound = { tier = tier_county }
	}

	is_valid = {
		completely_controls = var:earth_bound.duchy
	}

    effect = {
		var:earth_bound = {
			remove_variable = reliquarie
			duchy = { save_scope_as = target }
		}
			
		scope:target = {
			set_variable = { name = reliquarie value = root }
			every_de_jure_county = {
				if = {
					limit = { holder = { is_infernalist_trigger = yes } }
					add_county_modifier = earthbound_modifier
				}
				else = {
					add_county_modifier = earthbound_negative_modifier
				}
			}
		}
			
		set_variable = { name = earth_bound value = scope:target }
	}

	ai_will_do = {
		base = 0
	}
}
