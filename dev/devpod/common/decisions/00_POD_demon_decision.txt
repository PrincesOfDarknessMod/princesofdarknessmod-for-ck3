    picture = "gfx/interface/illustrations/decisions/decision_earthbound.dds"

    ai_check_interval = 0
	
	cost = {
		gold = {
			value = unearthbound_gold_cost
			if = {
				limit = {
					OR = {
						has_perk = nyarashahims_banishing_sign_perk
						has_perk = bind_the_demons_servant_perk
						has_perk = banishment_perk
					}
				}
				multiply = 0
			}
		}
	}
	
    is_shown = {
		is_infernalist_trigger = no
		any_held_title = { has_variable = reliquarie }
	}

	is_valid = {
	}

    effect = {
		show_as_tooltip = {
			add_piety = massive_piety_gain
					
			random_held_title = {
				limit = { has_variable = reliquarie }
				
				every_de_jure_county = { 
					remove_county_modifier = earthbound_modifier 
					remove_county_modifier = earthbound_negative_modifier 
				}
				var:reliquarie = {
					remove_trait = earthbound
				}
			}
		}
		trigger_event = POD_demon_maintenance.30
	}

	ai_will_do = {
		base = 0
	}
}

POD_demon_earthbound_duchy_decision = {
    picture = "gfx/interface/illustrations/decisions/decision_earthbound.dds"

    ai_check_interval = 0
	
	major = yes
	
	cost = {
		gold = 1000
		piety = 1000
		prestige = 1000
	}
	
    is_shown = {
		has_variable = earth_bound
		var:earth_bound = { tier = tier_county }
	}

	is_valid = {
		completely_controls = var:earth_bound.duchy
	}

    effect = {
		var:earth_bound = {
			remove_variable = reliquarie
			duchy = { save_scope_as = target }
		}
			
		scope:target = {
			set_variable = { name = reliquarie value = root }
			every_de_jure_county = {
				if = {
					limit = { holder = { is_infernalist_trigger = yes } }
					add_county_modifier = earthbound_modifier
				}
				else = {
					add_county_modifier = earthbound_negative_modifier
				}
				hidden_effect = { set_county_faith = root.faith }
			}
		}
			
		set_variable = { name = earth_bound value = scope:target }
	}

	ai_will_do = {
		base = 0
	}
}

		has_perk = twist_of_fate_perk
	}

		gui = "decision_view_widget_option_list_generic"

		show_from_start = yes

		item = {

			is_valid = {
				arcana_lifestyle_perk_points >= 2
			}
			current_description = wealth_and_influence_multiple_points_desc
			localization = wealth_and_influence_multiple_points_name
			ai_chance = {
				value = 0
			}
		}
		item = {
			value = single_point
			current_description = wealth_and_influence_single_point_desc
			localization = wealth_and_influence_single_point_name
			is_default = yes
			ai_chance = {
				value = 100
			}
		}
	}

	is_valid = {
		arcana_lifestyle_perk_points >= 1
	}

    effect = {
		if = {
			limit = { scope:multiple_points = yes }
			while = {
				count = arcana_lifestyle_perk_points
				add_arcana_lifestyle_perk_points = -1
				add_gold = minor_gold_value
				add_prestige = medium_prestige_gain
			}
		}
		else = {
			add_arcana_lifestyle_perk_points = -1
			add_gold = minor_gold_value
			add_prestige = medium_prestige_gain
		}
	}
	
	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}
