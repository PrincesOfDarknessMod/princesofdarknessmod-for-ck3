				}
				var:reliquarie = {

			}
		}
		trigger_event = POD_demon_maintenance.30
	}

	ai_will_do = {
		base = 0
	}
}

	}

	}

    effect = {

					limit = { holder = { POD_is_demonic_trigger = yes } }

				}
				else = {
					add_county_modifier = earthbound_negative_modifier
				}
				hidden_effect = { set_county_faith = root.faith }
			}
		}
			
		set_variable = { name = earth_bound value = scope:target }
	}

	ai_will_do = {
		base = 0
	}
}
