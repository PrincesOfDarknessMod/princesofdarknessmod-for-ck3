host_elysium_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}
	ai_check_interval = 0
	decision_group_type = pod_splat
	cost = { gold = standard_activity_cost }

	cooldown = { years = 10 }

	desc = host_elysium_decision_opening_desc
	selection_tooltip = host_elysium_decision_tooltip

	is_shown = {
		is_landed = yes
		is_vampire_trigger = yes
		highest_held_title_tier > tier_barony
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
		location = capital_province
		exists = cp:councillor_chancellor
	}

	effect = {

		#Stress loss
		if = {
			limit = {
				OR = {
					has_trait = shy
				}
			}
			custom_tooltip = host_elysium_decision_desc_shy
		}
		else_if = {
			limit = {
				OR = {
					has_trait = gregarious
					has_trait = lifestyle_reveler
				}
			}
			stress_impact = {
				base = feast_stress_loss_value
				gregarious = activity_stress_loss_impact
				lifestyle_reveler = activity_stress_loss_impact
			}
		}
		else = {
			stress_impact = {
				base = feast_stress_loss_value
			}
		}
		if = {
			limit = {
				faith = {
					has_doctrine_parameter = piety_from_feasts_active
					}
				}	
				add_piety = 50
			}
		if = {
			limit = { faith = { has_doctrine_parameter = reveler_from_elysium } }	
			if = {
				limit = { has_trait = lifestyle_reveler }
				add_trait_xp = {
					trait = lifestyle_reveler
					value = { 3 8 }
				}
			}
			else = { #progress towards reveler
				custom_tooltip = {
					text = progress_towards_reveler_tt
					reveler_points_gain_effect = {
						CHANGE = 10
						RANDOM_CHANGE = 10
					}
				}
			}
		}

		if = { limit = { is_target_in_global_variable_list = { name = unavailable_unique_decisions target = flag:malkavian_ecstasis_objective_decision } } 
			custom_tooltip = { 
				text = malkavian_pain_and_pleasure_modifier_powerup_tt
				if = { limit = { NOT = { has_variable = malkavian_pain_and_pleasure_scale } } set_variable = { name = malkavian_pain_and_pleasure_scale value = 2 } }
				else = { change_variable = { name = malkavian_pain_and_pleasure_scale add = 1 } }
				if = { limit = { var:malkavian_pain_and_pleasure_scale ?= 10 } trigger_event = pod1230objectives.1611 }
				if = { limit = { var:malkavian_pain_and_pleasure_scale ?= 20 } trigger_event = pod1230objectives.1612 }
				if = { limit = { var:malkavian_pain_and_pleasure_scale ?= 21 } set_variable = { name = malkavian_pain_and_pleasure_scale value = 20 } }
			}
		}
		custom_tooltip = elysium_decision_effect_tooltip
		trigger_event = POD_elysium_events.1000	

		add_legitimacy = minor_legitimacy_gain	
	}
	
	ai_potential = {
		short_term_gold > standard_activity_cost
		NOT = {
			has_trait = reclusive
		}
	}

	ai_will_do = {
		base = 0
	}
}
