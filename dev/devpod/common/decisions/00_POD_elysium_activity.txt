host_elysium_decision = {
	picture = "gfx/interface/illustrations/decisions/decision_misc.dds"
	ai_check_interval = 0

	cost = { gold = standard_activity_cost }

	cooldown = { years = 10 }

	desc = host_elysium_decision_opening_desc
	selection_tooltip = host_elysium_decision_tooltip

	is_shown = {
		is_landed = yes
		is_vampire_trigger = yes
		highest_held_title_tier > tier_barony
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
		location = capital_province
		has_council_position = councillor_chancellor
	}

	effect = {

		#Stress loss
		if = {
			limit = {
				OR = {
					has_trait = shy
				}
			}
			custom_tooltip = host_elysium_decision_desc_shy
		}
		else_if = {
			limit = {
				OR = {
					has_trait = gregarious
					has_trait = lifestyle_reveler
				}
			}
			stress_impact = {
				base = feast_stress_loss_value
				gregarious = activity_stress_loss_impact
				lifestyle_reveler = activity_stress_loss_impact
			}
		}
		else = {
			stress_impact = {
				base = feast_stress_loss_value
			}
		}
		if = {
			limit = {
				faith = {
					has_doctrine_parameter = piety_from_feasts_active
					}
				}	
				add_piety = 50
			}
		custom_tooltip = elysium_decision_effect_tooltip
		trigger_event = POD_elysium_events.1000			
	}
	
	ai_potential = {
		short_term_gold > standard_activity_cost
		NOT = {
			has_trait = reclusive
		}
	}

	ai_will_do = {
		base = 0
	}
}
