learning_kaja_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}


    ai_check_interval = 0
	decision_group_type = pod_splat
    is_shown = {
		NOT = { knows_language = language_kaja }
		OR = {
			has_POD_kj_blood_trigger = yes
			AND = {
				is_shih_trigger = yes
				has_global_variable = celestial_army_restored
			}
		}
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
		location = capital_province
	}

	cost = {
		prestige = {
			value = 200
			if = { 
				limit = {
					is_kueijin_trigger = no
				}
				multiply = 10
			}
			if = { 
				limit = {
					has_trait = dharma1
				}
				multiply = 5
			}
			if = { 
				limit = {
					has_trait = dharma2
				}
				multiply = 3
			}
			if = { 
				limit = {
					has_trait = dharma3
				}
				multiply = 2
			}
			if = { 
				limit = {
					has_trait = dharma4
				}
				multiply = 1.5
			}
		}
	}

    effect = {
		learn_language = language_kaja
		
		save_scope_as = player
		every_ruler = {
			limit = {
				knows_language = language_kaja
			}
			if = { 
				limit = {
					scope:player = { has_POD_kj_blood_trigger = no }
				}
				scope:player = {
					add_hook = {
						type = threat_hook
						target = prev
						years = 3
					}
				}
			}
			else = {
				add_opinion = {
					target = scope:player
					modifier = friendliness_opinion
					opinion = 20
				}
			}
		}
	}	

	ai_potential = {
		always = no
	}

	ai_will_do = {
		base = 0
	}
}

			num_of_relation_lover > 0

		}
	}

	}

						value = scope:duel_value

					}

					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
				}
			}
			if = {
				limit = { has_relation_soulmate = root }
				remove_relation_soulmate = root
			}
			else = { remove_relation_lover = root }
		}
		every_relation = {
			type = friend
			duel = {
				skill = intrigue
				target = root
				50 = {
					desc = board_games.0001.tt.failure
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier 
					}
					save_scope_as = victim
					execute_opinion_effect = { VICTIM = scope:victim EXECUTIONER = root }
					pod_death = {
						DEATH_REASON = death_sacrificed_to_gods
						KILLER = root
					}
				}
				50 = {
					desc = board_games.0001.tt.success
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
				}
			}
			if = {
				limit = { has_relation_best_friend = root }
				remove_relation_best_friend = root
			}
			else = { remove_relation_friend = root }
		}
		every_relation = {
			type = rival
			if = {
				limit = { has_relation_nemesis = root }
				remove_relation_nemesis = root
			}
			else = { remove_relation_rival = root }
		}

		trigger_event = POD_change_personality.100
	}

		always = no
	}

	ai_will_do = {
		base = 0
	}
}
