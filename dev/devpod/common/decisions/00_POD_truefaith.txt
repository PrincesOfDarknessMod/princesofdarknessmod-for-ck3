
conversion_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_dynasty_house.dds"
	}
	desc = conversion_decision_desc
	decision_group_type = pod_splat
	ai_check_interval = 0

	cooldown = { days = 30 }
	
	cost = {		
		piety = {
			value = realm_county_to_convert_value
			multiply = 50
		}
	}

	is_shown = {
		is_landed = yes
		OR = {
			has_perk = conversion_perk 
			has_trait = beatific
		}	
	}
		
	is_valid = {
		is_independent_ruler = yes
	}

	effect = {
		save_temporary_scope_as = inquisitor
		
		custom_tooltip = conversion_decision_effect_tooltip

		if = {
			limit = { has_trait = beatific }
			custom_tooltip = conversion_decision_effect_tooltip2
			if = {
				limit = { has_trait_xp = { trait = beatific value < 20 } }
				POD_advance_beatific_multiple_effect = { VALUE = root.mass_conversion_beatific_xp }
			}
		}
		
		if = {
			limit = { faith = faith:faceofthegods }
			POD_kj_advance_dharma_multiple_effect = { VALUE = realm_county_to_convert_value }
		}
		
		every_realm_county = { 
			set_county_faith = scope:inquisitor.faith
		}
	}

	ai_potential = {
		always = no
	}

	ai_will_do = {
		base = 0
	}
}

advance_beatific_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_dynasty_house.dds"
	}	
	decision_group_type = pod_splat

	ai_check_interval = 320
	
	cooldown = { years = 5 }

	is_shown = {
		is_landed = yes
		has_trait = beatific
		has_trait_xp = { trait = beatific value < 100 }
	}
	
	is_valid = {
		trigger_if = {
			limit = { has_trait_xp = { trait = beatific value < 20 } }
			has_trait_xp = { trait = beatific value >= 19 }
			piety_level >= 1
		}
		trigger_else_if = {
			limit = { has_trait_xp = { trait = beatific value < 40 } }
			has_trait_xp = { trait = beatific value >= 39 }
			piety_level >= 2
		}
		trigger_else_if = {
			limit = { has_trait_xp = { trait = beatific value < 60 } }
			has_trait_xp = { trait = beatific value >= 59 }
			piety_level >= 3
		}
		trigger_else_if = {
			limit = { has_trait_xp = { trait = beatific value < 80 } }
			has_trait_xp = { trait = beatific value >= 79 }
			piety_level >= 4
		}
		trigger_else = {
			has_trait_xp = { trait = beatific value >= 99 }
			piety_level >= 5
		}
	}
	
	effect = {
		add_trait_xp = { trait = beatific value = 1 }
		add_piety = minor_piety_gain
	}
	
	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}
