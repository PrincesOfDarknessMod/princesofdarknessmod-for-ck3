POD_sorcery_become_immortal_decision = {
    picture = "gfx/interface/illustrations/decisions/decision_misc.dds"

    ai_check_interval = 0

	cost = {
		gold = { value = omega_gold_value }
	}
	
    is_shown = {
		has_trait = sorcerer
		has_perk = alchemy_perk
		has_supernatural_race_trigger = no 
		is_immortal = no
	}
	
	is_valid = {
		custom_description = {
			text = POD_sorcery_become_immortal_decision_custom	
			any_prisoner = {
				count >= 5
				has_trait = mummy
			}
		}
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_at_war = no
	}

    effect = {
		add_trait = immortal
	}

	ai_will_do = {
		base = 0
	}
}

POD_sorcery_become_vampire_decision = {
    picture = "gfx/interface/illustrations/decisions/decision_misc.dds"

    ai_check_interval = 0

	cost = {
		gold = { value = medium_gold_value }
	}
	
    is_shown = {
		has_trait = sorcerer
		has_perk = alchemy_perk
		OR = {
			has_supernatural_race_trigger = no
			POD_is_any_kind_of_ghoul_trigger = yes
			has_trait = changeling
		}
	}
	
	is_valid = {
		custom_description = {
			text = POD_sorcery_become_vampire_decision_custom	
			any_prisoner = {
				count >= 5
				has_trait = vampire
			}
		}
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_at_war = no
	}

    effect = {
		save_scope_as = vampiresire
		POD_embrace_remove_incompatible_trait_effect = yes
		
		add_trait_force_tooltip = vampire
		add_trait = caitiff
		add_trait = vampgeneration12
		
		set_character_faith = faith:viahumanitatis
		change_government = free_state_government
		add_character_modifier = bloodpotency1_modifier
		add_character_modifier = bloodhunger0_modifier
		add_character_modifier = masquerade0_modifier
			
		hidden_effect = {
			trigger_event = POD_chargen.1020 
				
			add_character_flag = immune_to_disease
		}
	}

	ai_will_do = {
		base = 0
	}
}

	cost = {
		gold = { value = medium_gold_value }
	}
	
    is_shown = {

		}

			limit = { has_trait = kinfolk }
			remove_trait = kinfolk
		}
		if = {
			limit = { has_trait = ghoul }
			remove_trait = ghoul
		}
		if = {
			limit = { has_trait = revenant }
			remove_trait = revenant
		}
		if = {
			limit = { has_trait = gnosis }
			remove_trait = gnosis
		}
			
		add_trait_force_tooltip = werewolf
		POD_remove_wrong_supernatural_traits_effect = yes
		POD_remove_wrong_supernatural_modifiers_effect = yes
			
		hidden_effect = {
			trigger_event = POD_chargen.2600
		}
	}

	ai_will_do = {
		base = 0
	}
}
