POD_camarilla_ventrue_membership_decision = {
	picture = "gfx/interface/illustrations/decisions/decision_destiny_goal.dds"
	major = yes

	ai_check_interval = 0
	
    is_shown = {
		religion = religion:camarilla_religion
		NOT = { POD_is_member_clan_trigger = { CLAN = ventrue } }
		is_ventrue_trigger = no
	}

	is_valid = {
		custom_tooltip = {
			text = POD_camarilla_ventrue_membership_decision_custom
			any_independent_ruler = {
				count < 1
				is_ventrue_trigger = yes	
				is_landed = yes
				NOT = { religion = religion:camarilla_religion }
			}
		}
		custom_tooltip = {
			text = POD_camarilla_ventrue_membership_decision_custom2
			any_independent_ruler = {
				count >= 1
				is_ventrue_trigger = yes	
				is_landed = yes
				religion = religion:camarilla_religion
			}
		}
		trigger_if = {
			limit = { POD_is_inner_circle_trigger = no }
			POD_is_justicar_trigger = yes
		}
	}
	
	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_at_war = no
		faith:camahumanity = { 
			NOT = { has_doctrine = doctrine_ventrue_shunned }
			NOT = { has_doctrine = doctrine_ventrue_crime }
		}
	}

    effect = {
		custom_tooltip = POD_camarilla_ventrue_membership_decision_effect_tooltip
		trigger_event = POD_camarilla.2001
	}
}
