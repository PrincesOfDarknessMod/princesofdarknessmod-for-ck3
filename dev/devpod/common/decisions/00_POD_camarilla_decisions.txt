POD_camarilla_ventrue_membership_request_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_destiny_goal.dds"
	}
	decision_group_type = major

	ai_check_interval = 0
	
    is_shown = {
		POD_reached_endgame_date_trigger = { YEAR = 1504 }
		religion = religion:camarilla_religion
		NOT = { faith:camahumanity = { POD_is_member_clan_trigger = { CLAN = ventrue } } }
		is_ventrue_trigger = yes
	}

	is_valid = {
		trigger_if = {
			limit = { 
				any_independent_ruler = {
					is_ventrue_trigger = yes	
					is_landed = yes
					NOT = { religion = religion:camarilla_religion }
					save_temporary_scope_as = example
				}
			}
			custom_tooltip = {
				text = POD_camarilla_ventrue_membership_decision_custom_ex
				always = no
			}
		}
		trigger_else = {
			custom_tooltip = {
				text = POD_camarilla_ventrue_membership_decision_custom
			}
		}
	}
	
	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_at_war = no
	}

    effect = {
		custom_tooltip = POD_camarilla_ventrue_membership_decision_effect_tooltip
		religion:sabbat_religion = { remove_list_variable = { name = member_clan target = doctrine:doctrine_ventrue_accepted } }
		faith:camahumanity = { add_to_variable_list = { name = member_clan target = doctrine:doctrine_ventrue_accepted } }
		faith:camahumanity = {
			if= { # Only happen if multiplayer go mean
				limit = { has_doctrine = doctrine_ventrue_shunned }
				remove_doctrine = doctrine_ventrue_shunned
			}
			if= { # Only happen if multiplayer go mean
				limit = { has_doctrine = doctrine_ventrue_crime }
				remove_doctrine = doctrine_ventrue_crime
			}
		}
		trigger_event = POD_camarilla.2101
	}
}
