become_massregnant_decision = {
	picture = "gfx/interface/illustrations/decisions/decision_bloodsurge.dds"

	ai_check_interval = 480
	
	is_shown = {
		NOR = {
			has_trait = massregnant
			any_secret = { secret_type = secret_massregnant }
		}
		is_ai = yes #Clutter decision for players, should perhaps he consolidated in one
		has_trait = vampire
	}
	
	is_valid = {
		is_available_adult = yes
		is_landed = yes
		NOT = { has_character_modifier = bloodhunger5_modifier }	
	}	
	
	desc = become_massregnant_decision_desc

	effect = {
		POD_hunger_rouse_check_effect = yes
		POD_very_high_risk_of_increasing_masquerade_exposure = yes
		POD_piety_loss_from_massghouling = yes
		POD_give_massregnant_secret_effect = yes
		custom_tooltip = become_massregnant_decision_tooltip
	}

	# cost = { gold = 300 } 

	ai_will_do = {
		base = 0
		
		modifier = {
			add = 2
			piety > 300
		}
		modifier = {
			add = 10
			has_trait = tzimisce
		}
		modifier = {
			add = 5
			has_trait = vicissitudeadvanced
		}
		modifier = {
			add = 5
			AND = {
				exists = dynasty
				dynasty = { has_dynasty_perk = tzimisce_legacy_1 }
			}
		}
		modifier = {
			add = 5
			AND = {
				exists = dynasty
				dynasty = { has_dynasty_perk = tzimisce_legacy_3 }
			}
		}
		modifier = {
			add = 5
			AND = {
				exists = dynasty
				dynasty = { has_dynasty_perk = ashen_knights_legacy_3 }
			}
		}
		
		modifier = {
			factor = 5
			is_at_war = yes
		}
		modifier = {
			factor = 2
			has_trait = paranoid
			has_trait = craven
			has_trait = ambitious
		}
		
		modifier = {
			factor = 0.5
			OR = {
				has_trait = lazy
				has_trait = calm
				has_trait = arrogant
				has_trait = content
				has_trait = trusting
				has_trait = compassionate
			}
		}
	}
}

unlock_war_avatar_of_great_fenris_decision = {
	picture = "gfx/interface/illustrations/decisions/decision_destiny_goal.dds"

	ai_check_interval = 480
	
	is_shown = {
		POD_splat_has_gnosis_trigger = yes
		has_perk = call_great_fenris_perk
		NOR = { 
			has_character_flag = POD_permanent_maimed
			has_character_flag = POD_permanent_one_legged
		}
	}
	
	is_valid = {
		is_available_adult = yes
		is_landed = yes
		OR = {
			NOT = { has_trait = maimed }
			NOT = { has_trait = one_legged }
		}
	}

	effect = {
		POD_gnosis_loss_check_effect = yes
		if = {
			limit = { NOT = { has_trait = maimed } }
			add_trait = maimed
			add_character_flag = POD_permanent_maimed
		}
		else = {
			add_trait = one_legged
			add_character_flag = POD_permanent_one_legged
		}
		custom_tooltip = unlock_war_avatar_of_great_fenris_decision_tooltip
	}

	ai_will_do = {
		base = 0
		
		modifier = {
			add = 2
			piety > 1000
		}
		modifier = {
			factor = 5
			is_at_war = yes
		}
		modifier = {
			factor = 2
			has_trait = paranoid
			has_trait = craven
			has_trait = ambitious
		}
		
		modifier = {
			factor = 0.5
			OR = {
				has_trait = lazy
				has_trait = calm
				has_trait = arrogant
				has_trait = content
				has_trait = trusting
				has_trait = compassionate
			}
		}
	}
}
