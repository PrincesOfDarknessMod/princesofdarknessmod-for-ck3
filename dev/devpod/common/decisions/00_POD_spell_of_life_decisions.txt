# Needs a separate file from mummy decisions because these aren't objectives limited to a single splat, even in ST mode
spell_of_life_starter_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_sargon_codex.dds"
	}
	desc = spell_of_life_starter_decision_desc
	selection_tooltip = spell_of_life_starter_decision_tooltip
	decision_group_type = pod_splat

	ai_check_interval = 120

	is_shown = {
		NOR = {
			this = character:792600  # Cabirus
			this = character:792593  # Horus
			this = character:790512  # Kemintiri
		}
		OR = {
			NOT = { has_trait = eternal_rebirth_seeker }
			has_trait_xp = {
				trait = eternal_rebirth_seeker
				track = spell_of_life
				value = 0
			}
		}
		OR = {
			# Western mummy or Osirian League
			is_amenti_trigger = yes
			religion = religion:amenti_religion
			faith = {
				has_doctrine = special_doctrine_osirian_league
			}
			faith = {
				has_doctrine = special_doctrine_against_to_apophis_hostility
			}
			# Servant of Set
			is_setite_trigger = yes
			POD_is_aligned_with_set = yes
			# Hunter that completed mummy-related objective
			AND = {
				has_trait = supehunter
				faith = faith:judge
				is_target_in_global_variable_list = {
					name = unavailable_unique_decisions
					target = flag:judge_research_origins_mummies_objective_decision
				}
			}
			AND = {
				has_trait = supehunter
				faith = faith:ikhwan_al_safa
				is_target_in_global_variable_list = {
					name = unavailable_unique_decisions
					target = flag:ikhwan_al_safa_egypt_objective_decision
				}
			}
			# Independent hunter that is okay with magic & made necessary connections
			AND = {
				has_trait = supehunter
				faith = faith:independenthunters
				OR = {
					has_trait = sorcerer
					has_trait = theurgist
				}
				OR = {
					is_target_in_global_variable_list = {
						name = unavailable_unique_decisions
						target = flag:judges_landless_objective_decision
					}
					is_target_in_global_variable_list = {
						name = unavailable_unique_decisions
						target = flag:ikhwan_al_safa_landless_objective_decision
					}
				}
			}

			trigger_if = { 
				limit = { POD_splat_mixing_haight_mode = yes } 

				OR = {
					# Tremere or TMR
					is_tremere_trigger = yes
					religion = religion:talmahera_religion
					# Very scholarly/magical and in the region
					POD_is_scholarly = yes
					AND = {
						OR = {
							faith = { has_doctrine = tenet_pursuit_of_knowledge }
							faith = { has_doctrine = tenet_knowledge_above_all }
							has_trait_rank = {
								trait = education_learning
								rank >= 4
							}
							has_trait = sorcerer
							has_trait = theurgist
						}
						location = { geographical_region = pod_lands_of_faith }
					}
				}
			}
		}
	}

	is_valid = {
		OR = {
			learning >= 15
			has_trait = scholar
			has_trait = sorcerer
			has_trait = theurgist
		}
		any_courtier = {
			count >= 3
			OR = {
				learning >= 15
				has_trait = scholar
				has_trait = sorcerer
				has_trait = theurgist
			}
		}
		OR = {
			trigger_if = {
				limit = {
					# Can't use is_amenti_trigger as not all mummies (like Neferkheperu) have a subsplat
					has_trait = mummy
					NOR = {
						has_trait = wutian
						has_trait = wukuei
					}
				}
				AND = {
					has_perk = barring_the_gates_perk
					has_perk = aspect_of_duat_perk
					has_perk = promise_made_before_ammuts_scales_perk
					has_perk = heart_of_reincarnation_perk
				}
			}
			trigger_if = {
				limit = {
					POD_splat_is_fera_trigger = yes
				}
				AND = {
					custom_tooltip = {
						text = spell_of_life_stage1_hide_requirement
						any_character_trait = {
							has_trait_flag = fera_type
							save_temporary_scope_as = fera_trait
						}
					}
					has_trait_xp = { trait = scope:fera_trait value >= 40 }
					calc_true_if = {
						amount >= 3
						has_trait_xp = { trait = scope:fera_trait value >= 60 }
						has_trait_xp = { trait = scope:fera_trait value >= 80 }
						has_perk = age_of_sleep_perk
						has_perk = earth_heal_perk
						has_perk = moondream_perk
						has_perk = cleansing_flame_perk
						has_perk = madthought_perk
						has_perk = planetarygifts_ebon_binding_perk
						has_perk = bring_forth_the_future_perk
						has_perk = heart_of_the_ice_warrior_perk
						has_perk = pack_mind_perk
						has_perk = grand_gesture_perk
						has_perk = burden_of_knowledge_perk
						has_perk = diplomatic_immunity_perk
						has_perk = drown_perk
						has_perk = essential_transformation_perk
					}
				}
			}
			trigger_if = {
				limit = {
					OR = {
						is_vampire_trigger = yes
						has_trait = revenant
						has_trait = ghoul
						has_trait = dhampir
						has_character_flag = vamp_discipline_access
						has_trait = bloodsorcerydiscipline
					}
				}
				AND = {
					has_trait = bloodsorceryadvanced
					OR = {
						AND = {
							has_perk = the_sorcery_of_life_perk
							has_perk = hand_of_heka_perk
							has_perk = dark_invocation_perk
						}
						AND = {
							has_perk = mutate_perk
							has_perk = power_over_life_perk
							has_perk = duality_perk
						}
					}
				}
			}
			trigger_if = {
				limit = {
					has_POD_fae_blood_trigger = yes
				}
				AND = {
					has_trait = naming
					has_trait = realm_fae
					has_trait = realm_time
				}
			}
			trigger_if = {
				limit = {
					OR = {
						POD_splat_is_valid_for_sorcery = yes
						can_learn_sorcery_trigger = yes
						can_learn_theurgy_trigger = yes
					}
				}
				calc_true_if = {
					amount >= 3
					has_perk = numina_alchemy_5_perk
					has_perk = numina_ephemera_5_perk
					has_perk = numina_necromancy_5_perk
					has_perk = numina_oneiromancy_5_perk
					has_perk = numina_quintessence_manipulation_5_perk
					has_perk = numina_via_geniorum_5_perk
					has_perk = numina_via_necromantiae_5_perk
					has_perk = numina_via_medicamenti_5_perk
				}
			}
		}
	}
	
	is_valid_showing_failures_only = {
		is_capable_adult = yes
		is_imprisoned = no
		is_at_war = no
	}

	effect = {
		POD_add_spell_of_life_xp = {
			VALUE = 1
		}
	}
	
	ai_potential = {
		always = yes
	}
	ai_will_do = {
		base = 100
	}
}

spell_of_life_stage1_horus_phase2_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_a_new_era.dds"
	}
	desc = spell_of_life_stage1_horus_phase2_decision_desc
	selection_tooltip = spell_of_life_stage1_horus_phase2_decision_tooltip
	decision_group_type = pod_quest
	ai_check_interval = 120

	cooldown = { days = 1825 }
	
	is_shown = {
		has_character_flag = spell_of_life_stage1_horus_phase1_decision
		has_trait_xp = {
			trait = eternal_rebirth_seeker
			track = spell_of_life
			value >= 1
		}
		NOR = {
			has_character_flag = spell_of_life_stage1_horus_phase2_decision
			has_trait_xp = {
				trait = eternal_rebirth_seeker
				track = spell_of_life
				value >= 30
			}
		}
	}

	is_valid = {
		POD_character_supports_you_for_objectives = { CHARACTER = character:792593 }
	}

	is_valid_showing_failures_only = {
		is_capable_adult = yes
		is_imprisoned = no
		is_at_war = no
	}

	effect = {
		show_as_tooltip = {
			POD_add_spell_of_life_xp = {
				VALUE = 10
			}
		}
		custom_tooltip = {
			text = spell_of_life_stage1_unlock_next_phase_caveat
		}
		hidden_effect = {
			if = {
				limit = { is_ai = no}
				add_to_variable_list = {
					name = opponents_list
					target = character:792593
				}

				POD_start_debate_effect = {
					CHARACTER = this
					OPPONENTS = opponents_list
					STARTER_EVENT = POD_mummy_spell_of_life.10
				}
			}
			else = {
				duel = {
					skill = diplomacy
					value = high_skill_rating
					50 = { # You convince horus
						compare_modifier = {
							value = scope:duel_value
							multiplier = 0.5
						}
						add_character_flag = spell_of_life_stage1_horus_phase2_decision
						POD_add_spell_of_life_xp = {
							VALUE = 10
						}
					}
					50 = { # You fail
						compare_modifier = {
							value = scope:duel_value
							multiplier = -0.5
						}
					}
				}
			}
		}
	}

	ai_potential = {
		always = yes
	}
	ai_will_do = {
		base = 100
		modifier = {
			add = 10000
			has_character_flag = spell_of_life_stage1_horus_phase1_decision
		}
	}
}
