#AUH killed a lot of our govs maintenance scripts. Here is a tentative decision to make sure that the AI and players will have their appropriate one instead 

POD_convert_government_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_eternal_knowledge.dds" }

	decision_group_type = major
	
	sort_order = 1000

	ai_check_interval = 6

	is_shown = {
		exists = government_type
		NOT = {
			POD_government_is_valid_for_character = {
				GOVERNMENT = this.government_type
			}
		}		
	}

	is_valid = {
		always = yes
	}

	effect = {
	if = {
		limit = {
			government_type = government_type:landless_adventurer_government
		}
		# do nothing for adventurers
	}
	else_if = {
		limit = {
			POD_splat_is_valid_for_fera_tribe_government = yes
			NOT = { government_type = government_type:fera_tribe_government }
		}
		change_government = fera_tribe_government
		POD_restore_saved_haven_buildings = yes
	}
	else_if = {
		limit = {
			POD_splat_is_valid_for_infernal_court_government = yes
			NOT = { government_type = government_type:infernal_court_government }
		}
		change_government = infernal_court_government
		POD_restore_saved_haven_buildings = yes
	}
	else_if = {
		limit = {
			POD_splat_is_valid_for_cult_government = yes
			NOT = { government_type = government_type:cult_government }
		}
		change_government = cult_government
		POD_restore_saved_haven_buildings = yes
	}
	else_if = {
		limit = {
			POD_splat_is_valid_for_inquisition_government = yes
			NOT = { government_type = government_type:inquisition_government }
		}
		change_government = inquisition_government
		POD_restore_saved_haven_buildings = yes
	}
	else_if = {
		limit = {
			POD_splat_is_valid_for_cell_government = yes
			NOT = { government_type = government_type:cell_government }
		}
		change_government = cell_government
		POD_restore_saved_haven_buildings = yes
	}
	else_if = {
		limit = {
			POD_splat_is_valid_for_traditions_government = yes
			NOT = { government_type = government_type:traditions_government }
		}
		change_government = traditions_government
		POD_restore_saved_haven_buildings = yes
	}
	else_if = {
		limit = {
			POD_splat_is_valid_for_august_court_government = yes
			NOT = { government_type = government_type:august_court_government }
		}
		change_government = august_court_government
		POD_restore_saved_haven_buildings = yes
	}
	else_if = {
		limit = {
			POD_splat_is_valid_for_household_government = yes
			NOT = { government_type = government_type:household_government }
		}
		change_government = household_government
		POD_restore_saved_haven_buildings = yes
	}
	else_if = {
		limit = {
			POD_splat_is_valid_for_dark_kingdom_government = yes
			NOT = { government_type = government_type:dark_kingdom_government }
		}
		change_government = dark_kingdom_government
		POD_restore_saved_haven_buildings = yes
	}		
	}

	ai_potential = {
		always = yes
	}
	
	ai_will_do = {
		base = 100
	}
}
