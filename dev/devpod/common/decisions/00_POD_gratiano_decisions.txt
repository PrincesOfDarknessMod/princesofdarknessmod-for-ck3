######################################
# RIVAL PURSUIT
# Only one rival chain holds focus. Reappearing map rivals do not reclaim it.
######################################

POD_gratiano_pursue_giovanni_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_POD_giovanni.dds"
	}

	decision_group_type = pod_splat

	desc = POD_gratiano_pursue_giovanni_decision_desc

	is_shown = {
		POD_gratiano_lever_actor_trigger = yes
		NOT = { has_character_flag = POD_gratiano_giovanni_cleared }
		NOT = { has_character_flag = POD_gratiano_giovanni_arc_intro }
	}

	is_valid_showing_failures_only = {
		custom_description = {
			text = POD_gratiano_rival_focus_busy_desc
			POD_gratiano_rival_focus_free_trigger = yes
		}
	}

	effect = {
		custom_tooltip = POD_gratiano_pursue_giovanni_decision_effect_tt
		hidden_effect = {
			# Handle an already-cleared sphere without taking focus: its completion would never release the slot.
			if = {
				limit = { POD_gratiano_giovanni_cleared_trigger = yes }
				add_character_flag = POD_gratiano_giovanni_arc_intro
				add_character_flag = POD_gratiano_giovanni_cleared
				trigger_event = POD_gratiano.0073
			}
			else = {
				add_character_flag = POD_gratiano_rival_in_focus
				add_character_flag = POD_gratiano_giovanni_holding_slot
				add_character_flag = POD_gratiano_giovanni_arc_intro
				trigger_event = { id = POD_gratiano.0060 days = 1 }
			}
		}
		send_interface_toast = {
			type = event_generic_neutral
			title = POD_gratiano_pursue_giovanni_launch_toast
			left_icon = this
		}
	}

	ai_check_interval = 0
	ai_potential = { always = no }
	ai_will_do = { base = 0 }
}

		custom_tooltip = POD_gratiano_pursue_moncada_decision_effect_tt
		hidden_effect = {
			add_character_flag = POD_gratiano_rival_in_focus

				limit = { character:505154 = { is_alive = yes } }

				trigger_event = { id = POD_gratiano.0128 days = 365 }
			}
			else = {
				add_character_flag = POD_gratiano_moncada_arc_started
				if = {
					limit = {
						character:790178 = {
							is_alive = yes
							NOT = { has_character_modifier = caged_lucita_modifier }
						}
					}
					character:790178 = { add_character_modifier = caged_lucita_modifier }
				}

		}

######################################
# THE STRIKE GATE
# "Strike the Castle of Shadows"
######################################

POD_gratiano_strike_the_castle = {

	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_faith_in_shadows.dds"
	}

	decision_group_type = major

	ai_check_interval = 0

	desc = POD_gratiano_strike_the_castle_desc

	is_shown = {
		this = character:505144
		is_ai = no
		has_character_flag = POD_gratiano_started
		any_owned_story = {
			story_type = POD_gratiano_story
			var:phase >= 3
		}
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:POD_gratiano_strike_the_castle
			}
		}
		NOT = { has_global_variable = variable_POD_lasombra_dead }
	}

	is_valid = {
		POD_gratiano_strike_gate_trigger = yes
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_at_war = no
	}

	effect = {
		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:POD_gratiano_strike_the_castle
		}
		if = {
			limit = {
				NOT = {
					AND = {
						has_variable = POD_gratiano_conspiracy
						var:POD_gratiano_conspiracy >= 90
					}
				}
			}
			custom_tooltip = POD_gratiano_strike_knife_edge_warning_tt
		}
		custom_tooltip = POD_gratiano_strike_launch_tt
		trigger_event = POD_gratiano.0220
	}

	ai_potential = {
		always = no
	}
	ai_will_do = {
		base = 0
	}
}
