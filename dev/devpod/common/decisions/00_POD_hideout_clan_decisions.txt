# GANGREL — Rally the Beast Den Pack
# Recruits controlled animal MaA regiment
######################################################
POD_gangrel_recruit_beast_den_animals = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_recruitment.dds"
	}

	decision_group_type = pod_splat
	desc = POD_gangrel_recruit_beast_den_animals_desc
	selection_tooltip = POD_gangrel_recruit_beast_den_animals_tooltip

	ai_check_interval = 120

	is_shown = {
		is_playable_character = yes
		POD_is_vampire_trigger = yes
		is_gangrel_clan_or_bloodline_trigger = yes
		domicile ?= {
			has_domicile_building_or_higher = vampire_gangrel_beast_den_01
		}
	}

	cooldown = { years = 5 }

	is_valid_showing_failures_only = {
		is_available = yes
		is_at_war = no
	}

	cost = {
		gold = {
			value = 50
			if = {
				limit = {
					domicile ?= { has_domicile_building_or_higher = vampire_gangrel_beast_den_02 }
				}
				add = 50
			}
			if = {
				limit = {
					domicile ?= { has_domicile_building_or_higher = vampire_gangrel_beast_den_04 }
				}
				add = 50
			}
			if = {
				limit = {
					domicile ?= { has_domicile_building_or_higher = vampire_gangrel_beast_den_06 }
				}
				add = 50
			}
		}
	}

	effect = {
		custom_tooltip = POD_gangrel_recruit_beast_den_animals_effect_tooltip
		hidden_effect = {
			if = {
				limit = {
					domicile ?= { has_domicile_building_or_higher = vampire_gangrel_beast_den_06 }
				}
				spawn_army = {
					uses_supply = no
					inheritable = no
					men_at_arms = {
						type = POD_controlled_animals
						stacks = 4
					}
					location = root.capital_province
					name = gangrel_beast_den_pack
				}
			}
			else_if = {
				limit = {
					domicile ?= { has_domicile_building_or_higher = vampire_gangrel_beast_den_04 }
				}
				spawn_army = {
					uses_supply = no
					inheritable = no
					men_at_arms = {
						type = POD_controlled_animals
						stacks = 3
					}
					location = root.capital_province
					name = gangrel_beast_den_pack
				}
			}
			else_if = {
				limit = {
					domicile ?= { has_domicile_building_or_higher = vampire_gangrel_beast_den_02 }
				}
				spawn_army = {
					uses_supply = no
					inheritable = no
					men_at_arms = {
						type = POD_controlled_animals
						stacks = 2
					}
					location = root.capital_province
					name = gangrel_beast_den_pack
				}
			}
			else = {
				spawn_army = {
					uses_supply = no
					inheritable = no
					men_at_arms = {
						type = wolves
						stacks = 1
					}
					location = root.capital_province
					name = gangrel_beast_den_pack
				}
			}
		}
	}

	ai_potential = {
		always = yes
