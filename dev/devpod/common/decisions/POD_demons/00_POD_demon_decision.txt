POD_infernalist_wealth_and_influence_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}


	decision_group_type = pod_splat
	ai_check_interval = 240

	is_shown = {
		OR = {
			has_perk = backgrounds_perk
			has_perk = luck_of_yomi_perk
		}
	}

	widget = {
		gui = "decision_view_widget_option_list_generic"
		controller = decision_option_list_controller
		decision_to_second_step_button = "NEXT_LABEL"
		show_from_start = yes

		item = {
			value = wealth_and_influence_multiple_points
			is_valid = {
				investiture_lifestyle_perk_points >= 2
			}
			current_description = wealth_and_influence_multiple_points_tooltip
			localization = wealth_and_influence_multiple_points_name
			icon = "gfx/interface/icons/character_interactions/POD_infernalist_interaction.dds"
			ai_chance = {
				value = 0
			}
		}
		item = {
			value = wealth_and_influence_single_point
			current_description = wealth_and_influence_single_point_tooltip
			localization = wealth_and_influence_single_point_name
			icon = "gfx/interface/icons/character_interactions/POD_infernalist_interaction.dds"
			is_default = yes
			ai_chance = {
				value = 100
			}
		}
	}

	is_valid = {
		investiture_lifestyle_perk_points >= 1
	}

	effect = {
		if = {
			limit = { scope:wealth_and_influence_single_point = yes }
			add_investiture_lifestyle_perk_points = -1
			if = {
				limit = {
					has_perk = backgrounds_perk
					has_perk = luck_of_yomi_perk
				}
				add_gold = medium_gold_value
				add_prestige = medium_prestige_gain
			}
			else = {
				add_gold = minor_gold_value
				add_prestige = medium_prestige_gain
			}
		}
		else = {
			while = {
				count = investiture_lifestyle_perk_points
				add_investiture_lifestyle_perk_points = -1
				if = {
					limit = {
						has_perk = backgrounds_perk
						has_perk = luck_of_yomi_perk
					}
					add_gold = medium_gold_value
					add_prestige = medium_prestige_gain
				}
				else = {
					add_gold = minor_gold_value
					add_prestige = medium_prestige_gain
				}
			}
		}
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}
