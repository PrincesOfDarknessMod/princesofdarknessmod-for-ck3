# Start an event where to decide to confirm or not

# holds territory in the west of Iberia

hunters_st_james_compostela_decision = {
	picture = {
		reference = "gfx/interface/illustrations/event_scenes/pam_simple_church_western_interior.dds"
	}
	
	decision_group_type = pod_splat
	ai_check_interval = 0
	
	is_shown = {
		OR = {
			dynasty = dynasty:13000006
			AND = {
				faith = faith:shadowinquisition
				any_sub_realm_county = {
					any_county_province = {
						geographical_region = world_europe_west_iberia
					}
				}
			}
		}
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:hunters_st_james_compostela_decision
			}
		}	
	}

	cooldown = { days = 5 }

    is_valid = {
        completely_controls = title:c_santiago
    }

	desc = hunters_st_james_compostela_decision_desc

	effect = {
		trigger_event = PODstjames.100
		custom_tooltip = hunters_st_james_compostela_decision.tt
		show_as_tooltip = {
			add_piety = 1000
			add_trait_xp = { trait = pilgrim value = 20 }
		}
	}

	cost = { piety = 150 }

	ai_will_do = {
		base = 0
	}
}
