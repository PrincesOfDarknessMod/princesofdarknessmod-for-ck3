######################################
# Tablet Recovery
######################################
#
# Available when Mary holds one of the four tablet provinces and has not yet
# recovered its tablet. Each tablet recovered fires the next tablet event in
# the narrative arc (regardless of which province she took).
#
mary_recover_tablet_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_dynasty_house.dds"
	}

	desc = mary_recover_tablet_decision_desc
	decision_group_type = major

	ai_check_interval = 0

	is_shown = {
		this = character:790149
		# Hide the decision entirely once all four tablets are recovered.
		# Guarded by has_variable so the read short-circuits before
		# podintroductions.380 has initialized it.
		has_variable = mary_tablets_recovered
		var:mary_tablets_recovered < 4
	}

	is_valid = {
		custom_tooltip = {
			text = mary_recover_tablet_decision_tooltip
			OR = {
				AND = {
					completely_controls = title:c_aleppo
					NOT = { has_character_flag = mary_tablet_aleppo_recovered }
				}
				AND = {
					completely_controls = title:c_damascus
					NOT = { has_character_flag = mary_tablet_damascus_recovered }
				}
				AND = {
					completely_controls = title:c_mosul
					NOT = { has_character_flag = mary_tablet_mosul_recovered }
				}
				AND = {
					completely_controls = title:c_baghdad
					NOT = { has_character_flag = mary_tablet_baghdad_recovered }
				}
			}
		}
	}

	effect = {
		# Defensive init: if Mary somehow reaches this decision before
		# podintroductions.380 has set up her variables, init to 0 here.
		if = {
			limit = { NOT = { has_variable = mary_tablets_recovered } }
			set_variable = { name = mary_tablets_recovered value = 0 }
		}

		# Set the per-province recovered flag for the held tablet province.
		# Priority: process Aleppo first if available, then Damascus, etc. — the
		# memory delivered depends only on tablets-recovered count, not province.
		if = {
			limit = {
				completely_controls = title:c_aleppo
				NOT = { has_character_flag = mary_tablet_aleppo_recovered }
			}
			add_character_flag = mary_tablet_aleppo_recovered
		}
		else_if = {
			limit = {
				completely_controls = title:c_damascus
				NOT = { has_character_flag = mary_tablet_damascus_recovered }
			}
			add_character_flag = mary_tablet_damascus_recovered
		}
		else_if = {
			limit = {
				completely_controls = title:c_mosul
				NOT = { has_character_flag = mary_tablet_mosul_recovered }
			}
			add_character_flag = mary_tablet_mosul_recovered
		}
		else_if = {
			limit = {
				completely_controls = title:c_baghdad
				NOT = { has_character_flag = mary_tablet_baghdad_recovered }
			}
			add_character_flag = mary_tablet_baghdad_recovered
		}

		# Dispatch the appropriate tablet event based on count *before* increment.
		if = {
			limit = { var:mary_tablets_recovered = 0 }
			trigger_event = POD_maryb.100
		}
		else_if = {
			limit = { var:mary_tablets_recovered = 1 }
			trigger_event = POD_maryb.110
		}
		else_if = {
			limit = { var:mary_tablets_recovered = 2 }
			trigger_event = POD_maryb.120
		}
		else_if = {
			limit = { var:mary_tablets_recovered = 3 }
			trigger_event = POD_maryb.130
		}

		change_variable = { name = mary_tablets_recovered add = 1 }

		# Scale down Fragmented Self to the next tier.
		remove_character_modifier = mary_fragmented_self_modifier
		remove_character_modifier = mary_fragmented_self_modifier_t1
		remove_character_modifier = mary_fragmented_self_modifier_t2
		remove_character_modifier = mary_fragmented_self_modifier_t3
		if = {
			limit = { var:mary_tablets_recovered = 1 }
			add_character_modifier = mary_fragmented_self_modifier_t1
		}
		else_if = {
			limit = { var:mary_tablets_recovered = 2 }
			add_character_modifier = mary_fragmented_self_modifier_t2
		}
		else_if = {
			limit = { var:mary_tablets_recovered = 3 }
			add_character_modifier = mary_fragmented_self_modifier_t3
		}
		# At 4 tablets recovered, no Fragmented Self modifier is applied.

		# Mirror new tablet count into the story panel.
		hidden_effect = { POD_maryb_sync_story_effect = yes }
	}

	ai_potential = {
		always = no
	}

	ai_will_do = {
		base = 0
	}
}

######################################
# Deal with Michael
######################################
#
# Available once Tablet IV has fired and locked the outcome.
# The dispatcher event POD_maryb.200 reads var:mary_deal_outcome and triggers
# the matching outcome event.
#
mary_deal_with_michael_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_dynasty_house.dds"
	}

	desc = mary_deal_with_michael_decision_desc
	decision_group_type = major

	ai_check_interval = 0

	is_shown = {
		this = character:790149
		NOT = { has_character_flag = mary_deal_resolved }
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:mary_deal_with_michael_decision
			}
		}
	}

	is_valid = {
		custom_tooltip = {
			text = mary_deal_with_michael_decision_tooltip
			# Order matters: has_variable short-circuits when the variable
			# hasn't been initialized yet (e.g. before podintroductions.380).
			has_variable = mary_tablets_recovered
			var:mary_tablets_recovered = 4
			has_variable = mary_deal_outcome
		}
	}

	effect = {
		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:mary_deal_with_michael_decision
		}
		trigger_event = POD_maryb.200
	}

	ai_potential = {
		always = no
	}

	ai_will_do = {
		base = 0
	}
}

######################################
# Michael Path — Resurrect the Dream
######################################
#
# Second Dream path capstone. Mary herself is already on the Codex of
# Legacies — that conversion happens in POD_maryb.220 the moment Michael
# rises in her body. This decision is the *realm-wide* sequel: every
# county she holds and every non-zealous vassal under her follows her
# faith into the source doctrine. Mirrors pod1230objectives.6.a's mass-
# conversion + piety reward, then triggers the empire+nickname flavour
# event (POD_maryb.320).
#
mary_resurrect_the_dream_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_dreaming_again.dds"
	}

	desc = mary_resurrect_the_dream_decision_desc
	decision_group_type = major

	ai_check_interval = 120

	is_shown = {
		this = character:790149
		has_character_flag = mary_resolved_second_dream
		faith = faith:codexoflegacies
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:mary_resurrect_the_dream_decision
			}
		}
	}

	is_valid = {
		is_independent_ruler = yes
		completely_controls = title:k_thessalonika
		prestige >= 1000
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_at_war = no
	}

	effect = {
		# Realm-wide doctrine reclamation (pod1230objectives.6.a body,
		# minus the self-conversion — Mary is already codexoflegacies).
		hidden_effect = {
			every_held_title = {
				limit = { tier = tier_county }
				set_county_faith = faith:codexoflegacies
			}
			every_vassal_or_below = {
				limit = {
					NOT = { primary_title.tier = tier_barony }
					NOT = { has_trait = zealous }
				}
				set_character_faith = faith:codexoflegacies
			}
			# Codex of Legacies is the source doctrine; Via Archangeli was
			# always its descendant. When the Patriarchate is rebuilt, every
			# Archangeli adherent in the world is pulled back to the source.
			# No zealous filter here — the design intent is that the entire
			# Archangeli faith follows Michael home.
			every_living_character = {
				limit = { faith = faith:viaarchangeli }
				set_character_faith = faith:codexoflegacies
			}
		}
		add_piety = massive_piety_value
		add_piety_level = 2
		play_music_cue = "mx_cue_epic_sacral_moment"

		# Empire + nickname grant flows through the flavour event so the
		# player gets a proper narrative beat rather than a silent fait
		# accompli.
		trigger_event = POD_maryb.320

		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:mary_resurrect_the_dream_decision
		}
	}

	ai_potential = {
		always = no
	}

	ai_will_do = {
		base = 0
	}
}

######################################
# Mary & Synthesis Paths — Unholy Ascension: Decani Hunt
######################################
#
# Black Dream variant requires holding Jerusalem.
# Ash and Gold variant requires holding Constantinople.
# Kills Anoster.
#
mary_decani_hunt_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_dynasty_house.dds"
	}

	desc = mary_decani_hunt_decision_desc
	decision_group_type = major

	ai_check_interval = 120

	is_shown = {
		this = character:790149
		OR = {
			has_character_flag = mary_resolved_black_dream
			has_character_flag = mary_resolved_ash_and_gold
		}
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:mary_decani_hunt_decision
			}
		}
		# Anoster is a static canon character. He's trapped in the abyss at
		# 1229.1.1 so PoD's lifecycle keeps him alive but inaccessible until
		# Mary frees him in the Decani Hunt announcement (POD_maryb.300).
		exists = character:9201500
		character:9201500 = { is_alive = yes }
	}

	is_valid = {
		is_independent_ruler = yes

		# Path-specific seat requirement (per design doc).
		#
		# Ash and Gold path uses c_byzantion specifically rather than
		# e_byzantium. PoD treats e_byzantium as a title that may not exist
		# at runtime (every PoD reference to it is wrapped in `exists =
		# title:e_byzantium...`); reading completely_controls on it during
		# tooltip build with a null title throws
		# "landed title (completely_controls) was null". The design doc says
		# "holding Constantinople" — PoD names that county c_byzantion
		# (the Greek-form key used throughout PoD's landed titles).
		custom_tooltip = {
			text = mary_decani_hunt_decision_tooltip
			OR = {
				AND = {
					has_character_flag = mary_resolved_black_dream
					completely_controls = title:k_jerusalem
				}
				AND = {
					has_character_flag = mary_resolved_ash_and_gold
					completely_controls = title:c_byzantion
				}
			}
		}

		# Base PoD demonization_objective_decision requirements (minus the
		# faith = viadiabolis gate, since Mary may be viaadversarius on the
		# Ash and Gold path). Mary is Baali gen 4, so the non-Baali fallback
		# `vampgeneration_less_or_equal_to_trigger + piety_level` check
		# auto-passes and we omit it.
		trigger_if = { limit = { POD_can_pick_elder_perk_trigger = { GEN = 4 } } has_perk = spew_wide_the_pit_perk }
		trigger_else_if = { limit = { POD_can_pick_elder_perk_trigger = { GEN = 5 } } has_perk = adramelechs_wrath_perk }
		trigger_else_if = { limit = { POD_can_pick_elder_perk_trigger = { GEN = 6 } } has_perk = bring_the_plague_perk }
		trigger_else_if = { limit = { POD_can_pick_elder_perk_trigger = { GEN = 7 } } has_perk = hell_born_investiture_perk }
		trigger_else = { has_perk = advanceddaimonion_perk }
		POD_umbra_realm_known_localized_trigger = { NAME = flag:abyss }
		OR = {
			any_killed_character = { count >= 100 }
			num_of_relation_bound_demon >= 5
		}
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_at_war = no
	}

	effect = {
		trigger_event = POD_maryb.300
		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:mary_decani_hunt_decision
		}
	}

	ai_potential = {
		always = no
	}

	ai_will_do = {
		base = 0
	}
}
