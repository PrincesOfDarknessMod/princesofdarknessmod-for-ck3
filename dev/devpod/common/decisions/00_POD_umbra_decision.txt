return_to_earth_decision = {
	title = action_can_return_to_earth
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}
	decision_group_type = major

	ai_check_interval = 0

	selection_tooltip = action_can_return_to_earth_label
	desc = action_can_return_to_earth_desc
	confirm_text = action_can_return_to_earth_click

	is_shown = {
		has_variable = spirit_location
		NOT = { var:spirit_location = flag:earth }
		NOT = { in_shadow_realm_trigger = { LOCATION = var:spirit_location } }
		POD_is_currently_in_umbra_expedition_trigger = no
		NOT = { has_trait = wraith }
	}

	effect = {
		custom_tooltip = umbra_place_earth_tooltip
		set_variable = { name = spirit_destination value = flag:earth }
		show_as_tooltip = { POD_umbra_travel_effect = { LOCATION = flag:earth } }
		if = {
			limit = { POD_umbra_bypass_trigger = { LOCATION = flag:earth } }
			trigger_event = { on_action = spirit_travel_movement_end_pulse }
		}
		else = {
			trigger_event = { on_action = spirit_travel_movement_start_pulse }
		}
	}
}
