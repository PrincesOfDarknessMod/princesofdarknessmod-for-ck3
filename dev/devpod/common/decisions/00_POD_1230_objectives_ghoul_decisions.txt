ultimate_submission_objective_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_knight_kneeling.dds"
	}
	
	decision_group_type = major
	
	ai_check_interval = 0 # super unlikely

	desc = ultimate_submission_objective_decision_desc

	is_shown = {
		is_playable_character = yes
		POD_is_bound_ghoul_trigger = yes
	}
	
	is_valid = {
		custom_tooltip = { 
			text = ultimate_submission_objective_decision_custom1 
			POD_number_of_completed_ghoul_missions >= 10
		}
		custom_tooltip = { 
			text = ultimate_submission_objective_decision_custom2 
			any_relation = {
				type = ghoul_domitor
				opinion = {
					target = PREV
					value >= 100
				}
			}
		}
		trigger_if = {
			limit = { has_trait = fera }
			POD_fera_can_become_abomination = yes
		}
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_at_war = no
	}

	effect = {
		random_relation = {
			type = ghoul_domitor
			save_scope_as = vampiresire
		}
		POD_major_objective_finished_effect = yes
		show_as_tooltip = {
			save_scope_as = created_vampirechilde
			custom_tooltip = ultimate_submission_objective_decision_effect1
			if = {
				limit = { POD_fera_can_become_abomination = yes }
				fera_embrace_effect = yes
			}
			else = {
				POD_become_vampire_effect = yes
				POD_character_vampire_inherit_generation_effect = yes
				POD_character_vampire_inherit_clan_effect = yes
			}
			add_character_modifier = POD_fulfilled_lifes_goal_modifier
		}
		trigger_event = pod1230objectives_ghoul.1
	}
}
