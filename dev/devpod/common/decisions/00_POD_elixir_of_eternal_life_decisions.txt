elixir_of_eternal_life_starter_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_sargon_codex.dds"
	}
	desc = elixir_of_eternal_life_starter_decision_desc
	selection_tooltip = elixir_of_eternal_life_starter_decision_tooltip
	decision_group_type = pod_splat

	ai_check_interval = 120

	is_shown = {
		OR = {
			NOT = { has_trait = eternal_rebirth_seeker }
			has_trait_xp = {
				trait = eternal_rebirth_seeker
				track = elixir_of_eternal_life
				value = 0
			}
		}
		OR = {
			# Eastern mummy
			has_trait = wutian
			has_trait = wukuei
			# Mystical knowledge & in right region
			AND = {
				OR = {
					has_trait = sorcerer
					has_trait_xp = { trait = pod_crafter track = brewing value >= 20 }
				}
				location = { geographical_region = POD_region_middle_kingdom }
			}

			trigger_if = { 
				limit = { POD_splat_mixing_haight_mode = yes } 

				OR = {
					# Eastern religion so you care about the Wu T'ian
					faith = { is_POD_eastern_religion_trigger = yes }
					faith = { has_doctrine = doctrine_fera_beast_courts }
					# Wu T'ian as virtue
					faith = { has_doctrine = special_doctrine_against_to_yomi_wan_hostility }
					# Servant of YKs who care about Wu Kuei
					faith = { has_doctrine = special_doctrine_allied_to_yomi_wan_hostility }
				}
			}
		}
	}

	is_valid = {
		OR = {
			learning >= 15
			has_trait = scholar
			has_trait = sorcerer
			has_trait = theurgist
			has_trait_xp = { trait = pod_crafter track = brewing value >= 50 }
		}
		any_courtier = {
			count >= 3
			OR = {
				learning >= 15
				has_trait = scholar
				has_trait = sorcerer
				has_trait = theurgist
				has_trait_xp = { trait = pod_crafter track = brewing value >= 20 }
			}
		}
		OR = {
			trigger_if = {
				limit = {
					# Can't use is_amenti_trigger as not all mummies (like Neferkheperu) have a subsplat
					has_trait = mummy
					is_east_mummy_trigger = no
				}
				AND = {
					has_perk = barring_the_gates_perk
					has_perk = aspect_of_duat_perk
					has_perk = promise_made_before_ammuts_scales_perk
					has_perk = heart_of_reincarnation_perk
				}
			}
			trigger_if = {
				limit = {
					has_trait = mummy
					is_east_mummy_trigger = yes
				}
				calc_true_if = {
					amount >= 3
					has_perk = whispering_stones_perk
					has_perk = scroll_of_the_minds_eye_perk
					has_perk = tread_the_dragon_tracks_perk
					has_perk = plague_curing_method_perk
					has_perk = walk_the_barriers_perk
					has_perk = restoration_of_youths_vigor_perk
				}
			}
			trigger_if = {
				limit = {
					POD_splat_is_fera_trigger = yes
				}
				AND = {
					custom_tooltip = {
						text = spell_of_life_stage1_hide_requirement
						any_character_trait = {
							has_trait_flag = fera_type
							save_temporary_scope_as = fera_trait
						}
					}
					has_trait_xp = { trait = scope:fera_trait value >= 40 }
					calc_true_if = {
						amount >= 2
						has_trait_xp = { trait = scope:fera_trait value = 100 }
						has_perk = way_of_ten_thousand_cities_perk
						has_perk = age_of_sleep_perk
						has_perk = earth_heal_perk
						has_perk = moondream_perk
						has_perk = cleansing_flame_perk
						has_perk = madthought_perk
						has_perk = planetarygifts_ebon_binding_perk
						has_perk = bring_forth_the_future_perk
						has_perk = heart_of_the_ice_warrior_perk
						has_perk = pack_mind_perk
						has_perk = grand_gesture_perk
						has_perk = burden_of_knowledge_perk
						has_perk = diplomatic_immunity_perk
						has_perk = drown_perk
						has_perk = essential_transformation_perk
					}
				}
			}
			trigger_if = {
				limit = {
					OR = {
						is_vampire_trigger = yes
						has_trait = revenant
						has_trait = ghoul
						has_trait = dhampir
						has_character_flag = vamp_discipline_access
						has_trait = bloodsorcerydiscipline
					}
				}
				AND = {
					has_trait = bloodsorceryadvanced
					calc_true_if = {
						amount >= 3
						has_perk = mutate_perk
						has_perk = master_of_samsara_perk
						has_perk = power_over_life_perk
						has_perk = duality_perk
						has_perk = hunger_of_the_void_perk
						has_perk = heave_the_host_of_hell_perk
						has_perk = minions_of_deep_marmora_perk
					}
				}
			}
			trigger_if = {
				limit = {
					OR = {
						is_kueijin_trigger = yes
						POD_splat_has_access_to_chi_arts = yes
					}
				}
				AND = {
					has_perk = chi_arts_chi_mastery_perk
					OR = {
						has_perk = soul_arts_soul_nursery_perk
						has_perk = soul_arts_dark_cultivation_perk
						has_perk = soul_arts_joyful_spark_of_union_perk
						has_perk = soul_arts_spirited_away_perk
						has_perk = soul_arts_divinity_key_perk
						has_perk = soul_arts_death_punch_perk
					}
					OR = {
						has_perk = shintai_the_shintai_of_ash_and_fire_perk
						has_perk = shintai_the_shintai_of_blood_perk
						has_perk = shintai_the_shintai_of_bone_perk
						has_perk = shintai_the_shintai_of_devotion_perk
						has_perk = shintai_the_shintai_of_flesh_perk
						has_perk = shintai_the_shintai_of_illusion_perk
						has_perk = shintai_the_shintai_of_jade_perk
						has_perk = shintai_the_shintai_of_poxes_perk
						has_perk = shintai_the_shintai_of_storms_perk
						has_perk = shintai_the_shintai_of_smoke_perk
					}
				}
			}
			trigger_if = {
				limit = {
					OR = {
						POD_splat_is_valid_for_sorcery = yes
						can_learn_sorcery_trigger = yes
						can_learn_theurgy_trigger = yes
					}
				}
				calc_true_if = {
					amount >= 3
					has_perk = numina_alchemy_5_perk
					has_perk = numina_ephemera_5_perk
					has_perk = numina_necromancy_5_perk
					has_perk = numina_oneiromancy_5_perk
					has_perk = numina_quintessence_manipulation_5_perk
					has_perk = numina_via_geniorum_5_perk
					has_perk = numina_via_necromantiae_5_perk
					has_perk = numina_via_medicamenti_5_perk
				}
			}
			trigger_if = {
				limit = { 
					current_year >= 1588 
				}
				custom_tooltip = {
					text = elixir_of_eternal_life_starter_decision_requirement_dutchman
					any_equipped_character_artifact = {
						has_variable = flying_dutchman
						prev = { can_benefit_from_artifact = prev }
					}
				}
			}
			trigger_if = {
				limit = { 
					POD_bookmark_has_fae = yes
				}
				custom_tooltip = {
					text = elixir_of_eternal_life_starter_decision_requirement_quarters
					any_equipped_character_artifact = {
						has_variable = ship_of_the_four_quarters
						prev = { can_benefit_from_artifact = prev }
					}
				}
			}
			custom_tooltip = {
				text = elixir_of_eternal_life_starter_decision_requirement_redeyes
				any_equipped_character_artifact = {
					has_variable = redeyes
					prev = { can_benefit_from_artifact = prev }
				}
			}
		}
	}
	
	is_valid_showing_failures_only = {
		is_capable_adult = yes
		is_imprisoned = no
		is_at_war = no
	}

	effect = {
		POD_add_elixir_of_eternal_life_xp = {
			VALUE = 1
		}
	}
	
	ai_potential = {
		always = yes
	}
	ai_will_do = {
		base = 100
	}
}
