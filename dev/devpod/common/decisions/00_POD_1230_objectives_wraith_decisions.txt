transcendence_objective_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_destiny_goal.dds"
	}
	decision_group_type = major
	
	ai_check_interval = 0 # super unlikely

	desc = transcendence_objective_decision_desc
	
	is_shown = {
		has_trait = wraith
		NOT = { has_trait = spectre }
		NOT = { has_trait = risen }
	}
	
	is_valid = {
		has_trait_xp = {
			trait = wraith
			track = fetter_resolved
			value >= 100
		}
		piety_level >= 5
		num_of_relation_best_friend = 0
		num_of_relation_friend = 0
		num_of_relation_soulmate = 0
		num_of_relation_lover = 0
		num_of_relation_rival = 0
		num_of_relation_nemesis = 0
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_at_war = no
		NOT = { has_trait = shadow }
	}

	effect = {
		POD_major_objective_finished_effect = yes
		show_as_tooltip = {
			death = {
				death_reason = death_transcendence
			}
			custom_tooltip = transcendence_objective_tooltip
		}
		trigger_event = pod1230objectives_wraith.1
	}
}

enforce_the_dictum_mortuum_objective_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_destiny_goal.dds"
	}
	decision_group_type = major

	ai_check_interval = 120

	is_shown = {
		is_landed = yes
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:end_the_dictum_mortuum_objective_decision
			}
		}
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:enforce_the_dictum_mortuum_objective_decision
			}
		}
		has_trait = wraith
		faith = {
			OR = {
				has_doctrine = special_doctrine_is_legion_faith
				has_doctrine = special_doctrine_is_great_guild_faith
				has_doctrine = special_doctrine_is_working_guild_faith
			}
		}
	}
	
	is_valid = {
		is_independent_ruler = yes
		trigger_if = {
			limit = { 
				any_independent_ruler = {
					OR = {
						has_trait = wraith
						is_giovanni_trigger = yes	
					}
					is_landed = yes
					save_temporary_scope_as = example
				}
			}
			custom_tooltip = {
				text = enforce_the_dictum_mortuum_objective_decision_custom_ex
				always = no
			}
		}
		trigger_else = {
			custom_tooltip = {
				text = enforce_the_dictum_mortuum_objective_decision_custom
			}
		}
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_at_war = no
	}

	effect = {
		POD_major_objective_finished_effect = yes
		show_as_tooltip = {
			add_character_modifier = the_enforcer_of_dictum_mortuum
		}
		trigger_event = pod1230objectives.141
		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:enforce_the_dictum_mortuum_objective_decision
		}
	}
	
	ai_potential = {
		NOT = { any_player = { has_trait = wraith } }
	}

	ai_will_do = {
		base = 100
	}
}
war_with_the_tyrant_objective_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_destiny_goal.dds"
	}
	decision_group_type = major

	ai_check_interval = 120

	desc = war_with_the_tyrant_objective_decision_desc

	is_shown = {
		is_landed = yes
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:war_with_the_tyrant_objective_decision
			}
		}		
		has_trait = wraith
		character:POD_wraith_01 = { is_alive = yes }
		faith = {
			NOR = {
				has_doctrine = special_doctrine_is_legion_faith
				has_doctrine = special_doctrine_is_great_guild_faith
				has_doctrine = special_doctrine_is_working_guild_faith
			}
		}
	}
	
	is_valid = {
		is_independent_ruler = yes
		prestige_level >= 5
		custom_tooltip = { text = war_with_the_tyrant_objective_decision_skills sum_of_all_skills_value >= 60 }
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_at_war = no
	}

	effect = {
		POD_major_objective_finished_effect = yes
		character:POD_wraith_01 = {
			custom_tooltip = war_with_the_tyrant_objective_decision_custom
			hidden_effect = {
				start_war = {
					cb = POD_event_war
					target = root
				}
				random_character_war = {
					limit = { using_cb = POD_event_war }
					save_scope_as = this_war
				}
			}
			add_character_modifier = {
				modifier = charon_wraith_maa_modifier
				years = 50
			}
			spawn_army = {
				uses_supply = no
				inheritable = no
				men_at_arms = {
					type = wraiths
					stacks = 500 # NERFED FROM 1000
				}		
				location = root.capital_province
				war = scope:this_war
				name = charon_troops
			}
		}
		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:war_with_the_tyrant_objective_decision
		}
	}
	
	ai_potential = {
		NOT = { any_player = { has_trait = wraith } }
	}

	ai_will_do = {
		base = 100
	}
}

	picture = {

	}
	decision_group_type = major
	
	ai_check_interval = 0 # super unlikely

	desc = free_abel_objective_decision_desc
	
	cost = { prestige = 7777 }

	is_shown = {
		is_landed = yes
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:free_abel_objective_decision
			}
		}
		has_character_flag = find_abel
	}
	
	is_valid = {
		prowess >= 90
		calc_true_if = {
			amount >= 2
			diplomacy >= 50
			intrigue >= 50
			martial >= 50
			stewardship >= 50
			learning >= 50
		}
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_at_war = no
		NOT = { faith = faith:grand_maw }
	}

	effect = {
		POD_major_objective_finished_effect = yes
		show_as_tooltip = {
			random_list = {
				50 = {
					show_chance = no
					desc = pod1230objectives.90.a
					custom_tooltip = orpheus_good_ending_tooltip
				}
				50 = {
					show_chance = no
					desc = pod1230objectives.90.b
					dynasty = { add_dynasty_prestige = excessive_dynasty_prestige_value }

					declare_bloodline_holy_decision_effect = yes
					add_character_modifier = abel_modifier
				}
			}
		}
		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:free_abel_objective_decision
		}
		trigger_event = pod1230objectives.90
	}
}
