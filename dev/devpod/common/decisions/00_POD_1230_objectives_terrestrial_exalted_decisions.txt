transcendent_gaian_harmony_objective_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_demon_redemption.dds"
	}

	decision_group_type = major

	ai_check_interval = 120

	desc = {
		desc = transcendent_gaian_harmony_objective_decision_desc
		first_valid = {
			triggered_desc = {
				trigger = { has_variable = transcendent_gaian_harmony }
				desc = transcendent_gaian_harmony_objective_decision_discovered_desc
			}
			desc = transcendent_gaian_harmony_objective_decision_undiscovered_desc
		}
	}

	is_shown = {
		POD_is_terrestrial_exalted_trigger = yes
		NOT = { var:transcendent_gaian_harmony ?= 4 }
		has_trait_xp = {
			trait = terrestrial_exalted
			value = 100
		}
	}

	is_valid = {
		POD_has_enough_essence_for_charm = { AMOUNT = 10 }
		OR = {
			has_trait = whole_of_body
			has_trait = scholar
			has_trait = theologian
			has_trait = lifestyle_physician
		}
		save_temporary_scope_value_as = {
			name = learning_needed
			value = {
				value = 20
				if = {
					limit = { has_variable = transcendent_gaian_harmony }
					add = {
						value = var:transcendent_gaian_harmony
						multiply = 5
					}
				}
			}
		}
		learning >= scope:learning_needed
		save_temporary_scope_value_as = {
			name = dragon_nest_needed
			value = {
				value = 4
				if = {
					limit = { has_variable = transcendent_gaian_harmony }
					add = var:transcendent_gaian_harmony
				}
			}
		}
		any_directly_owned_province = {
			switch = {
				trigger = scope:dragon_nest_needed
				3 = { has_building_or_higher = dragon_nest_03 }
				4 = { has_building_or_higher = dragon_nest_04 }
				5 = { has_building_or_higher = dragon_nest_05 }
				6 = { has_building_or_higher = dragon_nest_06 }
				7 = { has_building_or_higher = dragon_nest_07 }
			}
		}
	}
	
	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_at_war = no
	}

	effect = {
		POD_major_objective_finished_effect = yes
		save_scope_value_as = {
			name = modifier_level
			value = {
				value = 0
				if = {
					limit = { has_variable = transcendent_gaian_harmony }
					value = var:transcendent_gaian_harmony
				}
				add = 1
			}
		}
		show_as_tooltip = {
			POD_essence_cost_effect = { COUNT = 10 }
			add_character_modifier = transcendent_gaian_harmony_modifier
			custom_tooltip = transcendent_gaian_harmony_objective_decision_effect1
			custom_tooltip = transcendent_gaian_harmony_objective_decision_effect2
		}
		set_variable = {
			name = transcendent_gaian_harmony
			value = {
				value = 0
				if = { 
					limit = { has_variable = transcendent_gaian_harmony } 
					value = var:transcendent_gaian_harmony 
				}
				add = 1
			}
		}
		trigger_event = pod1230objectives_terrestrial_exalted.1
	}
	
	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}

		reference = "gfx/interface/illustrations/decisions/decision_dreaming_again.dds"
	}
	
	decision_group_type = major

	ai_check_interval = 120

	is_shown = {
		POD_is_terrestrial_exalted_trigger = yes

				name = unavailable_unique_decisions

			}
		}
	}
	
	is_valid = {

				}
			}

			text = reaffirm_heavens_praise_objective_decision_custom1
			scope:total_spirit_companion_skill >= 200
		}
		save_temporary_scope_value_as = {
			name = total_spirit_companion_rank
			value = {
				every_relation = {
					type = spirit_companion
					add = POD_spirit_rank
				}
			}
		}
		custom_tooltip = {
			text = reaffirm_heavens_praise_objective_decision_custom2
			scope:total_spirit_companion_rank >= 10
		}
		OR = {
			custom_tooltip = {
				text = reaffirm_heavens_praise_objective_decision_custom3

			}
			custom_tooltip = {
				text = reaffirm_heavens_praise_objective_decision_custom4

					type = spirit_companion 
					count >= 3

					type = spirit_companion

				}
			}
		}
		prestige_level >= 4
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_at_war = no
	}

	effect = {
		POD_major_objective_finished_effect = yes
		show_as_tooltip = {
			faith = { add_doctrine = special_reaffirmed_heavens_praise_doctrine }
			custom_tooltip = reaffirm_heavens_praise_objective_decision_effect1
		}
		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:reaffirm_heavens_praise_objective_decision
		}
		trigger_event = pod1230objectives_terrestrial_exalted.3
	}
	
	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}
restore_mothers_favor_objective_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_get_of_fenris.dds"
	}
	
	decision_group_type = major

	ai_check_interval = 120

	is_shown = {
		POD_is_terrestrial_exalted_trigger = yes
		NOT = { faith = faith:immaculates }
		is_infernalist_trigger = no
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:restore_mothers_favor_objective_decision
			}
		}
	}
	
	is_valid = {
		NOT = { faith = faith:immaculates }
		is_infernalist_trigger = no
		POD_umbra_realm_known_localized_trigger = { NAME = flag:summer_country }
		save_temporary_scope_value_as = {
			name = gaian_fera_rulers_count
			value = {
				every_independent_ruler = {
					limit = {
						is_landed = yes
						POD_is_fera_trigger = yes
						POD_is_wyrm_servant_trigger = no
					}
					add = 1
				}
			}
		}
		save_temporary_scope_value_as = {
			name = positive_relation_count
			value = {
				every_relation = {
					type = friend
					type = best_friend
					type = lover
					type = soulmate
					limit = { POD_is_fera_trigger = yes POD_is_wyrm_servant_trigger = no }
					add = 1
				}
			}
		}
		save_temporary_scope_value_as = {
			name = average_development
			value = {
				add = total_domain_development
				divide = realm_size
			}
		}
		OR = {
			trigger_if = {
				limit = { POD_bookmark_has_fera = yes }
				OR = {
					custom_tooltip = {
						text = restore_mothers_favor_objective_decision_custom1
						scope:gaian_fera_rulers_count <= 15
					}
					custom_tooltip = {
						text = restore_mothers_favor_objective_decision_custom2
						scope:positive_relation_count >= 2
					}
				}
			}
			trigger_else = {
				realm_size >= 30 # You have to find unlanded fera.
			}
		}
		OR = {
			trigger_if = {
				limit = { POD_bookmark_has_fera = yes }
				custom_tooltip = {
					text = restore_mothers_favor_objective_decision_custom3
					scope:gaian_fera_rulers_count <= 10
				}
			}
			custom_tooltip = {
				text = restore_mothers_favor_objective_decision_custom4
				sum_of_all_skills_value >= 60
			}
		}
		OR = {
			trigger_if = {
				limit = { POD_bookmark_has_fera = yes }
				custom_tooltip = {
					text = restore_mothers_favor_objective_decision_custom5
					scope:gaian_fera_rulers_count <= 5
				}
			}
			AND = {
				custom_tooltip = {
					text = restore_mothers_favor_objective_decision_custom6
					scope:average_development <= 15
				}
				realm_size >= 15
			}
		}
		OR = {
			trigger_if = {
				limit = { POD_bookmark_has_fera = yes }
				custom_tooltip = {
					text = restore_mothers_favor_objective_decision_custom7
					scope:gaian_fera_rulers_count <= 0
				}
			}
			custom_tooltip = {
				text = restore_mothers_favor_objective_decision_custom8
				NOT = { has_character_flag = POD_killed_gaian_fera_last_one_hundred_years }
			}
		}
		
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_at_war = no
	}

	effect = {
		POD_major_objective_finished_effect = yes
		show_as_tooltip = {
			faith = { add_doctrine = special_restored_mothers_favor_doctrine }
			faith = { 
				remove_doctrine = doctrine_fera_shunned 
				add_doctrine = doctrine_fera_accepted 
			}
			faith = { add_doctrine = doctrine_fera_werewolf_pact }
		}
		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:restore_mothers_favor_objective_decision
		}
		trigger_event = pod1230objectives_terrestrial_exalted.4
	}
	
	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}
