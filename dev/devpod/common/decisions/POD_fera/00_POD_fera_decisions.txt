
fera_advance_rank_decision = {
	decision_group_type = major
	picture = { reference = "gfx/interface/illustrations/decisions/decision_personal_religious.dds" }

	desc = fera_advance_rank_decision_desc
	
	is_shown = {
		has_trait = fera
		POD_is_max_fera_rank = no
	}
	
	is_valid = {
		custom_tooltip = {
			text = fera_advance_rank_decision_effect
			always = yes
		}
		
		#Renown Types
		
		trigger_if = {
			limit = { POD_fera_uses_renown_glory = yes }
			custom_tooltip = { text = fera_advance_rank_renown_glory   always = yes }
		}
		trigger_if = {
			limit = { POD_fera_uses_renown_honor = yes }
			custom_tooltip = { text = fera_advance_rank_renown_honor   always = yes }
		}
		trigger_if = {
			limit = { POD_fera_uses_renown_wisdom = yes }
			custom_tooltip = { text = fera_advance_rank_renown_wisdom   always = yes }
		}
		trigger_if = {
			limit = { POD_fera_uses_renown_power = yes }
			custom_tooltip = { text = fera_advance_rank_renown_power   always = yes }
		}
		trigger_if = {
			limit = { POD_fera_uses_renown_cunning = yes }
			custom_tooltip = { text = fera_advance_rank_renown_cunning   always = yes }
		}
		trigger_if = {
			limit = { POD_fera_uses_renown_ferocity = yes }
			custom_tooltip = { text = fera_advance_rank_renown_ferocity   always = yes }
		}
		trigger_if = {
			limit = { POD_fera_uses_renown_infamy = yes }
			custom_tooltip = { text = fera_advance_rank_renown_infamy   always = yes }
		}
		
		##Add new here
		
		#Auspice and other bonuses
		custom_tooltip = {
			text = fera_advance_rank_renown_bonuses
			always = yes
		}
		
		POD_has_trait_xp_for_fera_rankup = yes
		
		trigger_if = {
			limit = { POD_fera_rank_value = 1 }
			any_powerful_vassal = {
				count = all

				opinion = {
					target = root
					value >= 0
				}
			}
		}
		trigger_else_if = {
			limit = { POD_fera_rank_value = 2 }
			OR = {
				prestige_level >= 2
				piety_level >= 2
			}
			any_powerful_vassal = {
				count = all

				opinion = {
					target = root
					value >= 25
				}
			}
		}
		trigger_else_if = {
			limit = { POD_fera_rank_value = 3 }
			OR = {
				prowess >= 40
				diplomacy >= 15
				martial >= 15
				stewardship >= 15
				intrigue >= 15
				learning >= 15
			}
			OR = {
				prestige_level >= 3
				piety_level >= 3
			}
			any_powerful_vassal = {
				count = all

				opinion = {
					target = root
					value >= 50
				}
			}
		}
		trigger_else_if = {
			limit = { POD_fera_rank_value = 4 }
			OR = {
				prowess >= 70
				diplomacy >= 25
				martial >= 25
				stewardship >= 25
				intrigue >= 25
				learning >= 25
			}
			OR = {
				prestige_level >= 4
				piety_level >= 4
			}
			any_powerful_vassal = {
				count = all

				opinion = {
					target = root
					value >= 75
				}
			}
		}
		trigger_else = {
			prestige_level >= 5
			piety_level >= 5
			legitimacy_level >= 5
			OR = {
				highest_held_title_tier >= tier_empire
				custom_tooltip = {
					text = fera_advance_rank_decision_requirement2
					any_tributary = {
						count = 3
						highest_held_title_tier >= tier_kingdom
					}
				}
			}
			any_powerful_vassal = {
				count = all

				opinion = {
					target = root
					value >= 75
				}
			}
		}
	}
	
	effect = {
		POD_fera_add_single_trait_xp_effect = yes
	}
	
	#AI skips this process
	ai_check_interval = 0
	ai_potential = { always = no }
	ai_will_do = { base = 0 }
}
