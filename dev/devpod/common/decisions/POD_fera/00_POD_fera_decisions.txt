

bastet_advance_rank_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_personal_religious.dds"
	}
		
	desc = {
		desc = bastet_advance_rank_decision_desc
		desc = line_break
		desc = line_break
		desc = advance_rank_decision_desc.cunning
		desc = line_break
		desc = advance_rank_decision_desc.ferocity
		desc = line_break
		desc = advance_rank_decision_desc.honor
	}
	
	decision_group_type = major

	ai_check_interval = 60
	
	is_shown = {
		is_playable_character = yes
		has_trait = bastet
		has_trait_xp = { trait = bastet value < 100 }
		NOT = { faith = { has_doctrine = special_doctrine_beast_courts } }
	}
	
	is_valid = {
		trigger_if = {
			limit = { has_trait_xp = { trait = bastet value < 20 } }
			has_trait_xp = { trait = bastet value >= 19 }
			OR = {
				piety_level >= 1
				legitimacy_level >= 1
				prestige_level >= 1
			}
			intrigue >= 10
		}
		trigger_else_if = {
			limit = { has_trait_xp = { trait = bastet value < 40 } }
			has_trait_xp = { trait = bastet value >= 39 }
			OR = {
				piety_level >= 2
				legitimacy_level >= 2
				prestige_level >= 2
			}
			intrigue >= 15
		}
		trigger_else_if = {
			limit = { has_trait_xp = { trait = bastet value < 60 } }
			has_trait_xp = { trait = bastet value >= 59 }
			OR = {
				piety_level >= 3
				legitimacy_level >= 3
				prestige_level >= 3
			}
			intrigue >= 20
		}
		trigger_else_if = {
			limit = { has_trait_xp = { trait = bastet value < 80 } }
			has_trait_xp = { trait = bastet value >= 79 }
			OR = {
				piety_level >= 4
				legitimacy_level >= 4
				prestige_level >= 4
			}
			intrigue >= 20
			diplomacy >= 10
		}
		trigger_else = {
			has_trait_xp = { trait = bastet value >= 99 }
			OR = {
				piety_level >= 5
				legitimacy_level >= 5
				prestige_level >= 5
			}
			intrigue >= 20
			diplomacy >= 15
		}
	}
	
	effect = {
		add_trait_xp = { trait = bastet value = 1 }
		add_piety = minor_piety_gain
		add_legitimacy = minor_legitimacy_gain
		add_prestige = minor_prestige_gain
	}	
	
	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}


changing_moon_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_personal_religious.dds"
	}
	decision_group_type = pod_splat	
	cost = { gold = { value = minor_gold_value } }
	cooldown = { years = 20 }

	is_shown = {
		is_playable_character = yes
		POD_is_gurahl_trigger = yes
		NOT = { has_trait = philodox }
	}

	is_valid = {	
		NOT = { has_trait = philodox } # tooltip only	
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
		is_at_war = no
	}
		
	effect = {
		if = {
			limit = { has_trait = ragabash }
			remove_trait = ragabash
			add_trait = ahroun
		}
		else_if = {
			limit = { has_trait = ahroun }
			remove_trait = ahroun
			add_trait = galliard
		}
		else_if = {
			limit = { has_trait = galliard }
			remove_trait = galliard
			add_trait = theurge
		}
		else = {
			remove_trait = theurge
			add_trait = philodox
		}
	}
	
	ai_check_interval = 360

	ai_potential = {
		years_from_game_start > 20
	}

	ai_will_do = {
		base = 100
	}
}
