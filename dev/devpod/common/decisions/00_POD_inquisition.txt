POD_landless_inquisition_victory_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/inquisitionfin.dds"
	}
	
	decision_group_type = adventurer
	ai_check_interval = 0
	sort_order = 1000
	
	is_shown = {
		is_ai = no
		faith = faith:shadowinquisition
		is_landless_ruler = yes
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:landless_inquisition_victory_objective_decision
			}
		}
	}

	is_valid = {
		trigger_if = {
			limit = { 
				any_independent_ruler = {
					is_landed = yes
					NOT = { religion = { is_in_family = rf_hunters } }
					OR = {
					capital_province = { geographical_region = world_europe }
					capital_province = { geographical_region = world_middle_east_jerusalem }
					}
					save_temporary_scope_as = example
				}
			}
			custom_tooltip = {
				text = POD_landless_inquisition_victory_decision_custom_ex
				always = no
			}
		}
		trigger_else = {
			custom_tooltip = {
				text = POD_landless_inquisition_victory_decision_custom
			}
		}		
	}
		
	desc = POD_landless_inquisition_victory_decision_desc

	effect = {
		POD_major_objective_finished_effect = yes
		save_scope_as = victorious_inquisitor
		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:landless_inquisition_victory_objective_decision
		}
		if = {
			limit = {
				has_trait = excommunicated
				}
			remove_trait = excommunicated
		}
		custom_tooltip = landless_inquisition_victory_objective_decision_effect_tooltip
		trigger_event = pod1230objectives.163
		show_as_tooltip = {
			if = {
				limit = { has_trait = beatific }
				POD_advance_beatific_multiple_effect = { VALUE = 5 }
			}
			add_character_modifier =  chosen_of_light_modifier			
		}
	}

	ai_will_do = {
		base = 0
	}
}
