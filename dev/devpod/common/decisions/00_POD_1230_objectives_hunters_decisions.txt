ikhwan_al_safa_landless_objective_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_destiny_goal.dds"
	}
	decision_group_type = major

	cooldown = { days = 5 } #Needed because you can actually refuse the reward

	ai_check_interval = 0

	sort_order = 100

	is_shown = {
		has_trait = supehunter
		NOT = { faith = faith:shadowinquisition }
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:ikhwan_al_safa_landless_objective_decision
			}
		}	
	}

	is_valid = {
		intrigue >= 20
		trigger_if = {
			limit = { 
				has_faith = faith:ikhwan_al_safa
			}
			OR = {
				POD_title_supports_you_for_objectives = { TITLE = title:c_tunis }
				POD_title_supports_you_for_objectives = { TITLE = title:c_cordoba }
				POD_title_supports_you_for_objectives = { TITLE = title:c_nikaea }
				POD_title_supports_you_for_objectives = { TITLE = title:c_SUM_perlak }
			}
		}
		trigger_else = {
			POD_title_supports_you_for_objectives = { TITLE = title:c_tunis }
		}
	}
	
	is_valid_showing_failures_only = {
		is_capable_adult = yes
		is_imprisoned = no
		is_independent_ruler = yes
	}

	effect = {
		POD_major_objective_finished_effect = yes
		#Just skip to the reward if you are of the right faith 
		if = {
			limit = {
				NOT = { has_faith = faith:ikhwan_al_safa }
			}
			show_as_tooltip = {
				dynasty = { add_dynasty_modifier = dynasty_hunter_alliances_ikhwan_modifier }
			}
		
			trigger_event = pod1230objectives.164
		}
		
		else = {
			dynasty = { add_dynasty_modifier = dynasty_hunter_alliances_ikhwan_modifier }
			add_to_global_variable_list = {
				name = unavailable_unique_decisions

			}
			custom_tooltip = {
				text = ikhwan_al_safa_landless_objectives_tt
			}
		}

	#	always = yes

judge_contact_magi_objective_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_destiny_goal.dds"
	}
	decision_group_type = major

	ai_check_interval = 120

	is_shown = {
		faith = faith:judge
		is_target_in_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:judges_landless_objective_decision
		}
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:judge_contact_magi_objective_decision
			}
		}
	}

	is_valid = {
		calc_true_if = {
			amount >= 5

			learning >= 20
			completely_controls = title:c_kufa
			completely_controls = title:c_baghdad
			has_trait = erudite
			has_trait = sorcerer
			has_trait = lifestyle_mystic
			has_trait_xp = {
				trait = lifestyle_mystic
				value >= 100
			}
			has_perk = numina_via_necromantiae_5_perk
			custom_tooltip = {
				text = judge_contact_magi_objective_decision_know_sorcerer
				OR = {
					any_relation = {
						type = coterie_member
						has_trait = sorcerer
					}
					any_relation = {
						type = friend
						has_trait = sorcerer
					}
					any_relation = {
						type = best_friend
						has_trait = sorcerer
					}
					any_relation = {
						type = lover
						has_trait = sorcerer
					}
					any_relation = {
						type = soulmate
						has_trait = sorcerer
					}
				}
			}
		}
	}
	
	is_valid_showing_failures_only = {
		is_capable_adult = yes
		is_imprisoned = no
		is_independent_ruler = yes
		is_at_war = no
	}

	effect = {
		POD_major_objective_finished_effect = yes
		show_as_tooltip = {
			if = {
				limit = { POD_splat_is_valid_for_sorcery = yes }
				faith:judge = {
					add_doctrine = special_doctrine_sorcerer_ordination
				}
				if = {
					limit = { NOT = { faith = faith:judge } }
					faith = {
						add_doctrine = special_doctrine_sorcerer_ordination
					}
				}
				add_trait = sorcerer
			}
			faith = {
				add_doctrine = doctrine_magicraft_accepted
			}
			POD_add_free_dynasty_perk = { PERK = hunter_judges_legacy_1 }
		}
		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:judge_contact_magi_objective_decision
		}
		trigger_event = pod1230objectives_judges.1

		# changed this to judge-only since sargon shards are for a judge objective
		if = {
			limit = { faith = faith:judge }
			custom_tooltip = quest_for_sargon_codex_objective_decision_at_effect1
		}

		if = {
			limit = {
				faith = faith:judge
				NOT = { any_character_artifact = { has_variable = judge_havanah_book } }
			}
			custom_tooltip = judge_contact_objective_decision_havanah_tt
		}
	}
	
	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}
