become_ghoul_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_destiny_goal.dds"
	}
	decision_group_type = pod_customization
    ai_check_interval = 0

    is_shown = {
		is_playable_character = yes
		OR = {
			has_character_flag = chargen_changing_splat_flag
			POD_charjoin_decisions_enabled = yes
		}
		OR = {
			has_trait = supehunter
			has_trait = revenant
			has_trait = fera
		}
		NOT = { has_trait = ghoul }
	}
	
	is_valid = {
		any_prisoner = {
			has_trait = vampire
		}
	}

	is_valid_showing_failures_only = {
		POD_has_true_faith_trigger = no
		NOT = { has_trait = orison_sanguine }
	}

    effect = {
		custom_tooltip = become_ghoul_decision_tooltip
		
		save_scope_as = ghoulthrall
		save_scope_as = domitor
		POD_player_character_create_ghoul_effect = yes
		random_prisoner = {
			limit = { has_trait = vampire }
			save_scope_as = victim
			execute_opinion_effect = { VICTIM = scope:victim EXECUTIONER = scope:domitor }
			death = {
	        	death_reason = death_bloodloss
				killer = scope:domitor
	        }
		}
		
		remove_character_flag = chargen_changing_splat_flag
		trigger_event = podintroductions.2002
	}
}

join_shih_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_destiny_goal.dds"
	}
	decision_group_type = pod_customization
	ai_check_interval = 120

	is_shown = {
		is_playable_character = yes
		has_character_flag = chargen_changing_order_flag
		OR = {
			has_trait = supehunter
			has_trait = mortal
		}
		NOT = { has_trait = shih }
	}

	effect = {
		custom_tooltip = join_shih_decision_tooltip
		if = { 
			limit = {
				has_trait = mortal 
			}
		remove_trait = mortal
		}
		if = {
			limit = {
				NOT = { has_trait = supehunter }
			}
		add_trait = supehunter
		}
		POD_shih_initialization_effect = yes
		remove_character_flag = chargen_changing_order_flag
	}
}
