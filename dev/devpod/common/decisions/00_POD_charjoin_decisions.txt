join_clan_decision = {
    picture = "gfx/interface/illustrations/decisions/decision_dynasty_house.dds"

    ai_check_interval = 0

    is_shown = {
		is_landed = yes
		has_character_flag = allow_charjoin_flag
		POD_can_be_embraced_scheme_trigger = yes
		NOT = { has_trait = shapeshifter }
		NOR = { 
			has_character_flag = independent_char_join_clan_flag
			has_character_flag = independent_char_join_order_flag
			has_character_flag = independent_char_join_wu_flag
			has_character_flag = independent_char_join_ghoulbros_flag
		}
	}

	is_valid_showing_failures_only = {
		NOR = { 
			has_trait = truefaith
			has_trait = beatific
			has_trait = orison_sanguine 
		}
	}

    effect = {
		custom_tooltip = join_clan_decision_tooltip
		add_character_flag = independent_char_join_clan_flag
	}
}

    picture = "gfx/interface/illustrations/decisions/decision_dynasty_house.dds"

    ai_check_interval = 120

    is_shown = {
		is_landed = yes
		OR = {
			has_character_flag = allow_charjoin_flag
			has_trait = dhampyr
		}
		OR = {
			has_trait = supehunter
			has_trait = mortal
			has_trait = dhampyr
		}
		NOR = { 
			has_character_flag = independent_char_join_clan_flag
			has_character_flag = independent_char_join_order_flag
			has_character_flag = independent_char_join_wu_flag
			has_character_flag = independent_char_join_ghoulbros_flag
		}
		NOT = { has_trait = shih }
	}

    effect = {
		custom_tooltip = join_shih_decision_tooltip
		remove_trait = mortal
		add_trait = supehunter
		POD_shih_initialization_effect = yes
	}
