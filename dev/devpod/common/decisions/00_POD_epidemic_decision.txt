
heal_chulorviosis_decision = {
	picture = "gfx/interface/illustrations/decisions/decision_misc.dds"

	ai_check_interval = 0

	is_shown = {
		has_trait = chulorviosis
	}

	is_valid = {
		OR = {

				}
			}

				type = lover

				}
			}
		}
	}

	is_valid_showing_failures_only = {
		is_available = yes
	}

	effect = {
		recover_from_disease_notify_effect = { DISEASE = chulorviosis }
	}
}
