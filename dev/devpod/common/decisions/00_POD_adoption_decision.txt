foster_sprite_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_recruitment.dds"
	}	

	desc = foster_sprite_decision_desc
	selection_tooltip = foster_sprite_decision_tooltip
	decision_group_type = pod_splat
	
	ai_check_interval = 120

	is_shown = {
		is_playable_character = yes
		has_trait = fae
	}

	cooldown = { days = 3650 }

	is_valid_showing_failures_only = {
		is_available = yes
		POD_reweaving_is_valid_showing_failures_only = yes
	}
	
	cost = {
		gold = 500
	}

	effect = {
		if = {
			limit = { is_ai = no }
			set_variable = { name = character_creator_type value = flag:sprite }
			set_variable = { name = character_creator_points value = 45 }
			custom_tooltip = foster_sprite_decision_effect_tooltip
		}
		else = {
			if = {
				limit = { is_female = yes }
				create_character = {
					mother = root
					age = { 18 60 }
					location = root.capital_province
					dynasty_house = root.house
					culture = root.capital_province.culture
					faith = root.faith
					gender_female_chance = 50
					trait = fae
					random_traits = yes
					after_creation = {
						POD_remove_all_terrible_traits_effect = yes
						POD_fae_initialization_effect = yes
					}
				}
			}	
			else = {
				create_character = {
					father = root
					age = { 18 60 }
					location = root.capital_province
					dynasty_house = root.house
					culture = root.capital_province.culture
					faith = root.faith
					gender_female_chance = 50
					trait = fae
					random_traits = yes
					after_creation = {
						POD_remove_all_terrible_traits_effect = yes
						POD_fae_initialization_effect = yes
					}
				}
			}
		}
	}
	
	ai_potential = {
		fertility <= 0.1
		any_child = { is_alive = yes count < 1 }
	}

	ai_will_do = {
		base = 100
	}
}
