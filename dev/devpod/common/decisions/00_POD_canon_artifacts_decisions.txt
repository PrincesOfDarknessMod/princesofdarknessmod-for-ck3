		reference = "gfx/interface/illustrations/decisions/decision_destiny_goal.dds"
	}

	decision_group_type = major

		OR = {
			any_character_artifact = {

				has_variable = spear_destiny
			}	
		}
		
		OR = {
			faith = { is_POD_christian_religion_trigger = yes }
			has_trait = werewolf
			POD_can_participate_in_vampire_politics = yes
			AND = {
				has_trait = supehunter
				NOT = { has_trait = shih }
			}
		}
		
		POD_has_true_faith_trigger = no
		POD_blocked_from_learning_trigger = no
	}
		
	is_valid = {
	}
	
	is_valid_showing_failures_only = {
		trigger_if = {
			limit = { POD_splat_mixing_haight_mode = no }
			POD_is_infernalist_trigger = no
		}	
	}
	
	effect = {	
		if = {
			limit = {
				faith = { is_POD_christian_religion_trigger = yes }
			}
		}
		
		else_if = {
			limit = {
				has_trait = werewolf
				is_female = no
			}
			set_character_faith = faith:crestofthehorn
		}

		else_if = {
			limit = {
				has_trait = werewolf
				is_female = yes
			}
			set_character_faith = faith:orderofourmercifulmother
		}

		else_if = {
			limit = {
				POD_can_participate_in_vampire_politics = yes
			}
			set_character_faith = faith:viachristi
		}

		else_if = {
			limit = {
			AND = {
				has_trait = supehunter
				NOT = { has_trait = shih }
			}			
			}
			set_character_faith = faith:akritai
		}

		add_character_modifier = {
			modifier = POD_faith_commited_modifier
			years = 100
		}
		
		add_trait = truefaith
	}

	ai_potential = {

	}

	ai_will_do = {
		base = 0
	}
}
