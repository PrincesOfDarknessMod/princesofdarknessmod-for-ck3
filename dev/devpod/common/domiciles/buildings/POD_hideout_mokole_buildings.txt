mokole_main_meditation_chamber_01 = {
	slot_type = internal
	allowed_domicile_types = { camp haven_landed }
	construction_time = @fera_domicile_internal_construction_duration_t1
	previous_building = camp_main_01

	cost = {
		gold = fera_domicile_internal_gold_cost_t1
		piety = fera_domicile_internal_piety_cost_t1
	}
	refund = {
		gold = {
			value = fera_domicile_internal_gold_cost_t1
			multiply = camp_refund_mult_value
			floor = yes
		}
		piety = {
			value = fera_domicile_internal_piety_cost_t1
			multiply = camp_refund_mult_value
			floor = yes
		}
	}

	can_construct = {
		domicile_building_requirement_mokole = yes
		trigger_if = {
			limit = { NOT = { is_target_in_variable_list = { name = saved_haven_buildings target = flag:mokole_main_meditation_chamber_01 } } }
			custom_tooltip = {
				text = mokole_main_meditation_chamber_01_requirement
				capital_province = { has_building = mokole_wallow_05 }
			}
		}
	}

	parameters = {
		mokole_building = yes
		domicile_gnosis_regen_2_5 = yes
	}

	character_modifier = {
		POD_terrain_gnosis_regen_supply_limit_mult = 0.025
		stress_loss_mult = 0.05
	}

	ai_value = { value = fera_domicile_ai_value }

	asset = {
		icon = "gfx/interface/icons/domicile_building/domicile_building_reference_corpus.dds"
		texture = "gfx/interface/window_domiciles/estate_building_temple_generic_byzantine.dds"
		intersectionmask_texture = "gfx/interface/window_domiciles/estate_building_temple_generic_mask.png"
		soundeffect = "event:/DLC/EP3/SFX/UI/camp_buildings/ep3_ui_domicile_buildings_mess_tent"
