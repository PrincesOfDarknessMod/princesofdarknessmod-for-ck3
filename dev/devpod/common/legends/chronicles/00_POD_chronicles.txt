	chapters = {
		opening = mithras_chronicle_opening
		motivation = mithras_chronicle_motivation
		famous_deed = mithras_chronicle_famous_deed
		extra_deed = mithras_chronicle_extra_deed
		conclusion = mithras_chronicle_conclusion
	}

	impact = {
		owner_modifier = {
		}
		
		on_complete = {
			add_pressed_claim = title:k_persia
			add_pressed_claim = title:k_romagna
			
			character:505071 = {
				if = {
					limit = {
						NOT = { has_trait = saint }
					}
				add_trait = saint
				}
				
				if = {
					limit = {
						NOT = { has_character_modifier = legend_chain_army_modifier }
						is_alive = yes
					}
				add_character_modifier = legend_chain_army_modifier
				}
				
				if = {
					limit = {
						has_trait = lazy
					}
					remove_trait = lazy
					add_trait = diligent
				}
			}
		}		
	}
}
