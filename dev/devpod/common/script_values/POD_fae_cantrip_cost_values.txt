realm_actor_level = {
	if = {
		limit = { has_perk = realm_actor_5_perk }
		add = 5
	}
	else_if = {
		limit = { has_perk = realm_actor_4_perk }
		add = 4
	}
	else_if = {
		limit = { has_perk = realm_actor_3_perk }
		add = 3
	}
	else_if = {
		limit = { has_perk = realm_actor_2_perk }
		add = 2
	}
	else_if = {
		limit = { has_perk = realm_actor_1_perk }
		add = 1
	}
}

# requires scope:cantrip_actor and scope:cantrip_recipient on the TopScope
realm_actor_interaction_requirement = {
	if = { 
		limit = {
			OR = {
				scope:cantrip_actor = scope:cantrip_recipient
				scope:cantrip_recipient = {
					has_relation_animantis_creator = scope:cantrip_actor
				}
			}
		}
		add = 0
	}
	else_if = { 
		limit = { 
			scope:cantrip_recipient = { 
				OR = {
					has_relation_rival = scope:cantrip_actor
					is_at_war_with = scope:cantrip_actor
				}
			}
		}
		add = 5
	}
	else_if = { 
		limit = { 
			scope:cantrip_recipient = { 
				has_any_bad_relationship_with_character_trigger = { CHARACTER = scope:cantrip_actor } 
			}
		}
		add = 4
	}
	else_if = { 
		limit = { 
			scope:cantrip_recipient = { 
				OR = {
					is_imprisoned_by = scope:cantrip_actor
					has_truce = scope:cantrip_actor
					NOR = { 
						has_any_good_relationship_with_character_trigger = { CHARACTER = scope:cantrip_actor }
						is_courtier_of = scope:cantrip_actor
						is_foreign_court_or_pool_guest_of = scope:cantrip_actor
						is_vassal_or_below_of = scope:cantrip_actor
						is_liege_or_above_of = scope:cantrip_actor
						is_allied_to = scope:cantrip_actor
						dynasty ?= scope:cantrip_actor.dynasty
						any_secret = {
							is_known_by = scope:cantrip_actor
						}
						any_held_title = {
							scope:cantrip_actor = { has_claim_on = prev }
						}
						any_character_artifact = {
							scope:cantrip_actor = { has_personal_artifact_claim = prev }
						}
						scope:cantrip_actor = { has_hook = prev }
					}
				}
			}
		}
		add = 3
	}
	else_if = { 
		limit = { 
			scope:cantrip_recipient = { 
				NOR = { 
					has_any_moderate_good_relationship_with_character_trigger = { CHARACTER = scope:cantrip_actor }
					is_courtier_of = scope:cantrip_actor
					is_foreign_court_or_pool_guest_of = scope:cantrip_actor
					is_vassal_or_below_of = scope:cantrip_actor
					is_liege_or_above_of = scope:cantrip_actor
					dynasty ?= scope:cantrip_actor.dynasty
					any_secret = {
						is_known_by = scope:cantrip_actor
					}
					any_held_title = {
						scope:cantrip_actor = { has_claim_on = prev }
					}
					scope:cantrip_actor = { has_hook = prev }
				}
			}
		}
		add = 2
	}
	else_if = {
		limit = {
			scope:cantrip_recipient = {
				NOR = {
					has_any_best_good_relationship_with_character_trigger = { CHARACTER = scope:cantrip_actor }
					is_close_family_of = scope:cantrip_actor
					is_knight_of = scope:cantrip_actor
					AND = {
						is_courtier_of = scope:cantrip_actor
						has_employed_any_court_position = yes
					}
				}
			}
		}
		add = 1
	}
}

# requires scope:cantrip_recipient on the TopScope
realm_fae_interaction_requirement = {
	scope:cantrip_recipient = {
		if = {
			limit = { this = scope:cantrip_actor }
			add = 0
		}
		else_if = {
			limit = {
				OR = {
					POD_is_demon_trigger = yes
					POD_is_mummy_trigger = yes
					POD_is_wraith_trigger = yes
					has_trait = spirit 
				}
			}
			add = 5
		}
		else_if = {
			limit = {
				OR = {
					is_kueijin_trigger = yes
					POD_is_vampire_trigger = yes
					POD_splat_is_fera_trigger = yes
					POD_is_mage_trigger = yes
					
					has_trait = beatific
					POD_is_possessed_trigger = yes
				}
			}
			add = 4
		}
		else_if = { 
			limit = {
				POD_is_fae_trigger = yes
			}
			add = 3
		}
		else_if = {
			limit = {
				OR = {
					POD_is_changeling_trigger = yes
					has_trait = automaton
				}
			}
			add = 2
		}
		else_if = {
			limit = {
				OR = {
					has_trait = asekh_sen
					has_trait = ghoul
					has_trait = revenant
					has_trait = dhampyr
					has_trait = nephilim
					has_trait = laham
					has_trait = kinain
					has_trait = kinfolk
					has_trait = alien
					
					is_sorcerer_trigger = yes
					has_trait = psychic
					has_trait = theurgist
					has_trait = truefaith
				}
			}
			add = 1
		}
		else = {
			add = 0
		}
	}
}

# requires scope:cantrip_target on the TopScope to either be an artifact or not exist
realm_prop_interaction_requirement = {
	if = { 
		limit = { 
			NOT = { exists = scope:cantrip_target }
		}
		add = 0
	}
	else_if = { 
		limit = { 
			scope:cantrip_target = { 
				rarity = illustrious 
				is_unique = yes
			}
		}
		add = 5
	}
	else_if = { 
		limit = { 
			scope:cantrip_target = { 
				rarity = famed 
				is_unique = yes
			}
		}
		add = 4
	}
	else_if = { 
		limit = { 
			scope:cantrip_target = { 
				OR = {
					rarity = illustrious 
					rarity = famed 
				}
			}
		}
		add = 3
	}
	else_if = { 
		limit = { 
			scope:cantrip_target = { 
				OR = {
					artifact_slot_type = primary_armament
					artifact_slot_type = armor
					artifact_slot_type = regalia
					artifact_slot_type = helmet
				}
			}
		}
		add = 2
	}
	else_if = { 
		limit = { 
			scope:cantrip_target = { 
				NOR = {
					has_variable = ai_to_ai_destroy_in_sieges 
					has_variable = unwanted_artifact 
					has_variable = cursed_artifact
				}
			}
		}
		add = 1
	}
}

# requires scope:cantrip_actor and scope:cantrip_recipient on the TopScope
realm_scene_interaction_requirement = {
	if = { 
		limit = { scope:cantrip_actor = scope:cantrip_recipient }
		add = 0
	}
	else_if = {
		# error suppression
		limit = {
			OR = {
				scope:cantrip_actor = { NOT = { exists = location } }
				scope:cantrip_recipient = { NOT = { exists = location } }
			}
		}
		add = 5
	}
	else_if = { 
		limit = { scope:cantrip_actor.location = { squared_distance = { target = scope:cantrip_recipient.location value > squared_distance_huge } } }
		add = 5
	}
	else_if = { 
		limit = { scope:cantrip_actor.location = { squared_distance = { target = scope:cantrip_recipient.location value > squared_distance_large } } }
		add = 4
	}
	else_if = { 
		limit = { scope:cantrip_actor.location = { squared_distance = { target = scope:cantrip_recipient.location value > squared_distance_medium } } }
		add = 3
	}
	else_if = { 
		limit = { scope:cantrip_actor.location = { squared_distance = { target = scope:cantrip_recipient.location value > squared_distance_small } } }
		add = 2
	}
	else_if = { 
		limit = { NOT = { scope:cantrip_actor.location = scope:cantrip_recipient.location } }
		add = 1
	}
}

# requires scope:cantrip_actor and scope:cantrip_target on the TopScope
realm_scene_area_interaction_requirement = {
	if = { 
		limit = { exists = scope:cantrip_target_continent }
		add = 5
	}
	else_if = { 
		limit = {
			OR = {
				NOT = { exists = scope:cantrip_target }
				scope:cantrip_target = { tier = tier_barony }
			}
		}
		add = 0
	}
	else_if = { 
		limit = { scope:cantrip_target = { tier = tier_county } }
		add = 1
	}
	else_if = { 
		limit = { scope:cantrip_target = { tier = tier_duchy } }
		add = 2
	}
	else_if = { 
		limit = { scope:cantrip_target = { tier = tier_kingdom } }
		add = 3
	}
	else_if = { 
		limit = { scope:cantrip_target = { tier = tier_empire } }
		add = 4
	}
	else = {
		limit = { scope:cantrip_target = { tier = tier_hegemony } }
		add = 5
	}
