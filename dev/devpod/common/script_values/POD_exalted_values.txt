POD_essence_fluctuation = {
	add = {
		value = POD_essence_fluctuation_base
		desc = BASE_VALUE
		format = "POD_FORMAT_PERCENTAGE_BASE"
	}
	if = {
		limit = { any_directly_owned_province = { has_building_or_higher = dragon_nest_01 } }
		add = {
			value = POD_essence_fluctuation_dragon_nest
			desc = POD_essence_fluctuation_dragon_nest
			format = "POD_FORMAT_PERCENTAGE_MODIFIER_POSITIVE_NO_DOT"
		}
	}
	if = {
		limit = { is_in_umbra_trigger = yes }
		add = {
			value = POD_essence_fluctuation_umbra
			desc = POD_essence_fluctuation_umbra
			format = "POD_FORMAT_PERCENTAGE_MODIFIER_POSITIVE_NO_DOT"
		}
	}
	if = {
		limit = { has_focus = charms_focus }
		add = {
			value = 5
			desc = charms_focus
			format = "POD_FORMAT_PERCENTAGE_MODIFIER_POSITIVE_NO_DOT"
		}
	}
	if = {
		limit = { 
			var:current_journey ?= { # Only active journey is counted.
				var:journey_type = flag:exalted_essence_meditation
			}
		}
		add = {
			value = POD_essence_fluctuation_meditation_journey
			desc = exalted_essence_meditation_name
			format = "POD_FORMAT_PERCENTAGE_MODIFIER_NEGATIVE_NO_DOT"
		}
	}
	if = {
		limit = { has_perk = exalted_elemental_empowerment_meditation_perk }
		add = {
			value = 10
			desc = exalted_elemental_empowerment_meditation_perk_name
			format = "POD_FORMAT_PERCENTAGE_MODIFIER_POSITIVE_NO_DOT"
		}
	}
	add = POD_essence_fluctuation_caste
	add = POD_essence_fluctuation_buildings
	add = {
		value = POD_essence_fluctuation_charms
		desc = POD_essence_fluctuation_active_passive_charms
		format = "POD_FORMAT_PERCENTAGE_MODIFIER_NEGATIVE_NO_DOT"
	}
}

POD_dragon_touched_exaltation_chance_child = {
	if = {
		limit = { exists = real_father exists = mother }
		if = {
			limit = { real_father.POD_dragon_touched_exaltation_chance > mother.POD_dragon_touched_exaltation_chance }
			value = real_father.POD_dragon_touched_exaltation_chance
		}
		else = {
			value = mother.POD_dragon_touched_exaltation_chance
		}
	}
	else_if = {
		limit = { exists = real_father }
		value = real_father.POD_dragon_touched_exaltation_chance
	}
	else_if = {
		limit = { exists = mother }
		value = mother.POD_dragon_touched_exaltation_chance
	}
}

			}
		}
		
		min = {

	}
