POD_dragon_touched_exaltation_chance_child = {
	if = {
		limit = { exists = real_father exists = mother }
		if = {
			limit = { real_father.POD_dragon_touched_exaltation_chance > mother.POD_dragon_touched_exaltation_chance }
			value = real_father.POD_dragon_touched_exaltation_chance
		}
		else = {
			value = mother.POD_dragon_touched_exaltation_chance
		}
	}
	else_if = {
		limit = { exists = real_father }
		value = real_father.POD_dragon_touched_exaltation_chance
	}
	else_if = {
		limit = { exists = mother }
		value = mother.POD_dragon_touched_exaltation_chance
	}
	if = {
		limit = {
			real_father ?= { is_ai = yes }
			mother ?= { is_ai = yes }
		}
		add = 10 # Ai gets additional 10%
	}
}
