POD_essence_fluctuation_base = 25
POD_essence_fluctuation_caste_condition = 25
POD_essence_fluctuation_dragon_nest = 30
POD_essence_fluctuation_umbra = 30

POD_essence_fluctuation = {
	add = {
		value = POD_essence_fluctuation_base
		desc = BASE_VALUE
		format = "POD_FORMAT_PERCENTAGE_BASE"
	}
	if = {
		limit = { any_directly_owned_province = { has_building_with_flag = dragon_nest } }
		add = {
			value = POD_essence_fluctuation_dragon_nest
			desc = POD_essence_fluctuation_dragon_nest
			format = "POD_FORMAT_PERCENTAGE_MODIFIER"
		}
	}
	if = {
		limit = { is_in_umbra_trigger = yes }
		add = {
			value = POD_essence_fluctuation_umbra
			desc = POD_essence_fluctuation_umbra
			format = "POD_FORMAT_PERCENTAGE_MODIFIER"
		}
	}
}
POD_essence_fluctuation_normalized = {
	value = POD_essence_fluctuation
	if = { limit = { POD_essence_fluctuation < 0 } multiply = -1 }
}

	}
	add = {
		value = POD_essence_fluctuation_umbra

		format = "POD_FORMAT_PERCENTAGE_MODIFIER"
	}
}
