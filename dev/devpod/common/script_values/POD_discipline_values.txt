max_ghoul_cap = {
	value = 3
	#divide = 8
	floor = yes
	#add = 3
	if = {
		limit = { has_character_modifier = ritual_brand_of_the_paramour_modifier }
		add = 1
	}
	if = {
		limit = { 
			has_trait_xp = {
				trait = ashen_adventurer
				track = ashen_cult
				value >= 25
			}		
			}
		add = 1
	}	
	if = {
		limit = { 
			has_trait_xp = {
				trait = ashen_adventurer
				track = ashen_cult
				value >= 75
			}		
			}
		add = 1
	}		
	if = { 
		limit = { 
			dynasty ?= { has_dynasty_perk = tzimisce_legacy_1 }
		}
		add = 2
	}	
	if = {
		limit = {
		domicile ?= {
			has_domicile_parameter = max_ghoul_size_3
		}
		}
		add = 3
	}
	else_if = {
		limit = {
		domicile ?= {
			has_domicile_parameter = max_ghoul_size_2
		}
		}
		add = 2
	}
	else_if = {
		limit = {
		domicile ?= {
			has_domicile_parameter = max_ghoul_size_1
		}
		}
		add = 1
	}		
}
