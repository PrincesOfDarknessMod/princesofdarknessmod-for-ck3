mummy_ressurection_piety_value = {
	value = 2

	add = num_sinful_traits
	subtract = num_virtuous_traits
	
	multiply = 500
	min = 500
}

mummy_resurrect_piety_loss = {
	value = 0
	subtract = mummy_ressurection_piety_value
	
	if = {
		limit = { has_perk = effigy_of_sin_perk }
		add = {
			value = mummy_ressurection_piety_value
			multiply = 0.4
		}
	}
	if = {
		limit = { has_perk = eating_the_heart_perk }
		add = {
			value = mummy_ressurection_piety_value
			multiply = 0.2
		}
	}
	
	ceiling = yes
}
