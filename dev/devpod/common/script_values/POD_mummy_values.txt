mummy_ressurection_piety_value = {
	value = 2

	add = num_sinful_traits
	subtract = num_virtuous_traits
	
	multiply = 500
	min = 500
	if = {
		limit = { 
			OR = {
				faith = { has_doctrine = mummy_special_kingdom_hades_doctrine }
				faith = { has_doctrine = mummy_special_kingdom_hellheim_doctrine }
				faith = { has_doctrine = mummy_special_kingdom_yomi_doctrine }
				faith = { has_doctrine = mummy_special_kingdom_hell_doctrine }
			} 
		}
		multiply = 2
	}
	if = {
		limit = { 
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:separation_from_maat_objective_decision
			}
			OR = {
				AND = {
					has_trait = shemsuheru
					NOR = {
						faith = { has_doctrine = mummy_special_kingdom_hades_doctrine }
						faith = { has_doctrine = mummy_special_kingdom_hellheim_doctrine }
						faith = { has_doctrine = mummy_special_kingdom_yomi_doctrine }
						faith = { has_doctrine = mummy_special_kingdom_hell_doctrine }
					} 
				}
				AND = {
					has_trait = cabiri
					NOR = {
						faith = { has_doctrine = mummy_special_kingdom_hades_doctrine }
						faith = { has_doctrine = mummy_special_kingdom_hellheim_doctrine }
						faith = { has_doctrine = mummy_special_kingdom_yomi_doctrine }
						faith = { has_doctrine = mummy_special_kingdom_hell_doctrine }
					} 
				}
			}
		}
		multiply = 2
	}
}

mummy_resurrect_piety_loss = {
	value = 0
	subtract = mummy_ressurection_piety_value
	
	if = {
		limit = { has_perk = effigy_of_sin_perk }
		add = {
			value = mummy_ressurection_piety_value
			multiply = 0.4
		}
	}
	if = {
		limit = { has_perk = eating_the_heart_perk }
		add = {
			value = mummy_ressurection_piety_value
			multiply = 0.2
		}
	}
	
	ceiling = yes
}
