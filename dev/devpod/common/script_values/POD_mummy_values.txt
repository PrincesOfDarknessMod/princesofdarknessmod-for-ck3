POD_mummy_spell_of_life_xp_passive = {
	value = { value = 0.05 desc = BASE }

	if = { limit = { has_character_flag = spell_of_life_stage1_setite_phase3_decision }
		add = { value = 1 desc = MUMMY_SOL_PASSIVE_EXPERIMENTATION }
	}

	# You are scholarly
	# 0.0002x^2 - 0.0007x
	if = { limit = { learning >= 20 }
		add = { 
			value = {
				value = learning
				multiply = learning
				multiply = 0.0002
				subtract = {
					value = learning
					multiply = 0.0007
				}
			} 
			desc = MUMMY_SOL_PASSIVE_LEARNING_STAT
		}
	}

	if = {
		limit = { has_trait = education_learning }
		add = {
			value = 0.01
			if = {
				limit = { has_trait = education_learning_2 }
				add = 0.09
			}
			else_if = {
				limit = { has_trait = education_learning_3 }
				add = 0.24
			}
			else_if = {
				limit = { has_trait = education_learning_4 }
				add = 0.49
			}
			else_if = {
				limit = { has_trait = education_learning_5 }
				add = 0.99
			}
			desc = MUMMY_SOL_PASSIVE_LEARNING_EDUCATION
		}
	}
	if = { limit = { has_trait = scholar }
		add = { value = 0.1 desc = MUMMY_SOL_PASSIVE_SCHOLAR_TRAIT }
	}
	if = { limit = { has_trait = theologian }
		add = { value = 0.1 desc = MUMMY_SOL_PASSIVE_THEOLOGIAN_TRAIT }
	}
	if = { limit = { has_trait = lifestyle_mystic }
		add = { value = 0.05 desc = MUMMY_SOL_PASSIVE_MYSTIC_TRAIT }
	}
	if = { limit = { has_trait = lifestyle_physician }
		add = { value = 0.05 desc = MUMMY_SOL_PASSIVE_PHYSICIAN_TRAIT }
	}
	if = { limit = { has_trait = journaller }
		add = { value = 0.05 desc = MUMMY_SOL_PASSIVE_JOURNALLER_TRAIT }
	}
	if = { limit = { has_trait = shrewd }
		add = { value = 0.05 desc = MUMMY_SOL_PASSIVE_SHREWD_TRAIT }
	}
	if = {
		limit = { has_trait = intellect_good }
		add = {
			value = 0.05
			if = {
				limit = { has_trait = intellect_good_2 }
				add = 0.2
			}
			else_if = {
				limit = { has_trait = intellect_good_3 }
				add = 0.45
			}
			desc = MUMMY_SOL_PASSIVE_INTELLECT_TRAIT
		}
	}

	# You have a magical trait
	if = { limit = { OR = {
		has_trait = sorcerer
		has_trait = theurgist
		has_trait = psychic
		has_trait = bloodsorcerydiscipline
		has_trait = necromancydiscipline
	} }
		add = { value = 0.1 desc = MUMMY_SOL_PASSIVE_MAGICAL_TRAIT }
	}

	# You come from a scholarly background
	if = { limit = { faith = { has_doctrine = tenet_knowledge_above_all } }
		add = { value = 0.05 desc = MUMMY_SOL_PASSIVE_FAITH_KNOWLEDGE_ABOVE_ALL }
	}
	if = { limit = { faith = { has_doctrine = tenet_pursuit_of_knowledge } }
		add = { value = 0.05 desc = MUMMY_SOL_PASSIVE_FAITH_PURSUIT_OF_KNOWLEDGE }
	}
	if = { limit = { culture = { has_cultural_tradition = tradition_philosopher_culture } }
		add = { value = 0.05 desc = MUMMY_SOL_PASSIVE_CULTURE_PHILOSOPHER_CULTURE }
	}

	# Your realm is advanced
	if = { limit = { total_your_realm_development >= 350 }
		add = { value = 1 desc = MUMMY_SOL_PASSIVE_HIGH_DEVELOPMENT }
	}
	# Broken for Goratrix
	if = { limit = { total_your_realm_development >= 700 }
		add = { value = 2 desc = MUMMY_SOL_PASSIVE_SUPER_HIGH_DEVELOPMENT }
	}

	# You can draw from your homeland's power
	if = { limit = { has_trait = mummy }
		if = { limit = { POD_is_in_homeland_trigger = yes }
			add = { value = 0.25 desc = MUMMY_SOL_PASSIVE_HOMELAND }
		}
	}

	# You own artifacts that grant some insight
	if = { 
		limit = { 
			POD_owns_all_egyptian_artifacts = yes 
			any_artifact = { has_variable = osiris_flail }
		}
		add = { value = 2 desc = MUMMY_SOL_PASSIVE_EGYPTIAN_ARTIFACTS }
	}
	if = { 
		limit = { POD_character_owns_linked_artifact = { LINK = writings_cabirus_link } }
		add = { value = 0.25 desc = MUMMY_SOL_PASSIVE_CABIRUS_ARTIFACT }
	}

	# True names hold power
	# 0.0004x^2 + 0.0012x
	add = { 
		value = {
			value = {
				value = known_trait_true_names_number
				add = known_nonstandard_true_names_number
			}
			multiply = {
				value = known_trait_true_names_number
				add = known_nonstandard_true_names_number
			}
			multiply = 0.0004
			add = {
				value = {
					value = known_trait_true_names_number
					add = known_nonstandard_true_names_number
				}
				multiply = 0.0012
			}
		} 
		desc = MUMMY_SOL_PASSIVE_KNOWN_TRUE_NAMES
	}

	# Knowing more languages makes understanding the scraps easier
	#0.0009x^2 + 0.0015x - 0.0024
	add = { 
		value = {
			value = num_of_known_languages
			multiply = num_of_known_languages
			multiply = 0.0009
			add = {
				value = num_of_known_languages
				multiply = 0.0015
			}
			subtract = 0.0024
		} 
		desc = MUMMY_SOL_PASSIVE_KNOWN_LANGUAGES
	}

	# Control the counties where Osiris' parts were scattered
	# These are the 14 places Osiris was scattered per Plutarch/Herodotus and the Mummy books.
	# Some historical sources list 16 places, but these aren't the cities listed in the Mummy books so I didn't use them.
	# Determining the right county for each spot was tricky. I checked for archeological digs if the name was known.
	# Otherwise, I hunted down other writings from around that time to find the nomes and went off of the capital city of each nome.
	# That being said, I'm not an egyptologist and translating these to CK3 was tricky for me so these are my best guesses.
	# For example, Het-sera in the city of Netert lacks an associated ruin or even a guess based on clues regarding nomes in other writings.
	# Also, these are really generic names with "netert" basically meaning "goddess" or at least "female divine energy". "Het" just means "house". "Sera" could be Hebrew for "princess" or "fiery".
	# It could also refer to the Serapeum of Saqqara or the Serapeum of Alexandria which honored the Greco-Egyptian syncretic god Serapis (mix of Zeus, Hades, Osiris, and Apis; consort of Isis).
	# Interestingly, the Talmud (well, one version) claims that Serapis was inspired by the patriarch Joseph.
	# Anyway, based on all of this, I decided that it might refer to a center of worship for Isis and/or Osiris which would point at Alexandria, Abydos, Aswan, or Behbeit El Hagar.
	# The counties of Aswan, Behbeit El Hagar, and Alexandria are already covered by other spots. As such, I decided to assume that it was Abydos and added the appropriate county.
	# The large size of CK3 counties thankfully means that quite a few places fall within the same counties which makes this source of advantages more balanced and made the historical aspect easier to track down.
	if = { limit = { has_title = title:c_aswan }
		add = { value = 0.1 desc = MUMMY_SOL_PASSIVE_ASWAN }
	}
	if = { limit = { has_title = title:c_asyut }
		add = { value = 0.1 desc = MUMMY_SOL_PASSIVE_ASYUT }
	}
	if = { limit = { has_title = title:c_cairo }
		add = { value = 0.1 desc = MUMMY_SOL_PASSIVE_CAIRO }
	}
	if = { limit = { has_title = title:c_al-buhaira }
		add = { value = 0.1 desc = MUMMY_SOL_PASSIVE_AL_BUHAIRA }
	}
	if = { limit = { has_title = title:c_giza }
		add = { value = 0.1 desc = MUMMY_SOL_PASSIVE_GIZA }
	}
	if = { limit = { has_title = title:c_al-gharbiya }
		add = { value = 0.1 desc = MUMMY_SOL_PASSIVE_AL_GHARBIYA }
	}
	if = { limit = { has_title = title:c_alexandria }
		add = { value = 0.1 desc = MUMMY_SOL_PASSIVE_ALEXANDRIA }
	}
	if = { limit = { has_title = title:c_bahnasa }
		add = { value = 0.1 desc = MUMMY_SOL_PASSIVE_BAHNASA }
	}
	if = { limit = { has_title = title:c_qus }
		add = { value = 0.1 desc = MUMMY_SOL_PASSIVE_QUS }
	}

	# Bonus for how many mummies (regardless of type) are friend/lover/coterie member/captured
	if = {
		limit = {
			any_prisoner = {
				count >= 1
				has_trait = mummy
			}
		}
		every_prisoner = {
			limit = { has_trait = mummy }
			add = { value = 0.01 desc = MUMMY_SOL_PASSIVE_MUMMY_PRISONER }
		}
	}
	if = {
		limit = {
			any_relation = {
				type = friend
				count >= 1
				has_trait = mummy
			}
		}
		every_relation = {
			limit = { has_trait = mummy }
			add = { value = 0.01 desc = MUMMY_SOL_PASSIVE_MUMMY_FRIEND }
		}
	}
	if = {
		limit = {
			any_relation = {
				type = best_friend
				count >= 1
				has_trait = mummy
			}
		}
		every_relation = {
			limit = { has_trait = mummy }
			add = { value = 0.01 desc = MUMMY_SOL_PASSIVE_MUMMY_BEST_FRIEND }
		}
	}
	if = {
		limit = {
			any_relation = {
				type = lover

	if = {

POD_mummy_spell_of_life_xp_mult = {
	value = 1
	# Having lots of memories makes things easier
	if = { limit = { has_trait_xp = { trait = mummy value >= 90 } } add = 0.5 }

	# Apply penalties for low years since maelstrom due to chaos
	if = { limit = { years_since_great_maelstrom <= 5 } subtract = 0.5 }
	else_if = { limit = { years_since_great_maelstrom <= 50 } subtract = 0.25 }
	# Apply penalties for every year that passes from 1230 as the trails grow colder and knowledge is lost
	subtract = {
		value = years_since_1230_thousandths
		divide = 100
	}

	# Apply penalty if not Amenti religion, Osirian League, Opposed to Apophis, one of the 2 setite pillar things, or Servant of Apophis
	# You don't really care
	if = { 
		limit = { 
			NOR = {
				faith = { religion_tag = amenti_religion }
				faith = { has_doctrine = special_doctrine_osirian_league }
				faith = { has_doctrine  = special_doctrine_against_to_apophis_hostility }
				faith = { has_doctrine  = special_doctrine_allied_to_apophis_hostility }
				faith = { has_doctrine = special_doctrine_set_pillars_kindred }
				faith = { has_doctrine = special_doctrine_set_corrupted_kindred }
			} 
		} 
		subtract = 0.25
	}

	# Duchy buildings - places of learning
	if = {
		limit = {
			OR = {
				any_directly_owned_province = { has_building_or_higher = prince_city_06 }
				any_directly_owned_province = { has_building_or_higher = occult_affiliation_03 }
				any_directly_owned_province = { has_building_or_higher = web_of_faith_03 }
				any_directly_owned_province = { has_building_or_higher = caern_wisdom_03 }
			}
		}
		add = 0.15
	}
	# Where mummies were made
	if = {
		limit = {
			any_directly_owned_province = { has_building_or_higher = mestha_farmstead_02 }
		}
		add = 0.15
	}
	# Universities and libraries
	if = {
		limit = {
			OR = {
				any_directly_owned_province = { has_building = generic_university }
				any_directly_owned_province = { has_building_or_higher = black_college_02 }
				any_directly_owned_province = { has_building_or_higher = library_malta_02 }
				any_directly_owned_province = { has_building = siena_university }
				any_directly_owned_province = { has_building_or_higher = vienna_chantry_02 }
				any_directly_owned_province = { has_building_or_higher = house_of_wisdom_01 }
				any_directly_owned_province = { has_building_or_higher = erciyes_02 }
			}
		}
		add = 0.25
	}
	
	min = 0
}
