	scope:actor = {
		switch = {
			trigger = has_perk
			realm_time_5_perk = { value = 0 }   # 0 means infinite
			realm_time_4_perk = { value = 120 } # 10 years
			realm_time_3_perk = { value = 60 }  # 5 years
			realm_time_2_perk = { value = 24 }  # 2 years
			realm_time_1_perk = { value = 12 }  # 1 year
			fallback          = { value = 4 }   # 4 months
		}
	}
}
