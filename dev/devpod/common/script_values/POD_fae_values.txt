					}

			}
			desc = YEARLY_BANALITY_FAITH_REALM
			format = "POD_FORMAT_NEGATIVE_PLUS"
		}
	}

		if = {

			}
			add = { 
				value = 5

			}
		}
	}

					name = unavailable_unique_decisions

				}
			}
		}

		add = { 
			value = {
				value = 0
				location ?= {
					culture = {
						switch = {
							trigger = has_cultural_era_or_later

				}
			}

			format = "POD_FORMAT_NEGATIVE_PLUS"
		}
	}
	if = {

			NOT = {
