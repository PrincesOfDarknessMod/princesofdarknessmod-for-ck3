POD_aging_multiplier = {
	if = {
		limit = { POD_life_expectancy_offset != 0 }
		add = POD_graceful_aging_default_period
		divide = {
			add = POD_graceful_aging_default_period
			add = POD_life_expectancy_offset
		}
	}
	else = {
		add = 1
	}
}

POD_graceful_aging_default_period = {
	value    = define:NPortrait|GRACEFUL_AGING_END
	subtract = define:NPortrait|GRACEFUL_AGING_START
}

POD_graceful_aging_adjusted_end = {
	add = define:NPortrait|GRACEFUL_AGING_END
	add = POD_life_expectancy_offset
}

POD_visual_age = {
	if = {
		limit = { is_immortal = yes }
		add = effective_age
	}
	# age normally before 25
	else_if = {
		limit = { age <= define:NPortrait|GRACEFUL_AGING_START }
		add = age
	}
	# start aging normally after reaching the equivalent of 70
	else_if = {
		limit = { age >= POD_graceful_aging_adjusted_end }
		add = define:NPortrait|GRACEFUL_AGING_END
		add = age
		subtract = POD_graceful_aging_adjusted_end
	}
	# interpolate values inbetween
	else = {
		add = define:NPortrait|GRACEFUL_AGING_START
		add = {
			value = age
			subtract = define:NPortrait|GRACEFUL_AGING_START
			multiply = POD_aging_multiplier
		}
	}
}

POD_needed_longevity = {
