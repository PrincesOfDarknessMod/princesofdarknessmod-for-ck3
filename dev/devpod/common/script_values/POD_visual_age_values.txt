POD_aging_multiplier = {
	if = {
		limit = { POD_life_expectancy_offset != 0 }
		add = 45 # GRACEFUL_AGING_END - GRACEFUL_AGING_START defines
		divide = {
			add = 45
			add = POD_life_expectancy_offset
		}
	}
	else = {
		add = 1
	}
}

POD_graceful_aging_end = {
	add = 70 # GRACEFUL_AGING_END define
	add = POD_life_expectancy_offset
}

POD_visual_age = {
	if = {
		limit = { is_immortal = yes }
		add = effective_age
	}
	# age normally before 25
	else_if = {
		limit = { age <= 25 } # GRACEFUL_AGING_START define
		add = age
	}
	# start aging normally after reaching the equivalent of 70
	else_if = {
		limit = { age >= POD_graceful_aging_end }
		add = 70
		add = age
		subtract = POD_graceful_aging_end
	}
	# interpolate values inbetween
	else = {
		add = 25
		add = {
			value = age
			subtract = 25
			multiply = POD_aging_multiplier
		}
	}
}
