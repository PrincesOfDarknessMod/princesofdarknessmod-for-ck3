POD_social_discipline_mult = {
	value = 1
	if = {
		limit = { 
			OR = {
				has_trait = daimoinonadvanced
				has_perk = as_below_so_above_perk
			}
		}
		add = 0.5
	}
	
	if = {
		limit = {
			exists = scope:target
			trigger_if = {
				limit = {
					exists = scope:target
				}
				scope:target = { has_character_flag = song_of_jupiter }
			}
		}
		subtract = 0.9
		
	}
	
}

		limit = { has_trait = dharma0 }

			value = 5

		}
	}
	if = {
		limit = {

			value = 0
			desc = gilgul_perk_name
		}
	}
	add = {
		desc = "BASE"
		value = POD_base_value
	}
}
