yin_difference = {
	value = 0
	if = { 
		limit = { 
			OR = {
				yin_cap > 10
			}
		}
		add = 10
	}
	else = {
		add = yin_cap
	}
	subtract = yin_points_value
}

sexagenary_year = {
	value = current_year
	subtract = 3
	modulo = 12
	if = {
		limit = { 
			current_month = 1
			sexagenary_day_lichun <= 15
			sexagenary_current_day_to_lichun > sexagenary_day_lichun
		}
		subtract = 1
	}
	else_if = {
		limit = { 
			current_month = 2
			OR = { 
				AND = {
					sexagenary_day_lichun < 5
					sexagenary_current_day_to_lichun > sexagenary_day_lichun
				}
				AND = {
					sexagenary_day_lichun > 15
					sexagenary_lichun_to_current_day < sexagenary_day_lichun
				}
			}
		}
		subtract = 1
	}
}
