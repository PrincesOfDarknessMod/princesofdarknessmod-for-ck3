patron_council_zeal = {
	value = global_var:patron_council_influence
	if = {
		limit = { 
			POD_is_philospher_king_trigger = no 
			POD_is_philospher_queen_trigger = no 
		}
		divide = 10
		floor = yes
	}
}

patron_council_influence_value = {
	value = 0
	add = {
		value = patron_council_county_value
		desc = patron_council_county_mod
	}
	
	if = {
		limit = { title:e_season_courts = { is_title_created = yes } }
		add = {
			value = 100
			desc = e_season_courts
		}
	}
	
	if = {
		limit = { global_banality = 0 }
		multiply = {
			value = 5
			desc = PATRON_COUNCIL_WINDOW_ERA
		}
	}
	else_if = {
		limit = { global_banality = 2 }
		multiply = {
			value = 0
			desc = PATRON_COUNCIL_WINDOW_ERA
		}
	}
	else = {
		multiply = {
			value = 2
			desc = PATRON_COUNCIL_WINDOW_ERA
		}
	}
}

patron_council_county_value = {
    value = 0
    every_province = {
        limit = {
            is_capital_barony = yes
            county = {
                holder = { POD_is_patron_trigger = yes }
            }
        }
        add = 1
    }
}
