# The number all other energy values are scaled against.
# 10 energy points are equivalent to a full "mana bar".
maximum_splat_agnostic_energy = 10

maximum_blood		= 5
maximum_angst		= 5
maximum_pathos		= 5
maximum_essence		= 10
maximum_resolve		= 10
maximum_gnosis		= 5
maximum_glamour		= 5
# used for GUI scaling purposes (assuming a "mana bar" that goes from 0 to 10: 10 dread = 1 energy)
maximum_dread		= 100
# used for GUI scaling purposes (assuming a "mana bar" that goes from 0 to 10: 10 stress = 1 energy)
maximum_willpower	= 100

maximum_gnosis_kinfolk = 3
maximum_essence_proper = {
	value = POD_exalted_rating
	multiply = 2
	max = 10
}

# also see the priority list in common\scripted_effects\POD_energy_effects.txt

# Used as reference for translating splat-specific energy values to splat-agnostic ones
maximum_splat_energy = {
	if = {
		limit = { POD_splat_has_hunger_trigger = yes }
		value = maximum_blood
	}
	else_if = {
		limit = { POD_splat_has_angst_trigger = yes }
		value = maximum_angst
	}
	else_if = {
		limit = { POD_splat_has_pathos_trigger = yes }
		value = maximum_pathos
	}
	else_if = {
		limit = { POD_splat_has_essence_trigger = yes }
		value = maximum_essence
	}
	else_if = {
		limit = { POD_splat_has_resolve_trigger = yes }
		value = maximum_resolve
	}
	else_if = {
		limit = { POD_splat_has_gnosis_trigger = yes }
		value = maximum_gnosis
	}
	else_if = {
		limit = { POD_splat_has_glamour_trigger = yes }
		value = maximum_glamour
	}
	else_if = {
		limit = { POD_splat_has_dread_as_resource_trigger = yes }
		# note that your actual maximum dread can change based on government settings
		value = maximum_dread
	}
	else = {
		# note that your actual maximum stress for ability triggers is different (stress_cap_for_willpower)
		value = maximum_willpower
	}
}

# The actual energy usable by splat abilities
# Main difference is that it uses 3 full stress bars (0-300) instead of just one
maximum_splat_energy_reserves = {
	if = {
		limit = { POD_splat_has_hunger_trigger = yes }
		value = maximum_blood
	}
	else_if = {
		limit = { POD_splat_has_angst_trigger = yes }
		value = maximum_angst
	}
	else_if = {
		limit = { POD_splat_has_pathos_trigger = yes }
		value = maximum_pathos
	}
	else_if = {
		limit = { POD_splat_has_essence_trigger = yes }
		value = maximum_essence_proper
	}
	else_if = {
		limit = { POD_splat_has_resolve_trigger = yes }
		value = maximum_resolve
	}
	else_if = {
		limit = { POD_splat_has_gnosis_trigger = yes }
		if = {
			limit = { POD_is_kinfolk_trigger = yes }
			value = maximum_gnosis_kinfolk
		}
		else = {
			value = maximum_gnosis
		}
	}
	else_if = {
		limit = { POD_splat_has_glamour_trigger = yes }
		value = maximum_glamour
	}
	else_if = {
		limit = { POD_splat_has_dread_as_resource_trigger = yes }
		# note that your actual maximum dread can change based on government settings
		# not sure if there's a way to check your actual max dread value
		value = maximum_dread
	}
	else = {
		value = stress_cap_for_willpower
	}
}

# Energy stored in energy vessels assumes a "mana bar" that goes from 0 to 10
# So e.g. for resources like blood that go from 0 to 5, the value is multiplied by 0.5
splat_energy_multiplier = {
	value = maximum_splat_energy
	divide = maximum_splat_agnostic_energy
}

inverse_splat_energy_multiplier = {
	value = 1
	divide = splat_energy_multiplier
}

splat_energy_points_value = {
	if = {
		limit = { POD_splat_has_hunger_trigger = yes }
		value = blood_level
	}
	else_if = {
		limit = { POD_splat_has_angst_trigger = yes }
		value = angst_level
	}
	else_if = {
		limit = { POD_splat_has_pathos_trigger = yes }
		value = pathos_level
	}
	else_if = {
		limit = { POD_splat_has_essence_trigger = yes }
		value = essence_level
	}
	else_if = {
		limit = { POD_splat_has_resolve_trigger = yes }
		value = resolve_level
	}
	else_if = {
		limit = { POD_splat_has_gnosis_trigger = yes }
		value = gnosis_level
	}
	else_if = {
		limit = { POD_splat_has_glamour_trigger = yes }
		value = glamour_level
	}
	else_if = {
		limit = { POD_splat_has_dread_as_resource_trigger = yes }
		value = dread
	}
	else = {
		value = willpower_level
	}
}

splat_agnostic_energy_points_value = {
	value = splat_energy_points_value
	multiply = inverse_splat_energy_multiplier
	floor = yes
}

splat_energy_deficit = {
	value = maximum_splat_energy_reserves
	subtract = splat_energy_points_value
}

splat_agnostic_energy_deficit = {
	value = splat_energy_deficit
	multiply = inverse_splat_energy_multiplier
	floor = yes
}

splat_agnostic_energy_deficit_including_vessels = {
	value = splat_agnostic_energy_deficit
	add = total_storable_vessel_empty_energy
}

total_stored_splat_energy = {
	value = splat_energy_multiplier
	save_temporary_value_as = energy_mult
	value = 0
	every_character_artifact = {
		limit = {
			has_variable = energy_stored
			artifact_owner = { can_benefit_from_artifact = prev }
		}
		add = {
			value = var:energy_stored
			multiply = scope:energy_mult
			floor = yes
		}
	}
}

# cannot spend static_stress if your stress would end up above this value
stress_cap_for_willpower = 300

willpower_level = {
	value = stress_cap_for_willpower
	subtract = stress
	min = 0
}

# splat-agnostic value
vessel_empty_energy = {
	value = var:max_energy_stored
	subtract = var:energy_stored
}

# to make sure that "half" energy points get filled
# splat-agnostic value
storable_vessel_empty_energy = {
	value = vessel_empty_energy
	add = {
		value = vessel_empty_energy
		modulo = artifact_owner.POD_full_energy_point_for_vessel
	}
}

# to make sure that "half" energy points get filled
# splat-agnostic value
total_storable_vessel_empty_energy = {
	# only calculate this once
	value = POD_full_energy_point_for_vessel
	save_temporary_value_as = full_energy_point
	value = 0
	every_character_artifact = {
		limit = {
			has_variable = max_energy_stored
			artifact_owner = { can_benefit_from_artifact = prev }
		}
		# based on storable_vessel_empty_energy
		add = vessel_empty_energy
		add = {
			value = vessel_empty_energy
			modulo = scope:full_energy_point
		}
	}
}

# to make sure that "half" energy points get ignored
# splat-agnostic value
total_usable_vessel_energy = {
	# only calculate this once
	value = POD_full_energy_point_for_vessel
	save_temporary_value_as = full_energy_point
	value = 0
	every_character_artifact = {
		limit = {
			has_variable = energy_stored
			artifact_owner = { can_benefit_from_artifact = prev }
		}
		if = {
			limit = { var:energy_stored < scope:full_energy_point }
			add = 0
		}
		else = {
			add = var:energy_stored
		}
	}
}

splat_adjusted_storable_vessel_empty_energy = {
	value = vessel_empty_energy
	multiply = artifact_owner.splat_energy_multiplier
	ceiling = yes
}

splat_adjusted_usable_vessel_energy = {
	value = var:energy_stored
	multiply = artifact_owner.splat_energy_multiplier
	floor = yes
}

# How much splat-agnostic energy counts as a "full" energy point for your splat?
POD_full_energy_point_for_vessel = {
	if = {
		limit = { inverse_splat_energy_multiplier > 1 }
		value = inverse_splat_energy_multiplier
	}
	else = {
		value = 1
	}
}
