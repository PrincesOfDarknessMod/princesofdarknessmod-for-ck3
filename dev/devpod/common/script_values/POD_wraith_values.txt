wraith_maintenance_stress_impact_gain = {
	value = 0
	
	every_relation = {
		type = bound_wraith
		if = {
			limit = {
				is_in_umbra_trigger = no
				is_landed = no
				NOT = { has_trait = risen }
				NOT = { has_trait = lament }
				NOT = { has_trait = torpor }
			}
			add = 10
		}
	}
}

}

	value = 0

				trait = wraith
				track = fetter

		}
		add = 10
	}
	else_if = {
		limit = { 
			has_trait_xp = {
				trait = wraith
				track = fetter

			} 
		}
