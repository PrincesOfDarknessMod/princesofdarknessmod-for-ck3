	value = 0
	
	every_relation = {
		type = bound_wraith
		if = {
			limit = {
				is_in_umbra_trigger = no
				is_playable_character = no
				NOT = { has_trait = risen }
				NOT = { has_trait = torpor }
			}
			add = {
				value = 10
				add = {
					value = -1
					multiply = prev.POD_necromancy_castigate_stacks
				}
			}
		}
	}
	if = { 
		limit = { has_perk = ns_reveal_the_catene_perk }
		divide = 2
	}
	if = { 
		limit = { has_perk = ns_tread_upon_the_grave_perk }
		divide = 2
	}
	if = { 
		limit = { 
			OR = {
				has_perk = ns_ephemeral_binding_perk 
				uses_cairo_osiris_alternate_reward = yes
			}
		}
		value = 0
	}
}
