defiled_quality = {
	add = {
		value = defilements_lifestyle_perks
		multiply = 2
	}
	add = {
		value = sum_of_all_skills_and_prowess_value
		divide = 2
		floor = yes
	}
	if = { limit = { highest_held_title_tier = tier_county } add = 1 }
	if = { limit = { highest_held_title_tier = tier_duchy } add = 3 }
	if = { limit = { highest_held_title_tier = tier_kingdom } add = 8 }
	if = { limit = { highest_held_title_tier = tier_empire } add = 15 }

	if = { 
		limit = { num_of_relation_worshipped_yamaking > 0 } 
		every_relation = {
			type = worshipped_yamaking
			multiply = {
				scope:yamaking = { value = "opinion(scope:owner)" }
				divide = 100
			}
		}
	}
}
