

#Base weights
POD_ai_weight_never = -100 #MaA we don't want the AI to us even if they can unlock it somehow
POD_ai_weight_swarm = 20 #swarms are annoying to deal for the player and not that efficient for AI
#vanilla cultural MaA are at 80 value. We want PoD special troops to have priority so bumping it a little higher 
POD_ai_weight_generic = 60
POD_ai_weight_special = 90

POD_ai_weight_small_boost = 5
POD_ai_weight_large_boost = 10
POD_ai_weight_massive_boost = 30

##Animals
POD_ai_weight_animals_generic = {
	value = POD_ai_weight_generic
	if = {
		limit = {
			OR = {
				has_trait = gangrel
				has_trait = fera
				has_trait = anh
			}
		}
		add = POD_ai_weight_large_boost

####Fera and Spirits already implemented
##Fera
#These use high initial values to force AI to use them but then scales down as their prestige maintenance decreases
POD_ai_weight_fera_generic = { 
	value = 85
	add = POD_ai_weight_fera_scale
}
POD_ai_weight_fera_special = { 
	value = 110
	add = POD_ai_weight_fera_scale
}
POD_ai_weight_fera_scale = {
	value = 0

	if = {
		limit = { character_men_at_arms_expense_prestige_relative >= 1 }
		add = -30
	}
	if = {
		limit = { character_men_at_arms_expense_prestige_relative >= 0.9 }
		add = -20
	}
	if = {
		limit = { character_men_at_arms_expense_prestige_relative >= 0.8 }
		add = -15
	}
	if = {
		limit = { character_men_at_arms_expense_prestige_relative >= 0.7 }
		add = -15
	}
	if = {
		limit = { character_men_at_arms_expense_prestige_relative >= 0.6 }
		add = -10
	}
	if = {
		limit = { character_men_at_arms_expense_prestige_relative >= 0.5 }
		add = -10
	}
	if = {
		limit = { character_men_at_arms_expense_prestige_relative >= 0.4 }
		add = -5
	}
	if = {
		limit = { character_men_at_arms_expense_prestige_relative >= 0.3 }
		add = -5
	}
}

		limit = { faith = { is_in_family = rf_wyrm } }
