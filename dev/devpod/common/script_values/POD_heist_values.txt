starting_suspicion = {
	value = suspicion_min
	if = {
		limit = { has_character_modifier = master_infiltrator_modifier }
		add = master_infiltrator_suspicion_offset
	}
	if = {
		limit = { has_character_modifier = skilled_infiltrator_modifier }
		add = skilled_infiltrator_suspicion_offset
	}
	if = {
		limit = { has_perk = journeying_the_iron_bridge_perk  }
		add = infiltration_small_perk_suspicion_offset
	}
	if = {
		limit = { has_perk = silent_stalking_perk  }
		add = infiltration_small_perk_suspicion_offset
	}
	if = {
		limit = { has_perk = ricepaper_walk_perk  }
		add = infiltration_small_perk_suspicion_offset
	}
	if = {
		limit = { uses_cairo_bast_alternate_reward = yes }
		add = infiltration_small_perk_suspicion_offset
	}
	if = {
		limit = { has_variable = infiltration_experience_modifier_stacks }
		add = {
			value = infiltration_experience_suspicion_offset
			multiply = var:infiltration_experience_modifier_stacks
		}
	}
	if = {
		limit = {
			has_variable = current_heist
			var:current_heist = {
				has_variable = court_owner
				var:court_owner = { has_character_modifier = wary_of_infiltrators_modifier }
			}
		}
		add = wary_of_infiltrators_suspicion_offset
	}
}

heist_suspicion_level = {
	value = 0
	if = {
		limit = {
			has_variable = current_heist
			var:current_heist = { has_variable = suspicion }
		}
		add = {
			value = var:current_heist.var:suspicion
			min = suspicion_min
			max = suspicion_max
			divide = suspicion_range
			ceiling = yes
		}
	}
}

pod_suspicion_arrow_position = {
	value = 0
	if = {
		limit = {
			has_variable = current_heist
			var:current_heist = {
				has_variable = suspicion
				var:suspicion > 0
			}
		}
		if = {
			limit = { var:current_heist.var:suspicion >= suspicion_max }
			add = suspicion_range
		}
		else = {
			add = var:current_heist.var:suspicion
			modulo = suspicion_range
		}
	}
}
