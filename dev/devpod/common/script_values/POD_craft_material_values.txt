water_cost = {
	value = 0
	
	location = { 
		if = {
			limit = {
				OR = {
					terrain = sea
					terrain = coastal_sea
					is_coastal = yes
					is_riverside_province = yes
				}
			}
			add = 1
		}
		else_if = {
			limit = {
				OR = {
					terrain = oasis
					terrain = wetlands
				}
			}
			add = 2
		}
		else_if = {
			limit = {
				OR = {
					terrain = desert
					terrain = desert_mountains
				}
			}
			add = 10
		}
		else = {
			add = 5
		}
	}
}
