POD_yearly_demonology_xp_gain = {
	value = 0
	add = {
		value = {
			value = learning
			divide = 50
		}
		desc = POD_journey_progress_learning
		format = "POD_PROGRESS_FORMAT_POSITIVE"
	}

	if = {
		limit = { has_character_modifier = darkfungus_totem_modifier }
		add = {
			value = 0.25
			desc = darkfungus_totem_modifier
			format = "POD_PROGRESS_FORMAT_POSITIVE"
		}
	}

	if = {
		limit = { has_character_modifier = POD_focus_on_demonology_modifier }
		multiply = {
			value = 2
			desc = POD_focus_on_demonology_modifier
			format = "POD_FORMAT_PERCENTAGE_DOUBLED"
		}
	}
}
