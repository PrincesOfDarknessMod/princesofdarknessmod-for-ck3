
enhance_success_chance_base = {
	value = martial

	if = {
		limit = { var:target_artifact = { rarity = illustrious } }
		multiply = 0.2
	}
	else_if = {
		limit = { var:target_artifact = { rarity = famed } }
		multiply = 0.5
	}
	else_if = {
		limit = { var:target_artifact = { rarity = common } }
		multiply = 2
	}

	if = {
		limit = { var:target_artifact = { is_unique = yes } }
		multiply = 0.5
	}
}

enhance_success_chance = {
	value = 0
	
	if = {
		limit = { exists = var:target_artifact }
		add = {
			value = enhance_success_chance_base
			desc = martial
			format = "BASE_VALUE_FORMAT_DECIMALS"
		}
	}
	
	multiply = enhance_success_chance_material_bonus
	
	max = 100
}

enhance_success_chance_material_bonus = {
	value = 100

	add = material_bonus
	
	divide = 100
}
