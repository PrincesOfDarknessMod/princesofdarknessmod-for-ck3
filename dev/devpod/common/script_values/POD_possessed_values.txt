POD_possessed_experience_mult = {
	if = {
		limit = {
			save_temporary_scope_value_as = {
				name = exponent
				value = {
					value = POD_total_possessed_perks
					# grace period since you start with one perk
					subtract = 1
					min = 0
				}
			}
		}
	}
	switch = {
		# faster xp gain for each "run"
		trigger = POD_possessed_prestige
		
		0 = { value = POD_065_pow }
		1 = { value = POD_07_pow }
		2 = { value = POD_075_pow }
		3 = { value = POD_08_pow }
		4 = { value = POD_085_pow }
		5 = { value = POD_09_pow }
		6 = { value = POD_095_pow }
		
		fallback = { value = 1 }
	}
}
