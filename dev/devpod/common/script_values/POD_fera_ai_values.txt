
fera_ai_renown_gain_chance = {
	value = 20

	if = {
		limit = { is_playable_character = no }
		subtract = 10
	}

	#The higher your rank, the harder it is to gain more
	if = {
		limit = { POD_fera_rank_value > 1 }
		subtract = {
			value = POD_fera_rank_value
			multiply = 6
		}
	}

	if = {
		limit = { num_virtuous_traits > 0 }
		add = {
			value = num_virtuous_traits
			multiply = 5
		}
	}

	#If you manage to get spirit companions, they should help
	if = {
		limit = { num_of_relation_spirit_companion >= 1 }
		add = {
            value = {
                every_relation = {
					type = spirit_companion
					limit = { is_alive = yes }

					if = {
						limit = { POD_spirit_rank = 5 }
						add = 8
					}
					else_if = {
						limit = { POD_spirit_rank = 4 }
						add = 6
					}
					else_if = {
						limit = { POD_spirit_rank = 3 }
						add = 4
					}
					else_if = {
						limit = { POD_spirit_rank = 2 }
						add = 2
					}
					else = {
						add = 1
					}
				}
            }
        }
	}

	if = {
		limit = { prestige_level > 1 }
		add = {
			value = prestige_level
			multiply = 5
		}
	}
	if = {
		limit = { piety_level > 1 }
		add = {
			value = prestige_level
			multiply = 5
		}
	}
	if = {
		limit = { dynasty ?= { dynasty_prestige_level > 1 } }
		add = {
			value = dynasty.dynasty_prestige_level
			multiply = 3 #Lower because it affects multiple characters at once
		}
	}

	if = {
		limit = { highest_held_title_tier >= tier_duchy }
		add = 5
	}
	if = {
		limit = { highest_held_title_tier >= tier_kingdom }
		add = 5
	}
	if = {
		limit = { highest_held_title_tier >= tier_empire }
		add = 10
	}
	if = {
		limit = { highest_held_title_tier >= tier_hegemony }
		add = 30 #If the AI gets this, it deserves to win
	}

	min = 0
	max = 100
}
