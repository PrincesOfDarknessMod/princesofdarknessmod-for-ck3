rarity_to_create = {
	value = 0
	if = {
		limit = { exists = scope:wealth }
		add = scope:wealth
	}
	if = {
		limit = { exists = scope:quality }
		add = scope:quality
	}
}

forge_time = {
	value = 0
	if = {
		limit = { has_variable = current_method  }
		switch = {
			trigger = var:current_method
			flag:fleshcraft = { add = 75 }
			flag:mold = { add = 75 }
			flag:catgut = { add = 50 }
			flag:soulforge = { add = 75 }
			flag:fetishsmith_forced = { add = 75 }
			flag:fetishsmith = { add = 50 }
			flag:shape_true_jade = { add = 25 }
			flag:frost_facsimile = { add = 10 }
			flag:eightfold_yin_mantle = { add = 10 }
			flag:corrupt_relic = { add = 10 }
			flag:tainted_relic = { add = 5 }
			flag:horrid_blade = { add = 10 }
			flag:demon_weapon = { add = 5 }
			flag:reweave_common = { add = 25 }
			flag:reweave_masterwork = { add = 50 }
			flag:reweave_famed = { add = 75 }
			flag:reweave_illustrious = { add = 100 }
			flag:enchant_object = { add = 100 }
		}
	}
	
	if = {
		limit = { NOT = { root = var:craft_creator } }
		multiply = 10
	}
}

forge_cost = {
	value = 0
	add = water_cost
	add = fuel_cost
	multiply = 10
}

forge_inspiration_average_skill_value = {	
	if = {
		limit = { is_armor_forge_trigger = { RECIPE = root.var:current_recipe } }
		value = armor_inspiration_average_skill_value
	}
	else_if = {
		limit = { is_weapon_forge_trigger = { RECIPE = root.var:current_recipe } }
		value = weapon_inspiration_average_skill_value
	}
	else_if = {
		limit = { is_shinies_forge_trigger = { RECIPE = root.var:current_recipe } }
		value = smith_inspiration_average_skill_value
	}
	else_if = {
		limit = { is_book_forge_trigger = { RECIPE = root.var:current_recipe } }
		value = book_inspiration_average_skill_value
	}
	else_if = {
		limit = { is_furniture_forge_trigger = { RECIPE = root.var:current_recipe } }
		value = weaver_inspiration_average_skill_value
	}
	else_if = {
		limit = { is_trinket_forge_trigger = { RECIPE = root.var:current_recipe } }
		value = artisan_inspiration_average_skill_value
	}
	else = {
		value = adventure_inspiration_average_skill_value
	}
	multiply = {
		value = quality_bonus_per_skill_level_value
		add = scope:random_inspired_skill_quality_bonus
	}
}
