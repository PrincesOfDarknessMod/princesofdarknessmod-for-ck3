# Laibon Aye/Orun System
# Script Values
##############################

@default_format = "POD_FORMAT_NO_DECIMALS"


POD_aye_xp = {
	value = "this.has_trait_xp(laibon_spirit|aye)"
}

POD_orun_xp = {
	value = "this.has_trait_xp(laibon_spirit|orun)"
}


POD_laibon_spirit_xp_for_rank_1 = 25
POD_laibon_spirit_xp_for_rank_2 = 50
POD_laibon_spirit_xp_for_rank_3 = 75


# needs scope:aye_affinity and scope:orun_affinity on the TopScope (either 1 or -1)
# it's janky but it's the best way to reuse this value and make it work with the ScriptValueDesc
POD_aye_orun_clan_affinity = {
	### Aye
	if = {
		limit = { is_guruhi_trigger = yes }
		add = {
			value = scope:aye_affinity
			desc = dynn_guruhi
			format = @default_format
		}
	}
	else_if = {
		limit = { is_ishtarri_trigger = yes }
		add = {
			value = scope:aye_affinity
			desc = dynn_ishtarri
			format = @default_format
		}
	}
	else_if = {
		limit = { is_nyayo_trigger = yes }
		add = {
			value = scope:aye_affinity
			desc = dynn_nyayo
			format = @default_format
		}
	}
	else_if = {
		limit = { house ?= house:house_wadjet }
		add = {
			value = scope:aye_affinity
			desc = dynn_wadjet
			format = @default_format
		}
	}
	### Orun
	else_if = {
		limit = { is_ramanga_trigger = yes }
		add = {
			value = scope:orun_affinity
			desc = dynn_ramanga
			format = @default_format
		}
	}
	else_if = {
		limit = { is_nagloper_trigger = yes }
		add = {
			value = scope:orun_affinity
			desc = dynn_nagloper
			format = @default_format
		}
	}
	else_if = {
		limit = { is_impundulu_trigger = yes }
		add = {
			value = scope:orun_affinity
			desc = trait_impundulu
			format = @default_format
		}
	}
	else_if = {
		limit = { is_mla_watu_trigger = yes }
		add = {
			value = scope:orun_affinity
			desc = dynn_mlawatu
			format = @default_format
		}
	}
	else_if = {
		limit = {
			OR = {
				is_osebo_trigger = yes
				is_shango_trigger = yes
				is_akunanse_trigger = yes
				is_kinyonyi_trigger = yes
			}
		}
		# balanced
	}
	# non-laibon clan
	else = {
		add = {
			value = -1
			desc = POD_aye_orun_value_outsider_clan
			format = @default_format
		}
	}
}

# needs scope:aye_affinity and scope:orun_affinity on the TopScope (either 1 or -1)
# it's janky but it's the best way to reuse this value and make it work with the ScriptValueDesc
POD_aye_orun_faith_affinity = {
	if = {
		limit = { faith = { has_doctrine = doctrine_enlightenment_virtues_aye } }
		add = {
			value = scope:aye_affinity
			desc = POD_aye_value_faith
			format = @default_format
		}
	}
	else_if = {
		limit = { faith = { has_doctrine = doctrine_enlightenment_virtues_orun } }
		add = {
			value = scope:orun_affinity
			desc = POD_orun_value_faith
			format = @default_format
		}
	}
}

# Expected annual net Aye XP change (decay + clan + faith + discipline passive)
pod_laibon_aye_net_value = {
	value = 0

	### Decay ###
	add = {
		value = {
			if = {
				limit = { has_aye_tier_3 = yes }
				value = -4
			}
			else_if = {
				limit = { has_aye_tier_2 = yes }
				value = -3
			}
			else = {
				value = -1
			}
		}
		desc = POD_aye_orun_value_decay
		format = @default_format
	}
	
	### Clan/Faith affinity ###
	if = {
		limit = {
			save_temporary_scope_value_as = {
				name = aye_affinity
				value = 1
			}
			save_temporary_scope_value_as = {
				name = orun_affinity
				value = -1
			}
		}
	}
	add = POD_aye_orun_clan_affinity
	add = POD_aye_orun_faith_affinity

	### Discipline passive (Presence / Dominate — both can apply) ###
	if = {
		limit = { has_trait = presenceadvanced }
		add = {
			value = 2
			desc = trait_presenceadvanced
			format = @default_format
		}
	}
	else_if = {
		limit = { has_trait = presence }
		add = {
			value = 1
			desc = trait_presence
			format = @default_format
		}
	}

	if = {
		limit = { has_trait = dominateadvanced }
		add = {
			value = 2
			desc = trait_dominateadvanced
			format = @default_format
		}
	}
	else_if = {
		limit = { has_trait = dominate }
		add = {
			value = 1
			desc = trait_dominate
			format = @default_format
		}
	}

	### Frenzy XP Adjustment ###
	if = {
		limit = {
			OR = {
				has_character_flag = had_hunger_frenzy
				has_character_flag = had_rage_frenzy
			}
		}
		add = {
			value = -3
			desc = POD_aye_orun_value_recent_frenzy
			format = @default_format
		}
	}

	# Careful feeding bonus: +1 Aye/year for having a herd
	if = {
		limit = {
			any_relation = {
				type = herd_member
				has_supernatural_race_trigger = no
			}
		}
		add = {
			value = 1
			desc = POD_aye_value_mortal_herd
			format = @default_format
		}
	}
}

# Expected annual net Orun XP change (decay + clan + faith + discipline passive)
pod_laibon_orun_net_value = {
	value = 0

	### Decay ###
	add = {
		value = {
			if = {
				limit = { has_orun_tier_3 = yes }
				value = -3
			}
			else_if = {
				limit = { has_orun_tier_2 = yes }

			}

			}

			desc = trait_obtenebrationadvanced
			format = @default_format
		}
	}
	else_if = {
		limit = { has_trait = obtenebration }
		add = {
			value = 1
			desc = trait_obtenebration
			format = @default_format
		}
	}

	if = {
		limit = { has_trait = necromancyadvanced }
		add = {
			value = 2
			desc = trait_necromancyadvanced
			format = @default_format
		}
	}
	else_if = {
		limit = { has_trait = necromancy }
		add = {
			value = 1
			desc = trait_necromancy
			format = @default_format
		}
	}

	### Frenzy XP Adjustment ###
	if = {
		limit = {
			OR = {
				has_character_flag = had_hunger_frenzy
				has_character_flag = had_rage_frenzy
			}
		}
		add = {
			value = 2
			desc = POD_aye_orun_value_recent_frenzy
			format = @default_format
		}
	}

	# Vampire feeding bonus: +1 Orun/year
	if = {
		limit = {
			OR = {
				any_relation = {
					type = herd_member
					POD_is_vampire_trigger = yes
				}
			}
		}
		add = {
			value = 1
			desc = POD_orun_value_vampire_herd
			format = @default_format
		}
	}
}
