POD_debug_squared_distance = {
	if = {
		limit = { scope:recipient = { exists = capital_province } }
		value = "scope:actor.capital_province.squared_distance(scope:recipient.capital_province)"
	}
	else = {
		value = "scope:actor.capital_province.squared_distance(scope:recipient.location)"
	}
}

POD_hud_portrait_scale = {
	if = {
		limit = { has_game_rule = POD_hud_portrait_size_vanilla }
		value = 1.3
	}
	else_if = {
		limit = { has_game_rule = POD_hud_portrait_size_larger }
		value = 1.7
	}
	else = {
		value = 1.5
	}
}
