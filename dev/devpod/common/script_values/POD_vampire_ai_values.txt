vampire_ai_herd_cap = {
	value = max_herd_cap
	subtract = 1
	if = {
		limit = {
			is_playable_character = no
			NOT = { any_parent ?= { is_ruler = yes } }
		}
		subtract = 1
	}

	if = {
		limit = { highest_held_title_tier >= tier_duchy }
		add = 1
	}
	if = {
		limit = { highest_held_title_tier >= tier_kingdom }
		add = 1
	}
	if = {
		limit = { highest_held_title_tier >= tier_empire }
		add = 1
	}
	if = {
		limit = { highest_held_title_tier = tier_hegemony }
		add = 1
	}

	add = {
		value = prestige_level
		divide = 1.5
		floor = yes
		min = 0
	}

	min = 0
}

vampire_ai_discipline_limit = {
	value = 3 

	#Age - the main factor here
	add = {
		value = age
		divide = 75
		floor = yes
		max = 7
	}

	#Generation - the second big factor here
	add = {
		value = generation_value #4th = 13 / 7th = 10 / 10th = 7
		divide = 4 #4th = 3 / 7th = 2 / 10th = 1
		floor = yes

		#4th gens get a boost
		if = {
			limit = { POD_vampgeneration == 4 }
			add = 2
		}
		#5th get a smaller boost
		if = {
			limit = { POD_vampgeneration == 5 }
			add = 1
		}
	}

	#Big important vamps
	if = {
		limit = {
			is_lowborn = no
			dynasty ?= { dynasty_prestige_level > 1 } #so no random generated vamps
			OR = {
				is_dynast = yes
				is_house_head = yes
			}
		}
		add = 1
	}
	if = {
		limit = { highest_held_title_tier >= tier_empire }
		add = 1
	}
}

vampire_ai_discipline_learning_chance = {
	value = 1

	add = {
		value = learning
		divide = 8
		floor = yes
		min = 0
	}

	if = {
		limit = { prestige_level > 1 }
		add = prestige_level
	}

	if = {
		limit = { piety_level > 1 }
		add = piety_level
	}

	if = {
		limit = { vampire_ai_unmastered_disciplines_value == 0 } #get a boost if you advanced all your disciplines
		add = 4
	}
	else = { subtract = vampire_ai_unmastered_disciplines_value }
	

	max = 100
}
