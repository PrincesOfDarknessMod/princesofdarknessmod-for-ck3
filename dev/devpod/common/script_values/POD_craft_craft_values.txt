forge_inspiration_average_skill_value = {	
	if = {
		limit = { is_armor_forge_trigger = { RECIPE = root.var:current_recipe } }
		value = armor_inspiration_average_skill_value
	}
	else_if = {
		limit = { is_weapon_forge_trigger = { RECIPE = root.var:current_recipe } }
		value = weapon_inspiration_average_skill_value
	}
	else_if = {
		limit = { is_shinies_forge_trigger = { RECIPE = root.var:current_recipe } }
		value = smith_inspiration_average_skill_value
	}
	else_if = {
		limit = { is_book_forge_trigger = { RECIPE = root.var:current_recipe } }
		value = book_inspiration_average_skill_value
	}
	else_if = {
		limit = { is_furniture_forge_trigger = { RECIPE = root.var:current_recipe } }
		value = weaver_inspiration_average_skill_value
	}
	else_if = {
		limit = { is_trinket_forge_trigger = { RECIPE = root.var:current_recipe } }
		value = artisan_inspiration_average_skill_value
	}
	else = {
		value = adventure_inspiration_average_skill_value
	}
	multiply = {
		value = quality_bonus_per_skill_level_value
		add = scope:random_inspired_skill_quality_bonus
	}
}

alchemy_success_chance = {
	value = 0
	
	if = {
		limit = { exists = var:current_recipe }
		add = {
			value = alchemy_success_chance_base
			desc = learning
			format = "BASE_VALUE_FORMAT_DECIMALS"
		}
	
		if = {
			limit = { has_perk = divine_quiddity_perk }
			multiply = {
				value = 5
				desc = divine_quiddity_perk_name
			}
		}
		else_if = {
			limit = { has_perk = kingly_quiddity_perk }
			multiply = {
				value = 4
				desc = kingly_quiddity_perk_name
			}
		}
		else_if = {
			limit = { has_perk = princely_quiddity_perk }
			multiply = {
				value = 3
				desc = princely_quiddity_perk_name
			}
		}
		else_if = {
			limit = { has_perk = greater_quiddity_perk }
			multiply = {
				value = 2
				desc = greater_quiddity_perk_name
			}
		}
		else_if = {
			limit = { has_perk = lesser_quiddity_perk }
			multiply = {
				value = 1.5
				desc = lesser_quiddity_perk_name
			}
		}
	}
	
	multiply = alchemy_success_chance_material_bonus
	
	if = {
		limit = { is_target_in_variable_list = { name = ingredient_list target = flag:sekhem } }
		add = {
			value = alchemy_success_chance_base_value
			desc = tears_of_isis_perk_name
			format = "POD_FORMAT_PERCENTAGE_DOUBLED"
		}
	}
	
	if = {
		limit = { var:craft_creator = { has_character_flag = craft_creator_tutor } }
		min = {
			value = 100
			desc = TUTORIAL
		}
	}
	
	max = 100
}
