mass_command_gold_potential_target_value = {
	value = num_of_relation_demonic_thrall
	subtract = num_of_relation_possessed_target
}

mass_command_gold_value = {
	ordered_relation = {
		type = demonic_thrall
		limit = { 
			gold > 0
			NOT = { has_relation_possessed_target = root }
		}
		order_by = gold
		max = mass_command_gold_thrall_list_max_size
		if = {
			limit = { gold >= root.major_gold_value }
			add = root.major_gold_value
		}
		else = {
			add = gold
		}
	}
}

mass_command_gold_thrall_list_max_size = {
	if = {
		limit = { root.num_of_relation_demonic_thrall >= root.arcana_lifestyle_perk_points }
		value = root.arcana_lifestyle_perk_points
	}
	else = {
		value = root.num_of_relation_demonic_thrall
	}
}
