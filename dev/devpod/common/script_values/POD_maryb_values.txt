#
# Zero-defaulting reads for Mary's maybe-absent counter variables. Reading
# through these lets every call site skip the has_variable guard / defensive
# init: an unset variable simply reads as 0.
#

# mary_tablets_recovered (0-3 tablet counter) as a number, 0 when unset.
POD_maryb_tablets_recovered_value = {
	value = 0
	if = {
		limit = { has_variable = mary_tablets_recovered }
		add = var:mary_tablets_recovered
	}
}

# mary_michael_balance (-100..+100) as a number. 0 (perfectly balanced) is the
# natural default when the variable is absent.
POD_maryb_michael_balance_value = {
	value = 0
	if = {
		limit = { has_variable = mary_michael_balance }
		add = var:mary_michael_balance
	}
}
