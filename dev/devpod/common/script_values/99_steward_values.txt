current_prosperity_penalty = {
	value = 0
	if = {
		limit = {
			exists = scope:county
		}
		add = scope:councillor.steward_develop_county_base_total
		add = scope:councillor.steward_develop_county_perk_bonus_monthly_increase
		add = scope:councillor.steward_develop_county_tribal_penalty
		multiply = {
			subtract = scope:county.prosperity_level
			multiply = {
				value = 9.13 # Maximum penalty at prosperity level 10
				# The formula is: Monthly Increase - (Monthly Increase(Prosperity Level/100)*Above Number)
				if = {
					limit = { culture = { has_innovation = innovation_prosperity_04 } }
					multiply = 0.1111 # With Renaissance Thought, this is increased to Prosperity Level 90.
				}
				else_if = {
					limit = { culture = { has_innovation = innovation_prosperity_03 } }
					multiply = 0.1818 # With Urbanization, Prosperity Level 55.
				}
				else_if = {
					limit = { culture = { has_innovation = innovation_prosperity_02 } }
					multiply = 0.2856 # With Communal Government, Prosperity Level 35.
				}
				else_if = {
					limit = { culture = { has_innovation = innovation_prosperity_01 } }
					multiply = 0.5000 # With the Public Works innovation, Prosperity Level 20.
				}
			}
			divide = 100
			min = -0.875 # This is the maximum penalty.
		}
	}
}
