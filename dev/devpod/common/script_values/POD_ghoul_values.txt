POD_years_to_lose_vitae_passively = {
	value = {
		value = POD_years_to_lose_vitae_passively_base
		desc = POD_base_value
		format = "POD_FORMAT_BASE_POSITIVE"
	}
	if = {
		limit = { age <= 80 }
		multiply = 1
	}
	else_if = {
		limit = { age <= 200 }
		multiply = {
			value = 2
			desc = POD_years_to_lose_vitae_passively_age_above_80
			format = "POD_FORMAT_POSITIVE"
		}
	}
	else_if = {
		limit = { age > 200 }
		multiply = {
			value = 3
			desc = POD_years_to_lose_vitae_passively_age_above_200
			format = "POD_FORMAT_POSITIVE"
		}
	}
}

}
