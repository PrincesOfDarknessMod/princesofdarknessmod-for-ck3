#This is so you can just do "add_X_lifestyle_xp = resonance_xp_value" instead of having to make a lot of ifs
resonance_xp_value = {
	value = 0 
	if = {
		limit = {
			has_character_flag = POD_resonance_negligible_flag
		}
		add = negligible_resonance_xp
	}
	if = {
		limit = {
			has_character_flag = POD_resonance_fleeting_flag
		}
		add = fleeting_resonance_xp
	}
	if = {
		limit = {
			has_character_flag = POD_resonance_intense_flag
		}
		add = intense_resonance_xp 
	}
	if = {
		limit = {
			has_character_flag = POD_resonance_acute_flag
		}
		add = acute_resonance_xp
	}
	
	if = {
		limit = {
			has_game_rule = resonance_xp_fast
		}
		multiply = 3
	}

	if = {
		limit = {
			has_game_rule = resonance_xp_half
		}
		divide = 2
	}
	
	if = {
		limit = {
			OR = {
				has_trait = demon
				has_trait = wraith
				has_trait = spirit
				has_trait = automaton
			}
		}
		multiply = 0
	}
}

resonance_xp_sip_value = {

	if = {

blood_potency_minimum = {
	if = {
		limit = { has_trait = vampgeneration }
		
		value = POD_vampgeneration
		save_temporary_value_as = generation
		
		if = {
			limit = { scope:generation >= 14 }
			value = 0
		}
		else_if = {
			limit = { scope:generation >= 10 }
			value = 1
		}
		else_if = {
			limit = { scope:generation >= 8 }
			value = 2
		}
		else_if = {
			limit = { scope:generation >= 6 }
			value = 3
		}
		else_if = {
			limit = { scope:generation >= 5 }
			value = 4
		}
		else = {
			value = 5
		}
	}
	else = {
		value = 0
	}
}

mortal_blood_needed_to_slake_1_hunger = {
	value = 0

	if = {
		limit = { blood_potency_value <= 3 }
		add = 1
	}
	else_if = {
		limit = { blood_potency_value <= 6 }
		add = 2
	}
	else_if = {
		limit = { blood_potency_value <= 9 }
		add = 3
	}
	else_if = {
		limit = { blood_potency_value = 10 }
		add = 4
	}

	if = {
		limit = { has_character_modifier = ritual_mourning_life_curse_modifier }
		subtract = 1
	}
	if = {
		limit = { has_character_modifier = ritual_stolen_kisses_modifier }
		subtract = 2
	}
	if = {
		limit = { any_directly_owned_province = { has_building_with_flag = building_improves_slaking_hunger } }
		subtract = 1
	}
	
	if = {
		limit = { has_character_modifier = abated_tooth_modifier }
		add = 200
	}

	min = 1

	divide = mortal_blood_nutrition_multiplier
}
