#I'm keeping the base idea but rewards more the higher resonance tiers to encourage people to really look for those high quality mortals -Karde
negligible_resonance_xp = 6
fleeting_resonance_xp = 6.67
intense_resonance_xp = 10 #3000 xp over 5 years, a little less than 2 perks
acute_resonance_xp = 13.33 #4000 xp over 5 years, so 2 perks

#This is so you can just do "add_X_lifestyle_xp = resonance_xp_value" instead of having to make a lot of ifs
resonance_xp_value = {
	value = 0 
	if = {
		limit = {
			has_character_flag = POD_resonance_negligible_flag
		}
		add = negligible_resonance_xp
	}
	if = {
		limit = {
			has_character_flag = POD_resonance_fleeting_flag
		}
		add = fleeting_resonance_xp
	}
	if = {
		limit = {
			has_character_flag = POD_resonance_intense_flag
		}
		add = intense_resonance_xp 
	}
	if = {
		limit = {
			has_character_flag = POD_resonance_acute_flag
		}
		add = acute_resonance_xp
	}
	
	if = {
		limit = {
			has_game_rule = resonance_xp_fast
		}
		multiply = 3
	}
	
	if = {
		limit = {
			OR = {
				has_trait = demon
				has_trait = wraith
				has_trait = spirit
				has_trait = automaton
			}
		}
		multiply = 0
	}
}

resonance_xp_sip_value = {
	value = resonance_xp_value 
	
	if = {
		limit = { POD_chi_user_trigger = no }
		multiply = 10
	}
	else = {
		multiply = chi_points_value
	}
}

	value = 1
	if = {
		limit = { num_of_relation_herd_member > max_herd_cap }

mortal_blood_needed_to_slake_1_hunger = {
	value = 0

	if = {
		limit = { blood_potency_value <= 3 }
		add = 1
	}
	else_if = {
		limit = { blood_potency_value <= 6 }
		add = 2
	}
	else_if = {
		limit = { blood_potency_value <= 9 }
		add = 3
	}
	else_if = {
		limit = { blood_potency_value = 10 }
		add = 4
	}

	if = { limit = { has_character_modifier = ritual_mourning_life_curse_modifier } subtract = 1 }
	if = { limit = { has_character_modifier = ritual_stolen_kisses_modifier } subtract = 2 }
	if = { limit = { has_character_modifier = abated_tooth_modifier } add = 200 }

	min = 1

	divide = mortal_blood_nutrition_multiplier
}
