##Build Time - these need to be placed in each domicile file manually. They do not work as regular script values

POD_domicile_tutor_aptitude = {
	value = 0

	if = {
		limit = { employer.domicile ?= { has_domicile_parameter = tutor_aptitude_t6 } }
		add = 24
	}
	else_if = {
		limit = { employer.domicile ?= { has_domicile_parameter = tutor_aptitude_t5 } }
		add = 20
	}
	else_if = {
		limit = { employer.domicile ?= { has_domicile_parameter = tutor_aptitude_t4 } }
		add = 16
	}
	else_if = {
		limit = { employer.domicile ?= { has_domicile_parameter = tutor_aptitude_t3 } }
		add = 12
	}
	else_if = {
		limit = { employer.domicile ?= { has_domicile_parameter = tutor_aptitude_t2 } }
		add = 8
	}
	else_if = {
		limit = { employer.domicile ?= { has_domicile_parameter = tutor_aptitude_t1 } }
		add = 4
	}
}
