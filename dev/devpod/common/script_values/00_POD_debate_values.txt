debate_opponent_selection_value = {
	value = sub_realm_size
	multiply = 1.75
	# Head of house/dynasty
	if = {
		limit = { is_house_head = yes }
		add = 5
	}
	if = {
		limit = { is_dynast = yes }
		add = 10
	}
	# Dynasty is important
	if = {
		limit = { dynasty ?= { dynasty_prestige_level >= 3 } }
		add = 10
	}
	if = {
		limit = { dynasty ?= { dynasty_prestige_level >= 5 } }
		add = 25
	}
	if = {
		limit = { dynasty ?= { dynasty_prestige_level >= 7 } }
		add = 50
	}
	# Emperor/Hegemon regardless of realm size
	if = {
		limit = { highest_held_title_tier >= tier_empire }
		add = 50
	}
	# Player
	if = {
		limit = { is_ai = no }
		add = 50
	}
	# Important person
	if = {
		limit = { prestige_level >= 2 }
		add = 5
	}
	if = {
		limit = { prestige_level >= 3 }
		add = 10
	}
	if = {
		limit = { prestige_level >= 4 }
		add = 15
	}
}
