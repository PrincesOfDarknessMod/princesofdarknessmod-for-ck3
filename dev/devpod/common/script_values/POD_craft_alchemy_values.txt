alchemy_success_chance = {
	value = 0
	
	if = {
		limit = { exists = var:current_recipe }
		add = {
			value = alchemy_success_chance_base
			desc = learning
			format = "BASE_VALUE_FORMAT_DECIMALS"
		}
	
		if = {
			limit = { has_perk = divine_quiddity_perk }
			multiply = {
				value = 5
				desc = divine_quiddity_perk_name
			}
		}
		else_if = {
			limit = { has_perk = kingly_quiddity_perk }
			multiply = {
				value = 4
				desc = kingly_quiddity_perk_name

	value = 0
	
	location = { 
		if = {
			limit = {
				OR = {
					terrain = sea
					terrain = coastal_sea
					is_coastal = yes
					is_riverside_province = yes
				}
			}
			add = 1
		}
		else_if = {
			limit = {
				OR = {
					terrain = oasis
					terrain = wetlands
				}

				OR = {
					terrain = desert
					terrain = desert_mountains
				}
			}
			add = 10
		}
		else = {
			add = 5
		}
	}
}
