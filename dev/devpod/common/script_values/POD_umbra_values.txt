	if = {

	# Werewolf
	if = {

		add = -5
	}
	if = {
		limit = { has_perk = determinism_perk }
		add = -5
	}
	
	# Possessed
	if = {
		limit = { 
			OR = { 
				has_perk = fomori_spirit_sideways_perk
				has_perk = gorgon_spirit_sideways_perk
				has_perk = drone_spirit_sideways_perk
				has_perk = kami_spirit_sideways_perk
			}
		}
		add = -5
	}
	
	# Artifact
	if = {
		limit = { any_equipped_character_artifact = { has_variable = ship_of_the_four_quarters } }
		add = -20
	}
	
	divide = 8
	round = yes
	min = 1
}

umbra_reweave_artifact_gold_value = {
	value = 0
	every_equipped_character_artifact = {
		limit = {
			artifact_slot_type = miscellaneous
			has_variable = artifact_pieces
		}
		if = {
			limit = { rarity = common }
			add = tiny_gold_value
		}
		else_if = {
			limit = { rarity = masterwork }
			add = minor_gold_value
		}
		else_if = {
			limit = { rarity = famed }
			add = medium_gold_value
		}
		else = {
			add = major_gold_value
		}
	}
}
