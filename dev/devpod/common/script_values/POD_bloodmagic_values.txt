POD_bloodmagic_success_chance = {
	value = 20
	add = {
		value = learning
		divide = 4
	}
	if = {
		limit = { NOT = { scope:target = ROOT } }
		if = {
			limit = {
				has_trait = vampire
				scope:target = { has_trait = vampire }
			} # Generation Difference if both vampires
			add = { # Generation Difference 
				value = generation_value
				subtract = scope:target.generation_value
				multiply = 10
			}
		}

		subtract = {
			value = scope:target.learning
			divide = 3
		}

		if = { limit = { scope:target = { has_character_modifier = ritual_warding_baptism_modifier } } subtract = 25 }
		if = { limit = { scope:target = { has_perk = arcane_knowledge_1_perk } } subtract = 10 }
		if = { limit = { scope:target = { has_perk = arcane_knowledge_2_perk } } subtract = 10 }
		if = { limit = { scope:target = { has_perk = arcane_knowledge_3_perk } } subtract = 10 }
		if = { limit = { scope:target = { has_perk = arcane_knowledge_4_perk } } subtract = 10 }
		if = { limit = { any_equipped_character_artifact = { has_variable = artifact_blood_magic_spell_defense_10 } } subtract = 10 }		
		if = { limit = { scope:target = { has_perk = wisdom_of_isis_perk } } subtract = 50 }
	}
	if = { limit = { has_perk = arcane_knowledge_1_perk } add = 15 }
	if = { limit = { has_perk = arcane_knowledge_2_perk } add = 15 }
	if = { limit = { has_perk = arcane_knowledge_3_perk } add = 15 }
	if = { limit = { has_perk = arcane_knowledge_4_perk } add = 15 }
	if = { limit = { any_equipped_character_artifact = { has_variable = artifact_blood_magic_spell_success_10 } } add = 10 }		
	if = { limit = { any_equipped_character_artifact = { has_variable = artifact_blood_magic_spell_success_25 } } add = 25 }		
	
	min = 0
	max = 100
}
