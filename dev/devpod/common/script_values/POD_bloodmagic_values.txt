POD_bloodmagic_success_chance = {
    value = 50 # Each spell have default 25% success chance
    add = learning # Flat learning bonus
    if = { limit = { NOT = { scope:target = ROOT } }
        if = { limit = { has_trait = vampire scope:target = { has_trait = vampire } } # Generation Difference if both vampires
            add = { # Generation Difference 
                value = generation_value
                subtract = scope:target.generation_value
                multiply = 10
            }
        }
        
        subtract = scope:target.learning # Targets Learning 

        subtract = { # Distance between Target and caster
            value = "ROOT.location.squared_distance(scope:target.location)" #
            divide = 40000
        }
    }
    else = {
        value = 50
        add = learning
    }
    if = { limit = { scope:target = { has_character_modifier = ritual_warding_baptism_modifier } } subtract = 25 }
    
    min = 0
    max = 100
}
POD_bloodmagic_success_chance_negated = {
    value = POD_bloodmagic_success_chance
    multiply = -1
}
