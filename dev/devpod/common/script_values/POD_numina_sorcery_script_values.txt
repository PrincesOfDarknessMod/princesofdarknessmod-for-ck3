sorcery_base_stress_cost = {
    value = 10
    if = {
        limit = { has_perk = mana_manipulation_1_perk  }
        subtract = { value = 10 multiply = 0.05 }
    }
    if = {
        limit = { has_perk = mana_manipulation_2_perk  }
        subtract = { value = 10 multiply = 0.05 }
    }
    if = {
        limit = { has_perk = mana_manipulation_3_perk  }
        subtract = { value = 10 multiply = 0.1 }
    }
    if = {
        limit = { has_perk = mana_manipulation_4_perk  }
        subtract = { value = 10 multiply = 0.1 }
    }
    if = {
        limit = { has_perk = mana_manipulation_5_perk  }
        subtract = { value = 10 multiply = 0.15 }
    }
    if = {
        limit = { has_perk = mana_manipulation_6_perk  }
        subtract = { value = 10 multiply = 0.25 }
    }
    
}
sorcery_low_stress_cost = { value = sorcery_base_stress_cost multiply = 2 }
