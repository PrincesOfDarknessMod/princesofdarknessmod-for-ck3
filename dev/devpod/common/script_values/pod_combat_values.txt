######################################
######   POD COMBAT VALUES   ######
######################################

# ─── COMBAT VARIABLE REFERENCE ─────────────────────────────────────────────
#
# Group combat uses a 10-slot system (slots 0-4 = player team, 5-9 = enemy
# team). Per-combatant state lives on POD_combat_slot story_cycle scopes —
# one story per slot, created at combat init and cleaned up at combat end.
# Pointer variables pod_combat_slot_0..9 on the combat host (the PC) hand
# back the matching story. The flat pod_combat_c{N}_* character-variable
# namespace this used to use is gone; reads/writes go through slot stories
# only.
#
# A shared WORKSPACE (pod_combat_actor_* / pod_combat_target_*) holds
# whichever combatant is acting and whichever one is the current target;
# the workspace is loaded from / saved back to the appropriate slot story
# at each action boundary by pod_combat_{load,save}_workspace_effect.
# Skill effects read and write only the workspace, so the same generated
# code runs for the PC, AI coterie, the primary opponent, and knights.
#
# ── Core state (on the combat host / PC character) ───────────────
#   pod_combat_active                 — flag: combat is in progress
#   pod_combat_is_over                — flag: combat is over (re-entry guard)
#   pod_combat_mode                   — 0=1v1, 1=1vN, 2=Nv1, 3=NvN
#   pod_combat_round                  — current round number
#   pod_combat_current_actor          — slot index of the acting combatant (0-9)
#   pod_combat_current_ap             — AP remaining this turn
#   pod_combat_current_turn_side      — 0 = player-team turn, 1 = enemy-team turn
#   pod_combat_enemy_team_size        — number of occupied enemy slots (5-9);
#                                       used by the slots-6-9 fill pass to
#                                       gate how many enemies still fit
#   pod_combat_pc_target              — the PC's persistent target slot (1-9)
#   pod_combat_selected_target        — slot loaded into opponent workspace
#   pod_combat_slot_0..9              — pointer var per slot; resolves to
#                                       the slot's POD_combat_slot story
#                                       scope. Unset for unfilled slots.
#   pod_combat_active_slots           — variable list of slot stories
#                                       currently in play (filled slots
#                                       only, in slot-index ascending
#                                       order). Iterated by every dispatch
#                                       loop instead of unrolling 0..9.
#
# ── Per-slot stats (vars on each POD_combat_slot story) ──────────
#   slot_index                        — 0..9, baked at slot creation
#   slot_char                         — character ref occupying the slot
#   slot_alive                        — 1 while in play, 0 when KO'd
#   (team affiliation is computed from slot_index — slots 0-4 = player
#    team, 5-9 = enemy team — not stored as a per-slot variable)
# Health is NOT stored on the slot story — it is live script values
# derived from each slot character's PoD wound vars (the slot story
# only holds the slot_char pointer; the SVs read POD_empty_damage_boxes
# / POD_total_damage_boxes / POD_health_aggravated_damage_num off the
# character):
#   pod_combat_slot_hp_value          — empty wound boxes (HP remaining)
#   pod_combat_slot_max_hp_value      — wound track length (max HP)
#   pod_combat_slot_agg_damage_sv     — aggravated wound boxes
#   pod_combat_slot_total_damage_sv   — total wound boxes filled
# Caching HP on the slot story was tried earlier and rejected — stale
# cache between turns vs PoD's wound writes was a constant source of
# desync; deriving HP live keeps display and gameplay reads in lockstep
# with the canonical PoD state.
#   slot_morale                       — current Morale
#   slot_prowess                      — current prowess
#   slot_soak                         — live soak chance (0-100)
#   slot_base_soak                    — soak before any tier effects
#   slot_hit_chance                   — live hit chance (20-95)
#   slot_base_hit_chance              — hit chance before any tier effects
#   slot_initiative                   — initiative roll this round
#   slot_init_bonus                   — next-round initiative modifier
#   slot_ap_drain                     — AP drained from next turn
#   slot_energy_used                  — counter for AI energy-attrition weighting
#   slot_turn_order                   — turn-rank within the round
#                                       (1 = first); stamped by
#                                       pod_combat_calc_turn_orders_effect
#                                       after initiative roll.
#   slot_has_acted                    — 1 once the slot's turn resolves
#                                       this round; reset to 0 by
#                                       pod_combat_calculate_initiative_effect
#                                       at the start of every round.
#
# ── Per-slot stance flags (vars on the POD_combat_slot story) ────
# All persist across rounds; only cleared when consumed or at combat end.
#   slot_is_guarding                  — Guard stance active
#   slot_is_dodging                   — Dodge stance active
#   slot_is_stealthed                 — Stealth active (includes +30 hit)
#   slot_retaliate_hit                — retaliation damage if hit
#   slot_retaliate_miss               — retaliation damage if missed
#   slot_retaliate_{hit,miss}_agg     — aggravated variants
#
# ── Per-slot spirit-damage capability ────────────────────────────
#   slot_incorporeal_spirit           — slot character is incorporeal;
#                                       physical attacks no-op unless the
#                                       attacker has _can_damage_spirit.
#   slot_can_damage_spirit            — slot character can land physical
#                                       hits on incorporeal targets.
#
# ── Tier variables (vars on the POD_combat_slot story) ───────────
# Tier = 1 (small, ±10), 2 (medium, ±20), 3 (large, ±30). Live stats are
# rebuilt from base + sum of active tier magnitudes. See pod_combat_tier_effects.txt.
#   slot_{stat}_{direction}_{duration}_tier
#     stat:      soak | hit
#     direction: buff | debuff
#     duration:  round (cleared at round end) | combat (lasts the fight)
#
# ── Workspace (acting combatant + current target) ────────────────
# Mirrors the per-slot fields for whoever is currently acting / targeted.
# Lives on the combat host so saved scopes / scripted-effect parameters
# can read it without crossing into the slot-story scope.
#   pod_combat_actor_*                — actor's stats (loaded from current_actor slot)
#   pod_combat_target_*               — target's stats (loaded from selected_target slot)
# The same tier, stance, retaliation, spirit, and base_* fields exist on
# both sides — load/save effects walk slot stories and write into / read
# back from the matching workspace mirror.
#   pod_combat_current_hit_chance     — active attacker's hit chance this attack
#
# ── Result / display ─────────────────────────────────────────────
#   pod_combat_winner                 — character ref: who won (or team leader)
#   pod_combat_loser                  — character ref: who lost
#   (the move story is no longer host-side; push_log receives it as
#    an explicit MOVE_STORY parameter from POD_execute_combat_move §16)
#   pod_combat_last_damage            — last HP damage amount dealt
#   pod_combat_last_heal_amount       — last HP heal amount applied
#   pod_combat_last_morale_damage_amount — last morale damage amount
#   pod_combat_last_morale_heal_amount   — last morale heal amount
#   pod_combat_last_result_code       — last action's resolved outcome
#                                       as a single int (B1):
#                                          -1 = no result yet (sentinel)
#                                           0 = HP damage
#                                           1 = guard-blocked
#                                           2 = soaked
#                                           3 = missed
#                                           5 = dodged
#                                           6 = HP damage + retaliation
#                                           7 = HP damage + retaliate soaked
#                                           8 = utility (no damage)
#                                           9 = morale damage only
#                                          10 = HP heal
#                                          11 = morale heal
#                                          12 = stealthed
#                                       Replaces the 11 pod_combat_last_was_*
#                                       presence flags this used to use.
#
# ── AI move-history (per-actor, NOT per-host) ────────────────────
# These live on the acting AI character object — saved by
# POD_execute_combat_move via save_actor_to_slot, cleared during
# slot cleanup (pod_combat_group_cleanup_effect). They are STORY
# REFERENCES (point at a POD_combat_move scope), not flags. The AI
# selector uses `scope:move = scope:ai_actor.var:pod_combat_ai_last_move`
# style scope-equality checks for anti-repetition penalties.
#   pod_combat_ai_last_move           — story ref: most recent move
#   pod_combat_ai_prev_move           — story ref: 2 turns ago
#   pod_combat_ai_prev2_move          — story ref: 3 turns ago
#
# Hit chance formula: 60 + (own prowess − target prowess) / 2, clamped 20-95.
# ────────────────────────────────────────────────────────────────────

###### CONSTANTS ######
# Tier magnitudes — small/medium/large = ±10/±20/±30, applied by the
# 8 tier-mag SVs (soak/hit × buff/debuff × round/combat) per slot.
@combat_tier_small_mag  = 10
@combat_tier_medium_mag = 20
@combat_tier_large_mag  = 30

# Hit chance bounds — every per-slot _hit_chance_value clamps to this
# window. Mirrors the bounds documented in the header comment.
@combat_hit_chance_min = 20
@combat_hit_chance_max = 95

# Percentage cap — used by morale_pct (×100/max_morale) and soak_value
# (already 0-100 by construction, clamped defensively).
@combat_pct_max = 100

###### BASE CONSTANTS ######

pod_combat_base_ap = 6

pod_combat_hit_chance_base = 60

# Morale is still shared across all slots (fixed 20 cap).
pod_combat_max_morale = 20

# Use to calculate the amount of morale damage taken when you suffer HP damage
# morale damage = HP damage * 2

pod_combat_morale_from_hp_ratio = {
	value = 2
}

POD_current_combat_version = 1

# Maximum equipped moves per combat_category (ATK/DEF/CTL/PWR).
pod_combat_max_per_category = 4

# Hard cap on total equipped moves across all categories.
pod_combat_max_equipped = 16


# Mental-strain scaling for morale damage in the combat minigame.
# Combat morale is 0-20, PoD stress is 0-100, so each point of morale
# damage translates to 5 stress applied as Mental Strain (static
# stress — unaffected by stress-gain modifiers). A full morale-broken
# character (20 morale damage) takes 100 stress.
# Adjust this knob to rebalance stress attrition.
pod_combat_morale_stress_multiplier = 5

###### CURRENT VALUE ACCESSORS ######
# Used by GUI and triggers to read combat state.
# Guarded with has_variable so CK3 doesn't log errors when the GUI
# evaluates these script values outside an active combat.

pod_combat_current_ap_value = {
	if = { limit = { has_variable = pod_combat_current_ap } value = var:pod_combat_current_ap }
}

pod_combat_round_value = {
	if = { limit = { has_variable = pod_combat_round } value = var:pod_combat_round }
}

pod_combat_actor_hp_value = {
	# Live-derived from the character's PoD wound track: HP = empty boxes.
	# The has_variable + is_alive guards keep tooltip walks quiet when 
	# no active combat / char is dead.
	if = {
		limit = {
			has_variable = pod_combat_actor_char
			var:pod_combat_actor_char = { is_alive = yes }
		}
		value = var:pod_combat_actor_char.POD_empty_damage_boxes
	}
}

pod_combat_actor_max_hp_value = {
	if = {
		limit = {
			has_variable = pod_combat_actor_char
			var:pod_combat_actor_char = { is_alive = yes }
		}
		value = var:pod_combat_actor_char.POD_health_track_length
	}
}

###### GROUP COMBAT ACTOR / TARGET ######
# Used by GUI to highlight the current actor and selected target

pod_combat_current_actor_value = {
	if = { limit = { has_variable = pod_combat_current_actor } value = var:pod_combat_current_actor }
}

pod_combat_pc_target_value = {
	# Drives the TARGETED badge on combat cards. Reads from the PC's
	# persistent target choice (not the transient selected_target, which can
	# be clobbered during AI turns). Defaults to -1 when unset so nothing
	# highlights before combat starts.
	value = -1
	if = { limit = { has_variable = pod_combat_pc_target } value = var:pod_combat_pc_target }
}

pod_combat_actor_sup_ceiling_value = {
	# The effective ceiling for SUPERFICIAL HP healing on the player:
	#   max_hp - agg_damage  (i.e. track_length - aggravated_boxes)
	# Aggravated damage blocks the top of the HP track, so only the space below
	# this ceiling can be refilled by pod_combat_restore_hp_superficial_effect.
	# The heal-availability trigger uses this to grey out superficial-heal buttons
	# when the player has nothing left to heal.
	if = {
		limit = {
			has_variable = pod_combat_actor_char
			var:pod_combat_actor_char = { is_alive = yes }
		}
		value = var:pod_combat_actor_char.POD_health_track_length
		subtract = var:pod_combat_actor_char.POD_aggravated_damage_boxes
		min = 0
	}
}

###### SLOT-STORY-SCOPED VALUE ACCESSORS ######
# Caller scope = the slot story (set externally via
# datacontext = "[GetPlayer.MakeScope.Var('pod_combat_slot_$N$')]"
# from a single-instance widget invocation, OR implicitly via a
# datamodel iteration over pod_combat_slots / pod_combat_active_slots —
# the iteration scope IS the iterated story).
#
# These read slot vars off the current scope (= slot story) directly
# via var:slot_X. Used by the datamodel-driven pod_combat_datamodel_slot
# widget for the GUI's slot card rendering, which iterates the host's
# pod_combat_active_slots variable list and emits one card per filled
# slot.
#
# All gates short-circuit on tooltip-build (where slot stories don't
# exist — pod_combat_create_one_slot_story_effect's create_story is
# wrapped in hidden_effect_new_object). The datamodel iteration
# itself is also a no-op during tooltip-build because pod_combat_slots
# is populated inside the same hidden_effect_new_object block.

pod_combat_slot_max_hp_value = {
	# Track length survives death (trait-based), so no is_alive gate.
	if = {
		limit = { has_variable = slot_char }
		value = var:slot_char.POD_health_track_length
	}
}
pod_combat_slot_total_damage_sv = {
	# Total wound boxes filled. Dead-char branch returns max_hp so the
	# GUI shows a fully-filled bar instead of zero (PoD's wound SVs
	# return 0 for dead chars via their own is_alive gate).
	if = {
		limit = { has_variable = slot_char var:slot_char = { is_alive = yes } }
		value = var:slot_char.POD_total_damage_boxes
		min = 0
	}
	if = {
		limit = { has_variable = slot_char var:slot_char = { is_alive = no } }
		value = pod_combat_slot_max_hp_value
	}
}
pod_combat_slot_agg_damage_sv = {
	# Aggravated boxes. Same alive/dead branching as total_damage_sv.
	if = {
		limit = { has_variable = slot_char var:slot_char = { is_alive = yes } }
		value = var:slot_char.POD_aggravated_damage_boxes
	}
	if = {
		limit = { has_variable = slot_char var:slot_char = { is_alive = no } }
		value = pod_combat_slot_max_hp_value
	}
	max = pod_combat_slot_total_damage_sv
}
pod_combat_slot_morale_pct = {
	value = var:slot_morale
	multiply = @combat_pct_max
	divide = pod_combat_max_morale
	min = 0
	max = @combat_pct_max
}
pod_combat_slot_soak_value = {
	value = var:slot_soak
	min = 0
	max = @combat_pct_max
}
pod_combat_slot_hit_chance_value = {
	# Two paths:
	#   1. slot_hit_chance is set → return it (post calculate_all_hit_chances).
	#   2. slot_hit_chance NOT set but slot_char is → synthesize a
	#      neutral baseline (the brief mid-init window before
	#      pod_combat_calculate_all_hit_chances_effect populates per-
	#      slot hit_chance).
	if = {
		limit = { has_variable = slot_hit_chance }
		value = var:slot_hit_chance
	}
	if = {
		limit = {
			NOT = { has_variable = slot_hit_chance }
			has_variable = slot_char
		}
		value = pod_combat_hit_chance_base
		# No prowess-delta synthesis here because the slot story doesn't
		# have a back-reference to the host (and thus no access to the
		# opposing slot's prowess). The current actor / target dance in
		# the workspace already handles correct hit chance during
		# combat — this fallback is for the brief init window only.
		min = @combat_hit_chance_min
		max = @combat_hit_chance_max
	}
}
pod_combat_slot_prowess_value = {
	value = var:slot_prowess
}
pod_combat_slot_alive_value = {
	value = var:slot_alive
}
pod_combat_slot_is_guarding_value = {
	if = { limit = { has_variable = slot_is_guarding  } value = 1 }
}
pod_combat_slot_is_dodging_value = {
	if = { limit = { has_variable = slot_is_dodging   } value = 1 }
}
pod_combat_slot_is_stealthed_value = {
	if = { limit = { has_variable = slot_is_stealthed } value = 1 }
}
pod_combat_slot_initiative_value = {
	if = { limit = { has_variable = slot_initiative } value = var:slot_initiative }
}
