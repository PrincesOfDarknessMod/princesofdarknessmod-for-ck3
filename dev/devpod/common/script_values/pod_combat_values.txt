# Host scope. Escalating AP reads the round through pod_combat_session;
# fight-scope evaluation incorrectly returns the base cap. No fight = base cap.

	value = pod_combat_base_ap

		limit = {
			exists = global_var:pod_combat_tune_ap_cap_increment
			has_variable = pod_combat_session
		}
		var:pod_combat_session = {
			if = {
				limit = { has_variable = pod_combat_round }
				value = 1
				if = { limit = { exists = global_var:pod_combat_tune_ap_cap_start } value = global_var:pod_combat_tune_ap_cap_start }
				add = {
					value = var:pod_combat_round
					subtract = 1
					multiply = global_var:pod_combat_tune_ap_cap_increment
				}
				max = pod_combat_base_ap
			}
		}
	}
}

	value = scope:ar_lose_score
	multiply = 1.5
}

	value = scope:ar_lose_score
	multiply = 2.5

# Requires scope:ar_win_score and scope:ar_lose_score. Spirit use wound_max if there isn't a way to deal damage.
# Zero-vs-zero counts as close, avoiding division and an incorrect highest band.
pod_combat_auto_resolve_loser_wounds_value = {
	value = pod_combat_auto_resolve_wound_base
	if = {
		limit = {
			scope:ar_win_score > 0
			scope:ar_win_score >= pod_combat_auto_resolve_margin_wide_value
		}
		add = 1
	}
	if = {
		limit = {
			scope:ar_win_score > 0
			scope:ar_win_score >= pod_combat_auto_resolve_margin_blowout_value
		}
		add = 1
	}
	max = pod_combat_auto_resolve_wound_max
}

###### MOVE-SHEET (spreadsheet pool view) ######
# Sort key for pod_combat_sheet_rebuild_effect's ordered_in_global_list.
# Evaluated in each move-story scope; the viewer is scope:pod_combat_sheet_owner
# (saved by the rebuild effect just before iterating).
# ordered_in_list iterates DESCENDING, so ascending = negate the raw stat.
pod_combat_sheet_sort_key_value = {
	value = 0
	if = {
		limit = { pod_combat_sheet_sort_is_trigger = { KEY = ap } }
		add = var:ap_cost
	}
	# Energy is signed like morale below: cost sorts positive, restore
	# negative, summed so a move carrying both ranks on its net.
	else_if = {
		limit = { pod_combat_sheet_sort_is_trigger = { KEY = energy } }
		if = {
			limit = { has_variable = energy_cost }
			add = var:energy_cost
		}
		if = {
			limit = { has_variable = energy_gain }
			subtract = var:energy_gain
		}
	}
	else_if = {
		limit = {
			pod_combat_sheet_sort_is_trigger = { KEY = damage }
			has_variable = hp_damage
		}
		add = var:hp_damage
	}
	else_if = {
		limit = {
			pod_combat_sheet_sort_is_trigger = { KEY = heal }
			has_variable = hp_heal
		}
		add = var:hp_heal
	}
	# Signed net values: damage/AP gain positive, healing/AP drain negative.
	# Sum both components for hybrid moves.
	else_if = {
		limit = { pod_combat_sheet_sort_is_trigger = { KEY = morale } }
		if = {
			limit = { has_variable = morale_damage }
			add = var:morale_damage
		}
		if = {
			limit = { has_variable = morale_heal }
			subtract = var:morale_heal
		}
	}
	else_if = {
		limit = {
			pod_combat_sheet_sort_is_trigger = { KEY = dot }
			has_variable = dot_amount
		}
		add = var:dot_amount
	}
	else_if = {
		limit = { pod_combat_sheet_sort_is_trigger = { KEY = apswing } }
		if = {
			limit = { has_variable = ap_gain }
			add = var:ap_gain
		}
		if = {
			limit = { has_variable = ap_drain }
			subtract = var:ap_drain
		}
	}
}

# Recipient-slot scope. Shared urgency for AI recipient scans and ally swaps;
# unset, dead or typeless slots return the base.

	value = pod_combat_ai_heal_base_value
	if = {
		limit = { has_variable = slot_char }
		if = {
			limit = { exists = var:slot_char }
			if = {
				limit = { var:slot_char = { is_alive = yes } }
				add = {
					value = var:slot_char.POD_health_track_length
					# Preserve PoD caps/floors through the canonical wound numerator.
					save_temporary_value_as = pod_combat_urgency_max_hp
					if = {
						limit = { scope:pod_combat_urgency_max_hp > 0 }
						value = var:slot_char.POD_total_damage_boxes
						divide = scope:pod_combat_urgency_max_hp
						multiply = pod_combat_ai_heal_span_value
					}
					else = { value = 0 }
				}
			}
		}
	}
}
