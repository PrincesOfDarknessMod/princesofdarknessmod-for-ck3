######################################
######   POD COMBAT VALUES   ######
######################################

# ─── COMBAT VARIABLE REFERENCE ─────────────────────────────────────────────
#
# All combat state is stored as character variables on the PLAYER character.
# Group combat uses a 10-slot system (slots 0-4 = player team, 5-9 = enemy
# team). A shared WORKSPACE (pod_combat_player_* / pod_combat_opponent_*)
# holds whichever combatant is acting and whichever one is the current
# target; the workspace is loaded from / saved back to the appropriate
# slot at each action boundary by pod_combat_{load,save}_workspace_effect.
# Skill effects read and write only the workspace, so the same generated
# code runs for the PC, AI coterie, the primary opponent, and knights.
#
# ── Core state ──────────────────────────────────────────────────
#   pod_combat_active                 — flag: combat is in progress
#   pod_combat_is_over                — flag: combat is over (re-entry guard)
#   pod_combat_mode                   — 0=1v1, 1=1vN, 2=Nv1, 3=NvN
#   pod_combat_round                  — current round number
#   pod_combat_current_actor          — slot index of the acting combatant (0-9)
#   pod_combat_current_ap             — AP remaining this turn
#   pod_combat_current_turn_side      — 0 = player-team turn, 1 = enemy-team turn
#   pod_combat_enemy_team_size        — number of occupied enemy slots (5-9);
#                                       used by the slots-6-9 fill pass to
#                                       gate how many enemies still fit
#   pod_combat_pc_target              — the PC's persistent target slot (1-9)
#   pod_combat_selected_target        — slot loaded into opponent workspace
#
# ── Per-slot stats (N = 0-9) ─────────────────────────────────────
#   pod_combat_c{N}_char              — character ref occupying the slot
#   pod_combat_c{N}_alive             — 1 if slot is active, 0 if KO'd
#   (team affiliation is computed from slot index — slots 0-4 = player
#    team, 5-9 = enemy team — not stored as a per-slot variable)
# Health values are NOT stored as per-slot variables — they are live
# script values derived from each slot character's PoD wound vars:
#   pod_combat_c{N}_hp_value          — empty wound boxes (HP remaining)
#   pod_combat_c{N}_max_hp_value      — wound track length (max HP)
#   pod_combat_c{N}_agg_damage_sv     — aggravated wound boxes
#   pod_combat_c{N}_total_damage_sv   — total wound boxes filled
# (see COMBAT_DESIGN_HISTORY.md §2 for the cache-removal rationale)
#   pod_combat_c{N}_morale            — current Morale
#   pod_combat_c{N}_prowess           — current prowess
#   pod_combat_c{N}_soak              — live soak chance (0-100)
#   pod_combat_c{N}_base_soak         — soak before any tier effects
#   pod_combat_c{N}_hit_chance        — live hit chance (20-95)
#   pod_combat_c{N}_base_hit_chance   — hit chance before any tier effects
#   pod_combat_c{N}_initiative        — initiative roll this round
#   pod_combat_c{N}_init_bonus        — next-round initiative modifier
#   pod_combat_c{N}_ap_drain          — AP drained from next turn
#   pod_combat_c{N}_energy_used       — counter for AI energy-attrition weighting
#
# ── Per-slot stance flags (N = 0-9) ─────────────────────────────
# All persist across rounds; only cleared when consumed or at combat end.
#   pod_combat_c{N}_is_guarding       — Guard stance active
#   pod_combat_c{N}_is_dodging        — Dodge stance active
#   pod_combat_c{N}_is_stealthed      — Stealth active (includes +30 hit)
#   pod_combat_c{N}_retaliate_hit     — retaliation damage if hit
#   pod_combat_c{N}_retaliate_miss    — retaliation damage if missed
#   pod_combat_c{N}_retaliate_{hit,miss}_agg  — aggravated variants
#
# ── Tier variables (per slot) ─────────────────────────────────────
# Tier = 1 (small, ±10), 2 (medium, ±20), 3 (large, ±30). Live stats are
# rebuilt from base + sum of active tier magnitudes. See pod_combat_tier_effects.txt.
#   pod_combat_c{N}_{stat}_{direction}_{duration}_tier
#     stat:      soak | hit
#     direction: buff | debuff
#     duration:  round (cleared at round end) | combat (lasts the fight)
#
# ── Workspace (acting combatant + current target) ────────────────
# Mirrors the per-slot fields for whoever is currently acting / targeted.
#   pod_combat_player_*               — actor's stats (loaded from current_actor slot)
#   pod_combat_opponent_*             — target's stats (loaded from selected_target slot)
# The same tier, stance, retaliation, and base_* fields exist on both sides.
#   pod_combat_current_hit_chance     — active attacker's hit chance this attack
#
# ── Result / display ─────────────────────────────────────────────
#   pod_combat_winner                 — character ref: who won (or team leader)
#   pod_combat_loser                  — character ref: who lost
#   pod_combat_executing_move         — story scope: the move that just ran
#                                       (log datamodel reads its vars directly)
#   pod_combat_last_damage            — last damage amount dealt
#   pod_combat_last_was_blocked       — flag: last attack was guard-blocked
#   pod_combat_last_was_soaked        — flag: last attack was soaked
#   pod_combat_last_was_missed        — flag: last attack missed the hit roll
#   pod_combat_last_was_dodged        — flag: last attack dodged or caught by stealth
#   pod_combat_last_was_retaliated    — flag: target fired a retaliation stance
#   pod_combat_last_was_heal          — flag: last action was a heal
#   pod_combat_last_was_morale_*      — flags: morale damage / heal variants
#
# ── AI move-history (per-actor, NOT per-host) ────────────────────
# These live on the acting AI character object — saved by
# POD_execute_combat_move via save_actor_to_slot, cleared during
# slot cleanup (pod_combat_group_cleanup_effect). They are STORY
# REFERENCES (point at a POD_combat_move scope), not flags. The AI
# selector uses `scope:move = scope:ai_actor.var:pod_combat_ai_last_move`
# style scope-equality checks for anti-repetition penalties.
#   pod_combat_ai_last_move           — story ref: most recent move
#   pod_combat_ai_prev_move           — story ref: 2 turns ago
#   pod_combat_ai_prev2_move          — story ref: 3 turns ago
#
# Hit chance formula: 60 + (own prowess − target prowess) / 2, clamped 20-95.
# ────────────────────────────────────────────────────────────────────

###### CONSTANTS ######
# Tier magnitudes — small/medium/large = ±10/±20/±30, applied by the
# 8 tier-mag SVs (soak/hit × buff/debuff × round/combat) per slot.
@combat_tier_small_mag  = 10
@combat_tier_medium_mag = 20
@combat_tier_large_mag  = 30

# Hit chance bounds — every per-slot _hit_chance_value clamps to this
# window. Mirrors the bounds documented in the header comment.
@combat_hit_chance_min = 20
@combat_hit_chance_max = 95

# Percentage cap — used by morale_pct (×100/max_morale) and soak_value
# (already 0-100 by construction, clamped defensively).
@combat_pct_max = 100

###### BASE CONSTANTS ######

pod_combat_base_ap = 6

pod_combat_hit_chance_base = 60

# Morale is still shared across all slots (fixed 20 cap).
pod_combat_max_morale = 20

# Use to calculate the amount of morale damage taken when you suffer HP damage
# morale damage = HP damage * 2

pod_combat_morale_from_hp_ratio = {
	value = 2
}

# Used by POD_reload_combat_moves to detect save-data mismatch.

POD_current_combat_version = 1

# Maximum equipped moves per combat_category (ATK/DEF/CTL/PWR).
pod_combat_max_per_category = 4

# Hard cap on total equipped moves across all categories.
pod_combat_max_equipped = 16


# Mental-strain scaling for morale damage in the combat minigame.
# Combat morale is 0-20, PoD stress is 0-100, so each point of morale
# damage translates to 5 stress applied as Mental Strain (static
# stress — unaffected by stress-gain modifiers). A full morale-broken
# character (20 morale damage) takes 100 stress.
# Adjust this knob to rebalance stress attrition.
pod_combat_morale_stress_multiplier = 5

###### CURRENT VALUE ACCESSORS ######
# Used by GUI and triggers to read combat state.
# Guarded with has_variable so CK3 doesn't log errors when the GUI
# evaluates these script values outside an active combat.

pod_combat_current_ap_value = {
	if = { limit = { has_variable = pod_combat_current_ap } value = var:pod_combat_current_ap }
}

pod_combat_round_value = {
	if = { limit = { has_variable = pod_combat_round } value = var:pod_combat_round }
}

pod_combat_player_hp_value = {
	# Live-derived from the character's PoD wound track: HP = empty boxes.
	# The has_variable + is_alive guards keep tooltip walks quiet when 
	# no active combat / char is dead.
	if = {
		limit = {
			has_variable = pod_combat_player_char
			var:pod_combat_player_char = { is_alive = yes }
		}
		value = var:pod_combat_player_char.POD_empty_damage_boxes
	}
}

pod_combat_player_max_hp_value = {
	if = {
		limit = {
			has_variable = pod_combat_player_char
			var:pod_combat_player_char = { is_alive = yes }
		}
		value = var:pod_combat_player_char.POD_health_track_length
	}
}

###### GROUP COMBAT ACTOR / TARGET ######
# Used by GUI to highlight the current actor and selected target

pod_combat_current_actor_value = {
	if = { limit = { has_variable = pod_combat_current_actor } value = var:pod_combat_current_actor }
}

pod_combat_pc_target_value = {
	# Drives the TARGETED badge on combat cards. Reads from the PC's
	# persistent target choice (not the transient selected_target, which can
	# be clobbered during AI turns). Defaults to -1 when unset so nothing
	# highlights before combat starts.
	value = -1
	if = { limit = { has_variable = pod_combat_pc_target } value = var:pod_combat_pc_target }
}

pod_combat_player_sup_ceiling_value = {
	# The effective ceiling for SUPERFICIAL HP healing on the player:
	#   max_hp - agg_damage  (i.e. track_length - aggravated_boxes)
	# Aggravated damage blocks the top of the HP track, so only the space below
	# this ceiling can be refilled by pod_combat_restore_hp_superficial_effect.
	# The heal-availability trigger uses this to grey out superficial-heal buttons
	# when the player has nothing left to heal.
	if = {
		limit = {
			has_variable = pod_combat_player_char
			var:pod_combat_player_char = { is_alive = yes }
		}
		value = var:pod_combat_player_char.POD_health_track_length
		subtract = var:pod_combat_player_char.POD_aggravated_damage_boxes
		min = 0
	}
}

pod_combat_c0_hp_value = {
	# HP = empty boxes on the character's PoD wound track. Returns 0
	# for dead chars (is_alive guard keeps the wound SVs from lying
	# about full HP post-death). Guard also keeps tooltip previews
	# quiet when no active combat populated the slot's char var.
	if = {
		limit = {
			has_variable = pod_combat_c0_char
			var:pod_combat_c0_char = { is_alive = yes }
		}
		value = var:pod_combat_c0_char.POD_empty_damage_boxes
	}
}
pod_combat_c0_max_hp_value = {
	# Track length survives death (trait-based), so no is_alive gate.
	if = {
		limit = { has_variable = pod_combat_c0_char }
		value = var:pod_combat_c0_char.POD_health_track_length
	}
}
pod_combat_c0_morale_value = {
	if = { limit = { has_variable = pod_combat_c0_morale } value = var:pod_combat_c0_morale }
}
pod_combat_c0_morale_pct = {
	if = { limit = { has_variable = pod_combat_c0_morale }
		value = var:pod_combat_c0_morale
		multiply = @combat_pct_max
		divide = pod_combat_max_morale
	}
	min = 0
	max = @combat_pct_max
}
pod_combat_c0_total_damage_sv = {
	# Total wound-boxes filled on the char's PoD track. Alive path
	# reads POD_total_damage_boxes; dead path returns max_hp so the
	# GUI shows a fully-filled "damaged" bar instead of the SV-zero
	# that dead-char wound boxes would return.
	if = {
		limit = {
			has_variable = pod_combat_c0_char
			var:pod_combat_c0_char = { is_alive = yes }
		}
		value = var:pod_combat_c0_char.POD_total_damage_boxes
		min = 0
	}
	if = {
		limit = {
			has_variable = pod_combat_c0_char
			var:pod_combat_c0_char = { is_alive = no }
		}
		value = pod_combat_c0_max_hp_value
	}
}
pod_combat_c0_agg_damage_sv = {
	# Aggravated wound boxes on the char. Same alive/dead branching
	# as total_damage_sv — dead chars render as fully aggravated.
	if = {
		limit = {
			has_variable = pod_combat_c0_char
			var:pod_combat_c0_char = { is_alive = yes }
		}
		value = var:pod_combat_c0_char.POD_aggravated_damage_boxes
	}
	if = {
		limit = {
			has_variable = pod_combat_c0_char
			var:pod_combat_c0_char = { is_alive = no }
		}
		value = pod_combat_c0_max_hp_value
	}
	max = pod_combat_c0_total_damage_sv
}
pod_combat_c0_soak_value = {
	if = { limit = { has_variable = pod_combat_c0_soak } value = var:pod_combat_c0_soak }
	min = 0
	max = @combat_pct_max

pod_combat_c0_hit_chance_value = {
	if = { limit = { has_variable = pod_combat_c0_hit_chance }
		value = var:pod_combat_c0_hit_chance
	}
	if = { limit = { NOT = { has_variable = pod_combat_c0_hit_chance } has_variable = pod_combat_c0_prowess }
		value = pod_combat_hit_chance_base
		add = {
			value = var:pod_combat_c0_prowess
			# Subtract currently targeted enemy's prowess
			if = { limit = { has_variable = pod_combat_pc_target var:pod_combat_pc_target = 5 has_variable = pod_combat_c5_prowess } subtract = var:pod_combat_c5_prowess }
			if = { limit = { has_variable = pod_combat_pc_target var:pod_combat_pc_target = 6 has_variable = pod_combat_c6_prowess } subtract = var:pod_combat_c6_prowess }
			if = { limit = { has_variable = pod_combat_pc_target var:pod_combat_pc_target = 7 has_variable = pod_combat_c7_prowess } subtract = var:pod_combat_c7_prowess }
			if = { limit = { has_variable = pod_combat_pc_target var:pod_combat_pc_target = 8 has_variable = pod_combat_c8_prowess } subtract = var:pod_combat_c8_prowess }
			if = { limit = { has_variable = pod_combat_pc_target var:pod_combat_pc_target = 9 has_variable = pod_combat_c9_prowess } subtract = var:pod_combat_c9_prowess }
			if = {
				limit = {
					NOT = { has_variable = pod_combat_pc_target }
					has_variable = pod_combat_c5_prowess
				}
				subtract = var:pod_combat_c5_prowess
			}
			divide = 2
		}
	}
	min = @combat_hit_chance_min
	max = @combat_hit_chance_max
}

pod_combat_c7_hp_value = {
	# HP = empty boxes on the character's PoD wound track. Returns 0
	# for dead chars (is_alive guard keeps the wound SVs from lying
	# about full HP post-death). Guard also keeps tooltip previews
	# quiet when no active combat populated the slot's char var.
	if = {
		limit = {
			has_variable = pod_combat_c7_char
			var:pod_combat_c7_char = { is_alive = yes }
		}
		value = var:pod_combat_c7_char.POD_empty_damage_boxes
	}
}

	}

pod_combat_c7_total_damage_sv = {
	# Total wound-boxes filled on the char's PoD track. Alive path
	# reads POD_total_damage_boxes; dead path returns max_hp so the
	# GUI shows a fully-filled "damaged" bar instead of the SV-zero
	# that dead-char wound boxes would return.
	if = {
		limit = {
			has_variable = pod_combat_c7_char
			var:pod_combat_c7_char = { is_alive = yes }
		}
		value = var:pod_combat_c7_char.POD_total_damage_boxes
		min = 0
	}
	if = {
		limit = {
			has_variable = pod_combat_c7_char
			var:pod_combat_c7_char = { is_alive = no }
		}
		value = pod_combat_c7_max_hp_value
	}
}

pod_combat_c9_hp_value = {
	# HP = empty boxes on the character's PoD wound track. Returns 0
	# for dead chars (is_alive guard keeps the wound SVs from lying
	# about full HP post-death). Guard also keeps tooltip previews
	# quiet when no active combat populated the slot's char var.
	if = {
		limit = {
			has_variable = pod_combat_c9_char
			var:pod_combat_c9_char = { is_alive = yes }
		}
		value = var:pod_combat_c9_char.POD_empty_damage_boxes
	}
}
pod_combat_c9_max_hp_value = {
	# Track length survives death (trait-based), so no is_alive gate.
	if = {
		limit = { has_variable = pod_combat_c9_char }
		value = var:pod_combat_c9_char.POD_health_track_length
	}
}

pod_combat_c9_total_damage_sv = {
	# Total wound-boxes filled on the char's PoD track. Alive path
	# reads POD_total_damage_boxes; dead path returns max_hp so the
	# GUI shows a fully-filled "damaged" bar instead of the SV-zero
	# that dead-char wound boxes would return.
	if = {
		limit = {
			has_variable = pod_combat_c9_char
			var:pod_combat_c9_char = { is_alive = yes }
		}
		value = var:pod_combat_c9_char.POD_total_damage_boxes
		min = 0
	}
	if = {
		limit = {
			has_variable = pod_combat_c9_char
			var:pod_combat_c9_char = { is_alive = no }
		}
		value = pod_combat_c9_max_hp_value
	}
}

}
