# AI target-selection thresholds, read by the finishable flag and by the shared
# target-selector core's kill-priority factor (both in pod_combat_group_effects.txt).
# Script values, not @-defines: @-defines are file-local and these are tuned here but
# read there.
#
# Two tracks, two thresholds each. FINISH = the low-track threshold that raises the
# finishable flag the intent lottery reads. That flag is a priority heuristic - the
# threshold plus ownership of an AP-affordable move of that kind - and proves nothing
# about energy, once-per availability, damage magnitude, soak, immunity or the hit
# roll; pod_combat_ai_flag_finishable_enemy_effect is the definition.
# KILL-PRIORITY = close enough to be worth converging on, and gates the selector's x3.
# Priority sits one step above finish on both tracks so the AI starts massing a hit
# before a target reaches the finish band. Keep priority >= finish: inverted, the
# selector would weight enemies further from a kill above the ones nearer it.

# AI soak-buff cap threshold (percent). At-or-above this workspace soak, a
# pure soak buff scores as waste (@ai_soak_buff_capped in
# pod_combat_ai_effects.txt): the recalc clamps soak at 100, so buffing past
# the cap does nothing. Script value, not an @-define: tuned here, read in the
# AI weight block.
# Since soak buffs went multiplicative they can only approach 100, never reach
# it, so at the default this fires just for characters already pinned at 100 by
# a flat source (Fae deja vu). A buff is never fully wasted below that, but it
# does shrink in absolute percentage points as soak climbs - lower the
# threshold if the AI should stop stacking soak on an already-tanky actor.
pod_combat_ai_soak_buff_cap_threshold = 100

# ── MORALE CAP ────────────────────────────────────────────────────
# The cap is per-character: it scales off the same PoD wound track that
# supplies max HP, so a sturdier creature carries a deeper morale pool. That
# makes it scope-dependent, which is why there is no single "the cap" value.
# Read the wrapper that matches the scope you are in:
#
#   character scope  -> pod_combat_char_max_morale_value
#   slot-story scope -> pod_combat_slot_max_morale_value
#   combat-host scope-> pod_combat_actor_max_morale_value
#                       pod_combat_target_max_morale_value
#
# pod_combat_max_morale_base is the BASELINE only (what a 6-box combatant
# gets). Comparing a combatant's morale against it directly is a bug.

pod_combat_max_morale_base = {
	value = 20
	if = { limit = { exists = global_var:pod_combat_tune_max_morale } value = global_var:pod_combat_tune_max_morale }
}

# CHARACTER scope. Exists so the slot- and host-scope wrappers can chain it off
# a character pointer, the same reason pod_combat_char_prowess_value does.
# POD_health_track_length is trait-derived and survives death, so no is_alive
# gate (matching pod_combat_slot_max_hp_value).
#
# 6 boxes = the baseline, and each box either side is worth 5 morale. PoD's
# ladder (POD_health_track_base_length) puts that at mortal 4 = -10 /
# ghoul, mummy, wraith 5 = -5 / vampire, kuei-jin, risen 6 = +-0 /
# fera, demon, fae 7 = +5 / werewolf 8 = +10, and the per-character track
# perks (Shintai of Bone, Age of Kings, ...) ride along for free.
pod_combat_char_max_morale_value = {
	value = pod_combat_max_morale_base
	add = {
		value = POD_health_track_length
		subtract = 6
		multiply = 5
	}
	min = 1
}

# Scope wrappers. Each falls back to the bare baseline when its character

	value = pod_combat_max_morale_base

# TOP-BAND WIDTH. The AI move roll admits only candidates scoring within this much of
# the best-scoring admitted move, then weights them by score as before. 0 = argmax
# only (plus ties); a value above the widest score spread is a no-op, which in practice
# is anything past ~100.
#
# 10 is measured, not chosen: across two shadow fights it redirects ~18-24% of picks -
# the ones where the AI took a move ten or more points below its own best. Re-measure
# with the band-width sweep in parse_ai_trace.py before changing it.
#
# min = 0 is load-bearing. The tune-override pattern below reads the global RAW, and a
# negative width would put the floor ABOVE the top score, emptying the band on every
# action and stalling every AI turn silently. The never-empties-the-pool guarantee
# holds only for width >= 0.
pod_combat_ai_band_width_value = {
	value = 10
	if = { limit = { exists = global_var:pod_combat_tune_ai_band_width } value = global_var:pod_combat_tune_ai_band_width }
	min = 0
}

# Effective per-turn AP cap. Equals the (overridable) base cap unless the
# escalating-cap knob is tuned: then min(base cap, start + (round-1) *
# increment) - e.g. start 1 / increment 1 = 1 AP in round 1, 2 in round 2...
# HOST-scope only (reads var:pod_combat_round); the guarded read keeps it
# safe when the GUI evaluates it outside combat (falls back to base cap).
pod_combat_effective_ap_cap_value = {
	value = pod_combat_base_ap
	if = {
		limit = {
			exists = global_var:pod_combat_tune_ap_cap_increment
			has_variable = pod_combat_round
		}
		value = 1
		if = { limit = { exists = global_var:pod_combat_tune_ap_cap_start } value = global_var:pod_combat_tune_ap_cap_start }
		add = {
			value = var:pod_combat_round
			subtract = 1
			multiply = global_var:pod_combat_tune_ap_cap_increment
		}
		max = pod_combat_base_ap
	}
}

###### DESIGNER EXTENSION HOOKS ######
# Three drop-in tuning knobs for handing a single character a flat bonus (or
# penalty) to combat INITIATIVE, to the morale they START a fight with, or to the
# action points (AP) they get each turn. All return 0, so out of the box they change
# nothing - a future developer fills in a body and it flows through the minigame
# automatically. Each is already summed into the one value that matters (the per-round
# initiative roll; the starting slot_morale; the per-turn AP budget), so there is
# nothing else to wire up.
#
# Keep each body a pure read (conditions + add / subtract, no effects). The SCOPE
# differs by hook - see each one's comment. The two slot-scoped hooks reach the
# character via var:slot_char; the AP hook is evaluated on the acting character
# directly (`this` is the character). Both var:slot_char and var:pod_combat_actor_char
# scope-switches are valid inside a script_value (engine precedent:
# 04_ep2_accolade_values.txt).

pod_combat_initiative_modifier_value = {
	# Summed into this combatant's initiative each round, alongside prowess, the
	# d[1..pod_combat_initiative_roll_max] roll, and any carried slot_init_bonus.
	# Positive = acts sooner. Re-evaluated every round, so it may read live state.
	value = 0
	# Tunable Celerity / Black Wind bonus (debug window): +N initiative per perk
	# owned in the celerity (Vampire) or black_wind (Kuei-jin) speed tree, under one
	# shared knob. Off (0 contribution) while the global is unset. Celerity's level-7+
	# perks are is_ai = no, so AI combatants count at most 9 of that tree's 15.
	if = {
		limit = { exists = global_var:pod_combat_tune_init_celerity_bonus }
		var:slot_char = {
			add = {
				value = "this.perks_in_tree(celerity)"
				multiply = global_var:pod_combat_tune_init_celerity_bonus
			}
		}
	}
	if = {
		limit = { exists = global_var:pod_combat_tune_init_celerity_bonus }
		var:slot_char = {
			add = {
				value = "this.perks_in_tree(black_wind)"
				multiply = global_var:pod_combat_tune_init_celerity_bonus
			}
		}
	}
	# Brave fighters always act a little sooner:
	# if = {
	# 	limit = { var:slot_char = { has_trait = brave } }
	# 	add = 5
	# }
	# Or scale by the character's martial (read in the character's own scope):
	# var:slot_char = { add = { value = martial  divide = 4 } }
}

	}
}

# Sort key for pod_combat_sheet_rebuild_effect's ordered_in_global_list.
# Evaluated in each move-story scope; the viewer is scope:pod_combat_sheet_owner
# (saved by the rebuild effect just before iterating).
# ordered_in_list iterates DESCENDING, so ascending = negate the raw stat.
pod_combat_sheet_sort_key_value = {
	value = 0
	if = {
		limit = { pod_combat_sheet_sort_is_trigger = { KEY = ap } }
		add = var:ap_cost
	}
	else_if = {
		limit = {
			pod_combat_sheet_sort_is_trigger = { KEY = energy }
			has_variable = energy_cost
		}
		add = var:energy_cost
	}
	else_if = {
		limit = {
			pod_combat_sheet_sort_is_trigger = { KEY = damage }
			has_variable = hp_damage
		}
		add = var:hp_damage
	}
	else_if = {
		limit = {
			pod_combat_sheet_sort_is_trigger = { KEY = heal }
			has_variable = hp_heal
		}
		add = var:hp_heal
	}
	# Morale and AP swing are signed: the column shows both polarities in
	# one cell, so the sort has to rank both. Offense sorts positive
	# (damage dealt, AP gained), support negative (morale restored, AP
	# taken off the target), which puts each polarity at its own end
	# instead of stranding half the column on the absent-stat sentinel.
	# Components are summed rather than chained, so a move carrying both
	# (cursed_eucharist deals AND restores morale) ranks on its net.
	else_if = {
		limit = { pod_combat_sheet_sort_is_trigger = { KEY = morale } }
		if = {
			limit = { has_variable = morale_damage }
			add = var:morale_damage
		}
		if = {
			limit = { has_variable = morale_heal }
			subtract = var:morale_heal
		}
	}
	else_if = {
		limit = {
			pod_combat_sheet_sort_is_trigger = { KEY = dot }
			has_variable = dot_amount
		}
		add = var:dot_amount
	}
	else_if = {
		limit = { pod_combat_sheet_sort_is_trigger = { KEY = apswing } }
		if = {
			limit = { has_variable = ap_gain }
			add = var:ap_gain
		}
		if = {
			limit = { has_variable = ap_drain }
			subtract = var:ap_drain
		}
	}
}

pod_combat_sheet_sort_value = {
	value = pod_combat_sheet_sort_key_value
	if = {
		limit = {
			scope:pod_combat_sheet_owner = { has_variable = pod_combat_sheet_asc }
		}
		multiply = -1
	}
	# Post-negation override: moves lacking the sorted stat sink to the
	# bottom in BOTH directions (negating a sentinel would float it to

	# any negated real stat.
	if = {
		limit = { pod_combat_sheet_sorted_stat_absent_trigger = yes }
		value = -100000
	}
}

# opponent. The one absolute percentage on screen is the combat window's top-row
# PC-vs-selected-target readout, which reads var:pod_combat_pc_slot.var:slot_hit_chance.
pod_combat_slot_hit_net_mod_value = {
	value = 0
	add      = pod_combat_slot_hit_buff_round_mag_sv
	add      = pod_combat_slot_hit_buff_combat_mag_sv
	subtract = pod_combat_slot_hit_debuff_round_mag_sv
	subtract = pod_combat_slot_hit_debuff_combat_mag_sv
}

# ── PC hit-chance breakdown (top-bar tooltip) ─────────────────────────
# Host-scoped: read on the PC via GetPlayer.MakeScope.ScriptValue. Guarded on
# the PC slot existing so the configure/preview passes (no slot stories) read 0
# instead of erroring on the chained slot-var derefs.
pod_combat_pc_stealth_bonus_value = {
	# Stealth's +hit contribution: the bonus when the PC slot is stealthed, else 0.
	if = {
		limit = {
			has_variable = pod_combat_pc_slot
			var:pod_combat_pc_slot = { has_variable = slot_is_stealthed }
		}
		value = pod_combat_stealth_hit_bonus
	}
}

pod_combat_pc_hit_net_mod_value = {
	# Net buff/debuff contribution: final - prowess base - stealth. The remainder
	# after the explicitly-shown terms, so the breakdown rows sum to the total.
	if = {
		limit = {
			has_variable = pod_combat_pc_slot
			var:pod_combat_pc_slot = { has_variable = slot_hit_chance  has_variable = slot_base_hit_chance }
		}
		value = var:pod_combat_pc_slot.var:slot_hit_chance
		subtract = var:pod_combat_pc_slot.var:slot_base_hit_chance
		subtract = pod_combat_pc_stealth_bonus_value
	}
}

# Loadout bookkeeping - counts the equipped moves per combat_category
# and in total by iterating the character's pod_combat_loadout variable
# list.
#
# Called on character scope. The GUI reads them via
# GetPlayer.MakeScope.ScriptValue('...').

# Two layers:
#   *_raw - live every_in_list compute, always correct; used by combat logic and loc.
#   public - while the loadout editor is open (pod_combat_config_open) read the
#            *_count_cache host var (refreshed on editor-open + every equip/unequip)
#            so skills-window pip strips don't re-walk the loadout ~36x/frame;
#            otherwise fall back to *_raw. config_open is never set mid-fight, so
#            combat logic never reads the cache and a stale cache can't be observed.
#            Display-mirror pattern (§7).

pod_combat_atk_count_raw = {
	every_in_list = {
		variable = pod_combat_loadout
		limit = { var:combat_category = flag:ATK }
		add = 1
	}
}

pod_combat_def_count_raw = {
	every_in_list = {
		variable = pod_combat_loadout
		limit = { var:combat_category = flag:DEF }
		add = 1
	}
}

pod_combat_ctl_count_raw = {
	every_in_list = {
		variable = pod_combat_loadout
		limit = { var:combat_category = flag:CTL }
		add = 1
	}
}

pod_combat_pwr_count_raw = {
	every_in_list = {
		variable = pod_combat_loadout
		limit = { var:combat_category = flag:PWR }
		add = 1
	}
}

pod_combat_equipped_count_raw = {

		add = 1
	}
}

# Alive combatant counts per team, read by the team-state move-condition
# predicates - `outnumbered` and the alive_enemies_* / alive_allies_* four
# (host scope, where pod_combat_active_slots lives). KO'd slots linger
# in the list, hence the explicit alive filter. These are whole-side counts:
# the ally predicates exclude the acting character by shifting their compare,
# since a script value cannot take the actor as a parameter.
pod_combat_alive_player_count_value = {
	every_in_list = {
		variable = pod_combat_active_slots
		limit = { pod_combat_slot_is_alive_trigger = yes  var:combat_team = flag:player }
		add = 1
	}
}
pod_combat_alive_enemy_count_value = {
	every_in_list = {
		variable = pod_combat_active_slots
		limit = { pod_combat_slot_is_alive_trigger = yes  var:combat_team = flag:enemy }
		add = 1
	}
}
