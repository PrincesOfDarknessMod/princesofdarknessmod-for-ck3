###### CONSTANTS ######
# Tier magnitudes — small/medium/large = ±10/±20/±30, applied by the
# 8 tier-mag SVs (soak/hit × buff/debuff × round/combat) per slot.
combat_tier_small_mag  = 10

combat_tier_large_mag  = 30

###### BASE CONSTANTS ######

pod_combat_base_ap = 6

pod_combat_hit_chance_base = 60

# Morale is still shared across all slots (fixed 20 cap).
pod_combat_max_morale = 20

# Soft per-team roster cap. Teams are otherwise unbounded — the duel
# builder appends combatants by team_position with no structural limit.
# This is a runaway guardrail (a buggy caller that loops add_ally/add_enemy
# forever would otherwise create unbounded slot stories and, via the O(N^2)
# turn-order pass, hang the game). Raise it freely; it is not a real design
# limit. The GUI panels scroll past it without issue.
pod_combat_max_team_size = 50

# Use to calculate the amount of morale damage taken when you suffer HP damage
# morale damage = HP damage * 2

pod_combat_morale_from_hp_ratio = 2

POD_current_combat_version = 1

# Maximum equipped moves per combat_category (ATK/DEF/CTL/PWR).
pod_combat_max_per_category = 4

# Hard cap on total equipped moves across all categories.
pod_combat_max_equipped = 16

# Initiative roll upper bound. Each round, every alive combatant's
# slot_initiative = slot_prowess + (random integer in [1, this value])
# + slot_init_bonus. See pod_combat_calculate_initiative_effect in
# pod_combat_group_effects.txt for the consuming integer_range call.
#
# Tuning:
#   higher → more chaos in turn order (prowess matters less)
#   lower  → more prowess-deterministic (small or no upsets)
#   == 1   → pure prowess ordering (debug only)
pod_combat_initiative_roll_max = 30

# Morale damage = HP damage * pod_combat_morale_from_hp_ratio. Used by
# pod_combat_route_to_pod_damage's spillover-into-morale calculation
# (see pod_combat_effects.txt). Kept as a script value (rather than an
# @ parse-time constant): `multiply = @morale_from_hp_ratio` inside a
# script value nested in a $PARAM$-using effect doesn't resolve and
# would crash the same way the @ min/max bounds in
# pod_combat_calc_one_hit_chance_effect's base_hit value once did
# (literal 20/95 there now; pod_combat_group_effects.txt). The
# per-damage-call SV-eval cost is negligible (one int read), so the
# script-value form stays.
pod_combat_morale_from_hp_ratio = {
	value = 2
}

# Mental-strain scaling for morale damage in the combat minigame.
# Combat morale is 0-20, each point of morale damage
# translates to 5 stress applied as Mental Strain (static
# stress: unaffected by stress-gain modifiers). A full
# morale-broken character (20 morale damage) takes 100 stress.
# Adjust this knob to rebalance stress attrition.
pod_combat_morale_stress_multiplier = 5

# Ally-down morale shock: morale every still-living teammate loses when a
# combatant on their team is knocked out (pod_combat_mark_slot_ko_effect in
# pod_combat_group_effects.txt). Designer-tunable. The loss floors at 1
# morale - an ally falling demoralizes but never morale-breaks anyone.
# Keep in sync with the plain-text amount in the
# pod_combat_log_system_ally_down_shock loc row (POD_combat_static_l_english.yml).
pod_combat_ally_down_morale_penalty = 2

# Stealth's hit-chance bonus while *_is_stealthed is set. Consumed by the
# recalc-core STEALTH_BONUS param and the slot-scope recalc mirror in
# pod_combat_tier_effects.txt. Designer-tunable. Keep in sync with the
# hardcoded "+30" texts in POD_combat_static_l_english.yml:
# pod_combat_status_stealth_tt, pod_combat_desc_stealth,
# game_concept_pod_combat_stealth_desc, and the Stealth bullet in
# pod_combat.0001.desc (help event).
pod_combat_stealth_hit_bonus = 30

# Maximum number of log entries kept in pod_combat_log. Each push
# rebuilds the list (CK3 lists are append-only and we need newest-
# first ordering), so without a cap the rebuild is O(N²) over the
# combat's lifetime. At 50 entries that bounds the per-push cost
# around 50 list-ops; for very long combats (NvN with 10+ actors
# and 12+ rounds) it prevents the list from growing unbounded too.
# Entries dropped beyond the cap get end_story'd so the underlying
# POD_combat_log_entry story_cycle doesn't leak on Charon for the
# rest of the session.
#
# Tune higher if players ask for more scroll-back history; tune
# lower if combat-log rendering ever becomes a perf bottleneck.
pod_combat_log_max_entries = 50

###### CURRENT VALUE ACCESSORS ######
# Used by GUI and triggers to read combat state.
# Guarded with has_variable so CK3 doesn't log errors when the GUI
# evaluates these script values outside an active combat.

pod_combat_current_ap_value = {
	if = { limit = { has_variable = pod_combat_current_ap } value = var:pod_combat_current_ap }
}

# Pip-row helper consumed by the AP-pip strip in pod_combat_window.gui.
# Returns the pip index above which pips render as supercharge instead
# of filled — i.e. the boundary between the two icon types. The pip
# strip's filled-visibility test is `N <= threshold`, supercharge is
# `N > threshold` (both gated by `N <= current_ap`).
#
# Default = pod_combat_base_ap (6) so when current AP is at or below
# the cap, the supercharge test `N > 6` never fires in a 6-pip row.
# When current AP overflows the cap, the threshold collapses to
#   max(0, 2 * base_ap - current_ap)
# which places exactly `current_ap - base_ap` pips into the supercharge
# band at the RIGHT edge of the row:
#   current = 7  -> threshold = 5  -> pip 6 supercharge, 1-5 filled
#   current = 8  -> threshold = 4  -> pips 5-6 supercharge, 1-4 filled
#   current = 9  -> threshold = 3  -> pips 4-6 supercharge, 1-3 filled
#   current = 13 -> threshold = 0  -> all 6 supercharge (excess beyond
#                                     2*base isn't expressible in 6 pips)
# The `min = 0` clamp on the if branch handles the extreme-overcap
# case so the threshold never goes negative.
pod_combat_pip_supercharge_threshold = {
	value = pod_combat_base_ap
	if = {
		limit = { pod_combat_current_ap_value > pod_combat_base_ap }
		value = pod_combat_base_ap
		multiply = 2
		subtract = pod_combat_current_ap_value
		min = 0
	}
}

}

	}
}

###### GROUP COMBAT ACTOR / TARGET ######
# Used by GUI to highlight the current actor and selected target

pod_combat_current_actor_value = {
	# A collision-free identity key for the acting slot: team_position * 2 +
	# (enemy ? 1 : 0), derived from the two-field host pointer. The GUI
	# compares it against each slot's pod_combat_slot_ordinal_value to
	# highlight the actor. The interleaved formula (player/p = 2p, enemy/p =
	# 2p+1) is a bijection for ANY team size, unlike the old (enemy?5:0)+pos.
	# -1 when no actor is set so nothing highlights between turns.
	value = -1
	if = {
		limit = { has_variable = pod_combat_current_actor_team }
		value = var:pod_combat_current_actor_position
		multiply = 2
		if = { limit = { var:pod_combat_current_actor_team = flag:enemy } add = 1 }
	}
}

pod_combat_pc_target_value = {
	# Drives the TARGETED badge on combat cards. Reads the PC's persistent
	# target choice (not the transient selected_target, which can be clobbered
	# during AI turns) as the same collision-free identity key the actor uses
	# (team_position * 2 + is_enemy), for comparison against each slot's
	# pod_combat_slot_ordinal_value. Defaults to -1 when unset so nothing
	# highlights before a target is chosen.
	value = -1
	if = {
		limit = { has_variable = pod_combat_pc_target_team }
		value = var:pod_combat_pc_target_position
		multiply = 2
		if = { limit = { var:pod_combat_pc_target_team = flag:enemy } add = 1 }
	}
}

pod_combat_actor_sup_ceiling_value = {
	# The effective ceiling for SUPERFICIAL HP healing on the player:
	#   max_hp - agg_damage  (i.e. track_length - aggravated_boxes)
	# Aggravated damage blocks the top of the HP track, so only the space below
	# this ceiling can be refilled by pod_combat_restore_hp_superficial_effect.
	# The heal-availability trigger uses this to grey out superficial-heal buttons
	# when the player has nothing left to heal.
	if = {
		limit = {
			has_variable = pod_combat_actor_char
			var:pod_combat_actor_char = { is_alive = yes }
		}
		value = var:pod_combat_actor_char.POD_health_track_length
		subtract = var:pod_combat_actor_char.POD_aggravated_damage_boxes
		min = 0
	}
}

###### SLOT-STORY-SCOPED VALUE ACCESSORS ######
# Caller scope = the slot story (set externally via
# datacontext = "[GetPlayer.MakeScope.Var('pod_combat_slot_$N$')]"
# from a single-instance widget invocation, OR implicitly via a
# datamodel iteration over pod_combat_active_slots — the iteration
# scope IS the iterated story).
#
# These read slot vars off the current scope (= slot story) directly
# via var:slot_X. Used by the datamodel-driven pod_combat_datamodel_slot
# widget for the GUI's slot card rendering, which iterates the host's
# pod_combat_active_slots variable list and emits one card per filled
# slot.
#
# All gates short-circuit on tooltip-build (where slot stories don't
# exist — pod_combat_set_slot_char_effect's lazy-create branch is
# wrapped in hidden_effect_new_object, and pod_combat_active_slots
# is populated inside the same wrap, so the list is empty during
# tooltip preview and the datamodel iteration is a no-op).

pod_combat_slot_max_hp_value = {
	# Track length survives death (trait-based), so no is_alive gate.
	if = {
		limit = { has_variable = slot_char }
		value = var:slot_char.POD_health_track_length
	}
}
pod_combat_slot_total_damage_sv = {
	# Total wound boxes filled. Dead-char branch returns max_hp so the
	# GUI shows a fully-filled bar instead of zero (PoD's wound SVs
	# return 0 for dead chars via their own is_alive gate).
	if = {
		limit = { has_variable = slot_char var:slot_char = { is_alive = yes } }
		value = var:slot_char.POD_total_damage_boxes
		min = 0
	}
	if = {
		limit = { has_variable = slot_char var:slot_char = { is_alive = no } }
		value = pod_combat_slot_max_hp_value
	}
}
pod_combat_slot_agg_damage_sv = {
	# Aggravated boxes. Same alive/dead branching as total_damage_sv.
	if = {
		limit = { has_variable = slot_char var:slot_char = { is_alive = yes } }
		value = var:slot_char.POD_aggravated_damage_boxes
	}
	if = {
		limit = { has_variable = slot_char var:slot_char = { is_alive = no } }
		value = pod_combat_slot_max_hp_value
	}
	max = pod_combat_slot_total_damage_sv
}
pod_combat_slot_morale_pct = {
	# has_variable guard for consistency with the other slot SVs. Safe in
	# practice (slot_morale is initialised at slot creation), but the guard

		value = var:slot_morale

# Signed net hit-chance modifier (buffs − debuffs). Sums the four
# hit-tier magnitude SVs above, adding buffs and subtracting debuffs,
# so a slot with a tier-2 round buff (+20) AND a tier-1 combat debuff
# (-10) reports +10. Returns 0 when no hit-tier mods are active. Used
# by NPC slot cards (slots 1-9) to display "this combatant is currently
# X% above/below their baseline hit chance" instead of the largely
# meaningless slot_hit_chance absolute value (which rotates between
# leader-baseline, last-attack-target, and last-attacker-retaliation
# depending on what happened most recently — see PODCombatSlotHitDisplay
# in 00_POD_combat_custom_loc.txt for the display dispatch). The PC slot
# (slot 0) ignores this SV and keeps the absolute readout because the
# load_workspace writeback explicitly refreshes slot 0's slot_hit_chance
# against the PC's currently-selected target every turn.
pod_combat_slot_hit_net_mod_value = {
	value = 0
	add      = pod_combat_slot_hit_buff_round_mag_sv
	add      = pod_combat_slot_hit_buff_combat_mag_sv
	subtract = pod_combat_slot_hit_debuff_round_mag_sv
	subtract = pod_combat_slot_hit_debuff_combat_mag_sv
}

# ────────────────────────────────────────────────────────────────────
# Move-story tier magnitudes for combat-log riders.
#
# The eight slot mag SVs above read slot-scoped tier vars
# (slot_X_buff/debuff_round/combat_tier) and feed the per-slot stat
# recalc. The four SVs below run against a MOVE STORY scope and
# collapse round-or-combat into a single per-stat magnitude — used by
# the combat log's pod_combat_log_buff_X_buff / _X_debuff loc strings
# to surface "+30% Soak" instead of just "+Soak".
#
# Move stories carry either soak_buff_round_tier OR soak_buff_combat_tier
# for a given stat (never both — a move's design is round-duration or
# combat-duration, not both at once). The SV chains six `if` branches ,
# over both duration vars each setting `value = @<size>_mag` based on 
# the matched tier. Plain`if` (not `else_if`) matches the existing slot-mag-SV 
# pattern; mutual exclusivity is guaranteed by the per-move tier-var pair invariant.
#
# Using @combat_tier_*_mag rather than tier × 10 keeps the magnitude
# linked to the same constants the slot SVs use, so a balance pass that
# changed medium from 20 to 25 would propagate to the log row text
# without needing this block to be re-edited.

pod_combat_move_soak_buff_mag_value = {
	if = { limit = { has_variable = soak_buff_round_tier  var:soak_buff_round_tier  = 1 } value = combat_tier_small_mag  }
	if = { limit = { has_variable = soak_buff_round_tier  var:soak_buff_round_tier  = 2 } value = combat_tier_medium_mag }
	if = { limit = { has_variable = soak_buff_round_tier  var:soak_buff_round_tier  = 3 } value = combat_tier_large_mag  }
	if = { limit = { has_variable = soak_buff_combat_tier var:soak_buff_combat_tier = 1 } value = combat_tier_small_mag  }
	if = { limit = { has_variable = soak_buff_combat_tier var:soak_buff_combat_tier = 2 } value = combat_tier_medium_mag }
	if = { limit = { has_variable = soak_buff_combat_tier var:soak_buff_combat_tier = 3 } value = combat_tier_large_mag  }

# ────────────────────────────────────────────────────────────────────
# Combat log: display gate for the elemental damage-source rider
# (the fire / silver / sun / holy / etc. icon that follows the wound
# box on damage rows).
#
# Suppresses the icon on attack-failure rows (guard_blocked / missed /
# dodged / stealthed). The icon describes the elemental flavor of
# damage that WAS delivered; on a miss / block / dodge no damage of
# any element landed, so showing "fire" next to a Blocked row adds
# noise without information. Damage that lands (hp_damage and its
# retaliation variants), morale damage, heals, and utility moves
# keep the icon if their move declared one.
#
# Caller scope must be a combat-log entry story (the var:attack_result
# read below resolves against entry.var). Returns 1 to show, 0 to
# hide; the consumer gates with GreaterThan_CFixedPoint > 0.
#
# Reads entry.var:attack_result (set by push_log from the host's
# pod_combat_last_attack_result) since the icon suppress logic only
# applies to the attack axis — heal / morale / utility rows don't
# have an attack_result var at all, so the trigger returns false for
# them and the icon stays visible.

pod_combat_log_show_damage_source_value = {
	# Single trigger call instead of an inline enum that has to stay
	# parallel to the partition trigger of the same set. Adding a
	# new no-damage attack_result flag is now a one-line edit in
	# pod_combat_result_indicates_no_damage_dealt_trigger; this SV
	# inherits the change automatically.
	value = 1
	if = {
		limit = {
			pod_combat_result_indicates_no_damage_dealt_trigger = { RESULT_VAR = attack_result }
		}
		value = 0
	}
}


# ─── Combat-log result-column visibility probes ────────────────────
# Each "rider" in the result column (heal / morale damage / morale
# heal / retal / buff) has its own customizable_loc dispatcher AND
# a separator widget that renders " · " before it. The SVs below
# return 1 if a given segment would actually render visible content,
# 0 otherwise — used uniformly by both the rider widget's `visible`
# binding AND the preceding separator's `visible` binding so the
# two stay in lockstep.
#
# Without these, the rider WIDGET's visibility (which gated only on
# `amount.IsSet`) could stay true while the rider's DISPATCHER
# returned blank (the heal / morale-damage / morale-heal dispatchers
# suppress themselves when outcome equals their primary, since the
# main result line already shows the same amount). That orphaned
# the surrounding separators around an empty widget — symptoms like
# "-5 Morale · · → Guard broken" with two dots.

# Main result has visible text. Returns 0 only when outcome=utility
# (loc literally returns ""); every other outcome renders something.
pod_combat_log_main_has_content_value = {
	value = 0
	if = {
		limit = {
			has_variable = outcome
			NOT = { var:outcome = flag:utility }
		}
		value = 1
	}
}

# Heal rider would render content. Mirrors the dispatcher gate in
# PODCombatLogHealRider: heal_amount > 0 AND outcome != flag:heal
# (the heal outcome's primary already shows the same amount, so
# the rider suppresses itself).
# NAND, not NOT: a multi-child NOT is NOR in Jomini, and since push_log
# always stamps an outcome, NOR(has_variable, compare) was permanently
# false - it silently killed this rider and the two below for any move
# row. NAND = "not (set and equal)", the intended suppression.
pod_combat_log_heal_rider_visible_value = {
	value = 0
	if = {
		limit = {
			has_variable = heal_amount
			var:heal_amount > 0
			NAND = {
				has_variable = outcome
				var:outcome = flag:heal
			}
		}
		value = 1
	}
}

# Morale-damage rider would render content. Same suppression
# pattern when outcome=morale_damage.
pod_combat_log_morale_damage_rider_visible_value = {
	value = 0
	if = {

			has_variable = morale_damage_amount
			var:morale_damage_amount > 0

				has_variable = outcome
				var:outcome = flag:morale_damage

	}
}

# Morale-heal rider would render content. Suppressed when

pod_combat_log_morale_heal_rider_visible_value = {
