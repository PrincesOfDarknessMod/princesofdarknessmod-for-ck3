camarilla_status = {
	if = {
		limit = { 
			faith:camahumanity = {
				is_target_in_variable_list = { 
					name = inner_circle
					target = prev
				} 
			}
		}
		value = 8
	}
	else_if = {
		limit = { 
			faith:camahumanity = {
				is_target_in_variable_list = { 
					name = justicars
					target = prev
				} 
			}
		}
		value = 7
	}
	else_if = {
		limit = { POD_is_archon_trigger = yes }
		value = 6
	}
	else_if = {
		limit = { is_ruler = yes }
		value = 5
	}
	else_if = {
		limit = { POD_is_primogen_trigger = yes }
		value = 4
	}
	else_if = {
		limit = { is_councillor = yes }
		value = 3
	}
	else_if = {
		limit = { has_any_court_position = yes }
		value = 2
	}
	else = {
		value = 1
	}
	
	if = {
		limit = {
			OR = {
				NOT = { has_trait = vampire }
				NOT = { faith = faith:camahumanity }
				has_trait = anathema
			}
		}
		multiply = 0
	}
}

justicar_bribe_value = {
	value = 50
	if = {
		limit = {
			scope:secondary_actor.ai_greed > 0
		}
		add = {
			value = scope:secondary_actor.ai_greed
			divide = 4
		}
	}
	
	if = {
		limit = {
			scope:secondary_actor = { is_landed = yes }
			scope:secondary_actor.primary_title.tier < tier_duchy
		}
		multiply = 2
	}
	else_if = {
		limit = {
			scope:secondary_actor.primary_title.tier >= tier_duchy
		}
		multiply = 3
	}
	
	if = {
		limit = {
			scope:secondary_actor = {
				liege = scope:actor
				is_theocratic_lessee = yes
			}
			scope:actor.primary_title.tier < tier_duchy
		}
		multiply = 2
	}
	else_if = {
		limit = {
			scope:secondary_actor = {
				liege = scope:actor
				is_theocratic_lessee = yes
			}
			scope:actor.primary_title.tier >= tier_duchy
		}
		multiply = 3
	}
	
	if = {
		limit = {
			scope:actor = {
				exists = dynasty
				dynasty = {
					has_dynasty_perk = guile_legacy_2
				}
			}
		}
		multiply = 0.5
	}
	
	if = {
		limit = {
			scope:secondary_actor = {
				OR = {
					AND = {
						is_landed = yes
						government_has_flag = government_is_tribal
					}
					AND = {
						is_landed = no
						exists = liege
						liege = {
							government_has_flag = government_is_tribal
						}
					}
					AND = {
						is_landed = no
						exists = host
						host = {
							government_has_flag = government_is_tribal
						}
					}
				}
			}
		}
		multiply = 0.25
		min = 5
	}

	# Discount from BP2 imaginary friend story cycle ending
	if = {
		limit = {
			scope:actor = { has_character_modifier = imaginary_friend_mad_modifier }
		}
		multiply = 0.5
		min = 5
	}
}
