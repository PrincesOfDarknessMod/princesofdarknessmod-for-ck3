camarilla_status = {
	if = {
		limit = { 
			faith:camahumanity = {
				is_target_in_variable_list = { 
					name = inner_circle
					target = prev
				} 
			}
		}
		value = 8
	}
	else_if = {
		limit = { 
			faith:camahumanity = {
				is_target_in_variable_list = { 
					name = justicars
					target = prev
				} 
			}
		}
		value = 7
	}
	else_if = {
		limit = { POD_is_archon_trigger = yes }
		value = 6
	}
	else_if = {
		limit = { is_ruler = yes }
		value = 5
	}
	else_if = {
		limit = { POD_is_primogen_trigger = yes }
		value = 4
	}
	else_if = {
		limit = { is_councillor = yes }
		value = 3
	}
	else_if = {
		limit = { has_any_court_position = yes }
		value = 2
	}
	else = {
		value = 1
	}
	
	if = {
		limit = {
			OR = {
				NOT = { has_trait = vampire }
				NOT = { faith = faith:camahumanity }
				has_trait = anathema
			}
		}
		multiply = 0
	}
}
