justicar_bribe_value = {
	value = 50
	if = {
		limit = {
			scope:secondary_actor.ai_greed > 0
		}
		add = {
			value = scope:secondary_actor.ai_greed
			divide = 4
		}
	}
	
	if = {
		limit = {
			scope:secondary_actor = { is_landed = yes }
			scope:secondary_actor.primary_title.tier < tier_duchy
		}
		multiply = 2
	}
	else_if = {
		limit = {
			scope:secondary_actor.primary_title.tier >= tier_duchy
		}
		multiply = 3
	}
	
	if = {
		limit = {
			scope:secondary_actor = {
				liege = scope:actor
				is_theocratic_lessee = yes
			}
			scope:actor.primary_title.tier < tier_duchy
		}
		multiply = 2
	}
	else_if = {
		limit = {
			scope:secondary_actor = {
				liege = scope:actor
				is_theocratic_lessee = yes
			}
			scope:actor.primary_title.tier >= tier_duchy
		}
		multiply = 3
	}
	
	if = {
		limit = {
			scope:actor = {
				exists = dynasty
				dynasty = {
					has_dynasty_perk = guile_legacy_2
				}
			}
		}
		multiply = 0.5
	}
	
	if = {
		limit = {
			scope:secondary_actor = {
				OR = {
					AND = {
						is_landed = yes
						government_has_flag = government_is_tribal
					}
					AND = {
						is_landed = no
						exists = liege
						liege = {
							government_has_flag = government_is_tribal
						}
					}
					AND = {
						is_landed = no
						exists = host
						host = {
							government_has_flag = government_is_tribal
						}
					}
				}
			}
		}
		multiply = 0.25
		min = 5
	}

	# Discount from BP2 imaginary friend story cycle ending
	if = {
		limit = {
			scope:actor = { has_character_modifier = imaginary_friend_mad_modifier }
		}
		multiply = 0.5
		min = 5
	}
}

sabbat_status = {
	if = {
		limit = { POD_is_sabbat_regent_trigger = yes }
		value = 8
	}
	else_if = {
		limit = { POD_is_cardinal_trigger = yes }
		value = 7
	}
	else_if = {
		limit = { is_ruler = yes } #Archbishop
		value = 6
	}
	else_if = {
		limit = { is_councillor = yes } #Priscus
		value = 5
	}
	else_if = {
		limit = { has_any_court_position = yes } #Bishop
		value = 4
	}
	else_if = {
		limit = { is_knight = yes } # Templar/Paladin(Accolade)
		value = 3
	}
	else_if = {
		limit = { POD_has_coterie_trigger = yes } # Ductus
		value = 2
	}
	else_if = {
		limit = { has_trait = zealous } # Pack Priest
		value = 1
	}
	else = { # True sabbat
		value = 0
	}
	
	if = {
		limit = {
			OR = {
				NOT = { has_trait = vampire }
				NOT = { religion = religion:sabbat_religion }
				has_trait = excommunicated
			}
		}
		multiply = 0
		add = -1
	}
}
