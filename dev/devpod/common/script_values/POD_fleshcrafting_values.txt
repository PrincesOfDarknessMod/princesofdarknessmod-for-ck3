POD_fleshcrafting_success_chance = {
	value = POD_fleshcrafting_success_chance_decimal
	multiply = 100
}

POD_fleshcrafting_success_chance_decimal = {
	if = {
		limit = {
			has_perk = perfect_essence_perk
			has_variable = fleshcraft_target
			this = var:fleshcraft_target
		}
		value = 1 # guaranteed success
	}
	else_if = {
		limit = {
			# only do this once, for performance
			POD_fleshcrafting_save_current_complexity = yes
		}
		value = POD_fleshcrafting_success_weight
		divide = {
			value = POD_fleshcrafting_success_weight
			add = POD_fleshcrafting_failure_weight
		}
	}
}


POD_fleshcrafting_current_complexity = {
	if = {
		limit = {
			POD_fleshcrafting_save_current_complexity = yes
		}
		value = scope:complexity
	}
}
