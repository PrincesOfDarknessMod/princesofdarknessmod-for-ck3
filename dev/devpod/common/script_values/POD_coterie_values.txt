
coterie_max_size = 5
coterie_rivalry_opinion = -25

coterie_size = {
	value = 0
	if = {
		limit = { exists = var:coterie }
		var:coterie = {
			every_in_list = {
				variable = members
				add = 1
			}
		}
	}
}

coterie_available_slots = {
	value = coterie_max_size
	subtract = coterie_size

	min = 0
}

coterie_harem_target_size = {
	value = coterie_max_size
	subtract = 1
}

coterie_diplomacy_gui = {
	value = 0
	if = {
		limit = { exists = var:coterie }
		var:coterie = {
			ordered_in_list = {
				variable = members
				order_by = diplomacy
				position = 0
				add = diplomacy
			}
		}
	}
	else = {
		add = diplomacy
	}
}

	value = 0
	if = {
		limit = { exists = var:coterie }
		var:coterie = {
			every_in_list = {
				variable = members
				limit = {
					POD_already_has_seek_knowledge_trait_trigger = {
						VAR = root.var:current_journey.var:journey_type
					}
				}
				add = 1
			}
		}
	}
}
