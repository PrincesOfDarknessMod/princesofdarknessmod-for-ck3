wheel_of_ages_progress_change = {
	value = 0
	if = {
		limit = { any_empire = { holder ?= { is_akuma_trigger = yes } } }
		add = { value = wheel_of_ages_progress_plus_small desc = wheel_of_ages_progress_change_akuma_anda }
	}
	if = {
		limit = { 
			any_empire = { 
				OR = {
					holder ?= { is_vampire_trigger = yes }
					holder ?= { is_akuma_trigger = yes }
				}
				title_capital_county = { title_province = { geographical_region = world_india } }
			} 
		}
		add = { value = wheel_of_ages_progress_plus_small desc = wheel_of_ages_progress_change_akuma_or_vampire_india }
	}
	if = {
		limit = { title:e_anda ?= { holder ?= { is_akuma_trigger = yes } } }
		add = { value = wheel_of_ages_progress_plus_small desc = wheel_of_ages_progress_change_akuma_anda }
	}
	title:e_anda ?= {
		if = { limit = { holder ?= { realm_size >= 100 } } add = { value = wheel_of_ages_progress_plus_small desc = wheel_of_ages_progress_change_huge_anda } }
	}
	if = { limit = { any_independent_ruler = { is_landed = yes is_vampire_trigger = yes } } add = { value = wheel_of_ages_progress_plus_small desc = wheel_of_ages_progress_change_vampire_ruler } }
	title:e_anda ?= {
		if = { limit = { is_title_created = no } add = { value = wheel_of_ages_progress_minus_medium desc = wheel_of_ages_progress_change_no_anda } }
	}
	if = { limit = { any_independent_ruler = { is_landed = yes is_akuma_trigger = yes count = 0 } } add = { value = wheel_of_ages_progress_minus_medium desc = wheel_of_ages_progress_change_no_akuma_ruler } }
	if = { limit = { any_independent_ruler = { is_landed = yes is_vampire_trigger = yes count = 0 } } add = { value = wheel_of_ages_progress_minus_medium desc = wheel_of_ages_progress_change_no_vampire_ruler } }
	if = { limit = { NOT = { game_start_date = 1230.1.4 } any_independent_ruler = { is_landed = yes is_demon_trigger = yes count = 0 } } add = { value = wheel_of_ages_progress_minus_medium desc = wheel_of_ages_progress_change_no_demon_ruler } }
}
