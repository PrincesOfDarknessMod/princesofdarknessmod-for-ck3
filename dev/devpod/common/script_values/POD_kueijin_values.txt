POD_kueijin_dharma_xp_passive = {
	value = { value = 0.1 desc = BASE }
	if = { limit = { religion = religion:boneflowers_religion  }  
		add = { value = bound_wraith_num desc = DHARMA_PASSIVE_BOUND_WRAITHS }
	}
	if = { limit = { religion = religion:deviltigers_religion  }  
		if = { limit = { dread >= 50 }  
			add = { value = 1 desc = DHARMA_PASSIVE_DREAD_ABOVE_50 }
		}
	}
	if = { limit = { religion = religion:resplendentcranes_religion  }  
		if = { limit = { prestige_level >= 5 }  
			add = { value = 1 desc = DHARMA_PASSIVE_PRESTIGE_LEVEL_ABOVE_5 }
		}
		if = { limit = { tyranny = 0 }  
			add = { value = 1 desc = DHARMA_PASSIVE_NO_TYRANNY }
		}
		if = { limit = { is_independent_ruler = no }  
			add = { value = 1 desc = DHARMA_PASSIVE_IS_VASSAL }
		}
	}
	if = { limit = { religion = religion:thousandwhispers_religion  }  
		if = { limit = { relations_younger_than_10_years >= 1 }  
			add = { value = relations_younger_than_10_years desc = DHARMA_PASSIVE_HAVE_NEW_RELATION }
		}
	}
	if = { limit = { religion = religion:thrashingdragons_religion  }  
		add = { value = num_of_relation_spirit_companion desc = DHARMA_PASSIVE_BOUND_SPIRITS }
	}
	if = { limit = { faith = faith:flameoftherisingphoenix }  
		if = { limit = { tyranny = 0 }  
			add = { value = 1 desc = DHARMA_PASSIVE_NO_TYRANNY }
		}
	}
	if = { limit = { faith = faith:faceofthegods }  
		if = { limit = { piety_level >= 5 }  
			add = { value = 1 desc = DHARMA_PASSIVE_PIETY_LEVEL_ABOVE_5 }
		}
		if = { limit = { any_courtier = { count >= 10 opinion = { target = PREV value >= 50 } } }  
			add = { value = 1 desc = DHARMA_PASSIVE_HAS_10_POSITIVE_COURTIERS }
		}
	}
	if = { limit = { faith = faith:inwardway }  
		if = { limit = { days_of_continuous_peace >= 3652 }  
			add = { value = 1 desc = DHARMA_PASSIVE_10_YEARS_WITHOUT_WAR }
		}
		add = { value = num_of_relation_spirit_companion desc = DHARMA_PASSIVE_BOUND_SPIRITS }
	}
	if = { limit = { faith = faith:wayofthetenthousandscreams  }  
		if = { limit = { num_of_relation_nemesis > 0 }  
			add = { value = 1 desc = DHARMA_PASSIVE_HAS_A_NEMESIS }
		}
		if = { limit = { has_trait = cultist } add = { value = 1 desc = DHARMA_PASSIVE_IS_A_AKUMA } }
	}

	if = { limit = { has_character_modifier = two_become_one_modifier } 
		add = {
			value = 1 desc = DHARMA_PASSIVE_TWO_BECOME_ONE_MODIFIER
		}
	}
	if = { limit = { has_character_modifier = harmony_with_the_all_modifier } 
		add = {
			value = 1 desc = DHARMA_PASSIVE_HARMONY_WITH_THE_ALL_MODIFIER
		}
	}
	if = { limit = { has_trait = dharma9 } divide = { value = 2 desc = DHARMA_PASSIVE_DHARMA9 } }
}
