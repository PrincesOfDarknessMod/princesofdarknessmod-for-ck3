

POD_perk_alliances_total = {
	value = 0

	if = {
		limit = { has_perk = defensive_negotiations_perk }
		add = 1
	}

	if = {
		limit = { POD_is_supehunter_trigger = yes }
		add = 1
	}

	if = {
		limit = { domicile ?= { has_domicile_parameter = negotiate_perk_alliance } }
		add = 1
	}
}

POD_perk_alliances_used = {
	value = 0

	add = {
		value = {
			every_ally = {
				limit = { has_opinion_modifier = { modifier = perk_negotiated_alliance_opinion target = scope:actor } }
				add = 1
			}
		}
	}
}

POD_perk_alliances_available = {
	value = POD_perk_alliances_total
	subtract = POD_perk_alliances_used

	min = 0
}
