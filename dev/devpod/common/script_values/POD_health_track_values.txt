		}

# this multiplier is for converting vanilla "wounds" to PoD health track boxes
# a "wound" means a one-step increase in the vanilla wound trait
# e.g. going from unhurt to wounded_1, or from wounded_1 to wounded_2, etc
#
# requires scope:death_reason to be saved on the TopScope
# (meaning a death reason flag, e.g. flag:death_greek_fire)
POD_wound_multiplier_for_vanilla_death_reason = {
	value = 1
	
	if = {
		limit = { scope:death_reason = flag:death_sun }
		multiply = POD_base_sun_damage
	}
	else_if = {
		limit = { scope:death_reason = flag:death_lightning }
		multiply = POD_base_lightning_damage
	}
	else_if = {
		limit = { scope:death_reason = flag:death_true_faith }
		multiply = POD_base_holy_damage
	}
	else_if = {
		limit = { scope:death_reason = flag:death_cold }
		multiply = POD_base_cold_damage
	}
	else_if = {
		limit = { scope:death_reason = flag:death_balefire }
		multiply = POD_base_balefire_damage
	}
	else_if = {
		limit = { POD_is_death_by_fire = { DEATH_REASON = scope:death_reason } }
		multiply = POD_base_fire_damage
	}
	else = {
		multiply = POD_base_wound_damage
	}
	
	ceiling = yes
}

# requires scope:death_reason to be saved on the TopScope
# (meaning a death reason flag, e.g. flag:death_greek_fire)
POD_wound_multiplier_from_splat_weaknesses = {
	value = 1
	
	if = {
		limit = {
			scope:death_reason = flag:death_true_faith
			OR = {
				is_baali_trigger = yes
				POD_is_demon_trigger = yes
				POD_is_wraith_trigger = yes
			}
			POD_suppress_clan_curse_trigger = no
		}
		multiply = POD_default_damage_multiplier_for_splat_weakness
	}
	
	# Sun weakness
	if = {
		limit = {
			scope:death_reason = flag:death_sun
			OR = {
				has_trait = setite
				is_bushi_trigger = yes
			}
			POD_suppress_clan_curse_trigger = no
		}
		multiply = POD_default_damage_multiplier_for_splat_weakness
	}
	
	# Fire weakness
	if = {
		limit = {
			POD_is_death_by_fire = { DEATH_REASON = scope:death_reason }
			#OR = {
				is_bushi_trigger = yes
			#}
			POD_suppress_clan_curse_trigger = no
		}
		multiply = POD_default_damage_multiplier_for_splat_weakness
	}
	
	ceiling = yes
}

# Cap (in soak %) on what any single STACKING source can contribute.
@soak_single_source_cap = 50

# Soak sources compose multiplicatively. Each source multiplies the running 
# chance-to-be-hit, so stacking diminishes and can never reach immunity. The 
# product is converted to an effective soak % at the end, keeping the 0-100 contract 
# every reader depends on (sidebar readout, maim rolls, the combat slot snapshot, the 
# Mokole objective). The combat tier bridge stays additive on that final percentage.
POD_soak_chance = {
	# Running chance-to-be-hit (a fraction). Starts at 1.0 (always hit), each
	# soak source multiplies it down. Converted to effective soak% at the end.
	value = 1

	# Armor artifact: rarity is one mutually-exclusive factor.
	if = {
		limit = {
			any_equipped_character_artifact = {
				artifact_slot_type = armor
				save_temporary_scope_as = armor
			}
		}
		scope:armor = {
			switch = {
				trigger = rarity
				common		= { multiply = @soak_mult_10 }
				masterwork	= { multiply = @soak_mult_20 }
				famed		= { multiply = @soak_mult_30 }
				illustrious	= { multiply = @soak_mult_40 }
			}
		}
	}

	if = { limit = { has_perk = defy_bane_perk }				multiply = @soak_mult_20 } # Vampires (Fortitude)
	if = { limit = { has_perk = armoured_flesh_perk }			multiply = @soak_mult_20 } # Vampires (Fortitude)
	if = { limit = { has_perk = toughness_perk }				multiply = @soak_mult_20 } # Vampires (Fortitude)
	if = { limit = { has_perk = flesh_of_marble_perk }			multiply = @soak_mult_20 } # Vampires (Fortitude); base gave 10 but the perk card displays 20

	if = { limit = { has_character_flag = granted_toughness }	multiply = @soak_mult_20 } # Vampires (Fortitude), can be used on others

		limit = { demon_arts_iron_mountain_modifier_stacks >= 1 }

	if = { limit = { has_perk = resist_pain_perk }						multiply = @soak_mult_05 } # Mokole (Setting Sun Gifts)
	if = { limit = { has_perk = armor_of_the_tortoise_perk }			multiply = @soak_mult_10 } # Mokole (Setting Sun Gifts)
	if = { limit = { has_character_modifier = agni_totem_modifier }		multiply = @soak_mult_05 } # Mokole (Totem)
	if = { limit = { domicile ?= { has_domicile_building_or_higher = mokole_edifice_mastery_05 } } multiply = @soak_mult_05 } # Mokole (Domicile)
	if = { limit = { faith = { has_doctrine = tenet_outlast_world } }	multiply = @soak_mult_10 } # Mokole (Faith Tenet)
	if = { limit = { has_character_modifier = archid_glittering_scales_modifier } multiply = @soak_mult_10 } # Mokole (Oogleforp Special Building)
	if = { limit = { has_character_modifier = archid_gale_cloak_modifier } multiply = @soak_mult_10 } # Mokole (Makara Objective Artifact)
	if = { limit = { has_character_modifier = archid_undying_flesh_modifier } multiply = @soak_mult_10 } # Mokole (Eternal Breath Objective)
	if = { limit = { has_character_modifier = greendragon_totem_modifier } multiply = @soak_mult_10 } # Fera (Wyrm Totem)
	if = { limit = { has_perk = howl_of_healing_perk }					multiply = @soak_mult_05 } # Werewolf (White Howler Gifts)

	# Investiture armor: stacks * 5% soak (capped at @soak_single_source_cap), one factor.
	# The Armor perk is repeatable, so its own pick cap is kept in step with this one
	# (see investiture_armor_perk in 00_investiture_01_demonic_investment_tree_perks.txt).
	if = {
		limit = { POD_investiture_armor_perk_modifier_stacks >= 1 }
		multiply = { value = 1 subtract = { value = { value = POD_investiture_armor_perk_modifier_stacks multiply = 5 max = @soak_single_source_cap } multiply = 0.01 } }
	}
	if = { limit = { has_perk = investiture_invulnerability_perk }		multiply = @soak_mult_25 } # Investiture
	if = { limit = { has_perk = malformed_flesh_perk }					multiply = @soak_mult_05 } # Maeljin Gifts

	# Bestial-form hide (Sorcery Shapeshift): scale * 5% soak (capped at @soak_single_source_cap), one factor.
	if = {
		limit = { has_character_modifier = POD_shapeshift_bestial_form_hide_modifier }
		multiply = { value = 1 subtract = { value = { value = POD_shapeshift_bestial_form_hide_scale multiply = 5 max = @soak_single_source_cap } multiply = 0.01 } }
	}

	if = { limit = { has_perk = common_arts_gimmix_perk } multiply = @soak_mult_15 } # Fae (Legerdemain)
	if = { limit = { has_perk = noble_arts_deja_vu_perk } multiply = @soak_mult_25 } # Fae (Chronos)
	if = { limit = { has_character_modifier = cantrip_basic_stone_skin_modifier } multiply = @soak_mult_05 } # Fae (Petros)
	if = { limit = { has_character_modifier = cantrip_confounding_coils_modifier } multiply = @soak_mult_10 } # Fae (Dragon's ire)
	if = { limit = { has_character_modifier = cantrip_dragonscales_modifier } multiply = @soak_mult_20 } # Fae (Dragon's ire)
	if = { limit = { has_character_modifier = cantrip_runic_imprint_survival_modifier } multiply = @soak_mult_10 } # Fae (Naming)
	if = { limit = { has_character_modifier = cantrip_uncontrolled_stone_skin_modifier } multiply = @soak_mult_10 } # Fae (Petros)
	if = { limit = { has_character_modifier = cantrip_unstable_stone_skin_modifier } multiply = @soak_mult_15 } # Fae (Petros)
	if = { limit = { has_character_modifier = cantrip_semi_stable_stone_skin_modifier } multiply = @soak_mult_20 } # Fae (Petros)
	if = { limit = { has_character_modifier = cantrip_stable_stone_skin_modifier } multiply = @soak_mult_25 } # Fae (Petros)
	if = { limit = { has_character_modifier = cantrip_perfect_stone_skin_modifier } multiply = @soak_mult_30 } # Fae (Petros)

	# Artifacts
	if = { limit = { any_character_artifact = { has_variable = leaf_armor is_equipped = yes prev = { can_benefit_from_artifact = prev } } } multiply = @soak_mult_10 } # Leaf Armor
	if = {
		limit = { any_character_artifact = { has_variable = artifact_soak_var is_equipped = yes prev = { can_benefit_from_artifact = prev } } }
		every_equipped_character_artifact = {
			limit = { has_variable = artifact_soak_var prev = { can_benefit_from_artifact = prev } }
			multiply = { value = 1 subtract = { value = var:artifact_soak_var  multiply = 0.01 } }
		}
	}
	if = { limit = { any_equipped_character_artifact = { has_variable = dipelta prev = { can_benefit_from_artifact = prev } } } multiply = @soak_mult_10 } # Dipelta Dinosaur
	if = { limit = { any_equipped_character_artifact = { has_variable = lunar_dragon_tide prev = { can_benefit_from_artifact = prev } } } multiply = @soak_mult_10 } # Lunar-Dragon Tide Katana
	if = { limit = { any_equipped_character_artifact = { has_variable = dragons_ichor prev = { can_benefit_from_artifact = prev } } } multiply = @soak_mult_15 } # Dragon's Ichor Wyrm Fetish

	# Porcupine Spirit Companion: rank value summed across companions, one aggregate factor,
	# capped like the other stacking sources.
	if = {
		limit = {
			any_relation = {
				type = spirit_companion
				var:spirit_type = flag:spirit_porcupine
			}
		}
		multiply = {
			value = 1
			subtract = {
				value = {
					every_relation = {
						type = spirit_companion
						limit = { is_alive = yes var:spirit_type = flag:spirit_porcupine }
						if = { limit = { POD_spirit_rank = 5 } add = 20 }
						else_if = { limit = { POD_spirit_rank = 4 } add = 16 }
						else_if = { limit = { POD_spirit_rank = 3 } add = 12 }
						else_if = { limit = { POD_spirit_rank = 2 } add = 8 }
						else = { add = 4 }
					}
				}
				max = @soak_single_source_cap
				multiply = 0.01
			}
		}
	}
	# Convert running chance-to-be-hit into effective soak percent.
	multiply = -100
	add = 100

	if = { limit = { has_character_modifier = POD_quest_of_deed_modifier } add = -30 } # Fae (Banality Quest)
	if = { limit = { has_perk = charm_comet_deflecting_defense_perk } add = -30 } # Exalted Fire Charms
	
	if = { # Exalted Water Charms
		limit = { scope:wound_attacker ?= { has_perk = charm_flow_past_defenses_perk } }
		add = -30
	} 
	if = { # Exalted Water Charms
		limit = { scope:wound_attacker ?= { has_character_modifier = POD_spring_follows_winter_modifier } }
		add = -100
	} 

	if = { limit = { has_character_modifier = cantrip_deja_vu_modifier } add = 100 } # Fae (Chronos)
	if = { limit = { has_character_modifier = POD_bottomless_depths_defense_modifier } add = 100 } # Exalted Water Charms

	# Clamp HERE, not only at the end: everything below this line is the combat
	# bridge, and an over-100 or negative intermediate would feed the buff factor
	# below and come back out wrong (the flip to chance-to-be-hit assumes a 0-100
	# input).
	min = 0
	max = 100

	# Combat-minigame tier soak BUFFS.
	# The tactical combat effects set pod_combat_tier_soak_buff_{round,combat}
	# = 1/2/3 on the character when a move grants a soak buff. Round-tier vars
	# are wiped at round end, combat-tier vars at combat end.
	# Buffs are multiplicative like every other soak source, so a tier is a
	# constant proportional cut to incoming damage rather than a flat point
	# gain worth ever more the soakier the target already is. Expressed by
	# flipping to chance-to-be-hit, scaling, and flipping back, because the flat
	# Fae terms above have already landed by this point, so folding these into
	# the product further up would not give the same answer. The factor cannot go
	# negative: the tuning knob clamps every tier magnitude to 0-100
	# (pod_combat_tuning_guis.txt).
	# The matching DEBUFFS stay flat percentage points, below.
	#
	# is_alive gates every branch: the combat-end clear has to skip the dead
	# (var mutation on a dead participant errors), so a combatant killed
	# mid-duel keeps these vars and the reader is what neutralises them. Slot
	# init re-clears them, so a resurrection cannot carry a stale tier into a
	# later fight either. The two flips either side are a no-op pair when no
	# branch fires, so gating the branches alone is enough.
	multiply = -1
	add = 100
	if = { limit = { is_alive = yes has_variable = pod_combat_tier_soak_buff_round   var:pod_combat_tier_soak_buff_round   = 1 } multiply = { value = 1 subtract = { value = pod_combat_tier_small_mag  multiply = 0.01 } } }
	if = { limit = { is_alive = yes has_variable = pod_combat_tier_soak_buff_round   var:pod_combat_tier_soak_buff_round   = 2 } multiply = { value = 1 subtract = { value = pod_combat_tier_medium_mag multiply = 0.01 } } }
	if = { limit = { is_alive = yes has_variable = pod_combat_tier_soak_buff_round   var:pod_combat_tier_soak_buff_round   = 3 } multiply = { value = 1 subtract = { value = pod_combat_tier_large_mag  multiply = 0.01 } } }
	if = { limit = { is_alive = yes has_variable = pod_combat_tier_soak_buff_combat  var:pod_combat_tier_soak_buff_combat  = 1 } multiply = { value = 1 subtract = { value = pod_combat_tier_small_mag  multiply = 0.01 } } }
	if = { limit = { is_alive = yes has_variable = pod_combat_tier_soak_buff_combat  var:pod_combat_tier_soak_buff_combat  = 2 } multiply = { value = 1 subtract = { value = pod_combat_tier_medium_mag multiply = 0.01 } } }
	if = { limit = { is_alive = yes has_variable = pod_combat_tier_soak_buff_combat  var:pod_combat_tier_soak_buff_combat  = 3 } multiply = { value = 1 subtract = { value = pod_combat_tier_large_mag  multiply = 0.01 } } }
	multiply = -1
	add = 100

	# Combat-minigame tier soak DEBUFFS.
	# Deliberately NOT multiplicative like the buff half above: a factor that
	# scales the chance to be hit can never meaningfully strip a stacked tank
	# (90% soak debuffed by a 0.3 factor is still 87%), whereas flat points
	# make soak-shredding moves the real answer to a soak build.
	if = { limit = { is_alive = yes has_variable = pod_combat_tier_soak_debuff_round   var:pod_combat_tier_soak_debuff_round   = 1 } subtract = pod_combat_tier_small_mag }
	if = { limit = { is_alive = yes has_variable = pod_combat_tier_soak_debuff_round   var:pod_combat_tier_soak_debuff_round   = 2 } subtract = pod_combat_tier_medium_mag }
	if = { limit = { is_alive = yes has_variable = pod_combat_tier_soak_debuff_round   var:pod_combat_tier_soak_debuff_round   = 3 } subtract = pod_combat_tier_large_mag }
	if = { limit = { is_alive = yes has_variable = pod_combat_tier_soak_debuff_combat  var:pod_combat_tier_soak_debuff_combat  = 1 } subtract = pod_combat_tier_small_mag }
	if = { limit = { is_alive = yes has_variable = pod_combat_tier_soak_debuff_combat  var:pod_combat_tier_soak_debuff_combat  = 2 } subtract = pod_combat_tier_medium_mag }
	if = { limit = { is_alive = yes has_variable = pod_combat_tier_soak_debuff_combat  var:pod_combat_tier_soak_debuff_combat  = 3 } subtract = pod_combat_tier_large_mag }

	min = 0
	max = 100
}
