POD_health_track_base_length = {
	if = {
		limit = { POD_is_werewolf_trigger = yes }
		min = 8
	}
	else_if = {
		limit = {
			OR = {
				POD_splat_is_fera_trigger = yes
				POD_is_demon_trigger = yes
				POD_is_fae_trigger = yes
			}
		}
		min = 7
	}
	else_if = {
		limit = {
			OR = {
				POD_is_vampire_trigger = yes
				is_kueijin_trigger = yes
				has_trait = risen
			}
		}
		min = 6
	}
	else_if = {
		limit = {
			OR = {
				POD_is_ghoul_trigger = yes
				POD_is_mummy_trigger = yes
				POD_is_wraith_trigger = yes
				has_trait = spirit
			}
		}
		min = 5
	}
	else = {
		min = 4
	}
}

POD_health_track_length = {
	value = POD_health_track_base_length
	
	if = {
		limit = { has_perk = shintai_the_shintai_of_bone_perk }
		add = 1
	}
	
	max = POD_max_health_track_size
	#min = 1
}

# this multiplier is for converting vanilla "wounds" to PoD health track boxes
# a "wound" means a one-step increase in the vanilla wound trait
# e.g. going from unhurt to wounded_1, or from wounded_1 to wounded_2, etc
#
# requires scope:death_reason to be saved on the TopScope
# (meaning a death reason flag, e.g. flag:death_greek_fire)
POD_wound_multiplier_for_vanilla_death_reason = {
	value = 1
	
	if = {
		limit = { scope:death_reason = flag:death_sun }
		multiply = POD_base_sun_damage
	}
	else_if = {
		limit = { scope:death_reason = flag:death_lightning }
		multiply = POD_base_lightning_damage
	}
	else_if = {
		limit = { POD_is_death_by_fire = { DEATH_REASON = scope:death_reason } }
		multiply = POD_base_fire_damage
	}
	else = {
		multiply = POD_wound_base_damage
	}
	
	ceiling = yes
}

# requires scope:death_reason to be saved on the TopScope
# (meaning a death reason flag, e.g. flag:death_greek_fire)
POD_wound_multiplier_from_splat_weaknesses = {
	value = 1
	
	# TODO: true faith weakness for baali/demons/infernalists/wraiths
	
	# Sun weakness
	if = {
		limit = {
			scope:death_reason = flag:death_sun
			OR = {
				has_trait = setite 
				is_bushi_trigger = yes
			}
			POD_suppress_clan_curse_trigger = no
		}
		multiply = POD_default_damage_multiplier_for_splat_weakness
	}
	
	# Fire weakness
	if = {
		limit = { 
			POD_is_death_by_fire = { DEATH_REASON = scope:death_reason }
			#OR = {
				is_bushi_trigger = yes
			#}
			POD_suppress_clan_curse_trigger = no 
		}
		multiply = POD_default_damage_multiplier_for_splat_weakness
	}
	
	ceiling = yes
}

	value = 0

		limit = {
			any_equipped_character_artifact = {
				artifact_slot_type = armor
				save_temporary_scope_as = armor
			}
		}
		scope:armor = {
			switch = {
				trigger = rarity
				common		= { add = 10 }
				masterwork	= { add = 20 }
				famed		= { add = 30 }
				illustrious	= { add = 40 }
			}
		}
	}
	if = { limit = { has_perk = defy_bane_perk } add = 20 } # Vampires (Fortitude)
	if = { limit = { has_perk = armoured_flesh_perk } add = 20 } # Vampires (Fortitude)

# health boxes are displayed in the loc keyed to POD_trait_health_track

# Trait xp corresponding to vanilla wound traits

POD_total_damage_boxes = {
	value = 0
	add = POD_superficial_damage_boxes
	add = POD_aggravated_damage_boxes
	floor = yes
}
POD_superficial_damage_boxes = {
	if = {
		limit = { has_variable = POD_health_damage_num }
		add = var:POD_health_damage_num
	}
	max = {
		value = POD_health_track_length
		subtract = POD_aggravated_damage_boxes
	}
	floor = yes
}
POD_aggravated_damage_boxes = {
	if = {
		limit = { has_variable = POD_health_aggravated_damage_num }
		add = var:POD_health_aggravated_damage_num
	}
	max = POD_health_track_length
	floor = yes
}
POD_empty_damage_boxes = {
	value = POD_health_track_length
	subtract = POD_total_damage_boxes
	#floor = yes
}
POD_health_track_percentage = {
	value = POD_total_damage_boxes
	divide = POD_health_track_length
	multiply = 100
	floor = yes
	max = 100
}
POD_aggravated_damage_threshold_for_death = {
	value = POD_health_track_length
	if = {
		limit = { has_perk = shintai_garden_of_bones_perk }
		multiply = 1
	}
	else_if = { 
		limit = { has_perk = shintai_relentless_bone_perk }
		multiply = 0.75
	}
	else = {
		multiply = 0.5
	}
}
