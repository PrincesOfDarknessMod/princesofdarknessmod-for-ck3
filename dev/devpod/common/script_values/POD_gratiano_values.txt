POD_gratiano_assault_edge_value = {
	value = 0
	if = {
		limit = { POD_gratiano_conspiracy_value >= 90 }
		add = 10
	}
	if = {
		limit = { has_character_flag = POD_gratiano_elders_framed }
		add = 5
	}
	add = {
		value = POD_gratiano_assamite_support_value
		multiply = 5
	}
	if = {
		limit = { has_character_flag = POD_gratiano_forced_strike }
		subtract = 20
	}
	If = {
		limit = { POD_gratiano_strike_doomed_trigger = YES }
		subtract = 30
	}
}
