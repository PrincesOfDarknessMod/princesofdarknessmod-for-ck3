mass_torture_num_of_prisoners = {
    value = 0
    every_prisoner = {
		limit = {
			NOT = { has_character_modifier = recently_tortured }
			NOT = { has_character_flag = is_being_tortured }
		}
		add = 1
	}
}

mass_torture_piety_gain = {
	value = 0
	
	if = {
		limit = {
			faith = {
				OR = {
					has_doctrine = tenet_exaltation_of_pain
					has_doctrine_parameter = torturing_prisoners_grants_piety
				}
			}
		}
		add = medium_piety_gain
	}
	else_if = {
		limit = {
			# Divine Retribution cancels out faith-based penalties
			NOT = { has_perk = divine_retribution_perk }
		}
		add = medium_piety_loss
	}

	multiply = mass_torture_num_of_prisoners
}

# refer to POD_mass_demand_payment_interaction in common\character_interactions\00_POD_qol_interactions.txt
mass_payment_num_of_characters = {
	value = 0
	every_hooked_character = {
		limit = {
			trigger_if = {
				limit = { scope:exclude_partial_payments = no }
				gold >= 1
			}
			trigger_else = {
				gold > golden_obligation_value
			}

			trigger_if = {
				limit = { scope:strong_hooks_only = yes }
				scope:actor = { has_strong_usable_hook = prev }
			}
			trigger_else = {
				scope:actor = { has_usable_hook = prev }
			}

			is_ai = yes
			is_imprisoned = no
			NOT = { is_at_war_with = scope:actor }

			# If payment was unlocked via struggle, check if both characters are in the same struggle
			trigger_if = {
				limit = {
					scope:actor = {
						NOT = { has_perk = golden_obligations_perk }
						any_character_struggle = {
							involvement = involved
							has_struggle_phase_parameter = unlocks_demand_payments_for_all
						}
					}
				}
				any_character_struggle = {
					involvement = involved
					has_struggle_phase_parameter = unlocks_demand_payments_for_all
					is_secondary_character_involvement_either_trigger = {
						CHAR = scope:actor
					}
				}
			}
		}
		add = 1
	}
}

mass_payment_total_gold = {
	value = 0
	every_hooked_character = {
		limit = {
			trigger_if = {
				limit = { scope:exclude_partial_payments = no }
				gold >= 1
			}
			trigger_else = {
				gold > golden_obligation_value
			}

			trigger_if = {
				limit = { scope:strong_hooks_only = yes }
				scope:actor = { has_strong_usable_hook = prev }
			}
			trigger_else = {
				scope:actor = { has_usable_hook = prev }
			}

			is_ai = yes
			is_imprisoned = no
			NOT = { is_at_war_with = scope:actor }

			# If payment was unlocked via struggle, check if both characters are in the same struggle
			trigger_if = {
				limit = {
					scope:actor = {
						NOT = { has_perk = golden_obligations_perk }
						any_character_struggle = {
							involvement = involved
							has_struggle_phase_parameter = unlocks_demand_payments_for_all
						}
					}
				}
				any_character_struggle = {
					involvement = involved
					has_struggle_phase_parameter = unlocks_demand_payments_for_all
					is_secondary_character_involvement_either_trigger = {
						CHAR = scope:actor
					}
				}
			}
		}

		if = {
			limit = { gold > golden_obligation_value }
			add = golden_obligation_value
		}
		else = {
			add = {
				value = gold
				floor = yes
			}
		}
	}
}
