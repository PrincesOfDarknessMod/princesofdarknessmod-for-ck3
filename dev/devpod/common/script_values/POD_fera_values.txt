
gnosis_level = {
	value = 0
	
	if = {
		limit = { has_character_modifier = POD_gnosis_1_modifier }
		add = 1
	}
	else_if = {
		limit = { has_character_modifier = POD_gnosis_2_modifier }
		add = 2
	}
	else_if = {
		limit = { has_character_modifier = POD_gnosis_3_modifier  }
		add = 3
	}
	else_if = {
		limit = { has_character_modifier = POD_gnosis_4_modifier }
		add = 4
	}
	else_if = {
		limit = { has_character_modifier = POD_gnosis_5_modifier }
		add = 5
	}
}

gnosis_difference = {
	value = 5

	subtract = gnosis_level
}

total_gnosis_reserves = {
	value = 0

	if = {
		limit = { POD_splat_has_gnosis_trigger = yes }
		add = gnosis_level
		add = total_stored_energy
	}
}

gnosis_regen_chance = {
	value = 0

	add = {
		value = 20
		desc = BASE_VALUE
		format = "POD_FORMAT_PERCENTAGE_BASE"
	}
	if = {
		limit = { has_trait = beast_born }
		add = {
			value = 10
			desc = POD_gnosis_regen_beast_born
			format = "POD_FORMAT_PERCENTAGE_MODIFIER"
		}
	}
	if = {
		limit = { piety_level = 2 }
		add = {
			value = 2.5
			desc = POD_gnosis_regen_piety_level
			format = "POD_FORMAT_PERCENTAGE_MODIFIER"
		}
	}
	if = {
		limit = { piety_level = 3 }
		add = {
			value = 5
			desc = POD_gnosis_regen_piety_level
			format = "POD_FORMAT_PERCENTAGE_MODIFIER"
		}
	}
	if = {
		limit = { piety_level = 4 }
		add = {
			value = 7.5
			desc = POD_gnosis_regen_piety_level
			format = "POD_FORMAT_PERCENTAGE_MODIFIER"
		}
	}
	if = {
		limit = { piety_level >= 5 }
		add = {
			value = 10
			desc = POD_gnosis_regen_piety_level
			format = "POD_FORMAT_PERCENTAGE_MODIFIER"
		}
	}

	if = {
		limit = {
			any_character_artifact = {
				has_variable = grandmothers_stones
				is_equipped = yes
				prev = { can_benefit_from_artifact = prev }
			}
		}
		add = {
			value = 5
			desc = POD_gnosis_regen_artifact
			format = "POD_FORMAT_PERCENTAGE_MODIFIER"
		}
	}

	###Buildings
	if = {
        limit = { is_landed = yes }
        add = {
            value = {
                every_held_title = {
                    limit = { 
						tier = tier_barony 
						is_landless_type_title = no
					}
                    title_province = {
                        if = {
                            limit = { has_building_with_flag = building_gnosis_regen_2_5 }
                            add = 2.5
                        }
                    }
                }
            }
            desc = POD_gnosis_regen_buildings
            format = "POD_FORMAT_PERCENTAGE_MODIFIER"
        }
    }

	###Perks
	if = {
		limit = {
			OR = {
				has_perk = fomori_spirit_ties_perk
				has_perk = gorgon_spirit_ties_perk
				has_perk = drone_spirit_ties_perk
				has_perk = kami_spirit_ties_perk
			}
		}
		add = {
			value = 5
			desc = POD_gnosis_regen_spirit_ties
			format = "POD_FORMAT_PERCENTAGE_MODIFIER"
		}
	}

	
	##################
	#### Negative ####
	##################

	if = {
		limit = { stress_level >= 1 }
		
		subtract = {
			value = stress_level
			multiply = 5

			desc = POD_gnosis_regen_stress
			format = "POD_FORMAT_PERCENTAGE_MODIFIER_NEGATIVE"
		}
	}


	if = {
		limit = {
			POD_is_kinfolk_trigger = yes
			gnosis_level >= 3
		}
		multiply = {
			value = 0
			desc = POD_gnosis_regen_kinfolk_cap
			format = "POD_FORMAT_PERCENTAGE_NULLIFIED"
		}
	}





	min = 0
	max = 100
}


fera_shapeshifter_chance = {
	value = 25


	if = {
		limit = { mother ?= { has_trait = fera } }
		add = 25
	}
	
	if = {
		limit = { father ?= { has_trait = fera } }
		add = 25
	}

	if = {
		limit = {
			real_father ?= {
				has_trait = fera
				is_ai = yes
				is_landed = yes
			}
		}
		add = 10
	}

	if = {
		limit = {
			mother ?= {
				has_trait = fera
				is_ai = yes
				is_landed = yes
			}
		}
		add = 10
	}

	if = {
		limit = {
			any_close_family_member = {
				is_landed = yes
				is_ai = yes
				has_trait = fera
			}
		}
		add = 20
	}

	if = {
		limit = {
			OR = {
				real_father ?= {
					has_trait = bastet
					any_character_artifact = { has_variable = lotus_petals }
				}
				mother ?= {
					has_trait = bastet
					any_character_artifact = { has_variable = lotus_petals }
				}
			}
		}
		add = 100
	}


	min = 0
	max = 100
}
