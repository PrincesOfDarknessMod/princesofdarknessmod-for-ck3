
gnosis_regen_chance = {
	value = 0

	add = {
		value = 20
		desc = BASE_VALUE
		format = "POD_FORMAT_PERCENTAGE_BASE"
	}
	if = {
		limit = { has_trait = beast_born }
		add = {
			value = 10
			desc = POD_gnosis_regen_beast_born
			format = "POD_FORMAT_PERCENTAGE_MODIFIER"
		}
	}

	if = {
		limit = { piety_level = 5 }
		add = {
			value = 10
			desc = POD_gnosis_regen_piety_level
			format = "POD_FORMAT_PERCENTAGE_MODIFIER"
		}
	}
	else_if = {
		limit = { piety_level = 4 }
		add = {
			value = 7.5
			desc = POD_gnosis_regen_piety_level
			format = "POD_FORMAT_PERCENTAGE_MODIFIER"
		}
	}
	else_if = {
		limit = { piety_level = 3 }
		add = {
			value = 5
			desc = POD_gnosis_regen_piety_level
			format = "POD_FORMAT_PERCENTAGE_MODIFIER"
		}
	}
	else_if = {
		limit = { piety_level = 2 }
		add = {
			value = 2.5
			desc = POD_gnosis_regen_piety_level
			format = "POD_FORMAT_PERCENTAGE_MODIFIER"
		}
	}

	###Spirit Companions
	if = {
		limit = { num_of_relation_spirit_companion >= 1 }
		add = {
            value = {
                every_relation = {
					type = spirit_companion
					limit = { is_alive = yes }

					if = {
						limit = { POD_spirit_rank = 5 }
						add = 10
					}
					else_if = {
						limit = { POD_spirit_rank = 4 }
						add = 8
					}
					else_if = {
						limit = { POD_spirit_rank = 3 }
						add = 6
					}
					else_if = {
						limit = { POD_spirit_rank = 2 }
						add = 4
					}
					else = {
						add = 2
					}
				}
            }
            desc = POD_gnosis_regen_spirit_companions
            format = "POD_FORMAT_PERCENTAGE_MODIFIER"
        }
	}

	###Buildings
	if = {
        limit = { is_landed = yes }
        add = {
            value = {
                every_held_title = {
                    limit = { 
						tier = tier_barony 
						is_landless_type_title = no
					}
                    title_province = {
                        if = {
                            limit = { has_building_with_flag = building_gnosis_regen_2_5 }
                            add = 2.5
                        }
                    }
                }
            }
            desc = POD_gnosis_regen_buildings
            format = "POD_FORMAT_PERCENTAGE_MODIFIER"
        }
    }

	###Perks
	if = {
		limit = {
			OR = {
				has_perk = fomori_spirit_ties_perk
				has_perk = gorgon_spirit_ties_perk
				has_perk = drone_spirit_ties_perk
				has_perk = kami_spirit_ties_perk
			}
		}
		add = {
			value = 5
			desc = POD_gnosis_regen_spirit_ties
			format = "POD_FORMAT_PERCENTAGE_MODIFIER"
		}
	}

	if = {
		limit = { has_perk = sleep_of_the_dragon_perk }
		add = {
			value = 10
			desc = POD_gnosis_regen_sleep_of_the_dragon
			format = "POD_FORMAT_PERCENTAGE_MODIFIER"
		}
	}
	if = {
		limit = { has_perk = fools_luck_perk }
		add = {
			value = 10
			desc = POD_gnosis_regen_fools_luck
			format = "POD_FORMAT_PERCENTAGE_MODIFIER"
		}
	}

	###Legacies
	if = {
		limit = { dynasty ?= { has_dynasty_perk = gumagan_legacy_5 } }
		add = {
			value = 5
			desc = POD_gnosis_regen_gumagan_legacy_5
			format = "POD_FORMAT_PERCENTAGE_MODIFIER"
		}
	}
	
	###Modifiers
	if = {
		limit = { has_character_modifier = pegasus_totem_modifier }
		add = {
			value = 5
			desc = pegasus_totem_modifier
			format = "POD_FORMAT_PERCENTAGE_MODIFIER"
		}
	}



	##################
	#### Negative ####
	##################

	#Fetish Upkeep
	if = {
		limit = {
			any_character_artifact = { 
				is_equipped = yes
				has_variable = fetish_artifact
			}
		}
		add = {
            value = {
                every_equipped_character_artifact = {
                    limit = { has_variable = fetish_artifact }
					add = -5
                }
            }
            desc = POD_gnosis_regen_fetish_upkeep
            format = "POD_FORMAT_PERCENTAGE_MODIFIER_NEGATIVE"
        }
	}

	#Stress
	if = {
		limit = { stress_level >= 1 }
		
		subtract = {
			value = stress_level
			multiply = 5

			desc = POD_gnosis_regen_stress
			format = "POD_FORMAT_PERCENTAGE_MODIFIER_NEGATIVE"
		}
	}

	#Kinfolk Limiter
	if = {
		limit = {
			POD_is_kinfolk_trigger = yes
			gnosis_level >= 3
		}
		multiply = {
			value = 0
			desc = POD_gnosis_regen_kinfolk_cap
			format = "POD_FORMAT_PERCENTAGE_NULLIFIED"
		}
	}





	min = 0
	max = 100
}


fera_shapeshifter_chance = {
	value = 25


	if = {
		limit = { mother ?= { has_trait = fera } }
		add = 25
	}
	
	if = {
		limit = { father ?= { has_trait = fera } }
		add = 25
	}

	if = {
		limit = {
			real_father ?= {
				has_trait = fera
				is_ai = yes
				is_landed = yes
			}
		}
		add = 10
	}

	if = {
		limit = {
			mother ?= {
				has_trait = fera
				is_ai = yes
				is_landed = yes
			}
		}
		add = 10
	}

	if = {
		limit = {
			any_close_family_member = {
				is_landed = yes
				is_ai = yes
				has_trait = fera
			}
		}
		add = 20
	}

	if = {
		limit = {
			OR = {
				real_father ?= {
					has_trait = bastet
					any_character_artifact = { has_variable = lotus_petals }
				}
				mother ?= {
					has_trait = bastet
					any_character_artifact = { has_variable = lotus_petals }
				}
			}
		}
		add = 100
	}


	min = 0
	max = 100
}
