
gnosis_regen_chance = {
	value = 0

	add = {
		value = 20
		desc = BASE_VALUE
		format = "POD_FORMAT_PERCENTAGE_BASE"
	}
	if = {
		limit = { has_trait = beast_born }
		add = {
			value = 10
			desc = POD_gnosis_regen_beast_born
			format = "POD_FORMAT_PERCENTAGE_MODIFIER"
		}
	}

	if = {
		limit = { piety_level = 5 }
		add = {
			value = 10
			desc = POD_gnosis_regen_piety_level
			format = "POD_FORMAT_PERCENTAGE_MODIFIER"
		}
	}
	else_if = {
		limit = { piety_level = 4 }
		add = {
			value = 7.5
			desc = POD_gnosis_regen_piety_level
			format = "POD_FORMAT_PERCENTAGE_MODIFIER"
		}
	}
	else_if = {
		limit = { piety_level = 3 }
		add = {
			value = 5
			desc = POD_gnosis_regen_piety_level
			format = "POD_FORMAT_PERCENTAGE_MODIFIER"
		}
	}
	else_if = {
		limit = { piety_level = 2 }
		add = {
			value = 2.5
			desc = POD_gnosis_regen_piety_level
			format = "POD_FORMAT_PERCENTAGE_MODIFIER"
		}
	}

	###Spirit Companions
	if = {
		limit = { num_of_relation_spirit_companion >= 1 }
		add = {
            value = {
                every_relation = {
					type = spirit_companion
					limit = { is_alive = yes }

					if = {
						limit = { POD_spirit_rank = 5 }
						add = 10
					}
					else_if = {
						limit = { POD_spirit_rank = 4 }
						add = 8
					}
					else_if = {
						limit = { POD_spirit_rank = 3 }
						add = 6
					}
					else_if = {
						limit = { POD_spirit_rank = 2 }
						add = 4
					}
					else = {
						add = 2
					}
				}
            }
            desc = POD_gnosis_regen_spirit_companions
            format = "POD_FORMAT_PERCENTAGE_MODIFIER"
        }
	}

	###Buildings
	if = {
        limit = { is_landed = yes }
        add = {
            value = {
                every_held_title = {
                    limit = { 
						tier = tier_barony 
						is_landless_type_title = no
					}
                    title_province = {
                        if = {
                            limit = { has_building_with_flag = building_gnosis_regen_2_5 }
                            add = 2.5
                        }
						else_if = {
							limit = { has_building_with_flag = building_gnosis_regen_5 }
                            add = 5
						}
                    }
                }
            }
            desc = POD_gnosis_regen_buildings
            format = "POD_FORMAT_PERCENTAGE_MODIFIER"
        }
    }

	###Domicile Buildings
	if = {
        limit = { domicile ?= { has_domicile_building_or_higher = mokole_main_meditation_chamber_01 } }
		domicile = {
			add = {
				value = {
					switch = {
						trigger = has_domicile_building

						mokole_main_meditation_chamber_06 = { add = 15 }
						mokole_main_meditation_chamber_05 = { add = 12.5 }
						mokole_main_meditation_chamber_04 = { add = 10 }
						mokole_main_meditation_chamber_03 = { add = 7.5 }
						mokole_main_meditation_chamber_02 = { add = 5 }
						mokole_main_meditation_chamber_01 = { add = 2.5 }
					}
				}
				desc = POD_gnosis_regen_domicile
				format = "POD_FORMAT_PERCENTAGE_MODIFIER"
			}
		}
    }

	###Artifacts, DO NOT PUSH A FETISH HERE
	if = {
        limit = { any_equipped_character_artifact = { has_variable = increases_gnosis_regen } }
        add = {
            value = {
                every_equipped_character_artifact = {
                    limit = { has_variable = increases_gnosis_regen }
					if = {
						limit = { has_variable = archeorex }
						add = 5
					}
                }
            }
            desc = POD_gnosis_regen_artifacts
            format = "POD_FORMAT_PERCENTAGE_MODIFIER"
        }
    }

	if = {
		limit = {
			OR = {
				has_perk = fomori_spirit_ties_perk

			value = 10

		}
	}

	#add an extra scaling value so higher lvl ahrouns don't struggle so much

		value = var:ahroun_target_prowess 
		divide = 4
	}

	#if they managed to beat someone much stronger, give an extra reward
	if = {
		limit = { var:ahroun_target_prowess >= POD_ahroun_target_prowess_worthy }
		add = var:ahroun_target_prowess
	}

	#if its a practice fight, reduce xp gain
	if = {
		limit = { pod_combat_is_practice_trigger = yes }
		divide = 2
	}

	divide = 30 #divide the whole thing

	min = 1
}
