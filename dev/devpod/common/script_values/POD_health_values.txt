POD_health_damage_limit = {
	value = 10
	
	max = 20 # In case of you wanting to increase that, you have to also increas it in `00_POD_misc_custom_loc`
	min = 1
}
POD_health_damage_limit_half = {
	value = POD_health_damage_limit
	divide = 2
}
POD_health_damage = {
	value = 0
	add = POD_health_normal_damage
	add = POD_health_aggravated_damage
	floor = yes
}
POD_health_normal_damage = {
	value = 0
	if = { limit = { has_variable = POD_health_damage_num } add = var:POD_health_damage_num }
	max = POD_health_damage_limit
	floor = yes
}

	value = 0
	if = { limit = { has_variable = POD_health_aggravated_damage_num } add = var:POD_health_aggravated_damage_num }
	max = POD_health_damage_limit
	floor = yes
}
POD_health_aggravated_damage_tt = {
	value = POD_health_aggravated_damage
	add = scope:amount
}

	value = POD_health_damage_limit

	floor = yes
}

	value = POD_health_damage_rest

}
