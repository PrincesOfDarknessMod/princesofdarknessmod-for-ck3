	every_county = {

	}

	multiply = hand_status

}

	if = {
		limit = { this = religion:kueijin_religion }
		value = kueijin_county_value
	}
	else = {
		value = religion_county_value
	}
	multiply = hand_status
	multiply = 0.2
	floor = yes
}
