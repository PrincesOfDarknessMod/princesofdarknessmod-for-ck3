POD_fera_weather_magic_interaction_cost = {
	if = { limit = { scope:county		?= yes } add = 1 }
	if = { limit = { scope:duchy		?= yes } add = 2 }
	if = { limit = { scope:kingdom		?= yes } add = 3 }
	if = { limit = { scope:empire		?= yes } add = 5 }

	if = { limit = { scope:greater_flood	?= yes } add = 1 }
	if = { limit = { scope:nourishing_tide	?= yes } add = 1 }

	if = {
		limit = { 
			OR = {
				scope:flood ?= yes
				scope:greater_flood ?= yes
			}
			OR = {
				AND = {
					has_perk = call_the_tides_perk
					has_perk = flood_the_land_perk
				}
				faith = { has_doctrine_parameter = reduces_flood_weather_cost }
			}
			
		}
		subtract = 1
	}
	
	if = {
		limit = { 
			scope:actor = { has_character_modifier = susanowo_totem_modifier }
			scope:storm = yes
			scope:recipient = { is_at_war_with = scope:actor }
		}
		add = 3
	}
	# no min cost to make the interaction tooltip properly
}
