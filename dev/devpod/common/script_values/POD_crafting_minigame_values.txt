        value = { value = 2 desc = BASE }
        if = { 
                limit = { has_variable = crafting_minigame var:crafting_minigame.var:type ?= flag:forge } 
                add = { value = var:crafting_minigame.var:crafter.crafting_minigame_moves_crafter_forging desc = crafting_minigame_moves_crafter }
                
                #if = { limit = { has_character_modifier = POD_one_time_forge_boost } add = { value = var:POD_one_time_forge_boost desc = crafting_minigame_one_time_boost } }
        }
        if = { 
                limit = { has_variable = crafting_minigame var:crafting_minigame.var:type ?= flag:brew } 
                add = { value = var:crafting_minigame.var:crafter.crafting_minigame_moves_crafter_brew desc = crafting_minigame_moves_crafter }
                if = { limit = { var:crafting_minigame.var:crafter ?= { has_perk = divine_quiddity_perk } } add = { value = 1 desc = DIVINE_QUIDDITY_EFFECT_DESC } }
                if = { limit = { var:crafting_minigame.var:crafter ?= { has_perk = kingly_quiddity_perk } } add = { value = 1 desc = KINGLY_QUIDDITY_EFFECT_DESC } }
                if = { limit = { var:crafting_minigame.var:crafter ?= { has_perk = princely_quiddity_perk } } add = { value = 1 desc = PRINCELY_QUIDDITY_EFFECT_DESC } }
                if = { limit = { var:crafting_minigame.var:crafter ?= { has_perk = greater_quiddity_perk } } add = { value = 1 desc = GREATER_QUIDDITY_EFFECT_DESC } }
                if = { limit = { var:crafting_minigame.var:crafter ?= { has_perk = lesser_quiddity_perk } } add = { value = 1 desc = LESSER_QUIDDITY_EFFECT_DESC } }

                #if = { limit = { has_character_modifier = POD_one_time_brew_boost } add = { value = var:POD_one_time_brew_boost desc = crafting_minigame_one_time_boost } }
        }
        if = { 
                limit = { has_variable = crafting_minigame var:crafting_minigame.var:type ?= flag:enhance } 
                add = { value = var:crafting_minigame.var:crafter.crafting_minigame_moves_crafter_enhance desc = crafting_minigame_moves_crafter }
        
                #if = { limit = { has_character_modifier = POD_one_time_enhance_boost } add = { value = var:POD_one_time_enhance_boost desc = crafting_minigame_one_time_boost } }
        }
        if = { 
                limit = { has_variable = crafting_minigame var:crafting_minigame = { has_variable = moves } } 
                subtract = { value = var:crafting_minigame.var:moves desc = POD_already_used_moves } 
        }
}
