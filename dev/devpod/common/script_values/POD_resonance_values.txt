#The numbers chosen here are chosen with the goal that a vampire with 5 herdmembers of level 2 (fleeting) resonance will gain xp at a rate that gives them one perk worth of xp per 5 years.

#I'm keeping the base idea but rewards more the higher resonance tiers to encourage people to really look for those high quality mortals -Karde
negligible_resonance_xp = 6
fleeting_resonance_xp = 6.67
intense_resonance_xp = 10   #3000 xp over 5 years, a little less than 2 perks
acute_resonance_xp = 13.33  #4000 xp over 5 years, so 2 perks

#Old system numbers
#negligible_resonance_xp = 1
#fleeting_resonance_xp = 10
#intense_resonance_xp = 100
#acute_resonance_xp = 1000


#This is so you can just do "add_X_lifestyle_xp = resonance_xp_value" instead of having to make a lot of ifs
resonance_xp_value = {
	value = 0 
	if = {
		limit = {
			has_character_flag = POD_resonance_negligible_flag
		}
		add = negligible_resonance_xp
	}
	if = {
		limit = {
			has_character_flag = POD_resonance_fleeting_flag
		}
		add = fleeting_resonance_xp
	}
	if = {
		limit = {
			has_character_flag = POD_resonance_intense_flag
		}
		add = intense_resonance_xp 
	}
	if = {
		limit = {
			has_character_flag = POD_resonance_acute_flag
		}
		add = acute_resonance_xp
	}
}

resonance_xp_sip_value = {
	value = resonance_xp_value 
	multiply = 10
}

resonance_xp_drink_value = { # Only used in event
	value = resonance_xp_sip_value 
	multiply = 2
}

resonance_xp_lot_value = { # Only used in event
	value = resonance_xp_drink_value 
	multiply = 2
}

resonance_xp_drain_value = {
	value = resonance_xp_lot_value 
	multiply = 2
}

resonance_rating_value = {
	value = 0 
	if = {
		limit = {
			has_character_flag = POD_resonance_negligible_flag
		}
		add = 1
	}
	if = {
		limit = {
			has_character_flag = POD_resonance_fleeting_flag
		}
		add = 2
	}
	if = {
		limit = {
			has_character_flag = POD_resonance_intense_flag
		}
		add = 3 
	}
	if = {
		limit = {
			has_character_flag = POD_resonance_acute_flag
		}
		add = 4
	}
}
