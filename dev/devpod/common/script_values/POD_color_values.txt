##### Clothing color values

POD_initial_color_max = 0.6
POD_initial_color_min = 0.1

# TopScope needs to contain scope:red, scope:green, and scope:blue (values between 0 and 1)

POD_color_darkest = {
	value = scope:red
	max   = scope:green
	max   = scope:blue
}

POD_color_lightest = {
	value = scope:red
	min   = scope:green
	min   = scope:blue
}

POD_color_rgb2chroma = {
	value    = POD_color_lightest
	subtract = POD_color_darkest
}

POD_color_hue = {
	if = {
		limit = { POD_color_rgb2chroma != 0 } # return 0 if chroma is 0
		if = {
			limit = { scope:red = POD_color_lightest }
			value    = scope:green
			subtract = scope:blue
			divide   = POD_color_rgb2chroma
			modulo   = 6
			divide   = 6
		}
		else_if = {
			limit = { scope:green = POD_color_lightest }
			value    = scope:blue
			subtract = scope:red
			divide   = POD_color_rgb2chroma
			add      = 2
			divide   = 6
		}
		else = {
			value    = scope:red
			subtract = scope:green
			divide   = POD_color_rgb2chroma
			add      = 4
			divide   = 6
		}
	}
}

POD_color_rgb2h = {
	if = {
		limit = { POD_color_hue < 0 }
		value = 1
		add   = POD_color_hue
	}
	else = {
		value = POD_color_hue
	}
}

POD_color_rgb2s = {
	if = {
		limit  = { POD_color_lightest != 0 } # return 0 if lightest color is 0
		value  = POD_color_rgb2chroma
		divide = POD_color_lightest
	}
}

POD_color_rgb2v = {
	value = POD_color_lightest
}

# TopScope needs to contain scope:hue, scope:saturation, and scope:value (values between 0 and 1)

POD_color_hsv2chroma = {
	value    = scope:saturation
	multiply = scope:value
}

POD_color_hue6 = {
	value    = scope:hue
	multiply = 6
}

POD_color_hue6_mod = {
	value    = scope:hue
	multiply = 6
	modulo   = 2
	subtract = 1
}

POD_color_hsv_intermediate = {
	value    = POD_color_hsv2chroma
	multiply = {
		value = 1
		if = {
			limit = { POD_color_hue6_mod < 0 }
			add = POD_color_hue6_mod
		}
		else = {
			subtract = POD_color_hue6_mod
		}
	}
}

POD_color_hsv2r = {
	if = {
		limit = { POD_color_hue6 < 1 }
		value = POD_color_hsv2chroma
	}
	else_if = {
		limit = { POD_color_hue6 < 2 }
		value = POD_color_hsv_intermediate
	}
	else_if = {
		limit = { POD_color_hue6 < 4 }
		value = 0
	}
	else_if = {
		limit = { POD_color_hue6 < 5 }
		value = POD_color_hsv_intermediate
	}
	else = {
		value = POD_color_hsv2chroma
	}

	add      = scope:value
	subtract = POD_color_hsv2chroma
}

POD_color_hsv2g = {
	if = {
		limit = { POD_color_hue6 < 1 }
		value = POD_color_hsv_intermediate
	}
	else_if = {
		limit = { POD_color_hue6 < 3 }
		value = POD_color_hsv2chroma
	}
	else_if = {
		limit = { POD_color_hue6 < 4 }
		value = POD_color_hsv_intermediate
	}
	else = {
		value = 0
	}

	add      = scope:value
	subtract = POD_color_hsv2chroma
}

POD_color_hsv2b = {
	if = {
		limit = { POD_color_hue6 < 2 }
		value = 0
	}
	else_if = {
		limit = { POD_color_hue6 < 3 }
		value = POD_color_hsv_intermediate
	}
	else_if = {
		limit = { POD_color_hue6 < 5 }
		value = POD_color_hsv2chroma
	}
	else = {
		value = POD_color_hsv_intermediate
	}

	add      = scope:value
	subtract = POD_color_hsv2chroma
}
