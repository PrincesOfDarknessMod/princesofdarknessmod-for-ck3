POD_baghatur_objective_military_strength = {
	value = prowess
	every_knight = {
		limit = { 
			has_trait = supehunter
			has_faith = faith:qorjin_baghatur
		}
		add = prowess
	}
	every_vassal = {
		limit = { 
			has_trait = supehunter
			has_faith = faith:qorjin_baghatur
		}
		add = prowess
		every_knight = {
			limit = { 
				has_trait = supehunter
				has_faith = faith:qorjin_baghatur
			}
			add = prowess
		}
	}
	if = {
		limit = { has_variable = POD_baghatur_objective_military_strength_additional }
		add = var:POD_baghatur_objective_military_strength_additional
	}
}

akritai_saint_worthiness_value = {
	value = 1

	# More virtuous traits = worthier.
	add = {
		value = num_virtuous_traits
		multiply = 3
	}
	if = {
		limit = { has_trait = zealous }
		add = 2
	}

	# Beatific is better than TF
	if = {
		limit = { has_trait = beatific }
		add = 9  # worth 3 virtuous traits
	}

	# Elders are more readily venerated.
	add = {
		value = age
		multiply = 0.1
	}

	# Theological learning lends itself to sainthood.
	add = {
		value = learning
		multiply = 0.2
	}
}

akritai_saint_relic_quality_value = {
	value = 20
	add = {
		value = akritai_saint_worthiness_value
		multiply = 1.5
	}
	min = 25
	max = 100
	round = yes
}

akritai_saint_count_value = {
	value = 0
	if = {
		limit = {
			exists = global_var:akritai_saint_count
		}
		add = global_var:akritai_saint_count
	}
}

akritai_canonize_saint_piety_cost = {
	value = 3000

	}

	}

}
