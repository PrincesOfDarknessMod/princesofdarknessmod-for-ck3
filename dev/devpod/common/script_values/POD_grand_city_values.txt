# values for passive influence gain
# base isn't 0 so is displayed but is still rounded down to 0
# implementation of POD_terrain_jerusalem_*_development_growth
POD_jerusalem_christian_influence_passive = {
	value = { value = 0.0001 desc = BASE }
	add = { 
		value = POD_jerusalem_equilibrium_difference_christian_influence
		multiply = -0.075
		desc = JERUSALEM_PULL_TO_EQUILIBRIUM
	}

	# Crown speaker
	if = {
		limit = {
			exists = court_position:jerusalem_crown_speaker_court_position
			court_position:jerusalem_crown_speaker_court_position ?= {
				has_variable = jerusalem_crown_speaker_court_position_improving_christian
			}
		}
		add = { 
			value = POD_jerusalem_crown_speaker_influence_passive_gain
			desc = JERUSALEM_CROWN_SPEAKER_BONUS
		}
	}

	# Buildings
	if = {
		limit = { any_directly_owned_province = { has_building = jerusalem_christian_holy_sepulcher_04 } }
		add = { 
			value = 2
			desc = JERUSALEM_CHRISTIAN_HOLY_SEPULCHER
		}
	}
	else_if = {
		limit = { any_directly_owned_province = { has_building = jerusalem_christian_holy_sepulcher_03 } }
		add = { 
			value = 1
			desc = JERUSALEM_CHRISTIAN_HOLY_SEPULCHER
		}
	}

	round = yes
}

# for converting influence into a modifier
# e.g. at 80% christian influence, select the rank 4 modifier
# needs scope:influence_ratio saved on the TopScope
POD_jerusalem_influence_rank = {
	if = {
		limit = { scope:influence_ratio >= POD_jerusalem_influence_rank_5_min }
		value = 5
	}
	else_if = {
		limit = { scope:influence_ratio >= POD_jerusalem_influence_rank_4_min }
		value = 4
	}
	else_if = {
		limit = { scope:influence_ratio >= POD_jerusalem_influence_rank_3_min }
		value = 3
	}
	else_if = {
		limit = { scope:influence_ratio >= POD_jerusalem_influence_rank_2_min }
		value = 2
	}
	else = {
		value = 1
	}
}
