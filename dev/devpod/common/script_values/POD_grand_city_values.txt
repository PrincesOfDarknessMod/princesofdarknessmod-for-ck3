POD_jerusalem_initial_christian_influence	= 65
POD_jerusalem_initial_muslim_influence		= 75
POD_jerusalem_initial_jewish_influence		= 25

POD_jerusalem_minor_influence_gain = 5
POD_jerusalem_major_influence_gain = 10
POD_jerusalem_minor_influence_loss = -5
POD_jerusalem_major_influence_loss = -10


# jerusalem variables are saved via POD_add_jerusalem_influence

POD_jerusalem_christian_influence = {
	title:c_jerusalem = {
		if = {
			limit = { has_variable = christian_influence }
			value = var:christian_influence
		}
		else = {
			value = POD_jerusalem_initial_christian_influence
		}
	}
}

POD_jerusalem_muslim_influence = {
	title:c_jerusalem = {
		if = {
			limit = { has_variable = muslim_influence }
			value = var:muslim_influence
		}
		else = {
			value = POD_jerusalem_initial_muslim_influence
		}
	}
}

POD_jerusalem_jewish_influence = {
	title:c_jerusalem = {
		if = {
			limit = { has_variable = jewish_influence }
			value = var:jewish_influence
		}
		else = {
			value = POD_jerusalem_initial_jewish_influence
		}
	}
}

POD_jerusalem_total_influence = {
	add = POD_jerusalem_christian_influence
	add = POD_jerusalem_muslim_influence
	add = POD_jerusalem_jewish_influence
}

POD_jerusalem_christian_influence_ratio = {
	value = POD_jerusalem_christian_influence
	divide = POD_jerusalem_total_influence
}

POD_jerusalem_muslim_influence_ratio = {
	value = POD_jerusalem_muslim_influence
	divide = POD_jerusalem_total_influence
}

POD_jerusalem_jewish_influence_ratio = {
	value = POD_jerusalem_jewish_influence
	divide = POD_jerusalem_total_influence
}


POD_jerusalem_influence_rank_5_min = 0.9
POD_jerusalem_influence_rank_4_min = 0.66
POD_jerusalem_influence_rank_3_min = 0.33
POD_jerusalem_influence_rank_2_min = 0.1

# for converting influence into a modifier
# e.g. at 80% christian influence, select the rank 4 modifier
# needs scope:influence_ratio saved on the TopScope
POD_jerusalem_influence_rank = {
	if = {
		limit = { scope:influence_ratio >= POD_jerusalem_influence_rank_5_min }
		value = 5
	}
	else_if = {
		limit = { scope:influence_ratio >= POD_jerusalem_influence_rank_4_min }
		value = 4
	}
	else_if = {
		limit = { scope:influence_ratio >= POD_jerusalem_influence_rank_3_min }
		value = 3
	}
	else_if = {
		limit = { scope:influence_ratio >= POD_jerusalem_influence_rank_2_min }
		value = 2
	}
	else = {
		value = 1
	}
}

# instead of reading the actual variables, this one reads from scope values on the TopScope
# so that influence changes can be previewed in tooltips
# saved values are scope:christian_influence, scope:muslim_influence, and scope:jewish_influence
POD_jerusalem_total_influence_preview = {
	add = scope:christian_influence
	add = scope:muslim_influence
	add = scope:jewish_influence
}

POD_jerusalem_christian_influence_rank_preview = {
	value = scope:christian_influence
	divide = POD_jerusalem_total_influence_preview
	save_temporary_value_as = influence_ratio
	value = POD_jerusalem_influence_rank
}

POD_jerusalem_jewish_influence_rank_preview = {

	divide = POD_jerusalem_total_influence_preview

	value = POD_jerusalem_influence_rank
}

# Reflects your capital's distance to each GC
POD_capitals_squared_distance_to_rome = {
	value = "this.capital_province.squared_distance(title:c_roma)"
}

POD_capitals_squared_distance_to_constantinople = {
	value = "this.capital_province.squared_distance(title:c_byzantion)"
}

POD_capitals_squared_distance_to_paris = {
	value = "this.capital_province.squared_distance(title:c_ile_de_france)"
}

POD_capitals_squared_distance_to_cairo = {
	value = "this.capital_province.squared_distance(title:c_cairo)"
}

POD_capitals_squared_distance_to_jerusalem = {
	value = "this.capital_province.squared_distance(title:c_jerusalem)"
}

POD_capitals_squared_distance_to_baghdad = {
	value = "this.capital_province.squared_distance(title:c_baghdad)"
}

POD_capitals_squared_distance_to_closest_grand_city = {
	value = POD_capitals_squared_distance_to_rome
	save_temporary_value_as = current_minimum
	if = {
		limit = { POD_capitals_squared_distance_to_constantinople < scope:current_minimum }
		value = POD_capitals_squared_distance_to_constantinople
		save_temporary_value_as = current_minimum
	}
	if = {
		limit = { POD_capitals_squared_distance_to_paris < scope:current_minimum }
		value = POD_capitals_squared_distance_to_paris
		save_temporary_value_as = current_minimum
	}
	if = {
		limit = { POD_capitals_squared_distance_to_cairo < scope:current_minimum }
		value = POD_capitals_squared_distance_to_cairo
		save_temporary_value_as = current_minimum
	}
	if = {
		limit = { POD_capitals_squared_distance_to_jerusalem < scope:current_minimum }
		value = POD_capitals_squared_distance_to_jerusalem
		save_temporary_value_as = current_minimum
	}
	if = {
		limit = { POD_capitals_squared_distance_to_baghdad < scope:current_minimum }
		value = POD_capitals_squared_distance_to_baghdad
		save_temporary_value_as = current_minimum
	}
}
