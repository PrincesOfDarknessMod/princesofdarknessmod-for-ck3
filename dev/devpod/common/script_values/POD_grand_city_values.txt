POD_jerusalem_influence_rank_5_min = 0.9
POD_jerusalem_influence_rank_4_min = 0.66
POD_jerusalem_influence_rank_3_min = 0.33
POD_jerusalem_influence_rank_2_min = 0.1

# for converting influence into a modifier
# e.g. at 80% christian influence, select the rank 4 modifier
# needs scope:influence_ratio saved on the TopScope
POD_jerusalem_influence_rank = {
	if = {
		limit = { scope:influence_ratio >= POD_jerusalem_influence_rank_5_min }
		value = 5
	}
	else_if = {
		limit = { scope:influence_ratio >= POD_jerusalem_influence_rank_4_min }
		value = 4
	}
	else_if = {
		limit = { scope:influence_ratio >= POD_jerusalem_influence_rank_3_min }
		value = 3
	}
	else_if = {
		limit = { scope:influence_ratio >= POD_jerusalem_influence_rank_2_min }
		value = 2
	}
	else = {
		value = 1
	}
}

# instead of reading the actual variables, this one reads from scope values on the TopScope
# so that influence changes can be previewed in tooltips
# saved values are scope:christian_influence, scope:muslim_influence, and scope:jewish_influence
POD_jerusalem_total_influence_preview = {
	add = scope:christian_influence
	add = scope:muslim_influence
	add = scope:jewish_influence
}

POD_jerusalem_christian_influence_rank_preview = {
	value = scope:christian_influence
	divide = POD_jerusalem_total_influence_preview
	save_temporary_value_as = influence_ratio
	value = POD_jerusalem_influence_rank
}

POD_jerusalem_muslim_influence_rank_preview = {

}

POD_jerusalem_jewish_influence_rank_preview = {
	value = scope:jewish_influence
	divide = POD_jerusalem_total_influence_preview
	save_temporary_value_as = influence_ratio
	value = POD_jerusalem_influence_rank
}
