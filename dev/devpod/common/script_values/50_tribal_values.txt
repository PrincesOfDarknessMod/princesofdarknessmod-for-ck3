feudalize_holding_interaction_cost = {
	value = 500
	if = {
		limit = {
			is_ai = yes
			capital_province ?= { has_holding_type = tribal_holding } #Cfv
		}
		add = -250
	}
	if = {
		limit = {
			scope:target.county ?= {
				has_variable = was_razed
			}
		}
		add = {
			value = 50
			desc = BASE
		}
	}
	else_if = {
		limit = {
			scope:target.county ?= {
				culture = scope:actor.culture
			}
		}
		add = {
			value = 200
			desc = BASE
		}
	}
	else = {
		add = {
			value = 500
			desc = BASE
		}
	}
	if = {
		limit = {
			scope:actor ?= {
				is_ai = yes
				capital_province = { has_holding_type = tribal_holding }
			}
		}
		multiply = 0.5
	}	
	if = {
		limit = {
			scope:target.county ?= {
				OR = {
					has_county_modifier = mpo_siberian_permafrost_modifier
					has_county_modifier = mpo_siberian_permafrost_modifier_bad
				} 
			}
		}
		add = {
			value = 2000
			desc = mpo_siberian_permafrost_modifier
		}
	}
	if = {
		limit = {
			scope:target.county ?= {
				OR = {
					has_county_modifier = pam_moving_faithfuls_to_frontiers
					has_county_modifier = pam_building_new_shrines
				}
			}
		}
		multiply = {
			value = 0.5
			desc = pam_mass_rite_conversion_feudalize_discount
		}
	}
	if = {
		limit = {
			government_has_mechanic = administrative
		}
		if = {
			limit = { realm_law_authority_level_3 = yes }
		}
		else_if = {
			limit = { realm_law_authority_level_2 = yes }
			add = {
				multiply = 0.15
				desc = has_admin_government
			}
		}
		else = {
			add = {
				multiply = 0.25
				desc = has_admin_government
			}
		}
	}
}
