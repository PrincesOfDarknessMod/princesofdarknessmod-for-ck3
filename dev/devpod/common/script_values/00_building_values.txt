ai_building_absolute_priority = @[ai_building_tier_1_top * 100]

##########################################
# AI Domicile Building Priority Tiers    #
##########################################

# Domicile ai_values are scored against holding buildings. One domicile unit equals one base unit at the first level of tier 4, so 10 units match a base-10 resource district building upgrade.
ai_building_domicile_factor = @ai_building_tier_4_top
# Merchant company values are calibrated around 280 for a normal internal upgrade, and this maps 280 to 10 domicile units
ai_building_merchant_company_domicile_factor = @[ai_building_tier_4_top * 10 / 280]

# Scope: domicile, expects scope:owner
ai_building_domicile_owner_factor = {
	value = 1
	if = { # Some characters should want to build in their domicile before they construct buildings in holdings
		limit = {
			scope:owner = {
				OR = {
					has_trait = arbitrary
					has_trait = gluttonous
					has_trait = greedy
				}
			}
		}
		multiply = 1.5
	}
	if = { # Characters loyal to the realm build in their domicile one tier later
		limit = {
			scope:owner = {
				has_trait = loyal
				OR = {
					has_trait = diligent
					has_trait = just
				}
			}
		}
		multiply = 0.8
	}
}

# Estate, east Asian estate, Japanese manor and ecclesiastical domiciles: new buildings, then slot building upgrades, then main building upgrades.
# Bands stay one strict step apart, below the 0.8 cutoff, even at the nudge extremes
# The income factor stays inside that gap, so income is favoured but not exclusive
@ai_building_domicile_new_base = 22
@ai_building_domicile_upgrade_base = 10
@ai_building_domicile_main_upgrade_base = 6
@ai_building_domicile_income_factor = 1.2

# Squeezes a building's own ai_value (about 3-15, with level decay and bonuses) into an in-band weight of 0.85-1.15, 1 at a value of 7.
ai_building_domicile_nudge_slope = 0.03
ai_building_domicile_nudge_offset = 0.79
ai_building_domicile_nudge_min = 0.85
ai_building_domicile_nudge_max = 1.15

# Scope: domicile, expects scope:owner
ai_building_domicile_new_value = {
	value = @ai_building_domicile_new_base
	multiply = ai_building_domicile_owner_factor
	multiply = ai_building_domicile_factor
}

# Scope: domicile, expects scope:owner
ai_building_domicile_new_income_value = {
	value = @[ai_building_domicile_new_base * ai_building_domicile_income_factor]
	multiply = ai_building_domicile_owner_factor
	multiply = ai_building_domicile_factor
}

# Scope: domicile, expects scope:owner
ai_building_domicile_upgrade_value = {
	value = @ai_building_domicile_upgrade_base
	multiply = ai_building_domicile_owner_factor
	multiply = ai_building_domicile_factor
}

# Scope: domicile, expects scope:owner
ai_building_domicile_upgrade_income_value = {
	value = @[ai_building_domicile_upgrade_base * ai_building_domicile_income_factor]
	multiply = ai_building_domicile_owner_factor
	multiply = ai_building_domicile_factor
}

# Scope: domicile, expects scope:owner
ai_building_domicile_main_upgrade_value = {
	value = @ai_building_domicile_main_upgrade_base
	multiply = ai_building_domicile_owner_factor
	multiply = ai_building_domicile_factor
}

# Scope: domicile, expects scope:owner
# For buildings the AI should avoid. Their own ai_value (1 or lower) is kept, so they stay below main building upgrades
ai_building_domicile_deprioritised_value = {
	value = ai_building_domicile_owner_factor
	multiply = ai_building_domicile_factor
}

			limit = {
				county = {

			}
			add = 1
		}
	}
}
