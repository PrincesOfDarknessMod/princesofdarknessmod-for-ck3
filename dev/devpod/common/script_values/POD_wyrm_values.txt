maeljingifts_xp_gain_value = {
    value = 100
	
	add = {
		value = learning
		multiply = 20
    }
	if = {
		limit = { piety_level >= 2 }
		add = {
			value = piety_level
			multiply = 100
		}
	}
	add = {
		value = POD_fera_rank_value
		multiply = 100
	}

    if = {
		limit = { faith = { has_doctrine_parameter = reduced_maeljingifts_xp } }
		multiply = 0.75
	}
    else_if = {
		limit = { faith = { has_doctrine_parameter = increased_maeljingifts_xp } }
		multiply = 1.25
	}

	max = 1000
}
