fera_renown_wisdom_value = {
	value = 1

	if = {
		limit = {
			OR = {
				has_trait = theurge
				has_trait = galliard
				has_trait = setting_sun
				has_trait = shrouded_sun
				has_trait = eclipsed_sun
				has_trait = zarad
				has_trait = sai_chau
				has_trait = pei_tung
			}
		}
		add = 1
	}

	if = {
		limit = {
			POD_is_on_journey_trigger = { TYPE = fera_spirit_quest }
		}
		if = {
			limit = { fera_spirit_quest_difficulty = 5 }
			add = 2
		}
		else_if = {
			limit = { fera_spirit_quest_difficulty = 4 }
			add = 1
		}
		else_if = {
			limit = { fera_spirit_quest_difficulty = 1 }
			subtract = 1
		}
	}
}

#Do not touch these unless you really know what you're doing
POD_fera_rank_xp_threshold = 20
POD_max_fera_rank = 6

POD_fera_rank_xp = {
	value = 0

	add = "this.has_trait_xp(werewolf)"
	add = "this.has_trait_xp(ajaba)"
	add = "this.has_trait_xp(ananasi)"
	add = "this.has_trait_xp(apis)"
	add = "this.has_trait_xp(bastet)"
	add = "this.has_trait_xp(camazotz)"
	add = "this.has_trait_xp(corax)"
	add = "this.has_trait_xp(grondr)"
	add = "this.has_trait_xp(gurahl)"
	add = "this.has_trait_xp(kitsune)"
	add = "this.has_trait_xp(mokole)"
	add = "this.has_trait_xp(nagah)"
	add = "this.has_trait_xp(nuwisha)"
	add = "this.has_trait_xp(ratkin)"
	add = "this.has_trait_xp(rokea)"
}

# value from 1-6
POD_fera_rank_value = {
	value = POD_fera_rank_xp
	divide = POD_fera_rank_xp_threshold
	floor = yes
	add = 1
}

POD_next_fera_rank = {
	value = POD_fera_rank_value
	add = 1
	max = POD_max_fera_rank
}

POD_xp_to_next_fera_rank = {
	value = POD_fera_rank_xp_threshold
	subtract = {
		value = POD_fera_rank_xp
		modulo = POD_fera_rank_xp_threshold
	}
}

POD_xp_to_next_fera_rank_minus_1 = {
	value = POD_xp_to_next_fera_rank
	subtract = 1
}

POD_xp_requirement_for_next_fera_rank = {
	value = POD_fera_rank_xp
	add = POD_xp_to_next_fera_rank_minus_1
}

POD_spendable_fera_renown_value = {
	value = 0

	#Max rank spends 100 prestige per point
	if = {
		limit = { POD_fera_rank_value = 6 }
		add = prestige
		divide = 100
		floor = yes
	}
	
	#others spend trait xp
	else = {
		add = POD_fera_rank_xp

		if = {
			limit = { POD_fera_rank_value = 5 }
			subtract = POD_fera_rank_xp_threshold
			subtract = POD_fera_rank_xp_threshold
			subtract = POD_fera_rank_xp_threshold
			subtract = POD_fera_rank_xp_threshold
		}
		else_if = {
			limit = { POD_fera_rank_value = 4 }
			subtract = POD_fera_rank_xp_threshold
			subtract = POD_fera_rank_xp_threshold
			subtract = POD_fera_rank_xp_threshold
		}
		else_if = {
			limit = { POD_fera_rank_value = 3 }
			subtract = POD_fera_rank_xp_threshold
			subtract = POD_fera_rank_xp_threshold
		}
		else_if = {
			limit = { POD_fera_rank_value = 2 }
			subtract = POD_fera_rank_xp_threshold
		}
	}

	min = 0
}
