	value = 1

	if = {
		limit = { has_trait = ahroun }
		add = 1
	}
}

	value = 1

		limit = { has_trait = philodox }
		add = 1
	}
}

fera_renown_wisdom_value = {
	value = 1

	if = {
		limit = {
			OR = {
				has_trait = theurge
				has_trait = galliard
			}
		}
		add = 1
	}

	if = {
		limit = {
			POD_is_on_journey_trigger = { TYPE = fera_spirit_quest }
		}
		if = {
			limit = { fera_spirit_quest_difficulty = 5 }
			add = 2
		}
		else_if = {
			limit = { fera_spirit_quest_difficulty = 4 }
			add = 1
		}
		else_if = {
			limit = { fera_spirit_quest_difficulty = 1 }
			subtract = 1
		}
	}
}
