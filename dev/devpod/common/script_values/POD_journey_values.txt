seek_knowledge_progress = {
	add = seek_knowledge_progress_not_current
	
	multiply = {
		value = 1
		desc = POD_journey_progress_seek_knowledge
		add = {
			value = 0
			var:coterie ?= {
				every_in_list = {
					variable = members
					limit = {
						POD_seek_knowledge_coterie_member_trigger = {
							VAR = root.var:current_journey.var:journey_type
						}
					}
					add = seek_knowledge_coterie_bonus
				}
			}
		}
	}
}
seek_knowledge_progress_not_current = {
	add = {
		value = seek_knowledge_base_progress
		desc = POD_journey_progress_base
		format = "BASE_VALUE_FORMAT_DECIMALS_MINUS_POSITIVE"
	}
	
	if = {
		limit = { NOT = { employs_court_position = court_tutor_court_position } }
		subtract = {
			value = seek_knowledge_base_progress
			multiply = 0.05
			desc = POD_journey_progress_court_tutor_none
			format = "POD_PROGRESS_FORMAT_NEGATIVE"
		}
	}
	else_if = {
		limit = {
			any_court_position_holder = {
				type = court_tutor_court_position
				aptitude = {
					court_position = court_tutor_court_position
					value = 2
				}
			}
		}
		add = {
			value = seek_knowledge_base_progress
			multiply = 0.05
			desc = POD_journey_progress_court_tutor_2
			format = "POD_PROGRESS_FORMAT_POSITIVE"
		}
	}
	else_if = {
		limit = {
			any_court_position_holder = {
				type = court_tutor_court_position
				aptitude = {
					court_position = court_tutor_court_position
					value = 3
				}
			}
		}
		add = {
			value = seek_knowledge_base_progress
			multiply = 0.1
			desc = POD_journey_progress_court_tutor_3
			format = "POD_PROGRESS_FORMAT_POSITIVE"
		}
	}
	else_if = {
		limit = {
			any_court_position_holder = {
				type = court_tutor_court_position
				aptitude = {
					court_position = court_tutor_court_position
					value = 4
				}
			}
		}
		add = {
			value = seek_knowledge_base_progress
			multiply = 0.15
			desc = POD_journey_progress_court_tutor_4
			format = "POD_PROGRESS_FORMAT_POSITIVE"
		}
	}
	else_if = {
		limit = {
			any_court_position_holder = {
				type = court_tutor_court_position
				aptitude = {
					court_position = court_tutor_court_position
					value = 5
				}
			}
		}
		add = {
			value = seek_knowledge_base_progress
			multiply = 0.2
			desc = POD_journey_progress_court_tutor_5
			format = "POD_PROGRESS_FORMAT_POSITIVE"
		}
	}
	
	if = {
		limit = { POD_has_coterie_members_trigger = yes }
		add = {
			value = {
				value = coterie_learning_gui
				max = coterie_skill_hardcap
				divide = 100
				multiply = seek_knowledge_base_progress
			}
			desc = POD_journey_progress_learning_coterie
			format = "POD_PROGRESS_FORMAT_POSITIVE"
		}
	}
	else = {
		add = {
			value = {
				value = learning
				max = coterie_skill_hardcap
				divide = 100
				multiply = seek_knowledge_base_progress
			}
			desc = POD_journey_progress_learning
			format = "POD_PROGRESS_FORMAT_POSITIVE"
		}
	}
}
