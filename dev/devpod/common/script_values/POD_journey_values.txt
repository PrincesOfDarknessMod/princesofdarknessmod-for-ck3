journey_cost_mult = {
	if = {
		limit = { has_game_rule = journey_cost_mult_0 }
		add = 0
	}
	else_if = {
		limit = { has_game_rule = journey_cost_mult_05 }
		add = 0.5
	}
	else_if = {
		limit = { has_game_rule = journey_cost_mult_2 }
		add = 2
	}
	else = {
		add = 1
	}
}
