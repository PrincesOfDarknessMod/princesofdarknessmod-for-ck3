POD_necromagic_success_chance = {
	value = 20
	add = {
		value = learning
		divide = 4
	}
	if = { limit = { NOT = { scope:target = THIS } }
		if = { limit = { has_trait = vampire scope:target = { has_trait = vampire } } # Generation Difference if both vampires
			add = { # Generation Difference 
				value = generation_value
				subtract = scope:target.generation_value
				multiply = 10
			}
		}
		
		subtract = {
			value = scope:target.learning
			divide = 3
		}
		
		if = { limit = { scope:target = { has_character_modifier = ritual_warding_baptism_modifier } } subtract = 25 }
		if = { limit = { scope:target = { has_perk = one_with_dead_1_perk } } subtract = 10 }
		if = { limit = { scope:target = { has_perk = one_with_dead_2_perk } } subtract = 10 }
		if = { limit = { scope:target = { has_perk = one_with_dead_3_perk } } subtract = 10 }
		if = { limit = { scope:target = { has_perk = one_with_dead_4_perk } } subtract = 10 }
		if = { limit = { scope:target = { has_perk = wisdom_of_isis_perk } } subtract = 50 }

		if = { limit = { scope:target = { has_character_modifier = ritual_tempesta_scudo_modifier } } subtract = 100 }
		if = { limit = { scope:target = { has_character_modifier = ritual_circle_of_cerberus_modifier } } subtract = 100 }
	}
	if = { limit = { has_perk = one_with_dead_1_perk } add = 15 }
	if = { limit = { has_perk = one_with_dead_2_perk } add = 15 }
	if = { limit = { has_perk = one_with_dead_3_perk } add = 15 }
	if = { limit = { has_perk = one_with_dead_4_perk } add = 15 }

	if = { limit = { has_character_modifier = ritual_deaths_communion_modifier } add = 15 }

	# Hideout Upgrades — Cappadocian Ossuary
	if = {
		limit = { domicile ?= { has_domicile_building_or_higher = vampire_cappadocian_ossuary_06 } }
		add = 15
	}
	else_if = {
		limit = { domicile ?= { has_domicile_building_or_higher = vampire_cappadocian_ossuary_05 } }
		add = 10
	}
	else_if = {
		limit = { domicile ?= { has_domicile_building_or_higher = vampire_cappadocian_ossuary_04 } }
		add = 5
	}

	min = 0
	max = 100
}
