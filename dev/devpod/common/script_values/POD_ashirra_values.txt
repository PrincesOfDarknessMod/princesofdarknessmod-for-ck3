willing_embrace_success_value = {
	value = 0

	#path to power
	if = {
		limit = { scope:target = { has_trait = ambitious } }
		add = 15
	}

	#life was good
	if = {
		limit = { scope:target = { has_trait = content } }
		add = -10
	}

	#old
	if = {
		limit = { scope:target = { age >= 55 NOT = { has_trait = content } } }
		add = 5
	}

	#injured
	if = {
		limit = { scope:target = { OR = { has_trait = wounded_3 has_trait = maimed } } }
		add = 10
	}
	else_if = {
		limit = { scope:target = { has_trait = wounded_2 } }
		add = 7
	}
	else_if = {
		limit = { scope:target = { has_trait = wounded_1 } }
		add = 4
	}

	#diseased, mortal trait to exclude revenants
	if = {
		limit = { scope:target = { has_trait = mortal has_any_mortal_disease_trigger = yes } }
		add = 10
	}

	#doesn't believe you
	if = {
		limit = { scope:target = { has_trait = cynical } }
		add = -10
	}
}

willing_embrace_great_success_value = {
	value = willing_embrace_success_value
	multiply = 0.5
}











ashirra_influence_value = {
	value = 0
	add = {
		value = ashirra_county_value
		desc = ashirra_county_mod
	}
}

ashirra_county_value = {
	value = 0
	every_county = {
		limit = { religion = religion:ashirra_religion }
		add = 1
	}
}

ashirra_zeal = {
	value = var:ashirra_influence
	if = {
		limit = { scope:player = { NOT = { has_title = title:e_ashirra } } }
		divide = 10
		floor = yes
	}
}
