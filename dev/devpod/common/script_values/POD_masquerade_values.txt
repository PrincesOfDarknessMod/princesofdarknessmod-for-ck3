masquerade_exposure_level = {
	value = 0
	if = {
		limit = { has_character_modifier = masquerade5_modifier }
		add = 5
	}
	else_if = {
		limit = { has_character_modifier = masquerade4_modifier }
		add = 4
	}
	else_if = {
		limit = { has_character_modifier = masquerade3_modifier }
		add = 3
	}
	else_if = {
		limit = { has_character_modifier = masquerade2_modifier }
		add = 2
	}
	else_if = {
		limit = { has_character_modifier = masquerade1_modifier }
		add = 1
	}
	else_if = {
		limit = { has_character_modifier = masquerade0_modifier }
		add = 0
	}
}

minimum_masquerade_exposure = {
	if = {
		limit = {
			OR = {
				has_trait = exposed
				POD_is_forever_infamous_trigger = yes
			}
		}
		add = 5
	}
	else_if = {
		limit = {
			OR = {
				is_predatortype_cultist_trigger = yes
				is_predatortype_familias_trigger = yes
				is_predatortype_consensualist_trigger = yes
			}
			POD_has_infinite_blood_trigger = no
		}
		add = 1
	}
	else = {
		add = 0
	}
}

masquerade_positive_factors_mult = {
	value = 1
	add   = masquerade_positive_factors_delta
}

masquerade_all_factors_mult = {
	value = 1
	add   = masquerade_positive_factors_delta
	add   = masquerade_negative_factors_delta
}

masquerade_positive_factors_delta = {
	value = 0
	if = {
		limit = {
			OR = {
				has_trait = dominateadvanced
				has_perk = cloud_memory_perk
			}
		}
		subtract = 0.2
	}
	if = {
		limit = {
			OR = {
				has_trait = dominateadvanced
				has_perk = the_forgetful_mind_perk
			}
		}
		subtract = 0.4
	}
	if = {
		limit = {
			OR = {
				AND = {
					has_trait = necromancyadvanced
					has_cappadocian_necromancy_trigger = yes
				}
				has_perk = ns_ashes_to_ashes_perk
			}
		}
		subtract = 0.2
	}
	if = {
		limit = {
			OR = {
				AND = {
					has_trait = necromancyadvanced
					has_lamia_necromancy_trigger = yes
				}
				has_perk = ns_tend_the_body_garden_perk
			}
		}
		subtract = 0.2
	}
	if = {
		limit = {
			has_sihr_bloodmagic_trigger = yes
			has_character_modifier = mask_of_allah_modifier
		}
		subtract = 0.2
	}
}

masquerade_negative_factors_delta = {
	value = 0
	if = {
		limit = { faith = { has_doctrine = tenet_continued_life } }
		add = 0.2
	}
	if = {
		limit = {
			has_character_modifier = POD_gangrel_frenzy_modifier_light
		}
		add = 0.2
	}
	if = {
		limit = {
			has_character_modifier = POD_gangrel_frenzy_modifier_medium
		}
		add = 0.4
	}
	if = {
		limit = {
			has_character_modifier = POD_gangrel_frenzy_modifier_heavy
		}
		add = 0.6
	}
}
