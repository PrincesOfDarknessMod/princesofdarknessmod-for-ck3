macro = {
	description = "Returns the description of a court position aptitude level (e.g. Excellent, Good, Average) corresponding to a numerical level (CFixedPoint)"
	definition = "GetAptitudeString(level)"
	replace_with = "SelectLocalization( GreaterThanOrEqualTo_CFixedPoint( level, '(CFixedPoint)4' ), 'skill_level_excellent', SelectLocalization( EqualTo_CFixedPoint( level, '(CFixedPoint)3' ), 'skill_level_good', SelectLocalization( EqualTo_CFixedPoint( level, '(CFixedPoint)2' ), 'skill_level_average', SelectLocalization( EqualTo_CFixedPoint( level, '(CFixedPoint)1' ), 'skill_level_poor', 'skill_level_terrible' ) ) ) )"
}
