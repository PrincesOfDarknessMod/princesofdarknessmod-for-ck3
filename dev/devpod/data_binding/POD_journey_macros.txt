}

macro = {
	description = "Base macro for checking how many years this Journey has left to go (s is the planning speed script value, p is the current progress)"
	definition = "JourneyYearsLeftBase(s,p)"
	replace_with = "GuiScope.SetRoot(GetPlayer.MakeScope).AddScope('total_progress',MakeScopeValue(p)).AddScope('monthly',MakeScopeValue(GetPlayer.MakeScope.ScriptValue(s))).ScriptValue('journey_progress_years')"
}

macro = {
	description = "How many years does this Journey have left to go if starting from 0? (need to pass the planning speed script value as s)"
	definition = "JourneyYearsLeft(s)"
	replace_with = "JourneyYearsLeftBase(s,'(CFixedPoint)0')"
}
