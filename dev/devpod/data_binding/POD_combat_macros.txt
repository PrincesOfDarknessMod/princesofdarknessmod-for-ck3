# Data-binding macros for the combat system. Move-tooltip shortcuts below
# (the PODSeg* segment macros are documented at their own banner):
#
#   PODCombatMoveEffect(scope) - dynamic stat block rendered by the
#                                pod_combat_move_desc SGUI's
#                                BuildTooltip. Takes a story_cycle
#                                SCOPE, so you must already have the
#                                story in hand. The SGUI declares
#                                `saved_scopes = { story }` and chains
#                                custom_tooltip = pod_combat_desc_X
#                                lines under has_variable guards;
#                                BuildTooltip carries the GuiScope's
#                                'story' saved scope into the effect
#                                evaluation so `[story.MakeScope.Var
#                                ('Y')]` data binding inside each
#                                pod_combat_desc_X loc string resolves
#                                against the move.
#
#   PODCombatMoveEffectStatic(scope) - the same card with clause hiding
#                                switched off, for surfaces that describe
#                                somebody else's action.
#
#   PODGetCombatMoveFromKey(k) - helper that resolves a decision id
#                                string to its story_cycle scope by
#                                looking up the named variable on
#                                Charon (where POD_init_combat_move
#                                stashes each story by decision name).
#                                Chained into PODCombatMoveDesc below.
#
#   PODCombatMoveDesc(key) - rich three-section inline tooltip
#                                (title + AP, dynamic stats, flavor).
#                                Used by decision _desc loc entries so
#                                hover tooltips and link pops both get
#                                the same full card.

macro = {
	description = "Dynamic stat block for a move. Takes a story_cycle scope. Renders via the pod_combat_move_desc SGUI's BuildTooltip."
	definition   = "PODCombatMoveEffect(combatmove)"
	# SetRoot is load-bearing even though every branch scope-switches to
	# scope:story: the SGUI declares scope = character, and the card's
	# viewer-relative lines (which weapon YOU are holding) read the root
	# directly. Building a bare GuiScope here left it unset, which is why
	# this is the one BuildTooltip call in the tree without a SetRoot.
	replace_with = "GetScriptedGui('pod_combat_move_desc').BuildTooltip( GuiScope.SetRoot( GetPlayer.MakeScope ).AddScope( 'story', combatmove ).End )"
}

macro = {
	description = "Move stat block with conditional-bonus hiding disabled. Takes a story_cycle scope."
	definition   = "PODCombatMoveEffectStatic(combatmove)"
	# PODCombatMoveEffect plus a bound `nohide` scope, which is the card's signal
	# to render every bonus clause including the ones flagged hide_if_unmet. The
	# hide test is viewer-relative, so on a combat-log row - which narrates
	# whoever acted, not the reader - an enemy's met hidden clause would be
	# missing from the card while the row's own segment prints the amount it
	# dealt. The scope is read only through exists = scope:nohide, so what it
	# points at carries no meaning; GetPlayer is simply always resolvable.
	replace_with = "GetScriptedGui('pod_combat_move_desc').BuildTooltip( GuiScope.SetRoot( GetPlayer.MakeScope ).AddScope( 'story', combatmove ).AddScope( 'nohide', GetPlayer.MakeScope ).End )"
}

macro = {
	description = "Resolve a decision key string to its story_cycle scope via the named variable on Charon."
	definition   = "PODGetCombatMoveFromKey(combatmove)"
	replace_with = "GetScriptedCharacterByHistoryID('POD_wraith_01').MakeScope.Var(combatmove).Story.MakeScope"
}

macro = {
	# Resolves the per-move POD_combat_move_<skill_key>_tooltip loc
	# dynamically by reading var:skill_key off the move story and
	# round-tripping it through GetFlagKey (returns the flag's name as
	# a string), then Concatenate3 builds the loc key and Localize
	# fetches the styled-italic flavor entry the generator emits.
	#
	# Bypasses the auto-loc <id>_tooltip lookup (Decision.GetCommandDescForPlayer
	# returns empty for hidden decisions). Used by the static loc
	# POD_combat_move_tooltip and POD_combat_log_move_tooltip,
	# wrapped in [AddLine(...)] so a missing / empty entry renders
	# nothing.
	description = "Flavor sentence for a move. Takes a story_cycle scope."
	definition   = "PODCombatMoveFlavor(combatmove)"
	replace_with = "Localize( Concatenate3( 'POD_combat_move_', combatmove.Var('skill_key').GetFlagKey, '_tooltip' ) )"
}

macro = {
	description = "Rich inline tooltip (title + stats + flavor). Takes a decision-id string."
	definition   = "PODCombatMoveDesc(combatmove)"
	replace_with = "GuiScope.SetRoot(GetPlayer.MakeScope).AddScope( 'story', PODGetCombatMoveFromKey(combatmove) ).Custom('PODCombatMoveInlineTooltip')"
}

macro = {

}

########## WEAPON OVERRIDE TEXT ##########
# What an equipped weapon does to a `uses_weapon` combat move, generated from
# the artifact's own override vars in one standardised voice, one line per
# override. The macro below binds the weapon as the scope `pod_weapon`, the
# convention the override dispatchers read (see
# POD_weapon_override_custom_loc.txt); the move card binds the same scope
# itself with `save_scope_as = pod_weapon` in pod_combat_move_guis.txt.
#
# ARTIFACT DESIGNERS stamp the override var in the artifact's creation effect
# and are done - the two [Artifact.GetDesc] surfaces (gui/shared/cooltip.gui,
# gui/window_artifact_details.gui) render pod_artifact_desc_and_weapon_override,
# which appends the Self variant below to every artifact automatically. It
# renders nothing when the artifact carries no override, so the unconditional
# append is safe.
#
# The Self variant reads the GUI's Artifact data context, so it resolves ONLY
# at widget level. Never place it inside a description loc key: GetDesc
# localizes the key in a fresh environment with no Artifact context (ROOT
# there is a character), so the reference errors and renders blank.

macro = {
	description = "Header + override block for the Artifact in GUI data context. Widget level only - never inside a description key."
	definition   = "PODWeaponOverrideSelfText"
	replace_with = "GuiScope.SetRoot( GetPlayer.MakeScope ).AddScope( 'pod_weapon', Artifact.MakeScope ).Custom( 'PODWeaponOverrideArtifactBlock' )"
}

# ── Combat-log segment contract ──
# Every result-column segment is a separator+content widget pair built
# from these three macros; the SV names follow
# pod_combat_log_seg_<name>_visible_value / pod_combat_log_before_<name>_value.

macro = {
	description = "Segment content visibility: 1-check on the segment's visible script value (entry-story scope)."
	definition   = "PODSegVisible(sv)"
	replace_with = "GreaterThan_CFixedPoint(Scope.ScriptValue(sv), '(CFixedPoint)0')"
}

macro = {
	description = "Segment separator visibility: this segment visible AND any content before it."
	definition   = "PODSegSepVisible(segsv,beforesv)"
	replace_with = "And(GreaterThan_CFixedPoint(Scope.ScriptValue(segsv), '(CFixedPoint)0'), GreaterThan_CFixedPoint(Scope.ScriptValue(beforesv), '(CFixedPoint)0'))"
}

macro = {
	description = "Segment renderer: the standard entry+move scope chain into a customizable_localization."
	definition   = "PODSegText(customloc)"
	replace_with = "GuiScope.SetRoot(GetPlayer.MakeScope).AddScope('entry', Scope.Story.MakeScope).AddScope('move', Scope.Var('move').Story.MakeScope).Custom(customloc)"
}

	definition   = "PODCharHasNoHumanBody"

}

macro = {
	# The fight the viewing player is in, as a scope: their own character's
	# pod_combat_session pointer, dereferenced to the POD_combat_fight story.
	# This is how the GUI reaches fight state - never by reading fight vars off
	# GetPlayer directly, and never by hopping through some other character.
	# Going via the viewer's own character is what lets a second client bind
	# the same fight later: GetPlayer resolves per client, so each one arrives
	# at the shared story from its own end.
	#
	# CONTRACT: this chain is not self-guarding, and a GUI raw_text is
	# evaluated even inside a `visible = no` subtree (COMBAT_REFERENCE.info
	# §7). Every use must sit under PODCombatFightIsSet - its own `visible`,
	# or an ancestor's - or terminate in a tolerant accessor (|0, GetFlagKey).
	description = "The viewing player's current fight (POD_combat_fight story) as a scope. Guard every use with PODCombatFightIsSet."
	definition   = "PODCombatFight"
	replace_with = "GetPlayer.MakeScope.Var('pod_combat_session').Story.MakeScope"
}

macro = {
	description = "True when the viewing player is in a fight, i.e. when PODCombatFight resolves."
	definition   = "PODCombatFightIsSet"
	replace_with = "GetPlayer.MakeScope.Var('pod_combat_session').IsSet"
}

macro = {
	# The COMBATANT this client is driving, which is not necessarily this client's
	# own character. Reached through pod_combat_my_control_slot, a var on the
	# VIEWER'S OWN character - re-pointed at every hand-off and deliberately not
	# cleared between turns, so it still names your combatant while an AI is acting.
	#
	# Per-viewer by construction. The fight-side twin pod_combat_active_control_slot
	# can only name one slot, so with two humans in one fight it would hand both
	# clients whichever of them was handed off most recently.
	#
	# Use this, NOT GetPlayer, for anything about the character being played:
	# their loadout, their unlocks, whose move this is. GetPlayer stays correct
	# for anything about the VIEWER - which window this is, who is clicking.
	# Those two coincide only while the one controlled slot is the host's own.
	#
	# Deliberately NOT the workspace actor (pod_combat_actor_char): that follows
	# whoever is acting right now, so a move bar rooted on it would show the
	# enemy's repertoire during their turn. The datamodels are not turn-gated.
	#
	# CONTRACT: same as PODCombatFight - not self-guarding. Guard with
	# PODCombatMyCombatantIsSet, an ancestor's PODCombatFightIsSet, or a tolerant
	# terminator.
	description = "The combatant the viewing player is driving, as a scope. Guard with PODCombatMyCombatantIsSet."
	definition   = "PODCombatMyCombatant"
	replace_with = "GetPlayer.MakeScope.Var('pod_combat_my_control_slot').Story.MakeScope.Var('slot_char').GetCharacter.MakeScope"
}

macro = {
	description = "True when the viewing player is driving a combatant in a live fight."
	definition   = "PODCombatMyCombatantIsSet"
	replace_with = "GetPlayer.MakeScope.Var('pod_combat_my_control_slot').IsSet"
}
