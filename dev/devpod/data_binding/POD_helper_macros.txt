macro = {
	description = "Applies the current GUI scaling to an unscaled value. Returns a float."
	definition = "ApplyGUIScaling(A0)"
	replace_with = "Multiply_float(GetInvParentScale(PdxGuiWidget.Self), A0)]"
}

macro = {
	description = "Returns the current scaling factor as a CVector2f. Requires the parent to use the POD_scaling_helper_widget template."
	definition = "GetScalingVector"
	replace_with = "Divide_CVector2f('(CVector2f)1,1', PdxGetWidgetScreenSize(PdxGuiWidget.FindChild('scaling_helper').Self))"
}

macro = {
	description = "Returns the int32 as a string."
	definition = "GetString_int32(Arg0)"
	replace_with = "GuiScope.AddScope( 'number', MakeScopeValue(IntToFixedPoint(Arg0))).Custom('PODDisplayNumber')"
}
