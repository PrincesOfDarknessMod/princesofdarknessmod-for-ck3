# data_binding macros can be used to create "shortcuts" for loc/GUI functions
# that you use frequently.
# this one calls the SGUI that auto-generates the descriptions for duel moves.

macro = {
	description = "Generates a link with the name of a duel move and a tooltip"
	definition = "PODDuelMove(duelmove)"
	replace_with = "GuiScope.SetRoot(GetPlayer.MakeScope).AddScope( 'flag', MakeScopeFlag(duelmove) ).Custom('PODDuelMoveLink')"
}

macro = {
	description = "Generates a description for all the effects of a duel move. The duelmove argument needs to be a scope (the story_cycle defining the duel move)"
	definition = "PODDuelMoveEffect(duelmove)"
	replace_with = "GetScriptedGui('POD_duel_move_desc').BuildTooltip( GuiScope.AddScope( 'story', duelmove ).End )"
}

# this is why we saved the duel moves as variables on charon:
# so that we can access them globally
macro = {
	description = "Receives a duel move's key (the database key of its decision) and returns a story_cycle scope that can be used by PODDuelMoveEffect"
	definition = "PODGetDuelMoveFromKey(duelmove)"
	replace_with = "GetScriptedCharacterByHistoryID('POD_wraith_01').MakeScope.Var(duelmove).Story.MakeScope"
}

macro = {
	description = "Generates a description for all the effects of a duel move. Used for inline tooltips"
	definition = "PODDuelMoveDesc(duelmove)"
	replace_with = "GuiScope.SetRoot(GetPlayer.MakeScope).AddScope( 'story', PODGetDuelMoveFromKey(duelmove) ).Custom('PODDuelMoveInlineTooltip')"
}
