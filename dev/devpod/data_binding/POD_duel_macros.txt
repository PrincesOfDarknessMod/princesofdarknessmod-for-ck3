# data_binding macros can be used to create "shortcuts" for loc/GUI functions
# that you use frequently.
# this one calls the SGUI that auto-generates the descriptions for duel moves.

macro = {
	description = "Generates a link with the name of a duel move and a tooltip"
	definition = "PODDuelMove(duelmove)"
	replace_with = "GuiScope.SetRoot(GetPlayer.MakeScope).AddScope( 'flag', MakeScopeFlag(duelmove) ).Custom('PODDuelMoveLoc')"
}

macro = {
	description = "Generates a description for all the effects of a duel move"
	definition = "PODDuelMoveDesc(duelmove)"
	replace_with = "GetScriptedGui('POD_duel_move_desc').BuildTooltip( GuiScope.AddScope( 'story', GetScriptedCharacterByHistoryID('POD_wraith_01').MakeScope.Var(duelmove).Story.MakeScope ).End )"
}
