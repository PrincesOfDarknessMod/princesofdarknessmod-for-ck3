macro = {
	description = "Return the default height of the Umbra map display in pixels"
	definition = "UmbraMapHeight"
	replace_with = "'(float)800'"
}

macro = {
	description = "Return the default width of the Umbra map display in pixels"
	definition = "UmbraMapWidth"
	replace_with = "'(float)600'"
}

macro = {
	description = "How off-center are the Umbra Node icons? (half of icon size)"
	definition = "UmbraNodeIconOffset"
	replace_with = "'(float)25'"
}

macro = {
	description = "Return the rows as a float"
	definition = "UmbraMapRows"
	replace_with = "FixedPointToFloat(GetPlayer.MakeScope.Var('umbra_map').Var('rows').GetValue)"
}

macro = {
	description = "Return the columns as a float"
	definition = "UmbraMapColumns"
	replace_with = "FixedPointToFloat(GetPlayer.MakeScope.Var('umbra_map').Var('columns').GetValue)"
}

macro = {
	description = "X scaling factor for Umbra maps"
	definition = "UmbraMapXScaling"
	replace_with = "Divide_float(UmbraMapWidth, UmbraMapColumns)"
}

macro = {
	description = "X offset for Umbra maps"
	definition = "UmbraMapXOffset"
	# (map_width / columns) / 2 = (map_width / (columns*2))
	# push items into the middle
	replace_with = "Divide_float(UmbraMapWidth, Multiply_float(UmbraMapColumns, '(float)2'))"
}

macro = {
	description = "X offset for a cell. This is centering and accounts for icon size."
	definition = "UmbraXOffset"
	replace_with = "Subtract_float(UmbraMapXOffset, UmbraNodeIconOffset)"
}

macro = {
	description = "Returns the X coordinate scaled and offset"
	definition = "UmbraXPosition(XCoord)"
	replace_with = "Add_float(Multiply_float(Subtract_float(UmbraMapColumns, Add_float(XCoord, '(float)1')), UmbraMapXScaling), UmbraXOffset)"
}

macro = {
	description = "Y scaling factor for Umbra maps"
	definition = "UmbraMapYScaling"
	replace_with = "Divide_float(UmbraMapHeight, UmbraMapRows)"
}

macro = {
	description = "Y offset for Umbra maps"
	definition = "UmbraMapYOffset"
	# (map_height / rows) / 2 = (map_height / (rows*2))
	# push items into the middle
	replace_with = "Divide_float(UmbraMapHeight, Multiply_float(UmbraMapRows, '(float)2'))"
}

macro = {
	description = "Total Y offset for a cell. This is centering and accounts for icon size."
	definition = "UmbraYOffset"
	replace_with = "Subtract_float(UmbraMapYOffset, UmbraNodeIconOffset)"
}

macro = {
	description = "Returns the Y coordinate scaled and offset"
	definition = "UmbraYPosition(YCoord)"
	replace_with = "Add_float(Multiply_float(Subtract_float(UmbraMapRows, Add_float(YCoord, '(float)1')), UmbraMapYScaling), UmbraYOffset)"
}
