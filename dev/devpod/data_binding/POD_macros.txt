macro = {
	description = "Return true if nothing blocks visibility of sidebar windows (e.g. switching characters)"
	definition = "IsSidebarVisibilityValid"
	replace_with = "And( Not( IsPauseMenuShown ), And(Or(Not(IsObserver), GetPlayer.IsValid), IsDefaultGUIMode))"
}
