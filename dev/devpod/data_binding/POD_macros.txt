macro = {
	description = "Return true if nothing blocks visibility of sidebar windows (e.g. switching characters)"
	definition = "IsSidebarVisibilityValid"
	replace_with = "And( Not( IsPauseMenuShown ), And(Or(Not(IsObserver), GetPlayer.IsValid), IsDefaultGUIMode))"
}

macro = {
	description = "Returns an artifact that was saved as a variable on Charon"
	definition = "GetPODArtifact(link)"
	replace_with = "GetScriptedCharacterByHistoryID('POD_wraith_01').MakeScope.Var(link).Artifact"
}

macro = {
	description = "Returns an artifact's name that was saved as a variable on Charon. If it doesn't exist, return a fallback string"
	definition = "GetPODArtifactNameWithFallback(link,fallback)"
	replace_with = "Select_CString( GetScriptedCharacterByHistoryID('POD_wraith_01').MakeScope.Var(link).IsSet, GetScriptedCharacterByHistoryID('POD_wraith_01').MakeScope.Var(link).Artifact.GetName, fallback )"
}

macro = {
	description = "Returns true if the DLC in this datacontext is PoD"
	definition = "DLCIsPOD"

}

macro = {
	description = "Returns true if a DLC is enabled. Makes sure the PoD 'DLC' always shows up as enabled even though it technically isn't"
	definition = "PODDLCIsEnabled"
	replace_with = "Or(Dlc.IsEnabled,DLCIsPOD)"
}
