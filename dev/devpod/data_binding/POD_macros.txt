macro = {
	description = "Return true if one of the POD sidebar windows is open (currently: Journeys, Camarilla)"
	definition = "IsPODSidebarOpen"
	replace_with = "Or(GetVariableSystem.Exists('journey_window_open'), GetVariableSystem.Exists('camarilla_window_open'))"
}

macro = {
	description = "Return true if nothing blocks visibility of sidebar windows (e.g. switching characters)"
	definition = "IsSidebarVisibilityValid"
	replace_with = "And(IsInGame, And( And(Or(Not(IsObserver), GetPlayer.IsValid), IsDefaultGUIMode), Not(IsPauseMenuShown) ) )"
}
