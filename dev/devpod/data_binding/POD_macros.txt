macro = {
	description = "Return true if nothing blocks visibility of sidebar windows (e.g. switching characters)"
	definition = "IsSidebarVisibilityValid"
	replace_with = "And( Not( IsPauseMenuShown ), And(Or(Not(IsObserver), GetPlayer.IsValid), IsDefaultGUIMode))"
}

macro = {
	description = "Return true if the Scripted GUI 'text' is shown, passing the scoped Trait as the root"
	definition = "TraitGuiIsShown(text)"
	replace_with = "GetScriptedGui(text).IsShown(GuiScope.SetRoot(Trait.MakeScope).End)"
}

macro = {
	description = "Return true if the player has selected this trait in the Ruler Designer"
	definition = "HasRDTrait(text)"
	replace_with = "RulerDesignerWindow.HasTrait(GetTrait(text).Self)"
}
