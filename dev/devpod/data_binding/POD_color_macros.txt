macro = {
	description = "For formatting numeric values that interpolate between green -> red as the value goes down. v is the current value and m is the maximum"
	definition = "PODValueGreen(v,m)"
	replace_with = "GuiScope.AddScope('value',MakeScopeValue(v)).AddScope('max',MakeScopeValue(m)).ScriptValue('POD_value_green')"
}

macro = {
	description = "Convert RGB to HSV (return hue)"
	definition = "RGB2H(r,g,b)"
	replace_with = "GuiScope.AddScope('red',r).AddScope('green',g).AddScope('blue',b).ScriptValue('POD_color_rgb2h')"
}

macro = {
	description = "Get a clothing color from the current root scope"
	definition = "PODClothingColor(channel,color)"
	replace_with = "color_root.MakeScope.Var(Concatenate(Concatenate('POD_portrait_clothing_',channel),Concatenate('_',color))).GetValue"
}

macro = {
	description = "If the current root scope has no clothing colors set, return 0. Otherwise, return 1"
	definition = "PODClothingColorAlpha(channel)"
	replace_with = "color_root.MakeScope.ScriptValue(Concatenate('POD_color_channel_alpha_',channel))"
}

macro = {
	description = "Currently selected tab in the Clothing Color GUI"
	definition = "PODClothingColorTab"
	replace_with = "GetVariableSystem.Get('POD_clothing_color_tab')"
}
