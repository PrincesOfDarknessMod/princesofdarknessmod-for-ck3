chinese = {
	trigger = {
		always = no # disabled pending a nighttime version
		has_tgp_dlc_trigger = yes
		OR = {
			capital_county ?= {
				OR = {
					culture = { has_graphical_chinese_culture_group_trigger = yes }
					# Also used for East Asian throne rooms
					title_province = { 
						geographical_region = world_asia
						NOT = { geographical_region = world_asia_japan }
						NOT = { geographical_region = world_asia_southeast }
					}
				}
			}
		}
		POD_can_use_non_CFP_throne_rooms = yes
		POD_uses_steppe_throne_room = no
	}
}

southeast_asia = {
	trigger = {
		has_tgp_dlc_trigger = yes
		NOT = {
			government_has_flag = government_is_tribal
		}
		capital_county ?= {
			OR = {
				title_province = { geographical_region = world_asia_southeast }
				culture = { has_building_gfx = southeast_asian_building_gfx }
			}
		}
		POD_can_use_non_CFP_throne_rooms = yes
		POD_uses_steppe_throne_room = no
	}
}

mediterranean = {
	trigger = {
		exists = capital_county
		capital_county = {
			OR = {
				culture = { has_graphical_mediterranean_culture_group_trigger = yes }
				AND = {
					culture = { has_graphical_iberian_culture_group_trigger = yes }
					faith = { is_POD_muslim_religion_trigger = no }
				}
				title_province = { steppe_building_gfx_use_mediterranean_trigger = yes }
			}
		}
		POD_can_use_non_CFP_throne_rooms = yes
		POD_uses_steppe_throne_room = no
	}
}

mena = {
	trigger = {
		exists = capital_county
		capital_county = {
			OR = {
				culture = {
					OR = {
						has_graphical_mena_culture_group_trigger = yes
						has_graphical_iranian_culture_group_trigger = yes
						has_graphical_african_culture_group_trigger = yes
						AND = {
							has_graphical_iberian_culture_group_trigger  = yes
							prev.faith = { is_POD_muslim_religion_trigger = yes }
						}
					}
				}
				AND = {
					culture = {
						has_graphical_steppe_culture_group_trigger = yes
					}
					title_province = {
						steppe_building_gfx_use_mediterranean_trigger = no
						steppe_building_gfx_use_indian_trigger = no
						steppe_building_gfx_use_western_trigger = no
					}
				}
			}
		}
		POD_can_use_non_CFP_throne_rooms = yes
		POD_uses_steppe_throne_room = no
	}
}
