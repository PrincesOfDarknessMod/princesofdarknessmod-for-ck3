
mediterranean = {
	trigger = {
		exists = capital_county
		OR = {
			capital_county.culture = {
				has_graphical_mediterranean_culture_group_trigger = yes
			}
			AND = {
				POD_CFP_has_graphical_iberian_culture_group_trigger = yes
				NOT = { exists = global_var:cfp_is_loaded }
			}
		}
		NOR = {
			faith.religion = religion:ashirra_religion
			faith.religion = religion:roadofblood_religion 
			POD_CFP_has_graphical_hre_culture_group_trigger = yes
			POD_CFP_has_graphical_norman_culture_group_trigger = yes
			POD_CFP_has_graphical_goidelic_culture_group_trigger = yes
			POD_CFP_has_graphical_papal_culture_group_trigger = yes	
			POD_CFP_has_graphical_west_african_culture_group_trigger = yes
			POD_CFP_has_graphical_east_african_culture_group_trigger = yes	
		}
	}
}

mena = {
	trigger = {
		exists = capital_county
		OR = {
			capital_county.culture = {
				OR = {
					has_graphical_mena_culture_group_trigger = yes
					has_graphical_african_culture_group_trigger = yes
					has_graphical_steppe_culture_group_trigger = yes
				}
			}
			AND = {
				POD_CFP_has_graphical_iberian_culture_group_trigger = yes
				OR = {
					faith.religion = religion:ashirra_religion 
				    faith.religion = religion:roadofblood_religion
				}
				NOT = { exists = global_var:cfp_is_loaded }
			}
		}

		NOR = {
			AND = { 
				has_graphical_west_african_culture_group_trigger = yes
				NOT = { faith.religion = religion:ashirra_religion }
			}
			AND = {
				has_graphical_east_african_culture_group_trigger = yes
				NOT = { faith.religion = religion:ashirra_religion }
			}
			AND = {
				exists = global_var:cfp_is_loaded
				POD_CFP_has_graphical_iberian_culture_group_trigger = yes
			}
		}
	}
}
