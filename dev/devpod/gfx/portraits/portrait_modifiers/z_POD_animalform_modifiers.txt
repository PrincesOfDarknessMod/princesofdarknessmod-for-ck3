######################################################################
# Naturalistic animal full-body portrait replacements.
#
# ONE group, because a group selects at most ONE of its entries and all eight
# animals are mutually exclusive body swaps - that gives mutual exclusion for
# free. No `fallback` key, which is what keeps the group a no-op for every
# character who matches nothing.
#
# interface_position 9 matches PoD's POD_fera_form and POD_beast_form, so the
# barbershop lists this alongside PoD's other form pickers.
######################################################################

POD_animalform_model = {
	usage = both
	priority = 200		# MUST exceed POD_clothes (100); see banner above
