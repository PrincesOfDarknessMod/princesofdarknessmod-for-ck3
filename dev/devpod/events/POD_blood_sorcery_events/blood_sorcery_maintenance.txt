namespace = POD_blood_sorcery_maintenance

POD_blood_sorcery_maintenance.100 = {
	hidden = yes
	
	trigger = {
		is_ai = no
		has_character_flag = got_BS_exp
	}
	
	immediate = {
		change_variable = {
			name = blood_sorcery_points 
			add = { value = learning divide = 2}
		}
	}

}

POD_blood_sorcery_maintenance.3000 = {
	hidden = yes
	
	immediate = {
		if = {
			limit = {
				is_target_in_variable_list = {
					name = centered_mind_botches
					target = flag:possessed
				}
				NOT = { has_character_flag = possessed_removal_counter }
			}
			remove_trait = possessed_1
			remove_list_variable = {
				name = centered_mind_botches
				target = flag:possessed
			}
		}
		if = {
			limit = {
				is_target_in_variable_list = {
					name = centered_mind_botches
					target = flag:lunatic
				}
				NOT = { has_character_flag = lunatic_removal_counter }
			}
			remove_trait = lunatic_1
			remove_list_variable = {
				name = centered_mind_botches
				target = flag:lunatic
			}
		}
		if = {
			limit = {
				is_target_in_variable_list = {
					name = centered_mind_botches
					target = flag:depressed
				}
				NOT = { has_character_flag = depressed_removal_counter }
			}
			remove_trait = depressed_1
			remove_list_variable = {
				name = centered_mind_botches
				target = flag:depressed
			}
		} 	
		if = {
			limit = {
				is_target_in_variable_list = {
					name = centered_mind_botches
					target = flag:dull
				}
				NOT = { has_character_flag = dull_removal_counter }
			}
			remove_trait = dull
			remove_list_variable = {
				name = centered_mind_botches
				target = flag:dull
			}
		} 
	}
		
	

}
