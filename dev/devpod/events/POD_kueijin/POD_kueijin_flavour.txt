		animation = grief
	}
	immediate = {
		random_religion_global = {
			limit = { is_in_family = rf_kueijin }
			random_faith = {
				save_scope_as = new_dharma
			}
		}
	}

	# Convert to random Dharma

		name = podkueijin.10.a
		
		add_internal_flag = dangerous

		}	
		
		set_character_faith = scope:new_dharma

	}
	# NO!

		trigger = { piety_level >= 3 }
		show_as_unavailable = { piety_level <= 2 }

		add_internal_flag = special
		reason = piety_level	

		name = podkueijin.10.b
		add_piety_level = -1
	}
}
