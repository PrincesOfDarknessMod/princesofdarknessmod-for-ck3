podkueijin.200 = {
	type = character_event
	content_source = dlc_POD
	title = podkueijin.200.t
	desc = podkueijin.200.desc
	theme = vampire

	left_portrait = {
		character = root
		animation = stress
	}
	
	widgets = {
		widget = {
			gui = "POD_shader_event_fog_alley_night"
			container = "background_shader_vfx_container"
		}
		widget = {
			gui = "event_window_widget_stress"
			container = "custom_widgets_container"
		}
	}
	
	trigger = {
		is_kueijin_trigger = yes
		NOT = { has_trait = yinyang_balanced }
		OR = {
			AND = {
				is_landed = yes
				primary_title.tier >= tier_county
			}
			has_character_flag = forced_to_frenzy
		}
	}

	immediate = {
		root = { save_scope_as = stress_character } # For stress widget
		
		stress_threshold_event_post_immediate = yes
	}
	
	option = {
		name = podkueijin.200.a	
		
		trigger = { has_trait = yin_imbalanced }
		
		#Account for stress.
		add_stress = major_stress_impact_loss
		custom_tooltip = podkueijin.200.a.tt
		
		POD_become_hideous_effect = yes
		
		ai_chance = { base = 0 }	#AI should never opt for stress.
	}
	
	option = {
		name = podkueijin.200.c	
		
		trigger = { has_trait = yang_imbalanced }
		
		#Account for stress.
		add_stress = major_stress_impact_loss
		custom_tooltip = podkueijin.200.c.tt
		
		random_list = {
			60 = {
				capital_province = {
					create_epidemic_outbreak = {
						type = crimson_fever
						intensity = minor
					}
				}
			}
			30 = {
				capital_province = {
					create_epidemic_outbreak = {
						type = crimson_fever
						intensity = major
					}
				}
			}
			10 = {
				capital_province = {
					create_epidemic_outbreak = {
						type = crimson_fever
						intensity = apocalyptic
					}
				}
			}
		}
	}
	
	# Option G (Always the 3rd option): Do not gain any coping traits, but major stress gain!
	option = {
		name = stress_threshold.1001.endure
		add_stress = mental_break_opt_out_stress_gain

		ai_chance = {
			base = 1
			modifier = {
				add = 15
				has_trait = stubborn
			}
		}
	}
	
	after = {
		stress_threshold_event_aftereffects = yes #Perform stress threshold event clean-up and preparation for future stress threshold events.
	}
}
