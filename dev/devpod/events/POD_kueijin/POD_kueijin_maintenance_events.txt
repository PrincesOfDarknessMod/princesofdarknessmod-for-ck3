		character = root
		animation = idle
	}
	
	trigger = {
		is_kueijin_trigger = yes
		is_imprisoned = no
		NOR = {
			has_trait = torpor
			# Not in duel
			exists = scope:sc_defender_last_move
			exists = scope:sc_attacker_last_move

	}
	
	option = {
		name = podkueijin.210.a

			limit = { faith = { has_doctrine_parameter = hurt_mortal_lose_piety } }
			add_piety = major_piety_loss
		}
		add_prestige = major_prestige_loss
		POD_reset_chi_effect = yes
	}
}
