namespace = podkueijin

podkueijin.1 = {
	hidden = yes
	immediate = {
		### KJ 2 Yearly Dharma Passive Gain
		if = {
			limit = {
				has_trait = dharma
				save_temporary_scope_value_as = {
					name = biyearly_countdown
					value = { value = current_year modulo = 2 }
				}
				scope:biyearly_countdown = 0
			}
			POD_give_dharma_xp_effect = { VALUE = POD_kueijin_dharma_xp_passive } 
		}
		### KJ Spirited Away Pulse
		if = {
			limit = {
				exists = capital_province
				capital_province ?= {
					has_building_or_higher = dragon_nest_01
				}
				save_temporary_scope_value_as = {
					name = biyearly_countdown
					value = { value = current_year modulo = 15 }
				}
				scope:biyearly_countdown = 0
			}
			capital_province ?= { POD_upgrade_ducal_building_effect = { MAX = 10 } }
		}
	}
}

podkueijin.200 = {
	type = character_event
	content_source = dlc_POD
	title = podkueijin.200.t
	desc = podkueijin.200.desc
	theme = vampire

	left_portrait = {
		character = root
		animation = stress
	}
	
	widgets = {
		widget = {
			gui = "POD_shader_event_fog_alley_night"
			container = "background_shader_vfx_container"
		}
		widget = {
			gui = "event_window_widget_stress"
			container = "custom_widgets_container"
		}
	}
	
	trigger = {
		is_kueijin_trigger = yes
		OR = {
			is_playable_character = yes
			has_character_flag = forced_to_frenzy
		}
	}

	immediate = {
		root = { save_scope_as = stress_character }
		
		stress_threshold_event_post_immediate = yes
	}
	
	option = {
		name = podkueijin.200.a	
				
		add_stress = major_stress_impact_loss
		custom_tooltip = podkueijin.200.a.tt
		
		POD_become_hideous_effect = yes
		
		ai_chance = { base = 0 }
	}
	
	option = {
		name = podkueijin.200.b
				
		add_stress = major_stress_impact_loss
		custom_tooltip = podkueijin.200.b.tt
		
		random_list = {
			60 = {
				location = {
					create_epidemic_outbreak = {
						type = crimson_fever
						intensity = minor
					}
				}
			}
			30 = {
				location = {
					create_epidemic_outbreak = {
						type = crimson_fever
						intensity = major
					}
				}
			}
			10 = {
				location = {
					create_epidemic_outbreak = {
						type = crimson_fever
						intensity = apocalyptic
					}
				}
			}
		}
	}
	
	option = {
		name = stress_threshold.1001.endure
		add_stress = mental_break_opt_out_stress_gain

		ai_chance = {
			base = 1
			modifier = {
				add = 15
				has_trait = stubborn
			}
		}
	}
	
	after = {
		stress_threshold_event_aftereffects = yes
	}
}
