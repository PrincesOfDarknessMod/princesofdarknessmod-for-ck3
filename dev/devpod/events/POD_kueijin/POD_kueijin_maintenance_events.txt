podkueijin.20 = {
	hidden = yes
	type = character_event
	
	trigger = {
		is_ai = no
		has_POD_kj_blood_trigger = yes
	}
	
	immediate = {
		# Change to the fifth age
		if = {
			limit = {
				NOT = { has_character_flag = age_degenerated }
				trigger_if = {
					limit = { has_trait = dharma9 }
					current_date >= 1992.1.1
				}
				trigger_else_if = {
					limit = { has_trait = dharma8 }
					current_date >= 1650.1.1
				}
				trigger_else_if = {
					limit = { has_trait = dharma7 }
					current_date >= 1600.1.1
				}
				trigger_else_if = {
					limit = { has_trait = dharma6 }
					current_date >= 1550.1.1
				}
				trigger_else_if = {
					limit = { has_trait = dharma5 }
					current_date >= 1450.1.1
				}
				trigger_else_if = {
					limit = { has_trait = dharma4 }
					current_date >= 1350.1.1
				}
				trigger_else = {
					current_date >= 1250.1.1
					NOT = { has_trait = dharma10 }
				}
			}
			trigger_event = podkueijin.250
		}
	}
}
