podkueijin.1 = {
	hidden = yes
	immediate = {
		### KJ 2 Yearly Dharma Passive Gain
		if = {
			limit = {
				has_trait = dharma
				save_temporary_scope_value_as = {
					name = biyearly_countdown
					value = { value = current_year modulo = 2 }
				}
				scope:biyearly_countdown = 0
			}
			POD_give_dharma_xp_effect = { VALUE = POD_kueijin_dharma_xp_passive } 
		}

		### KJ 10 years extra life
		if = { 
			limit = { 
				has_perk = holding_the_green_demon_at_bay_perk 
				save_temporary_scope_value_as = {
					name = biyearly_countdown
					value = { value = current_year modulo = 10 }
				}
				scope:biyearly_countdown = 0
				NOT = { has_character_modifier = extra_life_1_modifier }
			} 
			add_character_modifier = extra_life_1_modifier
		}
	}
}
