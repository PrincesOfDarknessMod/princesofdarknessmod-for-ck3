# War to Anda
PODquincunx.250 = {
	hidden = yes
	
	trigger = {
		title:e_anda = { is_title_created = yes }
		title:e_anda.holder = { faith = { has_doctrine = special_doctrine_is_anda_faith } }
	}
	
	on_trigger_fail = {
		if = {
			limit = { title:e_anda = { is_title_created = no } }	
			title:e_anda = {
				every_de_jure_county = {
					limit = { faith = faith:viayasaq }
					set_county_faith = faith:shining_ice_guardians
				}
				every_in_de_jure_hierarchy = {
					limit = { tier = tier_kingdom }
					set_de_jure_liege_title = title:e_quincunx_vanguard
				}
			}
		}
		add_character_flag = POD_quincunx_avenger
		trigger_event = PODquincunx.300
	}
	
	immediate = {
		title:e_anda.holder = { save_scope_as = anda_holder }
		add_character_flag = POD_quincunx_warrior
		start_war = {
			cb = POD_event_war
			target = scope:anda_holder
		}
		random_character_war = {
			limit = { using_cb = POD_event_war }
			save_scope_as = quincunxwar
		}
		every_independent_ruler = {
			limit = { 
				NOT = { this = root }
				faith = { has_doctrine = special_doctrine_against_to_anda_hostility } 
			}
			scope:quincunxwar = {
				hidden_effect = { set_called_to = prev }
				add_attacker = prev
			}
		}		
		every_independent_ruler = {
			limit = { 
				NOT = { this = scope:anda_holder }
				faith = { 
					OR = {
						has_doctrine = special_doctrine_is_anda_faith 
						has_doctrine = special_doctrine_allied_to_anda_hostility
					}
				} 
			}
			scope:quincunxwar = {
				hidden_effect = { set_called_to = prev }
				add_defender = prev
			}
		}
	}
}

PODquincunx.251 = {
	type = character_event
	content_source = dlc_POD
	title = great_holy_war.0050.title
	desc = {
		first_valid = {
			#Everybody else's flavor - Attackers.
			triggered_desc = {
				trigger = {
					OR = {
						faith = { has_doctrine = special_doctrine_against_to_anda_hostility }
						is_vassal_of = scope:attacker
					}
				}
				desc = great_holy_war.0071.generic.attackers.desc
			}
			#Everybody else's flavor - Defenders.
			triggered_desc = {
				trigger = {
					OR = {
						faith = { has_doctrine = special_doctrine_allied_to_anda_hostility }
						is_vassal_of = scope:defender
					}
				}
				desc = great_holy_war.0071.generic.defenders.desc
			}
		}
	}
	theme = faith
	override_background = {
		reference = armory
	}
	right_portrait = {
		character = scope:ghw_sponsor
		animation = war_attacker
	}
	left_portrait = {
		character = scope:ghw_defender
		animation = war_defender
	}
	
	immediate = {
		title:e_anda = {
			save_scope_as = target_kingdom # Is empire, but just not mess up the existent loca
		}
		scope:attacker = { #Loc keys for localization fluff.
			save_scope_as = ghw_sponsor
		}
		scope:defender = {
			save_scope_as = ghw_defender
		}
		scope:defender.faith = {
			save_scope_as = target_faith
		}
		scope:attacker.faith = {
			save_scope_as = the_faith
		}
		scope:attacker = {
			random_character_war = {
				limit = { using_cb = POD_event_war }
				save_scope_as = the_ghw
			}
		}
	}

	option = {
		name = great_holy_war.0050.a

		custom_tooltip = great_holy_war.0051.generic.start.tt
	}
}

		random_neighboring_and_across_water_top_liege_realm_owner = {

			}
			root = {
				start_war = {
					cb = POD_event_war
					target = prev
