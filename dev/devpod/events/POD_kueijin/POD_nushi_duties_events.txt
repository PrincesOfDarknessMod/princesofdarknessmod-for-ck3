namespace = POD_nushi_duties

POD_nushi_duties.1 = {
	hidden = yes
	
	trigger = {
		house ?= { has_nushi_modifier_trigger = yes }
		has_nushi_duty_trigger = no
	}
	
	immediate = {
		random = {
			chance = 10
			random_list = {
				75 = { 
					trigger = { dynasty = { any_dynasty_member = { NOT = { this = root } } } }
					trigger_event = POD_nushi_duties.101
				}
				20 = { trigger_event = POD_nushi_duties.111 }
				5 = { trigger_event = POD_nushi_duties.121 }
			}
		}
	}
}

POD_nushi_duties.101 = {
	type = character_event
	content_source = dlc_POD
	title = POD_nushi_duties.100.t
	desc = POD_nushi_duties.101.desc
	theme = spirit_wilds

	left_portrait = {
		character = root
		animation = boredom
	}
	
	immediate = {
		dynasty = {
			random_dynasty_member = {
				limit = { 
					NOT = { this = root }
					has_trait = cynical
				}
				alternative_limit = {
					NOT = { this = root }
				}
				save_scope_as = impious
			}
		}
		set_variable = { name = impious value = scope:impious }
	}

	option = {
		name = POD_nushi_duties.100.a
		custom_tooltip = POD_nushi_duties.100.a.tt
		
		add_character_flag = duty_punish_impious_flag
	}

	option = {
		name = POD_nushi_duties.100.b
		
		add_prestige = minor_prestige_loss
	}
}
