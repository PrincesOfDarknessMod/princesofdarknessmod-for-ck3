podkueijin.350 = {
	type = character_event
	content_source = dlc_POD
	title = podkueijin.350.t
	desc = podkueijin.350.desc
	theme = dynasty 
	
	left_portrait = {
		character = root
		animation = idle
	}
	
	right_portrait = {
		character = scope:optional_child
		animation = idle
	}	

	widgets = {
		widget = {
			gui = "event_window_widget_name_child"
			container = "dynamic_birth_name"
			controller = name_character
			setup_scope = { scope:optional_child = { save_scope_as = name_character_target } }
		}
	}

	# Adopt
	option = {
		name = podkueijin.350.a			

		hidden_effect = {
			scope:optional_child = {
				if = {
					limit = { root = { is_male = yes } }
					set_father = root
					set_house = father.house
				}
				else = {
					set_mother = root
					set_house = mother.house
				}
				
				POD_kueijin_initialization_effect = yes
			}
		
			reverse_add_opinion = {
				target = scope:optional_child
				modifier = hired_me_opinion
			}
		}
		
		trigger_event = {
			on_action = kueijin_personalty_education_random_pulse
			days = 1
		}
		trigger_event = {
			id = podkueijin.400
			months = { 6 9 }
		}
		
		ai_chance = {
			factor = 100
		}
	}
}

podkueijin.400 = {
	type = character_event
	content_source = dlc_POD
	title = podkueijin.400.t
	desc = podkueijin.400.desc

	theme = vampire
	left_portrait = {
		character = root
		animation = idle
	}
	
	right_portrait = {
		character = scope:optional_child
		animation = idle
	}	
	
	widget = {
		gui = "POD_shader_event_fog_alley_night"
		container = "background_shader_vfx_container"
	}
	
	trigger = {
		scope:optional_child = { is_alive = yes }
	}
	
	immediate = {
		scope:optional_child = { 
			if = {
				limit = { has_trait = dharma0 } 
				remove_trait = dharma0
				add_trait = dharma1
			}
		}
	}
	
	# Continue study
	option = {
		name = podkueijin.400.a
		remove_short_term_gold = 100
		custom_tooltip = podkueijin_education
		
		random_list = {
			1 = { scope:optional_child = { add_martial_skill = 1 } }
			1 = { scope:optional_child = { add_diplomacy_skill = 1 } }
			1 = { scope:optional_child = { add_stewardship_skill = 1 } }
			1 = { scope:optional_child = { add_intrigue_skill = 1 } }
			1 = { scope:optional_child = { add_learning_skill = 1 } }
			1 = { scope:optional_child = { add_prowess_skill = 1 } }
		}

		hidden_effect = {
			random_list = {
				1 = {
					trigger_event = {
						id = podkueijin.400
						years = 1
					}
				}
				1 = {
					trigger_event = {
						id = podkueijin.401
						years = 1
					}
				}
			}
		}
	}
	
	# Finish study
	option = {
		name = podkueijin.400.b
		
		scope:optional_child = { 
			add_character_flag = study_completed
			random_list = {
				1 = {
					add_trait = education_intrigue_1
				}
				1 = {
					add_trait = education_diplomacy_1
				}
				1 = {
					add_trait = education_learning_1
				}
				1 = {
					add_trait = education_stewardship_1
				}
				1 = {
					add_trait = education_martial_1
				}
				1 = {
					add_trait = education_martial_prowess_1
				}
			}
		}		
	}
}
