namespace = POD_personal_coat_of_arms

POD_personal_coat_of_arms.100 = {
	hidden = yes
	immediate = {
		if = {
			limit = {
				scope:exiled_dummy = {
					is_courtier_of = root
				}
			}
			remove_courtier_or_guest = scope:exiled_dummy
		}
		else = {
			scope:exiled_dummy.employer = { remove_courtier_or_guest = scope:exiled_dummy }
		}
		character:POD_wraith_01 = {
			add_courtier = scope:exiled_dummy
		}
	}
}

# Event for when the real character w/ the COA dies
POD_personal_coat_of_arms.200 = {
	type = character_event
	hidden = yes
	trigger = {
		exists = var:my_personal_coa
		var:my_personal_coa = {
			AND = {
				var:my_coa_belongs_to = root
				is_alive = yes
			}
		}
	}
	immediate = {
		var:my_personal_coa = {
			death = {
				death_reason = death_vanished
			}
		}
	}
}

# Event for when the dummy character w/ the COA dies
POD_personal_coat_of_arms.300 = {
	type = character_event
	hidden = yes
	trigger = {
		exists = var:my_coa_belongs_to
		var:my_coa_belongs_to = {
			is_alive = yes
		}
	}
	immediate = {
		root.house = {
			save_scope_as = dummy_house
		}
		var:my_coa_belongs_to = {
			save_scope_as = coa_owner
		}
		create_character = {
			#name = "Dummy"
			gender = male
			employer = character:POD_wraith_01
			trait = "content"
			trait = "shy"
			trait = "lazy"
			trait = "wraith"
			trait = "torpor"
			trait = "education_diplomacy_1"
			trait = "diplomatic_court_1"
			diplomacy = 0
			stewardship = 0
			martial = 0
			intrigue = 0
			learning = 0
			random_traits = no
			faith = faith:innocence
			culture = title:e_stygia.holder.culture
			dynasty = generate
			save_scope_as = new_dummy
		}
		scope:new_dummy = {
			set_house = root.house
			set_variable = {
				name = my_coa_belongs_to
				value = scope:coa_owner
			}
			add_character_flag = POD_dde_hre_war_low_nobles_flag
			add_character_flag = POD_portrait_hermetic_mage_hood_male_flag
			add_character_flag = POD_ep1_adventurer_cloak_flag
		}
		scope:coa_owner = {
			set_variable = {
				name = my_personal_coa
				value = scope:new_dummy
			}
			set_relation_pod_personal_coa_slave = scope:new_dummy
		}
	}
}
