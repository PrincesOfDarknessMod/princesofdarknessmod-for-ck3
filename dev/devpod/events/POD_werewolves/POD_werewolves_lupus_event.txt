POD_werewolves_lupus.101 = {
	type = character_event
	title = POD_werewolves_lupus.101.t
	desc = POD_werewolves_lupus.101.desc

	theme = faith
	left_portrait = {
		character = root
		animation = personality_honorable
	}
	
	trigger = {
		has_trait = werewolf
	}
	
	immediate = {
		hidden_effect = {
			if = {
				limit = {
					exists = capital_province
				}
				capital_province = { save_scope_as = target_location }
			}
			else = {
				location = { save_scope_as = target_location }
			}
			create_character = {
				location = scope:target_location
				template = lupus_maintenance_character
				save_scope_as = lupus_1
			}
			add_visiting_courtier = scope:lupus_1
		}
		
		random = {			
			chance = 20
			modifier = {
				add = 50
				has_faith = faith:redtalon
			}
			
		hidden_effect = {
			if = {
				limit = {
					exists = capital_province
				}
				capital_province = { save_scope_as = target_location }
			}
			else = {
				location = { save_scope_as = target_location }
			}
			create_character = {
				location = scope:target_location
				template = lupus_maintenance_character
				save_scope_as = lupus_2
			}
			add_visiting_courtier = scope:lupus_2
		}						
		}
		
		random = {			
			chance = 10
			modifier = {
				add = 50
				has_faith = faith:redtalon
			}
			
		hidden_effect = {
			if = {
				limit = {
					exists = capital_province
				}
				capital_province = { save_scope_as = target_location }
			}
			else = {
				location = { save_scope_as = target_location }
			}
			create_character = {
				location = scope:target_location
				template = lupus_maintenance_character
				save_scope_as = lupus_3
			}
			add_visiting_courtier = scope:lupus_3
		}						
		}		
	}
	
	option = {
		name = POD_werewolves_lupus.101.option.a	
		
		add_courtier = scope:lupus_1

		if = {
			limit = {
				exists = scope:lupus_2
				}
		add_courtier = scope:lupus_2
		}

		if = {
			limit = {
				exists = scope:lupus_3
				}
		add_courtier = scope:lupus_3
		}
	}
}


#Lupus pups for lupus ruler

POD_werewolves_lupus.102 = {
	type = character_event
	title = POD_werewolves_lupus.102.t
	desc = POD_werewolves_lupus.102.desc

	theme = faith
	left_portrait = {
		character = root
		animation = personality_content
	}
	
	trigger = {
		has_trait = beast_born
		has_trait = werewolf
	}
	
	immediate = {
		hidden_effect = {
			if = {
				limit = {
					exists = capital_province
				}
				capital_province = { save_scope_as = target_location }
			}
			else = {
				location = { save_scope_as = target_location }
			}
			create_character = {
				location = scope:target_location
				template = lupus_maintenance_character
				dynasty_house = root.house
				save_scope_as = lupus_1
			}
			add_visiting_courtier = scope:lupus_1
		}		
	}
	
	option = {
		name = POD_werewolves_lupus.102.option.a	
		
		trigger = {
			AND = {
				exists = dynasty
				dynasty = {
					has_dynasty_perk = redtalon_legacy_4
				}
			}
		}
		
		add_stress = minor_stress_gain
		
		custom_tooltip = cub_redtalon4_tt
		
		custom_tooltip = cub_recruited_tt
	
		add_courtier = scope:lupus_1

		scope:lupus_1 = {
			add_prowess_skill = 10
			add_learning_skill = 3
		}	

		random = {			
			chance = 10
			
		scope:lupus_1 = {
			add_stewardship_skill = 5
		}	
		}
		
		random = {			
			chance = 20
			
		scope:lupus_1 = {
			add_diplomacy_skill = 5
		}	
		}
		
		random = {			
			chance = 33
			
		scope:lupus_1 = {
			add_martial_skill = 5
		}	
		}
		
		hidden_effect = {
		scope:lupus_1 = {
		if = {
			limit = {
				root = { is_female = yes }
				}
				set_mother = root
		}
		else = {
			set_father = root
		}
		}
		}		
	}
	
	
	option = {
		name = POD_werewolves_lupus.102.option.b	
		
		trigger = {
			NAND = {
				exists = dynasty
				dynasty = {
					has_dynasty_perk = redtalon_legacy_4
				}
			}
		}
		
		add_stress = minor_stress_gain
		
		custom_tooltip = cub_recruited_tt
	
		add_courtier = scope:lupus_1

		scope:lupus_1 = {
			add_learning_skill = 3
		}	

		random = {			
			chance = 10
			
		scope:lupus_1 = {
			add_stewardship_skill = 5
		}	
		}
		
		random = {			
			chance = 20
			
		scope:lupus_1 = {
			add_diplomacy_skill = 5
		}	
		}
		
		random = {			
			chance = 33
			
		scope:lupus_1 = {
			add_martial_skill = 5
		}	
		}
		
		hidden_effect = {
		scope:lupus_1 = {
		if = {
			limit = {
				root = { is_female = yes }
				}
				set_mother = root
		}
		else = {
			set_father = root
		}
		}
		}
	}

	option = {
		name = POD_werewolves_lupus.102.option.c	

		scope:lupus_1 = { 
			death = { death_reason = death_disappearance }
		}		
	}	
}
