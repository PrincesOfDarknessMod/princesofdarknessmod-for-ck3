#First change

POD_werewolves_maintenance.2 = {
	type = character_event
	content_source = dlc_POD
	title = POD_werewolves_maintenance.2.t
	desc = POD_werewolves_maintenance.2.desc

	theme = faith
	left_portrait = {
		character = root
		animation = shock
	}
	
	trigger = {
		is_alive = yes
		has_trait = kinfolk
		NOT = { has_trait = changeling }
		NOT = { has_trait = revenant }
	}
	
	option = {
		name = POD_werewolves_maintenance.2.option.a	
		
		if = {
			limit = { has_trait = kinfolk }
			remove_trait = kinfolk
			if = {
				limit = { has_trait = gnosis }
				remove_trait = gnosis
			}
		}
		
		if = {
			limit = {
				has_trait = ghoul
				POD_splat_mixing_haight_mode = no
			}
			remove_trait = ghoul
			every_relation = {
				type = ghoul_domitor
				remove_hook = {
					target = root
					type = strong_bloodoath_hook
				}
				root = {
					remove_relation_ghoul_domitor = prev
					remove_character_flag = blocked_from_leaving
					remove_character_flag = blood_bonded
				}
			}
		}
		# if = {
		# 	limit = { has_trait = revenant }
		# 	remove_trait = revenant
		# }
		
		if = {
			limit = { has_trait = supehunter }
			hidden_effect = { 
				remove_trait = supehunter
				save_temporary_scope_as = is_sanbenito
			}
		}
		
		if = {
			limit = { exists = var:fera_type }
			var:fera_type ?= { save_scope_as = fera_type }
		}
		else = {
			# fallback
			trait:werewolf = { save_scope_as = fera_type }
		}
		
		add_trait_force_tooltip = scope:fera_type 
		add_trait = homid_born
		POD_fera_initialization_effect = yes
		
		hidden_effect = {
			if = {
				limit = { exists = scope:is_sanbenito }
				add_trait = supehunter
				set_character_faith = faith:sanbenito
			}
		}
		
		POD_remove_wrong_supernatural_traits_effect = yes
		POD_remove_wrong_supernatural_modifiers_effect = yes

		if = {
			limit = {
				exists = mother
				mother = {
					is_landed = yes
					primary_heir ?= { has_trait = kinfolk }
					is_ai = yes
				}
			}
			mother = { set_designated_heir = prev }
		}
		else_if = {
			limit = {
				exists = father
				father = {
					is_landed = yes
					primary_heir ?= { has_trait = kinfolk }
					is_ai = yes
				}
			}
			father = { set_designated_heir = prev }
		}
	}
}

# Werewolf Healing

POD_werewolves_maintenance.6 = {
	hidden = yes
	
	trigger = {
		has_trait = shapeshifter
		NOT = { has_trait = kitsune }
		NOT = { has_trait = entomos }
		POD_has_enough_gnosis_for_gift = { COUNT = 1 }
		OR = {
			AND = {
				has_trait = disfigured
				NOT = { has_character_flag = POD_permanent_disfigured_flag }
			}
			
			has_trait = wounded
			
			AND = {
				has_trait = blind
				NOT = { has_character_flag = POD_permanent_blind }
			}
			AND = {
				has_trait = maimed
				NOT = { has_character_flag = POD_permanent_maimed }
			}
			AND = {
				has_trait = one_legged
				NOT = { has_character_flag = POD_permanent_one_legged }
			}
			AND = {
				has_trait = one_eyed
				NOT = { has_character_flag = POD_permanent_one_eyed }
			}
			AND = {
				has_trait = eunuch_1
				NOT = { has_character_flag = POD_permanent_eunuch }
			}
		}	
		POD_can_be_healed_trigger = yes
	}

	immediate = {
		
		if = {
			limit = { has_character_modifier = infected_wound_modifier }
			remove_character_modifier = infected_wound_modifier

			if = {
				limit = { NOT = { has_trait = scarred } }
				add_character_flag = will_get_scar
			}
		}
		else_if = {
			limit = { NOT = { has_trait = scarred } }
			random = {
				chance = 10
				add_character_flag = will_get_scar
			}
		}
		send_interface_toast = {
			title = health.0001.desc

			POD_gnosis_loss_check_effect = yes

			remove_trait = flagellant
			remove_injury_trait_effect = yes
	
			if = {
				limit = {
					has_trait = disfigured
					NOT = { has_character_flag = POD_permanent_disfigured_flag }
				}
				remove_trait = disfigured
			}
			
			if = {
				limit = { has_character_flag = will_get_scar }
				add_trait = scarred
				remove_character_flag = will_get_scar
			}
		}
	}
}

	hidden = yes
	orphan = yes
	
	trigger = {
		has_trait = shapeshifter
		POD_has_enough_gnosis_for_gift = { COUNT = 1 }
		OR = {
			has_trait = ill
			has_trait = pneumonic
			has_trait = great_pox
			has_trait = early_great_pox
			has_trait = lovers_pox
			has_trait = leper

					has_trait = early_great_pox
				}
				remove_trait = early_great_pox

POD_werewolves_maintenance.9 = {
	hidden = yes
	orphan = yes
	
	trigger = {
		POD_splat_has_gnosis_trigger = yes

		OR = {
			AND = {
				NOT = { has_character_modifier = POD_gnosis_5_modifier }
				POD_is_kinfolk_trigger = no
			}
			AND = {
				NOT = { has_character_modifier = POD_gnosis_3_modifier }
				POD_is_kinfolk_trigger = yes				
			}
		}
	}

	immediate = {
		POD_gnosis_growing_maintenance_effect = yes
	}
}
