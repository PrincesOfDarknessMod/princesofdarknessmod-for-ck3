				}

			}			
			triggered_desc = {
				trigger = { has_faith = faith:bubasti }
				desc = POD_werewolves_adoption.10.bubasti
			}
			triggered_desc = {
				trigger = { has_trait = bastet }
				desc = POD_werewolves_adoption.10.bastet
			}
			#default to werewolves
			triggered_desc = {
				desc = POD_werewolves_adoption.10.desc
			}
		}
	}	
	theme = faith 

	trigger = { 
	}

	option = {
		name = POD_werewolves_adoption.10.a	

		trigger = {
			OR = {
				NOT = { has_faith = faith:bubasti }
				is_target_in_global_variable_list = {
					name = unavailable_unique_decisions
					target = flag:bubasti_objective_decision
				}
				AND = {
					uses_cairo_bast_alternate_reward = no
					any_directly_owned_province = { has_building_with_flag = building_allows_bubasti_returning has_building_with_flag = cairo }
				}
			}		
		}
		
		trigger_event = POD_werewolves_adoption.11
		
		custom_tooltip = start_beastborn_content_tt
		
	}
	
	option = {
		name = POD_werewolves_adoption.10.b

		trigger = {
			OR = {
				NOT = { has_faith = faith:bubasti }
				is_target_in_global_variable_list = {
					name = unavailable_unique_decisions
					target = flag:bubasti_objective_decision
				}

					uses_cairo_bast_alternate_reward = no
					any_directly_owned_province = { has_building_with_flag = building_allows_bubasti_returning has_building_with_flag = cairo }
				}
			}
		}
		
		custom_tooltip = delayed_start_beastborn_content_tt

		trigger_event = {
			id = POD_werewolves_adoption.11
			years = { 20 40 }
		}
	}
	
	option = {
		name = POD_werewolves_adoption.10.c

		trigger = {
			OR = {
				NOT = { has_faith = faith:bubasti }
				is_target_in_global_variable_list = {
					name = unavailable_unique_decisions
					target = flag:bubasti_objective_decision
				}

					uses_cairo_bast_alternate_reward = no
					any_directly_owned_province = { has_building_with_flag = building_allows_bubasti_returning has_building_with_flag = cairo }
				}
			}
		}
		
		custom_tooltip = end_beastborn_content_tt
	}
	
	option = {
		name = POD_werewolves_adoption.10.d

	type = character_event
	content_source = dlc_POD
	title = POD_werewolves_adoption.102.t

		character = root
		animation = personality_content
	}
	
	trigger = {
		has_trait = beast_born
		has_trait = werewolf
	}
	
	immediate = {
		hidden_effect = {
			if = {
				limit = {
					exists = capital_province
				}
				capital_province = { save_scope_as = target_location }
			}
			else = {
				location = { save_scope_as = target_location }
			}
			create_character = {
				location = scope:target_location
				template = lupus_maintenance_character
				dynasty_house = root.house
				save_scope_as = lupus_1
			}
			add_visiting_courtier = scope:lupus_1
		}

			add_prowess_skill = 10
			add_learning_skill = 3
		}

			chance = 10
			
		scope:lupus_1 = {

		}	
		}
		
		random = {			
			chance = 20
			
		scope:lupus_1 = {
			add_diplomacy_skill = 5
		}	
		}
		
		random = {			
			chance = 33
			
		scope:lupus_1 = {
			add_martial_skill = 5
		}	
		}
		
		hidden_effect = {
		scope:lupus_1 = {POD_set_parent_effect = { PARENT = root }}
		}		
	}
	
	
	option = {
		name = POD_werewolves_adoption.102.option.b	
		
		trigger = {
			NOT = {
				dynasty ?= {
					has_dynasty_perk = redtalon_legacy_4
				}
			}
		}
		
		add_stress = minor_stress_gain
		
		custom_tooltip = cub_recruited_tt
	
		add_courtier = scope:lupus_1

		scope:lupus_1 = {
			add_learning_skill = 3
		}

		scope:lupus_1 = {
			add_stewardship_skill = 5
		}
