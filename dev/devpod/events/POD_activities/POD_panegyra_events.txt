scripted_effect panegyra_clear_variables_effect = {
	if = {
		limit = { has_character_flag = panegyra_overdrive_flag }
		remove_character_flag = panegyra_overdrive_flag
	}
	if = {
		limit = { has_character_flag = panegyra_disqualified_flag }
		remove_character_flag = panegyra_disqualified_flag
	}	
	if = {
		limit = { has_character_flag = panegyra_underdrive_flag }
		remove_character_flag = panegyra_underdrive_flag
	}
	if = {
		limit = { has_character_flag = panegyra_success_flag }
		remove_character_flag = panegyra_success_flag
	}
	if = {
		limit = { has_character_flag = panegyra_failure_flag }
		remove_character_flag = panegyra_failure_flag
	}	
	if = {
		limit = { exists = var:contest_aptitude }
		remove_variable = contest_aptitude
	}
}

scripted_effect panegyra_set_current_aptitude_variable_effect = {
	set_variable = {
		name = contest_aptitude
		value = {
			value = { 1 10 } #Random factor to make competitions less predictable
			
			if = {
				limit = {
					scope:activity = {
						has_current_phase = brujah_panegyra_phase_boxing
					}
				}
				add = panegyra_boxing_aptitude_value
			}
			else_if = {
				limit = { scope:activity = { has_current_phase = brujah_panegyra_phase_board_game } }
				add = panegyra_board_game_aptitude_value
			}
			else_if = {
				limit = { scope:activity = { has_current_phase = brujah_panegyra_phase_kerykes } }
				add = panegyra_kerykes_aptitude_value
			}
			else_if = {
				limit = { scope:activity = { has_current_phase = brujah_panegyra_phase_horse_race } }
				add = panegyra_horse_race_aptitude_value
			}
			else_if = {
				limit = { scope:activity = { has_current_phase = brujah_panegyra_phase_archery } }
				add = panegyra_archery_aptitude_value
			}
			else = { add = panegyra_wrestling_aptitude_value }
		}
	}
}

scripted_effect panegyra_general_rewards_effect = {
		add_prestige = major_prestige_value

		if = {
			limit = {
				scope:activity = {
					has_activity_option = {
						category = brujah_panegyra_option_prizes
						option = brujah_panegyra_prizes_good
					}
				}
			}
			add_gold = scope:host.medium_gold_value
		}
		else_if = {
			limit = {
				scope:activity = {
					has_activity_option = {
						category = brujah_panegyra_option_prizes
						option = brujah_panegyra_prizes_normal
					}
				}
			}
			add_gold = scope:host.minor_gold_value
		}
		else = {
			add_gold = scope:host.tiny_gold_value
		}
		if = {
			limit = { has_trait = tourney_participant }
			show_as_tooltip = { add_trait = tourney_participant }
		}
		if = {
			limit = { NOT = { has_trait = tourney_participant } }
			random = {
				chance = 35		
			add_trait = tourney_participant 
			}
		}		
}

scripted_effect panegyra_determine_winner_effect = {
	scope:activity = {
	every_attending_character = {
		panegyra_set_current_aptitude_variable_effect = yes
	}
	}
	
	scope:activity = {
	if = {
		limit = {
			any_attending_character = {
				is_alive = yes
				exists = var:contest_aptitude
				is_ai = no
				save_temporary_scope_as = winner_temp
			}
			NOT = {
				any_attending_character = {
					is_alive = yes
					exists = var:contest_aptitude
					var:contest_aptitude > scope:winner_temp.var:contest_aptitude
				}
			}
		}
		random_attending_character = {
			limit = {
				is_alive = yes
				exists = var:contest_aptitude
				is_ai = no
				save_temporary_scope_as = winner_temp
				scope:activity  = {
					NOT = {
						any_attending_character = {
							is_alive = yes
							exists = var:contest_aptitude
							var:contest_aptitude > scope:winner_temp.var:contest_aptitude
						}
					}
				}
			}
			save_scope_as = contest_winner
		}
	}
	else = {
		random_attending_character = {
			limit = {
				is_alive = yes
				exists = var:contest_aptitude
				save_temporary_scope_as = winner_temp
			}
			weight = {
				base = 1
				
					modifier = {
						add = {
							value = var:contest_aptitude
						}
					}					
				# = { CHAR = scope:winner_temp CONTEST = $CONTEST$ }
			}
			save_scope_as = contest_winner
		}
	}
	}
}

brujah_panegyra_events.13 = {
	type = activity_event
	title = brujah_panegyra_events.13.t
	desc = brujah_panegyra_events.13.desc
	##window = widget_activity_locale_fullscreen_event	theme = tournament_grounds
	theme = tournament_grounds
	left_portrait = {
		character = root
		animation = throne_room_cheer_1
	}
	
	immediate = {
		scope:activity = {
		every_attending_character = {
			panegyra_clear_everything_effect = yes
		}	
		}		
	}
	
	option = { 
		name = brujah_panegyra_events.13.a
		
		scope:host = {
		if = {
			limit = {
				scope:activity = {
					has_activity_option = {
						category = brujah_panegyra_option_accommodations
						option = brujah_panegyra_accommodations_bad
					}
				}
			}
			root.dynasty = { add_dynasty_prestige = minor_dynasty_prestige_gain }		
		}
		else_if = {
			limit = {
				scope:activity = {
					has_activity_option = {
						category = brujah_panegyra_option_accommodations
						option = brujah_panegyra_accommodations_normal
					}
				}
			}
			root.dynasty = { add_dynasty_prestige = medium_dynasty_prestige_gain }		
		}
		else = {
			root.dynasty = { add_dynasty_prestige = major_dynasty_prestige_gain }		
		}
		}
	}	
}
