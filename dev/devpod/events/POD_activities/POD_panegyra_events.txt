scripted_effect panegyra_clear_variables_effect = {
	#if = {
	#	limit = { has_character_flag = contest_horse_race_war_horse }
	#	remove_character_flag = contest_horse_race_war_horse
	#}
	if = {
		limit = { exists = var:contest_aptitude }
		remove_variable = contest_aptitude
	}
}

scripted_effect panegyra_clear_everything_effect = {
	if = {
		limit = { exists = var:contest_aptitude }
		remove_variable = contest_aptitude
	}
	if = {
		limit = { exists = var:archery_winner }
		remove_variable = archery_winner
	}	
}

scripted_effect panegyra_set_current_aptitude_variable_effect = {
	set_variable = {
		name = contest_aptitude
		value = {
			value = { 1 10 } #Random factor to make competitions less predictable
			
			if = {
				limit = {
					scope:activity = {
						has_current_phase = brujah_panegyra_phase_boxing
					}
				}
				add = panegyra_boxing_aptitude_value
			}
			else_if = {
				limit = { scope:activity = { has_current_phase = brujah_panegyra_phase_board_game } }
				add = panegyra_board_game_aptitude_value
			}
			else_if = {
				limit = { scope:activity = { has_current_phase = brujah_panegyra_phase_kerykes } }
				add = panegyra_kerykes_aptitude_value
			}
			else_if = {
				limit = { scope:activity = { has_current_phase = brujah_panegyra_phase_horse_race } }
				add = panegyra_horse_race_aptitude_value
			}
			else_if = {
				limit = { scope:activity = { has_current_phase = brujah_panegyra_phase_archery } }
				add = panegyra_archery_aptitude_value
			}
			else = { add = panegyra_wrestling_aptitude_value }
		}
	}
}

brujah_panegyra_events.2 = {
	type = character_event
	hidden = yes
	
	immediate = {
		scope:activity = {
		every_attending_character = {
			panegyra_set_current_aptitude_variable_effect = yes
		}
		}
		
		scope:activity = {
		if = {
			limit = {
				any_attending_character = {
					is_alive = yes
					exists = var:contest_aptitude
					is_ai = no
					save_temporary_scope_as = winner_temp
				}
				NOT = {
					any_attending_character = {
						is_alive = yes
						exists = var:contest_aptitude
						var:contest_aptitude > scope:winner_temp.var:contest_aptitude
					}
				}
			}
			random_attending_character = {
				limit = {
					is_alive = yes
					exists = var:contest_aptitude
					is_ai = no
					save_temporary_scope_as = winner_temp
					scope:activity  = {
						NOT = {
							any_attending_character = {
								is_alive = yes
								exists = var:contest_aptitude
								var:contest_aptitude > scope:winner_temp.var:contest_aptitude
							}
						}
					}
				}
				save_scope_as = contest_winner
			}
		}
		else = {
			random_attending_character = {
				limit = {
					is_alive = yes
					exists = var:contest_aptitude
					save_temporary_scope_as = winner_temp
				}
				weight = {
					base = 1
					
						modifier = {
							add = {
								value = var:contest_aptitude
							}
						}					
					# = { CHAR = scope:winner_temp CONTEST = $CONTEST$ }
				}
				save_scope_as = contest_winner
			}
		}
		scope:contest_winner = {
			set_variable = {
				name = archery_winner
				value = scope:archery_winner
			}			
		}
		every_attending_character = {
			trigger_event = brujah_panegyra_events.20
		}
		}
	}
	
	after = {
	}
}

	type = activity_event
	title = ep2_tournament_events.4002.t
	desc = ep2_tournament_events.4002.desc
	window = widget_activity_locale_fullscreen_event
	theme = tournament_contest
	left_portrait = {
		character = scope:contest_winner
		animation = throne_room_cheer_1
	}	
	
	immediate = {
	}
	
	option = { 
		name = ep2_tournament_events.4002.a
	}
	
	after = {
		scope:activity = {
		every_attending_character = {
			panegyra_clear_variables_effect = yes
		}	
		}		
	}
}
