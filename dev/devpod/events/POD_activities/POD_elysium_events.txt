	left_portrait = {
		character = scope:duelingknight
		animation = personality_honorable
	}

	right_portrait = {
		character = scope:otherknight
		animation = personality_bold
	}		
	
	widget = {
		gui = "POD_shader_event_fog_alley_night"
		container = "background_shader_vfx_container"
	}
	
	immediate = {
		random_knight = {
			limit = {
				is_available_ai_adult = yes
			}
			save_scope_as = duelingknight
			}
		
		random_knight = {
			limit = {
				is_available_ai_adult = yes
				NOT = {
					this = scope:duelingknight
					}
				}
			save_scope_as = otherknight						
			}	
	}
	
	option = {
		name = POD_elysium_events.5.option.a
		
		scope:duelingknight = {
			add_opinion = {
				target = root
				modifier = favored_elysium_opinion
			}
		}
		
		scope:otherknight = {
			add_opinion = {
				target = root
				modifier = disfavored_elysium_opinion
			}
		}		

	}

	option = {
		name = POD_elysium_events.5.option.b
		
		scope:otherknight = {
			add_opinion = {
				target = root
				modifier = favored_elysium_opinion
			}
		}
		
		scope:duelingknight = {
			add_opinion = {
				target = root
				modifier = disfavored_elysium_opinion
			}
		}	
	}
	
	option = {
		name = POD_elysium_events.5.option.c

			add_opinion = {

			add_opinion = {
				target = root

			}
		}	
		
		add_prestige = 100
	}

	type = character_event
	content_source = dlc_POD
	title = POD_elysium_events.100.t
	desc = POD_elysium_events.100.desc

	theme = vampire
	left_portrait = {
		character = cp:councillor_chancellor
		animation = worry
	}
	
	right_portrait = {
		character = scope:neighbor
	}
	
	widget = {
		gui = "POD_shader_event_fog_alley_night"
		container = "background_shader_vfx_container"
	}
	
	immediate = {
		if = {
			limit = {
				is_independent_ruler = yes
			}
			random_neighboring_top_liege_realm = {
				holder = {
					save_scope_as = neighbor
				}
			}
		}
		else = {
			random_neighboring_realm_same_rank_owner = {
				save_scope_as = neighbor
			}
		}
	}
	
	trigger = {
		exists = cp:councillor_chancellor		
		}		
	
	option = {
		name = POD_elysium_events.100.option.a
		
		scope:neighbor = {
			add_opinion = {
				target = root
				modifier = chancellor_task_neighbor_increased_opinion
				days = chancellor_task_modifier_duration
			}
		}	
		
		remove_short_term_gold = 250
		
		}	


	option = {
		name = POD_elysium_events.100.option.b

		cp:councillor_chancellor = {
			add_opinion = {
				target = root
				modifier = ignored_my_plan_opinion
				days = 1825
			}
		}	
					
		}
