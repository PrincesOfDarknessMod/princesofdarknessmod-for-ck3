POD_exalted_ancient_sorcery_visit_events.3000 = {
	type = activity_event
	title = POD_exalted_ancient_sorcery_visit_events.3000.t
	desc = POD_exalted_ancient_sorcery_visit_events.3000.desc

	theme = witchcraft
	left_portrait = {
		character = root
		animation = personality_greedy
	}
	right_portrait = {
		character = scope:defender
		animation = personality_callous
	}

	immediate = {
		if = {
			limit = { 
				OR = {
					scope:province = { NOT = { has_variable = POD_vampire_for_ancient_sorcery } } 
					scope:province.var:POD_vampire_for_ancient_sorcery ?= { is_alive = no }
				}
			}
			create_character = {
				culture = scope:province.culture
				faith = scope:province.faith
				dynasty = none
				location = root.location
				age = { 18 60 }
				gender_female_chance = 50
				save_scope_as = defender
				after_creation = {
					POD_set_resonance_effect = yes
					add_trait = vampire
					
					random_list = {
						20 = { 
							add_trait = vampgeneration6 set_age = { 1000 2000 } 
							add_prowess_skill = 20
						}
						40 = { 
							add_trait = vampgeneration7 set_age = { 400 999 } 
							add_prowess_skill = 15
						}
						30 = { 
							add_trait = vampgeneration8 set_age = { 200 399 } 
							add_prowess_skill = 10
						}
						10 = { 
							add_trait = vampgeneration9 set_age = { 18 199 } 
							add_prowess_skill = 5
						}
					}
					POD_character_vampire_clan_initialization_effect = yes
					POD_character_initialization_random_background_effect = yes
					POD_character_vampire_discipline_initialization_effect = yes
					POD_character_vampire_clan_traits_effect = yes
					POD_character_vampire_faith_initialization_effect = yes
					POD_vampires_look_younger_effect = yes
					trigger_event = POD_character_canon_modifier_initalization.100
				}
			}
			hidden_effect_new_object = {
				scope:province = {
					set_variable = { name = POD_vampire_for_ancient_sorcery value = scope:defender }
				}
			}
		}
		else = {
			scope:province.var:POD_vampire_for_ancient_sorcery = { save_scope_as = defender }
		}
	}

	option = {
		name = POD_exalted_ancient_sorcery_visit_events.3000.a
		configure_start_single_combat_effect = {
			SC_INITIATOR = root
			SC_ATTACKER = root
			SC_DEFENDER = scope:defender
			FATALITY = always
			FIXED = no
			LOCALE = alley_night
			OUTPUT_EVENT = POD_exalted_ancient_sorcery_visit_events.3001
			INVALIDATION_EVENT = fp1_tbc.0102
		}
	}
	option = {
		name = POD_exalted_ancient_sorcery_visit_events.3000.b
	}
}
