namespace = POD_patron_council

POD_patron_council.1 = {
	scope = none
	hidden = yes
	
	trigger = {
		POD_bookmark_has_fae = yes
	}
	
	immediate = {
		set_global_variable = {
			name = patron_council_enabled
			value = yes
		}
		set_global_variable = {
			name = patron_council_influence
			value = 0
		}
		character:POD_fae_character_26 = {
			set_patron_council_status_philospher_king_effect = yes 
			set_patron_council_status_seeker_effect = yes
		}
		
		# TEST
		while = {
			count = 6
			random_ruler = {
				limit = { 
					has_trait = fae 
					NOT = { faith = { has_doctrine = special_doctrine_fae_militants_hostility } }
					POD_is_patron_trigger = no
				}
				set_patron_council_status_seeker_effect = yes
			}
		}
		while = {
			count = 6
			random_ruler = {
				limit = { 
					has_trait = fae 
					NOT = { faith = { has_doctrine = special_doctrine_fae_militants_hostility } }
					POD_is_patron_trigger = no
				}
				set_patron_council_status_vanguard_effect = yes
			}
		}
		while = {
			count = 6
			random_ruler = {
				limit = { 
					has_trait = fae 
					NOT = { faith = { has_doctrine = special_doctrine_fae_militants_hostility } }
					POD_is_patron_trigger = no
				}
				set_patron_council_status_austere_conclave_effect = yes
			}
		}
	}
}

POD_patron_council.2 = {
	scope = none
	hidden = yes
	
	trigger = {
		has_global_variable = patron_council_enabled
	}

	immediate = {
		change_global_variable = {
			name = patron_council_influence
			add = patron_council_influence_value
		}
	}
}

POD_patron_council.10 = {
	scope = none
	hidden = yes
	
	trigger = { has_global_variable = patron_council_enabled }
	
	immediate = {
		random_ruler = {
			limit = { POD_is_patron_trigger = yes }
			set_patron_council_status_philospher_king_effect = yes
			weight = {
				base = 0
				candidate_chance_modifier = yes
			}
		}
		trigger_event = {
			id = POD_patron_council.11
			days = 1
		}
	}
}

POD_patron_council.11 = {
	scope = none
	hidden = yes
	
	trigger = { global_var:patron_council_philospher_king = { is_alive = no } }
	
	immediate = {
		remove_global_variable = patron_council_philospher_king
		trigger_event = POD_patron_council.10
	}
}
