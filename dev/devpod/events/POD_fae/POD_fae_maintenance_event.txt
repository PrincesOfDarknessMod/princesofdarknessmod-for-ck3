POD_fae_maintenance.1 = {
	type = character_event
	content_source = dlc_POD
	title = POD_fae_maintenance.1.t
	desc = POD_fae_maintenance.1.desc

	theme = faith
	left_portrait = {
		character = root
		animation = shock
	}
	
	trigger = {
		is_alive = yes
		has_trait = kinain
		NOT = { has_trait = fera }
	}
	
	option = {
		name = POD_fae_maintenance.1.option.a	
		
		remove_trait = kinain
		
		if = {
			limit = { POD_splat_mixing_haight_mode = no }
			POD_stop_being_ghoul = yes
		}
		remove_trait = revenant
		
		if = {
			limit = { has_trait = supehunter }
			hidden_effect = { 
				remove_trait = supehunter
				save_temporary_scope_as = is_sanbenito
			}
		}
	
		add_trait_force_tooltip = changeling
		POD_remove_wrong_supernatural_traits_on_existing_character = yes
			
		POD_fae_initialization_effect = yes
		POD_realm_initiation_effect = yes
		POD_character_fae_art_initialization_effect = yes
		POD_character_fae_lifestyle_initialization_effect = yes
		POD_fae_dynasty_trait_assign_effect = yes

		hidden_effect = {
			if = {
				limit = { exists = scope:is_sanbenito }
				add_trait = supehunter
				set_character_faith = faith:sanbenito
			}
		}
		
		if = {
			limit = {
				exists = mother
				mother = {
					is_landed = yes
					primary_heir ?= { has_trait = kinain }
					is_ai = yes
				}
			}
			mother = { set_designated_heir = prev }
		}
		else_if = {
			limit = {
				exists = father
				father = {
					is_landed = yes
					primary_heir ?= { has_trait = kinain }
					is_ai = yes
				}
			}
			father = { set_designated_heir = prev }
		}	
	}
}

POD_fae_maintenance.20 = {
	hidden = yes
	scope = none
	
	immediate = {
		every_living_character = {
			limit = {
				OR = {
					has_trait = fae 
					has_trait = changeling
				}
				NOT = { has_trait = phyla }
				NOT = { has_trait = hsien }
			}
			switch = {
				trigger = has_trait
				fae = { trigger_event = POD_fae_maintenance.21 }
				changeling = { trigger_event = POD_fae_maintenance.22 }
			}
			trigger_event = POD_fae_maintenance.23
		}
		trigger_event = {
			id = POD_fae_maintenance.30
			years = 100
		}

		set_global_variable = {
			name = fae_era
			value = 3 # Shattering
		}
		if = {
			limit = { exists = situation:PoD_war_of_seasons }
			situation:PoD_war_of_seasons = {
				end_situation = yes
			}
		}
		
		random_ruler = {
			limit = { has_POD_fae_blood_trigger = yes }
			create_artifact_eye_of_opening_effect = { OWNER = this }
		}
		random_ruler = {
			limit = { has_POD_fae_blood_trigger = yes }
			create_artifact_waystone_effect = { OWNER = this }
		}
		random_ruler = {
			limit = { has_POD_fae_blood_trigger = yes }
			create_artifact_changestone_effect = { OWNER = this }
		}
		random_ruler = {
			limit = { has_POD_fae_blood_trigger = yes }
			create_artifact_shadowstone_effect = { OWNER = this }
		}
	}
}

POD_fae_maintenance.21 = {
	type = character_event
	content_source = dlc_POD
	title = POD_fae_maintenance.21.t
	desc = POD_fae_maintenance.21.desc
	theme = fae
	left_portrait = root

	option = { # So what?
		name = POD_fae_maintenance.21.a

		ai_chance = { base = 0 }
	}

	option = {
		name = POD_fae_maintenance.21.b
		remove_trait = fae
		add_trait = changeling
		add_character_modifier = POD_desprete_for_immortality
		
		random_courtier_or_guest = {
			limit = { has_trait = mortal }
			death = { 
				death_reason = death_souleaten 
				killer = root
			}
		}
		
		ai_chance = {
			base = 100
			modifier = {
				add = -99
				is_ruler = yes
			}
			modifier = {
				factor = 0
				this = character:POD_fae_character_08
			}
		}
	}

	option = {
		name = POD_fae_maintenance.21.c
		trigger = {
			is_imprisoned = no
		}
		# bandaid fix to make sure characters in arcadia don't end up with titular/hof titles
		every_held_title = {
			limit = {
				OR = {
					is_titular = yes
					is_head_of_faith = yes
				}
			}
			prev = { destroy_title = prev }	
		}
		POD_long_depose_effect = yes
		POD_send_to_arcadia_effect = yes
		
		if = {
			limit = {
				is_landed = yes 
			}
			random_neighboring_and_across_water_top_liege_realm_owner = {
				limit = {
					NOT = { has_trait = fae }
					NOT = { has_trait = changeling }
				}
				save_temporary_scope_as = canon_inheritor
			}
			POD_transfer_all_titles_to_character = {
				CHARACTER = scope:canon_inheritor
			}
		}
		
		ai_chance = {
			base = 90
			modifier = {
				add = -89
				is_ruler = no
			}
			modifier = {
				factor = 0
				this = character:POD_fae_character_08
			}
		}
	}

	option = {
		name = POD_fae_maintenance.21.d
		depose_effect = { DEPOSER = root }
		add_trait = torpor
		add_trait = lunatic_1
		add_trait = reclusive
		add_character_flag = POD_do_not_disturb_flag
		
		if = {
			limit = { this = character:POD_fae_character_08 }
			change_first_name = "Meilseoir"
		}

		if = {
			limit = {
				is_landed = yes 
			}
			random_neighboring_and_across_water_top_liege_realm_owner = {
				limit = {
					NOT = { has_trait = fae }
					NOT = { has_trait = changeling }
				}
				save_temporary_scope_as = canon_inheritor
			}
			POD_transfer_all_titles_to_character = {
				CHARACTER = scope:canon_inheritor
			}
		}
		
		ai_chance = {
			base = 10
			modifier = {
				add = -9
				is_ruler = no
			}
		}
	}
	
}

POD_fae_maintenance.23 = {
	hidden = yes

	immediate = {
		if = {
			limit = { has_character_modifier = POD_shattering_1_modifier }
			remove_character_modifier = POD_shattering_1_modifier
			add_character_modifier = { 
				modifier = POD_shattering_2_modifier 
				years = 50
			}
		}
		else_if = {
			limit = { has_character_modifier = POD_shattering_2_modifier }
			remove_character_modifier = POD_shattering_2_modifier
			add_character_modifier = { 
				modifier = POD_shattering_3_modifier 
				years = 50
			}
		}
		else_if = {
			limit = { has_character_modifier = POD_shattering_3_modifier }
			remove_character_modifier = POD_shattering_3_modifier
			add_character_modifier = { 
				modifier = POD_shattering_4_modifier 
				years = 50
			}
		}
		else_if = {
			limit = { has_character_modifier = POD_shattering_4_modifier }
			remove_character_modifier = POD_shattering_4_modifier
		}
		else = {
			add_character_modifier = { 
				modifier = POD_shattering_1_modifier 
				years = 50
			}
		}
	}
}
