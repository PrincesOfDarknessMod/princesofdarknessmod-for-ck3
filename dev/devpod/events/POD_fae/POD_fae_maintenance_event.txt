namespace = POD_fae_maintenance

POD_fae_maintenance.1 = {
	type = character_event
	title = POD_fae_maintenance.1.t
	desc = POD_fae_maintenance.1.desc

	theme = faith
	left_portrait = {
		character = root
		animation = shock
	}
	
	trigger = {
		has_trait = kinain 
	}
	
	option = {
		name = POD_fae_maintenance.1.option.a	
		
		if = {
			limit = {
				has_trait = kinain 
			}
			remove_trait = kinain
		}
		
		if = {
			limit = {
				has_trait = ghoul 
			}
			remove_trait = ghoul
			every_relation = {
				type = ghoul_domitor
				remove_hook = {
					target = scope:victim
					type = strong_bloodoath_hook
				}
				scope:victim = {
					remove_relation_ghoul_domitor = prev
					remove_character_flag = blocked_from_leaving
					remove_character_flag = blood_bonded
				}
			}
		}
		if = {
			limit = {
				has_trait = revenant 
			}
			remove_trait = revenant
		}	
	
		add_trait_force_tooltip = changeling
		POD_remove_wrong_supernatural_traits_effect = yes
		POD_remove_wrong_supernatural_modifiers_effect = yes
			
		POD_fae_initialization_effect = yes
		POD_realm_initiation_effect = yes
		POD_character_fae_art_initialization_effect = yes
		POD_character_fae_lifestyle_initialization_effect = yes
		POD_fae_dynasty_trait_assign_effect = yes

		if = {
			limit = {
				exists = mother
				mother = {
					any_heir = { has_trait = kinain }					
					is_landed = yes
					is_ai = yes
				}
			}
			mother = { set_designated_heir = prev }
		}
		else_if = {
			limit = {
				exists = father
				father = {
					any_heir = { has_trait = kinain }					
					is_landed = yes
					is_ai = yes
				}
			}
			father = { set_designated_heir = prev }
		}	
	}
}

POD_fae_maintenance.2 = {
	hidden = yes
	type = character_event
	
	trigger = {
		is_ai = no
		has_POD_fae_blood_trigger = yes
	}
	
	immediate = {
		every_character_artifact = {
			limit = { 
				has_variable = active_effigy
				var:active_effigy = yes
			}
			root = { 
				send_interface_message = {
					type = event_generic_bad
					title = POD_effigy_maintenance
					right_icon = root
					custom_tooltip = POD_effigy_maintenance.tt
					if = {
						limit = { glamour_level >= 1 }
						POD_cantrip_cost_effect = { COUNT = 1 }
					}
					else = {
						prev = { degrade_mummy_artifact_rarity_effect = yes }
					}
				}
			}
		}
	}
}

POD_fae_maintenance.20 = {
	hidden = yes
	type = empty
	
	trigger = {
		NOT = { has_global_variable = shattering_prevented }
	}
	
	immediate = {
		every_living_character = {
			limit = { has_trait = fae }
			trigger_event = POD_fae_maintenance.21
		}
		every_living_character = {
			limit = { has_trait = changeling }
			trigger_event = POD_fae_maintenance.22
		}
		set_global_variable = shattering
	}
}

POD_fae_maintenance.21 = {
	type = character_event
	title = POD_fae_maintenance.21.t
	desc = POD_fae_maintenance.21.desc
	theme = fae
	left_portrait = root

	option = {
		name = POD_fae_maintenance.21.a
		remove_trait = fae
		add_trait = changeling
		
		ai_chance = {
			base = 100
			modifier = {
				add = -99
				is_ruler = yes
			}
			modifier = {
				factor = 0
				this = character:POD_fae_character_08
			}
		}
	}

	option = {
		name = POD_fae_maintenance.21.b
		depose = yes
		add_trait = torpor
		# remove_variable = spirit_destination
		# set_variable = { name = spirit_location value = flag:arcadia }
		add_character_modifier = trapped_in_arcadia_modifier
		
		ai_chance = {
			base = 90
			modifier = {
				add = -89
				is_ruler = no
			}
			modifier = {
				factor = 0
				this = character:POD_fae_character_08
			}
		}
	}

	option = {
		name = POD_fae_maintenance.21.c
		depose = yes
        add_trait = torpor
		add_trait = lunatic_1
		add_trait = reclusive
		add_character_flag = POD_do_not_disturb_flag
		
		if = {
			limit = { this = character:POD_fae_character_08 }
			change_first_name = "Meilseoir"
		}
		
		ai_chance = {
			base = 10
			modifier = {
				add = -9
				is_ruler = no
			}
		}
	}
}
