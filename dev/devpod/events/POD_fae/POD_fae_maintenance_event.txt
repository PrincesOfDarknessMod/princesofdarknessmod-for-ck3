namespace = POD_fae_maintenance

POD_fae_maintenance.1 = {
	type = character_event
	title = POD_fae_maintenance.1.t
	desc = POD_fae_maintenance.1.desc

	theme = faith
	left_portrait = {
		character = root
		animation = shock
	}
	
	trigger = {
		has_trait = kinain 
	}
	
	option = {
		name = POD_fae_maintenance.1.option.a	
		
		if = {
			limit = {
				has_trait = kinain 
			}
			remove_trait = kinain
		}
		
		if = {
			limit = {
				has_trait = ghoul 
			}
			remove_trait = ghoul
			every_relation = {
				type = ghoul_domitor
				remove_hook = {
					target = scope:victim
					type = strong_bloodoath_hook
				}
				scope:victim = {
					remove_relation_ghoul_domitor = prev
					remove_character_flag = blocked_from_leaving
					remove_character_flag = blood_bonded
				}
			}
		}
		if = {
			limit = {
				has_trait = revenant 
			}
			remove_trait = revenant
		}
			
		add_trait = changeling
		POD_changeling_initialization_effect = yes

		if = {
			limit = {
				exists = mother
				mother = {
					any_heir = { has_trait = kinain }					
					is_landed = yes
					is_ai = yes
				}
			}
			mother = { set_designated_heir = prev }
		}
		else_if = {
			limit = {
				exists = father
				father = {
					any_heir = { has_trait = kinain }					
					is_landed = yes
					is_ai = yes
				}
			}
			father = { set_designated_heir = prev }
		}	
	}
}

POD_fae_maintenance.2 = {
	hidden = yes
	type = character_event
	
	trigger = {
		is_ai = no
		has_POD_fae_blood_trigger = yes
	}
	
	immediate = {
		every_character_artifact = {
			limit = { 
				has_variable = active_effigy
				var:active_effigy = yes
			}
			root = { 
				send_interface_message = {
					type = event_generic_bad
					title = POD_effigy_maintenance
					right_icon = root
					custom_tooltip = POD_effigy_maintenance.tt
					if = {
						limit = { glamour_level >= 1 }
						POD_glamour_decrease_effect = yes
					}
					else = {
						prev = { degrade_mummy_artifact_rarity_effect = yes }
					}
				}
			}
		}
	}
}
