namespace = POD_fae_maintenance

POD_fae_maintenance.1 = {
	type = character_event
	title = POD_fae_maintenance.1.t
	desc = POD_fae_maintenance.1.desc

	theme = faith
	left_portrait = {
		character = root
		animation = shock
	}
	
	trigger = {
		has_trait = kinain 
	}
	
	option = {
		name = POD_fae_maintenance.1.option.a	
		
		if = {
			limit = {
				has_trait = kinain 
			}
			remove_trait = kinain
		}
		
		if = {
			limit = {
				has_trait = ghoul 
			}
			remove_trait = ghoul
			every_relation = {
				type = ghoul_domitor
				remove_hook = {
					target = scope:victim
					type = strong_bloodoath_hook
				}
				scope:victim = {
					remove_relation_ghoul_domitor = prev
					remove_character_flag = blocked_from_leaving
					remove_character_flag = blood_bonded
				}
			}
		}
		if = {
			limit = {
				has_trait = revenant 
			}
			remove_trait = revenant
		}
			
		add_trait = changeling

		if = {
			limit = {
				exists = mother
				mother = {
					any_heir = { has_trait = kinain }					
					is_landed = yes
					is_ai = yes
				}
			}
			mother = { set_designated_heir = prev }
		}
		else_if = {
			limit = {
				exists = father
				father = {
					any_heir = { has_trait = kinain }					
					is_landed = yes
					is_ai = yes
				}
			}
			father = { set_designated_heir = prev }
		}	
	}
}
