					multiplier = high_positive_duel_skill_multiplier

				custom_tooltip = POD_fae_interactions.11.option.a.tt.success
				
				hidden_effect = {
					scope:cloned_character = { POD_create_phantom_shadow_effect = yes }
					scope:phantom_shadow = { 
						set_employer = root 
						add_character_flag = blocked_from_leaving
					}
					add_hook = {
						target = scope:phantom_shadow
						type = phantom_shadow_hook

					value = scope:duel_value

				}
				
				custom_tooltip = POD_fae_interactions.11.option.a.tt.failure
				
			    hidden_effect = {
				    trigger_event = {
					    id = POD_fae_interactions.13
						days = 2
					}
			    }
			}
		}
		add_piety = -800
	}
	
	option = {
