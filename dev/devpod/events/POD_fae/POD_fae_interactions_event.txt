	theme = fae

	    character = root
		animation = personality_bold 
	}
	
	right_portrait = {
	    character = scope:cloned_character
		animation = stress 
	}
	
	immediate = {
		
	}
	
	option = {
	    name = POD_fae_interactions.11.option.a
		custom_tooltip = POD_fae_interactions.11.option.a.tt
		
		duel = {
		    skill = learning
			value = 20
			desc = POD_fae_interactions.11.option.a.desc
		
			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = high_positive_duel_skill_multiplier
				}
				
				custom_tooltip = POD_fae_interactions.11.option.a.tt.success
				
				hidden_effect = {
					scope:cloned_character = { POD_create_phantom_shadow_effect = yes }
					scope:phantom_shadow = { 
						set_employer = root 
						add_character_flag = blocked_from_leaving
					}
					add_hook = {
						target = scope:phantom_shadow
						type = phantom_shadow_hook
					}
					trigger_event = {
					    id = POD_fae_interactions.12
						days = 2
					}
			    }
			}    
			
			50 = {

				compare_modifier = {
					value = scope:duel_value
					multiplier = low_negative_duel_skill_multiplier
				}
				
				custom_tooltip = POD_fae_interactions.11.option.a.tt.failure
				
			    hidden_effect = {
				    trigger_event = {
					    id = POD_fae_interactions.13
						days = 2
					}
			    }
			}
		}
		add_piety = -800
	}
	
	option = {
	    name = POD_fae_interactions.11.option.b
		custom_tooltip = POD_fae_interactions.11.option.b.tt
	}
}

POD_fae_interactions.100 = {
    type = character_event
	hidden = yes
	
	immediate = {
		if = {
			limit = { scope:actor = scope:sc_victor }
			scope:actor = { trigger_event = POD_fae_interactions.101 }
		}
		else = {
			scope:actor = {
				random_equipped_character_artifact = {
					limit = { has_artifact_modifier = singing_sword_of_madog_cynydd_modifier }
					set_owner = {
						target = scope:recipient
						history = {
							location = scope:recipient.location
							actor = scope:actor
							recipient = scope:recipient
							type = conquest
						}
					}
				}
				death = {
	        		death_reason = death_duel
					killer = scope:recipient
	        	}
			}
		}
	}
}
