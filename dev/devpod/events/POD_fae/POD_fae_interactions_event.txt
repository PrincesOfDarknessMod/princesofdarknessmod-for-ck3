POD_fae_interactions.300 = {
	type = character_event
	content_source = dlc_POD
	title = POD_fae_interactions.300.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:murder_method = flag:hired_thugs }
				desc = POD_fae_interactions.300.desc.thugs
			}
			triggered_desc = {
				trigger = { scope:murder_method = flag:riled_up_mob }
				desc = POD_fae_interactions.300.desc.mob
			}
			triggered_desc = {
				trigger = { scope:murder_method = flag:lured_into_forest }
				desc = POD_fae_interactions.300.desc.lured_into_forest
			}
			triggered_desc = {
				trigger = { scope:murder_method = flag:child_pushed }
				desc = POD_fae_interactions.300.desc.child_pushed
			}
			desc =  POD_fae_interactions.300.desc.assassin
		}
		triggered_desc = {
			trigger = { exists = local_var:oath_guardian_dies }
			desc = POD_fae_interactions.300.oath_guardian_dies
		}
	}
	theme = murder_scheme
	left_portrait = {
		character = scope:target
		animation = shock
	}
	right_portrait = {
		character = scope:oath_guardian
		animation = rage
	}
	widget = {
		gui = "event_window_widget_scheme"
		container = "custom_widgets_container"
	}
	
	trigger = {
		OR = {
			scope:murder_method = flag:hired_assassin
			scope:murder_method = flag:hired_thugs
			scope:murder_method = flag:riled_up_mob
			scope:murder_method = flag:lured_into_forest
			scope:murder_method = flag:child_pushed
		}
		scope:target = {
			has_variable = oath_guardian
			var:oath_guardian = {
				save_temporary_scope_as = prospective_oath_guardian
				NOR = {
					any_scheme_agent = { this = scope:prospective_oath_guardian }
					root = scope:prospective_oath_guardian
				}
			}
		}
	}

	weight_multiplier = {
		base = 1
	}

	immediate = {
		save_scope_value_as = {
			name = murder_saved
			value = flag:oath_guardian_stopped_assassin
		}
		scope:target = {
			var:oath_guardian = {
				save_scope_as = oath_guardian
			}
		}
		hidden_effect = {
			if = {
				limit = {
					NOR = {
						scope:murder_method = flag:lured_into_forest
						scope:murder_method = flag:child_pushed
					}
				}
				random = {
					chance = {
						value = 30
						subtract = scope:oath_guardian.prowess
					}
					set_local_variable = {
						name = oath_guardian_dies
						value = yes
					}
				}
			}
		}
		if = {
			limit = { exists = local_var:oath_guardian_dies }
			show_as_tooltip = {
				if = {
					limit = {
						exists = scope:scheme_discovered
					}
					known_murder_effect = {
						VICTIM = scope:oath_guardian
						MURDERER = root
						EXPOSER = scope:target
					}
				}
				else = {
					unknown_murder_effect = {
						VICTIM = scope:oath_guardian
						MURDERER = root
						REASON = death_duel
					}
				}
			}
		}
		random_dummy_gender_assassin_effect = yes
		murder_failure_effect = yes # Fires the response for the target
	}

	#It's over.
	option = {
		name = POD_fae_interactions.300.a
		show_as_tooltip = {
			if = {
				limit = { exists = scope:scheme_discovered }
				add_dread = medium_dread_gain
			}
			scope:scheme = {
				end_scheme = yes
			}
		}
		ai_chance = {
			base = 100
		}
	}

	#It's not over till I say it's over!
	option = {
		name = POD_fae_interactions.300.b
		show_as_tooltip = {
			if = {
				limit = { exists = scope:scheme_discovered }
				add_dread = medium_dread_gain
			}
			start_scheme = {
				type = murder
				target = scope:target
			}
		}
		hidden_effect = {
			save_scope_value_as = {
				name = restart_scheme
				value = yes
			}
		}
		ai_chance = {
			base = 0
		}
	}

	after = {
		hidden_effect = { trigger_event = murder_outcome.9999 }
	}
}

				desc = POD_fae_interactions.301.desc.assassin.oath_guardian_dies
			}
			triggered_desc = {
				trigger = {
					exists = local_var:oath_guardian_dies
					scope:murder_method = flag:hired_thugs
				}
				desc = POD_fae_interactions.301.desc.thugs.oath_guardian_dies
			}
			triggered_desc = {
				trigger = {
					exists = local_var:oath_guardian_dies
					scope:murder_method = flag:riled_up_mob
				}
				desc = POD_fae_interactions.301.desc.mob.oath_guardian_dies
			}
			triggered_desc = {
				trigger = {
					NOT = { exists = local_var:oath_guardian_dies }
					scope:murder_method = flag:riled_up_mob
				}
				desc = POD_fae_interactions.301.desc.mob.oath_guardian_lives
			}
		}
		# And who was responsible!?
		triggered_desc = {
			trigger = { exists = scope:scheme_discovered }
			desc = POD_fae_interactions.301.owner_discovered
		}
	}
	theme = murder_scheme
	override_background = { 
		trigger = { scope:murder_method = flag:hired_thugs }
		reference = wilderness_scope
	}
	override_background = { 
		trigger = {
			OR = {
				scope:murder_method = flag:riled_up_mob
				scope:murder_method = flag:child_pushed
			}
		}
		reference = courtyard
	}
	override_background = { 
		trigger = { scope:murder_method = flag:lured_into_forest }
		reference = wilderness_forest
	}

	left_portrait = {
		character = scope:oath_guardian
		animation = rage
	}
	right_portrait = {
		character = scope:owner_revealed
		animation = anger
	}

	trigger = {
		exists = scope:murder_saved
		scope:murder_saved = flag:oath_guardian_stopped_assassin
		scope:target = {
			NOT = { has_character_flag = recently_saved_by_oath_guardian }
		}
	}

	immediate = {
		add_character_flag = {
			flag = recently_saved_by_oath_guardian
			years = 10
		}
		if = {
			limit = { scope:murder_method = flag:hired_thugs }
			scope:home_province = {	save_scope_as = background_wilderness_scope }
		}
		# Handle the death of the oath_guardian if necessary
		if = {
			limit = { exists = local_var:oath_guardian_dies }
			if = {
				limit = {
					exists = scope:scheme_discovered
				}
				known_murder_effect = {
					VICTIM = scope:oath_guardian
					MURDERER = root
					EXPOSER = scope:target
				}
			}
			else = {
				unknown_murder_effect = {
					VICTIM = scope:oath_guardian
					MURDERER = root
					REASON = death_duel
				}
			}
		}
	}

	#Who could do such a thing?
	option = {
		name = POD_fae_interactions.301.a
		trigger = {
			NOT = {
				exists = scope:scheme_discovered
			}
		}
		custom_tooltip = murder_save.failure_unknown_owner_tt
	}

	#Vengeance!
	option = {
		name = POD_fae_interactions.301.b
		trigger = {
			exists = scope:scheme_discovered
		}
		custom_tooltip = murder_save.failure_known_owner_tt
	}

	after = {
		add_character_modifier = {
			modifier = watchful_modifier
			days = watchful_modifier_duration
		}
	}
}
