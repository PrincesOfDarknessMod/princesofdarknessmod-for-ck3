namespace = POD_fae_interactions


POD_fae_interactions.100 = {
	hidden = yes
	
	immediate = {
		if = {
			limit = { scope:actor = scope:sc_victor }
			scope:actor = { trigger_event = POD_fae_interactions.101 }
		}
		else = {
			scope:actor = {
				random_equipped_character_artifact = {
					limit = { has_variable = singing_sword_of_madog_cynydd }
					set_owner = {
						target = scope:recipient
						history = {
							location = scope:recipient.location
							actor = scope:actor
							recipient = scope:recipient
							type = conquest
						}
					}
				}
				death = {
	        		death_reason = death_duel
					killer = scope:recipient
	        	}
			}
		}
	}
}

POD_fae_interactions.101 = {
    type = character_event
	content_source = dlc_POD
	title = POD_fae_interactions.101.t
	desc = POD_fae_interactions.101.desc
	theme = fae
	
	option = {
	    name = POD_fae_interactions.101.option.a
		scope:recipient = {
			death = {
				death_reason = death_duel
				killer = scope:actor
			}
		}
		scope:actor = {
			random_equipped_character_artifact = { 
				limit = { has_variable = singing_sword_of_madog_cynydd }
				POD_add_modifier_and_track_stacks = {
					TYPE = artifact
					MODIFIER = artifact_life_expectancy_40_modifier
				}
			}
		}
	}
}
