					scope:cloned_character = { POD_create_phantom_shadow_effect = yes }

						set_employer = root

					}
					add_hook = {
						target = scope:phantom_shadow
						type = phantom_shadow_hook
					}

					    id = POD_fae_interactions.12

					}
			    }
			}    
			
			50 = {

				compare_modifier = {
					value = scope:duel_value
					multiplier = low_negative_duel_skill_multiplier
				}
				
				custom_tooltip = POD_fae_interactions.11.option.a.tt.failure
				
			    hidden_effect = {
				    trigger_event = {
					    id = POD_fae_interactions.13
						days = 2
					}
			    }
			}
		}
		add_piety = -800
	}
	
	option = {
	    name = POD_fae_interactions.11.option.b
