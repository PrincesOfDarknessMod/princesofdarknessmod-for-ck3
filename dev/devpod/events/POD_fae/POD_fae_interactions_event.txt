					scope:cloned_character = { POD_create_phantom_shadow_effect = yes }

					value = scope:duel_value

				    trigger_event = {
					    id = POD_fae_interactions.13

					}

			}
		}

POD_fae_interactions.100 = {
    type = character_event
	hidden = yes
	
	immediate = {
		if = {
			limit = { scope:actor = scope:sc_victor }

		}

			scope:actor = {
				random_equipped_character_artifact = {

	        	}
			}
		}
	}
