POD_fae_interactions.100 = {
	hidden = yes
	
	immediate = {
		if = {
			limit = { scope:actor = scope:sc_victor }
			scope:actor = { trigger_event = POD_fae_interactions.101 }
		}
		else = {
			scope:actor = {
				random_equipped_character_artifact = {
					limit = { has_variable = singing_sword_of_madog_cynydd }
					set_owner = {
						target = scope:recipient
						history = {
							location = scope:recipient.location
							actor = scope:actor
							recipient = scope:recipient
							type = conquest
						}
					}
				}
				death = {
	        		death_reason = death_duel
					killer = scope:recipient
	        	}
			}
		}
	}
}

POD_fae_interactions.600 = {
	hidden = yes
	
	immediate = {
		create_character = {
			faith = faith:wyrmbound
			culture = root.culture
			gender_female_chance = root.var:character_creator_sex
			age = root.var:character_creator_age
			diplomacy = root.var:character_creator_diplomacy
			martial = root.var:character_creator_martial
			stewardship = root.var:character_creator_stewardship
			intrigue = root.var:character_creator_intrigue
			learning = root.var:character_creator_learning
			prowess = root.var:character_creator_prowess
			random_traits = no
			trait = spirit
			dynasty = none
			employer = root
			save_scope_as = optional_child
			after_creation = {
				add_character_flag = optional_child
			}
		}
		POD_remove_character_creator_variable_effect = yes
		set_variable = {
			name = reweaving_recipient
			value = scope:optional_child
		}
		trigger_event = {
			id = POD_fae_interactions.601
			days = 1
		}
	}
}

POD_fae_interactions.601 = {
	type = character_event
	content_source = dlc_POD
	title = POD_fae_interactions.601.t
	desc = POD_fae_interactions.601.desc
	theme = dynasty 
	
	left_portrait = {
		character = root
		animation = idle
	}
	
	right_portrait = {
		character = scope:optional_child
		animation = idle
	}	

	widgets = {
		widget = {
			gui = "event_window_widget_name_child"
			container = "dynamic_birth_name"
			controller = name_character
			setup_scope = { scope:optional_child = { save_scope_as = name_character_target } }
		}
	}

	# Adopt
	option = {
		name = POD_fae_interactions.601.a			

		scope:optional_child = {
			remove_trait = spirit
			add_trait = fae
		}
		
		hidden_effect = {
			scope:optional_child = {
				if = {
					limit = { root = { is_male = yes } }
					set_father = root
					set_house = father.house
				}
				else = {
					set_mother = root
					set_house = mother.house
				}
				
				POD_fae_initialization_effect = yes
			}
		
			reverse_add_opinion = {
				target = scope:optional_child
				modifier = hired_me_opinion
			}
		}
		
		trigger_event = {
			on_action = fae_personalty_education_random_pulse
			days = 1
		}
	}
}
