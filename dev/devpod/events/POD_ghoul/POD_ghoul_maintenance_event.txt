	hidden = yes

		is_ai = yes
		POD_is_ghoul_trigger = yes
	}

    immediate = {
		POD_cease_being_ghoul_effect = yes
		if = {
			limit = { has_POD_race_trigger = no }
			add_trait = mortal
			if = {
				limit = { can_hold_title_as_mortal_trigger = no }
				POD_depose_effect = yes
			}
		}

		if = {
			limit = {
				age > 99
				is_immortal = no
			}
			death = { death_reason = death_old_age }
		}
	}
}

POD_ghoul_maintenance.300 = {
	type = character_event
	content_source = dlc_POD
	title = POD_ghoul_maintenance.300.t
	desc = POD_ghoul_maintenance.300.desc

	theme = vampire
	right_portrait = root

	widget = {
		gui = "POD_shader_event_fog_alley_night"
		container = "background_shader_vfx_container"
	}

#	trigger = {
#		has_character_flag = POD_will_become_revenant
#	}

	option = {
		name = POD_ghoul_maintenance.300.option.a

		hidden_effect = { remove_trait = mortal }
		add_trait = revenant

		if = {
			limit = {
				POD_is_supehunter_trigger = no
				POD_splat_is_fera_trigger = no
			}
			if = {
				limit = {
					NOT = { religion = { is_in_family = rf_kindred } }
				}
				POD_character_revenant_faith_initialization_effect = yes
			}
			if = {
				limit = {
					has_masquerade_risk_trigger = no
					POD_splat_has_masquerade_trigger = yes
				}
				POD_character_masquerade_initialization_effect = yes
			}
		}
		POD_character_revenant_discipline_initialization_effect = yes
		if = {
			limit = { POD_has_any_blood_hunger_modifier = no }
			add_character_modifier = bloodhunger0_modifier
		}

		if = {
			limit = { has_any_blood_potency_modifier = no }
			add_character_modifier = bloodpotency0_modifier
		}

		add_character_flag = "had_POD_character_initialization"

	}
}
