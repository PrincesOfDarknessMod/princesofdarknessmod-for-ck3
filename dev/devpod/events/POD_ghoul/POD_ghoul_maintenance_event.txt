# Fallback in case you chose to become a bound ghoul in the character creator but failed to pick a master in time
# (masterless ghoul setup)
POD_ghoul_maintenance.3 = {
	hidden = yes
	
	trigger = {
		POD_is_ghoul_trigger = yes
		POD_is_bound_ghoul_trigger = no
		NOT = { has_character_flag = charjoin_become_ghoul_flag }
	}
	
	immediate = {
		if = {
			limit = {
				OR = {
					NOR = {
						religion = { is_in_family = rf_kindred }
						POD_has_valid_splat_agnostic_faith = yes
					}
					# Character generator sets you to Via Humanitatis by default
					# make sure you still get to choose your faith in that case
					AND = {
						has_character_flag = POD_used_character_generator
						has_faith = faith:viahumanitatis
					}
				}
			}
			# set a vampire religion beforehand so the masquerade modifier gets initialized properly in POD_become_ghoul_effect
			POD_character_revenant_faith_initialization_effect = yes
			trigger_event = POD_chargen.710
		}
	
		POD_become_ghoul_effect = yes
		
		# If you managed to roll no disciplines (unlucky!), get Potence as a consolation prize
		if = {
			limit = { has_POD_vampire_discipline_trigger = no }
			add_trait = potence
		}
	
		if = {
			limit = { POD_has_illegal_government_trigger = yes }
			POD_adopt_correct_government_effect = yes
		}
		
		trigger_event = {
			id = POD_ghoul_maintenance.2
			days = 1
		}
	}
}

# die after a failed ghoul mission if your domitor had low opinion
# on a 1-day delay because otherwise the game crashes
# "it just works" - todd howard
POD_ghoul_maintenance.4 = {
	hidden = yes
	
	immediate = {
		pod_death = {
			death_reason = death_murder
			killer = scope:domitor
		}
		# hidden clause: if you had a resurrection ability, become a masterless ghoul after you "die"
		# because your domitor left you for dead
		hidden_effect = {
			if = {
				limit = { is_alive = yes }
				POD_clear_bound_ghoul_variables_and_relations = yes
			}
		}
	}
}

	}

POD_ghoul_maintenance.300 = {
	type = character_event
	content_source = dlc_POD
	title = POD_ghoul_maintenance.300.t
	desc = POD_ghoul_maintenance.300.desc

	theme = vampire
	right_portrait = root

	widget = {
		gui = "POD_shader_event_fog_alley_night"
		container = "background_shader_vfx_container"
	}

#	trigger = {
#		has_character_flag = POD_will_become_revenant
#	}

	option = {
		name = POD_ghoul_maintenance.300.option.a

		hidden_effect = { remove_trait = mortal }
		add_trait = revenant

		if = {
			limit = {
				POD_is_supehunter_trigger = no
				POD_is_fera_trigger = no
			}
			if = {
				limit = {
					NOT = { religion = { is_in_family = rf_kindred } }
				}
				POD_character_revenant_faith_initialization_effect = yes
			}
			if = {
				limit = {
					has_masquerade_risk_trigger = no
					POD_splat_has_masquerade_trigger = yes
				}
				POD_character_masquerade_initialization_effect = yes
			}
		}
		POD_character_revenant_discipline_initialization_effect = yes
		if = {
			limit = { POD_has_any_blood_hunger_modifier = no }
			add_character_modifier = bloodhunger0_modifier
		}

		if = {
			limit = { has_any_blood_potency_modifier = no }
			add_character_modifier = bloodpotency0_modifier
		}

		add_character_flag = "had_POD_character_initialization"

	}
}
