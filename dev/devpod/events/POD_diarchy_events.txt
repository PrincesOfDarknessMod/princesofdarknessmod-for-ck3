#Setting up starting diarchies
POD_diarchy_events.0001 = {
	hidden = yes
	
	immediate = {
		start_diarchy = pod_power_struggle
		
		## Hardestadt is dominant
		if = {
			limit = { this = character:505050 }
			set_diarchy_swing = 35
		}
		
		## Salianna is dominant
		if = {
			limit = { this = character:505072 }
			set_diarchy_swing = 30
		}

		## balanced for qarakh/deverra
		if = {
			limit = { this = character:790105 }
			set_diarchy_swing = 50
		}

		## Moncada dominates
		if = {
			limit = { this = character:505126 }
			set_diarchy_swing = 75
		}

		## Fang Liwan basically runs the court
		if = {
			limit = { this = character:791114 }
			set_diarchy_swing = 90
		}

		## Yuri slightly dominates
		if = {
			limit = { this = character:792537 }
			set_diarchy_swing = 45
		}			
	}

}

#a power struggle starrts (potentially)
POD_diarchy_events.0002 = {
	type = character_event
	theme = realm
	title = POD_diarchy_events.0002.t
	desc = POD_diarchy_events.0002.desc
	left_portrait = {
		character = scope:potential_diarch
		animation = chancellor
	}
	right_portrait = {
		character = root
		animation = disgust
	}
	
	immediate = {
		save_scope_as = contested_liege
	}

	option = {
		name = POD_diarchy_events.0002.a	
		
		custom_tooltip = liege_warning_diarchy_tt
		
		add_prestige = -500
		add_tyranny = 30
		
		reverse_add_opinion = {
			target = scope:potential_diarch
			modifier = insult_opinion
			opinion = -25
		}
		add_opinion = {
			target = scope:potential_diarch
			modifier = insult_opinion
			opinion = -25
		}

		scope:potential_diarch = {
			trigger_event = POD_diarchy_events.0003
		}
		
		ai_chance = {
			base = 10
			modifier = {
				add = 50
				OR = {
					has_trait = ambitious
					has_trait = arbitrary
					has_trait = greedy
				}
			}			
		}	
	}
		
	option = {
		name = POD_diarchy_events.0002.b	
		
		custom_tooltip = liege_warning_diarchy_tt

		scope:potential_diarch = {
			add_character_flag = has_had_diarch_intro_event_flag
		}
		
		designate_diarch = scope:potential_diarch

		start_diarchy = pod_power_struggle	
		
		set_diarchy_swing = 50

		scope:potential_diarch = {
			trigger_event = POD_diarchy_events.0004
		}

		ai_chance = {
			base = 50
			modifier = {
				add = 50
				OR = {
					has_trait = content
					has_trait = just
					has_trait = generous
				}
			}
			modifier = {
				add = 50
				scope:potential_diarch = { is_ai = no }
			}			
		}		
	}
	
}
