#A councilor has died, tremere loyalists players get first pick at trying to become the next one
pod_tremere_council.30 = {
	type = character_event
	content_source = dlc_POD
	title = pod_tremere_council.30.t
	desc = pod_tremere_council.30.desc
	theme = tremere_coa
	
	option = {
		name = pod_ventrue_ephorate.30.a
		
		custom_tooltip = entry_tremere_council_tt
		
		if = { 
			limit = {
				OR = {
					highest_held_title_tier >= tier_empire
					sub_realm_size >= 100
					piety >= 1500
				}
			}
		remove_short_term_gold = 100
		add_piety = -100
		}		
		
		else_if = { 
			limit = {
				OR = {
					highest_held_title_tier = tier_kingdom
					sub_realm_size >= 50
					piety >= 1000
				}
			}
		remove_short_term_gold = 500
		add_piety = -500
		}

		else = {
			remove_short_term_gold = 1000
			add_piety = -1000						
		}
		
		save_scope_as = future_councilor
		
		trigger_event = {
			id = pod_tremere_council.4
			days = 5
		}		
	}

	option = {
		name = pod_ventrue_ephorate.30.b
		
		add_character_flag = {
			flag = refused_tremere_council_flag
			years = 5
		}
		
		trigger_event = pod_tremere_council.3
	}	
}
