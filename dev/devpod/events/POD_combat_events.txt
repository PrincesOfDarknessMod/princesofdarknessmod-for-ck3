# pod_combat.0200 - once-per-character intro from pod_combat_sc_intro_or_launch_effect.
# Inherits the launcher's scope table; Begin must continue within this event chain.
# No event trigger: a failed trigger would discard the duel and leave flags.
# Revalidate on click. Saves no scopes here.
pod_combat.0200 = {
	type = character_event
	content_source = dlc_POD
	title = pod_combat.0200.t
	desc = pod_combat.0200.desc
	theme = war
	override_background = { reference = velvet }

	widgets = {
		widget = {
			gui = "POD_event_window_widget_bnw_portrait"
			container = "foreground_shader_vfx_container"
		}
		# Pause while already-engaged principals await the intro; the widget handles the fallback.
		widget = {
			gui = "pod_combat_intro_pause"
			container = "custom_widgets_container"
		}
	}

	option = {
		name = pod_combat.0200.a
		custom_tooltip = pod_combat_intro_review_tt

		hidden_effect = {
			# Fire before opening the editor so its saved scopes do not enter the ready-bar chain.
			trigger_event = { id = pod_combat.0201 }
			pod_combat_open_loadout_editor_effect = yes
		}
	}

	option = {
		name = pod_combat.0200.b
		custom_tooltip = pod_combat_duel_begins_tt
		pod_combat_intro_begin_fight_effect = yes
	}
}

# pod_combat_debug.0001 - OUTPUT_EVENT for POD_debug_interactions.txt launchers.
# Clear vanilla duel shirtlessness when the minigame did not intercept.
# pod_combat.9999 already releases engaged_in_single_combat.

pod_combat_debug.0001 = {
	hidden = yes

	immediate = {
		single_combat_clean_shirtlessness_effect = {
			ATTACKER = scope:sc_attacker
			DEFENDER = scope:sc_defender
		}

		# Draws/mutual KOs can leave victor or loser unset; guards portrait dereferences.
		if = {
			limit = {
				exists = scope:sc_victor
				scope:sc_victor = root
				exists = scope:sc_loser
			}
			send_interface_toast = {
				title = pod_combat_debug_victory_toast
				left_icon = root
				right_icon = scope:sc_loser
			}
		}
		else_if = {
			limit = {
				exists = scope:sc_victor
				NOT = { scope:sc_victor = root }
			}
			send_interface_toast = {
				title = pod_combat_debug_defeat_toast
				left_icon = root
				right_icon = scope:sc_victor
			}
		}
		else = {
			send_interface_toast = {
				title = pod_combat_debug_draw_toast
				left_icon = root
			}
		}
	}
}

# pod_combat.9995 - shell fired immediately by POD_combat_create_duel_effect.
# Keep open throughout combat to retain the caller's scope, including typed
# flags and arbitrary aliases. GUI clicks cannot carry that scope themselves.
# gui/event_windows/pod_combat_shell.gui reveals the card after resolution.
pod_combat.9995 = {
	type = character_event
	content_source = dlc_POD
	window = pod_combat_shell
	title = pod_combat.9995.t
	desc = pod_combat.9995.desc
	theme = war

	# No option trigger: it would evaluate before resolution and stay false.
	# Gate readiness live in the GUI and recheck in the option effect.
	option = {
		name = pod_combat.9995.a
		custom_tooltip = pod_combat_shell_conclude_tt

		hidden_effect = {
			if = {
				# Early click or multiplayer timeout: reopen within the same scope chain.
				limit = {
					is_alive = yes
					pod_combat_shell_owns_live_fight_trigger = yes
					NOT = { pod_combat_is_over_trigger = yes }
				}
				trigger_event = { id = pod_combat.9995 }
			}
			else_if = {
				# Dispatch .9999 from this chain to preserve inherited scopes.
				limit = {
					is_alive = yes
					pod_combat_shell_owns_live_fight_trigger = yes
				}
				pod_combat_end_combat_effect = yes
			}
			# A stale card only closes; it must not conclude a later fight.
		}
	}
}

# pod_combat.9990 - to-the-death aftermath for PoD interaction launchers.
# Vanilla tribal.1021 requires liege-chain scopes and penalizes the wrong survivor.
# Use only guaranteed sc_* scopes; finalization already applied wounds and deaths.
pod_combat.9990 = {
	hidden = yes

	immediate = {
		single_combat_clean_shirtlessness_effect = {
			ATTACKER = scope:sc_attacker
			DEFENDER = scope:sc_defender
		}
		if = {
			limit = { exists = scope:sc_victor }
			scope:sc_victor = {
				send_interface_toast = {
					type = event_generic_neutral
					title = pod_combat_death_duel_toast
					left_icon = scope:sc_victor
					right_icon = scope:sc_loser
				}
			}
		}
	}
}

# pod_combat.9999 - post-combat dispatch from shell Conclude or raw GUI fallback.
# Only the shell route retains caller scope; see pod_combat_sc_context_effects.txt.
pod_combat.9999 = {
	hidden = yes

	immediate = {
		# Distinguish successful dispatch from an unreached cleanup path in debug.log.
		debug_log = "pod_combat: .9999 dispatch entered"

		# Bind the context before either route; guard absent/ended fight pointers.
		var:pod_combat_session ?= { save_scope_as = sc_ctx }

		# GUI fallback can restore principals, but cannot recover caller aliases.
		if = {
			limit = { NOT = { exists = scope:sc_attacker } }
			POD_combat_restore_sc_context_effect = yes
			POD_combat_restore_sc_rule_flags_effect = yes
			# A builder-created fight losing its shell table is a dispatch defect.
			if = {
				limit = { scope:sc_ctx ?= { has_variable = pod_combat_sc_shell } }
				error_log = "pod_combat: .9999 reached without the shell scope table - concluding degraded, parent-event aliases unavailable to the follow-up"
			}
		}

		# Restore principals before deriving victor/loser.
		POD_combat_derive_sc_outcome_effect = yes

		POD_combat_apply_duel_finalization_effect = yes

		# Preserve vanilla recipient precedence: living initiator, defender, then attacker.
		POD_combat_fire_sc_followup_effect = yes

		# Finalization can kill the host.
		if = {
			limit = { is_alive = yes }
			pod_combat_cleanup_effect = yes
		}
		else = {
			pod_combat_host_died_release_effect = yes
		}
	}
}
