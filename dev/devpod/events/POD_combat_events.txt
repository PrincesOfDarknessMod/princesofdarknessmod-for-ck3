# pod_combat.0200 - the first-fight intro, shown once per character. Fired from
# Part 3 of configure_start_single_combat_effect (through
# pod_combat_sc_intro_or_launch_effect) INSTEAD of launching the fight, so it
# arrives holding that chain's whole saved-scope table: the sc_* principals, the
# flag-typed rule values, scope:follow_up_event and scope:invalidation_event, and
# whatever the original caller named for its own follow-up. Begin launches the
# fight from inside this chain, which is what keeps the hand-off identical to an
# uninterrupted duel.
#
# No trigger block on purpose: the deferral has already happened by the time this
# fires, so a trigger that failed would swallow the fight entirely and strand
# engaged_in_single_combat on both principals. Eligibility is decided at the
# Part-3 gate, and the click re-validates.
#
# Nothing is saved here, and there is no immediate. Any scope this event named
# would ride the table into the shell card and on to the follow-up, where a
# generic name would shadow the parent's own alias.
#
# Multiplayer picks an option automatically after MULTIPLAYER_EVENT_TIME_OUT
# days, so Begin is listed first; the review option converges on the same launch
# anyway, since pod_combat.0201's only option is Begin.
pod_combat.0200 = {
	type = character_event
	content_source = dlc_POD	# PoD corner icon (workspace convention)
	title = pod_combat.0200.t
	desc = pod_combat.0200.desc
	theme = war

	left_portrait = {
		character = root
		animation = personality_rational
	}

	# Force-pause carrier. Both principals are already stamped
	# engaged_in_single_combat by Part 3, so the world should not tick on while
	# the player reads. See the widget file for the fallback if the state does
	# not run from here.
	widgets = {
		widget = {
			gui = "pod_combat_intro_pause"
			container = "custom_widgets_container"
		}
	}

	option = {
		name = pod_combat.0200.a
		custom_tooltip = pod_combat_duel_begins_tt
		pod_combat_intro_begin_fight_effect = yes
	}

	option = {
		name = pod_combat.0200.b
		custom_tooltip = pod_combat_intro_review_tt

		hidden_effect = {
			# Fired BEFORE the editor opens: trigger_event copies the scope table
			# at the call site, so anything the open effect saved would otherwise
			# ride into the bar's table. Same ordering discipline as the duel
			# builder opening its shell card first.
			trigger_event = { id = pod_combat.0201 }
			pod_combat_open_loadout_editor_effect = yes
		}
	}
}

# pod_combat.0299 - never fires in a healthy build. The deferred launch has to
# hand the builder a literal OUTPUT_EVENT (macro args cannot carry a runtime
# value), so it passes this placeholder and writes the real follow-up id onto
# the fight story immediately afterwards. Reaching this event means that write
# was lost, and the caller's own outcome handling has been skipped.
pod_combat.0299 = {
	hidden = yes

	immediate = {
		error_log = "pod_combat: intro placeholder follow-up fired - the deferred launch's follow_up restore did not run, so the caller's follow-up event was replaced by this stub"
	}
}

# Hidden follow-up (OUTPUT_EVENT) for the combat debug launchers in
# common/character_interactions/POD_debug_interactions.txt. Clears any
# shirtlessness PoD's vanilla duel body applied (only the three launchers
# routed through configure_start_single_combat_effect can reach it, when the
# minigame declines to intercept) and shows a result toast.
# engaged_in_single_combat is already released by pod_combat.9999 (the
# post-combat dispatcher), so this follow-up leaves it alone.

pod_combat_debug.0001 = {
	hidden = yes

	immediate = {
		single_combat_clean_shirtlessness_effect = {
			ATTACKER = scope:sc_attacker
			DEFENDER = scope:sc_defender
		}

		# Guard the scope derefs: a draw / mutual-KO can leave sc_victor and
		# sc_loser unset, and an unguarded right_icon on an unset scope errors.
		if = {
			limit = {
				exists = scope:sc_victor
				scope:sc_victor = root
				exists = scope:sc_loser
			}
			send_interface_toast = {
				title = pod_combat_debug_victory_toast
				left_icon = root
				right_icon = scope:sc_loser
			}
		}
		else_if = {
			limit = {
				exists = scope:sc_victor
				NOT = { scope:sc_victor = root }
			}
			send_interface_toast = {
				title = pod_combat_debug_defeat_toast
				left_icon = root
				right_icon = scope:sc_victor
			}
		}
		else = {
			# Single-portrait toast: no second icon to dereference.
			send_interface_toast = {
				title = pod_combat_debug_draw_toast
				left_icon = root
			}
		}
	}
}

# pod_combat.9995 - the duel shell. Fired without delay from the duel builder
# (POD_combat_create_duel_effect, opening block), so it inherits the
# launching event's entire saved-scope table:
# the sc_* principals, the fatality / fixed / locale rule flags, and whatever
# the parent named for its own follow-up (scope:spirit, scope:task_contract,
# an artifact scope:target, and the flag-typed values a variable stash cannot
# carry at all). Holding this event open for the duration of the fight is
# what keeps that table alive, since the combat window's clicks are GUI
# executions with no scope table of their own. Concluding from the option
# therefore hands off inside the original chain, and the follow-up plus
# everything downstream of it reads the parent's scopes exactly as an
# uninterrupted vanilla duel would have delivered them.
#
# The card is gui/event_windows/pod_combat_shell.gui. Its window stays hidden
# for the length of the fight and appears centred once the fight resolves; the
# EVENT is open throughout regardless, which is the part that matters.
pod_combat.9995 = {
	type = character_event
	content_source = dlc_POD	# PoD corner icon (workspace convention)
	window = pod_combat_shell
	title = pod_combat.9995.t
	desc = pod_combat.9995.desc
	theme = war

	# Deliberately no option trigger: option triggers are evaluated when the
	# event fires, and the fight is unresolved by definition at that moment,
	# so any trigger would leave the option invalid for the whole fight.
	# Readiness is gated in the window GUI, which re-evaluates live, and
	# re-checked below for the paths the GUI cannot gate.
	option = {
		name = pod_combat.9995.a
		custom_tooltip = pod_combat_shell_conclude_tt

		hidden_effect = {
			if = {
				# Selected before the fight resolved. Multiplayer picks an
				# option automatically after MULTIPLAYER_EVENT_TIME_OUT days,
				# and a stray click can still slip past the GUI gate, so
				# reopen rather than concluding early: the re-fire carries
				# the same scope table onward and nothing is lost.
				limit = {
					is_alive = yes
					pod_combat_shell_owns_live_fight_trigger = yes
					NOT = { pod_combat_is_over_trigger = yes }
				}
				trigger_event = { id = pod_combat.9995 }
			}
			else_if = {
				# The normal route. Writeback runs here and its sc branch
				# fires .9999 from inside this event chain, which is what
				# carries the inherited scopes into the follow-up.
				limit = {
					is_alive = yes
					pod_combat_shell_owns_live_fight_trigger = yes
				}
				pod_combat_end_combat_effect = yes
			}
			# Neither branch: this card has outlived its fight (the host died
			# mid-fight and the release tore it down, or a later fight is
			# running that this card does not own). Clicking just closes it.
		}
	}
}

# To-the-death aftermath for the PoD interaction launchers. Vanilla's
# tribal.1021 (the liege-challenge output) cannot serve them: it reads
# liege-chain scopes these launchers never set, so a win garbles the toast,
# and its challenger-lost branch docks the SURVIVING victor a prestige
# level when the attacker dies. This one touches only the guaranteed sc_*
# set. Deaths and wounds are already applied by the finalization effect;
# the toast is pure player feedback.
pod_combat.9990 = {
	hidden = yes

	immediate = {
		single_combat_clean_shirtlessness_effect = {
			ATTACKER = scope:sc_attacker
			DEFENDER = scope:sc_defender
		}
		if = {
			limit = { exists = scope:sc_victor }
			scope:sc_victor = {
				send_interface_toast = {
					type = event_generic_neutral
					title = pod_combat_death_duel_toast
					left_icon = scope:sc_victor
					right_icon = scope:sc_loser
				}
			}
		}
	}
}

# Post-combat dispatch: the single funnel where a duel-launched fight hands
# back to the parent chain. Two routes arrive here and differ only in whether
# the parent's scope table came with them (file header of
# pod_combat_sc_context_effects.txt has the full model):
#   - the shell event's Conclude option, carrying the parent scopes intact;
#   - a raw GUI execution, carrying nothing.
pod_combat.9999 = {
	hidden = yes

	immediate = {
		# Positive dispatch marker. Nothing else on the end/cleanup path writes a
		# log line, so "did .9999 fire" was previously only answerable from an
		# ABSENCE - and absence reads identically for "fired cleanly", "fired
		# degraded" and "never reached". A silent regression here takes the
		# follow-up, the fatal-duel deaths and the whole teardown with it.
		debug_log = "pod_combat: .9999 dispatch entered"

		# The fight story carries the follow-up id, and for the fallback
		# route the principals too, so bind it ahead of either branch.
		# ?=-guarded: this single deref gates both routes, and an unset or
		# already-ended pointer here would error rather than resolve to
		# nothing, taking the whole dispatch down with it.
		var:pod_combat_session ?= { save_scope_as = sc_ctx }

		# An absent sc_attacker means the shell's table did not come along.
		# Rebuild what the story can supply; parent aliases are gone for
		# good on this route, so the follow-up runs on principals alone.
		if = {
			limit = { NOT = { exists = scope:sc_attacker } }
			POD_combat_restore_sc_context_effect = yes
			POD_combat_restore_sc_rule_flags_effect = yes
			# Every fight built through POD_combat_create_duel_effect opens a
			# card and carries the marker, so this is the defensive arm: a
			# fight seated without the builder, concluding through the combat
			# window's scope-less button. A marked fight arriving here lost a
			# table it did have, which is a defect worth logging.
			if = {
				limit = { scope:sc_ctx ?= { has_variable = pod_combat_sc_shell } }
				error_log = "pod_combat: .9999 reached without the shell scope table - concluding degraded, parent-event aliases unavailable to the follow-up"
			}
		}

		# The outcome is combat-owned, so it is derived the same way on both
		# routes - and after the restore, whose principals the victor snap
		# compares against.
		POD_combat_derive_sc_outcome_effect = yes

		POD_combat_apply_duel_finalization_effect = yes

		# Fire the follow-up. The three-step chain (initiator -> defender ->
		# attacker, gated on is_alive) matches vanilla's pattern so `root` of
		# the follow-up is the character vanilla would have used.
		POD_combat_fire_sc_followup_effect = yes

		# Full cleanup. pod_combat_cleanup_effect ends the context story and
		# removes the pointer var: atomic teardown, since every var that
		# lived on the story vanishes when end_story fires.
		#
		# Finalization can pod_death the host itself (sc_loser under fatality=always
		# / tribal-default); the full cleanup then mutates ~50 host vars on a
		# corpse, which errors. Route a dead host to the dead-safe release (ends
		# stories + clears living combatants' bridge vars; the host's own combat
		# vars stay inert in save data, the heir being a different character).
		if = {
			limit = { is_alive = yes }
			pod_combat_cleanup_effect = yes
		}
		else = {
			pod_combat_host_died_release_effect = yes
		}
	}
}
