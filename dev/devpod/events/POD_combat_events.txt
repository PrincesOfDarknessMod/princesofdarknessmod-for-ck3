# Simple hidden follow-up for debug interactions. Cleans the
# shirtlessness flag pod_combat_debug_interaction may have applied
# and shows a result toast. The engaged_in_single_combat flag is
# released earlier by pod_combat.9999 (the integrated bridge event)
# — this debug follow-up doesn't touch it.

pod_combat_debug.0001 = {
	hidden = yes

	immediate = {
		# Clean shirtlessness if it was applied
		single_combat_clean_shirtlessness_effect = {
			ATTACKER = scope:sc_attacker
			DEFENDER = scope:sc_defender
		}

		# Guard the scope derefs: a draw / mutual-KO can leave sc_victor and
		# sc_loser unset, and an unguarded right_icon on an unset scope errors.
		if = {
			# PC won — show the defeated opponent.
			limit = {
				exists = scope:sc_victor
				scope:sc_victor = root
				exists = scope:sc_loser
			}
			send_interface_toast = {
				title = pod_combat_debug_victory_toast
				left_icon = root
				right_icon = scope:sc_loser
			}
		}
		else_if = {
			# PC lost — show the victor.
			limit = {
				exists = scope:sc_victor
				NOT = { scope:sc_victor = root }
			}
			send_interface_toast = {
				title = pod_combat_debug_defeat_toast
				left_icon = root
				right_icon = scope:sc_victor
			}
		}
		else = {
			# Draw / mutual-KO / unresolved victor — single-portrait toast,
			# no second icon to dereference.
			send_interface_toast = {
				title = pod_combat_debug_draw_toast
				left_icon = root
			}
		}
	}
}

# Post-combat dispatch. Fires at the end of an sc-driven combat. CK3 does
# NOT propagate GUI-context saved scopes through event chains, so every
# scope is reconstructed from the POD_combat_sc_context story the launcher
# built (reached via the PC's pod_combat_sc_context pointer var). The six
# phases below carry the per-step contract; full rationale in
# pod_combat_sc_context_effects.txt's header.

pod_combat.9999 = {
	hidden = yes

	immediate = {
		# Six finalization phases, one helper each; per-phase contract in
		# pod_combat_sc_context_effects.txt's header.

		# Phase 1: Reconstruct sc_initiator / sc_attacker / sc_defender
		# + alias scopes (victim, recipient, target, attacker, challenger,
		# adversary) + sc_victor / sc_loser from the PC's
		# pod_combat_sc_context pointer.
		POD_combat_restore_sc_context_effect = yes

		# Phase 2: Rebuild parent-event aliases (scope:spirit, scope:actor,
		# scope:my_rival, etc. — 24 entries). Whitelist must stay
		# synchronised with pod_combat_sc_stash_parent_aliases_effect in
		# the launcher.
		pod_combat_sc_restore_parent_aliases_effect = yes

		# Phase 3: Round-trip the rule flags (fatality / fixed / locale)
		# from sc_ctx variables into saved-scope flag values. Also
		# captures sc_finished and victory_type for the follow-up.
		POD_combat_restore_sc_rule_flags_effect = yes

		# Phase 4: Side-effect finalization — release engaged flags,
		# bump victor's primary armament artifact_duel_var, apply
		# fatality-mode-aware death outcomes.
		POD_combat_apply_duel_finalization_effect = yes

		# Phase 5: Fire the follow-up event. Three-step fallback chain
		# (initiator → defender → attacker, gated on is_alive) matches
		# vanilla's pattern so `root` of the follow-up is the same
		# character vanilla would have used.
		POD_combat_fire_sc_followup_effect = yes

		# Phase 6: Full cleanup. pod_combat_cleanup_effect ends the
		# context story and removes the pointer var. Atomic teardown
		# — every var that lived on the story (sc_* refs, rule flags,
		# alias snapshots, follow-up id) vanishes when end_story fires.
		pod_combat_cleanup_effect = yes
	}
}
