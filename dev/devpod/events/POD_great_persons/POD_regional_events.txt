namespace = podregionalevents

podregionalevents.1 = {
	#Main event handling RE
	#Called from specific yearly pulse on action with a small delay
	hidden = yes
	scope = none
	
	trigger = {
	}
	
	immediate = {
		if = {
			limit = {
				has_game_rule = baybars_purge_difficulty_default
			}
			every_ruler = {
				limit = {
					any_realm_province = {
						geographical_region = ghw_region_egypt
					}
					is_supehunter_trigger = no
					is_child_of_supehunter_trigger = no
				}
				trigger_event = podregionalevents.2
			}
			
			random_list = {
				50 = {
					title:c_qus = {
						holder = {
							trigger_event = {
								id = podregionalevents.7
								years = 12
							}
						}
					}
				}
				50 = {
					title:c_bahriya = {
						holder = {
							trigger_event = {
								id = podregionalevents.8
								years = 12
							}
						}
					}
					
				}
			}
			trigger_event = {
				id = podregionalevents.11
				years = 17
			}
		}
	}
}

#Baybars' purge starts

podregionalevents.2 = {
	type = character_event
	content_source = dlc_POD
	title = podregionalevents.2.t
	desc = podregionalevents.2.desc
	theme = martial
	
	trigger = {
	}
	
	immediate = {
		every_held_county = {
			limit = {
				title_province = {
					geographical_region = ghw_region_egypt
				}
			}
			add_county_modifier = baybars_purge_small_modifier
		}
		hidden_effect = {
			random_list = {
				50 = {
					trigger_event = {
						id = podregionalevents.3
						years = 4
					}
				}
				50 = {
					trigger_event = {
						id = podregionalevents.4
						years = 4
					}
				}
			}
		}
	}
	
	option = {
		name = podregionalevents.2.a
		
		custom_tooltip = podregionalevents.2.custom
		
		if = {
			limit = {
				highest_held_title_tier = tier_county
			}
			add_gold = 150
			add_character_modifier = PODexile_modifier
		}
		
		else_if = {
			limit = {
				highest_held_title_tier = tier_duchy
			}
			add_gold = 300
			add_character_modifier = PODexile_modifier
		}
		
		else_if = {
			limit = {
				highest_held_title_tier = tier_kingdom
			}
			add_gold = 450
			add_character_modifier = PODexile_modifier
		}
		
		else_if = {
			limit = {
				highest_held_title_tier = tier_empire
			}
			add_gold = 600
			add_character_modifier = PODexile_modifier
		}
		
		ai_chance = {
			base = 75
			
			modifier = {
				add = 50
				has_trait = craven
			}
			
			modifier = {
				add = 50
				has_trait = fickle
			}
		}
	}
	
	option = {
		name = podregionalevents.2.b
		
		custom_tooltip = podregionalevents.2.customb
		
		ai_chance = {
			base = 25
			
			modifier = {
				add = 50
				has_trait = brave
			}
			
			modifier = {
				add = 50
				has_trait = stubborn
			}
		}
	}
	
}
