	type = character_event
	content_source = dlc_POD	
	title = podintroductions1500.105.t
	desc = podintroductions1500.105.desc
	theme = cappadocian 
	left_portrait = {
		character = character:505018
		animation = schadenfreude
		}		
	
	option = {
		name = podintroductions1500.105.a	

		custom_tooltip = podintroductions1500.105.tooltip

		if = {
			limit = {
				is_cappadocian_trigger = yes
				NOT = { this = character:790220 }
				}
		add_character_modifier = {
			modifier = kill_on_sight_modifier
				years = 50
				}	
		}

		hidden_effect = {
		every_living_character = {
		limit = {
			is_cappadocian_trigger = yes
			NOT = { this = character:790220 }			
			NOT = { house = house:house_giovanni }
			NOT = { house = house:house_premascine }			
		}	
		add_character_modifier = {
			modifier = kill_on_sight_modifier
				years = 50
				}			
		}
		}
	}
}
