	}	
		
	immediate = {
	}
	
	option = {
		name = podintroductionswerewolves.100.a	
		
		trigger_event = podintroductionswerewolves.103
	}
	option = {
		name = podintroductionswerewolves.100.b	
		
		trigger_event = podintroductionswerewolves.104
	}

		name = podintroductions.2001.e
		
		trigger_event = POD_chargen.1
	}	

	option = {
		name = podintroductionswerewolves.100.f
		
		custom_tooltip = continue_intro_tooltip
		
		trigger_event = {
			id = podintroductionswerewolves.110
			days = 1
			}
		trigger_event = POD_chargen.2600
	}
	
	option = {
		name = podintroductionswerewolves.100.d
		
		custom_tooltip = end_intro_tooltip
		trigger_event = POD_chargen.2600
	}	
}
