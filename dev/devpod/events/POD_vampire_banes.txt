# Events for vampire clan banes (obsessions and compulsions)
namespace = POD_vampire_banes

@toreador_bane_months = 6

# Toreador: Distracted by Beauty
POD_vampire_banes.0001 = {
	type = character_event
	content_source = dlc_POD

	left_portrait = {
		character = root
		animation = ecstasy
	}
	theme = toreador_coa

	# The event is fired on average every 20 years, so this cooldown makes the average 40
	cooldown = { years = 20 }

	trigger = {
		is_toreador_trigger = yes
		NOT = { has_character_modifier = POD_toreador_bane_modifier }
	}

	immediate = {
		random_list = {
			1 = {
				add_character_flag = {
					flag = toreador_distraction_painting
					months = @toreador_bane_months
				}
			}
			1 = {
				add_character_flag = {
					flag = toreador_distraction_song
					months = @toreador_bane_months
				}
			}
			1 = {
				add_character_flag = {
					flag = toreador_distraction_sculpture
					months = @toreador_bane_months
				}
			}
			1 = {
				add_character_flag = {
					flag = toreador_distraction_poem
					months = @toreador_bane_months
				}
			}
		}
	}

	title = POD_vampire_banes.0001.t
	desc = {
		desc = POD_vampire_banes.0001.desc.intro
		triggered_desc = {
			trigger = { has_character_flag = toreador_distraction_painting }
			desc = POD_vampire_banes.0001.desc.painting
		}
		triggered_desc = {
			trigger = { has_character_flag = toreador_distraction_song }
			desc = POD_vampire_banes.0001.desc.song
		}
		triggered_desc = {
			trigger = { has_character_flag = toreador_distraction_sculpture }
			desc = POD_vampire_banes.0001.desc.sculpture
		}
		triggered_desc = {
			trigger = { has_character_flag = toreador_distraction_poem }
			desc = POD_vampire_banes.0001.desc.poem
		}
		desc = POD_vampire_banes.0001.desc.end
	}

	option = {
		# "Beauty is the soul of the world"
		name = POD_vampire_banes.0001.a
		add_character_modifier = {
			modifier = POD_toreador_bane_modifier
			months = @toreador_bane_months
			desc = {
				triggered_desc = {
					trigger = { has_character_flag = toreador_distraction_painting }
					desc = POD_toreador_bane_modifier.painting
				}
				triggered_desc = {
					trigger = { has_character_flag = toreador_distraction_song }
					desc = POD_toreador_bane_modifier.song
				}
				triggered_desc = {
					trigger = { has_character_flag = toreador_distraction_sculpture }
					desc = POD_toreador_bane_modifier.sculpture
				}
				triggered_desc = {
					trigger = { has_character_flag = toreador_distraction_poem }
					desc = POD_toreador_bane_modifier.poem
				}
			}
		}
	}

	# TODO: option to acquire sculpture or painting as a court artifact, see issue #216
}
