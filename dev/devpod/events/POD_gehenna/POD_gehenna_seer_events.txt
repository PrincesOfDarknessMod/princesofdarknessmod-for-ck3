# Is the scoped character a possible Gehenna Seer?
scripted_trigger POD_qualify_gehenna_seer = {
	aptitude = {
		court_position = court_seer_court_position
		value >= 4
	}
}

# Can the scoped liege have Gehenna prophecies at court?
scripted_trigger POD_can_have_gehenna_seer = {
	NOT = { has_religion = religion:camarilla_religion }
}

# Event when the court seer isn't good enough to get gehenna prophecies
# This event nudges the player towards getting a seer with higher aptitude.
POD_gehenna_seer.100 = {
	type = character_event
	content_source = dlc_POD
	title = POD_gehenna_seer.100.t
	desc = {
		random_valid = {
			desc = POD_gehenna_seer.100.desc.1
			desc = POD_gehenna_seer.100.desc.2
			desc = POD_gehenna_seer.100.desc.3
		}
	}
	theme = gehennaseer
	
	left_portrait = {
		character = scope:gehenna_seer
		animation = personality_irrational
	}

	trigger = {
		has_trait = vampire
		is_available = yes
		is_ai = no
		NOT = { has_character_flag = denied_gehenna_seer }
		any_court_position_holder = {
			type = court_seer_court_position
			is_available_ai = yes
			POD_qualify_gehenna_seer = no
		}
		POD_is_available_for_events = yes
	}

	immediate = {
		random_court_position_holder = {
			type = court_seer_court_position
			limit = {
				is_available_ai = yes
				POD_qualify_gehenna_seer = no
			}
			save_scope_as = gehenna_seer
		}
	}

	option = { # I need a better seer
		name = POD_gehenna_seer.100.a
		revoke_court_position = {
			recipient = scope:gehenna_seer
			court_position = court_seer_court_position
		}
		# Some piety to compensate for the prestige loss of revoking the position
		add_piety = 100
		stress_impact = {
 			base = minor_stress_impact_gain
			lunatic_1 = medium_stress_impact_gain
			lunatic_genetic = medium_stress_impact_gain
 			paranoid = major_stress_impact_gain
 		}
 		ai_chance = {
 			base = 100
 		}
	}

	option = { # More drugged vessels for the seer
		name = POD_gehenna_seer.100.b
		stress_impact = {
			base = medium_stress_impact_gain
			lunatic_1 = medium_stress_impact_gain
			lunatic_genetic = medium_stress_impact_gain
			paranoid = major_stress_impact_gain
		}
		ai_chance = {
			base = 100
		}
		# The seer may get a trait that improves aptitude for court seer
		scope:gehenna_seer = {
			random_list = {
				50 = {}
				5 = {
					trigger = {
						NOT = { has_trait = lunatic }
					}
					add_trait = lunatic_1
				}
				5 = {
					trigger = {
						NOT = { has_trait = possessed }
					}
					add_trait = possessed_1
				}
				5 = {
					trigger = {
						NOT = { has_trait = blind }
					}
					add_trait = blind
				}
				#5 = {
				#	trigger = {
				#		has_trait = mystic_2
				#	}
				#	change_trait_rank = {
				#		trait = lifestyle_mystic
				#		rank = 1
				#		max = 3
				#	}
				#}
				#10 = {
				#	trigger = {
				#		has_trait = mystic_1
				#	}
				#	change_trait_rank = {
				#		trait = lifestyle_mystic
				#		rank = 1
				#		max = 3
				#	}
				#}
				20 = {
					trigger = {
						NOT = { has_trait = lifestyle_mystic }
					}
					add_trait = lifestyle_mystic
				}
			}
		}
	}

	option = { # This Gehenna nonsense is nonsense
		name = POD_gehenna_seer.100.c
 		ai_chance = {
 			base = 100
			ai_value_modifier = {
				ai_rationality = 1.5
			}
 		}
		hidden_effect = {
			add_character_flag = {
				flag = denied_gehenna_seer
				years = 50
			}
		}
	}
}
