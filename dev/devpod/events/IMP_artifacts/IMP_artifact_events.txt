### Impundulu Submod — Artifact Crafting Events
### One event per artifact — the ritual itself
### Reference: POD artifact creation pattern

namespace = POD_impundulu

### Event 1: Craft the Shew-Stone
POD_impundulu.3001 = {
	type = character_event
	content_source = dlc_POD
	title = POD_impundulu.3001.t
	desc = POD_impundulu.3001.desc
	theme = vampire
	override_background = { reference = bp1_bonfire }

	left_portrait = {
		character = root
		animation = personality_rational
	}
	
	artifact = {
		target = scope:newly_created_artifact
		position = lower_center_portrait
	}

	immediate = {
		create_artifact = {
			name = artifact_shewstone_name
			description = artifact_shewstone_description
			modifier = artifact_shewstone_modifier
			type = miscellaneous
			visuals = small_box
			wealth = 50
			quality = 60
			save_scope_as = newly_created_artifact
			decaying = no
		}
		### Artifact created and scoped — no variable tracking needed
	}

	option = {
		name = POD_impundulu.3001.a
		custom_tooltip = POD_impundulu_shewstone_crafted_tt
	}
}

### Event 3: Craft the Witch Eye
POD_impundulu.3003 = {
	type = character_event
	content_source = dlc_POD
	title = POD_impundulu.3003.t
	desc = POD_impundulu.3003.desc
	theme = vampire
	override_background = { reference = bp1_bonfire }

	left_portrait = {
		character = root
		animation = worry
	}
	
	artifact = {
		target = scope:newly_created_artifact
		position = lower_center_portrait
	}

	immediate = {
		create_artifact = {
			name = artifact_witch_eye_name
			description = artifact_witch_eye_description
			modifier = artifact_witch_eye_modifier
			type = miscellaneous
			visuals = small_box
			wealth = 90
			quality = 90
			save_scope_as = newly_created_artifact
			decaying = no
		}
		scope:newly_created_artifact = {
			set_variable = imp_witch_eye_artifact
		}
		### Apply the scarring modifier (self-mutilation cost)
		add_character_modifier = imp_witch_eye_scarred_modifier
	}

	option = {
		name = POD_impundulu.3003.a
		custom_tooltip = POD_impundulu_witch_eye_crafted_tt
	}
}
