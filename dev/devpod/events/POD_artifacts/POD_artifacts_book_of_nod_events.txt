POD_artifacts_book_of_nod.100 = {
	type = character_event
	content_source = dlc_POD
	title = POD_artifacts_book_of_nod.100.t
	desc = POD_artifacts_book_of_nod.100.desc
	theme = learning 
	
	left_portrait = {
		character = root
		animation = personality_rational
	}
	
	immediate = {
		every_character_artifact = {
			limit = {
				has_variable = chronicle_of_caine
			}
			save_scope_as = players_chronicle_of_caine
		}
		
		every_character_artifact = {
			limit = {
				has_variable = chronicle_of_shadows
			}
			save_scope_as = players_chronicle_of_shadows
		}
		
		every_character_artifact = {
			limit = {
				has_variable = chronicle_of_secrets
			}
			save_scope_as = players_chronicle_of_secrets
		}
		every_character_artifact = {
			limit = {
				has_variable = first_of_times 
			}
			save_scope_as = players_first_of_times 
		}
		every_character_artifact = {
			limit = {
				has_variable = coming_of_lilith
			}
			save_scope_as = players_coming_of_lilith
		}
		every_character_artifact = {
			limit = {
				has_variable = liliths_magick
			}
			save_scope_as = players_liliths_magick
		}		
		every_character_artifact = {
			limit = {
				has_variable = temptation_of_caine
			}
			save_scope_as = players_temptation_of_caine
		}
		every_character_artifact = {
			limit = {
				has_variable = zillahs_tale
			}
			save_scope_as = players_zillahs_tale
		}
		every_character_artifact = {
			limit = {
				has_variable = tale_of_the_crone
			}
			save_scope_as = players_tale_of_the_crone
		}
		every_character_artifact = {
			limit = {
				has_variable = tale_of_the_first_city
			}
			save_scope_as = players_tale_of_the_first_city
		}		
	}

	option = {
		name = POD_artifacts_book_of_nod.100.option.e
		trigger = {
			POD_has_unassembled_chronicle_of_caine_trigger = yes
		}
		if = {
			limit = { has_ep1_dlc_trigger = yes }
			create_artifact_chronicle_of_caine_effect = { OWNER = root TYPE = book }
		}
		else = {
			create_artifact_chronicle_of_caine_effect = { OWNER = root TYPE = miscellaneous }
		}
		add_prestige = minor_prestige_gain
		add_piety = minor_piety_gain
		POD_destroy_artifact = { ARTIFACT = scope:players_first_of_times }
		POD_destroy_artifact = { ARTIFACT = scope:players_coming_of_lilith }
		POD_destroy_artifact = { ARTIFACT = scope:players_liliths_magick }
		POD_destroy_artifact = { ARTIFACT = scope:players_temptation_of_caine }
		POD_destroy_artifact = { ARTIFACT = scope:players_zillahs_tale }
		POD_destroy_artifact = { ARTIFACT = scope:players_tale_of_the_crone }
		POD_destroy_artifact = { ARTIFACT = scope:players_tale_of_the_first_city }
	}

	option = {
		name = POD_artifacts_book_of_nod.100.option.f
		trigger = {
			POD_has_all_book_of_nod_chronicles_unassembled_trigger = yes
		}
		if = {
			limit = { has_ep1_dlc_trigger = yes }
			create_artifact_book_of_nod_effect = { OWNER = root TYPE = book }
		}
		else = {
			create_artifact_book_of_nod_effect = { OWNER = root TYPE = miscellaneous }
		}
		POD_overwrite_artifact_link = { LINK = book_of_nod_link }
		add_prestige = medium_prestige_gain
		add_piety = medium_piety_gain
		POD_destroy_artifact = { ARTIFACT = scope:players_chronicle_of_caine }
		POD_destroy_artifact = { ARTIFACT = scope:players_chronicle_of_shadows }
		POD_destroy_artifact = { ARTIFACT = scope:players_chronicle_of_secrets }
	}

	option = { # Open to the Chronicle of Caine
		name = POD_artifacts_book_of_nod.100.option.a
		trigger_event = { id = POD_artifacts_book_of_nod.200 }
	}	
	
	option = { # Open to the Chronicle of Shadows
		name = POD_artifacts_book_of_nod.100.option.b
		trigger_event = { id = POD_artifacts_book_of_nod.300 }
	}	

	option = { # Open to the Chronicle of Secrets
		name = POD_artifacts_book_of_nod.100.option.c
		trigger_event = { id = POD_artifacts_book_of_nod.400 }
	}	

	option = { # Close the Book of Nod
		name = POD_artifacts_book_of_nod.100.option.d
	}		
}
