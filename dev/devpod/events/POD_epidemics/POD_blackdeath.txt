#Black Death control event
#India and SubSaharan Africa are spared
POD_blackdeath.0001 = {
	hidden = yes
	scope = none

	on_trigger_fail = {
		#The scourge is a mutation of the BD. If the BD doesn't happen, the scourge doesn't as well
		#trigger_event = {
		#	id = POD_blackdeath.0006
		#	years = 5
		#}
	}
	
	trigger = {
		NOT = { has_global_variable = black_death_prevented }
		NOT = { has_game_rule = bd_frequency_never }
		#has_game_rule = black_death_default
	}
	
	immediate = {
		every_player = {
			trigger_event = POD_blackdeath.0004
		}
		
		province:562 = {
			create_epidemic_outbreak = {
				type = bubonic_plague
				intensity = apocalyptic
			}		
		}
		
		trigger_event = {
			id = POD_blackdeath.0006
			years = 5
		}
		
		#We use PDX stuff now
		#trigger_event = {
		#		id = POD_blackdeath.0003
		#		years = 3
		#	}	
	}

}

POD_blackdeath.0002 = {
	hidden = yes
	scope = none
	
	immediate = {
		every_living_character = {	
			limit = { has_trait = flaying_plague }
			remove_trait = flaying_plague
		}
		every_epidemic = {	
			limit = { epidemic_type = epidemic_type:flaying_plague }
			set_epidemic_outbreak_intensity = minor
		}
	}
}

POD_blackdeath.0003 = {
	hidden = yes

	trigger = {
		NOT = { has_global_variable = flaying_plague_stopped }
	}
	
	immediate = {
		random_county_in_region = {
			region = ghw_region_baltic
			title_province = { 
				create_epidemic_outbreak = {
					type = flaying_plague
					intensity = apocalyptic
				} 
			} 
		}
		every_player = {
			limit = {
				OR = {
					has_trait = werewolf
					has_trait = kinfolk
				}
			}
			trigger_event = POD_blackdeath.0007
		}
	}
}
