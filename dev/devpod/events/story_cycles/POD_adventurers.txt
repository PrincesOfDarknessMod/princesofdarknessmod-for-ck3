#Context events for Giangaleazzo


PODadventurers.0004 = {
	scope = none
	hidden = yes
	
	immediate = {
		if = { 
			limit = {
				any_player = {					
				any_realm_province = {
					geographical_region = ghw_region_italy
				}
				}
			}
		random_player = {
				limit = {
				any_realm_province = {
					geographical_region = ghw_region_italy
				}
				highest_held_title_tier > tier_county		
				}	
			trigger_event = PODadventurers.0003
		}
		
	}
	#If no player in Italy, skip to invasion	
	else = {
		trigger_event = PODadventurers.0001
	}	
	}
}

PODadventurers.0003 = {
	type = character_event
	content_source = dlc_POD
	title = PODadventurers.1002.t
	desc = PODadventurers.0003.desc
	theme = lasombra_coa
	left_portrait = {
		character = root
		animation = worry
	}
	
	immediate = {
		if = {
			limit = {
				POD_is_vampire_or_abomination_trigger = yes
			}
		trigger_event = { 
			id = PODadventurers.0005
			days = 15
		}
		}
		#Skip vampire specific negotiation
		else = {
		trigger_event = { 
			id = PODadventurers.0001
			days = 30
		}
		}		
	}	

	option = { 
		name = PODadventurers.0003.a
		
		custom_tooltip = PODadventurers.0003.a.tt
		
		hidden_effect = {
		title:c_lombardia.title_province = {			
			POD_remove_common_building_effect= yes
		}
		
		title:c_brescia.title_province = {			
			POD_remove_common_building_effect= yes
		}

		title:c_como.title_province = {			
			POD_remove_common_building_effect= yes
		}
		}
		
		custom_tooltip = PODadventurers.1001.player_warning
	}
	
	option = { 
		name = PODadventurers.0003.b
		
		remove_short_term_gold = 300	

		custom_tooltip = PODadventurers.0003.b.tt

		hidden_effect = {
		title:c_como.title_province = {			
			POD_remove_common_building_effect= yes
		}
		}
		
		custom_tooltip = PODadventurers.1001.player_warning
	}	
}

### Spawn Giangaleazzo
PODadventurers.0002 = {
	scope = none
	hidden = yes

	trigger = {
		scope:giangaleazzo_county = {
			holder = {
				OR = { # Last resort check to make sure that we don't automatically game over a player
					is_ai = yes # Either they have to be AI
					any_held_title = { # Or they have to hold land elsewhere
						tier = tier_county
						NOT = { this = scope:giangaleazzo_county }
					}
				}
			}
		}
	}

	immediate = {
		debug_log = "Giangaleazzo appeared!"
		debug_log_date = yes
		create_giangaleazzo_effect = yes
	}
}

PODadventurers.1002 = {
	type = character_event
	content_source = dlc_POD
	title = PODadventurers.1002.t
	desc = PODadventurers.1002.desc
	theme = lasombra_coa

	left_portrait = {
		character = scope:giangaleazzo
		animation = personality_bold
	}

	immediate = {
		send_interface_message = {
			type = POD_event_gehenna_omen
			title = POD_giangaleazzo.notification
			desc = {
				triggered_desc = {
					trigger = {
						has_trait = "vampire"
					}
					desc = POD_giangaleazzo.desc
				}
			}
		}	
	}

	option = { # Fight him off!
		name = PODadventurers.1002.a

			custom_tooltip = PODadventurers.1002.a.target.tt

	}

	option = { # Let's negotiate
		name = PODadventurers.1002.b
		
		trigger = {
			POD_is_vampire_or_abomination_trigger = yes 
		}
		
		add_prestige = massive_prestige_loss
		add_prestige_level = -1
		add_piety = massive_piety_loss
		add_piety_level = -1
		
		create_title_and_vassal_change = {
			type = created
			save_scope_as = change
			add_claim_on_loss = no
		}

			scope:giangaleazzo = {
				change_liege = {
					liege = root
					change = scope:change
				}
				resolve_title_and_vassal_change = scope:change
			}		
		
		scope:giangaleazzo = {
			add_opinion = {
				target = root
				modifier = respect_opinion
				opinion = 50
			}
		}		
		
		custom_tooltip = PODadventurers.1002.b.custom
		
	}

}

PODadventurers.1022 = {
	type = character_event
	content_source = dlc_POD
	title = PODadventurers.1022.t
	desc = PODadventurers.1022.desc
	theme = tzimisce_coa

	left_portrait = {
		character = scope:vlad
		animation = personality_bold
	}

	immediate = {
		send_interface_message = {
			type = POD_event_gehenna_omen
			title = POD_vlad.notification
			desc = {
				triggered_desc = {
					trigger = {
						has_trait = "vampire"
					}
					desc = POD_vlad.desc
				}
			}
		}
		root = {
			save_scope_as = sire
		}		
	}

	option = { # Fight him off!
		name = PODadventurers.1022.a

		custom_tooltip = PODadventurers.1022.a.target.tt
	}

	option = { # Let's negotiate
		name = PODadventurers.1022.b
		if = {
			limit = { is_female = yes }
			scope:vlad = { set_mother = root }
		}
		if = {
			limit = { is_female = no }
			scope:vlad = { set_father = root }
		}
		
		add_prestige = massive_prestige_loss
		add_prestige_level = -1
		add_piety = massive_piety_loss
		add_piety_level = -1
		remove_short_term_gold = 500
		
		scope:vlad = {
			add_opinion = {
				target = root
				modifier = respect_opinion
				opinion = 50
			}
			add_gold = 500
			set_house = scope:sire.house
		}		
		
		custom_tooltip = PODadventurers.1022.b.custom
	}
}
