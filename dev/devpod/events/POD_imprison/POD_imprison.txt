POD_imprison.001 = {
    hidden = yes
	
	trigger = {
		scope:imprisoner = { is_ai = no }
		prowess > 20
		trigger_if = {
			limit = { POD_custom_incompatible_splats_trigger = { ACTOR = scope:imprisoner RECIPIENT = root } }
			NOT = { is_vassal_of = scope:imprisoner }
		}
		trigger_else = {
			NOT = { scope:imprisoner = { has_imprisonment_reason = root } }
		}
		NOT = { has_character_flag = lost_duel }
		is_imprisoned = yes
        is_ai = yes
	}
	immediate = {
	    save_scope_as = targetdefender
		add_character_flag = {
			flag = escaped_from_prison_flag
			days = 1
		}
		add_character_flag = { #To make sure you don't get spammed about the release
			flag = block_for_prison_release_notification
			days = 1
		}
		release_from_prison = yes
	    scope:imprisoner = {
		    save_scope_as = attacker
			trigger_event = POD_imprison.002
		}
    }
}
