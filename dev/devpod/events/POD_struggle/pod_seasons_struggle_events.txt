#Set up for on_action struggle events
pod_seasons_struggle.0001 = {
	hidden = yes
	scope = struggle

	trigger = {
		exists = struggle:seasons_struggle
	}
	
	immediate = {
		#Get season courts holder to war	
		title:e_winter_court = { 
			if = { 
				limit = { exists = holder }
				holder = { trigger_event = pod_seasons_struggle.1000 }
			}
		}
		title:e_spring_court = { 
			if = { 
				limit = { exists = holder }
				holder = { trigger_event = pod_seasons_struggle.1000 }
			}
		}
		title:e_summer_court = { 
			if = { 
				limit = { exists = holder }
				holder = { trigger_event = pod_seasons_struggle.1000 }
			}
		}
		title:e_autumn_court = { 
			if = { 
				limit = { exists = holder }
				holder = { trigger_event = pod_seasons_struggle.1000 }
			}
		}
		#Starts events on_action struggle	
		trigger_event = {
			id = pod_seasons_struggle.0001
			years = { 3 5 }
		}
	}
}

pod_seasons_struggle.109 = {
	hidden = yes
	
	immediate = {
		create_title_and_vassal_change = {
			type = created
			save_scope_as = change
			add_claim_on_loss = no
		}
		if = {
			limit = { title:e_winter_court = { NOT = { exists = holder } } }
			random_ruler = {
				limit = { religion = religion:winter_court_religion }
				title:e_winter_court = {
					change_title_holder = {
						holder = prev
						change = scope:change
					}
				}
			}
		}
		if = {
			limit = { title:e_spring_court = { NOT = { exists = holder } } }
			random_ruler = {
				limit = { religion = religion:spring_court_religion }
				title:e_spring_court = {
					change_title_holder = {
						holder = prev
						change = scope:change
					}
				}
			}
		}
		if = {
			limit = { title:e_summer_court = { NOT = { exists = holder } } }
			random_ruler = {
				limit = { religion = religion:summer_court_religion }
				title:e_summer_court = {
					change_title_holder = {
						holder = prev
						change = scope:change
					}
				}
			}
		}
		if = {
			limit = { title:e_autumn_court = { NOT = { exists = holder } } }
			random_ruler = {
				limit = { religion = religion:autumn_court_religion }
				title:e_autumn_court = {
					change_title_holder = {
						holder = prev
						change = scope:change
					}
				}
			}
		}
		resolve_title_and_vassal_change = scope:change
	}
}

		}

			set_color_from_title = title:e_winter_court	
		}
	}
}
