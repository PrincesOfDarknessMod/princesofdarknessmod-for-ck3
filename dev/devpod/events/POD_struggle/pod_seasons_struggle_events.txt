#Set up for on_action struggle events
pod_seasons_struggle.0001 = {
	hidden = yes
	type = empty
	scope = struggle

	trigger = {
		exists = struggle:seasons_struggle
	}
	
	immediate = {
		#Get season courts holder to war	
		title:e_winter_court = { 
			if = { 
				limit = { exists = holder }
				holder = { trigger_event = pod_seasons_struggle.1000 }
			}
		}
		title:e_spring_court = { 
			if = { 
				limit = { exists = holder }
				holder = { trigger_event = pod_seasons_struggle.1000 }
			}
		}
		title:e_summer_court = { 
			if = { 
				limit = { exists = holder }
				holder = { trigger_event = pod_seasons_struggle.1000 }
			}
		}
		title:e_autumn_court = { 
			if = { 
				limit = { exists = holder }
				holder = { trigger_event = pod_seasons_struggle.1000 }
			}
		}
		#Starts events on_action struggle	
		trigger_event = {
			id = pod_seasons_struggle.0001
			years = { 3 5 }
		}
	}
}

pod_seasons_struggle.0501 = { # Fullscreen Intro Event
	type = fullscreen_event
	title = pod_seasons_struggle.0501.t
	desc = pod_seasons_struggle.0501.desc
	theme = realm
	trigger = { # we need to prevent this from firing for all players when a new player joins in MP
		is_ai = no
		NOT = { has_character_flag = pod_seasons_struggle_intro_event_flag }
		has_POD_fae_blood_trigger = yes
 	}

	override_sound = { reference = "event:/DLC/FP2/SFX/UI/fp2_struggle_ui_intro_animate" }

	widgets = {
		widget = {
 			gui = "event_window_widget_struggle_info"
 			container = "dynamic_content_widget"
 			controller = struggle_info
 			setup_scope = { struggle:seasons_struggle = { save_scope_as = struggle } }
		}
	}
	
	immediate = {
		play_music_cue = "mx_Struggle_Opening"
		add_character_flag = pod_seasons_struggle_intro_event_flag

			name = start
			value = yes
		}
	}
	
	option = {
		name = pod_seasons_struggle.0501.a
	}
}

scripted_effect set_season_courts_color_effect = {
	title:e_season_courts = { 
		if = {
			limit = { current_month < 2 }
			set_color_from_title = title:e_winter_court	
		}
		else_if = {
			limit = { current_month < 5 }
			set_color_from_title = title:e_spring_court	
		}
		else_if = {
			limit = { current_month < 8 }
			set_color_from_title = title:e_summer_court	
		}
		else_if = {
			limit = { current_month < 11 }
			set_color_from_title = title:e_autumn_court
		}
		else = {
			set_color_from_title = title:e_winter_court	
		}
	}
}
