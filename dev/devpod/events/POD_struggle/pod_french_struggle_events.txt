#Set up for on_action struggle events
pod_french_struggle.0001 = {
	hidden = yes
	type = empty
	scope = struggle

	trigger = {
		exists = struggle:french_struggle
	}
	
	immediate = {
		#Starts events on_action struggle
		struggle:french_struggle = {
			every_involved_ruler = {
				limit = {
					is_ai = no
					}
				trigger_event = {
					on_action = french_struggle_on_action_playable_pulse
					years = 2
				}
			}
		}
		struggle:french_struggle = {
			every_interloper_ruler = {
				limit = {
					is_ai = no
					}
				trigger_event = {
					on_action = french_struggle_on_action_playable_pulse
					years = 2
				}
			}
		}		
		trigger_event = {
			id = pod_french_struggle.0001
			years = { 3 5 }
		}		
	}
}
