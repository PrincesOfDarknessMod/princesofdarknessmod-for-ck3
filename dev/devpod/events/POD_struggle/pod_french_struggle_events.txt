#French victory
pod_french_struggle.102 = {
	type = character_event
	content_source = dlc_POD
	title = pod_french_struggle.102.t
	desc = pod_french_struggle.102.desc
	theme = vampire
	left_portrait = {
		character = scope:french_struggle_victor
		animation = personality_bold
	}
	
	widget = {
		gui = "POD_shader_event_fog_alley_night"
		container = "background_shader_vfx_container"
	}

	immediate = {
		play_music_cue = "mx_cue_epic_sacral_moment"	
		
		if = {
			limit = {
				is_ai = no
				this = character:505082
			}
			set_global_variable = pod_achiev_alexander # Achievement: Somehow, he has returned
		}
	}

	option = {
		name = pod_french_struggle.102.a
							
		add_prestige = massive_prestige_value
					
		hidden_effect = {
		every_player = { 
		limit = {
			NOT = { this = scope:french_struggle_victor }
			}
		trigger_event = pod_french_struggle.104
			}
		}		
		
		if = {
			limit = {
				title:e_love = {			
				holder = { NOT = { this = scope:french_struggle_victor } }
				}
				}
				create_title_and_vassal_change = {
					type = created
					save_scope_as = change
					add_claim_on_loss = no
				}
		
			title:e_love = {
				change_title_holder = {
					holder = root
					change = scope:change
				}
			}
		resolve_title_and_vassal_change = scope:change
		hidden_effect = { set_primary_title_to = title:e_love }
		}	
		
		# End Struggle
		struggle:french_struggle = {
			end_struggle = french_victory_objective_decision_title
		}

		title:b_rethel.title_province = {
		add_special_building = salianna_court_01
		}

		title:b_lyon.title_province = {
		add_special_building = salianna_court_01
		}	

		title:b_bordeaux.title_province = {
		add_special_building = salianna_court_01
		}

		title:b_blois.title_province = {
		add_special_building = salianna_court_01
		}

		title:b_poitiers.title_province = {
		add_special_building = salianna_court_01
		}

		if = {
			limit = {
				has_trait = brujah
			}
		title:e_love = { set_title_name = e_love_brujah }		
		}
		
		else_if = {
			limit = {
				has_trait = lasombra
			}
		title:e_love = { set_title_name = e_love_lasombra }		
		}

		else_if = {
			limit = {
				has_trait = gangrel
			}
		title:e_love = { set_title_name = e_love_gangrel }		
		}

		else = {}	
	}
}
