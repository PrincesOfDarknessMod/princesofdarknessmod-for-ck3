pod_assamite_struggle.0001 = {
	hidden = yes
	scope = none

	trigger = {
		exists = struggle:assamite_struggle
	}

	immediate = {		
		# Activate the catalyst.
		if = {
			limit = {
				exists = global_var:influence_wok
				global_var:influence_wok >= 5
			}
			struggle:assamite_struggle = { activate_struggle_catalyst = catalyst_massive_influence_wok }
		}
		if = {
			limit = {
				exists = global_var:influence_wok
				global_var:influence_wok <= 0
			}
			struggle:assamite_struggle = { activate_struggle_catalyst = catalyst_no_influence_wok }
		}
		if = {
			limit = {
				exists = global_var:influence_ashirra
				global_var:influence_ashirra >= 5
			}
			struggle:assamite_struggle = { activate_struggle_catalyst = catalyst_massive_influence_ashirra }
		}
		if = {
			limit = {
				exists = global_var:influence_ashirra
				global_var:influence_ashirra <= 0
			}
			struggle:assamite_struggle = { activate_struggle_catalyst = catalyst_no_influence_ashirra }
		}
		if = {
			limit = {
				exists = global_var:influence_council
				global_var:influence_council >= 5
			}
			struggle:assamite_struggle = { activate_struggle_catalyst = catalyst_massive_influence_council }
		}
		if = {
			limit = {
				exists = global_var:influence_council
				global_var:influence_council <= 0
			}
			struggle:assamite_struggle = { activate_struggle_catalyst = catalyst_no_influence_council }
		}
		if = {
			limit = {
				exists = global_var:influence_silsila
				global_var:influence_silsila >= 5
			}
			struggle:assamite_struggle = { activate_struggle_catalyst = catalyst_massive_influence_silsila }
		}
		if = {
			limit = {
				exists = global_var:influence_silsila
				global_var:influence_silsila <= 0
			}
			struggle:assamite_struggle = { activate_struggle_catalyst = catalyst_no_influence_silsila }
		}
		if = {
			limit = {
				exists = global_var:influence_losttribe
				global_var:influence_losttribe >= 5
			}
			struggle:assamite_struggle = { activate_struggle_catalyst = catalyst_massive_influence_losttribe }
		}
		if = {
			limit = {
				exists = global_var:influence_losttribe
				global_var:influence_losttribe <= 0
			}
			struggle:assamite_struggle = { activate_struggle_catalyst = catalyst_no_influence_losttribe }
		}

		#Simulate the AI doing stuff

		random_list = {
			40 = { }
			
			10 = { 
				if = {
					limit = { 
						exists = global_var:influence_wok 
						global_var:influence_wok < 5
						}
					change_global_variable = {
						name = influence_wok
						add = 1
					}
				}
			}
			
			10 = { 
				if = {
					limit = { exists = global_var:influence_wok }
					change_global_variable = {
						name = influence_wok
						add = -1
					}
				}
			}			

			20 = { 
				if = {
					limit = { 
						exists = global_var:influence_ashirra 
						global_var:influence_ashirra < 5
						}
					change_global_variable = {
						name = influence_ashirra
						add = 1
					}
				}
			}
			
			5 = { 
				if = {
					limit = { exists = global_var:influence_ashirra }
					change_global_variable = {
						name = influence_ashirra
						add = -1
					}
				}
			}
			
			10 = { 
				if = {
					limit = { 
						exists = global_var:influence_council 
						global_var:influence_council < 5
						}
					change_global_variable = {
						name = influence_council
						add = 1
					}
				}
			}
			
			10 = { 
				if = {
					limit = { exists = global_var:influence_council }
					change_global_variable = {
						name = influence_council
						add = -1
					}
				}
			}			

			5 = { 
				if = {
					limit = { 
						exists = global_var:influence_silsila 
						global_var:influence_silsila < 5
						}
					change_global_variable = {
						name = influence_silsila
						add = 1
					}
				}
			}
			
			10 = { 
				if = {
					limit = { exists = global_var:influence_silsila }
					change_global_variable = {
						name = influence_silsila
						add = -1
					}
				}
			}
			
			5 = { 
				if = {
					limit = { 
						exists = global_var:influence_losttribe 
						global_var:influence_losttribe < 5
						}
					change_global_variable = {
						name = influence_losttribe
						add = 1
					}
				}
			}
			
			15 = { 
				if = {
					limit = { exists = global_var:influence_losttribe }
					change_global_variable = {
						name = influence_losttribe
						add = -1
					}
				}
			}
		}

		trigger_event = {
			id = pod_assamite_struggle.0001
			years = 1
		}			
	}
}

pod_assamite_struggle.100 = {
	type = character_event
	content_source = dlc_POD
	title = pod_assamite_struggle.100.t
	desc = pod_assamite_struggle.100.desc
	
	theme = realm
	left_portrait = {
		character = root
		animation = war_over_win
	}

	trigger = {
	}

	immediate = {
		save_scope_as = iberian_footholder
		every_held_title = {
			limit = {
				# Any de jure kingdom of Iberia is completely controlled
				struggle_secure_assamite_foothold_assamite_kingdom_trigger = yes
			}
			add_to_list = iberian_footholder_list
		}
		hidden_effect  = {
			while = {
				limit = {
					any_in_list = {
						list = iberian_footholder_list
						# Borders held lands in a non-Iberian kingdom you hold
						any_de_jure_county = {
							any_title_to_title_neighboring_and_across_water_county = { struggle_secure_assamite_foothold_outsider_kingdom_trigger = yes }
						}
						NOT = { is_in_list = iberian_footholder_transfer_list }
					}
				}
				random_in_list = {
					list = iberian_footholder_list
					limit = {
						# Borders held lands in a non-Iberian kingdom you hold
						any_de_jure_county = {
							any_title_to_title_neighboring_and_across_water_county = { struggle_secure_assamite_foothold_outsider_kingdom_trigger = yes }
						}
						NOT = { is_in_list = iberian_footholder_transfer_list }
					}
					random_de_jure_county = {
						limit = {
							any_title_to_title_neighboring_and_across_water_county = { struggle_secure_assamite_foothold_outsider_kingdom_trigger = yes }
						}
						random_title_to_title_neighboring_and_across_water_county = {
							limit = { struggle_secure_assamite_foothold_outsider_kingdom_trigger = yes }
							empire = { save_scope_as = transfer_empire }
						}
					}
					add_to_list = iberian_footholder_transfer_list
					set_de_jure_liege_title = scope:transfer_empire
					custom_tooltip = iberian_footholder_transfer_tt
				}
			}
		}
		show_as_tooltip = {
			every_in_list = {
				list = iberian_footholder_transfer_list
				set_de_jure_liege_title = scope:transfer_empire
			}
		}
	}

	option = {
		name = pod_assamite_struggle.100.a
	}

	after = {
		struggle:assamite_struggle = {
			every_involved_ruler = {
				limit = { is_ai = no }
				trigger_event = pod_assamite_struggle.101
			}
		}
		# End Struggle
		struggle:assamite_struggle = {
			end_struggle = secure_assamite_foothold_decision_title
		}
	}
}

pod_assamite_struggle.102 = {
	type = character_event
	content_source = dlc_POD
	title = pod_assamite_struggle.102.t
	desc = pod_assamite_struggle.102.desc
	theme = banu_haqim_coa
	left_portrait = {
		character = scope:victorious_losttribe
		animation = personality_zealous
	}

	immediate = {
		play_music_cue = "mx_cue_epic_sacral_moment"	
	}

	option = {
		name = pod_assamite_struggle.102.a
			
		add_choleric_lifestyle_xp = 2000
		add_sanguine_lifestyle_xp = 2000
		
		
		add_prestige = massive_prestige_value
		add_prestige_level = 2
			
		custom_tooltip = pod_assamite_struggle.102.custom
	
		hidden_effect = {		
		every_living_character = {
		limit = {
			is_assamite_trigger = yes
			NOT = { faith = root.faith }
			NOT = { has_government = theocracy_government }
			NOT = { this = this.faith.religious_head }
		}
		trigger_event = pod_assamite_struggle.103
		}
		}
		
		hidden_effect = {
		every_player = { 
		limit = {
			is_assamite_trigger = no
			NOT = { this = scope:victorious_losttribe }
			}
		trigger_event = pod_assamite_struggle.104
			}
		}		
		
		add_character_flag = victorious_assamite_struggle
		custom_tooltip = assamite_struggle_victory_effect_tooltip
		if = {
			limit = {
				title:e_assamites = {			
				holder = { NOT = { this = scope:victorious_losttribe } }
				holder = { character_any_involvement_assamite_struggle_trigger = yes }
				}
				}
				create_title_and_vassal_change = {
					type = created
					save_scope_as = change
					add_claim_on_loss = no
				}
		
			title:e_assamites = {
				change_title_holder = {
					holder = root
					change = scope:change
				}
			}
		resolve_title_and_vassal_change = scope:change
		hidden_effect = { set_primary_title_to = title:e_assamites }
		}	
		
		# End Struggle
		struggle:assamite_struggle = {
			end_struggle = losttribe_victory_objective_decision_title
		}
		if = {
			limit = { exists = global_var:influence_wok }
			remove_global_variable = influence_wok
		}
		if = {
			limit = { exists = global_var:influence_silsila }
			remove_global_variable = influence_silsila
		}	
		if = {
			limit = { exists = global_var:influence_council }
			remove_global_variable = influence_council
		}
		if = {
			limit = { exists = global_var:influence_ashirra }
			remove_global_variable = influence_ashirra
		}
		if = {
			limit = { exists = global_var:influence_losttribe }
			remove_global_variable = influence_losttribe
		}		
	}
}

pod_assamite_struggle.1000 = {
	type = character_event
	content_source = dlc_POD
	title = pod_assamite_struggle.1000.t
	desc = pod_assamite_struggle.1000.desc
	
	theme = banu_haqim_coa
	left_portrait = root
	
	trigger = {		
		#Must hold a county with a city in it.
		any_held_title = {
			pod_assamite_struggle_1000_valid_county = yes
		}
		is_landless_adventurer = no
		character_any_involvement_assamite_struggle_trigger = yes
	}

	cooldown = { #this shouldn't fire too often
		years = 30
	}

	immediate = {
		random_held_title = {
			limit = {
				pod_assamite_struggle_1000_valid_county = yes
			}
			save_scope_as = heretic_county
		}
	}

	#Favor the Wok over the LT
	option = {
		name = pod_assamite_struggle.1000.a

		custom_tooltip = wok_gain_influence_tt

		if = {
			limit = { exists = global_var:influence_wok }
			change_global_variable = {
				name = influence_wok
				add = 1
			}
		}		

		scope:heretic_county = {
			add_county_modifier = {
				modifier = clashing_heretics_modifier
				days = 3650
			}
		}
	}
	
	#Favor the LT over the WoK
	option = {
		name = pod_assamite_struggle.1000.b

		custom_tooltip = losttribe_gain_influence_tt

		if = {
			limit = { exists = global_var:influence_wok }
			change_global_variable = {
				name = influence_losttribe
				add = 1
			}
		}		

		scope:heretic_county = {
			add_county_modifier = {
				modifier = clashing_heretics_modifier
				days = 3650
			}
		}
	}

	#Manipulate each faction to end the conflict
	option = {
		name = pod_assamite_struggle.1000.c

		duel = {
			skill = intrigue
			value = 15
		
			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
					min = -49
				}
				desc = pod_assamite_struggle.1000.a.success
				send_interface_toast = {
					type = event_toast_effect_good
					left_icon = scope:heretic_county
					title = pod_assamite_struggle.1000.a.success
					add_intrigue_lifestyle_xp = 150
				}
			}
			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
					min = -49
				}
				desc = pod_assamite_struggle.1000.a.failure
				send_interface_toast = {
					type = event_toast_effect_bad
					left_icon = scope:heretic_county
					title = pod_assamite_struggle.1000.a.failure
					scope:heretic_county = {
						add_county_modifier = {
							modifier = clashing_heretics_modifier
							days = 7300
						}
					}
				}
			}
		}
	}

	#None of them
	option = {
		name = pod_assamite_struggle.1000.d

		custom_tooltip = wok_lose_influence_tt

		custom_tooltip = losttribe_lose_influence_tt

		if = {
			limit = { exists = global_var:influence_wok }
			change_global_variable = {
				name = influence_wok
				add = -1
			}
		}

		if = {
			limit = { exists = global_var:influence_wok }
			change_global_variable = {
				name = influence_losttribe
				add = -1
			}
		}
		
		scope:heretic_county = {
			add_county_modifier = {
				modifier = clashing_heretics_modifier
				days = 7300
			}
		}
		scope:heretic_county = {
			change_county_control = -20
		}

	}
}
