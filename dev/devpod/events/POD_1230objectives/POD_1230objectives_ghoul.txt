	type = character_event
	title = pod1230objectives_ghoul.1.t
	desc = pod1230objectives_ghoul.1.desc
	theme = vampire

	left_portrait = { 
		character = ROOT
		animation = idle
	}
	right_portrait = scope:vampiresire	

	option = {
		name = pod1230objectives_ghoul.1.a
		POD_become_vampire_effect = yes
		custom_tooltip = {
			text = ultimate_submission_objective_decision_effect1
			POD_remove_supernatural_relation_effect = yes
			
			save_scope_as = created_vampirechilde
			scope:vampiresire = {
				add_hook = {
					target = PREV
					type = strong_bloodoath_hook
				}
			}
			POD_remove_child_relation_effect = yes
			POD_set_parent_effect = { PARENT = scope:vampiresire }
		
			set_house = scope:vampiresire.house	
			
			POD_character_vampire_inherit_generation_effect = yes
			POD_character_vampire_inherit_clan_effect = yes
			set_character_faith = scope:vampiresire.faith
			POD_character_vampire_bloodpotency_initialization_effect = yes
			hidden_effect = { POD_remove_hunger_modifiers_effect = yes }
			POD_assign_random_bloodhunger = yes
			POD_character_vampire_discipline_initialization_effect = yes
			POD_character_masquerade_initialization_effect = yes
			POD_character_vampire_clan_special_effect = yes
			POD_salubri_caste_new_vampire_initialization_effect = yes
			
			if = {
				limit = { POD_splat_has_disease_immunity = yes }
				POD_remove_disease_traits_effect = yes
			}
			
			POD_set_resonance_effect = yes	
			
			POD_commander_trait_initialization_effect = yes
			
			POD_side_effects_on_embrace = yes
		}
		add_character_modifier = POD_fulfilled_lifes_goal_modifier
	}

}

		}
		widget = {
			gui = "POD_shader_event_fire"
			container = "background_shader_vfx_container"
		}
	}
	left_portrait = {
		character = root
		animation = assassin
	}

	option = {
		name = pod1230objectives_ghoul.3.a
		custom_tooltip = the_vitae_rebellion_objective_decision_effect1
		custom_tooltip = the_vitae_rebellion_objective_decision_effect2
		custom_tooltip = the_vitae_rebellion_objective_decision_effect3
		remove_character_flag = declared_major_religious_war_flag
		custom_tooltip = {
			text = the_vitae_rebellion_objective_decision_effect4
			every_sub_realm_county = {
				add_to_list = checked_counties
			}
			every_living_character = {
				limit = {
					POD_is_masterless_ghoul_trigger = yes
					location ?= { county = { is_in_list = checked_counties } }
				}
				set_character_faith = faith:unmastered
			}
		}
		custom_tooltip = the_vitae_rebellion_objective_decision_effect5
		show_as_tooltip = {
			if = {
				limit = {
					is_target_in_global_variable_list = {
						name = unavailable_unique_decisions
						target = flag:contact_the_dispossessed_objective_decision
					}
				}
				faith:unmastered = { add_doctrine = special_masterless_ghoul_dispossessed_doctrine }
			}
			if = {
				limit = {
					is_target_in_global_variable_list = {
						name = unavailable_unique_decisions
						target = flag:contact_the_folch_cutthroats_objective_decision
					}
				}
				faith:unmastered = { add_doctrine = special_masterless_ghoul_folch_cutthroats_doctrine }
			}
			if = {
				limit = {
					is_target_in_global_variable_list = {
						name = unavailable_unique_decisions
						target = flag:contact_the_order_of_st_panteleimon_objective_decision
					}
				}
				faith:unmastered = { add_doctrine = special_masterless_ghoul_order_of_st_panteleimon_doctrine }
			}
			if = {
				limit = {
					is_target_in_global_variable_list = {
						name = unavailable_unique_decisions
						target = flag:contact_the_haqims_assassins_objective_decision
					}
				}
				faith:unmastered = { add_doctrine = special_masterless_ghoul_haqims_assassins_doctrine }
			}
		}
	}
}
