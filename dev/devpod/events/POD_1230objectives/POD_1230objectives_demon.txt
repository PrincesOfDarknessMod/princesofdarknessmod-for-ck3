pod1230objectives_demons.6 = {
	type = character_event
	content_source = dlc_POD
	title = pod1230objectives_demons.6.t
	desc = pod1230objectives_demons.6.desc
	theme = diplomacy_majesty_focus
	left_portrait = {
		character = root
		animation = personality_honorable
	}

	immediate = {
		play_music_cue = "mx_cue_epic_sacral_moment"	
	}

	option = {
		name = pod1230objectives_demons.6.a
				
		add_prestige = massive_prestige_value
		dynasty = { add_dynasty_prestige = 3000 }
		title:c_kufa.title_province = { add_special_building = hanging_gardens_01 }
		
		hidden_effect = {
			every_held_title = { #Should shift all dejure of all Empires owned at the time.
				limit = {
					tier = tier_empire
				}
				every_in_de_jure_hierarchy = {
					limit = {
						tier = tier_kingdom
					}
					set_de_jure_liege_title = title:e_babylon
					
				}
			}
			every_held_title = { #Completely Controlled Kingdoms as well.
				limit = {
					tier = tier_kingdom
					root = { completely_controls = prev }
				}
				set_de_jure_liege_title = title:e_babylon
			}
		}
	
		create_title_and_vassal_change = {
			type = created
			save_scope_as = change
			add_claim_on_loss = no
		}

		title:e_babylon = {
			change_title_holder = {
				holder = root
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change
		hidden_effect = { set_primary_title_to = title:e_babylon }
		every_held_title = { #Should destroy all other Empires owned at the time.
			limit = {
				tier = tier_empire
				NOT = { this = title:e_babylon }
			}
			root = { destroy_title = prev }
		}
	}
}
