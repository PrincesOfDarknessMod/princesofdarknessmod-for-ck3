pod1230objectives_judges.1 = {
	type = character_event
	content_source = dlc_POD
	title = pod1230objectives_judges.1.t
	desc = pod1230objectives_judges.1.desc
	theme = faith
	left_portrait = {
		character = root
		animation = personality_honorable
	}

	immediate = {
		play_music_cue = "mx_cue_epic_sacral_moment"
	}
	
	option = {
		name = pod1230objectives_judges.1.a
		
		if = {
			limit = { POD_splat_is_valid_for_sorcery = yes }
			add_trait = sorcerer
		}
		
		faith:judge = {
			POD_make_crime_doctrine_accepted_effect = { DOCTRINE = doctrine_magicraft }
			
			if = {
				limit = { prev = { POD_splat_is_valid_for_sorcery = yes } }
				add_doctrine = special_doctrine_sorcerer_ordination
			}
		}

		if = {
			limit = { NOT = { faith = faith:judge } }
			faith = {
				POD_make_crime_doctrine_accepted_effect = { DOCTRINE = doctrine_magicraft }
				
				if = {
					limit = { prev = { POD_splat_is_valid_for_sorcery = yes } }
					add_doctrine = special_doctrine_sorcerer_ordination
				}
			}
		}

		POD_add_free_dynasty_perk = { PERK = hunter_judges_legacy_1 }

		# Special Menahemite event if they reestablish contact with Magi
		if = {
			limit = { 
				house = house:house_menahemites
			}
			trigger_event = pod1230objectives_judges.111
		}

		#Grant Havanah artifact for advanced objectives
		if = {
			limit = {
				faith = faith:judge
				NOT = { any_character_artifact = { has_variable = judge_havanah_book }	 }
			}
			custom_tooltip = {
				text = judge_contact_objective_decision_havanah_tt
				create_artifact_pod_judges_havanah_book_effect = { OWNER = root }
				POD_save_new_artifact_as_link = { LINK = judge_havanah_book_link }
			}
		}
	}
}

		house ?= { add_house_modifier = house_menahemites_menahem_teachers_modifier }
		dynasty ?= { add_dynasty_prestige = 50 }
	}
}

pod1230objectives_judges.24 = {
	type = character_event
	title = pod1230objectives_judges.24.t
	desc = pod1230objectives_judges.24.desc
	theme = faith
	override_background = {
		reference = temple_generic
	}
	left_portrait = {
		character = scope:portrait_ruler
		triggered_animation = {
			trigger = { NOT = { scope:new_priest = scope:scoped_ruler } }
			animation = happiness
		}
		triggered_animation = {
			trigger = { scope:new_priest = scope:scoped_ruler }
			animation = personality_zealous
		}
	}
	right_portrait = {
		character = scope:new_priest
		animation = personality_zealous
		trigger = { scope:new_priest != scope:portrait_ruler }
	}

	immediate = {
		play_music_cue = "mx_cue_faith_conversion"
		
		save_scope_as = scoped_ruler
		save_scope_as = portrait_ruler
		
		#Allow head of faith
		faith = {
			if = {
				limit = { has_doctrine = special_doctrine_not_allowed_to_hof }
				remove_doctrine = special_doctrine_not_allowed_to_hof
			}
		}
		#Add a head of faith
		# if = { # Check if Temporal first.
		# 	limit = {
		# 		faith = {
		# 			OR = {
		# 				has_doctrine_parameter = lay_clergy
		# 				has_doctrine = doctrine_temporal_head
		# 			}
		# 		}
		# 	}
		# 	faith = {
		# 		if = {
		# 			limit = { has_doctrine = doctrine_no_head }
		# 			hidden_effect = {
		# 				remove_doctrine = doctrine_no_head
		# 				add_doctrine = doctrine_temporal_head
		# 			}
		# 		}
		# 	}
		# 	save_scope_as = new_priest
		# }
		if = { # Pick your chaplain first.
			limit = {
				exists = cp:councillor_court_chaplain
				cp:councillor_court_chaplain = { faith = scope:scoped_ruler.faith }
			}
			faith = {
				if = {
					limit = { has_doctrine = doctrine_no_head }
					hidden_effect = {
						remove_doctrine = doctrine_no_head
						add_doctrine = doctrine_spiritual_head
					}
				}
			}
			cp:councillor_court_chaplain = { save_scope_as = new_priest }
		}
		else = {
			hidden_effect_new_object = {
				create_character = {
					employer = root
					template = religious_leader_character
					random_traits = no
					save_scope_as = new_priest
					dynasty = generate
				}
				scope:new_priest = {
					add_trait = education_learning_3
					add_trait = zealous
					add_trait = intellect_good_1
					random_list = {
						1 = { add_trait = lustful }
						1 = { add_trait = chaste }
						1 = { add_trait = wrathful }
						1 = { add_trait = diligent }
						1 = { add_trait = impatient }
						1 = { add_trait = arrogant }
						1 = { add_trait = humble }
						1 = { add_trait = gregarious }
					}
					random_list = {
						1 = { add_trait = honest }
						1 = { add_trait = ambitious }
						1 = { add_trait = just }
						1 = { add_trait = cynical }
						1 = { add_trait = zealous }
						1 = { add_trait = compassionate }
						1 = { add_trait = stubborn }
					}
				}
			}
		}

		custom_description = {
			text = "new_religious_head_title"
			object = scope:new_priest
			subject = scope:new_priest

			set_up_dynamic_spiritual_hof_title_existing_hof_effect = {
				CREATOR = scope:scoped_ruler
				RELIGIOUS_HEAD = scope:new_priest
			}
		}
	}

	option = {
		name = pod1230objectives_judges.24.a
		give_nickname = nick_the_judge
		faith = {
			change_fervor = {
				value = 100
				desc = fervor_gain_priesthood_recreated
			}
		}
		if = { # Vassalize new HoF if possible/needed.
			limit = {
				highest_held_title_tier > scope:new_priest.highest_held_title_tier
				NOT = { scope:new_priest = { any_liege_or_above = { this = root } } }
			}
			create_title_and_vassal_change = {
				type = created
				save_scope_as = change
				add_claim_on_loss = no
			}
			scope:new_priest = {
				change_liege = {
					liege = root
					change = scope:change
				}
			}
			resolve_title_and_vassal_change = scope:change
		}

		create_title_and_vassal_change = {
			type = created
			save_scope_as = change
			add_claim_on_loss = no
		}

		title:e_judges_title = {
			change_title_holder = {
				holder = root
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change
		hidden_effect = { set_primary_title_to = title:e_judges_title }
	}
}
