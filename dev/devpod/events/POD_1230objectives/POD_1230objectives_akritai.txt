pod1230objectives_akritai.0001 = {
	type = character_event
	content_source = dlc_POD
	title = pod1230objectives_baghaturs.0001.t
	desc = pod1230objectives_baghaturs.0001.desc
	theme = christian
	left_portrait = {
		character = scope:canonizer
		animation = prayer_standing_eyes
	}
	right_portrait = {
		character = scope:saint_candidate
		animation = dead
	}

	immediate = {
		hidden_effect = {
			scope:saint_candidate = {
				save_scope_value_as = {
					name = quality
					value = akritai_saint_relic_quality_value
				}
			}
			POD_create_appropriate_akritai_relic_artifact_from_character_effect = {
				OWNER = scope:canonizer
				CREATOR = scope:saint_candidate
			}
			if = {
				limit = { exists = scope:newly_created_artifact }
				hidden_effect_new_object = {
					scope:newly_created_artifact = { save_scope_as = relic_artifact }
					scope:relic_artifact = {
						# Stamped onto the artifact because saint_candidate is only in scope during extraction
						# Resuses the PAM artifact template so I reuse its variable name
						set_variable = {
							name = akritai_saint_relic_saint
							value = scope:saint_candidate
						}
					}
				}
			}
		}
	}
	
	option = {
		name = pod1230objectives_baghaturs.0001.a
		custom_tooltip = pod1230objectives_baghaturs.0001.a.tt
		
		scope:saint_candidate = {
			add_trait = saint
		}
	}
}

pod1230objectives_akritai.20 = {
	type = character_event
	content_source = dlc_POD
	title = pod1230objectives_akritai.20.t
	desc = pod1230objectives_akritai.20.desc
	theme = faith
	left_portrait = root

	immediate = {
		play_music_cue = "mx_cue_epic_sacral_moment"
	}

	option = {
		name = pod1230objectives_akritai.20.a

		faith:akritai = {
			add_doctrine = special_doctrine_hunter_schism_mended
			change_fervor = {
				value = 25
				desc = fervor_gain_mended_schism
			}
		}
		faith:shadowinquisition = {
			change_fervor = {
				value = -25
				desc = fervor_loss_mended_schism
			}
		}

		custom_tooltip = {
			text = akritai_schism_tt

		every_living_character = {
			limit = {
				is_ai = yes
				has_faith = faith:shadowinquisition
			}
		set_character_faith = faith:akritai

		#Reset their buildings, since incompatible post conversion
		every_held_county = {
			random_county_province = {
				limit = {
				has_holding_type = castle_holding
				}
				set_holding_type = tribal_holding
				set_holding_type = castle_holding
			}

			random_county_province = {
				limit = {
				has_holding_type = city_holding
				}
				set_holding_type = tribal_holding
				set_holding_type = city_holding
			}

			random_county_province = {
				limit = {
				has_holding_type = church_holding
				}
				set_holding_type = tribal_holding
				set_holding_type = church_holding
			}

			random_county_province = {
				limit = {
				has_holding_type = grand_city_holding
				}
				POD_reset_grand_city_effect = yes
			}
		}
		}
		}

		set_nickname_effect = { NICKNAME = nick_the_ecumenist }
	}
}
