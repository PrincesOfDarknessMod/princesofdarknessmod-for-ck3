				}
			}
				
			religion:stargazer_religion = {
				every_faith = {
					remove_doctrine = doctrine_supehunting_crime
					add_doctrine = doctrine_supehunting_accepted
				}
			}
			
			religion:uktena_religion = {
				every_faith = {
					remove_doctrine = doctrine_supehunting_crime
					add_doctrine = doctrine_supehunting_accepted
				}
			}	
			
			# pod1230objectives_baghaturs.100 needs to occur 18 days before May 14, 1230

			trigger_event = {
				id = pod1230objectives_baghaturs.100
				days = 96
			}
		}
	}
}

	type = character_event

		animation = horse_sword_shield

	}
	override_background = { reference = eye_of_the_storm_background }

	immediate = {}
	
	option = {
		name = pod1230objectives_baghaturs.131.a
		set_relation_best_friend = { reason = best_friend_counsel_and_support target = character:792588 }

		if = {
			limit = { var:POD_qorjin_intro_bonus = 1 }

		}
	}

	after = {
		remove_variable = POD_qorjin_intro_bonus
