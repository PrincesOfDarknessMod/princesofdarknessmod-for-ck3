pod_lasombraevents.1 = {
	type = character_event
	content_source = dlc_POD
	title = pod_lasombraevents.1.n
	desc = { 
		first_valid = {
			triggered_desc = {
				trigger = {
					exists = scope:helper
				}
				desc = pod_lasombraevents.1.loverdesc
			}

			triggered_desc = {
				desc = pod_lasombraevents.1.desc
			}

		}
	}
	theme = lasombra
	left_portrait = {
		character = root
		animation = stress
		triggered_animation = {
			trigger = {
				has_trait = arrogant
			}
			animation = personality_bold
		}
	}	
	right_portrait = {
			character = scope:helper
			animation = personality_compassionate
		}

	trigger = {
	}

	immediate = {
		random_relation = {
			type = lover
			save_scope_as = helper
		}
		
	}

	option = { 
		name = pod_lasombraevents.1.a

		custom_tooltip = pod_lasombraevents.1.a.t
		add_character_modifier = {
            modifier = pareidolia_modifier
            years = 1
        }
        remove_short_term_gold = minor_gold_value
		add_stress = 30
	}

	option = { 
		name = pod_lasombraevents.1.b
		trigger = {
			NOT = { 
				has_trait = arrogant 
				exists = scope:helper
			}
		}

		custom_tooltip = pod_lasombraevents.1.b.t
        add_character_modifier = {
            modifier = pareidolia_modifier
            years = 5
        }
		add_stress = 70
		if = {
			limit = { POD_has_a_beast_trigger = yes }
			custom_tooltip = pod_lasombraevents.1.b.beast
		}
	}

	option = {
		name = pod_lasombraevents.1.c
		trigger = {
			has_trait = arrogant
		}

		custom_tooltip = pod_lasombraevents.1.c.t
		trait = arrogant
		add_character_modifier = {
			modifier = lasombraperfect_modifier
			years = 1
		}

	}

	option = {
		name = pod_lasombraevents.1.d
		trigger = {
			exists = scope:helper
			NOT = { has_trait = arrogant }
		}

		custom_tooltip = pod_lasombraevents.1.d.t
		scope:helper = {
			add_opinion = {
				target = root
				modifier = love_opinion
				opinion = 30
			}
		}
		scope:helper = {
			add_character_modifier = {
				modifier = lasombrahelper_modifier
				years = 1
			}
		}
	}
}

		name = pod_lasombraevents.2.a

		}

			target = scope:recipient
		}

	else = {}

	trigger_event = { id = pod_lasombraevents.3 days = 5 } 
	}
	
	option = {
		name = pod_lasombraevents.2.b
	}	
}

pod_lasombraevents.9 = {
	hidden = yes
	
	immediate = {
		if = {
			limit = {
				NOT = { has_character_flag = won_one_court_blood_flag }
				NOT = { has_character_flag = won_two_court_blood_flag }
			}
		trigger_event = pod_lasombraevents.10
		}
		else_if = {
			limit = {
				has_character_flag = won_one_court_blood_flag 
				has_character_flag = won_two_court_blood_flag 
			}
		trigger_event = pod_lasombraevents.11
		}		
		else = {
			random_list = {
				50 = { trigger_event = pod_lasombraevents.10 }
				50 = { trigger_event = pod_lasombraevents.11 }
			}
		}
	}
}
