namespace = pod_lasombraevents

#Control event for Creed flavor stuff

pod_lasombraevents.1001 = {
	type = character_event
	hidden = yes
	
	immediate = {
	
		if = {
			limit = {
					is_vampire_trigger = yes
					is_imprisoned = no
					NOT = {
						has_trait = torpor
						}
					has_trait = lasombra
					age > 150
				}
			trigger_event = pod_lasombraevents.1	
		}
	}
}


pod_lasombraevents.1 = {
	type = character_event
	title = pod_lasombraevents.1.n
	desc = { 
		first_valid = {
			triggered_desc = {
				trigger = {
					exists = scope:helper
				}
				desc = pod_lasombraevents.1.loverdesc
			}

			triggered_desc = {
				desc = pod_lasombraevents.1.desc
			}

		}
	}
	theme = lasombra
	left_portrait = {
		character = root
		animation = stress
		triggered_animation = {
			trigger = {
				has_trait = arrogant
			}
			animation = personality_bold
		}
	}	
	right_portrait = {
			character = scope:helper
			animation = personality_compassionate
		}

	trigger = {
	}

	immediate = {
		random_relation = {
			type = lover
			save_scope_as = helper
		}
		
	}

	option = { 
		name = pod_lasombraevents.1.a

		custom_tooltip = pod_lasombraevents.1.a.t
		add_character_modifier = {
            modifier = pareidolia_modifier
            years = 1
        }
        remove_short_term_gold = minor_gold_value
		add_stress = 30
	}

	option = { 
		name = pod_lasombraevents.1.b
		trigger = {
			NOT = { 
				has_trait = arrogant 
				exists = scope:helper
			}
		}

		custom_tooltip = pod_lasombraevents.1.b.t
        add_character_modifier = {
            modifier = pareidolia_modifier
            years = 5
        }
		add_stress = 70
		custom_tooltip = pod_lasombraevents.1.b.beast
	}

	option = {
		name = pod_lasombraevents.1.c
		trigger = {
			has_trait = arrogant
		}

		custom_tooltip = pod_lasombraevents.1.c.t
		trait = arrogant
		add_character_modifier = {
			modifier = lasombraperfect_modifier
			years = 1
		}

	}

	option = {
		name = pod_lasombraevents.1.d
		trigger = {
			exists = scope:helper
			NOT = { has_trait = arrogant }
		}

		custom_tooltip = pod_lasombraevents.1.d.t
		scope:helper = {
			add_opinion = {
				target = root
				modifier = love_opinion
				opinion = 30
			}
		}
		scope:helper = {
			add_character_modifier = {
				modifier = lasombrahelper_modifier
				years = 1
			}
		}
	}
}

pod_lasombraevents.2 = {
	type = character_event
	title = pod_lasombraevents.2.n
	desc = pod_lasombraevents.2.desc
	theme = lasombra
	left_portrait = {
		character = root
		animation = schadenfreude
	}	
	right_portrait = {
			character = scope:recipient
			animation = worry
	}
	
	option = {
		name = pod_lasombraevents.2.a

	if = {
		limit = {
			can_set_relation_nemesis_trigger = { CHARACTER = scope:recipient }
		}
		set_relation_nemesis = {
			reason = nemesis_historical
			target = scope:recipient
		}
	}

	custom_tooltip = courts_blood_tooltip
	
	trigger_event = { id = pod_lasombraevents.3 days = 5 } 
	}
	
	option = {
		name = pod_lasombraevents.2.b
	}	
}

		}
	}
		
		duel = {
			skill = diplomacy
			value = scope:recipient.diplomacy
				10 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = 0.5
					}
				desc = pod_lasombraevents.3.effect.success.desc
				
				add_character_flag = won_one_court_blood_flag
				
			send_interface_toast = {
				title = pod_lasombraevents.3.effect.success.desc
				left_icon = root
				}				
				
			}
				10 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = -0.5
					}

					desc = pod_lasombraevents.3.critical.failure.effect.desc
					
					send_interface_toast = {
						title = pod_lasombraevents.3.critical.failure.effect.desc
						left_icon = root
					}					
				}
			}		
		
	}
