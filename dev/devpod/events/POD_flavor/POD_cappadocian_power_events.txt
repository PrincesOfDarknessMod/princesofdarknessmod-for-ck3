POD_cappadocian_power.2000 = {
	type = character_event
	content_source = dlc_POD
	title = POD_cappadocian_power.2000.title
	desc = POD_cappadocian_power.2000.desc
	theme = ossuary
	
	left_portrait = {
		character = scope:recipient
		animation = ecstasy
	}
	right_portrait = {
		character = scope:secondary_recipient
		animation = personality_rational
	}
	
	immediate = {
		scope:recipient = { add_character_modifier = { modifier = councilor_diplomacy_modifier  years = 10 } }
		scope:secondary_recipient = { add_diplomacy_lifestyle_perk_points = 1 }
	}
	option = {
		name = POD_cappadocian_power.optiongood
		scope:recipient = { add_opinion = { modifier = POD_great_councilor_childe target = scope:actor } }
		scope:actor = { add_hook = { target = scope:recipient type = favor_hook } }
		scope:secondary_recipient = {
			remove_character_flag = diplomacy_job_flag
			remove_character_flag = steward_job_flag
			remove_character_flag = marshal_job_flag
			remove_character_flag = spymaster_job_flag
			remove_character_flag = chaplain_job_flag
		}
	}

}
