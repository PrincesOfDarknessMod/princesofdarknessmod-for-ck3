### ==========================================
### EVENT 3: "The Necromancer's Offer"
### Fires when an Impundulu has a necromantic vampire in diplomatic range
### Hints at the knowledge exchange interaction
### ==========================================

POD_impundulu.3 = {
	type = character_event
	orphan = yes # Deprecated since the decision searches for targets automatically
	content_source = dlc_POD
	title = POD_impundulu.3.t
	desc = POD_impundulu.3.desc
	theme = vampire
	override_background = { reference = bp1_bonfire }

	left_portrait = {
		character = root
		animation = personality_rational
	}

	right_portrait = {
		character = scope:imp_necromancer_contact
		animation = personality_rational
	}

	trigger = {
		is_impundulu_trigger = yes
		is_alive = yes
		NOT = { has_character_flag = imp_necromancer_offer_cooldown }
	}

	### Find a valid necromancer contact
	immediate = {
		random_ruler = {
			limit = {
				is_vampire_trigger = yes
				has_trait = necromancydiscipline
				NOT = { is_impundulu_trigger = yes }
				opinion = {
					target = root
					value >= 20
				}
				is_alive = yes
			}
			save_scope_as = imp_necromancer_contact
		}

		### Cooldown to prevent spam
		add_character_flag = {
			flag = imp_necromancer_offer_cooldown
			years = 15
		}
	}

	### Intriguing — I should approach them
	option = {
		name = POD_impundulu.3.a
		custom_tooltip = POD_impundulu_necromancer_offer_hint_tt
	}

	### I am not ready for such an exchange
	option = {
		name = POD_impundulu.3.b
	}
}

### Event 3: Hidden — Create the Bomkazi character based on outcome
POD_impundulu.1003 = {
	hidden = yes

	trigger = {
		is_alive = yes
	}

	immediate = {
		### Determine outcome: success, partial, or failure
		random_list = {
			70 = {
				### Full Success — create a proper Bomkazi witch
				create_character = {
					location = root.capital_province
					age = { 18 30 }
					culture = root.culture
					faith = root.faith
					gender_female_chance = 75
					trait = revenant
					learning = { 8 14 }
					diplomacy = { 4 8 }
					martial = { 2 6 }
					stewardship = { 4 8 }
					intrigue = { 4 10 }
					prowess = { 2 5 }
					random_traits = yes
					dynasty = dynasty:12000028
					save_scope_as = bomkazi_candidate
				}

				scope:bomkazi_candidate = {
					add_trait = bomkazi_witch
					add_trait = pure_blooded

					POD_character_revenant_discipline_initialization_effect = yes
					POD_learn_dur_an_ki_bloodmagic_effect = yes

					### Apply quality-specific bonuses
					if = {
						limit = { root = { has_character_flag = imp_bomkazi_seek_scholar } }
						add_trait = intellect_good_1
					}
					else_if = {
						limit = { root = { has_character_flag = imp_bomkazi_seek_healer } }
						add_trait = physique_good_1
					}
					else_if = {
						limit = { root = { has_character_flag = imp_bomkazi_seek_seer } }
						add_trait = beauty_good_1
					}
				}

				trigger_event = { id = POD_impundulu.1004 days = 3 }
			}

			20 = {
				### Partial Success — create a Bomkazi apprentice
				create_character = {
					location = root.capital_province
					age = { 16 25 }
					culture = root.culture
					faith = root.faith
					gender_female_chance = 75
					trait = mortal
					learning = { 5 10 }
					diplomacy = { 3 7 }
					martial = { 2 5 }
					stewardship = { 3 7 }
					intrigue = { 3 8 }
					prowess = { 2 4 }
					random_traits = yes
					dynasty = dynasty:12000028
					save_scope_as = bomkazi_candidate
				}

				scope:bomkazi_candidate = {
					add_trait = bomkazi_apprentice
				}

				trigger_event = { id = POD_impundulu.1005 days = 3 }
			}

			10 = {
				### Failure — no suitable candidate found
				trigger_event = { id = POD_impundulu.1006 days = 3 }
			}
		}

		### Clean up flags
		remove_character_flag = imp_bomkazi_seek_scholar
		remove_character_flag = imp_bomkazi_seek_healer
		remove_character_flag = imp_bomkazi_seek_seer
	}
}

### Impundulu Submod — Knowledge Exchange Events
### 3-event chain: Approach → Negotiation → Exchange
### Reference: POD_disciplinetrain_events.txt

### Event 1: "The Approach" — the negotiation begins
POD_impundulu.2001 = {
	type = character_event
	content_source = dlc_POD
	title = POD_impundulu.2001.t
	desc = POD_impundulu.2001.desc
	theme = vampire
	override_background = { reference = bp1_bonfire }

	left_portrait = {
		character = root
		animation = personality_rational
	}

	right_portrait = {
		character = scope:imp_exchange_target
		animation = personality_rational
	}

	trigger = {
		is_alive = yes
		any_ruler = {
			POD_is_valid_knowledge_exchange_target_for = {
				ACTOR = root
			}
		}
	}
	
	on_trigger_fail = {
		# fallback event if all valid rulers died or became unlanded in the 1-year waiting period
		trigger_event = POD_impundulu.2003
	}

	immediate = {
		ordered_ruler = {
			limit = {
				POD_is_valid_knowledge_exchange_target_for = {
					ACTOR = root
				}
			}
			order_by = {
				# closest rulers first
				value = "capital_province.squared_distance(root.capital_province)"
				multiply = -1
			}
			save_scope_as = imp_exchange_target
		}
	}

	### Option A: Offer gold — a straightforward exchange
	option = {
		name = POD_impundulu.2001.a
		custom_tooltip = POD_impundulu_exchange_offer_gold_tt
		hidden_effect = {
			save_scope_as = offered_gold
			trigger_event = { id = POD_impundulu.2002 days = 30 }
		}
	}

	### Option B: Offer a favor — build the relationship
	option = {
		name = POD_impundulu.2001.b
		custom_tooltip = POD_impundulu_exchange_offer_favor_tt
		hidden_effect = {
			save_scope_as = offered_favor
			trigger_event = { id = POD_impundulu.2002 days = 30 }
		}
	}

	### Option C: Share Impundulu secrets — mutual exchange
	option = {
		name = POD_impundulu.2001.c
		custom_tooltip = POD_impundulu_exchange_offer_secrets_tt
		trigger = {
			scope:imp_exchange_target = {
				has_impundulu_necromancy_trigger = no
			}
		}
		hidden_effect = {
			save_scope_as = offered_secrets
			trigger_event = { id = POD_impundulu.2002 days = 30 }
		}
	}
}

### Event 2: "The Negotiation" — outcome determined
POD_impundulu.2002 = {
	type = character_event
	content_source = dlc_POD
	title = POD_impundulu.2002.t
	desc = POD_impundulu.2002.desc
	theme = vampire
	override_background = { reference = bp1_bonfire }

	left_portrait = {
		character = root
		animation = personality_rational
	}

	right_portrait = {
		character = scope:imp_exchange_target
		animation = personality_rational
	}

	trigger = {
		is_alive = yes
	}

	immediate = {
		### Determine cost based on what was offered
		if = {
			limit = { exists = scope:offered_gold }
			remove_short_term_gold = 200
		}
		else_if = {
			limit = { exists = scope:offered_favor }
			### Create a hook for the target
			scope:imp_exchange_target = {
				add_hook = {
					target = root
					type = favor_hook
				}
			}
		}
		else_if = {
			limit = { exists = scope:offered_secrets }
			### The target also learns Impundulu necromancy
			scope:imp_exchange_target = {
				POD_learn_impundulu_necromancy_effect = yes
			}
		}
	}

	### Success — learn their path
	option = {
		name = POD_impundulu.2002.a
		
		### Ensure character has necromancy trait
		if = {
			limit = {
				NOT = {
					has_trait = necromancydiscipline
				}
			}
			add_trait = necromancy
		}

		### Determine which path the target knows and learn it
		if = {
			limit = {
				has_giovanni_necromancy_trigger = no
				scope:imp_exchange_target = { has_giovanni_necromancy_trigger = yes }
			}
			POD_learn_giovanni_necromancy_effect = yes
		}
		else_if = {
			limit = {
				has_cappadocian_necromancy_trigger = no
				scope:imp_exchange_target = { has_cappadocian_necromancy_trigger = yes }
			}
			POD_learn_cappadocian_necromancy_effect = yes
		}
		else_if = {
			limit = {
				has_harbinger_necromancy_trigger = no
				scope:imp_exchange_target = { has_harbinger_necromancy_trigger = yes }
			}
			POD_learn_harbinger_necromancy_effect = yes
		}
		else_if = {
			limit = {
				has_lamia_necromancy_trigger = no
				scope:imp_exchange_target = { has_lamia_necromancy_trigger = yes }
			}
			POD_learn_lamia_necromancy_effect = yes
		}
		else_if = {
			limit = {
				has_nagaraja_necromancy_trigger = no
				scope:imp_exchange_target = { has_nagaraja_necromancy_trigger = yes }
			}
			POD_learn_nagaraja_necromancy_effect = yes
		}
		else_if = {
			limit = {
				has_rosselini_necromancy_trigger = no
				scope:imp_exchange_target = { has_rosselini_necromancy_trigger = yes }
			}
			POD_learn_rosselini_necromancy_effect = yes
		}

		add_prestige = minor_prestige_value
	}
}
