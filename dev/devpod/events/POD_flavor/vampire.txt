namespace = pod_vampire_flavor

# Vampire misses seeing the sun
# Contributed by user VoltaDoLargo
pod_vampire_flavor.1000 = {
	type = character_event
	content_source = dlc_POD
	title = pod_vampire_flavor.1000.t
	desc = pod_vampire_flavor.1000.desc

	theme = vampire
	left_portrait = {
		character = root
		animation = boredom
	}
	
	widget = {
		gui = "POD_shader_event_fog_alley_night"
		container = "background_shader_vfx_container"
	}
	
	cooldown = { years = 30 }

	trigger = {
		POD_is_vampire_or_abomination_trigger = yes
		POD_fear_the_sun_trigger = yes
		NOT = { has_trait = callous }
	}

	option = {
		# Hire a painter
		name = pod_vampire_flavor.1000.a

		remove_short_term_gold = medium_gold_value
		trigger_event = {
			id = pod_vampire_flavor.1001
			days = { 3 12 }
		}
	}

	option = {
		# Possess an animal to see the sun
		name = pod_vampire_flavor.1000.b
		trigger = {
			has_animalism_trigger = yes
		}
		add_stress = medium_stress_loss
		POD_hunger_rouse_check_effect = yes
	}

	option = {
		# One peek won't hurt
		name = pod_vampire_flavor.1000.c
		trigger = {
			NOT = { has_trait = craven }
		}
		random_list = {
			50 = {
				if = {
					limit = { has_trait = brave }
					add_stress = major_stress_loss
				}
				else = {
					add_stress = medium_stress_loss
				}
			}
			49 = {
				trigger = { NOT = { has_perk = darkstatue_perk } }
				increase_wounds_no_death_effect = { REASON = sun }
			}
			1 = {
				trigger = { NOT = { has_perk = darkstatue_perk } }
				death = { death_reason = death_sun }
			}
		}
	}

	option = {
		# Get ahold of yourself
		name = pod_vampire_flavor.1000.d
		stress_impact = {
			base = 50
			diligent = minor_stress_impact_loss
			arrogant = minor_stress_impact_loss
		}
	}
}
