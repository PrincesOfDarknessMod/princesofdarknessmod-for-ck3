#Event looking for a ephor replacement
pod_ventrue_ephorate.3 = {
	scope = none
	hidden = yes
	
	trigger = {
	}
	
	immediate = {
		if = {
			limit = {
				any_player = {
					is_ventrue_trigger = yes
					NOT = { has_character_flag = refused_ephorate_flag }
					NOR = { 
						has_character_flag = support_eternal_flag
						has_character_flag = support_knights_flag
						has_character_flag = support_merchants_flag
						has_character_flag = support_patricians_flag
						has_character_flag = support_senate_flag			
					}					
				}
			}
			random_player = {
				limit = {
					is_ventrue_trigger = yes
					NOT = { has_character_flag = refused_ephorate_flag }					
					NOR = { 
					has_character_flag = support_eternal_flag
					has_character_flag = support_knights_flag
					has_character_flag = support_merchants_flag
					has_character_flag = support_patricians_flag
					has_character_flag = support_senate_flag			
					}
				}
				trigger_event = pod_ventrue_ephorate.30
			}
		}
		else = {
		random_ruler = {
			limit = {
			is_ventrue_trigger = yes
			NOT = { religion = religion:talmahera_religion }
			NOT = { religion = religion:raktasadhus_religion }
			NOR = { 
				has_character_flag = support_eternal_flag
				has_character_flag = support_knights_flag
				has_character_flag = support_merchants_flag
				has_character_flag = support_patricians_flag
				has_character_flag = support_senate_flag			
			}			
			is_vampire_trigger = yes
			}
			weight = {
				base = 0
				modifier = {
					add = 1000
						highest_held_title_tier >= tier_empire
							}													
				modifier = {
					add = 100
						highest_held_title_tier = tier_kingdom
							}
				modifier = {
					add = 1
						is_landed = yes
							}
				modifier = {
					add = 100
						sub_realm_size >= 50
							}
				modifier = {
					add = 1000
						prestige >= 1000
							}
				modifier = {
					add = 100
						prestige >= 500
							}
				modifier = {
					add = 1000
						sub_realm_size >= 100
							}							
				modifier = {
					add = 200
						is_ai = no
							}
					}		
		save_scope_as = future_ephor						
					}			
		
		character:POD_wraith_01 = {
			trigger_event = {
				id = pod_ventrue_ephorate.14
				days = 2
			}
		}
		}		
	}
	
}

#An ephor has died, ventrue players get first pick at trying to become the next one
pod_ventrue_ephorate.30 = {
	type = character_event
	content_source = dlc_POD
	title = pod_ventrue_ephorate.30.t
	desc = pod_ventrue_ephorate.30.desc
	theme = ventrue_coa
	
	option = {
		name = pod_ventrue_ephorate.30.a
		
		custom_tooltip = entry_ephorate_tt
		
		if = { 
			limit = {
				OR = {
					highest_held_title_tier >= tier_empire
					sub_realm_size >= 100
					prestige >= 1000
				}
			}
		remove_short_term_gold = 100
		add_prestige = -100
		}		
		
		else_if = { 
			limit = {
				OR = {
					highest_held_title_tier = tier_kingdom
					sub_realm_size >= 50
					prestige >= 500
				}
			}
		remove_short_term_gold = 500
		add_prestige = -500
		}

		else = {
			remove_short_term_gold = 1000
			add_prestige = -1000						
		}
		
		save_scope_as = future_ephor
		
		trigger_event = {
			id = pod_ventrue_ephorate.4
			days = 5
		}		
	}

	option = {
		name = pod_ventrue_ephorate.30.b
		
		add_character_flag = {
			flag = refused_ephorate_flag
			years = 5
		}
		
		trigger_event = pod_ventrue_ephorate.3
	}	
}
