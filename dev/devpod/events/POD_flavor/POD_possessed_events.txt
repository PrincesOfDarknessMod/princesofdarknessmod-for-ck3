# Bane Attractor
POD_possessed.1001 = {
	hidden = yes

	trigger = {
		has_perk = possessed_bane_attractor_perk
	}

	immediate = {
		POD_possessed_generate_fomor_goober = {
			CULTURE = root.culture
			LOCATION = root.capital_province
			# Start at 10 base prowess, +5 per possessed perk
			# -6 to counteract bonuses from BSD tenets
			PROWESS = 4
			SAVE_AS = goober
		}
		scope:goober = {
			add_prowess_skill = {
				value = root.POD_total_possessed_perks
				multiply = 5
			}
			
			if = {
				limit = { root.POD_total_possessed_perks >= POD_possessed_stronger_taint_threshold }
				add_trait = lifestyle_blademaster
			}
		}
		if = {
			limit = { POD_total_possessed_perks >= POD_possessed_stronger_taint_threshold }
			# lethal
			configure_start_single_combat_effect = {
				SC_INITIATOR = root
				SC_ATTACKER = scope:goober
				SC_DEFENDER = root
				FATALITY = always
				FIXED = no
				LOCALE = alley_night
				OUTPUT_EVENT = POD_possessed.1002
				INVALIDATION_EVENT = fp1_tbc.0102
			}
		}
		else = {
			# nonlethal
			configure_start_single_combat_effect = {
				SC_INITIATOR = root
				SC_ATTACKER = scope:goober
				SC_DEFENDER = root
				FATALITY = no
				FIXED = no
				LOCALE = alley_night
				OUTPUT_EVENT = POD_possessed.1002
				INVALIDATION_EVENT = fp1_tbc.0102
			}
		}
	}
}

POD_possessed.1022 = {
	content_source = dlc_POD
	type = character_event
	
	theme = wyrm
	override_background = { reference = shader_event_wyrm }
	
	widget = {
		gui = "POD_shader_event_wyrm"
		container = "background_shader_vfx_container"
	}

	title = POD_possessed.1022.t
	desc  = POD_possessed.1022.desc

	left_portrait = {
		character = root
		animation = anger
		camera = camera_event_very_left
	}

	right_portrait = {
		character = scope:target
	}
	
	immediate = {
		custom_tooltip = POD_possessed_urges_reason_tt
		hidden_effect = {
			random_list = {
				25 = {
					trigger = {
						any_courtier = {
							POD_is_available_for_events = yes
						}
					}
					random_courtier = {
						limit = { POD_is_available_for_events = yes }
						save_scope_as = target
					}
				}
				75 = {
					trigger = {
						any_vassal = {
							POD_is_available_for_events = yes
						}
					}
					random_vassal = {
						limit = { POD_is_available_for_events = yes }
						save_scope_as = target
					}
				}
				# fallback
				0 = {
					modifier = {
						NOR = {
							any_courtier = { POD_is_available_for_events = yes }
							any_vassal = { POD_is_available_for_events = yes }
						}
						add = 1
					}
					random_neighboring_and_across_water_top_liege_realm_owner = {
						limit = {
							NOT = {
								POD_custom_incompatible_splats_trigger = {
									ACTOR = root
									RECIPIENT = this
								}
							}
							POD_is_available_for_events = yes
						}
						alternative_limit = {
							POD_is_available_for_events = yes
						}
						save_scope_as = target
					}
				}
			}
		}
	}
	
	option = {
		name = POD_possessed_1021.a
		
		duel = {
			skill = prowess
			target = scope:target
		
			50 = {
				desc = POD_possessed.1022.success
				
				compare_modifier = {
					value = scope:duel_value
					multiplier = tiny_positive_duel_skill_multiplier
				}

				send_interface_toast = {
					title = POD_possessed.1022.success
					left_icon = this
					right_icon = scope:target

					scope:target = {
						POD_increase_wounds_hostile_effect = {
							ATTACKER = root
							COUNT = POD_base_wound_damage
							REASON = fight
						}
					}
					
					reverse_add_opinion = {
						target = scope:target
						modifier = possessed_urges_fight_opinion
					}
				}
			}
			
			50 = {
				desc = POD_possessed.1022.failure
				
				compare_modifier = {
					value = scope:duel_value
					multiplier = tiny_negative_duel_skill_multiplier
				}

				send_interface_toast = {
					title = POD_possessed.1022.failure
					left_icon = this
					right_icon = scope:target

					POD_increase_wounds_hostile_effect = {
						ATTACKER = scope:target
						COUNT = POD_base_wound_damage
						REASON = fight
					}
					
					reverse_add_opinion = {
						target = scope:target
						modifier = possessed_urges_fight_opinion
					}
				}
			}
		}
	}
	
	option = {
		name = POD_possessed_1021.b

# Debase yourself
POD_possessed.1024 = {
	content_source = dlc_POD
	type = character_event
	
	theme = wyrm
	override_background = { reference = shader_event_wyrm }
	
	widget = {
		gui = "POD_shader_event_wyrm"
		container = "background_shader_vfx_container"
	}

	title = POD_possessed.1024.t
	desc  = POD_possessed.1024.desc

	left_portrait = {
		character = root
		animation = throne_room_bow_1
		camera = camera_event_very_left
	}

	right_portrait = {
		character = scope:target
		animation = wedding_priest
	}
	
	immediate = {
		custom_tooltip = POD_possessed_urges_reason_tt
		
		random_neighboring_and_across_water_top_liege_realm_owner = {
			limit = {
				NOT = {
					POD_custom_incompatible_splats_trigger = {
						ACTOR = root
						RECIPIENT = this
					}
				}
				POD_is_available_for_events = yes
			}
			alternative_limit = {
				POD_is_available_for_events = yes
			}
			save_scope_as = target
		}
		if = {
			limit = { NOT = { exists = scope:target } }
			random_ruler = {
				limit = {
					NOT = {
						POD_custom_incompatible_splats_trigger = {
							ACTOR = root
							RECIPIENT = this
						}
					}
					POD_is_available_for_events = yes
					in_diplomatic_range = root
				}
				alternative_limit = {
					POD_is_available_for_events = yes
					in_diplomatic_range = root
				}
				alternative_limit = {
					POD_is_available_for_events = yes
				}
				save_scope_as = target
			}
		}
		
		POD_save_random_scope_value = {
			MIN = possessed_urges_prestige_min
			MAX = possessed_urges_prestige_max
			SAVE_AS = prestige_amount
		}
		save_scope_value_as = {
			name = opinion_gain
			value = {
				value = scope:prestige_amount
				divide = possessed_urges_prestige_max
				multiply = 30
			}
		}
	}
	
	option = {
		name = POD_possessed_1021.a
		add_prestige = {
			value = scope:prestige_amount
			multiply = -1
		}
		scope:target = {
			add_prestige = scope:prestige_amount
		}
		reverse_add_opinion = {
			target = scope:target
			modifier = possessed_urges_prestige_opinion
			opinion = scope:opinion_gain
			
		}
	}
	
	option = {
		name = POD_possessed_1021.b
		POD_possessed_urges_stress_option = yes
	}
}
