# Bane Attractor
POD_possessed.1001 = {
	hidden = yes

	trigger = {
		has_perk = possessed_bane_attractor_perk
	}

	immediate = {
		POD_possessed_generate_fomor_goober = {
			CULTURE = root.culture
			LOCATION = root.capital_province
			# Start at 10 base prowess, +5 per possessed perk
			# -6 to counteract bonuses from BSD tenets
			PROWESS = 4
			SAVE_AS = goober
		}
		scope:goober = {
			add_prowess_skill = {
				value = root.POD_total_possessed_perks
				multiply = 5
			}
			
			if = {
				limit = { root.POD_total_possessed_perks >= POD_possessed_stronger_taint_threshold_half }
				add_perk = possessed_regeneration_perk
				add_perk = possessed_shadowplay_perk
			}
			if = {
				limit = { root.POD_total_possessed_perks >= POD_possessed_stronger_taint_threshold }
				add_perk = possessed_thick_skinned_perk
				add_perk = possessed_hazardous_breath_perk
			}
		}
		if = {
			limit = { POD_total_possessed_perks >= POD_possessed_stronger_taint_threshold }
			# lethal
			configure_start_single_combat_effect = {
				SC_INITIATOR = root
				SC_ATTACKER = scope:goober
				SC_DEFENDER = root
				FATALITY = always
				FIXED = no
				LOCALE = alley_night
				OUTPUT_EVENT = POD_possessed.1002
				INVALIDATION_EVENT = fp1_tbc.0102
			}
		}
		else = {
			# nonlethal
			configure_start_single_combat_effect = {
				SC_INITIATOR = root
				SC_ATTACKER = scope:goober
				SC_DEFENDER = root
				FATALITY = no
				FIXED = no
				LOCALE = alley_night
				OUTPUT_EVENT = POD_possessed.1002
				INVALIDATION_EVENT = fp1_tbc.0102
			}
		}
		POD_combat_set_custom_title = { LOC = POD_combat_title_bane_attractor }
		POD_combat_set_enemy_team_name = { LOC = POD_enemy_team_bane_attractor }
	}
}
