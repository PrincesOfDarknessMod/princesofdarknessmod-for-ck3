podgiovanni.6 = {
	type = character_event
	content_source = dlc_POD
	title = podgiovanni.6.t
	desc = podgiovanni.6.desc
	theme = giovanni_coa 

	option = {
		name = podgiovanni.6.a
		
		custom_tooltip = giovanni_wait_time_tt
		
		hidden_effect = {
		random_list = {
			20 = {
			trigger_event = { 
				id = podgiovanni.9
				years = { 5 10 }
			}				
			}
			20 = {
			trigger_event = { 
				id = podgiovanni.10
				years = { 5 10 }
			}				
			}
			20 = {
			trigger_event = { 
				id = podgiovanni.11
				years = { 5 10 }
			}				
			}
			20 = {
			trigger_event = { 
				id = podgiovanni.12
				years = { 5 10 }
			}				
			}
			5 = {
			trigger_event = { 
				id = podgiovanni.13
				years = { 5 10 }
			}				
			}			
		}
		}
		
		if = {
			limit = {
				has_character_flag = giovanni_adopt_beauty_3
				has_character_flag = giovanni_adopt_intellect_3
				has_character_flag = giovanni_adopt_physique_3
			}	
		trigger_event = { 
			id = podgiovanni.7
			years = { 30 50 }
		}
		}
		else_if = {
			limit = {
				OR = {
				has_character_flag = giovanni_adopt_beauty_3
				has_character_flag = giovanni_adopt_intellect_3
				has_character_flag = giovanni_adopt_physique_3
				}
			}	
		trigger_event = { 
			id = podgiovanni.7
			years = { 25 45 }
		}
		}
		else_if = {
			limit = {
				has_character_flag = giovanni_adopt_beauty_2
				has_character_flag = giovanni_adopt_intellect_2
				has_character_flag = giovanni_adopt_physique_2
			}	
		trigger_event = { 
			id = podgiovanni.7
			years = { 20 40 }
		}
		}
		else_if = {
			limit = {
				OR = {
				has_character_flag = giovanni_adopt_beauty_2
				has_character_flag = giovanni_adopt_intellect_2
				has_character_flag = giovanni_adopt_physique_2
				}
			}	
		trigger_event = { 
			id = podgiovanni.7
			years = { 15 35 }
		}
		}
		else_if = {
			limit = {
				OR = {
				has_character_flag = giovanni_adopt_beauty_1
				has_character_flag = giovanni_adopt_intellect_1
				has_character_flag = giovanni_adopt_physique_1
				}
			}

		}
		}
		else = {

			years = { 10 15 }
		}
		}		
	}

podgiovanni.7 = {
	hidden = yes
	
	immediate = {
		create_character = {
			location = root.capital_province
			age = 18
			culture = root.culture
			faith = root.faith
			
			gender_female_chance = {
				value = 50
				if = {
					limit = { root = { has_character_flag = giovanni_adopt_man } }
					add = -50
				}
				if = {
					limit = { root = { has_character_flag = giovanni_adopt_woman } }
					add = 50
				}
			}
			diplomacy = { 5 10 }
			martial = { 5 10 }
			stewardship = { 5 10 }
			intrigue = { 5 10 }
			learning = { 5 10 }
			prowess = { 5 10 }
			trait = mortal
			random_traits = yes
			# trait = wight

			dynasty = none
			save_scope_as = optional_child
		}
		
		remove_character_flag = giovanni_adopt_man
		remove_character_flag = giovanni_adopt_woman

		if = {
			limit = { root.house = { has_house_modifier = giovanni_family_modifier } }
			
			scope:optional_child = {
				add_diplomacy_skill = 3
				add_learning_skill = 3
				add_martial_skill = 3
				add_intrigue_skill = 5
				add_stewardship_skill = 5 #they're mafia of course those 2 stats get buffed more than the others
			}
		}
		if = {
			limit = { root = { has_character_flag = giovanni_adopt_boosted } }
			remove_character_flag = giovanni_adopt_boosted
			
			scope:optional_child = {
				add_diplomacy_skill = 3
				add_learning_skill = 3
				add_martial_skill = 3
				add_intrigue_skill = 3
				add_stewardship_skill = 3
			}
		}		
		if = {
			limit = { root = { has_character_flag = giovanni_adopt_blademaster } }
			remove_character_flag = giovanni_adopt_blademaster
			
			scope:optional_child = {
				if = {
					limit = {
						NOT = { has_trait = lifestyle_blademaster }
					}
				add_trait = lifestyle_blademaster
				}
			}
		}
		if = {
			limit = { root = { has_character_flag = giovanni_adopt_schemer } }
			remove_character_flag = giovanni_adopt_schemer
			
			scope:optional_child = {
				if = {
					limit = {
						NOT = { has_trait = schemer }
					}
				add_trait = schemer
				}
			}
		}
		if = {
			limit = { root = { has_character_flag = giovanni_adopt_sorcerer } }
			remove_character_flag = giovanni_adopt_sorcerer
			
			scope:optional_child = {
				if = {
					limit = {
						NOT = { has_trait = sorcerer }
					}
				add_trait = sorcerer
				}
			}
		}		
		
		if = {
			limit = { root = { has_character_flag = giovanni_adopt_beauty_3 } }
			remove_character_flag = giovanni_adopt_beauty_3
			
			scope:optional_child = {
				add_trait = beauty_good_3
			}
		}
		else_if = {
			limit = { root = { has_character_flag = giovanni_adopt_beauty_2 } }
			remove_character_flag = giovanni_adopt_beauty_2
			
			scope:optional_child = {
				add_trait = beauty_good_2
			}
		}
		else_if = {
			limit = { root = { has_character_flag = giovanni_adopt_beauty_1 } }
			remove_character_flag = giovanni_adopt_beauty_1
			
			scope:optional_child = {
				add_trait = beauty_good_1
			}
		}
		
		if = {
			limit = { root = { has_character_flag = giovanni_adopt_intellect_3 } }
			remove_character_flag = giovanni_adopt_intellect_3
			
			scope:optional_child = {
				add_trait = intellect_good_3
			}
		}
		else_if = {
			limit = { root = { has_character_flag = giovanni_adopt_intellect_2 } }
			remove_character_flag = giovanni_adopt_intellect_2
			
			scope:optional_child = {
				add_trait = intellect_good_2
			}
		}
		else_if = {
			limit = { root = { has_character_flag = giovanni_adopt_intellect_1 } }
			remove_character_flag = giovanni_adopt_intellect_1
			
			scope:optional_child = {
				add_trait = intellect_good_1
			}
		}
		
		if = {
			limit = { root = { has_character_flag = giovanni_adopt_physique_3 } }
			remove_character_flag = giovanni_adopt_physique_3
			
			scope:optional_child = {
				add_trait = physique_good_3
			}
		}
		else_if = {
			limit = { root = { has_character_flag = giovanni_adopt_physique_2 } }
			remove_character_flag = giovanni_adopt_physique_2
			
			scope:optional_child = {
				add_trait = physique_good_2
			}
		}
		else_if = {
			limit = { root = { has_character_flag = giovanni_adopt_physique_1 } }
			remove_character_flag = giovanni_adopt_physique_1
			
			scope:optional_child = {
				add_trait = physique_good_1
			}
		}
		
		# Hideout Upgrades — Giovanni Counting House embrace stat bonus
		if = {
			limit = { domicile ?= { has_domicile_building_or_higher = vampire_giovanni_counting_house_06 } }
			scope:optional_child = {
				add_diplomacy_skill = 2
				add_intrigue_skill = 2
				add_stewardship_skill = 2
			}
		}

			scope:optional_child = {
				add_diplomacy_skill = 2

				add_stewardship_skill = 2
			}
		}
		else_if = {
			limit = { domicile ?= { has_domicile_building_or_higher = vampire_giovanni_counting_house_04 } }
			scope:optional_child = {
				add_diplomacy_skill = 1
				add_intrigue_skill = 1
				add_stewardship_skill = 2
			}
		}
		else_if = {
			limit = { domicile ?= { has_domicile_building_or_higher = vampire_giovanni_counting_house_03 } }
			scope:optional_child = {
				add_diplomacy_skill = 1
				add_intrigue_skill = 1
				add_stewardship_skill = 1
			}
		}

			limit = { domicile ?= { has_domicile_building_or_higher = vampire_giovanni_counting_house_02 } }
			scope:optional_child = {
				add_diplomacy_skill = 1
				add_intrigue_skill = 1
			}
		}

			scope:optional_child = {
				add_diplomacy_skill = 1
			}
		}
		scope:optional_child = {
			change_first_name = pod_giovanni_name
		}
		
		trigger_event = podgiovanni.8
	}
