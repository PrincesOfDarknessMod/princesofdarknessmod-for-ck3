podnosferatu.30 = {
	type = character_event
	content_source = dlc_POD
	title = podnosferatu.30.t
	desc = podnosferatu.30.desc
	theme = nosferatu_coa
	left_portrait = {
		character = root
		animation = fear
	}	
	
	option = {
		name = podnosferatu.30.a
	
		duel = {
			skill = stewardship
			value = 20
				10 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = 0.5
					}
				desc = podnosferatu.30.effect.success.desc
				
				
			send_interface_toast = {
				title = podnosferatu.30.effect.success.desc
				left_icon = root
				
				add_prestige = 200
				add_character_flag = nictuku_resist_flag #added these to every vicotry event for the nicktuku decision, you need to have fought one of these
				}				
				
			modifier = {
				has_character_flag = nictuku_well_prepared_flag
				add = 10
			}
			modifier = {
				has_character_flag = nictuku_prepared_flag
				add = 5
			}
			}
				10 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = -0.5
					}

					desc = podnosferatu.30.critical.failure.effect.desc
					
					death = {
						death_reason = death_murder
					}
				}
			}		
		
	}

	option = {
		name = POD_survey_events.110.option.d
	
		add_character_modifier = {
			modifier = broken_authority_modifier
			years = 5
		}
	}
	
	after = {
		hidden_effect = {
		if = {
			limit = {
				has_character_flag = nictuku_well_prepared_flag
			}
			remove_character_flag = nictuku_well_prepared_flag
		}
		else_if = {
			limit = {
				has_character_flag = nictuku_prepared_flag
			}
			remove_character_flag = nictuku_prepared_flag
		}
		else = {}
		
		trigger_event = { id = podnosferatu.2 years = 5 }
	}
	}
}
