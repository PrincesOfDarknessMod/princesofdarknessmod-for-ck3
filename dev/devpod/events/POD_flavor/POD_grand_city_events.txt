	type = character_event
	content_source = dlc_POD
	title = podgrandcity.01.t
	theme = default
	override_background = {
		reference = sitting_room
	}
	
	right_portrait = {
		character = root
	}

	trigger = {
		NOT = { has_character_flag = had_paris_chantry_event }
	}

	immediate = {
		add_character_flag = had_paris_chantry_event
	}
	

	desc = podgrandcity.01.desc
	
	option = {
		name = podgrandcity.01.option.a
		trigger = { has_thaumaturgy_bloodmagic_trigger = no }
		show_as_unavailable = { has_thaumaturgy_bloodmagic_trigger = yes }
		if = {
			limit = { NOT = { has_trait = bloodsorcerydiscipline } }
			add_trait = bloodsorcery
		}
		POD_learn_thaumaturgy_bloodmagic_effect = yes
	}
	
	option = {
		name = podgrandcity.01.option.b
		trigger = { has_trait = bloodsorcerydiscipline }
		trait = bloodsorcery
		trait = bloodsorceryadvanced
		add_blood_magic_lifestyle_perk_points = 5
	}

	option = {
		name = podgrandcity.01.option.c
		custom_tooltip = podgrandcity.01.tooltip
		add_character_flag = can_create_gargoyles
	}
}

# Baghdad modifier maintenance
podgrandcity.202 = {
	hidden = yes
	scope = none

	immediate = {
		title:b_baghdad.title_province = {
			save_scope_as = baghdad
			if = {
				limit = { has_province_modifier = POD_baghdad_looted_1_modifier }
				province_owner = {
					send_interface_message = {
						type = event_generic_good
						title = podgrandcity.202.t
						scope:baghdad = {
							remove_province_modifier = POD_baghdad_looted_1_modifier
							add_province_modifier = POD_baghdad_looted_2_modifier
							county = { change_development_level = 1 }
						}
					}
					if = {
						limit = { is_ai = yes }
						add_gold = 50
						add_prestige = 25
						add_piety = 25
					}
				}
				trigger_event = { id = podgrandcity.202 years = 5 }
			}
			else_if = {
				limit = { has_province_modifier = POD_baghdad_looted_2_modifier }
				province_owner = {
					send_interface_message = {
						type = event_generic_good
						title = podgrandcity.202.t
						scope:baghdad = {
							remove_province_modifier = POD_baghdad_looted_2_modifier
							add_province_modifier = POD_baghdad_looted_3_modifier
							county = { change_development_level = 2 }
						}
					}
					if = {
						limit = { is_ai = yes }
						add_gold = 50
						add_prestige = 25
						add_piety = 25
					}
				}
				trigger_event = { id = podgrandcity.202 years = 5 }
			}
			else_if = {
				limit = { has_province_modifier = POD_baghdad_looted_3_modifier }
				province_owner = {
					send_interface_message = {
						type = event_generic_good
						title = podgrandcity.202.t
						scope:baghdad = {
							remove_province_modifier = POD_baghdad_looted_3_modifier
							county = { change_development_level = 3 }
						}
					}
					if = {
						limit = { is_ai = yes }
						add_gold = 100
						add_prestige = 50
						add_piety = 50
					}
				}
			}
		}
	}
}
