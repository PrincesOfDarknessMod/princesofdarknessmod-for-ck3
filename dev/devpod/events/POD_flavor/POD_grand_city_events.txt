namespace = podgrandcity

#Paris Tremere Chantry reward
podgrandcity.01 = {
	type = character_event
	title = podgrandcity.01.t
	theme = default
	override_background = {
		reference = sitting_room
	}
	
	right_portrait = {
		character = root
	}

	trigger = {
		NOT = { has_character_flag = had_paris_chantry_event }
	}

	immediate = {
		add_character_flag = had_paris_chantry_event
	}
	

	desc = podgrandcity.01.desc
	
	option = {
		name = podgrandcity.01.option.a
		trigger = { has_thaumaturgy_bloodmagic_trigger = no }
		show_as_unavailable = { has_thaumaturgy_bloodmagic_trigger = yes }
		if = {
			limit = { NOT = { has_trait = bloodsorcerydiscipline } }
			add_trait = bloodsorcery
		}
		POD_learn_thaumaturgy_bloodmagic_effect = yes
	}
	
	option = {
		name = podgrandcity.01.option.b
		trigger = { has_trait = bloodsorcerydiscipline }
		trait = bloodsorcery
		trait = bloodsorceryadvanced
		add_blood_magic_lifestyle_perk_points = 5
	}

	option = {
		name = podgrandcity.01.option.c
		custom_tooltip = podgrandcity.01.tooltip
		add_character_flag = can_create_gargoyles
	}
}



#Cairo Dream Court
podgrandcity.02 = {
	type = character_event
	title = podgrandcity.02.t
	theme = default
	override_background = {
		reference = fp3_zoroastrian_temple
	}
	
	right_portrait = {
		character = root
	}

	trigger = {
		NOT = { has_character_flag = had_dreamcourt_event }
		NOT = { has_global_variable = pathofheliopolis_active }
	}

	immediate = {
		add_character_flag = had_dreamcourt_event
	}

	desc = podgrandcity.02.desc
	
	option = {
		name = podgrandcity.02.option.a
		custom_tooltip = podgrandcity.02.tooltip.a
		add_prestige = 500
		hidden_effect = {
			faith:pathofheliopolis = { 
				remove_doctrine = tenet_adaptive
				add_doctrine = tenet_carnal_exaltation
			}
		}
	}
	
	option = {
		name = podgrandcity.02.option.b
		custom_tooltip = podgrandcity.02.tooltip.b
		add_prestige = 500
		hidden_effect = {
			faith:pathofheliopolis = { 
				remove_doctrine = tenet_adaptive
				add_doctrine = tenet_hedonistic
			}
		}
	}

	option = {
		name = podgrandcity.02.option.c
		custom_tooltip = podgrandcity.02.tooltip.c
		add_prestige = 500
		hidden_effect = {
			faith:pathofheliopolis = { 
				remove_doctrine = tenet_adaptive
				add_doctrine = tenet_ritual_celebrations
			}
		}
	}

	after = {
		hidden_effect = {
			if = {
				limit = {
					has_character_flag = had_dreamcourt_event
					has_character_flag = had_dustandbones_event
					has_character_flag = had_crocodiletemple_event
				}
				trigger_event = { id = podgrandcity.05 }
			}
		}
	}
}
