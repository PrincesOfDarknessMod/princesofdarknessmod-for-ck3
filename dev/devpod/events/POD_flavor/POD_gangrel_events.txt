podgangrel.7 = {
	type = character_event
	content_source = dlc_POD
	title = podgangrel.7.t
	desc = podgangrel.7.desc
	theme = gangrel_coa 
	left_portrait = {
		character = root
		animation = personality_bold
	}	
	right_portrait = {
		character = scope:recipient
		animation = personality_content
	}
	
	trigger = { 
	}
	
	immediate = {
	}
	
	#Super friends
	option = {
		name = podgangrel.7.a
		
		trigger = {
			has_variable = gangrel_challenge_friendliness
			var:gangrel_challenge_friendliness = 6	
			NOR = {
				has_relation_friend = scope:recipient
				has_relation_best_friend = scope:recipient				
				has_relation_rival = scope:recipient
				has_relation_nemesis = scope:recipient				
			}
		}

		set_relation_friend = { reason = friend_generic_history target = scope:recipient }

		reverse_add_opinion = {
			target = scope:recipient
			modifier = respect_opinion
			opinion = 35
		}
		add_opinion = {
			target = scope:recipient
			modifier = respect_opinion
			opinion = 35
		}			
	}

	option = {
		name = podgangrel.7.b
		
		trigger = {
			has_variable = gangrel_challenge_friendliness
			var:gangrel_challenge_friendliness = 0	
			NOR = {
				has_relation_friend = scope:recipient
				has_relation_best_friend = scope:recipient				
				has_relation_rival = scope:recipient
				has_relation_nemesis = scope:recipient				
			}
		}
		
		set_relation_rival = { reason = rival_historical target = scope:recipient }

		reverse_add_opinion = {
			target = scope:recipient
			modifier = hate_opinion
			opinion = -20
		}
		add_opinion = {
			target = scope:recipient
			modifier = hate_opinion
			opinion = -20
		}		
	}
	
	option = {
		name = podgangrel.7.c

		trigger = {
		NOT = {	
			has_variable = gangrel_challenge_friendliness
			var:gangrel_challenge_friendliness = 6	
		}
		}
		
		reverse_add_opinion = {
			target = scope:recipient
			modifier = respect_opinion
			opinion = 10
		}
		add_opinion = {
			target = scope:recipient
			modifier = respect_opinion
			opinion = 10
		}		
	}
	
	after = {
		remove_variable = gangrel_challenge_friendliness	
	}
}
