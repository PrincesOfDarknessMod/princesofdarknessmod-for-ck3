		name = pod_embrace_offer_accepted.0001.a

}

# Ghoul Vassal that accepted embrace arrives on site
pod_landed_ghoul_embrace.101 = {
	hidden = yes
	immediate = {
		if = {
			limit = { exists = liege } # To stop errors
			liege = { save_scope_as = actor }
		}	
			scope:actor = {
				root = { save_scope_as = target }
				trigger_event = POD_embrace.100 
			}
		}
	current_travel_plan ?= {
		delay_travel_plan = { days = 90 }
	}
}

# Rogue landed ghoul selects sire.
pod_landed_ghoul_embrace.300 = {
	type = character_event
	title = pod_landed_ghoul_embrace.300.t
	theme = alliance
	desc = {
		desc =  pod_landed_ghoul_embrace.300.desc
		triggered_desc = {
			trigger = { exists = scope:domitor_rival }
			desc =  pod_landed_ghoul_embrace.300.desc.a
			}
		triggered_desc = {
			trigger = { exists = scope:adjacent_ruler }
			desc =  pod_landed_ghoul_embrace.300.desc.b
			}
		triggered_desc = {
			trigger = { exists = scope:interloper }
			desc =  pod_landed_ghoul_embrace.300.desc.c
			}
		}
			
	left_portrait = {
		character = scope:adjacent_ruler
		animation = personality_rational
	}
	#lower_left_portrait = scope:mystic_option
	right_portrait = {
		character = scope:domitor_rival
		animation = schadenfreude
	}
	lower_right_portrait = scope:interloper

	trigger = { #If I lost it before I got this event, I shouldn't qualify for the embrace
		exists = capital_province
	}

	immediate = {
		hidden_effect = {
			#Find some appropriate options				
			#DOMITOR ENEMY
			if = {
				limit = {
					any_relation = {
						type = ghoul_domitor
					}
				}
				random_relation = {
					type = ghoul_domitor
					save_scope_as = domitor
				}
				scope:domitor = {
					if = {
						limit = {
							any_relation = {
								type = rival
							}
						}
						random_relation = {
							type = rival
							limit = {
								is_ai = yes
								is_vampire_trigger = yes
								is_incapable = no
								}
							save_scope_as = domitor_rival
							}
						}
					}
				}
			#ADJACENT THREAT
			if = {
				limit = { is_landed = yes }
					if = {
						limit = { is_independent_ruler = yes }
						random_neighboring_and_across_water_top_liege_realm_owner = {
							limit = {
							is_ai = yes
							is_vampire_trigger = yes
							is_incapable = no
							NOR = { scope:domitor ?= this
									 scope:domitor_rival ?= this
								   }
							}
							weight = {
								base = 1
								
								modifier = {
									add = 10
									highest_held_title_tier > tier_county
								}
								modifier = {
									add = 10
									highest_held_title_tier > tier_duchy
								}
								modifier = {
									add = 10
									highest_held_title_tier > tier_kingdom
								}
							}
							save_scope_as = adjacent_ruler
						}
					}
					else = {
						scope:target.top_liege = {
							random_neighboring_and_across_water_top_liege_realm_owner = {
							limit = {
							is_ai = yes
							is_vampire_trigger = yes
							is_incapable = no
							NOR = { scope:domitor ?= this
								scope:domitor_rival ?= this
								}
							}
							weight = {
								base = 1
								
								modifier = {
									add = 10
									highest_held_title_tier > tier_county
								}
								modifier = {
									add = 10
									highest_held_title_tier > tier_duchy
								}
								modifier = {
									add = 10
									highest_held_title_tier > tier_kingdom
								}
							}
							save_scope_as = adjacent_ruler
						}
					}					
				}
			}
			#DREAD INTERLOPER
			random_pool_character = {
				province = root.capital_province
				limit = { 
					is_vampire_trigger = yes						
					is_landed = no
					is_incapable = no
					NOT = { scope:domitor ?= this }
					NOT = { scope:domitor_rival ?= this }
					NOT = { scope:adjacent_ruler ?= this }
					has_trait_rank = { trait = vampgeneration rank <= 5 } 
				}
				weight = {
					base = 1
					modifier = {
						age >= 500
						factor = 100
					}
					modifier = {
						faith = root.faith
						factor = 10
					}
				}
				save_scope_as = interloper
			}
		}
	}

	#ADJACENT RULER OPTION
	option = {
		name = pod_landed_ghoul_embrace.300.b
		trigger = {
			exists = scope:adjacent_ruler
			is_playable_character = yes #extra check to make sure they haven't lost their status
		}
		
		hidden_effect = {
			scope:adjacent_ruler = { save_scope_as = vampiresire }
			scope:target = { save_scope_as = created_vampirechilde }
			POD_player_character_create_vampirechilde_effect = yes
			}
	}

	#DOMITOR RIVAL OPTION
	option = {
		name = pod_landed_ghoul_embrace.300.a
		trigger = {
			is_playable_character = yes #extra check to make sure they haven't lost their status
			exists = scope:domitor_rival #Doesn't exists for AI
		}		
		hidden_effect = {
			scope:domitor_rival = { save_scope_as = vampiresire }
			scope:target = { save_scope_as = created_vampirechilde }
			POD_player_character_create_vampirechilde_effect = yes
			
			}
		}

	#DREAD INTERLOPER
	option = {
		name = pod_landed_ghoul_embrace.300.c
		trigger = {
			is_playable_character = yes #extra check to make sure they haven't lost their status
			exists = scope:interloper #Doesn't exists for AI
		}		
		hidden_effect = {
			scope:interloper = { save_scope_as = vampiresire }
			scope:target = { save_scope_as = created_vampirechilde }
			POD_player_character_create_vampirechilde_effect = yes
			}
		}
		
	#None...
	option = {
		name = {
			trigger = {
				NOR = {
					exists = scope:interloper
					exists = scope:adjacent_ruler
				}
			}
			text = pod_landed_ghoul_embrace.300.d.two
		}
		name = {
			trigger = {
				OR = {
					exists = scope:adjacent_ruler
					exists = scope:domitor_rival
				}
			}
			text = pod_landed_ghoul_embrace.300.d.more
		}
		custom_tooltip = pod_landed_ghoul_embrace.300.d.tt

		ai_chance = {
			factor = 1
		}
	}

	after = {
		hidden_effect = {
			scope:target = {		
				remove_character_flag = ambitious_ghoul
				}
				random_scheme = {
					limit = {
						scheme_type = PODghoulseekembrace
						scheme_target_character = scope:target
					}
					end_scheme = yes
				}	
		}
	}
}
