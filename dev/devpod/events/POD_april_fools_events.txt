# April Fools 2026: Yearly Maintenance
POD_april_fools.101 = {
	scope = none
	hidden = yes
	
	trigger = {
		has_global_variable = POD_marauder_mode
	}
	
	immediate = {
		change_global_variable = {
			name = POD_marauder_mode
			add  = 1
		}
		switch = {
			trigger = global_var:POD_marauder_mode
			2  = { every_player = { trigger_event = POD_april_fools.103 } } # Dragon
			4  = { every_player = { trigger_event = POD_april_fools.104 } } # Gold
			6  = { every_player = { trigger_event = POD_april_fools.105 } } # Prisoner
			8  = {
				every_player = { trigger_event = POD_april_fools.106 } # Bald
				every_independent_ruler = {
					limit = {
						is_ai = yes
						POD_no_hair_trigger = no
						POD_is_available_for_events = yes
					}
					add_character_flag = {
						years = 10
						flag = POD_bald_look_flag
					}
				}
			}
			10 = { every_player = { trigger_event = POD_april_fools.107 } } # Cheese
			12 = {
				every_player = { trigger_event = POD_april_fools.108 } # Inquisition
				# This is independent of the current_year based Inquisition spawn
				# So there could potentially be 2 Inquisitions
				trigger_event = {
					id = POD_endgame.2
					days = 1
				}
			}
			14 = { every_player = { trigger_event = POD_april_fools.109 } } # Raiders
			16 = { every_player = { trigger_event = POD_april_fools.110 } } # Tongues
			18 = { every_player = { trigger_event = POD_april_fools.111 } } # Rivals
			20 = { every_player = { trigger_event = POD_april_fools.112 } } # Finale
			fallback = { every_player = { trigger_event = POD_april_fools.112 } }
		}
		every_player = {
			POD_shuffle_personality_effect = yes
		}
	}
}
