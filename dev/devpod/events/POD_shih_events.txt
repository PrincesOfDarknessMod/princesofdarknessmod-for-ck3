POD_shih_events.0001 = {
	type = character_event
	content_source = dlc_POD
	title = POD_shih_events.0001.t
	desc = POD_shih_events.0001.desc
	theme = faith

	left_portrait = { character = root animation = idle }

	right_portrait = { character = var:current_journey.var:target animation = personality_bold }

	option = {
		name = POD_shih_events.0001.a
		trigger = { var:current_journey.var:target.diplomacy < shih_skill_limit }
		var:current_journey.var:target = { add_diplomacy_skill = 1 }
		if = { limit = { diplomacy < shih_skill_limit } random = { chance = 50 add_diplomacy_skill = 1 } }
	}

	option = {
		name = POD_shih_events.0001.b
		trigger = { var:current_journey.var:target.martial < shih_skill_limit }
		var:current_journey.var:target = { add_martial_skill = 1 }
		if = { limit = { martial < shih_skill_limit } random = { chance = 50 add_martial_skill = 1 } }
	}

	option = {
		name = POD_shih_events.0001.c
		trigger = { var:current_journey.var:target.stewardship < shih_skill_limit }
		var:current_journey.var:target = { add_stewardship_skill = 1 }
		if = { limit = { stewardship < shih_skill_limit } random = { chance = 50 add_stewardship_skill = 1 } }
	}

	option = {
		name = POD_shih_events.0001.d
		trigger = { var:current_journey.var:target.intrigue < shih_skill_limit }
		var:current_journey.var:target = { add_intrigue_skill = 1 }
		if = { limit = { intrigue < shih_skill_limit } random = { chance = 50 add_intrigue_skill = 1 } }
	}

	option = {
		name = POD_shih_events.0001.e
		trigger = { var:current_journey.var:target.learning < shih_skill_limit }
		var:current_journey.var:target = { add_learning_skill = 1 }
		if = { limit = { learning < shih_skill_limit } random = { chance = 50 add_learning_skill = 1 } }
	}

	option = {
		name = POD_shih_events.0001.f
		add_prestige = 100 # fallback
	}

	after = { POD_end_journey_effect = yes }
}
