POD_secrets.0101 = { 
	type = character_event
	content_source = dlc_POD
	title = POD_secrets.0101.t
	orphan = yes
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_negative_attitude_towards_trait_trigger = { TRAIT = diablerist GENDER_CHARACTER = scope:owner } }
				desc = POD_secrets.0101.negative.desc
			}
			desc = POD_secrets.0101.positive_1.desc
		}
		first_valid = {
			triggered_desc = {
				trigger = {
					any_secret = { secret_type = secret_diablerist }
				}
				desc = POD_secrets.0101.worried.desc
			}
			triggered_desc = {
				trigger = { NOT = { has_negative_attitude_towards_trait_trigger = { TRAIT = diablerist GENDER_CHARACTER = scope:owner } } }
				desc = POD_secrets.0101.positive_2.desc
			}
		}
	}
	theme = secret
	override_background = {
		trigger = { allowed_to_imprison_character_trigger = { CHARACTER = scope:owner } } # Show throne room if you can order imprisonment
		reference = throne_room
	}
	left_portrait = scope:owner
	right_portrait = scope:secret_exposer

	immediate = {
		secret_exposed_notification_effects_effect = { SECRET = scope:secret CHARACTER = scope:owner }

		play_music_cue = "mx_cue_secret"
	}

	#OK
	option = {
		trigger = {
			NAND = {
				allowed_to_imprison_character_trigger = { CHARACTER = scope:owner }
				trait_is_criminal_in_faith_trigger = { TRAIT = diablerist FAITH = root.faith GENDER_CHARACTER = scope:owner }
			}
		}
		name = {
			trigger = {
				has_negative_attitude_towards_trait_trigger = { TRAIT = diablerist GENDER_CHARACTER = scope:owner }
				opinion = {
					target = scope:owner
					value >= high_positive_opinion
				}
			}
			text = POD_secrets.0106.likes_negative
		}
		name = {
			trigger = {
				has_negative_attitude_towards_trait_trigger = { TRAIT = diablerist GENDER_CHARACTER = scope:owner }
				opinion = {
					target = scope:owner
					value < high_positive_opinion
				}
			}
			text = POD_secrets.0107.dislikes_negative
		}
		name = {
			trigger = {
				NOT = { has_negative_attitude_towards_trait_trigger = { TRAIT = diablerist GENDER_CHARACTER = scope:owner } }
				opinion = {
					target = scope:owner
					value >= high_positive_opinion
				}
			}
			text = POD_secrets.0106.likes_positive
		}
		name = {
			trigger = {
				NOT = { has_negative_attitude_towards_trait_trigger = { TRAIT = diablerist GENDER_CHARACTER = scope:owner } }
				opinion = {
					target = scope:owner
					value < high_positive_opinion
				}
			}
			text = POD_secrets.0106.dislikes_positive
		}	
	}

	#For liege: imprison
	option = {
		trigger = {
			allowed_to_imprison_character_trigger = { CHARACTER = scope:owner }
			trait_is_criminal_in_faith_trigger = { TRAIT = diablerist FAITH = root.faith GENDER_CHARACTER = scope:owner }
		}
		name = POD_secrets.0107.b

		imprison_character_effect = {
			TARGET = scope:owner
			IMPRISONER = root
		}

		ai_chance = {
			base = 100
			ai_value_modifier = {
				ai_zeal = low_chance_impact_positive_ai_value
			}
			modifier = {
				has_trait = zealous
				add = 100
			}
			modifier = {
				opinion = {
					target = scope:owner
					value < 0
				}
				add = 100
			}
			modifier = {
				OR = {
					is_close_family_of = scope:owner
					is_spouse_of = scope:owner
					has_relation_lover = scope:owner
					has_relation_friend = scope:owner
				}
				add = -100
			}
		}
	}

	#For liege: forgive
	option = {
		trigger = {
			allowed_to_imprison_character_trigger = { CHARACTER = scope:owner }
			trait_is_criminal_in_faith_trigger = { TRAIT = diablerist FAITH = root.faith GENDER_CHARACTER = scope:owner }
		}
		name = POD_secrets.0107.c

		reverse_add_opinion = {
			target = scope:owner
			modifier = grateful_opinion
			opinion = 20
		}

		add_piety_level = -1

		ai_chance = {
			base = 10
			ai_value_modifier = {
				ai_compassion = low_chance_impact_positive_ai_value
			}
			modifier = {
				has_trait = cynical
				add = 100
			}
			modifier = {
				scope:owner = { is_ai = no }
				add = 50
			}
			modifier = {
				NOT = { has_negative_attitude_towards_trait_trigger = { TRAIT = diablerist GENDER_CHARACTER = scope:owner } }
				add = 100
			}
		}
	}
}

#Someone I care about has been diablerized.
POD_secrets.0201 = {
    type = character_event
	content_source = dlc_POD
    title = POD_secrets.0201.t
    desc = {
        desc = POD_secrets.0201.desc
        first_valid = {
            triggered_desc = {
                trigger = {
                    scope:diablerist = { is_imprisoned = yes }
                }
                desc = POD_secrets.0201.desc.prisoner
            }
            desc = POD_secrets.0201.desc.free
        }
    }
    theme = unfriendly
    cooldown = { years = 10 }
    left_portrait = {
        character = root
        animation = rage
    }
    lower_left_portrait = scope:diablerist
    lower_right_portrait = scope:victim

    trigger = {
        scope:diablerist = { is_alive = yes }
        is_landed = yes
        NOT = {
            scope:diablerist = { is_imprisoned_by = root }
        }
        trigger_if = {
            limit = { exists = house }
            this != this.house.house_head
        }
    }

    option = { # Just form rivalry
        name = POD_secrets.0201.a
        if = {
            limit = {
                can_set_relation_rival_trigger = { CHARACTER = scope:diablerist }
            }
			if = {
				limit = { exists = scope:victim_family }
				set_relation_rival = {
					target = scope:diablerist
					reason =  rival_diablerized_family
				}
			}
			else_if = {
				limit = { exists = scope:victim_lover }
				set_relation_rival = {
					target = scope:diablerist
					reason =  rival_diablerized_lover
				}
			}
			else_if = {
				limit = { exists = scope:victim_friend }
				set_relation_rival = {
					target = scope:diablerist
					reason =  rival_diablerized_friend
				}
			}		
        }
        stress_impact = {
            base = minor_stress_impact_loss
            forgiving = medium_stress_impact_gain
        }
        ai_chance = {
            base = 50
            modifier = {
                factor = 2
                OR = {
                    scope:victim = { is_parent_of = root }
                    scope:victim = { is_child_of = root }
                }
            }
            ai_value_modifier = {
                ai_vengefulness = 1
                ai_boldness = 1
            }
        }
    }

    option = { # Forgive
        name = POD_secrets.0201.b
        stress_impact = {
            vengeful = major_stress_impact_gain
            paranoid = medium_stress_impact_gain
            wrathful = medium_stress_impact_gain
            just = minor_stress_impact_gain
        }
        ai_chance = {
            base = 50
            ai_value_modifier = {
                ai_vengefulness = -1
                ai_boldness = -1
            }
        }
    }
}
