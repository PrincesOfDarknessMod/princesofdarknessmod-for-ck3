# test event for checking which houses don't have clothing colors set
# meant to be used with the debug event button that copies event text to the clipboard
POD_debug.4 = {
	type = character_event
	content_source = dlc_POD
	orphan = yes
	
	theme = vampire

	title = blank_line
	desc = POD_debug.4.desc
	
	immediate = {
		every_living_character = {
			limit = {
				# for grabbing historical characters only (for best results, run at the start of the game)
				has_any_nickname = yes
				# uncomment and set other conditions if needed
				#has_trait = kueijin
				house ?= { NOT = { has_variable = POD_portrait_clothing_1_r } }
			}
			house = {
				root = {
					add_to_variable_list = {
						name = color_houses
						target = prev
					}
				}
			}
		}
	}

	option = {
		name = blank_line
	}
	
	after = {
		clear_variable_list = color_houses
	}
}
