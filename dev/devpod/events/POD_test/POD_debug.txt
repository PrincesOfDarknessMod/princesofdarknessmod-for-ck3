# Show the character's real prowess value
POD_debug.1 = {
	type = character_event
	content_source = dlc_POD
	
	theme = vampire
	override_background = { reference = shader_event_hypertrip }

	title = POD_debug.1.t
	desc = POD_debug.1.desc
	
	left_portrait = {
		character = root
		animation = sword_coup_degrace
	}
	
	widget = {
		gui = "POD_shader_event_hypertrip"
		container = "background_shader_vfx_container"
	}

	immediate = {
		POD_save_real_prowess = yes
	}

	option = {
		name = POD_debug.1.a
	}
}

# Portrait Shader Playground

	override_background = { reference = colorpicker }

	title = POD_debug.2.t
	desc = POD_debug.2.desc

		name = POD_debug.2.a
	}
}
