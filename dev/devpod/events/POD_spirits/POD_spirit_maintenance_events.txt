	hidden = yes
	
	immediate = { POD_update_spirit_pact_modifiers = yes }
}


POD_spirit_maintenance.02 = {
	hidden = yes
	
	immediate = {
		scope:dead_spirit = {
			POD_flag_injector_spirit_type = {
				EFFECT = POD_apply_spirit_form_from_scope_value
			}
		}
	}
}


#Find a dynasty member to pact with
POD_spirit_maintenance.03 = {
	hidden = yes

	trigger = {
		num_of_relation_human_companion = 0
		POD_is_ancestor_spirit_trigger = yes
	}
	
	on_trigger_fail = {
		# Keep the maintenance event going to check if the current human companion has died
		if = {
			limit = { 
				POD_is_ancestor_spirit_trigger = yes
				dynasty ?= {
					any_dynasty_member = {
						is_alive = yes
						POD_is_not_incorporeal_spirit = yes
					}
				}
			}
			trigger_event = { id = POD_spirit_maintenance.03 years = { 5 15 } }
		}
	}

	immediate = {
		save_scope_as = ancestor_spirit
		dynasty ?= {
			ordered_dynasty_member = {
				limit = {
					POD_fera_ai_can_spirit_pact_ancestors = yes
					POD_knows_spirit_language_custom_trigger = { ACTOR = this RECIPIENT = scope:ancestor_spirit }
					trigger_if = {
						limit = { scope:ancestor_spirit.religion = { is_in_family = rf_gaia } }
						religion = { is_in_family = rf_gaia }
					}
					trigger_else = { religion = { is_in_family = rf_wyrm } }
				}
				order_by = {
					value = 10

					add = {
						value = {
							add = diplomacy
							add = martial
							add = stewardship
							add = intrigue
							add = learning
							add = prowess

							multiply = 0.5
							floor = yes
						}
					}

					add = {
						value = {
							add = POD_fera_rank_value
							multiply = 3
						}
					}

					if = {
						limit = { is_close_family_of = scope:ancestor_spirit }
						add = 40
					}

					if = {
						limit = { is_child_of = scope:ancestor_spirit }
						add = 50
					}
					else_if = {
						limit = { is_sibling_of = scope:ancestor_spirit }
						add = 40
					}
					else_if = {
						limit = { is_grandchild_of = scope:ancestor_spirit }
						add = 30
					}
					
					if = {
						limit = { highest_held_title_tier = tier_hegemony }
						add = 100
					}
					else_if = {
						limit = { highest_held_title_tier = tier_empire }
						add = 50
					}
					else_if = {
						limit = { highest_held_title_tier = tier_kingdom }
						add = 30
					}
					else_if = {
						limit = { highest_held_title_tier = tier_duchy }
						add = 20
					}

					if = {
						limit = { faith = { has_doctrine = tenet_ancestor_worship } }
						add = 30
					}
				}
				max = 1
				save_scope_as = target
			}
		}
		
		scope:target ?= {
			POD_form_spirit_pact_effect = { TARGET = scope:ancestor_spirit }

			scope:ancestor_spirit = {
				#Leave if in another Coterie
				if = {
					limit = {	
						POD_has_coterie_trigger = yes
						NOT = { POD_is_in_same_coterie_trigger = { TARGET = scope:target } }
					}
					POD_leave_coterie_effect = yes

					#Only join if they currently have a coterie. If they do not, then wait until Fera AI handles that
					if = {
						limit = { scope:target = { POD_has_coterie_trigger = yes POD_coterie_at_max_size_trigger = no } }
						scope:target.var:coterie = { save_scope_as = coterie }
						POD_join_coterie_effect = yes
					}
				}
			}
		}

		
		#Try again after 5 years if your dynasty is still alive
		if = {
			limit = { 
				num_of_relation_human_companion = 0
				dynasty ?= {
					any_dynasty_member = {
						is_alive = yes
						POD_is_not_incorporeal_spirit = yes
					}
				}
			}
			trigger_event = { id = POD_spirit_maintenance.03 years = { 5 15 } }
		}
	}
}



#Remove AI Spirits
POD_spirit_maintenance.04 = {
	hidden = yes

	immediate = {
		scope:spirit = {
			if = {
				limit = {
					has_character_flag = ai_spirit
					num_of_relation_human_companion = 0
				}
				death = { death_reason = death_returned_umbra }
			}
		}
	}
}
