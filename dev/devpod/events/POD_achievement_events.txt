namespace = POD_achievements

# Events for achievements that need to be checked on a delay

POD_achievements.27 = {
	hidden = yes
	trigger = {
		is_ai = no
		POD_is_vampire_trigger = yes
		# contrary to what the achievement says, it's fine if the 3 relations are the same person
		any_relation = {
			type = coterie_member
			count >= 1
			POD_is_werewolf_trigger = yes
		}
		any_relation = {
			type = friend
			count >= 1
			POD_is_werewolf_trigger = yes
		}
		any_relation = {
			type = lover
			count >= 1
			POD_is_werewolf_trigger = yes
		}
	}
	immediate = {
		set_global_variable = pod_achievement_27 # Achievement: Making Fetch Happen
	}
}
