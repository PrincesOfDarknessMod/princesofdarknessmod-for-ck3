POD_yamaking_maintenance.1 = {
	hidden = yes

	trigger = {
		POD_is_yamaking_trigger = yes
	}

	immediate = {
		POD_yearly_defiled_xp_pulse = yes
		if = {
			limit = {
				domicile ?= { has_domicile_building_or_higher = yamaking_hells_of_rage_unyielding_rage }
				save_temporary_scope_value_as = {
					name = biyearly_countdown
					value = { value = current_year modulo = 25 }
				}
				scope:biyearly_countdown = 0
			}
			add_character_modifier = extra_life_1_modifier
		}
	}
}
