POD_mask_of_empathy.001 = {
    type = character_event
	content_source = dlc_POD
    title = POD_mask_of_empathy.001.t
	desc = POD_mask_of_empathy.001.desc
	theme = presence
	
	left_portrait = {
	    character = scope:presence_user
		animation = scheme
	}
	
	right_portrait = {
		character = scope:target
		animation = worry
	}
	lower_right_portrait = {
		character = scope:target_relation
		animation = boredome
	}
	
	immediate = {
		scope:actor = {
			save_scope_as = presence_user
		}
		
		scope:recipient = {
			save_scope_as = target
		}
		
		scope:secondary_recipient = {
			save_scope_as = target_relation
		}
	}
	
	option = {
		name = POD_mask_of_empathy.001.option.a
		custom_tooltip = POD_mask_of_empathy.001.option.a.tt
		
		POD_hunger_rouse_check_effect = yes
		POD_hunger_rouse_check_effect = yes
		
		duel = {
			skill = diplomacy
			value = scope:target.diplomacy
			desc = POD_mask_of_empathy.001.option.a.desc

			30 = {
				desc = POD_mask_of_empathy.001.option.a.success

				compare_modifier = {
					value = scope:duel_value
					multiplier = high_positive_duel_skill_multiplier
				}
				
				modifier = {
				    add = 25
					scope:presence_user = { has_perk = perfect_world_perk }
				}
				
				
				
				custom_tooltip = POD_mask_of_empathy.001.option.a.tt.attacker_wins
				
				hidden_effect = {
					scope:presence_user = {
						trigger_event = {
							id = POD_mask_of_empathy.002
							days = 2
						}
					}
				}
			}
			
			70 = {
				desc = POD_mask_of_empathy.001.option.a.failure

				compare_modifier = {
					value = scope:duel_value
					multiplier = low_negative_duel_skill_multiplier 
				}
				
				modifier = {
					add = 20
					scope:target = { 
					    OR = {
						    has_trait = fortitude
							has_trait = fortitudeadvanced
						}
					}
				}
				modifier = {
					add = 40
					scope:target = { has_trait = fortitudeadvanced }
				}
				
				custom_tooltip = POD_mask_of_empathy.001.option.a.tt.defender_wins
				
				hidden_effect = {	
					
					scope:presence_user = {
						trigger_event = {
							id = POD_mask_of_empathy.003
							days = 3
						}
					}
					
				}		
			}
		}
	}
	
	option = {
		name = POD_mask_of_empathy.001.option.b
		custom_tooltip = POD_mask_of_empathy.001.option.b.tt
	}
}
	
POD_mask_of_empathy.002 = {
    type = character_event
	content_source = dlc_POD
    title = POD_mask_of_empathy.002.t
	desc = POD_mask_of_empathy.002.desc
	theme = presence
	
	left_portrait = {
	    character = scope:presence_user
		animation = personality_dishonorable
	}
	
	right_portrait = {
		character = scope:target
		animation = shock
	}
	lower_right_portrait = {
		character = scope:target_relation
		animation = disapproval
	}
	
	immediate = {
		if = {
			limit = {
				scope:target = {
					has_character_flag = losing_friend
				}
			}
			scope:target = {
				remove_relation_friend = scope:target_relation
				remove_character_flag = losing_friend
			}			
		}
		if = {
			limit = {
				scope:target = {
					has_character_flag = losing_rival
				}
			}
			scope:target = {
				remove_relation_rival = scope:target_relation
				remove_character_flag = losing_rival
			}			
		}
		if = {
			limit = {
				scope:target = {
					has_character_flag = losing_lover
				}
			}
			scope:target = {
				remove_relation_lover = scope:target_relation
				remove_character_flag = losing_lover
			}			
		}
	}
	
	option = {
		name = POD_mask_of_empathy.002.option.a
		custom_tooltip = POD_mask_of_empathy.002.option.a.tt
		if = {
			limit = { POD_has_a_beast_trigger = yes }
			custom_tooltip = POD_mask_of_empathy.002.option.a.tt.beast
		}
		scope:target = {
			add_opinion = {
				target = scope:presence_user
				modifier = hurt_opinion
				opinion = -30
			}
		}		
	}
}
