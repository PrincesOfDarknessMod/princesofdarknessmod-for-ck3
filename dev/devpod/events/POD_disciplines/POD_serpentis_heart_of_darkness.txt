namespace = POD_heart_of_darkness

POD_heart_of_darkness.001 = {
    type = character_event
	content_source = dlc_POD 
	title = POD_heart_of_darkness.001.t
	desc = POD_heart_of_darkness.001.desc
	theme = serpentis
	
	left_portrait = {
	    character = root
		animation = personality_zealous
	}
	
	option = {
	    name = POD_heart_of_darkness.001.option.a
		custom_tooltip = POD_heart_of_darkness.001.option.a.tt
		if = {
			limit = { POD_has_a_beast_trigger = yes }
			custom_tooltip = POD_heart_of_darkness.001.option.a.tt.beast
		}

		POD_discipline_cost_effect = { COUNT = 2 }
		
		add_piety = -800
		
		hidden_effect = {
		    trigger_event = {
			    id = POD_heart_of_darkness.002
				days = 2
			}
	    }
		
		add_character_modifier = heart_of_darkness_modifier
	}
	
	option = {
	    name = POD_heart_of_darkness.001.option.b
		custom_tooltip = POD_heart_of_darkness.001.option.b.tt
		if = {
			limit = { POD_has_a_beast_trigger = yes }
			custom_tooltip = POD_heart_of_darkness.001.option.b.tt.beast
		}
		remove_interaction_cooldown = POD_heart_of_darkness_interaction
	}
	
}
