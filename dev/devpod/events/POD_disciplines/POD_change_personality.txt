POD_change_personality.100 = {
    type = character_event
	content_source = dlc_POD 
	title = POD_change_personality.100.t
	desc = POD_change_personality.100.desc
	theme = dominate
	
	left_portrait = {
	    character = scope:recipient
		animation = personality_zealous
	}

	option = {
	    name = POD_change_personality.100.option.a
		custom_tooltip = POD_change_personality.100.option.a.tt
		
		trigger = {
			scope:recipient = { number_of_personality_traits < personality_trait_limit }
		}

		trigger_event = {
			    id = POD_change_personality.200
			}
	}
	
	option = {
	    name = POD_change_personality.100.option.b

		trigger = {
			scope:recipient = { number_of_personality_traits > personality_trait_minimum }
		}

		trigger_event = {

			}
	}

	option = {
	    name = POD_change_personality.100.option.c
		custom_tooltip = POD_change_personality.100.option.c.tt

		#add_piety = major_piety_value
	}
	
}
