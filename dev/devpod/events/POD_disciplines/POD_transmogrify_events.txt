POD_transmogrify.1003 = {
	type = character_event
	content_source = dlc_POD
	title = POD_transmogrify.1003.t
	desc = POD_transmogrify.1003.desc
	theme = vicissitude

	left_portrait = {
		character = scope:actor
		animation = personality_rational
	}

	right_portrait = {
		character = scope:recipient
		animation = prisondungeon
	}
	
	widget = {
		gui = "POD_event_window_widget_learn_true_names"
		container = "custom_widgets_container"
	}

	immediate = {
		scope:recipient = {
			save_scope_as = cant_select_multiple_traits # jank to make the event widget work
			every_character_trait = {
				limit = {
					POD_can_study_anatomy_trait = { SCHOLAR = scope:actor }
				}
				scope:actor = {
					add_to_variable_list = {
						name   = potential_traits
						target = prev
					}
				}
			}
		}
	}
	
	option = {
		name = POD_transmogrify.1003.a

		scope:actor = {
			if = {
				limit = { any_in_list = { variable = selected_traits count = 1 } }

				every_in_list = {
					variable = selected_traits
					prev = {
						POD_start_anatomy_study_journey_effect = { TRAIT = prev }
					}
				}
	
				POD_torture_interaction_effect = yes
			}
		}
	}
	
	option = {
		name = POD_transmogrify.1003.b
	}

	after = {
		scope:actor = {
			clear_variable_list = potential_traits
			clear_variable_list = selected_traits
		}
	}
}
