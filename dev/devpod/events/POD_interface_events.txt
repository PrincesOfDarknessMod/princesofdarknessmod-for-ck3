# select a secret to expose
POD_interface.100 = {
	type = character_event
	content_source = dlc_POD
	
	title = POD_interface.100.t
	desc  = POD_interface.100.desc

	theme = secret
	
	widgets = {
		widget = {
			gui = "POD_event_window_widget_secret_selection"
			container = "custom_widgets_container"
		}
	}
	
	left_portrait = {
		character = scope:secret_target
		# idle animation
	}
	
	trigger = {
		# prevent the player from opening the event multiple times
		# and potentially screwing with the variable list
		POD_is_busy_in_secret_selector_event = no
	}
	
	immediate = {
		POD_on_secret_selector_event_immediate = yes
		POD_save_secrets_to_list_for_event = {
			TARGET = scope:secret_target
			FILTER_UNKNOWN = no
		}
	}

	option = {
		name = POD_interface.100.a
		
		if = {
			limit = { has_variable = event_secret_selection }
			
			random_courtier = {
				save_scope_as = exposer
			}
			if = {
				limit = { NOT = { exists = scope:exposer } }
				random_independent_ruler = {
					save_scope_as = exposer
				}
			}
			var:event_secret_selection = {
				expose_secret = scope:exposer
			}
			stress_and_fulfillment_impact = {
				honest = minor_stress_impact_loss
				just = minor_stress_impact_loss
				shy = miniscule_stress_impact_gain
			}
		}
	}

	option = {
		name = POD_interface.100.b
	}
	
	after = {
		POD_on_secret_selector_event_after = yes
	}
}
