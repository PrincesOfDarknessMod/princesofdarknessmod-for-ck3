								weight = { base = 20 }

								weight = { base = 40 }

								weight = { base = 30 }

								weight = { base = 30 }

							intellect_good_2 = {
								weight = { base = 30 }

							faith = faith:innocence
							culture = this.culture
							diplomacy = { 5 10 }
							stewardship = { 5 10 }
							save_scope_as = ddaughter
				}
			}
		}

		limit = {
			exists = scope:ddaughter

			scope:ddaughter = {
			POD_remove_all_terrible_traits_effect = yes
			}
		}

	option = {
		name = donors_daughter.1.a
		
		duel = {
			skill = diplomacy
			value = 15

				compare_modifier = {
					value = scope:duel_value

				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
					min = -49
				}
				trigger_event = { id = donors_daughter.2 }
				desc = donors_daughter.1.a.failure.ctt
				custom_tooltip = donors_daughter.1.a.failure.tt
				}
			}

		stress_and_fulfillment_impact = {
			arrogant = medium_stress_impact_gain
			honest = medium_stress_impact_gain
		}
	}
	
	option = {
	name = donors_daughter.1.b
	
	change_variable = {
		name = visit_outcome
		add = 1
		}
	
	add_character_modifier = {
		modifier = not_so_shadowy_modifier

	}
	
	trigger_event = { id = donors_daughter.2 }
	
	stress_and_fulfillment_impact = {

			diligent = major_stress_impact_gain
			deceitful = major_stress_impact_gain
			arrogant = major_stress_impact_gain
			honest = minor_stress_impact_loss
		}
	}
		
	option = {
		name = donors_daughter.1.c
		
		change_variable = {
		name = visit_outcome

		}
