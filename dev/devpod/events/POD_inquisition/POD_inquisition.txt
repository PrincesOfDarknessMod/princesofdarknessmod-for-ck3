namespace = PODinquisition

# Warning event

PODinquisition.0001 = {
	hidden = yes
	scope = none
	
	trigger = {
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:destroy_inquisition_objective_decision
			}
		}
	}
	
	immediate = {
		every_player = {
			trigger_event = PODinquisition.2000
			}
	}
}

PODinquisition.2000 = {
	type = character_event
	content_source = dlc_POD
	title = PODinquisition.0001.t
	desc = PODinquisition.0001.desc
	theme = inquisition
	left_portrait = {
		character = root
		animation = worry
	}
	
	immediate = {
		play_music_cue = "mx_cue_general_crusade_event_1"
	}

	trigger = {
	}

	option = {
		name = PODinquisition.0001.a
			trigger = {
				NOT = { has_faith = faith:shadowinquisition }
			}
		}
	option = {
		name = PODinquisition.0001.b
			trigger = {
				has_faith = faith:shadowinquisition	
			}
		custom_tooltip = PODinquisition.1001.player_warningb
		}	
}

PODinquisition.1001 = {
	type = character_event
	content_source = dlc_POD
	title = PODinquisition.1001.t
	desc = PODinquisition.1001.desc
	theme = inquisition
	left_portrait = {
		character = root
		animation = worry
	}
	
	immediate = {
		faith:viahumanitatis = {
			save_scope_as = viahumanitatis
		}
		show_as_tooltip = {
			scope:frenchinquisition_county = {
				set_county_faith = faith:viahumanitatis
			}
		}
	}	

	option = { #It'll probably be fine
		name = PODinquisition.1001.a
		custom_tooltip = PODinquisition.1001.player_warning
	}
}


### Spawn Inquisitor
PODinquisition.0003 = {
	scope = none
	hidden = yes

	trigger = {
		scope:frenchinquisition_county = {
			holder = {
				OR = { # Last resort check to make sure that we don't automatically game over a player
					is_ai = yes # Either they have to be AI
					any_held_title = { # Or they have to hold land elsewhere
						tier = tier_county
						NOT = { this = scope:frenchinquisition_county }
					}
				}
			}
		}
	}

	immediate = {
		debug_log = "French Grand Master appeared!"
		debug_log_date = yes
		create_frenchgm_effect = yes	
	}
}

	hidden = yes

	trigger = {
		scope:carpathianinquisition_county = {
			holder = {
				OR = { # Last resort check to make sure that we don't automatically game over a player
					is_ai = yes # Either they have to be AI
					any_held_title = { # Or they have to hold land elsewhere
						tier = tier_county
						NOT = { this = scope:carpathianinquisition_county }
					}
				}
			}
		}
	}

	immediate = {
		debug_log = "Carpathian Grand Master appeared!"
		debug_log_date = yes
		create_carpathiangm_effect = yes
	}
}

PODinquisition.1000 = {
	hidden = yes
	
	trigger = { 
		has_game_rule = age_of_inquisition_difficulty_hard
		has_character_modifier = POD_war_fervor_modifier
		any_neighboring_and_across_water_top_liege_realm_owner = {
			NOT = { is_at_war_with = root }
			POD_is_unholy_trigger = yes
			has_criminal_splat_trigger = { ACTOR = root RECIPIENT = this }
		}
	}
	
	immediate = {
		random_neighboring_and_across_water_top_liege_realm_owner = {
			limit = { 
				NOT = { is_at_war_with = root }
				POD_is_unholy_trigger = yes
				has_criminal_splat_trigger = { ACTOR = root RECIPIENT = this }
			}
			root = {
				start_war = {
					cb = POD_event_war
					target = prev
				}
			}
			weight = {
				base = 0.001
				modifier = {
					add = 1000
					AND = {
						is_ai = yes
						is_at_war = no
						realm_size >= major_realm_size
					}
				}
				modifier = {
					add = 100
					AND = {
						is_ai = yes
						is_at_war = no
						realm_size >= medium_realm_size
					}
				}
				modifier = {
					add = 10
					AND = {
						is_ai = yes
						is_at_war = no
						realm_size >= minor_realm_size
					}
				}
				modifier = {
					add = 1
					AND = {
						is_ai = yes
						is_at_war = no
						realm_size >= tiny_realm_size
					}
				}
			}
		}
	}
}

PODinquisition.100 = {
	type = character_event
	content_source = dlc_POD
	title = PODinquisition.100.t
	desc = PODinquisition.100.desc
	theme = inquisition
	left_portrait = {
		character = root
		animation = personality_zealous
	}
	
	immediate = {
		set_variable = {
			name = curia_support_var
			value = 5
		}		
	}	

	option = { 
		name = PODinquisition.100.a

		domicile ?= {
			change_provisions = 250
		}
		
		if = {
			limit = {
				gold <= 50
			}
		add_gold = 50
		}
		
		add_internal_flag = dangerous
	}

}
