beer_merchant.1 = {
	type = character_event
	content_source = dlc_POD
	title = beer_merchant.1.t
	desc = beer_merchant.1.desc
	
	theme = stewardship
	
	left_portrait = root
	
	right_portrait = {
		character = scope:merchant
	}
	
	trigger = {
		faith = faith:shadowinquisition
		NOT = { has_character_flag = had_event_beer_merchant }
		any_held_title = {
			tier = tier_county
		}
		exists = root.capital_province
	}
	
	weight_multiplier = {
		base = 1

		modifier = {
			add = 2
			has_lifestyle = stewardship_lifestyle

		modifier = {
			add = -1.9
			is_ai = yes
		}
	}
	
	immediate = {
		add_character_flag = {
			flag = had_event_beer_merchant
			years = 30
		}
		
		add_character_flag = {
			flag = had_recent_lifestyle_ongoing_event
			years = ongoing_lifestyle_event_delay
		}
		
		random_held_title = {
			limit = {
				tier = tier_county
			}
			
			save_scope_as = target_county
			
			hidden_effect = {
				create_character = {
							location = root.capital_province
							age = { 35 55 }
							gender = male
							trait = education_stewardship_3
							trait = shrewd
							trait = intellect_good_1
							trait = mortal
							faith = faith:innocence
							culture = root.culture
							stewardship = { 12 14 }
							save_scope_as = merchant
				}
			}
		}
	}

	#STEWARDSHIP
	
		option = {
		name = beer_merchant.a
		
		duel = {
			skill = stewardship
			value = 16

			desc = outcome_in_a_few_days

			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
					min = -49
				}
				trigger_event = { id = beer_merchant.2 }
				desc = beer_merchant.a.success.ctt
				custom_tooltip = beer_merchant.a.success.tt
				}
			
			50 = {		
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
					min = -49
				}
				trigger_event = { id = beer_merchant.3 }
				desc = beer_merchant.a.failure.ctt
				custom_tooltip = beer_merchant.a.failure.tt
			}
		}
		stress_and_fulfillment_impact = {
			lazy = medium_stress_impact_gain
			craven = minor_stress_impact_gain
		}

		ai_chance = {
			base = 100
			ai_value_modifier = {
				ai_boldness = 50
				ai_energy = 200
			}

			compare_modifier = {
				trigger = {
					ai_rationality >= 10
				}
				value = {
					value = stewardship
					add = -10
				}
				multiplier = 1000
			}
		}
	}
	
	#DIPLOMACY
		option = {
		name = beer_merchant.b
		
		duel = {
			skill = diplomacy
			value = 20

			desc = outcome_in_a_few_days

			50 = {		
				compare_modifier = {
					value = scope:duel_value

				}

				}

			}
		}
		stress_and_fulfillment_impact = {
			paranoid = major_stress_impact_gain

		}

		ai_chance = {
			base = 100
			ai_value_modifier = {
				ai_boldness = 200
				ai_energy = 50
			}

			compare_modifier = {
				trigger = {
					ai_rationality >= 10
				}
				value = {
					value = stewardship
					add = -10
				}
				multiplier = 1000
			}
		}
	}
	
	#INTRIGUE
	option = {

		}

			base = 100
