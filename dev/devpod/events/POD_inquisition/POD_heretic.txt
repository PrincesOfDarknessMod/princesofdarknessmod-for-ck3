namespace = heretic_event

heretic_event.1 = {
	type = character_event
	content_source = dlc_POD
	title = heretic_event.1.t
	desc = heretic_event.1.desc
	theme = diplomacy
	override_background = {
		reference = gallows
	}
	
	left_portrait = {
		character = scope:inquiruler
		animation = personality_rational
	}
	
	right_portrait = {
		character = scope:heretic
		animation = personality_bold
	}
	
	trigger = {
		faith = faith:shadowinquisition
		NOT = { has_character_flag = had_event_heretic }
		any_held_title = {
			tier = tier_county
		}
		exists = root.capital_province
	}

	weight_multiplier = {
		base = 1

		modifier = {
			add = 2
			has_lifestyle = learning_lifestyle	
		}
		
		modifier = {
			add = -1.9
			is_ai = yes
		}
	}

	immediate = {
		save_scope_as = inquiruler
		add_character_flag = {
			flag = had_event_heretic
			years = 30
		}
		
		add_character_flag = {
			flag = had_recent_lifestyle_ongoing_event
			years = ongoing_lifestyle_event_delay
		}
		
		random_held_title = {
			limit = {
				tier = tier_county
			}
			
			save_scope_as = target_county
			
			hidden_effect = {
				create_character = {
							location = root.capital_province
							age = { 25 35 }
							gender = male
							trait = education_learning_3
							trait = intellect_good_2
							trait = beauty_good_1
							trait = zealous
							trait = brave
							trait = gregarious
							trait = mortal
							faith = faith:viacommunitas
							culture = this.culture
							diplomacy = { 5 10 }
							learning = { 5 15 }
							save_scope_as = heretic
				}
			}
		}
	
	if = {
		limit = {
			exists = scope:heretic
				}
			scope:heretic = {
			POD_remove_all_terrible_traits_effect = yes
			add_character_flag = POD_is_wearing_rags_flag
			}
		}
	}
	
	option = {
		name = heretic_event.1.a
		
		duel = {
			skill = diplomacy
			value = 15
			

			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
					min = -49
				}
				trigger_event = { id = heretic_event.11 }
				desc = heretic_event.1.a.success.ctt
				custom_tooltip = heretic_event.1.a.success.tt
				}
			
			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
					min = -49
				}
				trigger_event = { id = heretic_event.12 }
				desc = heretic_event.1.a.failure.ctt
				custom_tooltip = heretic_event.1.a.failure.tt
				}
			}
		

		add_diplomacy_lifestyle_xp = major_lifestyle_experience

		hidden_effect = {
			scope:heretic = {
				death = {
					death_reason = death_disappearance
				}
			}
		}

		stress_and_fulfillment_impact = {
			zealous = medium_stress_impact_gain
			sadistic = minor_stress_impact_gain
			vengeful = minor_stress_impact_gain
		}
	}
	
	option = {
	name = heretic_event.1.b
		
	trigger_event = { id = heretic_event.2 }
	
	stress_and_fulfillment_impact = {
			shy = medium_stress_impact_gain
			lazy = minor_stress_impact_gain
			craven = minor_stress_impact_gain
			}
			
	add_learning_lifestyle_xp = major_lifestyle_experience
		
	}
		
	option = {
		name = heretic_event.1.c
		
		scope:heretic = {
			death = {
				death_reason = death_execution
				killer = root
			}
		}
		
		trigger_event = { id = heretic_event.3 }

		stress_and_fulfillment_impact = {
			just = major_stress_impact_gain
			forgiving = medium_stress_impact_gain
			compassionate = medium_stress_impact_gain
			callous = medium_stress_impact_loss
			sadistic = medium_stress_impact_loss
			arbitrary = medium_stress_impact_loss
		}
	}
}
