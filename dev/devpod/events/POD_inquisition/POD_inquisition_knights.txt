			location = root.capital_province
			template = debutante_character

		else_if = {
            limit = { has_contemplative_order_dynasty_trigger = yes }

			location = root.capital_province
			template = member_character

			}
		}
		
		else_if = {
            limit = { dynasty = dynasty:13000007 }
			create_character = {
			location = root.capital_province
			template = comrade_character
			prowess = { 2 8 }
			save_scope_as = prospect
			}
		}
	}

	scope:player = {
	add_visiting_courtier = scope:prospect
	}
	scope:prospect = {
		if = {
				limit = {
					scope:player = {
					has_trait = order_member
					}
					exists = scope:player.dynasty
					scope:player.dynasty = { has_dynasty_perk = meritocracy_legacy_1 }
					}
				meritocracy_1_effect = yes
			}
		
		if = {
				limit = {
					scope:player = {
					has_trait = order_member
					}
					exists = scope:player.dynasty
					scope:player.dynasty = { has_dynasty_perk = meritocracy_legacy_2 }
					is_female = no
					}
				meritocracy_2_effect_a = yes
			}
		
		if = {
				limit = {
					scope:player = {
					has_trait = order_member
					}
					exists = scope:player.dynasty
					scope:player.dynasty = { has_dynasty_perk = meritocracy_legacy_2 }
					is_female = yes
					}
				meritocracy_2_effect_b = yes
			}
		
		if = {
				limit = {
					scope:player = {
					has_trait = order_member
					}
