POD_martyrs.3 = {
    type = character_event
	content_source = dlc_POD
	title = POD_martyrs.3.t
	desc = POD_martyrs.3.desc
	theme = inquisition
	
	left_portrait = {
	    character = root
		animation = personality_zealous
	}
	
	option = {
	name = POD_martyrs.3.a
	
	if = {
		limit = { 
			POD_has_true_faith_trigger = no
			OR = { 
				POD_is_infernalist_trigger = no
				POD_splat_mixing_haight_mode = yes
			}
		}
		add_trait = truefaith
	}
	else_if = {
		limit = { 
		has_trait = truefaith
		NOT = { has_trait = beatific }
		}
	add_faith_lifestyle_xp = 1000
		}
	if = {
		limit = {
			NOT = { has_trait = zealous }
			}
	random = {
	chance = 25
	add_trait = zealous
			}
		}
	if = {
		limit = { has_trait = depressed_1 }
		remove_trait = depressed_1
		}
	}
}

POD_martyrs.5 = {
    type = character_event
	content_source = dlc_POD
	title = POD_martyrs.5.t
	desc = POD_martyrs.5.desc
	theme = inquisition
	override_background = {
		reference = armory
	}
	left_portrait = {
	    character = root
		animation = personality_honorable
	}
	
	option = {
		name = POD_martyrs.5.a
	
		if = {
			limit = {
				has_trait = lifestyle_blademaster
				NOT = { has_trait_xp = { trait = lifestyle_blademaster value = 100 } }
			}
			add_trait_xp = { trait = lifestyle_blademaster value = 50 }
		}
		else_if = {
			limit = {
				has_trait = lifestyle_hunter
				NOT = { has_trait_xp = { trait = lifestyle_hunter track = hunter value = 100 } }
			}
			add_trait_xp = { trait = lifestyle_hunter track = hunter value = 50 }
		}
		else_if = {
			limit = { NOT = { has_trait = strong } }
			add_trait = strong
		}
		else = {
			add_stress = medium_stress_impact_loss
		}
	}
}
