#Control event
PODask_pope.2 = {
	type = character_event
	hidden = yes
	
	trigger = {
		faith = faith:shadowinquisition
	}

	immediate = {
		if = {
			limit = {
				any_directly_owned_province = {
					has_building_with_flag = building_guarantees_pope_support
				}
			}
			trigger_event = PODask_pope.3
		}
		else = {
			random_list = {
				10 = {
					trigger_event = PODask_pope.3			
				}
				10 = {
					trigger_event = PODask_pope.4		
				}

		duel = {

				add_intrigue_lifestyle_xp = 300

					title = PODask_pope.6.effect.success.desc

						value = scope:duel_value
