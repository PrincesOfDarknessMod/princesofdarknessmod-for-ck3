PODask_pope.1 = {
	type = character_event
	content_source = dlc_POD
	title = PODask_pope.1.t
	desc = PODask_pope.1.desc
	
	theme = inquisition
	
	option = {
		name = PODask_pope.1.a
		
		if = { 
			limit = {
				NOT = { this = character:792899 }
			}
		add_gold = 200
		
		add_character_modifier = {
			modifier = inquisition_full_support_modifier
			years = 5
		}
		}

			limit = {
				 this = character:792899 
			}

	}
	
	after = {
		if = { 
			limit = {
				this = character:792899 
			}
			trigger_event = {
				id = PODask_pope.1003
				days = 2
			}
		}
	}
}

#Control event
PODask_pope.2 = {
	hidden = yes
	scope = none
	
	trigger = {
	}

	immediate = {
		every_player = {
			limit = { faith = faith:shadowinquisition }
		if = {
			limit = {
				OR = {
				any_directly_owned_province = {
					has_building_with_flag = building_guarantees_pope_support
				}
				is_target_in_global_variable_list = {
					name = unavailable_unique_decisions
					target = flag:landless_inquisition_victory_objective_decision
				}	
				}				
			}
			trigger_event = PODask_pope.3
		}
		else = {
			random_list = {
				10 = {
					trigger_event = PODask_pope.3			
				}
				10 = {
					trigger_event = PODask_pope.4		
				}			
				10 = {
					trigger_event = PODask_pope.5			
				}
				10 = {
					trigger_event = PODask_pope.6			
				}
				10 = {
					trigger_event = PODask_pope.7			
				}
				10 = {
					trigger_event = PODask_pope.8			
				}
				10 = {
					trigger_event = PODask_pope.9			
				}
				10 = {
					trigger_event = PODask_pope.10			
				}
				10 = {
					trigger_event = PODask_pope.11		
				}
				10 = {
					trigger_event = PODask_pope.12			
				}			
			}
		}	
	}
		trigger_event = {
			id = PODask_pope.2
			years = { 10 25 }
		}		
	}
}

#legate for a Full support pope
PODask_pope.30 = {
	type = character_event
	content_source = dlc_POD
	title = PODask_pope.30.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					NOT = { has_character_modifier = chosen_of_light_modifier }
					OR = {
						is_immortal = yes
						POD_is_possessed_trigger = yes
					}
				}
				desc = PODask_pope.30.desc.exco
			}
			triggered_desc = {
				trigger = { 
					NOT = { has_character_modifier = chosen_of_light_modifier }
					OR = {
						has_trait = sorcerer
						has_trait = demonology
					}
				}
				desc = PODask_pope.30.desc.witch
			}
			triggered_desc = {
				desc = PODask_pope.30.desc
			}
		}
	}	
	theme = inquisition
	
	immediate = {
		if = {
			limit = {
				NOT = { exists = root.var:curia_support_var }
				NOT = { has_trait = excommunicated }
			}
		set_variable = {
			name = curia_support_var
			value = 0
		}				
		}
	}
	
	option = {
		name = PODask_pope.30.a
		
		trigger = { 
			NOT = { has_character_modifier = chosen_of_light_modifier }
			OR = {
				is_immortal = yes
				POD_is_possessed_trigger = yes
			}
		}

		add_internal_flag = dangerous

		if = {
			limit = { exists = root.var:curia_support_var }
			remove_variable = curia_support_var
		}

		add_trait = excommunicated
		
		custom_tooltip = inquisition_excommunicated_tt
	}

	option = {
		name = PODask_pope.30.b
		
		trigger = {
			NOT = { has_character_modifier = chosen_of_light_modifier }	
			OR = {
				has_trait = sorcerer
				has_trait = demonology
			}
			NOR = {
				is_immortal = yes
				POD_is_possessed_trigger = yes
			}			
		}

		change_variable = {
			name = curia_support_var
			subtract = 100
		}

		custom_tooltip = curiasuppport_minushundred_tt		
	}

	option = {
		name = PODask_pope.1.a
		
		trigger = { 
		OR = {
			has_character_modifier = chosen_of_light_modifier
			AND = {
			NOR = {
				has_trait = sorcerer
				has_trait = demonology
			}
			NOR = {
				is_immortal = yes
				POD_is_possessed_trigger = yes
			}	
			}
		}			
		}

		change_variable = {
			name = curia_support_var
			add = 20
		}	

		custom_tooltip = curiasuppport_plustwenty_tt	
	}	
}
