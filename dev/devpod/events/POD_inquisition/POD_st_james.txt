#Compostella: The Compostella Chapter get invited by St. James
PODstjames.2 = {
	type = character_event
	content_source = dlc_POD
	title = PODstjames.2.t
    desc = {
        desc = PODstjames.2.desc_common
        desc = PODstjames.2.desc_2ndpart
    }
	theme = inquisition
	override_background = bp3_hills_winter
	widget = { gui = event_window_widget_vfx_snow container = foreground_shader_vfx_container }
	left_portrait = {
		character = scope:delgado
		animation = chancellor
		camera = camera_event_left_pointing_back_fov20
	}
	right_portrait = {
		character = character:791797
		animation = lantern
		camera = camera_event_center_away_pointing_left_fov25
	}

	lower_center_portrait = {
		character = scope:st_james_leader
	}

	immediate = {
		character:791725 = {
			save_scope_as = st_james_leader
		}
		character:791796 = {
			save_scope_as = delgado
		}
		character:790170 = {
			save_scope_as = de_rubio
		}
	}
	trigger = {
	}

	#Join Saint James Sword: base choice
	option = {
		name = PODstjames.2.a
		#AI will always do this option
		ai_chance = 100

		custom_tooltip = PODstjames.2.a.tt
		scope:delgado = {
			POD_transfer_all_titles_to_character = {
				CHARACTER = character:790170
			}
		}
		#Trigger the follow-up event to St.James
		scope:st_james_leader = {
			trigger_event = {
				id = PODstjames.4
				days = 1
			}
		}
		#If St James isn't a player and Delgado is, taking this option will switch the player to Rodrigue
		if ={ 
			limit = { 
				scope:st_james_leader = {
					is_ai = yes
				}
				scope:delgado = {
					is_ai = no
				}
			}
			set_player_character = scope:st_james_leader
		}

		add_internal_flag = special
	}

	#Adventure time
	option = {
		name = PODstjames.2.b

		custom_tooltip = pod_generic_landless_tooltip

		POD_transfer_all_titles_to_character = {
			CHARACTER = character:790170
		}
		
		create_landless_adventurer_title_effect = {
			REASON = flag:voluntary
			FLAVOR_CHAR = root
		}
		
		create_landless_adventurer_title_tooltip_effect = yes

		scope:delgado = {
			add_courtier = character:791797
			add_courtier = character:791748
			add_courtier = character:795193
		}
		#AI will never take this option
		ai_chance = 0
	}	
	#Get Unlanded
	option = {
		name = PODstjames.2.c
		custom_tooltip = PODstjames.2.c.tt
		add_internal_flag = dangerous

		#AI will never take this option
		ai_chance = 0
	}
}

# Event triggered by the Compostella decision
PODstjames.100 = {
	type = character_event
	content_source = dlc_POD
	title = PODstjames.100.t
	desc = PODstjames.100.desc
	theme = inquisition
	override_background = temple_church
	left_portrait = {
		character = scope:poor_knight_leader
		animation = divine_gesture_2
	}
	right_portrait = {
		character = scope:compostella_eyes_leader
		animation = divine_gesture_3
	}
	lower_center_portrait = {
		character= scope:st_james_leader
		animation = worry
	}

	immediate = {
		#Poor Knights
		if = { #Baltasar alive 
			limit = {
				character:791796 = {
					is_alive = yes 
				}
			}
			character:791796 = {
				save_scope_as = poor_knight_leader
			}
		}
		#Rafael alive
		else_if = {
			limit = {
				character:791797 = {
					is_alive = yes 
				}
			}
			character:791797 = {
				save_scope_as = poor_knight_leader
			}			
		}
		else = {
			limit = {
				exists = dynasty:13000002.dynast
			}
			dynasty:13000002.dynast = {
				save_scope_as = poor_knight_leader
			}
		}

		#Eyes of Compostella
		if = { #Ignacia alive 
			limit = {
				character:791748 = {
					is_alive = yes 
				}
			}
			character:791748 = {
				save_scope_as = compostella_eyes_leader
			}
		}
		#Fortunato alive
		else_if = {
			limit = {
				character:795193 = {
					is_alive = yes 
				}
			}
			character:795193 = {
				save_scope_as = compostella_eyes_leader
			}			
		}
		else = {
			limit = {
				exists = dynasty:13000005.dynast
			}
			dynasty:13000005.dynast = {
				save_scope_as = compostella_eyes_leader
			}
		}
		#Knights
		if = { #If you are not part of the knights
			limit = {
				AND = { 
					exists = dynasty:13000006.dynast
					NOT = {
						dynasty = dynasty:13000006
					}
				}

			}
			dynasty:13000006.dynast = {
				save_scope_as = st_james_leader
			}
		}		
	}

	trigger = {
	}

	#Knights of Acre choice
	option = {
		name = PODstjames.100.a
		reverse_add_opinion = {
			target = scope:poor_knight_leader
			modifier = pleased_opinion
			opinion = 20
		}
		reverse_add_opinion = {
			target = scope:compostella_eyes_leader
			modifier = annoyed_opinion
			opinion = -20
		}
		if = {
			limit = {
				dynasty = dynasty:13000006
			}
			custom_tooltip = {
				text = POD_hunters_story_st_james_paranoia_or_trust_decrease_5_tt
				change_variable = { name = paranoia_or_trust add = -5 }
			}
		}
		#TODO: Add Acre Modifier
	}
	#Eyes of Compostella choice
	option = {
		name = PODstjames.100.b
		reverse_add_opinion = {
			target = scope:poor_knight_leader
			modifier = pleased_opinion
			opinion = 20
		}
		reverse_add_opinion = {
			target = scope:compostella_eyes_leader
			modifier = annoyed_opinion
			opinion = -20
		}
		if = {
			limit = {
				dynasty = dynasty:13000006
			}
			custom_tooltip = {
				text = POD_hunters_story_st_james_paranoia_or_trust_decrease_5_tt
				change_variable = { name = paranoia_or_trust add = -5 }
			}
		}
		#TODO: Add Oculus Modifier
	}
	#St. James choice [you are St. James]
	option = {
		name = PODstjames.100.c
		trigger = {
			dynasty = dynasty:13000006
		}
		reverse_add_opinion = {
			target = scope:poor_knight_leader
			modifier = annoyed_opinion
			opinion = -20
		}
		reverse_add_opinion = {
			target = scope:compostella_eyes_leader
			modifier = annoyed_opinion
			opinion = -20
		}
		add_piety = 150
		custom_tooltip = {
			text = POD_hunters_story_st_james_paranoia_or_trust_increase_5_tt
			change_variable = { name = paranoia_or_trust add = 5 }
		}
		#TODO: Add St. James Modifier
	}
	#St. James choice, but you are not a member of St. James
	option = {	
		name = PODstjames.100.d
		trigger = {
			NOT = {
				dynasty = dynasty:13000006
			}
		}
		reverse_add_opinion = {
			target = scope:st_james_leader
			modifier = pleased_opinion
			opinion = 20
		}
		#TODO: Add St James Modifier
	}
	after = {
		# We update the paranoia if playing st james.
		if = {
			limit = {
				dynasty = dynasty:13000006
			}
			hidden_effect = { POD_hunters_st_james_story_effect = yes }
		}
		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:hunters_st_james_compostela_decision
		}
	}
}
