### ==========================================
### MAJOR OBJECTIVE: "Cure the Withering Blood" — Completion Event
### ==========================================
### 2-way branch choice: Covenant Renewed or Ishologu Rite

POD_impundulu_objectives.1 = {
	type = character_event
	content_source = dlc_POD
	title = POD_impundulu_objectives.1.t
	desc = POD_impundulu_objectives.1.desc
	theme = vampire
	override_background = { reference = bp1_bonfire }

	left_portrait = {
		character = scope:victorious_impundulu
		animation = personality_rational
	}

	trigger = {
		is_alive = yes
	}

	immediate = {
	}

	### Option A: "Covenant Renewed" — preserve the Bomkazi bond
	option = {
		name = POD_impundulu_objectives.1.a

		POD_impundulu_apply_covenant_renewed_effect = yes

		### Announce to other vampire players
		hidden_effect = {
			every_player = {
				limit = {
					is_vampire_trigger = yes
					NOT = { this = scope:victorious_impundulu }
				}
				trigger_event = { id = POD_impundulu_objectives.2 days = 3 }
			}
		}
	}

	### Option B: "The Ishologu Rite" — break free from the dependency
	option = {
		name = POD_impundulu_objectives.1.b

		POD_impundulu_apply_ishologu_rite_effect = yes

		### Announce to other vampire players
		hidden_effect = {
			every_player = {
				limit = {
					is_vampire_trigger = yes
					NOT = { this = scope:victorious_impundulu }
				}
				trigger_event = { id = POD_impundulu_objectives.2 days = 3 }
			}
		}
	}
}
