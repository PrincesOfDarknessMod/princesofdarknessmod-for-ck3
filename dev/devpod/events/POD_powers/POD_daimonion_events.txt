#infection event, vector prespective
POD_daimonion_events.9100 = {
	type = character_event
	hidden = yes
	
	immediate = {
		if = {
			limit = { 
				OR = {
					has_trait = demonic_plague 
					has_trait = demonic_plague_non_lethal
				}
				var:waves_remaining > 0
			}
			save_health_court_owner_effect  = { SCOPE_NAME = contagion_court_owner }
			create_contagion_list_effect = { SICK_CHARACTER = root CONTAGION_COURT_OWNER = scope:contagion_court_owner }
			scope:contagion_court_owner = {
				every_councillor = {
					add_to_temporary_list = contagion_list
				}
				
				every_war_enemy = {
					add_to_temporary_list = contagion_list
				}
				
				every_war_ally = {
					add_to_temporary_list = contagion_list
				}
				
				  
				every_vassal = {
					add_to_temporary_list = contagion_list
				}
				every_liege_or_above = {
					add_to_temporary_list = contagion_list
				}
			}
			
			random_in_list = {
				
				list = contagion_list
				limit = {
					NOR = { 
						has_trait = demonic_plague 
						has_trait = demonic_plague_non_infectious
						has_trait = demonic_plague_non_lethal 
						has_trait = demonic_plague_non_lethal_and_infectious
						has_character_flag = overcame_demon_plague 
					}
				}
				weight = {
					base = 100
					modifier = {
						add = 200
						is_ruler = yes
					}
				}
				set_variable = { name = infector value = root }
				set_variable = { name = waves_remaining value = { value = root.var:waves_remaining subtract = 1 }}
				trigger_event = POD_daimonion_events.9200
				
			}			
			random_in_list = {
				
				list = contagion_list
				limit = {
					NOR = { 
						has_trait = demonic_plague 
						has_trait = demonic_plague_non_infectious
						has_trait = demonic_plague_non_lethal 
						has_trait = demonic_plague_non_lethal_and_infectious
						has_character_flag = overcame_demon_plague 
					}
				}
				weight = {
					base = 100
					modifier = {
						add = 200
						is_ruler = yes
					}
				}
				set_variable = { name = infector value = root }
				set_variable = { name = waves_remaining value = { value = root.var:waves_remaining subtract = 1 }}
				trigger_event = POD_daimonion_events.9200
				
			}
			random_in_list = {
				
				list = contagion_list
				limit = {
					NOR = { 
						has_trait = demonic_plague 
						has_trait = demonic_plague_non_infectious
						has_trait = demonic_plague_non_lethal 
						has_trait = demonic_plague_non_lethal_and_infectious
						has_character_flag = overcame_demon_plague 
					}
				}
				weight = {
					base = 100
					modifier = {
						add = 200
						is_ruler = yes
					}
				}
				set_variable = { name = infector value = root }
				set_variable = { name = waves_remaining value = { value = root.var:waves_remaining subtract = 1 }}
				trigger_event = POD_daimonion_events.9200
				
			}
		}
		
		random_list = {
			15 = {
				set_variable = { name = overcomes value = 0 }
				if = {
					limit = {
						OR = {
							has_trait = demonic_plague 
							has_trait = demonic_plague_non_infectious
						}
					}
					send_interface_toast = {
						title = demonic_plague_wound
						left_icon = root
				
						increase_wounds_effect = { REASON =  wounds }
					}
				}
				
				
			}
			0 = {
				modifier = {
					add = prowess
					always = yes
				}
				change_variable = {
					name = overcomes
					add = 1
				}
				send_interface_toast = {
					title = demonic_plague_overcome
					left_icon = root
					custom_tooltip = demonic_plague_overcome_ct
				}
				if = {
					limit = { var:overcomes >= 3 }
					remove_variable	= waves_remaining			
					send_interface_toast = {
						title = demonic_plague_overcome_complete
						left_icon = root
						custom_tooltip = demonic_plague_overcome_complete_ct
						remove_trait = demonic_plague 
						remove_trait = demonic_plague_non_infectious
						remove_trait = demonic_plague_non_lethal 
						remove_trait = demonic_plague_non_lethal_and_infectious
						add_character_flag = { flag = overcame_demon_plague years = 10 }
					}
				}
			}
			
			
		}
		if = {
			limit = {
				OR = { 
						has_trait = demonic_plague 
						has_trait = demonic_plague_non_infectious
						has_trait = demonic_plague_non_lethal 
						has_trait = demonic_plague_non_lethal_and_infectious

					}
			}
			trigger_event = { months = 1 id = POD_daimonion_events.9100 }
		}
		
	}
	
}
