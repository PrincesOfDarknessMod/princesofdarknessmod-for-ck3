	}

		container = "dynamic_birth_name"

}

true_magick.3000 = {
	hidden = yes
	immediate = {
		save_scope_as = target
		add_piety = 200
		scope:target_title = {
			true_magick_remove_modifiers_effect = yes
			true_magick_revert_traits_effect = yes
		}
		remove_list_variable = {
			name = spells_on_character
			target = scope:target_title
		}
		
	}
}

true_magick.4100 = {
	#type = character_event
	#title = true_magick.4100.t
	#desc = true_magick.4100.desc
	hidden = yes
	
	#theme = tremere
	#left_portrait = {
	#	character = root
	#	animation = personality_cynical
	#}

	immediate = { 
		#name = true_magick.4100.a
		hidden_effect = {
			save_scope_as = caster
			if = {
				limit = {
					any_in_list = {
						variable = casting_spells
						this = scope:spell
					}
				}
				set_variable = { name = roll_successes value = 0 }
				set_while_counter_variable_effect = yes
				while = {
					limit = { var:while_counter < casting_dice }
					scope:spell = {
						random = {
							chance = 50
							change_variable = { name = current_successes add = 1 } 
							scope:caster = { change_variable = { name = roll_successes add = 1 } }
						}
					}
					increase_while_counter_variable_effect = yes
				}
				remove_while_counter_variable_effect = yes
				
				if = {
					limit = {
						scope:spell.var:current_successes = { compare_value >= scope:spell.var:spell_difficulty }
					}
					add_piety = scope:spell.var:current_successes
					trigger_event = true_magick.4000
					
				}
				else = {
					trigger_event = { id = true_magick.4100 months = 1 }
				}
			}
		}
		
	}
	
}
