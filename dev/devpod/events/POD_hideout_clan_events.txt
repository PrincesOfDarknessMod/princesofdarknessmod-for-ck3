######################################################
# CLAN HIDEOUT UPGRADE EVENTS
# Namespace: POD_hideout_clan
# Event ID allocation:
#   001: Hideout Event Trigger
#   0002-0009: Malkavian Oracle's Chamber
#   0010-0019: Ravnos Jati Refuge visitors
######################################################

namespace = POD_hideout_clan

#Called for hideout events
POD_hideout_clan.0001 = {
	type = character_event
	hidden = yes

	immediate = {
		if = {
			limit = { domicile ?= { has_domicile_building_or_higher = vampire_malkavian_oracles_chamber_01 } }
			trigger_event = POD_hideout_clan.0002
		}
		else_if = {
			limit = { domicile ?= { has_domicile_building_or_higher = vampire_ravnos_jati_refuge_01 } }
			trigger_event = POD_hideout_clan.0010
		}
	}
}

######################################################
# MALKAVIAN — The Whispers Speak
# Periodic secret discovery event
# ID: POD_hideout_clan.0002
######################################################
POD_hideout_clan.0002 = {
	type = character_event
	content_source = dlc_POD
	title = POD_hideout_clan.0002.t
	desc = POD_hideout_clan.0002.desc
	theme = intrigue
	override_background = { reference = shader_event_kaleidoscope }
	
	widget = {
		gui = "POD_shader_event_kaleidoscope"
		container = "background_shader_vfx_container"
	}

	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		character = scope:secret_target
		animation = worry
	}

	trigger = {
		is_available = yes
		is_malkavian_trigger = yes
		domicile ?= {
			has_domicile_building_or_higher = vampire_malkavian_oracles_chamber_01
		}
		NOT = { has_character_flag = malkavian_whispers_cooldown }
	}

	immediate = {
		# Find a target with secrets
		# Priority 1: Rival or nemesis
		random_relation = {
			type = rival
			limit = {
				any_secret = {
					NOT = { is_known_by = root }
				}
			}
			save_scope_as = secret_target
		}
		# Priority 2: Powerful vassal
		if = {
			limit = { NOT = { exists = scope:secret_target } }
			random_vassal = {
				limit = {
					highest_held_title_tier >= tier_duchy
					any_secret = {
						NOT = { is_known_by = root }
					}
				}
				# Priority 3: Any vassal
				alternative_limit = {
					any_secret = {
						NOT = { is_known_by = root }
					}
				}
				save_scope_as = secret_target
			}
		}
		# Priority 4: Any courtier
		if = {
			limit = { NOT = { exists = scope:secret_target } }
			random_courtier = {
				limit = {
					any_secret = {
						NOT = { is_known_by = root }
					}
				}
				save_scope_as = secret_target
			}
		}
	}

	# Only show if we found a valid target
	option = {
		name = POD_hideout_clan.0002.a
		trigger = { exists = scope:secret_target }
		# Learn a random secret
		scope:secret_target = {
			random_secret = {
				limit = {
					NOT = { is_known_by = root }
				}
				reveal_to = root
			}
		}
		stress_and_fulfillment_impact = {
			base = 10
		}
		hidden_effect = {
			if = {
				limit = { domicile ?= { has_domicile_building_or_higher = vampire_malkavian_oracles_chamber_06 } }
				add_character_flag = { flag = malkavian_whispers_cooldown years = 3 }
			}
			else_if = {
				limit = { domicile ?= { has_domicile_building_or_higher = vampire_malkavian_oracles_chamber_05 } }
				add_character_flag = { flag = malkavian_whispers_cooldown years = 4 }
			}
			else_if = {
				limit = { domicile ?= { has_domicile_building_or_higher = vampire_malkavian_oracles_chamber_04 } }
				add_character_flag = { flag = malkavian_whispers_cooldown years = 5 }
			}
			else_if = {
				limit = { domicile ?= { has_domicile_building_or_higher = vampire_malkavian_oracles_chamber_03 } }
				add_character_flag = { flag = malkavian_whispers_cooldown years = 6 }
			}
			else_if = {
				limit = { domicile ?= { has_domicile_building_or_higher = vampire_malkavian_oracles_chamber_02 } }
				add_character_flag = { flag = malkavian_whispers_cooldown years = 8 }
			}
			else = {
				add_character_flag = { flag = malkavian_whispers_cooldown years = 10 }
			}
		}
	}

	# Reject option — always available
	option = {
		name = POD_hideout_clan.0002.b
		add_piety = 25
		hidden_effect = {
			if = {
				limit = { domicile ?= { has_domicile_building_or_higher = vampire_malkavian_oracles_chamber_03 } }
				add_character_flag = { flag = malkavian_whispers_cooldown years = 3 }
			}
			else_if = {
				limit = { domicile ?= { has_domicile_building_or_higher = vampire_malkavian_oracles_chamber_02 } }
				add_character_flag = { flag = malkavian_whispers_cooldown years = 4 }
			}
			else = {
				add_character_flag = { flag = malkavian_whispers_cooldown years = 5 }
			}
		}
	}

	# Fallback — no valid target found
	option = {
		name = POD_hideout_clan.0002.c
		trigger = { NOT = { exists = scope:secret_target } }
		custom_tooltip = POD_hideout_clan_0001_no_target_tt
		hidden_effect = {
			add_character_flag = {
				flag = malkavian_whispers_cooldown
				years = 3
			}
		}
	}
}

######################################################
# RAVNOS — A Traveler at the Threshold
# Periodic Jati visitor event
# ID: POD_hideout_clan.0010
######################################################
POD_hideout_clan.0010 = {
	type = character_event
	content_source = dlc_POD
	title = POD_hideout_clan.0010.t
	desc = POD_hideout_clan.0010.desc
	theme = diplomacy

	left_portrait = {
		character = root
		animation = personality_honorable
	}

	trigger = {
		is_available = yes
		is_ravnos_trigger = yes
		domicile ?= {
			has_domicile_building_or_higher = vampire_ravnos_jati_refuge_01
		}
		NOT = { has_character_flag = ravnos_jati_visitor_cooldown }
	}

	# Welcome them
	option = {
		name = POD_hideout_clan.0010.a
		remove_short_term_gold = 50
		hidden_effect = {
			if = {
				limit = { domicile ?= { has_domicile_building_or_higher = vampire_ravnos_jati_refuge_06 } }
				add_character_flag = { flag = ravnos_jati_visitor_cooldown years = 3 }
			}
			else_if = {
				limit = { domicile ?= { has_domicile_building_or_higher = vampire_ravnos_jati_refuge_04 } }
				add_character_flag = { flag = ravnos_jati_visitor_cooldown years = 6 }
			}
			else_if = {
				limit = { domicile ?= { has_domicile_building_or_higher = vampire_ravnos_jati_refuge_02 } }
				add_character_flag = { flag = ravnos_jati_visitor_cooldown years = 9 }
			}
			else = {
				add_character_flag = { flag = ravnos_jati_visitor_cooldown years = 12 }
			}
			# Fire appropriate sub-event based on jati type
			random_list = {
				20 = { trigger_event = POD_hideout_clan.0011 } # Phuri Dae
				20 = { trigger_event = POD_hideout_clan.0012 } # Kshatriya
				20 = { trigger_event = POD_hideout_clan.0013 } # Vaisyas
				20 = { trigger_event = POD_hideout_clan.0014 } # Sybarites
				20 = { trigger_event = POD_hideout_clan.0015 } # Brahmin
			}
		}
	}

	# Turn them away
	option = {
		name = POD_hideout_clan.0010.b
		add_prestige = -50
		hidden_effect = {
			add_character_flag = {
				flag = ravnos_jati_visitor_cooldown
				years = 5
			}
		}
	}
}
