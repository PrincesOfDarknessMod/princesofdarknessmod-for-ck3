POD_wraith_binding.100 = {
	type = character_event
	content_source = dlc_POD

	desc =  POD_wraith_binding.100.desc		
	theme = wraith
	right_portrait = scope:actor
	left_portrait = scope:recipient
	
	option = {
		name = POD_wraith_binding.100.option.a
		trigger = {
			stress < very_high_stress
		}
		
		POD_willpower_rouse_check_effect = yes
		
		save_scope_value_as = {
			name = mental_duel_defender_will
			value = flag:willpower_spent
		}
		
		scope:actor = {
			trigger_event = POD_wraith_binding.200
		}
		
		ai_chance = {
			base = 80
			modifier = {
				add = 100
				has_trait = paranoid
			}
		}	
	}

	option = {
		name = POD_wraith_binding.100.option.b
		save_scope_value_as = {
			name = mental_duel_defender_will
			value = flag:willpower_unspent
		}
		scope:actor = {
			trigger_event = POD_wraith_binding.200
		}		
	}	
	
	option = {
		name = POD_wraith_binding.100.option.c
		scope:actor = {
			trigger_event = POD_wraith_binding.300
		}
		
		stress_impact = {
			reclusive = minor_stress_gain
		}

			base = 5
			opinion_modifier = {
				opinion_target = scope:actor
			}

				add = -100

			}
