POD_wraith_maintenance.2 = {
	type = character_event
	content_source = dlc_POD
	title = POD_wraith_maintenance.2.t
	desc = POD_wraith_maintenance.2.desc

	theme = wraith
	left_portrait = root
	trigger = {
		has_trait = shadow
	}
	immediate = {
		custom_tooltip = {
			text = POD_wraith_maintenance.2.i
			random_in_list = {
				variable = owned_fetter_list
				limit = { var:level >= 1 }
				save_scope_as = fetter
			}
		}
	}
	option = { # Yes, We don't need it.
		name = POD_wraith_maintenance.2.a
		if = { limit = { scope:fetter.var:level = 1 }  
			scope:fetter.var:object = { POD_destroy_fetter = yes }
		}
		else = {
			scope:fetter.var:object = {
				POD_increase_fetter_level = { VALUE = 1 ACTION = subtract }
			}
		}
		ai_chance = {
			base = 20
		}
	}
	option = { # Yes, We don't need it.
		name = POD_wraith_maintenance.2.b
		if = { limit = { has_perk = tainted_relic_perk } POD_static_stress_gain = { COUNT = 150 } }
		else = { POD_static_stress_gain = { COUNT = 100 } }
		ai_chance = {
			base = 5
		}
	}
}

POD_wraith_maintenance.31 = {
	hidden = yes
	
	trigger = {
		is_ai = no
		has_trait = wraith
		NOT = { has_trait = spectre }
		NOT = { has_trait = ferryman }
		NOT = { has_trait = shadow }
		has_perk = devils_dare_perk
	}
	
	immediate = {
		if = {
			limit = {
				OR = {
					has_character_flag = devils_dare_spend_gold_flag
					has_character_flag = devils_dare_spend_prestige_flag
					has_character_flag = devils_dare_spend_piety_flag
				}
			}
			add_stress = monumental_stress_gain
		
			remove_character_flag = devils_dare_spend_gold_flag
			remove_character_flag = devils_dare_spend_prestige_flag
			remove_character_flag = devils_dare_spend_piety_flag
		}

		random_list = {
			10 = { add_character_flag = devils_dare_spend_gold_flag }
			10 = { add_character_flag = devils_dare_spend_prestige_flag }
			10 = { add_character_flag = devils_dare_spend_piety_flag }
			10 = { trigger = { has_perk = nullify_thorns_perk } }
		}
	}
}
