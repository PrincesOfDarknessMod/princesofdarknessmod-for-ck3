POD_wraith_maintenance.10 = {
	hidden = yes
	
	trigger = {
		has_trait = wraith
		NOT = { has_trait = risen }
		NOT = { has_trait = lament }
		num_of_relation_necromancer = 0
	}
	
	immediate = {
		if = {
			limit = { is_in_umbra_trigger = no }
			send_interface_toast = {
				title = POD_wraith_maintenance
				left_icon = root
				POD_wraith_willpower_check_effect = yes
				custom_tooltip = POD_wraith_maintenance.tt
			}
		}
		else_if = {
			limit = {
				has_variable = spirit_location
				var:spirit_location = flag:shadowlands
				NOT = { has_character_modifier = temporary_fetter_modifier }
			}
			random = {
				chance = {
					value = 0
					add = unset_fetter_point
				}
				send_interface_toast = {
					title = POD_wraith_maintenance
					left_icon = root
					POD_wraith_willpower_check_effect = yes
					custom_tooltip = shadowlands_fetter_maintenance_tooltip
				}
				trigger_event = POD_wraith_maintenance.21 # Reuse event
			}
		}
	}
}
