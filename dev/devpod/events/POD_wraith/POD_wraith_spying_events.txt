POD_wraith_spying.300 = {
	type = character_event
	content_source = dlc_POD
	title = POD_wraith_spying.300.t
	desc =  POD_wraith_spying.300.desc		
	theme = wraith
	right_portrait = scope:actor
	left_portrait = scope:recipient

	immediate = {
		scope:recipient = {
			random_secret = {
				limit = { NOT = {is_known_by = scope:actor}}
				save_scope_as = secret_to_learn
			}
		}
	}
	option = {
		name = POD_wraith_spying.300.option.a
		
		scope:secret_to_learn = {
			reveal_to = scope:actor
		}
		scope:actor = {
			if = {
				limit = { num_of_relation_necromancer >= 1 }
				every_relation = {
					type = necromancer
					send_interface_toast = {
						title = POD_wraith_spying.300.toasttitle
						custom_tooltip = POD_wraith_spying.300.toasttext
						left_icon = scope:actor
						right_icon = scope:recipient
						scope:secret_to_learn = {
							reveal_to = prev
						}

					}
				}
			}
			
		}
		

		scope:recipient = {
			
			trigger_event = POD_wraith_spying.400
		
		}	
	}	
}
