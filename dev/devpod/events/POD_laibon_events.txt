# Annual maintenance event for the Aye/Orun system
# Registered in POD_laibon_on_action
pod_laibon.1 = {
	hidden = yes

	trigger = {
		is_laibon_vampire = yes
	}

	immediate = {
		### Initialization check ###
		# If character qualifies but hasn't been initialized yet, do it now
		POD_laibon_spirit_initialize = yes
		POD_laibon_add_aye_and_orun_xp = {
			AYE  = pod_laibon_aye_net_value
			ORUN = pod_laibon_orun_net_value
		}
	}
}

######################################################################
# pod_laibon.100: The Mortal Petitioner
# A mortal comes to the Laibon seeking judgment in a local dispute.
# Requires: landed, mortal courtier alive
######################################################################
pod_laibon.100 = {
	type = character_event
	title = pod_laibon.100.t
	desc = pod_laibon.100.desc
	theme = vampire
	content_source = dlc_POD
	override_background = { reference = daitya }

	left_portrait = {
		character = root
		animation = throne_room_conversation_3
	}

	trigger = {
		is_laibon_vampire = yes
		NOT = { has_character_flag = had_pod_laibon_general_event }
		is_landed = yes
		any_courtier = {
			is_alive = yes
			POD_is_vampire_trigger = no
		}
	}

	immediate = {
		add_character_flag = {
			flag = had_pod_laibon_general_event
			years = 10
		}
	}

	# Option A - Judge the dispute personally (Diplomacy check)
	option = {
		name = pod_laibon.100.a
		duel = {
			skill = diplomacy
			value = medium_skill_rating
			target = root
			50 = {
				desc = pod_laibon.100.a.success
				compare_modifier = {
					value = scope:duel_value
					multiplier = high_positive_duel_skill_multiplier
				}
				send_interface_toast = {
					type = event_toast_effect_good
					title = pod_laibon.100.a.success
					POD_laibon_add_aye_and_orun_xp = {
						AYE  = 4
						ORUN = 0
					}
					add_character_modifier = {
						modifier = mortal_petitioner_success_modifier
						years = 5
					}
					add_prestige = minor_prestige_value
				}
			}
			50 = {
				desc = pod_laibon.100.a.failure
				compare_modifier = {
					value = scope:duel_value
					multiplier = low_negative_duel_skill_multiplier
				}
				send_interface_toast = {
					type = event_toast_effect_bad
					title = pod_laibon.100.a.failure
					POD_laibon_add_aye_and_orun_xp = {
						AYE  = 2
						ORUN = 0
					}
					add_character_modifier = {
						modifier = mortal_petitioner_failure_modifier
						years = 2
					}
				}
			}
		}
	}

	# Option B - Delegate to a mortal advisor
	option = {
		name = pod_laibon.100.b
		POD_laibon_add_aye_and_orun_xp = {
			AYE  = 2
			ORUN = 0
		}
		remove_short_term_gold = tiny_gold_value
	}

	# Option C - Send them away
	option = {
		name = pod_laibon.100.c
		POD_laibon_add_aye_and_orun_xp = {
			AYE  = -2
			ORUN = 0
		}
		stress_and_fulfillment_impact = {
			base = -15
		}
	}
}

######################################################################
# pod_laibon.101: Voices on the Wind
# Spectral whispers carry knowledge from the spirit world.
######################################################################
pod_laibon.101 = {
	type = character_event
	title = pod_laibon.101.t
	desc = pod_laibon.101.desc
	theme = vampire
	override_background = { reference = bp1_bonfire }
	content_source = dlc_POD

	left_portrait = {
		character = root
		animation = eavesdrop
		camera = camera_activity_horse
	}

	trigger = {
		is_laibon_vampire = yes
		NOT = { has_character_flag = had_pod_laibon_general_event }
	}

	immediate = {
		add_character_flag = {
			flag = had_pod_laibon_general_event
			years = 10
		}
		# save the secret beforehand to lock in RNG
		random_neighboring_top_liege_realm_owner = {
			limit = {  
				any_secret = { 
					NOT = { is_known_by = root } 
				}
			}
			random_secret = {
				limit = { NOT = { is_known_by = root } }
				save_scope_as = secret
			}
		}
	}

	# Option A - Listen and follow (Learning check)
	option = {
		name = pod_laibon.101.a
		duel = {
			skill = learning
			value = medium_skill_rating
			target = root
			50 = {
				desc = pod_laibon.101.a.success
				compare_modifier = {
					value = scope:duel_value
					multiplier = high_positive_duel_skill_multiplier
				}
				# preview that doesn't reveal the secret beforehand
				show_as_tooltip = {
					POD_laibon_add_aye_and_orun_xp = {
						AYE  = 0
						ORUN = 5
					}
					custom_tooltip = pod_laibon.101.a.success
				}
				# actual effect
				hidden_effect = {
					send_interface_toast = {
						type = event_toast_effect_good
						title = pod_laibon.101.a.success
						POD_laibon_add_aye_and_orun_xp = {
							AYE  = 0
							ORUN = 5
						}
						# only reveal the secret if one was found
						scope:secret ?= {
							reveal_to = root
						}
					}
				}
			}
			50 = {
				desc = pod_laibon.101.a.failure
				compare_modifier = {
					value = scope:duel_value
					multiplier = low_negative_duel_skill_multiplier
				}
				send_interface_toast = {
					type = event_toast_effect_bad
					title = pod_laibon.101.a.failure
					POD_laibon_add_aye_and_orun_xp = {
						AYE  = 0
						ORUN = 2
					}
					add_stress = 15
				}
			}
		}
	}

	# Option B - Acknowledge but remain
	option = {
		name = pod_laibon.101.b
		POD_laibon_add_aye_and_orun_xp = {
			AYE  = 0
			ORUN = 2
		}
	}

	# Option C - Shut it out
	option = {
		name = pod_laibon.101.c
		POD_laibon_add_aye_and_orun_xp = {
			AYE  = 1
			ORUN = 0
		}
		stress_and_fulfillment_impact = {
			base = -15
		}
	}
}
