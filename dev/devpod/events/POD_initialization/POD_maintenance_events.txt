#Incompatible landholders are ousted when win a war 
POD_maintenance.6 = {
	hidden = yes
	type = character_event

	trigger = {		
		}

	immediate =	
	{
	every_sub_realm_county = {
		limit = {
				holder = {
					POD_custom_incompatible_splats_trigger = { ACTOR = root RECIPIENT = this }
						is_ai = yes	
					}
				}
			
			save_scope_as = target_title
			add_to_list = target_title_counties
			}
		
		create_title_and_vassal_change = {
			type = revoked
			save_scope_as = change
			add_claim_on_loss = yes
		}	
		
		every_in_list = {
			list = target_title_counties
			limit = {
				holder = {
					target_is_liege_or_above = root
				}
			}
			change_title_holder = {
				holder = root
				change = scope:change
			}
		}
		
		resolve_title_and_vassal_change = scope:change		
	}
}

#Vampire realms get feudal elective if not already active
POD_maintenance.7 = {
	hidden = yes
	type = character_event

	trigger = {	
		is_landed = yes # No merc
		highest_held_title_tier >= tier_duchy
		is_vampire_trigger = yes
		}

	immediate =	
	{

		every_held_title = {
			limit = {
			OR = {
				tier = tier_empire
				tier = tier_duchy
				tier = tier_kingdom
				}
				NOT = {
					has_title_law = feudal_elective_succession_law
				}
			}
			add_title_law = feudal_elective_succession_law
		}
	}
}

#Eco aid
POD_maintenance.16 = {
	hidden = yes
	type = character_event

	trigger = {
		is_landed = yes # No merc
		# highest_held_title_tier >= tier_county# Already called by "playable" (count+) on_action
		gold < 0
		OR = {
			has_trait = vampire
			has_trait = shapeshifter
		}
		is_at_war = no
		is_ai = yes
	}

	immediate =	{
		if = {
			limit = { years_from_game_start < 50 }
			add_gold = { 
				value = root.gold
				multiply = -0.5
			}
		}
		else = {
			add_gold = { 
				value = root.gold
				multiply = -1
			}
		}
	}
}
