#AI vampires lose stress traits, wound traits (useful for unlanded chars) and get a hunger fix
#Also stress fix (since no feasts or hunts anymore)
#Also prestige fix (since the AI needs to build rabble and thugs for levies, but that costs prestige)
#Also money fix (since the AI always rises all its levies while at war, which is very costly and not efficient in PoD)
##This event is now officially the 'Fix all the AI issues in one ugly package of bandaids' event
POD_maintenance.5 = {
	hidden = yes
	type = character_event

	trigger = {
		OR = {
			is_vampire_trigger = yes
			POD_is_bloodgorged_trigger = yes
			}
			is_ai = yes			
		}

	immediate =	
	{	
		if = {
			limit = {
				is_landed = yes
				highest_held_title_tier >= tier_county
				is_at_war = no
				years_from_game_start < 50
				gold < 500
				}
		if = {
			limit = {
				gold < 0
				gold > -500
				}
			add_gold = 250
		}
		
		else_if = {
			limit = {
				gold < -500
				gold > -1000
				}
			add_gold = 750
		}
		
		else = {
			add_gold = 75
		}
		}

		else_if = {
			limit = {
				is_landed = yes
				highest_held_title_tier >= tier_county
				is_at_war = no
				years_from_game_start < 200	
				gold < 500				
				}
		if = {
			limit = {
				gold < 0
				gold > -500
				}
			add_gold = 500
		}
		
		else_if = {
			limit = {
				gold < -500
				gold > -1000
				}
			add_gold = 750
		}
		
		else = {
			add_gold = 150
		}
		}
		
		else = {
			limit = {
				is_landed = yes
				highest_held_title_tier >= tier_county
				is_at_war = no
				gold < 500
				}
		if = {
			limit = {
				gold < 0
				gold > -500
				}
			add_gold = 500
		}
		
		else_if = {
			limit = {
				gold < -500
				gold > -1000
				}
			add_gold = 750
		}
		
		else = {
			add_gold = 250
		}
		}
		
		if = {
			limit = {
				NOT = { has_character_modifier = prison_of_the_mind }
			}
			remove_stress_trait_effect = yes
		}
		
		if = {
			limit = {
				is_landed = yes
				prestige < 0
			}
			add_prestige = 500
		}
		
		if = {
			limit = {
				has_character_modifier = bloodhunger1_modifier
			}
			remove_character_modifier = bloodhunger1_modifier
			add_character_modifier = {
				modifier = bloodhunger0_modifier
				}
			}
		else_if = {
			limit = {
				has_character_modifier = bloodhunger2_modifier
			}
			remove_character_modifier = bloodhunger2_modifier
			add_character_modifier = {
				modifier = bloodhunger1_modifier
				}
			}	
		else_if = {
			limit = {
				has_character_modifier = bloodhunger3_modifier
			}
			remove_character_modifier = bloodhunger3_modifier
			add_character_modifier = {
				modifier = bloodhunger1_modifier
				}
			}	
		else_if = {
			limit = {
				has_character_modifier = bloodhunger4_modifier
			}
			remove_character_modifier = bloodhunger4_modifier
			add_character_modifier = {
				modifier = bloodhunger1_modifier
				}
			}
		else_if = {
			limit = {
				has_character_modifier = bloodhunger5_modifier
			}
			remove_character_modifier = bloodhunger5_modifier
			add_character_modifier = {
				modifier = bloodhunger1_modifier
				}
			}					
	}
}


#Vampire and WW landholders are ousted when a hunter win a war 
POD_maintenance.6 = {
	hidden = yes
	type = character_event

	trigger = {		
		}

	immediate =	
	{
	every_sub_realm_county = {
		limit = {
				holder = {
					OR = {
						is_vampire_trigger = yes
						has_trait = ghoul
						has_trait = revenant
						has_trait = bloodgorged
						POD_is_werewolf_trigger = yes
						has_demon_trait_trigger = yes
						POD_is_wraith_andco_trigger = yes					
						}
						is_ai = yes	
					}
				}
			
			save_scope_as = target_title
			add_to_list = target_title_counties
			}
		
		create_title_and_vassal_change = {
			type = revoked
			save_scope_as = change
			add_claim_on_loss = yes
		}	
		
		every_in_list = {
			list = target_title_counties
			limit = {
				holder = {
					target_is_liege_or_above = root
				}
			}
			change_title_holder = {
				holder = root
				change = scope:change
			}
		}
		
		resolve_title_and_vassal_change = scope:change		
	}
}


#Vampire realms get feudal elective if not already active

	hidden = yes
	type = character_event

	trigger = {	
		is_landed = yes
		highest_held_title_tier >= tier_duchy
		is_vampire_trigger = yes
		}

	immediate =	
	{

		every_held_title = {
			limit = {
			OR = {
				tier = tier_empire
				tier = tier_duchy
				tier = tier_kingdom
				}
				NOT = {
					has_title_law = feudal_elective_succession_law
				}
			}
			add_title_law = feudal_elective_succession_law
		}
	}
}
