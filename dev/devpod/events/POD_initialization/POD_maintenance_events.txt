#Incompatible landholders are ousted when win a war 
POD_maintenance.6 = {
	hidden = yes

	trigger = {
		scope:vassal = { # Fix for inquisition attack
			days_as_ruler > 1 
			NOT = { is_at_war_with = root }
		}
		POD_custom_incompatible_splats_trigger = { ACTOR = root RECIPIENT = scope:vassal }
	}

	immediate = {
		scope:vassal = {
			every_held_title = {
				save_scope_as = target_title
				if = {
					limit = { tier <= tier_county }
					if = {
						limit = { is_capital_barony = no } # we'll use the county in that case
						add_to_list = target_title_counties
					}
					else = {
						county = { add_to_list = target_title_counties }
					}
				}
				else = {
					root = { destroy_title = prev }
				}
			}
			
			create_title_and_vassal_change = {
				type = revoked
				save_scope_as = change
				add_claim_on_loss = yes
			}
			
			every_in_list = {
				list = target_title_counties
				change_title_holder = {
					holder = root
					change = scope:change
				}
			}
			
			resolve_title_and_vassal_change = scope:change		
		}
	}
}

# Incompatible building clean
POD_maintenance.9 = {
	hidden = yes

	trigger = {
	}

	immediate =	{	
		if = {
			limit = {
				is_landed = yes
				has_trait = mortal
				can_hold_title_as_mortal_trigger = no
			}
			if = {
				limit = {
					NOT = { exists = father } 
					NOT = { exists = mother } 
				}
				remove_trait = mortal
			}
			else = {
				create_character = {
					save_temporary_scope_as = random_usurper
					location = root.capital_province
					culture = root.culture
					faith = root.faith
					gender = root
					age = { 18 50 }
					random_traits = yes
					after_creation = {
						POD_character_initialization_for_generated_baron_effect = yes
					}
				}
				create_title_and_vassal_change = {
					type = usurped
					save_scope_as = title_change
					add_claim_on_loss = no
				}
				scope:title = {
					change_title_holder = {
						holder = scope:random_usurper
						change = scope:title_change
					}
				}
				resolve_title_and_vassal_change = scope:title_change
			}
		}
		POD_character_initialization_for_generated_baron_effect = yes
		if = {
			limit = { NOT = { exists = scope:random_usurper } }
			scope:title = { 
				if = {
					limit = { tier = tier_county }
					title_province = {
						POD_remove_incompatible_building_effect = yes
						POD_replace_special_building_effect = yes
					}
					every_county_province = {
						limit = { 
							is_county_capital = no 
							barony.holder ?= root
						}
						POD_remove_incompatible_building_effect = yes
					}
				}
				else_if = {
					limit = { tier = tier_barony }
					title_province = { POD_remove_incompatible_building_effect = yes }
				}
			}
		}
	}
}

# Building aid
POD_maintenance.12 = {
	hidden = yes
	
	trigger = {
		is_landed = yes # No merc
		is_ai = yes
	}
		
	immediate = {
		every_held_county = {
			if = {
				limit = {
					duchy.title_capital_county = this 
					title_province = { 
						NOT = { has_building_with_flag = fully_upgraded_duchy_capital_building }
						has_ongoing_construction = no
						NOT = { barony = title:b_constantinople }
						NOT = { barony = title:b_roma }
					}
				}
				title_province = { 
					if = {
						limit = { years_from_game_start < 50 }
						POD_add_ducal_building_capped_effect = { MAX = 1 }
					}
					else_if = {
						limit = { years_from_game_start < 100 }
						POD_add_ducal_building_capped_effect = { MAX = 2 }
					}
					else_if = {
						limit = { years_from_game_start < 150 }
						POD_add_ducal_building_capped_effect = { MAX = 3 }
					}
					else = {
						POD_add_ducal_building_effect = yes
					}
				}
			}
			every_county_province = {
				limit = {
					trigger_if = {
						limit = { years_from_game_start < 50 }
						num_buildings <= 1 # Actually 0
					}
					trigger_else_if = {
						limit = { years_from_game_start < 100 }
						num_buildings <= 2 # Actually 1
					}
					trigger_else_if = {
						limit = { years_from_game_start < 150 }
						num_buildings <= 3 # Actually 2
					}
					trigger_else_if = {
						limit = { years_from_game_start < 200 }
						num_buildings <= 4 # Actually 3
					}
					trigger_else = {
						free_building_slots > 0
					}
				}
				generate_building = yes
			}
		}
	}
}
