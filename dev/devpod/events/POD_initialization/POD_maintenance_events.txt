#AI vampires lose get a hunger fix
#Also prestige fix (since the AI needs to build rabble and thugs for levies, but that costs prestige)
##This event is now officially the 'Fix all the AI issues in one ugly package of bandaids' event
POD_maintenance.5 = {
	hidden = yes
	type = character_event

	trigger = {
		OR = {
			is_vampire_trigger = yes
			POD_is_bloodgorged_trigger = yes
			}
			is_ai = yes			
		}

	immediate =	
	{	
		if = {
			limit = { prestige < 0 }
			add_prestige = { 
				value = root.prestige
				multiply = -1
			}
		}
		
		if = {
			limit = {
				has_character_modifier = bloodhunger1_modifier
			}
			remove_character_modifier = bloodhunger1_modifier
			add_character_modifier = {
				modifier = bloodhunger0_modifier
				}
			}
		else_if = {
			limit = {
				has_character_modifier = bloodhunger2_modifier
			}
			remove_character_modifier = bloodhunger2_modifier
			add_character_modifier = {
				modifier = bloodhunger1_modifier
				}
			}	
		else_if = {
			limit = {
				has_character_modifier = bloodhunger3_modifier
			}
			remove_character_modifier = bloodhunger3_modifier
			add_character_modifier = {
				modifier = bloodhunger1_modifier
				}

		else_if = {

#Incompatible landholders are ousted when win a war 
POD_maintenance.6 = {
	hidden = yes
	type = character_event

	trigger = {		
		}

	immediate =	
	{
	every_sub_realm_county = {
		limit = {
				holder = {
					POD_custom_incompatible_splats_trigger = { ACTOR = root RECIPIENT = this }
						is_ai = yes	
					}
				}
			
			save_scope_as = target_title
			add_to_list = target_title_counties
			}
		
		create_title_and_vassal_change = {
			type = revoked
			save_scope_as = change
			add_claim_on_loss = yes
		}	
		
		every_in_list = {
			list = target_title_counties
			limit = {
				holder = {
					target_is_liege_or_above = root
				}
			}
			change_title_holder = {
				holder = root
				change = scope:change
			}
		}
		
		resolve_title_and_vassal_change = scope:change		
	}
}

		}
	}
}

# Incompatible building clean
POD_maintenance.9 = {
	hidden = yes
	type = character_event

	trigger = {
		is_ai = yes
	}

	immediate =	{	 
		every_held_title = {
			limit = { tier = 2 }
			every_county_province = {
				if = {
					limit = { 
						OR = { 
							trigger_if = {
								limit = { has_holding_type = tribal_holding }
								barony_controller = { NOT = { government_has_flag = government_is_tribal } }
							}
							trigger_else = { barony_controller = { government_has_flag = government_is_tribal } }
							POD_has_wrong_type_of_building_trigger = yes 
						}
					}
					set_local_variable = {
						name = pod_building_level
						value = combined_building_level
						days = 1
					}
					POD_remove_common_building_effect = yes
					while = {
						limit = {
							local_var:pod_building_level > 1
						}
						generate_building = yes
						change_local_variable = {
							name = pod_building_level
							subtract = 1
						}
					}
					remove_local_variable = pod_building_level
				}
				if = {
					limit = { is_county_capital = yes }
					POD_remove_incompatible_ducal_building_effect = yes 
				}
			}
		}
	}
}

# Building aid
POD_maintenance.12 = {
	hidden = yes
	type = character_event
	
	trigger = {
		is_landed = yes # No merc
		is_ai = yes
	}
		
	immediate = {
		every_held_title = {
			limit = { tier = 2 }
			every_county_province = {
				limit = {
					trigger_if = {
						limit = { years_from_game_start < 50 }
						num_buildings <= 1 # Actually 0
					}
					trigger_else_if = {
						limit = { years_from_game_start < 100 }
						num_buildings <= 2 # Actually 1
					}
					trigger_else_if = {
						limit = { years_from_game_start < 150 }
						num_buildings <= 3 # Actually 2
					}
					trigger_else_if = {
						limit = { years_from_game_start < 200 }
						num_buildings <= 4 # Actually 3
					}
					trigger_else_if = {
						limit = { years_from_game_start < 250 }
						num_buildings <= 5 # Actually 4
					}
					trigger_else = {
						free_building_slots > 0
					}
				}
				generate_building = yes
			}
		}
	}
}
