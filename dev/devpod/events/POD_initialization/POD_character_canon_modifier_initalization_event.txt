namespace = POD_character_canon_modifier_initalization

POD_character_canon_modifier_initalization.1 = {
	hidden = yes
	scope = none
	
	immediate = {	
		every_living_character = {
			trigger_event = POD_character_canon_modifier_initalization.10
		}
	}
}

POD_character_canon_modifier_initalization.10 = {
	type = character_event
	hidden = yes
	
	immediate = {
		if = {
			limit = {
				has_trait = vampire
			}
			trigger_event = POD_character_canon_modifier_initalization.100
		}
		else_if = {
			limit = {
				has_trait = kueijin
			}
			trigger_event = POD_character_canon_modifier_initalization.200
		}
		else_if = {
			limit = {
				has_trait = supehunter
			}
			if = {
				limit = { has_trait = shih }
				trigger_event = POD_character_canon_modifier_initalization.210
			}
			else_if = {
				limit = { NOT = { religion = { is_in_family = rf_hunters } } }
				POD_character_hunter_faith_initialization_effect = yes
			}
		}
		else_if = {
			limit = {
				has_trait = mummy
			}
			trigger_event = POD_character_canon_modifier_initalization.300
		}
		else_if = {
			limit = {
				has_trait = wraith
			}
			trigger_event = POD_character_canon_modifier_initalization.400
		}
		else_if = {
			limit = {
				has_trait = shapeshifter
			}
			trigger_event = POD_character_canon_modifier_initalization.500
		}
		else_if = {
			limit = {
				has_trait = demon
			}
			trigger_event = POD_character_canon_modifier_initalization.700
		}
		else_if = {
			limit = {
				has_trait = mage
			}
			trigger_event = POD_character_canon_modifier_initalization.800
		}
		else_if = {
			limit = {
				has_trait = fae 
			}
			trigger_event = POD_character_canon_modifier_initalization.900
		}
		else_if = {
			limit = {
				has_trait = changeling
			}
			trigger_event = POD_character_canon_modifier_initalization.910
		}
		else_if = {
			limit = {
				POD_is_any_kind_of_ghoul_trigger = yes
			}
			trigger_event = POD_character_canon_modifier_initalization.110
		}
		else_if = {
			limit = {
				POD_is_kinfolk_trigger = yes
			}
			trigger_event = POD_character_canon_modifier_initalization.600
		}
		else = {
			trigger_event = POD_character_initialization.1
		}
	
		#initialize lifestyle perks
		if = {
			limit = {
				NOT = { has_character_flag = had_POD_character_lifestyle }
			}
			POD_character_vampire_lifestyle_initialization_effect = yes
		}
	}
}

POD_character_canon_modifier_initalization.100 = {
	type = character_event
	hidden = yes
	
	trigger = {
		POD_is_vampire_trigger = yes
	}
	
	immediate = {
		if = {
			limit = {
				has_blood_potency_trigger = no
			}
			POD_character_vampire_bloodpotency_initialization_effect = yes	
		}
		
		if = {
			limit = {
				has_masquerade_risk_trigger = no
			}
			add_character_modifier = masquerade0_modifier
		}
		
		if = {
			limit = {
				has_blood_hunger_trigger = no
			}
			add_character_modifier = bloodhunger1_modifier		
		}

		if = {
			limit = { is_ai = no }
			if = {
				limit = { 
					has_trait = bloodsorceryadvanced
					NOT = { has_character_flag = got_BS_exp} 
				}
				set_variable = { name = blood_sorcery_points value = 600 }
				add_character_flag = got_BS_exp
			}
			else_if = {
				limit = { 
					has_trait = bloodsorcery
					NOT = { has_character_flag = got_BS_exp} 
				}
				set_variable = { name = blood_sorcery_points value = 100 }
				add_character_flag = got_BS_exp
			}
			if = {
				limit = {
					has_predator_type_trigger = no
				}
				if = {
					limit = {has_trait = ventrue}
					trigger_event = { 
						id = POD_chargen.1100 
						days = 5
					}
				}
				else = {
					trigger_event = { 
						id = POD_chargen.1020 
						days = 5
					}
				}
			}
			if = {
				limit = {
					has_POD_vampire_clan_trigger = no
				}
				POD_character_vampire_clan_initialization_effect = yes
			}
			if = {
				limit = {
					has_POD_vampire_generation_trigger = no
				}
				add_trait = vampgeneration12
			}
		}
		
		POD_character_vampire_clan_special_effect = yes
		
		#not allowing torpored canon characters to wake up
		if = {
			limit = { has_trait = torpor }
			POD_torpor_longsleep_effect = yes
		}
		
		#cooldown on bloodpotency increase/decrease
		POD_bloodpotency_cooldown_effect = yes
	}
}

POD_character_canon_modifier_initalization.110 = {
	type = character_event
	hidden = yes
	
	trigger = {
		POD_is_any_kind_of_ghoul_trigger = yes
	}
	
	immediate = {
		if = {
			limit = {
				has_blood_potency_trigger = no
			}
			add_character_modifier = bloodpotency0_modifier 
		}
		
		if = {
			limit = {
				has_masquerade_risk_trigger = no
				is_forsaken_or_gantenbein_trigger = no
				POD_splat_is_shifter_trigger = no
			}
			add_character_modifier = masquerade0_modifier
		}
		
		if = {
			limit = {
				has_blood_hunger_trigger = no
			}
			add_character_modifier = bloodhunger0_modifier		
		}

		if = {
			limit = {
				is_ai = no
				has_trait = bloodsorcery
				NOT = { has_character_flag = got_BS_exp }
			}
			set_variable = { name = blood_sorcery_points value = 100}
			add_character_flag = got_BS_exp
			
		}
		
		if = {
			limit = {
				is_ai = no
				has_trait = bloodsorceryadvanced
				NOT = { has_character_flag = got_BS_exp}
			}
			set_variable = { name = blood_sorcery_points value = 600}
			add_character_flag = got_BS_exp
		}
		
		POD_character_vampire_infernalist_initialization_effect = yes
	}
}
