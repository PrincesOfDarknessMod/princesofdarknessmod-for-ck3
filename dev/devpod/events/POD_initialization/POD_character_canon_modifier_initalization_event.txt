namespace = POD_character_canon_modifier_initalization

POD_character_canon_modifier_initalization.1 = {
	hidden = yes
	type = empty
	
	trigger = {
	}
	
	immediate = {	
		every_living_character = {
			if = {
				limit = {
					has_trait = vampire
				}
				trigger_event = POD_character_canon_modifier_initalization.100
			}
			else_if = {
				limit = {
					OR = {
						has_trait = kueijin
						has_trait = shih
						has_trait = dhampyr
					}
				}
				trigger_event = POD_character_canon_modifier_initalization.200
			}
			else_if = {
				limit = {
					has_trait = mummy
				}
				trigger_event = POD_character_canon_modifier_initalization.300
			}
			else_if = {
				limit = {
					has_trait = wraith
				}
				trigger_event = POD_character_canon_modifier_initalization.400
			}
			else_if = {
				limit = {
					POD_is_werewolf_trigger = yes
				}
				trigger_event = POD_character_canon_modifier_initalization.500
			}
			else_if = {
				limit = {
					has_trait = demon
				}
				trigger_event = POD_character_canon_modifier_initalization.700
			}
			else_if = {
				limit = {
					has_trait = mage
				}
				trigger_event = POD_character_canon_modifier_initalization.800
			}
			else_if = {
				limit = {
					OR = { 
						has_trait = fae 
						has_trait = changeling
					}
				}
				trigger_event = POD_character_canon_modifier_initalization.900
			}
			else_if = {
				limit = {
					has_POD_race_trait_trigger = no
				}
				trigger_event = POD_character_initialization.1
			}
			
			if = {
				limit = {
					POD_is_any_kind_of_ghoul_trigger = yes
				}
				trigger_event = POD_character_canon_modifier_initalization.100
			}
			if = {
				limit = {
					POD_is_kinfolk_trigger = yes
				}
				trigger_event = POD_character_canon_modifier_initalization.600
			}
			if = {
				limit = {
					OR = {
						has_trait = laham
						has_trait = nephilim
					}
				}
				trigger_event = POD_character_canon_modifier_initalization.700
			}
			if = {
				limit = {
					has_trait = kinain
				}
				trigger_event = POD_character_canon_modifier_initalization.900
			}
			
			if = {
				limit = {
					is_from_ruler_designer = yes
				}
				# Add missing staff (by random) and change wrong staff
				POD_remove_wrong_supernatural_traits_effect = yes
				POD_add_missing_supernatural_traits_effect = yes
				POD_adopt_correct_religion_effect = yes
				POD_adopt_correct_goverment_effect = yes
			}
		}
	}
}

POD_character_canon_modifier_initalization.100 = {
	type = character_event
	hidden = yes
	
	trigger = {
		has_POD_vamp_blood_trigger = yes
	}
	
	immediate = {
	
		#POD_clear_resonance_effect = yes
		POD_set_resonance_effect = yes
		
		if = {
			limit = {
				has_blood_potency_trigger = no
			}
			
			if = { 
				limit = {
					POD_is_vampire_trigger = yes 
				}
				POD_character_vampire_bloodpotency_initialization_effect = yes
			}
			if = { 
				limit = {
					OR = {
						POD_is_revenant_trigger = yes
						POD_is_ghoul_trigger = yes
						POD_is_bloodgorged_trigger = yes
					}
				}
				add_character_modifier = {
					modifier = bloodpotency0_modifier 
				}
			}		
		}
		
		if = {
			limit = {
				has_masquerade_risk_trigger = no
				is_forsaken_or_gantenbein_trigger = no
				POD_splat_is_shifter_trigger = no
			}
			add_character_modifier = {
				modifier = masquerade0_modifier 
			}	
			#set_variable = { name = masquerade_level value = 5 }
		}
		
		if = {
			limit = {
				has_blood_hunger_trigger = no
			}
			if = {
				limit = {
					POD_is_vampire_trigger = yes
				}
				add_character_modifier = {
					modifier = bloodhunger1_modifier
				}
				#set_variable = { name = blood_level value = 4 }
			}
			if = {
				limit = {
					OR = {
						POD_is_revenant_trigger = yes
						POD_is_ghoul_trigger = yes
						POD_is_bloodgorged_trigger = yes
					}
				}
				add_character_modifier = {
					modifier = bloodhunger0_modifier
					
				}
				#set_variable = { name = blood_level value = 5 }
			}					
		}
		
		if = {
			limit = {
				POD_is_vampire_trigger = yes
				has_predator_type_trigger = no
				is_ai = yes 
			}
			POD_character_vampire_predatortype_initialization_effect = yes
		}

		if = {
			limit = {
				is_ventrue_trigger = yes 
				is_ai = yes
				has_prey_exclusion_trigger = no
			}
			POD_character_ventrue_prey_exclusion_initialization_effect = yes
		}

		if = {
			limit = {
				POD_is_vampire_trigger = yes
				has_predator_type_trigger = no
				is_ai = no
			}
			if = {
				limit = {has_trait = ventrue}
				trigger_event = { 
					id = POD_chargen.1100 
					days = 5
				}
			}
			else = {
				trigger_event = { 
					id = POD_chargen.1020 
					days = 5
				}
			}
		}		
		
		if = {
			limit = { has_trait = torpor }
			POD_torpor_longsleep_effect = yes
		}

		#cooldown on bloodpotency increase/decrease
		if = {
			limit = { is_vampire_trigger = yes }
			POD_bloodpotency_cooldown_effect = yes
		}

		if = {
			limit = {
				is_ai = no
				has_trait = bloodsorcery
				NOT = { has_character_flag = got_BS_exp }
			}
			set_variable = { name = blood_sorcery_points value = 100}
			add_character_flag = got_BS_exp
			
		}
		
		if = {
			limit = {
				is_ai = no
				has_trait = bloodsorceryadvanced
				NOT = { has_character_flag = got_BS_exp}
			}
			set_variable = { name = blood_sorcery_points value = 600}
			add_character_flag = got_BS_exp
			
		}		
		
		if = {
			limit = {
				is_nosferatu_trigger = yes 
				NOT = { has_character_flag = had_POD_character_nosferatu_looks }
			}
			POD_character_vampire_nosferatu_initialization_effect = yes
		}
		
		#initialize lifestyle perks
		if = {
			limit = {
				NOT = { has_character_flag = had_POD_character_lifestyle }
			}
			POD_character_vampire_lifestyle_initialization_effect = yes
		}
		
		#not allowing torpored canon characters to wake up
		if = {
			limit = { has_trait = torpor }
			POD_torpor_longsleep_effect = yes
		}
		
		#cooldown on bloodpotency increase/decrease
		if = {
			limit = { is_vampire_trigger = yes }
			POD_bloodpotency_cooldown_effect = yes
		}
	}
}

POD_character_canon_modifier_initalization.200 = {
	type = character_event
	hidden = yes
	
	trigger = {
		OR = {
			has_trait = kueijin
			has_trait = shih
			has_trait = dhampyr
		}
	}
	
	immediate = {
		if = {
			limit = { is_kueijin_trigger = yes }
			POD_kueijin_initialization_effect = yes
		}
		if = {
			limit = { has_trait = dhampyr }
			POD_dhampyr_initialization_effect = yes
		}
		if = {
			limit = { has_trait = shih }
			POD_shih_initialization_effect = yes
		}
		
		#initialize lifestyle perks
		if = {
			limit = {
				NOT = { has_character_flag = had_POD_character_lifestyle }
			}
			POD_character_vampire_lifestyle_initialization_effect = yes
		}
		if = {
			limit = {
				NOT = { has_character_flag = had_POD_character_lifestyle_random }
			}
			POD_character_random_lifestyle_initialization_effect = yes
		}
	}
}
