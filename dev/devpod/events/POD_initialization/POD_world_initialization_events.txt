POD_world_initialization.10 = {
	hidden = yes
	scope = none
	
	trigger = {
		NOT = { has_global_variable = variable_POD_world_county_faith_initialization }
	}
	
	immediate = {
		POD_county_faith_initialization_effect = yes
		
		set_global_variable = {
			name = variable_POD_world_county_faith_initialization
			value = yes
		}
			
		#an effect to automatically set county faith to their holder
		#why? mostly because assigning county faith in history files is a huge pain that only works when it feels like it		
		#Could be removed if we go other all the new provinces to manually change county religion but until then... it works 
		
		hidden_effect = {
			every_county_in_region = {
				region = world_asia
				
				limit = {
					NOT = { county.faith = county.holder.faith }
				}
				set_county_faith = county.holder.faith
			}
		}
	}	
}

#Kill characters of unwanted splats depending on game start
#Happens before lobby
#Otherwise, all those characters start as councilors of existing characters, spamming the player will "x has been killed" on game start
POD_world_initialization.3 = {
	hidden = yes
	scope = none
	
	immediate = {
		#WW and hunters and co are killed in vamps startdate
		if = {
			limit = {
				POD_bookmark_is_vampire_only = yes
			}
			hidden_effect = {
				every_living_character = {
					limit = {
						POD_is_any_kind_of_ghoul_trigger = yes
						is_ai = yes
						POD_is_not_apocalyptic_background_character = yes
					}
					if = {
						limit = { has_trait = kinfolk }
						remove_trait = kinfolk
					}
					if = {
						limit = { has_trait = supehunter }
						remove_trait = supehunter
					}
				}
				every_living_character = {
					limit = {
						has_trait = demon
						is_ai = yes
						POD_is_not_apocalyptic_background_character = yes
					}
					POD_send_to_abyss_effect = yes
					character:POD_wraith_01 = {
						add_courtier = prev
					}
					return_to_court = yes
				}
				every_living_character = {
					limit = {
						OR = {
							has_trait = supehunter
							has_trait = fera
							has_trait = mummy
							has_trait = wraith
							has_trait = fae
							has_trait = changeling
						}
						is_ai = yes
						POD_is_not_apocalyptic_background_character = yes
					}
					death = { death_reason = death_disappearance }
				}
			}	
		}
		#Fae are killed in hunters startdate
		if = {
			limit = {
				POD_bookmark_has_fae = no
			}
			hidden_effect = {
				every_living_character = {
					limit = {
						OR = {
							has_trait = fae
							has_trait = changeling
						}
						is_ai = yes
					}
					death = { death_reason = death_disappearance }
				}
			}	
		}
		if = {
			limit = {
				POD_is_expod_unleashed = no
			}
			hidden_effect = {
				every_living_character = {
					limit = {
						POD_is_exalted_trigger = yes
						POD_is_terrestrial_exalted_trigger = no
						is_ai = yes
					}
					death = { death_reason = death_disappearance }
				}
			}	
		}
		if = {
			limit = {
				POD_is_expod_enabled = no
			}
			hidden_effect = {
				every_living_character = {
					limit = {
						POD_is_exalted_trigger = yes
						is_ai = yes
					}
					death = { death_reason = death_disappearance }
				}
			}	
		}
	}
}
