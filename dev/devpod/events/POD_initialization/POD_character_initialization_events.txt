	trigger = {
		}
	
	immediate = {
		every_living_character = {
			trigger_event = POD_character_initialization.1
		}
	}	
}

POD_character_initialization.1 = {
	hidden = yes
	
	trigger = {
		is_ai = yes 
		NOT = { has_character_flag = had_POD_character_initialization }
		#exists = capital_province
		#is_adult = yes
		#exists = location		
		#capital_province = { has_supernatural_religion_trigger = yes }
		#is_landed = yes
	}
	
	immediate = {
	
	#POD_clear_resonance_effect = yes
	POD_set_resonance_effect = yes
		
		if = {
			limit = {
				has_POD_race_trigger = yes
			}
			add_character_flag = had_POD_character_initialization
		}
		
		else_if = {
			limit = {
				is_ruler = yes
				is_landed = no
			}
			POD_character_initialization_for_merc_commanders_effect = yes
		}
		
		else_if = {
			limit = {
				OR = {
					primary_title.tier = tier_county
					primary_title.tier = tier_duchy
					primary_title.tier = tier_kingdom
					primary_title.tier = tier_empire
				}
			}
			POD_character_initialization_for_nobles_effect = yes
		}
		
		else_if = {
			limit = {
				primary_title.tier = tier_barony	
			}
			POD_character_initialization_for_barons_effect = yes				
			POD_character_new_dynasty_effect = yes
			
			scope:title ?= { title_province = { POD_remove_incompatible_building_effect = yes } }
		}
		
		else_if = {
			limit = {
				is_a_faction_leader = yes
			}
			POD_character_initialization_for_merc_commanders_effect = yes
		}
		
		else = {
			POD_character_initialization_for_commoners_effect = yes
		}
		
	}	
}

POD_character_initialization.100 = {
	hidden = yes
	scope = none
	
	trigger = {	
		title:k_shadowinquisition = { 
			exists = holder
			holder = { NOT = { has_trait = supehunter } }
		}
	}
	
	immediate = {
		title:k_shadowinquisition.holder = { 
		if = {
			limit = {
				NOT = { has_trait = supehunter }
			}
			if = {
				limit = {
					has_trait = mortal
				}
				remove_trait = mortal
			}
			add_trait = supehunter
		}
		POD_character_new_dynasty_effect = yes		
	}
	}
}

POD_character_initialization.1001 = {
	hidden = yes
	scope = none
	
	immediate = {
		# Extra-spicy easter eggs (based on Haight mode)
		if = {
			# TODO: split the patron/easteregg gamerule into two separate ones, with "Immersive/Silly easter eggs" options
			limit = { POD_is_samuel_haight_mode_enabled = no }
			
			if = {
				limit = { character:easteregg_flandre = { is_landed = no } }
				character:easteregg_flandre = { death = { death_reason = death_disappearance } }
			}
			if = {
				limit = { character:easteregg_remilia = { is_landed = no } }
				character:easteregg_remilia = { death = { death_reason = death_disappearance } }
			}
			if = {
				limit = { character:easteregg_roselia = { is_landed = no } }
				character:easteregg_roselia = { death = { death_reason = death_disappearance } }
			}
			if = {
				limit = { character:easteregg_shinobu_oshino = { is_landed = no } }
				character:easteregg_shinobu_oshino = { death = { death_reason = death_disappearance } }
			}
			if = {
				limit = { character:easteregg_marceline = { is_landed = no } }
				character:easteregg_marceline = { death = { death_reason = death_disappearance } }
			}
		}
		# Patrons and Easter Eggs (based on game rule)
		switch = {
			trigger = has_game_rule

			both_patron_and_easter_egg_on = {
			}
			patron_only = {
				every_living_character = {
					limit = {
						has_character_flag = easter_egg_character
					}
					death = { death_reason = death_disappearance }
				}
			}
			easter_egg_only = {
				every_living_character = {
					limit = {
						has_trait = patron
					}
					death = { death_reason = death_disappearance }
				}
			}
			canon_only = {
				every_living_character = {
					limit = {
						OR = {
							has_trait = patron
							has_character_flag = easter_egg_character
						}
					}
					death = { death_reason = death_disappearance }
				}
			}
		}
	}	
}
