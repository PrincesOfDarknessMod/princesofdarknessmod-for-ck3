POD_character_initialization.1 = {
	hidden = yes
	type = character_event
	
	trigger = {
		is_ai = yes 
		NOT = { has_character_flag = had_POD_character_initialization }
		#exists = capital_province
		#is_adult = yes
		#exists = location		
		#capital_province = { has_supernatural_religion_trigger = yes }
		#is_landed = yes
	}
	
	immediate = {
	
	#POD_clear_resonance_effect = yes
	POD_set_resonance_effect = yes
		
		if = {
			limit = {
				has_POD_race_trigger = yes
			}
			add_character_flag = had_POD_character_initialization
		}
		
		else_if = {
			limit = {
				is_ruler = yes
				OR = {
					has_government = holy_order_government
					has_government = mercenary_government
				}
			}
			POD_character_initialization_for_merc_commanders_effect = yes
		}
		
		else_if = {
			limit = {
				OR = {
					primary_title.tier = tier_county
					primary_title.tier = tier_duchy
					primary_title.tier = tier_kingdom
					primary_title.tier = tier_empire
				}
			}
			POD_character_initialization_for_nobles_effect = yes
		}
		
		else_if = {
			limit = {
				primary_title.tier = tier_barony	
			}
			POD_character_initialization_for_barons_effect = yes				
			POD_character_new_dynasty_effect = yes
		}
		
		else = {
			POD_character_initialization_for_commoners_effect = yes
		}
	}	
}

POD_character_initialization.100 = {
	hidden = yes
	type = empty
	
	trigger = {	
		title:k_shadowinquisition = { 
			exists = holder
			holder = { NOT = { has_trait = beatific } }
		}
	}
	
	immediate = {
		title:k_shadowinquisition.holder = { POD_character_create_shadow_cardinal_effect = yes }
	}
}

POD_character_initialization.1001 = {
	hidden = yes
	type = empty
	
	trigger = {
		OR = {
			has_game_rule = easter_egg_only
			has_game_rule = patron_only
			has_game_rule = canon_only
		}
	}
	
	immediate = {
		switch = {
			trigger = has_game_rule

			both_patron_and_easter_egg_on = {
			}
			patron_only = {
				every_living_character = {
					limit = {
						has_character_flag = easter_egg_character
					}
					death = { death_reason = death_disappearance }
				}
			}
			easter_egg_only = {
				every_living_character = {
					limit = {
						has_trait = patron
					}
					death = { death_reason = death_disappearance }
				}
			}
			canon_only = {
				every_living_character = {
					limit = {
						OR = {
							has_trait = patron
							has_character_flag = easter_egg_character
						}
					}
					death = { death_reason = death_disappearance }
				}
			}
		}
	}	
}
