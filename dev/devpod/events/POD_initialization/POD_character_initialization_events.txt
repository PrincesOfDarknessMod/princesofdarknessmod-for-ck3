POD_character_initialization.1 = {
	hidden = yes
	
	trigger = {
		is_ai = yes 
		NOT = { has_character_flag = had_POD_character_initialization }
		#exists = capital_province
		#is_adult = yes
		#exists = location		
		#is_landed = yes
	}
	
	immediate = {
	
	#POD_clear_resonance_effect = yes
	POD_set_resonance_effect = yes
		
		if = {
			limit = {
				has_POD_race_trigger = yes
			}
			add_character_flag = had_POD_character_initialization
		}
		
		else_if = {
			limit = {
				is_ruler = yes
				is_playable_character = no
			}
			POD_character_initialization_for_merc_commanders_effect = yes
		}
		
		else_if = {
			limit = {
				OR = {
					primary_title.tier >= tier_county
				}
			}
			POD_character_initialization_for_nobles_effect = yes
		}
		
		else_if = {
			limit = {
				primary_title.tier = tier_barony	
			}
			POD_character_initialization_for_barons_effect = yes				
			POD_character_new_dynasty_effect = yes
			
			scope:title ?= {
				if = {
					limit = { holder ?= prev }
					title_province = { POD_remove_incompatible_building_effect = yes }
				}
			}
		}
		
		else_if = {
			limit = {
				is_a_faction_leader = yes
			}
			POD_character_initialization_for_merc_commanders_effect = yes
		}
		
		else = {
			POD_character_initialization_for_commoners_effect = yes
		}
		
	}	
}

POD_character_initialization.100 = {
	hidden = yes
	scope = none
	
	trigger = {	
		title:k_shadowinquisition = { 
			exists = holder
			holder = { NOT = { has_trait = supehunter } }
		}
	}
	
	immediate = {
		title:k_shadowinquisition.holder = { 
		if = {
			limit = {
				NOT = { has_trait = supehunter }
			}
			if = {
				limit = {
					has_trait = mortal
				}
				remove_trait = mortal
			}
			add_trait = supehunter
		}
		POD_character_new_dynasty_effect = yes		
	}
	}
}

POD_character_initialization.1002 = {
	hidden = yes
	type = character_event
	
	immediate = {
		
		if = {
			limit = {
				is_ruler = yes
				primary_title.tier > tier_barony
				is_independent_ruler = no
			}
				liege ?= { save_temporary_scope_as = canon_inheritor }
					
				POD_transfer_all_titles_to_character = {
					CHARACTER = scope:canon_inheritor
				}
				death = { death_reason = death_disappearance }
			}	
		else_if = {
			limit = {
			primary_title.tier > tier_barony
			is_independent_ruler = yes
			}
			random_neighboring_and_across_water_realm_same_rank_owner = {
					limit = {
						NOT = { has_trait = patron }
						NOT = { has_character_flag = easter_egg_character }
					}
				save_temporary_scope_as = canon_inheritor
				}
			POD_transfer_all_titles_to_character = {
				CHARACTER = scope:canon_inheritor
			}
			death = { death_reason = death_disappearance }			
		}
		else = {
			death = { death_reason = death_disappearance }	
		}	
		
	}
	
}
