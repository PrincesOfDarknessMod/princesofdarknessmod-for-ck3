# pod_combat.0001 — Combat system overview help event.
# Fired by the "Help" button on the combat window. Shows a reference
# sheet covering the basic flow, damage types, and skill modifiers.
pod_combat.0001 = {
	type = character_event
	title = pod_combat.0001.t
	desc = pod_combat.0001.desc
	theme = war

	left_portrait = {
		character = root
		animation = personality_rational
	}

	option = {
		name = pod_combat.0001.a
	}
}

# Simple hidden follow-up for debug interactions. Just cleans up the
# engaged_in_single_combat flag and shows a toast with the result.

pod_combat_debug.0001 = {
	hidden = yes

	immediate = {
		# Clean shirtlessness if it was applied
		single_combat_clean_shirtlessness_effect = {
			ATTACKER = scope:sc_attacker
			DEFENDER = scope:sc_defender
		}

		if = {
			limit = { scope:sc_victor = root }
			send_interface_toast = {
				title = pod_combat_debug_victory_toast
				left_icon = root
				right_icon = scope:sc_loser
			}
		}
		else = {
			send_interface_toast = {
				title = pod_combat_debug_defeat_toast
				left_icon = root
				right_icon = scope:sc_victor
			}
		}
	}
}

# Post-combat dispatch event — fires immediately when an sc-driven combat
# ends (no delay). CK3 does NOT propagate saved scopes from GUI execute
# contexts through normal event chains, so this event does not rely on
# the scope chain: instead it reconstructs every scope it needs from
# character variables stashed on the PC during combat (sc_initiator,
# sc_attacker, sc_defender, fatality, fixed, follow_up, etc.). Applies
# wounds / death per the duel's fatality setting, fires the original
# PoD follow-up event, and then cleans up all combat variables.
#
# The old version used trigger_event { days = 1 } to escape the GUI
# context entirely via a fresh tick, but since all scopes are now
# reconstructed from stashed vars anyway, the delay is no longer needed.

pod_combat.9999 = {
	hidden = yes

	immediate = {
		# ── Reconstruct character scopes from stash ──
		var:pod_combat_sc_initiator = { save_scope_as = sc_initiator }
		var:pod_combat_sc_attacker  = { save_scope_as = sc_attacker }
		var:pod_combat_sc_defender  = { save_scope_as = sc_defender }

		# Defender aliases (covers victim, recipient, defender, target, adversary)
		var:pod_combat_sc_defender = { save_scope_as = victim }
		var:pod_combat_sc_defender = { save_scope_as = recipient }
		var:pod_combat_sc_defender = { save_scope_as = defender }
		var:pod_combat_sc_defender = { save_scope_as = target }
		var:pod_combat_sc_defender = { save_scope_as = adversary }

		# Attacker aliases (covers attacker, challenger)
		var:pod_combat_sc_attacker = { save_scope_as = attacker }
		var:pod_combat_sc_attacker = { save_scope_as = challenger }

		# Victor / loser from minigame result
		if = {
			limit = {
				has_variable = pod_combat_winner
				has_variable = pod_combat_loser
			}
			var:pod_combat_winner = { save_scope_as = sc_victor }
			var:pod_combat_loser  = { save_scope_as = sc_loser }
		}

		# ── Reconstruct fatality scope value ──
		if = {
			limit = { var:pod_combat_sc_fatality = flag:always }
			save_scope_value_as = { name = fatality value = flag:always }
		}
		else_if = {
			limit = { var:pod_combat_sc_fatality = flag:default }
			save_scope_value_as = { name = fatality value = flag:default }
		}
		else_if = {
			limit = { var:pod_combat_sc_fatality = flag:possible }
			save_scope_value_as = { name = fatality value = flag:possible }
		}
		else_if = {
			limit = { var:pod_combat_sc_fatality = flag:practice }
			save_scope_value_as = { name = fatality value = flag:practice }
		}
		else = {
			save_scope_value_as = { name = fatality value = flag:no }
		}

		# Fixed
		if = {
			limit = { var:pod_combat_sc_fixed = flag:sc_attacker }
			save_scope_value_as = { name = fixed value = flag:sc_attacker }
		}
		else_if = {
			limit = { var:pod_combat_sc_fixed = flag:sc_defender }
			save_scope_value_as = { name = fixed value = flag:sc_defender }
		}
		else = {
			save_scope_value_as = { name = fixed value = flag:no }
		}

		save_scope_value_as = { name = locale      value = flag:battlefield }
		save_scope_value_as = { name = sc_finished  value = yes }
		save_scope_value_as = { name = victory_type value = flag:skill }

		# ── Release engaged flags ──
		scope:sc_attacker = {
			if = {
				limit = { is_alive = yes }
				remove_variable = engaged_in_single_combat
			}
		}
		scope:sc_defender = {
			if = {
				limit = { is_alive = yes }
				remove_variable = engaged_in_single_combat
			}
		}

		# ── Apply wounds/death (inlined work_out_wounds_or_death_effect) ──
		if = {
			limit = { exists = scope:sc_victor }

			if = {
				limit = { scope:fatality = flag:practice }
			}
			else = {
				POD_fera_add_renown_effect = { TYPE = ferocity }
			}

			if = {
				limit = { scope:fatality = flag:no }
				scope:sc_loser = {
					increase_wounds_no_death_effect = { REASON = duel }
				}
			}
			else_if = {
				limit = { scope:fatality = flag:possible }
				scope:sc_loser = {
					POD_increase_wounds_hostile_effect = {
						REASON = duel_aggravated
						ATTACKER = scope:sc_victor
						COUNT = POD_base_wound_damage
					}
				}
			}
			else_if = {
				limit = { scope:fatality = flag:default }
				if = {
					limit = {
						scope:sc_attacker = {
							OR = {
								AND = {
									is_ruler = yes
									government_has_flag = government_is_tribal
								}
								AND = {
									is_ruler = no
									exists = court_owner
									scope:sc_attacker.court_owner = { government_has_flag = government_is_tribal }
								}
								AND = {
									is_ruler = no
									exists = host
									scope:sc_attacker.host = { government_has_flag = government_is_tribal }
								}
								AND = {
									is_ruler = no
									NOR = { exists = court_owner exists = host }
									scope:sc_defender = {
										OR = {
											AND = { is_ruler = yes government_has_flag = government_is_tribal }
											AND = { is_ruler = no exists = court_owner scope:sc_defender.court_owner = { government_has_flag = government_is_tribal } }
											AND = { is_ruler = no exists = host scope:sc_defender.host = { government_has_flag = government_is_tribal } }
										}
									}
								}
							}
						}
					}
					scope:sc_loser = {
						pod_death = { killer = scope:sc_victor death_reason = death_duel }
					}
					if = {
						limit = { scope:sc_victor = { faith = { has_doctrine = tenet_cranial_trophies } } }
						tgp_cranial_trophies_beheading_effect = { KILLER = scope:sc_victor DEAD = scope:sc_loser }
					}
				}
				else = {
					scope:sc_loser = {
						POD_increase_wounds_hostile_effect = {
							REASON = duel_aggravated
							ATTACKER = scope:sc_victor
							COUNT = POD_base_wound_damage
						}
					}
				}
			}
			else_if = {
				limit = { scope:fatality = flag:always }
				scope:sc_victor = {
					if = {
						limit = {
							this = scope:sc_attacker
							NOT = { has_execute_reason = scope:sc_loser }
						}
						add_kinslayer_trait_or_nothing_effect = { VICTIM = scope:sc_loser }
					}
				}
				scope:sc_loser = {
					pod_death = { killer = scope:sc_victor death_reason = death_duel }
				}
				if = {
					limit = { scope:sc_victor = { faith = { has_doctrine = tenet_cranial_trophies } } }
					tgp_cranial_trophies_beheading_effect = { KILLER = scope:sc_victor DEAD = scope:sc_loser }
				}
			}
		}

		# ── Fire the follow-up event ──
		# Fire on ROOT (= the PC, who is sc_initiator in all current callers).
		# No scope:X = { } wrapping — scopes saved at the event's top level
		# propagate without any scope-block interference.
		trigger_event = { saved_event_id = var:pod_combat_sc_follow_up }

		# ── Full cleanup ──
		pod_combat_cleanup_effect = yes
	}
}
