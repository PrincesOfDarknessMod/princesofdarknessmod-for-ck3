# Simple hidden follow-up for debug interactions. Just cleans up the
# engaged_in_single_combat flag and shows a toast with the result.

pod_combat_debug.0001 = {
	hidden = yes

	immediate = {
		# Clean shirtlessness if it was applied
		single_combat_clean_shirtlessness_effect = {
			ATTACKER = scope:sc_attacker
			DEFENDER = scope:sc_defender
		}

		if = {
			limit = { scope:sc_victor = root }
			send_interface_toast = {
				title = pod_combat_debug_victory_toast
				left_icon = root
				right_icon = scope:sc_loser
			}
		}
		else = {
			send_interface_toast = {
				title = pod_combat_debug_defeat_toast
				left_icon = root
				right_icon = scope:sc_victor
			}
		}
	}
}

# Post-combat dispatch event — fires immediately when an sc-driven combat
# ends (no delay). CK3 does NOT propagate saved scopes from GUI execute
# contexts through normal event chains, so this event does not rely on
# the scope chain: instead it reconstructs every scope it needs from
# character variables stashed on the PC during combat (sc_initiator,
# sc_attacker, sc_defender, fatality, fixed, follow_up, etc.). Applies
# wounds / death per the duel's fatality setting, fires the original
# PoD follow-up event, and then cleans up all combat variables.
#
# The old version used trigger_event { days = 1 } to escape the GUI
# context entirely via a fresh tick, but since all scopes are now
# reconstructed from stashed vars anyway, the delay is no longer needed.

pod_combat.9999 = {
	hidden = yes

	immediate = {
		# ── Reconstruct character scopes from stash ──
		var:pod_combat_sc_initiator = { save_scope_as = sc_initiator }
		var:pod_combat_sc_attacker  = { save_scope_as = sc_attacker }
		var:pod_combat_sc_defender  = { save_scope_as = sc_defender }

		# Defender aliases (covers victim, recipient, defender, target, adversary)
		var:pod_combat_sc_defender = { save_scope_as = victim }
		var:pod_combat_sc_defender = { save_scope_as = recipient }
		var:pod_combat_sc_defender = { save_scope_as = defender }
		var:pod_combat_sc_defender = { save_scope_as = target }
		var:pod_combat_sc_defender = { save_scope_as = adversary }

		# Attacker aliases (covers attacker, challenger)
		var:pod_combat_sc_attacker = { save_scope_as = attacker }
		var:pod_combat_sc_attacker = { save_scope_as = challenger }

		# ── Restore custom parent-event aliases ──
		# Rebuild any caller-specific scope names (scope:spirit,
		# scope:actor, scope:my_rival, etc.) that the parent event
		# saved before handing off to the minigame. Stash is populated
		# by pod_combat_sc_stash_parent_aliases_effect in the launcher;
		# the matching whitelist lives there too. Missing stashes are
		# skipped — any follow-up that reads an un-stashed alias will
		# still throw the same "Undefined event target" error, and the
		# alias needs to be added to the launcher whitelist.
		if = { limit = { exists = var:pod_combat_sc_alias_spirit           } var:pod_combat_sc_alias_spirit           = { save_scope_as = spirit } }
		if = { limit = { exists = var:pod_combat_sc_alias_alastor          } var:pod_combat_sc_alias_alastor          = { save_scope_as = alastor } }
		if = { limit = { exists = var:pod_combat_sc_alias_wyrm_champion    } var:pod_combat_sc_alias_wyrm_champion    = { save_scope_as = wyrm_champion } }
		if = { limit = { exists = var:pod_combat_sc_alias_vampire_attacker } var:pod_combat_sc_alias_vampire_attacker = { save_scope_as = vampire_attacker } }
		if = { limit = { exists = var:pod_combat_sc_alias_actor            } var:pod_combat_sc_alias_actor            = { save_scope_as = actor } }
		if = { limit = { exists = var:pod_combat_sc_alias_parent           } var:pod_combat_sc_alias_parent           = { save_scope_as = parent } }
		if = { limit = { exists = var:pod_combat_sc_alias_aggressor_vassal } var:pod_combat_sc_alias_aggressor_vassal = { save_scope_as = aggressor_vassal } }
		if = { limit = { exists = var:pod_combat_sc_alias_distant_knight   } var:pod_combat_sc_alias_distant_knight   = { save_scope_as = distant_knight } }
		if = { limit = { exists = var:pod_combat_sc_alias_snitch           } var:pod_combat_sc_alias_snitch           = { save_scope_as = snitch } }
		if = { limit = { exists = var:pod_combat_sc_alias_strong_leader    } var:pod_combat_sc_alias_strong_leader    = { save_scope_as = strong_leader } }
		if = { limit = { exists = var:pod_combat_sc_alias_warrior_monk     } var:pod_combat_sc_alias_warrior_monk     = { save_scope_as = warrior_monk } }
		if = { limit = { exists = var:pod_combat_sc_alias_attacked_rival   } var:pod_combat_sc_alias_attacked_rival   = { save_scope_as = attacked_rival } }
		if = { limit = { exists = var:pod_combat_sc_alias_my_rival         } var:pod_combat_sc_alias_my_rival         = { save_scope_as = my_rival } }
		if = { limit = { exists = var:pod_combat_sc_alias_rival_burner     } var:pod_combat_sc_alias_rival_burner     = { save_scope_as = rival_burner } }
		if = { limit = { exists = var:pod_combat_sc_alias_rival_on_sight   } var:pod_combat_sc_alias_rival_on_sight   = { save_scope_as = rival_on_sight } }
		if = { limit = { exists = var:pod_combat_sc_alias_rivals_child     } var:pod_combat_sc_alias_rivals_child     = { save_scope_as = rivals_child } }
		if = { limit = { exists = var:pod_combat_sc_alias_og_root          } var:pod_combat_sc_alias_og_root          = { save_scope_as = og_root } }
		if = { limit = { exists = var:pod_combat_sc_alias_root_for_loc     } var:pod_combat_sc_alias_root_for_loc     = { save_scope_as = root_for_loc } }
		if = { limit = { exists = var:pod_combat_sc_alias_recidivist       } var:pod_combat_sc_alias_recidivist       = { save_scope_as = recidivist } }
		if = { limit = { exists = var:pod_combat_sc_alias_shadowflame      } var:pod_combat_sc_alias_shadowflame      = { save_scope_as = shadowflame } }
		if = { limit = { exists = var:pod_combat_sc_alias_burned_artifact  } var:pod_combat_sc_alias_burned_artifact  = { save_scope_as = burned_artifact } }

		# Victor / loser from minigame result
		if = {
			limit = {
				has_variable = pod_combat_winner
				has_variable = pod_combat_loser
			}
			var:pod_combat_winner = { save_scope_as = sc_victor }
			var:pod_combat_loser  = { save_scope_as = sc_loser }
		}

		# ── Reconstruct fatality scope value ──
		if = {
			limit = { var:pod_combat_sc_fatality = flag:always }
			save_scope_value_as = { name = fatality value = flag:always }
		}
		else_if = {
			limit = { var:pod_combat_sc_fatality = flag:default }
			save_scope_value_as = { name = fatality value = flag:default }
		}
		else_if = {
			limit = { var:pod_combat_sc_fatality = flag:possible }
			save_scope_value_as = { name = fatality value = flag:possible }
		}
		else_if = {
			limit = { var:pod_combat_sc_fatality = flag:practice }
			save_scope_value_as = { name = fatality value = flag:practice }
		}
		else = {
			save_scope_value_as = { name = fatality value = flag:no }
		}

		# Fixed
		if = {
			limit = { var:pod_combat_sc_fixed = flag:sc_attacker }
			save_scope_value_as = { name = fixed value = flag:sc_attacker }
		}
		else_if = {
			limit = { var:pod_combat_sc_fixed = flag:sc_defender }
			save_scope_value_as = { name = fixed value = flag:sc_defender }
		}
		else = {
			save_scope_value_as = { name = fixed value = flag:no }
		}

		save_scope_value_as = { name = locale      value = flag:battlefield }
		save_scope_value_as = { name = sc_finished  value = yes }
		save_scope_value_as = { name = victory_type value = flag:skill }

		# ── Release engaged flags ──
		scope:sc_attacker = {
			if = {
				limit = { is_alive = yes }
				remove_variable = engaged_in_single_combat
			}
		}
		scope:sc_defender = {
			if = {
				limit = { is_alive = yes }
				remove_variable = engaged_in_single_combat
			}
		}

		# ── Apply death-only outcomes ──
		# NOTE: Wound damage has already been written back to PoD health
		# tracks by pod_combat_apply_health_track_effect (in pod_combat_end_combat_effect).
		# Re-applying POD_increase_wounds_hostile_effect here would double-damage
		# the loser and — more importantly — trigger the PoD death-check / depose
		# cascade, which in turn runs on_player_select_destiny_setup and evaluates
		# every CB group's allowed_for_character trigger in a scope where
		# scope:attacker has no valid liege, producing thousands of "Failed context
		# switch" errors per duel. We trust the minigame's own damage numbers:
		# if the loser's track was filled, PoD's normal periodic health check
		# will kill them on the next tick.
		#
		# For fatality modes that guarantee a kill (always, default-tribal),
		# we still call pod_death directly so the outcome is deterministic.
		if = {
			limit = { exists = scope:sc_victor }

			if = {
				limit = { scope:fatality = flag:practice }
			}
			else = {
				POD_fera_add_renown_effect = { TYPE = ferocity }
			}

			if = {
				limit = { scope:fatality = flag:default }
				# Tribal governments: tribal duels are always to the death.
				if = {
					limit = {
						scope:sc_attacker = {
							OR = {
								AND = {
									is_ruler = yes
									government_has_flag = government_is_tribal
								}
								AND = {
									is_ruler = no
									exists = court_owner
									scope:sc_attacker.court_owner = { government_has_flag = government_is_tribal }
								}
								AND = {
									is_ruler = no
									exists = host
									scope:sc_attacker.host = { government_has_flag = government_is_tribal }
								}
								AND = {
									is_ruler = no
									NOR = { exists = court_owner exists = host }
									scope:sc_defender = {
										OR = {
											AND = { is_ruler = yes government_has_flag = government_is_tribal }
											AND = { is_ruler = no exists = court_owner scope:sc_defender.court_owner = { government_has_flag = government_is_tribal } }
											AND = { is_ruler = no exists = host scope:sc_defender.host = { government_has_flag = government_is_tribal } }
										}
									}
								}
							}
						}
					}
					scope:sc_loser = {
						pod_death = { killer = scope:sc_victor death_reason = death_duel }
					}
					if = {
						limit = { scope:sc_victor = { faith = { has_doctrine = tenet_cranial_trophies } } }
						tgp_cranial_trophies_beheading_effect = { KILLER = scope:sc_victor DEAD = scope:sc_loser }
					}
				}
				# Non-tribal default: trust minigame damage (no forced kill).
			}
			else_if = {
				limit = { scope:fatality = flag:always }
				scope:sc_victor = {
					if = {
						limit = {
							this = scope:sc_attacker
							NOT = { has_execute_reason = scope:sc_loser }
						}
						add_kinslayer_trait_or_nothing_effect = { VICTIM = scope:sc_loser }
					}
				}
				scope:sc_loser = {
					pod_death = { killer = scope:sc_victor death_reason = death_duel }
				}
				if = {
					limit = { scope:sc_victor = { faith = { has_doctrine = tenet_cranial_trophies } } }
					tgp_cranial_trophies_beheading_effect = { KILLER = scope:sc_victor DEAD = scope:sc_loser }
				}
			}
			# flag:no       — writeback already capped loser's track at 1 empty box; no further action.
			# flag:possible — trust minigame damage; periodic death check handles full-track kills.
		}

		# ── Fire the follow-up event ──
		# Fire on ROOT (= the PC, who is sc_initiator in all current callers).
		# No scope:X = { } wrapping — scopes saved at the event's top level
		# propagate without any scope-block interference.
		trigger_event = { saved_event_id = var:pod_combat_sc_follow_up }

		# ── Full cleanup ──
		pod_combat_cleanup_effect = yes
	}
}
