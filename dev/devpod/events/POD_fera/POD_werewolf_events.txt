	}

	}
	
	option = {
		name = POD_werewolves.2.option.a
		custom_tooltip = POD_werewolves.2.option.a.tt
		
		POD_gift_cost_effect = { COUNT = 5 }
		
		duel = {
			skill = learning
			value = 20
			desc = POD_werewolves.2.option.a.desc
		
			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = high_positive_duel_skill_multiplier
				}
				
				custom_tooltip = POD_werewolves.2.option.a.tt.success
				
				hidden_effect = {
					POD_create_shadowy_duplicate_effect = { SAVE_AS = shadowy_duplicate }
					trigger_event = {
						id = POD_werewolves.3
						days = 2
					}
				}
			}
			
			50 = {

				compare_modifier = {
					value = scope:duel_value
					multiplier = low_negative_duel_skill_multiplier
				}
				
				custom_tooltip = POD_werewolves.2.option.a.tt.failure
				
				hidden_effect = {
					trigger_event = {
						id = POD_werewolves.4
						days = 2
					}
				}
			}
		}
		add_piety = -800
	}
	
	option = {
		name = POD_werewolves.2.option.b
		custom_tooltip = POD_werewolves.2.option.b.tt
		remove_interaction_cooldown = POD_shadow_pack_interaction
		remove_interaction_cooldown = POD_shadow_brethren_interaction
	}
}

#Rhythm of War, reached character
POD_werewolves.100 = {
	type = character_event
	content_source = dlc_POD
	title = POD_werewolves.100.t
	desc = POD_werewolves.100.desc
	theme = fera
	
	left_portrait = {
		character = root
		animation = personality_bold
	}
	right_portrait = scope:recipient
	
	
	option = {
		name = POD_werewolves.100.option.a
		if = {
			limit = { scope:lethality = yes }
			POD_combat_create_duel_effect = {
				INITIATOR    = scope:actor
				ATTACKER     = scope:actor
				DEFENDER     = scope:recipient
				FATALITY     = possible
				FIXED        = no
				LOCALE       = terrain_scope
				OUTPUT_EVENT = POD_werewolves.102
			}
			POD_combat_duel_add_enemy_effect = { CHARACTER = scope:recipient }
			POD_combat_finalize_duel_effect = yes
		}
		else = {
			POD_combat_create_duel_effect = {
				INITIATOR    = scope:actor
				ATTACKER     = scope:actor
				DEFENDER     = scope:recipient
				FATALITY     = practice
				FIXED        = no
				LOCALE       = terrain_scope
				OUTPUT_EVENT = POD_werewolves.102
			}
			POD_combat_duel_add_enemy_effect = { CHARACTER = scope:recipient }
			POD_combat_finalize_duel_effect = yes
		}
	}

	option = {
		name = POD_werewolves.100.option.b

		if = {
			limit = { scope:lethality = yes }
			add_prestige = minor_prestige_loss
		}

		stress_impact = {
			brave = medium_stress_impact_gain
			arrogant = minor_stress_impact_gain
		}
	}
}

#Check battle outcome
POD_werewolves.102 = {
	type = character_event
	hidden = yes

	immediate = {
		if = {
			limit = {
				scope:pod_combat_winner = root
				scope:lethality = yes
			}
			trigger_event = POD_werewolves.103
		}
		else_if = {
			limit = {
				NOT = { scope:pod_combat_winner = root }
				scope:lethality = yes
			}
			trigger_event = POD_werewolves.104
		}
		else_if = {
			limit = {
				scope:pod_combat_winner = root
				scope:lethality = no
			}
			trigger_event = POD_werewolves.105
		}
		else_if = {
			limit = {
				NOT = { scope:pod_combat_winner = root }
				scope:lethality = no
			}
			trigger_event = POD_werewolves.106
		}
	}
}

POD_werewolves.108 = {
	type = character_event
	content_source = dlc_POD
	title = POD_werewolves.108.t
	desc = {
		triggered_desc = {
			trigger = { scope:pod_combat_winner = root }
			desc = POD_werewolves.108.desc.a
		}
		triggered_desc = {
			trigger = { NOT = { scope:pod_combat_winner = root } }
			desc = POD_werewolves.108.desc.b
		}
	}
	theme = battle
	
	#Won
	option = {
		name = POD_werewolves.108.option.a
		trigger = { scope:pod_combat_winner = root }
		scope:ahroun_target = {
			if = {
				limit = { is_alive = yes }
				set_knight_status = forbid
				set_knight_status = allow
			}
		}
	}

	#Lost
	option = {
		name = POD_werewolves.108.option.a
		trigger = { NOT = { scope:pod_combat_winner = root } }
	}
}
