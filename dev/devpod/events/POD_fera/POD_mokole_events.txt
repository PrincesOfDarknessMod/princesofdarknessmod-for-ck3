#Yearly Maintenance
POD_mokole.1 = {
	hidden = yes

	trigger = {
		POD_is_mokole_trigger = yes
	}

	immediate = {
		#Give Barons Archid Options
		every_vassal = {
			limit = { highest_held_title_tier = tier_barony POD_is_mokole_trigger = yes }
			POD_mokole_assign_archid_options_effect = yes
		}
		#Give Knights and Guests Archid Options
		every_courtier_or_guest = {
			limit = { POD_is_mokole_trigger = yes }
			POD_mokole_assign_archid_options_effect = yes
		}

		if = {
			limit = { num_of_relation_innocent_tormentor >= 1 }
			save_scope_as = tormented
			every_relation = {
				type = innocent_tormentor

				trigger_event = { id = POD_mokole.2 days = { 1 300 } }
			}
			
		}

		###AI###
		if = {
			limit = { is_ai = yes }

			#Fill out Archid Options
			POD_mokole_assign_archid_options_effect = yes
		}
	}
}

#Walk into my eyes
POD_mokole.500 = {
	hidden = yes

	trigger = {
		is_alive = yes
	}

	immediate = {
		random_character_trait = {
			limit = { has_trait_category = personality }
			save_scope_as = changed_trait
		}

		send_interface_message = {
			type = POD_mokole_message_bad
			title = POD_walk_into_my_eyes_interaction_trait_target_notification
			right_icon = scope:actor

			scope:changed_trait = {
				scope:recipient = { remove_trait = prev }
				random_opposite_trait = { scope:recipient = { add_trait_force_tooltip = prev } }
			}
		}

		scope:actor = {
			send_interface_message = {
				type = POD_mokole_message_good
				title = POD_walk_into_my_eyes_interaction_trait_actor_notification
				right_icon = scope:recipient
				show_as_tooltip = {
					scope:changed_trait = {
						scope:recipient = { remove_trait = prev }
						random_opposite_trait = { scope:recipient = { add_trait_force_tooltip = prev } }
					}
				}
			}
		}
	}
}


#Landspeak
POD_mokole.501 = {
	hidden = yes

	trigger = {
		has_perk = landspeak_perk
	}

	immediate = {

		duel = {
			skills = { intrigue learning }
			value = 12
		
			40 = {
				compare_modifier = { value = scope:duel_value multiplier = high_positive_duel_skill_multiplier }

				ordered_vassal_or_below = {
					limit = {
						has_any_secrets = yes
						any_secret = { NOT = { is_known_by = root } }
					}
					order_by = {
						value = 100
						subtract = intrigue
						if = {
							limit = { highest_held_title_tier >= tier_duchy }
							subtract = 20
						}
						if = {
							limit = { has_trait = paranoid }
							subtract = 30
						}
						min = 0
					}
					max = 1
					save_scope_as = secret_owner
				}
				if = {
					limit = { exists = scope:secret_owner }
					send_interface_message = {
						type = POD_mokole_message_good
						title = landspeak_perk_name
						right_icon = scope:secret_owner

						scope:secret_owner = {
							random_secret = {
								limit = { NOT = { is_known_by = root } }
								reveal_to = root
							}
						}
					}
				}
			}
			
			60 = {
				compare_modifier = { value = scope:duel_value multiplier = low_negative_duel_skill_multiplier }
			}
		}
	}
}

	hidden = yes

	immediate = {
		POD_remove_character_modifier = { MODIFIER = POD_mokole_bloody_mindedness_nemesis_modifier }
		POD_remove_character_modifier = { MODIFIER = POD_mokole_bloody_mindedness_rival_modifier }

			limit = { has_perk = bloody_mindedness_perk }

			if = {
				limit = { any_relation = { type = nemesis } }
				add_character_modifier = POD_mokole_bloody_mindedness_nemesis_modifier
			}
			else_if = {
				limit = { any_relation = { type = rival } }
				add_character_modifier = POD_mokole_bloody_mindedness_rival_modifier
			}
		}
	}

	type = character_event
	content_source = dlc_POD

	left_portrait = root
	right_portrait = scope:dream_hunter

			skills = { diplomacy }
			target = scope:dream_hunter

		}
	}

#Journey end event
POD_mokole.690 = {
	type = character_event
	content_source = dlc_POD
	title = POD_mokole.690.t
	desc = POD_mokole.690.desc
	theme = fera
	right_portrait = root

	widget = {
		gui = "POD_event_window_widget_mokole_mnesis_quest"
		container = "custom_widgets_container"
	}

	trigger = {
		POD_is_mokole_trigger = yes
	}
	
	on_trigger_fail = {
		POD_end_journey_effect = yes # safeguard
	}

	option = {
		name = POD_mokole.690.option.a

		add_mnesis_lifestyle_xp = mokole_mnesis_lifestyle_reward_value

		if = {
			limit = { domicile ?= { has_domicile_building_or_higher = mokole_ancient_studies_01 } }
			random_list = {
				20 = { add_diplomacy_lifestyle_xp	= mokole_mnesis_vanilla_lifestyle_reward }
				20 = { add_martial_lifestyle_xp		= mokole_mnesis_vanilla_lifestyle_reward }
				20 = { add_stewardship_lifestyle_xp	= mokole_mnesis_vanilla_lifestyle_reward }
				20 = { add_intrigue_lifestyle_xp	= mokole_mnesis_vanilla_lifestyle_reward }
				20 = { add_learning_lifestyle_xp	= mokole_mnesis_vanilla_lifestyle_reward }
			}
		}

		POD_end_journey_effect = yes
	}


	after = {
		# Repeat
		if = {
			limit = { has_variable = continue_mokole_mnesis_quest }
			add_piety = -100
			custom_tooltip = {
				text = mokole_mnesis_quest_widget_tt
				POD_find_and_start_journey_effect = {
					TYPE = mokole_mnesis_quest
				}
			}
		}
		else = {
			# actual effects handled via journey on_actions
			show_as_tooltip = {
				POD_mokole_mnesis_clear_sleep_effect = yes
			}
		}
	}
}

#Spirit is sapping your strength
POD_mokole.720 = {
	type = character_event
	content_source = dlc_POD
	title = POD_mokole.720.t
	desc = POD_mokole.720.desc
	theme = spirit_world

	left_portrait = { character = root animation = paranoia }
	right_portrait = { character = scope:innocent animation = rage }
	
	immediate = {
		POD_mokole_spawn_innocent = { SAVE_AS = innocent }
	}

	#Resist
	option = {
		name = POD_mokole.720.a
		POD_willpower_rouse_check_effect = yes
		POD_gift_cost_effect = { COUNT = 1 }
		add_stress = medium_stress_gain
		add_piety = -250
	}

	#Redirect it
	option = {
		name = POD_mokole.720.b
		trigger = { any_vassal = { highest_held_title_tier >= tier_county } }
		#Target most relevant vassals
		ordered_vassal = {
			limit = { highest_held_title_tier >= tier_county }
			order_by = {
				value = 10
				if = {
					limit = { is_powerful_vassal_of = root }
					add = 40
				}
				if = {
					limit = { is_close_family_of = root }
					add = 30
				}
				if = {
					limit = { highest_held_title_tier = tier_duchy }
					add = 20
				}
				if = {
					limit = { highest_held_title_tier = tier_kingdom }
					add = 50
				}
				if = {
					limit = { is_councillor_of = root }
					add = 20
				}
			}
			max = 1
			save_scope_as = vassal_victim
		}
		stress_impact = {
			callous = medium_stress_impact_loss
			sadistic = major_stress_impact_loss
			compassionate = medium_stress_impact_gain
		}
		
		add_tyranny = minor_tyranny_gain
		POD_secondary_effects_from_torturing = yes
		scope:vassal_victim = {
			add_character_modifier = { modifier = recently_tortured years = 5 }
			add_opinion = { target = root modifier = tortured_me }
		}
	}

	#Succumb
	option = {
		name = POD_mokole.720.c
		POD_increase_wounds_hostile_effect = { ATTACKER = scope:innocent REASON = silver COUNT = 1 }
		add_character_modifier = { modifier = innocent_drained_modifier years = 5 }
	}


	after = { POD_mokole_become_tormentor = { INNOCENT = scope:innocent } }
}

#Archid Form Innocent attacking you
POD_mokole.721 = {
	type = character_event
	content_source = dlc_POD
	title = POD_mokole.721.t
	desc = POD_mokole.721.desc
	theme = spirit_world

	left_portrait = { character = root animation = paranoia }
	right_portrait = { character = scope:innocent animation = rage }
	
	immediate = {
		POD_mokole_spawn_innocent = { SAVE_AS = innocent }
	}

	#Fight it
	option = {
		name = POD_mokole.721.a
		POD_willpower_rouse_check_effect = yes
		POD_gift_cost_effect = { COUNT = 1 }
		duel = {
			skill = prowess
			target = scope:innocent
		
			30 = {
				compare_modifier = { value = scope:duel_value multiplier = high_positive_duel_skill_multiplier }
			}
			
			70 = {
				compare_modifier = { value = scope:duel_value multiplier = low_negative_duel_skill_multiplier }

				POD_increase_wounds_hostile_effect = { ATTACKER = scope:innocent COUNT = 2 REASON = fight }
				add_legitimacy = medium_legitimacy_loss
				add_prestige = -200
			}
		}
	}

	#Sacrifice an artifact
	option = {
		name = POD_mokole.721.b
		trigger = {
			any_equipped_character_artifact = {
				POD_artifact_is_destructible_by_events_trigger = yes
				NOR = {
					is_unique = yes
					rarity = illustrious
				}
			}
		}
		ordered_equipped_character_artifact = {
			limit = {
				POD_artifact_is_destructible_by_events_trigger = yes
				NOR = {
					is_unique = yes
					rarity = illustrious
				}
			}
			order_by = {
				value = 10
				if = {
					limit = { rarity = masterwork } #yes, masterwork has higher prio
					add = 20
				}
				if = {
					limit = { rarity = famed }
					add = 10
				}
			}
			max = 1
			save_scope_as = sacrificed_artifact
		}
		stress_impact = {
			greedy = major_stress_impact_gain
			avaricious = medium_stress_impact_gain
			ambitious = minor_stress_impact_gain
			arrogant = minor_stress_impact_gain
		}
		scope:sacrificed_artifact = { destroy_artifact = this }
	}

	#Flee
	option = {
		name = POD_mokole.721.c
		stress_impact = {
			brave = medium_stress_impact_gain
		}
		add_legitimacy = minor_legitimacy_loss
		add_prestige = -100
		add_character_modifier = { modifier = innocent_coward_modifier years = 5 }
	}


	after = { POD_mokole_become_tormentor = { INNOCENT = scope:innocent } }
}
