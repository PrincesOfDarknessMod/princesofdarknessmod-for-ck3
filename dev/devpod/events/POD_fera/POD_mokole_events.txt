		}
	}
}

#Walk into my eyes
POD_mokole.500 = {
	hidden = yes

	trigger = {
		is_alive = yes
	}

	immediate = {
		random_character_trait = {
			limit = { has_trait_category = personality }
			save_scope_as = changed_trait
		}

		send_interface_message = {
			type = POD_mokole_message_bad
			title = POD_walk_into_my_eyes_interaction_trait_target_notification
			right_icon = scope:actor

			scope:changed_trait = {
				scope:recipient = { remove_trait = prev }
				random_opposite_trait = { scope:recipient = { add_trait_force_tooltip = prev } }
			}
		}

		scope:actor = {
			send_interface_message = {
				type = POD_mokole_message_good
				title = POD_walk_into_my_eyes_interaction_trait_actor_notification
				right_icon = scope:recipient
				show_as_tooltip = {
					scope:changed_trait = {
						scope:recipient = { remove_trait = prev }
						random_opposite_trait = { scope:recipient = { add_trait_force_tooltip = prev } }
					}
				}
			}
		}
	}
}


#Landspeak
POD_mokole.501 = {
	hidden = yes

	trigger = {
		has_perk = landspeak_perk
	}

	immediate = {

		duel = {
			skills = { intrigue learning }
			value = 12
		
			40 = {
				compare_modifier = { value = scope:duel_value multiplier = high_positive_duel_skill_multiplier }

				ordered_vassal_or_below = {
					limit = {
						has_any_secrets = yes
						any_secret = { NOT = { is_known_by = root } }
					}
					order_by = {
						value = 100
						subtract = intrigue
						if = {
							limit = { highest_held_title_tier >= tier_duchy }
							subtract = 20
						}
						if = {
							limit = { has_trait = paranoid }
							subtract = 30
						}
						min = 0
					}
					save_scope_as = secret_owner
				}

				send_interface_message = {
					type = POD_mokole_message_good
					title = landspeak_perk_name
					right_icon = scope:secret_owner

					scope:secret_owner = {
						random_secret = {
							limit = { NOT = { is_known_by = root } }
							reveal_to = root
						}
					}
				}	
			}
			
			60 = {
				compare_modifier = { value = scope:duel_value multiplier = low_negative_duel_skill_multiplier }
			}
		}
	}
}

	type = character_event
	hidden = yes

	trigger = {
		var:current_journey = { var:journey_type = flag:mokole_mnesis_quest }
	}
	weight_multiplier = {
		base = 1

		## Global ##
		modifier = {
			var:current_journey.var:dreamhunter_counter >= 2
			add = -0.25
		}
		modifier = {
			var:current_journey.var:dreamhunter_counter >= 4
			add = -0.25
		}

		## Age of Beasts+ ##
		modifier = {
			mokole_mnesis_age_value >= 2
			var:current_journey.var:dreamhunter_counter < 1
			add = 0.5
		}

		## Age of Dreams+ ##
		modifier = {
			mokole_mnesis_age_value >= 3
			var:current_journey.var:dreamhunter_counter < 3
			add = 0.5
		}

		## Age of Kings ##
		modifier = {
			mokole_mnesis_age_value = 4
			var:current_journey.var:dreamhunter_counter < 6
			add = 1
		}
	}

	immediate = {
		var:current_journey = { change_variable = { name = dreamhunter_counter add  = 1 } }
		
		random = {
			chance = mokole_dream_hunter_encounter_chance

			POD_mokole_spawn_dream_hunter_effect = yes

			random_list = {
				50 = { trigger_event = POD_mokole.681 } #Diplomacy Challenge
				50 = {  } #Martial Challenge
				50 = {  } #Stewardship Challenge
				50 = {  } #Intrigue Challenge
				50 = {  } #Learning Challenge
				50 = {  } #Misc Challenge
				50 = {  } #Misc Challenge
			}
		}
	}
}

#Dream Hunter is lost spirit, try to talk it through
POD_mokole.681 = {
	type = character_event
	content_source = dlc_POD
	title = POD_mokole.681.t
	desc = POD_mokole.681.desc

	theme = faith
	left_portrait = root
	right_portrait = scope:dream_hunter


	#Talk it out
	option = {
		name = POD_mokole.681.option.a
		duel = {
			skills = { diplomacy }
			target = scope:dream_hunter
		
			50 = {
				compare_modifier = { value = scope:duel_value multiplier = high_positive_duel_skill_multiplier }
				
				add_mnesis_lifestyle_xp = mokole_dream_hunter_reward_value
			}
			
			50 = {
				compare_modifier = { value = scope:duel_value multiplier = low_negative_duel_skill_multiplier }
				
				POD_mokole_dream_hunter_damage_effect = yes
			}
		}
	}


	#Fight
	option = {
		name = POD_mokole.681.option.b
		configure_start_single_combat_effect = {
			SC_INITIATOR = root
			SC_ATTACKER = scope:dream_hunter
			SC_DEFENDER = root
			FATALITY = possible
			FIXED = no
			LOCALE = terrain_scope
			OUTPUT_EVENT = POD_mokole.689
			INVALIDATION_EVENT = fp1_tbc.0102
		}
	}
}
