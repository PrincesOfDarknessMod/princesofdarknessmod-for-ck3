namespace = POD_fera_maintenance



# Fera Yearly Healing
POD_fera_maintenance.1 = {
	hidden = yes
	
	trigger = {
		has_trait = fera
		NOR = {
			has_trait = kitsune
			has_trait = entomos
		}
		POD_has_enough_gnosis_for_gift = { COUNT = 1 }
		OR = {
			AND = {
				has_trait = disfigured
				POD_is_trait_not_permanent_trigger = { TRAIT = disfigured }
			}
			AND = {
				has_trait = blind
				POD_is_trait_not_permanent_trigger = { TRAIT = blind }
			}
			AND = {
				has_trait = maimed
				POD_is_trait_not_permanent_trigger = { TRAIT = maimed }
			}
			AND = {
				has_trait = one_legged
				POD_is_trait_not_permanent_trigger = { TRAIT = one_legged }
			}
			AND = {
				has_trait = one_eyed
				POD_is_trait_not_permanent_trigger = { TRAIT = one_eyed }
			}
			AND = {
				has_trait = eunuch_1
				POD_is_trait_not_permanent_trigger = { TRAIT = eunuch_1 }
			}
		}
		POD_can_be_healed_trigger = yes
	}

	immediate = {
		
		if = {
			limit = { has_character_modifier = infected_wound_modifier }
			remove_character_modifier = infected_wound_modifier

			if = {
				limit = { NOT = { has_trait = scarred } }
				add_character_flag = will_get_scar
			}
		}
		else_if = {
			limit = { NOT = { has_trait = scarred } }
			random = {
				chance = 10
				add_character_flag = will_get_scar
			}
		}
		send_interface_toast = {
			title = health.0001.desc

			POD_gift_cost_effect = { COUNT = 1 }

			remove_trait = flagellant
			POD_remove_major_injury_traits_effect = yes
	
			if = {
				limit = {
					has_trait = disfigured
					POD_is_trait_not_permanent_trigger = { TRAIT = disfigured }
				}
				remove_trait = disfigured
			}
			
			if = {
				limit = { has_character_flag = will_get_scar }
				add_trait = scarred
				remove_character_flag = will_get_scar
			}
		}
	}
}


#First change
POD_fera_maintenance.2 = {
	type = character_event
	content_source = dlc_POD
	title = POD_fera_maintenance.2.t
	desc = POD_fera_maintenance.2.desc

	theme = faith
	left_portrait = {
		character = root
		animation = shock
	}
	
	trigger = {
		has_character_flag = fera_first_change
		has_trait = kinfolk
		NOT = { has_trait = changeling }
		NOT = { has_trait = revenant }
	}
	
	option = {
		name = POD_fera_maintenance.2.option.a
		
		remove_trait = kinfolk
		remove_trait = gnosis
		
		if = {
			limit = { POD_splat_mixing_haight_mode = no }
			POD_stop_being_ghoul = yes
		}
		
		if = {
			limit = { has_trait = supehunter }
			hidden_effect = {
				remove_trait = supehunter
				save_temporary_scope_as = is_sanbenito
			}
		}
		
		if = {
			limit = { exists = var:fera_type }
			var:fera_type = { save_scope_as = fera_type }
		}
		else = {
			# fallback
			trait:werewolf = { save_scope_as = fera_type }
		}
		
		add_trait_force_tooltip = scope:fera_type
		add_trait = homid_born
		POD_fera_initialization_effect = yes
		
		hidden_effect = {
			if = {
				limit = { exists = scope:is_sanbenito }
				add_trait = supehunter
				set_character_faith = faith:sanbenito
			}
		}
		
		POD_remove_wrong_supernatural_traits_on_existing_character = yes


		#AI help
		if = {
			limit = {
				exists = mother
				mother = {
					is_landed = yes
					primary_heir ?= { has_trait = kinfolk }
					is_ai = yes
				}
				is_valid_designated_heir = mother
			}
			mother = { set_designated_heir = prev }
		}
		else_if = {
			limit = {
				exists = father
				father = {
					is_landed = yes
					primary_heir ?= { has_trait = kinfolk }
					is_ai = yes
				}
				is_valid_designated_heir = father
			}
			father = { set_designated_heir = prev }
		}
	}
}
