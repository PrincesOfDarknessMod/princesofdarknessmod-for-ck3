#First change
POD_fera_maintenance.2 = {
	type = character_event
	content_source = dlc_POD
	title = POD_fera_maintenance.2.t
	desc = POD_fera_maintenance.2.desc

	theme = faith
	left_portrait = {
		character = root
		animation = shock
	}
	
	trigger = {
		has_character_flag = fera_first_change
		has_trait = kinfolk
		NOT = { has_trait = changeling }
		NOT = { has_trait = revenant }
	}
	
	option = {
		name = POD_fera_maintenance.2.option.a
		
		remove_trait = kinfolk
		remove_trait = gnosis
		
		if = {
			limit = { POD_splat_mixing_haight_mode = no }
			POD_stop_being_ghoul = yes
		}
		
		if = {
			limit = { has_trait = supehunter }
			hidden_effect = {
				remove_trait = supehunter
				save_temporary_scope_as = is_sanbenito
			}
		}
		
		if = {
			limit = { exists = var:fera_type }
			var:fera_type = { save_scope_as = fera_type }
		}
		else = {
			# fallback
			trait:werewolf = { save_scope_as = fera_type }
		}
		
		add_trait_force_tooltip = scope:fera_type
		add_trait = homid_born
		POD_fera_initialization_effect = yes
		
		hidden_effect = {
			if = {
				limit = { exists = scope:is_sanbenito }
				add_trait = supehunter
				set_character_faith = faith:sanbenito
			}
		}
		
		POD_remove_wrong_supernatural_traits_on_existing_character = yes


		#AI help
		if = {
			limit = {
				exists = mother
				mother = {
					is_landed = yes
					primary_heir ?= { has_trait = kinfolk }
					is_ai = yes
				}
				is_valid_designated_heir = mother
			}
			mother = { set_designated_heir = prev }
		}
		else_if = {
			limit = {
				exists = father
				father = {
					is_landed = yes
					primary_heir ?= { has_trait = kinfolk }
					is_ai = yes
				}
				is_valid_designated_heir = father
			}
			father = { set_designated_heir = prev }
		}
	}
}

	hidden = yes
	
	trigger = {
		has_trait = mokole #currently only mokole use this
	}

	immediate = {

		#Mnesis
		if = {
			limit = { has_trait = mokole }

# Fera Umbra
POD_fera_maintenance.103 = {
	hidden = yes
	
	trigger = {
		POD_fera_rank_value >= 2
		has_umbra_access_trigger = yes
	}

	immediate = {
		
		#Find MAA

		#Planetary Gifts
		if = {
			limit = {
				POD_splat_has_access_to_planetary_gifts = yes
				has_access_to_spirit_wilds_trigger = yes
				planetarygifts_lifestyle_perks < 65
			}
			random = {
				chance = fera_ai_planetary_gift_gain_chance
				add_planetarygifts_lifestyle_perk_points = 1
			}
		}

		#Find a Spirit
		if = {
			limit = {
				POD_fera_ai_can_spirit_pact = yes
				POD_can_summon_spirit_trigger = no #Summoning spirits directly is far superior, so AI should do that instead
				has_access_to_spirit_wilds_trigger = yes
			}
			random = {
				chance = fera_ai_spirit_pact_umbra_chance
				POD_spawn_friendly_spirit_effect = yes
				POD_form_spirit_pact_effect = { TARGET = scope:spawned_spirit }
				if = {
					limit = { NOT = { POD_knows_spirit_language_custom_trigger = { ACTOR = this RECIPIENT = scope:spawned_spirit } } }
					POD_learn_spirit_language_effect = { TARGET = scope:spawned_spirit }
				}
			}
		}
	}
}


# Fera Spirit management
POD_fera_maintenance.104 = {
	hidden = yes
	
	trigger = {
		POD_fera_rank_value >= 3
	}

	immediate = {
		#Summon a Spirit
		if = {
			limit = { POD_fera_ai_can_summon_spirit_pact = yes }
			random = {
				chance = fera_ai_spirit_pact_summon_chance
				POD_spawn_friendly_spirit_effect = yes
				POD_form_spirit_pact_effect = { TARGET = scope:spawned_spirit }
				if = {
					limit = { NOT = { POD_knows_spirit_language_custom_trigger = { ACTOR = this RECIPIENT = scope:spawned_spirit } } }
					POD_learn_spirit_language_effect = { TARGET = scope:spawned_spirit }
				}
			}
		}


		#Perform Chiminage/Empowerment
		#AI cheats and does chiminage for Wyrm spirits as well. It also pays less
		if = {
			limit = {
				num_of_relation_spirit_companion > 0
				any_relation = {
					type = spirit_companion
					NOT = { POD_spirit_rank = 5 }
				}
				NOT = { has_character_flag = ai_chiminage_cooldown }
			}

			add_character_flag = { flag = ai_chiminage_cooldown years = 8 }
			ordered_relation = {
				type = spirit_companion
				limit = { NOT = { POD_spirit_rank = 5 } }
				order_by = {
					value = 0
					add = "this.has_trait_xp(spirit)"
				}
				max = 1
				save_scope_as = chiminage_target
			}
			scope:chiminage_target ?= {
				save_temporary_scope_value_as = {
					name = old_spirit_rank
					value = POD_spirit_rank
				}
				add_trait_xp = { trait = spirit value = POD_spirit_chiminage_gain }
				if = {
					limit = { religion = { is_in_family = rf_wyrm } }
					root = { add_stress = miniscule_stress_gain }
				}
				else = { root = { add_character_modifier = { modifier = spirit_short_chiminage_modifier years = 3 } } }

				if = {
					limit = { POD_spirit_rank > scope:old_spirit_rank }
					root = { POD_update_spirit_pact_modifiers = yes }
				}
			}
		}
	
		#Spirit Quests
		#AI only gets Fetishes and Wisdom renown from these
		random = {
			chance = fera_ai_spirit_quest_chance
			POD_fera_ai_spirit_quest_effect = yes
		}
	}
}
