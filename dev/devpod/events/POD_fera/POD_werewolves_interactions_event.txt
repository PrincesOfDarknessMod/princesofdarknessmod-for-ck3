	theme = witchcraft
	
	left_portrait = {
	    character = root
		animation = personality_bold 
	}
	
	immediate = {
	
	}
	
	option = {
	    name = POD_werewolves_interactions.11.option.a
		custom_tooltip = POD_werewolves_interactions.11.option.a.tt
		
		duel = {
		    skill = learning
			value = 20
			desc = POD_werewolves_interactions.11.option.a.desc
		
			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = high_positive_duel_skill_multiplier
				}
				
				custom_tooltip = POD_werewolves_interactions.11.option.a.tt.success
				
				hidden_effect = {
				    POD_create_shadowy_duplicate_effect = yes
					trigger_event = {
					    id = POD_werewolves_interactions.12
						days = 2
					}
			    }
			}    
			
			50 = {

				compare_modifier = {
					value = scope:duel_value
					multiplier = low_negative_duel_skill_multiplier
				}
				
				custom_tooltip = POD_werewolves_interactions.11.option.a.tt.failure
				
			    hidden_effect = {
				    trigger_event = {
					    id = POD_werewolves_interactions.13
						days = 2
					}
			    }
			}
		}
		add_piety = -800
	}
	
	option = {
	    name = POD_werewolves_interactions.11.option.b
		custom_tooltip = POD_werewolves_interactions.11.option.b.tt
		remove_interaction_cooldown = POD_shadow_pack_interaction
		remove_interaction_cooldown = POD_shadow_brethren_interaction
	}
}

POD_werewolves_interactions.15 = {
    type = character_event
	content_source = dlc_POD
	title = POD_werewolves_interactions.15.t
	desc = POD_werewolves_interactions.15.desc
	theme = witchcraft
	
	left_portrait = {
	    character = root
		animation = stress
	}
	right_portrait = scope:kinfolk_child
	
	trigger = {
		scope:kinfolk_child = { is_alive = yes }
		has_trait = werewolf
	}
	
	option = {
	    name = POD_werewolves_interactions.15.option.a
		
		add_stress = minor_stress_gain
		add_piety = minor_piety_loss
		
		hidden_effect = {
			if = {
				limit = { scope:kinfolk_child = { has_character_flag = will_have_first_change } }
				send_interface_toast = {
					title = POD_werewolves_interactions.15.a.sucess
					left_icon = root
					right_icon = scope:kinfolk_child
					custom_tooltip = POD_werewolves_interactions.15.a.sucess.tt
				}
			}
			else = {
				send_interface_toast = {
					title = POD_werewolves_interactions.15.a.failure
					left_icon = root
					right_icon = scope:kinfolk_child
					custom_tooltip = POD_werewolves_interactions.15.a.failure.tt
				}
			}
		}
	}
	
	option = {
	    name = POD_werewolves_interactions.15.option.b
	}
}
