	#Capital development decline

		name = POD_fera_pack.1.option.a

			compassionate = medium_stress_impact_gain

			wrathful = minor_stress_impact_loss

			callous = medium_stress_impact_loss
		}

			limit = { highest_held_title_tier >= kingdom }

		}
		else = { add_legitimacy = medium_legitimacy_loss }

			change_development_level = -4

	}

	#War modifier
	option = {
		name = POD_fera_pack.1.option.b

			craven = medium_stress_impact_gain

			limit = { highest_held_title_tier >= kingdom }
			add_prestige = major_prestige_loss
		}
		else = { add_prestige = medium_prestige_loss }
		add_character_modifier = { modifier = fera_pack_war_modifier years = 5 }
	}
	
	#Train prowess
	option = {
		name = POD_fera_pack.1.option.c
		
		stress_impact = {
			craven = major_stress_impact_gain
			brave = minor_stress_impact_loss
			shy = minor_stress_impact_gain
			lazy = medium_stress_impact_gain
		}

			33 = {
				add_prowess_skill = 2
				if = {
					limit = { faith = { has_doctrine_parameter = pack_training_blademaster } }
					POD_increase_blademaster_xp = { VALUE = 3 }
				}
				POD_increase_wounds_effect = { REASON = fight COUNT = 1 }
			}
			33 = {
				add_prowess_skill = 3
				if = {
					limit = { faith = { has_doctrine_parameter = pack_training_blademaster } }
					POD_increase_blademaster_xp = { VALUE = 6 }
				}
				POD_increase_wounds_effect = { REASON = fight COUNT = 2 }
			}
			33 = {
				trigger = { prowess <= 50 }
				add_prowess_skill = 4
				if = {
					limit = { faith = { has_doctrine_parameter = pack_training_blademaster } }
					POD_increase_blademaster_xp = { VALUE = 10 }
				}
				POD_increase_wounds_effect = { REASON = fight COUNT = 3 }
			}
		}
		if = {
			limit = {
				faith = { has_doctrine_parameter = pack_training_blademaster }
				NOT = { has_trait = lifestyle_blademaster }
			}
			add_trait = lifestyle_blademaster
		}
		custom_tooltip = fera_pack_coterie_rewards_tt
		
		hidden_effect = {
			every_relation = {
				type = coterie_member
				add_prowess_skill = 3
			}
		}
	}
