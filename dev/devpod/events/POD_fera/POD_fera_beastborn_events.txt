# Intro Event
POD_fera_beastborn.1 = {
	type = character_event
	content_source = dlc_POD
	title = POD_fera_beastborn.1.t
	desc = {
		first_valid = {			
			#Bubasti after their objective
			triggered_desc = {
				trigger = {
					OR = {
						AND = {
							has_faith = faith:bubasti 			
							is_target_in_global_variable_list = {
								name = unavailable_unique_decisions
								target = flag:bubasti_objective_decision
							}
						}
						uses_cairo_bast_main_reward = yes
					}
				}
				desc = POD_fera_beastborn.1.bubastivictorious
			}
			#Bastet
			triggered_desc = {
				trigger = { has_trait = bastet }
				desc = POD_fera_beastborn.1.bastet
			}

			#Werewolves
			triggered_desc = {
				desc = POD_fera_beastborn.1.werewolf
			}
		}
	}	
	theme = faith 

	trigger = {}

	option = {
		name = POD_fera_beastborn.1.a	
		custom_tooltip = start_beastborn_content_tt
		trigger_event = POD_fera_beastborn.10
	}
	
	option = {
		name = POD_fera_beastborn.1.b
		custom_tooltip = delayed_start_beastborn_content_tt
		trigger_event = {
			id = POD_fera_beastborn.10
			years = { 20 40 }
		}
	}
	
	option = {
		name = POD_fera_beastborn.1.c
		custom_tooltip = end_beastborn_content_tt
	}
}

#End of Chain
POD_fera_beastborn.20 = {
	type = character_event
	content_source = dlc_POD
	title = POD_fera_beastborn.20.t
	desc = 	POD_fera_beastborn.20.desc
	theme = faith 
	
	right_portrait = {
		character = scope:beastborn_1
		animation = personality_honorable
	}

	left_portrait = {
		trigger = { exists = scope:beastborn_2 }
		character = scope:beastborn_2
		animation = personality_content
	}
	
	lower_right_portrait = {
		trigger = { exists = scope:beastborn_3 }
		character = scope:beastborn_3
	}


	immediate = {
		hidden_effect = {
			save_scope_as = spawning_character
			create_character = {
				location = scope:spawning_character.capital_province
				template = beastborn_character
				dynasty_house = none
				save_scope_as = beastborn_1
			}
			if = {
				limit = { has_character_flag = numerous_beast_born_flag }
				create_character = {
					location = scope:spawning_character.capital_province
					template = beastborn_character
					dynasty_house = none
					save_scope_as = beastborn_2
				}
				create_character = {
					location = scope:spawning_character.capital_province
					template = beastborn_character
					dynasty_house = none
					save_scope_as = beastborn_3
				}
			}
		}
	}
	
	option = {
		name = POD_fera_beastborn.20.a

		add_courtier = scope:beastborn_1
		if = {
			limit = { has_character_flag = numerous_beast_born_flag }
			add_courtier = scope:beastborn_2
			add_courtier = scope:beastborn_3
		}
		trigger_event = { id = POD_fera_beastborn.10 years = { 2 5 } }
	}

	option = {
		name = POD_fera_beastborn.20.b
		
		add_courtier = scope:beastborn_1
		if = {
			limit = { has_character_flag = numerous_beast_born_flag }
			add_courtier = scope:beastborn_2
			add_courtier = scope:beastborn_3
		}
		custom_tooltip = end_beastborn_content_tt		
	}
	
	after = {
		remove_character_flag = numerous_beast_born_flag
		remove_character_flag = beastborn_no_interference
		remove_character_flag = beastborn_full_interference
		remove_character_flag = beastborn_more_diplomacy
		remove_character_flag = beastborn_more_martial
		remove_character_flag = beastborn_more_stewardship
		remove_character_flag = beastborn_more_intrigue
		remove_character_flag = beastborn_more_learning
	}		
}

#Lupus pups for lupus ruler
POD_fera_beastborn.102 = {
	type = character_event
	content_source = dlc_POD
	title = POD_fera_beastborn.102.t
	desc = POD_fera_beastborn.102.desc

	theme = faith
	left_portrait = {
		character = root
		animation = personality_content
	}
	
	trigger = {
		has_trait = beast_born
		has_trait = werewolf
	}
	
	immediate = {
		hidden_effect = {
			if = {
				limit = {
					exists = capital_province
				}
				capital_province = { save_scope_as = target_location }
			}
			else = {
				location = { save_scope_as = target_location }
			}
			create_character = {
				location = scope:target_location
				template = fera_beastborn_character
				dynasty_house = root.house
				save_scope_as = lupus_1
			}
			add_visiting_courtier = scope:lupus_1
		}		
	}
	
	option = {
		name = POD_fera_beastborn.102.option.a	
		
		trigger = {
			dynasty ?= { has_dynasty_perk = redtalon_legacy_4 }
		}
		
		add_stress = minor_stress_gain
		
		custom_tooltip = cub_redtalon4_tt
		
		custom_tooltip = cub_recruited_tt
	
		add_courtier = scope:lupus_1

		scope:lupus_1 = {
			add_prowess_skill = 10
			add_learning_skill = 3
		}	

		random = {			
			chance = 10
			
		scope:lupus_1 = {
			add_stewardship_skill = 5
		}	
		}
		
		random = {			
			chance = 20
			
		scope:lupus_1 = {
			add_diplomacy_skill = 5
		}	
		}
		
		random = {			
			chance = 33
			
		scope:lupus_1 = {
			add_martial_skill = 5
		}	
		}
		
		hidden_effect = {
		scope:lupus_1 = {POD_set_parent_effect = { PARENT = root }}
		}		
	}
	
	
	option = {
		name = POD_fera_beastborn.102.option.b	
		
		trigger = {
			NOT = {
				dynasty ?= {
					has_dynasty_perk = redtalon_legacy_4
				}
			}
		}
		
		add_stress = minor_stress_gain
		
		custom_tooltip = cub_recruited_tt
	
		add_courtier = scope:lupus_1

		scope:lupus_1 = {
			add_learning_skill = 3
		}	

		random = {			
			chance = 10
			
		scope:lupus_1 = {
			add_stewardship_skill = 5
		}	
		}
		
		random = {			
			chance = 20
			
		scope:lupus_1 = {
			add_diplomacy_skill = 5
		}	
		}
		
		random = {			
			chance = 33
			
		scope:lupus_1 = {
			add_martial_skill = 5
		}	
		}
		
		hidden_effect = {
		scope:lupus_1 = {POD_set_parent_effect = { PARENT = root }}
		}
	}

	option = {
		name = POD_fera_beastborn.102.option.c
