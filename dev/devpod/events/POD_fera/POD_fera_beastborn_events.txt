		}
		add_character_flag = beastborn_no_interference

	}

	option = {
		name = POD_fera_beastborn.11.b
		if = {
			limit = { is_ai = no }
			add_piety = -100
		}
		add_character_flag = beastborn_full_interference
		custom_tooltip = beastborn_full_interference_tooltip
		ai_chance = { base = 50 }
	}

	option = {
		name = POD_fera_beastborn.11.c
		ai_chance = { base = 0 }
	}

	after = {
		custom_tooltip = wait_time_beastborn_tooltip
		hidden_effect = {
			if = {
				limit = { is_ai = no }
				random_list = {
					20 = { trigger_event = { id = POD_fera_beastborn.12 years = { 5 10 } } }
					20 = { trigger_event = { id = POD_fera_beastborn.13 years = { 5 10 } } }
					20 = { trigger_event = { id = POD_fera_beastborn.14 years = { 5 10 } } }
					20 = { trigger_event = { id = POD_fera_beastborn.15 years = { 5 10 } } }
					20 = { trigger_event = { id = POD_fera_beastborn.16 years = { 5 10 } } }
				}
			}
			else = {
				random_list = {
					20 = { add_character_flag = beastborn_more_diplomacy }
					20 = { add_character_flag = beastborn_more_martial }
					20 = { add_character_flag = beastborn_more_stewardship }
					20 = { add_character_flag = beastborn_more_intrigue }
					20 = { add_character_flag = beastborn_more_learning }
				}
				trigger_event = { id = POD_fera_beastborn.20 years = { 3 5 } }
			}
		}
	}
}

POD_fera_beastborn.20 = {
	type = character_event
	content_source = dlc_POD
	title = POD_fera_beastborn.20.t
	desc = 	POD_fera_beastborn.20.desc
	theme = fera
	
	right_portrait = {
		character = scope:beastborn_1
		animation = personality_honorable
	}

	left_portrait = {
		trigger = { exists = scope:beastborn_2 }
		character = scope:beastborn_2
		animation = personality_content
	}
	
	lower_right_portrait = {
		trigger = { exists = scope:beastborn_3 }
		character = scope:beastborn_3
	}


	immediate = {
		hidden_effect = {
			save_scope_as = spawning_character
			if = {
				limit = { is_landed = yes }
				capital_province ?= { save_scope_as = location }
			}
			else = { location ?= { save_scope_as = location } }
			create_character = {
				location = scope:location
				template = fera_beastborn_character
				dynasty = none
				save_scope_as = beastborn_1
			}
			if = {
				limit = { has_character_flag = numerous_beast_born_flag }
				create_character = {
					location = scope:location
					template = fera_beastborn_character
					dynasty = none
					save_scope_as = beastborn_2
				}
				if = {
					limit = { is_ai = no } #AI don't get a 3rd character
					create_character = {
						location = scope:location
						template = fera_beastborn_character
						dynasty = none
						save_scope_as = beastborn_3
					}
				}
			}
		}
	}
	
	#Add all
	option = {
		name = POD_fera_beastborn.20.a

		if = {
			limit = { is_playable_character = yes }
			add_courtier = scope:beastborn_1
		}
		
		if = {
			limit = { has_character_flag = numerous_beast_born_flag is_playable_character = yes }
			add_courtier = scope:beastborn_2
			if = {
				limit = { is_ai = no }
				add_courtier = scope:beastborn_3
			}
		}
		ai_chance = {
			base = 25
			modifier = {
				has_trait = beast_born
				POD_fera_is_long_lived_trigger = no
				any_child = {
					count <= 2
					is_alive = yes
				}
				add = -25
			}
		}
	}


	#Adopt 1st
	option = {
		name = POD_fera_beastborn.20.b
		if = {
			limit = { is_ai = no }
			add_prestige = -250
			add_legitimacy = -30
		}
		if = {
			limit = { is_playable_character = yes }
			add_courtier = scope:beastborn_1
		}

		if = {
			limit = { dynasty ?= { has_dynasty_perk = redtalon_legacy_4 } }
			custom_tooltip = POD_fera_beastborn.20.tooltip_redtalon
			scope:beastborn_1 = { add_prowess_skill = 10 }
		}
		scope:beastborn_1 = { POD_set_parent_effect = { PARENT = root } }


		if = {
			limit = { has_character_flag = numerous_beast_born_flag is_playable_character = yes }
			add_courtier = scope:beastborn_2
			if = {
				limit = { is_ai = no } #3rd character only created for players
				add_courtier = scope:beastborn_3
			}
		}

		ai_chance = {
			base = 50
			modifier = {
				any_child = {
					count >= 3
					is_alive = yes
				}
				add = -50
			}
			modifier = {
				POD_fera_is_long_lived_trigger = yes
				age <= 100
				add = -40
			}
			modifier = {
				age <= 40
				add = -1000
			}
		}
	}

	#Adopt 2nd
	option = {
		name = POD_fera_beastborn.20.c
		trigger = { has_character_flag = numerous_beast_born_flag }
		if = {
			limit = { is_ai = no }
			add_prestige = -250
			add_legitimacy = -30
		}
		if = {
			limit = { is_playable_character = yes }
			add_courtier = scope:beastborn_2
		}

		if = {
			limit = { dynasty ?= { has_dynasty_perk = redtalon_legacy_4 } }
			custom_tooltip = POD_fera_beastborn.20.tooltip_redtalon
			scope:beastborn_2 = { add_prowess_skill = 10 }
		}
		scope:beastborn_2 = { POD_set_parent_effect = { PARENT = root } }

		if = {
			limit = { is_playable_character = yes }
			add_courtier = scope:beastborn_1
			if = {
				limit = { is_ai = no }
				add_courtier = scope:beastborn_3
			}
		}

		ai_chance = {
			base = 50
			modifier = {
				any_child = {
					count >= 3
					is_alive = yes
				}
				add = -50
			}
			modifier = {
				POD_fera_is_long_lived_trigger = yes
				age <= 100
				add = -40
			}
			modifier = {
				age <= 40
				add = -1000
			}
		}
	}


	#Adopt 3rd
	option = {
		name = POD_fera_beastborn.20.d
		trigger = { has_character_flag = numerous_beast_born_flag is_ai = no }
		if = {
			limit = { is_ai = no }
			add_prestige = -250
			add_legitimacy = -30
		}
		if = {
			limit = { is_playable_character = yes }
			add_courtier = scope:beastborn_3
		}

		if = {
			limit = { dynasty ?= { has_dynasty_perk = redtalon_legacy_4 } }
			custom_tooltip = POD_fera_beastborn.20.tooltip_redtalon
			scope:beastborn_3 = { add_prowess_skill = 10 }
		}
		scope:beastborn_3 = { POD_set_parent_effect = { PARENT = root } }

		if = {
			limit = { is_playable_character = yes }
			add_courtier = scope:beastborn_1
			add_courtier = scope:beastborn_2
		}
	}


	
	after = {
		remove_character_flag = numerous_beast_born_flag
		remove_character_flag = beastborn_no_interference
		remove_character_flag = beastborn_full_interference
		remove_character_flag = beastborn_more_diplomacy
		remove_character_flag = beastborn_more_martial
		remove_character_flag = beastborn_more_stewardship
		remove_character_flag = beastborn_more_intrigue
		remove_character_flag = beastborn_more_learning
		if = {
			limit = { is_ai = yes }
			remove_decision_cooldown = fera_recruit_beastborn_decision
		}
	}
}
