POD_fera_spirit_quest.1 = {
	type = character_event
	content_source = dlc_POD
	title = POD_fera_spirit_quest.1.t
	desc = POD_fera_spirit_quest.1.desc

	theme = faith
	right_portrait = root


	#Easy Quest
	option = {
		name = POD_fera_spirit_quest.1.option.a
		custom_tooltip = spirit_quest_difficulty_1
		save_scope_value_as = {
			name = quest_difficulty
			value = 1
		}
	}

	#Regular Quest
	option = {
		name = POD_fera_spirit_quest.1.option.b
		custom_tooltip = spirit_quest_difficulty_2
		save_scope_value_as = {
			name = quest_difficulty
			value = 2
		}
	}

	#Difficult Quest
	option = {
		name = POD_fera_spirit_quest.1.option.c
		custom_tooltip = spirit_quest_difficulty_3
		save_scope_value_as = {
			name = quest_difficulty
			value = 3
		}
	}

	#Hard Quest
	option = {
		name = POD_fera_spirit_quest.1.option.d
		custom_tooltip = spirit_quest_difficulty_4
		save_scope_value_as = {
			name = quest_difficulty
			value = 4
		}
	}

	#You asked for it
	option = {
		name = POD_fera_spirit_quest.1.option.e
		custom_tooltip = spirit_quest_difficulty_5
		save_scope_value_as = {
			name = quest_difficulty
			value = 5
		}
		stress_impact = {
			arrogant = minor_stress_impact_loss
		}
	}

	####Add unique quests here


	#Exit
	option = {
		name = POD_fera_spirit_quest.1.option.exit
	}


	after = {
		if = {
			limit = { exists = scope:quest_difficulty }
			POD_start_journey_from_template_effect = { TEMPLATE = scope:story }
			var:current_journey = {
				set_variable = {
					name = quest_difficulty
					value = scope:quest_difficulty
				}
			}
		}
	}
}

#Spirit gives you a riddle
POD_fera_spirit_quest.10 = {
	type = character_event
	hidden = yes


	trigger = {
		fera_spirit_quest_difficulty >= 1
	}
	weight_multiplier = {
		base = 1
	}

	immediate = {
		duel = {
			skills = { diplomacy }
			value = 5
		
			50 = {
				compare_modifier = { value = scope:duel_value multiplier = high_positive_duel_skill_multiplier }
				modifier = { add = { value = fera_spirit_quest_progress_success_value } }
				
				send_interface_message = {
					type = POD_spirit_quest_good
					desc = POD_fera_spirit_quest.10.good
					left_icon = root

					POD_journey_progress = { AMOUNT = fera_spirit_quest_difficulty_1_progress }
				}
			}
			
			50 = {
				compare_modifier = { value = scope:duel_value multiplier = low_positive_duel_skill_multiplier }
				
				send_interface_message = {
					type = POD_spirit_quest_bad
					desc = POD_fera_spirit_quest.10.bad
					left_icon = root

					add_legitimacy = -15
					stress_impact = {
						arrogant = minor_stress_impact_gain
						wrathful = minor_stress_impact_gain
						irritable = minor_stress_impact_gain
					}
				}
			}
		}
	}
}

POD_fera_spirit_quest.1000 = {
	type = character_event
	content_source = dlc_POD
	title = POD_fera_spirit_quest.1000.t
	desc = POD_fera_spirit_quest.1000.desc

	theme = faith
	right_portrait = root


	#Breed Gifts xp
	option = {
		name = POD_fera_spirit_quest.1000.option.a
		trigger = {
			OR = {
				POD_is_werewolf_trigger = yes
				POD_is_bastet_trigger = yes
			}
		}

		add_breedgifts_lifestyle_xp = fera_spirit_quest_lifestyle_reward_value
	}

	#Auspice or equivalent Gifts xp
	option = {
		name = POD_fera_spirit_quest.1000.option.b
		trigger = {
			POD_is_werewolf_trigger = yes
		}

		add_lunar_auspicegifts_lifestyle_xp = fera_spirit_quest_lifestyle_reward_value
	}

	#Tribal or equivalent Gifts xp
	option = {
		name = POD_fera_spirit_quest.1000.option.c
		trigger = {
			OR = {
				POD_is_werewolf_trigger = yes
				POD_is_bastet_trigger = yes
			}
		}

		add_tribalgifts_lifestyle_xp = fera_spirit_quest_lifestyle_reward_value
	}

	#Hengeyokai Gifts xp
	option = {
		name = POD_fera_spirit_quest.1000.option.d
		trigger = {
			faith = { has_doctrine = special_doctrine_beast_courts }
		}

		add_hengeyokaigifts_lifestyle_xp = fera_spirit_quest_lifestyle_reward_value
	}

	#Fetish TODO Spirits: Move other minor fetishes here and create new ones
	option = {
		name = POD_fera_spirit_quest.1000.option.e
		trigger = {
			NAND = {
				any_character_artifact = { has_variable = lepers_rags }
				any_character_artifact = { has_variable = trail_branch }
				any_character_artifact = { has_variable = feather_of_the_waters }
				any_character_artifact = { has_variable = heart_of_fire }
			}
		}

		add_piety = -500
		POD_static_stress_gain = { COUNT = 30 }

			25 = {
				trigger = { NOT = { any_character_artifact = { has_variable = lepers_rags } } }
				create_artifact_lepers_rags_effect = { OWNER = root }
			}
			25 = {
				trigger = { NOT = { any_character_artifact = { has_variable = trail_branch } } }
				create_artifact_trail_branch_effect = { OWNER = root }
			}
			25 = {
				trigger = { NOT = { any_character_artifact = { has_variable = feather_of_the_waters } } }
				create_artifact_feather_of_the_waters_effect = { OWNER = root }
			}
			25 = {
				trigger = { NOT = { any_character_artifact = { has_variable = heart_of_fire } } }
				create_artifact_heart_of_fire_effect = { OWNER = root }
			}
		}
	}

	#Spirit Companion TODO Spirits: Make this

			always = no #TODO Spirits

	}
