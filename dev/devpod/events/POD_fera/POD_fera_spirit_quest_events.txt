#Pick difficulty

#If adding unique quests, make sure you also give it a difficulty value
POD_fera_spirit_quest.1 = {
	type = character_event
	content_source = dlc_POD
	title = POD_fera_spirit_quest.1.t
	desc = POD_fera_spirit_quest.1.desc

	theme = faith
	right_portrait = root


	#Easy Quest
	option = {
		name = POD_fera_spirit_quest.1.option.a
		custom_tooltip = spirit_quest_difficulty_1
		save_scope_value_as = {
			name = quest_difficulty
			value = 1
		}
	}

	#Regular Quest
	option = {
		name = POD_fera_spirit_quest.1.option.b
		custom_tooltip = spirit_quest_difficulty_2
		save_scope_value_as = {
			name = quest_difficulty
			value = 2
		}
	}

	#Difficult Quest
	option = {
		name = POD_fera_spirit_quest.1.option.c
		custom_tooltip = spirit_quest_difficulty_3
		trigger = { has_access_to_spirit_wilds_trigger = yes }
		show_as_unavailable = { custom_tooltip = spirit_quest_difficulty_spirit_wilds_unavailable }
		save_scope_value_as = {
			name = quest_difficulty
			value = 3
		}
	}

	#Hard Quest TODO Spirits: enable after adding more rewards
	option = {
		name = POD_fera_spirit_quest.1.option.d
		trigger = {
			always = no
		}
		custom_tooltip = spirit_quest_difficulty_4
		save_scope_value_as = {
			name = quest_difficulty
			value = 4
		}
	}

	#You asked for it TODO Spirits: enable after adding more rewards
	option = {
		name = POD_fera_spirit_quest.1.option.e
		trigger = {
			always = no
		}
		custom_tooltip = spirit_quest_difficulty_5
		save_scope_value_as = {
			name = quest_difficulty
			value = 5
		}
		stress_impact = {
			arrogant = minor_stress_impact_loss
		}
	}

	####Add unique quests here


	#Exit
	option = {
		name = POD_fera_spirit_quest.1.option.exit
		add_piety = 250 #refund
	}


	after = {
		if = {
			limit = { exists = scope:quest_difficulty }
			POD_start_journey_from_template_effect = { TEMPLATE = scope:story }
			var:current_journey = {
				set_variable = {
					name = quest_difficulty
					value = scope:quest_difficulty
				}
			}
		}
	}
}

#Spirit gives you a riddle
POD_fera_spirit_quest.10 = {
	type = character_event
	hidden = yes

	trigger = {
		fera_spirit_quest_difficulty >= 1
	}
	weight_multiplier = {
		base = 1
		POD_fera_spirit_quest_weight_easy_modifier = yes
	}

	immediate = {
		duel = {
			skills = { diplomacy }
			value = 5
		
			50 = {
				compare_modifier = { value = scope:duel_value multiplier = high_positive_duel_skill_multiplier }
				modifier = { add = { value = fera_spirit_quest_progress_success_value } }
				
				send_interface_message = {
					type = POD_spirit_quest_good
					desc = POD_fera_spirit_quest.10.good
					left_icon = root

					POD_journey_progress = { AMOUNT = fera_spirit_quest_difficulty_1_progress }
				}
			}
			
			50 = {
				compare_modifier = { value = scope:duel_value multiplier = low_negative_duel_skill_multiplier }
				
				send_interface_message = {
					type = POD_spirit_quest_bad
					desc = POD_fera_spirit_quest.10.bad
					left_icon = root

					add_legitimacy = -15
					stress_impact = {
						arrogant = miniscule_stress_impact_gain
						wrathful = minor_stress_impact_gain
						irritable = minor_stress_impact_gain
					}
				}
			}
		}
	}
}
