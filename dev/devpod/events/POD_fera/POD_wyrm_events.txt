POD_wyrm.1 = {
	type = character_event
	content_source = dlc_POD

	desc = POD_wyrm.1.desc
	theme = spirit_world
	right_portrait = root
	
	
	#Rank xp

		name = POD_wyrm.1.option.a
		trigger = { POD_fera_rank_value < 6 }
		
		POD_fera_add_renown_effect = { TYPE = power }
		add_stress = medium_stress_gain
	}
	
	#Learn Primal Tongue
	option = {
		name = POD_wyrm.1.option.b
		trigger = { NOT = { knows_language = language_ancestor_werewolf } }

		learn_language = language_ancestor_werewolf
		add_stress = massive_stress_gain
	}

	#Urge Taints perk
	option = {
		name = POD_wyrm.1.option.c
		trigger = {
			NAND = {
				has_perk = unholy_heart_perk
				has_perk = embrace_oblivion_perk
			}
		}
		add_urgetaints_lifestyle_perk_points = 1
		add_stress = major_stress_gain
	}

	#Teach about the wyrm
	option = {
		name = POD_wyrm.1.option.d
		
		random_list = {
			45 = { add_piety = 250 }
			45 = { add_learning_lifestyle_xp = 250 }
			10 = { add_learning_skill = 1 }
		}
	}

	#Exit
	option = {
		name = POD_wyrm.1.option.e
	}
}

#Room of Loyalty
POD_wyrm.507 = {
	type = character_event
	content_source = dlc_POD
	title = POD_wyrm.507.t
	desc = POD_wyrm.507.desc
	theme = labyrinth

	trigger = { POD_is_wyrm_servant_trigger = yes }
	lower_right_portrait = {
		trigger = { exists = scope:sacrifice }
		character = scope:sacrifice
		animation = personality_coward
	}

	immediate = {
		if = {
			limit = {
				any_relation = {
					type = soulmate
					OR = {
						is_landed = no
						top_liege ?= root
					}
				}
			}
			random_relation = {
				type = soulmate
				limit = {
					OR = {
						is_landed = no
						top_liege ?= root
					}
				}
				save_scope_as = sacrifice
			}
		}
		else_if = {
			limit = {
				any_relation = {
					type = best_friend
					OR = {
						is_landed = no
						top_liege ?= root
					}
				}
			}
			random_relation = {
				type = best_friend
				limit = {
					OR = {
						is_landed = no
						top_liege ?= root
					}
				}
				save_scope_as = sacrifice
			}
		}
		else_if = {
			limit = { primary_heir ?= { is_child_of = root } }
			primary_heir = { save_scope_as = sacrifice }
		}
		else_if = {
			limit = {
				any_relation = {
					type = lover
					OR = {
						is_landed = no
						top_liege ?= root
					}
				}
			}
			random_relation = {
				type = lover
				limit = {
					OR = {
						is_landed = no
						top_liege ?= root
					}
				}
				save_scope_as = sacrifice
			}
		}
		else_if = {
			limit = {
				any_relation = {
					type = friend
					OR = {
						is_landed = no
						top_liege ?= root
					}
				}
			}
			random_relation = {
				type = friend
				limit = {
					OR = {
						is_landed = no
						top_liege ?= root
					}
				}
				save_scope_as = sacrifice
			}
		}
	}


	option = {
		name = POD_wyrm.507.option.a
		trigger = { exists = scope:sacrifice }
		scope:sacrifice = { death = { death_reason = death_sacrificed_to_gods } }
	}

	option = {
		name = POD_wyrm.507.option.b
		POD_give_semi_permanent_injury_effect = { TRAIT = maimed TIME = 15 }
	}

	option = {
		name = POD_wyrm.507.option.c
		remove_short_term_gold = monumental_gold_value
	}

	option = {
		name = POD_wyrm.507.option.d
		trigger = {
			is_landed = yes
			NOT = { has_trait = fera }
		}
		capital_county = {
			change_development_level = -10
			change_county_control = -100
		}
	}

	after = {
		POD_spawn_bane_lieutenant_effect = yes
		if = {
			limit = { POD_is_unknowing_wyrm_servant_trigger = no }
			add_piety = 500
		}
		if = {
			limit = { has_trait = fomori }
			add_possessedpower_lifestyle_perk_points = 2
		}
	}
}
