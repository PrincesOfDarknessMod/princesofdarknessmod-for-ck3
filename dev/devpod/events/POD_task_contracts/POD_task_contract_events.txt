namespace = POD_task_contracts

# Maintenance event: Remove this contract once time runs out
POD_task_contracts.1 = {
	hidden = yes
	scope = task_contract
	trigger = { exists = this }
	immediate = {
		invalidate_contract = yes
	}
}

# Maintenance event: "Un-activate" a contract by deleting and recreating it
# (needs scope:task_contract saved on the TopScope)
POD_task_contracts.2 = {
	hidden = yes
	scope = none
	immediate = {
		if = {
			limit = {
				any_character_active_contract = {
					scope:task_contract ?= this
				}
			}
			scope:task_contract = {
				invalidate_contract = yes
			}
		}
	}
}

# Maintenance event: Delete a contract and make sure it doesn't get recreated
# (needs scope:task_contract saved on the TopScope)
POD_task_contracts.3 = {
	hidden = yes
	scope = none
	immediate = {
		if = {
			limit = {
				any_character_active_contract = {
					scope:task_contract ?= this
				}
			}
			scope:task_contract = {
				set_variable = do_not_recreate
				invalidate_contract = yes
			}
		}
	}
}

# Maintenance event: Generic starting event for travel plans that started AFTER a contract was accepted
# The travel plan and contract are saved on each other. This lets the variables act as a double safeguard.
# If the travel plan is aborted, the contract gets invalidated.
# If the contract is abandoned, the travel plan gets rerouted to home.
POD_task_contracts.4 = {
	hidden = yes
	immediate = {
		current_travel_plan ?= {
			set_variable = {
				name  = task_contract
				value = scope:task_contract
			}
			scope:task_contract ?= {
				set_variable = {
					name  = travel_plan
					value = prev
				}
			}
		}
		if = {
			limit = { has_variable = traveling_with_coterie }
			every_relation = {
				type = coterie_member
				limit = { POD_can_join_journey_trigger = yes }
				prev.current_travel_plan ?= {
					add_companion = prev
				}
			}
		}
	}
}
