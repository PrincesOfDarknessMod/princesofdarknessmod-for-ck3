# Events podnagaraja.1-2

# Victory Celebration
# Triggered by: purge_serpent_india_decision effect block
podnagaraja.1 = {
	type = character_event
	content_source = dlc_POD
	window = fullscreen_event
	title = podnagaraja.1.t
	desc = podnagaraja.1.desc
	theme = vampire
	override_background = { reference = throne_room }

	immediate = {
	}

	option = {
		name = podnagaraja.1.a

		add_character_modifier = scourge_of_sutekh_modifier
		add_prestige = 1500
		dynasty = {
			add_dynasty_prestige = 1000
		}

		# Setite/Daitya rulers hate you
		hidden_effect = {
			every_ruler = {
				limit = {
					has_trait = setite
				}
				add_opinion = {
					modifier = hate_opinion
					target = scope:serpent_purger
					opinion = -100
				}
			}
		}

		# Notify other players in India/Tibet
		hidden_effect = {
			every_player = {
				limit = {
					NOT = { this = root }
					capital_province = {
						OR = {
							geographical_region = world_india
							geographical_region = world_tibet
						}
					}
				}
				trigger_event = podnagaraja.2
			}
		}
	}
}

# FEATURE 2: PILGRIMAGE TO ENOCH (Journey)
# Events podnagaraja.10-21
# Started via pilgrimage_to_enoch_decision -> Journey system
# Travel phase triggers podnagaraja.10 on arrival
#
# Flow:
#   10 (arrival/setup) -> 11 (dispatch) -> [pool 12-19] -> 11 -> ... -> 20 (final)
#   Pool: 8 events, 4-5 randomly selected per journey
#   21: Fellow Traveler spy follow-up
#####################################################

# ===== ARRIVAL & DISPATCH =====

# Arrival event — sets up the journey and chains to dispatch
# Triggered by: pilgrimage_to_enoch_travel_phase SGUI
podnagaraja.10 = {
	type = character_event
	content_source = dlc_POD
	title = podnagaraja.10.t
	desc = podnagaraja.10.desc
	theme = vampire
	override_background = { reference = wilderness }

	left_portrait = {
		character = root
		animation = personality_content
	}

	immediate = {
		POD_pause_journey_travel_plan_effect = yes
		hidden_effect = {
			# Initialize event counter
			# Local variables are confined to this event chain, which cleans them up automatically afterwards
			set_local_variable = {
				name = pilgrimage_events_fired
				value = 0
			}
			# Randomly decide how many pool events (4 or 5)
			random_list = {
				50 = {
					set_local_variable = {
						name = pilgrimage_events_target
						value = 4
					}
				}
				50 = {
					set_local_variable = {
						name = pilgrimage_events_target
						value = 5
					}
				}
			}
		}
	}

	option = {
		name = podnagaraja.10.a
		custom_tooltip = podnagaraja.10.a.tt
		hidden_effect = {
			trigger_event = {
				id = podnagaraja.12
				days = { 30 60 }
			}
		}
	}
}

# Hidden dispatch event — picks next random pool event or fires final
podnagaraja.11 = {
	hidden = yes

	trigger = {
		is_alive = yes
	}

	immediate = {
		if = {
			limit = {
				local_var:pilgrimage_events_fired >= local_var:pilgrimage_events_target
			}
			# All pool events done — fire the final Gates of Enoch event
			trigger_event = {
				id = podnagaraja.20
				days = { 60 120 }
			}
		}
		else = {
			# Pick a random unfired pool event
			random_list = {
				# Entering the Shadowlands

				# The Tempest
				10 = {
					trigger = { NOT = { has_local_variable = pilgrimage_evt_tempest } }
					set_local_variable = pilgrimage_evt_tempest
					trigger_event = {
						id = podnagaraja.13
						days = { 30 90 }
					}
				}
				# Whispers of the Drowned
				10 = {
					trigger = { NOT = { has_local_variable = pilgrimage_evt_whispers } }
					set_local_variable = pilgrimage_evt_whispers
					trigger_event = {
						id = podnagaraja.14
						days = { 30 90 }
					}
				}
				# The False Necropolis (blocked if warned by wraiths)
				10 = {
					trigger = {
						NOT = { has_local_variable = pilgrimage_evt_false_necropolis }
						NOT = { has_local_variable = nagaraja_wraith_market_warned }
					}
					set_local_variable = pilgrimage_evt_false_necropolis
					trigger_event = {
						id = podnagaraja.15
						days = { 30 90 }
					}
				}
				# A Spectral Anomaly
				10 = {
					trigger = { NOT = { has_local_variable = pilgrimage_evt_anomaly } }
					set_local_variable = pilgrimage_evt_anomaly
					trigger_event = {
						id = podnagaraja.16

# Pool Event 1: Entering the Shadowlands
podnagaraja.12 = {
	type = character_event
	content_source = dlc_POD
	title = podnagaraja.12.t
	desc = podnagaraja.12.desc
	theme = vampire
	override_background = { reference = wilderness }

	left_portrait = {
		character = root
		animation = personality_content
	}

	immediate = {
		POD_save_coterie_duelists = yes
	}

	# Option A: Focus on the path (safe)
	option = {
		name = podnagaraja.12.a
	}

	# Option B: Open senses to the dead (Learning check)
	option = {
		name = podnagaraja.12.b
		
		scope:learning_duelist = {
			duel = {
				skill = learning
				# arbitrary value, actually unused. the duel is just to display "Learning challenge"
				value = 10
				40 = {
					desc = podnagaraja.12.b.success
					modifier = {
						add = {
							value = learning
							multiply = 2
						}
					}
					# Success
					root = {
						add_character_modifier = {
							modifier = nagaraja_attuned_to_dead_modifier
							years = 1
						}
					}
				}
				60 = {
					desc = podnagaraja.12.b.failure
					modifier = {
						add = {
							value = learning
							multiply = -2
							min = -55 # at least 5% chance of failure
						}
					}
					# Failure
					root = {
						add_stress = minor_stress_gain
					}
				}
			}
		}
	}

	after = {
		hidden_effect = {
			change_local_variable = {
				name = pilgrimage_events_fired
				add = 1
			}
			trigger_event = {
				id = podnagaraja.11
				days = { 14 30 }
			}
		}
	}
}

					modifier = {
						add = {
							value = prowess

						}
					}

				}
				60 = {
					desc = podnagaraja.13.a.failure
					modifier = {

							value = prowess
							multiply = -1
							# prowess tends to be easy to increase, so no multiplier
							#multiply = -3
							min = -55 # at least 5% chance of failure
						}
					}
					# Failure: stress
					root = {
						add_stress = medium_stress_gain
					}
				}
			}
		}
	}

	# Option B: Use necromancy (Necromancy rank check)
	option = {
		name = podnagaraja.13.b
		custom_tooltip = podnagaraja.13.b.tt
		if = {
			limit = { has_trait = necromancyadvanced }
			# Rank 3+: auto-success
			custom_tooltip = podnagaraja.13.b.success
			add_prestige = 50
			add_character_modifier = {
				modifier = nagaraja_tempered_by_tempest_modifier
				years = 5
			}
		}
		else = {
			# Rank 2 or lower: 50/50
			random = {
				chance = 50
				custom_tooltip = podnagaraja.13.b.success
				add_prestige = 50
			}
			random = {
				chance = 50
				custom_tooltip = podnagaraja.13.b.failure
				add_stress = medium_stress_gain
			}
		}
	}

	# Option C: Wait for it to pass (safe)
	option = {
		name = podnagaraja.13.c
		add_stress = minor_stress_gain
	}

	after = {
		hidden_effect = {
			change_local_variable = {
				name = pilgrimage_events_fired
				add = 1
			}
			trigger_event = {
				id = podnagaraja.11
				days = { 14 30 }
			}
		}
	}
}
