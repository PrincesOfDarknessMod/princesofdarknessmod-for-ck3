namespace = PODbirthsabbat

#Hidden event calculating the two most powerful ruler of the Anarch Revolt

PODbirthsabbat.0001 = {
	hidden = yes
	scope = none
	
	immediate = {
			
		random_independent_ruler = {
			limit = {
				is_vampire_trigger = yes
				has_faith = faith:freeanarch
				NOT = { has_trait = torpor }	
#				NOT = { has_faith = faith:laibon }	
#				NOT = { has_faith = faith:pathofdevaraja }	
#				NOT = { has_faith = faith:raktasadhus }	
			}
			weight = {
				base = 1
				modifier = {
					add = 1000
					highest_held_title_tier >= tier_empire
				}													
				modifier = {
					add = 10
					highest_held_title_tier = tier_kingdom
				}
				modifier = {
					add = 100
					sub_realm_size >= 50
				}
				modifier = {
					add = 1000
					sub_realm_size >= 100
				}
				modifier = {
					add = 2000
					is_ai = no
				}
			}		
			save_scope_as = most_powerful_anarch					
		}			
		if = {
			limit = {
				exists = scope:most_powerful_anarch
			}		
			scope:most_powerful_anarch = {
				trigger_event = {
					id = PODbirthsabbat.0002
					days = 5
				}
			}
		}
		random_independent_ruler = {
			limit = {
				is_vampire_trigger = yes
				has_faith = faith:freeanarch
				NOT = { this = scope:most_powerful_anarch }
				NOT = { has_trait = torpor }	
#				NOT = { has_faith = faith:laibon }	
#				NOT = { has_faith = faith:pathofdevaraja }	
#				NOT = { has_faith = faith:raktasadhus }	
			}
			weight = {
				base = 1
				modifier = {
					add = 1000
					highest_held_title_tier >= tier_empire
				}													
				modifier = {
					add = 10
					highest_held_title_tier = tier_kingdom
				}
				modifier = {
					add = 100
					sub_realm_size >= 50
				}
				modifier = {
					add = 1000
					sub_realm_size >= 100
				}
				modifier = {
					add = 2000
					is_ai = no
					}
				}		
				save_scope_as = most_powerful_anarch2					
		}			
		if = {
			limit = {
				exists = scope:most_powerful_anarch2
			}		
			scope:most_powerful_anarch2 = {
				trigger_event = {
					id = PODbirthsabbat.0003
					days = 15
				}
			}
		}
		
		trigger_event = {
			id = PODbirthsabbat.300
			days = 60
		}
	}
}


PODbirthsabbat.0002 = {
	type = character_event
	content_source = dlc_POD
	title = PODbirthsabbat.0002.t
	desc = PODbirthsabbat.0002.desc
	theme = sabbat
	left_portrait = {
		character = scope:most_powerful_anarch
		animation = personality_honorable
	}
	
	immediate = {
		play_music_cue = "mx_cue_ck3_vocal_maintheme"
		
		hidden_effect = {
			every_living_character = {
				limit = {
					is_vampire_trigger = yes
					has_faith = faith:freeanarch	
					is_ai = yes
					NOT = { has_trait = torpor }
					NOT = { has_trait = zealous }	
#					NOT = { has_faith = faith:laibon }
#					NOT = { has_faith = faith:pathofdevaraja }	
#					NOT = { has_faith = faith:raktasadhus }				
				}
				set_character_faith = faith:pathofhonorableaccord
			}		
			every_player = {
				limit = {
					NOT = { this = scope:most_powerful_anarch }
					NOT = { this = scope:most_powerful_anarch2 }
				}		
				trigger_event = {
					id = PODbirthsabbat.0004
					days = 30
				}
			}	
		}
	}	


	option = {
		name = PODbirthsabbat.0002.a
		
		trigger = { is_vampire_trigger = yes }
		
		custom_tooltip = sabbat_begins_effect_message

		give_nickname = nick_founder_of_the_sabbat
		
		set_character_faith = faith:pathofhonorableaccord

		if = {
			limit = {
				is_tremere_trigger = yes
			}
		set_global_variable = tremere_sabbat_variable
		}

		add_character_modifier = {
			modifier = sabbat_founder_modifier
			years = 50
		}
		save_scope_as = sabbat_regent
		religion:sabbat_religion = { 
			set_variable = {
				name = regent
				value = scope:sabbat_regent
			}
		}
		
		trigger_event = {
			id = PODbirthsabbat.100
			days = 15
		}		
		
		ai_chance = {
			base = 100	
			}
		}	
		
		option = {
			name = PODbirthsabbat.0002.b
			
			custom_tooltip = sabbat_begins_effect_message
			
		ai_chance = {
			base = 0	
			}			
		}		

}
