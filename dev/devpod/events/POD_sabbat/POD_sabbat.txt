POD_sabbat.110 = {
	title = POD_sabbat.110.title
	desc = POD_sabbat.110.desc
	theme = sabbat
	override_background = {
		reference = armory
	}
	right_portrait = {
		character = scope:attacker
		animation = war_attacker
	}
	left_portrait = {
		character = scope:defender
		animation = war_defender
	}
	
	trigger = {
		scope:defender = {
			any_neighboring_and_across_water_top_liege_realm_owner = {
				NOT = { is_at_war_with = scope:defender }
				religion = religion:sabbat_religion
			}
		}
	}
	
	immediate = {
		scope:attacker = {
			random_character_war = {
				limit = { using_cb = POD_wild_hunt_war }
				save_scope_as = wild_hunt
			}
		}
	}

	option = {
		name = POD_sabbat.110.a
		
		scope:defender = {
			every_neighboring_and_across_water_top_liege_realm_owner = {
				limit = {
					NOT = { is_at_war_with = scope:defender }
					religion = religion:sabbat_religion
				}
				if = {
					limit = { is_ai = yes }
					scope:wild_hunt = {
						hidden_effect = { set_called_to = prev }
						add_attacker = prev
					}
				}
				else = {
					run_interaction = {
						interaction = wild_hunt_call_interaction
						actor = scope:attacker
						recipient = this
						execute_threshold = maybe
					}
				}
			}
		}
	}

	option = {
		name = POD_sabbat.110.b
	}
}

		}
	}

			}
		}
	}
