#Name your dragon
POD_umbra_pet.101 = {
	type = character_event
	title = POD_umbra_pet.101.t
	desc = POD_umbra_pet.101.desc
	theme = pet
	left_portrait = {
		character = root
		animation = personality_cynical
	}

	immediate = {
		play_music_cue = "mx_cue_low_key_positive"
	}

	widget = {
		gui = "event_window_widget_enter_text"
		container = "dynamic_birth_name"
		controller = {
			type = text
			data = {
				key = story_cycle_dragon_name
				default = cat_name_mite
			}
		}
		setup_scope = {
			root = { save_scope_as = text_target }
		}
	}

	option = {
		name = pet_animal.0001.a
	}

	after = {
		remove_character_flag = is_naming_dragon

		if = {
			# We entered custom text into the pop up
			limit = { NOT = { has_variable = story_cycle_dragon_name } }
			set_variable = {
				name = story_cycle_dragon_name
				value = flag:custom
			}
		}
	}
}

#Bygone must bygone
POD_umbra_pet.199 = {
	type = character_event
	title = POD_umbra_pet.199.t
	desc = POD_umbra_pet.199.desc
	theme = pet
	left_portrait = {
		character = root
		animation = sadness
	}
	artifact = {
		position = lower_right_portrait
		target = scope:newly_created_artifact
	}

	immediate = {
		play_music_cue = "mx_cue_prison"
		create_artifact_animal_hide_effect = {
			OWNER = root
			HUNTER = root
			LEGENDARY = yes
			ANIMAL = flag:dragon
		}
		scope:newly_created_artifact = { set_artifact_name = artifact_animal_hide_name_dragon }
	}

	option = {
		name = POD_umbra_pet.199.a
		if = {
			limit = { exists = scope:newly_created_artifact }
			hidden_effect = { destroy_artifact = scope:newly_created_artifact }
		}
		hidden_effect = {  # secret reward
			add_character_modifier = { modifier = blessed_by_great_celestial_dragons_modifier years = 5 } 
			add_character_flag = released_your_dragon
		}
		ai_chance = {
			base = 100
		}
	}
	
	option = {
		name = pet_animal.0199.b
		remove_short_term_gold = { 10 20 }
		show_as_tooltip = {
			scope:newly_created_artifact = { set_owner = root }
		}
		ai_chance = {
			base = 10
		}
	}

	after = {
		show_as_tooltip = {
			remove_dragon_story_modifiers_effect = yes
		}
		scope:story = {
			hidden_effect = { end_story = yes }
		}
		remove_character_flag = dragon_is_dying
	}
}
