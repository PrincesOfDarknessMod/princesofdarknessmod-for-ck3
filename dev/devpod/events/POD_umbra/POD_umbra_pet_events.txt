#Name your dragon
POD_umbra_pet.101 = {
	type = character_event
	content_source = dlc_POD
	title = POD_umbra_pet.101.t
	desc = POD_umbra_pet.101.desc
	theme = pet
	left_portrait = {
		character = root
		animation = personality_cynical
	}

	immediate = {
		play_music_cue = "mx_cue_low_key_positive"
	}

	widget = {
		gui = "event_window_widget_enter_text"
		container = "dynamic_birth_name"
		controller = {
			type = text
			data = {
				key = story_cycle_dragon_name
				default = cat_name_mite
			}
		}
		setup_scope = {
			root = { save_scope_as = text_target }
		}
	}

	option = {
		name = pet_animal.0001.a
	}

	after = {
		remove_character_flag = is_naming_dragon

		if = {
			# We entered custom text into the pop up
			limit = { NOT = { has_variable = story_cycle_dragon_name } }
			set_variable = {
				name = story_cycle_dragon_name
				value = flag:custom
			}
		}
	}
}

POD_umbra_pet.211 = {
	type = character_event
	content_source = dlc_POD
	title = POD_umbra_pet.211.t
	desc = POD_umbra_pet.211.desc
	theme = martial_chivalry_focus
	override_background = {
		reference = courtyard
	}
	left_portrait = root

	cooldown = { years = 5 }
	
	trigger = {
		NOR = { 
			has_character_modifier = unicorn_strength_modifier
			has_character_modifier = unicorn_agility_modifier
			has_character_modifier = unicorn_endurance_modifier
			has_character_modifier = unicorn_fierceness_modifier
		}
		minor_gold_value_trigger = yes
	}

	immediate = {
		if = {
			limit = {
				has_royal_court = yes
				employs_court_position = master_of_horse_court_position
				any_court_position_holder = {
					type = master_of_horse_court_position
					is_available_ai_adult = yes
				}
			}
			random_court_position_holder = {
				type = master_of_horse_court_position
				limit = { is_available_ai_adult = yes }
				save_scope_as = m_horse
			}
		}
	}

	option = { #Strength
		name = POD_umbra_pet.211.a
		remove_short_term_gold = minor_gold_value
		add_character_modifier = {
			modifier = unicorn_strength_modifier
			years = 5
		}
		ai_chance = {
			base = 50
		}
	}

	option = { #Agility
		name = POD_umbra_pet.211.b
		remove_short_term_gold = minor_gold_value
		add_character_modifier = {
			modifier = unicorn_agility_modifier
			years = 5
		}
		ai_chance = {
			base = 50
		}
	}

	option = { #Endurance
		name = POD_umbra_pet.211.c
		remove_short_term_gold = minor_gold_value
		add_character_modifier = {
			modifier = unicorn_endurance_modifier
			years = 5
		}
		ai_chance = {
			base = 50
		}
	}

	option = { #Fierceness
		name = POD_umbra_pet.211.d
		remove_short_term_gold = minor_gold_value
		add_character_modifier = {
			modifier = unicorn_fierceness_modifier
			years = 5
		}
		ai_chance = {
			base = 50
		}
	}

	option = {	#Master of Horse
		name = POD_umbra_pet.211.f
		trigger = {
			exists = scope:m_horse
		}
		custom_tooltip = POD_umbra_pet.211.f.tt
		scope:m_horse = {
			add_opinion = {
				target = root
				modifier = pleased_opinion
				opinion = 15
			}
		}
		hidden_effect = {
			random_list = {
				50 = {	# nothing happens
					modifier = {
						scope:m_horse.aptitude:master_of_horse_court_position = 4
						factor = 0.1
					}
					modifier = {
						scope:m_horse.aptitude:master_of_horse_court_position = 3
						factor = 0.2
					}
					modifier = {
						scope:m_horse.aptitude:master_of_horse_court_position = 2
						factor = 0.3
					}
					modifier = {
						scope:m_horse.aptitude:master_of_horse_court_position = 1
						factor = 0.5
					}
					send_interface_toast = {
						title = POD_umbra_pet.211.f.failure
						left_icon = scope:m_horse
						custom_tooltip = POD_umbra_pet.211.f.failure.tt
					}
				}
				5 = {	#Your master of the horse makes your horse better
					add_character_modifier = {
						modifier = unicorn_strength_modifier
						years = 5
					}
				}
				5 = {	#Your master of the horse makes your horse better
					add_character_modifier = {
						modifier = unicorn_agility_modifier
						years = 5
					}
				}
				5 = {	#Your master of the horse makes your horse better
					add_character_modifier = {
						modifier = unicorn_endurance_modifier
						years = 5
					}
				}
				5 = {	#Your master of the horse makes your horse better
					add_character_modifier = {
						modifier = unicorn_fierceness_modifier
						years = 5
					}
				}
			}
		}
		ai_chance = {
			base = 70
		}
	}

	option = { #Nah
		name = POD_umbra_pet.211.e
		custom_tooltip = POD_umbra_pet.211.e.tt
		ai_chance = {
			base = 0
			ai_value_modifier = {
				ai_greed = 0.75
			}
		}
	}
}

#Another horse-enthusiast compliments your horse
POD_umbra_pet.213 = {
	type = character_event
	content_source = dlc_POD
	title = POD_umbra_pet.213.t
	desc = POD_umbra_pet.213.desc
	theme = martial_chivalry_focus
	override_background = {
		reference = courtyard
	}
	left_portrait = {
		character = scope:horse_vassal
		animation = happiness
	}

	cooldown = { years = 5 }
	
	trigger = {
		any_vassal = {
			is_available_healthy_ai_adult = yes
			OR = {
				any_owned_story = {
					story_type = story_cycle_martial_lifestyle_warhorse
				}
				has_lifestyle = martial_lifestyle
				has_court_position = master_of_horse_court_position
			}
		}
	}

	immediate = {
		random_vassal = {
			limit = {
				is_available_healthy_ai_adult = yes
				is_powerful_vassal = yes
				any_owned_story = {
					story_type = story_cycle_martial_lifestyle_warhorse
				}
			}
			alternative_limit = {
				is_available_healthy_ai_adult = yes
				has_court_position = master_of_horse_court_position
			}
			alternative_limit = {
				is_available_healthy_ai_adult = yes
				is_powerful_vassal = yes
				has_lifestyle = martial_lifestyle
			}
			alternative_limit = {
				is_available_healthy_ai_adult = yes
				any_owned_story = {
					story_type = story_cycle_martial_lifestyle_warhorse
				}
			}
			alternative_limit = {
				is_available_healthy_ai_adult = yes
				has_lifestyle = martial_lifestyle
			}
			save_scope_as = horse_vassal
		}
	}

	option = { #Let them ride
		name = POD_umbra_pet.213.a
		reverse_add_opinion = {
			target = scope:horse_vassal
			modifier = friendliness_opinion
			opinion = 20
		}
		if = {
			limit = { #They can't handle the horse
				has_character_modifier = unicorn_fierceness_modifier
				scope:horse_vassal = {
					prowess < medium_skill_rating
				}
			}
			random = {
				chance = 50
				scope:horse_vassal = {
					increase_wounds_effect = { REASON = horse_riding_accident }
				}
			}
			custom_tooltip = POD_umbra_pet.213.a.tt
		}
		ai_chance = {
			base = 50
			ai_value_modifier = {
				ai_compassion = 0.5
			}

		}
	}

	option = { #Don't let them ride
		name = POD_umbra_pet.213.b
		add_prestige = medium_prestige_gain
		ai_chance = {
			base = 50
			ai_value_modifier = {
				ai_vengefulness = 0.25
				ai_greed = 0.25
			}

		}
	}
}
