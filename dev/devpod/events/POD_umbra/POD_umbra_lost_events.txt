POD_umbra_lost.100 = {
	type = character_event
	content_source = dlc_POD
	title = POD_umbra_lost.100.t
	desc = POD_umbra_lost.100.desc
	
	left_portrait = root
	theme = oblivion
	
	immediate = {
		set_variable = { name = spirit_location value = flag:oblivion }
	}

	option = {
		name = POD_umbra_lost.100.a
		add_internal_flag = dangerous
		
		change_first_name = "No_Character"
		remove_nickname = yes
		add_diplomacy_skill = -1000
		add_intrigue_skill = -1000
		add_learning_skill = -1000
		add_martial_skill = -1000
		add_prowess_skill = -1000
		add_stewardship_skill = -1000
		hidden_effect = { clear_traits = yes }
		add_piety = -9999999
		add_prestige = -9999999
		depose_effect = { DEPOSER = root }
		set_to_lowborn = yes
		move_to_pool = yes
		set_character_faith = faith:innocence
		every_claim = { root = { remove_claim = prev } }
		every_spouse = { divorce = root }
		every_relation = { type = friend remove_relation_friend = root }
		every_relation = { type = best_friend remove_relation_best_friend = root }
		every_relation = { type = rival remove_relation_rival = root }
		every_relation = { type = nemesis remove_relation_nemesis = root }
		every_relation = { type = lover remove_relation_lover = root }
		every_relation = { type = soulmate remove_relation_soulmate = root }
		death = { death_reason = death_existence_erased }
	}
}
