POD_umbra_lost.104 = {
	type = character_event
	content_source = dlc_POD
	title = POD_umbra_lost.104.t
	
	desc = {
		first_valid = {
			triggered_desc = {
			    trigger = {
					has_trait = demon
			    }
			    desc = POD_umbra_lost.104.desc_demon
			}
			desc = POD_umbra_lost.104.desc_infernalist
		}
	}
	
	left_portrait = root
	theme = spirit_world
	
	trigger = { 
		var:spirit_location = flag:abyss 
	}

	option = {
		name = POD_umbra_lost.104.a
		
		trigger = { has_trait = demon }
	}

	option = {
		name = POD_umbra_lost.104.b
		
		trigger = { 
			NOT = { has_trait = demon } 
			no_resurrecting_prisoner_count >= 1
		}
		
		show_as_unavailable = {
			custom_description = { 
				text = POD_infernalist_sacrifice_prisoner_interaction_valid 
				no_resurrecting_prisoner_count = 0
			}
		}
		
		random_prisoner = {
			limit = { has_POD_resurrection_trigger = no }
			trigger_event = char_interaction.0171

				VICTIM = this

			}
		}
		give_infernalist_secret_or_trait_effect = yes
		if = {
			limit = { NOT = { has_trait = demonology } }
			add_trait = demonology
		}
		add_trait_xp = { trait = demonology track = thrall value = 5 }
		
		POD_umbra_location_initiate_effect = { LOCATION = var:spirit_destination }
	}

	option = {
		name = POD_umbra_lost.104.c
		
		trigger = { NOT = { has_trait = demon } }
		
		POD_umbra_adv_clean_effect = yes
	}
