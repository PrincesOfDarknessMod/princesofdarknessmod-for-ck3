namespace = POD_umbra_movement
	
POD_umbra_movement.10 = {
	hidden = yes

	trigger = {
		has_variable = spirit_destination
	}
	
	immediate = {
		POD_change_umbra_modifier_effect = { START = var:spirit_location END = var:spirit_destination }
		
		set_variable = { name = spirit_location value = var:spirit_destination }
		POD_umbra_location_initiate_effect = { LOCATION = var:spirit_destination }
		
		remove_variable = spirit_destination
	}
}

POD_umbra_movement.20 = {
	hidden = yes
	
	immediate = {
		if = {
			limit = { var:spirit_destination = flag:oblivion }
			trigger_event = POD_umbra_lost.100
		}
		else = {
			trigger_event = {
				on_action = spirit_travel_movement_end_pulse
				days = spirit_travel_movement_time_value
			}
		}
	}
}
