namespace = POD_umbra_crisis

POD_umbra_crisis.1 = {
	hidden = yes
	
	immediate = {
		set_local_variable = {
			name = umbra_crisis_location
			value = scope:this_character.var:umbra_stationed_location
		}
		switch = {
			trigger = var:umbra_interest_type
			flag:wyrmish_summoning = {
				trigger_event = POD_umbra_crisis.100
			}
		}
	}
}

POD_umbra_crisis.100 = {
	hidden = yes
	
	immediate = {
		capital_province = {
			save_scope_as = crisis_location
		}
		trigger_event = { id = POD_umbra_crisis.101 years = umbra_interest_duration } 
	}
}

POD_umbra_crisis.101 = {
	hidden = yes
	
	immediate = {
		if = {
			limit = { is_alive = yes }
			trigger_event = POD_umbra_crisis.102
		}
		else = {
			scope:crisis_location.province_owner = { trigger_event = POD_umbra_crisis.102 }
		}
	}
}
	
POD_umbra_crisis.102 = {
	type = character_event
	content_source = dlc_POD
	title = POD_umbra_crisis.102.t
	desc = POD_umbra_crisis.102.desc
	
	left_portrait = root
	theme = wyrm_reaches
	
	trigger = { POD_umbra_crisis_valid_trigger = { LOCATION = local_var:umbra_crisis_location } }

	on_trigger_fail = { remove_list_variable = { name = umbra_secured_list target = local_var:umbra_crisis_location } }
	
	option = {
		name = POD_umbra_crisis.102.a
		
		scope:crisis_location = {
			custom_tooltip = {
				text = POD_umbra_crisis.102.a.tt
				POD_remove_common_building_effect = yes
			}
		}
		scope:this_character = {
			if = {
				limit = { 
					is_alive = yes 
					var:umbra_stationed_location ?= local_var:umbra_crisis_location
				}
				POD_umbra_remove_expedition_interest_effect = { LOCATION = local_var:umbra_crisis_location }
			}
		}
	}
}

POD_umbra_crisis.110 = {
	type = character_event
	content_source = dlc_POD
	title = POD_umbra_crisis.110.t
	desc = POD_umbra_crisis.110.desc
	
	left_portrait = root
	right_portrait = scope:spirit
	theme = wyrm_reaches
	
	immediate = {
		create_character = {
			location = root.location
			template = POD_werewolf_teaser_character
			faith = faith:wyrm
			save_scope_as = spirit
		}
	}

	option = {
		name = POD_umbra_crisis.110.a
		
		if = {
			limit = { POD_is_on_golconda_trigger = yes }
			configure_start_single_combat_effect = {
				SC_INITIATOR = scope:spirit
				SC_ATTACKER = scope:spirit
				SC_DEFENDER = root
				FATALITY = no
				FIXED = no
				LOCALE = terrain_scope
				OUTPUT_EVENT = POD_umbra_crisis.111
				INVALIDATION_EVENT = tribal.1022
			}
		}
		else = {
			configure_start_single_combat_effect = {
				SC_INITIATOR = scope:spirit
				SC_ATTACKER = scope:spirit
				SC_DEFENDER = root
				FATALITY = always
				FIXED = no
				LOCALE = terrain_scope
				OUTPUT_EVENT = POD_umbra_crisis.111
				INVALIDATION_EVENT = tribal.1022
			}
		}
	}
}

POD_umbra_crisis.111 = {
	hidden = yes
	
	immediate = {
		# Did win?
		if = {
			limit = { scope:sc_victor = scope:spirit }
			scope:sc_loser = {
				pod_death = {
					DEATH_REASON = death_duel
					KILLER = scope:sc_victor
				}
			}
		}
		# Did lose?
		else = {
			scope:sc_victor = {
				add_to_variable_list = { name = umbra_secured_list target = var:spirit_location }
			}
		}
		scope:spirit = { 
			if = {
				limit = { is_alive = yes }
				POD_disappearance_effect = yes
			}
		}
	}
}
