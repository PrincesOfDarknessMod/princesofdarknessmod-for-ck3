POD_umbra_kj.1002 = {
	type = character_event
	title = POD_umbra_kj.1002.t
	desc = {
		desc = POD_umbra_kj.1002.desc_opening
		first_valid = {
			triggered_desc = {
				trigger = {
					OR = {
						has_trait = dharma9 
						has_trait = dharma10
					}
				}
				desc = POD_umbra_kj.1002.desc_high
			}
			triggered_desc = {
				trigger = {
					OR = {
						has_trait = dharma8
						has_trait = dharma7
						has_trait = dharma6
					}
					OR = {
						in_underworld_trigger = { LOCATION = var:spirit_location }
						in_underworld_trigger = { LOCATION = var:umbra_stationed_location }
					}
				}
				desc = POD_umbra_kj.1002.desc_yin
			}
			triggered_desc = {
				trigger = {
					OR = {
						has_trait = dharma8
						has_trait = dharma7
						has_trait = dharma6
					}
				}
				desc = POD_umbra_kj.1002.desc_yang
			}
			desc = POD_umbra_kj.1002.desc_low
		}
		desc = POD_umbra_kj.1002.desc_ending
	}
	
	left_portrait = root
	theme = spirit_world
	
	cooldown = { years = 1 }
	
	trigger = {
		has_trait = kueijin
		NOT = { in_dangerous_umbra_location_trigger = { LOCATION = var:umbra_stationed_location } }
	}
	
	option = {
		name = POD_umbra_kj.1002.a
		if = {
			limit = {
				OR = {
					has_trait = dharma9 
					has_trait = dharma10
				}
			}
			add_piety = major_piety_gain
		}
		else_if = {
			limit = {
				OR = {
					has_trait = dharma6
					has_trait = dharma7
					has_trait = dharma8
				}
			}
			add_piety = medium_piety_gain
		}
		else = {
			add_piety = minor_piety_gain
		}
	}
}
