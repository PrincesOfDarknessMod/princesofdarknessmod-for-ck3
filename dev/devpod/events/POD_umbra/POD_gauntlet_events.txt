namespace = POD_gauntlet

POD_gauntlet.101 = {
	type = character_event
	content_source = dlc_POD
	title = POD_gauntlet.101.t
	desc = POD_gauntlet.101.desc
	theme = spirit_wilds

	left_portrait = {
		character = root
		animation = idle
	}
		
	trigger = { 
		POD_chi_user_trigger = no
	}

	option = {
		name = POD_gauntlet.101.a
		
		trigger = { 
			scope:target_location = { 
				NOR = { 
					has_province_modifier = gauntlet10_modifier 
					has_province_modifier = gauntlet0_modifier 
				} 
			} 
			can_manipulate_the_gauntlet_trigger = yes
		}
		
		show_as_unavailable = { scope:target_location = { has_province_modifier = gauntlet10_modifier } }
		
		POD_universal_energy_decrease_multiple_effect = { COUNT = 2 }
		POD_anger_the_local_spirit_effect = yes
		scope:target_location = { POD_increase_gauntlet_modifier_effect = yes }
	}
	
	option = {
		name = POD_gauntlet.101.b
		
		trigger = { 
			scope:target_location = { 
				NOR = { 
					has_province_modifier = gauntlet1_modifier 
					has_province_modifier = gauntlet0_modifier 
				} 
			} 
			can_manipulate_the_gauntlet_trigger = yes
		}
		
		show_as_unavailable = { scope:target_location = { has_province_modifier = gauntlet1_modifier } }
		
		POD_universal_energy_decrease_multiple_effect = { COUNT = 2 }
		random = {
			chance = 5
			POD_calm_the_local_spirit_effect = yes
		}
		scope:target_location = { POD_decrease_gauntlet_modifier_effect = yes }
	}
	
	option = {
		name = POD_gauntlet.101.c
		
		trigger = { 
			scope:target_location = { 
				NOT = { has_province_modifier = gauntlet0_modifier }
			} 
			can_tear_down_the_gauntlet_trigger = yes
		}
		
		show_as_unavailable = { scope:target_location = { has_province_modifier = gauntlet0_modifier } }

		POD_universal_energy_decrease_multiple_effect = { COUNT = 10 }
		random = {
			chance = 5
			POD_calm_the_local_spirit_effect = yes
		}
		scope:target_location = { POD_tear_down_the_gauntlet_effect = yes }
	}
	
	option = {
		name = POD_gauntlet.101.d
	}
}
