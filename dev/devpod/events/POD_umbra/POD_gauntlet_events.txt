namespace = POD_gauntlet

POD_gauntlet.101 = {
	type = character_event
	content_source = dlc_POD
	title = POD_gauntlet.101.t
	desc = POD_gauntlet.101.desc
	theme = spirit_wilds

	left_portrait = {
		character = root
		animation = idle
	}
		
	trigger = { 
		POD_middle_kingdom_splat_trigger = no
	}

	option = {
		name = POD_gauntlet.101.a
		
		trigger = { 
			scope:target_location = { 
				custom_description = { text = gauntlet_too_strong_trigger_tooltip gauntlet_level != 10 }
				custom_description = { text = gauntlet_too_weak_trigger_tooltip gauntlet_level != 0 }
			} 
			can_manipulate_the_gauntlet_trigger = yes
		}
		
		show_as_unavailable = { scope:target_location = { gauntlet_level = 10 } }
		
		POD_splat_agnostic_energy_cost = { COUNT = 2 }
		POD_anger_the_local_spirit_effect = yes
		scope:target_location = { POD_increase_gauntlet_modifier_effect = yes }
	}
	
	option = {
		name = POD_gauntlet.101.b
		
		trigger = { 
			scope:target_location = { 
				custom_description = { text = gauntlet_1_trigger_tooltip gauntlet_level != 1 }
				custom_description = { text = gauntlet_too_weak_trigger_tooltip gauntlet_level != 0 }				
			} 
			can_manipulate_the_gauntlet_trigger = yes
		}
		
		show_as_unavailable = { scope:target_location = { gauntlet_level = 1 } }
		
		POD_splat_agnostic_energy_cost = { COUNT = 2 }
		random = {
			chance = 5
			POD_calm_the_local_spirit_effect = yes
		}
		scope:target_location = { POD_decrease_gauntlet_modifier_effect = yes }
	}
	
	option = {
		name = POD_gauntlet.101.c
		
		trigger = { 
			scope:target_location = { 
				custom_description = { text = gauntlet_too_weak_trigger_tooltip gauntlet_level != 0 }
			} 
			can_tear_down_the_gauntlet_trigger = yes
		}
		
		show_as_unavailable = { scope:target_location = { gauntlet_level = 0 } }

		POD_splat_agnostic_energy_cost = { COUNT = 10 }
		random = {
			chance = 5
			POD_calm_the_local_spirit_effect = yes
		}
		scope:target_location = { POD_tear_down_the_gauntlet_effect = yes }
	}
	
	option = {
		name = POD_gauntlet.101.d
	}
}

POD_gauntlet.201 = {
	type = character_event
	content_source = dlc_POD
	title = POD_gauntlet.201.t
	desc = POD_gauntlet.201.desc
	theme = spirit_wilds
	
	left_portrait = {
		character = root
		animation = idle
	}

	option = {
		name = POD_gauntlet.201.a
		
		remove_short_term_gold = minor_gold_value
		duel = {
			skill = diplomacy
			value = high_skill_rating
			30 = {
				desc = POD_shared_loca_success
				compare_modifier = {
					value = scope:duel_value
					multiplier = high_positive_duel_skill_multiplier
				}
				POD_calm_the_local_spirit_effect = yes
			}
			70 = {
				desc = POD_shared_loca_failure
				compare_modifier = {
					value = scope:duel_value
					multiplier = low_negative_duel_skill_multiplier 
				}
			}
		}
	}

	option = {
		name = POD_gauntlet.201.b
		trigger = { has_trait = possessed }
		
		add_piety = medium_piety_loss 
		duel = {
			skill = diplomacy
			value = high_skill_rating
			50 = {
				desc = POD_shared_loca_success
				compare_modifier = {
					value = scope:duel_value
					multiplier = high_positive_duel_skill_multiplier
				}
				POD_calm_the_local_spirit_effect = yes
			}
			50 = {
				desc = POD_shared_loca_failure
				compare_modifier = {
					value = scope:duel_value
					multiplier = low_negative_duel_skill_multiplier 
				}
			}
		}
	}

	option = {
		name = POD_gauntlet.201.c
		trigger = { has_trait = truepossessed }
		
		add_prestige = minor_prestige_loss 
		POD_calm_the_local_spirit_effect = yes
	}

	option = {
		name = POD_gauntlet.201.d
	}
}
