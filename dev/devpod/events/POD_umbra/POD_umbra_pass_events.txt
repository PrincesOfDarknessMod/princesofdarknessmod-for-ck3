# Fire at start
POD_umbra_pass.101 = {
	type = character_event
	content_source = dlc_POD
	title = POD_umbra_pass.101.t
	desc = POD_umbra_pass.101.desc
	
	left_portrait = root
	theme = shadowlands
	
	trigger = {
		var:spirit_location ?= flag:shadowlands
		var:spirit_destination = flag:stormwall	
		NOT = { has_perk = enshroud_perk }
		location ?= { NOT = { has_province_modifier = nihil_modifier } }
	}

	option = {
		name = POD_umbra_pass.101.a
		
		stress_impact = {
			compassionate = major_stress_impact_gain
			just = medium_stress_impact_gain
		}
		
		location = {
			county = {
				change_county_control = -10
				change_development_progress = major_development_progress_loss	#Always deduct progress.
				if = {	#If they don't have enough progress made, burn a level of development.
					limit = { development_towards_level_increase < major_development_progress_gain }
					change_development_level = -1
				}
			}
			add_province_modifier = {
				modifier = nihil_modifier
				months = 1
			}
		}
		trigger_event = { on_action = spirit_travel_movement_end_pulse }
	}

	option = {
		name = POD_umbra_pass.101.b
		trigger = { any_character_artifact = { has_variable = instant_nihil } }
		random_character_artifact = { 
			limit = { has_variable = instant_nihil }
			destroy_artifact = this
		}
		trigger_event = { on_action = spirit_travel_movement_end_pulse }
	}

	option = {
		name = POD_umbra_pass.101.c
		remove_variable = spirit_destination
		if = {
			limit = { POD_is_currently_in_umbra_expedition_trigger = yes }
			custom_tooltip = {
				text = umbra_retreat_tt
				POD_umbra_expedition_cleanup_effect = yes
			}
		}
	}
}

POD_umbra_pass.202 = {
	type = character_event
	content_source = dlc_POD
	title = POD_umbra_pass.202.t
	desc = POD_umbra_pass.202.desc
	
	left_portrait = root
	theme = labyrinth
		
	trigger = {
		var:spirit_location ?= flag:central_duchy
		var:spirit_destination = flag:black_spiral_labyrinth
	}

	option = {
		name = POD_umbra_pass.202.a
		
		trigger = {
			POD_is_wyrm_servant_trigger = yes
		}
		
		custom_tooltip = POD_umbra_pass.202.a.tt
		POD_umbra_send_effect = { LOCATION = var:spirit_destination }
	}

	option = {
		name = POD_umbra_pass.202.b
		
		trigger = {
			POD_is_wyrm_servant_trigger = no
		}
		
		duel = {
		    skill = intrigue
			value = 40
		
			50 = {
				desc = POD_umbra_pass.202.b.duel.tt.success
				compare_modifier = {
					value = scope:duel_value
					multiplier = high_positive_duel_skill_multiplier
				}
				min = 1
				POD_umbra_send_effect = { LOCATION = var:spirit_destination }
			}    
			
			50 = {
				desc = POD_umbra_pass.202.b.duel.tt.failure
				compare_modifier = {
					value = scope:duel_value
					multiplier = low_negative_duel_skill_multiplier
				}
				min = 1
				
				if = {
					limit = { has_variable = physical_spirit_travel }
					maimed_in_battle_effect = yes
				}
				else = {
					add_trait = lunatic_1
					add_stress = massive_stress_impact_gain
				}
				
				remove_variable = spirit_destination
				if = {
					limit = { POD_is_currently_in_umbra_expedition_trigger = yes }
					custom_tooltip = {
						text = umbra_retreat_tt
						POD_umbra_expedition_cleanup_effect = yes
					}
				}
			}
		}
	}

	option = {
		name = POD_umbra_pass.202.c
		remove_variable = spirit_destination
		if = {
			limit = { POD_is_currently_in_umbra_expedition_trigger = yes }
			custom_tooltip = {
				text = umbra_retreat_tt
				POD_umbra_expedition_cleanup_effect = yes
			}
		}
	}
}
