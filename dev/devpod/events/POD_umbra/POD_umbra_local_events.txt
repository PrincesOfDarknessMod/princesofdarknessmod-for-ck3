namespace = POD_umbra_local
	
POD_umbra_local.1 = {
	scope = none
	hidden = yes
	
	immediate = {
		every_county = {
			POD_set_ai_spirit_wilds_controller_effect = { CHRACTER = county_controller }
			POD_set_ai_shadowlands_controller_effect = { CHRACTER = county_controller }
		}
		
		trigger_event = {
			id = POD_umbra_local.2
			years = 5
		}
	}
}
	
POD_umbra_local.2 = {
	scope = none
	hidden = yes
	
	immediate = {
		every_county = {
			if = {
				limit = { var:spirit_wilds_controller ?= { is_alive = no } }
				if = {
					limit = { has_variable = local_spirit_wilds_buff }
					POD_remove_local_umbra_buff_effect = {
						BUFF = var:local_spirit_wilds_buff
						VICTIM = var:spirit_wilds_controller
					}
					remove_variable = local_spirit_wilds_buff
				}
				remove_variable = spirit_wilds_controller
			}
			if = {
				limit = { NOT = { has_variable = spirit_wilds_controller } }
				POD_set_ai_spirit_wilds_controller_effect = { CHRACTER = county_controller }
			}
			else_if = {
				limit = { var:spirit_wilds_controller = { is_ai = yes } }
				if = {
					limit = { NOT = { has_variable = local_spirit_wilds_buff } }
					random_list = {
						33 = {
							trigger = { var:spirit_wilds_controller = { has_access_to_spirit_wilds_trigger = yes } }
							POD_random_local_spirit_wilds_buff_effect = yes
						}
						33 = {
							trigger = { var:spirit_wilds_controller = { has_access_to_dreaming_trigger = yes } }
							POD_random_local_dreaming_buff_effect = yes
						}
						33 = {
							trigger = { var:spirit_wilds_controller = { has_access_to_astral_reaches_trigger = yes } }
							POD_random_local_astral_reaches_buff_effect = yes
						}
					}
				}
				else = {
					random = {
						chance = 5
						random_list = {
							33 = {
								trigger = { var:spirit_wilds_controller = { has_access_to_spirit_wilds_trigger = yes } }
								POD_update_local_spirit_wilds_buff_effect = yes
							}
							33 = {
								trigger = { var:spirit_wilds_controller = { has_access_to_dreaming_trigger = yes } }
								POD_update_local_dreaming_buff_effect = yes
							}
							33 = {
								trigger = { var:spirit_wilds_controller = { has_access_to_astral_reaches_trigger = yes } }
								POD_update_local_astral_reaches_buff_effect = yes
							}
						}
					}
				}
			}
			
			if = {
				limit = { var:shadowlands_controller ?= { is_alive = no } }
				if = {
					limit = { has_variable = local_shadowlands_buff }
					POD_remove_local_umbra_buff_effect = {
						BUFF = var:local_shadowlands_buff
						VICTIM = var:shadowlands_controller
					}
					remove_variable = local_shadowlands_buff
				}
				remove_variable = shadowlands_controller
			}
			if = {
				limit = { NOT = { has_variable = shadowlands_controller } }
				POD_set_ai_shadowlands_controller_effect = { CHRACTER = county_controller }
			}
			else_if = {
				limit = { var:shadowlands_controller = { is_ai = yes } }
				if = {
					limit = { NOT = { has_variable = local_shadowlands_buff } }
					POD_random_local_shadowlands_buff_effect = yes
				}
				else = {
					random = {
						chance = 5
						POD_update_local_shadowlands_buff_effect = yes
					}
				}
			}
		}
		
		trigger_event = {
			id = POD_umbra_local.2
			years = 5
		}
	}
}

				scope:victim = {
					add_opinion = {

						opinion = {
							value = revoke_title_tyranny_gain
							multiply = -1
						}
					}
				}
				scope:target_county = { POD_umbra_change_spirit_wilds_controller_effect = { CHRACTER = root } }
			}
			70 = {

						modifier = angry_opinion

						opinion = -10
					}
				}
				add_stress = major_stress_gain
			}
		}
	}

	option = {
