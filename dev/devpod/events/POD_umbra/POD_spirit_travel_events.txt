POD_spirit_travel.10 = {
	type = character_event
	hidden = yes

	trigger = {
		has_variable = spirit_destination
	}
	
	immediate = {
		if = {
			limit = { var:spirit_location = flag:skinlands }
			send_interface_message = {
				type = character_arrived
				title = spirit_travel_arrived_title
				desc = spirit_travel_arrived_desc
				right_icon = root
				
				pay_for_spirit_travel_effect = yes
				if = {
					limit = { 
						has_POD_shapeshifted_form_trigger = yes 
						NOT = { has_perk = aspect_of_duat_perk }
					}
					POD_shapeshift_revert_effect = yes
				}
			}
			POD_add_umbra_modifier_effect = yes
		}
		else = {
			POD_set_umbra_modifier_effect = yes
		}
		
		set_variable = { name = spirit_location value = var:spirit_destination }
		POD_set_local_spirit_list_effect = yes
		
		random = {
			chance = 1
			add_character_flag = { flag = mirror_zone_pass days = 1 }
		}
		random = {
			chance = 1
			add_character_flag = { flag = null_zone_pass days = 1 }
		}
		
		remove_variable = spirit_destination
	}
}
