POD_umbra_adv.1 = {
	hidden = yes
	
	immediate = {
		if = {
			limit = { var:spirit_destination = flag:abyss }
			create_character = {
				location = root.location
				template = POD_common_demon_character
				save_scope_as = spirit
			}
		}
		else_if = {
			limit = { in_dreaming_trigger = { LOCATION = var:spirit_destination } }
			create_character = {
				location = root.location
				template = POD_common_fae_character
				save_scope_as = spirit
			}
		}
		else_if = {
			limit = { in_underworld_trigger = { LOCATION = var:spirit_destination } }
			create_character = {
				location = root.location
				template = POD_common_wraith_character
				save_scope_as = spirit
			}
		}
		else = {
			create_character = {
				location = root.location
				template = POD_common_spirit_character
				save_scope_as = spirit
			}
		}
		configure_start_single_combat_effect = {
			SC_INITIATOR = scope:spirit
			SC_ATTACKER = scope:spirit
			SC_DEFENDER = root
			FATALITY = always
			FIXED = no
			LOCALE = terrain_scope
			OUTPUT_EVENT = POD_umbra_adv.100
			INVALIDATION_EVENT = tribal.1022
		}
	}
}
