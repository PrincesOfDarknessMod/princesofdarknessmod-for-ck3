namespace = POD_umbra_adv

POD_umbra_adv.1 = {
	hidden = yes
	
	immediate = {
		if = {
			limit = { var:spirit_destination = flag:abyss }
			create_character = {
				location = root.location
				template = POD_common_demon_character
				save_scope_as = spirit
			}
		}
		else_if = {
			limit = { in_dreaming_trigger = { LOCATION = var:spirit_destination } }
			create_character = {
				location = root.location
				template = POD_common_fae_character
				save_scope_as = spirit
			}
		}
		else_if = {
			limit = { in_underworld_trigger = { LOCATION = var:spirit_destination } }
			create_character = {
				location = root.location
				template = POD_common_wraith_character
				save_scope_as = spirit
			}
		}
		else = {
			create_character = {
				location = root.location
				template = POD_common_spirit_character
				save_scope_as = spirit
			}
		}
		configure_start_single_combat_effect = {
			SC_INITIATOR = scope:spirit
			SC_ATTACKER = scope:spirit
			SC_DEFENDER = root
			FATALITY = always
			FIXED = no
			LOCALE = terrain_scope
			OUTPUT_EVENT = POD_umbra_adv.100
			INVALIDATION_EVENT = tribal.1022
		}
	}
}

#Who won the final duel?
POD_umbra_adv.100 = {
	hidden = yes

	immediate = {
		# Did win?
		if = {
			limit = { scope:sc_victor = scope:spirit }
			scope:sc_victor = { POD_disappearance_effect = yes }
		}
		# Did lose?
		else = {
			scope:sc_victor = { trigger_event = POD_umbra_adv.101 }
		} 
	}	
}

	type = character_event

	left_portrait = root
	theme = spirit_world

	option = {
		name = POD_umbra_adv.101.fallback
		
		add_prestige = medium_prestige_gain
	}
}
