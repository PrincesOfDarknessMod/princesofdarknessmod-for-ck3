POD_umbra_gp.103 = {
	type = character_event
	content_source = dlc_POD
	title = POD_umbra_gp.103.t
   	desc = POD_umbra_gp.103.desc
	theme = greatperson 
	left_portrait = {
		character = scope:gp_aristotle
		animation = personality_rational
	}
	
	immediate = {		
		create_character = {
			name = "Aristotle" 
			age = 62 
			gender = male
			dynasty = none
			learning = 10
			diplomacy = 5
			intrigue = 2
			stewardship = 6
			faith = faith:innocence
			culture = culture:classical_greek
			trait = education_learning_5
			trait = intellect_good_3
			trait = diligent
			trait = patient
			trait = gregarious
			trait = just
			trait = scholar
			trait = theologian
			trait = poet
			trait = shrewd
			trait = scholarly_court_2
			trait = lifestyle_physician
			trait = lifestyle_mystic
			trait = wraith
			random_traits = no		
			location = root.capital_province	
			save_scope_as = gp_aristotle
			after_creation = {
				set_age = current_year
				change_age = 322
				add_trait_xp = {
					trait = lifestyle_mystic
					value = 100
				}
				add_trait_xp = {
					trait = wraith
					track = fetter_resolved
					value = 90
				}
				give_nickname = nick_philosopher
				add_character_flag = POD_great_person_flag
				POD_set_resonance_effect = yes
				if = {
					limit = { has_ep1_dlc_trigger = yes }
					create_inspiration = book_inspiration
				}
				POD_wraith_initialization_effect = yes
				POD_arcanoi_initiation_effect = yes
			}
		}
		
		add_to_global_variable_list = {
			name = special_character_list
			target = flag:spawned_gp_aristotle
		}
	
		play_music_cue = "mx_cue_positive_effect"	
	}
	
	option = {
		name = POD_umbra_gp.103.a
		add_courtier = scope:gp_aristotle
		
		hidden_effect = {
			POD_masquerade_exposure_increase_effect = yes
		
			scope:gp_aristotle = {
				add_character_flag = {
					flag = blocked_from_leaving
					years = 2
				}
			}
		}
	}
	
	option = {
		name = POD_umbra_gp.103.b
		
		scope:gp_aristotle = { death = { death_reason = death_disappearance } }
	}
}

			add_stress = -1000
		}		
	}

		name = POD_umbra_gp.109.c

	}

	}

	option = {
