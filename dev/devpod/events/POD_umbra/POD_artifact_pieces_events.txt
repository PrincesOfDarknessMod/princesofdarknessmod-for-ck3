namespace = POD_artifact_pieces

POD_artifact_pieces.100 = {
	type = character_event
	content_source = dlc_POD
	title = POD_artifact_pieces.100.t
	desc = POD_artifact_pieces.100.desc
	theme = stewardship_wealth_focus
	
	left_portrait = root
	
	option = {
		name = POD_artifact_pieces.100.a
		trigger_event = POD_artifact_pieces.200
	}
	
	option = {
		name = POD_artifact_pieces.100.b
		trigger = { scope:creator = { has_perk = reliquary_perk } }
		show_as_unavailable = { always = yes }

		trigger_event = POD_artifact_pieces.300
	}
	
	option = {
		name = POD_artifact_pieces.100.c
		trigger = {
			trigger_if = {
				limit = { restore_artifact_check <= 1 }
				scope:creator = { has_perk = major_restoration_perk }
			}
			trigger_else_if = {
				limit = { restore_artifact_check = 2 }
				scope:creator = { has_perk = greater_restoration_perk }
			}
			trigger_else = {
				scope:creator = { has_perk = princely_restoration_perk }
			}
		}
		show_as_unavailable = { always = yes }
		
		custom_tooltip = POD_artifact_pieces.100.c.tt
		remove_short_term_gold = umbra_reweave_artifact_gold_value
		
		scope:creator = { 
			duel = {
				skill = stewardship
				value = root.restore_artifact_check
				50 = {
					desc = board_games.0001.tt.success
					compare_modifier = {
						value = scope:duel_value
						multiplier = high_positive_duel_skill_multiplier
					}
					root = {
						send_interface_toast = {
							title = board_games.0001.tt.success
							left_icon = root
							right_icon = scope:creator
							
							custom_tooltip = {
								text = POD_artifact_pieces.100.c.success
								create_artifact_restored_artifact_pieces_effect = { OWNER = root }
							}
						}
					}
				}
				50 = {
					desc = board_games.0001.tt.failure
					compare_modifier = {
						value = scope:duel_value
						multiplier = low_negative_duel_skill_multiplier 
					}
					root = {
						every_equipped_character_artifact = {
							limit = {
								artifact_slot_type = miscellaneous
								has_variable = artifact_pieces
							}
							destroy_artifact = this
						}
					}
				}
			}
		}
	}
	
	option = {
		name = POD_artifact_pieces.100.d
	}
}
