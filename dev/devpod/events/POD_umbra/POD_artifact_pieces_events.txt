namespace = POD_artifact_pieces

POD_artifact_pieces.100 = {
	type = character_event
	content_source = dlc_POD
	title = POD_artifact_pieces.100.t
	desc = POD_artifact_pieces.100.desc
	theme = stewardship_wealth_focus
	
	left_portrait = root
	
	option = {
		name = POD_artifact_pieces.100.b
		trigger = { scope:creator = { has_perk = reliquary_perk } }
		show_as_unavailable = { always = yes }

		trigger_event = POD_artifact_pieces.300
	}
	
	option = {
		name = POD_artifact_pieces.100.c
		trigger = { 	
			custom_tooltip = {
				text = enough_common_artifact_piece
				common_artifact_piece_reserve >= 3
			}
			scope:creator = { has_perk = major_restoration_perk } 
		}
		show_as_unavailable = { always = yes }
		
		custom_tooltip = POD_artifact_pieces.100.c.tt
		remove_short_term_gold = minor_gold_value
		
		send_interface_toast = {
			title = board_games.0001.tt.success
			left_icon = root
			right_icon = scope:creator
			
			lose_common_artifact_pieces_effect = { VALUE = 3 }
			gain_masterwork_artifact_pieces_effect = { VALUE = 1 }
		}
	}
	
	option = {
		name = POD_artifact_pieces.100.c
		trigger = { 
			custom_tooltip = {
				text = enough_masterwork_artifact_piece
				masterwork_artifact_piece_reserve >= 3
			}
			scope:creator = { has_perk = greater_restoration_perk } 
		}
		show_as_unavailable = { always = yes }
		
		custom_tooltip = POD_artifact_pieces.100.c.tt
		remove_short_term_gold = minor_gold_value
		
		send_interface_toast = {
			title = board_games.0001.tt.success
			left_icon = root
			right_icon = scope:creator
			
			lose_masterwork_artifact_pieces_effect = { VALUE = 3 }
			gain_famed_artifact_pieces_effect = { VALUE = 1 }
		}
	}
	
	option = {
		name = POD_artifact_pieces.100.c
		trigger = { 
			custom_tooltip = {
				text = enough_famed_artifact_piece
				famed_artifact_piece_reserve >= 3
			}
			scope:creator = { has_perk = princely_restoration_perk } 
		}
		show_as_unavailable = { always = yes }
		
		custom_tooltip = POD_artifact_pieces.100.c.tt
		remove_short_term_gold = minor_gold_value
		
		send_interface_toast = {
			title = board_games.0001.tt.success
			left_icon = root
			right_icon = scope:creator
					
			lose_famed_artifact_pieces_effect = { VALUE = 3 }
			gain_illustrious_artifact_pieces_effect = { VALUE = 1 }
		}
	}
	
	option = {
		name = POD_artifact_pieces.100.d
	}
}

POD_artifact_pieces.300 = {
	type = character_event
	content_source = dlc_POD
	title = POD_artifact_pieces.300.t
	desc = POD_artifact_pieces.300.desc
	theme = stewardship_wealth_focus
	
	left_portrait = root
	
	option = {
		name = POD_artifact_pieces.300.a
		trigger = { 
			custom_tooltip = {
				text = POD_artifact_pieces.300.a.invalid
				any_equipped_character_artifact = {
					artifact_slot_type = primary_armament
					NOT = { has_variable = strengthened_artifact }
				}
			}
		}
		show_as_unavailable = { always = yes }
		
		remove_short_term_gold = minor_gold_value
		random_equipped_character_artifact = {
			limit = { artifact_slot_type = primary_armament }
			set_artifact_pieces_modifier_effect = yes

		}
	}
