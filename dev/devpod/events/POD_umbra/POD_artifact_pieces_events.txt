namespace = POD_artifact_pieces

POD_artifact_pieces.100 = {
	type = character_event
	content_source = dlc_POD
	title = POD_artifact_pieces.100.t
	desc = POD_artifact_pieces.100.desc
	theme = stewardship_wealth_focus
	
	left_portrait = root
	
	option = {
		name = POD_artifact_pieces.100.b
		trigger = { scope:creator = { has_perk = reliquary_perk } }
		show_as_unavailable = { always = yes }

		trigger_event = POD_artifact_pieces.300
	}
	
	option = {
		name = POD_artifact_pieces.100.d
	}
}

POD_artifact_pieces.300 = {
	type = character_event
	content_source = dlc_POD
	title = POD_artifact_pieces.300.t
	desc = POD_artifact_pieces.300.desc
	theme = stewardship_wealth_focus
	
	left_portrait = root
	
	option = {
		name = POD_artifact_pieces.300.a
		trigger = { 
			custom_tooltip = {
				text = POD_artifact_pieces.300.a.invalid
				any_equipped_character_artifact = {
					artifact_slot_type = primary_armament
					NOT = { has_variable = strengthened_artifact }
				}
			}
		}
		show_as_unavailable = { always = yes }
		
		remove_short_term_gold = minor_gold_value
		random_equipped_character_artifact = {
			limit = { artifact_slot_type = primary_armament }
			set_artifact_pieces_modifier_effect = yes
			set_variable = strengthened_artifact
		}
	}
	
	option = {
		name = POD_artifact_pieces.300.b

		}

			limit = { artifact_slot_type = helmet }
