#Salvation through Pain
PoD_erebus.3 = {
	type = character_event
	content_source = dlc_POD
	title = PoD_erebus.3.t
	desc = PoD_erebus.3.desc
	
	left_portrait = {
		character = root
		animation = fear
	}
	theme = spirit_world
	
	cooldown = { years = 1 }
	
	trigger = {
		POD_umbra_event_available_trigger = { LOCATION = flag:erebus }
	}

	option = {
		name = PoD_erebus.3.a
		
		if = {
			limit = {
				piety > 500
			}
			increase_wounds_no_death_effect = {REASON = whipping}
		}
		else_if = {
			limit = {
				piety > 0
				piety <= 500
			}
			increase_wounds_no_death_multiple_effect = {REASON = whipping AMOUNT = 2}
		}
		else_if = {
			limit = {
				piety <= 0
			}	
			increase_wounds_no_death_multiple_effect = { REASON = whipping AMOUNT = 3 }
		}
			
	}
}
