PoD_pangaea.4 = {
	type = character_event
	content_source = dlc_POD
	title = PoD_pangaea.4.t
	desc = PoD_pangaea.4.desc
	
	left_portrait = {
		character = root
		animation = personality_bold
	}
	theme = spirit_world
	
	cooldown = { years = 1 }
	
	trigger = {
		POD_umbra_event_available_trigger = { LOCATION = flag:pangaea }
		POD_can_equip_fetish_trigger = yes
		NOR = {
			any_character_artifact = {
				has_variable = belt_of_the_giants
			}
			any_character_artifact = {
				has_variable = cloak_of_woven_mists
			}
			any_character_artifact = {
				has_variable = staff_of_life
			}
			any_character_artifact = {
				has_variable = bone_flute
			}
		}		
	}
	
	option = {
		name = PoD_pangaea.4.a
		
		remove_short_term_gold = 150

		duel = {
			skill = prowess
			value = 60
			10 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = 0.5
				}
				desc = POD_spirit_travel.4403.effect.success.desc
				
				send_interface_toast = {
					title = POD_spirit_travel.4403.success
					left_icon = root
					create_artifact_belt_of_the_giants_effect = { OWNER = root }
					create_artifact_cloak_of_woven_mists_effect = { OWNER = root }
					create_artifact_staff_of_life_effect = { OWNER = root }
					create_artifact_bone_flute_effect = { OWNER = root }
				}				
				
				min = 1
			}
			10 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = 0.5
				}

				desc = POD_spirit_travel.4403.failure.effect.desc
				
				send_interface_toast = {
					title = POD_spirit_travel.4403.failure
					left_icon = root
					add_prestige = -300
					increase_wounds_effect = {REASON = fight}
				}	
			
				min = 1
			}
		}	
	}

	option = {
		name = POD_spirit_travel.5403.e
		trigger_event = POD_umbra_expedition.200
	}	
}
