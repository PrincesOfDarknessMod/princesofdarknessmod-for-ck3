#Love is a Mountain
PoD_spires.1 = {
	type = character_event
	content_source = dlc_POD
	title = PoD_spires.1.t
	desc = PoD_spires.1.desc
	
	left_portrait = {
		character = root
		animation = flirtation
	}
	theme = spires
	
	cooldown = { years = 1 }
	
	trigger = {
		POD_umbra_event_available_trigger = { LOCATION = flag:spires }
	}
	
	option = {
		name = PoD_spires.1.a
				
		trigger = {
		 OR = {	 
		 	any_relation = {
		 		type = lover
		 		count >= 1
		 	}
		 	any_relation = {
		 		type = soulmate
		 		count >= 1
		 	}			
		 }
		 }
		 
		 custom_tooltip = love_spires_tt
	}
	
	option = {
		name = PoD_spires.1.b
				
		trigger = {
		 OR = {	 
		 	any_relation = {
		 		type = lover
		 		count < 1
		 	}
		 	any_relation = {
		 		type = soulmate
		 		count < 1
		 	}			
		 }
		 }
		 
		 add_stress = medium_stress_gain
		 custom_tooltip = unlove_spires_tt
		 
	}	
}
