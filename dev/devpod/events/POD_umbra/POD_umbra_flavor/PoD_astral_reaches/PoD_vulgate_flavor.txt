#The River of Language
PoD_vulgate.1 = {
	type = character_event
	content_source = dlc_POD
	title = PoD_vulgate.1.t
	desc = PoD_vulgate.1.desc
	
	left_portrait = {
		character = root
		animation = personality_content
	}
	theme = spires
	
	cooldown = { years = 1 }
	
	trigger = {
		POD_umbra_event_available_trigger = { LOCATION = flag:vulgate }
		has_ep1_dlc_trigger = yes
		OR = {
			NOT = { knows_language = language_punic }
			NOT = { knows_language = language_ancient_egyptian }
			NOT = { knows_language = language_coptic }
		}
	}

	option = {
		name = PoD_vulgate.1.a
		
		trigger = {
			NOT = { knows_language = language_punic }
		}
		
		learn_language	= language_punic
		add_stress = medium_stress_gain
		
	}

	option = {
		name = PoD_vulgate.1.b
		
		trigger = {
			NOT = { knows_language = language_ancient_egyptian }
		}
		
		learn_language	= language_ancient_egyptian
		add_stress = medium_stress_gain
						
	}
	
	option = {
		name = PoD_vulgate.1.c
		
		trigger = {
			NOT = { knows_language = language_coptic }
		}
		
		learn_language	= language_coptic
		add_stress = medium_stress_gain
						
	}

	option = {
		name = PoD_vulgate.1.d
						
	}	
}
