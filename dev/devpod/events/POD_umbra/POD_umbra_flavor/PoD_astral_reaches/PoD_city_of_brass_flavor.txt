PoD_city_of_brass.1 = {
	type = character_event
	content_source = dlc_POD

	desc = PoD_city_of_brass.1.desc
	
	left_portrait = {
		character = root
		animation = worry
	}
	theme = spires
	
	cooldown = { years = 1 }
	
	trigger = {
		POD_umbra_event_available_trigger = { LOCATION = flag:city_of_brass }
	}
	
	option = {
		name = PoD_city_of_brass.1.a
		
	duel = {
	skill = diplomacy
	value = 20
		10 = {
			compare_modifier = {
				value = scope:duel_value

			}
		desc = PoD_city_of_brass.1.effect.success.desc
		
		send_interface_toast = {
			title = PoD_city_of_brass.1.success
			left_icon = root	
		}		
		min = 1		
		
	}
		10 = {
			compare_modifier = {
				value = scope:duel_value
				multiplier = 0.5
			}

				title = PoD_city_of_brass.1.failure
				left_icon = root

				}

		}
	}
		
	}

}

PoD_city_of_brass.200 = {
	type = character_event
	content_source = dlc_POD
	title = PoD_city_of_brass.200.t
	desc = PoD_city_of_brass.200.desc
	
	left_portrait = {
		character = root
		animation = personality_greedy
	}
	theme = spires
	
	option = {
		name = PoD_city_of_brass.200.a
		
		trigger = { 
		NOT = {
		any_character_artifact = {
			has_variable = book_of_the_spirits_secrets
		}
		}		
		}
		
		remove_short_term_gold = 200
		
		if = {
			limit = { has_ep1_dlc_trigger = yes }
			create_artifact_book_of_the_spirits_secrets_effect = { OWNER = this TYPE = book }
		}
		else = {
			create_artifact_book_of_the_spirits_secrets_effect = { OWNER = this TYPE = miscellaneous_when_not_court }
		}
			
	}
	
	option = {
		name = PoD_city_of_brass.200.b
		
		trigger = { 
		NOT = {
		any_character_artifact = {
			has_variable = feathered_cloak
		}
		}		
		}
		
		remove_short_term_gold = 200
		
		create_artifact_feathered_cloak_effect = { OWNER = root }
			
	}

	option = {
		name = PoD_city_of_brass.200.c
		
		trigger = { 
		NOT = {
		any_character_artifact = {
			has_variable = tangling_lance
		}
		}		
		}
		
		remove_short_term_gold = 200
		
		create_artifact_tangling_lance_effect = { OWNER = root }
			
	}

	option = {
		name = PoD_city_of_brass.200.d
		
		trigger = { 
		NOT = {
		any_character_artifact = {
			has_variable = time_piece
		}
		}		
		}
		
		remove_short_term_gold = 200
		
		create_artifact_time_piece_effect = { OWNER = root }
			
	}

	option = {
		name = POD_spirit_travel.5403.e
		trigger_event = POD_umbra_expedition.200
	}	
}
