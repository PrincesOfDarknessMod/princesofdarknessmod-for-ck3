		character = root
		animation = personality_callous
	}
	theme = wyrm_reaches
	
	cooldown = { years = 1 }
	
	trigger = {
		POD_umbra_event_available_trigger = { LOCATION = flag:wyrm_reaches }

		add_prestige = -200
			
	}	
}
