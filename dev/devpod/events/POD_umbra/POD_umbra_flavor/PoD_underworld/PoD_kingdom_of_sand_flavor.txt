	content_source = dlc_POD

	desc = PoD_kingdom_of_sand.1.desc
	
	left_portrait = {
		character = root
		animation = personality_content
	}
	theme = dark_kingdom
	
	cooldown = { years = 1 }
	
	trigger = {
		POD_umbra_event_available_trigger = { LOCATION = flag:dark_kingdom_of_sand }
		has_trait = mummy
	}
	
	option = {
		name = PoD_kingdom_of_sand.1.a
				
		trigger = {
			NOT = { culture = culture:ancient_egyptian }

		set_culture = culture:ancient_egyptian
		add_prestige = 200
	}
	
	option = {
		name = PoD_kingdom_of_sand.1.b

			culture = culture:ancient_egyptian 
		}
		
		random_list = {
			50 = { add_prestige = 200 }
			50 = { add_stress = medium_stress_gain }
		}
	}

	option = {
		name = PoD_kingdom_of_sand.1.c		

		trigger = {
			NOT = { culture = culture:ancient_egyptian } 
		}
		
		add_stress = medium_stress_gain
	}	
}
