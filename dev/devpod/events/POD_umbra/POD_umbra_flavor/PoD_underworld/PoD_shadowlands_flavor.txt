		animation = worry

	desc = PoD_shadowlands.2.desc
	
	left_portrait = {
		character = root
		animation = personality_rational
	}
	theme = shadowlands

	trigger = {
		POD_umbra_event_available_trigger = { LOCATION = flag:shadowlands }
	}
	
	immediate = {
		random_culture_global = {
			limit = {
				NOT = { this = root.culture }

		name = PoD_shadowlands.2.a
		
		set_culture = scope:randomculture

		name = PoD_shadowlands.2.b
		
	}
}
