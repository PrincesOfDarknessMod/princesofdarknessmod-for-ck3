POD_custom_nickname.100 = {
	type = character_event
	content_source = dlc_POD
	title = POD_custom_nickname.100.t
	desc = POD_custom_nickname.100.desc
	theme = court
	left_portrait = {
		character = root

	}

	immediate = {
		if = {
			limit = {
				has_any_nickname = yes

	widget = {
		gui = "POD_event_window_widget_enter_nickname"
		container = "dynamic_birth_name"

			type = text
			data = {
				key = POD_custom_nickname_text

					localization_key = nick_POD_custom_nickname1
				}
				#default = {
				#	localization_key = nick_POD_custom_nickname2
				#}
			}
		}
		setup_scope = {
			root = { save_scope_as = text_target }
		}
	}

	option = { 
		name = POD_custom_nickname.100.a
	}

	after = {
		set_variable = {
			name = POD_custom_nickname_text
			value = flag:custom
		}
		hidden_effect = {
			set_nickname_effect = { NICKNAME = nick_POD_custom_nickname2 }
			add_character_flag = POD_custom_nickname
		}
		store_localized_text_in_death = POD_custom_nickname_text
