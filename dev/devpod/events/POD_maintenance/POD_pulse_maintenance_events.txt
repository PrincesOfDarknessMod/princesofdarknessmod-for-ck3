# Global yearly maintenance, called from yearly_global_pulse on_action

				}
			}

				chance = 1
				modifier = {
					factor = 0.25
					current_date < 1300.1.1
				}

					add = 3
					current_date > 1350.1.1
				}
				modifier = {
					add = 5
					current_date > 1400.1.1
				}

					add = 5
					current_date > 1450.1.1
				}

					name = visceratika_created
					value = yes
				}

					limit = { character:790870 = { is_alive = yes } }

						add_trait = visceratika
						save_scope_as = visceratika_creator
					}
				}
				else = {
					random_ruler = {
						limit = { is_gargoyle_trigger = yes }
						alternative_limit = { is_tremere_trigger = yes }
						alternative_limit = { is_vampire_trigger = yes }
						add_trait = visceratika
						save_scope_as = visceratika_creator
					}
				}
				every_living_character = {
					limit = { is_gargoyle_trigger = yes }
					add_trait = visceratika
				}
				every_player = {
					trigger_event = pod1230objectives.137
				}
			}
		}
		
		trigger_event = POD_ashirra.2
		trigger_event = POD_talmahera.2
		trigger_event = POD_patron_council.2
		trigger_event = PODquincunx.2
		
		if = {
			limit = { POD_has_conqueror_ai_game_rule = yes }
			# bi-monthly conqueror maintenance
			trigger_event = { id = POD_maintenance.301 }
			trigger_event = { id = POD_maintenance.301 days = 60 }
			trigger_event = { id = POD_maintenance.301 days = 120 }
			trigger_event = { id = POD_maintenance.301 days = 180 }
			trigger_event = { id = POD_maintenance.301 days = 240 }
			trigger_event = { id = POD_maintenance.301 days = 300 }
		}
	}
}

# Yearly maintenance event for player characters
# Replacement for old on_actions since they don't advance the RNG
POD_maintenance.100 = {
	hidden = yes
	
	trigger = {
		is_ai = no
	}

	immediate = {
		### Set up QUARTERLY maintenance events
		trigger_event = { id = POD_maintenance.101 days =  45 }
		trigger_event = { id = POD_maintenance.101 days = 135 }
		trigger_event = { id = POD_maintenance.101 days = 225 }
		trigger_event = { id = POD_maintenance.101 days = 315 }

		### Yearly chance to get a nushi quest
		trigger_event = POD_nushi_duties.1

		### Yearly chance (per demonic master) to get a demon quest
		trigger_event = POD_demon_demand.1

		### Yearly downgrade of fetter as shadow
		if = { limit = { has_trait = shadow } trigger_event = { id = POD_wraith_maintenance.2 months = { 1 10 } } }

		every_character_artifact = {
			limit = { has_variable = max_energy_stored NOT = { has_artifact_modifier = energy_vessel_modifier } }
			while = {
				count = var:max_energy_stored
				add_artifact_modifier = energy_vessel_modifier
			}
		}
		### Yearly heal from arcana
		if = { limit = { POD_can_be_healed_trigger = yes has_perk = arcana_regeneration_perk } 
			remove_injury_trait_effect = yes 
			remove_disease_trait_effect = yes
			remove_supernatural_disease_trait_effect = yes
		}
	}
}

# Quarterly maintenance event for player characters
POD_maintenance.101 = {
	hidden = yes
	
	trigger = {
		is_ai = no
	}

	immediate = {
		### Set up MONTHLY maintenance events
		trigger_event = { id = POD_maintenance.102 }
		trigger_event = { id = POD_maintenance.102 months = 1 }
		trigger_event = { id = POD_maintenance.102 months = 2 }

		### Set up QUARTERLY maintenance on_actions
		trigger_event = { on_action = POD_coterie_maintenance_quarterly_pulse }
		trigger_event = { on_action = POD_umbra_location_event_random_pulse }
		trigger_event = { on_action = POD_umbra_lost_maintenance_quarterly_pulse }
		trigger_event = { on_action = POD_umbra_stationed_maintenance_quarterly_pulse }

		### Stress gain for unfinished quests
		if = {
			limit = { has_nushi_duty_trigger = yes }
			send_interface_message = {
				type = event_generic_bad
				title = POD_nushi_duty_unfinished
				custom_tooltip = POD_nushi_duty_unfinished.tt
				add_stress = minor_stress_gain
			}
		}
		if = {
			limit = { has_demon_demand_trigger = yes }
			send_interface_message = {
				type = event_generic_bad
				title = POD_demon_demand_unfinished
				custom_tooltip = POD_demon_demand_unfinished.tt
				add_stress = minor_stress_gain
			}
		}

		### Quarterly passive wound healing (doesn't trigger for AI)
		if = {
			limit = {
				OR = {
					has_perk = regeneration_perk
					has_perk = fomori_regeneration_perk
					has_perk = gorgon_regeneration_perk
					has_perk = kami_regeneration_perk
					has_perk = numina_healing_6_perk
					has_trait = drone
				}
				POD_can_be_healed_trigger = yes
			}
			remove_trait = flagellant
			remove_injury_trait_effect = yes
		}
		
        if = { 
			limit = { 
				has_perk = biocontrol_1_perk
				has_trait = wounded	
				POD_can_be_healed_trigger = yes
			}
			remove_trait = flagellant
			random = {
				chance = 33
				modifier = {
					add = 33
					has_perk = biocontrol_2_perk
				}
				modifier = {
					add = 34
					has_perk = biocontrol_3_perk
				}
				send_interface_toast = {
					title = health.0001.desc
					if = { limit = { has_trait = wounded_1 } remove_trait = wounded_1 }
					else = { change_trait_rank = { trait = wounded rank = -1 } }
				}
			}
		}

		### Herd members regenerate blood every quarter
		every_relation = {
			type = herd_member
			limit = { POD_splat_has_hunger_trigger = yes }
			POD_remove_hunger_modifiers_effect = yes
			add_character_modifier = bloodhunger0_modifier
		}

		### Enchanted Glamour decay every quarter
		if = {
			limit = {
				has_trait = enchanted
				NOT = { has_trait = kinain }
				NOT = { any_equipped_character_artifact = { has_variable = caliburn } }
			}
			POD_glamour_decrease_effect = yes
		}

		if = { limit = { tyranny >= 25 } POD_pulse_dark_passion = { TYPE = dark_passion_cruelity } }

		# MUMMY
		if = { limit = { has_variable = POD_command_effigy_gold }  
			send_interface_message = {
				type = POD_mummy_effigy
				title = POD_command_effigy_gold
				add_gold = command_effigy_gold
				POD_hekau_cost_effect = { COUNT = 2 }
			}
		}
		if = { limit = { has_variable = POD_command_effigy_prestige }  
			send_interface_message = {
				type = POD_mummy_effigy
				title = POD_command_effigy_prestige
				add_prestige = command_effigy_prestige
				POD_hekau_cost_effect = { COUNT = 2 }
			}
		}
		if = { limit = { has_variable = POD_command_effigy_piety }  
			send_interface_message = {
				type = POD_mummy_effigy
				title = POD_command_effigy_piety
				add_piety = command_effigy_piety
				POD_hekau_cost_effect = { COUNT = 2 }
			}
		}
		if = { limit = { has_variable = POD_command_effigy_renown }  
			send_interface_message = {
				type = POD_mummy_effigy
				title = POD_command_effigy_renown
				dynasty = { add_dynasty_prestige = PREV.command_effigy_renown }
				POD_hekau_cost_effect = { COUNT = 2 }
			}
		}
		### Quaterly heal from arcana
		if = { limit = { POD_can_be_healed_trigger = yes has_perk = arcana_increased_health_perk } 
			remove_injury_trait_effect = yes 
			remove_disease_trait_effect = yes
			remove_supernatural_disease_trait_effect = yes
		}
	}
}

# Monthly maintenance event for player characters
POD_maintenance.102 = {
	hidden = yes
	
	trigger = {
		is_ai = no
	}

	immediate = {
		### Set up MONTHLY maintenance on_actions
		#trigger_event = { on_action = POD_dark_passion_maintenance_monthly_pulse }
		trigger_event = { on_action = POD_umbra_maintenance_monthly_playable_pulse }

		if = {
			limit = { has_trait = wraith }
			POD_wraith_monthly_tick = yes	
		}

		### For powers that shouldn't trigger for imprisoned players
		if = {
			limit = { is_imprisoned = no }
			# Remove Wounds
			trigger_event = POD_werewolves_maintenance.6
			# Mana regen
			trigger_event = POD_maintenance.14
		}

		if = {
			limit = { has_variable = current_journey }
			POD_journey_monthly_tick = yes
		}

		if = { limit = { has_perk = augment_perk } 
			add_stress = -10
		}
		if = { limit = { has_perk = ebon_fountain_perk } 
			add_stress = -10
		}
		if = { limit = { has_perk = name_of_sekhem_perk } 
			add_stress = -5
		}
		if = { limit = { has_perk = name_of_the_will_perk } 
			add_stress = -7
		}
		
		if = {
			limit = {
				has_POD_vamp_blood_trigger = yes
				domicile ?= { has_domicile_building_or_higher = vampire_stable_01 }
				NOT = { has_character_modifier = excessive_animal_modifier }
			}
			# coupling this to the specific buildings instead of the building flags is probably faster
			# not like the flags are stackable anyway (unless we wanna use terrain modifier jank instead)
			if = {
				limit = { domicile = { has_domicile_building = vampire_stable_06 } } 
				add_animal_lifestyle_xp = 30
			}
			else_if = {
				limit = { domicile = { has_domicile_building = vampire_stable_05 } } 
				add_animal_lifestyle_xp = 25
			}
			else_if = {
				limit = { domicile = { has_domicile_building = vampire_stable_04 } } 
				add_animal_lifestyle_xp = 20
			}
			else_if = {
				limit = { domicile = { has_domicile_building = vampire_stable_03 } } 
				add_animal_lifestyle_xp = 15
			}
			else_if = {
				limit = { domicile = { has_domicile_building = vampire_stable_02 } } 
				add_animal_lifestyle_xp = 10
			}
			else_if = {
				limit = { domicile = { has_domicile_building = vampire_stable_01 } } 
				add_animal_lifestyle_xp = 5
			}
		}
	}
