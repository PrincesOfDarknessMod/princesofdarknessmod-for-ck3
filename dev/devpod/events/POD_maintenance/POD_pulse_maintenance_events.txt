# Yearly maintenance event for player characters
# Replacement for old on_actions since they don't advance the RNG
POD_maintenance.100 = {
	hidden = yes

	trigger = {
		is_ai = no
	}

	immediate = {
		### Set up BIANNUAL maintenance events
		trigger_event = { id = POD_maintenance.103 days = 100 }
		trigger_event = { id = POD_maintenance.103 days = 200 }

		### Set up QUARTERLY maintenance events
		trigger_event = { id = POD_maintenance.101 days =  45 }
		trigger_event = { id = POD_maintenance.101 days = 135 }
		trigger_event = { id = POD_maintenance.101 days = 225 }
		trigger_event = { id = POD_maintenance.101 days = 315 }

		trigger_event = { on_action = POD_umbra_stationed_mokole_yearly_player_pulse }

		### Yearly downgrade of fetter as shadow
		if = { limit = { has_trait = shadow } trigger_event = { id = POD_wraith_maintenance.2 months = { 1 10 } } }

		if = {
			limit = { has_perk = shintai_the_flow_of_worship_perk }
			every_relation = {
				type = coterie_member
				every_known_secret = {
					limit = { NOT = { is_known_by = root } }
					reveal_to = root
				}
			}
		}

		if = { 
			limit = { has_perk = shintai_eyes_in_the_blood_perk }
			POD_heal_permanent_injury_effect = { TRAIT = blind }
			POD_heal_permanent_injury_effect = { TRAIT = one_eyed }
		}

		### Infernalism: Demon Sabotage
		POD_bound_demon_sabotage_effect = yes

		if = {
			limit = { has_trait = ghoul }
			POD_vitae_dependency_pulse_effect = yes
			# VERY unlikely edge case fix for players who had an existing bound ghoul as an heir, and inherit.
			# normally, you get the first ghoul mission in podintroductions.2002
			trigger_event = {
				id = POD_task_contracts.6
				days = 1
			}
		}
		if = {
			limit = {
				OR = {
					has_trait = dhampir
					has_trait = revenant
				}
				NOT = { has_trait = ghoul }
			}
			POD_blood_income_effect = { COUNT = 1 }
		}

		POD_sorcery_fascination_pulse_effect = yes
	}
}

# Monthly maintenance event for player characters
POD_maintenance.102 = {
	hidden = yes

	trigger = {
		is_ai = no
	}

	immediate = {
		### Set up MONTHLY maintenance on_actions
		#trigger_event = { on_action = POD_dark_passion_maintenance_monthly_pulse }
		trigger_event = { on_action = POD_umbra_maintenance_monthly_playable_pulse }

		if = { limit = { has_trait = wraith } POD_static_stress_gain = { COUNT = wraith_monthly_stress_gain } }

		if = {
			limit = { employs_court_position = cult_leader_court_position }
			POD_cult_leader_pulse_effect = yes
		}

		### For powers that shouldn't trigger for imprisoned players
		if = {
			limit = { is_imprisoned = no }
			# Mana regen
			trigger_event = POD_maintenance.14
		}

		if = {
			limit = { has_variable = current_journey }
			POD_journey_monthly_tick = yes
		}

		# For Safety
		remove_variable = saved_stress
		remove_variable = saved_stress_level

		# MUMMY
		if = {
			limit = {
				OR = {
					has_character_modifier = POD_command_effigy_interaction_gold
					has_character_modifier = POD_command_effigy_interaction_prestige
					has_character_modifier = POD_command_effigy_interaction_piety
					has_character_modifier = POD_command_effigy_interaction_renown
				}
			}
			POD_static_stress_gain = { COUNT = 2 }
		}

		if = { limit = { has_perk = ebon_fountain_perk }
			add_stress = -10
		}
		if = { limit = { has_perk = name_of_sekhem_perk }
			add_stress = -5
		}
		if = { limit = { has_perk = name_of_the_will_perk }
			add_stress = -7
		}
		
		if = {
			limit = {
				has_perk = shintai_festival_night_perk
				POD_has_coterie_trigger = yes
			}
			save_scope_value_as = {
				name = stress_gain
				value = {
					value = coterie_size
					subtract = 1
					multiply = -1
				}
			}
			POD_static_stress_gain = { COUNT = scope:stress_gain }
		}

		if = {
			limit = {
				has_POD_vamp_blood_trigger = yes
				domicile ?= { has_domicile_building_or_higher = vampire_stable_01 }
				POD_is_hungerless_trigger = no
				NOT = { has_character_modifier = excessive_animal_modifier }
				NAND = {
					is_kishu_trigger = yes
					POD_suppress_clan_curse_trigger = no
				}
			}
			POD_save_resonance_gain_multiplier = yes
			add_animal_lifestyle_xp = {
				# coupling this to the specific buildings instead of the building flags is probably faster
				# not like the flags are stackable anyway (unless we wanna use terrain modifier jank instead)
				domicile = {
					switch = {
						trigger = has_domicile_building
						vampire_stable_06 = { value = 20 }
						vampire_stable_05 = { value = 15 }
						vampire_stable_04 = { value = 10 }
						vampire_stable_03 = { value = 7 }
						vampire_stable_02 = { value = 5 }
						vampire_stable_01 = { value = 3 }
					}
				}
				if = {
					limit = { has_perk = gift_of_apis_perk }
					multiply = 1.5
				}
				multiply = scope:resonance_gain_multiplier
			}
		}
		
		if = {
			limit = { has_global_variable = POD_marauder_mode }
			POD_marauder_mode_shuffle_stats = yes
		}
	}
}
