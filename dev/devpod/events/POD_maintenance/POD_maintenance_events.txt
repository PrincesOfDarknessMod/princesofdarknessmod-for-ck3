#Incompatible Guest are kicked out of courts
#Also just kicks out torpored/do not disturb guests
POD_maintenance.4 = {
	hidden = yes

	trigger = {
		OR = {	
			POD_custom_incompatible_splats_trigger = { ACTOR = scope:host RECIPIENT = scope:guest }
			scope:guest = {
				OR = {
					has_character_flag = POD_do_not_disturb_flag
					has_trait = torpor 
				}
			}
			AND = {
				scope:host = { primary_title = { is_mercenary_company = yes } } 
				scope:guest = { NOT = {	has_trait = mortal } }
			}
		}
	}

	immediate = {	
		scope:guest = { 
			select_and_move_to_pool_effect = yes
		}
	}
}

#Incompatible landholders are ousted when win a war 
POD_maintenance.6 = {
	hidden = yes
	
	trigger = { 
		scope:vassal = { is_ruler = yes }	
		scope:vassal = { # Fix for inquisition attack
			days_as_ruler > 1 
			NOT = { is_at_war_with = root }
		}
		POD_custom_incompatible_splats_trigger = { ACTOR = root RECIPIENT = scope:vassal }
	}
		

	immediate = {
		scope:vassal = {
			every_held_title = {
				save_scope_as = target_title
				if = {
					limit = { tier <= tier_county }
					if = {
						limit = { is_capital_barony = no } # we'll use the county in that case
						add_to_list = target_title_counties
					}
					else = {
						county = { add_to_list = target_title_counties }
					}
				}
				else = {
					root = { destroy_title = prev }
				}
			}
			
			create_title_and_vassal_change = {
				type = revoked
				save_scope_as = change
				add_claim_on_loss = yes
			}
			
			every_in_list = {
				list = target_title_counties
				change_title_holder = {
					holder = root
					change = scope:change
				}
			}
			
			resolve_title_and_vassal_change = scope:change		
		}
	}
}

# Incompatible building clean
POD_maintenance.9 = {
	hidden = yes

	trigger = {
	}

	immediate =	{	
		if = {
			limit = {
				is_landed = yes
				has_trait = mortal
				can_hold_title_as_mortal_trigger = no
				is_ai = yes # TODO MORTAL: trigger mortal maintenance event otherwise
			}
			if = {
				limit = {
					NOT = { exists = father } 
					NOT = { exists = mother } 
				}
				remove_trait = mortal
			}
			else = {
				create_character = {
					save_temporary_scope_as = random_usurper
					location = root.capital_province
					culture = root.culture
					faith = root.faith
					gender = root
					age = { 18 50 }
					random_traits = yes
					after_creation = {
						POD_character_initialization_for_generated_baron_effect = yes
					}
				}
				create_title_and_vassal_change = {
					type = usurped
					save_scope_as = title_change
					add_claim_on_loss = no
				}
				scope:title = {
					change_title_holder = {
						holder = scope:random_usurper
						change = scope:title_change
					}
				}
				resolve_title_and_vassal_change = scope:title_change
			}
		}
		POD_character_initialization_for_generated_baron_effect = yes
		scope:title = { 
			if = {
				limit = {
					NOT = { exists = scope:random_usurper }
					exists = title_province
					title_province.province_owner ?= root
				}
				if = {
					limit = { tier = tier_county }
					scope:title.title_province = {
						POD_remove_incompatible_building_effect = yes
						POD_replace_special_building_effect = yes
					}
					every_county_province = {
						limit = {
							is_county_capital = no 
							barony.holder ?= root
						}
						POD_remove_incompatible_building_effect = yes
					}
				}
				else_if = {
					limit = { tier = tier_barony }
					title_province = { POD_remove_incompatible_building_effect = yes }
				}
			}
		}
	}
}

# Building aid
POD_maintenance.12 = {
	hidden = yes
	
	trigger = {
		is_landed = yes # No merc
		is_ai = yes
	}
		
	immediate = {
		every_held_county = {
			if = {
				limit = {
					duchy.title_capital_county = this 
					title_province = { 
						NOT = { has_building_with_flag = fully_upgraded_duchy_capital_building }
						has_ongoing_construction = no
						is_grand_city_location_trigger = no
					}
				}
				title_province = { 
					if = {
						limit = { years_from_game_start < 50 }
						POD_upgrade_ducal_building_effect = { MAX = 2 }
					}
					else_if = {
						limit = { years_from_game_start < 100 }
						POD_upgrade_ducal_building_effect = { MAX = 2 }
					}
					else_if = {
						limit = { years_from_game_start < 150 }
						POD_upgrade_ducal_building_effect = { MAX = 3 }
					}
					else = {
						POD_upgrade_ducal_building_effect = { MAX = 4 }
					}
				}
			}
			every_county_province = {
				limit = {
					trigger_if = {
						limit = { years_from_game_start < 50 }
						num_buildings <= 1 # Actually 0
					}
					trigger_else_if = {
						limit = { years_from_game_start < 100 }
						num_buildings <= 2 # Actually 1
					}
					trigger_else_if = {
						limit = { years_from_game_start < 150 }
						num_buildings <= 3 # Actually 2
					}
					trigger_else_if = {
						limit = { years_from_game_start < 200 }
						num_buildings <= 4 # Actually 3
					}
					trigger_else = {
						free_building_slots > 0
					}
				}
				generate_building = yes
			}
		}
	}
}

#Control event to see if need to change overextension law
POD_maintenance.20 = {
	hidden = yes
	#orphan = yes

	trigger = {
		is_ai = no
		NOT = { has_game_rule = pod_overextension_off }
		years_from_game_start >= 10
		NOT = { government_has_flag = government_is_landless_adventurer }
	}
	
	immediate = {
		if = {
			limit = {
				sub_realm_size <= 50
				NOT = { has_realm_law = overextension_0 }
			}
			trigger_event = POD_maintenance.18
		}
		else_if = {
			limit = {
				sub_realm_size > 50
				sub_realm_size <= 100
				NOT = { has_realm_law = overextension_1 }
			}
			trigger_event = POD_maintenance.18
		}
		else_if = {
			limit = {
				sub_realm_size > 100
				sub_realm_size <= 150
				NOT = { has_realm_law = overextension_2 }
			}
			trigger_event = POD_maintenance.18
		}
		else_if = {
			limit = {
				sub_realm_size > 150
				NOT = { has_realm_law = overextension_3 }
			}
			trigger_event = POD_maintenance.18
		}	
		else = {}	
	}
}
