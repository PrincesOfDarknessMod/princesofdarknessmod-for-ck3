						location = root.capital_province
						template = POD_mortal_choleric_character

				}
				25 = {
					create_character = {
						location = root.capital_province
						template = POD_mortal_sanguine_character
						save_scope_as = wraith_one
					}
				}
			}

			#Turn mortal into a Wraith and make sure they're in the skinlands so they can be interacted with
			scope:wraith_one = {
				POD_become_wraith_effect = yes

					remove_trait = torpor
