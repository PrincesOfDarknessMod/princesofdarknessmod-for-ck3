######################################
# Arrival dispatcher - POD_gratiano.0159
# Every pact journey arrives here. The partner var outlives journey teardown
# so 0151 can recover the partner for the next-night rite.
######################################

POD_gratiano.0159 = {
	type = character_event
	hidden = yes

	immediate = {
		if = {
			limit = {
				exists = var:current_journey.var:target
				NOT = { POD_vampire_is_defeated = { CHARACTER = var:current_journey.var:target } }
				var:current_journey.var:target = {
					is_assamite_trigger = yes
					is_landed = yes
					is_imprisoned = no
				}
			}
			var:current_journey.var:target = { save_scope_as = pact_partner }
			set_variable = {
				name = POD_gratiano_pact_partner
				value = scope:pact_partner
				days = 60
			}
			if = {
				limit = {
					NOR = {
						has_character_flag = POD_gratiano_assamite_pact
						has_character_flag = POD_gratiano_pact_gold_paid
					}
				}
				trigger_event = POD_gratiano.0150
			}
			else_if = {
				limit = {
					has_character_flag = POD_gratiano_pact_gold_paid
					NOT = { has_character_flag = POD_gratiano_assamite_pact }
				}
				trigger_event = POD_gratiano.0152
			}
			else = { trigger_event = POD_gratiano.0153 }
		}
		else = { trigger_event = POD_gratiano.0154 }
	}
}

######################################
# The audience - POD_gratiano.0150
# First trip; gold seals the pact on the return trip, blood seals it immediately.
######################################

POD_gratiano.0150 = {
	type = character_event
	content_source = dlc_POD
	title = POD_gratiano.0150.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:pact_partner ?= { this = character:790530 }
					# A deposed Nakurtum gets the neutral audience.
					title:e_assamites.holder ?= { this = character:790530 }
				}
				desc = POD_gratiano.0150.desc.nakurtum
			}
			desc = POD_gratiano.0150.desc
		}
	}
	theme = cainite

	left_portrait = { character = root animation = personality_bold }
	right_portrait = { character = scope:pact_partner animation = personality_rational }

	immediate = {
		# Allow calls outside the arrival dispatcher.
		if = {
			limit = { NOT = { exists = scope:pact_partner } }
			var:current_journey.var:target ?= { save_scope_as = pact_partner }
		}
	}

	option = {
		name = POD_gratiano.0150.a
		trigger = { gold >= POD_gratiano_pact_price_value }
		remove_short_term_gold = POD_gratiano_pact_price_value
		add_character_flag = POD_gratiano_pact_gold_paid
		custom_tooltip = POD_gratiano.0150.a.tt
		hidden_effect = {
			send_interface_message = {
				type = event_generic_neutral
				title = POD_gratiano_pact_gold_paid_msg
				left_icon = root
				right_icon = scope:pact_partner
				custom_tooltip = POD_gratiano_pact_gold_paid_msg_tt
			}
		}
	}

	option = {
		name = POD_gratiano.0150.b
		custom_tooltip = POD_gratiano.0150.b.tt
		# Hide seal_pact's +1 support tooltip; this option grants the full 3.
		hidden_effect = {
			POD_gratiano_seal_pact_effect = yes
			set_variable = { name = POD_gratiano_assamite_support value = 3 }
		}
		add_character_flag = POD_gratiano_hearts_blood_promised
		trigger_event = { id = POD_gratiano.0151 days = 1 }
		trigger_event = { id = POD_gratiano.0180 days = { 7 14 } }
	}

	option = {
		name = POD_gratiano.0150.c
	}

	after = { POD_end_journey_effect = yes }
}

######################################
# The pawn-tier choice - POD_gratiano.0181
# Called synchronously from on_phase_completed with scope:frame_scheme.
######################################

POD_gratiano.0181 = {
	type = character_event
	content_source = dlc_POD
	title = POD_gratiano.0181.t
	desc = POD_gratiano.0181.desc
	theme = dominate

	left_portrait = { character = root animation = scheme }
	right_portrait = { character = scope:frame_scheme.scheme_target_character animation = worry }

	trigger = { exists = scope:frame_scheme }

	option = {
		name = POD_gratiano.0181.a
		trigger = {
			# The recruit pool retains dead members; size alone cannot guarantee a pawn.
			any_in_list = {
				variable = POD_gratiano_recruit_pool
				is_alive = yes
			}
		}
		custom_tooltip = POD_gratiano.0181.a.tt
		hidden_effect = {
			random_in_list = {
				variable = POD_gratiano_recruit_pool
				limit = { is_alive = yes }
				save_scope_as = spent_pawn
			}
		}
		if = {
			limit = { exists = scope:spent_pawn }
			POD_gratiano_remove_from_pool_effect = { TARGET = scope:spent_pawn }
			scope:frame_scheme = { add_scheme_progress = POD_gratiano_frame_pawn_progress_value }
			hidden_effect = {
				random_list = {
					75 = {
						scope:spent_pawn = { death = { death_reason = death_vanished } }
						send_interface_toast = {
							type = event_generic_neutral
							title = POD_gratiano.0181.a.spent_toast
							left_icon = scope:spent_pawn
							custom_tooltip = POD_gratiano.0181.a.spent_toast_tt
						}
					}
					25 = {
						scope:spent_pawn = {
							if = {
								limit = { can_set_relation_rival_trigger = { CHARACTER = root } }
								set_relation_rival = { target = root reason = rival_usurper }
							}
							move_to_pool = yes
						}
						send_interface_toast = {
							type = event_generic_bad
							title = POD_gratiano.0181.a.escape_toast
							left_icon = scope:spent_pawn
							custom_tooltip = POD_gratiano.0181.a.escape_toast_tt
						}
					}
				}
			}
		}
	}

	option = {
		name = POD_gratiano.0181.b
		trigger = { POD_has_coterie_members_trigger = yes }
		custom_tooltip = POD_gratiano.0181.b.tt
		hidden_effect = {
			var:coterie = {
				random_in_list = {
					variable = members
					limit = {
						NOT = { this = root }
						is_alive = yes
					}
					save_scope_as = spent_friend
				}
			}
		}
		if = {
			limit = { exists = scope:spent_friend }
			scope:spent_friend = { POD_leave_coterie_effect = yes }
			scope:frame_scheme = {
				add_scheme_progress = POD_gratiano_frame_coterie_progress_value
				add_scheme_modifier = { type = outside_help_modifier }
			}
			hidden_effect = {
				scope:spent_friend = { move_to_pool = yes }
				send_interface_toast = {
					type = event_generic_neutral
					title = POD_gratiano.0181.b.toast
					left_icon = scope:spent_friend
					custom_tooltip = POD_gratiano.0181.b.toast_tt
				}
			}
		}
	}

	option = {
		name = POD_gratiano.0181.c
		custom_tooltip = POD_gratiano.0181.c.tt
		POD_gratiano_change_trust_effect = { DELTA = -5 }
		scope:frame_scheme = { add_scheme_progress = POD_gratiano_frame_self_progress_value }
	}

	option = {
		name = POD_gratiano.0181.d
	}
}
