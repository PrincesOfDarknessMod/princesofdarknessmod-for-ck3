# 0003: exposure at zero Trust; forced strike or purge.
POD_gratiano.0003 = {
	type = character_event
	content_source = dlc_POD
	title = POD_gratiano.0003.t
	desc = POD_gratiano.0003.desc
	theme = lasombra_coa

	left_portrait = {
		character = root
		animation = fear
	}

	trigger = {
		is_ai = no
		POD_gratiano_arc_active_trigger = yes
		POD_gratiano_trust_value <= 0
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:POD_gratiano_strike_the_castle
			}
		}
	}

	immediate = {
		# Start the lockout on delivery, as in 0002.
		hidden_effect = {
			add_character_flag = {
				flag = POD_gratiano_exposure_cooldown
				days = 3650		# [TEAM-TUNABLE] re-exposure lockout
			}
		}
	}

	option = {
		name = POD_gratiano.0003.a
		trigger = { POD_gratiano_conspiracy_value >= 80 }
		add_internal_flag = dangerous
		custom_tooltip = POD_gratiano_forced_strike_tt
		if = {
			limit = { POD_gratiano_strike_doomed_trigger = yes }
			custom_tooltip = POD_gratiano_doomed_strike_warning_tt
		}
		hidden_effect = {
			# The Strike decision can commit an assault while this window is open. Recheck on click.
			if = {
				limit = {
					NOT = {
						is_target_in_global_variable_list = {
							name = unavailable_unique_decisions
							target = flag:POD_gratiano_strike_the_castle
						}
					}
				}
				add_character_flag = POD_gratiano_forced_strike
				add_to_global_variable_list = {
					name = unavailable_unique_decisions
					target = flag:POD_gratiano_strike_the_castle
				}
				trigger_event = {
					id = POD_gratiano.0220
					days = { 3 5 }
				}
				send_interface_message = {
					type = event_generic_bad
					title = POD_gratiano_forced_strike_message

			}

				send_interface_message = {
					type = event_generic_neutral
					title = POD_gratiano_strike_already_committed_message
					left_icon = root
				}
			}
		}
	}

	option = {
		name = POD_gratiano.0003.b
		add_internal_flag = dangerous
		custom_tooltip = POD_gratiano_face_purge_tt
		hidden_effect = {
			trigger_event = {
				id = POD_gratiano.0004
				days = { 5 10 }
			}
			send_interface_message = {
				type = event_generic_bad
				title = POD_gratiano_purge_coming_message
				left_icon = root
			}
		}
	}

	option = {
		name = POD_gratiano.0003.c
		custom_tooltip = POD_gratiano_flee_court_tt
		hidden_effect = {
			add_character_flag = {
				flag = POD_gratiano_crash_fled
				days = 60
			}

				id = POD_gratiano.0004
				days = { 5 10 }
			}

				type = event_generic_bad
				title = POD_gratiano_purge_coming_message

# 0004: purge, scheduled by the exposure event (0003).
POD_gratiano.0004 = {
	type = character_event
	content_source = dlc_POD
	title = POD_gratiano.0004.t
	desc = POD_gratiano.0004.desc
	theme = lasombra_coa

	left_portrait = {
		character = root
		animation = fear
	}

	trigger = {
		is_ai = no
		POD_gratiano_arc_active_trigger = yes
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:POD_gratiano_strike_the_castle
			}
		}
	}

	immediate = {
		# Judgment uses Conspiracy before the reset.
		save_scope_value_as = {
			name = pod_gratiano_crash_heat
			value = POD_gratiano_conspiracy_value
		}
		hidden_effect = {
			# The post-crash Trust restore would instantly re-trip the low-Trust warning.
			add_character_flag = {
				flag = POD_gratiano_trust_warning_cooldown
				days = 730
			}
			# Defer clearing the pool until all member fates resolve.
			every_in_list = {
				variable = POD_gratiano_recruit_pool
				limit = { is_alive = yes }
				random_list = {
					50 = {
						move_to_pool = yes
					}
					30 = {
						death = { death_reason = death_disappearance }
					}
					20 = {
						set_relation_rival = { reason = rival_historical target = root }
					}
				}
			}
			clear_variable_list = POD_gratiano_recruit_pool
			POD_gratiano_change_conspiracy_effect = { DELTA = -100 }
		}
	}

	option = {
		name = POD_gratiano.0004.a
		trigger = { NOT = { has_character_flag = POD_gratiano_crash_fled } }
		add_internal_flag = dangerous
		custom_tooltip = POD_gratiano_purge_faced_tt
		add_prestige = -1000		# [TEAM-TUNABLE] heavy standing loss
		add_character_modifier = {
			modifier = POD_gratiano_exposed_conspirator_modifier
			years = 10
		}
		hidden_effect = {
			POD_gratiano_change_trust_effect = { DELTA = 25 }		# [TEAM-TUNABLE] chastened floor
			random = {
				chance = 10		# [TEAM-TUNABLE] the court's judgment: Final Death
				modifier = {
					add = 10
					scope:pod_gratiano_crash_heat >= 60
				}
				death = { death_reason = death_final_death }
			}
			if = {
				limit = { is_alive = yes }
				send_interface_message = {
					type = event_generic_bad
					title = POD_gratiano_purge_survived_message
					left_icon = root
				}
			}
		}
	}

	option = {
		name = POD_gratiano.0004.b
		trigger = { has_character_flag = POD_gratiano_crash_fled }
		custom_tooltip = POD_gratiano_purge_fled_tt
		add_prestige = -1500		# [TEAM-TUNABLE]
		add_character_modifier = {
			modifier = POD_gratiano_exposed_conspirator_modifier
			years = 20
		}
		hidden_effect = {
			POD_gratiano_change_trust_effect = { DELTA = 10 }		# [TEAM-TUNABLE]
			send_interface_message = {
				type = event_generic_bad
				title = POD_gratiano_purge_fled_message
				left_icon = root
			}
		}
	}
}

					chance = POD_gratiano_academy_rumor_chance_value

				limit = {
					OR = {

							variable = POD_gratiano_recruit_pool

		name = POD_gratiano.0021.a
		custom_tooltip = POD_gratiano.0021.a.tt

			POD_gratiano_change_conspiracy_effect = {
