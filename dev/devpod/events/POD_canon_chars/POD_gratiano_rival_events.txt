					add_character_flag = POD_gratiano_bargain_accepted

				70 = {
					modifier = { add = 25  has_character_flag = POD_gratiano_giovanni_bargain_refused }
				}
			}
			trigger_event = { id = POD_gratiano.0065 days = { 30 60 } }
		}

# 0065: embassy answer, 30-60 days after 0064.
# Acceptance can outlive Augustus's control of Veneto.
POD_gratiano.0065 = {
	type = character_event
	title = POD_gratiano.0065.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_character_flag = POD_gratiano_bargain_accepted }
				desc = POD_gratiano.0065.desc_yes
			}
			desc = POD_gratiano.0065.desc_no
		}
	}
	theme = giovanni_coa
	content_source = dlc_POD
	left_portrait = { character = character:505018 animation = personality_greedy }

	trigger = { character:505018 ?= { is_alive = yes  is_independent_ruler = yes  has_title = title:c_veneto } }

	immediate = {
		hidden_effect = {
			# Guilelmo (505099) has first claim to the ceded lands.
			if = {
				limit = { character:505099 ?= { is_alive = yes  has_title = title:d_venice } }
				character:505099 = { save_scope_as = POD_gratiano_beneficiary }
			}
			else = { save_scope_as = POD_gratiano_beneficiary }
		}
	}

	option = {
		name = POD_gratiano.0065.a
		trigger = {
			has_character_flag = POD_gratiano_bargain_accepted
			# This option names the Prince of Venice as beneficiary.
			character:505099 ?= { is_alive = yes  has_title = title:d_venice }
		}
		remove_short_term_gold = POD_gratiano_giovanni_bargain_cost_value
		character:505018 = { add_gold = POD_gratiano_giovanni_bargain_cost_value }
		POD_gratiano_augustus_exile_effect = yes
		hidden_effect = {
			remove_character_flag = POD_gratiano_bargain_accepted
			trigger_event = POD_gratiano.0066
			if = {
				limit = {
					has_character_flag = POD_gratiano_ambrogino_bought
					character:790140 ?= { is_alive = yes }
				}
				trigger_event = { id = POD_gratiano.0081 days = { 30 60 } }
			}
		}
	}
	option = {
		name = POD_gratiano.0065.b
		trigger = { has_character_flag = POD_gratiano_bargain_accepted }
		remove_short_term_gold = POD_gratiano_giovanni_bargain_cost_value
		character:505018 = { add_gold = POD_gratiano_giovanni_bargain_cost_value }
		hidden_effect = { save_scope_as = POD_gratiano_beneficiary }
		POD_gratiano_augustus_exile_effect = yes
		hidden_effect = {
			remove_character_flag = POD_gratiano_bargain_accepted
			trigger_event = POD_gratiano.0066
			if = {
				limit = {
					has_character_flag = POD_gratiano_ambrogino_bought
					character:790140 ?= { is_alive = yes }
				}
				trigger_event = { id = POD_gratiano.0081 days = { 30 60 } }
			}
		}
	}
	# On account: Augustus leaves now and calls the debt in later (0079) at the price stored here.
	option = {
		name = POD_gratiano.0065.d
		trigger = { has_character_flag = POD_gratiano_bargain_accepted }
		character:505018 = { add_hook = { type = majorboon_hook target = root } }
		POD_gratiano_augustus_exile_effect = yes
		custom_tooltip = POD_gratiano.0065.account_tt
		hidden_effect = {
			# Store the price now; the bargain cost value changes with later flags and modifiers.
			set_variable = {
				name = POD_gratiano_giovanni_debt
				value = {
					value = POD_gratiano_giovanni_bargain_cost_value
					multiply = 2
				}
			}
			add_character_flag = POD_gratiano_bargain_on_account
			remove_character_flag = POD_gratiano_bargain_accepted
			trigger_event = POD_gratiano.0066
			trigger_event = { id = POD_gratiano.0079 days = { 1095 2190 } }
			if = {
				limit = {
					has_character_flag = POD_gratiano_ambrogino_bought
					character:790140 ?= { is_alive = yes }
				}
				trigger_event = { id = POD_gratiano.0081 days = { 30 60 } }
			}
		}
	}
	option = {
		name = POD_gratiano.0065.c
		trigger = { NOT = { has_character_flag = POD_gratiano_bargain_accepted } }
		add_character_flag = POD_gratiano_giovanni_bargain_refused
		hidden_effect = { trigger_event = { id = POD_gratiano.0078 days = { 60 120 } } }
	}
}

POD_gratiano.0067 = {
	type = character_event
	title = POD_gratiano.0067.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:POD_gratiano_witness }
				desc = POD_gratiano.0067.desc
			}
			desc = POD_gratiano.0067.desc_no_witness
		}
		triggered_desc = {
			trigger = { exists = scope:POD_gratiano_laura }
			desc = POD_gratiano.0067.desc_laura
		}
	}
	theme = giovanni_coa
	content_source = dlc_POD
	left_portrait = { character = root animation = scheme }
	right_portrait = {
		trigger = { exists = scope:POD_gratiano_witness }
		character = scope:POD_gratiano_witness
		animation = personality_cynical
	}

	trigger = { character:505018 ?= { is_alive = yes } }
	immediate = {
		add_character_flag = POD_gratiano_dossier_started
		if = {
			limit = { character:790965 ?= { is_alive = yes } }
			character:790965 = { save_scope_as = POD_gratiano_witness }
		}
		# .desc_laura names her as ruler of Grosseto, so offer her only while she holds it.

			limit = { character:792157 ?= { is_alive = yes  has_title = title:c_grosseto } }
			character:792157 = { save_scope_as = POD_gratiano_laura }
		}
	}

	option = {
		name = POD_gratiano.0067.a
		trigger = { exists = scope:POD_gratiano_witness }
		remove_short_term_gold = 150		# [TEAM-TUNABLE C-4]
		add_character_flag = POD_gratiano_dossier_witness
		custom_tooltip = POD_gratiano.0067.witness_tt
		hidden_effect = { trigger_event = { id = POD_gratiano.0068 days = { 90 180 } } }
	}
	# The protection flag is read by 0070 and cleared by POD_gratiano_giovanni_reeval_effect.

		name = POD_gratiano.0067.c
		trigger = { exists = scope:POD_gratiano_laura }

		add_character_flag = POD_gratiano_laura_protected
		custom_tooltip = POD_gratiano.0067.laura_tt

		name = POD_gratiano.0067.b
		custom_tooltip = POD_gratiano.0067.forge_tt
		hidden_effect = { trigger_event = { id = POD_gratiano.0068 days = { 30 60 } } }

	hidden = yes

	trigger = {
		this = character:505144
		is_ai = no
		NOT = { has_character_flag = POD_gratiano_nahir_dealt }
	}

	immediate = {
		if = {
			limit = { POD_gratiano_nahir_objective_done_trigger = yes }
			add_character_flag = POD_gratiano_nahir_dealt
			remove_character_flag = POD_gratiano_rival_in_focus
			if = {
				limit = { has_character_flag = POD_gratiano_nahir_grudge }
				remove_character_flag = POD_gratiano_nahir_grudge
			}
			if = {
				limit = { has_character_modifier = POD_gratiano_nahir_hunted_modifier }
				remove_character_modifier = POD_gratiano_nahir_hunted_modifier
			}
			send_interface_toast = {
				type = event_toast_effect_good
				title = POD_gratiano_nahir_done_toast
				left_icon = root
			}
			trigger_event = { id = POD_gratiano.0056 days = 7 }
		}
		else = {
			if = {
				limit = {
					has_character_flag = POD_gratiano_nahir_activated
					NOT = { has_character_flag = POD_gratiano_nahir_grudge }
				}
				# Preserve completed amends when retrying a reconciliation lost to torpor.
				if = {
					limit = { POD_gratiano_nahir_amends_value >= 3 }
					trigger_event = { id = POD_gratiano.0140 days = 30 }
				}
				# Keep one audience queued at a time.
				else_if = {
					limit = { NOT = { has_character_flag = POD_gratiano_nahir_offer_pending } }
					add_character_flag = { flag = POD_gratiano_nahir_offer_pending days = 730 }
					trigger_event = { id = POD_gratiano.0136 days = 730 }
				}
			}
			if = {
				limit = { has_character_flag = POD_gratiano_nahir_grudge }
				trigger_event = { id = POD_gratiano.0144 days = 30 }
			}
			trigger_event = { id = POD_gratiano.0148 days = 365 }
		}
	}
