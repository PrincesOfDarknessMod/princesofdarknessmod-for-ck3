# Acceptance can outlive Augustus's control of Veneto.
POD_gratiano.0065 = {
	type = character_event
	title = POD_gratiano.0065.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_character_flag = POD_gratiano_bargain_accepted }
				desc = POD_gratiano.0065.desc_yes
			}
			desc = POD_gratiano.0065.desc_no
		}
	}
	theme = giovanni_coa
	content_source = dlc_POD
	left_portrait = { character = character:505018 animation = personality_greedy }

	trigger = { character:505018 = { is_alive = yes  is_independent_ruler = yes  has_title = title:c_veneto } }

	immediate = {
		hidden_effect = {
			# Guilelmo (505099) has first claim to the ceded lands.
			if = {
				limit = { character:505099 ?= { is_alive = yes  has_title = title:d_venice } }
				character:505099 = { save_scope_as = POD_gratiano_beneficiary }
			}
			else = { save_scope_as = POD_gratiano_beneficiary }
		}
	}

	option = {
		name = POD_gratiano.0065.a
		trigger = {
			has_character_flag = POD_gratiano_bargain_accepted
			# This option names the Prince of Venice as beneficiary.
			character:505099 ?= { is_alive = yes  has_title = title:d_venice }
		}
		remove_short_term_gold = POD_gratiano_giovanni_bargain_cost_value
		POD_gratiano_augustus_exile_effect = yes
		hidden_effect = {
			remove_character_flag = POD_gratiano_bargain_accepted
			trigger_event = POD_gratiano.0066
		}
	}
	option = {
		name = POD_gratiano.0065.b
		trigger = { has_character_flag = POD_gratiano_bargain_accepted }
		remove_short_term_gold = POD_gratiano_giovanni_bargain_cost_value
		hidden_effect = { save_scope_as = POD_gratiano_beneficiary }
		POD_gratiano_augustus_exile_effect = yes
		hidden_effect = {
			remove_character_flag = POD_gratiano_bargain_accepted
			trigger_event = POD_gratiano.0066
		}
	}
	option = {
		name = POD_gratiano.0065.c
		trigger = { NOT = { has_character_flag = POD_gratiano_bargain_accepted } }
		add_character_flag = POD_gratiano_giovanni_bargain_refused
