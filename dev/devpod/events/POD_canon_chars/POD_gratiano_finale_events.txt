# 0220: assault launcher, called by the strike decision or 0003.a's forced strike.
POD_gratiano.0220 = {
	type = character_event
	content_source = dlc_POD
	title = POD_gratiano.0220.t
	desc = POD_gratiano.0220.desc
	theme = lasombra_coa

	left_portrait = {
		character = root
		animation = personality_bold
	}

	trigger = {
		is_ai = no
		NOT = { has_global_variable = variable_POD_lasombra_dead }
		# The strike decision and 0003.a can both reach this launcher.
		NOT = { has_character_flag = POD_gratiano_assault_launched }
	}

	immediate = {
		hidden_effect = {
			add_character_flag = POD_gratiano_assault_launched
			# Blocks energy restoration until strike_concluded; expiry releases the
			# lock if the vault relay stalls.
			add_character_flag = {
				flag = POD_gratiano_castle_strike_underway
				days = 365
			}
		}
		# The assault lacks co-travel context; the non-FORCE wrapper would select root for every duelist.
		POD_force_save_coterie_duelists = yes
	}

	option = {
		name = POD_gratiano.0220.a

		send_interface_message = {
			type = event_generic_neutral
			title = POD_gratiano.0220.launch
			left_icon = root
			custom_tooltip = POD_gratiano.0220.launch.tt
		}

		hidden_effect = {
			if = {
				limit = { POD_vampire_can_show_up_in_events = { CHARACTER = character:505149 } }
				trigger_event = { id = POD_gratiano.0221 days = 15 }
			}
			if = {
				limit = { POD_vampire_can_show_up_in_events = { CHARACTER = character:790210 } }
				trigger_event = { id = POD_gratiano.0222 days = 30 }
			}
			if = {
				limit = { POD_vampire_can_show_up_in_events = { CHARACTER = character:505145 } }
				trigger_event = { id = POD_gratiano.0223 days = 45 }
			}
			if = {
				limit = { POD_vampire_can_show_up_in_events = { CHARACTER = character:790449 } }
				trigger_event = { id = POD_gratiano.0224 days = 60 }
			}
			if = {
				limit = { POD_vampire_can_show_up_in_events = { CHARACTER = character:790573 } }
				trigger_event = { id = POD_gratiano.0225 days = 75 }
			}
			if = {
				limit = { POD_vampire_can_show_up_in_events = { CHARACTER = character:505070 } }
				trigger_event = { id = POD_gratiano.0226 days = 90 }
			}
			trigger_event = { id = POD_gratiano.0227 days = 105 }
		}
	}
}

# 0221: Boukephos, first assault stage from 0220.

	type = character_event

		}
	}
	theme = lasombra_coa

		character = character:505149
		animation = personality_cynical
	}

		is_ai = no

					title = POD_gratiano.0221.critical.failure

######################################
# Karotos duel outcome - POD_gratiano.0228
# Duel callbacks can run on the defender; use named scopes throughout.
# Fatality is already resolved; a draw leaves both alive.
######################################

POD_gratiano.0228 = {
	type = character_event
	hidden = yes
	content_source = dlc_POD

	immediate = {
		if = {
			limit = {
				exists = scope:sc_victor
				scope:sc_victor = scope:sc_attacker
			}
			scope:sc_attacker = {
				send_interface_toast = {
					title = POD_gratiano.0223.success
					left_icon = this
					right_icon = character:505145
				}
			}
		}
		else_if = {
			# The champion-failure loc requires martial_duelist, absent from this branch.
			limit = { NOT = { exists = scope:sc_victor } }
			scope:sc_attacker = {
				send_interface_toast = {
					title = POD_gratiano.0223.b.draw
					left_icon = this
					right_icon = character:505145
					custom_tooltip = POD_gratiano.0223.b.draw.tt
				}
			}
		}
	}
}

		}
		else = {
			scope:sc_attacker = { POD_gratiano_montano_escapes_effect = yes }
		}
	}

				send_interface_toast = {
					title = POD_gratiano.0291.a.success
					left_icon = this
					right_icon = scope:assamite_blade
				}
				scope:assamite_blade ?= {
					death = { death_reason = death_battle killer = scope:sc_attacker }
				}
				POD_set_or_change_variable = { VAR = POD_gratiano_assassins_slain VAL = 1 }
				if = {
					limit = { POD_gratiano_assassins_slain_value >= 3 }
					trigger_event = { id = POD_gratiano.0292 days = { 10 30 } }
				}
			}
		}
		# Only a concluded loss is fatal; sc_finished guard as in 0273.
		else_if = {
			limit = { exists = scope:sc_finished }
			# Dispose of the blade before killing the callback recipient.
			scope:assamite_blade ?= { move_to_pool = yes }
			scope:sc_attacker = {
				pod_death = {
					killer = scope:assamite_blade
					death_reason = death_duel
				}
			}
		}
		else = {
			scope:assamite_blade ?= { move_to_pool = yes }
		}
	}
}
