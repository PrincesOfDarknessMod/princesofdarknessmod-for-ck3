######################################
# .100 — Tablet I: The Devotee
######################################
#
# Restores Mary's memory of Ebla, Michael's early influence, her original
# devotion. The first response inflection of the arc.
#
POD_maryb.100 = {
	type = character_event
	content_source = dlc_POD
	title = POD_maryb.100.t
	desc = POD_maryb.100.desc
	theme = baali_coa

	left_portrait = {
		character = root
		animation = personality_compassionate
	}

	immediate = {}

	option = {	# Mary-shift
		name = POD_maryb.100.a
		change_variable = { name = mary_michael_balance add = 25 }
		add_character_modifier = angel_soul_modifier
	}

	option = {	# No-shift
		name = POD_maryb.100.b
	}

	option = {	# Michael-shift
		name = POD_maryb.100.c
		custom_tooltip = POD_maryb.100.c.tt
		change_variable = { name = mary_michael_balance subtract = 25 }
		if = {
			limit = { has_character_modifier = angel_soul_modifier }
			remove_character_modifier = angel_soul_modifier
		}
	}

	after = {
		hidden_effect = { POD_maryb_sync_story_effect = yes }
	}
}

######################################
# .130 — Tablet IV: The Empty Revenge
######################################
#
# Sack of Constantinople, the attack on Michael, the failed finality of
# diablerie, the shared body. Locks the Deal with Michael outcome.
#
POD_maryb.130 = {
	type = character_event
	content_source = dlc_POD
	title = POD_maryb.130.t
	desc = POD_maryb.130.desc
	theme = baali_coa

	left_portrait = {
		character = root
		animation = personality_bold
	}

	immediate = {}

	# Tablet IV options are gated by the accumulated Mary/Michael balance so
	# the player is locked into the outcome the arc has been pointing at all
	# along. Balance range after three tablets: [-85, +85] (start ±25 plus
	# three ±20 shifts). Threshold ±30 cuts the range into three mutually
	# exclusive bands so exactly one option ever shows:
	#
	#   balance >=  30 → Mary-heavy   → The Black Dream
	#   balance <= -30 → Michael-heavy → The Second Dream
	#   otherwise      → Balanced      → Ash and Gold
	#
	# The has_variable guard short-circuits tooltip-build evaluation when
	# the variable hasn't been set yet (e.g. before podintroductions.380).
	option = {	# Mary-shift — locks The Black Dream
		name = POD_maryb.130.a
		trigger = {
			has_variable = mary_michael_balance
			var:mary_michael_balance >= 30
		}
		set_variable = { name = mary_deal_outcome value = flag:black_dream }
	}

	option = {	# No-shift / Synthesis — locks Ash and Gold
		name = POD_maryb.130.b
		trigger = {
			has_variable = mary_michael_balance
			var:mary_michael_balance > -30
			var:mary_michael_balance < 30
		}
		custom_tooltip = POD_maryb.130.b.tt
		set_variable = { name = mary_deal_outcome value = flag:ash_and_gold }
	}

	option = {	# Michael-shift — locks The Second Dream
		name = POD_maryb.130.c
		trigger = {
			has_variable = mary_michael_balance
			var:mary_michael_balance <= -30
		}
		custom_tooltip = POD_maryb.130.c.tt
		if = {
			limit = { has_character_modifier = angel_soul_modifier }
			remove_character_modifier = angel_soul_modifier
		}
		set_variable = { name = mary_deal_outcome value = flag:second_dream }
	}

	after = {
		hidden_effect = { POD_maryb_sync_story_effect = yes }
	}
}
