######################################
# .10 — A Lingering Debt
######################################
#
# Fires once, when Tanith's Protection has been invalidated by death, broken
# vassalage, opinion collapse, or a diarchy that has resolved with Mary on top.
#
POD_maryb.10 = {
	type = character_event
	content_source = dlc_POD
	title = POD_maryb.10.t
	desc = POD_maryb.10.desc
	theme = baali_coa
	override_background = { reference = study }

	left_portrait = {
		character = root
		animation = personality_callous
	}

	immediate = {
		# The arrangement that set them is over, so any commission still open
		# dies with it. mary_active_commission is the canonical "a commission is
		# live" marker; clear it and whichever commission modifier it applied
		# (the removes are safe no-ops for the three that aren't active).
		if = {
			limit = { has_variable = mary_active_commission }
			remove_variable = mary_active_commission
			remove_character_modifier = mary_tanith_commission_diyarbakr_modifier
			remove_character_modifier = mary_tanith_commission_cyprus_modifier
			remove_character_modifier = mary_tanith_commission_cyprus_brujah_modifier
			remove_character_modifier = mary_tanith_commission_olives_modifier
			remove_character_flag = mary_commission_cyprus_brujah_variant
		}
	}

	option = {
		trigger = {
			POD_maryb_tanith_protection_invalidated_trigger = no
		}
		name = POD_maryb.10.a		# "Her protection was earned. I deserve it."
		add_prestige = -100
		remove_character_modifier = mary_tanith_protection_modifier
		add_character_modifier = mary_tanith_legacy_modifier
	}

	option = {
		name = POD_maryb.10.b		# "I owe her nothing now."
		add_prestige = 100
		remove_character_modifier = mary_tanith_protection_modifier
	}
}

######################################
# .50 — Yearly maintenance dispatcher (called from POD_yearly_playable_pulse)
######################################
#
# Decides whether to fire a commission, whether A Lingering Debt should fire,
# and whether to issue a new commission at the 15-year cadence.
#
POD_maryb.50 = {
	type = character_event
	hidden = yes

	trigger = {
		this = character:790149
		is_alive = yes
	}

	immediate = {
		# Track years since the last commission. Initialize on first run.
		if = {
			limit = { NOT = { has_variable = mary_years_since_last_commission } }
			set_variable = { name = mary_years_since_last_commission value = 15 }
		}
		else = {
			change_variable = { name = mary_years_since_last_commission add = 1 }
		}

		# Tanith's Protection lapses if Mary abandons her holdings to wander. A
		# landless adventurer is outside Tanith's political shelter, so the
		# protection ends here. This runs BEFORE the A Lingering Debt check below 
		if = {
			limit = {
				is_landless_adventurer = yes
				has_character_modifier = mary_tanith_protection_modifier
			}
			remove_character_modifier = mary_tanith_protection_modifier
			remove_character_modifier = mary_tanith_commission_diyarbakr_modifier
			remove_character_modifier = mary_tanith_commission_cyprus_modifier
			remove_character_modifier = mary_tanith_commission_cyprus_brujah_modifier
			remove_character_modifier = mary_tanith_commission_olives_modifier
			send_interface_toast = {
				type = event_generic_neutral
				title = mary_tanith_protection_lapsed_landless.toast
				left_icon = character:505095
			}
		}

		# A Lingering Debt — fires once if Tanith's Protection is invalidated.
		# The invalidation OR block lives in POD_maryb_triggers.txt as
		# POD_maryb_tanith_protection_invalidated_trigger so future events
		# and decisions can reuse the same definition.
		if = {
			limit = {
				has_character_modifier = mary_tanith_protection_modifier
				OR = {
					POD_maryb_tanith_protection_invalidated_trigger = yes
					# Diarchy resolved — Mary was once a diarch with Tanith, and now isn't.
					AND = {
						has_character_flag = mary_was_diarch_with_tanith
						NOT = { is_diarch_of_target = character:505095 }
					}
				}
			}
			trigger_event = {
				id = POD_maryb.10
				days = { 7 30 }
			}
		}

		# Track diarchy state so we know if Mary was diarch with Tanith.
		if = {
			limit = {
				exists = character:505095
				is_diarch_of_target = character:505095
				NOT = { has_character_flag = mary_was_diarch_with_tanith }
			}
			add_character_flag = mary_was_diarch_with_tanith
		}

		# Fire a new commission if conditions are met.
		if = {
			limit = {
				NOT = { has_variable = mary_active_commission }
				var:mary_years_since_last_commission >= 15
				exists = character:505095
				character:505095 = { is_alive = yes }
				OR = {
					liege = character:505095
					is_diarch_of_target = character:505095
				}
			}
			# Pick an available commission whose target still exists.
			random_list = {
				100 = {
					trigger = {
						NOT = { has_character_flag = mary_commission_diyarbakr_disabled }
						OR = {
							title:d_diyarrabia = { exists = holder holder = { is_baali_trigger = yes NOT = { this = character:790149 } NOT = { is_vassal_or_below_of = character:790149 } } }
							title:d_diyarbakr = { exists = holder holder = { is_baali_trigger = yes NOT = { this = character:790149 } NOT = { is_vassal_or_below_of = character:790149 } } }
						}
					}
					trigger_event = {
						id = POD_maryb.1
						days = { 7 60 }
					}
				}
				# Cyprus commission — standard variant (vs Knights of Acre).
				# Fires when the Knights are top liege of Cyprus's holder.
				100 = {
					trigger = {
						NOT = { has_character_flag = mary_commission_cyprus_disabled }
						title:d_cyprus = { holder.top_liege = { has_title = title:k_knights_of_acre } }
					}
					trigger_event = {
						id = POD_maryb.3
						days = { 7 60 }
					}
				}
				# Cyprus commission — vampire-only variant. Fires when
				# Cyprus is held by Nehemiah (790239, Brujah Prince) instead
				# of the Knights. The two triggers are mutually exclusive
				# given the world state, so at most one branch ever matches.
				100 = {
					trigger = {
						NOT = { has_character_flag = mary_commission_cyprus_disabled }
						title:d_cyprus = { holder = character:790239 }
					}
					trigger_event = {
						id = POD_maryb.7
						days = { 7 60 }
					}
				}
				100 = {
					trigger = {
						NOT = { has_character_flag = mary_commission_olives_disabled }
						OR = {
							title:d_antioch = { exists = holder holder = { NOT = { this = character:790149 } NOT = { is_vassal_or_below_of = character:790149 } } }
							title:d_lebanon = { exists = holder holder = { NOT = { this = character:790149 } NOT = { is_vassal_or_below_of = character:790149 } } }
						}
					}
					trigger_event = {
						id = POD_maryb.5
						days = { 7 60 }
					}
				}
			}
		}

		# Ash and Gold — ascendancy oscillation. On the Synthesis path, the
		# body lurches between Mary and Michael on its own clock. Every 3 years
		# the ascendant half is re-rolled (POD_maryb_ash_gold_reroll_effect).
		# The cadence counter is seeded to 0 at resolution (POD_maryb.230) so
		# the first oscillation lands a few years after the deal, not instantly.
		if = {
			limit = { has_character_flag = mary_resolved_ash_and_gold }
			if = {
				limit = { NOT = { has_variable = mary_ash_gold_years } }
				set_variable = { name = mary_ash_gold_years value = 0 }
			}
			else = {
				change_variable = { name = mary_ash_gold_years add = 1 }
			}
			if = {
				limit = { var:mary_ash_gold_years >= 3 }
				set_variable = { name = mary_ash_gold_years value = 0 }
				POD_maryb_ash_gold_reroll_effect = yes
			}
		}
	}

	option = {
		name = POD_maryb.50.a
	}
}

######################################
# .51 — Commission completion checker (called from on_title_gain)
######################################
#
# Fires for every title gain (since on_title_gain is global). Trigger gates to
# Mary holding an active commission and the new title being in the matching
# duchy.
#
POD_maryb.51 = {
	type = character_event
	hidden = yes

	trigger = {
		root = { this = character:790149 }
		scope:title ?= { tier = tier_county }
		root = { has_variable = mary_active_commission }
	}

	immediate = {
		if = {
			limit = {
				root.var:mary_active_commission = flag:commission_diyarbakr
				OR = {
					scope:title.duchy ?= title:d_diyarbakr
					scope:title.duchy ?= title:d_diyarrabia
				}
			}
			root = { trigger_event = POD_maryb.2 }
		}
		else_if = {
			limit = {
				root.var:mary_active_commission = flag:commission_cyprus
				scope:title.duchy ?= title:d_cyprus
			}
			# Route to the matching completion event: .8 for the
			# vampire-only Nehemiah variant, .4 for the standard
			# Knights-of-Acre variant.
			if = {
				limit = { root = { has_character_flag = mary_commission_cyprus_brujah_variant } }
				root = { trigger_event = POD_maryb.8 }
			}
			else = {
				root = { trigger_event = POD_maryb.4 }
			}
		}
		else_if = {
			limit = {
				root.var:mary_active_commission = flag:commission_olives
				OR = {
					scope:title.duchy ?= title:d_antioch
					scope:title.duchy ?= title:d_lebanon
				}
			}
			root = { trigger_event = POD_maryb.6 }
		}
	}
}

	type = character_event
	content_source = dlc_POD

		animation = aggressive_sword
	}

		character = character:505051
		animation = flirtation
	}

	immediate = {}

	option = {
		name = POD_maryb.140.a

		# reveals what that smile meant. No delay — it is one continuous memory.
		trigger_event = { id = POD_maryb.141 }
	}

	after = {
		hidden_effect = { POD_maryb_sync_story_effect = yes }
	}
}

######################################
# .200 — Write the Last Tablet
######################################
#
# Balance is the sum of the opening choice (podintroductions.380 sets ±25) and
# the three tablet shifts (each ±25, or 0 on the No-shift option), so the reachable values are
# {-100, -75, -50, -25, 0, +25, +50, +75, +100}. Threshold ±30 trisects them cleanly:
#
#   balance >=  30 → Mary-heavy    → The Black Dream   (.210)
#   balance <= -30 → Michael-heavy → The Second Dream  (.220)
#   otherwise (±25/0) → Balanced      → Ash and Gold      (.230)
#
# The has_variable guard short-circuits tooltip-build evaluation if the
# variable was never set (defensive against console/out-of-order paths).
#
POD_maryb.200 = {
	type = character_event
	content_source = dlc_POD
	title = POD_maryb.200.t
	desc = POD_maryb.200.desc
	theme = baali_coa
	override_background = { reference = study }

	left_portrait = {
		character = root
		animation = writing
	}
	right_portrait = {
		character = character:505051
		animation = flirtation
		camera = camera_event_duel_left
	}

	immediate = {
		# Defensive: if mary_michael_balance was never set (console / out-of-order),
		# default to balanced so this event is never optionless.
		if = {
			limit = { NOT = { has_variable = mary_michael_balance } }
			set_variable = { name = mary_michael_balance value = 0 }
		}
	}

	option = {	# Mary-heavy — The Black Dream
		name = POD_maryb.200.a
		trigger = {
			has_variable = mary_michael_balance
			var:mary_michael_balance >= 30
		}
		trigger_event = POD_maryb.210
	}

	option = {	# Balanced — Ash and Gold
		name = POD_maryb.200.b
		trigger = {
			has_variable = mary_michael_balance
			var:mary_michael_balance > -30
			var:mary_michael_balance < 30
		}
		custom_tooltip = POD_maryb.200.b.tt
		trigger_event = POD_maryb.230
	}

	option = {	# Michael-heavy — The Second Dream
		name = POD_maryb.200.c
		trigger = {
			has_variable = mary_michael_balance
			var:mary_michael_balance <= -30
		}
		custom_tooltip = POD_maryb.200.c.tt
		trigger_event = POD_maryb.220
	}
}

######################################
# .220 — Outcome II: The Second Dream
######################################
#
# Michael dominates. The voice that has been whispering threats now is the
# character. Name, traits, culture, and faith are reshaped.
#
POD_maryb.220 = {
	type = character_event
	content_source = dlc_POD
	title = POD_maryb.220.t
	desc = POD_maryb.220.desc
	theme = baali_coa
	override_background = { reference = temple }

	left_portrait = {
		character = character:505051
		animation = ecstasy
	}

	immediate = {
		add_character_flag = mary_resolved_second_dream
	}

	option = {
		name = POD_maryb.220.a
		change_first_name = name_michael
		if = { limit = { has_trait = vengeful } remove_trait = vengeful }
		if = { limit = { has_trait = ambitious } remove_trait = ambitious }
		if = { limit = { has_trait = sadistic } remove_trait = sadistic }
		add_trait = arrogant
		add_trait = zealous
		add_trait = fickle
		add_character_flag = POD_portrait_monster_angel_wings_folded_flag
		set_culture = culture:greek
		if = {
			limit = { has_trait = diablerist }
			remove_trait = diablerist
		}
		if = {
			limit = { has_character_modifier = mary_most_hated_modifier }
			remove_character_modifier = mary_most_hated_modifier
		}
		if = {
			limit = { has_character_modifier = angel_soul_modifier }
			remove_character_modifier = angel_soul_modifier
		}
		# Michael's resurrection lands on the source doctrine directly — the
		# Codex of Legacies, his proper home. Mary's realm follows separately
		# via POD_maryb_resurrect_the_dream_decision once she controls Thessalonika.
		#
		# Permanent faith lock.
		add_character_modifier = POD_faith_commited_modifier
		add_character_flag = converted_by_forced_conversion_interaction
		set_character_faith = faith:codexoflegacies
		add_character_modifier = mary_second_dream_modifier

		# Mary's soul leaves; her bargain with Anoster leaves with her.
		# Refund every investiture lifestyle perk she ever spent (PoD helper —
		# also unwinds stacking perk modifiers), end both pact stories so they
		# stop ticking consequences against the new body, and net out the 5
		# perk points the two starting pacts originally granted (sacrifice_of_blood
		# = 2 + tribute_of_agony = 3) so Michael doesn't inherit a free
		# investiture surplus he was never owed.
		POD_refund_investiture = yes
		hidden_effect = {
			every_in_list = {
				variable = demonic_pacts
				end_story = yes
			}
			clear_variable_list = demonic_pacts
		}
		if = {
			limit = { has_trait = witch }
			remove_trait = witch
		}
		add_investiture_lifestyle_perk_points = -5

		# Michael's signature discipline. now that he's in the 
		# driver's seat again it surfaces in the body he wears.
		add_trait = chimerstry

		# Move into Michael's actual house instead. house_nephilim sits under
		# the Toreador founder dynasty (10000013) and already carries
		# house_nephilim_modifier (defined in PoD's 00_POD_houses_modifiers.txt).
		set_house = house:house_nephilim

		# Cleanup
		remove_character_modifier = mary_fragmented_self_modifier
		remove_character_modifier = mary_fragmented_self_modifier_t1
		remove_character_modifier = mary_fragmented_self_modifier_t2
		hidden_effect = { POD_maryb_sync_story_effect = yes }
		if = {
			limit = { has_variable = mary_michael_balance }
			remove_variable = mary_michael_balance
		}
	}
}

# .300 — Decani Hunt: Announcement
######################################
#
# Fired by POD_maryb_decani_hunt_decision. Sets the stage; the binding ritual itself
# happens in POD_maryb.310 after a short delay.
#
POD_maryb.300 = {
	type = character_event
	content_source = dlc_POD
	title = POD_maryb.300.t
	desc = POD_maryb.300.desc
	theme = baali_coa
	override_background = { reference = dungeon }

	# Defensive guard: the Decani Hunt belongs strictly to the Black Dream
	# and Ash and Gold paths. The Second Dream path settles Anoster
	# differently (Mary's soul takes the pacts with her in POD_maryb.220),
	# so this chain must not fire on that branch even via console / future
	# code refactors. The decision's is_shown already excludes Second Dream.
	trigger = {
		OR = {
			has_character_flag = mary_resolved_black_dream
			has_character_flag = mary_resolved_ash_and_gold
		}
	}

	left_portrait = {
		character = root
		animation = scheme
	}

	option = {
		name = POD_maryb.300.a
		# The binding ritual pulls Anoster out of the abyss so the resolution
		# event has a target standing inside the circle. Free him here, kill
		# him in POD_maryb.310.
		if = {
			limit = {
				exists = character:9201500
				character:9201500 = { has_character_modifier = trapped_in_abyss_modifier }
			}
			character:9201500 = { POD_free_from_abyss_effect = yes }
		}
		trigger_event = {
			id = POD_maryb.310
			days = { 14 30 }
		}
	}
}

######################################
# .310 — Decani Hunt: Resolution (the duel)
######################################
#
# Mirrors base PoD's pod1230objectives.1581 *option B* — the prowess duel
# against the Decani, not a guaranteed settling. Win and Anoster is unmade and
# the pact ledger closes for good; lose and he collects what the pact always
# promised him: Mary herself.
#
# Pact stories do NOT auto-end when their demon dies. The success branch closes 
# them explicitly and clears the list.
#
POD_maryb.310 = {
	type = character_event
	content_source = dlc_POD
	title = POD_maryb.310.t
	desc = POD_maryb.310.desc
	theme = baali_coa
	override_background = { reference = dungeon }

	# Same guard as POD_maryb.300 — Second Dream characters must never
	# reach this event. Their pacts are already gone and Mary's name no
	# longer belongs to anyone who could "settle the ledger" with Anoster.
	trigger = {
		OR = {
			has_character_flag = mary_resolved_black_dream
			has_character_flag = mary_resolved_ash_and_gold
		}
	}

	left_portrait = {
		character = root
		animation = anger
	}
	# Anoster stands in the binding circle, freed from the abyss by POD_maryb.300.
	# He is alive here (only the success branch kills him), but guard anyway.
	right_portrait = {
		trigger = { exists = character:9201500 }
		character = character:9201500
		animation = disgust
	}

	immediate = {
		play_music_cue = "mx_cue_epic_sacral_moment"
	}

	option = {
		name = POD_maryb.310.a
		custom_tooltip = POD_maryb.310.a.tt

		# Option B: a prowess duel against the Decani (Anoster). value = 150
		# matches the canonical hunt — a hard, deliberately dangerous contest;
		# scope:duel_value weights the 50/50 by Mary's prowess.
		duel = {
			skill = prowess
			value = 150

			50 = {	# Success — she breaks him.
				compare_modifier = {
					value = scope:duel_value
					multiplier = 0.5
				}
				desc = POD_maryb.310.success

				# Kill Anoster — freed from the abyss by POD_maryb.300, standing
				# in the binding circle.
				if = {
					limit = {
						exists = character:9201500
						character:9201500 = { is_alive = yes }
					}
					character:9201500 = {
						death = {
							death_reason = death_souleaten
							killer = root
						}
					}
				}

				# Close every pact story she still carries, then clear the list.
				hidden_effect = {
					every_in_list = {
						variable = demonic_pacts
						end_story = yes
					}
					clear_variable_list = demonic_pacts
				}

				# Option B's reward: the Decani modifier (not option A's
				# investiture boon), plus Mary's own reward for surviving.
				add_character_modifier = decani_modifier
				add_prestige = 500
				add_piety = 250
			}

			50 = {	# Failure — he collects the ledger.
				compare_modifier = {
					value = scope:duel_value
					multiplier = -0.5
				}
				desc = POD_maryb.310.failure

				# Anoster eats her soul. He is alive (the success branch is the
				# only thing that kills him), so he is the killer; the else is a
				# defensive fallback only.
				if = {
					limit = {
						exists = character:9201500
						character:9201500 = { is_alive = yes }
					}
					death = {
						death_reason = death_souleaten
						killer = character:9201500
					}
				}
				else = {
					death = { death_reason = death_souleaten }
				}
			}
		}
	}
}
