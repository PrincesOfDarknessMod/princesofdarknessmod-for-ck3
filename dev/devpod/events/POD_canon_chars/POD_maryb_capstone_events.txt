namespace = POD_maryb

######################################
# .300 — Decani Hunt: Announcement
######################################
#
# Fired by mary_decani_hunt_decision. Sets the stage; the binding ritual itself
# happens in POD_maryb.310 after a short delay.
#
POD_maryb.300 = {
	type = character_event
	content_source = dlc_POD
	title = POD_maryb.300.t
	desc = POD_maryb.300.desc
	theme = baali_coa

	# Defensive guard: the Decani Hunt belongs strictly to the Black Dream
	# and Ash and Gold paths. The Second Dream path settles Anoster
	# differently (Mary's soul takes the pacts with her in POD_maryb.220),
	# so this chain must not fire on that branch even via console / future
	# code refactors. The decision's is_shown already excludes Second Dream;
	# this is belt-and-suspenders.
	trigger = {
		OR = {
			has_character_flag = mary_resolved_black_dream
			has_character_flag = mary_resolved_ash_and_gold
		}
	}

	left_portrait = {
		character = root
		animation = personality_callous
	}

	option = {
		name = POD_maryb.300.a
		# The binding ritual pulls Anoster out of the abyss so the resolution
		# event has a target standing inside the circle. Free him here, kill
		# him in POD_maryb.310.
		if = {
			limit = {
				exists = character:9201500
				character:9201500 = { has_character_modifier = trapped_in_abyss_modifier }
			}
			character:9201500 = { POD_free_from_abyss_effect = yes }
		}
		trigger_event = {
			id = POD_maryb.310
			days = { 14 30 }
		}
	}
}

######################################
# .310 — Decani Hunt: Resolution
######################################
#
# Mirrors base PoD's pod1230objectives.1581 outcome (Decani Hunt rewards)
# without the demon transformation — Mary stays Mary. Kills Anoster, ends the
# two pact stories explicitly (pact stories do NOT auto-end when their demon
# dies — they keep ticking against a dead target unless we close them), and
# applies the same investiture/access benefits the canonical hunt grants.
#
POD_maryb.310 = {
	type = character_event
	content_source = dlc_POD
	title = POD_maryb.310.t
	desc = POD_maryb.310.desc
	theme = baali_coa

	# Same guard as POD_maryb.300 — Second Dream characters must never
	# reach this event. Their pacts are already gone and Mary's name no
	# longer belongs to anyone who could "settle the ledger" with Anoster.
	trigger = {
		OR = {
			has_character_flag = mary_resolved_black_dream
			has_character_flag = mary_resolved_ash_and_gold
		}
	}

	left_portrait = {
		character = root
		animation = personality_bold
	}

	immediate = {
		play_music_cue = "mx_cue_epic_sacral_moment"
	}

	option = {
		name = POD_maryb.310.a

		# 1) Kill Anoster. He's been freed from the abyss by the announcement
		# event (POD_maryb.300) and now stands in the binding circle.
		if = {
			limit = {
				exists = character:9201500
				character:9201500 = { is_alive = yes }
			}
			character:9201500 = {
				death = {
					death_reason = death_souleaten
					killer = root
				}
			}
		}

		# 2) End every pact story Mary still carries (Anoster's pacts, plus any
		# other diabolist pacts she may have signed in the meantime). Pact
		# stories tick on a 1-year story pulse independent of the demon, so
		# we must close them explicitly and clear the list.
		hidden_effect = {
			every_in_list = {
				variable = demonic_pacts
				end_story = yes
			}
			clear_variable_list = demonic_pacts
		}

		# 3) Apply the canonical Decani Hunt rewards (per pod1230objectives.1581
		# option A — the investiture path, since the duel path's failure mode
		# is death and Mary has earned a clean resolution by this point).
		add_investiture_lifestyle_perk_points = 10
		add_character_modifier = decani_boon_modifier
		if = {
			limit = { NOT = { has_character_flag = access_to_investiture } }
			add_character_flag = access_to_investiture
		}

		# 4) Mary's own narrative reward for the hunt.
		add_prestige = 500
		add_piety = 250
	}
}
