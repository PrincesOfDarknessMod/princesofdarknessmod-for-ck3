######################################

namespace = POD_maryb

######################################
# .200 — Dispatcher
######################################
#
# Fired by mary_deal_with_michael_decision. Reads var:mary_deal_outcome and
# triggers the locked outcome event.
#
POD_maryb.200 = {
	type = character_event
	hidden = yes

	# Guard against being fired by console / out-of-order paths.
	trigger = {
		has_variable = mary_deal_outcome
	}

	immediate = {
		add_character_flag = mary_deal_resolved
		if = {
			limit = { var:mary_deal_outcome = flag:black_dream }
			trigger_event = POD_maryb.210
		}
		else_if = {
			limit = { var:mary_deal_outcome = flag:second_dream }
			trigger_event = POD_maryb.220
		}
		else_if = {
			limit = { var:mary_deal_outcome = flag:ash_and_gold }
			trigger_event = POD_maryb.230
		}
	}

	option = {
		name = POD_maryb.200.a
	}
}

######################################
# .210 — Outcome I: The Black Dream
######################################
#
# Mary dominates. Michael goes silent forever. No gold-italic line in the
# outcome flavor — the absence is the point.
#
POD_maryb.210 = {
	type = character_event
	content_source = dlc_POD
	title = POD_maryb.210.t
	desc = POD_maryb.210.desc
	theme = baali_coa

	left_portrait = {
		character = root
		animation = personality_bold
	}

	immediate = {
		add_character_flag = mary_resolved_black_dream
	}

	option = {
		name = POD_maryb.210.a
		set_character_faith = faith:viadiabolis
		add_character_modifier = mary_black_dream_modifier
		# Cleanup
		remove_character_modifier = mary_fragmented_self_modifier
		remove_character_modifier = mary_fragmented_self_modifier_t1
		remove_character_modifier = mary_fragmented_self_modifier_t2
		remove_character_modifier = mary_fragmented_self_modifier_t3
		if = {
			limit = { has_character_modifier = angel_soul_modifier }
			remove_character_modifier = angel_soul_modifier
		}
		# Sync to story before clearing the balance variable so the journey
		# panel keeps the final value on display post-resolution.
		hidden_effect = { POD_maryb_sync_story_effect = yes }
		if = {
			limit = { has_variable = mary_michael_balance }
			remove_variable = mary_michael_balance
		}
	}
}

######################################
# .220 — Outcome II: The Second Dream
######################################
#
# Michael dominates. The voice that has been whispering threats now is the
# character. Name, traits, culture, and faith are reshaped.
#
POD_maryb.220 = {
	type = character_event
	content_source = dlc_POD
	title = POD_maryb.220.t
	desc = POD_maryb.220.desc
	theme = baali_coa

	left_portrait = {
		character = root
		animation = personality_zealous
	}

	immediate = {
		add_character_flag = mary_resolved_second_dream
	}

	option = {
		name = POD_maryb.220.a
		change_first_name = name_michael
		if = { limit = { has_trait = vengeful } remove_trait = vengeful }
		if = { limit = { has_trait = ambitious } remove_trait = ambitious }
		if = { limit = { has_trait = sadistic } remove_trait = sadistic }
		add_trait = arrogant
		add_trait = zealous
		add_trait = fickle
		set_culture = culture:greek
		if = {
			limit = { has_character_modifier = mary_most_hated_modifier }
			remove_character_modifier = mary_most_hated_modifier
		}
		if = {
			limit = { has_character_modifier = angel_soul_modifier }
			remove_character_modifier = angel_soul_modifier
		}
		# Michael's resurrection lands on the source doctrine directly — the
		# Codex of Legacies, his proper home. Mary's realm follows separately
		# via mary_resurrect_the_dream_decision once she controls Thessalonika.
		#
		# Permanent faith lock. Two complementary defenses:
		#
		#   1. mary_michael_faith_locked — our own flag, read by POD_maryb.221
		#      (hooked on on_character_faith_change). Snaps Michael back to
		#      codexoflegacies if anything still manages to convert him.
		#
		#   2. converted_by_forced_conversion_interaction — vanilla CK3 flag.
		#      Vanilla `recently_converted_faith_trigger` checks for this
		#      (along with the two heresy-event flags), and PoD reuses that
		#      trigger to block AI heresy events from re-converting characters.
		#      Setting it permanently means AI heresy chains stay off Michael
		#      entirely, so the snap-back rarely has to fire in practice.
		#
		# No vanilla character_modifier exists that blocks player-driven
		# self-conversion at the UI level; overriding convert_to_religion_interaction
		# would be invasive. The flag-plus-snap-back combination is the
		# cleanest non-invasive lock.
		add_character_flag = mary_michael_faith_locked
		add_character_flag = converted_by_forced_conversion_interaction
		set_character_faith = faith:codexoflegacies
		add_character_modifier = mary_second_dream_modifier

		# Mary's soul leaves; her bargain with Anoster leaves with her.
		# Refund every investiture lifestyle perk she ever spent (PoD helper —
		# also unwinds stacking perk modifiers), end both pact stories so they
		# stop ticking consequences against the new body, and net out the 5
		# perk points the two starting pacts originally granted (sacrifice_of_blood
		# = 2 + tribute_of_agony = 3) so Michael doesn't inherit a free
		# investiture surplus he was never owed.
		POD_refund_investiture = yes
		hidden_effect = {
			every_in_list = {
				variable = demonic_pacts
				end_story = yes
			}
			clear_variable_list = demonic_pacts
		}
		add_investiture_lifestyle_perk_points = -5

		# Michael's signature discipline. Chimerstry is how he kept up the
		# illusion of angelic beauty for four thousand years; now that he's
		# in the driver's seat again it surfaces in the body he wears.
		add_trait = chimerstry

		# Found a cadet branch off house_moloch for the new identity. The
		# body and Mary's dynasty are still hers, but the soul running it
		# is Michael — a new house under the same bloodline. Applies the
		# bespoke house modifier (Michael's nephilim effects + Baali
		# opinion shape, defined in POD_maryb_modifiers.txt).
		create_cadet_branch = {
			name = dynn_house_maryb_dream
			save_scope_as = maryb_new_house
		}
		scope:maryb_new_house ?= {
			add_house_modifier = house_maryb_dream_modifier
		}

		# Cleanup
		remove_character_modifier = mary_fragmented_self_modifier
		remove_character_modifier = mary_fragmented_self_modifier_t1
		remove_character_modifier = mary_fragmented_self_modifier_t2
		remove_character_modifier = mary_fragmented_self_modifier_t3
		hidden_effect = { POD_maryb_sync_story_effect = yes }
		if = {
			limit = { has_variable = mary_michael_balance }
			remove_variable = mary_michael_balance
		}
	}
}

######################################
# .221 — Michael's faith lock (snap-back)
######################################
#
# Hooked into on_character_faith_change. Fires for every character whose
# faith just changed; the trigger filters to Michael (the character carrying
# mary_michael_faith_locked) and only catches the case where the new faith
# is NOT codexoflegacies. The immediate sets the faith back to codex,
# which fires the on-action again — but the trigger fails on the bounce
# (faith IS codex now), so no infinite loop.
#
# PoD's broader pattern blocks conversion on a timer (just_converted_to_*
# flags read by specific decisions). Michael's lock is permanent: the soul
# in this body does not entertain other doctrines.
#
POD_maryb.221 = {
	type = character_event
	hidden = yes

	trigger = {
		has_character_flag = mary_michael_faith_locked
		NOT = { faith = faith:codexoflegacies }
	}

	immediate = {
		set_character_faith = faith:codexoflegacies
		# Player-facing notification so the bounce reads as intentional
		# rather than as a buggy "I clicked Convert but nothing changed."
		# Only surfaces to the player who owns Michael (is_ai = no), since
		# AI characters don't need the explanation.
		if = {
			limit = { is_ai = no }
			send_interface_toast = {
				type = event_generic_neutral
				title = POD_maryb.221.toast.t
				desc = POD_maryb.221.toast.desc
				left_icon = root
			}
		}
	}

	option = {
		name = POD_maryb.221.a
	}
}
