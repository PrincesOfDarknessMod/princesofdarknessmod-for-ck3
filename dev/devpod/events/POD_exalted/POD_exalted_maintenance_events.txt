namespace = POD_exalted_maintenance_events
#Exaltation!
POD_exalted_maintenance_events.1 = {
	type = character_event
	content_source = dlc_POD
	title = POD_exalted_maintenance_events.1.t
	desc = POD_exalted_maintenance_events.1.desc

	theme = witchcraft
	
	left_portrait = {
		character = root
		animation = shock
	}
	
	trigger = {
		has_character_flag = dragon_touched_exaltation
		has_trait = dragon_touched
	}
	
	option = {
		name = POD_exalted_maintenance_events.1.a
		
		POD_remove_all_supernatural_atribute_effect = yes

		remove_trait = dragon_touched
		POD_cease_being_ghoul_effect = yes
		
		if = {
			limit = { has_trait = supehunter }
			hidden_effect = { remove_trait = supehunter }
		}

		POD_become_terrestrial_exalted_effect = yes
		POD_remove_wrong_supernatural_traits_on_existing_character = yes

		#AI help
		if = {
			limit = {
				mother ?= {
					is_landed = yes
					primary_heir ?= { has_trait = dragon_touched }
					is_ai = yes
					is_valid_designated_heir = prev
				}
			}
			mother = { set_designated_heir = prev }
		}
		else_if = {
			limit = {
				father ?= {
					is_landed = yes
					primary_heir ?= { has_trait = dragon_touched }
					is_ai = yes
					is_valid_designated_heir = prev
				}
			}
			father = { set_designated_heir = prev }
		}

		save_scope_as = exalt

		mother ?= { trigger_event = POD_exalted_maintenance_events.2 }
		father ?= { trigger_event = POD_exalted_maintenance_events.2 }
	}
}
POD_exalted_maintenance_events.2 = {
	type = character_event
	content_source = dlc_POD
	title = POD_exalted_maintenance_events.2.t
	desc = POD_exalted_maintenance_events.2.desc

	theme = witchcraft
	
	left_portrait = {
		character = root
		animation = happiness
	}
	
	right_portrait = {
		character = scope:exalt
		animation = shock
	}

	immediate = {
		show_as_tooltip = {
			scope:exalt ?= {
				remove_trait = dragon_touched
				POD_cease_being_ghoul_effect = yes
				
				POD_become_terrestrial_exalted_effect = yes
				POD_remove_wrong_supernatural_traits_on_existing_character = yes
			}
		}
	}
	
	option = {
		name = POD_exalted_maintenance_events.2.a
		add_stress = -100
		add_prestige = 1000
	}
}
POD_exalted_maintenance_events.3 = {
	type = character_event
	content_source = dlc_POD
	title = POD_exalted_maintenance_events.3.t
	desc = POD_exalted_maintenance_events.3.desc
	theme = court
	left_portrait = {
		character = root
		animation = personality_bold
	}

	immediate = {
		create_character = {
			name = POD_passive_charms_saved_loadout
			dynasty = none
			age = 16
			gender_female_chance = 50
			random_traits = no
			location = location
			faith = faith
			culture = culture
			save_scope_as = POD_saved_loadout_nameholder
		}
		scope:POD_saved_loadout_nameholder = {
			death = { death_reason = death_mysterious }
			make_unprunable = yes
		}
	}

	widget = {
		gui = "POD_event_window_widget_name_charm_loadout"
		container = "dynamic_birth_name"
		controller = name_character
		setup_scope = {
			scope:POD_saved_loadout_nameholder = { save_scope_as = name_character_target }
		}
	}

	option = {
		name = POD_exalted_maintenance_events.3.a
		create_story = {
			type = POD_story_passive_charm_gui
			save_scope_as = story
		}
		ordered_in_list = {
			variable = charm_list
			limit = { var:activated = flag:yes }
			max = 1000
			check_range_bounds = no
			scope:story ?= {
				add_to_variable_list = {
					name = POD_passive_charms
					target = PREV.var:charm_perk
				}
			}
			custom_tooltip = POD_exalted_maintenance_events.3.a.tt
		}
		scope:story ?= {
			set_variable = {
				name = nameholder
				value = scope:POD_saved_loadout_nameholder
			}
		}
		add_to_variable_list = {
			name = POD_passive_charms_loadouts
			target = scope:story
		}
	}

	option = {
		name = POD_exalted_maintenance_events.3.b
	}
}
