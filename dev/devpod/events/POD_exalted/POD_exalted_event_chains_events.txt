scripted_effect POD_erembour_increase = {
	POD_set_or_change_variable = { VAR = erembour_additional_diff VAL = 3 }
	custom_tooltip = POD_erembour_increase
}
scripted_effect POD_erembour_decrease = {
	POD_set_or_change_variable = { VAR = erembour_additional_diff VAL = -3 }
	custom_tooltip = POD_erembour_decrease
}
scripted_effect POD_erembour_massively_increase = {
	POD_set_or_change_variable = { VAR = erembour_additional_diff VAL = 6 }
	custom_tooltip = POD_erembour_massively_increase
}
scripted_effect POD_erembour_massively_decrease = {
	POD_set_or_change_variable = { VAR = erembour_additional_diff VAL = -6 }
	custom_tooltip = POD_erembour_massively_decrease
}

scripted_effect POD_erembour_cleanup = {
	remove_list_global_variable = {
		name = unavailable_unique_decisions
		target = flag:erembour_the_new_fetich_soul_objective_decision
	}
	remove_variable = erembour_additional_diff
	custom_tooltip = POD_erembour_cleanup
}

scripted_effect POD_erembour_partial_failure = {
	custom_tooltip = {
		text = POD_erembour_partial_failure
		random_relation = {
			type = coterie_member
			save_scope_as = sacrifice
		}
		if = {
			limit = { exists = scope:sacrifice }
			scope:sacrifice = { death = { death_reason = $DEATH_REASON$ } }
			trigger_event = $EVENT$
		}
		else = {
			death = { death_reason = $DEATH_REASON$ }
		}
	}
}

scripted_effect POD_erembour_failure_event = {
	show_as_tooltip = {
		POD_erembour_failure = { DEATH_REASON = $DEATH_REASON$ }
	}
	trigger_event = $EVENT$
}
scripted_effect POD_erembour_failure = {
	death = { death_reason = $DEATH_REASON$ }
	every_relation = {
		type = coterie_member
		death = { death_reason = $DEATH_REASON$ }
	}
	POD_erembour_cleanup = yes
}

scripted_effect POD_erembour_start = {
	custom_tooltip = {
		text = erembour_the_new_fetich_soul_objective_decision_effect1
		every_religion_global = {
			every_faith = {
				limit = {
					OR = {
						has_doctrine = doctrine_demonic_evil
						has_doctrine = doctrine_demonic_crime
					}
				}
				if = { 
					limit = { has_doctrine = doctrine_exaltation_evil } 
					remove_doctrine = doctrine_exaltation_evil 
				}
				if = { 
					limit = { has_doctrine = doctrine_exaltation_crime } 
					remove_doctrine = doctrine_exaltation_crime 
				}
				if = { 
					limit = { has_doctrine = doctrine_exaltation_shunned } 
					remove_doctrine = doctrine_exaltation_shunned 
				}
				if = { 
					limit = { has_doctrine = doctrine_exaltation_accepted } 
					remove_doctrine = doctrine_exaltation_accepted 
				}
				if = { 
					limit = { has_doctrine = doctrine_exaltation_ignorant } 
					remove_doctrine = doctrine_exaltation_ignorant 
				}
				add_doctrine = doctrine_exaltation_evil
			}
		}
	}
}
