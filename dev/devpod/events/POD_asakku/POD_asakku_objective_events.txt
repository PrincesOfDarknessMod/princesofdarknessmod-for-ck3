	left_portrait = {
		character = root
		animation = personality_callous
	}

	immediate = { }

	option = {
		name = POD_asakku.0403.a
		add_prestige = 200
		add_piety = 100
		add_gold = 300
		add_character_modifier = POD_asakku_phase3_reward
		# Remove Phase 1 reward (replaced by Brood)
		if = {
			limit = { has_character_modifier = POD_asakku_phase1_reward }
			remove_character_modifier = POD_asakku_phase1_reward
		}
		# Fire the Eastern Shadows introduction event, then the Watcher's Voice
		hidden_effect = {
			trigger_event = {
				id = POD_asakku.0215
				days = { 3 7 }
			}
			if = {
				limit = { has_trait = POD_asakku_iryi_ascendant}
				trigger_event = {
					id = POD_asakku.0404
					days = { 30 60 }
				}
			}
		}
	}
}
