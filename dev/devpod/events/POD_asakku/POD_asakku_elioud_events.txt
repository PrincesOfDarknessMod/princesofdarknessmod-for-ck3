######################################################
# 0205 - "A Curious Investigation" (Depth 25 Elioud response)
# Fires for Andeleon when depth reaches 25
######################################################

POD_asakku.0205 = {
	type = character_event
	title = POD_asakku.0205.t
	desc = POD_asakku.0205.desc
	theme = tzimisce
	content_source = dlc_POD

	left_portrait = {
		character = root
		animation = worry
	}

	trigger = {
		POD_asakku_is_andeleon_trigger = yes
		POD_has_asakku_depth = { DEPTH = 25 }
		NOT = { has_character_flag = POD_asakku_curious_investigation_fired }
	}

	immediate = {
		add_character_flag = POD_asakku_curious_investigation_fired
		# Find near by Elioud
		random_living_character = {
			limit = { has_trait = POD_asakku_elioud }
			weight = {
				base = 1
				modifier = {
					add = 10
					in_diplomatic_range = root
				}
			}
			save_scope_as = nearest_elioud
		}
	}

	# Option A - Find them (reveal the Elioud)
	option = {
		name = POD_asakku.0205.a
		trigger = { exists = scope:nearest_elioud }
		# Reveal the Elioud
		scope:nearest_elioud = {
			set_relation_potential_rival = {
				reason = rival_asakku_threat
				target = root
			}
		}
		# Grant intrigue XP for the investigation
		add_intrigue_lifestyle_xp = 200
		custom_tooltip = POD_asakku_reveal_elioud_tt
	}

	# Option B - Arrange an accident (start murder scheme)
	option = {
		name = POD_asakku.0205.b
		trigger = { exists = scope:nearest_elioud }
		hidden_effect = {
			start_scheme = {
				type = murder
				target_character = scope:nearest_elioud
			}
		}
		custom_tooltip = POD_asakku_assassinate_elioud_tt
	}
}

	}

	right_portrait = {
		character = scope:phalanx_leader
		animation = personality_bold
	}

	immediate = {
		# Find the Elioud leader and store on root so 0211 picks the same character
		random_living_character = {
			limit = { has_trait = POD_asakku_elioud }

				base = 1
				modifier = {
					add = 10

				}
			}
			save_scope_as = phalanx_leader
		}
		if = {
			limit = { exists = scope:phalanx_leader }
			set_variable = {
				name = POD_asakku_phalanx_leader_var
				value = scope:phalanx_leader
			}
		}
	}

	option = {
		name = POD_asakku.0210.a
		hidden_effect = {
			trigger_event = {
				id = POD_asakku.0211
				days = { 7 14 }
			}
		}
	}
}

# 0211 - The Assault
POD_asakku.0211 = {
	type = character_event
	title = POD_asakku.0211.t
	desc = POD_asakku.0211.desc
	theme = vampire
	content_source = dlc_POD

	left_portrait = {
		character = root
		animation = personality_bold
	}

	right_portrait = {
		character = scope:phalanx_leader
		animation = personality_bold
	}

	immediate = {
		# Restore the same leader saved during 0210 warning event
		if = {
			limit = { has_variable = POD_asakku_phalanx_leader_var }
			var:POD_asakku_phalanx_leader_var = {
				save_scope_as = phalanx_leader
			}
		}
		# Fallback: re-find if variable is missing
		if = {
			limit = { NOT = { exists = scope:phalanx_leader } }
			random_living_character = {
				limit = { has_trait = POD_asakku_elioud }
				weight = {
					base = 1
					modifier = { add = 10 prowess >= 10 }
				}
				save_scope_as = phalanx_leader
			}
		}
		# Compute stat diffs for duel modifiers
		save_scope_value_as = {
			name = prowess_diff
			value = {
				value = root.prowess
				subtract = scope:phalanx_leader.prowess
			}
		}
		save_scope_value_as = {
			name = intrigue_diff
			value = {
				value = root.intrigue
				subtract = scope:phalanx_leader.intrigue
			}
		}
	}

	# Option A - Fight (prowess duel vs. the leader)
	option = {
		name = POD_asakku.0211.a
		duel = {
			skill = prowess
			value = scope:prowess_diff
			50 = {
				desc = POD_asakku.0211.a.success.desc
				max = 85
				modifier = {
					add = {
						value = scope:prowess_diff
						multiply = 0.75
					}
				}
				send_interface_toast = {
					title = POD_asakku.0211.a.success
					left_icon = root
					right_icon = scope:phalanx_leader
					add_character_flag = POD_asakku_phalanx_survived
					add_character_flag = POD_asakku_phalanx_fight_clean
					add_prestige = 100
				}
			}
			25 = {
				desc = POD_asakku.0211.a.wounded.desc
				send_interface_toast = {
					title = POD_asakku.0211.a.wounded
					left_icon = root
					add_character_flag = POD_asakku_phalanx_survived
					add_character_flag = POD_asakku_phalanx_fight_wounded
					increase_wounds_effect = { REASON = fight }
				}
			}
			25 = {
				desc = POD_asakku.0211.a.grave.desc
				min = 10
				modifier = {
					add = {
						value = scope:prowess_diff
						multiply = -0.5
					}
				}
				send_interface_toast = {
					title = POD_asakku.0211.a.grave
					left_icon = root
					add_character_flag = POD_asakku_phalanx_survived
					add_character_flag = POD_asakku_phalanx_fight_grave
					POD_increase_wounds_effect = {
						REASON = fight
						COUNT = POD_double_wound_damage
					}
				}
			}
		}
		hidden_effect = {
			trigger_event = { id = POD_asakku.0212 days = { 1 3 } }
		}
	}

	# Option B - Outmaneuver (intrigue duel vs. the leader)
	option = {
		name = POD_asakku.0211.b
		duel = {
			skill = intrigue
			value = scope:intrigue_diff
			60 = {
				desc = POD_asakku.0211.b.success.desc
				max = 90
				modifier = {
					add = {
						value = scope:intrigue_diff
						multiply = 0.75
					}
				}
				send_interface_toast = {
					title = POD_asakku.0211.b.success
					left_icon = root
					right_icon = scope:phalanx_leader
					add_character_flag = POD_asakku_phalanx_survived
					add_character_flag = POD_asakku_phalanx_outmaneuvered
					add_prestige = 75
					add_intrigue_lifestyle_xp = 150
				}
			}
			40 = {
				desc = POD_asakku.0211.b.failure.desc
				min = 15
				modifier = {
					add = {
						value = scope:intrigue_diff
						multiply = -0.45
					}
				}
				send_interface_toast = {
					title = POD_asakku.0211.b.failure
					left_icon = root
					right_icon = scope:phalanx_leader
					add_character_flag = POD_asakku_phalanx_survived
					add_character_flag = POD_asakku_phalanx_intrigue_failed
					stress_impact = { base = minor_stress_gain }
					scope:phalanx_leader = {
						set_relation_rival = {
							reason = rival_asakku_threat
							target = root
						}
					}
				}
			}
		}
		hidden_effect = {
			trigger_event = { id = POD_asakku.0212 days = { 1 3 } }
		}
	}

	# Option C - Vanish (guaranteed survival at cost)
	option = {
		name = POD_asakku.0211.c
		add_character_flag = POD_asakku_phalanx_survived
		add_character_flag = POD_asakku_phalanx_fled
		add_character_modifier = POD_asakku_hunted
		POD_asakku_change_depth_effect = { AMOUNT = -5 }
		stress_impact = { base = minor_stress_gain }
		custom_tooltip = POD_asakku_phalanx_vanish_tt
		hidden_effect = {
			trigger_event = { id = POD_asakku.0212 days = { 1 3 } }
		}
	}
}

# 0212 - Aftermath
POD_asakku.0212 = {
	type = character_event
	title = POD_asakku.0212.t
	desc = {
		triggered_desc = {
			trigger = { has_character_flag = POD_asakku_phalanx_fight_clean }
			desc = POD_asakku.0212.desc.fight_clean
		}
		triggered_desc = {
			trigger = { has_character_flag = POD_asakku_phalanx_fight_wounded }
			desc = POD_asakku.0212.desc.fight_wounded
		}
		triggered_desc = {
			trigger = { has_character_flag = POD_asakku_phalanx_fight_grave }
			desc = POD_asakku.0212.desc.fight_grave
		}
		triggered_desc = {
			trigger = { has_character_flag = POD_asakku_phalanx_outmaneuvered }
			desc = POD_asakku.0212.desc.outmaneuvered
		}
		triggered_desc = {
			trigger = { has_character_flag = POD_asakku_phalanx_intrigue_failed }
			desc = POD_asakku.0212.desc.intrigue_failed
		}
		triggered_desc = {
			trigger = { has_character_flag = POD_asakku_phalanx_fled }
			desc = POD_asakku.0212.desc.fled
		}
	}
	theme = vampire
	content_source = dlc_POD
	override_background = { reference = throne_room_west }

	left_portrait = {
		character = root
		animation = personality_callous
	}

	immediate = { }

	# Option A - Fight path (clean, wounded, or grave)
	option = {
		name = POD_asakku.0212.a
		trigger = {
			NOT = { has_character_flag = POD_asakku_phalanx_outmaneuvered }
			NOT = { has_character_flag = POD_asakku_phalanx_intrigue_failed }
			NOT = { has_character_flag = POD_asakku_phalanx_fled }
		}
		POD_asakku_change_depth_effect = { AMOUNT = 5 }
		hidden_effect = {
			POD_asakku_check_depth_thresholds_effect = yes
		}
	}

	# Option B - Intrigue victory path
	option = {
		name = POD_asakku.0212.b
		trigger = { has_character_flag = POD_asakku_phalanx_outmaneuvered }
		POD_asakku_change_depth_effect = { AMOUNT = 5 }
		hidden_effect = {
			POD_asakku_check_depth_thresholds_effect = yes
		}
	}

	# Option C - Intrigue failure path
	option = {
		name = POD_asakku.0212.c
		trigger = { has_character_flag = POD_asakku_phalanx_intrigue_failed }
		# No depth bonus - they escaped but exposed
	}

	# Option D - Fled path
	option = {
		name = POD_asakku.0212.d
		trigger = { has_character_flag = POD_asakku_phalanx_fled }
		# Depth already decremented on the flee choice - no bonus here
	}

	after = {
		hidden_effect = {
			remove_character_flag = POD_asakku_phalanx_fight_clean
			remove_character_flag = POD_asakku_phalanx_fight_wounded
			remove_character_flag = POD_asakku_phalanx_fight_grave
			remove_character_flag = POD_asakku_phalanx_outmaneuvered
			remove_character_flag = POD_asakku_phalanx_intrigue_failed
			remove_character_flag = POD_asakku_phalanx_fled
			remove_variable = POD_asakku_phalanx_leader_var
		}
	}
}
