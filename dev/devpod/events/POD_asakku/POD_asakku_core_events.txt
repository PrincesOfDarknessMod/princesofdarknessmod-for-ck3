######################################################
# 0001 - "What Has Always Been There" (Opening Event)
# Fires 10 days after game start for Andeleon only
######################################################

POD_asakku.0001 = {
	type = character_event
	title = POD_asakku.0001.t
	desc = {
		desc = POD_asakku.0001.desc
		desc = POD_asakku.0001.desc.mechanics
	}
	theme = vampire
	content_source = dlc_POD

	left_portrait = {
		character = root
		animation = personality_callous
	}

	trigger = {
		POD_asakku_is_andeleon_trigger = yes
		NOT = { POD_asakku_has_any_asakku_trait_trigger = yes }
		NOT = { has_character_flag = POD_asakku_suppressed }
	}

	immediate = { }

	# Option A - Accept the presence
	option = {
		name = POD_asakku.0001.a
		POD_asakku_set_depth_effect = { VALUE = 1 }
		POD_asakku_grant_soul_infection_effect = yes
		POD_asakku_elioud_setup = yes
		custom_tooltip = POD_asakku_soul_infection_accepted_tt
	}

	# Option B - Suppress it
	option = {
		name = POD_asakku.0001.b
		add_character_flag = POD_asakku_suppressed
		custom_tooltip = POD_asakku_suppressed_tt
	}
}

######################################################
# 0010 - "The Voice Becomes Louder" (Depth 25)
# Grants Asakku trait, possession, Corrupted Way faith
# Applies Sinister Mastery tier 1
######################################################

POD_asakku.0010 = {
	type = character_event
	title = POD_asakku.0010.t
	desc = POD_asakku.0010.desc
	theme = vampire
	content_source = dlc_POD

	left_portrait = {
		character = root
		animation = personality_callous
	}

	immediate = { }

	option = {
		name = POD_asakku.0010.a
		POD_asakku_grant_asakku_effect = yes
		POD_asakku_apply_sinister_mastery_effect = yes
		# Set faith to Corrupted Way
		hidden_effect = {
			set_character_faith = faith:corruptedway
		}
		# Elioud rivalry and escalation
		hidden_effect = {
			POD_asakku_elioud_rivalry_with_andeleon_effect = yes
			# Fire Curious Investigation event for Andeleon
			if = {
				limit = { POD_asakku_is_andeleon_trigger = yes }
				trigger_event = {
					id = POD_asakku.0205
					days = { 14 30 }
				}
			}
		}
		custom_tooltip = POD_asakku_depth_25_tt
	}
}

######################################################
# 0012 - "The Voice Becomes Louder" (Depth 75)
# Unlocks Soul Whispers, grants Inner Babble
# Strips Lilim Pillars tolerance, grants Del'Roh Elioud
######################################################

POD_asakku.0012 = {
	type = character_event
	title = POD_asakku.0012.t
	desc = POD_asakku.0012.desc
	theme = vampire
	content_source = dlc_POD

	left_portrait = {
		character = root
		animation = personality_callous
	}

	immediate = { }

	option = {
		name = POD_asakku.0012.a
		add_character_modifier = POD_asakku_inner_babble
		# Grant Elioud trait to the Del'Roh (holder of e_talmahera)
		hidden_effect = {
			if = {
				limit = {
					exists = title:e_talmahera.holder
				}
				title:e_talmahera.holder = {
					if = {
						limit = {
							is_alive = yes
							NOT = { has_trait = POD_asakku_elioud }
							NOT = { POD_asakku_has_any_asakku_trait_trigger = yes }
						}
						add_trait = POD_asakku_elioud
					}
				}
			}
			POD_asakku_elioud_rivalry_with_andeleon_effect = yes
			# Fire The Final Accounting event
			trigger_event = {
				id = POD_asakku.0220
				days = { 30 90 }
			}
		}
		custom_tooltip = POD_asakku_depth_75_tt
	}
}

######################################################
# 0102 - "A New Eye" (Vicissitude carrier enters court)
# Triggered via on_join_court on_action
######################################################

POD_asakku.0101 = {
	type = character_event
	title = POD_asakku.0101.t
	desc = POD_asakku.0101.desc
	theme = vampire
	content_source = dlc_POD

	left_portrait = {
		character = root
		animation = personality_callous
	}

	right_portrait = {
		character = scope:new_vicissitude_carrier
		animation = personality_rational
	}

	immediate = { }

	# Option A - Cultivate their susceptibility (start Soul Infection scheme)
	option = {
		name = POD_asakku.0101.a
		trigger = {
			exists = scope:new_vicissitude_carrier
			can_start_scheme = {
				type = POD_asakku_soul_infection_scheme
				target_character = scope:new_vicissitude_carrier
			}
		}
		hidden_effect = {
			start_scheme = {
				type = POD_asakku_soul_infection_scheme
				target_character = scope:new_vicissitude_carrier
			}
		}
		custom_tooltip = POD_asakku_new_eye_cultivate_tt
	}

	# Option B - Direct measures (Asakku only, target imprisoned)
	option = {
		name = POD_asakku.0101.b
		trigger = {
			POD_asakku_has_asakku_trigger = yes
			exists = scope:new_vicissitude_carrier
			scope:new_vicissitude_carrier = { is_imprisoned_by = root }
		}
		scope:new_vicissitude_carrier = {
			POD_asakku_grant_asakku_effect = yes
		}
		custom_tooltip = POD_asakku_new_eye_blood_inception_tt
	}

	# Option C - Keep your distance
	option = {
		name = POD_asakku.0101.c
		POD_asakku_decrement_depth_effect = { AMOUNT = 1 }
		stress_impact = {
			base = 5
		}
	}
}
