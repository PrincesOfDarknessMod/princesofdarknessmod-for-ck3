######################################################
# 0030 - Soul Infection: Target is approached
# Fires on target via on_phase_completed
# Target chooses to resist or submit
######################################################

POD_asakku.0030 = {
	type = character_event
	title = POD_asakku.0030.t
	desc = POD_asakku.0030.desc
	theme = intrigue
	content_source = dlc_POD

	right_portrait = scope:scheme.scheme_owner
	left_portrait = scope:scheme.scheme_target_character

	immediate = {
		scope:scheme.scheme_owner = { save_scope_as = attacker }
		scope:scheme.scheme_target_character = { save_scope_as = defender }
	}

	# Option A - Resist (goes to duel)
	option = {
		name = POD_asakku.0030.a
		scope:scheme.scheme_owner = {
			trigger_event = POD_asakku.0031
		}
	}

	# Option B - Submit willingly (goes to success)
	option = {
		name = POD_asakku.0030.b
		scope:scheme.scheme_owner = {
			trigger_event = POD_asakku.0032
		}

		stress_impact = {
			paranoid = minor_stress_gain
		}
		ai_chance = {
			base = -40
			opinion_modifier = {
				opinion_target = scope:attacker
			}
			modifier = {
				add = -100
				has_trait = paranoid
			}
		}
	}
}

######################################################
# 0031 - Soul Infection: The Duel
# Intrigue + Prowess duel between attacker and defender
######################################################

POD_asakku.0031 = {
	type = character_event
	title = POD_asakku.0031.t
	desc = POD_asakku.0031.desc
	theme = intrigue
	content_source = dlc_POD

	right_portrait = scope:scheme.scheme_owner
	left_portrait = scope:scheme.scheme_target_character

	immediate = {
		save_scope_value_as = {
			name = attacker_avg
			value = {
				value = scope:attacker.intrigue
				multiply = 2
				add = scope:attacker.prowess
				divide = 2
			}
		}
		save_scope_value_as = {
			name = defender_avg
			value = {
				value = scope:defender.intrigue
				multiply = 2
				add = scope:defender.prowess
				divide = 2
			}
		}
		save_scope_value_as = {
			name = stat_diff
			value = {
				value = scope:attacker_avg
				subtract = scope:defender_avg
			}
		}
	}

	option = {
		name = POD_asakku.0031.a
		duel = {
			skills = { intrigue prowess }
			value = scope:stat_diff
			45 = {
				desc = POD_asakku.0031.a.success.desc
				max = 90
				modifier = {
					add = {
						value = scope:stat_diff
						multiply = 0.75
					}
				}
				# Asakku bonus: target has Vicissitude
				modifier = {
					add = 10
					scope:defender = { has_trait = vicissitudeadvanced }
				}
				modifier = {
					add = 5
					scope:defender = {
						has_trait = vicissitude
						NOT = { has_trait = vicissitudeadvanced }
					}
				}
				# Success - infection takes hold
				hidden_effect = {
					scope:scheme.scheme_owner = {
						trigger_event = POD_asakku.0032
					}
				}
				show_as_tooltip = {
					scope:defender = {
						POD_asakku_grant_soul_infection_effect = yes
					}
				}
			}
			55 = {
				desc = POD_asakku.0031.a.failure.desc
				min = 25
				modifier = {
					add = {
						value = scope:stat_diff
						multiply = -0.45
					}
				}
				# Failure - target resists
				hidden_effect = {
					scope:scheme.scheme_owner = {
						trigger_event = POD_asakku.0033
					}
				}
				show_as_tooltip = {
					scope:defender = {
						add_opinion = {
							target = scope:attacker
							opinion = -50
							modifier = hate_opinion
						}
					}
				}
			}
		}
	}

	# Option B - Cancel the scheme
	option = {
		name = POD_asakku.0031.b
		scope:scheme = {
			end_scheme = yes
		}
		custom_tooltip = POD_asakku_cancel_scheme_tt
	}
}

######################################################
# 0032 - Soul Infection: Success (attacker notified)
######################################################

POD_asakku.0032 = {
	type = character_event
	title = POD_asakku.0032.t
	desc = POD_asakku.0032.desc
	theme = intrigue
	content_source = dlc_POD

	right_portrait = scope:scheme.scheme_owner
	left_portrait = scope:scheme.scheme_target_character

	option = {
		name = POD_asakku.0032.a
		# Apply the infection
		scope:scheme.scheme_target_character = {
			POD_asakku_grant_soul_infection_effect = yes
		}
		# Applies a weak bloodbond hook
		scope:scheme.scheme_owner = {
			add_hook = {
				target = scope:scheme.scheme_target_character 
				type = weak_bloodoath_hook
			}
		}
		# Notify target
		scope:scheme.scheme_target_character = {
			trigger_event = POD_asakku.0034
		}
		# Notify Elioud
		hidden_effect = {
			POD_asakku_change_depth_effect = { AMOUNT = 1 }
		}
		# Stress relief for the schemer
		add_stress = medium_stress_loss
	}
}
