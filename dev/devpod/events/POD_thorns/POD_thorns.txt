#I'm doing it by including regions instead of excluding them so that it's failproof for Asia
scripted_trigger POD_is_valid_for_thorns_trigger = {
	is_vampire_trigger = yes
	NOR = {
		has_trait = torpor	
		has_religion = religion:demons_religion
		POD_is_anathema_trigger = yes
	}
	OR = {
		capital_province = {
			geographical_region = world_europe
		}
		capital_province = {
			geographical_region = world_asia_minor
		}		
		capital_province = {
			geographical_region = world_middle_east
		}
		capital_province = {
			geographical_region = world_africa_north
		}		
		capital_province = {
			geographical_region = world_steppe_west
		}
		capital_province = {
			geographical_region = world_siberia
		}
	}		
}
