namespace = POD_sabbat
	
POD_sabbat.1 = {
	scope = none
	hidden = yes
	
	trigger = {
		game_start_date >= 1510.1.3
	}
	
	immediate = {
		religion:sabbat_religion = {
			# Membership
			add_to_variable_list = { name = member_clan target = doctrine:doctrine_lasombra_accepted }
			add_to_variable_list = { name = member_clan target = doctrine:doctrine_tzimisce_accepted }
			
			# Cardinal
			character:505154 = { set_cardinal_effect = { REGION = world_europe_west_iberia } }
			character:790456 = { set_cardinal_effect = { REGION = world_europe_south_italy } }
			character:790105 = { set_cardinal_effect = { REGION = ghw_region_baltic } }
			character:790114 = { set_cardinal_effect = { REGION = ghw_region_russia } }
			
			set_variable = {
				name = regent
				value = character:792372 #Gorschist has travel the time
			}
		}
	}
}

# New Cardinal
POD_sabbat.1001 = {
	type = character_event
	title = POD_sabbat.1001.t
	desc = POD_sabbat.1001.desc
	theme = camarilla
	left_portrait = {
		character = root
		animation = personality_honorable
	}

	immediate = {
		play_music_cue = "mx_cue_epic_sacral_moment"
	}

	option = {
		name = POD_sabbat.1001.a
		custom_tooltip = POD_sabbat_greece_cardinal_decision_effect_tooltip
		
		set_cardinal_effect = { REGION = POD_region_greece }
		save_scope_as = new_cardinal
		every_player = { 
			limit = { NOT = { this = root } }
			trigger_event = POD_sabbat.1101
		}
	}
}
