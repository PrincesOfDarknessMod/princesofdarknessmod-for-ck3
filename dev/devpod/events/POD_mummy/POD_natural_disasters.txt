POD_natural_disasters.0100 = {
	hidden = yes
	scope = none

	immediate = {
		scope:target_title = {
			if = {
				limit = {
					OR = {
						var:disaster_outcome ?= flag:stopped
						var:disaster_outcome ?= flag:prosperity
						var:disaster_outcome ?= flag:great_relief
					}
				}
				# Do nothing
			}
			else_if = {
				limit = { var:disaster_outcome ?= flag:relief }
				scope:target_title = {
					every_in_de_jure_hierarchy = {
						limit = { tier = tier_county }
						change_county_control = -10
						change_development_level = -1
					}
				}
			}
			else_if = {
				limit = { var:disaster_outcome ?= flag:aggravated }
				scope:target_title = {
					every_in_de_jure_hierarchy = {
						limit = { tier = tier_county }
						change_county_control = -50
						change_development_level = -5
						if = {
							limit = { scope:disaster_type ?= flag:POD_weather_magic_storm }
							add_county_modifier = {
								modifier = weather_magic_storm_modifier
								years = 5
							}
						}
						if = {
							limit = { scope:disaster_type ?= flag:POD_weather_magic_earthquake }
							add_county_modifier = {
								modifier = weather_magic_earthquake_modifier
								years = 5
							}
						}
						if = {
							limit = { scope:disaster_type ?= flag:POD_weather_magic_flood }
							add_county_modifier = {
								modifier = weather_magic_flood_modifier
								years = 5
							}
						}
					}
					every_de_jure_county_holder = {
						limit = {
							is_ai = yes
						}
						every_courtier_or_guest = {
							limit = { has_any_nickname = no }
							pod_death_no_killer = { DEATH_REASON = death_accident }
						}
					}
				}
			}
			else = {
				scope:target_title = {
					every_in_de_jure_hierarchy = {
						limit = { tier = tier_county }
						change_county_control = -25
						change_development_level = -3
					}
					every_de_jure_county_holder = {
						limit = {
							is_ai = yes
						}
						random_courtier_or_guest = {
							limit = { has_any_nickname = no }
							pod_death_no_killer = { DEATH_REASON = death_accident }
						}
						random_courtier_or_guest = {
							limit = { has_any_nickname = no }
							pod_death_no_killer = { DEATH_REASON = death_accident }
						}
						random_courtier_or_guest = {
							limit = { has_any_nickname = no }
							pod_death_no_killer = { DEATH_REASON = death_accident }
						}
					}
				}
			}
		}
		every_player = {
			limit = {
				OR = {
					is_target_in_variable_list = {
						name   = tracked_disasters
						target = scope:target_title
					}
					capital_county = {
						scope:target_title = {
							is_de_jure_liege_or_above_target = prev
						}
					}
				}
			}
			trigger_event = POD_natural_disasters.0101
		}
	}
}
