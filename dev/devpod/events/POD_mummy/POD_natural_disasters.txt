POD_natural_disasters.0003 = {
	type = character_event
	content_source = dlc_POD
	title = POD_natural_disasters.0003.t
	desc = POD_natural_disasters.0003.desc
	theme = mummy
	
	option = {
		name = POD_natural_disasters.0003.a
		
		trigger = {
			has_perk = becalm_perk 
			trigger_if = {
				limit = { exists = scope:disaster_flood }
				has_perk = inundation_of_the_nile_perk
			}
			trigger_if = {
				limit = { exists = scope:disaster_earthquake }
				has_perk = rousing_apophis_perk
			}
			trigger_if = {
				limit = { exists = scope:disaster_storm }
				has_perk = sandstorm_perk
			}
			# trigger_if = {
				# limit = { exists = scope:disaster_eruption }
				# has_perk = call_the_stars_perk
			# }
		}
		show_as_unavailable = { has_trait = mummy }
		
		scope:target_title = {
			title_capital_county = {
				title_province = {
					save_scope_as = disaster_location
				}
			}
		}
		if = {
			limit = { NOT = { location = scope:disaster_location } }
			current_travel_plan ?= {
				if = {
					limit = { travel_plan_owner = root }
					abort_travel_plan = yes
				}
				else = {
					remove_character = root
				}
			}
			start_travel_plan = {
				destination = scope:disaster_location
				on_arrival_event = POD_natural_disasters.0005
				on_arrival_destinations = all_but_last
			}
			trigger_event = {
				id = POD_natural_disasters.0004
				days = 1
			}
		}
		else = {
			trigger_event = POD_natural_disasters.0005
		}
		add_character_flag = goto_the_disaster
	}
	
	option = {
		name = POD_natural_disasters.0003.a
		
		trigger = { has_perk = manipulate_weather_perk }
		
		scope:target_title = {
			title_capital_county = {
				title_province = {
					save_scope_as = disaster_location
				}
			}
		}
		if = {
			limit = { NOT = { location = scope:disaster_location } }
			current_travel_plan ?= {
				if = {
					limit = { travel_plan_owner = root }
					abort_travel_plan = yes
				}
				else = {
					remove_character = root
				}
			}
			start_travel_plan = {
				destination = scope:disaster_location
				on_arrival_event = POD_natural_disasters.0005
				on_arrival_destinations = all_but_last
			}
			trigger_event = {
				id = POD_natural_disasters.0004
				days = 1
			}
		}
		else = {
			trigger_event = POD_natural_disasters.0005
		}
		add_character_flag = goto_the_disaster
	}
	
	option = {
		name = POD_natural_disasters.0003.c
	}
}
