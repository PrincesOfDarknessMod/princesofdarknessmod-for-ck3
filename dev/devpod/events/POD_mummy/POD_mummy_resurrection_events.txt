namespace = POD_mummy_resurrection

POD_mummy_resurrection.100 = {
	type = character_event
	title = POD_mummy_resurrection.100.t
	desc = POD_mummy_resurrection.100.desc
	right_portrait = root
	theme = mummy
	
	immediate = {
		if = {
			limit = { has_trait = cabiri }
			set_variable = { name = spirit_location value = flag:hades }
		}
		else_if = {
			limit = { has_trait = wutian }
			set_variable = { name = spirit_location value = flag:gates_of_heaven }
		}
		else_if = {
			limit = { has_trait = wukuei }
			set_variable = { name = spirit_location value = flag:yomi_wan }
		}
		else = {
			set_variable = { name = spirit_location value = flag:shadow_realm }
		}
		remove_variable = spirit_destination
		POD_add_umbra_modifier_effect = yes
	}
	
	option = {
		name = POD_mummy_resurrection.100.a
		add_piety = mummy_resurrect_piety_loss
		
		trigger_event = POD_mummy_resurrection.101
	}
}

POD_mummy_resurrection.102 = {
	type = character_event
	hidden = yes
	
	immediate = {
		if = {
			limit = { var:spirit_location = flag:hades }
			set_variable = { name = spirit_destination value = flag:dark_kingdom_of_iron }
		}
		if = {
			limit = { var:spirit_location = flag:gates_of_heaven }
			set_variable = { name = spirit_destination value = flag:elemental_courts }
		}
		if = {
			limit = { var:spirit_location = flag:yomi_wan }
			set_variable = { name = spirit_destination value = flag:dark_kingdom_of_jade }
		}
		if = {
			limit = { var:spirit_location = flag:shadow_realm }
			set_variable = { name = spirit_destination value = flag:dark_kingdom_of_sand }
		}
		
		if = {
			limit = { has_perk = empower_ba_perk }
			set_variable = { name = spirit_destination value = flag:skinlands }
		}
		trigger_event = POD_spirit_travel.10
	}
}
