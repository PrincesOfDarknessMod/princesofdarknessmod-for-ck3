namespace = POD_mummy_resurrection

POD_mummy_resurrection.1 = {
	type = character_event
	content_source = dlc_POD
	title = POD_mummy_resurrection.1.t
	desc = POD_mummy_resurrection.1.desc
	theme = mummy

	left_portrait = {
		character = root
		animation = sadness 
	}

	option = {
		name = POD_mummy_resurrection.1.a
		custom_tooltip = POD_mummy_resurrection.1.a.tt
		trigger = { POD_should_have_integrity_trigger = yes }
		trigger_event = POD_mummy_resurrection.2
	}

	option = {
		name = POD_mummy_resurrection.1.b
		custom_tooltip = POD_mummy_resurrection.1.b.tt
		trigger = { 
			OR = {
				AND = {
					has_trait = cabiri 
					NOT = { is_target_in_global_variable_list = { name = unavailable_unique_decisions target = flag:secure_hades_objective_decision } }
				}
				AND = {
					has_trait = custom_mummy_dynasty 
					NOT = { is_target_in_global_variable_list = { name = unavailable_unique_decisions target = flag:spell_of_life_stabilize_new_realm_decision } }
				}
			}
		}
		add_trait_xp = { trait = mummy value = -10 }
	}

	option = {
		name = POD_mummy_resurrection.1.c
		custom_tooltip = POD_mummy_resurrection.1.c.tt
		trigger = {
			OR = {
				has_trait = banemummy
				has_trait = wukuei
			}
		}
		add_stress = 200
	}

	option = {
		name = POD_mummy_resurrection.1.d
		custom_tooltip = POD_mummy_resurrection.1.d.tt
		trigger = { 
			has_trait = cabiri
			is_target_in_global_variable_list = { name = unavailable_unique_decisions target = flag:secure_hades_objective_decision }
		}
	}
}

POD_mummy_resurrection.2 = {
	type = character_event
	content_source = dlc_POD
	title = {
		first_valid = {
			triggered_desc = {
				trigger = {
					has_trait = wutian
				}
				desc = POD_mummy_resurrection.2.t.wutian
			}
			triggered_desc = {
				trigger = {
					has_trait = custom_mummy_dynasty
				}
				desc = POD_mummy_resurrection.2.t.custom
			}
			desc = POD_mummy_resurrection.2.t
		}
	}
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_trait = wutian }
				desc = POD_mummy_resurrection.2.desc.wutian
			}
			triggered_desc = {
				trigger = { has_trait = custom_mummy_dynasty }
				desc = POD_mummy_resurrection.2.desc.custom
			}
			desc = POD_mummy_resurrection.2.desc
		}
	}
	theme = mummy

	left_portrait = {
		character = root
		animation = sadness 
	}

	option = {
		name = POD_mummy_resurrection.2.a
		custom_tooltip = POD_mummy_resurrection.2.a.tt
		if = { limit = { has_trait_xp = { trait = integrity value = 0 } } 
			death = {
				death_reason = death_final_death
			}
		}
		if = { limit = { has_trait_xp = { trait = integrity value > 1 } has_trait_xp = { trait = integrity value < 10 } } 
			random_list = {
				1 = { add_diplomacy_skill = -1 }
				1 = { add_martial_skill = -1 }
				1 = { add_stewardship_skill = -1 }
				1 = { add_intrigue_skill = -1 }
				1 = { add_learning_skill = -1 }
			}
		}
		if = { limit = { has_trait_xp = { trait = integrity value > 1 } has_trait_xp = { trait = integrity value < 20 } } 
			random_list = {
				1 = { trigger = { hekau_alchemy_lifestyle_perk_points > 0 } add_hekau_alchemy_lifestyle_perk_points = -5 }
				1 = { trigger = { hekau_amulets_lifestyle_perk_points > 0 } add_hekau_amulets_lifestyle_perk_points = -5 }
				1 = { trigger = { hekau_celestial_lifestyle_perk_points > 0 } add_hekau_celestial_lifestyle_perk_points = -5 }
				1 = { trigger = { hekau_effigy_lifestyle_perk_points > 0 } add_hekau_effigy_lifestyle_perk_points = -5 }
				1 = { trigger = { hekau_necromancy_lifestyle_perk_points > 0 } add_hekau_necromancy_lifestyle_perk_points = -5 }
				1 = { trigger = { hekau_nomenclature_lifestyle_perk_points > 0 } add_hekau_nomenclature_lifestyle_perk_points = -5 }
				1 = { trigger = { tan_lifestyle_perk_points > 0 } add_tan_lifestyle_perk_points = -5 }
				1 = { 
					trigger = { 
						hekau_alchemy_lifestyle_perk_points = 0 
						hekau_amulets_lifestyle_perk_points = 0 
						hekau_celestial_lifestyle_perk_points = 0 
						hekau_effigy_lifestyle_perk_points = 0 
						hekau_necromancy_lifestyle_perk_points = 0 
						hekau_nomenclature_lifestyle_perk_points = 0 
						tan_lifestyle_perk_points = 0 
					}
					add_stress = 200
				}
			}
		}
		if = { limit = { has_trait_xp = { trait = integrity value > 1 } has_trait_xp = { trait = integrity value < 30 } } 
			add_character_modifier = {
				modifier = POD_mummy_integrity_punishment_modifier
				years = 5
			}
		}
	}

	option = {
		name = POD_mummy_resurrection.2.b
		custom_tooltip = POD_mummy_resurrection.2.b.tt
		trigger = { has_trait_xp = { trait = integrity value = 0 } }
		if = { limit = { has_trait = wutian } POD_mummy_initialization_wukuei_effect = yes }
		else = { POD_mummy_initialization_banemummy_effect = yes }
	}

	option = {
		trigger = { has_perk = eating_the_heart_perk NOT = { has_character_flag = ate_the_heart_flag } }
		name = POD_mummy_resurrection.2.c
		custom_tooltip = POD_mummy_resurrection.2.c.tt
		add_character_flag = {
			flag = ate_the_heart_flag
			years = 50
		}
	}

	option = {
		trigger = { has_character_flag = stabilized_custom_shadow_realm }
		name = POD_mummy_resurrection.2.d
		custom_tooltip = POD_mummy_resurrection.2.d.tt
	}

	option = {
		trigger = { has_character_modifier = wutian_ninth_immortal }
		name = POD_mummy_resurrection.2.e
		custom_tooltip = POD_mummy_resurrection.2.d.tt
	}
}
