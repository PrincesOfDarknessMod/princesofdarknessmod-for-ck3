namespace = POD_mummy_resurrection

POD_mummy_resurrection.100 = {
	type = character_event
	content_source = dlc_POD
	title = {
		first_valid = {
			triggered_desc = {
				trigger = {
					OR = {
						faith = { has_doctrine = mummy_special_kingdom_hades_doctrine }
						faith = { has_doctrine = mummy_special_kingdom_hellheim_doctrine }
						faith = { has_doctrine = mummy_special_kingdom_yomi_doctrine }
						faith = { has_doctrine = mummy_special_kingdom_hell_doctrine }
					}
				}
				desc = POD_mummy_resurrection.100.t2
			}
			desc = POD_mummy_resurrection.100.t
		}
	}
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					OR = {
						faith = { has_doctrine = mummy_special_kingdom_hades_doctrine }
						faith = { has_doctrine = mummy_special_kingdom_hellheim_doctrine }
						faith = { has_doctrine = mummy_special_kingdom_yomi_doctrine }
						faith = { has_doctrine = mummy_special_kingdom_hell_doctrine }
					}
				}
				desc = POD_mummy_resurrection.100.desc2
			}
			desc = POD_mummy_resurrection.100.desc
		}
	}
	right_portrait = root
	theme = mummy
	
	immediate = {
		if = {
			limit = { faith = { has_doctrine = mummy_special_kingdom_hades_doctrine } }
			POD_umbra_send_effect = { LOCATION = flag:dark_kingdom_of_hades }
		}
		else_if = {
			limit = { faith = { has_doctrine = mummy_special_kingdom_hellheim_doctrine } }
			POD_umbra_send_effect = { LOCATION = flag:dark_kingdom_of_hellheim }
		}
		else_if = {
			limit = { faith = { has_doctrine = mummy_special_kingdom_yomi_doctrine } }
			POD_umbra_send_effect = { LOCATION = flag:dark_kingdom_of_yomi }
		}
		else_if = {
			limit = { faith = { has_doctrine = mummy_special_kingdom_hell_doctrine } }
			POD_umbra_send_effect = { LOCATION = flag:dark_kingdom_of_hell }
		}
		else_if = {
			limit = { has_trait = cabiri }
			POD_umbra_send_effect = { LOCATION = flag:hades }
		}
		else_if = {
			limit = { has_trait = wutian }
			POD_umbra_send_effect = { LOCATION = flag:gates_of_heaven }
		}
		else_if = {
			limit = { has_trait = wukuei }
			add_character_modifier = trapped_in_yomi_modifier
			POD_umbra_send_effect = { LOCATION = flag:yomi_wan }
		}
		else = {
			POD_umbra_send_effect = { LOCATION = flag:shadow_realm }
		}
	}
	
	option = {
		name = POD_mummy_resurrection.100.a
		trigger = {
			NOT = {
				faith = { has_doctrine = mummy_special_kingdom_hades_doctrine }
				faith = { has_doctrine = mummy_special_kingdom_hellheim_doctrine }
				faith = { has_doctrine = mummy_special_kingdom_yomi_doctrine }
				faith = { has_doctrine = mummy_special_kingdom_hell_doctrine }
			}
		}
		add_piety = mummy_resurrect_piety_loss
		
		trigger_event = POD_mummy_resurrection.101
	}
	option = {
		name = POD_mummy_resurrection.100.b
		trigger = {
			OR = {
				faith = { has_doctrine = mummy_special_kingdom_hades_doctrine }
				faith = { has_doctrine = mummy_special_kingdom_hellheim_doctrine }
				faith = { has_doctrine = mummy_special_kingdom_yomi_doctrine }
				faith = { has_doctrine = mummy_special_kingdom_hell_doctrine }
			}
		}
		add_piety = mummy_resurrect_piety_loss
		if = {
			limit = { has_perk = empower_ba_perk }
			custom_tooltip = "empower_ba_perk_resurrection"
			POD_umbra_send_effect = { LOCATION = flag:earth }
		}
	}
	
	option = {
		name = POD_mummy_resurrection.100.c
		trigger = {
			has_trait = judge_of_maat
		}
		add_piety = mummy_resurrect_piety_loss
		
		trigger_event = POD_mummy_resurrection.102
	}
}

POD_mummy_resurrection.102 = {
	type = character_event
	content_source = dlc_POD
	title = POD_mummy_resurrection.102.t
	desc = POD_mummy_resurrection.102.desc
	right_portrait = root
	theme = mummy
	
	option = {
		name = POD_mummy_resurrection.102.a
		trigger = { has_perk = empower_ba_perk }
		show_as_unavailable = { NOT = { has_perk = empower_ba_perk } }
		POD_umbra_send_effect = { LOCATION = flag:earth }
	}
	
	option = {
		name = POD_mummy_resurrection.102.b
		add_stress = major_stress_gain
		add_prestige = minor_prestige_loss

		switch = {
			trigger = var:spirit_location
			flag:hades = { POD_umbra_send_effect = { LOCATION = flag:dark_kingdom_of_iron } }
			flag:gates_of_heaven = { POD_umbra_send_effect = { LOCATION = flag:elemental_courts } }
			flag:yomi_wan = {
				remove_character_modifier = trapped_in_yomi_modifier
				POD_umbra_send_effect = { LOCATION = flag:dark_kingdom_of_jade } 
			}
			flag:shadow_realm = { POD_umbra_send_effect = { LOCATION = flag:dark_kingdom_of_sand } }
		}
	}
}
