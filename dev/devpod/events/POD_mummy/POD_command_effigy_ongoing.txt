#effigy is damaged by brigands
POD_command_effigy_ongoing.1000 = {
	type = character_event
	content_source = dlc_POD
	title = POD_command_effigy_ongoing.1000.t 
	desc = POD_command_effigy_ongoing.1000.desc
	theme = mummy
	cooldown = { years = 1 }

	artifact = { # To display the artifact in the event-window
		target = scope:this_effigy
		position = lower_center_portrait
	}
	
	immediate = {
		random_character_artifact = {			
			limit = {
				has_variable = creature
				var:active_effigy = yes
			}
			save_scope_as = this_effigy
		} 
	}

	option = {
		name = POD_command_effigy_ongoing.1000.option.a
		scope:this_effigy = {
			random_list = {
				60 = {
					modifier = {
						OR = {
							scope:this_effigy = { var:creature = flag:artifact_divine_beast }
							scope:this_effigy = { var:creature = flag:artifact_princely_creature }
						}
						add = 20
					}
					modifier = {
						OR = {
							scope:this_effigy = { var:creature = flag:artifact_kingly_beast }
							scope:this_effigy = { var:creature = flag:artifact_superior_creature }
						}
						add = 10
					}
					# Remove 25% of remaining durability
					desc = POD_command_effigy_ongoing.1000.option.a.LIGHT_DAMAGE
					add_durability = {
						value = artifact_durability
						multiply = -0.25
					}
					hidden_effect = {
						scope:owner = {
							send_interface_toast = {
								title = POD_command_effigy_ongoing.1000.option.a.t
								custom_tooltip = POD_command_effigy_ongoing.1000.option.a.tt.LIGHT_DAMAGE
								left_icon = scope:owner
								right_icon = scope:this_effigy
							}
						}
					}		
				}
				30 = {
					# Remove 50% of remaining durability
					desc = POD_command_effigy_ongoing.1000.option.a.MODERATE_DAMAGE
					add_durability = {
						value = artifact_durability
						multiply = -0.50
					}
					hidden_effect = {
						scope:owner = {
							send_interface_toast = {
								title = POD_command_effigy_ongoing.1000.option.a.t
								custom_tooltip = POD_command_effigy_ongoing.1000.option.a.tt.MODERATE_DAMAGE
								left_icon = scope:owner
								right_icon = scope:this_effigy
							}
						}
					}
				}
				10 = {
					# Remove 75% of remaining durability 
					desc = POD_command_effigy_ongoing.1000.option.a.HEAVY_DAMAGE
					add_durability = {
						value = artifact_durability
						multiply = -0.75
					}
					hidden_effect = {
						scope:owner = {
							send_interface_toast = {
								title = POD_command_effigy_ongoing.1000.option.a.t
								custom_tooltip = POD_command_effigy_ongoing.1000.option.a.tt.HEAVY_DAMAGE
								left_icon = scope:owner
								right_icon = scope:this_effigy
							}
						}
					}
				}
			}
		}     
	}
}

scripted_trigger command_effigy_appropriately_evil_rival = {
	is_available_ai_adult = yes
	ai_compassion <= low_positive_ai_value
}
#effigy is stolen by a rival
POD_command_effigy_ongoing.1030 = {
	type = character_event
	content_source = dlc_POD
	title = POD_command_effigy_ongoing.1030.t
	desc = POD_command_effigy_ongoing.1030.desc
	theme = generic_intrigue_scheme
	cooldown = { years = 5 }

	left_portrait = {
		character = root
		animation = shock
	}
	#show rival
	right_portrait = {
		character = scope:sabotage_rival
		animation = schadenfreude
	}  

	artifact = { # To display the artifact in the event-window
		target = scope:this_effigy
		position = lower_center_portrait
	}

	widget = {
		gui = "event_window_widget_scheme"
		container = "custom_widgets_container"
	}

	trigger = {
		OR = {
			any_relation = {
				type = rival
				command_effigy_appropriately_evil_rival = yes
				trigger_if = {
					limit = { #make sure vassals don't get stupid
						OR = {
							is_vassal_or_below_of = scope:owner
							is_courtier_of = scope:owner
						}
					}
					dread_modified_ai_boldness = {
						dreaded_character = scope:owner
						value >= very_high_positive_ai_value
					}
				}
				trigger_else = {
					dread_modified_ai_boldness = {
						dreaded_character = scope:owner
						value >= very_low_positive_ai_value
					}
				}
			}
			any_relation = {
				type = potential_rival
				command_effigy_appropriately_evil_rival = yes
				trigger_if = {
					limit = { #make sure vassals don't get stupid
						OR = {
							is_vassal_or_below_of = scope:owner
							is_courtier_of = scope:owner
						}
					}
					dread_modified_ai_boldness = {
						dreaded_character = scope:owner
						value >= very_high_positive_ai_value
					}
				}
				trigger_else = {
					dread_modified_ai_boldness = {
						dreaded_character = scope:owner
						value >= very_low_positive_ai_value
					}
				}
			}
		}
	}

	immediate = {
		random_character_artifact = {
			limit = {
				has_variable = creature
				var:active_effigy = yes
			}			
			save_scope_as = this_effigy
		}
		every_relation = {
			type = rival
			limit = {
				command_effigy_appropriately_evil_rival = yes
				trigger_if = {
					limit = { #make sure vassals don't get stupid
						OR = {
							is_vassal_or_below_of = scope:owner
							is_courtier_of = scope:owner
						}
					}
					dread_modified_ai_boldness = {
						dreaded_character = scope:owner
						value >= very_high_positive_ai_value
					}
				}
				trigger_else = {
					dread_modified_ai_boldness = {
						dreaded_character = scope:owner
						value >= very_low_positive_ai_value
					}
				}
			}
			add_to_list = evil_fellas
		}
		every_relation = {
			type = potential_rival
			limit = {
				command_effigy_appropriately_evil_rival = yes
				trigger_if = {
					limit = { #make sure vassals don't get stupid
						OR = {
							is_vassal_or_below_of = scope:owner
							is_courtier_of = scope:owner
						}
					}
					dread_modified_ai_boldness = {
						dreaded_character = scope:owner
						value >= very_high_positive_ai_value
					}
				}
				trigger_else = {
					dread_modified_ai_boldness = {
						dreaded_character = scope:owner
						value >= very_low_positive_ai_value
					}
				}
			}
			add_to_list = evil_fellas
		}
		
		random_in_list = {
			list = evil_fellas
			weight = {
				base = 1
				modifier = {
					add = 10
					is_of_major_interest_to_root_trigger = yes
				}
				modifier = {
					add = 5
					is_of_minor_interest_to_root_trigger = yes
				}
				modifier = {
					add = 5
					has_relation_rival = root
				}
				modifier = {
					add = {
						value = ai_vengefulness
						divide = 5
					}
					ai_vengefulness > 0
				}
			}
			save_scope_as = sabotage_rival
		}
		add_prestige = minor_prestige_loss 
		add_opinion = { #give imprisonment reason
			target = scope:sabotage_rival
			modifier = stole_artifact_from_me
		}
	}

	option = {
		name = POD_command_effigy_ongoing.1030.option.a #you may keep it
		stress_impact = {
			generous = minor_stress_impact_loss	
			forgiving = minor_stress_impact_loss
			vengeful = medium_stress_impact_gain
			greedy = medium_stress_impact_gain
		}
		trigger = {
			has_perk = scepter_of_the_vizier_perk
		}
		#change ownership of effigy
		scope:this_effigy = { 
			set_owner = {
				target = scope:sabotage_rival 
				history = {
					location = scope:owner.location
					actor = scope:owner
					recipient = scope:sabotage_rival
					type = inherited
				}
			}
			#transfer cost
			if = {
				limit = { 
					var:creature = flag:artifact_simple_creature
				}
				scope:owner = { POD_hekau_cost_effect = { COUNT = 1 } }
			}
			if = {
				limit = { 
					OR = {
						var:creature = flag:artifact_major_creature
						var:creature = flag:artifact_lesser_creature
					}             
				}
				scope:owner = { POD_hekau_cost_effect = { COUNT = 2 } }
			}
			if = {
				limit = { 
					OR = {
						var:creature = flag:artifact_superior_creature
						var:creature = flag:artifact_princely_creature
						var:creature = flag:artifact_viziers_beast
						var:creature = flag:artifact_kingly_beast
						var:creature = flag:artifact_divine_beast
					}            
				}
				scope:owner = { POD_hekau_cost_effect = { COUNT = 3 } }
			}
		}
		scope:sabotage_rival = {
			add_stress = major_stress_impact_loss
		}
		ai_chance = {
			base = 0
			modifier = {
				#TODO
			}
		}
	}

	option = {
		name = POD_command_effigy_ongoing.1030.option.b #try to negotiate ransom
		duel = {
			skill = stewardship
			target = scope:sabotage_rival
			12 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = tiny_positive_duel_skill_multiplier
				}
				send_interface_toast = {
					title = POD_command_effigy_ongoing.1030.option.b.t.PAY_MINOR
					pay_short_term_gold = {
						target = scope:sabotage_rival
						gold = minor_gold_value
					}
					stress_impact = {
						base = miniscule_stress_impact_gain
						greedy = miniscule_stress_impact_gain
					}
				}
			}
			24 = {
				send_interface_toast = {
					title = POD_command_effigy_ongoing.1030.option.b.t.PAY_MEDIUM
					pay_short_term_gold = {
						target = scope:sabotage_rival
						gold = medium_gold_value
					}
					stress_impact = {
						base = minor_stress_impact_gain
						greedy = minor_stress_impact_gain
					}
				}
			}
			12 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = tiny_negative_duel_skill_multiplier
				}
				send_interface_toast = {
					title = POD_command_effigy_ongoing.1030.option.b.t.PAY_MAJOR
					pay_short_term_gold = {
						target = scope:sabotage_rival
						gold = major_gold_value
					}
					stress_impact = {
						base = medium_stress_impact_gain
						greedy = medium_stress_impact_gain
					}
				}
			}
		}
		scope:sabotage_rival = {
			add_stress = major_stress_impact_loss
			progress_towards_rival_effect = {
				CHARACTER = root
                REASON = rival_artifact_destroyer
				OPINION = 0
			}
		}
		ai_chance = {
			base = 0
			modifier = {
				#TODO:greediness?
			}
		}
	}

	option = {
		name = POD_command_effigy_ongoing.1030.option.c #mine!
		#change ownership of effigy
		scope:this_effigy = { 
			set_owner = {
				target = scope:sabotage_rival 
				history = {
					location = scope:owner.location
					actor = scope:owner
					recipient = scope:sabotage_rival
					type = stolen
				}
			}
		}
		add_personal_artifact_claim = scope:this_effigy
		scope:sabotage_rival = {
			add_stress = major_stress_impact_loss
			progress_towards_rival_effect = {
				CHARACTER = root
                REASON = rival_artifact_destroyer
				OPINION = 0
			}
		}
		stress_impact = {	
			base = medium_stress_impact_gain
			forgiving = massive_stress_impact_gain
			compassionate = major_stress_impact_gain
			calm = major_stress_impact_gain
		}
		ai_chance = {
			base = 0
			modifier = {
				add = ai_vengefulness
			}
		}
	}
}

#effigy discovers a treasure
POD_command_effigy_ongoing.1050 = {
	type = character_event
	content_source = dlc_POD
	title = POD_command_effigy_ongoing.1050.t
	desc = POD_command_effigy_ongoing.1050.desc
	theme = mummy
	cooldown = { years = 1 }

	immediate = {
		#random list of crap - decent artifact pieces
		hidden_effect = {
			scope:owner = {
				random_list = {
					65 = {
						save_temporary_scope_as = common
					}
					30 = { 
						save_temporary_scope_as = masterwork
					} 
					5 = { 
						modifier = {
							any_character_artifact = {
								has_variable = creature
								var:active_effigy = yes
								OR = {
									var:creature = flag:artifact_kingly_beast
									var:creature = flag:artifact_superior_creature
								}
							}
							add = 5
						}
						modifier = {
							any_character_artifact = {
								has_variable = creature
								var:active_effigy = yes
								OR = {
									var:creature = flag:artifact_divine_beast
									var:creature = flag:artifact_princely_creature
								}
							}
							add = 10
						}
						save_temporary_scope_as = famed
					}
				}
			}
		}	
	}

	option = {
		name = POD_command_effigy_ongoing.1050.option.a #sell it
		#give money
		if = {
			limit = { 
				exists = scope:common
			}
			add_gold = tiny_gold_value
		}
		else_if = {
			limit = {
				exists = scope:masterwork
			}
			add_gold = minor_gold_value
		}
		else_if = {
			limit = {
				exists = scope:famed
			}
			add_gold = medium_gold_value
		}
		else = { add_gold = 10 }
	}

	option = {
		name = POD_command_effigy_ongoing.1050.option.b #keep it
		send_interface_toast = {
			title = POD_command_effigy_ongoing.1050.option.b.t.KEEP_IT
			if = {
				limit = { 
					exists = scope:common
				}
				gain_common_artifact_pieces_effect = { VALUE = 1 }
			}
			if = {
				limit = {
					exists = scope:masterwork
				}
				gain_masterwork_artifact_pieces_effect = { VALUE = 1 }
			}
			if = {
				limit = {
					exists = scope:famed
				}
				gain_famed_artifact_pieces_effect = { VALUE = 1 }
			}
		}
		add_prestige = 20
	}
}
