# Request access to Secret Writings
POD_mummy_spell_of_life.1 = {
	type = character_event
	content_source = dlc_POD
	theme = culture_change

	title = POD_mummy_spell_of_life.1.t
	desc = POD_mummy_spell_of_life.1.desc
	
	left_portrait = {
		character = root
		animation = holding_scrolls
	}
	right_portrait = {
		character = scope:writings_cabirus_owner
		animation = interested
	}

	immediate = {
		every_artifact = {
			limit = {
				OR = {
					artifact_type = miscellaneous_when_not_court
					artifact_type = book
					artifact_slot_type = journal
				}
				has_variable = writings_cabirus
			}
			artifact_owner = {
				save_scope_as = writings_cabirus_owner
			}
		}
	}

	option = {
		name = POD_mummy_spell_of_life.1.a
		trigger = {
			any_character_artifact = {
				count >= 3
				OR = {
					artifact_type = miscellaneous_when_not_court
					artifact_type = book
					artifact_slot_type = journal
				}
				OR = {
					rarity = illustrious
					rarity = famed
					rarity = masterwork
				}
				NOT = { has_variable = ungiftable }
			}
		}
		ordered_character_artifact = {
			limit = {
				OR = {
					artifact_type = miscellaneous_when_not_court
					artifact_type = book
					artifact_slot_type = journal
				}
				OR = {
					rarity = illustrious
					rarity = famed
					rarity = masterwork
				}
				NOT = { has_variable = ungiftable }
			}
			max = 3
			order_by = artifact_rarity 
			
			set_owner = {
				target = scope:writings_cabirus_owner
				history = {
					type = given
					actor = root
					recipient = scope:writings_cabirus_owner
					location = root.location
				}
			}
		}
		custom_tooltip = {
			text = request_access_writings_cabirus_decision_effect
			add_character_flag = access_to_writings_cabirus
		}
	}
	option = {
		name = POD_mummy_spell_of_life.1.b
		custom_tooltip = {
			text = request_access_writings_cabirus_decision_effect
			add_character_flag = access_to_writings_cabirus
		}
		add_character_modifier = {
			modifier = drained_troops_modifier
			years = 10
		}
	}
	option = {
		name = POD_mummy_spell_of_life.1.c
		trigger = {
			highest_held_title_tier >= tier_duchy
			completely_controls = root.capital_county.duchy
			root.capital_county.duchy = {
				any_in_de_jure_hierarchy = {
					count = all
					trigger_if = {
						limit =  { tier = tier_county }
						development_level > 5
					}
					trigger_else = {
						always = yes
					}
				}
			}
		}
		custom_tooltip = {
			text = request_access_writings_cabirus_decision_effect
			add_character_flag = access_to_writings_cabirus
		}
		custom_tooltip = {
			text = POD_mummy_spell_of_life.1.c.fuel_experiments
			every_sub_realm_county = {
				limit = {
					tier = tier_county
					duchy = root.capital_county.duchy
				}
				add_county_modifier = {
					modifier = extinguish_life_modifier
					years = 1
				}
				change_development_level = -5
			}
		}
	}
	option = {
		name = POD_mummy_spell_of_life.1.d
		trigger = {
			any_in_global_list = {
				variable = tremere_council_list
				this = root
			}
			scope:writings_cabirus_owner = {
				has_religion = religion:codeoftremere_religion
			}
		}
		custom_tooltip = {
			text = request_access_writings_cabirus_decision_effect
			add_character_flag = access_to_writings_cabirus
		}
	}
	option = {
		name = POD_mummy_spell_of_life.1.e
	}
}

POD_mummy_spell_of_life.10 = {
	type = character_event
	content_source = dlc_POD
	theme = vassal

	title = POD_mummy_spell_of_life.10.t
	desc = POD_mummy_spell_of_life.10.desc
	
	left_portrait = {
		character = root
		animation = schadenfreude
	}

	immediate = {
		hidden_effect = {
			# Unique modifiers for just horus rather than generic debate
			if = {
				limit = {
					has_trait = vampire
					NOR = {
						house = house:house_osiris
						faith = { has_doctrine = special_doctrine_osirian_league }
					}
				}
				POD_change_opponents_debate_force_effect = { COUNT = 250 }
			}
			if = {
				limit = {
					has_trait = mummy
					OR = {
						NOT = { has_trait = shemsuheru }
						faith = faith:ishmaelite
					}
				}
				POD_change_opponents_debate_force_effect = { COUNT = 250 }
			}
			if = {
				limit = {
					OR = {
						faith = faith:bubasti
						house = house:house_osiris
						has_religion = religion:silentstrider_religion
						faith = { has_doctrine = special_doctrine_osirian_league }
						AND = {
							has_trait = shemsuheru
							NOT = { faith = faith:ishmaelite }
						}
					}
				}
				POD_change_character_debate_force_effect = { COUNT = 250 }
			}
			if = {
				limit = {
					this = character:505071  # Mithras
				}
				POD_change_character_debate_force_effect = { COUNT = 100 }
			}
			if = {
				limit = {
					has_trait = supehunter
					faith = faith:ikhwan_al_safa
					is_target_in_global_variable_list = {
						name = unavailable_unique_decisions
						target = flag:ikhwan_al_safa_egypt_objective_decision
					}
				}
				POD_change_character_debate_force_effect = { COUNT = 100 }
			}
			if = {
				limit = {
					OR = {
						has_trait = mummy
						faith = { has_doctrine = special_doctrine_osirian_league }
					}
					is_target_in_global_variable_list = { name = unavailable_unique_decisions target = flag:ending_the_snake_objective_decision }
				}
				POD_change_character_debate_force_effect = { COUNT = 100 }
			}
		}
	}

	option = {
		name = POD_mummy_spell_of_life.10.a
		trigger_event = {
			id = POD_mummy_spell_of_life.11
			days = { 10 15 }
		}
	}
	option = {
		name = POD_mummy_spell_of_life.10.b
	}
}

					limit = { has_character_flag = spell_of_life_stage1_kemintiri_phase3_decision }

						name = total_opponent_coterie_prowess

		custom_tooltip = POD_mummy_spell_of_life_total_prowess_check
