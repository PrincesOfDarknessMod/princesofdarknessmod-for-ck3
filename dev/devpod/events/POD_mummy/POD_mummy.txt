podmummy.1 = {
	hidden =yes
	trigger = {
		has_trait = mummy
	}
	immediate = {
		if = { 
			limit = { POD_is_in_homeland_trigger = no }
			send_interface_toast = {
				type = event_toast_effect_good
				title = podmummy.1.homeland
				left_icon = root

				add_trait_xp = { trait = mummy value = 1 } 
			}
		}
		if = { limit  = { has_trait = integrity } add_trait_xp = { trait = integrity value = 1 } }
	}
}

podmummy.2 = {
	hidden =yes
	immediate = {
		if = { limit = { NOT = { has_character_modifier = extra_life_1_modifier } } add_character_modifier = extra_life_1_modifier }
	}
}
