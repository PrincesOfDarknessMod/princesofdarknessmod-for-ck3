scripted_effect increase_tomb_quality_effect = {
	
	custom_tooltip = increase_tomb_quality_tt
	
	if = {
		limit = {
			has_character_flag = low_tomb_quality_flag
		}
		remove_character_flag = low_tomb_quality_flag
		add_character_flag = medium_tomb_quality_flag
	}
	else_if = {
		limit = {
			has_character_flag = medium_tomb_quality_flag
		}
		remove_character_flag = medium_tomb_quality_flag
		add_character_flag = high_tomb_quality_flag
	}
	else_if = {
		limit = {
			has_character_flag = high_tomb_quality_flag
		}
		remove_character_flag = high_tomb_quality_flag
		add_character_flag = highest_tomb_quality_flag
	}
	else_if = {
		limit = {
			has_character_flag = highest_tomb_quality_flag
		}
	}
	else = { 
		add_character_flag = low_tomb_quality_flag
	}
}


scripted_effect decrease_tomb_quality_effect = {
	
	custom_tooltip = decrease_tomb_quality_tt
	
	if = {
		limit = {
			has_character_flag = medium_tomb_quality_flag
		}

		add_character_flag = low_tomb_quality_flag
	}
	else_if = {
		limit = {
			has_character_flag = high_tomb_quality_flag
		}
		remove_character_flag = high_tomb_quality_flag
		add_character_flag = medium_tomb_quality_flag
	}
	else_if = {
		limit = {
			has_character_flag = highest_tomb_quality_flag
		}
		remove_character_flag = highest_tomb_quality_flag
		add_character_flag = high_tomb_quality_flag
	}
	else = { 
	}
}
