podmummy.2000 = {
	type = character_event
	content_source = dlc_POD
	title = podmummy.2000.t
	desc = podmummy.2000.desc
	right_portrait = root
	theme = mummy
	
	option = {
		name = podmummy.1000.a
		change_variable = {
			name = amulets_craft_outcome
			add = 0.5
		}
	}
	
	option = {
		name = podmummy.1000.b
		trigger = { 
			prestige >= major_prestige_value
			prestige_level >= medium_prestige_level
		}
		show_as_unavailable = { 
			OR = {
				prestige < major_prestige_value 
				prestige_level < medium_prestige_level
			}
		}
		add_prestige = major_prestige_loss
		change_variable = {
			name = amulets_craft_outcome
			add = 1
		}
	}
	
	option = {
		name = podmummy.1000.c
		trigger = { 
			prestige >= massive_prestige_value
			prestige_level >= high_prestige_level
		}
		show_as_unavailable = { 
			OR = {
				prestige < massive_prestige_value 
				prestige_level < high_prestige_level
			}
		}
		add_prestige = massive_prestige_loss
		change_variable = {
			name = amulets_craft_outcome
			add = 1.5
		}
	}
	
	option = {
		name = podmummy.1000.d
		trigger = { 
			prestige >= monumental_prestige_value
			prestige_level >= very_high_prestige_level
		}
		show_as_unavailable = { 
			OR = {
				prestige < monumental_prestige_value 
				prestige_level < very_high_prestige_level
			}
		}
		add_prestige = monumental_prestige_loss
		change_variable = {
			name = amulets_craft_outcome
			add = 2
		}
	}
	
	option = {
		name = podmummy.1000.e
		trigger = { 
			prestige >= 3000
			prestige_level = max_prestige_level
		}
		show_as_unavailable = { 
			OR = {
				prestige < 3000 
				NOT = { prestige_level = max_prestige_level }
			}
		}
		add_prestige = -3000
		change_variable = {
			name = amulets_craft_outcome
			add = 2.5
		}
	}
}
