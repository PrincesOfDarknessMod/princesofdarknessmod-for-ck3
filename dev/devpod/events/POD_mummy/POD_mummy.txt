podmummy.1 = {
	hidden = yes
	trigger = {
		has_trait = mummy
	}
	immediate = {
		
		POD_gain_integrity = { VALUE = 1 }
		send_interface_toast = {
			type = POD_mummy_effigy
			title = podmummy.1.homelandandtomb
			left_icon = root
			if = { 
				limit = { POD_is_in_homeland_trigger = yes }
				custom_tooltip = podmummy.1.homeland
				add_trait_xp = { trait = mummy value = 1 } 
			}
			else = { custom_tooltip = POD_not_in_homelands_tt }
			if = { 
				limit = { NOT = { POD_mummy_yearly_memory_change = 0 } }
				custom_tooltip = podmummy.1.tomb
				add_trait_xp = { trait = mummy value = POD_mummy_yearly_memory_change } 
			}
			else = { custom_tooltip = POD_no_tomb_memory_tt }
		}
		
	}
}

podmummy.4000 = {
	hidden = yes
	immediate = {
		save_temporary_scope_as = liege
		every_courtier_or_guest = {
			limit = {
				NOT = { has_faith = scope:liege.faith }
			}
			random = {
				chance = 15
				save_temporary_scope_value_as = {
					name = saved_immortal_age
					value = POD_visual_age
				}
				hidden_effect = { POD_remove_splat_traits_effect = yes }
				add_trait_force_tooltip = asekh_sen
				if = {
					limit = { exists = scope:saved_immortal_age }
					hidden_effect = { set_immortal_age = scope:saved_immortal_age }
				}
				hidden_effect = {
					POD_remove_wrong_supernatural_traits_on_existing_character = yes
					POD_remove_supernatural_relation_effect = yes
					POD_remove_physical_coping_strategies = yes

					POD_side_effects_from_changing_splats = yes
				}
				POD_mummy_initialization_asekh_sen_effect = yes
			}
			neteru_faith_initiation_effect = { LIEGE = scope:liege }
		}
		every_vassal_or_below = {
			limit = {
				primary_title.tier >= tier_county
				NOT = { has_faith = scope:liege.faith }
				is_ai = yes
			}
			random = {
				chance = 15
				hidden_effect = { POD_remove_splat_traits_effect = yes }
				add_trait_force_tooltip = asekh_sen
				if = {
					limit = { exists = scope:saved_immortal_age }
					hidden_effect = { set_immortal_age = scope:saved_immortal_age }
				}
				hidden_effect = {
					POD_remove_wrong_supernatural_traits_on_existing_character = yes
					POD_remove_supernatural_relation_effect = yes
					POD_remove_physical_coping_strategies = yes

					POD_side_effects_from_changing_splats = yes
				}
				POD_mummy_initialization_asekh_sen_effect = yes
			}
			neteru_faith_initiation_effect = { LIEGE = scope:liege }
		}
	}
}
