POD_mummy_maintenance.20 = {
	hidden = yes
	type = character_event
	
	trigger = {
		is_ai = no
		has_trait = mummy
	}
	
	immediate = {
		every_relation = {
			type = effigy_character
			limit = {
				is_alive = yes
				NOT = { has_trait = torpor }
			}
			root = { 
				send_interface_message = {
					type = event_generic_bad
					title = POD_effigy_maintenance
					left_icon = root
					right_icon = prev
					if = { 
						limit = { energy_points_value >= 1 }
						POD_hekau_cost_effect = { COUNT = 1 }
					}
					else = { 
						prev = { increase_wounds_effect = { REASON = mysterious } }
					}
					custom_tooltip = POD_effigy_maintenance.tt
				}
			}
		}
		every_character_artifact = {
			limit = { 
				has_variable = active_effigy
				var:active_effigy = yes
			}
			root = { 
				send_interface_message = {
					type = event_generic_bad
					title = POD_effigy_maintenance
					right_icon = root
					if = { 
						limit = { 
							prev = { 
								OR = { 
									has_variable = viziers_beast_effigy
									has_variable = kingly_beast_effigy
									has_variable = divine_beast_effigy
								}
							}
							gold >= tiny_gold_value
						}
						remove_short_term_gold = tiny_gold_value
					}
					else_if = {
						limit = { energy_points_value >= 1 }
						POD_hekau_cost_effect = { COUNT = 1 }
					}
					else = {
						prev = { degrade_mummy_artifact_rarity_effect = yes }
					}
					custom_tooltip = POD_effigy_maintenance.tt
				}
			}
		}
	}
}
