POD_mummy_hekau.100 = {
	type = character_event
	title = POD_mummy_hekau.100.t
	desc = POD_mummy_hekau.100.desc
	theme = stewardship_wealth_focus
	left_portrait = {
		character = root
		animation = happiness
	}
	
	right_portrait = {
		character = scope:cloned_character
		animation = disapproval
	}
	
	artifact = { 
	    target = scope:this_artifact
	    position = lower_left_portrait
	}

	immediate = {
		scope:target = {
			save_scope_as = this_artifact
		}
		scope:cloned_character = { 
			random_equipped_character_artifact = {
				limit = { artifact_slot_type = primary_armament }
				save_scope_as = that_primary_armament_artifact
			}
			random_equipped_character_artifact = {
				limit = { artifact_slot_type = armor }
				save_scope_as = that_armor_artifact
			}
			random_equipped_character_artifact = {
				limit = { artifact_slot_type = regalia }
				save_scope_as = that_regalia_artifact
			}
			random_equipped_character_artifact = {
				limit = { artifact_slot_type = helmet }
				save_scope_as = that_helmet_artifact
			}
		}
	}

	option = {
		name = POD_mummy_hekau.100.a
		
		trigger = {
			exists = scope:that_primary_armament_artifact
			OR = {
				AND = {
					scope:this_artifact = { has_variable = lesser_chattel_effigy }
					scope:that_primary_armament_artifact = { 
						rarity = common
					}
				}
				AND = {
					scope:this_artifact = { has_variable = complex_chattel_effigy }
					scope:that_primary_armament_artifact = { 
						OR = {
							rarity = masterwork
							rarity = common
						}
					}
				}
				AND = {
					scope:this_artifact = { has_variable = superior_chattel_effigy }
					scope:that_primary_armament_artifact = { 
						OR = {
							rarity = famed
							rarity = masterwork
							rarity = common
						}
					}
				}
				scope:this_artifact = { has_variable = princely_chattel_effigy }
			}
		}
		
		scope:this_artifact = {
			set_variable = {
				name = active_effigy
				value = yes
			}
			hidden_effect = { copy_artifact_modifiers = scope:that_primary_armament_artifact }
		}
	}

	option = {
		name = POD_mummy_hekau.100.b
		
		trigger = {
			exists = scope:that_armor_artifact
			OR = {
				AND = {
					scope:this_artifact = { has_variable = lesser_chattel_effigy }
					scope:that_armor_artifact = { 
						rarity = common
					}
				}
				AND = {
					scope:this_artifact = { has_variable = complex_chattel_effigy }
					scope:that_armor_artifact = { 
						OR = {
							rarity = masterwork
							rarity = common
						}
					}
				}
				AND = {
					scope:this_artifact = { has_variable = superior_chattel_effigy }
					scope:that_armor_artifact = { 
						OR = {
							rarity = famed
							rarity = masterwork
							rarity = common
						}
					}
				}
				scope:this_artifact = { has_variable = princely_chattel_effigy }
			}
		}
		
		scope:this_artifact = {
			set_variable = {
				name = active_effigy
				value = yes
			}
			hidden_effect = { copy_artifact_modifiers = scope:that_armor_artifact }
		}
	}

	option = {

		trigger = {

			OR = {
				AND = {
					scope:this_artifact = { has_variable = lesser_chattel_effigy }
					scope:that_regalia_artifact = { 
						rarity = common
					}
				}
				AND = {
					scope:this_artifact = { has_variable = complex_chattel_effigy }
					scope:that_regalia_artifact = { 
						OR = {
							rarity = masterwork
							rarity = common

					scope:this_artifact = { has_variable = superior_chattel_effigy }
					scope:that_regalia_artifact = { 
						OR = {
							rarity = famed
							rarity = masterwork
							rarity = common
						}
					}
				}
				scope:this_artifact = { has_variable = princely_chattel_effigy }

POD_mummy_hekau.201 = {
    type = character_event
    title = POD_mummy_hekau.201.t
	desc = POD_mummy_hekau.201.desc
	theme = mummy
	
	left_portrait = {
	    character = root
		animation = personality_bold 
	}
	
	immediate = {
	
	}
	
	option = {
	    name = POD_mummy_hekau.201.option.a
		custom_tooltip = POD_mummy_hekau.201.option.a.tt
		
		duel = {
		    skill = learning
			value = 20
			desc = POD_mummy_hekau.201.option.a.desc
		
			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = high_positive_duel_skill_multiplier
				}
				
				custom_tooltip = POD_mummy_hekau.201.option.a.tt.success
				
				hidden_effect = {
				    POD_create_separated_khaibit_effect = yes
					every_relation = {
                        type = separated_khaibit_servant
                        save_scope_as = separated_khaibit
                    }
					trigger_event = {
					    id = POD_mummy_hekau.202
						days = 2
					}
			    }
			}    
			
			50 = {

				compare_modifier = {
					value = scope:duel_value
					multiplier = low_negative_duel_skill_multiplier
				}
				
				custom_tooltip = POD_mummy_hekau.201.option.a.tt.failure
				
			    hidden_effect = {
				    trigger_event = {
					    id = POD_mummy_hekau.203
						days = 2
					}
			    }
			}
		}
		add_piety = -800
	}
	
	option = {
	    name = POD_mummy_hekau.201.option.b
		custom_tooltip = POD_mummy_hekau.201.option.b.tt
	}
}
