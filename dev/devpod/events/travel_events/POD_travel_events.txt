namespace = POD_travel_events

POD_travel_events.1 = {
	type = character_event
	content_source = dlc_POD
	title = POD_travel_events.1.t
	desc = POD_travel_events.1.desc

	theme = travel
	left_portrait = {
		character = root
		animation = thinking
	}

	trigger = {
		location = province:4323 #Alamut
		current_travel_plan.travel_plan_owner = root
		is_playable_character = yes
		is_location_valid_for_travel_event_on_land = yes #Should not be needed...
		NOT = { location.county.holder = root } #If I own it then it does not make sense to visit it.
		NOT = {
			any_memory = { #Did I come here before?
				has_memory_type = memory_big_city_visited
				has_variable = city_location
				var:city_location = root.location
			}
		}
		
	}

	immediate = {
		play_music_cue = mx_cue_travel_arrival
		location = {
			save_scope_as = city_province
		}
	}

	option = { 
		name = POD_travel_events.1.a
	
		current_travel_plan = {
			delay_travel_plan = {
				days = 7
			}
		}
		
		add_stress = medium_stress_gain
		
		if = {
			limit = {
				has_trait = lifestyle_blademaster
				is_assamite_trigger = yes
			}
			add_trait_xp = {
				trait = lifestyle_blademaster
				value = 10
			}
		}
		
		else_if = {
			limit = {
				is_assamite_trigger = yes
			}			
			add_trait = lifestyle_blademaster
		}
		
		else = {
			add_prestige = 150
		}
	}

	option = { 
		name = POD_travel_events.1.b
	
	}
	
	after = {
		create_character_memory = {
			type = memory_big_city_visited
		}
		scope:new_memory = {
			save_scope_as = city_memory
		}
		scope:city_memory = {
			set_variable = {
				name = city_location
				value = scope:city_province
			}
		}
	}
}
