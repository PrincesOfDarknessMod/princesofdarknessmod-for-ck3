# COMMON : ANEZKA 
#1: Before freeing her
POD_christof_death_events.1 = {
	type = character_event
	content_source = dlc_POD
	title = POD_christof_death_events.1.t
	desc = POD_christof_death_events.1.desc
	theme = death
    override_background = ce1_catacombs
    override_effect_2d = { reference = smoke }
    widget = { gui = event_window_widget_vfx_fog container = foreground_shader_vfx_container }
    #Christoff
	left_portrait = {
		character = character:790802
        animation = disbelief
        camera = camera_event_left_crowning_observation
	}
    #Anezka 
	right_portrait = {
		character = character:792754
        animation = dead
        camera = camera_event_horse_right
	}

    immediate = {
        play_music_cue = "mx_cue_death"
        character:790802 = {
		    add_character_flag = POD_christof_anezka_is_dead_flag
        }
    }

	option = {
		name = POD_christof_death_events.1.a
		add_character_modifier = {
			modifier = stress_frozen_grief
			years = 20
		}
    }
	option = {
		name = POD_christof_death_events.1.b
        death = {
            death_reason = death_suicide
        }
    }
    option = {
		name = POD_christof_death_events.1.c
		add_character_modifier = {
			modifier = stress_frozen_grief
			years = 15
		}
        add_stress=50
        add_trait = golconda_seeker
    }
}

#ECATERINA
#If part of the Ecaterina Path and you have not freed Anezka 
POD_christof_death_events.5 = {
	type = character_event
	content_source = dlc_POD
	title = POD_christof_death_events.5.t
	desc = POD_christof_death_events.5.desc
	theme = death
    override_background = bp1_relaxing_room_western
    #Christoff
	right_portrait = {
		character = character:790802
        animation = frontend_center_idle
        camera = camera_event_far_right_very_right
	}
	lower_left_portrait = {
		character = character:790602
        animation = dead
	}
    left_portrait = {
        trigger = {
            is_alive = yes
            NOT = { has_trait = torpor }
        }
        animation = disapproval
        camera = camera_event_right_away_3_4
        character = character:790850
    }
    immediate = {
        play_music_cue = "mx_cue_death"
    }

    #Reporting to Cosmos instead
	option = {
		trigger = {
			character:790850 = {
                is_alive = yes
                NOT = { has_trait = torpor }
			}
		}
        custom_tooltip = POD_christof_death_events.5.a.tt
		name = POD_christof_death_events.5.a

		add_contact = character:790850 #Cosmos

        #We set Cosmos as the new patron
        custom_tooltip = set_relation_ashen_ambassadorial_patron

        every_relation = {
			type = ashen_ambassadorial_patron
			remove_relation_ashen_ambassadorial_patron = root
					
		} 

		hidden_effect = {
            character:790850 = {
                set_relation_ashen_ambassadorial_patron = root
            }
        }
    }

    #Go to the independant path
	option = {
        custom_tooltip = POD_christof_death_events.5.b.tt
        remove_character_flag = POD_christof_Ecaterinapath_flag
        add_character_flag = POD_christof_Independant_flag
		name = POD_christof_death_events.5.b
    }
}
