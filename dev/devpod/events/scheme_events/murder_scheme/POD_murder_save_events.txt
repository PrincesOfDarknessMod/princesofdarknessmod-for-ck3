# Events for saving from a murder attempt

namespace = POD_murder_save

#Save against sun : owner

POD_murder_save.0001 = {
	type = character_event
	title = POD_murder_save.0001.t
	desc = POD_murder_save.0001.desc
	
	theme = murder_scheme
	left_portrait = {
		character = scope:target
		animation = shock
	}
	widget = {
		gui = "event_window_widget_scheme"
		container = "custom_widgets_container"
	}
	
	trigger = {
		#scope:murder_method = flag:vampire_sun
		scope:target = {
			OR = {
				has_trait = fortitudeadvanced
				has_trait = celerityadvanced
				has_trait = temporisadvanced
				has_perk = darkstatue_perk
			}
		}
	}

	weight_multiplier = {
		base = 1
		modifier = {
			add = 1
			scope:target = {
				has_trait = fortitudeadvanced
				has_trait = celerityadvanced
				has_trait = temporisadvanced
				has_perk = darkstatue_perk
			}	
		}		
		modifier = {
			add = -1
			scope:target ?= { has_trait = setite }	
		}
	}

	immediate = {
		save_scope_value_as = {
			name = murder_saved
			value = flag:vampire_sun_saved
		}
		murder_failure_effect = yes # Fires the response for the target
	}

	option = {
		name = POD_murder_save.0001.a
			if = {
				limit = { exists = scope:scheme_discovered }
				add_dread = medium_dread_gain
			}
			scope:scheme = {
				end_scheme = yes
			}
		ai_chance = {
			base = 100
		}
	}

	option = {
		name = POD_murder_save.0001.b
			if = {
				limit = { exists = scope:scheme_discovered }
				add_dread = medium_dread_gain
			}
			restart_murder_scheme_effect = yes
		ai_chance = {
			base = 0
		}
	}
	after = {
		scope:target = { trigger_event = murder_save.1001 }
	}
}


#Save against sun : target

POD_murder_save.1001 = {
	type = character_event
	title = POD_murder_save.1001.t
	desc = POD_murder_save.1001.desc
	
	left_portrait = {
		character = root
		animation = anger
	}
	right_portrait = {
		character = scope:owner_revealed
		animation = shock
	}
	theme = intrigue

	trigger = {
		exists = scope:murder_saved
		scope:murder_saved = flag:vampire_sun_saved
	}

	immediate = {
		if = {
			limit = {
				exists = scope:scheme_discovered
			}
			scope:owner = {
				save_scope_as = owner_revealed
			}
		}
	}

	option = { #Who could do such a thing?
		name = POD_murder_save.1001.a
		trigger = {
			NOT = {
				exists = scope:scheme_discovered
			}
		}
		custom_tooltip = murder_save.failure_unknown_owner_tt
	}

	option = { #Vengeance!
		name = POD_murder_save.1001.b
		trigger = {
			exists = scope:scheme_discovered
		}
		custom_tooltip = murder_save.failure_known_owner_tt
	}

	after = {
		add_character_modifier = {
			modifier = watchful_modifier
			days = watchful_modifier_duration
		}
	}
}
