namespace = POD_murder_outcome

#The events for the target are handled in modded vanilla files, these are only for the schemer
##Vampire murder : sun

#Setup
POD_murder_outcome.0001 = {
	hidden = yes
	
	trigger = { 
		scope:scheme.scheme_target_character = { 
			POD_fear_the_sun_trigger = yes
			POD_takes_sun_damage_as_superficial = no
		}
	}

	immediate = {
		save_scope_value_as = {
			name = murder_method
			value = flag:vampire_sun
		}
		#trigger_event = { on_action = murder_setup }
	}
}

#Launch it
POD_murder_outcome.1001 = {
	type = character_event
	content_source = dlc_POD
	title = POD_murder_outcome.1001.t
	desc = POD_murder_outcome.1001.desc
	theme = murder_scheme
	left_portrait = scope:target
	widget = {
		gui = "event_window_widget_scheme"
		container = "custom_widgets_container"
	}

	trigger = {
		#Will have been set by murder_outcome.0001 or a special setup event.
		scope:murder_method = flag:vampire_sun
	}

	immediate = {
		#murder_outcome_roll_setup_effect = yes
	}

	option = {
		name = POD_murder_outcome.1001.a
		#murder_outcome_roll_effect = yes
		stress_impact = {
			deceitful = minor_stress_impact_loss
			paranoid = medium_stress_impact_gain
		}
		custom_tooltip = POD_murder_save_physical		
	}
	option = {
		name = POD_murder_outcome.1001.b
		ai_chance = {
			base = 0
		}
		stress_impact = {
			impatient = medium_stress_impact_gain
		}
		scope:scheme = {
			add_scheme_progress = decline_execution_setback
		}
		custom_tooltip = do_not_execute_murder_tooltip
	}
}

#Success 
POD_murder_outcome.2001 = {
	type = character_event
	content_source = dlc_POD
	title = POD_murder_outcome.2001.t
	### START DESC
	desc = {
		# Open with a statement celebrating success
		# Remember to name the Target character
		random_valid = {
			desc = POD_murder_outcome.2001.opening_1
		}

		first_valid = {
			# If the Scheme Owner is known:
			triggered_desc = {
				trigger = {
					exists = scope:scheme_discovered
				}
				desc = {
					random_valid = {
						desc = POD_murder_outcome.2001.owner_is_known_1
					}
				}
			}
			desc = {
				random_valid = {
					desc = POD_murder_outcome.2001.no_awareness_1
				}
			}
		}
	}
	### END OF DESC
	theme = murder_scheme
	left_portrait = {
		character = scope:target
		animation = fear
	}

	trigger = {
		scope:scheme.scheme_target_character = { 
			POD_fear_the_sun_trigger = yes
			POD_takes_sun_damage_as_superficial = no
		}	
	}

	immediate = {
		scheme_owner_pov_murder_success_effect = { REASON = death_mysterious }
	}

	option = {
		name = {
			trigger = {
				exists = scope:scheme_discovered
			}
			text = POD_murder_outcome.2001.a.discovered
		}
		name = {
			trigger = {
				NOT = { exists = scope:scheme_discovered }
			}
			text = POD_murder_outcome.2001.a
		}
	}
	
	after = {
		scope:target = { trigger_event = POD_murder_outcome.2022 }
	}
}

# Target POV — Success
POD_murder_outcome.2022 = {
	type = character_event

}

#Failure
POD_murder_outcome.4001 = {
	type = character_event
	content_source = dlc_POD
	title = POD_murder_outcome.4001.t
	### START DESC
	desc = {
	
		# Start off with declaration of failure
		# Include Target's name!
		random_valid = {
			desc = POD_murder_outcome.4001.failure_declaration_1
		}
	
		first_valid = {
			# "Regular" scheme execution
			desc = {
				desc = POD_murder_outcome.4001.failed_attempt_1
				first_valid = {
					triggered_desc = {
						trigger = {
							exists = scope:scheme_discovered
						}
						desc = POD_murder_outcome.4001.disastrous_attempt_1
					}
					desc = POD_murder_outcome.4001.unknown_so_far
				}
			}
		}
	}
	### END DESC

	theme = murder_scheme
	left_portrait = scope:target

	trigger = {
		scope:scheme.scheme_target_character = { 
			POD_fear_the_sun_trigger = yes
			POD_takes_sun_damage_as_superficial = no
		}	
	}

	immediate = {
		show_as_tooltip = { murder_failure_effect = yes }
	}

	option = {
		name = POD_murder_outcome.4001.a
		trigger = {
			exists = scope:scheme_discovered
		}		
			scope:scheme = {
				end_scheme = yes
			}
			if = {
				limit = { exists = scope:scheme_discovered }
				add_dread = minor_dread_gain
			}
	}

	option = {
		name = POD_murder_outcome.4001.b
		trigger = {
			NOT = { exists = scope:scheme_discovered }
		}
			if = {
				limit = { exists = scope:scheme_discovered }
				add_dread = minor_dread_gain
			}
			restart_murder_scheme_effect = yes
	}

	after = {
		scope:target = { trigger_event = POD_murder_outcome.4022 }
	}
}
