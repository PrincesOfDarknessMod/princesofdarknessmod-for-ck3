######################################################
# SETITE — SERPENT'S EMBRACE: SUCCESS
# Fires on the scheme owner when the Serpent's Embrace
# scheme converts its target to the owner's faith.
# Scopes: root = owner (Setite), scope:target = converted
######################################################
namespace = pod_setite_serpent_embrace

pod_setite_serpent_embrace.0001 = {
	type = character_event
	title = pod_setite_serpent_embrace.0001.t
	desc = pod_setite_serpent_embrace.0001.desc
	theme = vampire

	left_portrait = {
		character = root
		animation = personality_deceitful
	}
	right_portrait = {
		character = scope:target
		animation = ecstasy
	}

	option = {
		name = pod_setite_serpent_embrace.0001.a
		add_opinion = {
			target = scope:target
			modifier = grateful_opinion
			opinion = 20
		}
	}
}

pod_setite_serpent_embrace.0002 = {
	type = character_event
	title = pod_setite_serpent_embrace.0002.t
	desc = pod_setite_serpent_embrace.0002.desc
	theme = vampire

	left_portrait = {
		character = root
		animation = worry
	}
	right_portrait = {
		character = scope:scheme.scheme_owner
		animation = personality_deceitful
	}

	# Accept the Path of Set
	option = {
		name = pod_setite_serpent_embrace.0002.a
		set_character_faith = scope:scheme.scheme_owner.faith
		scope:scheme.scheme_owner = {
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 20
			}
		}
	}
	
	# Resist — pay piety to throw off the influence
	option = {
		name = pod_setite_serpent_embrace.0002.b
		add_piety = -300
		scope:scheme.scheme_owner = {
			add_opinion = {
				target = root
				modifier = hate_opinion
				opinion = -20
			}
		}
		add_character_flag = {
			flag = block_pod_setite_serpent_embrace
			years = 10
		}
	}
}
