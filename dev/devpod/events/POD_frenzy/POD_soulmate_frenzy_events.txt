POD_soulmate_frenzy.001 = {
    type = character_event
	content_source = dlc_POD
	title = POD_soulmate_frenzy.001.t
	desc = POD_soulmate_frenzy.001.desc
	theme = frenzy 
	
	left_portrait = {
	    character = scope:frenziedvamp
		animation = grief 
	}
	
	widget = {
		gui = "POD_shader_event_fog_alley_night"
		container = "background_shader_vfx_container"
	}
	
	immediate = {
	    save_scope_as = frenziedvamp
		scope:dead_character = {
		    save_scope_as = lost_soulmate
		}
	}
	
	trigger = {
	    is_vampire_trigger = yes
		cant_frenzy_trigger = no
		OR = {
			NOT = { exists = scope:dead_character.killer }
			NOT = { scope:dead_character.killer = root } # Error: set_relation_rival effect [ Trying to set a relation with ourselves ]
		}
	}
	
	option = {
	    name = POD_soulmate_frenzy.001.option.a
		custom_tooltip = POD_soulmate_frenzy.001.option.a.tt
		
		hidden_effect ={
            random_list = {
			    3 = {
				    trigger_event = {
                        id = POD_soulmate_frenzy.002
						days = 2
					}					
				}
				7 = {
				    trigger_event = {
                        id = POD_soulmate_frenzy.004
						days = 2
					} 
					show_as_tooltip = {
						if = {
							limit = { faith = { has_doctrine_parameter = frenzy_lose_piety } }
							add_piety = major_piety_loss
						}
					}
					
					modifier = {
					   add = 6
					   has_trait = brujah
					}
					modifier = {
					    add = 2
						piety_level < 0
					}
					modifier = {
					    add = 2
						blood_level = 0
					}
					modifier = {
					    add = 1
						blood_level = 1
					}
					modifier = {
			            factor = 0.5
				        has_character_modifier = heart_of_darkness_modifier
			        }
				}
			}
		}
	}
	option = {
	    name = POD_soulmate_frenzy.001.option.b
		custom_tooltip = POD_soulmate_frenzy.001.option.b.tt
		
		hidden_effect = {
		    random_list = {
			    4 = {
				    trigger_event = {
                        id = POD_soulmate_frenzy.003
						days = 2
					}
					show_as_tooltip = {
						if = {
							limit = { faith = { has_doctrine_parameter = frenzy_lose_piety } }
							add_piety = medium_piety_loss
						}
					}
  					
					modifier = {
					    add = 2
						has_trait = celerity
					}
					modifier = {
					    add = 4
						has_trait = celerityadvanced
					}
					modifier = {
					    add = 2
						has_trait = temporis
					}
					modifier = {
					    add = 4
						has_trait = temporisadvanced
					}
				}
				6 = {
				    trigger_event = {
                        id = POD_soulmate_frenzy.004
						days = 2
					}
					show_as_tooltip = {
						if = {
							limit = { faith = { has_doctrine_parameter = frenzy_lose_piety } }
							add_piety = major_piety_loss
						}
					}

					modifier = {
					   add = 6

					    add = 2

POD_soulmate_frenzy.303 = {
    type = character_event
	content_source = dlc_POD
	title = POD_soulmate_frenzy.303.t
	desc = POD_soulmate_frenzy.303.desc
	theme = frenzy #gangrel
	
	left_portrait = {
	    character = scope:frenziedvamp 
		animation = rage
	}
	
	right_portrait = {
		character = scope:teaser_werewolf
		animation = idle
	}
	
	widget = {
		gui = "POD_shader_event_fog_alley_night"
		container = "background_shader_vfx_container"
	}
	
	trigger = {
		is_ruler = yes
	}
	
	immediate = {
		# add_character_flag = {
		# 	flag = had_event_POD_werewolf_teaser_100_flag
		# 	years = { 500 1000 }
		# }

		hidden_effect = {
			create_character = {
				location = root.capital_province
				template = POD_werewolf_teaser_character
				save_scope_as = teaser_werewolf
			}		
		}
		
		scope:teaser_werewolf = {
			hidden_effect = { 
				add_character_modifier = POD_werewolf_form_crinos_modifier
				set_variable = {
					name = signature_weapon
					value = flag:fangsclaws
				}
				add_character_flag = has_scripted_appearance
				#add_character_flag = POD_resonance_acute_flag 
				#POD_set_resonance_effect = yes 
			}
		}

		scope:teaser_werewolf = {
			save_scope_as = defender
		}
		
		root = {
			save_scope_as = attacker
		}	
	
	    if = {
		    limit = {
			    any_vassal_or_below = {
				    OR = {
				        has_relation_friend = scope:frenziedvamp
						has_relation_lover = scope:frenziedvamp
				    }
			    }
		    }
		    random_vassal_or_below = {
				limit = {
				    OR = {
					    has_relation_friend = scope:frenziedvamp
					    has_relation_lover = scope:frenziedvamp
					}
				}
				save_scope_as = friend_savior
		    }
	    }
		
		save_scope_as =	soulmate_frenzy_garou_fight	
	}
	option = {
	    name = POD_soulmate_frenzy.303.option.a
		custom_tooltip = POD_soulmate_frenzy.303.option.a.tt
		
		
		duel = {
		    skill = prowess
		    value = 45
		    desc = POD_soulmate_frenzy.303.option.a.desc
			
		    50 = {
		        desc = POD_soulmate_frenzy.303.option.a.success
				
			    compare_modifier = {
				    value = scope:duel_value
				    multiplier = 1.5
			    }
					
				custom_tooltip = POD_soulmate_frenzy.303.option.a.tt.win
				
			    hidden_effect = {	
					trigger_event = {
					    id = POD_soulmate_frenzy.403
					    days = 2
				    }
			    }
		    }	
		  
		    50 = {
		        desc = POD_soulmate_frenzy.303.option.a.failure

			    compare_modifier = {
				    value = scope:duel_value
				    multiplier = -1.5
			    }
					
				custom_tooltip = POD_soulmate_frenzy.303.option.a.tt.lose

			    hidden_effect = {
				    if = {
					    limit = {
						    exists = scope:friend_savior
						}
				        trigger_event = {
						    id = POD_soulmate_frenzy.603
						    days = 2
					    }
			        }
					else = {
					    trigger_event = {
						    id = POD_soulmate_frenzy.703
						    days = 2
					    }  
					}
				}
		    }
	    }
		
		hidden_effect = {
			scope:defender = {
				trigger_event = {
					id = POD_werewolf_teaser.500
					days = 5
				}
			}
		}		
		
	}
}
