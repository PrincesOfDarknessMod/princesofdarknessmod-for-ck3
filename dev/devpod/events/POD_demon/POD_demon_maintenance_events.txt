namespace = POD_demon_maintenance

POD_demon_maintenance.1 = {
	hidden = yes
	type = character_event
	
	trigger = {
		has_trait = demon
	}
	
	immediate = {
		if = {
			limit = { 
				is_ai = no
				has_trait = truefallen 
			}
			if = {
				limit = { NOT = { has_variable = host } }
				trigger_event = POD_demon_maintenance.20
			}
			else = {
				if = {
					limit = { var:host = { is_alive = no } }
					send_interface_message = {
						type = event_generic_bad
						title = POD_demon_maintenance
						left_icon = root
						right_icon = var:host
						var:host = {
							if = { 
								limit = { has_trait = wounded_2 }
								root = { 
									show_as_tooltip = { POD_remove_host_leech_modifier_effect = yes }
									remove_list_variable = { name = corpse_list target = this }
									POD_new_demon_vessel_effect = yes 
								}
							}
							change_trait_rank = {
								trait = wounded
								rank = 1
								max = 3
							}
						}
					}
				}
				else = {
					send_interface_message = {
						type = event_generic_bad
						title = POD_demon_maintenance
						left_icon = root
						right_icon = var:host
						var:host = {
							POD_health_level_decrease_effect = yes
						}
						custom_tooltip = POD_demon_maintenance.tt
					}
				}
			}
		}
		
		if = {
			limit = {
				NOT = { has_trait = truefallen }
				NOT = { has_trait = earthbound }
				NOT = {
					is_target_in_global_variable_list = {
						name = unavailable_unique_decisions
						target = flag:breakdown_the_abyss_objective_decision
					}
				}
			}
			remove_variable = spirit_destination
			remove_character_modifier = traveling_spirit_world_modifier
			set_variable = { name = spirit_location value = flag:abyss }
			add_character_modifier = trapped_in_abyss_modifier
			trigger_event = POD_umbra_lost.104
		}
		
		send_interface_message = {
			type = event_generic_good
			title = POD_arcana_maintenance
			custom_tooltip = POD_arcana_maintenance.tt
			add_arcana_lifestyle_xp = arcana_xp_gain_value
		}
	}
}

POD_demon_maintenance.10 = {
	title = POD_demon_maintenance.10.t
	desc = POD_demon_maintenance.10.desc
	theme = demon

	left_portrait = {
		character = root
		animation = idle
	}
	
	trigger = {
		has_trait = demon
		NOR = {
			has_trait = truefallen
			has_variable = stone_prison
			has_variable = earth_bound
		}
	}

	option = {
		name = POD_demon_maintenance.10.option.a
		trigger = { NOT = { has_trait = earthbound } }
		
		add_trait = truefallen
		
		if = {
			limit = { num_of_relation_demonic_thrall = 0 }
			random_courtier_or_guest = {
				limit = {
					religion = religion:demons_religion
					can_be_thrall = yes
				}
				alternative_limit = { can_be_thrall = yes }
				set_relation_demonic_master = root
			}
		}
	}
	
	option = {
		name = POD_demon_maintenance.10.option.b
		
		custom_tooltip = earthbound_artifact_choice_tt
		
		add_trait = earthbound
		trigger_event = POD_demon_maintenance.11
	}
	
	option = {
		name = POD_demon_maintenance.10.option.c
		
		add_trait = earthbound
		custom_tooltip = POD_demon_maintenance.10.option.c.tt
		set_variable = {
			name = earth_bound
			value = capital_county
		}
		capital_county = { 
			add_county_modifier = earthbound_modifier 
			set_variable = { name = reliquarie value = root } 
		}
	}
}

POD_demon_maintenance.20 = {
	title = POD_demon_maintenance.20.t
	desc = POD_demon_maintenance.20.desc
	theme = demon

	left_portrait = {
		character = root
		animation = idle
	}
	
	right_portrait = scope:thrall
	
	immediate = {
		random_relation = {
			type = demonic_thrall
			limit = { can_be_demonic_possesed = yes }
			save_scope_as = thrall
		}
	}
	
	trigger = {
		has_trait = demon
		has_trait = truefallen
	}

	# Auto-choose host
	option = {
		name = POD_demon_maintenance.20.option.a
		
		trigger = { exists = scope:thrall }
		
		POD_first_demon_vessel_effect = { HOST = scope:thrall }
		
		ai_chance = {
			base = 100
		}
	}
	
	option = {
		name = POD_demon_maintenance.20.option.b
		
		add_internal_flag = dangerous
		
		custom_tooltip = demon_without_host_tt
		
		POD_remove_host_leech_modifier_effect = yes
		remove_trait = truefallen
		remove_variable = host
		if = {
			limit = { is_ai = yes }
			depose = yes
			trigger_event = POD_demon_maintenance.1
		}
	}
}

		}
	}
