# a demon broke free from binding
# needs scope:demon (the demon) saved on the TopScope
POD_demon_maintenance.3 = {
	type = character_event
	content_source = dlc_POD
	title = POD_demon_maintenance.3.t
	desc = POD_demon_maintenance.3.desc
	theme = demon

	override_background = { reference = shader_event_ritual }
	
	widget = {
		gui = "POD_shader_event_ritual"
		container = "background_shader_vfx_container"
	}

	left_portrait = {
		character = scope:demon
	}
	
	immediate = {
		scope:demon = {
			set_relation_rival = {
				target = prev
				reason = rival_diabolist
			}
			add_opinion = {
				target = prev
				modifier = POD_demon_binding_opinion
				opinion = -100
			}
		}
		if = {
			limit = { has_relation_bound_demon = scope:demon }
			remove_relation_bound_demon = scope:demon
		}
		POD_refund_investiture = yes
		add_investiture_lifestyle_perk_points = {
			value = scope:demon.var:diabolist_perks_gained
			multiply = -1
		}
	}

	option = {
		trigger = { scope:demon = { POD_is_available = yes } }
		name = POD_demon_maintenance.3.a
		custom_tooltip = POD_demon_maintenance.3.a.tt
	}

	option = {
		trigger = { scope:demon = { POD_is_available = no } }
		name = POD_demon_maintenance.3.b
	}
}

# Insolent Thrall Player
POD_demon_maintenance.70 = {
	type = character_event
	content_source = dlc_POD
	title = POD_demon_maintenance.70.t
	desc = POD_demon_maintenance.70.desc
	theme = demon

	left_portrait = {
		character = root
		animation = fear
	}
	
	right_portrait = {
		character = scope:demon
		animation = anger
	}
	
	immediate = { 
		scope:demon = { 
			if = {
				limit = { is_imprisoned = yes }
				release_from_prison = yes
			}
		} 
	}

	option = {
		name = POD_demon_maintenance.70.option.a
		
		duel = {
			skill = learning
			target = scope:demon
			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = low_positive_duel_skill_multiplier
				}
				add_prestige = minor_prestige_loss
			}
			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = high_negative_duel_skill_multiplier 
				}
				death = { 
					death_reason = death_souleaten 
					killer = scope:demon
				}
			}
		}
	}
}
