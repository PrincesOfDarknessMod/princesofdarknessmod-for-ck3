namespace = POD_demon_maintenance

POD_demon_maintenance.1 = {
	hidden = yes
	type = character_event
	
	trigger = {
		has_trait = demon
	}
	
	immediate = {
		if = {
			limit = { 
				is_ai = no
				has_trait = truefallen 
			}
			ordered_relation = {
				type = possessed_target
				order_by = age
				save_scope_as = vessel
			}
			if = {
				limit = { 
					NOT = { exists = scope:vessel }
					has_perk = cerement_of_walking_dust_perk
				}
				ordered_in_list = {
					variable = corpse_list
					order_by = age
					save_scope_as = corpse
				}
			}
			if = {
				limit = { 
					exists = scope:corpse
				}
				send_interface_message = {
					type = event_generic_bad
					title = POD_demon_maintenance
					left_icon = root
					right_icon = scope:corpse
					scope:corpse = {
						if = { 
							limit = { 
								has_trait = wounded_2
							}
							remove_trait = wounded_2
							add_trait_force_tooltip = wounded_3
							remove_list_variable = { name = corpse_list target = this }
						}
						else_if = { 
							limit = { 
								has_trait = wounded_1
							}
							remove_trait = wounded_1
							add_trait_force_tooltip = wounded_2
						}
						else = {
							add_trait = wounded_1
						}
					}
				}
			}
			else_if = {
				limit = { 
					exists = scope:vessel
				}
				send_interface_message = {
					type = event_generic_bad
					title = POD_demon_maintenance
					left_icon = root
					right_icon = scope:vessel
					scope:vessel = {
						POD_health_level_decrease_effect = yes
					}
					custom_tooltip = POD_demon_maintenance.tt
				}
			}
			else = {
				trigger_event = POD_demon_maintenance.20
			}
		}
		
		if = {
			limit = {
				NOT = { has_trait = truefallen }
				NOT = { has_trait = earthbound }
				NOT = {
					is_target_in_global_variable_list = {
						name = unavailable_unique_decisions
						target = flag:breakdown_the_abyss_objective_decision
					}
				}
			}
			remove_variable = spirit_destination
			remove_character_modifier = traveling_spirit_world_modifier
			set_variable = { name = spirit_location value = flag:abyss }
			add_character_modifier = trapped_in_abyss_modifier
			trigger_event = POD_umbra_lost.104
		}
		
		send_interface_message = {
			type = event_generic_good
			title = POD_arcana_maintenance
			custom_tooltip = POD_arcana_maintenance.tt
			add_arcana_lifestyle_xp = arcana_xp_gain_value
		}
	}
}

POD_demon_maintenance.10 = {
	title = POD_demon_maintenance.10.t
	desc = POD_demon_maintenance.10.desc
	theme = demon

	left_portrait = {
		character = root
		animation = idle
	}
	
	trigger = {
		has_trait = demon
		NOR = {
			has_trait = truefallen
			has_variable = stone_prison
			has_variable = earth_bound
		}
	}

	option = {
		name = POD_demon_maintenance.10.option.a
		trigger = { NOT = { has_trait = earthbound } }
		
		add_trait = truefallen
	}
	
	option = {
		name = POD_demon_maintenance.10.option.b
		
		add_trait = earthbound

	option = {
		name = POD_demon_maintenance.10.option.c
		
		add_trait = earthbound
		custom_tooltip = POD_demon_maintenance.10.option.c.tt
		set_variable = {
			name = earth_bound
			value = capital_county
		}
	}
}

POD_demon_maintenance.20 = {
	title = POD_demon_maintenance.20.t
	desc = POD_demon_maintenance.20.desc
	theme = demon

	left_portrait = {
		character = root
		animation = idle
	}
	
	immediate = {
		ordered_relation = {
			type = possessed_target
			order_by = age
			save_scope_as = vessel
		}
		if = {
			limit = { 
				NOT = { exists = scope:vessel }
				has_perk = cerement_of_walking_dust_perk
			}
			ordered_in_list = {
				variable = corpse_list
				order_by = age
				save_scope_as = corpse
			}
		}
		if = {
			limit = { 
				NOT = { exists = scope:vessel }
				NOT = { exists = scope:corpse }
			}
			random_relation = {
				type = demonic_thrall
				limit = { can_be_demonic_possesed = yes }
				save_scope_as = thrall
			}
		}
	}
	
	trigger = {
		has_trait = demon
		has_trait = truefallen
	}

	option = {
		name = POD_demon_maintenance.20.option.a
	
		trigger = {
			OR = {
				exists = scope:vessel
				exists = scope:corpse
			}
		}
		
		if = {
			limit = { 
				exists = scope:corpse
			}
			send_interface_message = {
				type = event_generic_bad
				title = POD_demon_maintenance
				left_icon = root
				right_icon = scope:corpse
				scope:corpse = {
					add_trait_force_tooltip = wounded_3
					remove_list_variable = { name = corpse_list target = this }
				}
			}
		}
		else = {
			send_interface_message = {
				type = event_generic_bad
				title = POD_demon_maintenance
				left_icon = root
				right_icon = scope:vessel
				scope:vessel = { death = { death_reason = death_suicide } }
				custom_tooltip = POD_demon_maintenance.tt
			}
		}
		
		ai_chance = {
			base = 100
		}
	}

	option = {
		name = POD_demon_maintenance.20.option.a
	
		trigger = {
			exists = scope:thrall
		}
		
		POD_resolve_decrease_multiple_effect = { COUNT = 5 }
		set_relation_possessed_target = scope:thrall
		
		scope:thrall = {
			add_trait = possessed_genetic
			copy_inheritable_appearance_from = root
		}
		
		ai_chance = {
			base = 100
		}
	}
	
	option = {
		name = POD_demon_maintenance.20.option.b
		
		remove_trait = truefallen
		if = {
			limit = { is_ai = yes }
			depose = yes
			set_variable = { name = spirit_location value = flag:abyss }
			add_character_modifier = trapped_in_abyss_modifier
		}
	}
}
