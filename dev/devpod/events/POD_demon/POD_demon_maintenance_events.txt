namespace = POD_demon_maintenance

POD_demon_maintenance.1 = {
	hidden = yes
	type = character_event
	
	trigger = {
		has_trait = demon
	}
	
	immediate = {
		if = {
			limit = { 
				is_ai = no
				has_trait = truefallen 
			}
			if = {
				limit = { NOT = { has_variable = host } }
				trigger_event = POD_demon_maintenance.20
			}
			else = {
				if = {
					limit = { var:host = { is_alive = no } }
					send_interface_message = {
						type = event_generic_bad
						title = POD_demon_maintenance
						left_icon = root
						right_icon = var:host
						var:host = {
							if = { 
								limit = { has_trait = wounded_2 }
								root = { 
									show_as_tooltip = { POD_remove_host_leech_modifier_effect = yes }
									remove_list_variable = { name = corpse_list target = var:host }
									POD_new_demon_vessel_effect = yes 
								}
							}
							change_trait_rank = {
								trait = wounded
								rank = 1
								max = 3
							}
						}
					}
				}
				else = {
					send_interface_message = {
						type = event_generic_bad
						title = POD_demon_maintenance
						left_icon = root
						right_icon = var:host
						var:host = {
							add_prowess_skill = host_maintenance_loss
							if = {
								limit = { prowess <= 0 }
								no_health_left_effect = yes
							}
						}
						custom_tooltip = POD_demon_maintenance.tt
					}
				}
			}
		}
		
		if = {
			limit = {
				NOT = { has_trait = truefallen }
				NOT = { has_trait = earthbound }
				NOT = {
					is_target_in_global_variable_list = {
						name = unavailable_unique_decisions
						target = flag:breakdown_the_abyss_objective_decision
					}
				}
			}
			POD_send_to_abyss_effect = yes
		}
		
		send_interface_message = {
			type = event_generic_good
			title = POD_arcana_maintenance
			custom_tooltip = POD_arcana_maintenance.tt
			add_arcana_lifestyle_xp = arcana_xp_gain_value
		}
	}
}

POD_demon_maintenance.20 = {
	title = POD_demon_maintenance.20.t
	desc = POD_demon_maintenance.20.desc
	theme = demon

	left_portrait = {
		character = root
		animation = idle
	}
	
	right_portrait = scope:thrall
	
	immediate = {
		random_relation = {
			type = demonic_thrall
			limit = { 
				can_be_demonic_possesed = yes 
				is_ai = yes
			}
			save_scope_as = thrall
		}
	}
	
	trigger = {
		has_trait = demon
		has_trait = truefallen
	}

	# Auto-choose host
	option = {
		name = POD_demon_maintenance.20.option.a
		
		trigger = { exists = scope:thrall }
		
		POD_first_demon_vessel_effect = { HOST = scope:thrall }
		
		ai_chance = {
			base = 100
		}
	}
	
	option = {
		name = POD_demon_maintenance.20.option.b
		
		add_internal_flag = dangerous
		
		custom_tooltip = demon_without_host_tt
		
		POD_remove_host_leech_modifier_effect = yes
		remove_trait = truefallen
		remove_variable = host
		if = {
			limit = { is_ai = yes }
			depose = yes
			trigger_event = POD_demon_maintenance.1
		}
	}
}

POD_demon_maintenance.51 = {
	hidden = yes
	type = character_event
	
	trigger = { 
		any_neighboring_and_across_water_top_liege_realm_owner = {
			highest_held_title_tier > 3 #Kings and higher.
			NOT = { is_at_war_with = root }
			NOR = { 
				faith = { has_doctrine = doctrine_infernalism_accepted } 
				religion = { is_in_family = rf_demons }
				POD_is_infernalist_trigger = yes
			}
		}
	}
	
	immediate = {
		every_neighboring_and_across_water_top_liege_realm_owner = {
			limit = { 
				highest_held_title_tier > 3 #Kings and higher.
				NOR = { 
					faith = { has_doctrine = doctrine_infernalism_accepted } 
					religion = { is_in_family = rf_demons }
					POD_is_infernalist_trigger = yes
				}
				NOT = { is_at_war_with = root }
			}
			root = {
				start_war = {
					cb = POD_event_war
					target = prev
				}
			}
		}
	}
}

POD_demon_maintenance.52 = {
	hidden = yes
	type = character_event
	
	trigger = { is_at_war = no }
	
	immediate = { 
		random_independent_ruler = {
			limit = {
				highest_held_title_tier > 3 #Kings and higher.
				is_landed = yes
				NOR = { 
					faith = { has_doctrine = doctrine_infernalism_accepted } 
					religion = { is_in_family = rf_demons }
					POD_is_infernalist_trigger = yes
				}
			}
			save_scope_as = new_target
		}
		if = {
			limit = { NOT = { exists = scope:new_target } }
			trigger_event = pod1230objectives.120
		}
		else = {
			start_war = {
				cb = POD_event_war
				target = scope:new_target
			}
		}
	}
