namespace = POD_demon_maintenance

POD_demon_maintenance.1 = {
	hidden = yes
	
	trigger = {
		has_trait = demon
	}
	
	immediate = {
		if = {
			limit = { 
				has_variable = host
			}
			if = {
				limit = { var:host = { has_trait = automaton } }
			}
			else_if = {
				limit = { var:host = { is_alive = yes } }
				random = {
					chance = root.host_kill_chance
					send_interface_message = {
						type = event_generic_bad
						title = POD_demon_maintenance
						left_icon = root
						right_icon = var:host
						var:host = {
							pod_death_no_killer = {
								DEATH_REASON = death_lost_all_life_force
							}
						}
						custom_tooltip = POD_demon_maintenance.tt
					}
				}
			}
			else_if = {
				limit = { has_perk = cerement_of_walking_dust_perk }
				send_interface_message = {
					type = event_generic_bad
					title = POD_demon_maintenance
					left_icon = root
					right_icon = var:host
					var:host = {
						change_trait_rank = {
							trait = wounded
							rank = 1
							max = 3
						}
						if = { 
							limit = { has_trait = wounded_3 }
							root = { 
								show_as_tooltip = { POD_remove_host_leech_modifier_effect = yes }
								remove_list_variable = { name = corpse_list target = var:host }
								POD_new_demon_vessel_effect = yes 
							}
						}
					}
					custom_tooltip = cerement_of_walking_dust.tt
				}
			}
			else = {
				POD_new_demon_vessel_effect = yes
			}
			if = {
				limit = { NOT = { has_trait = earthbound } }
				send_interface_message = {
					type = event_generic_good
					title = POD_torment_maintenance
					custom_tooltip = POD_torment_maintenance.tt
					add_trait_xp = { trait = demon value = -1 }
				}
			}
		}
		
		else_if = {
			limit = { 
				has_trait = fallen
			}
			send_interface_message = {
				type = event_generic_good
				title = POD_torment_maintenance
				custom_tooltip = POD_torment_maintenance.tt
				add_trait_xp = { trait = demon value = -1 }
			}
		}
		
		else = { 
			trigger_event = POD_demon_maintenance.20
		 }
		
		send_interface_message = {
			type = event_generic_good
			title = POD_arcana_maintenance
			custom_tooltip = POD_arcana_maintenance.tt
			add_arcana_lifestyle_xp = arcana_xp_gain_value
		}
	}
}

POD_demon_maintenance.20 = {
	type = character_event
	content_source = dlc_POD
	title = POD_demon_maintenance.20.t
	desc = POD_demon_maintenance.20.desc
	theme = demon

	left_portrait = root
	right_portrait = scope:potential_host
	
	artifact = { 
	    target = scope:this_artifact
	    position = lower_left_portrait
	}
	
	immediate = {
		POD_save_potential_host_effect = yes
		if = {
			limit = {
				is_ai = yes
				NOT = { exists = scope:potential_host }
			}
			create_character = {
				save_scope_as = potential_host
				faith = this.faith
				culture = this.culture
				gender_female_chance = 50
				dynasty = none
				location = root.location
			}
		}
		random_character_artifact = {
			limit = {
				NOT = { has_variable = possessed_artifact }
				NOT = { has_variable = summoned_artifact }
				NOT = { has_variable = temporary_vessel }
				NOT = { has_variable = wraith_jade }
				NOT = { has_variable = soultaker } # Probably not a good idea
				NOT = { has_variable = soulforge } # Probably not a good idea
				NOT = { artifact_type = animal }
			}
			save_scope_as = this_artifact
		}
	}
	
	trigger = {
		has_trait = demon
		NOT = { has_trait = earthbound }
		NOT = { has_global_variable = abyss_broken }
	}

	# Auto-choose host
	option = {
		name = POD_demon_maintenance.20.option.a
		
		trigger = { exists = scope:potential_host }
		
		POD_first_demon_vessel_effect = { HOST = scope:potential_host }
		
		ai_chance = {
			base = 0
			modifier = {
				add = 100
				NOT = { exists = scope:reliquarie_destroyed }
			}
			modifier = {
				add = 100
				NOT = { exists = scope:this_artifact }
				is_landed = no
			}
		}
	}

	# Auto-choose artifact
	option = {
		name = POD_demon_maintenance.20.option.b
		
		trigger = { exists = scope:this_artifact }
		
		scope:this_artifact = { 
			set_variable = { name = reliquarie value = root } 
			set_variable = possessed_artifact 
			ROOT ?= { POD_crafting_give_modifier_based_on_character = { ARTIFACT = PREV } }
			set_artifact_name = artifact_new_reliquarie_name
			set_artifact_description = artifact_new_reliquarie_desc
		}
		
		ai_chance = {
			base = 0
			modifier = {
				add = 100
				exists = scope:reliquarie_destroyed
			}
		}
	}
	
	option = {
		name = POD_demon_maintenance.20.option.c
		
		trigger = { is_landed = yes }
		
		capital_county = {
			set_variable = { name = reliquarie value = root } 
			add_county_modifier = earthbound_modifier
		}
		
		ai_chance = {
			base = 0
			modifier = {
				add = 100
				exists = scope:reliquarie_destroyed
			}
		}
	}
	
	option = {
		name = POD_demon_maintenance.20.option.d
		
		custom_tooltip = demon_without_host_tt
		
		add_internal_flag = dangerous
		
		trigger_event = {
			id = POD_demon_maintenance.21
			months = 1
		}
		
		ai_chance = {
			base = 0
		}
	}
}

				faith = { has_doctrine = doctrine_demonic_accepted } 
				religion = { is_in_family = rf_demons }
			}
		}
	}
	
	immediate = {
		every_neighboring_and_across_water_top_liege_realm_owner = {

					faith = { has_doctrine = doctrine_demonic_accepted }

POD_demon_maintenance.52 = {
	hidden = yes
	
	trigger = { is_at_war = no }
	
	immediate = { 
		random_independent_ruler = {
			limit = {
				highest_held_title_tier > 3 #Kings and higher.
				is_landed = yes
				NOR = { 
					faith = { has_doctrine = doctrine_demonic_accepted } 
					religion = { is_in_family = rf_demons }
					POD_is_demonic_trigger = yes
				}
			}
			save_scope_as = new_target
		}
		if = {
			limit = { NOT = { exists = scope:new_target } }
			trigger_event = pod1230objectives_demons.3
		}
		else = {
			start_war = {
				cb = POD_event_war
				target = scope:new_target
			}
		}
	}
}
