namespace = POD_demon_maintenance

POD_demon_maintenance.1 = {
	hidden = yes
	type = character_event
	
	trigger = {
		has_trait = demon
	}
	
	immediate = {
		if = {
			limit = { 
				is_ai = no
				has_trait = truefallen 
			}
			if = {
				limit = { NOT = { has_variable = host } }
				trigger_event = POD_demon_maintenance.20
			}
			else = {
				if = {
					limit = { var:host = { is_alive = no } }
					send_interface_message = {
						type = event_generic_bad
						title = POD_demon_maintenance
						left_icon = root
						right_icon = var:host
						var:host = {
							if = { 
								limit = { has_trait = wounded_2 }
								root = { 
									show_as_tooltip = { POD_remove_host_leech_modifier_effect = yes }
									remove_list_variable = { name = corpse_list target = var:host }
									POD_new_demon_vessel_effect = yes 
								}
							}
							change_trait_rank = {
								trait = wounded
								rank = 1
								max = 3
							}
						}
					}
				}
				else = {
					send_interface_message = {
						type = event_generic_bad
						title = POD_demon_maintenance
						left_icon = root
						right_icon = var:host
						var:host = {
							add_prowess_skill = host_maintenance_loss
							if = {
								limit = { prowess <= 0 }
								pod_death_no_killer = {
									DEATH_REASON = death_lost_all_life_force
								}
							}
						}
						custom_tooltip = POD_demon_maintenance.tt
					}
				}
			}
		}
		
		if = {
			limit = {
				NOT = { has_trait = truefallen }
				NOT = { has_trait = earthbound }
				NOT = { has_global_variable = abyss_broken }
			}
			POD_send_to_abyss_effect = yes
		}
		
		send_interface_message = {
			type = event_generic_good
			title = POD_arcana_maintenance
			custom_tooltip = POD_arcana_maintenance.tt
			add_arcana_lifestyle_xp = arcana_xp_gain_value
		}
	}
}

POD_demon_maintenance.20 = {
	title = POD_demon_maintenance.20.t
	desc = POD_demon_maintenance.20.desc
	theme = demon

	left_portrait = root
	right_portrait = scope:thrall
	
	artifact = { 
	    target = scope:target
	    position = lower_left_portrait
	}
	
	immediate = {
		random_relation = {
			type = demonic_thrall
			limit = { 
				can_be_demonic_possesed = yes 
				is_ai = yes
			}
			save_scope_as = thrall
		}
		random_character_artifact = {
			limit = { 
				NOT = { has_variable = reliquarie } 
				NOT = { has_variable = artifact_pieces } 
				NOT = { has_variable = possessed_artifact }
				NOT = { has_variable = summoned_artifact }
				NOT = { has_variable = wraith_jade }
				NOT = { artifact_type = animal }
			}
			save_scope_as = this_artifact
		}
	}
	
	trigger = {
		has_trait = demon
	}

	# Auto-choose host
	option = {
		name = POD_demon_maintenance.20.option.a
		
		trigger = { exists = scope:thrall }
		
		POD_first_demon_vessel_effect = { HOST = scope:thrall }
		
		ai_chance = {
			base = 100
		}
	}

	# Auto-choose artifact
	option = {
		name = POD_demon_maintenance.20.option.b
		
		trigger = { exists = scope:this_artifact }
		
		scope:this_artifact = { 
			set_variable = { name = reliquarie value = root } 
			add_soulforged_artifact_modifier_effect = { KILLED = root }
			set_artifact_name = artifact_new_reliquarie_name
			set_artifact_description = artifact_new_reliquarie_desc
		}
		
		ai_chance = {
			base = 100
		}
	}
	
	option = {
		name = POD_demon_maintenance.20.option.c
		
		trigger = { is_landed = yes }
		
		capital_county = {
			set_variable = { name = reliquarie value = root } 
			add_county_modifier = earthbound_modifier
		}
		
		ai_chance = {
			base = 100
		}
	}
	
	option = {
		name = POD_demon_maintenance.20.option.d
		
		custom_tooltip = demon_without_host_tt
		
		add_internal_flag = dangerous
		
		show_as_tooltip = {
			# Scary the noob or perdition of what will come
			death = { death_reason = death_stress }
		}
		
		if = {
			limit  = { has_trait = truefallen }
			POD_remove_host_leech_modifier_effect = yes
			remove_trait = truefallen
			remove_variable = host
		}

		if = {
			limit = { is_ai = yes }
			if = {
				limit = { is_ruler = yes }
				depose = yes
			}
			trigger_event = POD_demon_maintenance.1
		}
	}
}

POD_demon_maintenance.51 = {
	hidden = yes
	type = character_event
	
	trigger = { 
		any_neighboring_and_across_water_top_liege_realm_owner = {
			highest_held_title_tier > 3 #Kings and higher.
			NOT = { is_at_war_with = root }
			NOR = { 
				faith = { has_doctrine = doctrine_infernalism_accepted } 
				religion = { is_in_family = rf_demons }
				POD_is_infernalist_trigger = yes
			}
		}
	}
	
	immediate = {
		every_neighboring_and_across_water_top_liege_realm_owner = {
			limit = { 
				highest_held_title_tier > 3 #Kings and higher.
				NOR = { 
					faith = { has_doctrine = doctrine_infernalism_accepted } 
					religion = { is_in_family = rf_demons }
					POD_is_infernalist_trigger = yes
				}
				NOT = { is_at_war_with = root }
			}
			root = {
				start_war = {
					cb = POD_event_war
					target = prev
				}
			}
		}
	}
}
