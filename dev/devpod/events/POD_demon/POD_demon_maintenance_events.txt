namespace = POD_demon_maintenance

POD_demon_maintenance.1 = {
	hidden = yes
	type = character_event
	
	trigger = {
		has_trait = demon
	}
	
	immediate = {
		if = {
			limit = { 
				is_ai = no
				has_trait = truefallen 
			}
			POD_get_demon_vessel_effect = yes
			if = {
				limit = { NOT = { exists = scope:vessel } }
				trigger_event = POD_demon_maintenance.20
			}
			else = {
				if = {
					limit = { scope:vessel = { is_alive = no } }
					send_interface_message = {
						type = event_generic_bad
						title = POD_demon_maintenance
						left_icon = root
						right_icon = scope:vessel
						scope:vessel = {
							if = { 
								limit = { 
									has_trait = wounded_2
								}
								remove_trait = wounded_2
								add_trait_force_tooltip = wounded_3
								remove_list_variable = { name = corpse_list target = this }
								root = { show_as_tooltip = { POD_remove_host_leech_modifier_effect = yes } }
							}
							else_if = { 
								limit = { 
									has_trait = wounded_1
								}
								remove_trait = wounded_1
								add_trait_force_tooltip = wounded_2
							}
							else = {
								add_trait = wounded_1
							}
						}
					}
				}
				else = {
					send_interface_message = {
						type = event_generic_bad
						title = POD_demon_maintenance
						left_icon = root
						right_icon = scope:vessel
						scope:vessel = {
							POD_health_level_decrease_effect = yes
						}
						custom_tooltip = POD_demon_maintenance.tt
					}
				}
			}
		}
		
		if = {
			limit = {
				NOT = { has_trait = truefallen }
				NOT = { has_trait = earthbound }
				NOT = {
					is_target_in_global_variable_list = {
						name = unavailable_unique_decisions
						target = flag:breakdown_the_abyss_objective_decision
					}
				}
			}
			remove_variable = spirit_destination
			remove_character_modifier = traveling_spirit_world_modifier
			set_variable = { name = spirit_location value = flag:abyss }
			add_character_modifier = trapped_in_abyss_modifier
			trigger_event = POD_umbra_lost.104
		}
		
		send_interface_message = {
			type = event_generic_good
			title = POD_arcana_maintenance
			custom_tooltip = POD_arcana_maintenance.tt
			add_arcana_lifestyle_xp = arcana_xp_gain_value
		}
	}
}
