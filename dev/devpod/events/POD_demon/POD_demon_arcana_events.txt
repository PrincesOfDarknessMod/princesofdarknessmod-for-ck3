namespace = POD_demon_arcana

scripted_trigger POD_can_pay_for_arsenal = {
	piety >= 250
	POD_has_enough_resolve_for_power = { COUNT = 1 }
}

scripted_effect POD_arsenal_cost_effect = {
	add_piety = -250
	POD_resolve_cost_effect = { COUNT = 1 }
}
