namespace = POD_demon_demand

POD_demon_demand.1 = {
	hidden = yes
	
	trigger = {
		is_a_cultist = yes
		is_demon_trigger = no
		has_demon_demand_trigger = no
	}
	
	immediate = {
		random_relation = {
			type = worshipped_demon
			save_scope_as = demon
		}
		random = {
			chance = {
				value = num_of_relation_worshipped_demon
				multiply = 10
			}
			trigger_event = { on_action = POD_demon_demand_events }
		}
	}
}

POD_demon_demand.101 = {
	type = character_event
	content_source = dlc_POD
	title = POD_demon_demand.101.t
	desc = POD_demon_demand.101.desc
	theme = demon

	left_portrait = {
		character = root
		animation = worry
	}

	right_portrait = {
		character = scope:demon
		animation = scheme
	}
	
	trigger = {
		any_held_title = {
			tier = tier_county
			count > 1
		}
	}
	
	immediate = {
		random_held_title = {
			limit = {
				trigger_if = {
					limit = { scope:demon = { is_ruler = yes } }
					tier >= tier_county
				}
				trigger_else = {
					tier = tier_county
				}
			}
			save_scope_as = title 
		}
	}

	option = {
		name = POD_demon_demand.100.a
		
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = title_change
			add_claim_on_loss = no
		}
		scope:title = {
			change_title_holder = {
				holder = scope:demon
				change = scope:title_change
			}
		}
		resolve_title_and_vassal_change = scope:title_change
		
		add_investiture_lifestyle_xp = 100
	}

	option = {
		name = POD_demon_demand.100.b
		
		pay_short_term_gold = {
			gold = major_gold_value
			target = scope:demon
		}
		
		add_investiture_lifestyle_xp = 25
	}

	option = {
		name = POD_demon_demand.100.c
		trigger = { NOT = { has_relation_demonic_master = scope:demon } }
		show_as_unavailable = {has_relation_demonic_master = scope:demon }
		
		add_stress = medium_stress_gain
	}
}

POD_demon_demand.104 = {
	type = character_event
	content_source = dlc_POD
	title = POD_demon_demand.104.t
	desc = POD_demon_demand.104.desc
	theme = demon

	left_portrait = {
		character = root
		animation = worry
	}

	right_portrait = {
		character = scope:demon
		animation = scheme
	}
	
	artifact = {
		target = scope:this_artifact
		position = lower_center_portrait
	}
	
	trigger = {
		any_character_artifact = {
			NOT = { has_variable = possessed_artifact }
			NOT = { has_variable = summoned_artifact }
			NOT = { has_variable = temporary_vessel }
			NOT = { has_variable = wraith_jade }
			NOT = { has_variable = soultaker } # Probably not a good idea
			NOT = { has_variable = soulforge } # Probably not a good idea
			NOT = { artifact_type = animal }
		}
	}
	
	immediate = {
		random_character_artifact = {
			limit = {
				NOT = { has_variable = possessed_artifact }
				NOT = { has_variable = summoned_artifact }
				NOT = { has_variable = temporary_vessel }
				NOT = { has_variable = wraith_jade }
				NOT = { has_variable = soultaker } # Probably not a good idea
				NOT = { has_variable = soulforge } # Probably not a good idea
				NOT = { artifact_type = animal }
			}
			save_scope_as = this_artifact
		}
	}

	option = {
		name = POD_demon_demand.100.a
		
		scope:this_artifact = { 
			set_owner = {
				target = scope:demon
				history = {
					location = scope:demon.location
					actor = root
					recipient = scope:demon
					type = given
				}
			}
		}
		
		add_investiture_lifestyle_xp = 100
	}

	option = {
		name = POD_demon_demand.100.b
		
		pay_short_term_gold = {
			gold = major_gold_value
			target = scope:demon
		}
		
		add_investiture_lifestyle_xp = 25
	}

	option = {
		name = POD_demon_demand.100.c
		trigger = { NOT = { has_relation_demonic_master = scope:demon } }
		show_as_unavailable = {has_relation_demonic_master = scope:demon }
		
		add_stress = medium_stress_gain
	}
}
