namespace = POD_wake_from_torpor

# Wrapper event for the old awaketorpor scheme
POD_wake_from_torpor.1 = {
	hidden = yes
	
	immediate = {
		if = {
			limit = { var:current_journey.var:target = { is_alive = no } }
			POD_clear_journeys_with_target_effect = { TARGET = var:current_journey.var:target }
		}
		else = {
			current_travel_plan ?= {
				pause_travel_plan = yes
			}
			save_scope_as = attacker
			var:current_journey.var:target = {
				save_scope_as = defender
			}
			random_list = {
				20 = { trigger_event = POD_awaketorpor.1 }
				20 = { trigger_event = POD_awaketorpor.2 }
				20 = { trigger_event = POD_awaketorpor.3 }
			}
		}
	}
}
