scripted_effect hunt_in_capital_cleanup_effect = {
	POD_journey_cleanup_effect = yes
	every_in_list = {
		variable = hunt_in_capital_mortals
		limit = { NOT = { has_character_flag = hunted_mortal } }
		death = { death_reason = death_vanished }
	}
	clear_variable_list = hunt_in_capital_mortals
	remove_variable = hunt_in_capital_selected_mortal
}

	type = character_event
	content_source = dlc_POD

	desc  = POD_journey_maintenance.1.desc
	theme = journey
	
	left_portrait = {
		character = root
		animation = personality_bold 
	}

	artifact = {
		target = var:current_journey.var:target_artifact
		position = lower_center_portrait
		trigger = { var:current_journey = { has_variable = target_artifact } }
	}
	
	widget = {
		gui = "POD_event_window_widget_journey"
		container = "custom_widgets_container"
	}
	
	# First option is actually a "fake" button in the event widget
	# because datafunctions work better for this
	
	option = {
		name = POD_journey_maintenance.1.option.b
		custom_tooltip = POD_journey_maintenance.1.option.b.tt
		custom_tooltip = POD_journey_maintenance.1.option.b.tt.2

			remove_character_flag = POD_is_in_travel_phase

				limit = { POD_is_in_travel_phase_trigger = no }
				POD_pause_journey_effect = yes
				POD_auto_resume_journey_effect = yes
			}
		}
	}
}

# Start the travel phase!
POD_journey_maintenance.100 = {
	type = character_event
	content_source = dlc_POD
	title = POD_journey_maintenance.100.t
	theme = travel
	
	immediate = {
		add_character_flag = POD_is_in_travel_phase
		var:current_journey = {
			set_variable = {
				name = travel_plan
				value = prev.current_travel_plan
			}
		}
		if = {
			limit = { has_variable = traveling_with_coterie }
			current_travel_plan = {
				save_scope_as = journey_travel_phase
			}
			var:coterie ?= {
				# add coterie members to travel plan
				every_in_list = {
					variable = members
					limit = {
						NOT = { this = root } # skip player
						POD_can_join_journey_trigger = yes
					}
					scope:journey_travel_phase = {
						add_companion = prev
					}
				}
				# select a coterie member for event flavor
				random_in_list = {
					variable = members
					limit = {
						NOT = { this = root } # skip player
						POD_can_join_journey_trigger = yes
						is_ai = yes  # skip other players in multiplayer
					}
					save_scope_as = companion
				}
			}
		}
	}
	
	left_portrait = {
		character = root
		#animation = personality_bold 
	}
	
	right_portrait = {
		trigger = { exists = scope:companion }
		character = scope:companion
	}
	
	desc = {
		desc = POD_journey_maintenance.100.desc
		triggered_desc = {
			trigger = { exists = scope:companion }
			desc = POD_journey_maintenance.100.desc.companion
		}
	}
	
	option = {
		name = POD_journey_maintenance.100.option.a
		hidden_effect = {
			#
		}
	}
}
