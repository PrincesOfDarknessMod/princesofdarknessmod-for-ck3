namespace = POD_seek_knowledge

scripted_trigger POD_seek_knowledge_is_valid_teacher = {
	is_adult = yes
	is_imprisoned = no
	NOT = { has_trait = torpor }
	NOT = { has_trait = wight }

	save_temporary_scope_as = temp_recipient
	NOT = { POD_custom_incompatible_splats_trigger = { ACTOR = scope:actor RECIPIENT = scope:temp_recipient } }
}

scripted_effect POD_seek_knowledge_find_teacher_numina = {
	hidden_effect = {
		# check coterie members
		if = {
			limit = {
				var:coterie ?= {
					any_in_list = {
						variable = members
						count > 0
						NOT = { this = scope:actor }

						has_trait = $TRAIT$
					}
				}
			}
			var:coterie = {
				random_in_list = {
					variable = members
					limit = {
						NOT = { this = scope:actor }
						has_trait = $TRAIT$
					}
					save_scope_as = recipient
				}
				save_scope_value_as = {
					name  = teacher_flag
					value = flag:coterie_member
				}
			}
		}
		# check local ruler
		else_if = {
			limit = {
				scope:actor.location.county_controller = {
					has_trait = $TRAIT$
						
					opinion = {
						target = scope:actor
						value >= 0
					}
					
					POD_seek_knowledge_is_valid_teacher = yes
				}
			}
			scope:temp_recipient ?= { save_scope_as = recipient }
			save_scope_value_as = {
				name  = teacher_flag
				value = flag:local_ruler
			}
		}
		# check wanderers
		else_if = {
			limit = {
				any_pool_character = {
					province = root.capital_province
					
					has_trait = $TRAIT$
					
					opinion = {
						target = scope:actor
						value >= 0
					}
					
					POD_seek_knowledge_is_valid_teacher = yes
				}
			}
			random_pool_character = {
				province = root.capital_province
				limit = {
					has_trait = $TRAIT$
					
					opinion = {
						target = scope:actor
						value >= 0
					}
					
					POD_seek_knowledge_is_valid_teacher = yes
				}
				save_scope_as = recipient
			}
			save_scope_value_as = {
				name  = teacher_flag
				value = flag:wanderer
			}
		}
		# create new character
		else = {
			create_character = {
				template = POD_journey_teacher_template
				dynasty = none
				location = root.location
				save_scope_as = recipient
			}
			
			if = {
				limit = { exists = scope:recipient } # error suppression

				scope:recipient = {
					POD_character_initialization_for_mortal_commoners_effect = yes
					add_trait = $TRAIT$
					add_character_flag = temporary_teacher
					
					set_employer = character:POD_wraith_01 # TODO: is there a better way to do this?
					return_to_court = yes
					add_character_flag = blocked_from_leaving
				}
			}
			
			save_scope_value_as = {
				name  = teacher_flag
				value = flag:mortal
			}
		}
	}
}

scripted_effect POD_seek_knowledge_find_teacher_discipline = {
	hidden_effect = {
		# check coterie members
		if = {
			limit = {
				var:coterie ?= {
					any_in_list = {
						variable = members
						count > 0
						NOT = { this = scope:actor }

						OR = {
							has_trait = $TRAIT$
							has_trait = $TRAIT$advanced
						}
					}
				}
			}
			var:coterie = {
				random_in_list = {
					variable = members
					limit = {
						NOT = { this = scope:actor }
						OR = {
							has_trait = $TRAIT$
							has_trait = $TRAIT$advanced
						}
					}
					save_scope_as = recipient
				}
				save_scope_value_as = {
					name  = teacher_flag
					value = flag:coterie_member
				}
			}
		}
		# check local ruler
		else_if = {
			limit = {
				scope:actor.location.county_controller = {
					OR = {
						has_trait = $TRAIT$
						has_trait = $TRAIT$advanced
					}
						
					opinion = {
						target = scope:actor
						value >= 0
					}
					
					POD_seek_knowledge_is_valid_teacher = yes
				}
			}
			scope:temp_recipient ?= { save_scope_as = recipient }
			save_scope_value_as = {
				name  = teacher_flag
				value = flag:local_ruler
			}
		}
		# check wanderers
		else_if = {
			limit = {
				any_pool_character = {
					province = root.capital_province
					
					OR = {
						has_trait = $TRAIT$
						has_trait = $TRAIT$advanced
					}
					
					opinion = {
						target = scope:actor
						value >= 0
					}
					
					POD_seek_knowledge_is_valid_teacher = yes
				}
			}
			random_pool_character = {
				province = root.capital_province
				limit = {
					
					OR = {
						has_trait = $TRAIT$
						has_trait = $TRAIT$advanced
					}
					
					opinion = {
						target = scope:actor
						value >= 0
					}
					
					POD_seek_knowledge_is_valid_teacher = yes
				}
				save_scope_as = recipient
			}
			save_scope_value_as = {
				name  = teacher_flag
				value = flag:wanderer
			}
		}
		# create new character
		else = {
			create_character = {
				template = POD_journey_teacher_template
				dynasty = none
				location = root.location
				save_scope_as = recipient
			}

			if = {
				limit = { exists = scope:recipient } # error suppression

				if = {
					# Different setup for gargoyles
					limit = { trait:$TRAIT$ = trait:visceratika }
	
					scope:recipient = {
						add_trait = vampire
						add_trait = vampgeneration10
						add_trait = gargoyle
						add_trait = flight
						add_trait = fortitude
						add_trait = potence
						POD_assign_random_portrait_horn_effect = yes
						add_character_flag = POD_portrait_monster_dragon_wings_flag
						add_character_flag = POD_portrait_claws_form_flag
						add_character_modifier = POD_gargoyle_caste_sentinel_modifier
						add_character_modifier = bloodpotency1_modifier
						POD_character_masquerade_initialization_effect = yes
						add_character_modifier = bloodhunger1_modifier
						POD_set_resonance_effect = yes
						remove_disease_trait_effect = yes
						add_character_flag = immune_to_disease
					}
	
					save_scope_value_as = {
						name  = teacher_flag
						value = flag:gargoyle
					}
				}
				else = {
					# Every other discipline gets an independent ghoul
					scope:recipient = { POD_character_initialization_for_forsaken_commoners_effect = yes }
	
					save_scope_value_as = {
						name  = teacher_flag
						value = flag:ghoul
					}
				}
				
				scope:recipient = {
					set_character_faith = root.location.faith
					add_trait = $TRAIT$
					add_character_flag = temporary_teacher
					
					set_employer = character:POD_wraith_01
					return_to_court = yes
					add_character_flag = blocked_from_leaving
				}
			}
		}
	}
}

# Safeguard for not getting stuck, until a better bugfix can be found
POD_seek_knowledge.1 = {
	hidden = yes
	
	immediate = {
		if = {
			limit = { has_variable = current_journey }
			# Turn back if the player already knows the ability they're trying to learn
			if = {
				limit = {
					POD_already_has_seek_knowledge_trait_trigger = {
						VAR = var:current_journey.var:journey_type
					}
				}
				var:current_journey.var:travel_plan ?= {
					resume_travel_plan = yes
					prev = { return_home = yes }
				}
				POD_journey_cleanup_effect = yes
				send_interface_message = {
					type  = POD_journey_abandoned
					title = POD_journey_abandoned_title
					desc  = POD_journey_abandoned_desc_trait
				}
			}
			else = {
				trigger_event = POD_seek_knowledge.2
			}
		}
		else = {
			trigger_event = POD_seek_knowledge.3
		}
	}
}
