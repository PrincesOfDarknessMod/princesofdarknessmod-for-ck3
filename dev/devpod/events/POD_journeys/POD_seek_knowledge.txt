scripted_effect POD_seek_knowledge_find_teacher_numina = {
	# check local ruler
	if = {
		limit = {
			scope:actor.location.county_controller = {
				has_trait = $TRAIT$
				
				opinion = {
					target = scope:actor
					value >= 0
				}
				
				is_available_ai_adult = yes
			}
		}
		scope:actor.location.county_controller = { save_scope_as = recipient }
		save_scope_value_as = {
			name  = teacher_flag
			value = flag:local_ruler
		}
	}
	else = {
		# check wanderers
		if = {
			limit = {
				any_pool_character = {
					province = root.capital_province
					
					has_trait = $TRAIT$
					
					opinion = {
						target = scope:actor
						value >= 0
					}
					
					is_available_ai_adult = yes
				}
			}
			random_pool_character = {
				province = root.capital_province
				limit = {
					has_trait = $TRAIT$
					
					opinion = {
						target = scope:actor
						value >= 0
					}
					
					is_available_ai_adult = yes
				}
				save_scope_as = recipient
			}
			save_scope_value_as = {
				name  = teacher_flag
				value = flag:wanderer
			}
		}
		# create new character
		else = {
			create_character = {
				template = POD_journey_teacher_template
				dynasty = none
				location = root.location
				save_scope_as = recipient
			}
			
			scope:recipient = {
				POD_character_initialization_for_mortal_commoners_effect = yes
				add_trait = $TRAIT$
				add_character_flag = temporary_teacher
				
				set_employer = character:POD_wraith_01 # TODO: is there a better way to do this?
				return_to_court = yes
				add_character_flag = {
					flag = blocked_from_leaving
					years = 5
				}
			}
			
			save_scope_value_as = {
				name  = teacher_flag
				value = flag:mortal
			}
		}
	}
}

# Wrapper event for selecting the right teacher
POD_seek_knowledge.1 = {
	hidden = yes
	
	immediate = {
		save_scope_as = actor
		
		current_travel_plan ?= {
			pause_travel_plan = yes
		}
		
		POD_seek_knowledge_injector = {
			PREFIX = POD_seek_knowledge_find_teacher
			START_MENTORING = yes
		}
		
		trigger_event = POD_seek_knowledge.2
	}
}
