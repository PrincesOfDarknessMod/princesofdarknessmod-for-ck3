    custom_tooltip = {
        text = POD_hunt_for_sargon_codex_shard.tt

        POD_copy_artifact_effect = yes
        hidden_effect_new_object = { 
            scope:newly_created_artifact = {
                if = {
                    limit = { has_ep1_dlc_trigger = yes }
                    reforge_artifact = {
                        name = artifact_shard_copy_name
                        description = artifact_shard_copy_desc
                        type = book
                        visuals = book_POD_black
                        history = {
                            type = created_before_history
                        }		
                        decaying = no
                    }
                }
                else = {
                    reforge_artifact = {
                        name = artifact_shard_copy_name
                        description = artifact_shard_copy_desc
                        type = miscellaneous_when_not_court
                        visuals = book_POD_black
                        history = {
                            type = created_before_history
                        }	
                        decaying = no
                    }
                }
            }
        }
    }
}
# Shard is sold as 'magic tablet' in some obscure village 
POD_hunt_for_sargon_codex_shard.3 = {
    type = character_event
	content_source = dlc_POD
    title = POD_hunt_for_sargon_codex_shard.3.t
	theme = travel

	desc = POD_hunt_for_sargon_codex_shard.3.desc

	left_portrait = {
		character = root
		animation = personality_zealous
	}

	artifact = {
		target = scope:target
		position = lower_center_portrait
	}

	option = { # Buy normally and hope no know will know if I don't care about kine
		name = POD_hunt_for_sargon_codex_shard.3.a
        remove_short_term_gold = massive_gold_max_value
        if = { 
            limit = { faith = { has_doctrine_parameter = conviction_opinion_active } }
            random = {
                chance = 25
                add_prestige = -750
            }
        }
        sargon_copy_effect = yes
	}

	option = { # Steal
		name = POD_hunt_for_sargon_codex_shard.3.b
        random = {
            chance = 50
            add_prestige = -1500
        }
        POD_commit_theft_effect = yes
        sargon_copy_effect = yes
	}

	option = { # Kill everyone
		name = POD_hunt_for_sargon_codex_shard.3.c
        if = { 
            limit = { faith = { NOT = { has_doctrine_parameter = conviction_opinion_active } } }
            random = {
                chance = 25
                add_prestige = -1500
            }
            add_piety = -1500
        }
        POD_very_high_risk_of_increasing_masquerade_exposure = yes
        POD_commit_murder_effect = yes
        sargon_copy_effect = yes
	}

	option = { # Use presence or dominate
		name = POD_hunt_for_sargon_codex_shard.3.d
        trait = dominate
        trait = dominateadvanced
        trait = presence
        trait = presenceadvanced
        trigger = {
            has_trait = dominatediscipline
            has_trait = presencediscipline
        }
		POD_hunger_rouse_check_effect = yes
        sargon_copy_effect = yes
	}
    after = {
		POD_end_journey_effect = yes
    }
}
