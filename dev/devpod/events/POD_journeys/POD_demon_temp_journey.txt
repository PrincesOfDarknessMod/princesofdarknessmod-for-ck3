namespace = POD_demon_temp_journey

scripted_effect demon_temp_journey_effect = {
	faith = {
		switch = {
			trigger = has_doctrine
			special_doctrine_vices_avarice  = { trait:greedy     = { save_scope_as = sin_trait } }
			special_doctrine_vices_envy     = { trait:vengeful   = { save_scope_as = sin_trait } }
			special_doctrine_vices_gluttony = { trait:gluttonous = { save_scope_as = sin_trait } }
			special_doctrine_vices_lust     = { trait:lustful    = { save_scope_as = sin_trait } }
			special_doctrine_vices_pride    = { trait:arrogant   = { save_scope_as = sin_trait } }
			special_doctrine_vices_sloth    = { trait:lazy       = { save_scope_as = sin_trait } }
			special_doctrine_vices_wrath    = { trait:wrathful   = { save_scope_as = sin_trait } }
		}
	}
	
	if = {
		limit = { 
			OR = {
				AND = {
					exists = scope:sin_trait
					scope:target = { has_trait = scope:sin_trait }
				}
				NOT = { exists = scope:sin_trait }
				has_character_flag = can_tempt_from_any_sin
			}
		}
		scope:target = { set_character_faith = prev.faith }
	}
	else = {
		scope:target = {
			every_character_trait = {
				limit = { is_opposite_of_trait = scope:sin_trait }
				prev = { remove_trait = prev }
			}
			add_trait_force_tooltip = scope:sin_trait
		}
	
		custom_tooltip = POD_demon_temp_journey.1.a.tt
	}
}

POD_demon_temp_journey.1 = {
	content_source = dlc_POD
	type = character_event
	
	theme = demon
	override_background = { reference = shader_event_ritual }
	
	widget = {
		gui = "POD_shader_event_ritual"
		container = "background_shader_vfx_container"
	}

	title = POD_demon_temp_journey.1.t
	desc = {
		desc = POD_demon_temp_journey.1.desc
	}

	left_portrait = {
		character = root
	}
	
	right_portrait = {
		character = scope:target
	}
	
	immediate = {
		var:current_journey.var:target = {
			save_scope_as = target
		}
		POD_save_coterie_duelists = yes
	}

	option = {
		name = POD_demon_temp_journey.1.a
		scope:intrigue_duelist = {
			duel = {
				skills = { intrigue }
				target = scope:target
				1 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = 0.5
					}
					root = {
						send_interface_toast = {
							title = POD_shared_loca_success
							demon_temp_journey_effect = yes
						}
					}
				}
				3 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = -0.5
					}
					root = {
						send_interface_toast = {
							title = POD_shared_loca_failure
							add_stress = 20
							scope:target = {
								add_character_modifier = {
									modifier = POD_failed_tempting_modifier
									years = 25
								}
								add_opinion = {
									target = root
									modifier = angry_opinion
									opinion = -50
								}
							}
						}
					}
				}
			}
		}
	}

	option = {
		name = POD_demon_temp_journey.1.b
		scope:diplomacy_duelist = {
			duel = {
				skills = { diplomacy }

						value = scope:duel_value
						multiplier = 0.5
					}
					root = {
						send_interface_toast = {
							title = POD_shared_loca_success
							demon_temp_journey_effect = yes
						}
					}
				}
				3 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = -0.5
					}
					root = {
						send_interface_toast = {
							title = POD_shared_loca_failure
							add_stress = 20
							scope:target = {
								add_character_modifier = {
									modifier = POD_failed_tempting_modifier
									years = 25
								}
								add_opinion = {
									target = root
									modifier = angry_opinion
									opinion = -50
								}
							}
						}
					}
				}
			}
		}
	}

	after = { 
		POD_end_journey_effect = yes
	}
