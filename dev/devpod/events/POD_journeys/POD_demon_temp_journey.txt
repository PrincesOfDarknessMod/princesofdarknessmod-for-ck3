namespace = POD_demon_temp_journey
scripted_effect demon_temp_journey_effect = {
	if = {
		limit = { 
			OR = {
				AND = { faith = { has_doctrine = special_doctrine_vices_avarice } var:current_journey.var:target = { has_trait = greedy } }
				AND = { faith = { has_doctrine = special_doctrine_vices_envy } var:current_journey.var:target = { has_trait = vengeful } }
				AND = { faith = { has_doctrine = special_doctrine_vices_gluttony } var:current_journey.var:target = { has_trait = gluttonous } }
				AND = { faith = { has_doctrine = special_doctrine_vices_lust } var:current_journey.var:target = { has_trait = lustful } }
				AND = { faith = { has_doctrine = special_doctrine_vices_pride } var:current_journey.var:target = { has_trait = arrogant } }
				AND = { faith = { has_doctrine = special_doctrine_vices_sloth } var:current_journey.var:target = { has_trait = lazy } }
				AND = { faith = { has_doctrine = special_doctrine_vices_wrath } var:current_journey.var:target = { has_trait = wrathful } }
				scope:demon = { NOT = { has_trait = demon } }
			}
		}
		save_scope_as = demon
		var:current_journey.var:target = { set_character_faith = scope:demon.faith }
	}	
	else = {
		custom_tooltip = POD_demon_temp_journey.1.a.tt
		if = {
			limit = { faith = { has_doctrine = special_doctrine_vices_avarice } var:current_journey.var:target = { NOT = { has_trait = greedy } } }
			if = { limit = { var:current_journey.var:target = { has_trait = generous } } var:current_journey.var:target = { remove_trait = generous } var:current_journey.var:target = { add_trait_force_tooltip = greedy } }
			else = { var:current_journey.var:target = { add_trait = greedy } }
		}		
		else_if = {
			limit = {faith = { has_doctrine = special_doctrine_vices_envy } var:current_journey.var:target = { NOT = { has_trait = vengeful } } }
			if = { limit = { var:current_journey.var:target = { has_trait = forgiving } } var:current_journey.var:target = { remove_trait = forgiving } var:current_journey.var:target = { add_trait_force_tooltip = vengeful } }
			else = { var:current_journey.var:target = { add_trait = vengeful } }
		}				
		else_if = {
			limit = { faith = { has_doctrine = special_doctrine_vices_gluttony } var:current_journey.var:target = { NOT = { has_trait = gluttonous } } }
			if = { limit = { var:current_journey.var:target = { has_trait = temperate } } var:current_journey.var:target = { remove_trait = temperate } var:current_journey.var:target = { add_trait_force_tooltip = gluttonous } }
			else = { var:current_journey.var:target = { add_trait = gluttonous } }
		}			
		else_if = {
			limit = { faith = { has_doctrine = special_doctrine_vices_lust } var:current_journey.var:target = { NOT = { has_trait = lustful } } }
			if = { limit = { var:current_journey.var:target = { has_trait = chaste } } var:current_journey.var:target = { remove_trait = chaste } var:current_journey.var:target = { add_trait_force_tooltip = lustful } }
			else = { var:current_journey.var:target = { add_trait = lustful } }
		}	
		else_if = {
			limit = { faith = { has_doctrine = special_doctrine_vices_pride } var:current_journey.var:target = { NOT = { has_trait = arrogant } } }
			if = { limit = { var:current_journey.var:target = { has_trait = humble } } var:current_journey.var:target = { remove_trait = humble } var:current_journey.var:target = { add_trait_force_tooltip = arrogant } }
			else = { var:current_journey.var:target = { add_trait = arrogant } }
		}	
		else_if = { 
			limit = { faith = { has_doctrine = special_doctrine_vices_sloth } var:current_journey.var:target = { NOT = { has_trait = lazy } } }
			if = { limit = { var:current_journey.var:target = { has_trait = diligent } } var:current_journey.var:target = { remove_trait = diligent } var:current_journey.var:target = { add_trait_force_tooltip = lazy } }
			else = { var:current_journey.var:target = { add_trait = lazy } }
		}
		else_if = {
			limit = { faith = { has_doctrine = special_doctrine_vices_wrath } var:current_journey.var:target = { NOT = { has_trait = wrathful } } }
			if = { limit = { var:current_journey.var:target = { has_trait = calm } } var:current_journey.var:target = { remove_trait = calm } var:current_journey.var:target = { add_trait_force_tooltip = wrathful } }
			else = { var:current_journey.var:target = { add_trait = wrathful } }
		}
	}
}
