namespace = POD_heists

# Arrival via artifact_heist Journey
POD_heists.1 = {
	type = character_event
	content_source = dlc_POD
	title = POD_heists.1.t
	desc  = POD_heists.1.desc
	theme = heist
	
	left_portrait = {
		character = root
		animation = personality_bold 
	}
	
	option = {
		name = POD_heists.1.a
		if = {
			limit = { POD_is_on_heist_trigger = no }

			play_music_cue = "mood_twilights_approach"

			POD_heist_setup_effect = {
				MAP     = star
				ENDING  = POD_heists.100
				SAVE_AS = heist
			}
			POD_heist_set_target_artifact = {
				HEIST    = scope:heist
				ARTIFACT = var:current_journey.var:target_artifact
			}
			POD_heist_setup_suspicion_effect = {
				HEIST  = scope:heist
				EVENT1 = POD_heists.1001
				EVENT2 = POD_heists.1002
				EVENT3 = POD_heists.1003
				EVENT4 = POD_heists.101
			}
		}
	}
	
	option = {
		name = POD_heists.1.b
		POD_end_journey_can_retry_effect = yes
	}
}
