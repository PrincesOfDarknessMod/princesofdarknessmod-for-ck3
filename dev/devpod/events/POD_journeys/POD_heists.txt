# Arrival via artifact_heist Journey
POD_heists.1 = {
	type = character_event
	content_source = dlc_POD
	title = POD_heist_arrival_title
	theme = heist
	override_background = { reference = terrain_travel }
	override_sound = { reference = "event:/DLC/EP2/SFX/Events/Travel/ep2_evn_travel_end_neutral" }
	
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					POD_custom_incompatible_splats_trigger = {
						ACTOR = root
						RECIPIENT = scope:target_court
					}
				}
				desc = {
					first_valid = {
						triggered_desc = {
							trigger = { POD_has_coterie_members_present_trigger = yes }
							desc = POD_heists.1.desc.hostile.coterie
						}
						desc = POD_heists.1.desc.hostile.nocoterie
					}
				}
			}
			desc = {
				first_valid = {
					triggered_desc = {
						trigger = { POD_has_coterie_members_present_trigger = yes }
						desc = POD_heists.1.desc.coterie
					}
					desc = POD_heists.1.desc.nocoterie
				}
			}
		}
	}
	
	left_portrait = {
		character = root
		animation = personality_bold 
	}

	artifact = {
		target = var:current_journey.var:target_artifact
		position = lower_center_portrait
	}

	immediate = {
		hidden_effect = {
			POD_pause_journey_travel_plan_effect = yes
			var:current_journey.var:target_artifact.artifact_owner = { save_scope_as = target_court }
		}
	}
	
	option = {
		name = POD_heist_arrival_confirm

		play_music_cue = "mood_twilights_approach"

		POD_heist_setup_and_select_map = {
			COURT   = scope:target_court
			ENDING  = POD_heists.100
			SAVE_AS = heist
		}
		POD_heist_select_suspicion_events = {
			HEIST         = scope:heist
			ON_FAILURE    = POD_heists.102
			FORCE_HOSTILE = no
		}
		POD_heist_initialize_artifact_heist = {
			HEIST    = scope:heist
			ARTIFACT = var:current_journey.var:target_artifact
		}
	}
	
	option = {
		name = POD_heist_arrival_decline
		scope:target_court = {
			add_character_modifier = {
				modifier = wary_of_infiltrators_modifier
				years = 20
			}
		}
		POD_end_journey_effect = yes
	}
}

# Successful escape during an artifact_heist Journey
POD_heists.100 = {
	type = character_event
	content_source = dlc_POD
	title = POD_heists.100.t
	theme = heist
	override_background = { reference = terrain_travel }
	
	override_sound = {
		trigger = { POD_heists_100_is_successful = yes }
		reference = "event:/DLC/EP1/SFX/UI/Notifications/inspiration_project_complete"
	}
	override_sound = { reference = "event:/DLC/EP2/SFX/Events/Travel/ep2_evn_travel_end_neutral" }
	
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { POD_has_coterie_members_present_trigger = yes }
				desc = POD_heists.100.desc.coterie
			}
			desc = POD_heists.100.desc.nocoterie
		}
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:success_artifact }
				desc = POD_heists.100.desc.artifact
			}
			triggered_desc = {
				trigger = {
					var:current_heist = { has_variable = target_artifact }
					NOT = { exists = scope:success_artifact }
				}
				desc = POD_heists.100.desc.noartifact
			}
			triggered_desc = {
				trigger = { exists = scope:success_hunt }
				desc = POD_heists.100.desc.hunt
			}
			triggered_desc = {
				trigger = {
					var:current_heist = { has_variable = hunt_target }
					NOT = { exists = scope:success_hunt }
				}
				desc = POD_heists.100.desc.nohunt
			}
			triggered_desc = {
				trigger = { exists = scope:success_gold }
				desc = POD_heists.100.desc.gold
			}
			desc = POD_heists.100.desc.nogold
		}
	}
	
	left_portrait = {
		character = root
		animation = personality_bold 
	}

	artifact = {
		target = scope:displayed_artifact
		position = lower_center_portrait
		trigger = { exists = scope:displayed_artifact }
	}

	immediate = {
		var:current_heist ?= {
			random_in_list = {
				variable = loot_artifacts
				limit = { prev.var:target_artifact ?= this }
				save_scope_as = displayed_artifact
				save_scope_value_as = { name = success_artifact value = yes }
			}
			var:court_owner = {
				if = {
					limit = { is_alive = yes }
					add_character_modifier = {
						modifier = wary_of_infiltrators_modifier
						years = 20
					}
				}
			}
			var:gold_owner ?= {
				save_scope_as = gold_owner
			}
		}
		if = {
			limit = {
				var:current_heist.var:stolen_gold > 0
			}
			save_scope_value_as = { name = success_gold value = yes }
		}
		if = {
			limit = {
				var:current_heist = { has_variable = success_hunt }
			}
			save_scope_value_as = { name = success_hunt value = yes }
		}

		POD_close_heist = yes
		POD_heist_gain_loot_effect = yes
		if = {
			limit = { POD_heists_100_is_successful = yes }
			POD_heist_increase_infiltration_skill = yes
			POD_fera_add_renown_effect = { TYPE = cunning }
			POD_heist_perk_attack_effect = yes
			random_character_active_contract = {
				task_contract_type = POD_ghoul_mission_steal_rivals_gold
				limit = {
					exists = scope:gold_owner
					POD_ghoul_mission_can_be_finished = yes
					task_contract_employer = {
						has_relation_rival = scope:gold_owner
					}
				}
				complete_task_contract = success_standard
			}
		}
		POD_heist_display_expenditures = yes
	}
	
	option = {
		name = {
			trigger = {
				POD_heists_100_is_successful = yes
				POD_has_coterie_members_present_trigger = yes
			}
			text = POD_heists.100.a.success.coterie
		}
		name = {
			trigger = {
				POD_heists_100_is_successful = yes
				POD_has_coterie_members_present_trigger = no
			}
			text = POD_heists.100.a.success.nocoterie
		}
		name = {
			trigger = {
				POD_heists_100_is_successful = no
			}
			text = POD_heists.100.a.failure
		}
		POD_heist_cleanup_effect = yes
		POD_end_journey_effect = yes
	}
}


# Hidden event when failing an artifact_heist Journey (select outcome based on splat hostility)
POD_heists.102 = {
	hidden = yes

	immediate = {
		if = {
			limit = {
				POD_heist_is_hostile_trigger = { HEIST = var:current_heist }
				var:current_heist.var:court_owner = { POD_is_willing_to_kill_trigger = yes }
			}
			# Hostile splat: Kill the infiltrators
			trigger_event = POD_heists.103
		}
		else = {
			# Friendly splat: Imprison the infiltrators and give them a lasting debuff
			trigger_event = POD_heists.101
		}
	}
}


# Failed an artifact_heist on a friendly splat, resulting in imprisonment and a debuff
POD_heists.101 = {
	type = character_event
	content_source = dlc_POD
	title = POD_heists.101.t
	theme = heist_suspicion
	override_sound = { reference = "event:/SFX/Events/Themes/sfx_event_theme_type_health_bad" }
	
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:heist_imprisoner = { is_ruler = no } }
				desc = POD_heists.101.desc.landless
			}
			triggered_desc = {
				trigger = {
					NOT = {
						POD_damage_spirit_custom_trigger = {
							RECIPIENT = this
							ACTOR = scope:heist_imprisoner
						}
					}
				}
				desc = POD_heists.101.desc.spirit
			}
			desc = POD_heists.101.desc
		}
	}
	
	left_portrait = {
		character = root
		animation = prisondungeon
	}

	right_portrait = {
		character = scope:heist_imprisoner
		animation = war_defender
	}

	widget = {
		gui = "POD_event_window_widget_heist_suspicion_threshold_3"
		container = "custom_widgets_container"
	}

	immediate = {
		POD_setup_heist_event_background = yes
		POD_heist_suspicion_threshold_4_tooltip = yes
		POD_close_heist = yes
		var:current_heist.var:court_owner = {
			save_scope_as = heist_imprisoner
			add_character_modifier = {
				modifier = wary_of_infiltrators_modifier
				years = 20
			}
		}

		POD_heist_give_stolen_gold_back = yes

		add_character_modifier = {
			modifier = heist_caught_modifier
			years = 20
		}

		if = {
			limit = {
				var:current_heist.var:court_owner = {
					is_vassal_or_below_of = root
				}
			}
			custom_tooltip = {
				text = POD_heist_legitimacy_loss_tt
				add_legitimacy = major_legitimacy_loss
			}
		}
	}
	
	option = {
		name = POD_heists.101.a

		scope:heist_imprisoner = {
			progress_towards_rival_effect = {
				CHARACTER = root
				REASON    = rival_infiltration
				OPINION   = default_rival_opinion
			}
		}

		if = {
			limit = { POD_has_coterie_members_trigger = yes }
			var:coterie = {
				every_in_list = {
					variable = members
					limit = {
						POD_is_in_same_journey_trigger = { TARGET = root }
					}
					POD_heist_get_imprisoned_if_possible = {
						IMPRISONER = scope:heist_imprisoner
					}
				}
			}
		}
		else = {
			POD_heist_get_imprisoned_if_possible = {
				IMPRISONER = scope:heist_imprisoner
			}
		}

		POD_heist_cleanup_effect = yes
		hidden_effect = { POD_end_journey_effect = yes }
	}
}

	title = {
		first_valid = {

			}

		}
	}

		first_valid = {

					OR = {
						scope:prisoner_type = flag:relation

			}
			triggered_desc = {
				trigger = { scope:prisoner_type = flag:friendly_splat }

			}
			triggered_desc = {
				trigger = { scope:prisoner_type = flag:hostile_splat }

			}
		}
	}

		character = root
	}
	
	right_portrait = {
		character = scope:prisoner_to_release
		animation = prisondungeon
	}

	immediate = {
		POD_setup_heist_event_background = yes
		POD_heist_pause_for_node_event = yes
	}

	option = {
		name = POD_heists.200.a
		scope:prisoner_to_release = {
			if = {
				limit = { is_imprisoned = yes } # safeguard
				release_from_prison = yes
			}
			add_opinion = {
				modifier = released_from_prison
				target = root
			}
		}
		stress_impact = {
			compassionate = medium_stress_impact_loss
			sadistic = medium_stress_impact_gain
			callous = minor_stress_impact_gain
		}
	}

	option = {
		name = POD_heists.200.b
		trigger = { scope:prisoner_to_release = { gold >= 1 } }
		scope:prisoner_to_release = {
			if = {
				limit = { is_imprisoned = yes } # safeguard
				release_from_prison = yes
			}
			pay_short_term_gold = {
				gold = {
					if = {
						limit = { gold >= ransom_cost_value }
						add = ransom_cost_value
					}
					else = {
						add = gold
					}
				}
				target = root
			}
			add_opinion = {
				target = root
				modifier = requested_money_opinion
				opinion = -20
			}
		}
		stress_impact = {
			compassionate = major_stress_impact_gain
			forgiving = major_stress_impact_gain
			generous = medium_stress_impact_gain
			greedy = medium_stress_impact_loss
		}
	}

	option = {
		name = POD_heists.200.c
		trigger = {
			scope:prisoner_to_release = {
				is_ruler = no
				NOT = { is_courtier_of = root }
			}

					ACTOR = root
					RECIPIENT = scope:prisoner_to_release
				}
			}
		}

			if = {
				limit = { is_imprisoned = yes } # safeguard

			}

			}

			}

			}

			compassionate = medium_stress_impact_gain
			sadistic = medium_stress_impact_loss
		}
	}

	after = {
		POD_heist_resume_after_node_event = yes
	}
}


# Outcome event when dueling an elite knight
POD_heists.201 = {
	hidden = yes

	immediate = {
		if = {
			limit = { root = scope:sc_victor }
			trigger_event = POD_heists.202
		}
		else = {
			POD_heist_resume_after_node_event = yes
			if = {
				limit = { scope:sc_victor = { POD_is_willing_to_kill_trigger = yes } }
				POD_die_during_heist_effect = {
					DEATH_REASON = death_duel
					KILLER = scope:sc_victor
				}
			}
			else = {
				# If the knight isn't willing to kill, deliver the Coterie to their liege instead
				POD_heist_trigger_suspicion_failure_event = yes
			}
		}
		
		var:current_heist ?= {
			var:current_location = {
				POD_heist_clear_node_type = yes
				POD_heist_set_node_type_prowess = yes
			}
		}
	}
}

				limit = { NOT = { exists = scope:potential_rival } }

						root = { can_set_relation_rival_trigger = { CHARACTER = prev } }
						NOT = { this = scope:spymaster }
						POD_is_valid_heist_adversary = yes
					}
					order_by = sum_of_all_skills_value
					save_scope_as = potential_rival
				}
			}

		}

		name = {
			trigger = { POD_has_coterie_members_present_trigger = yes }
			text = POD_heists.1011.a.coterie
		}
		name = {
			trigger = { POD_has_coterie_members_present_trigger = no }
			text = POD_heists.1011.a.nocoterie
		}
		POD_heist_suspicion_event_skill_check = {
			EVENT   = 1
			SKILL   = diplomacy
			LOC_KEY = POD_heists.1011.a
			ON_SUCCESS = ""
			ON_FAILURE = "POD_heist_raise_suspicion_effect = { VALUE = minor_suspicion_value }"
		}
	}

			EVENT   = 1
			SKILL   = intrigue
			LOC_KEY = POD_heists.1011.b
			ON_SUCCESS = ""
			ON_FAILURE = "POD_heist_raise_suspicion_effect = { VALUE = minor_suspicion_value }"
