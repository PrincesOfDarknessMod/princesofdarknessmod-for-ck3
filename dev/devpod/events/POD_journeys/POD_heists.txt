	content_source = dlc_POD

	desc  = POD_heists.1.desc

	override_sound = { reference = "event:/DLC/EP2/SFX/Events/Travel/ep2_evn_travel_end_neutral" }

	}

				pause_travel_plan = yes

		play_music_cue = "mood_twilights_approach"

			MAP     = star
			ENDING  = POD_heists.100

		}
		POD_heist_initialize_artifact_heist = {
			HEIST    = scope:heist
			ARTIFACT = var:current_journey.var:target_artifact
		}
		POD_heist_setup_suspicion_effect = {
			HEIST  = scope:heist
			EVENT1 = POD_heists.1001
			EVENT2 = POD_heists.1002
			EVENT3 = POD_heists.1003
			EVENT4 = POD_heists.101
		}
	}
	
	option = {
		name = POD_heists.1.b
		POD_end_journey_can_retry_effect = yes
	}
}

# Successful escape during an artifact_heist Journey
POD_heists.100 = {
	type = character_event
	content_source = dlc_POD
	title = POD_heists.100.t
	desc  = POD_heists.100.desc
	theme = heist
	override_background = { reference = terrain_travel }
	override_sound = { reference = "event:/DLC/EP1/SFX/UI/Notifications/inspiration_project_complete" }
	
	left_portrait = {
		character = root
		animation = personality_bold 
	}

	artifact = {
		target = scope:displayed_artifact
		position = lower_center_portrait
		trigger = { exists = scope:displayed_artifact }
	}

	immediate = {
		var:current_heist ?= {
			random_in_list = {
				variable = loot_artifacts
				limit = { this = prev.var:target_artifact }
				save_scope_as = displayed_artifact
			}
		}
		POD_close_heist = yes
		POD_heist_gain_loot_effect = yes
	}
	
	option = {
		name = POD_heists.100.a
		POD_heist_cleanup_effect = yes
		POD_end_journey_effect = yes
	}
}


# Got caught during an artifact_heist Journey
POD_heists.101 = {
	type = character_event
	content_source = dlc_POD
	title = POD_heists.101.t
	desc  = POD_heists.101.desc
	theme = heist_suspicion
	override_sound = { reference = "event:/SFX/Events/Themes/sfx_event_theme_type_health_bad" }
	
	left_portrait = {
		character = root
		animation = prisondungeon
	}

	right_portrait = {
		character = scope:heist_imprisoner
		animation = war_defender
	}

	widget = {
		gui = "POD_event_window_widget_heist_suspicion_threshold_3"
		container = "custom_widgets_container"
	}

	immediate = {
		POD_heist_suspicion_threshold_4_tooltip = yes
		POD_close_heist = yes
		var:current_heist.var:court_owner = { save_scope_as = heist_imprisoner }
	}
	
	option = {
		name = POD_heists.101.a

		scope:heist_imprisoner = {
			progress_towards_rival_effect = {
				CHARACTER = root
				REASON    = rival_infiltration
				OPINION   = default_rival_opinion
			}
		}

		if = {
			limit = { POD_has_coterie_members_trigger = yes }
			var:coterie = {
				every_in_list = {
					variable = members
					limit = {
						POD_is_in_same_journey_trigger = { TARGET = root }
					}
					rightfully_imprison_character_effect = {
						TARGET = this
						IMPRISONER = scope:heist_imprisoner
					}
				}
			}
		}
		else = {
			rightfully_imprison_character_effect = {
				TARGET = root
				IMPRISONER = scope:heist_imprisoner
			}
		}

		POD_heist_cleanup_effect = yes
		hidden_effect = { POD_end_journey_effect = yes }

	}
}
