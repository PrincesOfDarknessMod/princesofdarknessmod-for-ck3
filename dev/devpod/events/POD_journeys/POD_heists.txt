# Arrival via artifact_heist Journey
POD_heists.1 = {
	type = character_event
	content_source = dlc_POD
	title = POD_heist_arrival_title
	theme = heist
	override_background = { reference = terrain_travel }
	override_sound = { reference = "event:/DLC/EP2/SFX/Events/Travel/ep2_evn_travel_end_neutral" }
	
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { POD_has_coterie_members_present_trigger = yes }
				desc = POD_heists.1.desc.coterie
			}
			desc = POD_heists.1.desc.nocoterie
		}
	}
	
	left_portrait = {
		character = root
		animation = personality_bold 
	}

	artifact = {
		target = var:current_journey.var:target_artifact
		position = lower_center_portrait
	}

	immediate = {
		hidden_effect = {
			current_travel_plan ?= {
				pause_travel_plan = yes
			}
		}
	}
	
	option = {
		name = POD_heist_arrival_confirm

		play_music_cue = "mood_twilights_approach"

		POD_heist_setup_and_select_map = {
			COURT   = var:current_journey.var:target_artifact.artifact_owner
			ENDING  = POD_heists.100
			SAVE_AS = heist
		}
		POD_heist_initialize_artifact_heist = {
			HEIST    = scope:heist
			ARTIFACT = var:current_journey.var:target_artifact
		}
		POD_heist_select_suspicion_events = {
			HEIST      = scope:heist
			ON_FAILURE = POD_heists.101
		}
	}
	
	option = {
		name = POD_heist_arrival_decline
		POD_end_journey_can_retry_effect = yes
	}
}

# Successful escape during an artifact_heist Journey
POD_heists.100 = {
	type = character_event
	content_source = dlc_POD
	title = POD_heists.100.t
	theme = heist
	override_background = { reference = terrain_travel }
	
	override_sound = {
		trigger = { POD_heists_100_is_successful = yes }
		reference = "event:/DLC/EP1/SFX/UI/Notifications/inspiration_project_complete"
	}
	override_sound = { reference = "event:/DLC/EP2/SFX/Events/Travel/ep2_evn_travel_end_neutral" }
	
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { POD_has_coterie_members_present_trigger = yes }
				desc = POD_heists.100.desc.coterie
			}
			desc = POD_heists.100.desc.nocoterie
		}
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:success_artifact }
				desc = POD_heists.100.desc.artifact
			}
			triggered_desc = {
				trigger = {
					var:current_heist = { has_variable = target_artifact }
					NOT = { exists = scope:success_artifact }
				}
				desc = POD_heists.100.desc.noartifact
			}
			triggered_desc = {
				trigger = { exists = scope:success_gold }
				desc = POD_heists.100.desc.gold
			}
			desc = POD_heists.100.desc.nogold
		}
	}
	
	left_portrait = {
		character = root
		animation = personality_bold 
	}

	artifact = {
		target = scope:displayed_artifact
		position = lower_center_portrait
		trigger = { exists = scope:displayed_artifact }
	}

	immediate = {
		var:current_heist ?= {
			random_in_list = {
				variable = loot_artifacts
				limit = { prev.var:target_artifact ?= this }
				save_scope_as = displayed_artifact
				save_scope_value_as = { name = success_artifact value = yes }
			}
		}
		POD_close_heist = yes
		POD_heist_gain_loot_effect = yes
		POD_heist_display_expenditures = yes

		if = {
			limit = {
				var:current_heist.var:stolen_gold > 0
			}
			save_scope_value_as = { name = success_gold value = yes }
		}
	}
	
	option = {
		name = {
			trigger = {
				POD_heists_100_is_successful = yes
				POD_has_coterie_members_present_trigger = yes
			}
			text = POD_heists.100.a.success.coterie
		}
		name = {
			trigger = {
				POD_heists_100_is_successful = yes
				POD_has_coterie_members_present_trigger = no
			}
			text = POD_heists.100.a.success.nocoterie
		}
		name = {
			trigger = {
				POD_heists_100_is_successful = no
			}
			text = POD_heists.100.a.failure
		}
		POD_heist_cleanup_effect = yes
		POD_end_journey_effect = yes
	}
}


# Got caught during an artifact_heist Journey
POD_heists.101 = {
	type = character_event
	content_source = dlc_POD
	title = POD_heists.101.t
	desc  = POD_heists.101.desc
	theme = heist_suspicion
	override_sound = { reference = "event:/SFX/Events/Themes/sfx_event_theme_type_health_bad" }
	
	left_portrait = {
		character = root
		animation = prisondungeon
	}

	right_portrait = {
		character = scope:heist_imprisoner
		animation = war_defender
	}

	widget = {
		gui = "POD_event_window_widget_heist_suspicion_threshold_3"
		container = "custom_widgets_container"
	}

	immediate = {
		POD_heist_suspicion_threshold_4_tooltip = yes
		POD_close_heist = yes
		var:current_heist.var:court_owner = { save_scope_as = heist_imprisoner }

		POD_heist_give_stolen_gold_back = yes

		add_character_modifier = {
			modifier = heist_caught_modifier
			years = 10
		}
	}
	
	option = {
		name = POD_heists.101.a

		scope:heist_imprisoner = {
			progress_towards_rival_effect = {
				CHARACTER = root
				REASON    = rival_infiltration
				OPINION   = default_rival_opinion
			}
		}

		if = {
			limit = { POD_has_coterie_members_trigger = yes }
			var:coterie = {
				every_in_list = {
					variable = members
					limit = {
						POD_is_in_same_journey_trigger = { TARGET = root }
					}
					rightfully_imprison_character_less_verbose_effect = {
						TARGET = this
						IMPRISONER = scope:heist_imprisoner
					}
				}
			}
		}
		else = {
			rightfully_imprison_character_less_verbose_effect = {
				TARGET = root
				IMPRISONER = scope:heist_imprisoner
			}
		}

		POD_heist_cleanup_effect = yes
		hidden_effect = { POD_end_journey_effect = yes }
	}
}

POD_heists.1001 = {
	type = character_event
	content_source = dlc_POD
	title = POD_heists.1001.t
	theme = heist_suspicion
	override_background = { reference = throne_room_scope }

	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { POD_has_coterie_members_present_trigger = yes }
				desc = POD_heists.1001.desc.coterie
			}
			desc = POD_heists.1001.desc.nocoterie
		}
		desc = POD_heists.1001.desc.end
	}
	
	left_portrait = {
		character = root
		animation = boredom
	}
	
	right_portrait = {
		character = scope:diplomacy_courtier
		animation = thinking
	}

	widget = {
		gui = "POD_event_window_widget_heist_suspicion_threshold_1"
		container = "custom_widgets_container"
	}

	immediate = {
		POD_heist_suspicion_threshold_1_tooltip = yes

		POD_save_coterie_duelists = yes

		var:current_heist.var:court_owner = {
			save_scope_as = background_throne_room_scope # for event background

			ordered_courtier_or_guest = {
				limit = {
					trigger_if = {
						limit = { exists = prev.cp:councillor_chancellor }
						NOT = { this = prev.cp:councillor_chancellor }
					}
				}
				order_by = diplomacy
				save_scope_as = diplomacy_courtier
			}
			if = {
				limit = { NOT = { exists = scope:diplomacy_courtier } }
				save_scope_as = diplomacy_courtier
			}

			if = {
				limit = { exists = capital_province }
				capital_province = { save_scope_as = flavortext_province }
			}
			else = {
				location = { save_scope_as = flavortext_province }
			}
			scope:flavortext_province = {
				county_controller = {
					random_neighboring_and_across_water_top_liege_realm = {
						save_scope_as = flavortext_realm
					}
				}
			}
		}

	}
	
	option = {
		name = POD_heists.1001.a
		scope:diplomacy_duelist = {
			duel = {
				skill = diplomacy
				value = heist_suspicion_event_1_difficulty
				15 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = 0.5
					}
					desc = POD_heists.1001.a.success
				}
				15 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = -0.5
					}
					desc = POD_heists.1001.a.failure
					root = { POD_heist_raise_suspicion_effect = { VALUE = minor_suspicion_value } }
				}
			}
		}
	}
	
	option = {
		name = {
			trigger = { POD_has_coterie_members_present_trigger = yes }
			text = POD_heists.1001.b.coterie
		}
		name = {
			trigger = { POD_has_coterie_members_present_trigger = no }
			text = POD_heists.1001.b.nocoterie
		}
		add_character_modifier = {
			modifier = heist_diploroute_1_modifier
			years = heist_negative_modifier_duration
		}
	}

	after = {
		POD_heist_resume_after_event = yes
	}
}
