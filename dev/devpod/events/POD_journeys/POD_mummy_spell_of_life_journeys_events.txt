POD_mummy_spell_of_life_journeys.12 = {
	type = character_event
	content_source = dlc_POD
	theme = education

	title = POD_mummy_spell_of_life_journeys.12.t
	desc = POD_mummy_spell_of_life_journeys.12.desc
	
	left_portrait = {
		character = root
		animation = chancellor
	}
	right_portrait = {
		character = character:792858
		animation = holding_scrolls
	}

	immediate = {
		POD_save_coterie_duelists = yes
	}

	option = {
		name = POD_mummy_spell_of_life_journeys.12.a
		trigger = {
			any_character_artifact = {
				count >= 3
				OR = {
					rarity = illustrious
					rarity = famed
					rarity = masterwork
				}
				NOT = { has_variable = ungiftable }
			}
		}
		ordered_character_artifact = {
			limit = {
				OR = {
					rarity = illustrious
					rarity = famed
					rarity = masterwork
				}
				NOT = { has_variable = ungiftable }
			}
			max = 3
			order_by = artifact_rarity 
			
			set_owner = {
				target = character:792858
				history = {
					type = given
					actor = root
					recipient = character:792858
					location = root.location
				}
			}
		}
		POD_add_spell_of_life_xp = {
			VALUE = 10
		}
	}
	option = {
		name = POD_mummy_spell_of_life_journeys.12.b
		remove_short_term_gold = 1000
		POD_add_spell_of_life_xp = {
			VALUE = 10
		}
	}
	option = {
		name = POD_mummy_spell_of_life_journeys.12.c
		custom_tooltip = spell_of_life_coterie_prowess_duel
		scope:prowess_duelist = {
			duel = {
				skill = prowess
				value = 60
				10 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = 0.5
					}

					desc = POD_mummy_spell_of_life_journeys.12.effect.success.desc

					root = {
						send_interface_toast = {
							title = POD_mummy_spell_of_life_journeys.12.effect.success.desc
							left_icon = root
							add_gold = character:792858.gold
						}
						character:792858 = {
							remove_short_term_gold = character:792858.gold
							every_character_artifact = {
								set_owner = {
									target = root
									history = {
										type = taken_in_battle
										actor = character:792858
										recipient = root
										location = character:792858.location
									}
								}
							}
						}
					}
				}
				10 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = -0.5
					}

					desc = POD_mummy_spell_of_life_journeys.12.critical.failure.effect.desc

					root = {
						send_interface_toast = {
							title = POD_mummy_spell_of_life_journeys.12.critical.failure.effect.desc
							left_icon = root
						}
					}
				}

	}
