namespace = POD_exalted_wyld_hunt

POD_exalted_wyld_hunt.1 = {
	hidden = yes
	immediate = {
		if = {
			limit = { var:current_journey ?= { has_variable = wyld_hunt_loud } }
			trigger_event = POD_exalted_wyld_hunt.1000
		}
		else = {
			trigger_event = POD_exalted_wyld_hunt.2000
		}
	}
}

POD_exalted_wyld_hunt.1000 = {
	type = character_event
	content_source = dlc_POD
	title = POD_exalted_wyld_hunt.1000.t
	desc = POD_exalted_wyld_hunt.1000.desc
	
	theme = witchcraft
	left_portrait = {
		character = root
		animation = anger
	}

	immediate = {
		var:current_journey.var:target = { save_scope_as = wyld_target }
	}

	option = {
		name = POD_exalted_wyld_hunt.1000.a
		custom_tooltip = POD_exalted_wyld_hunt.1000.a.effect
		POD_combat_create_duel_effect = {
			INITIATOR    = ROOT
			ATTACKER     = ROOT
			DEFENDER     = scope:wyld_target
			FATALITY     = possible
			FIXED        = no
			LOCALE       = alley_night
			OUTPUT_EVENT = POD_exalted_wyld_hunt.1001
		}
		POD_combat_duel_add_enemy_effect = { CHARACTER = scope:wyld_target }
		every_relation = {
			type = coterie_member
			limit = { POD_is_in_same_journey_trigger = { TARGET = ROOT } }
			save_scope_as = duel_ally
			ROOT = { POD_combat_duel_add_ally_effect = { CHARACTER = scope:duel_ally } }
		}
		scope:wyld_target ?= {
			save_scope_as = target
			ordered_knight = {
				order_by = { value = prowess }
				limit = {
					is_alive = yes
					POD_is_available = yes
				}
				max = 1
				save_scope_as = duel_enemy
			}
			ordered_knight = {
				order_by = { value = prowess }
				limit = {
					is_alive = yes
					POD_is_available = yes
					THIS != scope:duel_enemy
				}
				max = 1
				save_scope_as = duel_enemy2
			}
		}
		POD_combat_duel_add_enemy_effect = { CHARACTER = scope:duel_enemy }
		POD_combat_duel_add_enemy_effect = { CHARACTER = scope:duel_enemy2 }
		POD_combat_finalize_duel_effect = yes
		set_variable = {
			name = prestige_gained
			value = scope:wyld_target.POD_wyld_hunt_prestige_value
		}
	}
}

POD_exalted_wyld_hunt.2000 = {
	type = character_event
	content_source = dlc_POD
	title = POD_exalted_wyld_hunt.2000.t
	theme = heist
	override_background = { reference = terrain_travel }
	override_sound = { reference = "event:/DLC/EP2/SFX/Events/Travel/ep2_evn_travel_end_neutral" }
	
	desc = POD_exalted_wyld_hunt.2000.desc
	
	left_portrait = {
		character = root
		animation = assassin 
	}

	immediate = {
		hidden_effect = {
			POD_pause_journey_travel_plan_effect = yes
			var:current_journey.var:target = { save_scope_as = target_court }
		}
	}
	
	option = {
		name = POD_exalted_wyld_hunt.2000.a

		play_music_cue = "mood_twilights_approach"

		POD_heist_setup_and_select_map = {
			COURT   = var:current_journey.var:target
			ENDING  = POD_heists.100 # A small tweak inside it
			SAVE_AS = heist
		}
		POD_heist_select_suspicion_events = {
			HEIST         = scope:heist
			ON_FAILURE    = POD_exalted_wyld_hunt.2001
			FORCE_HOSTILE = yes
		}
		POD_heist_initialize_wyld_hunt = {
			HEIST  = scope:heist
			TARGET = var:current_journey.var:target
		}
		set_variable = {
			name = prestige_gained
			value = scope:target_court.POD_wyld_hunt_prestige_value
		}
	}
	
	option = {
		name = POD_heist_arrival_decline
		scope:target_court = {
			add_character_modifier = {
				modifier = wary_of_infiltrators_modifier
				years = 20
			}
		}
		POD_end_journey_effect = yes
	}
}

POD_exalted_wyld_hunt.2001 = {
	type = character_event
	content_source = dlc_POD
	title = POD_exalted_wyld_hunt.2001.t
	desc  = POD_exalted_wyld_hunt.2001.desc
	theme = heist_suspicion
	override_sound = { reference = "event:/SFX/Events/Themes/sfx_event_theme_type_health_bad" }
	
	left_portrait = {
		character = ROOT
		animation = shock
	}

	widget = {
		gui = "POD_event_window_widget_heist_suspicion_threshold_3"
		container = "custom_widgets_container"
	}

	immediate = {
		POD_setup_heist_event_background = yes
		POD_heist_suspicion_threshold_4_tooltip = yes
		POD_close_heist = yes
		var:current_heist.var:court_owner = {
			save_scope_as = heist_imprisoner
			add_character_modifier = {
				modifier = wary_of_infiltrators_modifier
				years = 20
			}
		}
		POD_save_coterie_duelists = yes
		remove_variable = prestige_gained
	}
	
	option = {
		name = POD_exalted_wyld_hunt.2001.a

		add_prestige = -4000

		scope:prowess_duelist ?= {
			duel = {
				skill = prowess
				value = 60
				1 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = 0.5
					}
					desc = POD_exalted_wyld_hunt.2001.a.success
				}
				3 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = -0.5
					}
					desc = POD_exalted_wyld_hunt.2001.a.failure
					ROOT = {
						if = {
							limit = { POD_has_coterie_members_trigger = yes }
							var:coterie = {
								every_in_list = {
									variable = members
									limit = {
										POD_is_in_same_journey_trigger = { TARGET = root }
									}
									pod_death = {
										death_reason = death_execution
										killer = scope:heist_imprisoner
									}
								}
							}
						}
						else = {
							pod_death = {
								death_reason = death_execution
								killer = scope:heist_imprisoner
							}
						}
					}
				}
			}
		}

		POD_heist_cleanup_effect = yes
		hidden_effect = { POD_end_journey_effect = yes }
	}
}
