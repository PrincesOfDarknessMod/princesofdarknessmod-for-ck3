POD_learn_true_name.2 = {
	content_source = dlc_POD
	type = character_event

	title = POD_learn_true_name.2.t
	desc = POD_learn_true_name.2.desc
	theme = spires

	left_portrait = {
		character = root
	}

	immediate = {
		hidden_effect = { POD_learn_random_true_name_of_nonstandard_effect = yes }
		set_variable = { name = true_name_variable value = scope:true_name_scope }
	}

	option = {
		name = POD_learn_true_name.2.a
		custom_tooltip = POD_LEARN_TRUE_NAME_OF_NONSTANDARD_EFFECT
		trigger = { piety >= 250 }
		show_as_unavailable = { piety < 250 }
		
		custom_tooltip = {
			text = POD_learn_true_name.2.a.tt
			POD_journey_cleanup_effect = yes
			remove_variable = true_name_variable
			add_piety = -250
			POD_find_and_start_journey_effect = {
				TYPE = study_true_name_journey
			}
		}
	}

	option = {
		name = POD_learn_true_name.2.b
		custom_tooltip = POD_LEARN_TRUE_NAME_OF_NONSTANDARD_EFFECT
		custom_tooltip = {
			text = POD_end_journey_tt
			POD_journey_cleanup_effect = yes
		}
		remove_variable = true_name_variable
	}
}

POD_learn_true_name.3 = {
	content_source = dlc_POD
	type = character_event

	title = POD_learn_true_name.3.t
	desc = POD_learn_true_name.3.desc
	theme = spires

	left_portrait = {
		character = scope:actor
		animation = personality_rational
	}

	right_portrait = {
		character = scope:recipient
		trigger = { scope:actor != scope:recipient }
	}
	
	widget = {
		gui = "POD_event_window_widget_learn_true_names"
		container = "custom_widgets_container"
	}

	immediate = {
		scope:recipient = {
			save_scope_as = cant_select_multiple_traits # jank to make the event widget work
			every_character_trait = {
				limit = {
					POD_can_study_true_name_of_trait_trigger = { SCHOLAR = scope:actor }
				}
				scope:actor = { add_to_variable_list = { name = potential_traits target = PREV } }
			}
		}
	}
	
	option = {
		name = POD_learn_true_name.3.a

		scope:actor = {
			random_in_list = {
				variable = selected_traits
				PREV = { 
					POD_start_trait_journey_effect = {
						TRAIT   = PREV
						TOOLTIP = study_true_name_trait_journey_tt
						TYPE    = study_true_name_trait_journey
						MONTHLY = study_true_name_journey
					}
				}
			}
		}
	}
	
	option = {
		name = POD_learn_true_name.3.b
	}

	after = {
		scope:actor = {
			clear_variable_list = potential_traits
			clear_variable_list = selected_traits
		}
	}
}
