# Azhi Dahaka Learning
POD_azhi_dahaka_journey.1 = {
	type = character_event
	content_source = dlc_POD
	title = POD_azhi_dahaka_journey.1.t
	desc = POD_azhi_dahaka_journey.1.desc
	theme = vampire
	override_background = { reference = sitting_room }
	override_sound = { reference = "event:/DLC/FP2/SFX/UI/fp2_struggle_ui_intro_animate" }

	trigger = { POD_is_vampire_trigger = yes }
	on_trigger_fail = {
		POD_abandon_journeys_of_type_effect = { TYPE = azhi_dahaka_1 }
	}

	left_portrait = {
		character = root
		animation = personality_zealous
	}

	widget = {
		gui = "POD_event_window_widget_azhi_dahaka"
		container = "custom_widgets_container"
	}

	immediate = {

		#Randomize challenge
		random_list = {
			25 = { save_scope_value_as = { name = skill value = flag:diplomacy } }
			25 = { save_scope_value_as = { name = skill value = flag:martial } }
			25 = { save_scope_value_as = { name = skill value = flag:stewardship } }
			25 = { save_scope_value_as = { name = skill value = flag:intrigue } }
			#No learning, it already determines speed and rewards
		}
		#Get difficulty
		save_scope_value_as = {
			name = difficulty
			value = {
				value = sum_of_all_skills_value
				add = { -7 0 }
			}
		}
	}

	#Multiple options but they're all essentially the same
	option = {
		name = POD_azhi_dahaka_journey.1.a
		trigger = { scope:skill = flag:diplomacy }

		duel = {
			skills = { diplomacy }
			value = scope:difficulty
		
			40 = {
				compare_modifier = { value = scope:duel_value multiplier = high_positive_duel_skill_multiplier }

				add_trait_xp = { trait = azhi_dahaka_scholar value = POD_azhi_dahaka_scholar_boosted_xp }
			}
			
			60 = {
				compare_modifier = { value = scope:duel_value multiplier = low_negative_duel_skill_multiplier }

				add_trait_xp = { trait = azhi_dahaka_scholar value = POD_azhi_dahaka_scholar_base_xp }
				add_stress = medium_stress_gain 
			}
		}
	}

	option = {
		name = POD_azhi_dahaka_journey.1.a
		trigger = { scope:skill = flag:martial }

		duel = {
			skills = { martial }
			value = scope:difficulty
		
			40 = {
				compare_modifier = { value = scope:duel_value multiplier = high_positive_duel_skill_multiplier }

				add_trait_xp = { trait = azhi_dahaka_scholar value = POD_azhi_dahaka_scholar_boosted_xp }
			}
			
			60 = {
				compare_modifier = { value = scope:duel_value multiplier = low_negative_duel_skill_multiplier }

				add_trait_xp = { trait = azhi_dahaka_scholar value = POD_azhi_dahaka_scholar_base_xp }
				add_stress = medium_stress_gain 
			}
		}
	}

	option = {
		name = POD_azhi_dahaka_journey.1.a
		trigger = { scope:skill = flag:stewardship }

		duel = {
			skills = { stewardship }
			value = scope:difficulty
		
			40 = {
				compare_modifier = { value = scope:duel_value multiplier = high_positive_duel_skill_multiplier }

				add_trait_xp = { trait = azhi_dahaka_scholar value = POD_azhi_dahaka_scholar_boosted_xp }
			}
			
			60 = {
				compare_modifier = { value = scope:duel_value multiplier = low_negative_duel_skill_multiplier }

				add_trait_xp = { trait = azhi_dahaka_scholar value = POD_azhi_dahaka_scholar_base_xp }
				add_stress = medium_stress_gain 
			}
		}
	}

	option = {
		name = POD_azhi_dahaka_journey.1.a
		trigger = { scope:skill = flag:intrigue }

		duel = {
			skills = { intrigue }
			value = scope:difficulty
		
			40 = {
				compare_modifier = { value = scope:duel_value multiplier = high_positive_duel_skill_multiplier }

				add_trait_xp = { trait = azhi_dahaka_scholar value = POD_azhi_dahaka_scholar_boosted_xp }
			}
			
			60 = {
				compare_modifier = { value = scope:duel_value multiplier = low_negative_duel_skill_multiplier }

				add_trait_xp = { trait = azhi_dahaka_scholar value = POD_azhi_dahaka_scholar_base_xp }
				add_stress = medium_stress_gain 
			}
		}
	}

	after = {
		POD_end_journey_effect = yes
		if = {
			limit = { has_trait_xp = { trait = azhi_dahaka_scholar value >= 100 } }
			remove_variable = continue_azhi_dahaka
			trigger_event = POD_azhi_dahaka_journey.2
		}
		else_if = {
			limit = { has_variable = continue_azhi_dahaka }
			if = {
				limit = { piety >= 1000 }
				custom_tooltip = {
					text = azhi_dahaka_widget_tt
					POD_find_and_start_journey_effect = { TYPE = azhi_dahaka_1 }
				}
				add_piety = -1000
			}
			else = { custom_tooltip = azhi_dahaka_not_enough_piety }
		}
	}
}

POD_azhi_dahaka_journey.3 = {
	type = character_event
	content_source = dlc_POD
	title = POD_azhi_dahaka_journey.3.t
	desc = POD_azhi_dahaka_journey.3.desc
	theme = vampire
	override_background = { reference = shader_event_blood }
	override_sound = { reference = "event:/DLC/FP2/SFX/UI/fp2_struggle_ui_intro_animate" }

	trigger = { POD_is_vampire_trigger = yes }
	on_trigger_fail = {
		POD_abandon_journeys_of_type_effect = { TYPE = azhi_dahaka_2 }
	}

	
	widget = {
		gui = "POD_shader_event_blood"
		container = "background_shader_vfx_container"
	}

	left_portrait = {
		character = root
		animation = personality_zealous
		triggered_outfit = {
			trigger = { always = yes }
			outfit_tags = { no_clothes }
		}
	}

	immediate = {
		hidden_effect = {
			POD_apply_temporary_portrait_flag = { FLAG = POD_ironskin_form_flag }
			POD_apply_temporary_portrait_flag = { FLAG = POD_azhi_dahaka_form_flag }
		}
	}

	option = {
		name = POD_azhi_dahaka_journey.3.a
		remove_trait = azhi_dahaka_scholar
		add_trait_force_tooltip = azhi_dahaka
		#To make the stats randomization more fun!
		add_diplomacy_skill = 1
		add_martial_skill = 1
		add_stewardship_skill = 1
		add_intrigue_skill = 1
		add_learning_skill = 1

		POD_end_journey_effect = yes

		hidden_effect = {
			POD_remove_temporary_portrait_flag = { FLAG = POD_ironskin_form_flag }
			POD_remove_temporary_portrait_flag = { FLAG = POD_azhi_dahaka_form_flag } #you still keep the look after this event
		}
	}
}

#Yearly stats randomization
POD_azhi_dahaka_journey.4 = {
	type = character_event
	hidden = yes

	trigger = { has_trait = azhi_dahaka }

	immediate = {
		POD_get_base_diplomacy_effect = yes
		POD_get_base_martial_effect = yes
		POD_get_base_stewardship_effect = yes
		POD_get_base_intrigue_effect = yes
		POD_get_base_learning_effect = yes


		save_scope_value_as = {
			name = total_base_skills
			value = {
				value = scope:base_diplomacy
				add = scope:base_martial
				add = scope:base_stewardship
				add = scope:base_intrigue
				add = scope:base_learning
			}
		}

		add_diplomacy_skill = -1000
		add_martial_skill = -1000
		add_stewardship_skill = -1000
		add_intrigue_skill = -1000
		add_learning_skill = -1000
	
		while = {
			count = scope:total_base_skills
			POD_add_random_skill_point_no_prowess_effect = yes
		}
	}
}
