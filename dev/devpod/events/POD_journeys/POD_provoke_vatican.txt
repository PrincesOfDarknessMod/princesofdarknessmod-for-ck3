scripted_effect POD_vatican_massacre_effect = {
	save_temporary_scope_as = owner
	
	stress_impact = {
		gluttonous = massive_stress_impact_loss
		compassionate = massive_stress_impact_gain
	}
	hidden_effect_new_object = {
		while = {
			count = 27 # DCSS
			
			create_character = {
				template = $TEMPLATE$
				location = title:b_roma.title_province
				culture = title:b_roma.title_province.culture
				faith = faith:innocence
				save_scope_as = victim
			}
			
			scope:victim = {
				add_trait = mortal
				POD_set_resonance_effect = yes
			}
			
			POD_drain_victim_effect = { VICTIM = scope:victim }
		}
	}
}

scripted_effect POD_vatican_hunter_kill_effect = {
	set_variable = {
		name = vatican_hunters_killed
		value = { value = var:vatican_hunters_killed add = 1 }
		days = 30
	}
	
	if = {
		limit = { var:vatican_hunters_killed ?= 3 }
		
		custom_tooltip = POD_provoke_vatican_escape_complete_tt
		
		remove_variable = vatican_hunters_killed
		remove_variable = vatican_times_debuffed

		add_character_flag = spawn_the_lamb
		
		if = {
			limit = { POD_has_coterie_trigger = yes }
			var:coterie = {
				every_in_list = {
					variable = members
					if = {
						limit = { has_character_modifier = vatican_hunter_tf_modifier }
						remove_all_character_modifier_instances = vatican_hunter_tf_modifier
					}
				}
			}
		}
		else = {
			remove_all_character_modifier_instances = vatican_hunter_tf_modifier
		}
		
		if = {
			limit = { exists = current_travel_plan }
			current_travel_plan = { resume_travel_plan = yes }
		}
		else = {
			POD_clear_topscope_and_trigger_event = {
				ID = POD_provoke_vatican.30
				DAYS = 14
			}
		}
	}
	else = {
		custom_tooltip = POD_provoke_vatican_escape_tt
		POD_clear_topscope_and_trigger_event = {
			ID = POD_provoke_vatican.20
			DAYS = 7
		}
	}
}

# The ruler of Rome bars your way
POD_provoke_vatican.2 = {
	type = character_event
	content_source = dlc_POD
	title = POD_provoke_vatican.2.t
	theme = vampire
	override_background = { reference = alley_night }
	
	desc = {
		first_valid = {
			# Different lines if someone already died
			triggered_desc = {
				trigger = { exists = scope:coterie_loser }
				desc = POD_provoke_vatican.2.desc.repeat
			}
			# Default lines
			desc = {
				desc = POD_provoke_vatican.2.desc
				first_valid = {
					triggered_desc = {
						trigger = { scope:adversary = { POD_is_vampire_trigger = yes } }
						desc = POD_provoke_vatican.2.desc.vampire
					}
					desc = POD_provoke_vatican.2.desc.nonvampire
				}
				first_valid = {
					triggered_desc = {
						trigger = { exists = scope:selected_neighbor }
						desc = POD_provoke_vatican.2.desc.neighbor
					}
					desc = POD_provoke_vatican.2.desc.ruler
				}
			}
		}
	}
	
	widget = {
		gui = "POD_shader_event_fog_alley_night"
		container = "background_shader_vfx_container"
	}
	
	left_portrait = {
		character = root
		triggered_animation = {
			trigger = { NOT = { exists = scope:coterie_loser } }
			animation = war_attacker
		}
		# Different animation if a coterie member has died
		triggered_animation = {
			trigger = { exists = scope:coterie_loser }
			animation = fear
		}
	}
	
	right_portrait = {
		character = scope:adversary
		triggered_animation = {
			trigger = { NOT = { exists = scope:coterie_loser } }
			animation = war_defender
		}
		# Different animation if a coterie member has died
		triggered_animation = {
			trigger = { exists = scope:coterie_loser }
			animation = marshal
		}
	}
	
	# If Coterie member died in a previous iteration, show them
	lower_center_portrait = {
		trigger = { exists = scope:coterie_loser }
		character = scope:coterie_loser
	}
	
	immediate = {
		hidden_effect = {
			POD_save_coterie_duelists_except_root = yes
			# Select an adversary
			if = {
				limit = { NOT = { exists = scope:adversary } }
				# Select ruler of Rome if it's not the player
				if = {
					limit = {
						title:c_roma.county_controller = {
							NOT = { this = root }
							POD_is_eligible_rome_adversary_trigger = yes
						}
					}
					title:c_roma.county_controller = {
						save_scope_as = adversary
					}
				}
				# Otherwise, select the neighboring ruler with the highest prowess
				else = {
					ordered_neighboring_and_across_water_top_liege_realm_owner = {
						limit = { POD_is_eligible_rome_adversary_trigger = yes }
						order_by = prowess
						position = 0
						save_scope_as = adversary
					}
					save_scope_value_as = {
						name  = selected_neighbor
						value = yes
					}
				}
			}
		}
	}
	
	# Duel the ruler of Rome yourself (Single Combat)
	option = {
		name = {
			trigger = { NOT = { exists = scope:coterie_loser } }
			text = POD_provoke_vatican.2.option.a
		}
		# Different line if a coterie member has already died
		name = {
			trigger = { exists = scope:coterie_loser }
			text = POD_provoke_vatican.2.option.a.2
		}
		custom_tooltip = POD_provoke_vatican.2.option.a.tt
		
		configure_start_single_combat_effect = {
			SC_INITIATOR = root
			SC_ATTACKER = root
			SC_DEFENDER = scope:adversary
			FATALITY = no
			FIXED = no
			LOCALE = alley_night
			OUTPUT_EVENT = POD_provoke_vatican.100
			INVALIDATION_EVENT = fp1_tbc.0102
		}
	}
	
	# Let a coterie member fight them off
	option = {
		trigger = { NOT = { scope:prowess_duelist = root } }
		name = POD_provoke_vatican.2.option.b
		scope:prowess_duelist = {
			duel = {
				skill = prowess
				target = scope:adversary
				# Success: Coterie member wins
				30 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = 0.5
					}
					custom_tooltip = POD_vatican_coterie_duel_success_tt
					save_scope_as = coterie_victor
					root = {
						trigger_event = {
							id = POD_provoke_vatican.3
							days = 1
						}
					}
				}
				# Failure: Coterie member and adversary both die
				20 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = -0.25
					}
					custom_tooltip = {
						text = POD_vatican_coterie_duel_failure_tt
						save_scope_as = coterie_loser
						root = {
							trigger_event = {
								id = POD_provoke_vatican.5
								days = 1
							}
						}
					}
				}
				# Critical Failure: Coterie member dies so fast, they don't even buy time
				10 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = -0.25
					}
					custom_tooltip = {
						text = POD_vatican_coterie_duel_critfail_tt
						save_scope_as = coterie_loser
						pod_death = {
							DEATH_REASON = death_murder
							KILLER = scope:adversary
						}
						root = {
							trigger_event = {
								id = POD_provoke_vatican.2
								days = 1
							}
						}
					}
				}
			}
		}
	}
	
	# Just leave
	option = {
		name = {
			trigger = { NOT = { exists = scope:coterie_loser } }
			text = POD_provoke_vatican.2.option.c
		}
		# Different line if a coterie member has already died
		name = {
			trigger = { exists = scope:coterie_loser }
			text = POD_provoke_vatican.2.option.c.2
		}
		custom_tooltip = POD_provoke_vatican.2.option.c.tt
		POD_end_journey_effect = yes
	}
}

# Outcome of duel #1: Victory (Player or Coterie member)
POD_provoke_vatican.3 = {
	type = character_event
	content_source = dlc_POD
	title = POD_provoke_vatican.3.t
	theme = vampire
	override_background = { reference = alley_night }
	
	desc = {
		desc = POD_provoke_vatican.3.desc
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:root_won_duel_1 }
				desc = POD_provoke_vatican.3.desc.root
			}
			desc = POD_provoke_vatican.3.desc.coterie
		}
		desc = POD_provoke_vatican.3.desc.end
	}
	
	widget = {
		gui = "POD_shader_event_fog_alley_night"
		container = "background_shader_vfx_container"
	}
	
	left_portrait = {
		character = scope:coterie_victor
		animation = sword_coup_degrace
	}
	
	right_portrait = {
		character = scope:adversary
		animation = severelywounded
	}
	
	immediate = {
		if = {
			limit = { NOT = { exists = scope:coterie_victor } }
			save_scope_as = coterie_victor
			save_scope_value_as = {
				name  = root_won_duel_1
				value = yes
			}
		}
		hidden_effect = {
			scope:adversary = {
				increase_wounds_no_death_effect = { REASON = fight }
				increase_wounds_no_death_effect = { REASON = fight }
			}
		}
	}
	
	# Kill them
	option = {
		name = POD_provoke_vatican.3.option.a
		scope:adversary = {
			pod_death = {
				DEATH_REASON = death_murder
				KILLER = root
			}
		}
	}
	
	# Leave them alive
	option = {
		name = POD_provoke_vatican.3.option.b
		if = {
			limit = {
				NOT = { has_relation_rival = scope:adversary }
				NOT = { has_relation_nemesis = scope:adversary }
			}
			set_relation_rival = scope:adversary
		}
		else = {
			set_relation_nemesis = scope:adversary
		}
	}
	
	after = {
		custom_tooltip = POD_provoke_vatican.1.option.a.tt
		if = {
			limit = { root = character:790839 } # Caelestius branch
			POD_clear_topscope_and_trigger_event = {
				ID = POD_provoke_vatican.12
				DAYS = 30
			}
		}
		else = {
			POD_clear_topscope_and_trigger_event = {
				ID = POD_provoke_vatican.10
				DAYS = 7
			}
		}
	}
}
