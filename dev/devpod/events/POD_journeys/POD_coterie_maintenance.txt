# Name your coterie
POD_coterie_maintenance.2 = {
	type = character_event
	content_source = dlc_POD
    title = POD_coterie_maintenance.2.t
	desc  = POD_coterie_maintenance.2.desc
	theme = coterie
	
	left_portrait = {
	    character = root
		#animation = personality_bold 
	}
	
	widget = {
		gui = "POD_event_window_widget_enter_nickname"
		container = "dynamic_birth_name"
		controller = {
			type = text
			data = {
				key = POD_coterie_name_text
				default = {
					#localization_key = POD_coteriename_default
				}
			}
		}
		setup_scope = {
			var:coterie.var:root_char = { save_scope_as = text_target }
		}
	}
	
	immediate = {
		hidden_effect = {
			add_character_flag = coterie_creation_in_progress
			var:coterie = {
				if = {
					limit = { has_variable = custom_name }
					
					remove_variable = custom_name
					
					var:root_char = {
						remove_localized_text = POD_coterie_name_text
						# remove_variable = POD_coterie_name_text
					}
				}
				if = {
					limit = { var:root_char = { is_alive = no } }
					set_variable = {
						name = root_char
						value = root
					}
				}
			}
		}
	}
	
	option = {
	    name = POD_coterie_maintenance.2.option.a
		hidden_effect = {
			remove_character_flag = coterie_creation_in_progress
			var:coterie = {
				set_variable = custom_name
				
				var:root_char = {
					# set_variable = {
						# name = POD_coterie_name_text
						# value = flag:custom
					# }
					store_localized_text_in_death = POD_coterie_name_text
				}
			}
		}
	}
	
	option = {
	    name = POD_coterie_maintenance.2.option.b
		hidden_effect = {
			remove_character_flag = coterie_creation_in_progress
		}
	}
}

# Your predecessor had a Coterie
POD_coterie_maintenance.20 = {
	type = character_event
	content_source = dlc_POD
    title = POD_coterie_maintenance.20.t
	theme = coterie
	
	desc = {
		desc = POD_coterie_maintenance.20.desc
		triggered_desc = {
			trigger = { POD_has_coterie_trigger = yes }
			desc = POD_coterie_maintenance.20.desc.2
		}
	}
	
	trigger = {
		exists = scope:coterie
	}
	
	left_portrait = {
	    character = root
	}
	
	right_portrait = {
	    character = scope:dead_member
	}
	
	option = {
	    name = POD_coterie_maintenance.20.option.a
		custom_tooltip = POD_coterie_maintenance.20.option.a.tt
		trigger = { POD_has_coterie_trigger = no }
		hidden_effect = {
			POD_join_coterie_effect = yes
			send_interface_toast = {
				title = POD_joined_coterie_title
				left_icon = root
				custom_tooltip = POD_joined_coterie_tt
			}
		}
	}
	
	option = {
	    name = POD_coterie_maintenance.20.option.b
		custom_tooltip = POD_coterie_maintenance.20.option.b.tt
		trigger = {
			POD_has_coterie_trigger = no
			# can't disband if other players are present
			NOT = {
				scope:coterie = {
					any_in_list = {
						variable = members
						is_ai = no
					}
				}
			}
		}
		hidden_effect = { scope:coterie = { end_story = yes } }
	}
	
	option = {
	    name = POD_coterie_maintenance.20.option.c
		custom_tooltip = POD_coterie_maintenance.20.option.c.tt
		trigger = { POD_has_coterie_trigger = no }
	}
	
	option = {
	    name = POD_coterie_maintenance.20.option.d
		trigger = { POD_has_coterie_trigger = yes }
		hidden_effect = {
			POD_leave_coterie_effect = yes
			POD_join_coterie_effect = yes
		}
	}
	
	option = {
	    name = POD_coterie_maintenance.20.option.e
		trigger = {
			POD_has_coterie_trigger = yes
			# can't disband if other players are present
			NOT = {
				var:coterie = {
					any_in_list = {
						variable = members
						NOT = { this = root }
						is_ai = no
					}
				}
			}
		}
		var:coterie = {
			every_in_list = {
				variable = members
				limit = { NOT = { this = root } }
				POD_leave_coterie_with_rivalry_effect = { RIVAL = root }
			}
		}
		hidden_effect = {
			POD_disband_coterie_effect = yes
			POD_join_coterie_effect = yes
		}
	}
	
	option = {
	    name = POD_coterie_maintenance.20.option.f
		trigger = { POD_has_coterie_trigger = yes }
		hidden_effect = { scope:coterie = { end_story = yes } }
	}
	
	option = {
	    name = POD_coterie_maintenance.20.option.g
		trigger = { POD_has_coterie_trigger = yes }
	}
}

	}

	    name = POD_coterie_maintenance.30.option.a

	}
