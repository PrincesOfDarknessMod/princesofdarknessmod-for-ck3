namespace = POD_realm_mastery

# Safeguard for not getting stuck, until a better bugfix can be found
POD_realm_mastery.1 = {
	hidden = yes
	
	immediate = {
		if = {
			limit = { has_variable = current_journey }
			trigger_event = POD_realm_mastery.2
		}
		else = {
			POD_end_journey_effect = yes
		}
	}
}

# Arriving at the freehold

		character = root

# Gaining the trait
POD_realm_mastery.3 = {
	type = character_event
	content_source = dlc_POD
	title = {
		first_valid = {
			triggered_desc = {
				trigger = {
					OR = {
						scope:petition_result = flag:gold_success
						scope:petition_result = flag:prestige_success
						scope:petition_result = flag:piety_success
					}
				}
				desc = POD_realm_mastery.3.t.success
			}
			triggered_desc = {
				trigger = {
					OR = {
						scope:petition_result = flag:gold_fail
						scope:petition_result = flag:prestige_fail
						scope:petition_result = flag:piety_fail
					}
				}
				desc = POD_realm_mastery.3.t.fail
			}
			desc = POD_realm_mastery.3.t.critfail
		}
	}
	theme = fae

	override_background = { reference = shader_event_unmoored }
	
	widget = {
		gui = "POD_shader_event_unmoored"
		container = "background_shader_vfx_container"
	}
	
	left_portrait = {
		character = root
		animation = throne_room_bow_1
	}

	immediate = {
		switch = {
			trigger = scope:petition_result

			flag:gold_success  = { remove_short_term_gold = massive_gold_value    }
			flag:gold_fail     = { remove_short_term_gold = monumental_gold_value }
			flag:gold_critfail = { remove_short_term_gold = omega_gold_value      }

			flag:prestige_success  = { add_prestige = -250  }
			flag:prestige_fail     = { add_prestige = -750  }
			flag:prestige_critfail = { add_prestige = -1500 }

			flag:piety_success  = { add_piety = -250  }
			flag:piety_fail     = { add_piety = -750  }
			flag:piety_critfail = { add_piety = -1500 }

			fallback = {}
		}
	}
	
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { fae_era >= 3 }
				desc = POD_realm_mastery.3.desc.postshattering
			}
			desc = POD_realm_mastery.3.desc.preshattering
		}
		desc = POD_realm_mastery.3.desc
		first_valid = {
			triggered_desc = {
				trigger = { root.var:current_journey.var:journey_type = flag:realm_mastery_actor }
				desc = POD_realm_mastery.3.desc.actor
			}
			triggered_desc = {
				trigger = { root.var:current_journey.var:journey_type = flag:realm_mastery_fae }
				desc = POD_realm_mastery.3.desc.fae
			}
			triggered_desc = {
				trigger = { root.var:current_journey.var:journey_type = flag:realm_mastery_prop }
				desc = POD_realm_mastery.3.desc.prop
			}
			triggered_desc = {
				trigger = { root.var:current_journey.var:journey_type = flag:realm_mastery_nature }
				desc = POD_realm_mastery.3.desc.nature
			}
			triggered_desc = {
				trigger = { root.var:current_journey.var:journey_type = flag:realm_mastery_scene }
				desc = POD_realm_mastery.3.desc.scene
			}
			triggered_desc = {
				trigger = { root.var:current_journey.var:journey_type = flag:realm_mastery_time }
				desc = POD_realm_mastery.3.desc.time
			}
		}
		desc = POD_realm_mastery.3.desc.end
	}
	
	option = {
		name = POD_realm_mastery.3.option.a
		
		switch = {
			trigger = root.var:current_journey.var:journey_type
			
			flag:realm_mastery_actor  = { add_trait = realm_actor  }
			flag:realm_mastery_fae    = { add_trait = realm_fae    }
			flag:realm_mastery_prop   = { add_trait = realm_prop   }
			flag:realm_mastery_nature = { add_trait = realm_nature }
			flag:realm_mastery_scene  = { add_trait = realm_scene  }
			flag:realm_mastery_time   = { add_trait = realm_time   }
			
			fallback = {}
		}

		POD_end_journey_effect = yes
	}
}
