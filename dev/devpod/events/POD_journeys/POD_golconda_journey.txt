namespace = POD_golconda_journey

scripted_trigger POD_golconda_teacher_requirements = {
	POD_knows_about_golconda_trigger = yes
	opinion = {
		target = root
		value >= 0
	}
	is_adult = yes
	POD_is_available_for_events = yes
}

# You find a teacher
POD_golconda_journey.1 = {
	type = character_event
	content_source = dlc_POD
	title = POD_golconda_journey.1.t
	theme = vampire
	override_background = { reference = garden }
	override_sound = { reference = "event:/SFX/Events/Themes/sfx_event_theme_type_health_good" }

	desc = POD_golconda_journey.1.desc

	left_portrait = {
		character = root
		animation = personality_content
	}

	right_portrait = {
		character = scope:teacher
		animation = personality_zealous
	}

	immediate = {
		POD_pause_journey_travel_plan_effect = yes
		# check local ruler
		if = {
			limit = {
				root.location.county_controller = {
					POD_golconda_teacher_requirements = yes
				}
			}
			root.location.county_controller = { save_scope_as = teacher }
		}
		else = {
			# check wanderers
			if = {
				limit = {
					any_pool_character = {
						province = root.capital_province
						POD_golconda_teacher_requirements = yes
					}
				}
				random_pool_character = {
					province = root.capital_province
					limit = {
						POD_golconda_teacher_requirements = yes
					}
					save_scope_as = teacher
				}
			}
			# create new character
			else = {
				create_character = {
					template = POD_journey_teacher_template
					dynasty = none
					faith = faith:viahumanitatis
					location = root.location
					save_scope_as = teacher
				}
				
				scope:teacher = {
					add_trait = salubri
					add_trait = salubri_healer
					POD_character_initialization_for_vampire_barons_effect = yes
					add_character_flag = temporary_teacher
					
					set_employer = character:POD_wraith_01
					return_to_court = yes
					add_character_flag = blocked_from_leaving
				}
			}
		}
	}

	option = {
		name = POD_golconda_journey.1.option.a
		trigger_event = {
			id = POD_golconda_journey.2
			days = 7
		}
	}
}

		gui = "POD_event_window_widget_golconda"
		container = "custom_widgets_container"
	}

	immediate = {
		add_stress = medium_stress_gain
		if = {
			limit = { NOT = { has_trait = golconda_seeker } }
			add_trait = golconda_seeker
		}
		add_trait_xp = {
			trait = golconda_seeker
			value = scope:golconda_xp
		}
	}

	option = {
		name = POD_golconda_journey.10.option.a
		if = {
			limit = { NOT = { has_character_flag = ready_for_suspire } }
			random_list = {
				0 = {
					modifier = { add = { value = suspire_success_chance multiply = 0.5 } }
					custom_tooltip = POD_golconda_journey.10.option.a.success
					trigger_event = POD_golconda_journey.11
				}
				100 = {
					modifier = { add = { value = 0 subtract = suspire_success_chance multiply = 0.5 } }
					modifier = { factor = 0 suspire_success_chance = 100 }
					#custom_tooltip = POD_golconda_journey.1000.option.a.failure
					POD_end_journey_effect = yes
					if = {
						limit = {
							has_variable = continue_golconda
							piety >= 1000
						}
						custom_tooltip = {
							text = golconda_widget_tt
							add_piety = -1000
							POD_find_and_start_journey_effect = {
								TYPE = golconda_2
							}
						}
					}
				}
			}
		}
		else = {
			POD_end_journey_effect = yes
			if = {
				limit = {
					has_variable = continue_golconda
					piety >= 1000
				}
				custom_tooltip = {
					text = golconda_widget_tt
					add_piety = -1000
					POD_find_and_start_journey_effect = {
						TYPE = golconda_2
					}
				}
			}
		}
	}
}

# Golconda mentoring automation (wrapper event)
POD_golconda_journey.2000 = {
	hidden = yes

	immediate = {
		if = {
			limit = { has_variable = golconda_mentee }
			var:golconda_mentee = { save_scope_as = golconda_mentee }
			if = {
				limit = { scope:golconda_mentee = { has_trait = golconda } }
				POD_golconda_end_mentorship = {
					REASON_TOOLTIP = POD_golconda_journey.2000.toast.golconda
				}
			}
			else_if = {
				limit = { scope:golconda_mentee = { suspire_success_chance >= prev.suspire_success_chance } }
				POD_golconda_end_mentorship = {
					REASON_TOOLTIP = POD_golconda_journey.2000.toast.insight
				}
			}
			else_if = {
				limit = { suspire_success_chance < 20 }
				POD_golconda_end_mentorship = {
					REASON_TOOLTIP = POD_golconda_journey.2000.toast.lowinsight
				}
			}
			else_if = {
				limit = {
					scope:golconda_mentee = {
						OR = {
							POD_is_available = no
							is_alive = no
							is_imprisoned = yes
						}
					}
				}
				POD_golconda_end_mentorship = {
					REASON_TOOLTIP = POD_golconda_journey.2000.toast.invalid
				}
			}
			else = {
				trigger_event = POD_golconda_journey.2001
			}
		}
	}
}

# Maintenance event for AI Golconda Seekers
POD_golconda_journey.3000 = {
	hidden = yes

	trigger = {
		has_trait = golconda_seeker
	}

	immediate = {
		if = {
			limit = {
				is_ai = yes
				POD_is_available = yes
			}
			
			POD_golconda_mentoring_effect = {
				STUDENT = this
			}

			if = {
				limit = {
					suspire_success_chance >= suspire_threshold
					NOR = {
						# Don't attempt the Suspire if in a player's Coterie
						any_relation = {
							type = coterie_member
							is_ai = no
						}
						# Don't attempt the Suspire if being mentored
						has_variable = golconda_mentor
					}
				}
				random_list = {
					0 = {
						modifier = { add = { value = suspire_success_chance } }
						POD_achieve_golconda_effect = yes
					}
					100 = {
						modifier = { add = { value = 0 subtract = suspire_success_chance } }
						POD_become_wight_effect = yes
						POD_golconda_seeker_maintenance_effect = yes
					}
				}
			}
			else = {
				POD_golconda_seeker_maintenance_effect = yes
			}
		}
		else = {
			POD_golconda_seeker_maintenance_effect = yes
		}
	}
}
