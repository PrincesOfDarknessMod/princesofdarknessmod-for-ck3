namespace = POD_golconda_journey

scripted_trigger POD_golconda_teacher_requirements = {
	POD_knows_about_golconda_trigger = yes
	opinion = {
		target = root
		value >= 0
	}
	is_adult = yes
	NOT = { has_trait = torpor } # disqualify Byzar
}

# You find a teacher
POD_golconda_journey.1 = {
	type  = character_event
    title = POD_golconda_journey.1.t
	theme = vampire
	override_background = { reference = garden }
	override_sound = { reference = "event:/SFX/Events/Themes/sfx_event_theme_type_health_good" }

	desc = POD_golconda_journey.1.desc

	left_portrait = {
	    character = root
		animation = personality_content
	}

	right_portrait = {
	    character = scope:teacher
		animation = personality_zealous
	}

	immediate = {
		hidden_effect = {
			current_travel_plan ?= {
				pause_travel_plan = yes
			}
		}
		# check local ruler
		if = {
			limit = {
				root.location.county_controller = {
					POD_golconda_teacher_requirements = yes
				}
			}
			root.location.county_controller = { save_scope_as = teacher }
		}
		else = {
			# check wanderers
			if = {
				limit = {
					any_pool_character = {
						province = root.capital_province
						POD_golconda_teacher_requirements = yes
					}
				}
				random_pool_character = {
					province = root.capital_province
					limit = {
						POD_golconda_teacher_requirements = yes
					}
					save_scope_as = teacher
				}
			}
			# create new character
			else = {
				create_character = {
					template = POD_journey_teacher_template
					dynasty = none
					faith = faith:viahumanitatis
					location = root.location
					save_scope_as = teacher
				}
				
				scope:teacher = {
					add_trait = salubri
					add_trait = salubri_healer
					POD_character_initialization_for_vampire_barons_effect = yes
					add_character_flag = temporary_teacher
					
					set_employer = character:POD_wraith_01
					return_to_court = yes
					add_character_flag = blocked_from_leaving
				}
			}
		}
	}

	option = {
	    name = POD_golconda_journey.1.option.a
		trigger_event = {
			id = POD_golconda_journey.2
			days = 7
		}
	}
}

	left_portrait = {
	    character = root
		animation = thinking
	}

	right_portrait = {
	    character = scope:teacher
		animation = personality_zealous
	}

	immediate = {
		custom_tooltip = {
			text = POD_golconda_journey.2.effect

			add_character_flag = knows_about_golconda

			var:coterie ?= {
				every_in_list = {
					variable = members
					limit = {
						POD_is_in_same_journey_trigger = { TARGET = root }
						NOT = { has_character_flag = knows_about_golconda }
					}
					add_character_flag = knows_about_golconda
				}
			}
		}
	}

	option = {
	    name = POD_golconda_journey.2.option.a
		POD_end_journey_effect = yes
		hidden_effect = {
			scope:teacher = {
				if = {
					limit = { has_character_flag = temporary_teacher }
					trigger_event = {
						id = POD_disappearance.1 
						days = 1
					}
				}
			}
		}
	}
}
