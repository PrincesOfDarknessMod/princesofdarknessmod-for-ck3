namespace = PODanarchrevolt

scripted_trigger POD_is_valid_ante_slayer_trigger = {
	is_ai = yes
	NOT = { has_faith = faith:laibon }
	POD_is_on_golconda_trigger = no
	POD_is_available_for_events = yes
}

PODanarchrevolt.0001 = {
	hidden = yes
	scope = none
	
	immediate = {
		faith:freeanarch = {
			add_doctrine = anarch_revolt_doctrine
			}
		every_living_character = {
			limit = {
				is_vampire_trigger = yes
				NOT = { has_faith = faith:laibon }		
				NOT = { has_trait = torpor }	
				# NOT = { is_at_location = { geographical_region = world_india } }
				# NOT = { is_at_location = { geographical_region = world_africa } }
				# NOT = { is_at_location = { geographical_region = world_tibet } }		
			}
			trigger_event = PODanarchrevolt.0002
		}
		if = {
			limit = {
				character:790512 = { 
					is_ai = no 
					is_vampire_trigger = no
					POD_is_available_for_events = yes
				}
			}
			character:790512 = { 
				trigger_event = {
					id = POD_anteshuntset.100
					years = 1 
				} 
			}
		}
		
		trigger_event = {
				id = PODanarchrevolt.0015
				years = { 5 6 }
			}
		
		trigger_event = {
				id = PODanarchrevolt.0016
				years = { 10 15 }
			}		
			
	}

}

PODanarchrevolt.0002 = {
	type = character_event
	content_source = dlc_POD
	title = PODanarchrevolt.0002.t
	desc = PODanarchrevolt.0002.desc
	theme = lasombra_coa
	left_portrait = {
		character = root
		animation = worry
	}
	
	immediate = {
		play_music_cue = "mx_cue_general_crusade_event_1"
			if = {	
				limit = {
					is_ai = no
					}
				trigger_event = {
						id = POD_anteshuntabsi.1
						years = 1 
					}	
			}		
	}

	trigger = {
	}

	option = {
		name = PODanarchrevolt.0002.a
		
		ai_chance = { 
		
			base = 55 
		
			modifier = { 
				factor = 0
				has_faith = faith:viafurores
			}		
		}
		
		custom_tooltip = anarch_revolt_effect_message
		
	}
		
	option = {
		name = PODanarchrevolt.0002.b
		
		set_character_faith = faith:freeanarch
		add_piety = massive_piety_value
		add_piety_level = 2		
		every_held_title = {
			limit = { tier = tier_county }
			set_county_faith = faith:freeanarch
		}
		ai_chance = {
			base = 1
			modifier = { # Where the wind blows, I sail.
				add = 20
				has_trait = cynical
			}
			modifier = { 
				add = 50
				has_trait = brujah
			}			
			modifier = { 
				add = 90
				age < 150
			}			
			modifier = { 
				factor = 0.5
				highest_held_title_tier > tier_duchy
			}			
			modifier = { 
				factor = 0
				highest_held_title_tier > tier_kingdom
			}
			modifier = { 
				factor = 0
				age > 1000
			}
			modifier = { 
				factor = 0
				religion = religion:demons_religion
			}			
		}		
		
		custom_tooltip = anarch_revolt_effect_message
		
	}
		
	option = {
		name = PODanarchrevolt.0002.b
		
		trigger = {
			is_ai = yes
			has_faith = faith:viafurores
		}	

		set_character_faith = faith:freeanarch
		add_piety = massive_piety_value
		add_piety_level = 2		
		every_held_title = {
			limit = { tier = tier_county }
			set_county_faith = faith:freeanarch
		}
		
		ai_chance = 1000
	}
}

	}
