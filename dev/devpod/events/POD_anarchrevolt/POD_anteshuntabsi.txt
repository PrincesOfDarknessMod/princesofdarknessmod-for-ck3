#First Nictuku attack event, unlock decisions
POD_anteshuntabsi.3 = {
	type = character_event
	content_source = dlc_POD
	title = POD_anteshuntabsi.3.t
	desc = POD_anteshuntabsi.3.desc
	theme = nosferatu_coa
	left_portrait = {
		character = root
		animation = worry
	}

	right_portrait = {
		character = scope:target
		animation = fear
	}

	immediate = {
		if = {
			limit = { any_vassal_or_below = { always = yes } }
			random_vassal = {
				limit = {
					POD_is_available_for_events = yes
				}
				save_scope_as = target
			}
		}
	}

	option = {
		name = POD_anteshuntabsi.3.a

		custom_tooltip = anteabsi_nictuku

		add_character_flag = hunted_by_nictuku

		scope:target = {
			death = { death_reason = death_murder }
		}

		trigger_event = {
				id = POD_anteshuntabsi.5
				days = 30
				}
		}

}

POD_anteshuntabsi.5 = {
	type = character_event
	content_source = dlc_POD
	title = POD_anteshuntabsi.5.t
	desc = POD_anteshuntabsi.5.desc
	theme = nosferatu_coa

	option = {
		name = POD_anteshuntabsi.5.a

			limit = {
				is_unbondable_trigger = yes
			}
			add_character_modifier = {
				modifier = babayaga_bound_modifier

			}
		}

		else = {
			add_character_modifier = {
				modifier = babayaga_bound_modifier
				years = 50
			}
		}

		trigger_event = {
			id = POD_anteshuntabsi.6
			days = { 10 15 }
		}
	}

	option = {
		name = POD_anteshuntabsi.5.b

		duel = {
			skill = prowess
			value = 70
			20 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = 0.5
				}
				desc = POD_anteshuntabsi.5.effect.success.desc
			}
			20 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = -0.5
				}
				desc = POD_anteshuntabsi.5.critical.failure.effect.desc

				send_interface_toast = {
					title = POD_anteshuntabsi.5.critical.failure
					left_icon = root

					death = {
						death_reason = death_execution
					}
				}
			}
		}
		custom_tooltip = anteabsi_end_baba
	}
}

#The forest : set up event
POD_anteshuntabsi.6 = {
	type = character_event
	content_source = dlc_POD
	title = POD_anteshuntabsi.6.t
	desc = POD_anteshuntabsi.6.desc
	theme = nosferatu_coa

	immediate = {
		remove_character_flag = hunted_by_nictuku
	}

	option = {
		name = POD_anteshuntabsi.6.a

		#Darvag
		if = {
			limit = {
				POD_vampire_can_show_up_in_events = { CHARACTER = character:790113 }
			}
			trigger_event = {
				id = POD_anteshuntabsi.7
				days = 15
			}
		}
		else = {
			trigger_event = {
				id = POD_anteshuntabsi.8
				days = 15
			}
		}

		#Gutka
		if = {
			limit = {
				POD_vampire_can_show_up_in_events = { CHARACTER = character:790249 }
			}
			trigger_event = {
				id = POD_anteshuntabsi.9
				days = 30
			}
		}
		else = {
			trigger_event = {
				id = POD_anteshuntabsi.10
				days = 30
			}
		}

		#Siamko
		if = {
			limit = {
				POD_vampire_can_show_up_in_events = { CHARACTER = character:791047 }
			}
			trigger_event = {
				id = POD_anteshuntabsi.11
				days = 45
			}
		}
		else = {
			trigger_event = {
				id = POD_anteshuntabsi.12
				days = 45
			}
		}

		#Abraxes
		if = {
			limit = { NOT = { has_character_flag = dealt_with_abraxes } }
			trigger_event = {
				id = POD_anteshuntabsi.13
				days = 60
			}
		}

		#Echidna
		if = {
			limit = { NOT = { has_character_flag = dealt_with_echidna } }
			trigger_event = {
				id = POD_anteshuntabsi.14
				days = 75
			}
		}

		#Gorgo
		if = {
			limit = { NOT = { has_character_flag = dealt_with_gorgo } }
			trigger_event = {
				id = POD_anteshuntabsi.15
				days = 90
			}
		}

		#Nuckalavee
		if = {
			limit = { NOT = { has_character_flag = dealt_with_nuckalavee } }
			trigger_event = {
				id = POD_anteshuntabsi.16
				days = 105
			}
		}

		#Gayomart
		if = {
			limit = { NOT = { has_character_flag = dealt_with_gayomart } }
			trigger_event = {
				id = POD_anteshuntabsi.17
				days = 120
			}
		}

		#Ending
		trigger_event = {
			id = POD_anteshuntabsi.18
			days = 135
		}
	}
}
