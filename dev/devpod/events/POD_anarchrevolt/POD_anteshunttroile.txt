scripted_trigger POD_can_get_carthage_insights_from = {
	POD_vampire_can_show_up_in_events = { CHARACTER = $CHARACTER$ }
	OR = {
		this = $CHARACTER$
		reverse_opinion = {
			target = $CHARACTER$
			value >= 25
		}
	}
}
