# Victory
POD_anteshuntset.18 = {
	type = character_event
	content_source = dlc_POD
	title = POD_anteshuntset.18.t
	desc = POD_anteshuntset.18.desc
	theme = setite_coa

	option = {
		name = POD_anteshuntset.18.a
			
		if = {
			limit = {
				has_character_flag = kemintiriredemption_flag
			}
			set_global_variable = pod_achiev_kemintiri # Achievement: Isis' Revenge
		}
			
		add_prestige = massive_prestige_value
		add_prestige_level = 2				
		
		set_global_variable = {
			name = variable_POD_set_dead
			value = yes
		}
		
		POD_remove_vampire_generation_traits_effect = yes	
		add_trait_force_tooltip = vampgeneration3

		assign_obfuscate_effect = yes
		assign_presence_effect = yes
		assign_serpentis_effect = yes
		
		POD_give_diablerist_secret_or_trait_effect = yes
		
		if = {
			limit = { has_character_modifier = pseudo_third_modifier }
			remove_character_modifier = pseudo_third_modifier
		}
		
		if = {
			limit = { NOT = { has_character_modifier = bloodpotency10_modifier } }
			POD_remove_bloodpotency_modifiers_effect = yes
			add_character_modifier = bloodpotency10_modifier
		}
		
		add_character_modifier = {
			modifier = slayer_of_gods
			years = 50
			}	

		add_character_modifier = {
			modifier = young_third_modifier
			}		
			
		character:505012 = {
			set_death_reason = {
				death_reason = death_diablerie
				killer = root
			}
		}
		
		POD_achiev_nomnom_check = yes
	}
}
