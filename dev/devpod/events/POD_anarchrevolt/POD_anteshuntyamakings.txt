#MS check IF you have made Emma o mad by having jizo and threaten the ainu underworld
POD_anteshuntyamakings.17 = {
	type = character_event
	content_source = dlc_POD
	title = POD_anteshuntyamakings.17.t
	desc = {
		first_valid = {
			desc = POD_anteshuntyamakings.17.desc
		}
		first_valid = {
			triggered_desc = {
				trigger = {
					has_character_flag = yk_hunt_ainu_flag
					has_character_flag = yk_hunt_jizo_flag
				}
				desc = POD_anteshuntyamakings.17.desc.b
			}
		}
		first_valid = {
			triggered_desc = {
				trigger = {
					NOT = {
					has_character_flag = yk_hunt_ainu_flag
					has_character_flag = yk_hunt_jizo_flag
					}
				}
				desc = POD_anteshuntyamakings.17.desc.c
			}
		}
	}
	theme = vampire

	option = {
		name = POD_anteshuntyamakings.17.a

		trigger = {
			NAND = {
				has_character_flag = yk_hunt_ainu_flag
				has_character_flag = yk_hunt_jizo_flag
			}
		}

		custom_tooltip = anteabsi_end
	}

	option = {
		name = POD_anteshuntyamakings.17.b

		trigger = {
			has_character_flag = yk_hunt_ainu_flag
			has_character_flag = yk_hunt_jizo_flag
			var:character_ykhunt_force >= var:yamaking_ykhunt_force
		}

		trigger_event = {
			id = POD_anteshuntyamakings.18
			days = { 10 15 }
		}
	}

	option = {
		name = POD_anteshuntyamakings.17.c

		trigger = {
			has_character_flag = yk_hunt_ainu_flag
			has_character_flag = yk_hunt_jizo_flag
			var:character_ykhunt_force < var:yamaking_ykhunt_force
		}

		death = { death_reason = death_battle }

		random_knight = {
			limit = { is_available_ai_adult = yes }
			death = { death_reason = death_battle }
		}

		random_knight = {
			limit = { is_available_ai_adult = yes }
			death = { death_reason = death_battle }
		}
	}

	after = {
		if = {
			limit = {
				has_character_flag = yk_hunt_ainu_flag
			}
			remove_character_flag = yk_hunt_ainu_flag
		}
		if = {
			limit = {
				has_character_flag = yk_hunt_jizo_flag
			}
			remove_character_flag = yk_hunt_jizo_flag
		}
		remove_variable = character_ykhunt_force
		remove_variable = yamaking_ykhunt_force
	}
}
