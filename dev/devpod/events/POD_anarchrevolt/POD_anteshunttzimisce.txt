POD_anteshunttzimisce.15 = {
	type = character_event
	content_source = dlc_POD
	title = POD_anteshunttzimisce.15.t
	desc = POD_anteshunttzimisce.15.desc
	theme = tzimisce_coa
	
	trigger = {
		POD_vampire_can_show_up_in_events = { CHARACTER = character:790457 }
	}
	
	on_trigger_fail = {
		if = {
			limit = { has_character_flag = antetzimisce_lambach }
			remove_character_flag = antetzimisce_lambach
		}
		trigger_event = POD_anteshunttzimisce.16
	}

	immediate = {
		if = {
			limit = { has_character_flag = antetzimisce_lambach }
			remove_character_flag = antetzimisce_lambach
		}
	}

	option = {
		name = POD_anteshunttzimisce.15.a
			
		add_prestige = massive_prestige_value
		add_prestige_level = 2				
		
		set_global_variable = {
			name = variable_POD_tzimisce_dead
			value = yes
		}
		
		if = { 
			limit = {
				NOT = { has_character_modifier = kupala_avatar_modifier }
				}
		trigger_event = {
				id = POD_anteshunttzimisce.17
				days = 15 
				}
		}	
		
		POD_remove_vampire_generation_traits_effect = yes	
		add_trait_force_tooltip = vampgeneration3

		assign_animalism_effect = yes
		assign_auspex_effect = yes
		assign_vicissitude_effect = yes
				
		POD_give_diablerist_secret_or_trait_effect = yes	
		
		if = {
			limit = { has_character_modifier = pseudo_third_modifier }
			remove_character_modifier = pseudo_third_modifier
		}
		
		if = {
			limit = { NOT = { has_character_modifier = bloodpotency10_modifier } }
			POD_remove_bloodpotency_modifiers_effect = yes
			add_character_modifier = bloodpotency10_modifier
		}
		
		add_character_modifier = {
			modifier = slayer_of_gods
			years = 50
			}	

		add_character_modifier = {
			modifier = young_third_modifier
			}	
			
		character:505021 = {
			set_death_reason = {
				death_reason = death_diablerie
				killer = root
			}
		}
		
		POD_achiev_nomnom_check = yes
	}
}
