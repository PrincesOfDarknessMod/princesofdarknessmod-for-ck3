namespace = POD_charjoin

POD_charjoin.100 = {           
	type = character_event
	title = POD_charjoin.100.t
	desc = POD_charjoin.100.desc
	theme = vampire
	left_portrait = scope:created_vampirechilde
	right_portrait = scope:vampiresire
	
	# Become Vampire
	option = {
		name = POD_charjoin.100.option.a
		scope:vampiresire = { POD_player_character_create_vampirechilde_effect = yes }
		
		if = {
			limit = {
				scope:vampiresire = { has_government = free_state_government }
			}
			change_government = free_state_government
		}
		else = {
			change_government = traditions_government
		}

		trigger_event = POD_chargen.610
	}
	
	# Become Ghoul
	option = {
		name = POD_charjoin.100.option.b
		trigger = { 
			POD_can_be_ghouled_scheme_trigger = yes
		 } 
		
		scope:created_vampirechilde = {
			save_scope_as = ghoulthrall
		}
		scope:vampiresire = {
			save_scope_as = domitor
		}
		
		POD_player_character_create_ghoul_effect = yes
		
		scope:ghoulthrall = {
			add_character_flag = blood_bonded 
			set_relation_ghoul_domitor = scope:domitor
		}
		
		scope:domitor = {
			add_hook = {
				target = scope:ghoulthrall
				type = strong_bloodoath_hook
			}
			POD_hunger_rouse_check_effect = yes
			POD_high_risk_of_increasing_masquerade_exposure = yes
			
			POD_give_regnant_secret_effect = yes
		}
		
		if = {
			limit = {
				scope:vampiresire = { has_government = free_state_government }
			}
			change_government = free_state_government
		}
		else = {
			change_government = traditions_government
		}
	}

	# Regret
	option = {
		name = POD_charjoin.100.option.c
		custom_tooltip = POD_charjoin.100.option.c.t
	}
}

POD_charjoin.300 = {           
	type = character_event
	title = POD_charjoin.300.t
	desc = POD_charjoin.300.desc
	theme = faith
	left_portrait = scope:hunterpupil
	right_portrait = scope:huntermentor
	
	# Mortal become Inquisitor
	option = {
		name = POD_charjoin.100.option.a
		
		if = {
			limit = {
				exists = scope:huntermentor.dynasty
				scope:huntermentor.dynasty = { has_dynasty_perk = unity_legacy_4 }
			}
			add_martial_skill = 4
		}
		else = {
			add_martial_skill = 2
		}
		remove_trait = mortal
		if = {
			limit = {
				has_trait = bastard
			}
			remove_trait = bastard
		}
		add_trait_force_tooltip = supehunter
		if = {
			limit = {
				scope:huntermentor = { has_trait = order_member }
				scope:hunterpupil = { NOT = { has_trait = order_member } }
			}
			add_trait = order_member
		}
		set_character_faith = scope:huntermentor.faith
		set_house = scope:huntermentor.house
		if = {
			limit = {
			has_orison_trigger = no
			}
			POD_give_orison_effect = yes
		}
		random = {
			chance = 10
			add_trait = truefaith
		}
		
		change_government = inquisition_government
	}

	# Regret
	option = {
		name = POD_charjoin.100.option.c
		custom_tooltip = POD_charjoin.100.option.c.t
	}
}

POD_charjoin.510 = {           
	type = character_event
	title = POD_charjoin.510.t
	desc = POD_charjoin.510.desc
	theme = vampire
	left_portrait = scope:kueijindisciple
	right_portrait = scope:kueijinsifu
	
	# Present
	option = {
		name = POD_charjoin.510.option.a

		if = {
			limit = {
				scope:kueijinsifu = { has_trait = dharma9 }
			}
			scope:kueijindisciple = { 
				pay_short_term_gold = {
					target = scope:kueijinsifu
					gold = 1000
				} 
			}
		}
		else_if = {
			limit = {
				scope:kueijinsifu = { has_trait = dharma8 }
			}
			scope:kueijindisciple = { 
				pay_short_term_gold = {
					target = scope:kueijinsifu
					gold = 500
				} 
			}
		}
		else_if = {
			limit = {
				scope:kueijinsifu = { has_trait = dharma7 }
			}
			scope:kueijindisciple = { 
				pay_short_term_gold = {
					target = scope:kueijinsifu
					gold = 300
				} 
			}
		}
		else_if = {
			limit = {
				scope:kueijinsifu = { has_trait = dharma6 }
			}
			scope:kueijindisciple = { 
				pay_short_term_gold = {
					target = scope:kueijinsifu
					gold = 200
				} 
			}
		}
		else = {
			scope:kueijindisciple = { 
				pay_short_term_gold = {
					target = scope:kueijinsifu
					gold = 100
				} 
			}
		}
		
		trigger_event = POD_charjoin.500
	}

	# Lose Prestige
	option = {
		name = POD_charjoin.510.option.a

		if = {
			limit = {
				scope:kueijinsifu = { has_trait = dharma9 }
			}
			scope:kueijindisciple = { add_prestige = -1000 }
		}
		else_if = {
			limit = {
				scope:kueijinsifu = { has_trait = dharma8 }
			}
			scope:kueijindisciple = { add_prestige = -500 }
		}
		else_if = {
			limit = {
				scope:kueijinsifu = { has_trait = dharma7 }
			}
			scope:kueijindisciple = { add_prestige = -300 }
		}
		else_if = {
			limit = {
				scope:kueijinsifu = { has_trait = dharma6 }
			}
			scope:kueijindisciple = { add_prestige = -200 }
		}
		else = {
			scope:kueijindisciple = { add_prestige = -100 }
		}
		
		trigger_event = POD_charjoin.500
	}

	# Give up
	option = {
		name = POD_charjoin.100.option.c
		custom_tooltip = POD_charjoin.100.option.c.t
	}
}
