namespace = POD_education_events

scripted_trigger POD_has_vassal_or_follower = {
	trigger_if = {
		limit = { is_landless_adventurer = yes }
		any_courtier = { count >= 1 }
	}
	trigger_else = {
		any_vassal = { count >= 1 }
	}
}

scripted_effect POD_save_random_vassal_or_follower = {
	if = {
		limit = { is_landless_adventurer = yes }
		random_courtier = { save_scope_as = $SAVE_AS$ }
	}
	else = {
		random_vassal = { save_scope_as = $SAVE_AS$ }
	}
}
