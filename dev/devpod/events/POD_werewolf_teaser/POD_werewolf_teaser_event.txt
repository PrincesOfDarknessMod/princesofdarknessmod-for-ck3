namespace = POD_werewolf_teaser

# While hunting animtals, the player character encounters a werewolf in the wilderness 

POD_werewolf_teaser.100 = {
	type = character_event
	content_source = dlc_POD
	title = POD_werewolf_teaser.100.t
	theme = hunting #gangrel

#	right_portrait = {
#		character = root
#		animation = shock
#	}
	
	right_portrait = {
		character = scope:teaser_werewolf
		animation = idle
	}
	
	desc = POD_werewolf_teaser.100.desc
	
	trigger = {
		is_ai = no
		is_playable_character = yes
		is_imprisoned = no
		NOT = { has_trait = torpor }
		NOT = { has_character_flag = had_event_POD_werewolf_teaser_100_flag }
	}	
	
	immediate = {
		
		add_character_flag = {
			flag = had_event_POD_werewolf_teaser_100_flag
			years = { 500 1000 }
		}

		hidden_effect = {
			create_character = {
				location = root.capital_province
				template = POD_werewolf_teaser_character
				save_scope_as = teaser_werewolf
			}		
		}
		
		scope:teaser_werewolf = {
			hidden_effect = { 
				add_character_modifier = POD_werewolf_form_crinos_modifier
				set_variable = {
					name = signature_weapon
					value = flag:fangsclaws
				}
				add_character_flag = has_scripted_appearance
				#add_character_flag = POD_resonance_acute_flag 
				#POD_set_resonance_effect = yes 
			}
		}

		scope:teaser_werewolf = {
			save_scope_as = defender
		}
		
		root = {
			save_scope_as = attacker
		}

	}
	
	# Run Away
	option = {
		name = POD_werewolf_teaser.100.option.a
		
		hidden_effect = {
			scope:defender = {
				trigger_event = {
					id = POD_werewolf_teaser.500
					days = 1
				}
			}
		}		
	}	

	# Fight It
	option = {
		name = POD_werewolf_teaser.100.option.b
		
		# The fight!
		configure_start_single_combat_effect = {
			SC_INITIATOR = scope:attacker
			SC_ATTACKER = scope:attacker
			SC_DEFENDER = scope:defender
			FATALITY = no
			FIXED = no
			LOCALE = terrain_scope
			OUTPUT_EVENT = POD_werewolf_teaser.200
			INVALIDATION_EVENT = single_combat.1006
		}	
		
	}	

}

POD_werewolf_teaser.300 = {
	type = character_event
	content_source = dlc_POD
	title = POD_werewolf_teaser.300.t
	theme = hunting #gangrel

#	left_portrait = {
#		character = root
#		animation = personality_bold
#	}
	
	left_portrait = {
		character = scope:teaser_werewolf
		animation = idle
	}
	
	desc = POD_werewolf_teaser.300.desc

	immediate = {
		scope:defender = {
			death = {
	        	 death_reason = death_duel
					killer = scope:attacker
	         }
		 }
	}	

	option = {
		name = POD_werewolf_teaser.300.option.a
		
		add_character_modifier = {
			modifier = POD_blood_of_a_werewolf_modifier 
			years = 10
		}

		# Lupine Hide Artifact
		if = {
			limit = { has_ep1_dlc_trigger = yes }
			create_artifact_lupine_hide_effect = { 
				OWNER = root 
				CREATOR = scope:teaser_werewolf
				DESKULLED = scope:teaser_werewolf 
				TYPE = animal_hide_big 
			}
		}
		else = {
			create_artifact_lupine_hide_effect = { 
				OWNER = root 
				CREATOR = scope:teaser_werewolf
				DESKULLED = scope:teaser_werewolf 
				TYPE = miscellaneous 
			}
		}
		
		# Lupine Skull Artifact
		if = {
			limit = { has_ep1_dlc_trigger = yes }
			create_artifact_lupine_skull_effect = { 
				OWNER = root 
				CREATOR = root
				DESKULLED = scope:teaser_werewolf 
				TYPE = animal_skull 
			}
		}
		else = {
			create_artifact_lupine_skull_effect = { 
				OWNER = root 
				CREATOR = root
				DESKULLED = scope:teaser_werewolf 
				TYPE = miscellaneous 
			}
		}
		
	}
}
