# Sent by rite_yearly_hor_divergence_drift_notify_effect to every ruler of the diverged rite + the religious head of the parent faith.
pam_tenet_events.0004 = {
	type = letter_event

	opening = {
		first_valid = {
			triggered_desc = {
				trigger = { faith.religious_head ?= root }
				desc = pam_tenet_events.0004.opening.hof
			}
			desc = pam_tenet_events.0004.opening.ruler
		}
	}

	desc = {
		# Sender only fires once diverged_added_tenet or diverged_added_doctrine exists
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:diverged_change >= 1
					exists = scope:diverged_added_tenet
				}
				desc = pam_tenet_events.0004.desc.forward_tenet
			}
			triggered_desc = {
				trigger = {
					scope:diverged_change >= 1
					exists = scope:diverged_added_doctrine
				}
				desc = pam_tenet_events.0004.desc.forward_doctrine
			}
			triggered_desc = {
				trigger = {
					scope:diverged_change <= -1
					exists = scope:diverged_added_tenet
				}
				desc = pam_tenet_events.0004.desc.backward_tenet
			}
			triggered_desc = {
				trigger = {
					scope:diverged_change <= -1
					exists = scope:diverged_added_doctrine
				}
				desc = pam_tenet_events.0004.desc.backward_doctrine
			}
		}
	}

	sender = scope:diverged_hor

	option = {
		name = pam_tenet_events.0004.a
		if = {
			limit = { scope:diverged_change >= 1 }
			custom_tooltip = pam_tenet_events.0004.a.tt.diverged
		}
		else_if = {
			limit = { scope:diverged_change <= -1 }
			custom_tooltip = pam_tenet_events.0004.a.tt.consolidated
		}
		show_as_tooltip = {
			scope:diverged_rite = {
				change_rite_divergence = {
					value = scope:diverged_change
				}
			}
			if = {
				limit = { scope:diverged_change <= -1 }
				scope:diverged_rite.faith = {
					change_fervor = {
						value = notable_fervor_value
						desc = fervor_gain_pam_rite_consolidated
					}
				}
			}
			else_if = {
				limit = { scope:diverged_change >= 1 }
				scope:diverged_rite.faith = {
					change_fervor = {
						value = significant_fervor_loss
						desc = fervor_loss_pam_rite_diverged
					}
				}
			}
		}
	}
}

# Sent by redistribute_realm_wealth_decision to every landed vassal with coin to give.
pam_tenet_events.0005 = {
	type = letter_event

	opening = {
		desc = pam_tenet_events.0005.opening
	}

	desc = pam_tenet_events.0005.desc

	sender = scope:redistributing_liege

	trigger = {
		is_landed = yes
		exists = scope:redistributing_liege
		scope:redistributing_liege = { is_alive = yes }
	}

	# Open the chests
	option = {
		name = pam_tenet_events.0005.a

		if = {
			limit = { gold > 0 }
			remove_short_term_gold = pam_communal_redistribution_gold_value
		}

		every_held_county = {
			custom = redistribute_realm_wealth_every_county_tt
			add_county_modifier = {
				modifier = tenet_communal_possessions_redistribution_modifier
				years = 10
			}
		}

		if = {
			limit = {
				rite = {
					has_tenet_status = {
						tenet = tenet_communal_possessions
						status = core
					}
				}
			}
			add_piety = medium_piety_gain
		}
		if = {
			limit = { has_personal_tenet = tenet_communal_possessions }
			change_spiritual_fulfillment = medium_spiritual_fulfillment_value
		}


		ai_chance = {
			base = 100
		}
	}

	# Keep them shut
	option = {
		name = pam_tenet_events.0005.b

		custom_tooltip = pam_tenet_events.0005.b.tt

		scope:redistributing_liege = {
			add_opinion = {
				modifier = refused_wealth_redistribution_opinion
				target = root
			}
		}

		# The faithful always comply; only a player may refuse their liege.
		ai_chance = {
			base = 0
		}
	}
}

# Sent by seek_spiritual_mysteries_decision.
pam_tenet_events.0006 = {
	type = character_event
	title = pam_tenet_events.0006.t
	desc = {
		desc = pam_tenet_events.0006.desc_intro
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:mystic_witness }
				desc = pam_tenet_events.0006.desc_witness
			}
			desc = pam_tenet_events.0006.desc_alone
		}
	}
	theme = learning_theology_focus

	left_portrait = {
		character = root
		animation = reading
	}
	right_portrait = {
		character = scope:mystic_witness
		animation = worry
	}

	immediate = {
		if = {
			limit = { has_trait = lifestyle_mystic }
			add_trait_xp = {
				trait = lifestyle_mystic
				value = lifestyle_mystic_xp_gain_minor_value
			}
		}
		else = {
			add_trait = lifestyle_mystic
		}
		# A learned witness keeps the vigil, if one is at hand
		if = {
			limit = {
				exists = cp:councillor_court_chaplain
				cp:councillor_court_chaplain = {
					is_available = yes
					is_adult = yes
				}
			}
			cp:councillor_court_chaplain = { save_scope_as = mystic_witness }
		}
		else_if = {
			limit = {
				any_courtier = {
					is_available = yes
					is_adult = yes
					learning >= 10
				}
			}
			random_courtier = {
				limit = {
					is_available = yes
					is_adult = yes
					learning >= 10
				}
				weight = {
					base = 1
					modifier = { add = learning }
				}
				save_scope_as = mystic_witness
			}
		}
	}

	# Delve deeper
	option = {
		name = pam_tenet_events.0006.a
		add_piety = medium_piety_gain
		add_learning_lifestyle_xp = medium_lifestyle_xp
		change_spiritual_fulfillment = medium_spiritual_fulfillment_value
		add_stress = major_stress_gain
		ai_chance = {
			base = 0
			modifier = { factor = 0 has_trait = craven }
			modifier = { factor = 0 has_trait = content }
		}
	}

	# Tread carefully
	option = {
		name = pam_tenet_events.0006.b
		add_learning_lifestyle_xp = minor_lifestyle_xp
		change_spiritual_fulfillment = minor_spiritual_fulfillment_value
		ai_chance = {
			base = 100
		}
	}
}

# Sent by inherit_ancestral_trait_decision.
pam_tenet_events.0008 = {
	type = character_event
	title = pam_tenet_events.0008.t
	desc = pam_tenet_events.0008.desc
	theme = faith

	left_portrait = {
		character = root
		scripted_animation = pam_reincarnation_personality_animation
	}
	right_portrait = {
		trigger = { exists = scope:remembered_forebear }
		character = scope:remembered_forebear
		scripted_animation = pam_reincarnation_personality_animation
	}

	immediate = {
		# A single forebear supplies every nature on offer
		ordered_ancestor = {
			even_if_dead = yes
			limit = {
				is_alive = no
				OR = {
					is_parent_of = root
					is_grandparent_of = root
				}
				any_character_trait = {
					has_trait_category = personality
					NOT = { root = { has_trait = prev } }
				}
			}
			order_by = pam_recallable_forebear_quality_value
			save_scope_as = remembered_forebear
		}

		if = {
			limit = { exists = scope:remembered_forebear }
			scope:remembered_forebear = {
				every_character_trait = {
					limit = {
						has_trait_category = personality
						NOT = { root = { has_trait = prev } }
					}
					add_to_list = recallable_personality_traits
				}
			}
		}

		if = {
			limit = { any_in_list = { list = recallable_personality_traits } }
			random_in_list = {
				list = recallable_personality_traits
				save_scope_as = remembered_trait_1
			}
		}
		if = {
			limit = {
				exists = scope:remembered_trait_1
				any_in_list = {
					list = recallable_personality_traits
					NOT = { this = scope:remembered_trait_1 }
				}
			}
			random_in_list = {
				list = recallable_personality_traits
				limit = { NOT = { this = scope:remembered_trait_1 } }
				save_scope_as = remembered_trait_2
			}
		}
		if = {
			limit = {
				exists = scope:remembered_trait_2
				any_in_list = {
					list = recallable_personality_traits
					NOR = {
						this = scope:remembered_trait_1
						this = scope:remembered_trait_2
					}
				}
			}
			random_in_list = {
				list = recallable_personality_traits
				limit = {
					NOR = {
						this = scope:remembered_trait_1
						this = scope:remembered_trait_2
					}
				}
				save_scope_as = remembered_trait_3
			}
		}
	}

	option = {
		name = pam_tenet_events.0008.a
		trigger = { exists = scope:remembered_trait_1 }
		pam_claim_remembered_personality_trait_effect = { TRAIT = scope:remembered_trait_1 }
		ai_chance = { base = 100 }
	}

	option = {
		name = pam_tenet_events.0008.b
		trigger = { exists = scope:remembered_trait_2 }
		pam_claim_remembered_personality_trait_effect = { TRAIT = scope:remembered_trait_2 }
		ai_chance = { base = 100 }
	}

	option = {
		name = pam_tenet_events.0008.c
		trigger = { exists = scope:remembered_trait_3 }
		pam_claim_remembered_personality_trait_effect = { TRAIT = scope:remembered_trait_3 }
		ai_chance = { base = 100 }
	}

	# Recall nothing
	option = {
		name = pam_tenet_events.0008.d
		change_spiritual_fulfillment = medium_spiritual_fulfillment_value
		ai_chance = { base = 0 }
	}
}
