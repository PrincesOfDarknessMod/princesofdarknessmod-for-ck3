namespace = POD_ai_vampire_ghouling_event

POD_ai_vampire_ghouling_event.100 = {
	type = character_event
	hidden = yes

    trigger = {
		highest_held_title_tier > tier_county
		is_playable_character = yes
        is_vampire_trigger = yes
		is_adult = yes
		is_imprisoned = no
		is_ai = yes 
		NOT = { has_character_flag = POD_delay_ghouling_flag }
		NOT = { has_trait = torpor }
		NOT = { vassal_contract_has_flag = progeny_rights_none }
		NOT = { vassal_contract_has_flag = progeny_rights_childer }
		OR = {
			faith = { has_doctrine_parameter = vampireghouling_accepted }
			faith = { has_doctrine_parameter = vampireghouling_mass }
		}
		num_of_relation_ghoul_thrall < max_ghoul_cap
        POD_is_not_clone_trigger = yes
    }

    immediate = {
		
		save_scope_as = POD_new_domitor

		scope:POD_new_domitor = {
			save_scope_as = domitor
		}

		#Gender
		random_list = {
			50 = { #Female
				dummy_female = { save_temporary_scope_as = POD_new_ghoul_gender }
			}
			50 = { #Male
				dummy_male = { save_temporary_scope_as = POD_new_ghoul_gender }
			}
		}
		
		create_character = {
			save_temporary_scope_as = POD_new_ghoul
			age = { 18 50 }
			location = scope:POD_new_domitor.capital_province
			culture = scope:POD_new_domitor.capital_province.culture
			faith = scope:POD_new_domitor.faith
			gender = scope:POD_new_ghoul_gender 
			random_traits = yes
			trait = ghoul
		}				
		
		scope:POD_new_ghoul = {
			POD_remove_all_terrible_traits_effect = yes	
			add_character_modifier = {
				modifier = masquerade0_modifier 
			}
			add_character_modifier = {
				modifier = bloodhunger0_modifier
			}	
			add_character_modifier = {
				modifier = bloodpotency0_modifier 
			}
			set_relation_ghoul_domitor = scope:POD_new_domitor
			add_character_flag = blocked_from_leaving	
			add_character_flag = blood_bonded 
			POD_character_ghoul_discipline_initialization_effect = yes
			POD_set_resonance_effect = yes
			if = { 
				limit = {
					scope:POD_new_domitor = { has_perk = heavens_gate_perk }
				}
				add_trait = revenant
			}
		}
		
		add_hook = {
				target = scope:POD_new_ghoul
				type = strong_bloodoath_hook
		}
		add_courtier = scope:POD_new_ghoul
		add_character_flag = {
			years = 25
			flag = POD_delay_ghouling_flag
		}
		POD_give_regnant_secret_effect = yes
	}
}
