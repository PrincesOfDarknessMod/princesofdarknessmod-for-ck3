namespace = POD_ai_vampire_ghouling_event

POD_ai_vampire_ghouling_event.100 = {
	type = character_event
	hidden = yes

    	trigger = {
		highest_held_title_tier > tier_county
		is_playable_character = yes
      		is_vampire_trigger = yes
		is_adult = yes
		is_imprisoned = no
		is_ai = yes 
		NOT = { has_character_flag = POD_delay_ghouling_flag }
		NOT = { has_trait = torpor }
		NOT = { vassal_contract_has_flag = progeny_rights_none }
		NOT = { vassal_contract_has_flag = progeny_rights_childer }
		OR = {
			faith = { has_doctrine_parameter = vampireghouling_accepted }
			faith = { has_doctrine_parameter = vampireghouling_mass }
		}
		num_of_relation_ghoul_thrall < max_ghoul_cap
		POD_is_not_clone_trigger = yes
	}

	immediate = {		
		create_character = {
			save_temporary_scope_as = POD_new_ghoul
			age = { 18 50 }
			location = THIS.capital_province
			culture = THIS.capital_province.culture
			faith = THIS.faith
			gender_female_chance = 50
			random_traits = yes
			after_creation = {
				POD_become_ghoul_effect = yes
			}
		}				
		
		scope:POD_new_ghoul = {
			POD_remove_all_terrible_traits_effect = yes
			POD_become_bound_ghoul_effect = { DOMITOR = PREV }
		}
		
		add_courtier = scope:POD_new_ghoul
		add_character_flag = {
			years = 25
			flag = POD_delay_ghouling_flag
		}
		POD_give_regnant_secret_effect = yes
	}
}
