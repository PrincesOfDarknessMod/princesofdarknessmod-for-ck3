namespace = find_lair_scheme_maintenance

find_lair_scheme_maintenance.1 = {
	type = character_event
	
	hidden = yes

	trigger = {
		is_pool_character = no
		POD_is_supehunter_trigger = yes
		any_child = { is_alive = yes has_trait = supehunter } # Just stop suiciding without successor
		is_ai = yes
		NOT = {
			any_scheme = {
				scheme_type = inquihunt
			}
		}

		build_find_lair_targets_trigger = yes

		any_in_list = {
			list = find_lair_targets
			root = {
				can_start_scheme = {
					target_character = prev
					type = murder
				}
			}
			NOT = { this = root }
		}
	}

	immediate = {
		if = {
			limit = {
			build_find_lair_targets_trigger = yes 
			POD_is_supehunter_trigger = yes
			}
		
		random_in_list = {
			list = find_lair_targets
			limit = {
				root = {
					can_start_scheme = {
						target_character = prev
						type = murder
						}
					}
				NOR = {
					is_child_of = root
					is_grandchild_of = root
					is_great_grandchild_of = root
					}
				NOT = { this = root }
				}
				root = {
					start_scheme = {
						type = inquihunt
						target_character = prev
					}
				}
			}
		}
	}
}
