namespace = POD_ai_vampire_embrace

POD_ai_vampire_embrace.1 = {
	type = character_event
	hidden = yes

    trigger = {
		is_landed = yes
        is_vampire_trigger = yes
		is_adult = yes
		is_imprisoned = no
		is_ai = yes 
		NOT = { has_character_flag = had_event_POD_ai_vampire_embrace_1 }
		# is_vampgeneration1_trigger = no
		# is_vampgeneration2_trigger = no
		# is_vampgeneration3_trigger = no
		NOT = { has_trait = torpor }
		# NOT = { has_title = title:e_enoch } # Stop checking unexistent stuff to every ai !
		NOT = { vassal_contract_has_flag = progeny_rights_none }
		NOT = { vassal_contract_has_flag = progeny_rights_ghouls }
		OR = {
			any_child = { is_alive = yes count < 1 }
			count_has_two_childer_or_less_trigger = yes
			duke_has_three_childer_or_less_trigger = yes
			king_has_four_childer_or_less_trigger = yes
			emperor_has_five_childer_or_less_trigger = yes
		}
        POD_is_clone_trigger = no		
    }

    immediate = {
		if = {
			limit = {
				any_child = { is_alive = yes count < 1 }
			}
			add_character_flag = {
				flag = had_event_POD_ai_vampire_embrace_1
				years = 1
			}
		}	
		if = {
			limit = {
				any_child = { is_alive = yes count < 2 }
			}
			add_character_flag = {
				flag = had_event_POD_ai_vampire_embrace_1
				years = 2
			}
		}		
		
		if = {
			limit = {
				any_child = { is_alive = yes count < 3 }
			}
			add_character_flag = {
				flag = had_event_POD_ai_vampire_embrace_1
				years = 5
			}
		}	

		if = {
			limit = {
				any_child = { is_alive = yes count < 4 }
			}
			add_character_flag = {
				flag = had_event_POD_ai_vampire_embrace_1
				years = 10
			}
		}		

		if = {
			limit = {
				any_child = { is_alive = yes count < 5 }
			}
			add_character_flag = {
				flag = had_event_POD_ai_vampire_embrace_1
				years = 25
			}
		}	
		
		save_scope_as = vampiresire
		scope:vampiresire = {
			POD_character_create_vampirechilde_effect = yes		
		}
	}
}
