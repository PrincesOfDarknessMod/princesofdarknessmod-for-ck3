#Repurpose of old psychic invasion
POD_ai_disciplines.2 = {
	hidden = yes

	immediate = {
		random_vassal = {
			limit = {
				is_available = yes
				POD_has_strong_mental_defenses_trigger = no
				reverse_opinion = {
					target = root
					value < 35
				}
			}
			weight = {
				base = 1
				modifier = {
					has_relation_rival = root
					factor = 20
				}
				modifier = {
					factor = 20
					reverse_opinion = {
						target = root
						value <= medium_negative_opinion
					}
				}
				modifier = {
					factor = 10
					reverse_opinion = {
						target = root
						value <= low_negative_opinion
					}
				}
				modifier = {
					factor = 30
					reverse_opinion = {
						target = root
						value <= high_negative_opinion
					}
				}	
				modifier = {
					factor = 5
					is_powerful_vassal = yes
				}
			}
		save_scope_as = ai_disciplines_target			
		}
		
		if = {
			limit = {
				exists = scope:ai_disciplines_target
			}
		root = { save_scope_as = actor }
		scope:ai_disciplines_target = { save_scope_as = recipient }
		scope:recipient = { trigger_event = POD_auspex.100 }
		POD_hunger_rouse_check_effect = yes	
		}
	}
}

				}
			}			
		}
	}
}
