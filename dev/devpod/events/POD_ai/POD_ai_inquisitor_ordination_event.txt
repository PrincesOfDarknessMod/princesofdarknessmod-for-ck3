POD_ai_inquisitor_ordination.1 = {
	type = character_event
	hidden = yes

    trigger = {
		is_landed = yes
		has_trait = supehunter
		age > 20
		is_imprisoned = no
		is_ai = yes
		OR = {
			any_child = { is_landed = no count < 2 }
			any_child = { is_adult = yes count < 2 }
			any_child = { POD_is_supehunter_trigger = yes count < 2 }
		}
		NOR = { 
			dynasty = dynasty:13000004 
			dynasty = dynasty:13000005	
			AND = {
				exists = dynasty
				dynasty = { has_dynasty_modifier = custom_lay_order_modifier }
			}
		}
    }

    immediate = {
		save_scope_as = masterhunter
		scope:masterhunter = {
			POD_ai_inquisitor_ordination_effect = yes	
		}
		scope:created_apprenticehunter = {
			random = {
				chance = 5
				add_trait = truefaith
			}
			POD_endowments_reward_large_effect = yes
			if = {
				limit = {
					scope:masterhunter = {
					has_trait = order_member
					}
					exists = scope:masterhunter.dynasty
					scope:masterhunter.dynasty = { has_dynasty_perk = meritocracy_legacy_1 }
					}
				meritocracy_1_effect = yes
			}
		
		if = {
				limit = {
					scope:masterhunter = {
					has_trait = order_member
					}
					exists = scope:masterhunter.dynasty
					scope:masterhunter.dynasty = { has_dynasty_perk = meritocracy_legacy_2 }
					is_female = no
					}
				meritocracy_2_effect_a = yes
			}
		
		if = {
				limit = {
					scope:masterhunter = {
					has_trait = order_member
					}
					exists = scope:masterhunter.dynasty
					scope:masterhunter.dynasty = { has_dynasty_perk = meritocracy_legacy_2 }
					is_female = yes
					}
				meritocracy_2_effect_b = yes
			}
		
		if = {
				limit = {
					scope:masterhunter = {
					has_trait = order_member
					}
					exists = scope:masterhunter.dynasty
					scope:masterhunter.dynasty = { has_dynasty_perk = meritocracy_legacy_3 }
					}
				meritocracy_3_effect = yes
			}
		
		if = {
				limit = {
					scope:masterhunter = {
					has_trait = order_member
					}
					exists = scope:masterhunter.dynasty
					scope:masterhunter.dynasty = { has_dynasty_perk = meritocracy_legacy_4 }
					}
				meritocracy_4_effect = yes
			}
			
		if = {
				limit = {
					scope:masterhunter = {
					has_trait = order_member
					}
					exists = scope:masterhunter.dynasty
					scope:masterhunter.dynasty = { has_dynasty_perk = meritocracy_legacy_5 }
					}
				meritocracy_5_effect = yes
			}
			if = {
				limit = {
					scope:masterhunter = {
					dynasty = dynasty:13000001
					}
					exists = scope:masterhunter.dynasty
					scope:masterhunter.dynasty = {
					has_dynasty_perk = unity_legacy_4
						}
					}
				add_learning_skill = 2
				add_martial_skill = 2
			}
			if = {
				limit = {
					exists = scope:masterhunter.dynasty
					scope:masterhunter.dynasty = { has_dynasty_perk = unity_legacy_4 }
					scope:masterhunter = {
						OR = {
							dynasty = dynasty:13000002
							dynasty = dynasty:13000006
							dynasty = dynasty:13000008
							}
						}
					}
				add_martial_skill = 2
				add_prowess_skill = 5
			}
			if = {
				limit = {
					scope:masterhunter = {
					dynasty = dynasty:13000003
					}
					exists = scope:masterhunter.dynasty
					scope:masterhunter.dynasty = {
					has_dynasty_perk = unity_legacy_4
						}
					}
				add_stewardship_skill = 2
				add_learning_skill = 2
			}
		}
	}
}

POD_ai_inquisitor_ordination.2 = {
	type = character_event
	hidden = yes

    trigger = {
        is_mortal_trigger = yes
		trigger_if = {
			limit = {
				exists = dynasty
				OR = {
					dynasty = dynasty:13000001
					dynasty = dynasty:13000002
					dynasty = dynasty:13000003
					dynasty = dynasty:13000006
					dynasty = dynasty:13000008
					dynasty = { has_dynasty_modifier = custom_mendicant_order_modifier }
					dynasty = { has_dynasty_modifier = custom_contemplative_order_modifier }
					dynasty = { has_dynasty_modifier = custom_knightly_order_modifier }
				}
			}
			has_trait = legitimized_bastard
		}
		trigger_else_if = {
			limit = {
				exists = dynasty
				OR = {
					dynasty = dynasty:13000004
					dynasty = dynasty:13000005
					dynasty = { has_dynasty_modifier = custom_lay_order_modifier }
				}
			}
		}
		trigger_else = {
			OR = {
				AND = {
					exists = father
					father = { has_trait = supehunter }
				}
				AND = {
					exists = mother
					mother = { has_trait = supehunter }
				}
				AND = {
					exists = liege
					liege = { has_trait = supehunter }
				}
			}
			is_a_neteru = no
		}
        is_adult = yes
		is_a_thrall = no
    }

    immediate = {
		remove_trait = mortal
		add_trait = supehunter
		if = {
			limit = { can_be_shih_trigger = no }
			POD_give_orison_effect = yes
		}
		else = {
			POD_shih_initialization_effect = yes
			POD_shih_random_perk_initialization_for_noble_effect = yes
		}
		
		random = {
			chance = 5
			add_trait = truefaith
		}
		
		if = {
			limit = {
				NOT = { religion = { is_in_family = rf_hunters } }
			}
			if = {
				limit = {
					exists = mother
					mother = { religion = { is_in_family = rf_hunters } }
				}
				set_character_faith = mother.faith
			}
			else_if = {
				limit = {
					exists = father
					father = { religion = { is_in_family = rf_hunters } }
				}
				set_character_faith = father.faith
			}
			else = {
				POD_character_hunter_faith_initialization_effect = yes
			}
		}
		
		if = {
			limit = { 
				faith = { has_doctrine_parameter = theurgist_ordination }
			}
			add_trait = theurgist
		}
		if = {
			limit = { 
				faith = { has_doctrine_parameter = sorcerer_ordination }
			}
			add_trait = sorcerer
		}
		if = {
			limit = { exists = dynasty }
			if = {
				limit = {
					OR = {
						dynasty = dynasty:13000001
						dynasty = dynasty:13000002
						dynasty = dynasty:13000003
						dynasty = dynasty:13000006
						dynasty = dynasty:13000008
						dynasty = { has_dynasty_modifier = custom_mendicant_order_modifier }
						dynasty = { has_dynasty_modifier = custom_contemplative_order_modifier }
						dynasty = { has_dynasty_modifier = custom_knightly_order_modifier }
					}
				}
				add_trait = order_member
			}
			if = {
				limit = {
					dynasty = dynasty:13000004
					dynasty = { has_dynasty_perk = unity_legacy_4 }
				}
				add_stewardship_skill = 2
				add_diplomacy_skill = 2
			}
			if = {
				limit = {
					dynasty = dynasty:13000005
					dynasty = { has_dynasty_perk = unity_legacy_4 }
				}
				add_intrigue_skill = 2
				add_prowess_skill = 5
			}
		}
	}
}

POD_ai_inquisitor_ordination.3 = {
	type = character_event
	hidden = yes

    trigger = {
		OR = {
			has_trait = mortal
			has_trait = ghoul
		}
		dynasty = dynasty:13000007
        is_adult = yes
		is_a_thrall = no
    }

    immediate = {
		remove_trait = mortal
		add_trait = ghoul
		add_trait = supehunter
		POD_clear_resonance_intensity_effect = yes
		POD_set_resonance_effect = yes
			
		random = {
			chance = 5
			add_trait = truefaith
		}
			
		if = {
			limit = {
				NOT = { religion = religion:calomenasforsaken_religion }
			}
			if = {
				limit = {
					mother ?= { religion = religion:calomenasforsaken_religion }
				}
				set_character_faith = mother.faith
			}
			else_if = {
				limit = {
					father ?= { religion = religion:calomenasforsaken_religion }
				}
				set_character_faith = father.faith
			}
			else = {
				set_character_faith = faith:calomenasforsaken
			}
		}
		
		if = {
			limit = { dynasty = { has_dynasty_perk = unity_legacy_4 } }
			add_martial_skill = 2
			add_intrigue_skill = 2
		}
	}
}
