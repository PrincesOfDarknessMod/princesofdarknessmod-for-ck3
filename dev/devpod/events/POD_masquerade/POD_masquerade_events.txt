#Bad events happening if your Masquerade Exposure gets too high

pod_masquerade.9 = {
	type = character_event
	content_source = dlc_POD
	title = pod_masquerade.9.t
	desc = pod_masquerade.9.desc
	theme = masquerade
	left_portrait = {
		character = root
		animation = worry
	}
	
	widget = {
		gui = "POD_shader_event_fog_alley_night"
		container = "background_shader_vfx_container"
	}

	trigger = {
		is_vampire_trigger = yes
		is_imprisoned = no
		NOT = {
			has_trait = torpor
			}
		is_ai = no
		OR = {
			has_character_modifier = masquerade2_modifier
			has_character_modifier = masquerade3_modifier
			has_character_modifier = masquerade4_modifier
			has_character_modifier = masquerade5_modifier
			}
	}

	weight_multiplier = {
		base = 1
		modifier = { 
			add = 0.5
			has_character_modifier = masquerade2_modifier
		}
		modifier = { 
			add = 1.5
			has_character_modifier = masquerade4_modifier
		}
		modifier = { 
			add = 2
			has_character_modifier = masquerade5_modifier
		}		
	}
	
	immediate = {
	}

	option = { 
		name = pod_masquerade.9.a	
	
		custom_tooltip = high_masquerade_exposed_tt
	
		duel = {
			skill = stewardship
			value = 20
				60 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = 0.5
					}
				desc = pod_masquerade.9.a.effect.success.desc
				
				send_interface_toast = {
					title = pod_masquerade.9.a.success
					left_icon = root
				}				
				
			}
				35 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = 0.5
					}

					desc = pod_masquerade.9.a.failure.effect.desc
					
					send_interface_toast = {
						title = pod_masquerade.9.a.failure
						left_icon = root
						remove_short_term_gold = medium_gold_value
						}					
				}
				15 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = -2.5
					}

					desc = pod_masquerade.9.a.critical.failure.effect.desc
					
					send_interface_toast = {
						title = pod_masquerade.9.a.critical.failure
						left_icon = root
						add_character_modifier = {
							modifier = broken_authority_modifier
							years = 20
							}	
						}					
				}
			}		
	}
	option = { 
		name = pod_masquerade.9.b
	
		custom_tooltip = high_masquerade_exposed_tt
	
		remove_short_term_gold = medium_gold_value
		stress_impact = {
			base = medium_stress_gain
		}	
		
	}
}
