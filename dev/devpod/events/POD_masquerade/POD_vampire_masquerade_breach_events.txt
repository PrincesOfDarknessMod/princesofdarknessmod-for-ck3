# Exposed Hunt

POD_vampire_masquerade_breach.100 = {
	title = POD_vampire_masquerade_breach.100.title
    type = character_event
	content_source = dlc_POD
    desc = POD_vampire_masquerade_breach.100.desc
	theme = masquerade
	left_portrait = {
		character = scope:mortal
		animation = shock
	}
	
	widget = {
		gui = "POD_shader_event_fog_alley_night"
		container = "background_shader_vfx_container"
	}

	trigger = {
		is_vampire_trigger = yes
		is_ai = no		
		is_ruler = yes
		NOT = { has_character_flag = is_in_POD_vampire_masquerade_breach_100 }
	}		
	
	immediate = {
		hidden_effect = {
			add_character_flag = { 
				flag = is_in_POD_vampire_masquerade_breach_100
				days = 1825
			}
			create_character = {
				location = root.capital_province
				template = POD_mortal_hunter_template
				save_scope_as = mortal
			}
			scope:mortal = {
				add_trait = "mortal"
				set_to_lowborn = yes
				POD_set_resonance_effect = yes
				add_character_flag = {
					flag = blocked_from_leaving
					years = 2
				}
				add_trait = POD_knows_too_much
				save_scope_as = victim
			}
			add_courtier = scope:mortal
			scope:mortal = { hidden_effect = { return_to_court = yes } }
			add_opinion = {
				target = scope:mortal
				modifier = POD_opinion_breached_masquerade
			}				
		}	
	}
	
# Dominate Forgetful Mind
    option = {
        name = POD_vampire_masquerade_breach.100.a
		
		trigger = {
			POD_has_enough_blood_for_discipline = { AMOUNT = 1 }
			OR = { 
				has_trait = dominateadvanced
				has_perk = the_forgetful_mind_perk
			}	
		  
		} 		  
		root = {
			POD_hunger_rouse_check_effect = yes 
			remove_opinion = { 
				target = scope:mortal
				modifier = POD_opinion_breached_masquerade
			}
		}	
		scope:mortal = {
			remove_trait = POD_knows_too_much
		}
		POD_high_chance_of_reducing_masquerade_exposure = yes
    }

# Ignore them	
    option = {
	
        name = POD_vampire_masquerade_breach.100.b
		
		scope:mortal = {
				add_stress = medium_stress_gain
				add_trait = POD_knows_too_much		
		}
		POD_moderate_risk_of_increasing_masquerade_exposure = yes		
    }	
	
	
# Bribe them	
    option = {
        name = POD_vampire_masquerade_breach.100.c
		scope:mortal = { 	
			if = { limit = { has_trait = honest }
				remove_trait = honest	
				}
			if = { limit = { NOT = { has_trait = deceitful } } 
				add_trait = deceitful	}
			}			
		root = { 
			pay_short_term_gold = {
					gold = 20
					target = scope:mortal
				}
				remove_opinion = { 
				target = scope:mortal
				modifier = POD_opinion_breached_masquerade }
		}
		POD_small_chance_of_reducing_masquerade_exposure = yes		
    }

# Kill them	
    option = {
        name = POD_vampire_masquerade_breach.100.d
		if = {
			limit = { POD_has_a_beast_trigger = yes }
			custom_tooltip = POD_vampire_masquerade_breach.100.d.tt.beast
		}
		POD_commit_murder_effect = yes
		duel = {
			skill = prowess
			target = scope:mortal
			95 = {		
				compare_modifier = {
					value = scope:duel_value
					multiplier = 2				
				}
				desc = POD_vampire_masquerade_breach.100.d.notwounded
			}
			0 = {		
				compare_modifier = {
					value = scope:duel_value
					multiplier = 0.5
					max = 15				
				}
				desc = POD_vampire_masquerade_breach.100.d.wounded
				root = {
					increase_wounds_effect = { REASON = fight }
					hidden_effect = {
						send_interface_toast = {
							title = POD_vampire_masquerade_breach.100.d.woundedtoastitle
							custom_tooltip = POD_vampire_masquerade_breach.100.d.woundedtoast
							left_icon = root
							right_icon = scope:mortal

						}
					}
				}
			}
		}	
		stress_impact = {
			compassionate = medium_stress_impact_gain
		}
		if = {
			limit = {
				faith = {
					OR = {
						has_doctrine_parameter = vampirism_illegal
						has_doctrine_parameter = vampirism_shunned
						has_doctrine_parameter = hurt_mortal_lose_piety
					}
				}
			}
			add_piety = major_piety_loss
		} 	
		scope:mortal = { 
			death = { death_reason = death_mysterious }
		}		
		POD_moderate_chance_of_reducing_masquerade_exposure = yes
	}
	#Drain victim
	option = {
		name = POD_vampire_masquerade_breach.100.e
		if = {
			limit = { POD_has_a_beast_trigger = yes }
			custom_tooltip = POD_vampire_masquerade_breach.100.e.tt.beast
		}
		
		duel = {
			skill = prowess
			target = scope:mortal
			95 = {		
				compare_modifier = {
					value = scope:duel_value
					multiplier = 2
					
				}
				desc = POD_vampire_masquerade_breach.100.d.notwounded
			}
			0 = {		
				compare_modifier = {
					value = scope:duel_value
					multiplier = 0.5
					max = 15
					
				}
				desc = POD_vampire_masquerade_breach.100.d.wounded
				root = {
					increase_wounds_effect = { REASON = fight }
                    hidden_effect = {
						send_interface_toast = {
							title = POD_vampire_masquerade_breach.100.d.woundedtoastitle
							custom_tooltip = POD_vampire_masquerade_breach.100.d.woundedtoast
							left_icon = root
							right_icon = scope:mortal

						}
					}
				}
			}
		}
		POD_drain_victim_effect = { VICTIM = scope:mortal }
		POD_moderate_chance_of_reducing_masquerade_exposure = yes
	}	
}

POD_vampire_masquerade_breach.200 = {
    hidden = yes
	
	trigger = {
		is_courtier = yes
		liege ?= { is_ai = no }
		is_vampire_trigger = yes
		liege ?= { is_vampire_trigger = yes } 
		is_imprisoned = no
		is_ai = yes
		NOT = { has_trait = torpor }
		is_at_war = no
		intrigue <= 10 #High intrigue characters aren't sloppy
	}		
	
	immediate = {
			
		liege = { save_scope_as = vampire_judge_masquerade } 
		root = { save_scope_as = vampire_exposes_masquerade }	
		scope:vampire_judge_masquerade = {
			add_opinion = {
				target = scope:vampire_exposes_masquerade
				modifier = POD_opinion_breached_masquerade 
			}	
		}
		hidden_effect = {
			create_character = {
				location = root.capital_province
				template = POD_mortal_hunter_template
				save_scope_as = mortal
				
			}
			scope:mortal = {
				add_trait = "mortal"
				set_to_lowborn = yes
				POD_set_resonance_effect = yes
				add_character_flag = {
					flag = blocked_from_leaving
					years = 2
				}
				add_trait = POD_knows_too_much
			}		
		}
		scope:vampire_judge_masquerade = {
			add_opinion = {
				target = scope:mortal
				modifier = POD_opinion_breached_masquerade 
			}	
		}
		scope:vampire_judge_masquerade = { 
			trigger_event = {
				id = POD_vampire_masquerade_breach.300
				days = 1
			}
				
		}
		scope:vampire_judge_masquerade = { 
			trigger_event = POD_vampire_masquerade_breach.400
		}		
				
	}	
}

POD_vampire_masquerade_breach.400 = {
    type = character_event
	content_source = dlc_POD
    desc = POD_vampire_masquerade_breach.400.desc
	theme = masquerade
	title = POD_vampire_masquerade_breach.400.title
	left_portrait = {
		character = scope:vampire_exposes_masquerade
		animation = shame
	}		
	
	widget = {
		gui = "POD_shader_event_fog_alley_night"
		container = "background_shader_vfx_container"
	}

# Forgive them for a favor

    option = {
		name = POD_vampire_masquerade_breach.400.a
		trigger = {
			scope:vampire_judge_masquerade = {
				NOT = {
					has_hook_of_type = {
						target =  scope:vampire_exposes_masquerade 
						type = indebted_hook
					}
					faith = { has_doctrine = doctrine_vampirethemasquerade_none }
				}	
			}
		}
		scope:vampire_judge_masquerade = {
			add_hook = {
				target = scope:vampire_exposes_masquerade
				type = indebted_hook
			}	
		}
		
		scope:vampire_judge_masquerade = {
			remove_opinion = { 
				target = scope:vampire_exposes_masquerade
				modifier = POD_opinion_breached_masquerade 
			}
		}	
		

		
    }
	
# Forgive them	

    option = {
        name = POD_vampire_masquerade_breach.400.b
		trigger = { scope:vampire_judge_masquerade = {
			NOT = {	faith = { has_doctrine = doctrine_vampirethemasquerade_none } } 
			}
		}
		scope:vampire_judge_masquerade = {
			remove_opinion = { 
				target = scope:vampire_exposes_masquerade
				modifier = POD_opinion_breached_masquerade 
			}
		}

    }

# Order the Sheriff to Arrest them 

    option = {
        name = POD_vampire_masquerade_breach.400.c
		trigger = { scope:vampire_judge_masquerade = {
			NOT = {	faith = { has_doctrine = doctrine_vampirethemasquerade_none }} 
			}
		}
		duel = {
			skill = intrigue
			target = scope:vampire_exposes_masquerade
			50 = {		
				compare_modifier = {
					value = scope:duel_value
					multiplier = 2	
				}
				desc = POD_vampire_masquerade_breach.400.c.Arrest
				
				if = { 
					limit = {
						faith = {
							NOT = { has_doctrine = doctrine_vampirethemasquerade_none }
						}
					}
					rightfully_imprison_character_effect = {
						TARGET = scope:vampire_exposes_masquerade
						IMPRISONER = root
					}
				}
				else = { 
					imprison_character_effect = {
						TARGET = scope:vampire_exposes_masquerade
						IMPRISONER = root
					}
				}		
				
				hidden_effect = {
					send_interface_toast = {
						title = POD_vampire_masquerade_breach.400.c.ArrestNotificationtitle
						custom_tooltip = POD_vampire_masquerade_breach.400.c.ArrestNotification
						left_icon = root
						right_icon = scope:vampire_exposes_masquerade

					}
				}
			}
			50 = {		
				compare_modifier = {
					value = scope:duel_value
					multiplier = 1
					
				}
				desc = POD_vampire_masquerade_breach.400.c.Escape
				root = {
					hidden_effect = {
						send_interface_toast = {
							title = POD_vampire_masquerade_breach.400.c.EscapeNotificationtitle
							custom_tooltip = POD_vampire_masquerade_breach.400.c.EscapeNotification
							left_icon = root
							right_icon = scope:vampire_exposes_masquerade

						}
					}
				}
			}
		}
		
	}

# This is not forgiven or forgotten
	
    option = {
        name = POD_vampire_masquerade_breach.400.d
		trigger = { scope:vampire_judge_masquerade = {
			NOT = {	faith = { has_doctrine = doctrine_vampirethemasquerade_none }} 
			}
		}
    }

# no masquerade
	option = {
		name = POD_vampire_masquerade_breach.400.e
		trigger = { scope:vampire_judge_masquerade = {
			faith = { has_doctrine = doctrine_vampirethemasquerade_none }
			}
		}
		scope:vampire_judge_masquerade = {
			remove_opinion = { 
				target = scope:vampire_exposes_masquerade
				modifier = POD_opinion_breached_masquerade 
			}
		}
	}

}
