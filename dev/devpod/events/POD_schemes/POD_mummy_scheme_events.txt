namespace = POD_mummy_scheme

POD_mummy_scheme.100 = {
	type = character_event
	content_source = dlc_POD
	title = POD_mummy_scheme.100.t
	desc = POD_mummy_scheme.100.desc
	theme = mummy

	left_portrait = {
		character = root
		animation = anger
	}
	
	right_portrait = {
		character = scope:recipient
		animation = fear
	}
	
	immediate = {
		if = {
			limit = {
				exists = scope:scheme
			}
			scope:scheme.scheme_target_character = {
				save_scope_as = recipient	
			}
		}
	}

	option = {
		name = POD_mummy_scheme.100.a
		
		POD_willpower_rouse_check_effect = yes
		
		if = {
			limit = {
				scope:recipient = {
					has_trait = wounded_3
				}
			}
			POD_hekau_cost_effect = { COUNT = 3 }
		}
		
		if = {
			limit = {
				scope:recipient = {
					has_trait = wounded_2
				}
			}
			POD_hekau_cost_effect = { COUNT = 2 }
		}
		
		if = {
			limit = {
				scope:recipient = {
					has_trait = wounded_1
				}
			}
			POD_hekau_cost_effect = { COUNT = 1 }
		}
		
		scope:recipient = {
			send_interface_toast = {
				title = POD_power_attack_toast
				left_icon = root
				right_icon = this
				add_character_modifier = {
					modifier = eye_of_sakhmet_modifier
					years = 20
				}
				add_opinion = {
					target = root
					modifier = hurt_opinion
					opinion = -20
				}
			}
		}
	}
	
	after = {
		if = {
			limit = {
				exists = scope:scheme
			}
			scope:scheme = {
				end_scheme = yes
			}
		}
	}
}

POD_mummy_scheme.101 = {
	hidden = yes
	
	immediate = {
		if = {
			limit = { scope:actor = scope:sc_victor }
			scope:actor = { trigger_event = POD_mummy_scheme.100 }
		}
		else_if = {
			limit = { exists = scope:scheme }
			scope:scheme = {
				end_scheme = yes
			}
		}
	}
}
