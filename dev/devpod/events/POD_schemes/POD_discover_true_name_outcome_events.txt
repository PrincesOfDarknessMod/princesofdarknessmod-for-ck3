# Hidden event to resolve success
POD_discover_true_name_outcome.1000 = {
	hidden = yes
	
	immediate = {
		if = {
			limit = { scope:owner = { NOT = { has_hook_of_type = { target = scope:target type = true_name_hook } } } }
			scope:owner = {
				send_interface_toast = {
					title = POD_shared_loca_success
					left_icon = scope:target
					right_icon = scope:owner
			
					POD_discover_true_name_effect = {
						MASTER = scope:owner
						VICTIM = scope:target
					}
				}
			}
		}
		if = {
			limit = { 
				scope:target = { exists = house }
				scope:owner = { has_character_modifier = nian_shun_modifier } 
			}
			scope:target = { 
				every_close_family_member = {
					limit = { 
						NOT = { this = scope:owner }
						scope:owner = { 
							NOR = { 
								has_hook_of_type = { target = prev type = true_name_hook } 
								has_hook_of_type = { target = prev type = half_true_name_hook }
								has_hook_of_type = { target = prev type = partial_true_name_hook } 
							}
						} 
					}
					scope:owner = { 
						add_hook = {
							target = prev
							type = partial_true_name_hook
						}
					}
				}
			}
		}
		if = {
			limit = { 
				scope:target = { is_ruler = yes }
				scope:owner = { has_perk = echo_the_lesser_name_perk } 
			}
			set_while_counter_variable_effect = yes
			while = {
				limit = { var:while_counter < echo_the_lesser_name_perk_max }
				scope:target = { 
					random_courtier_or_guest = { 
						limit = {
							NOT = { scope:owner = { has_hook = prev } } 
						}
						scope:owner = { 
							add_hook = {
								target = prev
								type = partial_true_name_hook
							}
						}
					}
				}
				increase_while_counter_variable_effect = yes
			}
			remove_while_counter_variable_effect = yes
		}
		if = {
			limit = { 
				scope:target = { is_ruler = yes }
				scope:owner = { has_perk = echo_the_greater_name_perk } 
			}
			set_while_counter_variable_effect = yes
			while = {
				limit = { var:while_counter < echo_the_greater_name_perk_max }
				scope:target = { 
					random_vassal_or_below = { 
						limit = {
							NOT = { scope:owner = { has_hook = prev } } 
						}
						scope:owner = { 
							add_hook = {
								target = prev
								type = partial_true_name_hook
							}
						}
					}
				}
				increase_while_counter_variable_effect = yes
			}
			remove_while_counter_variable_effect = yes
		}
		scope:scheme = {
			end_scheme = yes
		}
	}
}
