scripted_trigger POD_inquihunt_scheme_1_trigger = {
	any_scheme = {
		scheme_type = inquihunt
		scheme_target_character = {
			this = scope:defender
		}
	}
}

scripted_effect POD_end_inquihunt_contract_or_journey_effect = {
	if = {
		limit = { exists = scope:task_contract }
		scope:task_contract = {
			var:travel_plan ?= {
				if = {
					limit = { is_paused = yes }
					resume_travel_plan = yes
				}
			}
		}
		if = {
			limit = {
				any_character_active_contract = { scope:task_contract ?= this }
			}
			hidden_effect = {
				scope:task_contract = { complete_task_contract = failure_abandoned_hunt }
			}
		}
	}
	else_if = {
		limit = { exists = scope:scheme }
		scope:scheme = { end_scheme = yes }
	}
	else_if = {
		limit = {
			var:current_journey ?= {
				var:journey_type = flag:assault_lair
				var:target = scope:defender
			}
		}
		POD_end_journey_effect = yes
	}
}

scripted_effect POD_inquihunt_bonus_rewards_effect = {
	save_scope_value_as = {
		name = multiplier
		value = {
			if = {
				limit = { exists = scope:task_contract }
				if = {
					limit = { scope:task_contract.task_contract_tier = tier_empire }
					value = 1
				}
				else_if = {
					limit = { scope:task_contract.task_contract_tier = tier_kingdom }
					value = 0.75
				}
				else_if = {
					limit = { scope:task_contract.task_contract_tier = tier_duchy }
					value = 0.5
				}
				else = {
					value = 0.25
				}
			}
			else = {
				value = 1
			}
		}
	}
	add_prestige = {
		value = $PRESTIGE$
		multiply = scope:multiplier
	}
	add_piety = {
		value = $PIETY$
		multiply = scope:multiplier
	}
}

scripted_effect POD_inquihunt_success_effect = {
	POD_inquihunt_success_side_effects = yes
	
	scope:defender = {
		pod_death = {
			death_reason = $DEATH_REASON$
			killer = scope:attacker
		}
	}
	
	hidden_effect = {
		scope:task_contract ?= { complete_task_contract = success_standard }
	}
	
	POD_end_inquihunt_contract_or_journey_effect = yes
}

# Journey wrapper event for the old inquihunt scheme
POD_inquihunt.7000 = {
	hidden = yes
	
	immediate = {
		if = {
			limit = { var:current_journey.var:target = { is_alive = no } }
			POD_clear_journeys_with_target_effect = { TARGET = var:current_journey.var:target }
		}
		else_if = {
			limit = {
				var:current_journey.var:target = {
					highest_held_title_tier >= tier_kingdom
					is_demon_trigger = no
				}
			}

			trigger_event = POD_heists.3
		}
		else = {
			POD_pause_journey_travel_plan_effect = yes

			save_scope_as = attacker
			var:current_journey.var:target = {
				save_scope_as = defender
			}
			
			trigger_event = POD_inquihunt.7002
		}
	}
}

scripted_effect POD_age_up_bloodsucker_for_contract = {
	if = {
		limit = { scope:task_contract.task_contract_tier = tier_empire }
		set_age = { 1000 2000 }
	}
	else_if = {
		limit = { scope:task_contract.task_contract_tier = tier_kingdom }
		set_age = { 400 999 }
	}
	else_if = {
		limit = { scope:task_contract.task_contract_tier = tier_duchy }
		set_age = { 200 399 }
	}
	else = {
		set_age = { 18 199 }
	}
}

# task_contract wrapper event for the old inquihunt scheme
POD_inquihunt.7001 = {
	hidden = yes
	
	immediate = {
		POD_pause_task_contract_travel_plan = yes
		
		save_scope_as = attacker
		
		create_character = {
			culture = scope:task_contract.task_contract_employer.culture
			faith = scope:task_contract.task_contract_employer.faith
			dynasty = none
			location = root.location
			age = { 18 60 }
			gender_female_chance = 50
			save_scope_as = defender
			after_creation = {
				POD_set_resonance_effect = yes
			}
		}
		
		hidden_effect_new_object = {
			scope:defender = {
				if = {
					limit = {
						scope:task_contract.task_contract_employer = {
							is_werewolf_trigger = yes
						}
					}
					add_trait = werewolf
					POD_werewolf_initialization_effect = yes
				}
				else_if = {
					limit = {
						scope:task_contract.task_contract_employer = {
							is_fae_trigger = yes
						}
					}
					add_trait = fae
					POD_custom_copy_kith_traits_effect = {
						ORIGIN = scope:task_contract.task_contract_employer
					}
					POD_fae_initialization_effect = yes
				}
				else_if = {
					limit = {
						scope:task_contract.task_contract_employer = {
							is_kueijin_trigger = yes
						}
					}
					add_trait = kueijin
					POD_age_up_bloodsucker_for_contract = yes
					POD_kueijin_initialization_effect = yes
				}
				else = {
					add_trait = vampire
					add_trait = vampgeneration8
					POD_age_up_bloodsucker_for_contract = yes
					add_character_flag = immune_to_disease
					POD_character_vampire_clan_initialization_effect = yes
					POD_character_initialization_random_background_effect = yes
					POD_character_vampire_discipline_initialization_effect = yes
					POD_character_vampire_clan_traits_effect = yes
					POD_character_vampire_faith_initialization_effect = yes
					POD_vampires_look_younger_effect = yes
					trigger_event = POD_character_canon_modifier_initalization.100
				}
				
				if = {
					limit = {
						scope:task_contract.task_contract_employer = {
							is_demon_trigger = yes
						}
					}
					give_infernalist_secret_or_trait_effect = yes
					give_cultist_secret_or_trait_effect = yes
				}
			}
		}
		
		trigger_event = POD_inquihunt.7002
	}
}

# trigger the right event by splat
POD_inquihunt.7002 = {
	hidden = yes
	immediate = {
		if = {
			limit = { scope:defender = { is_werewolf_trigger = yes } }
			trigger_event = POD_inquihunt.50
		}
		else_if = {
			limit = { scope:defender = { is_demon_trigger = yes } }
			trigger_event = POD_inquihunt.80
		}
		else_if = {
			limit = { scope:defender = { is_fae_trigger = yes } }
			trigger_event = POD_inquihunt.110
		}
		else = {
			# Fallback for Vampires, Kuei-Jin, etc
			trigger_event = POD_inquihunt.1
		}
	}
}
