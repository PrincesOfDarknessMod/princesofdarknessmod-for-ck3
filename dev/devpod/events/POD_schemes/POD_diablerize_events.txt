namespace = POD_diablerize

scripted_trigger POD_diablerie_scheme_100_trigger = {
	any_scheme = {
		scheme_type = diablerize
		scheme_target = {
			this = scope:defender
		}
	}
}

POD_diablerize.100 = {
	type = character_event
	title = POD_diablerize.100.t
	desc =  POD_diablerize.100.desc		
	theme = vampire
	left_portrait = {
		character = root
		animation = anger
	}
	
	right_portrait = {
		character = scope:scheme.scheme_target
		animation = fear
	}
	
	immediate = {
	
		scope:scheme.scheme_target = {
			save_scope_as = defender	
		}
		
		scope:scheme.scheme_owner= {
			save_scope_as = attacker
		}
		
		scope:defender = {
			location = { save_scope_as = diablerie_location }
			#capital_province = { save_scope_as = diablerie_location }
		}
	}
	
# Proceed with the Attack
	option = {
		name = POD_diablerize.100.option.a
		custom_tooltip = POD_diablerize.100.option.a.tt
		scope:defender = {
			trigger_event = {
				id = POD_diablerize.200
				days = 7
			}
		}
		POD_hunger_rouse_check_effect = yes 
		POD_willpower_rouse_check_effect = yes
		if = {
			limit = { POD_has_a_beast_trigger = yes }
			custom_tooltip = POD_diablerize.100.option.a.tt.beast
		}
		
		ai_chance = {
			base = 50
			modifier = {
				factor = 2
				is_diablerist_trigger = yes
			}
			modifier = {
				factor = 1.5
				has_trait = ambitious
			}			
		}
	}

# Give up	
	option = {
		name = POD_diablerize.100.option.b
		
		stress_impact = {
			ambitious = medium_stress_impact_gain
			#compassionate = medium_stress_impact_loss
		}		
			
		scope:attacker = {
			scope:scheme = {
				end_scheme = yes
			}
		}	
		
		if = {
			limit = { POD_has_a_beast_trigger = yes }
			custom_tooltip = POD_diablerize.100.option.b.tt.beast
		}
		
		ai_chance = {
			base = 0
			modifier = {
				factor = 2
				has_trait = compassionate
			}	
			modifier = {
				factor = 1.25
				has_trait = craven
			}	
			modifier = {
				factor = 1.5
				is_diablerist_trigger = no
			}
		}
	}

	after = {
		trigger_event = {
			id = POD_diablerize.3000
			days = 150
		}
	}
}
