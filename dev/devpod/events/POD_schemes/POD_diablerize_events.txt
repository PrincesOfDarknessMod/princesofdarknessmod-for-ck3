namespace = POD_diablerize

scripted_trigger POD_diablerie_scheme_100_trigger = {
	any_scheme = {
		scheme_type = diablerize
		scheme_target_character = {
			this = scope:defender
		}
	}
}

POD_diablerize.100 = {
	type = character_event
	content_source = dlc_POD
	title = POD_diablerize.100.t
	desc =  POD_diablerize.100.desc		
	theme = vampire
	left_portrait = {
		character = root
		animation = anger
	}
	
	right_portrait = {
		character = scope:scheme.scheme_target_character
		animation = fear
	}
	
	widget = {
		gui = "POD_shader_event_fog_alley_night"
		container = "background_shader_vfx_container"
	}
	
	immediate = {
	
		scope:scheme.scheme_target_character = {
			save_scope_as = defender	
		}
		
		scope:scheme.scheme_owner= {
			save_scope_as = attacker
		}
		
		scope:defender = {
			location = { save_scope_as = diablerie_location }
			#capital_province = { save_scope_as = diablerie_location }
		}
	}
	
# Proceed with the Attack
	option = {
		name = POD_diablerize.100.option.a
		custom_tooltip = POD_diablerize.100.option.a.tt
		scope:defender = {
			trigger_event = {
				id = POD_diablerize.200
				days = 7
			}
		}
		POD_hunger_rouse_check_effect = yes 
		POD_willpower_rouse_check_effect = yes
		if = {
			limit = { POD_has_a_beast_trigger = yes }
			custom_tooltip = POD_diablerize.100.option.a.tt.beast
		}
		
		ai_chance = {
			base = 50
			modifier = {
				factor = 2
				is_diablerist_trigger = yes
			}
			modifier = {
				factor = 1.5
				has_trait = ambitious
			}			
		}
	}

# Give up	
	option = {
		name = POD_diablerize.100.option.b
		
		stress_impact = {
			ambitious = medium_stress_impact_gain
			#compassionate = medium_stress_impact_loss
		}		
			
		scope:attacker = {
			scope:scheme = {
				end_scheme = yes
			}
		}	
		
		if = {
			limit = { POD_has_a_beast_trigger = yes }
			custom_tooltip = POD_diablerize.100.option.b.tt.beast
		}
		
		ai_chance = {
			base = 0
			modifier = {
				factor = 2
				has_trait = compassionate
			}	
			modifier = {
				factor = 1.25
				has_trait = craven
			}	
			modifier = {
				factor = 1.5
				is_diablerist_trigger = no
			}
		}
	}

	after = {
		trigger_event = {
			id = POD_diablerize.3000
			days = 150
		}
	}
}

POD_diablerize.600 = {
	type = character_event
	content_source = dlc_POD
	title = POD_diablerize.600.t
	desc =  POD_diablerize.600.desc		
	theme = diablerie
	left_portrait = {
		character = scope:attacker
		animation = rage
	}
	
	right_portrait = {
		character = scope:defender
		animation = severelywounded
	}
	
	widget = {
		gui = "POD_shader_event_blood"
		container = "background_shader_vfx_container"
	}
	
	immediate = {

		scope:defender = {
			save_scope_as = victim	
		}
		
		scope:victim = {
			#POD_clear_resonance_effect = yes
			if = {
				limit = {
					NOT = {	has_character_flag = POD_resonance_negligible_flag }
					NOT = {	has_character_flag = POD_resonance_fleeting_flag }
					NOT = {	has_character_flag = POD_resonance_intense_flag }
					NOT = {	has_character_flag = POD_resonance_acute_flag }
					NOT = {	has_character_flag = choleric_resonance }
					NOT = {	has_character_flag = melancholic_resonance }
					NOT = {	has_character_flag = phlegmatic_resonance }
					NOT = {	has_character_flag = sanguine_resonance }
				}
				POD_set_resonance_effect = yes
			}
		}			
	}	
	
	# diablerize them
	option = {
		name = POD_diablerize.600.option.a
		custom_tooltip = POD_diablerize.600.option.a.tt
		
		POD_drain_victim_effect = { VICTIM = scope:defender }
		
		if = {
			limit = {
				exists = scope:scheme_discovered
				scope:victim = { 
					NOT = { has_character_modifier = sentenced_court_blood_modifier }
				}				
			}
			murder_consequences_known_or_exposed_effect = {
				MURDERER = scope:murderer
				VICTIM = scope:victim
				POV = root
			}
		}
	
		scope:scheme = {
			end_scheme = yes
		}
		
		if = {
			limit = { POD_has_a_beast_trigger = yes }
			custom_tooltip = POD_diablerize.600.option.a.tt.beast
		}

		ai_chance = {
			base = 90
			
			modifier = {
				factor = 1.5
				has_trait = vengeful
			}	
			modifier = {
				factor = 3
				has_trait = wrathful
			}	
			
			modifier = {
				factor = 3
				has_trait = ambitious
			}	
			
			modifier = {
				factor = 5
				is_diablerist_trigger = yes
			}
			
			modifier = {
				factor = 0.1
				has_trait = compassionate
			}
			modifier = {
				factor = 0.1
				faith = { has_doctrine = doctrine_enlightenment_virtues_conscience_and_instinct }
			}
			modifier = {
				factor = 0.1
				faith = { has_doctrine = doctrine_enlightenment_virtues_conscience_and_selfcontrol }
			}			
			modifier = {
				factor = 5
				faith = {
						has_doctrine = doctrine_vampirediablerie_accepted
				}
			}			
			modifier = {
				factor = 0.5
				faith = {
						has_doctrine = doctrine_vampirediablerie_crime
				}
			}	
			modifier = {
				factor = 0.7
				faith = {
						has_doctrine = doctrine_vampirediablerie_shunned
				}
			}			
		}
	}

	option = {
		name = POD_diablerize.600.option.c
		
		scope:attacker = {
			save_scope_as = murderer	
		}
		
		hidden_effect = {
			scope:victim = {
				murder_success_kill_effect = { REASON = death_mysterious }
			}
		}
		
		if = {
			limit = { POD_has_a_beast_trigger = yes }
			custom_tooltip = POD_diablerize.600.option.c.tt.beast
		}	
	
		scope:scheme = {
			end_scheme = yes
		}		
	}
	
	option = {
		name = POD_diablerize.600.option.b
		
		stress_impact = {
			ambitious = medium_stress_impact_gain
			#compassionate = medium_stress_impact_loss
		}		
		
		hidden_effect = {
			random = {
				chance = 10
				stress_impact = {
					compassionate = medium_stress_impact_loss
				}			
			}
		}
		
		if = {
			limit = { POD_has_a_beast_trigger = yes }
			custom_tooltip = POD_diablerize.600.option.b.tt.beast
		}

		hidden_effect = {
			scope:defender = {
				trigger_event = {
					id = POD_diablerize.650
					days = 7
				}
			}
		}
		
		scope:scheme = {
			end_scheme = yes
		}

		ai_chance = {
			base = 10
			modifier = {
				factor = 2
				has_trait = forgiving
			}	
			modifier = {
				factor = 1.25
				has_trait = arbitrary
			}
			modifier = {
				factor = 1.25
				has_trait = compassionate
			}	
			modifier = {
				factor = 1.1
				is_diablerist_trigger = no
			}
			modifier = {
				factor = 1.1
				faith = { has_doctrine = doctrine_enlightenment_virtues_conscience_and_selfcontrol }
			}	
			modifier = {
				factor = 0.1
				faith = { has_doctrine = doctrine_enlightenment_virtues_conscience_and_instinct }
			}			
			modifier = {
				factor = 0.1
				faith = {
						has_doctrine = doctrine_vampirediablerie_crime
				}
			}	
			modifier = {
				factor = 1.1
				faith = {
						has_doctrine = doctrine_vampirediablerie_shunned
				}
			}				
		}	
		
	}
