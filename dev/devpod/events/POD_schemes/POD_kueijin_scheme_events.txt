	type = character_event
	content_source = dlc_POD
	title = {

			triggered_desc = {

				}

	desc = {

	option = {
		name = kueijin_scheme.130.a
		random = {
			chance = 5
			add_diplomacy_lifestyle_xp = 50
		}
		random = {
			chance = 5
			add_martial_lifestyle_xp = 50
		}
		random = {
			chance = 5
			add_stewardship_lifestyle_xp = 50
		}
		random = {
			chance = 5
			add_intrigue_lifestyle_xp = 50
		}
		random = {
			chance = 5
			add_learning_lifestyle_xp = 50
		}
	}
	
	option = {
		name = kueijin_scheme.130.b
		trigger = {
			OR = {
				AND = {
					scope:actor = {
						has_trait = education_intrigue
					}
					scope:recipient = {
						has_trait_rank = {
							trait = education_intrigue
							character > scope:actor
						}
					}
				}
				AND = {
					scope:actor = {
						has_trait = education_diplomacy
					}
					scope:recipient = {
						has_trait_rank = {
							trait = education_diplomacy
							character > scope:actor
						}
					}
				}
				AND = {
					scope:actor = {
						has_trait = education_stewardship
					}
					scope:recipient = {
						has_trait_rank = {
							trait = education_stewardship
							character > scope:actor
						}
					}
				}
				AND = {
					scope:actor = {
						has_trait = education_martial
					}
					scope:recipient = {
						has_trait_rank = {
							trait = education_martial
							character > scope:actor
						}
					}
				}
				AND = {
					scope:actor = {
						has_trait = education_learning
					}
					scope:recipient = {
						has_trait_rank = {
							trait = education_learning
							character > scope:actor
						}
					}
				}
				AND = {
					scope:actor = {
						has_trait = education_martial_prowess
					}
					scope:recipient = {
						has_trait_rank = {
							trait = education_martial_prowess
							character > scope:actor
						}
					}
				}
			}
		}
		
		scope:actor = {
			add_stress = massive_stress_impact_gain
			if = {
				limit = {
					has_trait = education_intrigue
				}
				change_trait_rank = {
					trait = education_intrigue
					rank = 1
					max = 5
				}
			}
			if = {
				limit = {
					has_trait = education_diplomacy
				}
				change_trait_rank = {
					trait = education_diplomacy
					rank = 1
					max = 5
				}
			}
			if = {
				limit = {
					has_trait = education_stewardship
				}
				change_trait_rank = {
					trait = education_stewardship
					rank = 1
					max = 5
				}
			}
			if = {
				limit = {
					has_trait = education_martial
				}
				change_trait_rank = {
					trait = education_martial
					rank = 1
					max = 5
				}
			}
			if = {
				limit = {
					has_trait = education_learning
				}
				change_trait_rank = {
					trait = education_learning
					rank = 1
					max = 5
				}
			}
			if = {
				limit = {
					has_trait = education_martial_prowess
				}
				change_trait_rank = {
					trait = education_martial_prowess
					rank = 1
					max = 5
				}
			}
		}
	}

kueijin_scheme.200 = {
	type = character_event
	content_source = dlc_POD
	title = kueijin_scheme.200.t
	desc = kueijin_scheme.200.desc
	theme = corruption

	left_portrait = {
		character = root
		animation = anger
	}
	
	right_portrait = {
		character = scope:recipient
		animation = fear
	}
	
	immediate = {
		if = {
			limit = {
				exists = scope:scheme
			}
			scope:scheme.scheme_target_character = {
				save_scope_as = recipient	
			}
		}
		save_scope_as = caster
	}

	# Increase
	option = {
		name = kueijin_scheme.200.a
		
		scope:caster = {
			POD_kj_art_cost_effect = { COUNT = 1 }
		}
		
		trigger_event = kueijin_scheme.210
	}

	# Decrease
	option = {
		name = kueijin_scheme.200.b
		
		scope:caster = {
			POD_kj_art_cost_effect = { COUNT = 1 }
		}
		
		trigger_event = kueijin_scheme.220
	}
}

	hidden = yes
	
	immediate = {
		if = {
			limit = { scope:actor = scope:sc_victor }
			scope:actor = { trigger_event = kueijin_scheme.200 }
		}
		else_if = {
			limit = { exists = scope:scheme }
			scope:scheme = {
				end_scheme = yes
			}
		}
	}
}
