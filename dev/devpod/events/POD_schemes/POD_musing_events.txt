namespace = POD_musing

POD_musing.1 = {
	type = character_event
	title = POD_musing.1.t
	desc =  POD_musing.1.desc		
	theme = fae
	left_portrait = scope:target
	
	option = {
		name = POD_musing.1.option.a
		scope:target = {
			set_relation_dreamer_owner = scope:owner
			add_character_flag = blocked_from_leaving
			trigger_event = {
				id = POD_musing.2
				days = { min_musing_days max_musing_days }
			}
		}
		scope:owner = {
			add_hook = {
				type = dreamer_hook
				target = scope:target
			}
			add_courtier = scope:target
		}
	}

	after = {
		if = {
			limit = {
				exists = scope:scheme
			}
			scope:scheme = {
				end_scheme = yes
			}
		}
	}
}

POD_musing.2 = {
	type = character_event
	hidden = yes
	
	trigger = { num_of_relation_dreamer_owner > 0 }
	
	immediate = {
		if = {
			limit = { has_ep1_dlc_trigger = yes }
			POD_create_random_inspiration_effect = yes
			trigger_event = {
				id = POD_musing.3
				days = 5
			}
		}
		else = {
			add_character_modifier = gave_inspiring_speech
			trigger_event = {
				id = POD_musing.4
				days = 5
			}
		}
		trigger_event = {
			id = POD_musing.2
			days = { min_musing_days max_musing_days }
		}
	}
}

POD_musing.3 = {
	type = character_event
	hidden = yes
	
	trigger = { exists = inspiration }
	
	immediate = {
		every_relation = {
			type = dreamer_owner
			send_interface_message = {
				type = event_generic_good
				title = POD_dreamer_maintenance
				left_icon = this
				right_icon = root
				custom_tooltip = POD_dreamer_maintenance.tt
				add_realm_lifestyle_xp = 25
				add_common_arts_lifestyle_xp = 45
				if = {
					limit = { has_trait = kithain }
					add_kith_arts_lifestyle_xp = 45
				}
				if = {
					limit = { has_trait = phyla }
					add_slivers_lifestyle_xp = 45
				}
				if = {
					limit = { has_trait = hsien }
					add_wu_tan_lifestyle_xp = 45
				}
				random = {
					chance = 10
					POD_glamour_or_chi_increase_effect = yes
				}
				root = {
					destroy_inspiration = inspiration
					add_stress = minor_stress_gain
				}
			}
		}
	}
}
