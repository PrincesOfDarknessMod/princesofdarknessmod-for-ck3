namespace = POD_musing

POD_musing.1 = {
	type = character_event
	content_source = dlc_POD
	title = POD_musing.1.t
	desc =  POD_musing.1.desc		
	theme = fae
	left_portrait = scope:target
	
	option = {
		name = POD_musing.1.option.a
		scope:target = {
			set_relation_dreamer_owner = scope:owner
			add_character_flag = blocked_from_leaving
			trigger_event = {
				id = POD_musing.2
				days = { min_musing_days max_musing_days }
			}
			POD_become_enchanted_effect = yes
		}
		scope:owner = {
			add_hook = {
				type = dreamer_hook
				target = scope:target
			}
			add_courtier = scope:target
			scope:target = { hidden_effect = { return_to_court = yes } }
		}
	}

	after = {
		if = {
			limit = {
				exists = scope:scheme
			}
			scope:scheme = {
				end_scheme = yes
			}
		}
	}
}

POD_musing.2 = {
	hidden = yes
	
	trigger = { num_of_relation_dreamer_owner > 0 }
	
	immediate = {
		if = {
			limit = { has_ep1_dlc_trigger = yes }
			POD_create_random_inspiration_effect = yes
			trigger_event = {
				id = POD_musing.3
				days = 5
			}
		}
		else = {
			add_character_modifier = tour_smooth_talker_modifier
			trigger_event = {
				id = POD_musing.4
				days = 5
			}
		}
		trigger_event = {
			id = POD_musing.2
			days = { min_musing_days max_musing_days }
		}
	}
}

POD_musing.3 = {
	hidden = yes
	
	trigger = { exists = inspiration }
	
	immediate = {
		every_relation = {
			type = dreamer_owner
			send_interface_message = {
				type = event_generic_good
				title = POD_dreamer_maintenance
				left_icon = this
				right_icon = root
				custom_tooltip = POD_dreamer_maintenance.tt
				if = {
					limit = { POD_splat_has_access_to_fae_arts = yes }
					add_realm_lifestyle_xp = 50
					add_common_arts_lifestyle_xp = 100
				}
				if = {
					limit = { POD_splat_has_access_to_kith_arts = yes }
					add_kith_arts_lifestyle_xp = 50
				}
				if = {
					limit = { POD_splat_has_access_to_slivers = yes }
					add_slivers_lifestyle_xp = 50
				}
				if = {
					limit = { POD_splat_has_access_to_wu_tan = yes }
					add_wu_tan_lifestyle_xp = 50
				}
				random = {
					chance = 10
					modifier = {
						has_character_modifier = POD_useless_dreamers
						factor = 0
					}
					POD_glamour_income_effect = { COUNT = 1 }
				}
				root = {
					destroy_inspiration = inspiration
					add_stress = minor_stress_gain
				}
			}
		}
	}
}
