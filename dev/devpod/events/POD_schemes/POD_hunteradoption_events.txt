POD_hunteradoption.100 = {
	type = character_event
	content_source = dlc_POD
	title = POD_hunteradoption.100.t
	desc =  POD_hunteradoption.100.desc		
	theme = faith
	left_portrait = {
		character = root
		animation = personality_honorable
	}
	
	right_portrait = {
		character = scope:target
		animation = personality_bold
	}
	
	immediate = {
		scope:target = {
			save_scope_as = victim		 
		}
		
		root = {
			save_scope_as = vampiresire 
		}
		
		scope:victim = { POD_set_parent_effect = { PARENT = scope:vampiresire } }
	}
	
	option = {
		name = POD_hunteradoption.100.option.a
		custom_tooltip = POD_hunteradoption.101.option.c.tt

		
		stress_impact = {
			base = medium_stress_gain
		}

		scope:target = {
			if = {
				limit = { scope:vampiresire.dynasty ?= { has_dynasty_perk = unity_legacy_4 } }
			add_martial_skill = 4
			}
			else = {
			add_martial_skill = 2
			}
			remove_trait = mortal
			remove_trait = turnedhunter
			# Remove bastard traits
			if = {
				limit = {
					has_any_bastard_trait_trigger = yes
				}
				remove_all_bastard_traits = yes
			}
			add_trait = supehunter
			if = {
				limit = {
				scope:vampiresire = { has_trait = order_member }
				scope:target = {
				NOT = { has_trait = order_member }
						}
					}
			add_trait = order_member
				}
			set_character_faith = scope:vampiresire.faith
			set_house = scope:vampiresire.house
			hidden_effect = {
			if = {
				limit = {
				has_orison_trigger = no
				}
				POD_give_orison_effect = yes
			}
			}
			if = {
				limit = {
					has_trait = sorcerer
				}
			remove_trait = sorcerer
			}
			if = {
				limit = { 
					POD_has_true_faith_trigger = no
					OR = { 
						POD_is_infernalist_trigger = no
						POD_splat_mixing_haight_mode = yes
					}
				}
				if = {
					limit = {
						NOT = { scope:vampiresire.dynasty = dynasty:13000011 }
						has_trait = theurgist
					}	
					remove_trait = theurgist
					add_trait = truefaith
				}
				else = {
					random = {
						chance = 5
						add_trait = truefaith
					}
				}
			}
			POD_remove_spouse_relation_effect = yes
			every_child = { add_trait = bastard }
		}		
		
		hidden_effect = {
			scope:scheme = {
				end_scheme = yes
			}
		}	
	}
