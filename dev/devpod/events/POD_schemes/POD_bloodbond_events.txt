POD_bloodbond.100 = {
	type = character_event
	title = POD_bloodbond.100.t
	desc =  POD_bloodbond.100.desc		
	theme = intrigue
	right_portrait = scope:scheme.scheme_owner
	left_portrait = scope:scheme.scheme_target
	
	
	option = {
		name = POD_bloodbond.100.option.a
		trigger = {
			stress < very_high_stress
		}
		
		POD_willpower_rouse_check_effect = yes
		
		save_scope_value_as = {
			name = mental_duel_defender_will
			value = flag:willpower_spent
		}
		
		scope:scheme.scheme_owner = {
			trigger_event = POD_bloodbond.200
		}
		
		ai_chance = {
			base = 95
			modifier = {
				add = 1000
				has_trait = paranoid
			}
		}	
	}

	option = {
		name = POD_bloodbond.100.option.b
		save_scope_value_as = {
			name = mental_duel_defender_will
			value = flag:willpower_unspent
		}
		scope:scheme.scheme_owner = {
			trigger_event = POD_bloodbond.200
		}		
	}	
	
	option = {
		name = POD_bloodbond.100.option.c
		scope:scheme.scheme_owner = {
			trigger_event = POD_bloodbond.300
		}
		
		stress_impact = {
			reclusive = minor_stress_gain
		}			
		ai_chance = {
			base = -20
			opinion_modifier = {
				opinion_target = scope:sender
			}			
			modifier = {
				add = -100
				has_trait = paranoid
			}
		}			
	}	
}

	type = character_event

	desc =  POD_bloodbond.600.desc		
	theme = intrigue
	right_portrait = scope:scheme.scheme_owner
	left_portrait = scope:scheme.scheme_target

	immediate = {
		scope:scheme.scheme_owner = {
			scope:scheme = {
				end_scheme = yes
			}
		}	
	}
	
	option = {
		name = POD_bloodbond.600.option.a

			add_opinion = {
				target = scope:scheme.scheme_owner
				opinion = -100
				modifier = hate_opinion
				}
			}
			
		ai_chance = {
			base = 100
		}			
	}
}
