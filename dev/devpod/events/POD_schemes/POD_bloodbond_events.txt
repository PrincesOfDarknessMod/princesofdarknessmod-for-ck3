POD_bloodbond.200 = {
	type = character_event
	content_source = dlc_POD
	title = POD_bloodbond.200.t
	desc =  POD_bloodbond.200.desc		
	theme = intrigue
	right_portrait = scope:scheme.scheme_owner
	left_portrait = scope:scheme.scheme_target_character

	immediate = {
		save_scope_value_as = {
			name = attacker_avg
			value = {
				value = scope:attacker.intrigue
				multiply = 2
				add = scope:attacker.prowess
				divide = 2
			}
		}
		save_scope_value_as = {
			name = defender_avg
			value = {
				value = scope:defender.intrigue
				multiply = 2
				add = scope:defender.prowess
				divide = 2
			}
		}
		save_scope_value_as = {
			name = stat_diff
			value = {
				value = scope:attacker_avg
				subtract = scope:defender_avg
			}
		}
	}
	option = {
		name = POD_bloodbond.200.option.a		
		duel = { #using duel for unique tt effects
			skills = { intrigue prowess } #not acutally used in calculations, but gives a nice tt
			value = scope:stat_diff #not used, but avoids errors and removes need for target
			45 = {
				desc = POD_bloodbond.200.a.success.desc
				max = 75
				modifier = {
					add = {
						value = scope:stat_diff
						multiply = 0.45
					}
				}
				# Success 
				hidden_effect = {
					scope:scheme.scheme_owner = {
						trigger_event = POD_bloodbond.300
					}
				}
				show_as_tooltip = {
					POD_bloodbond_effect = { 
						REGENT = scope:attacker
						THRALL = scope:defender
					}
				}	
			}
			55 = {
				desc = POD_bloodbond.200.a.failure.desc
				min = 25
				modifier = {
					add = {
						value = scope:stat_diff
						multiply = -0.45
					}
				}
				# Failure
				hidden_effect = {
					scope:scheme.scheme_owner = { 
						trigger_event = POD_bloodbond.500
					}
				}
				show_as_tooltip = {
					scope:defender = {
						add_opinion = {
							target = scope:attacker
							opinion = -100
							modifier = hate_opinion
						}
					}
				}
			}				
		}	
	}
	
	option = {
		name = POD_bloodbond.200.option.c

		#Cancel the scheme 
		scope:scheme = {
			end_scheme = yes
		}
		custom_tooltip = POD_bloodbond.200.option.c.tt
	}		
}

POD_bloodbond.600 = {
	type = character_event
	content_source = dlc_POD
	title = POD_bloodbond.600.t
	desc =  POD_bloodbond.600.desc		
	theme = intrigue
	right_portrait = scope:scheme.scheme_owner
	left_portrait = scope:scheme.scheme_target_character

	immediate = {
		scope:scheme.scheme_target_character = {
			add_opinion = {
				target = scope:scheme.scheme_owner
				opinion = -100
				modifier = hate_opinion
			}
		}
	}
	
	option = {
		name = POD_bloodbond.600.option.a

		scope:scheme.scheme_owner = {
			scope:scheme = {
				end_scheme = yes
			}
		}					
	}

	option = { #revenge
		name = POD_embrace_ongoing.1000.option.c
		progress_towards_rival_effect = {
			CHARACTER = scope:scheme.scheme_owner
			REASON = rival_attempted_bloodbond
			OPINION = 0
		}
		ai_chance = {
			base = 0
			modifier = {
				add = 20
				ai_vengefulness >= medium_positive_ai_value
				ai_vengefulness < very_high_positive_ai_value
			}
			modifier = {
				add = 60
				ai_vengefulness >= very_high_positive_ai_value
			}
		}
	}

	after = {
		scope:scheme.scheme_owner ?= {
			scope:scheme ?= {
				end_scheme = yes
			}
		}	
	}
}
