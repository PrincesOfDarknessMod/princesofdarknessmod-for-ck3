namespace = POD_skuldfulfilled

POD_skuldfulfilled.001 = {
    type = character_event
	content_source = dlc_POD
	title = POD_skuldfulfilled.001.t
	desc = POD_skuldfulfilled.001.desc
	theme = vampire
	
	left_portrait = {
	    character = root 
		animation = personality_cynical 
	}
	
	right_portrait = {
	    character = scope:scheme.scheme_target_character
		animation = fear
	}
	
	widget = {
		gui = "POD_shader_event_fog_alley_night"
		container = "background_shader_vfx_container"
	}
	
	immediate = {
	    scope:scheme.scheme_target_character = {
		     save_scope_as = defender
		}
		scope:scheme.scheme_owner = {
		    save_scope_as = attacker
		}
	}
	option = {
	    name = POD_skuldfulfilled.001.option.a
		custom_tooltip = POD_skuldfulfilled.001.option.a.tt
		
		duel = {
		    skill = learning
			value = 10
			desc = POD_skuldfulfilled.option.001.a.desc
			
			45 = {
			    desc = POD_skuldfulfilled.001.option.success
				
				compare_modifier = {
					value = scope:duel_value
					multiplier = 1.5
				}
				modifier = {
				    add = 30
					has_trait = oblivionadvanced
				}
				
				custom_tooltip = POD_skuldfulfilled.001.option.a.tt.attacker_wins
				scope:defender = {
				    if = {
					    limit = {
						    has_trait = ghoul
						}
						remove_trait = ghoul
					}
					if = {
					    limit = {
						    has_trait = revenant 
						}
						remove_trait = revenant
					}
					if = {
					    limit = {
						    has_POD_race_trigger = no
						}
						add_trait_force_tooltip = mortal
						if = {
							limit = { can_hold_title_as_mortal_trigger = no }
							POD_depose_effect = yes
						}
					}
                }
				
				hidden_effect = {
				  scope:defender = {
				    POD_remove_bloodpotency_modifiers_effect = yes
					POD_remove_masquerade_modifiers_effect = yes
					POD_remove_vampire_discipline_traits_effect = yes
					POD_remove_hunger_modifiers_effect = yes
					
				
					
				    if = {
					    limit = {
						    scope:defender.liege = {
							    has_hook_of_type = {
					                target = scope:defender
					                type = strong_bloodoath_hook
				                }
							}
						}
						scope:defender.liege = {
						    save_scope_as = master
							remove_hook = {
						        target = scope:defender
						        type = strong_bloodoath_hook
					        }
						}
						if = {
						    limit = {
							    has_relation_ghoul_domitor = scope:master
							}
							remove_relation_ghoul_domitor = scope:master
						}
					}
				    remove_character_flag = blocked_from_leaving
					remove_character_flag = blood_bonded
					if = {
					    limit = {
						    age > 99
						}
						death = { death_reason = death_old_age }
					}
				 }
				 if = {
				     limit = {
					     exists = scope:master
					 }
				 
				    scope:master = {
				        trigger_event = {
					        id = POD_skuldfulfilled.002
						    days = 2
					    }
				    }
			     }	
			    }
			}
			
			50 = {
			    desc = POD_skuldfulfilled.001.option.failure

				compare_modifier = {
					value = scope:duel_value
					multiplier = -1.5
				}
				modifier = {
				    add = 15
					has_trait = fortitude
				}
				
				
				custom_tooltip = POD_skuldfulfilled.001.option.a.tt.attacker_lost
			    scope:attacker = {
				    stress_impact = {
			            base = medium_stress_gain
		            }	
				}
			}
			
		}
		scope:scheme = {
			end_scheme = yes
		}
		POD_hunger_rouse_check_effect = yes 
	}
	option = {
	    name = POD_skuldfulfilled.001.option.b
		custom_tooltip = POD_skuldfulfilled.001.option.b.tt
		scope:scheme = {
			end_scheme = yes
		}
	}
}
