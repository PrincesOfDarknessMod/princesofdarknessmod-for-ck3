namespace = POD_ghoulify

POD_ghoulify.100 = {
	type = character_event
	title = POD_ghoulify.100.t
	desc =  POD_ghoulify.100.desc		
	theme = ghoul
	left_portrait = scope:ghoulthrall
	
	immediate = {
		scope:scheme.scheme_target = {
			save_scope_as = ghoulthrall
		}
		scope:scheme.scheme_owner = {
			save_scope_as = domitor
		}
		
	}
	
	option = 
	{
		name = POD_ghoulify.100.option.a
		
		if = {
			limit = { scope:domitor = { has_character_flag = drunk_grail_flag } }
			custom_tooltip = drink_grail_tt4

			scope:ghoulthrall = {
				add_prowess_skill = 5
				if = {
					limit = { NOT = { has_trait = physique_good } }
					add_trait = physique_good_1
				}
			}
		}
		add_prestige = -100
		
		POD_player_character_create_ghoul_effect = yes
		
		scope:ghoulthrall = {
			add_character_flag = blocked_from_leaving
			add_character_flag = blood_bonded 
			set_relation_ghoul_domitor = scope:domitor
		}
		
		scope:domitor = {
			add_hook = {
				target = scope:ghoulthrall
				type = strong_bloodoath_hook
			}
			add_courtier = scope:ghoulthrall
			POD_hunger_rouse_check_effect = yes
			POD_high_risk_of_increasing_masquerade_exposure = yes
			
			POD_give_regnant_secret_effect = yes
		}
		
		scope:scheme = {
			end_scheme = yes
		}
	}
	
	option = 
	{
		name = POD_ghoulify.100.option.b
		
		add_stress = minor_stress_gain
		
		scope:scheme = {
			end_scheme = yes
		}
	}
}

POD_ghoulify.200 = {
	type = character_event
	title = POD_ghoulify.200.t
	desc =  POD_ghoulify.200.desc		
	theme = ghoul
	left_portrait = scope:ghoulthrall
	
	immediate = {
		scope:scheme.scheme_target = {
			save_scope_as = ghoulthrall
		}
		scope:scheme.scheme_owner = {
			save_scope_as = domitor
		}
		var:bloodbag = {
			save_scope_as = bloodbag
		}
	}
	
	option = {
		name = POD_ghoulify.200.option.a
			
		POD_player_character_create_ghoul_effect = yes
		
		scope:domitor = {
			add_courtier = scope:ghoulthrall
			if = {
				limit = { NOT = { dynasty = dynasty:13000007 } }
				POD_small_risk_of_increasing_masquerade_exposure = yes
			}		
			
			if = {
				limit = { faith = {has_doctrine_parameter = vampirediablerie_illegal } }
				add_piety = massive_piety_loss
			}
			else_if = {
				limit = { faith = {has_doctrine_parameter = vampirediablerie_shunned } }
				add_piety = major_piety_loss
			}
		
			POD_give_diablerist_secret_or_trait_effect = yes
		}
		
		var:bloodbag = {
			death = {
	        	death_reason = death_diablerie
				killer = scope:domitor
	        }
		}		
		
		scope:scheme = {
			end_scheme = yes
		}
	}
	
	option = {
		name = POD_ghoulify.200.option.b
		
		add_stress = minor_stress_gain
		
		scope:scheme = {
			end_scheme = yes
		}
	}
}
