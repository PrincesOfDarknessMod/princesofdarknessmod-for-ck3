namespace = POD_ghoulify

POD_ghoulify.100 = {
	type = character_event
	content_source = dlc_POD
	title = POD_ghoulify.100.t
	desc =  POD_ghoulify.100.desc		
	theme = ghoul
	left_portrait = scope:ghoulthrall
	
	immediate = {
		scope:target = {
			save_scope_as = ghoulthrall
		}
		scope:owner = {
			save_scope_as = domitor
		}
		
	}
	
	option = {
		name = POD_ghoulify.100.option.a
		add_prestige = -100
		
		scope:ghoulthrall = {
			POD_become_ghoul_effect = yes
			POD_become_bound_ghoul_effect = { DOMITOR = scope:domitor }
		}
		
		scope:domitor = {
			add_courtier = scope:ghoulthrall
			scope:ghoulthrall = { hidden_effect = { return_to_court = yes } }
			POD_hunger_rouse_check_effect = yes
			POD_high_risk_of_increasing_masquerade_exposure = yes
			
			POD_give_regnant_secret_effect = yes
		}
	}
	
	option = {
		name = POD_ghoulify.100.option.b
		
		add_stress = minor_stress_gain
	}

	after = {
		if = {
			limit = {
				exists = scope:scheme
			}
			scope:scheme = {
				end_scheme = yes
			}
		}	
	}
}

POD_ghoulify.200 = {
	type = character_event
	content_source = dlc_POD
	title = POD_ghoulify.200.t
	desc =  POD_ghoulify.200.desc		
	theme = ghoul
	left_portrait = scope:ghoulthrall
	
	immediate = {
		scope:target = {
			save_scope_as = ghoulthrall
		}
		scope:owner = {
			save_scope_as = domitor
		}
		if = { 
			limit = { NOT = { exists = scope:bloodbag } }
			scope:owner = { 
				random_prisoner = {
					limit = { POD_is_vampire_trigger = yes }
					save_scope_as = bloodbag 
				}
			}
		}
	}
	
	option = {
		name = POD_ghoulify.200.option.a

		scope:ghoulthrall = {
			POD_become_ghoul_effect = yes
			POD_ghoul_new_generation_ingested_effect = { GENERATION = scope:bloodbag.POD_vampgeneration }
		}
		
		scope:domitor = {
			add_courtier = scope:ghoulthrall
			scope:ghoulthrall = { hidden_effect = { return_to_court = yes } }
			
			if = {
				limit = { NOT = { dynasty = dynasty:13000007 } }
				POD_small_risk_of_increasing_masquerade_exposure = yes
			}		
			
			if = {
				limit = { faith = {has_doctrine_parameter = vampirediablerie_illegal } }
				add_piety = massive_piety_loss
			}
			else_if = {
				limit = { faith = {has_doctrine_parameter = vampirediablerie_shunned } }
				add_piety = major_piety_loss
			}
		}
		
		scope:bloodbag = {
			death = {
	        		death_reason = death_bloodloss
				killer = scope:domitor
			}
		}
	}
	
	option = {
		name = POD_ghoulify.200.option.b
		
		add_stress = minor_stress_gain
	}

	after = {
		if = {
			limit = {
				exists = scope:scheme
			}
			scope:scheme = {
				end_scheme = yes
			}
		}	
	}
}
