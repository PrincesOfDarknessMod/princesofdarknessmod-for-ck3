scripted_effect ai_embrace_target = {
	hidden_effect = {
		scope:target = { save_scope_as = victim	}
		this = { save_scope_as = vampiresire }
		scope:victim = { POD_set_resonance_effect = yes	}	
		scope:target = { save_scope_as = created_vampirechilde }
		scope:created_vampirechilde = {
			random_relation = {
				type = ghoul_domitor
				limit = { is_alive = yes }
				save_scope_as = former_domitor
			} 
		}
		ai_embrace_2 = yes
	}
	custom_tooltip = POD_embrace_ongoing.tt.ai_embraces_target
}
scripted_effect ai_embrace_2 = {
	POD_hunger_feeding_drain_effect = yes
	POD_small_risk_of_increasing_masquerade_exposure = yes	
	POD_resonance_feeding_drain_effect = yes
	POD_hunger_rouse_check_effect = yes
	POD_player_character_create_vampirechilde_effect = yes
	scope:created_vampirechilde = {
		every_relation = {
			type = herd_owner
			remove_hook = {
				target = scope:created_vampirechilde
				type = herd_hook
			}
			scope:created_vampirechilde = {
				remove_relation_herd_owner = prev
				remove_character_flag = blocked_from_leaving
			}
		}
	}
	if = {
		 limit = { exists = scope:former_domitor }
		scope:created_vampirechilde = {
			remove_relation_ghoul_domitor = scope:former_domitor
			remove_character_flag = blocked_from_leaving
			remove_character_flag = blood_bonded
		}
		 scope:former_domitor = {
			remove_hook = {
				 target = scope:created_vampirechilde
				 type = strong_bloodoath_hook
			}
		}
	}
	scope:created_vampirechilde = {
		set_employer = scope:vampiresire
		return_to_court = yes
	}
}
scripted_trigger appropriately_evil_rival = {
	is_available_ai_adult = yes
	ai_compassion <= low_positive_ai_value
	is_ruler = yes
	#POD_is_vampire_trigger = yes #not needed as any splat now has option to foil your hunt
	trigger_if = {
		limit = { #make sure vassals don't get stupid
			OR = {
				is_vassal_or_below_of = root
				is_courtier_of = root
			}
		}
		dread_modified_ai_boldness = {
			dreaded_character = root
			value >= very_high_positive_ai_value
		}
	}
	trigger_else = {
		dread_modified_ai_boldness = {
			dreaded_character = root
			value >= very_low_positive_ai_value
		}
	}
}

					add = 5

						value = ai_vengefulness

					}

				}
			}
			save_scope_as = rival
		}
		random_list = {
			#kill target
			0 = {
				scope:scheme = { set_variable = target_killed }
				if = {
					limit = { scope:rival = { POD_is_vampire_trigger = yes }}
					show_as_tooltip = {
						scope:target = {
							pod_death_effect = {

							}
						}
					}

						}

					add = {
						value = scope:rival.ai_compassion
						divide = -2
					}
				}
			}

			80 = {
				scope:scheme = { set_variable = target_kidnapped }
				scope:rival = {
					imprison = {
						target = scope:target
						type = dungeon
					}
				}

					add = -50

						scope:rival = { POD_is_vampire_trigger = no }
						scope:rival.faith = { has_doctrine_parameter = willing_embrace }
					}
				}
			}
		}
		add_prestige = minor_prestige_loss
		scope:rival = {
			add_stress = major_stress_impact_loss
		}
		scope:target = { save_scope_as = relationship_reason_involved_character }
	}

			skill = stewardship

					right_icon = scope:rival

						target = scope:rival

						base = minor_stress_impact_gain
						greedy = minor_stress_impact_gain
					}
				}
			}
			12 = {
				modifier = {
					add = {
						value = scope:duel_value
						multiply = tiny_negative_duel_skill_multiplier
					}
				}
				send_interface_toast = {
					left_icon = root

			id = POD_embrace_ongoing.1001 #delivery of kidnapped target

	option = { #just a mortal
		name = POD_embrace_ongoing.1000.option.b
		stress_impact = {
			wrathful = minor_stress_impact_gain
			impatient = minor_stress_impact_gain
			compassionate = minor_stress_impact_gain
			vengeful = medium_stress_impact_gain
			callous = miniscule_stress_impact_loss
		}
	}
	option = { #vow revenge
		name = POD_embrace_ongoing.1000.option.c
		if = {
			limit = { scope:scheme = { has_variable = target_kidnapped }}
			progress_towards_rival_effect = {
				CHARACTER = scope:rival
				OPINION = no
				REASON = rival_kinapped_potential_childe
			}
		}
		else_if = {
			limit = { scope:scheme = { has_variable = target_embraced }}
			progress_towards_rival_effect = {
				CHARACTER = scope:rival
				OPINION = no
				REASON = rival_embraced_potential_childe

			limit = { scope:scheme = { has_variable = target_killed }}

scripted_effect AI_start_rivalry_chance_effect = { #use from AI scope
	if = {
		limit = {
			ai_vengefulness >= medium_positive_ai_value
			ai_vengefulness < very_high_positive_ai_value
		}
		random = {
			chance = 20
			progress_towards_rival_effect = {
				CHARACTER = root
				REASON = $REASON$
				OPINION = no
			}
		}
	}
	else_if = {
		limit = {
			ai_vengefulness >= very_high_positive_ai_value
		}

scripted_effect turn_ai_into_supehunter = {
	#if liege is hunter use adoption events without making liege parent
	if = {
		limit = { liege_or_court_owner = { POD_is_supehunter_trigger = yes }}
		remove_trait = mortal
		add_trait = supehunter
		if = {
			limit = {
				liege_or_court_owner = { has_trait = order_member }
				NOT = { has_trait = order_member }
			}
			add_trait = order_member
		}
		set_character_faith = liege_or_court_owner.faith
		set_house = liege_or_court_owner.house
		if = {
			limit = {
				has_trait = sorcerer
			}
			remove_trait = sorcerer
		}
		if = {
			limit = {
				has_trait = theurgist
				NOT = { has_trait = truefaith }
			}	
			remove_trait = theurgist
			add_trait = truefaith
		}
		else_if = {
			limit = {
				has_trait = theurgist
				has_trait = truefaith
			}	
			remove_trait = theurgist
		}
		if = {
			limit = {
				NOT = { has_trait = truefaith }
			}
			random = {
				chance = 10
				add_trait = truefaith
			}
		}
	}
	else = {
		#else make independent/regional hunter
		POD_character_initialization_for_supehunter_commoners_effect = yes
		#add chance to be order memeber
		if = {
			limit = {
				NOT = { has_trait = order_member }
			}
			random = {
				chance = 10
				add_trait = order_member
			}
		}
	}
}
