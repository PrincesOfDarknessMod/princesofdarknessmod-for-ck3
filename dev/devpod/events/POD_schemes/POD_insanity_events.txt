			desc = POD_insanity.001.option.a.desc

					value = scope:duel_value
					multiplier = high_positive_duel_skill_multiplier
				}
				modifier = {
					add = 10
					scope:attacker = { has_trait = dominateadvanced }
				}
				modifier = {
					add = 10

							has_perk = submerged_directive_perk

					add = 10

					}

				scope:defender = {
				    add_trait = lunatic_1
				}
				
				send_interface_toast = {
					title = POD_insanity.toast.success
					left_icon = root
					right_icon = scope:defender 
				}				
				
			}
			
			50 = {
			    desc = POD_insanity.001.option.failure

				compare_modifier = {
					value = scope:duel_value
					multiplier = low_negative_duel_skill_multiplier
				}
				
				modifier = {
					add = 10
					scope:defender = { has_trait = fortitudeadvanced }
				}
				modifier = {
					add = 20
					scope:defender = { 
						OR = {
							has_perk = unswayable_mind_perk
							has_trait = fortitudeadvanced
						}
					}
				}

					add = 25
					scope:defender = { 
						OR = {
							has_perk = fortify_the_inner_facade_perk

					}

				    stress_and_fulfillment_impact = {
