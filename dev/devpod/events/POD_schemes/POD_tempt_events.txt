namespace = POD_tempt_scheme

#Rolls success and discovery, and triggers on_actions (or sends "player's choice" event)
POD_tempt_scheme.0002 = {
	hidden = yes
	orphan = yes # obsolete after the RTP scheme rework

	immediate = {
		#SUCCESS ROLL
		random = {
			chance = scope:scheme.scheme_success_chance

			save_scope_value_as = {
				name = scheme_successful
				value = yes
			}
		}

		#DISCOVERY ROLL
		save_scope_value_as = {
			name = discovery_chance
			value = {
				value = 100
				subtract = scope:scheme.scheme_secrecy
			}
		}

		random = {
			chance = scope:discovery_chance
			save_scope_value_as = {
				name = scheme_discovered
				value = yes
			}
		}

			if = {
				limit = {
					exists = scope:scheme_successful
				}
				trigger_event = POD_tempt_scheme.0003
			}
			else = {
				trigger_event = POD_tempt_scheme.0004
			}
		}
	
}

POD_tempt_scheme.0003 = { 
	type = character_event
	content_source = dlc_POD
	title = POD_tempt_scheme.0003.t
	desc = POD_tempt_scheme.0003.desc
	theme = seduce_scheme
	left_portrait = scope:owner
	right_portrait = scope:target
	
	immediate = {
		scope:owner = { scope:scheme = { end_scheme = yes } }
		scope:owner = { add_stress = medium_stress_loss }
	}
	
	option = {
		name = POD_tempt_scheme.0003.a

	if = {
		limit = {
			scope:owner = { faith = { has_doctrine = special_doctrine_vices_avarice } }
			scope:target = { NOT = { has_trait = greedy } }
		}
		if = {
			limit = {
				scope:target = { has_trait = generous }
			}
				scope:target = { remove_trait = generous }
				scope:target = { add_trait_force_tooltip = greedy }
		}
		else = {
				scope:target = { add_trait = greedy }
		}
	}		
	else_if = {
		limit = {
			scope:owner = { faith = { has_doctrine = special_doctrine_vices_envy } }
			scope:target = { NOT = { has_trait = vengeful } }
		}
		if = {
			limit = {
				scope:target = { has_trait = forgiving }
			}
				scope:target = { remove_trait = forgiving }
				scope:target = { add_trait_force_tooltip = vengeful }
		}
		else = {
				scope:target = { add_trait = vengeful }
		}
	}				
	else_if = {
		limit = {
			scope:owner = { faith = { has_doctrine = special_doctrine_vices_gluttony } }
			scope:target = { NOT = { has_trait = gluttonous } }
		}
		if = {
			limit = {
				scope:target = { has_trait = temperate }
			}
				scope:target = { remove_trait = temperate }
				scope:target = { add_trait_force_tooltip = gluttonous }
		}
		else = {
				scope:target = { add_trait = gluttonous }
		}
	}			
	else_if = {
		limit = {
			scope:owner = { faith = { has_doctrine = special_doctrine_vices_lust } }
			scope:target = { NOT = { has_trait = lustful } }
		}
		if = {
			limit = {
				scope:target = { has_trait = chaste }
			}
				scope:target = { remove_trait = chaste }
				scope:target = { add_trait_force_tooltip = lustful }
		}
		else = {
				scope:target = { add_trait = lustful }
		}
	}	
	else_if = {
		limit = {
			scope:owner = { faith = { has_doctrine = special_doctrine_vices_pride } }
			scope:target = { NOT = { has_trait = arrogant } }
		}
		if = {
			limit = {
				scope:target = { has_trait = humble }
			}
				scope:target = { remove_trait = humble }
				scope:target = { add_trait_force_tooltip = arrogant }
		}
		else = {
				scope:target = { add_trait = arrogant }
		}
	}	
	else_if = {
		limit = {
			scope:owner = { faith = { has_doctrine = special_doctrine_vices_sloth } }
			scope:target = { NOT = { has_trait = lazy } }
		}
		if = {
			limit = {
				scope:target = { has_trait = diligent }
			}
				scope:target = { remove_trait = diligent }
				scope:target = { add_trait_force_tooltip = lazy }
		}
		else = {
				scope:target = { add_trait = lazy }
		}
	}
	else_if = {
		limit = {
			scope:owner = { faith = { has_doctrine = special_doctrine_vices_wrath } }
			scope:target = { NOT = { has_trait = wrathful } }
		}
		if = {
			limit = {
				scope:target = { has_trait = calm }
			}
				scope:target = { remove_trait = calm }
				scope:target = { add_trait_force_tooltip = wrathful }
		}
		else = {
				scope:target = { add_trait = wrathful }
		}
	}
	}
}
	
POD_tempt_scheme.0004 = {

	desc = POD_tempt_scheme.0004.desc

	left_portrait = scope:owner
	right_portrait = scope:target

	immediate = {

	}
	
	option = {
		name = POD_tempt_scheme.0004.a

	}
}

POD_tempt_scheme.0006 = { 
	type = character_event
	content_source = dlc_POD
	title = POD_tempt_scheme.0006.t
	desc = POD_tempt_scheme.0006.desc
	theme = seduce_scheme
	left_portrait = scope:owner
	right_portrait = scope:target

	immediate = {
		scope:owner = { scope:scheme = { end_scheme = yes } }
		scope:owner = { add_stress = major_stress_loss }
		if = {
			limit = {
				scope:owner  = {
					is_ai = no
					POD_is_demon_trigger = yes
				}
				scope:target = { highest_held_title_tier >= tier_kingdom }
			}
			set_global_variable = pod_achiev_demontutorial # Achievement: Master of Puppets
		}
	}
	
	option = {
		name = POD_tempt_scheme.0006.a
		
		scope:target = { set_character_faith = scope:owner.faith }
	}
}
