POD_auspex.100 = {
	type = character_event
	content_source = dlc_POD
	title = POD_auspex.100.t
	desc =  POD_auspex.100.desc
	theme = auspex
	right_portrait = scope:actor
	left_portrait = scope:recipient
	
	
	option = {
		name = POD_auspex.100.option.a
		trigger = {
			stress < very_high_stress
		}
		
		POD_willpower_rouse_check_effect = yes
		
		save_scope_value_as = {
			name = mental_duel_defender_will
			value = flag:willpower_spent
		}
		
		scope:actor = {
			trigger_event = POD_auspex.200
		}
		
		ai_chance = {
			base = 95
			modifier = {
				add = 1000
				has_trait = paranoid
			}
		}	
	}

	option = {
		name = POD_auspex.100.option.b
		save_scope_value_as = {
			name = mental_duel_defender_will
			value = flag:willpower_unspent
		}
		scope:actor = {
			trigger_event = POD_auspex.200
		}		
	}	
	
	option = {
		name = POD_auspex.100.option.c
		scope:actor = {
			trigger_event = POD_auspex.300
		}
		
		stress_and_fulfillment_impact = {
			reclusive = minor_stress_gain
		}			
		ai_chance = {
			base = 5
			opinion_modifier = {
				opinion_target = scope:actor
			}			
			modifier = {
				add = -100
				has_trait = paranoid
			}
		}			
	}	
}
