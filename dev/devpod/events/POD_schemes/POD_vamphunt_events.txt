namespace = POD_vamphunt

POD_vamphunt.1 =
{
	type = character_event
	title = POD_vamphunt.1.t
	desc =  POD_vamphunt.1.desc
	theme = vamphunt

	left_portrait = {
		character = root
		animation = anger
	}	
	
	right_portrait = {
		character = scope:target
		animation = fear
	}	
	
	immediate = {
		scope:target = {
			save_scope_as = victim	
		}
		
		scope:victim = {
			if = {
				limit = {
					NOT = {	has_character_flag = POD_resonance_negligible_flag }
					NOT = {	has_character_flag = POD_resonance_fleeting_flag }
					NOT = {	has_character_flag = POD_resonance_intense_flag }
					NOT = {	has_character_flag = POD_resonance_acute_flag }
					NOT = {	has_character_flag = choleric_resonance }
					NOT = {	has_character_flag = melancholic_resonance }
					NOT = {	has_character_flag = phlegmatic_resonance }
					NOT = {	has_character_flag = sanguine_resonance }
				}
				POD_set_resonance_effect = yes
			}
		}
	}
	
	# Take no blood
	option = 
	{
		name = POD_vamphunt.1.option.a
		
		stress_impact = {
			gluttonous = medium_stress_impact_gain
			#compassionate = medium_stress_impact_loss
		}	

		if = {
			limit = {
				has_trait = compassionate
				}
		random = {
			chance = 10
			stress_impact = {
				compassionate = miniscule_stress_impact_loss
			}			
		}
		}
		
		scope:scheme = {
			end_scheme = yes
		}		
	}	
	
	# A sip
	option = {
		name = POD_vamphunt.1.option.b

		if = {
			limit = {
				faith = {
					has_doctrine_parameter = vampirism_illegal
				}
			}
			add_piety = minor_piety_loss
		}

		# Resolve Blood Hunger Slaking from Sip
		if = {
			limit = { scope:victim = { POD_chi_user_trigger = no } }
			POD_hunger_feeding_sip_effect = yes
			POD_resonance_feeding_sip_effect = yes
		}	
		else = {
			if = {
				limit = { scope:victim = { chi_points_value >= 2 } }
				scope:victim = { POD_chi_decrease_effect = { COUNT = 2 } }
				POD_hunger_feeding_sip_effect = yes
				POD_resonance_feeding_sip_effect = yes 
			}
		}

		# The Victim
		scope:victim = {
			add_character_modifier = {
				modifier = slightbloodloss_modifier
				days = 30	
			}
		}
		
		scope:scheme = {
			end_scheme = yes
		}	
	}
	
	# A drink
	option = {
		name = POD_vamphunt.1.option.c

		if = {
			limit = {
				faith = {
					has_doctrine_parameter = vampirism_illegal
				}
			}
			add_piety = major_piety_loss
		}

		if = {
			limit = {
				faith = {
					has_doctrine_parameter = vampirism_shunned
				}
			}
			add_piety = medium_piety_loss
		}
		
		if = {
			limit = { scope:victim = { POD_chi_user_trigger = no } }
			POD_hunger_feeding_drink_effect = yes
			POD_resonance_feeding_drink_effect = yes
		}	
		else = {
			if = {
				limit = { scope:victim = { chi_points_value >= 4 } }
				scope:victim = { POD_chi_decrease_effect = { COUNT = 4 } }
				POD_hunger_feeding_drink_effect = yes
				POD_resonance_feeding_drink_effect = yes 
			}
			else_if = {
				limit = { scope:victim = { chi_points_value >= 2 } }
				scope:victim = { POD_chi_decrease_effect = { COUNT = 2 } }
				POD_hunger_feeding_sip_effect = yes
				POD_resonance_feeding_sip_effect = yes 
			}
		}

		#The Victim
		scope:victim = {
			add_character_modifier = {
				modifier = smallbloodloss_modifier
				months = 2	
			}
		}
		
		scope:scheme = {
			end_scheme = yes
		}
	}
	
	# A lot
	option = {
		name = POD_vamphunt.1.option.d

		if = {
			limit = {
				faith = {
					has_doctrine_parameter = vampirism_illegal
				}
			}
			add_piety = massive_piety_loss
		}

		if = {
			limit = {
				OR = {
					faith = {
						has_doctrine_parameter = vampirism_shunned
					}
					faith = {
						has_doctrine = doctrine_enlightenment_virtues_conscience_and_selfcontrol
					}
					faith = {
						has_doctrine = doctrine_enlightenment_virtues_conscience_and_instinct
					}
				}
			}
			add_piety = major_piety_loss
		}
		
		POD_minimal_risk_of_increasing_masquerade_exposure = yes

		stress_impact = {
			gluttonous = medium_stress_impact_loss
			compassionate = medium_stress_impact_gain
		}
		
		if = {
			limit = { scope:victim = { POD_chi_user_trigger = no } }
			POD_hunger_feeding_lot_effect = yes
			POD_resonance_feeding_lot_effect = yes
		}	
		else = {
			if = {
				limit = { scope:victim = { chi_points_value >= 6 } }
				scope:victim = { POD_chi_decrease_effect = { COUNT = 6 } }
				POD_hunger_feeding_lot_effect = yes
				POD_resonance_feeding_lot_effect = yes 
			}
			else_if = {
				limit = { scope:victim = { chi_points_value >= 4 } }
				scope:victim = { POD_chi_decrease_effect = { COUNT = 4 } }
				POD_hunger_feeding_drink_effect = yes
				POD_resonance_feeding_drink_effect = yes 
			}
			else_if = {
				limit = { scope:victim = { chi_points_value >= 2 } }
				scope:victim = { POD_chi_decrease_effect = { COUNT = 2 } }
				POD_hunger_feeding_sip_effect = yes
				POD_resonance_feeding_sip_effect = yes 
			}
		}

		scope:victim = {
			add_character_modifier = {
				modifier = majorbloodloss_modifier
				months = 7	
			}
		}

		scope:scheme = {
			end_scheme = yes
		}
	}
	
	# Kill them
	option = {
		name = POD_vamphunt.1.option.e

		if = {
			limit = {
				OR = {
					faith = {
						has_doctrine_parameter = vampirism_illegal
					}
					faith = {
						has_doctrine_parameter = vampirism_shunned
					}
					faith = {
						has_doctrine = doctrine_enlightenment_virtues_conscience_and_selfcontrol
					}
					faith = {
						has_doctrine = doctrine_enlightenment_virtues_conscience_and_instinct
					}
				}
			}
			add_piety = massive_piety_loss
		}
		
		POD_small_risk_of_increasing_masquerade_exposure = yes
		
		stress_impact = {
			gluttonous = major_stress_impact_loss
			compassionate = major_stress_impact_gain
		}	
		
		if = {
			limit = { scope:victim = { POD_chi_user_trigger = no } }
			POD_hunger_feeding_drain_effect = yes
			POD_resonance_feeding_drain_effect = yes
		}	
		else = {
			if = {
				limit = { scope:victim = { chi_points_value >= 10 } }
				scope:victim = { POD_chi_decrease_effect = { COUNT = 10 } }
				POD_hunger_feeding_drain_effect = yes
				POD_resonance_feeding_drain_effect = yes 
			}
			else_if = {
				limit = { scope:victim = { chi_points_value >= 6 } }
				scope:victim = { POD_chi_decrease_effect = { COUNT = 6 } }
				POD_hunger_feeding_lot_effect = yes
				POD_resonance_feeding_lot_effect = yes 
			}
			else_if = {
				limit = { scope:victim = { chi_points_value >= 4 } }
				scope:victim = { POD_chi_decrease_effect = { COUNT = 4 } }
				POD_hunger_feeding_drink_effect = yes
				POD_resonance_feeding_drink_effect = yes 
			}
			else_if = {
				limit = { scope:victim = { chi_points_value >= 2 } }
				scope:victim = { POD_chi_decrease_effect = { COUNT = 2 } }
				POD_hunger_feeding_sip_effect = yes
				POD_resonance_feeding_sip_effect = yes 
			}
		}
		
		save_scope_as = murderer
		scope:victim = { 
			if = {
				limit = {
					scope:murderer = { has_perk = cthonic_womb_perk }
				}
				set_variable = {
					name = marked_for_return
					value = scope:murderer
					days = 1
				}
				custom_tooltip = POD_cthonic_womb_perk_effect.tt
			}
			murder_success_kill_effect = yes
		}
	
		scope:scheme = {
			end_scheme = yes
		}
	}
	
	#Herd them
	option = {
		name = POD_vamphunt.1.option.f
		trigger = {
			OR = {
				has_perk = lingering_kiss_perk 
				has_trait = presenceadvanced
			}
			num_of_relation_herd_member < max_herd_cap
			scope:victim = { 
				POD_can_be_herd_trigger = yes
			}
		}
		show_as_unavailable = {
			custom_description = {
				text = POD_herdify_vampire_interaction_limit
				num_of_relation_herd_member >= max_herd_cap
			}
		}
		if = {
			limit = {
				faith = {
					has_doctrine_parameter = vampirism_illegal
				}
			}
			add_piety = major_piety_loss
		}

		if = {
			limit = {
				faith = {
					has_doctrine_parameter = vampirism_shunned
				}
			}
			add_piety = medium_piety_loss
		}

		#The Victim
		scope:victim = {
			add_character_modifier = {
				modifier = smallbloodloss_modifier
				months = 2	
			}
		}		
		
		POD_hunger_feeding_drink_effect = yes
		POD_resonance_feeding_drink_effect = yes
		scope:victim = {
			set_relation_herd_owner = root
			add_character_flag = blocked_from_leaving
			POD_herd_resonance_setup = yes

			#Saves the herdifying vampire as the herd_owner and victim as the herdmember
			root = { save_scope_as = herd_owner }
			scope:victim = { save_scope_as = herdmember }

			#Starts the autofeeding process by triggering the recurring feeding event.
			trigger_event = POD_herd_automation.100
		}
		root = {
			add_hook = {
				type = herd_hook
				target = scope:scheme.scheme_target
			}
		}
		
		
		scope:scheme = {
			end_scheme = yes
		}
	
	
	
	}
}
