namespace = POD_vamphunt

POD_vamphunt.1 = {
	type = character_event
	title = POD_vamphunt.1.t
	desc =  POD_vamphunt.1.desc
	theme = vamphunt

	left_portrait = {
		character = root
		animation = anger
	}	
	
	right_portrait = {
		character = scope:victim
		animation = fear
	}
	
	widget = {
		gui = "POD_event_window_widget_hunt"
		container = "custom_widgets_container"
	}
	
	immediate = {
		scope:target = {
			save_scope_as = victim	
		}
		
		scope:victim = {
			if = {
				limit = {
					NOT = {	has_character_flag = POD_resonance_negligible_flag }
					NOT = {	has_character_flag = POD_resonance_fleeting_flag }
					NOT = {	has_character_flag = POD_resonance_intense_flag }
					NOT = {	has_character_flag = POD_resonance_acute_flag }
					NOT = {	has_character_flag = choleric_resonance }
					NOT = {	has_character_flag = melancholic_resonance }
					NOT = {	has_character_flag = phlegmatic_resonance }
					NOT = {	has_character_flag = sanguine_resonance }
				}
				POD_set_resonance_effect = yes
			}
		}
	}
	
	# Take no blood
	option = 
	{
		name = POD_vamphunt.1.option.a
		
		stress_impact = {
			gluttonous = medium_stress_impact_gain
			#compassionate = medium_stress_impact_loss
		}	

		if = {
			limit = {
				has_trait = compassionate
			}
			random = {
				chance = 10
				stress_impact = {
					compassionate = miniscule_stress_impact_loss
				}			
			}
		}	
	}	
	
	# A sip
	option = {
		name = POD_vamphunt.1.option.b

		if = {
			limit = {
				faith = {
					has_doctrine_parameter = vampirism_illegal
				}
			}
			add_piety = minor_piety_loss
		}

		#Display resonance type
		if = {
			limit = {
				OR = {
					is_ventrue_trigger = no
					ventrue_weakness_compatible_trigger = {
						ACTOR = root
						RECIPIENT = scope:victim
					}
				}
			}
			POD_show_resonance_tooltip_effect = yes
		}

		# Resolve Blood Hunger Slaking from Sip
		POD_hunger_feeding_sip_from_victim_effect = { VICTIM = scope:victim }
		POD_resonance_feeding_sip_effect = yes

		# The Victim
		scope:victim = {
			add_character_modifier = {
				modifier = slightbloodloss_modifier
				days = 30	
			}
		}
	}
	
	# A drink
	option = {
		name = POD_vamphunt.1.option.c

		if = {
			limit = {
				faith = {
					has_doctrine_parameter = vampirism_illegal
				}
			}
			add_piety = major_piety_loss
		}

		#Display resonance type
		if = {
			limit = {
				OR = {
					is_ventrue_trigger = no
					ventrue_weakness_compatible_trigger = {
						ACTOR = root
						RECIPIENT = scope:victim
					}
				}
			}
			POD_show_resonance_tooltip_effect = yes
		}

		if = {
			limit = {
				faith = {
					has_doctrine_parameter = vampirism_shunned
				}
			}
			add_piety = medium_piety_loss
		}
		
		POD_hunger_feeding_drink_from_victim_effect = { VICTIM = scope:victim }
		POD_resonance_feeding_drink_effect = yes

		#The Victim
		scope:victim = {
			add_character_modifier = {
				modifier = smallbloodloss_modifier
				months = 2	
			}
		}
	}
	
	# A lot
	option = {
		name = POD_vamphunt.1.option.d

		if = {
			limit = {
				faith = {
					has_doctrine_parameter = vampirism_illegal
				}
			}
			add_piety = massive_piety_loss
		}

		#Display resonance type
		if = {
			limit = {
				OR = {
					is_ventrue_trigger = no
					ventrue_weakness_compatible_trigger = {
						ACTOR = root
						RECIPIENT = scope:victim
					}
				}
			}
			POD_show_resonance_tooltip_effect = yes
		}

		if = {
			limit = {
				faith = {
					OR = {
						has_doctrine_parameter = vampirism_shunned
						has_doctrine_parameter = hurt_mortal_lose_piety
					}
				}
			}
			add_piety = major_piety_loss
		}
		
		POD_minimal_risk_of_increasing_masquerade_exposure = yes

		stress_impact = {
			gluttonous = medium_stress_impact_loss
			compassionate = medium_stress_impact_gain
		}
		
		POD_hunger_feeding_lot_from_victim_effect = { VICTIM = scope:victim }
		POD_resonance_feeding_lot_effect = yes

		scope:victim = {
			add_character_modifier = {
				modifier = majorbloodloss_modifier
				months = 7	
			}
		}
	}
	
	# Kill them
	option = {
		name = POD_vamphunt.1.option.e

		if = {
			limit = {
				faith = {
					OR = {
						has_doctrine_parameter = vampirism_illegal
						has_doctrine_parameter = vampirism_shunned
						has_doctrine_parameter = hurt_mortal_lose_piety
					}
				}
			}
			add_piety = massive_piety_loss
		}

		#Display resonance type
		if = {
			limit = {
				OR = {
					is_ventrue_trigger = no
					ventrue_weakness_compatible_trigger = {
						ACTOR = root
						RECIPIENT = scope:victim
					}
				}
			}
			POD_show_resonance_tooltip_effect = yes
		}
		
		POD_small_risk_of_increasing_masquerade_exposure = yes
		
		stress_impact = {
			gluttonous = major_stress_impact_loss
			compassionate = major_stress_impact_gain
		}	
		
		POD_hunger_feeding_drain_from_victim_effect = { VICTIM = scope:victim }
		POD_resonance_feeding_drain_effect = yes
		
		save_scope_as = murderer
		save_scope_as = owner
		
		scope:victim = {
			save_scope_as = target
			if = {
				limit = {
					scope:murderer = {
						has_perk = cthonic_womb_perk
						has_variable = cthonic_womb_enabled
					}
				}
				set_variable = {
					name = marked_for_return
					value = scope:murderer
					days = 1
				}
				custom_tooltip = POD_cthonic_womb_perk_effect.tt
			}
			murder_success_kill_effect = yes
		}
	}
	
	#Herd them
	option = {
		name = POD_vamphunt.1.option.f
		trigger = {
			OR = {
				has_perk = lingering_kiss_perk 
				has_trait = presenceadvanced
			}
			POD_is_hungerless_trigger = no
			num_of_relation_herd_member < max_herd_cap
			scope:victim = { 
				POD_can_be_herd_trigger = yes
			}
			trigger_if = {
				limit = {
					POD_is_any_kind_of_ghoul_trigger = yes
				}
				scope:victim = { POD_is_vampire_trigger = yes }
			}
			OR = {
				is_ventrue_trigger = no
				ventrue_weakness_compatible_trigger = {
					ACTOR = root
					RECIPIENT = scope:victim
				}
			}
		}
		show_as_unavailable = {
			custom_description = {
				text = POD_herdify_vampire_interaction_limit
				num_of_relation_herd_member >= max_herd_cap
			}
		}
		if = {
			limit = {
				faith = {
					has_doctrine_parameter = vampirism_illegal
				}
			}
			add_piety = major_piety_loss
		}

		if = {
			limit = {
				faith = {
					has_doctrine_parameter = vampirism_shunned
				}
			}
			add_piety = medium_piety_loss
		}

		#The Victim
		scope:victim = {
			add_character_modifier = {
				modifier = smallbloodloss_modifier
				months = 2	
			}
		}		
		
		POD_hunger_feeding_drink_from_victim_effect = { VICTIM = scope:victim }
		POD_resonance_feeding_drink_effect = yes
		scope:victim = {
			set_relation_herd_owner = root
			add_character_flag = blocked_from_leaving
			POD_herd_resonance_setup = yes

			#Saves the herdifying vampire as the herd_owner and victim as the herdmember
			root = { save_scope_as = herd_owner }
			scope:victim = { save_scope_as = herdmember }

			#Starts the autofeeding process by triggering the recurring feeding event.
			trigger_event = POD_herd_automation.100
		}
		root = {
			add_hook = {
				type = herd_hook
				target = scope:victim
			}
		}
	}
	
	after = {
		scope:scheme ?= {
			end_scheme = yes
		}
		POD_end_journey_effect = yes
		
		# Keep hunting (hunt_in_capital Journey)
		if = {
			limit = {
				exists = scope:use_widget
				has_variable = continue_hunting
			}
			custom_tooltip = {
				text = hunt_in_capital_widget_tt
				POD_find_and_start_journey_effect = {
					TYPE = hunt_in_capital
				}
				POD_hunt_in_capital_cost_effect = yes
			}
		}
	}
}

POD_vamphunt.11 = {
	type = character_event
	title = POD_vamphunt.11.t
	desc =  POD_vamphunt.11.desc
	theme = vamphunt

	left_portrait = {
		character = root
		animation = anger
	}	
	
	right_portrait = {
		character = scope:victim
		animation = fear
	}	
	
	immediate = {
		scope:target = { save_scope_as = victim	 }
	}
	
	# Take no blood
	option = {
		name = POD_vamphunt.11.option.a
		
		stress_impact = {
			gluttonous = medium_stress_impact_gain
			#compassionate = medium_stress_impact_loss
		}	

		if = {
			limit = { has_trait = compassionate	}
			random = {
				chance = 10
				stress_impact = {
					compassionate = miniscule_stress_impact_loss
				}			
			}
		}		
	}	
	
	# Devore Yin chi
	option = {
		name = POD_vamphunt.11.option.b
		trigger = {
			has_midhigh_dharma_trigger = yes
			scope:victim = { yin_energy_points_value >= 1 }
		}
		POD_absorb_yinchi_from_breath_effect = { MURDERER = root VICTIM = scope:victim }
	}
	
	# Devore Yang chi
	option = {
		name = POD_vamphunt.11.option.c
		trigger = {
			has_midhigh_dharma_trigger = yes
			scope:victim = { yang_energy_points_value >= 1 }
		}
		POD_absorb_yangchi_from_breath_effect = { MURDERER = root VICTIM = scope:victim }
	}
	
	# Devore blood
	option = {
		name = POD_vamphunt.11.option.d
		trigger = {
			root.chi_difference <= scope:victim.lifeforce_points_value
		}

		if = {
			limit = {
				faith = { has_doctrine_parameter = hurt_mortal_lose_piety }
			}
			add_piety = major_piety_loss
		}

		stress_impact = {
			gluttonous = medium_stress_impact_loss
			compassionate = medium_stress_impact_gain
		}

		POD_absorb_chi_from_flesh_or_blood_effect = { MURDERER = root VICTIM = scope:victim }
	}
	
	# Kill them
	option = {
		name = POD_vamphunt.11.option.e

		if = {
			limit = {
				faith = { has_doctrine_parameter = hurt_mortal_lose_piety }
			}
			add_piety = massive_piety_loss
		}
		
		stress_impact = {
			gluttonous = major_stress_impact_loss
			compassionate = major_stress_impact_gain
		}

		POD_absorb_chi_from_flesh_or_blood_effect = { MURDERER = root VICTIM = scope:victim }
			
		save_scope_as = murderer
		save_scope_as = owner
		
		scope:victim = {
			save_scope_as = target
			murder_success_kill_effect = yes
		}
	}
	
	after = {
		scope:scheme ?= {
			end_scheme = yes
		}
		POD_end_journey_effect = yes
	}
}
