# Journey wrapper event for the old awaketorpor scheme
POD_awaketorpor.7000 = {
	hidden = yes
	
	immediate = {
		if = {
			limit = { var:current_journey.var:target = { is_alive = no } }
			POD_clear_journeys_with_target_effect = { TARGET = var:current_journey.var:target }
		}
		else = {
			POD_pause_journey_travel_plan_effect = yes
			save_scope_as = attacker
			var:current_journey.var:target = {
				save_scope_as = defender
			}
			random_list = {
				20 = { trigger_event = POD_awaketorpor.1 }
				20 = { trigger_event = POD_awaketorpor.2 }
				20 = { trigger_event = POD_awaketorpor.3 }
			}
		}
	}
}

POD_awaketorpor.1 = {
	type = character_event
	content_source = dlc_POD
	title = POD_awaketorpor.1.t
	desc =  POD_awaketorpor.1.desc		
	theme = ossuary
	right_portrait = {
		character = root
		animation = scheme
	}
	
	left_portrait = {
		character = scope:defender
		animation = dead
	}
	
	immediate = {
	
		scope:scheme.scheme_target_character ?= {
			save_scope_as = defender	
		}
		
		scope:scheme.scheme_owner ?= {
			save_scope_as = attacker
		}
		
		POD_save_coterie_duelists = yes
		
	}
	
# Diplomacy contest
	option = {
		name = POD_awaketorpor.1.option.a
		#custom_tooltip = POD_diablerize.100.option.a.tt
		
		scope:diplomacy_duelist = {
			duel = {
				skill = diplomacy
				value = 20
				13 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = 0.5
					}
					desc = POD_awaketorpor.1.option.aa.effect.success.desc
					
					root = {
						if = {
							limit = {
								NOT = {
									POD_custom_incompatible_splats_trigger = {
										ACTOR = root
										RECIPIENT = scope:defender
									}
								}
							}
							trigger_event = { 	
								id = POD_awaketorpor.10
								days = 7	
							}
						}
						else = {
							trigger_event = { 	
								id = POD_awaketorpor.13
								days = 7	
							}
						}
					}
				}
				13 = {
					# compare_modifier = {
						# value = scope:duel_value
						# #multiplier = 0.5
					# }

					desc = POD_awaketorpor.1.option.ab.effect.success.desc
					
					root = {
						send_interface_toast = {
							title = POD_awaketorpor.1.success
							left_icon = root
							remove_short_term_gold = medium_gold_value
						}
					
						if = {
							limit = {
								NOT = {
									POD_custom_incompatible_splats_trigger = {
										ACTOR = root
										RECIPIENT = scope:defender
									}
								}
							}
							trigger_event = { 	
								id = POD_awaketorpor.10
								days = 7	
							}
						}
						else = {
							trigger_event = { 	
								id = POD_awaketorpor.13
								days = 7	
							}
						}
					}
				}
				13 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = -0.5
					}

					desc = POD_awaketorpor.1.option.ac.effect.success.desc
					
					root = {
						trigger_event = {
							id = POD_awaketorpor.11
							days = 7	
						}
					}
				}
			}
		}
	}

# Give up
	option = {
		name = POD_awaketorpor.1.option.b
		
		stress_impact = {
			ambitious = medium_stress_impact_gain
			#compassionate = medium_stress_impact_loss
		}		
			
		scope:attacker = {
			scope:scheme ?= {
				end_scheme = yes
			}
			POD_end_journey_can_retry_effect = yes
		}	
		
		#custom_tooltip = POD_awaketorpor.1.option.b.custom	
		
	}

	after = {
		trigger_event = {
			id = POD_awaketorpor.666
			days = 150
		}
	}
}
