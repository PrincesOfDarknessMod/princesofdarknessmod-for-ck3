 namespace = POD_awaketorpor

scripted_trigger POD_awaketorpor_scheme_1_trigger = {
	any_scheme = {
		scheme_type = awaketorpor
		scheme_target = {
			this = scope:defender
		}
	}
}

POD_awaketorpor.1 = {
	type = character_event
	title = POD_awaketorpor.1.t
	desc =  POD_awaketorpor.1.desc		
	theme = ossuary
	right_portrait = {
		character = root
		animation = scheme
	}
	
	left_portrait = {
		character = scope:scheme.scheme_target
		animation = vampsleep
	}
	
	immediate = {
	
		scope:scheme.scheme_target = {
			save_scope_as = defender	
		}
		
		scope:scheme.scheme_owner= {
			save_scope_as = attacker
		}
		
	}
	
# Diplomacy contest
	option = {
		name = POD_awaketorpor.1.option.a
		#custom_tooltip = POD_diablerize.100.option.a.tt
		
		duel = {
			skill = diplomacy
			value = 20
				13 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = 0.5
					 }
				desc = POD_awaketorpor.1.option.aa.effect.success.desc
				
				trigger_event = { 	
					id = POD_awaketorpor.10
					days = 7	
					}
				
			}
				13 = {
					# compare_modifier = {
						# value = scope:duel_value
						# #multiplier = 0.5
					# }

				desc = POD_awaketorpor.1.option.ab.effect.success.desc

				send_interface_toast = {
					title = POD_awaketorpor.1.success
					left_icon = root
					remove_short_term_gold = medium_gold_value
				}
				
				trigger_event = { 	
					id = POD_awaketorpor.10
					days = 7	
					}				
				}
				13 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = -0.5
					 }

				desc = POD_awaketorpor.1.option.ac.effect.success.desc
				
				trigger_event = { 	
					id = POD_awaketorpor.11
					days = 7	
					}					
				}
			}	
	}

# Give up	
	option = {
		name = POD_awaketorpor.1.option.b
		
		stress_impact = {
			ambitious = medium_stress_impact_gain
			#compassionate = medium_stress_impact_loss
		}		
			
		scope:attacker = {
			scope:scheme = {
				end_scheme = yes
			}
		}	
		
		custom_tooltip = POD_awaketorpor.1.option.b.custom	
		
	}

	after = {
		trigger_event = {
			id = POD_awaketorpor.666
			days = 150
		}
	}
}

POD_awaketorpor.14 = {
	type = character_event
	title = POD_awaketorpor.14.t
	desc =  POD_awaketorpor.14.desc		
	theme = ossuary
	
	right_portrait = {
		character = root
		animation = rage
	}
	
	left_portrait = {
		character = scope:defender
		animation = vampsleep
	}

	immediate = {

		scope:defender = {
			save_scope_as = victim	
		}
		
		scope:attacker = {
			save_scope_as = diablerist
		}
		
		scope:victim = {
			if = {
				limit = {
					NOT = {	has_character_flag = POD_resonance_negligible_flag }
					NOT = {	has_character_flag = POD_resonance_fleeting_flag }
					NOT = {	has_character_flag = POD_resonance_intense_flag }
					NOT = {	has_character_flag = POD_resonance_acute_flag }
					NOT = {	has_character_flag = choleric_resonance }
					NOT = {	has_character_flag = melancholic_resonance }
					NOT = {	has_character_flag = phlegmatic_resonance }
					NOT = {	has_character_flag = sanguine_resonance }
				}
				POD_set_resonance_effect = yes
			}
		}			
	}	
	
	option = {
		name = POD_awaketorpor.14.option.a

		custom_tooltip = POD_diablerize.600.option.a.tt
		
		POD_give_diablerist_secret_or_trait_effect = yes
		POD_diablerie_piety_loss_effect = yes

		stress_impact = {
			ambitious = major_stress_impact_loss
			compassionate = major_stress_impact_gain
		}
		
		POD_hunger_feeding_drain_effect = yes
		POD_resonance_feeding_drain_effect = yes
		POD_diablerie_V5_effect = yes	
		POD_diablerie_gain_discipline_effect = yes
		POD_bloodpotency_increase_effect = yes
		
		hidden_effect = {
		
			scope:victim = {
				add_character_flag = {
					flag = POD_corpse_look_flag
				}
			}
		
			if = {
				limit = {
					NOT = { has_character_flag = POD_in_unconquered_soul_event }
				}
				POD_diablerie_unconquered_soul_effect = yes
			}
		}
				
# Murder success
		
		scope:attacker = {
			save_scope_as = murderer	
		}
		
		hidden_effect = {
			scope:victim = {
				murder_success_kill_effect = yes	
			}
		}
		
		custom_tooltip = POD_diablerize.600.option.a.tt.beast				
	}	
	
}
