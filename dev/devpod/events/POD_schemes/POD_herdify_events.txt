# Success
POD_herdify.1 = {
	type = character_event
	content_source = dlc_POD
	title = POD_herdify.1.t
	desc =  POD_herdify.1.desc		
	theme = vampire
	left_portrait = scope:target
	
	trigger = {
		POD_splat_can_herd_trigger = yes
	}
	
	on_trigger_fail = {
		trigger_event = POD_herdify.2
	}
	
	widget = {
		gui = "POD_shader_event_fog_alley_night"
		container = "background_shader_vfx_container"
	}
	
	option = {
		name = POD_herdify.1.option.a
		POD_herdify_effect = { TARGET = scope:target }
	}

	after = {
		if = {
			limit = { exists = scope:scheme }
			scope:scheme = { end_scheme = yes }
		}
	}
}
