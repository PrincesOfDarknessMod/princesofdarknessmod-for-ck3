# Fallen Hunter
POD_negotiation.4000 = {
	type = character_event
	content_source = dlc_POD
	
	theme = inquisition

	title = POD_negotiation.4000.t
	desc  = POD_negotiation.4000.desc
	
	trigger = {
		is_landed = yes
		# If you're already in the process of getting deposed, skip this event
		NOT = {
			any_character_war = {
				using_cb = depose_war
				casus_belli = {
					OR = {
						primary_defender = root
						primary_attacker = root
					}
				}
			}
		}
		# TODO MORTAL: what if you have no buildings left?
	}

	immediate = {
		# show the trait again if an earlier effect added it, for UX/clarity purposes
		if = {
			limit = { has_trait = turnedhunter }
			show_as_tooltip = {
				add_trait_force_tooltip = turnedhunter
			}
		}
		# save possible usurpers (county+ Hunter vassals or your liege)
		# if none exist, you're safe
		liege ?= {
			if = {
				limit = {
					this != root
					POD_is_supehunter_trigger = yes
					NOT = { is_imprisoned_by = root }
				}
				save_scope_as = usurper
			}
		}
		if = {
			limit = { NOT = { exists = scope:usurper } }
			ordered_vassal = {
				order_by = current_military_strength
				limit = {
					highest_held_title_tier >= tier_county
					POD_is_supehunter_trigger = yes
					NOT = { is_imprisoned_by = root }
				}
				save_scope_as = usurper
			}
		}
	}
	
	left_portrait = {
		character = root
		animation = personality_bold
	}

	# Lose duchy+ titles to most powerful Hunter vassal
	option = {
		name = POD_negotiation.4000.a
		trigger = {
			exists = scope:usurper
			is_independent_ruler = yes
		}
		
		# Transfer landed titles
		every_held_title = {
			limit = {
				is_landless_type_title = no
				tier > tier_county
			}
			add_to_list = target_titles
		}
		create_title_and_vassal_change = {
			type = usurped
			save_scope_as = change
			add_claim_on_loss = yes
		}
		every_in_list = {
			list = target_titles
			change_title_holder_include_vassals = {
				holder = scope:usurper
				change = scope:change
			}
		}
		becomes_independent = {
			change = scope:change
		}
		resolve_title_and_vassal_change = scope:change
		
		ai_chance = { base = 0 }
	}

	# If no count+ Hunter vassals: do nothing
	option = {
		name = POD_negotiation.4000.b
		trigger = { NOT = { exists = scope:usurper } }
		
		ai_chance = { base = 100 }
	}

	# Become landless
	option = {
		name = POD_negotiation.4000.c
		trigger = { has_ep3_dlc_trigger = yes }
		
		POD_depose_effect = yes
		
		ai_chance = { base = 0 }
	}

	# Start a war with EVERYONE
	# (If you're a vassal and can't go landless, this is your only option)
	# (But at that point, you kinda deserve it)
	option = {
		name = POD_negotiation.4000.d
		trigger = { exists = scope:usurper }
		add_internal_flag = dangerous
		
		top_liege = {
			every_vassal = {
				add_to_list = potential_war_joiners
			}
		}
		every_vassal = {
			add_to_list = potential_war_joiners
		}
		
		scope:usurper = {
			start_war = {
				cb = depose_war
				target = root
			}
			random_character_war = {
				limit = {
					using_cb = depose_war
					casus_belli = {
						primary_attacker = scope:usurper
						primary_defender = root
					}
				}
				save_scope_as = war
			}
		}
		
		hidden_effect_new_object = {
			every_in_list = {
				list = potential_war_joiners
				limit = {
					highest_held_title_tier >= tier_county
					POD_is_supehunter_trigger = yes
					NOT = { is_imprisoned_by = root }
					this != scope:usurper
					this != root
				}
				scope:war ?= {
					set_called_to = prev
					add_attacker = prev
				}
			}
		}
		
		ai_chance = { base = 100 }
	}
}
