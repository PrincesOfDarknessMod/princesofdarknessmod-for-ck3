namespace = pod_knights

scripted_trigger is_valid_root_for_POD_knight_events = {
	is_ai = no
	is_vampire_trigger = yes
	is_landed = yes
	is_available = yes
}

scripted_trigger is_valid_target_for_POD_knight_events = {
	is_landed = no
	is_vampire_trigger = yes
	NOT = { is_child_of = root }
	POD_is_clone_trigger = no
}

# Some events depending on your knight traits

pod_knights.1 = {
	type = character_event
	content_source = dlc_POD
	title = pod_knights.1.t
	desc = pod_knights.1.desc
	theme = masquerade
	left_portrait = {
		character = root
		animation = dismissal
	}
	right_portrait = {
		character = scope:knight
		animation = scheme
	}	
	
	widget = {
		gui = "POD_shader_event_fog_alley_night"
		container = "background_shader_vfx_container"
	}

	trigger = {
		is_valid_root_for_POD_knight_events = yes
		any_knight = {
			is_valid_target_for_POD_knight_events = yes
			has_trait = arrogant
		}
	}

	immediate = {
		random_knight = {
			limit = {
				is_valid_target_for_POD_knight_events = yes
				has_trait = arrogant
			}
			save_scope_as = knight
		}
	}

	option = { 
		name = pod_knights.1.a
		
		scope:knight = { 
			select_and_move_to_pool_effect = yes
		}
	}
	
	option = { 
		name = pod_knights.1.b
		
		POD_masquerade_exposure_increase_effect = yes
	}
}

pod_knights.3 = {
	type = character_event
	content_source = dlc_POD
	title = pod_knights.3.t
	desc = pod_knights.3.desc
	theme = masquerade
	left_portrait = {
		character = root
		animation = stress
	}
	right_portrait = {
		character = scope:knight
		animation = rage
	}	
	
	widget = {
		gui = "POD_shader_event_fog_alley_night"
		container = "background_shader_vfx_container"
	}

	trigger = {
		is_valid_root_for_POD_knight_events = yes
		any_knight = {
			is_valid_target_for_POD_knight_events = yes
			has_trait = wrathful
			prowess > 40
		}
	}

	immediate = {
		random_knight = {
			limit = {
				is_valid_target_for_POD_knight_events = yes
				has_trait = wrathful
				prowess > 40
			}
			save_scope_as = knight
		}
		if = {
			limit = {
				any_vassal_or_below = {
					is_available = yes
					prowess <= 40
				}
			}
			random_vassal = {
				limit = {
					is_available = yes
					prowess <= 40
				}
				save_scope_as = target
			}
		}
		else = {
			hidden_effect_new_object = {
				create_character = {
					location = root.capital_province
					template = POD_mortal_hunter_template
					save_scope_as = mortal
				}
				scope:mortal = {
					add_trait = "mortal"
					set_to_lowborn = yes
					add_character_flag = {
						flag = blocked_from_leaving
						years = 2
					}
					add_trait = POD_knows_too_much
					save_scope_as = target
				}
				add_courtier = scope:mortal
			}
		}			
	}

	option = { 
		name = pod_knights.3.a
		
		scope:knight = { 
			select_and_move_to_pool_effect = yes
		}
	}

	option = { 
		name = pod_knights.3.b
	
		unknown_murder_effect = {
			VICTIM = scope:target
			MURDERER = scope:knight
			REASON = death_execution
		}
	}
}

# Knight loyalty

pod_knights.99 = {
	hidden = yes
	immediate = {
		every_knight = {
			limit = {
				NOT = {
					prev = {
						is_target_in_variable_list = {
							name = potential_disloyal_knights
							target = prev
						}
					}
				}
			}
			# prevent event spam from checking knights that were employed just now
			add_to_list = knights_to_check
		}

		every_in_list = {
			variable = potential_disloyal_knights
			limit = {
				is_knight_of = prev
				POD_knight_should_leave_due_to_loyalty_trigger = { EMPLOYER = prev }
			}
			prev = {
				send_interface_message = {
					type = event_generic_bad
					title = pod_knights.100.message
					left_icon = prev
					prev = {
						custom_description = {
							text = "knight_became_wanderer"
							subject = this
							move_to_pool = yes
							if = {
								limit = { 
									has_relation_worshipped_demon = root 
									NOT = { has_relation_demonic_master = root }
									NOT = { has_relation_possessesing_wraith = root }
								}
								remove_relation_worshipped_demon = root
							}
							if = {
								limit = { 
									has_relation_worshipped_reborn = root 
									NOT = { house ?= root.house }
								}
								remove_relation_worshipped_reborn = root
							}
							POD_clear_discipline_battle_effect = yes
						}
					}
				}
			}
		}

		clear_variable_list = potential_disloyal_knights

		every_in_list = {
			list = knights_to_check
			limit = {
				POD_knight_should_leave_due_to_loyalty_trigger = { EMPLOYER = prev }
			}
			prev = {
				add_to_variable_list = {
					name = potential_disloyal_knights
					target = prev
				}
			}
		}

		if = {
			limit = {
				has_variable_list = potential_disloyal_knights
				variable_list_size = {
					name = potential_disloyal_knights
					value >= 1
				}
			}
			trigger_event = pod_knights.100
		}
	}
}

pod_knights.100 = {
	type = character_event
	content_source = dlc_POD
	title = pod_knights.100.t
	desc = pod_knights.100.desc
	theme = masquerade
	
	left_portrait = {
		character = root
		animation = stress
	}
	
	right_portrait = {
		character = scope:portrait1
		trigger = {
			exists = scope:portrait1
			NOT = { exists = scope:portrait2 }
		}
	}
	
	lower_left_portrait = {
		character = scope:portrait1
		trigger = {
			exists = scope:portrait1
			exists = scope:portrait2
		}
	}
	
	lower_center_portrait = {
		character = scope:portrait2
		trigger = { exists = scope:portrait2 }
	}
	
	lower_right_portrait = {
		character = scope:portrait3
		trigger = { exists = scope:portrait3 }
	}
	
	widget = {
		gui = "POD_shader_event_fog_alley_night"
		container = "background_shader_vfx_container"
	}

	immediate = {
		ordered_in_list = {
			variable = potential_disloyal_knights
			position = 0
			save_scope_as = portrait1
		}
		if = {
			limit = {
				variable_list_size = {
					name = potential_disloyal_knights
					value > 1
				}
			}
			ordered_in_list = {
				variable = potential_disloyal_knights
				position = 1
				save_scope_as = portrait2
			}
			if = {
				limit = {
					variable_list_size = {
						name = potential_disloyal_knights
						value > 2
					}
				}
				ordered_in_list = {
					variable = potential_disloyal_knights
					position = 2
					save_scope_as = portrait3
				}
			}
		}
	}

	option = { 
		name = pod_knights.100.a
		
		custom_tooltip = pod_knights.100.tt
		every_in_list = {
			variable = potential_disloyal_knights
			custom_description = {
				text = "character_getfirstnamenicknamed"
				subject = this
			}
		}
		custom_tooltip = pod_knights.100.tt2
	}

	option = { 
		name = pod_knights.100.b

		every_in_list = {
			variable = potential_disloyal_knights
			set_knight_status = forbid
		}
	}
}
