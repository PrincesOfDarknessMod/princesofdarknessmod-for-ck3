#### Vampire AI ####

#The big master event, fired for all playable characters
POD_vampire_maintenance.100 = {
	hidden = yes
	
	trigger = {
		has_POD_vamp_blood_trigger = yes #Vampires, Abominations, Ghouls
		is_playable_character = yes
		NOR = { #These two don't need any special maintenance
			POD_is_revenant_trigger = yes
			has_trait = dhampir
		}
		POD_is_clone_trigger = no
		is_in_torpor_trigger = no
		is_imprisoned = no
		is_ai = yes
	}

	immediate = {
		#Force end these
		if = {
			limit = {
				has_focus = necro_magic_focus
				necro_magic_lifestyle_perk_points >= 2
			}
			set_focus = no
		}
		if = {
			limit = {
				has_focus = blood_magic_focus
				blood_magic_lifestyle_perk_points >= 2
			}
			set_focus = no
		}


		#### Embrace ####
		trigger_event = POD_vampire_maintenance.101

		#### Herd ####
		trigger_event = POD_vampire_maintenance.102
		every_courtier_or_guest = {
			limit = { is_playable_character = no is_ai = yes }
			trigger_event = POD_vampire_maintenance.102
		}

		#### Ghouling ####
		trigger_event = POD_vampire_maintenance.103

		#### Gargoyles ####
		trigger_event = POD_vampire_maintenance.104

		#### Blood Heal Traits ####
		trigger_event = POD_vampire_maintenance.105
		every_courtier_or_guest = {
			limit = { is_playable_character = no is_ai = yes }
			trigger_event = POD_vampire_maintenance.105
		}

		#### Blood Surge ####
		trigger_event = POD_vampire_maintenance.106
		every_courtier_or_guest = {
			limit = { is_playable_character = no is_ai = yes }
			trigger_event = POD_vampire_maintenance.106
		}
		

		#### Elder Perks and Disciplines ####
		trigger_event = POD_vampire_maintenance.107

		#### Ghoul Management ####
		trigger_event = POD_vampire_maintenance.108

	}
}

# Embrace
POD_vampire_maintenance.101 = {
	type = character_event
	hidden = yes

    trigger = {
        is_vampire_trigger = yes
		NOT = { has_character_flag = ai_embrace_cd }
		trigger_if = {
			limit = { highest_held_title_tier >= tier_county }
			any_child = { is_alive = yes count < 2 }
		}
		trigger_else = { any_child = { is_alive = yes count < 1 } }
		NOT = { has_trait = vampgeneration16 }
		NOR = {
			vassal_contract_has_flag = progeny_rights_none
			vassal_contract_has_flag = progeny_rights_ghouls
		}
    }

    immediate = {
		add_character_flag = { flag = ai_embrace_cd years = { 3 10 } }
		save_scope_as = vampiresire
		POD_character_create_vampirechilde_effect = yes
	}
}

# Ghouling
POD_vampire_maintenance.103 = {
	type = character_event
	hidden = yes

    trigger = {
		POD_is_vampire_or_abomination_trigger = yes
		NOT = { has_character_flag = ai_ghouling_cd }
		highest_held_title_tier > tier_county
		num_of_relation_ghoul_thrall < vampire_ai_ghoul_limit
		faith = {
			OR = {
				has_doctrine_parameter = vampireghouling_accepted
				has_doctrine_parameter = vampireghouling_mass
			}
		}
		NOR = {
			vassal_contract_has_flag = progeny_rights_none
			vassal_contract_has_flag = progeny_rights_childer
		}
	}

	immediate = {
		create_character = {
			save_temporary_scope_as = POD_new_ghoul
			age = { 18 50 }
			location = THIS.capital_province
			culture = THIS.capital_province.culture
			faith = THIS.faith
			gender_female_chance = 50
			random_traits = yes
			after_creation = { POD_become_ghoul_effect = yes }
		}
		
		scope:POD_new_ghoul = {
			POD_remove_all_terrible_traits_effect = yes
			POD_become_bound_ghoul_effect = { DOMITOR = PREV }
			add_prowess_skill = root.vampire_ai_ghoul_prowess_boost
			while = {
				count = root.vampire_ai_ghoul_stat_boost
				POD_add_random_skill_point_no_prowess_effect = yes
			}
		}
		add_courtier = scope:POD_new_ghoul
		add_character_flag = { flag = ai_ghouling_cd years = { 5 15 } }
	}
}
