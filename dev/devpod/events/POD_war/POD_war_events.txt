namespace = POD_war

# Beneficiary selection after a selfless war
POD_war.1 = {
	type = character_event
	content_source = dlc_POD
	title = POD_war.1.t

	theme = crown
	
	desc = {
		desc = POD_war.1.desc
		desc = double_line_break
		desc = line_break
		desc = POD_saved_tooltip_weak
		triggered_desc = {
			trigger = {
				any_relation = {
					type = ashen_ambassadorial_patron
					count >= 1
				}
			}
			desc = {
				desc = line_break
				desc = POD_selfless_invasion_victory_ambassador_event
			}
		}
	}
	
	widget = {
		gui = "POD_event_window_widget_character_selection"
		container = "foreground_shader_vfx_container"
	}
	
	immediate = {
		# Variable setup happens in POD_selfless_invasion_victory_effect
	}

	option = {
		trigger = { exists = var:selected_event_character }
		name = POD_war.1.a
		
		POD_give_titles_to_beneficiary = {
			BENEFICIARY = var:selected_event_character
		}
	}

	option = {
		name = POD_war.1.b
		custom_tooltip = POD_war.1.b.tt
		custom_tooltip = POD_selfless_invasion_victory_patron
		
		POD_create_new_beneficiary = { SAVE_AS = beneficiary }
		hidden_effect_new_object = {
			POD_give_titles_to_beneficiary = {
				BENEFICIARY = scope:beneficiary
			}
		}
	}
	
	after = {
		POD_clear_character_select_vars = yes
	}
}

POD_war.2 = {
	type = character_event
	content_source = dlc_POD
	title = POD_war.2.t
	desc = POD_war.2.desc	
	theme = inquisition
	
	immediate = {
	}
	
	option = {
		name = POD_war.2.a

		add_prestige = 50
		root.dynasty = { add_dynasty_prestige = minor_dynasty_prestige_gain }
	}

	option = {
		name = POD_war.2.b

		domicile ?= {
			change_provisions = 100
		}
		add_gold = 50
	}

	option = {
		name = POD_war.2.c
		
		add_piety = 50
		
		POD_add_curia_support = { VALUE = 5 }
	}
}
