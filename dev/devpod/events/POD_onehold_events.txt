namespace = POD_onehold

# POD_onehold.1 - deferred capacity snapshot.
# Fired a day after POD_onehold_request_recalc_effect uncaps the character,
# so the read reflects the natural (uncapped) domain limit with all modifiers
POD_onehold.1 = {
	type = character_event
	hidden = yes

	immediate = {
		if = {
			limit = {
				POD_onehold_should_cap_trigger = yes
			}
			POD_onehold_refresh_tax_rung_effect = yes
			add_character_modifier = POD_onehold_cap
		}
	}
}
